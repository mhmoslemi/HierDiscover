"""
Propose a non‑negative step‑function (list of heights) that minimizes
evaluate_sequence().  The algorithm proceeds in three stages:

1.  A quick scan of a few analytic “edge‑weighted’’ families to obtain a
    decent seed (the families are heavy at the two ends of the interval).

2.  Refine the seed with projected gradient descent on a *smooth* version of
    the max‑convolution objective.  The smooth approximation is the
    log‑sum‑exp of the convolution entries; its gradient can be computed
    exactly by a single FFT‑based convolution (O(n log n)).  The smoothing
    parameter λ is increased gradually (continuation) so that the algorithm
    finally optimises the true max‑convolution.

3.  A final greedy pairwise mass‑transfer hill‑climbing pass removes any
    remaining tiny defects.

All operations respect the simplex constraint Σa = 1, a_i ≥ 0, which is
enforced by Euclidean projection after each update.  The whole search
fits comfortably into the 1000 s budget.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # already feasible?
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT (returns length len(a)+len(b)-1)."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used during the search.
    The vector a is assumed to be normalised (Σa = 1) and non‑negative.
    """
    n = a.shape[0]
    return float(2.0 * n * _fft_self_convolve(a).max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    Parameter p>0 controls how much mass sits at the two ends.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:                     # fallback – uniform
        w.fill(1.0 / n)
    else:
        w /= w_sum
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective ---------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    For a given weight vector w (same length as the full convolution),
    returns the gradient of Σ_k w_k·(a*a)_k with respect to a.
    Formula: grad = 2·(a ⋆ w) where ⋆ denotes correlation.
    Implemented via FFT convolution of a with the reversed weight vector.
    """
    n = a.shape[0]                     # length of a
    # reverse w for correlation
    w_rev = w[::-1]
    full = _fft_convolve(a, w_rev)    # length = n + (2n-1) - 1 = 3n-2
    start = len(w_rev) - 1             # = 2n-2, the index of the first valid correlation term
    grad = 2.0 * full[start:start + n]
    return grad

def _smooth_max_pgd(
    a0: np.ndarray,
    lam: float,
    step0: float,
    max_iter: int,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth approximation
    f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k).
    The gradient is obtained analytically (see _gradient_from_weights).
    """
    a = a0.copy()
    n = a.shape[0]
    for it in range(1, max_iter + 1):
        if time.time() >= time_limit:
            break

        conv = _fft_self_convolve(a)                     # shape 2n‑1
        # stable soft‑max weights
        max_c = conv.max()
        exp_shifted = np.exp(lam * (conv - max_c))
        w = exp_shifted / exp_shifted.sum()

        # gradient of the smooth max
        grad = _gradient_from_weights(a, w)

        # simple diminishing step size
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        if it % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}]  PGD λ={lam:.2f} it={it:<5d}"
                    f"  new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: move a random fraction of the mass from one index
    to another and keep the move only if it improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move up to 45 % of the source weight
        delta = a[i] * rng.random() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}]  swap i={i:>4d}→j={j:>4d} δ={delta:.2e}"
                f"   new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point: returns a plain Python list of non‑negative heights.
    The routine respects the given time budget (default 1000 s).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) quick analytic scan (very limited – enough to get a good seed)
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 16000]               # moderate sizes (fast enough)
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 9),          # 0.01 … 0.5  (edge‑heavy)
            np.linspace(0.5, 2.5, 8),          # moderate
        ]
    )
    lam_grid = [0.0, 0.1, 0.3, 0.6, 0.9]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))

            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                cand /= cand.sum()                     # enforce Σ=1
                sc = _objective(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] new seed len={n} p={p:.3g}"
                        f" lam={lam:.2f} score={sc:.6f}"
                    )

    # fallback – should never be needed
    if best_seq is None:
        n = 5000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 2) continuation‑style smooth‑max projected gradient descent
    # ------------------------------------------------------------------
    # schedule of smoothing parameters (starting gentle, ending sharp)
    lam_schedule = [0.5, 1.0, 2.0, 5.0, 10.0]
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a few thousand iterations per λ (fast because FFT is cheap)
        max_iter = min(5000, int(remaining * 10))   # very rough heuristic
        best_seq, best_score = _smooth_max_pgd(
            best_seq,
            lam=lam,
            step0=0.25,
            max_iter=max_iter,
            time_limit=deadline,
            best_seq=best_seq,
            best_score=best_score,
            eval_every=250,
        )
        print(
            f"[{time.time():.1f}] after λ={lam:.2f} best_score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 3) final greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        best_seq, best_score = _pairwise_swap(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
            max_trials=None,
        )
        print(
            f"[{time.time():.1f}] after polishing best_score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4) report and return
    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)) = {best_score:.6f}"
        f"   length = {len(best_seq)}"
    )
    # The caller will apply its own normalisation/clipping, so we return the raw vector.
    return best_seq.tolist()
