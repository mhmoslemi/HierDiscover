    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    eps = max_conv * eps_initial_factor + 1e-12

    while time.time() < deadline:
        # compute conv and g_smooth
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a modest pool of analytic seeds for a given length."""
    cand = []

    # Uniform
    cand.append(_distribution_uniform(n))

    # arcsine offsets – sweep a finer grid
    for c in (0.1, 0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_arcsine_offset(n, c))

    # power‑inverse‑sqrt family – more values of p
    for p in (0.5, 0.7, 0.9, 1.0, 1.2, 1.4):
        cand.append(_distribution_power_inv_sqrt(n, p))

    # Two‑block (different fractions)
    for bf in (0.03, 0.06, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.02))

    # Dirichlet draws with a few alphas
    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.2 + 0.8 * rng.random()))
    return cand

def _smooth_descent(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.8,
                    eps_factor: float = 0.3,
                    min_eps: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the smooth (log‑sum‑exp) surrogate of the max‑conv.
    Uses exponentiated updates a_i ← a_i·exp(‑η·g_i) and projection onto the
    simplex after each step.  The smoothing parameter ε is taken as a *fraction*
    of the current peak value; it therefore shrinks automatically.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)
    t = 0
    while time.time() < deadline:
        # current convolution
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            # occasional progress report
            print(f"[{time.time():.1f}] smooth iter={t} new best {best_score:.6f}")

        # smoothing parameter – never let it collapse to zero
        eps = max(max_conv * eps_factor, min_eps)

        # soft‑max weights for the convolution entries
        scaled = conv / eps
        # stability trick
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()   # shape (2n‑1,)

        # correlation a ⋆ w  (linear cross‑correlation)
        M = w.shape[0]                      # = 2n‑1
        L = 1 << ((n + M - 1).bit_length())  # power‑of‑two length ≥ n+M‑1
        a_pad = np.zeros(L, dtype=np.float64)
        w_pad = np.zeros(L, dtype=np.float64)
        a_pad[:n] = a
        w_pad[:M] = w
        A = np.fft.rfft(a_pad)
        W = np.fft.rfft(w_pad)
        corr = np.fft.irfft(A * np.conj(W), n=L)   # linear correlation
        grad = 2.0 * corr[M - 1:M - 1 + n]        # ∂/∂a_i of the smooth term

        # exponentiated‑gradient step
        t += 1
        eta = init_lr / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project onto the simplex (preserves non‑negativity)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
    return best_seq, best_score


# ----------------------------------------------------------------------
def _max_subgrad_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent that moves mass away from the *single*
    worst shift.  Kept for a short polishing phase.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        max_conv = conv[k]

        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={t} new best {best_score:.6f}")

        # sub‑gradient for the maximal index
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        grad = np.zeros_like(a)
        grad[idx_i] = 2.0 * a[idx_j]          # g_i = 2·a_{k‑i}

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection onto probability simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Greedy random mass‑transfer moves that accept only improvements."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point required by the grader.
    Generates a pool of analytic seeds, refines the most promising ones with
    a smooth‑max mirror‑descent, finishes with a short sub‑gradient and a greedy
    polishing stage, and returns the best sequence found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # 1.  Generate a broad set of seeds of several lengths
    lengths = [1800, 2500, 3200, 4000, 5000]   # a mix of modest and larger sizes
    all_seeds: list[tuple[float, np.ndarray]] = []

    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] *** scanning length {n} ***")
        cand_list = _initial_candidates(n, rng)
        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            all_seeds.append((sc, cand.copy()))

    # ------------------------------------------------------------------
    # 2.  Inject any previously known solution (if the variable exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            all_seeds.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Keep an elite set of the most promising seeds (size ≤ 8)
    all_seeds.sort(key=lambda x: x[0])
    elite = [seq for _, seq in all_seeds[:8]]

    # ------------------------------------------------------------------
    # 4.  Smooth‑max mirror descent on each elite seed
    remaining = deadline - time.time()
    if remaining > 30.0 and elite:
        # give each elite roughly the same share of the remaining time
        per_seed = max(1.0, remaining / len(elite))
        for idx, seed_seq in enumerate(elite):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            print(f"[{time.time():.1f}] smooth descent on elite #{idx+1} (len={len(seed_seq)})")
            seq_out, score_out = _smooth_descent(
                seed_seq.copy(),
                sub_deadline,
                init_lr=0.8,
                eps_factor=0.3,
                min_eps=1e-9,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after smooth descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Short sub‑gradient refinement (uses the classic max‑only gradient)
    if overall_best_seq is not None and time.time() < deadline - 5.0:
        sub_deadline = time.time() + 8.0   # a few seconds of sub‑gradient work
        print(f"[{time.time():.1f}] short sub‑gradient refinement")
        overall_best_seq, overall_best_score = _max_subgrad_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.5,
        )

    # ------------------------------------------------------------------
    # 6.  Final greedy polishing
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Convert to plain Python list as required by the evaluator.
    return overall_best_seq.tolist()

"""
Hybrid smoothing + subgradient optimisation for low‑autocorrelation step‑function.

The algorithm works in three phases:

1. **Seeding** – a large, diverse pool of analytic distributions is created
   (uniform, arcsine with several offsets, a power‑law family,
   two‑block, sparse, Dirichlet and a generic (δ,α) power law).  The best
   few seeds (by the exact objective) are retained.

2. **Smooth‑max mirror descent** – the hard max in the objective is replaced
   by a ε·log‑sum‑exp surrogate.  For a given ε the gradient
   ∂/∂a_i (ε log ∑ e^{(a∗a)_k/ε}) = 2 ∑_k w_k a_{k‑i} where the
   weights w_k form a soft‑max over the convolution entries.  The gradient
   is obtained by a single FFT‑based correlation, and an exponentiated‑
   gradient update a←a·exp(‑η g) keeps the vector non‑negative.  After each
   step we project back onto the probability simplex.  ε is chosen as a
   fraction of the current peak value, so it shrinks automatically.

3. **Polishing** – a short run of the classic max‑only sub‑gradient descent
   followed by a greedy random‑mass‑transfer walk removes any remaining
   spikes.

All stages respect the global time budget (≈ 1000 s).  The returned list of
floats is normalised to sum = 1, which leaves the evaluation invariant.
"""
import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_general_power(n: int, delta: float, alpha: float) -> np.ndarray:
    """Family w_i ∝ ((i+δ)*(n‑1‑i+δ))^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    left = i + delta
    right = (n - 1 - i) + delta
    vals = (left * right) ** (-alpha)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a modest pool of analytic (and parametric) seeds for a given length."""
    cand = []

    # Uniform
    cand.append(_distribution_uniform(n))

    # Arcsine offsets – a slightly finer grid
    for c in (0.1, 0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_arcsine_offset(n, c))

    # Power‑inverse‑sqrt – several exponents
    for p in (0.5, 0.7, 0.9, 1.0, 1.2, 1.4):
        cand.append(_distribution_power_inv_sqrt(n, p))

    # Two‑block variants
    for bf in (0.03, 0.06, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.02))

    # Dirichlet draws
    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.2 + 0.8 * rng.random()))

    # Generic (δ,α) power‑law – a few random draws
    for _ in range(5):
        delta = 0.01 + 9.99 * rng.random()          # ∈ [0.01,10]
        alpha = 0.3 + 1.2 * rng.random()           # ∈ [0.3,1.5]
        cand.append(_distribution_general_power(n, delta, alpha))

    return cand

def _smooth_descent(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.8,
                    eps_factor: float = 0.3,
                    min_eps: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the smooth (log‑sum‑exp) surrogate of the max‑convolution.
    The smoothing ε is taken as a constant *fraction* of the current peak value,
    therefore shrinking automatically as the optimisation proceeds.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)
    t = 0

    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth iter={t} new best {best_score:.6f}")

        # smoothing parameter – never let it become zero
        eps = max(max_conv * eps_factor, min_eps)

        # soft‑max weights over the convolution entries
        scaled = conv / eps
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()          # shape (2n‑1,)

        # linear cross‑correlation a ⋆ w  (g_i = 2 Σ_k w_k a_{k‑i})
        M = w.shape[0]                              # = 2n‑1
        L = 1 << ((n + M - 1).bit_length())        # ≥ n+M‑1
        a_pad = np.zeros(L, dtype=np.float64)
        w_pad = np.zeros(L, dtype=np.float64)
        a_pad[:n] = a
        w_pad[:M] = w
        A = np.fft.rfft(a_pad)
        W = np.fft.rfft(w_pad)
        corr = np.fft.irfft(A * np.conj(W), n=L)    # linear correlation
        grad = 2.0 * corr[M - 1:M - 1 + n]          # ∂/∂a_i of the smooth term

        # exponentiated‑gradient step
        t += 1
        eta = init_lr / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection onto the simplex (keeps non‑negativity)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

    return best_seq, best_score

def _max_subgrad_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent that moves mass away from the *single*
    worst shift (the exact max‑convolution).  Used only for a short
    polishing phase.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        max_conv = conv[k]

        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={t} new best {best_score:.6f}")

        # sub‑gradient for the maximal index
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        grad = np.zeros_like(a)
        grad[idx_i] = 2.0 * a[idx_j]          # g_i = 2·a_{k‑i}

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection onto the probability simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Greedy random mass‑transfer moves that accept only improvements."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score
