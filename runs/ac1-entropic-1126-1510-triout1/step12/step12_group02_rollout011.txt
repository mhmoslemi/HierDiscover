def _smooth_grad_max(a, alpha):
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    # subtract maxc for stable exponent
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    # compute correlation via FFT

    return grad_M, M_alpha

def _smooth_exponentiated_gradient_descent(a, deadline, best_seq, best_score, alpha_start, alpha_end, eta0=0.6, eval_every=50):

```
t=0
start = time.time()
while time.time() < deadline:
    t +=1
    elapsed = time.time() - start
    frac = elapsed / (deadline - start)
    alpha = alpha_start * ((alpha_end/alpha_start) ** frac) if alpha_start>0 else alpha_end
    grad_M, M_alpha = _smooth_grad_max(a, alpha)
    grad = 2.0 * a.shape[0] * grad_M
    eta = eta0 / np.sqrt(t)
    a = a * np.exp(-eta * grad)
    a_sum = a.sum()
    if a_sum == 0.0:
         a = np.full_like(a, 1.0 / a.shape[0])
    else:
         a = a / a_sum
    if t % eval_every == 0:
        cur = _objective(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SMGD it={t} alpha={alpha:.2f} new best {best_score:.6f}")

Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Rich seed generation** – families: uniform, arcsine, Hann,
   symmetric Beta, edge‑weighted, generalized normal, power‑law,
   random Dirichlet and sparse vectors.  Several lengths (2k–16k)
   are tried and the 30 best seeds are retained.

2. **Smooth exponential‑gradient descent** – instead of using the
   gradient of only the active (max) convolution entry we minimise a
   smooth approximation of the max using the log‑sum‑exp trick.
   The temperature parameter α is increased from 1 to ≈100 so the
   objective gradually focuses on the true supremum.  Updates are
   performed by the exponentiated‑gradient rule
   ``a ← a·exp(‑η·∇obj)`` and normalised onto the probability simplex.

3. **Exact max‑gradient descent** – a short burst of the classic
   exponentiated‑gradient descent that uses the exact active
   convolution entry (as in the previous state‑of‑the‑art).

4. **Upsampling & polishing** – the current best is linearly
   interpolated to a larger length (up to 64 k) and refined again
   with both smooth and exact gradient steps.

5. **Long‑run pairwise annealing** – random mass transfers between
   bins with a simple simulated‑annealing schedule to escape any
   remaining local minima.

All stages respect a total wall‑clock budget of 1000 s (a 2 s safety

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution via real FFT (full linear self‑convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective used by evaluate_sequence (sum is assumed to be 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a normalised vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – all return a probability vector (sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x‑0.5|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law: w_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    w = i ** (-p) + (n - i + 1) ** (-p)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density (Dirichlet‑like)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination (lam ∈ [0,1]) of a base distribution with the uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth max‑gradient (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def _smooth_grad_max(a: np.ndarray, alpha: float):
    """
    Compute gradient of the α‑smoothed maximum convolution entry.
    Returns (grad_M, M_alpha) where
        M_alpha = (1/α)·log Σ exp(α·c_k)   (c = a*a)
        grad_M = ∂M_alpha/∂a (size n numpy array)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    maxc = conv.max()
    # stable softmax on the convolution entries
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp                           # softmax probabilities
    M_alpha = maxc + np.log(sum_exp) / alpha

    # correlation g_i = Σ_k p_k·a_{k‑i}  (zero for out‑of‑range indices)
    size = 1 << ((3 * n - 2) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    P = np.fft.rfft(p, n=size)
    corr = np.fft.irfft(A.conj() * P, n=size)        # linear correlation
    g = corr[:n]                                     # lags 0 … n‑1

    grad_M = (2.0 / alpha) * g
    return grad_M, M_alpha


def _smooth_exponentiated_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    alpha_start: float = 1.0,
    alpha_end: float = 100.0,
    eta0: float = 0.6,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth approximation.
    The smoothing parameter α grows from ``alpha_start`` to ``alpha_end``.
    """
    n = a.shape[0]
    t = 0
    start = time.time()
    while time.time() < deadline:
        t += 1
        # linear schedule in the (log) ratio of alphas
        frac = (time.time() - start) / (deadline - start)
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)

        grad_M, M_alpha = _smooth_grad_max(a, alpha)
        # full objective gradient (sum is kept =1)
        grad = 2.0 * n * grad_M                     # ∇(2·n·M)
        # clipping to avoid overflow in exp()
        grad = np.clip(grad, -50.0, 50.0)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SMGD it={t} α={alpha:.2f} new best {best_score:.6f}")
    return best_seq, best_score

# Exact max‑gradient descent (as in the previous solution)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray):
    """Gradient of the *currently* maximal convolution entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


def _exact_exponentiated_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """Classic exponentiated‑gradient descent using the *active* max entry."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact GD it={t} new best {best_score:.6f}")
    return best_seq, best_score

def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random mass transfers with a simple simulated‑annealing schedule."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new
