def _grad_obj(a):
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    M = conv[k]
    S = a.sum()
    # compute rev
    rev = np.zeros_like(a)
    if k < n:
        rev[:k+1] = a[k::-1]
    else:
        start = k - (n - 1)
        length = 2 * n - 1 - k
        rev[start:] = a[:length][::-1]
    factor = (4.0 * n) / (S ** 3)
    grad = factor * (S * rev - M)
    return grad, k, M, S

Now gradient descent function:

```
def _gradient_descent_obj(a, deadline, best_seq, best_score, eta0=0.2, eval_every=20):
    n = len(a)
    t = 0
    while time.time() < deadline:
        t += 1
        grad, k, M, S = _grad_obj(a)
        eta = eta0 / math.sqrt(t)  # decreasing schedule
        # multiply step: a_i = a_i * exp(-eta * grad_i)
        a = a * np.exp(-eta * grad)
        # enforce non-negative and renorm
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum
        # symmetrize
        a = (a + a[::-1]) / 2.0
        if t % eval_every == 0:
            cur = _objective(a)  # same as evaluate function ignoring clamps
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD obj it={t} new global best {best_score:.6f}")
    return best_seq, best_score
```

"""
Search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (∑a)²

The algorithm combines a broad pool of analytic seeds (Beta, edge‑weighted,
generalised‑normal, sparse, Dirichlet …) with a *gradient descent on the true
objective* (including the denominator) followed by a long pairwise annealing
polish.  The gradient of the objective is known in closed form:

    ∂O/∂a_i = (4·n / S³)·(S·a_{k‑i} – M)

where k is the index attaining the maximal convolution entry M
and S = Σ a_i.  An exponentiated‑gradient step keeps the variables non‑negative
and a final normalisation restores the simplex constraint.
The whole routine respects the 1000 s budget (2 s safety margin).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (identical to the supplied evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Objective used by the scorer – includes the same clipping as evaluate_sequence."""
    n = a.shape[0]
    a = np.clip(a, 0.0, 1000.0)            # same security checks as the grader
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    """Classic Hann window (cosine taper)."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x‑0.5|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the *exact* objective (including denominator)
# ----------------------------------------------------------------------
def _grad_of_objective(a: np.ndarray):
    """
    Returns (grad, k, M, S) where grad_i = ∂O/∂a_i,
    k is the active convolution index,
    M = (a*a)[k] and S = Σ a_i.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    M = conv[k]
    S = a.sum()
    # build the shifted vector a_{k-i} (zero outside the support)
    rev = np.zeros_like(a)
    if k < n:
        # i = 0..k
        rev[: k + 1] = a[k::-1]
    else:
        start = k - (n - 1)                # first i with a_{k-i} in range
        length = 2 * n - 1 - k             # how many such i's
        rev[start:] = a[:length][::-1]
    factor = (4.0 * n) / (S ** 3)
    grad = factor * (S * rev - M)
    return grad, k, M, S


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _gradient_descent_obj(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.2,
    eval_every: int = 20,
) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient descent on the exact objective O.
    The update is a ← a·exp(-η·∇O) followed by normalisation and a
    symmetry projection (a ≔ (a + a[::-1])/2).  The best sequence seen
    during the run is returned.
    """
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        grad, k, M, S = _grad_of_objective(a)
        # step size decreasing like 1/√t
        eta = eta0 / math.sqrt(it)
        # multiplicative update (exponentiated gradient)
        a = a * np.exp(-eta * grad)
        # renormalise to the simplex (Σa = 1)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total
        # enforce symmetry (helps the landscape)
        a = (a + a[::-1]) * 0.5

        if it % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] GD it={it} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers (polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins, accepting improvements
    or, early on, hill‑climbs with a temperature schedule."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        trials += 1
        src = rng.choice(np.where(a > 1e-12)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        # move up to 40 % of the source bin
        delta = a[src] * rng.random() * 0.4
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01
        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal src={src}→dst={dst} Δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score

    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Build a diverse pool of seeds (moderate lengths first)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000]
    # p‑values for Beta / edge‑weighted families – go well below 0.01
    p_vals = np.concatenate(
        [
            np.logspace(-4, -2, 6),   # 1e‑4 … 1e‑2
            np.logspace(-2, 0, 6),    # 1e‑2 … 1
        ]
    )
    lam_vals = np.linspace(0.0, 1.0, 5)   # mixing with uniform
    q_vals = np.linspace(0.5, 3.0, 6)     # generalized‑normal

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds, length={n}")

        # deterministic families
        families = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
        ]

        # parameterised families
        for p in p_vals:
            families.append(_distribution_beta_sym(n, p))
            families.append(_distribution_edge_weighted(n, p))

        for q in q_vals:
            families.append(_distribution_generalized_normal(n, q))

        # random Dirichlet seeds
        for _ in range(4):
            alpha = rng.uniform(0.01, 2.0)
            families.append(rng.dirichlet(np.full(n, alpha)))

        # sparse random seeds
        for _ in range(4):
            families.append(_distribution_random_sparse(n, rng, density=0.01))

        # mix each base with the uniform distribution
        all_seeds = []
        for base in families:
            for lam in lam_vals:
                all_seeds.append(_mix_with_uniform(base, lam))

        # evaluate and keep the very best
        for cand in all_seeds:
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # keep only the top‑30 seeds overall (pre‑selection)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  If a previously found construction exists, add it to the pool
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – guarantee a non‑empty answer
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Gradient descent on a few elite seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # take the 6 best (but not more than we have time for)
        elite = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(elite))
        print(f"[{time.time():.1f}] gradient descent on {len(elite)} elite seeds")
        for idx, seed_seq in enumerate(elite):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            # enforce symmetry from the start
            seed_seq = (seed_seq + seed_seq[::-1]) * 0.5
            seed_seq /= seed_seq.sum()
            a_opt, sc_opt = _gradient_descent_obj(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.2,
                eval_every=25,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after GD seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final long‑run annealing polish (uses the full remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Normalise once more and report
    # ------------------------------------------------------------------
    overall_best_seq = np.maximum(overall_best_seq, 0.0)
    if overall_best_seq.sum() == 0.0:
        overall_best_seq = _distribution_uniform(len(overall_best_seq))
    overall_best_seq /= overall_best_seq.sum()

    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
