"""
Improved optimizer for the self‑convolution L∞‑ratio problem.

Strategy
--------
1. Seed generation – a diverse collection of analytic families (uniform,
   sine, beta, Dirichlet, power‑law, two‑block, arcsine) plus many random
   Dirichlet draws (different concentration parameters) for a *moderate*
   length (n≈3000).  All seeds are normalised to sum = 1 and the exact
   objective 2·n·max(conv)/sum² is evaluated.

2. Selection – the best few seeds (elite) are retained (size ≤ 8).

3. Long refinement – each elite seed is refined with a two‑phase exponentiated‑
   gradient descent:
   * Phase A – soft‑max smoothing of the max‑convolution entry.  The temperature
     α grows exponentially from a small value to a larger one (≈ 200) while a
     decaying learning‑rate η = η₀/√t is used.  The update is performed on the
     log‑weights (a ← a·exp(−η·grad)) and the vector is projected back onto the
     simplex by simple renormalisation.
   * Phase B – hard‑max sub‑gradient descent (gradient of the single maximal
     convolution entry) with the same learning‑rate schedule.

   Both phases run until the per‑seed time budget is exhausted (the loop
   checks the global deadline after every iteration).

4. Final polishing – a cheap random‑mass‑transfer walk (pairwise swaps) is
   performed until the total time budget expires.  A move transfers a random
   fraction (≤ 30 %) of source mass to a random destination and is kept only if
   the true objective improves.

All heavy work (FFT based convolution, gradient computation) is vectorised
with NumPy.  The routine respects the provided `budget_s` seconds and returns
the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution via real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (a is assumed non‑negative)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_i w_i·conv_i w.r.t a (soft‑max weights)."""
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the single maximal convolution entry `conv[k]`."""
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]

    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a broad pool of analytic and random seeds for a given length."""
    cand = []

    # analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))

    for p in np.linspace(0.05, 1.0, 12):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # random sparse draws
    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws with several concentration parameters
    for a in (0.02, 0.05, 0.1, 0.3, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    return cand


# ----------------------------------------------------------------------
def _softmax_grad_descent(a: np.ndarray, deadline: float,
                          best_seq: np.ndarray, best_score: float,
                          lr0: float = 0.9,
                          max_iter: int = 300000,
                          eval_interval: int = 400,
                          alpha_start: float = 0.4,
                          alpha_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using a soft‑max approximation of the max.
    Temperature α grows exponentially: α = α_start·(α_end/α_start)^{t/max_iter}.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # exponential temperature schedule
        alpha = alpha_start * (alpha_end / alpha_start) ** (t / max_iter)

        # stable soft‑max weights
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max it={t} α={alpha:.2f} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hardmax_grad_descent(a: np.ndarray, deadline: float,
                          best_seq: np.ndarray, best_score: float,
                          lr0: float = 0.5,
                          max_iter: int = 300000,
                          eval_interval: int = 400) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximal convolution entry.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        grad = _gradient_max_index(a, k)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t} idx={k} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         max_fraction: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Cheap random mass‑transfer walk.  A random fraction of the source mass
    (up to `max_fraction`) is moved to another index.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        delta = a[i] * rng.random() * max_fraction
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-15:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Entry point for the competition.
    Generates a set of seeds, refines the most promising ones with a
    two‑phase exponentiated‑gradient descent, and finishes with a cheap
    random walk.  Returns the best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Pick a working length – a trade‑off between quality and speed.
    #    3000 gives a good balance for the budget.
    n = 3000

    # ------------------------------------------------------------------
    # 2. Build a large pool of seeds and keep the elite.
    elite = []          # list of (score, sequence) tuples
    overall_best_seq = None
    overall_best_score = float("inf")

    # analytic + random seeds
    candidates = _initial_candidates(n, rng)

    # evaluate all raw candidates
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] !!! new global best (seed) score={sc:.6f}")
        elite.append((sc, cand.copy()))

    # keep only a few elite members
    elite.sort(key=lambda x: x[0])
    elite = elite[:8]

    # ------------------------------------------------------------------
    # 3. Bring in the previous run if available (height_sequence_1)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:8]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Long refinement of the elite seeds.
    remaining = deadline - time.time()
    if remaining > 30.0 and elite:
        per_seed_budget = remaining / (len(elite) * 2.0)   # split equally between two phases
        for seq0, (score0, seq) in enumerate(elite):
            if time.time() >= deadline:
                break

            # soft‑max phase
            sub_deadline_a = time.time() + per_seed_budget
            seq_a, score_a = _softmax_grad_descent(
                seq.copy(),
                sub_deadline_a,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                max_iter=500000,
                eval_interval=500,
                alpha_start=0.4,
                alpha_end=200.0,
            )
            if score_a < overall_best_score:
                overall_best_score = score_a
                overall_best_seq = seq_a.copy()
                print(f"[{time.time():.1f}] after soft‑max new best {overall_best_score:.6f}")

            # hard‑max phase
            if time.time() >= deadline:
                break
            sub_deadline_b = time.time() + per_seed_budget
            seq_b, score_b = _hardmax_grad_descent(
                seq_a.copy(),
                sub_deadline_b,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                max_iter=500000,
                eval_interval=500,
            )
            if score_b < overall_best_score:
                overall_best_score = score_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after hard‑max new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final cheap polishing (pairwise random swaps)
    if time.time() < deadline and overall_best_seq is not None:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_fraction=0.30,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Convert to python list for the checker
    return overall_best_seq.tolist()
