    # initialize logits as log(a+eps)

    v = np.zeros_like(z)

        maxz = np.max(z)
        a = np.exp(z - maxz)

def _simulated_annealing_swaps(a, deadline, best_seq, best_score, rng, max_trials=1e8, init_temp=1e-1, final_temp=1e-6):
    n = a.shape[0]
    cur_a = a.copy()
    temperature = init_temp
    decay = (final_temp / init_temp) ** (1.0 / (deadline - time.time()))
    while time.time() < deadline:
        # choose source from non-zero entries (including small)
        src_idx = rng.integers(n)
        if cur_a[src_idx] < 1e-15:
            continue
        dst_idx = rng.integers(n)
        if dst_idx == src_idx:
            continue
        delta = cur_a[src_idx] * rng.random() * 0.5  # move up to 50%
        new_a = cur_a.copy()
        new_a[src_idx] -= delta
        new_a[dst_idx] += delta
        cur_val = _objective(new_a)
        # accept if better or with probability exp(-(cur_val - best_score)/temperature)
        if cur_val < best_score - 1e-14:
            cur_a = new_a
            best_score = cur_val
            best_seq = cur_a.copy()
            print(f"[{time.time():.1f}] anneal i={src_idx}→j={dst_idx} δ={delta:.3g} new best {best_score:.6f}")
        else:
            # optional uphill acceptance
            if temperature > 0 and rng.random() < math.exp(-(cur_val - best_score) / temperature):
                cur_a = new_a
        # update temperature
        temperature *= decay
    return best_seq, best_score

The optimiser tries to minimise  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2}\qquad a_i\ge 0 .
\]

The value of *F* does **not** depend on a global scale – only on the shape of the
step‑function.  
The programme therefore works on the simplex  
\(\{a\mid a_i\ge0,\;\sum a_i=1\}\).

The search consists of three stages

1. **Diverse seeding** – many analytic families (uniform, arcsine, Beta,
   Dirichlet, exponential, power‑law, sparse, two‑block, sine, random monotone)
   are generated for several lengths.  Their exact objective values are
   computed and the best few are kept as *elite* seeds.

2. **Two‑phase Adam‑based descent** – each elite seed is refined first with a
   *soft‑max* (smooth) approximation of the max‑convolution using an Adam optimiser
   on the log‑probability (mirror‑descent) parameters, then with a *hard‑max*
   sub‑gradient (the true maximal entry).  Adam’s adaptive moments give a
   much more stable and faster convergence than the pure exponentiated‑gradient
   used previously.

3. **Simulated‑annealing polishing** – a cheap random mass‑transfer walk is
   performed while a slowly cooling temperature occasionally accepts uphill
   moves.  This helps to escape tiny local minima that the gradient phases miss.

All steps respect the global time budget (`budget_s`).  The best sequence found
is returned as a plain list of floats; it is already normalised to sum = 1, so
`evaluate_sequence` can be called on it directly.

"""
Hybrid optimiser for the self‑convolution max‑ratio problem.

- A large pool of analytic/random seeds (uniform, arcsine, Beta, Dirichlet,
  exponential, power‑law, sparse, two‑block, sine, random monotone) is built
  for a handful of lengths.
- The most promising seeds are refined by two Adam‑based phases:
    * soft‑max (smooth) descent,
    * hard‑max (true maximal entry) descent.
- Finally a simulated‑annealing random‑swap walk polishes the result.
- The routine respects the given time budget and returns the best sequence.
"""

import time
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (a is non‑negative)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """∇_a (∑_k w_k·conv_k) = 2·a⋆w  (cross‑correlation via FFT)."""
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the single maximal convolution entry conv[k]."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i
    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]
    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]
    return g

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_power_decreasing(n: int, p: float) -> np.ndarray:
    # a_i ∝ (n-i)^p
    i = np.arange(n, dtype=np.float64)
    vals = (n - i) ** p
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals

    i = np.arange(n, dtype=np.float64)

    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_random_monotone(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random decreasing sequence by sorting uniform draws."""
    vals = np.sort(rng.random(n))[::-1]
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic and random seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))

    # symmetric beta on a fine grid
    for p in np.linspace(0.05, 1.0, 12):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric beta
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power laws
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))
        cand.append(_distribution_power_decreasing(n, p))

    # exponentials
    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))
        cand.append(_distribution_exponential_decreasing(n, lam))

    # sparse random draws
    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws with various concentrations
    for a in (0.1, 0.3, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # random monotone sequences
    for _ in range(4):
        cand.append(_distribution_random_monotone(n, rng))

    return cand

def _adam_softmax_descent(a: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          lr: float = 0.4,
                          max_iter: int = 120000,
                          eval_interval: int = 400,
                          alpha_start: float = 0.5,
                          alpha_end: float = 80.0,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑probability (mirror descent) using a soft‑max
    approximation of max(conv).  Parameters are tuned for stability.
    """
    n = a.shape[0]
    # initialise logits (log‑probabilities) – safe from zeros
    z = np.log(a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0

    while t < max_iter and time.time() < deadline:
        t += 1
        # recover probability vector from logits
        maxz = np.max(z)
        a = np.exp(z - maxz)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
            a_sum = 1.0
        a /= a_sum

        # convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t. a (smooth objective)
        grad_a = _gradient_from_weights(a, w)

        # transform to gradient w.r.t. logits (mirror‑descent)
        avg = np.dot(a, grad_a)
        g_z = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z * g_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= step

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam‑soft it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _adam_hardmax_descent(a: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          lr: float = 0.2,
                          max_iter: int = 120000,
                          eval_interval: int = 400,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser using the *true* maximal convolution entry (hard‑max).
    """
    n = a.shape[0]
    z = np.log(a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0

    while t < max_iter and time.time() < deadline:
        t += 1
        maxz = np.max(z)
        a = np.exp(z - maxz)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
            a_sum = 1.0
        a /= a_sum

        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        grad_a = _gradient_max_index(a, k)

        avg = np.dot(a, grad_a)
        g_z = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z * g_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= step

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam‑hard it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score

def _simulated_annealing_swaps(a: np.ndarray,
                               deadline: float,
                               best_seq: np.ndarray,
                               best_score: float,
                               rng: np.random.Generator,
                               init_temp: float = 0.1,
                               final_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer walk with a temperature schedule (accepts
    occasional uphill moves).  Very cheap, suitable for the last seconds.
    """
    n = a.shape[0]
    cur = a.copy()
    temperature = init_temp
    # exponential temperature decay so that we reach final_temp exactly at deadline
    remaining = max(1e-6, deadline - time.time())
    decay = (final_temp / init_temp) ** (1.0 / remaining)

    while time.time() < deadline:
        # pick source (must have non‑zero mass) and destination
        src = rng.integers(n)
        if cur[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if dst == src:
            continue

        # move a random fraction (up to 50 %) of the source mass
        delta = cur[src] * rng.random() * 0.5
        if delta < 1e-15:
            continue

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        cur_val = _objective(new)

        # accept if better, otherwise with Metropolis probability
        if cur_val < best_score - 1e-14:
            cur = new
            best_score = cur_val
            best_seq = cur.copy()
            print(f"[{time.time():.1f}] anneal i={src}→j={dst} δ={delta:.3g} new best {best_score:.6f}")
        else:
            if temperature > 0 and rng.random() < math.exp(-(cur_val - best_score) / temperature):
                cur = new

        temperature *= decay

    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point used by the checker.
    Generates a diverse pool of seeds, refines the best ones with two‑phase
    Adam descent, and finishes with a cheap annealed swap walk.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths – longer vectors give a tighter bound
    lengths = [4000, 8000, 12000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    elite: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic / random families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length = {n} ---")
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep only the best few across all processed lengths
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]

    # ------------------------------------------------------------------
    # 3. Optionally include a previously discovered sequence
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:12]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback (should never be needed)
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation – two‑phase Adam descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0:
        # take the best elite seeds (up to 5)
        seeds = [seq for _, seq in sorted(elite, key=lambda x: x[0])[:5]]
        per_seed = max(0.5, remaining / (len(seeds) * 3.0))

        for seed_seq in seeds:
            if time.time() >= deadline:
                break

            # ---- Phase A: soft‑max Adam descent
            sub_deadline_a = time.time() + per_seed
            seq_a, score_a = _adam_softmax_descent(
                seed_seq.copy(),
                sub_deadline_a,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.4,
                max_iter=150000,
                eval_interval=400,
                alpha_start=0.5,
                alpha_end=80.0,
            )
            if score_a < overall_best_score:
                overall_best_score = score_a
                overall_best_seq = seq_a.copy()
                print(f"[{time.time():.1f}] after soft‑max new best {overall_best_score:.6f}")

            if time.time() >= deadline:
                break

            # ---- Phase B: hard‑max Adam descent
            sub_deadline_b = time.time() + per_seed
            seq_b, score_b = _adam_hardmax_descent(
                seq_a.copy(),
                sub_deadline_b,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.2,
                max_iter=150000,
                eval_interval=400,
            )
            if score_b < overall_best_score:
                overall_best_score = score_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after hard‑max new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final cheap polishing with simulated‑annealing swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _simulated_annealing_swaps(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # convert to plain Python list (evaluate_sequence expects a list)
    return overall_best_seq.tolist()
