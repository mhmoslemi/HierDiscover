def _fft_convolve(x, y):
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 2).bit_length())
    X = rfft(x, n=size)
    Y = rfft(y, n=size)

"""
Propose a step‑function using a continuation method.
A smoothed version of the maximum of the self‑convolution is
minimised by exponentiated‑gradient descent on the simplex.
The smoothing parameter α is increased gradually (annealing);
the update pushes mass away from positions that contribute most
to the large convolution entries.  After the descent a few cheap
pairwise‐swap and simulated‑annealing refinements are applied.
The routine respects the supplied time budget (default 1000 s) and
returns a plain Python list of heights.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    X = rfft(x, n=size)
    Y = rfft(y, n=size)
    conv = irfft(X * Y, n=size)
    return conv[:need]

def _objective(a: np.ndarray) -> float:
    """Exact objective for sum(a)=1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# ----------------------------------------------------------------------
# Analytic families (all return a vector that already sums to 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)

def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Lam (0‑1) is weight of the uniform component."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# ----------------------------------------------------------------------
# Smoothed‑max gradient (continuation) --------------------------------

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Returns (grad, max_conv) where
    grad_i = 2 * Σ_j a_j * softmax_α(conv)[i+j]
    and max_conv = max(conv).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum
    # gradient via convolution of w_norm with reversed a
    a_rev = a[::-1]
    conv2 = _fft_convolve(w_norm, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * conv2[start:start + n]
    return grad, max_c

def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 10_000_000,
    init_alpha: float = 1.0,
    max_alpha: float = 2000.0,
    alpha_factor: float = 1.025,
    eta: float = 0.2,
    eval_interval: int = 250,

"""
Propose a step‑function using a continuation (annealing) method.

The max of the self‑convolution is a non‑smooth objective.  We replace it
by a smooth soft‑max (log‑sum‑exp) with a temperature parameter α.
Starting from a few analytic seeds we run an exponentiated‑gradient
descent on the simplex, gradually increasing α (so the smoothed
objective approaches the true max).  The update pushes mass away from
positions that contribute most to the large convolution entries.
After the continuation stage we apply cheap pairwise‑swap and a short
simulated‑annealing local search.  The routine respects the supplied
time budget (default 1000 s) and returns a plain Python list of heights.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n, m = len(x), len(y)
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    X = rfft(x, n=size)
    Y = rfft(y, n=size)
    conv = irfft(X * Y, n=size)
    return conv[:need]

# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective for sum(a)=1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# ----------------------------------------------------------------------
# Analytic families (all already normalised to sum 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """return lam·U + (1‑lam)·base, U the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# ----------------------------------------------------------------------
# Smoothed‑max gradient (continuation)

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Returns (grad, max_conv) where
        grad_i = 2 * Σ_j a_j * softmax_α(conv)[i+j]
    and max_conv = max(conv) for the current a (which sums to 1).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                 # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (stable)
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum
    # gradient via convolution of w_norm with reversed a
    a_rev = a[::-1]
    conv2 = _fft_convolve(w_norm, a_rev)        # length 3n‑2
    start = n - 1
    grad = 2.0 * conv2[start:start + n]
    return grad, max_c

    a0: np.ndarray,
