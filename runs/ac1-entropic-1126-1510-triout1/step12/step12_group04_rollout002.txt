def _smooth_gradient_descent(a, deadline, best_seq, best_score, lr0=0.5, beta0=0.5, beta_max=20.0):
    t = 0
    beta = beta0
    n = a.shape[0]
    while time.time() < deadline:
        t += 1
        # compute conv
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv / (a.sum() ** 2)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            print(...)

        # compute smoothed max weights
        # compute beta schedule; can gradually increase
        beta = min(beta_max, beta * 1.001)  # slowly increase each iteration

        # compute softmax weights
        # avoid overflow: subtract max
        scaled = beta * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)   # stable
        w = exp_scaled / exp_scaled.sum()
        # compute gradient
        # grad = 2 * correlation(w, a)
        # use convolution with reversed w
        w_rev = w[::-1]
        # convolution linear
        grad = _fft_convolve(w_rev, a)  # returns length len(w_rev)+len(a)-1
        # extract gradient entries at indices (len(w_rev)-1 - np.arange(n))
        # len(w_rev) = 2*n -1
        idx_start = (2*n - 1) - 1  # 2n-2
        grad_vec = 2.0 * grad[idx_start - np.arange(n)]

        # exponentiated gradient update
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad_vec)

        # projection onto simplex
        # ensure sum >0
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
    return best_seq, best_score
```

def _fft_convolve(x, y):
    size = 1 << ((len(x) + len(y) - 1) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:len(x)+len(y)-1]
```

"""
Hybrid optimizer with a smoothed‑max gradient.

The algorithm consists of three phases:

1️⃣  **Rich seeding** – Generate many analytic families (uniform,
    arcsine with a fine grid of offsets, a power‑inverse‑sqrt family,
    two‑block shapes, sparse random, Dirichlet draws) for several
    lengths.  All candidates are normalised to a unit sum and the
    exact objective 2·n·max(conv(a,a))/sum(a)² is evaluated.  The best
    few (global) seeds are kept.

2️⃣  **Smoothed‑max exponentiated‑gradient descent** –  For each elite
    seed we minimise the maximum self‑convolution via a smooth
    approximation of the max:
        Sβ(a)= (1/β)·log Σi exp(β·(a∗a)_i) .
    The gradient of Sβ is a weighted sum of the individual
    convolution gradients, which we compute efficiently as a
    correlation by FFTs.  An exponentiated‑gradient step
        a←a·exp(‑η·∇Sβ)
    is performed, followed by a Euclidean projection onto the
    probability simplex.  The smoothing parameter β is slowly
    increased (β←min(β·1.001, β_max)) so that early iterations move
    mass away from many large peaks while later iterations focus on
    the true maximum.  The best iterate seen for each seed is kept.

3️⃣  **Greedy polishing** – Random mass‑transfer moves (δ‑fraction
    from one index to another) are accepted only if they improve the
    exact objective.  This removes tiny spikes left by the continuous
    optimisation.

The whole routine respects the supplied time budget (default 1000 s)
and returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y via FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families – richer parameter grids
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_power_inv_sqrt(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 1 - i) + 0.5
    vals = (left * right) ** (-p / 2.0)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = _distribution_uniform(n)
    else:
        vals /= total
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(
    n: int, rng: np.random.Generator, alpha: float = 0.5
) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse set of analytic seeds for length n."""
    cand = []

    cand.append(_distribution_uniform(n))

    # arcsine – geometric sweep of the offset
    for c in np.geomspace(0.1, 5.0, num=7):
        cand.append(_distribution_arcsine_offset(n, float(c)))

    # power‑inverse‑sqrt – linear sweep of the exponent
    for p in np.linspace(0.5, 2.0, num=9):
        cand.append(_distribution_power_inv_sqrt(n, float(p)))

    # two‑block – several block fractions
    for bf in (0.02, 0.05, 0.07, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # sparse random – several densities
    for density in (0.001, 0.003, 0.006, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=density))

    # Dirichlet draws with varied concentration
    for _ in range(4):
        alpha = 0.1 + 0.9 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    return cand


# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    beta0: float = 0.5,
    beta_max: float = 20.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed max of the
    self‑convolution.  Returns the best vector encountered.
    """
    n = a.shape[0]
    t = 0
    beta = beta0
    while time.time() < deadline:
        t += 1

        # exact objective (for bookkeeping)
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t} new best {best_score:.6f}")

        # ----- smoothed max -----
        #  w_i ∝ exp(beta * conv_i)
        scaled = beta * conv
        # stability shift
        max_s = np.max(scaled)
        exp_scaled = np.exp(scaled - max_s)
        w = exp_scaled / exp_scaled.sum()   # weights sum to 1

        # ----- gradient of the smoothed max -----
        # ∇Sβ = 2 * correlation(w, a)
        w_rev = w[::-1]
        corr = _fft_convolve(w_rev, a)                 # length = (2n-1)+n-1 = 3n-2
        # correlation index mapping: g_j = 2 * corr[ (2n-2) - j ]
        idx_start = 2 * n - 2
        grad_vec = 2.0 * corr[idx_start - np.arange(n)]

        # ----- exponentiated‑gradient step -----
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad_vec)

        # ----- simplex projection -----
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # slowly increase the smoothing parameter
        beta = min(beta_max, beta * 1.001)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Greedy random mass‑transfer moves that improve the exact objective."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30          # up to 30 % of source
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}→{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Entry point for the optimisation.  All work respects the total
    time budget and the best sequence found (as a list of floats) is
    returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # 1️⃣  Generate a pool of analytic seeds for several lengths
    lengths = [1800, 2500, 3200, 4000]          # trade‑off between quality and speed
    elite: list[tuple[float, np.ndarray]] = []  # (score, vector)
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length {n} ---")
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")

            elite.append((sc, cand.copy()))

        # keep only the top few across all processed lengths
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]

    # --------------------------------------------------------------
    # 2️⃣  If a previous solution is available, inject it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3️⃣  Smoothed‑max exponentiated‑gradient descent on the elite seeds
    if elite and time.time() < deadline:
        # allocate time roughly equally among the current elite seeds
        remaining = deadline - time.time()
        per_seed = max(5.0, remaining / len(elite))
        for _, seed_seq in elite[:5]:  # a few strongest seeds
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _smooth_max_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq if overall_best_seq is not None else seed_seq,
                best_score=overall_best_score,
                lr0=0.5,
                beta0=0.5,
                beta_max=20.0,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after smoothed descent new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final greedy polishing
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # --------------------------------------------------------------
    if overall_best_seq is None:
        # fallback – a simple uniform distribution (should never happen)
        n = 2000
        overall_best_seq = _distribution_uniform(n)

    # Ensure the returned list complies with the evaluator's expectations
    result = [float(x) for x in overall_best_seq]
    print(f"Final best score: {overall_best_score:.6f} (length={len(result)})")
    return result
