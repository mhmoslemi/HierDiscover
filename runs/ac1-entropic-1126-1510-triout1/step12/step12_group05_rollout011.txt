def _gradient_max_index_fast(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    # Compute overlapping region indices
    i_start = max(0, k - (n-1))
    i_end = min(n-1, k)
    # length of overlap L = i_end - i_start +1
    i = np.arange(i_start, i_end+1)
    j = k - i
    # now compute diag mask
    diag = i == j
    if np.any(diag):
        g[i[diag]] += 2.0 * a[i[diag]]
    # off-diagonal pairs: i!=j
    off_i = i[~diag]
    off_j = j[~diag]
    g[off_i] += a[off_j]
    g[off_j] += a[off_i]
    return g

"""
Mass‑transfer gradient descent for the self‑convolution max‑ratio problem.

The algorithm works directly on the discrete step heights a_0,…,a_{n‑1}
(kept on the probability simplex ∑a_i=1).  In each iteration the
index k where the self‑convolution a∗a attains its maximum is found.
The exact sub‑gradient of this maximal entry with respect to the
coefficients is computed (a simple O(n) routine).  Mass is then moved
from the index with the largest gradient contribution to the index
with the smallest one.  The convolution is updated by an O(n)
vectorised slice operation, i.e. without recomputing an FFT.  The step
size decays as 1/√t, guaranteeing convergence and allowing many
iterations within the 1000 s budget.  Several analytic seeds (uniform,
arcsine, Dirichlet, sparse, two‑block) are tried for a few problem
sizes; the best sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (∑a is assumed = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry conv[k].
    Returns a non‑negative vector of the same length as a.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq = (i == j)
    if np.any(eq):
        g[i[eq]] += 2.0 * a[i[eq]]

    if np.any(~eq):
        ii = i[~eq]
        jj = j[~eq]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g


# ----------------------------------------------------------------------
# Analytic families used as seeds ---------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.3) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.015) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _mass_transfer_descent(a: np.ndarray,
                           deadline: float,
                           max_iters: int = 200_000,
                           min_step: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Direct mass‑transfer / sub‑gradient descent.
    The convolution is updated by O(n) slice operations, which makes the
    inner loop extremely cheap compared to an FFT.
    """
    n = a.shape[0]
    # initial convolution
    c = _fft_self_convolve(a)
    cur_score = 2.0 * n * c.max()
    best_local_seq = a.copy()
    best_local_score = cur_score

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1

        # maximal entry of the current convolution
        k = int(c.argmax())
        g = _gradient_max_index(a, k)

        # choose source (large gradient) and destination (small gradient)
        src_candidates = np.where(a > 1e-13)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]

        dest = np.argmin(g)
        if dest == src:
            # second smallest gradient if the first coincides with src
            order = np.argsort(g)
            for d in order:
                if d != src:
                    dest = d
                    break

        # step size – decaying fraction of the source mass
        frac = 0.3 / (1.0 + math.sqrt(t))   # roughly 0.3·t^{-1/2}
        delta = min(frac * a[src], a[src])
        if delta < min_step:
            continue

        # ---- compute tentative new state --------------------------------
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta

        c_new = c.copy()
        # contribution of the source (negative) and destination (positive)
        c_new[src:src + n] -= 2.0 * delta * a
        c_new[dest:dest + n] += 2.0 * delta * a
        # quadratic delta^2 corrections
        c_new[2 * src] += delta * delta
        c_new[2 * dest] += delta * delta
        c_new[src + dest] -= 2.0 * delta * delta

        new_score = 2.0 * n * c_new.max()

        # ---- accept only if the true objective does not increase ---------
        if new_score <= cur_score + 1e-15:
            a = a_new
            c = c_new
            cur_score = new_score
            if new_score < best_local_score - 1e-14:
                best_local_score = new_score
                best_local_seq = a.copy()
        # otherwise reject the move (keep current a, c)

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Entry point for the optimiser.  A handful of analytic seeds are
    generated for a few problem sizes, each one is refined with the
    mass‑transfer descent described above, and the globally best
    sequence is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. problem sizes (larger n usually yields a lower bound)
    lengths = [8000, 20000, 50000]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] === exploring length {n} ===")

        # ----------------------------------------------------------------
        # 2. generate a small diverse pool of seeds for this length
        seeds = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_two_blocks(n, block_frac=0.07),
            _distribution_dirichlet(n, rng, alpha=0.3),
            _distribution_random_sparse(n, rng, density=0.015),
        ]

        # optional: start from the previously submitted sequence if available
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                if prev.sum() == 0.0:
                    prev = _distribution_uniform(prev.shape[0])
                prev = prev / prev.sum()
                if prev.shape[0] == n:
                    seeds.append(prev)
        except Exception:
            pass

        # ----------------------------------------------------------------
        # 3. loop over seeds, improve each and keep the best
        for idx, a in enumerate(seeds):
            if time.time() >= deadline:
                break

            a = np.maximum(a, 0.0)
            a = a / a.sum()                           # enforce simplex

            # evaluate the raw seed
            score = _objective(a)
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best (seed) len={n} score={score:.6f}")

            # ------------------------------------------------------------
            # mass‑transfer descent
            per_seed_deadline = min(deadline, time.time() + 30.0)   # at most 30 s per seed
            a_opt, score_opt = _mass_transfer_descent(
                a,
                per_seed_deadline,
                max_iters=200_000,
            )
            if score_opt < overall_best_score:
                overall_best_score = score_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] improved after descent len={n} score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 4. final sanity check – ensure the sequence is non‑negative and summed to 1
    if overall_best_seq is None:
        # fallback – a uniform distribution of modest size
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
