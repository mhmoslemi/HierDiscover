def _distribution_power_left_right(n, p_left, p_right):
    i = np.arange(n, dtype=np.float64)
    left = (i + 0.5) ** (-p_left)  # or if p_left==0 => 1
    right = (n - i - 0.5) ** (-p_right)
    vals = left * right
    total = vals.sum()
    if total == 0:
        vals = np.full_like(vals, 1.0/n)
    else:
        vals /= total
    return vals
```

"""
Hybrid random‑search + Adam optimiser for the step‑function “max‑convolution”
problem.

The routine works in three phases:

1.  **Broad stochastic seeding** –  a large collection of analytical families
    (uniform, arcsine, asymmetric Beta, left/right power‑law, exponential,
    etc.) is sampled for several lengths.  Every candidate is evaluated with the
    exact objective (FFT based) and the best few are kept.

2.  **Adam smooth‑max refinement** – each kept seed is optimised on the
    smooth‑max surrogate using a soft‑max (log‑parameter) representation.
    The temperature α is increased geometrically up to a very large value,
    thus the surrogate converges to the true max.  Whenever the exact
    objective improves the best solution is updated.

3.  **Polishing** – a short exponentiated‑gradient multimax descent followed
    by a cheap greedy pairwise‑swap walk extracts the final tiny improvements.

All vectors are kept on the probability simplex (sum = 1, a_i ≥ 0); the
objective is scale‑invariant, so this incurs no loss of optimality.
The function respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import math
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·conv_k(a)   w.r.t. a.
    Here w is a probability vector on the convolution indices.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (assumes sum(a)>0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Beta(alpha=beta=p) – symmetric variant."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_left_right(n: int, p_left: float, p_right: float) -> np.ndarray:
    """Weight ∝ (i+½)^(-p_left)·(n‑i‑½)^(-p_right)."""
    i = np.arange(n, dtype=np.float64)
    left = np.power(i + 0.5, -p_left, where=(i + 0.5) > 0, out=np.ones_like(i))
    right = np.power(n - i - 0.5, -p_right, where=(n - i - 0.5) > 0, out=np.ones_like(i))
    vals = left * right
    if vals.sum() == 0.0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_exponential_decreasing(n: int, lam: float = 0.005) -> np.ndarray:
    """Discrete exponential decay a_i ∝ exp(-λ·i)."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i)
    if vals.sum() == 0.0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s

def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 5000.0,
                   growth_factor: float = 1.05,
                   lr: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 200000,
                   eval_interval: int = 250) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best sequence (on the simplex) seen before the deadline.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        a = _softmax(z)                     # a_i ≥ 0, Σ a_i = 1
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad_a = _gradient_from_weights(a, w)          # ∂/∂a Σ w·conv
        dot = np.dot(a, grad_a)                        # softmax Jacobian term
        grad_z = a * (grad_a - dot)                    # chain rule

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 30000,
                                 eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution objective.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk; only accept improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best list found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0       # small safety margin

    # ------------------------------------------------------------------
    # Lengths to explore – moderate sizes keep each iteration cheap
    lengths = [2000, 3000, 4000, 5000]

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    all_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Stochastic generation of seeds
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # deterministic baselines
        for name, fun in [
            ("uniform", _distribution_uniform),
            ("arcsine", _distribution_arcsine),
        ]:
            if time.time() >= deadline:
                break
            cand = fun(n)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] **** new global best ({name}) score={sc:.6f}")

        # random asymmetric Beta draws
        n_beta = 250
        for _ in range(n_beta):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 2)   # 0.01 … 100
            beta = 10 ** rng.uniform(-2, 2)
            cand = _distribution_beta_asym(n, alpha, beta)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best (beta) {sc:.6f}")

        # random left‑right power‑law families
        n_pow = 250
        for _ in range(n_pow):
            if time.time() >= deadline:
                break
            p_left = rng.uniform(0.1, 3.0)
            p_right = rng.uniform(0.1, 3.0)
            cand = _distribution_power_left_right(n, p_left, p_right)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best (powLR) {sc:.6f}")

        # random decreasing exponential
        n_exp = 250
        for _ in range(n_exp):
            if time.time() >= deadline:
                break
            lam = rng.uniform(1e-4, 0.05)
            cand = _distribution_exponential_decreasing(n, lam)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best (exp) {sc:.6f}")

    # ------------------------------------------------------------------
    # Keep only the most promising seeds overall
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:12]]
    if overall_best_seq is None:
        # fallback – uniform of the first length
        n = lengths[0]
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)

    # ------------------------------------------------------------------
    # 2. Adam refinement on the selected seeds
    for seed_seq in top_seeds:
        if time.time() >= deadline:
            break
        slice_time = min(40.0, deadline - time.time())
        if slice_time <= 5.0:
            break
        sub_deadline = time.time() + slice_time

        z0 = np.log(np.maximum(seed_seq, 1e-300))

        seq_opt, score_opt = _adam_optimize(
            z0,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            alpha0=1.0,
            alpha_max=5000.0,
            growth_factor=1.05,
            lr=0.12,
            max_iter=200000,
            eval_interval=250,
        )
        if score_opt < overall_best_score:
            overall_best_score, overall_best_seq = score_opt, seq_opt.copy()
            print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Global polishing – multimax sub‑gradient descent
    if time.time() < deadline - 5.0:
        sub_deadline = time.time() + (deadline - time.time()) * 0.5
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            max_iter=50000,
            eval_interval=300,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final greedy pairwise‑swap walk
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
