"""
A fast Adam‑based optimizer that directly smooths the “max‑convolution”
objective.
The algorithm works in three stages:

1.  Build a modest pool of analytic and random seeds (uniform, arcsine,
    beta, edge‑weighted, power‑law, Hann, random Dirichlet, sparse …)
    for a few lengths (≈2k‒8k).  Keep the ≈15 best seeds.

2.  For each seed run a *smooth‑max* Adam optimisation.
    The true objective 2·n·max(conv)/sum² is approximated by the
    log‑sum‑exp of the convolution (parameter β).  β is increased
    gradually (annealing) so the optimizer first sees a smooth landscape
    and finally focuses on the true maximum.
    The probability vector is parametrised by soft‑max of a free vector
    θ, guaranteeing non‑negativity and unit sum at every step.

3.  A short pair‑wise mass‑transfer simulated‑annealing polish is run on
    the current best candidate.

All work respects the 1000 s budget (a 2 s safety margin) and returns a
plain ``list[float]`` ready for the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y using a real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (sum is assumed = 1).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    if a.sum() < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Analytic families – all return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_sparse_random(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Very sparse random vector (non‑zero entries ≈ density·n)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    return lam * np.full(n, 1.0 / n, dtype=np.float64) + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max helpers
# ----------------------------------------------------------------------
def _softmax(theta: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    max_t = np.max(theta)
    e = np.exp(theta - max_t)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) value, gradient w.r.t. a and true max for logging
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, beta: float):
    """
    Returns
        grad_a : ∂/∂a ( log‑sum‑exp_β (a*a) )
        smooth_max :   (1/β)·log∑exp(β·conv)
        true_max   :   max(conv)
    """
    conv = _fft_self_convolve(a)                     # length 2n-1
    maxc = conv.max()
    # stable soft‑max over the convolution entries
    shifted = beta * (conv - maxc)
    w = np.exp(shifted)
    w_sum = w.sum()
    w_norm = w / w_sum                               # ∂ smooth_max / ∂conv

    # gradient w.r.t. a:
    #   ∂/∂a_i  smooth_max = Σ_k w_norm[k]·∂conv_k/∂a_i  = 2·Σ_j w_norm[i+j]·a_j
    w_rev = w_norm[::-1]
    conv_wr_a = _fft_convolve(w_rev, a)              # length 3n‑2
    offset = len(conv) - 1                           # = 2n‑2
    grad_a = 2.0 * conv_wr_a[offset - np.arange(a.shape[0])]

    smooth_max = maxc + np.log(w_sum) / beta
    return grad_a, smooth_max, maxc


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    beta_start: float = 0.5,
    beta_max: float = 200.0,
    beta_ramp_iters: int = 2000,
    lr: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate.
    Returns the best sequence (probability vector) seen before `deadline`.
    """
    n = a0.shape[0]
    theta = np.log(np.clip(a0, 1e-16, None))          # initialise θ = log a
    m = np.zeros_like(theta)
    v = np.zeros_like(theta)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    pow1, pow2 = 1.0, 1.0

    while time.time() < deadline:
        it += 1

        # ----- β schedule (linear ramp) -----
        if it <= beta_ramp_iters:
            beta = beta_start + (beta_max - beta_start) * it / beta_ramp_iters
        else:
            beta = beta_max

        # ----- forward pass -----
        a = _softmax(theta)                           # stays on the simplex
        grad_a, smooth_val, true_max = _smooth_max_grad(a, beta)

        # ----- back‑prop to θ (soft‑max Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_theta = a * (grad_a - dot)

        # ----- Adam update -----
        pow1 *= beta1
        pow2 *= beta2
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)
        m_hat = m / (1.0 - pow1)
        v_hat = v / (1.0 - pow2)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional exact evaluation -----
        if it % 50 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] Adam it={it:>5d} β={beta:5.1f} "
                    f"smooth={smooth_val:.6f} max={true_max:.6f} → best {best_score:.6f}"
                )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple pair‑wise mass‑transfer annealing (as in the previous baseline)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # pick a source with non‑trivial mass
        src = rng.choice(np.where(a > 1e-12)[0])
        # pick a distinct destination
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move a random fraction (up to 40 %) of source mass
        delta = a[src] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal src={src}→dst={dst} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that (almost) minimises
    ``evaluate_sequence``.  The routine respects the 1000 s total budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Seed generation (several analytic families + random draws)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]          # moderate sizes – keep FFT cheap
    p_grid = np.concatenate(
        [
            np.logspace(-2.0, -0.5, 8, base=10),
            np.logspace(-0.5, 0.0, 8, base=10),
            np.logspace(0.0, 0.7, 6, base=10),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)   # mixing with uniform
    power_grid = np.linspace(0.2, 2.0, 7)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_hann(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # beta‑sym and edge‑weighted families, mixed with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}"
                        )

        # power‑law family
        for p in power_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, p)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best power p={p:.2f} len={n} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_sparse_random(n, rng, density=0.015)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the very best seeds (≈15)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # 2.  If a previous construction is present, include it.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev[:] = 1.0 / prev.size
            else:
                prev /= prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – guarantee a non‑empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam optimisation on the most promising seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # sort by current score, keep the top few for a longer run
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] Adam optimisation on {len(seed_seqs)} seeds (≈{per_seed:.1f}s each)")

        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta_start=0.5,
                beta_max=150.0,
                beta_ramp_iters=1500,
                lr=0.4,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] seed {idx} improved global best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Optional up‑sampling (larger n) if time permits.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] up‑sampling {cur_n} → {target_n}")
            # linear interpolation, then renormalise
            x_old = (np.arange(cur_n, dtype=np.float64) + 0.5) / cur_n
            x_new = (np.arange(target_n, dtype=np.float64) + 0.5) / target_n
            up = np.interp(x_new, x_old, overall_best_seq, left=0.0, right=0.0)
            up = np.maximum(up, 0.0)
            if up.sum() == 0.0:
                up[:] = 1.0 / target_n
            else:
                up /= up.sum()
            up_score = _objective(up)
            if up_score < overall_best_score:
                overall_best_score, overall_best_seq = up_score, up
                print(f"[{time.time():.1f}] up‑sampled new best {overall_best_score:.6f}")

            # a brief Adam pass on the up‑sampled vector
            sub_deadline = time.time() + (deadline - time.time()) * 0.3
            a_opt, sc_opt = _adam_optimize(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta_start=0.5,
                beta_max=120.0,
                beta_ramp_iters=1200,
                lr=0.35,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt
                print(f"[{time.time():.1f}] post‑upsample Adam improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final long‑run pairwise annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)/sum²): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # return plain Python list (the caller will feed it to evaluate_sequence)
    return overall_best_seq.tolist()
