#!/usr/bin/env python
"""
Hybrid optimiser for the max‑convolution problem.

* Generates a rich seed set (uniform, arcsine, many Beta‑families,
  Dirichlet draws, sparse patterns and the previous best if present).
* Starts from the smallest length, finds a good shape and *upsamples*
  it to larger lengths, refining each size separately.
* Uses three optimisation phases:
    1. Smooth‑max Adam (soft‑max approximation of the max, slowly increasing
       the temperature to focus on the true max).
    2. Classic multimax sub‑gradient descent for fine tuning.
    3. Simple stochastic hill‑climbing (Gaussian perturbations + simplex
       projection) to escape possible local minima.
* Ends with a cheap greedy pair‑wise mass‑transfer polishing step.
* All updates stay on the probability simplex (∑a = 1, a ≥ 0).
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, ∑x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of 〈w, a⋆a〉 w.r.t. a.
    w must have length 2·n‑1 and ∑w = 1.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/∑a² (∑a = 1 → 2·n·max(conv))."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_custom_triangular(n: int) -> np.ndarray:
    """Linearly increasing up to the centre then decreasing (area normalized)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = np.where(x <= 0.5, 4.0 * x, 4.0 * (1.0 - x))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates_extended_fine(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """
    Produce a diverse list of seed vectors for length n.
    """
    cand = []

    # basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric Beta – log‑spaced exponents
    for p in np.geomspace(0.01, 5.0, num=20):
        cand.append(_distribution_beta_sym(n, float(p)))

    # random asymmetric Beta draws
    for _ in range(12):
        alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power‑law increasing
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential families
    for lam in (0.001, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))

    # two‑block spikes
    for bf in (0.03, 0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Dirichlet draws
    for a in (0.1, 0.3, 0.7, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.03))

    # custom triangular (smooth, centred)
    cand.append(_distribution_custom_triangular(n))

    # few completely random probability vectors
    for _ in range(8):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    return cand

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.7,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.2,
                             alpha_mul: float = 1.3,
                             alpha_max: float = 800.0,
                             max_iter: int = 120000,
                             eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smooth‑max surrogate
    f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)).
    α is increased geometrically, yielding a tighter approximation of the true max.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # full self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update (mirror‑descent on the simplex)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # re‑project onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule
        if t % 350 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.7,
                                 max_iter: int = 90000,
                                 eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient descent where the sub‑gradient is the
    uniform average over all indices attaining the maximal convolution value.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pair‑wise mass transfer: move a small amount from i to j if it improves the objective."""
    n = a.shape[0]
    rng = np.random.default_rng(12345)
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = int(rng.choice(src))

        # pick a different destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # random amount up to 30 % of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_hill_climb(a: np.ndarray,
                       deadline: float,
                       best_score: float,
                       step_ini: float = 0.05) -> Tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: Gaussian perturbations followed by projection onto the simplex.
    The step size decays slowly to allow fine‑grained exploration.
    """
    n = a.shape[0]
    rng = np.random.default_rng(987654)
    sigma = step_ini
    while time.time() < deadline:
        cand = a + sigma * rng.normal(size=n)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        cur = _objective(cand)
        if cur < best_score - 1e-13:
            best_score = cur
            a = cand
            print(f"[{time.time():.1f}] hill‑climb new best {best_score:.6f}")
        # exponential decay of the perturbation magnitude
        sigma *= 0.9999
        if sigma < 1e-6:
            sigma = step_ini * 0.1
    return a, best_score


# ----------------------------------------------------------------------
def _upsample(a_old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation (preserving total mass) from length len(a_old) → n_new."""
    n_old = a_old.shape[0]
    x_old = (np.arange(n_old, dtype=np.float64) + 0.5) / n_old
    x_new = (np.arange(n_new, dtype=np.float64) + 0.5) / n_new
    a_new = np.interp(x_new, x_old, a_old, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new = _distribution_uniform(n_new)
    else:
        a_new /= a_new.sum()
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects ``budget_s`` seconds and returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1. Length schedule – start small, finish with larger sizes.
    lengths = [4000, 8000, 12000]
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Scan seeds (cheap analytic families) for the first length.
    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] ---------- Length {n} ----------")

        # ------------------------------------------------------------------
        #   a) Gather seeds
        if idx == 0:
            seed_list = _initial_candidates_extended_fine(n, rng)
        else:
            # Upsample the current global best to the new length and add a few fresh random seeds.
            if overall_best_seq is not None:
                up = _upsample(overall_best_seq, n)
                seed_list = [up]
            else:
                seed_list = [_distribution_uniform(n)]

            # some extra randomness for robustness
            for _ in range(6):
                seed_list.append(_distribution_dirichlet(n, rng, alpha=0.2))
            for _ in range(4):
                seed_list.append(_distribution_random_sparse(n, rng, density=0.015))
            seed_list.append(_distribution_arcsine(n))
            seed_list.append(_distribution_uniform(n))

        # ------------------------------------------------------------------
        #   b) Quick evaluation of all seeds, keep top few
        top_local: List[Tuple[float, np.ndarray]] = []
        for cand in seed_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_local.append((sc, cand.copy()))

        # keep only a small elite set for deep optimisation
        top_local.sort(key=lambda x: x[0])
        elite = [seq for _, seq in top_local[:6]]

        # ------------------------------------------------------------------
        #   c) Deep optimisation of the elite seeds
        # allocate remaining time proportionally to remaining lengths
        remaining_lengths = len(lengths) - idx
        time_per_len = max(0.0, (deadline - time.time()) / remaining_lengths)

        # Phase 1 – smooth‑max Adam
        for seed_seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + time_per_len * 0.55
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                max_iter=120000,
                eval_interval=500,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

        # Phase 2 – multimax sub‑gradient descent
        for seed_seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + time_per_len * 0.30
            seq_out, score_out = _subgradient_multimax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                max_iter=90000,
                eval_interval=500,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

        # Phase 3 – stochastic hill‑climbing (if time permits)
        remaining = deadline - time.time()
        if remaining > 5.0:
            seq_out, score_out = _random_hill_climb(
                overall_best_seq.copy(),
                deadline,
                overall_best_score,
                step_ini=0.07,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after hill‑climb new best {overall_best_score:.6f}")

        # Phase 4 – cheap greedy pairwise swaps (final polishing for this length)
        if time.time() < deadline:
            overall_best_seq, overall_best_score = _pairwise_swap_local(
                overall_best_seq.copy(),
                deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            print(f"[{time.time():.1f}] after pairwise best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Incorporate previously discovered solution if present.
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev_arr = np.asarray(prev, dtype=np.float64).ravel()
            if prev_arr.size > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                if prev_arr.sum() == 0.0:
                    prev_arr = _distribution_uniform(prev_arr.size)
                prev_arr /= prev_arr.sum()
                sc = _objective(prev_arr)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev_arr.copy()
                    print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Final safety check
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(4000)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
