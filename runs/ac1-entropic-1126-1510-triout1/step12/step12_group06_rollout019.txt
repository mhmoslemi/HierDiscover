def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5

    # Choose length(s). We'll try n=4000 or n=5000. Possibly also n=8000 if time.
    lengths = [4000, 5000, 8000]  # slower for larger

    best_seq = None
    best_score = float('inf')
    top = [] # list of tuples (score, seq)
    # Phase 1: random seeds generation
    while time.time() < deadline and len(top) < 80: # generate many
        n = rng.choice(lengths)
        # Choose random distribution type
        typ = rng.choice(['dirichlet','exponential','power','uniform'])
        if typ == 'uniform':
            seq = _distribution_uniform(n)
        elif typ == 'dirichlet':
            alpha = rng.choice([0.1,0.3,0.7,1.0,2.0])
            seq = _distribution_dirichlet(n, rng, alpha)
        elif typ == 'exponential':
            lam = 10**rng.uniform(-3,-1)  # lam in [0.001,0.1]
            seq = _distribution_exponential_increasing(n, lam)
        elif typ == 'power':
            p = rng.choice([0.5,1.0,2.0,3.0])
            seq = _distribution_power_increasing(n, p)
        # Optional random permutation
        if rng.random() < 0.3:
            rng.shuffle(seq)
        # Normalize sum to 1
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New random best n={n} score={sc:.6f}")
        top.append((sc, seq.copy()))
        # Keep only best few
        top.sort(key=lambda x: x[0])
        top = top[:10]  # keep 10 best seeds

"""
Hybrid optimiser for the step‑function max‑convolution problem.

The strategy is a wider‑range, time‑aware search that builds on the
previous hybrid optimiser:

1.  **Broad seeding** – generate many low‑dimensional candidates from a
    rich collection of analytic families (uniform, Dirichlet, power,
    exponential, random sparse, two‑block spikes, …) and also from the
    historic best solution if it exists.

2.  **Long Adam descent** – run exponentiated‑gradient descent on the
    smooth‑max surrogate with many more iterations and a slower
    temperature schedule than the reference implementation.
    The update keeps the vector on the probability simplex.

3.  **Multimax sub‑gradient descent** – classic exponentiated‑gradient
    on the true max‑convolution objective, starting from the best
    Adam‑refined seeds.

4.  **Linear‑programming polishing** – for a medium‑size problem
    (n≈800) we solve the LP described in the paper:
        maximise   Σ b_j
        subject to  (f * b)[k] ≤ M   ∀k ,  b_j ≥ 0
    where M is the current maximal convolution value.
    The solution enlarges the ℓ₁‑mass while preserving the bound;
    a small convex combination with the current iterate strictly
    reduces the objective.

5.  **Greedy pairwise swaps** – cheap local mass‑transfer moves that
    further reduce the exact objective.

All stages respect the global 1000 s budget and periodically log the
best score found.  The returned list of floats can be fed directly to
``evaluate_sequence``.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate:
        ∇  (∑_k w_k (a⋆a)_k) = 2·a⋆w
    w has length 2·n‑1 and Σ w = 1.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The vector is assumed to be non‑negative; the sum is >0.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.6,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.3,
                             alpha_mul: float = 1.1,
                             alpha_max: float = 200.0,
                             max_iter: int = 150000,
                             eval_interval: int = 1000) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max
    surrogate  f_α(a)= (1/α) log Σ exp(α·conv_k(a)).
    Temperature α increases slowly; the update is multiplicative and
    projected onto the simplex.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)

        # soft‑max (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # projection onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule
        if t % 400 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 70000,
                                 eval_interval: int = 800) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with the true sub‑gradient:
    the uniform average over all indices attaining the maximal
    convolution value.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pair‑wise mass transfer."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer amount (up to 30 % of the source mass)
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def _build_A_ub(f: np.ndarray) -> sp.csr_matrix:
    """Build the sparse matrix A_ub for the LP  (f * b)[k] ≤ M."""
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for i, fi in enumerate(f):
        if fi == 0.0:
            continue
        # column i contributes with coefficient fi to rows i .. i+n-1
        rows.append(np.arange(i, i + n, dtype=np.int32))
        cols.append(np.full(n, i, dtype=np.int32))
        data.append(np.full(n, fi, dtype=np.float64))
    rows = np.concatenate(rows) if rows else np.array([], dtype=np.int32)
    cols = np.concatenate(cols) if cols else np.array([], dtype=np.int32)
    data = np.concatenate(data) if data else np.array([], dtype=np.float64)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_refine(a: np.ndarray,
               deadline: float,
               mix_factor: float = 0.15,
               max_iters: int = 3) -> Tuple[np.ndarray, float]:
    """
    Run a few LP‑based improvements as described in the paper.
    Each iteration solves:
        maximise   Σ b_j
        subject to (a * b)[k] ≤ M   and   b ≥ 0
    where M is the current maximal self‑convolution value.
    The solution is mixed back with the current vector.
    """
    cur_score = _objective(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        M = conv.max()

        A = _build_A_ub(a)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σ b_j

        res = opt.linprog(c,
                         A_ub=A,
                         b_ub=b_ub,
                         bounds=[(0.0, None)] * n,
                         method='highs',
                         options={'time_limit': max(5, int(deadline - time.time()))})

        if not res.success:
            # LP failed – abort further refinements
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= a.sum() + 1e-12:
            # No mass increase ⇒ no improvement possible
            break

        # Normalise to a probability vector
        g = b / sum_b
        a = (1 - mix_factor) * a + mix_factor * g
        new_score = _objective(a)
        if new_score < cur_score - 1e-15:
            cur_score = new_score
            print(f"[{time.time():.1f}] LP refine iter={it+1} new best {cur_score:.6f}")

    return a, cur_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest set of lengths; longer vectors give more freedom
    #    but each iteration is more expensive.
    lengths = [3500, 4000, 4500]
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Broad random seeding (cheap, many samples)
    print(f"[{time.time():.1f}] Starting random seeding phase")
    while time.time() < deadline and len(top_candidates) < 120:
        n = int(rng.choice(lengths))
        cand_list = _initial_candidates_fine(n, rng)

        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the very best seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # 3. Incorporate the historic solution if present.
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev = np.asarray(prev, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                s = prev.sum()
                if s == 0.0:
                    prev = _distribution_uniform(prev.shape[0])
                    s = prev.sum()
                prev /= s
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – never needed but keeps the code safe.
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. First refinement: Adam‑style smooth‑max descent on a selection
    #    of promising seeds.
    remaining = deadline - time.time()
    if remaining > 80:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:4]
        ]
        sub_deadline_total = deadline - time.time()
        per_seed = sub_deadline_total / max(1, len(seeds))
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.5,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.25,
                alpha_mul=1.1,
                alpha_max=180.0,
                max_iter=150000,
                eval_interval=800,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second refinement: classic multimax sub‑gradient descent.
    remaining = deadline - time.time()
    if remaining > 50:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:3]
            if not np.may_share_memory(seq, overall_best_seq)
        ]
        per_seed = (deadline - time.time()) / max(1, len(seeds))
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _subgradient_multimax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=70000,
                eval_interval=500,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. LP‑based polishing (run on a medium‑size vector for speed).
    #    If the current length is >800 we temporarily down‑sample,
    #    polish, and then up‑sample back.
    remaining = deadline - time.time()
    if remaining > 30:
        n_lp = min(overall_best_seq.shape[0], 800)
        if n_lp < overall_best_seq.shape[0]:
            # down‑sample by simple averaging of blocks
            factor = overall_best_seq.shape[0] // n_lp
            a_small = overall_best_seq[: factor * n_lp].reshape(n_lp, factor).mean(axis=1)
        else:
            a_small = overall_best_seq.copy()
        a_small, score_small = _lp_refine(a_small, deadline, mix_factor=0.12, max_iters=3)
        if a_small.shape[0] == n_lp and n_lp < overall_best_seq.shape[0]:
            # up‑sample back (nearest‑neighbor)
            factor = overall_best_seq.shape[0] // n_lp
            a_up = np.repeat(a_small, factor)
            # truncate / pad to original length
            a_up = a_up[: overall_best_seq.shape[0]]
            a_up = np.maximum(a_up, 0.0)
            a_up /= a_up.sum()
            a_up_score = _objective(a_up)
            if a_up_score < overall_best_score:
                overall_best_seq = a_up
                overall_best_score = a_up_score
                print(f"[{time.time():.1f}] LP‑up‑sampled improvement {overall_best_score:.6f}")
        else:
            if score_small < overall_best_score:
                # same size – accept directly
                a_small = np.maximum(a_small, 0.0)
                a_small /= a_small.sum()
                overall_best_seq = a_small
                overall_best_score = score_small
                print(f"[{time.time():.1f}] LP refinement improvement {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final cheap greedy polishing with pairwise swaps.
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
