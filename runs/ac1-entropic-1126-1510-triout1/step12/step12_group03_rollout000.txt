def _distribution_triangle(n):
    i = np.arange(n)
    mid = (n-1)/2
    w = 1 - np.abs(i - mid) / (mid+1)  # like linearly decreasing from centre to zero at edges.
    w = np.clip(w,0,None)
    w /= w.sum()
    return w

"""
Propose a non‑negative step function (vector) that minimizes
evaluate_sequence().  The algorithm works in three phases:

1. **Massive analytic scan** – generate many candidate shapes for a range
   of lengths (up to 120 000) and a dense grid of shape parameters.
   Families include uniform, arcsine, symmetric beta, edge‑weighted,
   quadratic, triangular and a few random Dirichlet / sparse seeds.
   All candidates are normalized to sum‑to‑1, their exact objective
   2·n·max(conv) is evaluated with an FFT, and the best ~30 seeds are kept.

2. **Mirror‑descent refinement** – on the top seeds we run a projected
   exponentiated‑gradient (mirror) descent on the true L∞‑convolution
   objective.  The gradient uses the `top_k` largest convolution entries,
   a decaying learning rate, and Euclidean projection onto the probability
   simplex.

3. **Greedy pairwise polishing** – a simple hill‑climbing that repeatedly
   transfers a random fraction of mass between two positions and keeps
   any improvement.

The whole routine respects the 1000 s budget, prints progress, and returns
a plain Python list suitable for evaluate_sequence().
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    Input `a` must be non‑negative and already normalised.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid {0,…,n‑1} (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ ((i+½)*(n‑i‑½))^{‑p}, heavy at the edges for p>0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    """Parabolic shape peaking in the middle (non‑negative)."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_triangle(n: int) -> np.ndarray:
    """Symmetric triangular hat: 1 at centre, linearly to 0 at edges."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = 1.0 - np.abs(i - mid) / (mid + 1.0)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    top_k: int = 10,
    eval_every: int = 150,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on the exact max‑convolution
    objective using the `top_k` largest convolution entries.
    Returns the best sequence ever seen.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        # indices of the `top_k` largest entries (including centre)
        idxs = np.argpartition(-conv, range(top_k))[:top_k]

        grad = np.zeros_like(a)
        for idx in idxs:
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)          # decaying step size
        a = a * np.exp(-eta * grad)       # EG update (keeps positivity)

        # project back onto the simplex
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing: move a random fraction of mass from one index to another.
    Keep the move only if the exact objective strictly improves.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes
    evaluate_sequence().  The method combines a very broad analytic scan,
    mirror‑descent refinement of the most promising seeds and a final
    greedy polishing.  It respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    best_seq = None
    best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)
    MAX_TOP = 30

    def _maybe_update(seq: np.ndarray, sc: float):
        """Update global best and the top‑k list."""
        nonlocal best_seq, best_score, top_candidates
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST score={sc:.6f} len={len(seq)}")
        # keep a bounded list of good seeds
        top_candidates.append((sc, seq.copy()))
        if len(top_candidates) > MAX_TOP:
            top_candidates.sort(key=lambda x: x[0])
            top_candidates = top_candidates[:MAX_TOP]

    # ------------------------------------------------------------------
    # 1.  Large‑scale analytic family scan
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 12000, 20000, 30000, 50000, 80000, 120000]

    # parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 12),   # 0.001 … 0.316 edge‑heavy
            np.linspace(0.5, 2.0, 12),   # moderate regime
            np.linspace(2.0, 6.0, 12),   # centre‑heavy
        ]
    )
    lam_grid = np.linspace(0.0, 0.5, 6)   # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ----- deterministic basic families -----
        for fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_quadratic,
            _distribution_triangle,
        ):
            cand = fn(n)
            sc = _objective(cand)
            _maybe_update(cand, sc)

        # ----- parametric families (beta / edge‑weighted) -----
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta_seq = _distribution_beta_sym(n, p)
            edge_seq = _distribution_edge_weighted(n, p)
            for base in (beta_seq, edge_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    _maybe_update(cand, sc)

        # ----- random Dirichlet seeds -----
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            _maybe_update(a, sc)

        # ----- sparse random seeds -----
        for _ in range(3):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            _maybe_update(a, sc)

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously discovered construction (if present)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            _maybe_update(prev, sc)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Mirror‑descent refinement on the very best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # select the 5 most promising seeds for deeper work
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start subgradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent(
                seed_seq,
                sg_deadline,
                best_seq=best_seq,
                best_score=best_score,
                lr0=0.9,
                top_k=10,
                eval_every=150,
            )
            if sc_opt < best_score - 1e-12:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {idx} new global best {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 4.  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        if sc_opt < best_score - 1e-12:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"FINAL BEST score (2·n·max(conv)): {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list – evaluate_sequence will perform its own normalisation.
    return best_seq.tolist()
