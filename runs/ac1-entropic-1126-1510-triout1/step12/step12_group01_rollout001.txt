def _subgradient_descent(a, deadline, best_seq, best_score):
    n = len(a)
    t=0
    while time.time()<deadline and t < max_iter:
        t+=1
        conv = _fft_self_convolve(a)
        maxv = conv.max()
        mask = conv >= maxv - 1e-12
        # compute gradient
        grad = np.zeros_like(a)
        active_idxs = np.where(mask)[0]
        for k in active_idxs:
            # compute shift contribution
            i0 = max(0, k - (n-1))
            i1 = min(n-1, k)
            # indices i from i0..i1 inclusive, each maps to a_{k-i}
            # vectorized: i_range = np.arange(i0,i1+1); grad[i_range] += a[k - i_range]
            i_range = np.arange(i0,i1+1)
            grad[i_range] += a[k - i_range]
        grad *= 2.0 / len(active_idxs)  # average across active
        # step size: some eta decreasing
        eta = 0.5 / np.sqrt(t)   # or adapt
        a = a - eta * grad
        # ensure nonnegative
        a = np.maximum(a, 0.0)
        # project onto simplex (sum=1)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0/n)
        else:
            a /= a_sum
        # evaluate occasionally
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score:
                best_score=cur; best_seq=a.copy()
    return best_seq,best_score
```

"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm Overview
-----------------
1. **Generate a diverse pool of seeds** – several analytic families
   (uniform, arcsine, beta‑type, edge‑weighted, sine‑weighted) over a
   few different lengths, plus random Dirichlet and sparse vectors.
   The best ~12 seeds are kept.

2. **Two‑phase local optimisation** for each seed  
   a) *Smooth exponential descent* – the existing soft‑max based mirror
      descent (fast convergence to a good neighbourhood).  
   b) *Exact sub‑gradient descent* – a projected sub‑gradient method that
      uses the true active set of the max‑convolution (no smoothing).  
      This sharpens the result obtained by the first phase.

3. **Greedy pairwise swaps** – move a small amount of mass between two
   positions only if the objective improves.

4. **Final stochastic local search** – cheap simulated‑annealing walk
   that can escape shallow local minima.

The routine respects a time budget (default 1000 s) and finally returns
a plain Python list.  All intermediate vectors are kept normalised to
sum = 1, which makes the objective scale‑invariant and matches the
behaviour of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smooth exponential descent (soft‑max gradient) ------------------------
def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 3.0,
    tau_mult: float = 1.001,
    eval_interval: int = 120,
    active_k: int = 12,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true objective using a soft‑max
    weighting of the convolution entries.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 400_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # ---- top active_k entries (or all) ----
        if active_k >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -active_k)[-active_k:]

        # soft‑max weights (stable)
        w_raw = np.exp(tau * (conv[top_idx] - max_c))
        w = w_raw / w_raw.sum()

        # ---- weighted gradient ----
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for weight, idx in zip(w, top_idx):
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * weight * a[ind[mask]]

        # ---- update ----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        tau *= tau_mult

        # ---- occasional evaluation ----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}"
                )
            else:
                no_improve += 1

        # ---- jitter on stagnation ----
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Exact sub‑gradient descent (projected onto simplex) -------------------
def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    eval_interval: int = 200,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the *true* active set of the
    max‑convolution (no softening).  Very effective as a final refinement.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # active set: all indices within a tiny absolute tolerance
        tol = maxc * 1e-12 + 1e-14
        act_idx = np.where(conv >= maxc - tol)[0]

        # ----- gradient (average over active set) -----
        grad = np.zeros_like(a)
        for k in act_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            grad[idx] += a[k - idx]
        if act_idx.size > 0:
            grad *= 2.0 / act_idx.size

        # ----- step size (diminishing) -----
        eta = eta0 / math.sqrt(t)
        a = a - eta * grad
        a = np.maximum(a, 0.0)

        # ----- projection onto simplex (sum = 1) -----
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # ----- occasional evaluation -----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] sub‑grad it={t} new best {best_score:.6f}"
                )
            else:
                no_improve += 1

        # ----- jitter on stagnation -----
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Greedy pairwise refinement --------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small amount of mass; keep it only if it improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Simulated‑annealing style random local search -------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects a time budget (default 1000 s) and returns a plain

    """

    overall_best_score = float("inf")

            np.logspace(-1, 0, 9, base=10),    # 0.1 .. 1
            np.logspace(0, 0.5, 7, base=10),   # 1 .. 3.16
            np.logspace(0.5, 1.0, 5, base=10), # 3.16 .. 10

        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)

            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

            beta = _distribution_beta_sym(n, p)

            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, sine):

                    if time.time() >= deadline:

                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"

            if time.time() >= deadline:

            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 .. 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a_vec = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a_vec)
            top_candidates.append((sc, a_vec.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a_vec.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a_vec = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a_vec)
            top_candidates.append((sc, a_vec.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a_vec.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the most promising seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. Include previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard: ensure we have at least one candidate
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Two‑phase optimisation on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # select a handful of seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting two‑phase optimisation on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed

            # a) smooth exponential descent (fast coarse optimisation)
            a_opt, sc_opt = _smooth_exponential_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.55,
                tau0=2.5,
                tau_mult=1.0015,
                eval_interval=100,
                active_k=14,
                patience=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth descent seed {idx} new global best {overall_best_score:.6f}"
                )

            # b) exact sub‑gradient descent for sharp refinement
            if time.time() < sub_deadline:
                a_opt, sc_opt = _subgradient_descent(
                    a_opt.copy(),
                    sub_deadline,
                    best_seq=overall_best_seq,
                    best_score=overall_best_score,
                    eta0=0.5,
                    eval_interval=150,
                    patience=2000,
                )
                if sc_opt < overall_best_score:
                    overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                    print(
                        f"[{time.time():.1f}] after sub‑grad seed {idx} new global best {overall_best_score:.6f}"
                    )

    # ------------------------------------------------------------------
    # 5. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 6. Final stochastic local search (simulated annealing style)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; ``evaluate_sequence`` will perform final checks.
    return overall_best_seq.tolist()
