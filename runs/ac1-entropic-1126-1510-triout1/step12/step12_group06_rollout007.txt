#!/usr/bin/env python
"""
Hybrid optimiser for the max‑convolution problem (improved version).

Main ideas
~~~~~~~~~~
1.  Produce a very large pool of seed distributions (uniform, arcsine,
    a dense grid of symmetric/asymmetric Beta families, Dirichlet draws,
    sparse patterns and a few completely random vectors).  The pool is
    cheap to evaluate and we keep the top‑k seeds.
2.  For each seed we run a *smooth‑max* Adam mirror‑descent with a
    much higher temperature schedule (α up to 500) and many more
    iterations.  The Adam loop works on the soft‑max surrogate
    f_α(a) = (1/α)·log Σ exp(α·conv_i(a)) and keeps the vector on the
    probability simplex.
3.  The best few seeds are refined by a classic sub‑gradient “multimax”
    descent (the same update used in the reference solution) with a
    longer budget.
4.  A deterministic *gradient‑mass‑transfer* phase follows: the
    sub‑gradient of the true max‑convolution is computed, the index with
    highest gradient (most responsible for the peak) donates a small
    fraction of its mass to the index with the lowest gradient.  This
    greedy step is performed until no improvement is observed or the
    time budget expires.
5.  Finally a cheap greedy pair‑wise swap (mass transfer) is applied as
    polishing.

All updates respect non‑negativity and the ℓ₁‑normalisation
(∑a_i = 1).  The total run‑time is bounded by the supplied budget
(`budget_s` seconds).  The routine prints intermediate scores and
returns the best sequence it has found.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    We keep sum(a)=1, so this reduces to 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Distribution factories -------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on the discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    """i^p, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Few non‑zero coordinates, random values."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Two‑block spikes
    for bf in (0.02, 0.04, 0.08, 0.12):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Dirichlet draws with varied concentration
    for a in (0.05, 0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.03))

    # Completely random probability vectors
    for _ in range(8):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    # Additional random Dirichlet draws (different alphas)
    for _ in range(6):
        a = 10 ** rng.uniform(-2, 0.5)      # 0.01 … 3.16
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    return cand

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.6,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.2,
                             alpha_mul: float = 1.20,
                             alpha_max: float = 500.0,
                             max_iter: int = 120000,
                             eval_interval: int = 600) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max surrogate
    f_α(a) = (1/α)·log Σ exp(α·conv_i(a)).
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # self‑convolution (exact)
        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # projection onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule (increase smoothness)
        if t % 400 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 80000,
                                 eval_interval: int = 800) -> Tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient descent using the exact sub‑gradient
    of the max‑convolution (average over all active indices).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-13
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _gradient_mass_transfer(a: np.ndarray,
                           deadline: float,
                           best_seq: np.ndarray,
                           best_score: float,
                           transfer_frac: float = 0.15,
                           max_no_improve: int = 3000) -> Tuple[np.ndarray, float]:
    """
    Greedy deterministic mass transfer guided by the sub‑gradient of the
    current max‑convolution.  On each step we move a fraction of the mass
    from the index with largest gradient to the one with smallest
    gradient.  The loop stops after ``max_no_improve`` iterations without
    progress or when the time budget ends.
    """
    n = a.shape[0]
    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # active set for the true max
        mask = conv >= max_val - 1e-12
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] < 1e-12:
            no_improve += 1
            continue

        delta = a[src] * transfer_frac
        # ensure we do not create near‑zero negativity
        delta = min(delta, a[src] * 0.99)

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] mass‑transfer src={src}→dst={dst} δ={delta:.3e} new best {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Cheap greedy pairwise mass‑transfer polishing."""
    n = a.shape[0]
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer amount (up to 30 % of the source mass)
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects ``budget_s`` seconds and returns the best sequence found
    (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest set of lengths.  Longer vectors give more freedom
    #    but are slower; we start with 4000 and 8000 and adapt if time permits.
    lengths = [4000, 8000, 12000]
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan a very rich set of analytic families (cheap).
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Scanning families, n={n}")

        cand_list = _initial_candidates(n, rng)

        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only a handful of promising seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ------------------------------------------------------------------
    # 3. Incorporate the previous solution if it exists.
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev = np.asarray(prev, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, 0.0)
                s = prev.sum()
                if s == 0.0:
                    prev = _distribution_uniform(prev.shape[0])
                    s = prev.sum()
                prev /= s
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – should never be needed.
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. First refinement: Adam‑style smooth‑max descent on the best seeds.
    remaining = deadline - time.time()
    if remaining > 80:
        # choose a few promising seeds (including the current global best)
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:4]
        ]
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # allocate a slice of the remaining time to each seed
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds) - idx)
            seq_out, score_out = _smooth_max_adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha0=0.2,
                alpha_mul=1.20,
                alpha_max=500.0,
                max_iter=120000,
                eval_interval=800,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam descent new best {overall_best_score:.6f}")
            # early stop if we already beat the target
            if overall_best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # 6. Second refinement: classic multimax sub‑gradient descent.
    remaining = deadline - time.time()
    if remaining > 60:
        seeds = [overall_best_seq.copy()] + [
            seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:2]
            if not np.may_share_memory(seq, overall_best_seq)
        ]
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds) - idx)
            seq_out, score_out = _subgradient_multimax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=80000,
                eval_interval=1000,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")
            if overall_best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # 7. Deterministic gradient‑guided mass transfer.
    if time.time() < deadline and overall_best_score > 1.503:
        overall_best_seq, overall_best_score = _gradient_mass_transfer(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            transfer_frac=0.12,
            max_no_improve=4000,
        )
        print(f"[{time.time():.1f}] after mass‑transfer best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final cheap polishing with pairwise swaps.
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
