def _optimize_sequence(a_init, deadline):

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family: w_i ∝ (i+0.5)^(−p) + (n−i−0.5)^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _cross_corr(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the cross‑correlation g_i = Σ_k p_k a_{k-i}
    (full convolution of p with reversed a, sliced to length n).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    need = len(p) + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    size = 1 << ((need - 1).bit_length())
    P = np.fft.rfft(p, n=size)
    A = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(P * A, n=size)
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
def _optimize_sequence(a_init: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Soft‑max mirror descent with a slowly increasing temperature α.
    Returns the best vector seen (ℓ₁‑normalised) and its exact objective.
    """
    a = _project_onto_simplex(a_init, s=1.0)   # guarantee ℓ₁=1
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    eta0 = 0.9            # base step‑size
    alpha = 0.5           # start temperature
    alpha_max = 2000.0    # do not let α explode

    while time.time() < deadline and t < 200_000:
        t += 1

        # ---- convolution & soft‑max ----
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # stable soft‑max
        exp_vals = np.exp(alpha * (conv - maxc))
        p = exp_vals / exp_vals.sum()

        # ---- gradient (2 * cross‑correlation) ----
        grad = 2.0 * _cross_corr(p, a)

        # ---- exponentiated‑gradient step ----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # ---- keep feasibility (ℓ₁=1, non‑negative) ----
        a = _project_onto_simplex(a, s=1.0)

        # ---- occasional evaluation ----
        if t % 100 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_a = a.copy()
                no_improve = 0
                # progress output
                print(f"[{time.time():.1f}] it={t:,} new local best {best_score:.6f}")
            else:
                no_improve += 1

        # ---- temperature schedule ----
        if t % 200 == 0:
            alpha = min(alpha * 1.02, alpha_max)

        # ---- jitter to escape plateaus ----
        if no_improve >= 2500:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep the move only if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style local search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Respects a wall‑clock budget (default 1000 s) and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0        # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    # ------------------------------------------------------------------
    # 1. Build a diversified pool of seeds (analytic + random)
    lengths = [2500, 4000, 6000, 8000, 12000, 16000]
    p_grid = np.concatenate(
        [
            np.logspace(-4, -1, 7),   # very heavy edge
            np.logspace(-1, 0, 7),
            np.logspace(0, 1, 7),
        ]
    )
    lam_grid = np.linspace(0.0, 0.7, 5)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # deterministic families
        seed_vecs = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
        ]

        # parametrised families + mixing with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    seed_vecs.append(cand)

        # random Dirichlet seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            seed_vecs.append(_distribution_dirichlet(n, rng))

        # sparse random seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            seed_vecs.append(_distribution_random_sparse(n, rng, density=0.005))

        # incorporate a previously discovered construction if present
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                prev = prev / prev.sum()
                seed_vecs.append(prev)
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 2. Optimise each seed (soft‑max mirror descent) with a slice of time
        per_seed_budget = min(3.0 + n / 5000.0, deadline - time.time())
        for base in seed_vecs:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_budget
            a_opt, sc_opt = _optimize_sequence(base.copy(), sub_deadline)
            if sc_opt < overall_best_score:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] new global best len={n} score={overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3. Fallback guard – should never happen but keeps the pipeline safe
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Greedy pairwise refinement (deterministic local optimisation)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 5. Final cheap simulated‑annealing style local search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
