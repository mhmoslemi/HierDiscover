"""
Improved candidate generator for the max‐convolution objective.

Strategy
--------
1. Choose a relatively large length (≈ 65 k) so that the discrete
   distribution can approximate the continuous optimum more closely.
2. Initialise the vector from several analytic families
   (uniform, arcsine, symmetric beta, edge–weighted) and also by a
   cheap random search over the two continuous parameters
   *p* (shape) and *λ* (mixing with the uniform distribution).
   The best seed obtained is kept as ``best_seq``.
3. Run a long exponentiated‑gradient (mirror) descent on the exact
   objective 2·n·max(conv(a,a)) while the sum of the vector is fixed
   to 1.  The gradient is taken only from convolution entries that are
   within 0.5 % of the current maximum (this focuses the update on the
   true bottleneck).
   The learning rate decays as 1/√t.
4. Periodically evaluate the objective and keep the globally best
   vector seen.
5. Finish with a greedy “pairwise‑swap’’ hill‑climber that moves a
   random fraction of mass from one coordinate to another whenever it
   strictly improves the objective.
6. Return the best vector (as a plain Python list).  The caller will
   invoke ``evaluate_sequence`` which applies the same normalisation,
   so the algorithm works on the scale‑invariant formulation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    ``a`` must be non‑negative and already normalised.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1} (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Keep the move only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.
    The routine builds a high‑resolution seed, runs many exponentiated‑gradient
    updates and finishes with a greedy hill‑climber.  It respects the supplied
    time budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin for cleanup

    # ------------------------------------------------------------------
    # 1.   Choose a large dimension (good trade‑off between resolution and speed)
    # ------------------------------------------------------------------
    n = 65536                # power‑of‑two comfortably handled by FFT

    # Initialise best objects
    best_seq = _distribution_uniform(n)
    best_score = _objective(best_seq)
    print(f"[{time.time():.1f}] start uniform  n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.   Deterministic families (very cheap, give good baselines)
    # ------------------------------------------------------------------
    for base_fn in (
        _distribution_arcsine,
        _distribution_uniform,
    ):
        cand = base_fn(n)
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.1f}] new best deterministic {base_fn.__name__} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.   Parameterised families – quick random scan
    # ------------------------------------------------------------------
    # we will explore ``edge_weighted`` and ``beta_sym`` with random p, λ
    # Keep scanning until we run out of a few seconds (reserve > 200 s for descent)
    while time.time() < deadline - 250:
        family = rng.choice(['edge', 'beta'])
        # log‑uniform sampling of p ∈ [0.01, 5]
        logp = rng.uniform(-2.0, math.log10(5.0))
        p = 10.0 ** logp
        lam = rng.random()
        if family == 'edge':
            base = _distribution_edge_weighted(n, p)
        else:
            base = _distribution_beta_sym(n, p)
        cand = _mix_with_uniform(base, lam)
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time():.1f}] new best {family} p={p:.4g} λ={lam:.3f} score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 4.   Exponentiated‑gradient (mirror) descent on the best seed
    # ------------------------------------------------------------------
    a = best_seq.copy()
    t = 0
    # we keep a small fraction of the total budget for the final swap phase
    descent_deadline = deadline - 30.0
    while time.time() < descent_deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # indices whose value is within 0.5 % of the max – these are the bottleneck spots
        thresh = max_val * 0.995
        idxs = np.where(conv >= thresh)[0]

        # build sub‑gradient from the selected indices
        grad = np.zeros_like(a)
        n_len = a.shape[0]
        for idx in idxs:
            ind = idx - np.arange(n_len)
            mask = (ind >= 0) & (ind < n_len)
            grad[mask] += 2.0 * a[ind[mask]]

        # EG update (multiplicative) with a diminishing step size
        eta = 0.6 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection onto the probability simplex (just normalise because a≥0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n_len, 1.0 / n_len, dtype=np.float64)
        else:
            a = a / s

        # occasional evaluation
        if t % 500 == 0:
            cur = _objective(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] iter {t} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.   Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] start final greedy polishing")
        best_seq, best_score = _pairwise_swap(
            best_seq.copy(),
            deadline,
            best_seq.copy(),
            best_score,
        )
        print(f"[{time.time():.1f}] after polishing best {best_score:.6f}")

    print(
        f"Final best score (2·n·max(conv)) = {best_score:.6f}   length = {len(best_seq)}"
    )
    # Return a plain Python list – the external ``evaluate_sequence`` will
    # apply its own clipping and normalisation.
    return best_seq.tolist()
