def _smooth_descent(a, deadline, best_seq, best_score, beta_start=0.5, beta_end=200.0, lr0=0.7):

   n=len(a)
   t=0
   beta=beta_start
   while time.time() < deadline:
        t+=1
        # compute conv of a (maybe not needed each iteration, but compute for objective)
        conv_a = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv_a.max() / (a.sum()**2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth iter={t} best {best_score:.6f}")
        # compute softmax weights
        max_c = conv_a.max()
        # scale conv by beta, compute softmax
        exp_vals = np.exp(beta * (conv_a - max_c))
        p = exp_vals / exp_vals.sum()
        # compute gradient via convolution p and a_rev
        a_rev = a[::-1]
        conv_pr = _fft_convolve(p, a_rev)
        grad = conv_pr[n-1:n-1+len(a)]  # length n
        # optionally scale grad by average
        grad_scaled = grad / (grad.mean() + 1e-12)
        eta = lr0 / np.sqrt(t)
        a = a * np.exp(- eta * grad_scaled)
        # renorm
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0/len(a))
        else:
            a = a / s
        # increase beta gradually to approach max
        if beta < beta_end:
            beta *= 1.02  # increase 2% each iteration
        else:
            beta = beta_end
   return best_seq, best_score

"""
Hybrid analytic seeding + exponential‑gradient subgradient +
soft‑max smoothed gradient descent + greedy polishing.

The search explores several families (uniform, arcsine, Beta, power,
two‑block, sparse, Dirichlet) for a few lengths (up to 12000).  The best
few seeds are refined by an exponentiated‑gradient sub‑gradient method,
then by a smooth approximation of the max‑operator using a soft‑max.
Finally a cheap random mass–transfer walk tightens the objective.
All iterates stay on the probability simplex (sum = 1, non‑negative).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real arrays via real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (sum is assumed >0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    """Flat distribution (probability simplex)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine_offset(n: int, c: float) -> np.ndarray:
    """Arcsine‑like density with a positive offset c."""
    i = np.arange(n, dtype=np.float64)
    left = i + c
    right = (n - 1 - i) + c
    vals = 1.0 / np.sqrt(left * right)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:

def _max_subgrad_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using the true sub‑gradient
    g_i = 2·a_{k‑i} where k is a maximal convolution index.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        max_conv = conv[k]

        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={t} new best {best_score:.6f}")

        # sub‑gradient for the chosen shift k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        grad = np.zeros_like(a)
        grad[idx_i] = 2.0 * a[idx_j]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # keep on the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

    return best_seq, best_score

def _smooth_gradient_descent(a: np.ndarray, deadline: float,
                             best_seq: np.ndarray, best_score: float,
                             beta_start: float = 0.5,
                             beta_end: float = 200.0,
                             lr0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth surrogate
        F(a)=2·n·(1/β)·log Σ_k exp(β·conv_k).
    The gradient is 4·n·(p * a_rev) where p is the soft‑max
    over conv_k.  An exponentiated‑gradient update keeps the iterate
    in the probability simplex.
    """
    n = a.shape[0]
    t = 0
    beta = beta_start
    a_rev = a[::-1]

    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth iter={t} new best {best_score:.6f}")

        # soft‑max weights over the convolution values
        max_c = conv.max()
        # avoid overflow/underflow
        exp_vals = np.exp(beta * (conv - max_c))
        p = exp_vals / exp_vals.sum()

        # gradient = 4·n·(p * a_rev)
        conv_pr = _fft_convolve(p, a_rev)
        grad = conv_pr[n - 1:n - 1 + n]          # length n
        # normalise the magnitude to avoid too large steps
        grad = grad / (grad.mean() + 1e-12)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection onto simplex (renormalisation)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # slowly increase the temperature (β) towards the target value
        if beta < beta_end:
            beta = min(beta_end, beta * 1.02)

        # keep a_rev up‑to‑date
        a_rev = a[::-1]

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a random fraction of mass from
    one index to another and keep the move if it strictly improves the
    true (non‑smoothed) objective.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30  # move at most 30 % of source
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real arrays via real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]

    beta = beta_start
    a_rev = a[::-1]

    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smooth iter={t} new best {best_score:.6f}")

        # soft‑max weights
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))
        p = exp_vals / exp_vals.sum()

        # gradient = 4·n·(p * a_rev)
        conv_pr = _fft_convolve(p, a_rev)
        grad = conv_pr[n - 1:n - 1 + n]       # length n
        grad = grad / (grad.mean() + 1e-12)   # normalise magnitude

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # slowly raise β toward its final value
        if beta < beta_end:
            beta = min(beta_end, beta * 1.02)

        a_rev = a[::-1]

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a random fraction of mass from
    one index to another and keep the move only if the true objective
    strictly improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30   # move up to 30 % of source
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur_score = _objective(new_a)
        if cur_score < best_score - 1e-13:
            a = new_a
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """Main entry point. Returns the best sequence discovered within the budget."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 3.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Analytic seeding – diversified pool for several lengths
    lengths = [2000, 4000, 8000, 12000]
    seed_pool: list[tuple[float, np.ndarray]] = []

    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- generating seeds for length {n} ---")
        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                cand = _distribution_uniform(n)
            else:
                cand = cand / s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            seed_pool.append((sc, cand.copy()))

        # retain a modest elite across all processed lengths
        seed_pool.sort(key=lambda x: x[0])
        seed_pool = seed_pool[:20]

    # ------------------------------------------------------------------
    # 2. Incorporate any previously known solution (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
            seed_pool.append((sc, prev.copy()))
            seed_pool.sort(key=lambda x: x[0])
            seed_pool = seed_pool[:20]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Sub‑gradient refinement on the elite seeds
    remaining = deadline - time.time()
    if remaining > 20.0 and seed_pool:
        seed_pool.sort(key=lambda x: x[0])
        elite = [seq for _, seq in seed_pool[:5]]
        per_seed = max(1.0, remaining / len(elite))
        print(f"[{time.time():.1f}] sub‑gradient stage: {len(elite)} seeds, {per_seed:.1f}s each")
        for seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _max_subgrad_descent(
                seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smoothed‑gradient descent on the top candidates
    remaining = deadline - time.time()
    if remaining > 30.0 and overall_best_seq is not None:
        seed_pool.sort(key=lambda x: x[0])
        smooth_seeds = [overall_best_seq] + [seq for _, seq in seed_pool[:2]]
        per_seed = max(2.0, remaining / len(smooth_seeds))
        print(f"[{time.time():.1f}] smooth‑gradient stage: {len(smooth_seeds)} seeds, {per_seed:.1f}s each")
        for seq in smooth_seeds:
            if time.time() >= deadline:
                break
            smooth_deadline = time.time() + per_seed
            seq_out, score_out = _smooth_gradient_descent(
                seq.copy(),
                smooth_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta_start=0.5,
                beta_end=200.0,
                lr0=0.7,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after smooth new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final greedy polishing
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] polishing stage")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # Fallback: uniform distribution of moderate size
        overall_best_seq = _distribution_uniform(1000)
        overall_best_score = _objective(overall_best_seq)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
