"""
Hybrid optimizer with Adam‑based exponentiated gradient.

The algorithm proceeds in three stages:

1. **Broad seed generation** – for several lengths (8000 and 20000) a
   collection of analytic families (uniform, arcsine, Beta, power‑law,
   exponential, two‑block, sine) and random Dirichlet draws is created.
   All candidates are normalised to sum = 1 and the exact objective
   `2·n·max(conv(a,a))/sum(a)^2` is evaluated.  The best few (≤ 12)
   seeds are kept as the elite set.

2. **Adam‑based exponentiated‑gradient descent** – each elite seed is
   refined by two phases.
      *Phase A* uses a soft‑max approximation of the maximal convolution
      entry and an Adam update on the log‑probabilities (exponentiated
      gradient on the simplex).  The temperature α is increased from a
      small value to a large one (≈ 200) so that the approximation tightens.
      *Phase B* switches to the true maximal entry (hard‑max) while still
      using the Adam update.  Both phases run until the allocated time per
      seed expires.

3. **Greedy polishing** – a cheap random mass‑transfer walk (pairwise
   swaps) runs until the global deadline, keeping any improvement.

The whole routine respects the supplied time budget (`budget_s`) and
returns the best sequence (list of non‑negative floats) found.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    # next power of two >= 2*n-1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The vector a is assumed to be non‑negative.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, sum(w)=1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.ifft(W * A_rev).real
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry `conv[k]`.
    Returns a vector of size len(a) with non‑negative entries.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]
    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]
    return g

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))

    # symmetric beta on a coarse grid
    for p in np.linspace(0.05, 1.0, 12):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric beta
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power law
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential
    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # sparse random draws
    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws with various concentrations
    for a in (0.1, 0.3, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # a few pure Dirichlet samples
    for _ in range(6):
        cand.append(_distribution_dirichlet(n, rng, alpha=rng.uniform(0.05, 2.0)))

    return cand


# ----------------------------------------------------------------------
def _softmax_adam(a: np.ndarray, deadline: float,
                  best_seq: np.ndarray, best_score: float,
                  lr: float = 0.1,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  max_iter: int = 200000,
                  eval_interval: int = 400,
                  alpha_start: float = 0.5,
                  alpha_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam‑based exponentiated‑gradient descent on a soft‑max approximation
    of the maximal convolution entry.
    """
    n = a.shape[0]
    # keep a strictly positive to stay in log‑space
    a = np.maximum(a, 1e-30)
    a /= a.sum()
    y = np.log(a)                     # log‑probabilities
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        a = np.exp(y)
        a /= a.sum()                  # renormalise (simple simplex projection)

        # convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed max
        grad = _gradient_from_weights(a, w)

        # gradient w.r.t. y (log‑space) : ∂L/∂y = a * ∂L/∂a
        grad_y = a * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        y -= step                     # exponentiated‑gradient in log‑space

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max‑adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    # final projection to guarantee sum=1 and non‑negativity
    a = np.exp(y)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _hardmax_adam(a: np.ndarray, deadline: float,
                  best_seq: np.ndarray, best_score: float,
                  lr: float = 0.08,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  max_iter: int = 200000,
                  eval_interval: int = 400) -> tuple[np.ndarray, float]:
    """
    Adam‑based exponentiated‑gradient descent on the true maximal
    convolution entry.
    """
    n = a.shape[0]
    a = np.maximum(a, 1e-30)
    a /= a.sum()
    y = np.log(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        a = np.exp(y)
        a /= a.sum()

        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        grad = _gradient_max_index(a, k)

        grad_y = a * grad

        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        y -= step

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max‑adam it={t} idx={k} new best {best_score:.6f}")

    a = np.exp(y)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap random mass‑transfer walk (pairwise swaps) that keeps a change
    only if the true objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        # move up to 30 % of source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, seed: int = 123) -> list[float]:
    """
    Entry point for the checker.  Generates a diverse set of seeds,
    refines the elite few with Adam‑based soft‑max then hard‑max
    descent, and finally polishes them with a greedy local walk.
    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1. Seed generation for two problem sizes
    lengths = [8000, 20000]   # longer sequences give better theoretical bounds
    elite = []                # (score, seq) tuples across all lengths

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length = {n}")

        # analytic families
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            elite.append((sc, cand.copy()))
            if sc < (elite[0][0] if elite else float('inf')):
                pass  # just keep for sorting later

        # a few pure Dirichlet random draws (extra diversity)
        for _ in range(12):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng, alpha=rng.uniform(0.05, 2.0))
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            elite.append((sc, cand.copy()))

    # keep only the best few seeds
    elite.sort(key=lambda x: x[0])
    elite = elite[:12]

    # ------------------------------------------------------------------
    # 2. Main optimisation loop on elite seeds
    overall_best_seq = None
    overall_best_score = float("inf")
    if elite:
        overall_best_seq, overall_best_score = elite[0][1].copy(), elite[0][0]
        print(f"[{time.time():.1f}] initial global best score={overall_best_score:.6f}")

    # use the top six seeds for deep refinement
    seeds = [seq for _, seq in elite[:6]]

    remaining_time = deadline - time.time()
    if remaining_time > 20.0 and seeds:
        per_seed_budget = remaining_time / (len(seeds) * 2.0)  # two phases per seed
        for seed_seq in seeds:
            if time.time() >= deadline:
                break

            # Phase A – soft‑max with Adam
            sub_deadline_a = time.time() + per_seed_budget
            seq_a, score_a = _softmax_adam(
                seed_seq.copy(),
                sub_deadline_a,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.12,
                max_iter=200000,
                eval_interval=400,
                alpha_start=0.3,
                alpha_end=200.0,
            )
            if score_a < overall_best_score:
                overall_best_score = score_a
                overall_best_seq = seq_a.copy()
                print(f"[{time.time():.1f}] after soft‑max new best {overall_best_score:.6f}")

            if time.time() >= deadline:
                break

            # Phase B – hard‑max with Adam
            sub_deadline_b = time.time() + per_seed_budget
            seq_b, score_b = _hardmax_adam(
                seq_a.copy(),
                sub_deadline_b,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.09,
                max_iter=200000,
                eval_interval=400,
            )
            if score_b < overall_best_score:
                overall_best_score = score_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after hard‑max new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final greedy polishing
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 4. Return the best sequence found
    if overall_best_seq is None:
        # fallback: uniform distribution of moderate length
        overall_best_seq = _distribution_uniform(8000)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform, score={overall_best_score:.6f}")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
