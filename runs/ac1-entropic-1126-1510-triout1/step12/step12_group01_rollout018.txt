def _active_lp_refinement(a, deadline, best_seq, best_score, active_k=15, max_refine_iters=20, epsilon=1e-7):
   while time.time() < deadline and max_refine_iters >0:
       conv = _fft_self_convolve(a)
       M = conv.max()
       # compute improvement threshold: require new t < M - eps.
       # Determine active set: top active_k indices
       if active_k>= len(conv):
           idxs = np.arange(len(conv))
       else:
           idxs = np.argpartition(conv, -active_k)[-active_k:]

       # Build LP
       n = a.shape[0]
       m = len(idxs)
       # Build A_ub matrix (m x (n+1))
       # We will create a dense matrix? For n=20000 and m=15, it's 15*20001 ~ 300k entries, okay.
       A_ub = np.zeros((m, n+1), dtype=np.float64)
       b_ub = np.empty(m, dtype=np.float64)
       for row, k in enumerate(idxs):
           # fill coefficients for Δ_i
           # for i from max(0, k-n+1) to min(n-1, k)
           i_start = max(0, k - n + 1)
           i_end = min(n-1, k)  # inclusive
           # indices i_start..i_end inclusive; number = i_end-i_start+1
           # For each i: g_i = 2 * a[k-i]
           # So we can set slice of A_ub[row, i_start:i_end+1] = 2 * a[k - np.arange(i_start,i_end+1)] (i.e., reversed segment)
           indices_i = np.arange(i_start, i_end+1)
           a_indices = k - indices_i
           A_ub[row, i_start:i_end+1] = 2.0 * a[a_indices]  # vector
           # coefficient for t
           A_ub[row, -1] = -1.0
           b_ub[row] = -conv[k]

       # Equality for sum Δ = 0
       A_eq = np.zeros((1, n+1), dtype=np.float64)
       A_eq[0, :n] = 1.0
       b_eq = np.array([0.0])

       # variable bounds
       bounds = []
       for i in range(n):
           bounds.append((-a[i], None))
       bounds.append((0.0, None))  # t >= 0

       # Solve LP
       res = scipy.optimize.linprog(c=np.concatenate([np.zeros(n), [1.0]]), A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(5, int((deadline - time.time())))} )
       if not res.success:
           # abort
           break

       delta = res.x[:n]
       t_opt = res.x[-1]
       if t_opt < M - epsilon:
           a = a + delta
           # clamp small negatives to zero, renorm
           a = np.maximum(a, 0.0)
           sum_a = a.sum()
           if sum_a != 0.0:
               a = a / sum_a
           new_score = 2 * a.shape[0] * _fft_self_convolve(a).max()  # compute objective
           if new_score < best_score - 1e-12:
               best_score = new_score
               best_seq = a.copy()
               print(f"[{time.time():.1f}] LP refine iteration: new best {best_score:.6f}")
       else:
           # no improvement, break
           break

       max_refine_iters -= 1

   return best_seq, best_score

def _active_set_lp_refine(a, deadline, best_seq, best_score, active_k=15, max_iter=20, epsilon=1e-8):
    n = a.shape[0]
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a)
        M = conv.max()
        # early exit if improvement cannot be more than epsilon (abs)
        if best_score <= 2.0 * n * (M - epsilon): # approximate?
            pass
        # get top indices:
        if active_k >= len(conv):
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(conv, -active_k)[-active_k:]

        # Build LP:
        m = len(idxs)
        # Build A_ub
        A_ub = np.zeros((m, n+1), dtype=np.float64)
        b_ub = np.empty(m, dtype=np.float64)
        for ri, k in enumerate(idxs):
            i_start = max(0, k - n + 1)
            i_end = min(n-1, k)
            # indices range inclusive
            # assign slice
            # reversed indexing:
            # a_slice = a[k - np.arange(i_start,i_end+1)]
            idx_range = np.arange(i_start, i_end+1)
            a_slice = a[k - idx_range]  # size L
            A_ub[ri, i_start:i_end+1] = 2.0 * a_slice
            A_ub[ri, -1] = -1.0
            b_ub[ri] = -conv[k]

        A_eq = np.zeros((1, n+1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        c = np.concatenate([np.zeros(n), np.array([1.0])], dtype=np.float64)

        bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]

        # solve
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(5, int(deadline - time.time())) })
        if not res.success:
            # if fails break
            break

        delta = res.x[:n]
        t_opt = res.x[-1]

        # Check improvement
        if t_opt < M - epsilon:
            a = a + delta
            # numeric issues: clip to min 0, renorm
            a = np.maximum(a, 0.0)
            # renorm to ensure sum 1
            s = a.sum()

                best_score = new_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] LP refinement iter {it} new best {best_score:.6f}")
            # continue loop
        else:
            # no improvement or stagnated
            break
    return best_seq, best_score

Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.  The algorithm

1. builds a diverse pool of analytic and random seeds;
2. runs a fast exponentiated‑gradient descent (soft‑max weighting) on the
   exact objective;
3. refines the result with a small active‑set linear‑program that directly
   reduces the current maximal convolution entry using a linearisation of
   the constraints;
4. finishes with cheap pairwise swaps and a short simulated‑annealing walk.

All steps respect the 1000 s budget.
"""

from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid underflow
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))  # shift for stability
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 3.0,
    tau_mult: float = 1.001,
    eval_interval: int = 120,
    active_k: int = 12,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact objective.
    Uses a soft‑max weighting of the top active_k convolution entries.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # ---- soft‑max weighting of the top active_k entries ----
        if active_k >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -active_k)[-active_k:]

        # scale by the current temperature (subtract max_c to avoid overflow)
        w_raw = np.exp(tau * (conv[top_idx] - max_c))
        w = w_raw / w_raw.sum()

        # ---- compute weighted gradient ----
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for weight, idx in zip(w, top_idx):
            # g_i = 2 * a_{idx-i} when index valid
            i_start = max(0, idx - n + 1)
            i_end = min(n - 1, idx)
            # i runs from i_start..i_end inclusive
            # corresponding a indices: idx - i
            # form slice of grad
            grad[i_start:i_end + 1] += 2.0 * weight * a[idx - np.arange(i_start, i_end + 1)]

        # ---- exponentiated‑gradient update ----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # increase temperature slowly (focuses more on the true max)
        tau *= tau_mult

        # ---- occasional evaluation ----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # ---- jitter on stagnation ----
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _active_set_lp_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    active_k: int = 14,
    max_iters: int = 12,
    eps: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Refines the current vector by solving a small linear programme that
    linearises the max‑convolution constraints around the current point.
    Only the top ``active_k`` convolution entries are kept as constraints.
    """
    n = a.shape[0]
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        M = conv.max()

        # indices of the (up to) active_k largest entries
        if active_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -active_k)[-active_k:]

        # Build LP data
        m = len(idxs)
        # A_ub * [Δ, t] <= b_ub
        A_ub = np.zeros((m, n + 1), dtype=np.float64)
        b_ub = np.empty(m, dtype=np.float64)

        for row, k in enumerate(idxs):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # slice of a that appears in the gradient for this k
            # g_i = 2 * a_{k-i}
            idx_range = np.arange(i_start, i_end + 1)
            a_slice = a[k - idx_range]           # a_{k-i}
            A_ub[row, i_start:i_end + 1] = 2.0 * a_slice
            A_ub[row, -1] = -1.0                 # -t
            b_ub[row] = -conv[k]

        # Equality: sum Δ_i = 0 (mass is preserved)
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        # Objective: minimise t
        c = np.concatenate([np.zeros(n), np.array([1.0])])

        # Variable bounds: Δ_i >= -a_i ; t >= 0
        bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]

        # Solve LP using HiGHS (fast for modest size)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # if the LP fails we abort the refinement
            print(f"[{time.time():.1f}] LP failed at it={it}: {res.message}")
            break

        delta = res.x[:n]
        t_opt = res.x[-1]

        if t_opt < M - eps:
            a = a + delta
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a /= s
            new_score = _objective(a)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] LP refine it={it} new best {best_score:.6f}")
        else:
            # no appreciable improvement – stop
            break

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def _interpolate_sequence(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Resize a vector to ``target_n`` by linear interpolation and normalise."""
    if seq.size == target_n:
        return seq.copy()
    x_src = np.linspace(0.0, 1.0, seq.size, endpoint=False)
    x_tgt = np.linspace(0.0, 1.0, target_n, endpoint=False)
    interp = np.interp(x_tgt, x_src, seq)
    interp = np.maximum(interp, 0.0)
    s = interp.sum()
    if s == 0.0:
        return np.full(target_n, 1.0 / target_n, dtype=np.float64)
    return interp / s

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

The algorithm proceeds in four stages:

1. **Seed generation** – a large, diverse pool of analytic families
   (uniform, arcsine, symmetric/asymmetric Beta, edge‑weighted, sine‑weighted),
   mixtures with the uniform distribution, and random sparse vectors is built.
   For several lengths (up to 20 000) the best ~50 seeds are retained.

2. **Exponentiated‑gradient descent** – each promising seed is refined by a
   smooth exponential descent that uses a soft‑max over the top convolution
   entries (the “temperature’’ τ is slowly increased).  The iterate is
   projected onto the simplex after each step.

3. **Active‑set LP refinement** – the current vector is locally improved by
   solving a tiny linear programme that linearises the active convolution
   constraints (the indices attaining the maximal value) and directly reduces
   the maximal convolution entry.  Only the top ~20 entries are kept as
   constraints, making the LP cheap even for n≈20 000.

4. **Final local polishing** – a greedy pairwise‑mass move and a short
   simulated‑annealing walk are applied to squeeze out any remaining gain.

All steps respect a 1000 s budget.  The returned list is plain Python; the
framework will later call ``evaluate_sequence`` which handles clipping,
normalisation, etc.
"""

import time
import math
import numpy as np
from typing import Optional
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 3.0,
    tau_mult: float = 1.001,
    eval_interval: int = 120,
    active_k: int = 12,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact objective.
    Uses a soft‑max weighting of the top ``active_k`` convolution entries.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # ---- soft‑max weighting of the top active_k entries ----
        if active_k >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -active_k)[-active_k:]

        w_raw = np.exp(tau * (conv[top_idx] - max_c))
        w = w_raw / w_raw.sum()

        # ---- weighted gradient (exact linearisation of each active entry) ----
        grad = np.zeros_like(a)
        for weight, idx in zip(w, top_idx):
            i_start = max(0, idx - n + 1)
            i_end = min(n - 1, idx)
            # a_{idx-i} for i in [i_start,i_end]
            indices = np.arange(i_start, i_end + 1)
            grad[indices] += 2.0 * weight * a[idx - indices]

        # ---- exponentiated‑gradient step ----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        tau *= tau_mult

        # periodic evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter if we have been stuck for a while
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _active_set_lp_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    active_k: int = 20,
    max_iters: int = 12,
    eps: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Linearise the active convolution constraints and solve a tiny LP that
    directly reduces the maximal convolution value.
    """
    n = a.shape[0]
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        M = conv.max()

        # choose the active set – the largest entries of the convolution
        if active_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -active_k)[-active_k:]

        m = len(idxs)
        # A_ub @ [Δ, t] ≤ b_ub  where Δ ∈ ℝⁿ, t ∈ ℝ
        A_ub = np.zeros((m, n + 1), dtype=np.float64)
        b_ub = np.empty(m, dtype=np.float64)

        for r, k in enumerate(idxs):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1)
            # gradient coefficient 2·a_{k-i}
            A_ub[r, i_start:i_end + 1] = 2.0 * a[k - idx_range]
            A_ub[r, -1] = -1.0                # coefficient of -t
            b_ub[r] = -conv[k]

        # Equality: total mass is preserved (sum Δ = 0)
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        # Objective: minimise t
        c = np.concatenate([np.zeros(n), np.array([1.0])])

        # Bounds: Δ_i ≥ -a_i (so a_i+Δ_i ≥0),   t ≥ 0
        bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]

        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            print(f"[{time.time():.1f}] LP failed at it={it}: {res.message}")
            break

        delta = res.x[:n]
        t_opt = res.x[-1]

        if t_opt < M - eps:
            a = a + delta
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a /= s
            new_score = _objective(a)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] LP refine it={it} new best {best_score:.6f}")
        else:
            # no appreciable improvement – stop early
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: Optional[int] = None,
) -> tuple[np.ndarray, float]:
    """Greedy random mass transfer; keep the move if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing walk that occasionally accepts worsening moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _interpolate_sequence(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Resize a vector to ``target_n`` by linear interpolation and renormalise."""
    if seq.size == target_n:
        return seq.copy()
    x_src = np.linspace(0.0, 1.0, seq.size, endpoint=False)
    x_tgt = np.linspace(0.0, 1.0, target_n, endpoint=False)
    interp = np.interp(x_tgt, x_src, seq)
    interp = np.maximum(interp, 0.0)
    s = interp.sum()
    if s == 0.0:
        return np.full(target_n, 1.0 / target_n, dtype=np.float64)
    return interp / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the supplied time budget (default 1000 s) and
    returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 4.0  # safety margin

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    candidate_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Assemble a diverse pool of seeds
    lengths = [4000, 8000, 12000, 20000]
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.2, 8),   # 0.01 … 0.63
            np.logspace(-0.2, 0.4, 12), # 0.63 … 2.5
            np.logspace(0.4, 0.8, 8),   # 2.5 … 6.3
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for base in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(base)
            candidate_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"   new global best len={n} score={sc:.6f}")

        # parametrised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    candidate_pool.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"   new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta random draws (log‑uniform over [0.1,10])
        for _ in range(6):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-1, 1)   # 0.1 … 10
            b_param = 10 ** rng.uniform(-1, 1)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"   new global best asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"   new global best sparse len={n} score={sc:.6f}")

        # keep only the most promising seeds overall
        candidate_pool.sort(key=lambda x: x[0])
        candidate_pool = candidate_pool[:50]

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            target_len = max(lengths)
            up = _interpolate_sequence(prev, target_len)
            sc = _objective(up)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, up.copy()
                print(f"   imported previous construction score={sc:.6f}")
            candidate_pool.append((sc, up.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback guard
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth exponential descent on the top seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and candidate_pool:
        seeds = [seq for _, seq in sorted(candidate_pool, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] smooth descent on {len(seeds)} seeds, {per_seed:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_exponential_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.55,
                tau0=2.5,
                tau_mult=1.0015,
                eval_interval=200,
                active_k=15,
                patience=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Active‑set LP refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] active‑set LP refinement")
        overall_best_seq, overall_best_score = _active_set_lp_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            active_k=20,
            max_iters=12,
            eps=1e-9,
        )

    # ------------------------------------------------------------------
    # 6. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 7. Final simulated‑annealing style local search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; ``evaluate_sequence`` will handle the final checks.
    return overall_best_seq.tolist()
