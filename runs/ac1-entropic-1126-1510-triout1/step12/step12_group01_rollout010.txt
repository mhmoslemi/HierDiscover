        grad = 2.0 * np.convolve(w, a[::-1], mode='full')[n-1:n-1+n]
        # update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # renorm to simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s
        # increase temperature
        tau *= tau_mul
        # eval occasionally
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] expdescent iter={t} new best {best_score:.6f}")
            else:
                no_improve += 1
        if no_improve >= jitter_patience:
            # add jitter
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.clip(a, 0, None)
            a = a / a.sum()
            no_improve = 0
    return best_seq, best_score

"""
Propose a low‑autocorrelation step function.

Algorithm
---------
1. Build a diverse library of analytic “seed” distributions (uniform,
   arcsine, beta‑family, edge‑weighted, sine‑weighted, random Dirichlet,
   sparse) for a relatively large length (default 12 000).  All seeds are
   normalised to sum = 1 and the best few are kept.

2. For each promising seed run a *soft‑max* exponentiated‑gradient descent.
   The smoothed objective is the log‑sum‑exp of the self‑convolution, i.e.

        Fτ(a) = (1/τ)·log Σexp(τ·conv(a,a))

   Its gradient can be written as a convolution and is computed with one
   FFT per iteration.  The update is multiplicative
        a ← a·exp(‑η·∇Fτ)   followed by a projection onto the simplex.
   The temperature τ is increased slowly (τ←τ·τ_mul) so that the
   approximation tightens around the true max.  A jitter step is
   injected when progress stalls.

3. The best sequence from the descent phase is refined by cheap
   greedy pairwise mass moves and a very short simulated‑annealing walk.

The whole routine respects the 1000 s budget and returns a plain Python
list; `evaluate_sequence` will perform the final checks.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Discrete symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b_: float) -> np.ndarray:
    """Discrete asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b_ - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+0.5)*(n‑i‑0.5))^(‑p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, U = uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) objective.
    Returns ∇Fτ(a) = 2·conv(w , rev(a)) where w_i ∝ exp(τ·conv_i).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (stable)
    w_raw = np.exp(tau * (conv - max_c))
    w = w_raw / w_raw.sum()
    # gradient via a single convolution
    # grad[i] = 2·Σ_k w_k·a_{k‑i}
    grad = 2.0 * np.convolve(w, a[::-1], mode='full')[n - 1 : n - 1 + n]
    return grad


def _exp_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    tau0: float = 1.0,
    tau_mul: float = 1.001,
    max_tau: float = 10.0,
    eval_interval: int = 120,
    jitter_patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the smoothed max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 400_000:
        t += 1

        # gradient of the smooth approximation
        grad = _softmax_gradient(a, tau)

        # multiplicative update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project back onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # temperature schedule
        tau = min(tau * tau_mul, max_tau)

        # occasional evaluation on the true objective
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] exp‑descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter when stuck
        if no_improve >= jitter_patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.clip(a, 0, None)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy move of a small mass between two positions."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

    # 3. Evaluate all seeds, keep the best few
    evaluated = []
    overall_best_seq = None
    overall_best_score = float("inf")
    for cand in seeds:
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s
        sc = _objective(cand)
        evaluated.append((sc, cand))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new global seed score {sc:.6f}")
        if time.time() >= deadline:
            break

    # keep only the most promising seeds (top‑8)
    evaluated.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in evaluated[:8]]

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient descent on the selected seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining / max(1, len(top_seeds))
        print(f"[{time.time():.1f}] start exp‑gradient descent on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            overall_best_seq, overall_best_score = _exp_gradient_descent(
                seed_seq,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.9,
                tau0=1.0,
                tau_mul=1.0015,
                max_tau=8.0,
                eval_interval=150,
                jitter_patience=1800,
            )
            print(
                f"[{time.time():.1f}] after descent seed {idx} best={overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 5. Final cheap refinement on the full vector
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=120_000,
        )

    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.006,
            decay=0.9996,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain list – evaluate_sequence will perform final checks.
    return overall_best_seq.tolist()
