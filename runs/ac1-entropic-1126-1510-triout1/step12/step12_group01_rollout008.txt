def _build_convolution_matrix(f):

    n = len(f)

    rows = []

    cols = []

    data = []

    for d, fd in enumerate(f):

        # rows from d to d + (n - d) - 1

        length = n - d

        if length <= 0:
            continue

        rows.extend(range(d, d + length))

        cols.extend(range(0, length))

        data.extend([fd] * length)

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    return A

"""
Optimise a non‑negative step‑function (probability distribution) to
minimise 2·n·max(conv(a,a))/(sum(a)^2).  The algorithm combines three
ideas that improve on the previous baseline:

1. **LP refinement** – for a fixed distribution f we solve the linear
   programme  max∑g  subject to  f*g ≤ M  (M = max(f*f))  and g ≥ 0.
   If the optimal ∑g exceeds 1 we can move a little towards that g and
   strictly lower the objective.  The LP is built with a sparse Toeplitz
   matrix and solved with ``scipy.optimize.linprog`` (highs).

2. **Exact sub‑gradient exponential descent** – instead of the soft‑max
   used previously we compute the true sub‑gradient of the max‑convolution
   (all indices whose value is within a tiny tolerance of the maximum) and
   perform a mirror‑descent step on the simplex (Kullback‑Leibler geometry).

3. **Greedy local moves** – pairwise mass transfers and a cheap simulated‑annealing
   walk finish the optimisation.

The routine respects the supplied time budget (default 1000 s) and
returns a plain Python list; ``evaluate_sequence`` will perform the final
normalisation, clipping, etc.
"""

# Helper: projection onto simplex (used only for safety)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Convolution utilities
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2n‑1) using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a normalised probability vector)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Discrete symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Discrete asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from each side, controlled by λ."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base (U is uniform)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# LP improvement (requires SciPy)
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _build_convolution_matrix(f: np.ndarray) -> _sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) == conv(f,g) (full convolution).
    The matrix is lower‑triangular Toeplitz: each diagonal d (0 ≤ d < n)
    contains the constant f[d].
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for d, fd in enumerate(f):
        length = n - d
        if length <= 0:
            continue
        rows.extend(range(d, d + length))
        cols.extend(range(0, length))
        data.extend([fd] * length)
    A = _sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)
    return A


def _lp_refine(a: np.ndarray, deadline: float,
               max_iter: int = 8, step: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinements as described in the statement.
    Returns the improved vector and its objective.
    """
    if not _SCIPY_AVAILABLE:
        return a, _objective(a)

    best_seq = a.copy()
    best_score = _objective(a)

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        n = a.shape[0]
        conv = _fft_self_convolve(a)
        M = conv.max()

        A = _build_convolution_matrix(a)

        c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  → minimise -sum(g)
        b = np.full(2 * n - 1, M, dtype=np.float64)  # conv(f,g) ≤ M

        res = _opt.linprog(c, A_ub=A, b_ub=b,
                           bounds=[(0, None)] * n,
                           method='highs',
                           options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        # If we cannot increase the total mass while respecting the bound, stop.
        if sum_g <= 1.0 + 1e-12:
            break

        # Scale to a proper probability vector.
        g = g / sum_g

        # Convex combination (step ∈ (0,1]).
        a = (1.0 - step) * a + step * g
        a = a / a.sum()

        cur = _objective(a)
        if cur < best_score - 1e-14:
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] LP refine improved to {best_score:.6f}")

    return best_seq, best_score

# Exact sub‑gradient exponential descent
def _exponential_subgradient_descent(a: np.ndarray,
                                     deadline: float,
                                     best_seq: np.ndarray,
                                     best_score: float,
                                     eta0: float = 0.6,
                                     eval_interval: int = 150,
                                     patience: int = 2500,
                                     eps_active: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true objective.
    Uses the exact sub‑gradient (all indices within eps_active of the max)
    and the KL‑projection (exponentiated‑gradient) to stay on the simplex.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # active set: indices whose value is within eps_active (absolute) of the max
        active = np.where(conv >= max_c - eps_active)[0]

        # ---- sub‑gradient ----
        grad = np.full_like(a, len(active) * a)            # the +a part from each active index
        for s in active:
            lo = max(0, s - (n - 1))
            hi = min(s, n - 1)
            i_idx = np.arange(lo, hi + 1, dtype=np.int32)
            j_idx = s - i_idx
            grad[i_idx] += a[j_idx]

        # ---- mirror‑descent (exponentiated‑gradient) update ----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= total

        # ---- occasional evaluation ----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] sub‑grad descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # ---- jitter if stuck ----
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Pairwise swap local refinement (identical to the baseline)
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep the move only if it improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing local search (identical to the baseline)
def _random_local_search(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         init_temp: float = 1e-2,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by re‑normalisation."""
    n = len(seq)
    if target_len == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, target_len, dtype=np.float64)
    y_new = np.interp(x_new, x_old, seq)
    y_new = np.maximum(y_new, 0.0)
    if y_new.sum() == 0.0:
        y_new = np.full_like(y_new, 1.0 / target_len)
    else:
        y_new /= y_new.sum()
    return y_new

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s)
    and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0               # a little safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Generate a diverse pool of seed vectors (moderate lengths)
    lengths = [1200, 1500, 1800, 2100]               # chose lengths that keep LP cheap
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.5, 6),                # 0.01 … 0.316
            np.logspace(-0.5, 0.0, 6),               # 0.316 … 1
            np.logspace(0.0, 0.5, 5),                # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)               # 0, 0.25,…,1

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families (symmetric beta, edge‑weighted, sine‑weighted,
        # exponential) mixed with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            expo = _distribution_exponential(n, p)  # λ ~ p for convenience

            for base in (beta, edge, sine, expo):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # asymmetric beta family (randomly sampled)
        for _ in range(5):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # keep only the most promising seeds overall (size limited)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. If a previous construction is available, add it to the pool
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard in case nothing was generated (should never happen)
    if overall_best_seq is None:
        n = 1500
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine the most promising seeds:
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] refining {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed * 0.9   # a little slack for the final polishing steps
            a = seed_seq.copy()

            # a) LP refinement (fast, improves the large‑scale structure)
            a, sc_lp = _lp_refine(a, sub_dead,
                                  max_iter=8,
                                  step=0.3)
            if sc_lp < overall_best_score:
                overall_best_score, overall_best_seq = sc_lp, a.copy()
                print(f"[{time.time():.1f}] after LP seed {idx} new best {overall_best_score:.6f}")

            # b) Exact sub‑gradient exponential descent
            a, sc_des = _exponential_subgradient_descent(
                a, sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                eval_interval=120,
                patience=2500,
                eps_active=1e-12)
            if sc_des < overall_best_score:
                overall_best_score, overall_best_seq = sc_des, a.copy()
                print(f"[{time.time():.1f}] after sub‑grad seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Greedy pairwise refinement on the current best
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5. Final stochastic local search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    # 6. Upsample to a longer length (if time permits) and polish again
    final_len = 6000
    if time.time() < deadline - 30.0 and len(overall_best_seq) < final_len:
        print(f"[{time.time():.1f}] upsampling to length {final_len}")
        a_up = _upsample(overall_best_seq, final_len)
        # a brief polish (no LP – it would be too expensive at this size)
        a_up, sc_up = _exponential_subgradient_descent(
            a_up, deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            eval_interval=150,
            patience=2000,
            eps_active=1e-12)
        if sc_up < overall_best_score:
            overall_best_score, overall_best_seq = sc_up, a_up.copy()
            print(f"[{time.time():.1f}] after upsample polish new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
