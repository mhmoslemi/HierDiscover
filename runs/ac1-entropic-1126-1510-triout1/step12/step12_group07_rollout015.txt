def _random_logit_hill_climb(a0, deadline, best_seq, best_score):

    n = a0.shape[0]

    scale = 0.5

        # number of proposals per scale, maybe 200 for each scale

                break

"""
Hybrid random‑seed + smooth‑max Adam + local polishing optimizer.

The routine builds a diverse pool of analytic and random step‑functions
(uniform, arcsine, β‑distributions, power‑law, Gaussian, sparse, etc.),
keeps the best few, and then refines them with

  1. Adam optimisation on a log‑sum‑exp smooth‑max surrogate.
  2. Sub‑gradient multimax descent.
  3. Stochastic log‑it hill‑climbing.
  4. A greedy mass‑transfer walk.

All vectors are kept on the probability simplex (non‑negative,
sum = 1) – this makes the exact objective
    2·n·max(conv(a,a))/sum(a)² = 2·n·max(conv)
independent of scaling.
The function respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of a weighted max‑convolution surrogate:
        grad = 2·a ⋆ w   (where w sums to 1)
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (assumes sum>0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    logw = (p - 1) * np.log(np.clip(x, eps, 1.0 - eps)) \
         + (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0 - eps))
    logw -= np.max(logw)                     # avoid overflow
    w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isinf(w_sum):
        w = np.full(n, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    logw = (alpha - 1) * np.log(np.clip(x, eps, 1.0 - eps)) \
         + (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0 - eps))
    logw -= np.max(logw)
    w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isinf(w_sum):
        w = np.full(n, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals

def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 2000.0,
                   growth_factor: float = 1.015,
                   lr: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 500000,
                   eval_interval: int = 150) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    The surrogate is f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)).
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # probability vector on simplex
        a = _softmax(z)

        # smooth‑max weights
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient w.r.t. a for the surrogate
        grad_a = _gradient_from_weights(a, w)

        # chain rule to obtain gradient w.r.t. logits z
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # temperature schedule
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.8,
                                 max_iter: int = 50000,
                                 eval_interval: int = 250) -> Tuple[np.ndarray, float]:
    """
    Classic multimax sub‑gradient descent (exponentiated‑gradient on the simplex).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_logit_hill_climb(a0: np.ndarray,
                            deadline: float,
                            best_seq: np.ndarray,
                            best_score: float,
                            rng: np.random.Generator,
                            eval_interval: int = 100) -> Tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing in logit space.
    Temperature (step size) is reduced geometrically.
    """
    n = a0.shape[0]
    z = np.log(np.maximum(a0, 1e-300))
    scale = 0.5
    it = 0
    while time.time() < deadline and scale > 1e-4:
        for _ in range(200):
            if time.time() >= deadline:
                break
            it += 1
            z_new = z + rng.normal(scale=scale, size=n)
            a_new = _softmax(z_new)
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a_new.copy()
                z = z_new
                print(f"[{time.time():.1f}] hill‑climb it={it} new best {best_score:.6f}")
        scale *= 0.9
    return best_seq, best_score


# ----------------------------------------------------------------------
def _greedy_mass_transfer(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """
    Simple pairwise mass‑transfer walk.  Randomly chooses a source index i
    (probability proportional to a[i]) and a destination j, transfers a
    random fraction of a[i] and accepts only improving moves.
    """
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src, p=a[src] / a[src].sum())
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random()          # can move the whole mass
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] mass‑transfer i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s) and returns
    the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 0.  Choose a set of lengths (moderate sizes keep each iteration cheap)
    lengths = [2048, 4096, 8192, 12000]

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Broad seeding across all lengths
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")
        for cand in _generate_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
        # retain only the most promising seeds seen so far
        candidates.sort(key=lambda x: x[0])
        candidates = candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Bring in a previously found solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            sp = prev.sum()
            if sp == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                sp = prev.sum()
            prev = prev / sp
            sc = _objective(prev)
            candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback in the unlikely event we have no candidate at all
    if overall_best_seq is None:
        n = lengths[0]
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam smooth‑max descent on a handful of top seeds
    candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidates[:10]]
    if overall_best_seq is not None and not any(np.allclose(overall_best_seq, s) for s in top_seeds):
        top_seeds.append(overall_best_seq)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a rough equal slice of the remaining budget
        slice_time = min(30.0, remaining / max(1, len(top_seeds) - idx))
        sub_deadline = time.time() + slice_time
        print(f"[{time.time():.1f}] Adam optimisation on seed {idx+1}/{len(top_seeds)} (len={seed_seq.shape[0]}) – allocated {slice_time:.1f}s")
        z0 = np.log(np.maximum(seed_seq, 1e-300))
        overall_best_seq, overall_best_score = _adam_optimize(
            z0,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            alpha0=1.0,
            alpha_max=2000.0,
            growth_factor=1.015,
            lr=0.15,
            max_iter=500000,
            eval_interval=150,
        )
        print(f"[{time.time():.1f}] after Adam best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Multimax sub‑gradient polishing (if we still have time)
    if time.time() < deadline - 5.0:
        sub_deadline = time.time() + (deadline - time.time()) * 0.45
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.8,
            max_iter=50000,
            eval_interval=250,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Stochastic hill‑climbing in logit space
    if time.time() < deadline - 5.0:
        sub_deadline = time.time() + (deadline - time.time()) * 0.35
        overall_best_seq, overall_best_score = _random_logit_hill_climb(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        print(f"[{time.time():.1f}] after hill‑climb best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final greedy pairwise mass‑transfer walk (cheap but often yields tiny gains)
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _greedy_mass_transfer(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        print(f"[{time.time():.1f}] after final swaps best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
