def _lp_step(a, t, deadline):
    # a normalized to sum=1
    n = len(a)
    conv = _fft_self_convolve(a)   # length 2n-1
    M = conv.max()
    # Build A matrix
    # Use loops to fill A.
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    # Solve LP maximize sum(b)
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a # fallback
    b = res.x
    # Scale b to sum=1
    if b.sum() <= 0:
        b = a
    else:
        b = b / b.sum()
    # Mix
    new_a = (1 - t) * a + t * b
    # Ensure sum 1 again
    new_a = new_a / new_a.sum()
    return new_a

"""
Hybrid search for a low‑max‑convolution step function.

The algorithm consists of four stages:

1. **Broad seeding** – generate a large variety of analytic families and
   many random Dirichlet / sparse patterns for a moderate length (≈2000).
   The best few are kept as seeds.

2. **Adam smooth‑max descent** – each seed is refined by optimizing the
   smooth‑max surrogate of ``max(conv(a,a))`` using Adam on the soft‑max
   parametrisation.  The temperature is increased gradually so the surrogate
   converges to the true max.

3. **LP‑based improvement** – for the current best vector we solve a linear
   program that maximises the total mass of a secondary vector `b` under the
   constraint ``conv(a,b) ≤ max(conv(a,a))``.  Mixing the original vector with
   `b` (with a small step size) provably never increases the maximum
   convolution and often strictly reduces it.

4. **Polishing** – a short multimax sub‑gradient descent followed by a greedy
   pairwise‑swap local search are applied.

All vectors stay on the probability simplex (non‑negative, sum = 1) which
matches the behaviour of ``evaluate_sequence`` (it rescales negative values
to 0 and caps large ones).  The routine respects a user‑supplied time budget
(and uses a small safety margin) and returns the best sequence found.
"""

import time

from typing import List, Tuple, Optional

import numpy as np

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective used in the competition (sum(a) is assumed >0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families (all return a probability vector, i.e. sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """
    Produce a rich collection of seeds for a given length.  The function
    returns probability vectors (sum = 1) and never includes a zero‑sum vector.
    """
    cand: List[np.ndarray] = []

    # Simple analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_gaussian(n))

    # Symmetric beta family (grid of exponents)
    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric beta draws
    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law families
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # Exponential families
    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes
    for block_frac in (0.05, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac))

    # Random Dirichlet draws
    for _ in range(30):
        alpha = 0.1 + 0.9 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # Random sparse patterns
    for _ in range(30):
        cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # Completely random probability vectors
    for _ in range(20):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        if v.sum() == 0.0:
            v[0] = 1.0
        v /= v.sum()
        cand.append(v)

    # Ensure non‑negativity and unit sum (defensive)
    cleaned = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s == 0.0:
            continue
        cleaned.append(c / s)

    return cleaned

def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) with respect to a.
    Implements 2·a⋆w via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 500.0,
                   growth_factor: float = 1.02,
                   lr: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 200000,
                   eval_interval: int = 250) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best sequence seen (as a probability vector) and its score.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # probability vector on the simplex
        a = _softmax(z)

        # convolution and smooth‑max weights
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient wrt a (exact for the surrogate)
        grad_a = _gradient_from_weights(a, w)

        # chain rule through soft‑max
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # temperature schedule
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 30000,
                                 eval_interval: int = 300) -> Tuple[np.ndarray, float]:
    """
    Classic multimax sub‑gradient descent (exponentiated‑gradient on the simplex).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk accepting only improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


def _lp_step(a: np.ndarray,
             mix_factor: float,
             deadline: float) -> Tuple[np.ndarray, bool]:
    """
    Solve the LP
        maximise   Σ_j b_j
        subject to  a * b ≤ M   (component‑wise)
                   b_j ≥ 0
    where M = max(conv(a,a)).
    Return a new vector ``(1−mix)·a + mix·b`` if it strictly improves the
    max‑convolution; otherwise return the original ``a`` unchanged.
    """
    if not _SCIPY_AVAILABLE:
        return a, False

    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()

    rows = 2 * n - 1
    # Build matrix A where A[k, j] = a_{k‑j} (zero when out of range)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a

    c = -np.ones(n)                     # minimise –Σ b_j  == maximise Σ b_j
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Linprog with the robust “highs” method
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:  # pragma: no cover
        return a, False

    if not res.success:
        return a, False

    b = res.x
    if b.sum() == 0.0:
        return a, False

    b = b / b.sum()                     # normalise to the simplex
    new_a = (1.0 - mix_factor) * a + mix_factor * b
    new_a = new_a / new_a.sum()         # re‑normalise (numerical safety)

    new_M = _fft_self_convolve(new_a).max()
    if new_M < M * (1.0 - 1e-12):
        return new_a, True
    else:
        return a, False

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine follows the four‑stage strategy described in the module
    docstring and respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a single moderate length – large enough to beat the 2‑uniform bound,
    #     small enough for the LP step to stay tractable.
    n = 2000

    # ------------------------------------------------------------------
    # 2.  Broad seeding
    print(f"[{time.time():.1f}] generating seeds for length {n}")
    seed_vectors = _initial_candidates(n, rng)

    # Evaluate all seeds and keep the best few
    scored: List[Tuple[float, np.ndarray]] = []
    for cand in seed_vectors:
        if time.time() >= deadline:
            break
        score = _objective(cand)
        scored.append((score, cand))
        if score < 1.55:  # early sensible threshold to keep the log tidy
            print(f"[{time.time():.1f}] seed score={score:.6f}")

    scored.sort(key=lambda x: x[0])
    top_seeds = [v for _, v in scored[:10]]
    overall_best_seq = top_seeds[0].copy()
    overall_best_score = scored[0][0]
    print(f"[{time.time():.1f}] initial best score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Adam smooth‑max descent on the best seeds
    remaining = deadline - time.time()
    if remaining > 30.0:
        print(f"[{time.time():.1f}] starting Adam optimisation on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            # allocate a time slice for each seed (minimum 6 s)
            slice_time = min(45.0, (deadline - time.time()) / max(1, len(top_seeds) - idx))
            sub_deadline = time.time() + slice_time

            z0 = np.log(np.maximum(seed_seq, 1e-300))
            best_seq, best_score = _adam_optimize(
                z0,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha0=1.0,
                alpha_max=500.0,
                growth_factor=1.02,
                lr=0.12,
                max_iter=200000,
                eval_interval=200,
            )
            if best_score < overall_best_score:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] after Adam new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  LP‑based improvement (if SciPy is available)
    if _SCIPY_AVAILABLE:
        print(f"[{time.time():.1f}] LP‑based refinement phase")
        mix = 0.15          # start with a moderate mixing factor
        while time.time() < deadline - 5.0:
            new_seq, improved = _lp_step(overall_best_seq, mix, deadline)
            if not improved:
                # reduce the mixing factor if we cannot improve
                mix *= 0.7
                if mix < 1e-4:
                    break
                continue
            # accept the improvement
            overall_best_seq = new_seq
            overall_best_score = _objective(overall_best_seq)
            print(f"[{time.time():.1f}] LP step improved score to {overall_best_score:.6f}")
            # gradually decay the step size
            mix *= 0.9
    else:
        print(f"[{time.time():.1f}] SciPy not available – skipping LP stage")

    # ------------------------------------------------------------------
    # 5.  Multimax sub‑gradient polishing
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + remaining * 0.5
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            max_iter=20000,
            eval_interval=200,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final greedy pairwise‑swap walk
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Return a plain Python list as required by the evaluation harness.
    return overall_best_seq.tolist()
