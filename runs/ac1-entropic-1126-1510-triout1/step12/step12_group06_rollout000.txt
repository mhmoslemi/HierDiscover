"""
Hybrid optimiser for the step‑function max‑convolution problem (enhanced).

Strategy
--------
1. Generate a diverse set of seed distributions for a moderate length (≈3000–4000)
   using analytic families (uniform, arcsine, Beta, Dirichlet, sparse spikes, …).
2. Run a long Adam‑style exponentiated‑gradient descent on the smooth‑max
   surrogate of the max‑convolution.  The temperature α is increased slowly
   up to a large value (≥300) so that the optimisation focuses on the true
   objective.
3. Upsample the currently best distribution to a larger length (≈8000–12000)
   and continue the Adam descent – this transfers the shape while giving
   more degrees of freedom.
4. Refine the best candidate with classic multimax (sub‑gradient) descent.
5. Finish with an aggressive greedy pair‑wise mass‑transfer local search
   (moving mass from indices with large gradient to those with small
   gradient).  All updates keep the vector on the probability simplex.

The routine respects the supplied time budget (default 1000 s) and prints
intermediate scores.  The final list of non‑negative heights is returned.
"""


import time

from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2·n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(a⋆a)_i  w.r.t. a.
    Implemented as 2·a⋆′w, where “⋆′” is the cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 2).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The vector is assumed to be non‑negative; sum is usually 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families (used for seeding)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0))
               + (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Very sparse random pattern."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Two symmetric spikes of equal mass."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _initial_candidates_extended_fine(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Produce a rich collection of seed vectors for a given length."""
    cand: List[np.ndarray] = []

    # Basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric Beta – log‑spaced exponents
    for p in np.geomspace(0.01, 5.0, num=12):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Random asymmetric Beta draws
    for _ in range(6):
        alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law increasing
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # Exponential families
    for lam in (0.001, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes
    for bf in (0.03, 0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # Dirichlet draws with varied concentration
    for a in (0.1, 0.3, 0.7, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # Random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.03))

    # Completely random probability vectors (few of them)
    for _ in range(4):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    return cand


# ----------------------------------------------------------------------
def _upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    up = np.interp(x_new, x_old, arr)
    # Clip tiny negatives due to interpolation noise
    up = np.maximum(up, 0.0)
    if up.sum() == 0.0:
        up = np.full_like(up, 1.0 / new_n)
    else:
        up /= up.sum()
    return up

def _smooth_max_adam_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             eta0: float = 0.8,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             alpha0: float = 0.1,
                             alpha_mul: float = 1.15,
                             alpha_max: float = 300.0,
                             max_iter: int = 80000,
                             eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth‑max surrogate

        f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a))

    The temperature α starts from ``alpha0`` and grows geometrically.
    """
    n = a.shape[0]
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # full self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate (via cross‑correlation)
        grad = _gradient_from_weights(a, w)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update (exponentiated gradient)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # projection onto the probability simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # temperature schedule – increase every few hundred iterations
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.7,
                                 max_iter: int = 50000,
                                 eval_interval: int = 250) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent for  max(conv(a,a)).
    The sub‑gradient is the average gradient over all active (max‑attaining) indices.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """
    Greedy pairwise mass‑transfer.  Use the sub‑gradient to decide a promising
    source (large positive gradient) and destination (large negative gradient).
    """
    n = a.shape[0]
    while time.time() < deadline:
        # compute the sub‑gradient using the active set of max indices
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt
        grad = _gradient_from_weights(a, w)

        # source = index with biggest positive gradient (decrease it)
        i = int(np.argmax(grad))
        # destination = index with most negative gradient (increase it)
        j = int(np.argmin(grad))
        if i == j:
            # fall back to a random move if gradients are uniform
            src = np.where(a > 1e-12)[0]
            if src.size == 0:
                break
            i = int(np.random.choice(src))
            j = int(np.random.randint(n - 1))
            if j >= i:
                j += 1

        # transfer a fraction of source mass
        delta = a[i] * np.random.rand() * 0.35  # up to 35 % of source
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects ``budget_s`` seconds and returns the best coefficient
    list found (plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Small‑scale seeding (n ≈ 3000–4000)
    small_lengths = [2800, 3200, 3600]
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    candidate_pool: List[Tuple[float, np.ndarray]] = []

    print(f"[{time.time():.1f}] Starting seed generation (small lengths)")
    for n in small_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating candidates for n={n}")
        seeds = _initial_candidates_extended_fine(n, rng)

        # add a few extra Dirichlet draws with random concentration
        for _ in range(6):
            alpha = rng.uniform(0.05, 2.0)
            seeds.append(_distribution_dirichlet(n, rng, alpha=alpha))

        for cand in seeds:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best n={n} score={sc:.6f}")
            candidate_pool.append((sc, cand.copy()))

    # keep only a limited number of promising seeds
    candidate_pool.sort(key=lambda x: x[0])
    candidate_pool = candidate_pool[:8]

    # ------------------------------------------------------------------
    # 2. Incorporate previous solution if present.
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.size > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s == 0.0:
                    prev_arr = _distribution_uniform(prev_arr.shape[0])
                    s = prev_arr.sum()
                prev_arr = prev_arr / s
                sc = _objective(prev_arr)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev_arr.copy()
                    print(f"[{time.time():.1f}] previous solution improved score={sc:.6f}")
                candidate_pool.append((sc, prev_arr.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Adam‑style smooth‑max descent on the best seeds
    if overall_best_seq is None:
        # fallback to uniform if everything failed
        n = small_lengths[0]
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)

    print(f"[{time.time():.1f}] Starting Adam descent on {len(candidate_pool)} seeds")
    seeds_for_adam = [overall_best_seq.copy()] + [seq for _, seq in candidate_pool[:4]]
    for seed_seq in seeds_for_adam:
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + max(5.0, (deadline - time.time()) / max(1, len(seeds_for_adam)))
        seq_out, score_out = _smooth_max_adam_descent(
            seed_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.8,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha0=0.1,
            alpha_mul=1.18,
            alpha_max=350.0,
            max_iter=90000,
            eval_interval=250,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

    if time.time() < deadline:
        current_n = overall_best_seq.shape[0]
        target_n = max(8000, int(current_n * 1.5))

        up_seq = up_seq / up_seq.sum()

        seq_out, score_out = _smooth_max_adam_descent(
            up_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.8,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha0=0.1,
            alpha_mul=1.2,
            alpha_max=400.0,
            max_iter=110000,
            eval_interval=300,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] after upsample Adam new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Classic multimax (sub‑gradient) refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Starting multimax refinement")
        seq_out, score_out = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.7,
            max_iter=60000,
            eval_interval=300,
        )
        if score_out < overall_best_score:
            overall_best_score = score_out
            overall_best_seq = seq_out.copy()
            print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final greedy pairwise polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final greedy pairwise polishing")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 7. Normalise once more and report final score
    overall_best_seq = np.maximum(overall_best_seq, 0.0)
    s = overall_best_seq.sum()
    if s == 0.0:
        overall_best_seq = np.full_like(overall_best_seq, 1.0 / overall_best_seq.shape[0])
        s = overall_best_seq.sum()
    overall_best_seq = overall_best_seq / s

    final_score = _objective(overall_best_seq)
    print(f"[{time.time():.1f}] Final best score: {final_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
