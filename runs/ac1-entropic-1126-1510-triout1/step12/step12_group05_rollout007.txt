def _adam_descent(a0, deadline, best_seq, best_score, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=200000, eval_interval=500):
    n = a0.shape[0]
    y = np.log(np.maximum(a0, 1e-30))  # log of a
    # Actually we can start y = np.log(a0) after adding epsilon.
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # compute a from y via softmax
        # To avoid overflow, subtract max(y)
        max_y = np.max(y)
        exp_y = np.exp(y - max_y)
        a = exp_y / np.sum(exp_y)
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # temperature (alpha)
        # maybe linear schedule to final value alpha_end
        # we can compute fraction = t / max_iter
        # alpha = alpha_start + (alpha_end - alpha_start) * fraction
        # choose start=0.1,end=120
        alpha = 0.1 + (120.0 - 0.1) * t / max_iter
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        grad_a = _gradient_from_weights(a, w)   # shape n
        # convert gradient to y-space
        # grad_y_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr_t = lr / (np.sqrt(v_hat) + eps)
        y = y - lr_t * m_hat
        if t % eval_interval == 0:
            # evaluate objective exactly
            cur = _objective(a) # we already have a
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} cur={cur:.6f}")
    return best_seq, best_score
```

"""
Hybrid multi‑start optimiser with Adam‑based soft‑max descent.

The algorithm works as follows:

1. **Seed generation** – For a few problem sizes (4000‒20000) a rich
   collection of analytic families (uniform, arcsine, Beta, Dirichlet,
   power‑law, exponential, sparse, two‑block, sine) and many random
   Dirichlet draws are created.  All candidates are normalised to sum = 1
   and their exact objective `2·n·max(conv(a,a))` is measured.

2. **Elite selection** – The best few candidates (across all lengths)
   are kept.

3. **Adam‑based soft‑max optimisation** – Each elite seed is further
   refined by a mirror‑descent on the simplex using the
   log‑parameterisation `a = softmax(y)`.  A temperature‑controlled
   soft‑max over the convolution entries yields a smooth approximation
   of the true max; the temperature is increased linearly from a small
   start value (≈0.1) to a large final value (≈120).  Adam’s adaptive
   steps are employed on the dual variables `y`.  The loop runs until
   the per‑seed time budget expires, continuously updating the global
   best sequence.

4. **Greedy polishing** – A cheap random mass‑transfer walk (pairwise
   swaps) is performed on the current best sequence until the overall
   time budget runs out.

The routine respects the provided time budget (default 1000 s) and
returns the best sequence (a plain list of floats) it discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities (kept for compatibility, not used in the main flow)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient with respect to a (soft‑max weights)
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, sum(w)=1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))

    # symmetric beta on a fine grid
    for p in np.linspace(0.05, 1.0, 12):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric beta
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power law
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential
    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # sparse random draws
    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws with various concentrations
    for a in (0.1, 0.3, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # a few pure random Dirichlet samples (different α)
    for _ in range(6):
        a = 10 ** rng.uniform(-3, 2)   # wide range
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    return cand

# Adam‑based soft‑max optimisation ---------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float,
                  lr: float = 0.07,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  max_iter: int = 400000,
                  eval_interval: int = 500,
                  alpha_start: float = 0.1,
                  alpha_end: float = 120.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation of a.
    A temperature‑controlled soft‑max (alpha) provides a smooth surrogate
    of the max‑convolution entry.
    """
    n = a0.shape[0]
    # initialise log‑probabilities safely
    y = np.log(np.maximum(a0, 1e-30))

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        # a = softmax(y)
        max_y = np.max(y)
        exp_y = np.exp(y - max_y)
        a = exp_y / exp_y.sum()

        # convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linearly increase temperature
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t a (sub‑gradient of the soft‑max surrogate)
        grad_a = _gradient_from_weights(a, w)

        # map to gradient w.r.t y (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # Adam update on y
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + eps)
        y = y - step * m_hat

        if t % eval_interval == 0:
            cur = 2.0 * n * max_conv   # exact objective (sum(a)=1)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} score={best_score:.6f}")

    # final a from the last y
    max_y = np.max(y)
    exp_y = np.exp(y - max_y)
    a = exp_y / exp_y.sum()
    return a, _objective(a) if best_score == float("inf") else best_score, best_seq

# Random pairwise swap polishing ----------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap random mass‑transfer walk.  Randomly moves a fraction of mass
    from one index to another and keeps the change only if the true
    objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        # move up to 30 % of source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Entry point for the optimiser.  Returns the best sequence (list of floats)
    found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # 1. Choose problem lengths (larger lengths give tighter bounds)
    lengths = [4000, 8000, 12000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    elite = []  # list of (score, seq) tuples

    # --------------------------------------------------------------
    # 2. Seed generation & quick evaluation
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length = {n} ---")
        candidates = _initial_candidates_extended(n, rng)
        for cand in candidates:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep only the very best few across all lengths examined so far
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]

    # --------------------------------------------------------------
    # 3. Incorporate a previously known solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:12]
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4. Fallback (should never be needed)
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Main refined optimisation (Adam‑based soft‑max descent)
    remaining = deadline - time.time()
    if remaining > 30.0:
        # take the very best elite seeds
        seeds = [seq for _, seq in sorted(elite, key=lambda x: x[0])[:5]]
        per_seed = max(5.0, remaining / (len(seeds) * 2.0))

        for idx, seed_seq in enumerate(seeds, 1):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            print(f"[{time.time():.1f}] Adam optimisation on seed {idx}/{len(seeds)} (len={len(seed_seq)})")
            seq_opt, score_opt, best_seq_candidate = _adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.07,
                max_iter=500000,
                eval_interval=400,
                alpha_start=0.1,
                alpha_end=120.0,
            )
            # _adam_descent returns the (possibly updated) best sequence and its score
            if score_opt < overall_best_score:
                overall_best_score = score_opt
                overall_best_seq = seq_opt.copy()
                print(f"[{time.time():.1f}] After Adam new global best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final greedy polishing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # --------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
```

This implementation expands the search space, employs an Adam‑based soft‑max descent for more precise optimisation, and finishes with a cheap greedy polishing step, all while respecting the 1000‑second budget.
