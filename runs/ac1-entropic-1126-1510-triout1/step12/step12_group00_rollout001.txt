def improve_with_lp(a, time_limit, max_iter=5):
    best = a.copy()
    best_score = _objective(a)
    start = time.time()
    for itr in range(max_iter):
        if time.time() > time_limit:
            break
        # compute convolution, M.
        conv = _fft_self_convolve(a)
        M = conv.max()
        # solve LP
        g = solve_lp_max_sum(a, M)
        if g is None:
            break
        # combine
        t = 0.5  # mixing weight
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        # renormalize
        sum_a = a_new.sum()
        if sum_a == 0:
            a_new = np.full_like(a_new, 1.0 / len(a_new))
        else:
            a_new /= sum_a
        cur_score = _objective(a_new)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a_new.copy()
            a = a_new
            print(f"LP iter {itr} improvement: {best_score}")
        else:
            # no improvement, break
            break
    return best, best_score

def _solve_lp_max_sum(f, M):
    n = len(f)
    # Build constraint matrix A via scipy.sparse.diags
    import scipy.sparse as sp
    diag_data = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(diag_data, offsets, shape=(2*n-1, n), format='csc')
    # bound vector
    b = np.full(2*n-1, M)
    # objective c = -1 for maximize sum(g)
    c = -np.ones(n)
    # bounds
    bounds = [(0, None)] * n
    res = sp.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        print("LP failed", res.message)
        return None
```

"""
Propose a step‑function (list of heights) that minimises ``evaluate_sequence``.

Algorithm
---------

1️⃣ **Generate diverse seeds** – analytic families (uniform, arcsine,
   symmetric beta, edge‑weighted, edge‑inverse‑power, sine‑weighted) and a few
   random Dirichlet / sparse vectors are created for several moderate lengths
   (1500‑3000).  The best few (by the exact objective `2·n·max(conv)`) are kept.

2️⃣ **Mirror‑descent refinement** – each seed is improved by an exponentiated‑gradient
   (mirror) descent that works on the exact objective.  Only the convolution
   entries close to the current maximum are taken as an active set, which makes
   the sub‑gradient cheap to compute.

3️⃣ **Linear‑programming boost** – for a candidate `a` (already normalised to sum = 1)
   we compute `M = max(conv(a,a))`.  The LP

        maximise   Σ g_i
        subject to  conv(a,g) ≤ M      (2·n‑1 linear constraints)
                    g_i ≥ 0

   is solved (SciPy HiGHS).  The optimal `g` has a larger total mass while keeping
   the convolution bound unchanged, therefore mixing `a` with `g` and renormalising
   strictly reduces the objective.  A few such LP steps are applied.

4️⃣ **Local clean‑up** – a cheap greedy pairwise mass‑transfer followed by a
   simulated‑annealing style random walk is performed while time permits.

All steps respect the overall time budget (default 1000 s).  The final
distribution is returned as a plain Python list; ``evaluate_sequence`` will
handle the harmless clipping and scaling automatically.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

import scipy.optimize  # linear programming (HiGHS)
import scipy.sparse as sp  # sparse matrix for convolution constraints


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a real vector using an FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())        # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1.
    The function is scale‑invariant, so we renormalise internally.
    """
    s = a.sum()
    if s == 0.0:
        return float("inf")
    a = a / s
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family: w_i ∝ (i+0.5)^(−p) + (n−i−0.5)^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p) + j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

# Mirror‑descent with momentum (exponentiated‑gradient) ----------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    active_frac: float = 0.03,
    max_active: int = 30,
    beta: float = 0.85,
    eval_interval: int = 200,
    patience: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact objective.
    Only convolution entries within ``active_frac`` of the current maximum are
    used to build a sub‑gradient.  Momentum (beta) and a 1/√t step size are used.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 200_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for idx in act:
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy random pairwise mass transfer."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style local search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   subject to   conv(f,g) ≤ M,  g_i ≥ 0.
    Returns the optimal vector g (already non‑negative) or None on failure.
    """
    n = f.shape[0]
    # Build the Toeplitz convolution matrix A such that A @ g = conv(f,g)
    diag_data = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)               # non‑negative offsets only
    A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format="csc")
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)    # maximise sum(g)  → minimise -sum(g)
    bounds = [(0, None)] * n

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    # Failure is unlikely but we guard against it
    print("LP failed:", res.message)
    return None

    # 2️⃣  Mirror‑descent refinement on the most promising seeds
    if time.time() < deadline and top_seeds:
        per_seed = (deadline - time.time()) / max(1, len(top_seeds))
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            print(f"[{time.time():.1f}] start mirror descent on seed {idx}")
            a_opt, sc_opt = _subgradient_descent_momentum(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                active_frac=0.03,
                max_active=30,
                beta=0.85,
                eval_interval=200,
                patience=1500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Linear‑programming boost (few iterations)
    if time.time() < deadline and overall_best_seq is not None:
        a = overall_best_seq.copy()
        lp_iters = 4
        for it in range(lp_iters):
            if time.time() >= deadline:
                break
            conv = _fft_self_convolve(a)
            M = conv.max()
            g = _solve_lp_max_sum(a, M)
            if g is None:
                break
            # Convex combination (mixing weight 0.5) and renormalisation
            a = 0.5 * a + 0.5 * g
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / a.shape[0])
            else:
                a /= s
            cur_score = _objective(a)
            print(f"[{time.time():.1f}] LP step {it} score {cur_score:.6f}")
            if cur_score < overall_best_score - 1e-14:
                overall_best_score, overall_best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] new global best after LP {overall_best_score:.6f}")
            else:
                # No improvement – stop the LP loop early
                break

    # ------------------------------------------------------------------
    # 4️⃣  Greedy pairwise refinement
    if time.time() < deadline and overall_best_seq is not None:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final cheap simulated‑annealing style local search
    if time.time() < deadline and overall_best_seq is not None:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # Fallback: uniform distribution of a moderate size
        n = 2500
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
