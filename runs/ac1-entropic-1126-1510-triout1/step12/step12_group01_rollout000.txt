"""
Propose a non‑negative step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Random sparse mask** – For several lengths (up to 40 000) generate a
   binary mask with a moderate density (≈0.4‑0.6).  All non‑zero entries receive
   the same weight 1/m (m = number of ones) so the vector sums to 1.
   Convolution of such a mask is close to flat, giving an objective close to
   the theoretical lower bound 1.

2. **Multiple seeds** – For each length try a few deterministic densities and
   a handful of random densities.  Keep the best vector seen so far.

3. **Optional tiny local refinement** – If time permits a few rounds of
   pairwise mass transfers are attempted to shave off the last few
   micro‑improvements.

The whole routine respects the supplied time budget (default 1000 s) and
returns a plain ``list[float]``.  ``evaluate_sequence`` will perform the
final clipping/normalisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector `a` whose sum is 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple deterministic refinement: move a random fraction of mass from a
    source index to a destination index if it improves the objective.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline and trials < max_trials:
        trials += 1
        # choose a source with positive mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # choose a different destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest amount of mass
        delta = a[i] * rng.random() * 0.2
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The function respects the time budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Scan a modest set of lengths and densities
    lengths = [5000, 10000, 20000, 40000]
    fixed_densities = [0.4, 0.5, 0.6]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] trying length {n}")

        # deterministic densities
        for dens in fixed_densities:
            if time.time() >= deadline:
                break
            mask = rng.random(n) < dens
            m = mask.sum()
            if m < 2:
                continue
            a = np.zeros(n, dtype=np.float64)
            a[mask] = 1.0 / m                     # equal weights, sum = 1
            sc = _objective(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new best len={n} dens={dens:.2f} m={m} score={sc:.6f}")

        # a few random densities
        for _ in range(8):
            if time.time() >= deadline:
                break
            dens = rng.uniform(0.1, 0.9)
            mask = rng.random(n) < dens
            m = mask.sum()
            if m < 2:
                continue
            a = np.zeros(n, dtype=np.float64)
            a[mask] = 1.0 / m
            sc = _objective(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new best len={n} rnd_dens={dens:.3f} m={m} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform len={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Tiny local refinement if time permits (pairwise swaps)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] local refinement (pairwise swaps)")
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
