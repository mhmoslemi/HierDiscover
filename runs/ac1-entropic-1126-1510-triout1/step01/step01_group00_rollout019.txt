def _optimise_lp(n, deadline):
    a = _arcsine_distribution(n)  # sum=1
    best_seq = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()
    t = 0.5  # initial blending factor
    iter = 0
    while time.time() < deadline:
        iter += 1
        # compute convolution and M
        b = np.convolve(a, a)
        M = b.max()
        # Construct constraints matrix
        A = _conv_matrix_dense(a)  # shape (2n-1, n)
        # RHS vector b_ub = M repeated
        rhs = np.full(2*n-1, M, dtype=np.float64)
        # Objective: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        # Solve LP
        res = linprog(c, A_ub=A, b_ub=rhs, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            # fallback to gradient descent
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        # Scale to have L1=1 (same as a)
        g = g0 / sum_g0
        # Blend
        a_new = (1 - t) * a + t * g
        # Project onto simplex (as we keep sum=1)
        # Actually a_new sum = (1-t)*1 + t*1 = 1, so sum preserved. Ensure non-negativity.
        a_new = np.maximum(a_new, 0.0)
        # Evaluate
        cur_score = 2.0 * n * np.convolve(a_new, a_new).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"{time.time():.2f} n={n} iter={iter} score={best_score:.6f}")
        # update
        a = a_new
        # adapt t: maybe reduce if no improvement
        if iter % 5 == 0:
            t = max(0.05, t * 0.9)
    return best_seq, best_score

def _refine_subgradient(seq, deadline):
    a = seq.copy()
    best_seq = a.copy()
    best_score = 2*n*np.convolve(a,a).max()
    n = len(a)
    it = 0

        b = np.convolve(a, a)
        k = np.argmax(b)

def _subgradient_multi(a, deadline):
    n = len(a)
    best_seq = a.copy()
    best_score = 2*n*np.convolve(a,a).max()
    it = 0
    eta0 = 0.5
    while time.time()<deadline:
        it+=1
        # conv via FFT
        size = 2*n-1
        N = 1 << (size-1).bit_length()
        A = np.fft.rfft(a, n=N)
        b = np.fft.irfft(A*A, n=N)[:size]
        max_b = b.max()
        cur_score = 2*n*max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] iter={it} score={best_score:.6f}")
        # active set:
        eps = max_b * 0.01  # 1% tolerance
        active = np.where(b >= max_b - eps)[0]
        # compute gradient as sum of subgradients for each active k
        g = np.zeros_like(a)
        for k in active:
            idx = k - np.arange(n)
            mask = (idx >=0) & (idx < n)
            g[mask] += 2*a[idx[mask]]
        # step size
        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        # project onto simplex (sum to 1)
        a = _project_onto_simplex(a, s=1.0)
        # occasional jitter
        if it % 500 == 0:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)
    return best_seq, best_score

"""...
Algorithm:
- Start from the best known sequence (or an arcsine distribution) and normalise to unit sum.
- Phase 1: projected gradient descent on the squared‑deviation from the constant autocorrelation.
  The gradient is computed by a fast FFT‑based convolution, and after each step the vector
  is projected onto the probability simplex.
- Phase 2: a classic sub‑gradient descent on the max‑convolution (the original objective),
  again with FFT convolution and simplex projection.
Both phases run as long as time permits (up to the 1000 s budget).  The best sequence
encountered is returned as a plain Python list.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _arcsine_distribution(n: int) -> np.ndarray:
    """Length‑n probability vector approximating the arcsine law."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _fft_convolution(x: np.ndarray, y: np.ndarray, N: int) -> np.ndarray:
    """Return the linear convolution x*y using FFT of size N (real inputs)."""
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)
    # truncate to the exact linear length
    return conv[: x.size + y.size - 1]


def _ls_gradient_step(a: np.ndarray, C: float, N: int, eta: float) -> tuple[np.ndarray, np.ndarray]:
    """One projected gradient step for the least‑squares objective
       sum_k ( (a*a)[k] - C )**2.
       Returns the new vector and the current self‑convolution.
    """
    n = a.size
    # self‑convolution a*a
    b = _fft_convolution(a, a, N)
    diff = b - C

    # gradient = 4 * conv(diff, a_rev)
    a_rev = a[::-1]
    grad = 4.0 * _fft_convolution(diff, a_rev, N)[n - 1 : n - 1 + n]

    a_new = a - eta * grad
    a_new = _project_onto_simplex(a_new, s=1.0)
    return a_new, b


def _subgradient_step(a: np.ndarray, N: int, eta: float) -> tuple[np.ndarray, np.ndarray]:
    """One projected sub‑gradient step for the original max‑convolution objective."""
    n = a.size
    b = _fft_convolution(a, a, N)
    k = int(np.argmax(b))

    g = np.zeros_like(a)
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g[mask] = 2.0 * a[idx[mask]]

    a_new = a - eta * g
    a_new = _project_onto_simplex(a_new, s=1.0)
    return a_new, b

    """One projected sub‑gradient step for the original max‑convolution objective."""
    n = a.size
    b = _fft_convolution(a, a, N)
    k = int(np.argmax(b))

    g = np.zeros_like(a)
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g[mask] = 2.0 * a[idx[mask]]

    a_new = a - eta * g
    a_new = _project_onto_simplex(a_new, s=1.0)
    return a_new, b


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Produce a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine combines a fast least‑squares
    flattening phase with a final sub‑gradient phase, staying inside the
    1000 s time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Initialise
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.size == 0:
            raise ValueError
    except Exception:
        init = _arcsine_distribution(2000)   # fallback length

    init = np.maximum(init, 0.0)
    if init.sum() == 0.0:
        init = np.full_like(init, 1.0 / init.size)
    else:
        init = init / init.sum()               # unit ℓ₁‑mass

    n = init.size
    C = 1.0 / (2 * n - 1)                      # target constant value of the autocorrelation
    N = 1 << ((3 * n - 2).bit_length())        # FFT length (next power of two ≥ 3n‑2)

    a = init.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_convolution(a, a, N))
    print(f"Start: length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Phase 1 – least‑squares flattening (many cheap iterations)
    eta0_ls = 1.0
    it = 0
    phase1_deadline = deadline - 60.0          # keep the last minute for phase 2
    while time.time() < phase1_deadline:
        it += 1
        eta = eta0_ls / np.sqrt(it)
        a, b = _ls_gradient_step(a, C, N, eta)

        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LS iter={it}  score={best_score:.6f}")

        # tiny random jitter to shake out plateaus
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Phase 2 – sub‑gradient descent on the true max‑convolution objective
    eta0_sg = 0.5
    it2 = 0
    while time.time() < deadline:
        it2 += 1
        eta = eta0_sg / np.sqrt(it2)
        a, b = _subgradient_step(a, N, eta)

        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] SG iter={it2}  score={best_score:.6f}")

        if it2 % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    print(f"Finished. Best score found: {best_score:.6f} (length={n})")
    return best_seq.tolist()
