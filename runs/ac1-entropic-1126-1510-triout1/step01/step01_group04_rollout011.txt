def internal_eval(seq):
    # Clip sequence values between 0 and 1000
    seq = np.clip(seq, 0.0, 1000.0)
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a * sum_a))

def make_double_power_law(n, a, b, delta):
    idx = np.arange(n, dtype=np.float64)
    # avoid zero when delta=0 and exponent negative
    term1 = (idx + delta) ** a
    term2 = (n - 1 - idx + delta) ** b
    vec = term1 * term2
    # Ensure non-negative (should be)
    total = vec.sum()
    if total == 0:
        # fallback uniform
        vec = np.ones(n,dtype=np.float64)
        total = n
    return vec / total

import time, numpy as np, math

def _eval(seq):
    # fast internal evaluation matching evaluate_sequence
    seq = np.clip(seq, 0.0, 1000.0)
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a * sum_a))

def make_double_power_law(n, a, b, delta):
    idx = np.arange(n, dtype=np.float64)
    # add small epsilon to avoid zero at exponent negative
    vec = (idx + delta) ** a * (n - 1 - idx + delta) ** b
    total = vec.sum()
    if total == 0:
        return np.full(n, 1.0 / n)
    return vec / total

    # returns best seq, params, score
    a_low, a_high = -0.9, -0.1
    b_low, b_high = -0.9, -0.1
    d_low, d_high = 0.0, 5.0

"""
Propose a step‑function (list of non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three stages:

1. **Parametric annealing** – we look for the best triple
   (α, β, δ) in the family  

        a_i ∝ (i+δ)^α · (n‑1‑i+δ)^β

   on a modest size (n≈1800).  This three‑parameter family already contains
   the classical arcsine distribution (α=β=‑½) and allows the ends of the
   interval to be weighted more heavily, which experimentally reduces the
   peak of the self‑convolution.

2. **Scaling up** – the best parameters are used to build a longer sequence
   (n=5000).  The sequence is normalised to sum to 1, which is the only
   scale‑invariant normalization required.

3. **Stochastic hill‑climbing** – a large number of tiny mass‑transfer
   moves (keep the total sum fixed) are performed on the full‑size vector.
   Only moves that improve the objective are kept.  The loop runs
   until the time budget is exhausted.

All calculations are performed with NumPy.  An internal replica of
`evaluate_sequence` is used for speed; the final vector is returned as a
plain Python list, which the external evaluator will accept.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast replica of the provided evaluate_sequence (no type checks)
def _eval(seq: np.ndarray) -> float:
    """Evaluate a candidate vector using the same formula as evaluate_sequence."""
    # Clip to the limits imposed inside the official evaluator
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.shape[0]
    conv = np.convolve(seq, seq)          # length 2*n‑1
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric family: (i+δ)^α · (n‑1‑i+δ)^β  (normalised)
def _make_double_power_law(n: int, alpha: float, beta: float, delta: float) -> np.ndarray:
    """Return a normalised vector a_i ∝ (i+δ)^α·(n‑1‑i+δ)^β."""
    idx = np.arange(n, dtype=np.float64)
    # avoid exact zero when delta is 0 and exponents are negative
    term1 = (idx + delta) ** alpha
    term2 = (n - 1 - idx + delta) ** beta
    vec = term1 * term2
    total = vec.sum()
    if total == 0.0:
        # fallback to uniform – should never happen with the chosen bounds
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vec / total


# ----------------------------------------------------------------------
# Phase 1 – simulated annealing on (α,β,δ) for a modest n
def _anneal_parameters(n: int, deadline: float) -> tuple[np.ndarray, tuple[float, float, float], float]:
    """Return best sequence, its parameters and score found before the deadline."""
    # Search ranges
    a_lo, a_hi = -0.90, -0.10      # α
    b_lo, b_hi = -0.90, -0.10      # β
    d_lo, d_hi = 0.0, 5.0          # δ (absolute offset)

    # Start from a sensible point (close to the arcsine law)
    cur_alpha, cur_beta, cur_delta = -0.55, -0.55, 0.30
    cur_seq = _make_double_power_law(n, cur_alpha, cur_beta, cur_delta)
    cur_score = _eval(cur_seq)

    best_seq, best_score = cur_seq, cur_score
    best_params = (cur_alpha, cur_beta, cur_delta)

    temperature = 0.20
    it = 0
    # a few‑tens of thousands of proposals – stop early if time runs out
    while time.time() < deadline and it < 50000:
        # Gaussian proposals
        a_new = np.clip(cur_alpha + np.random.normal(scale=0.022), a_lo, a_hi)
        b_new = np.clip(cur_beta  + np.random.normal(scale=0.022), b_lo, b_hi)
        d_new = np.clip(cur_delta + np.random.normal(scale=0.30), d_lo, d_hi)

        seq_new = _make_double_power_law(n, a_new, b_new, d_new)
        score_new = _eval(seq_new)

        # Metropolis acceptance
        if score_new < cur_score or np.random.rand() < math.exp(-(score_new - cur_score) / max(temperature, 1e-12)):
            cur_alpha, cur_beta, cur_delta = a_new, b_new, d_new
            cur_score = score_new
            if score_new < best_score:
                best_seq, best_score = seq_new, score_new
                best_params = (a_new, b_new, d_new)

        # exponential cooling
        temperature *= 0.99995
        it += 1
        if it % 5000 == 0:
            print(f"[anneal] it={it:,}  best_score={best_score:.6f}")

    return best_seq, best_params, best_score


# ----------------------------------------------------------------------
# Phase 3 – stochastic hill‑climbing on the full‑size vector
def _hill_climb(seq: np.ndarray, best_score: float, deadline: float, max_iters: int = 100000) -> tuple[np.ndarray, float]:
    """Mass‑transfer hill‑climbing that keeps the total sum fixed."""
    n = seq.shape[0]
    cur = seq.copy()
    cur_score = best_score
    best = cur.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            it += 1
            continue

        # move up to 10 % of the source mass
        delta = cur[i] * np.random.rand() * 0.10
        cur[i] -= delta
        cur[j] += delta

        score = _eval(cur)
        if score < cur_score:
            cur_score = score
            if score < best_score:
                best_score = score
                best = cur.copy()
        else:
            # reject – revert the move
            cur[i] += delta
            cur[j] -= delta

        it += 1
        if it % 5000 == 0:
            print(f"[hill] it={it:,}  cur_score={cur_score:.6f}  best_score={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑scoring step‑function.
    The routine respects the 1000 s time budget and returns a Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # tiny safety margin

    # -------- Phase 1 : quick parameter optimisation on a reduced size ----------
    n_small = 1800
    print(f"Phase 1 – annealing on n={n_small}")
    _, best_params, best_score = _anneal_parameters(n_small, deadline)
    print(f"[phase 1] best_params={best_params}  score={best_score:.6f}")

    # -------- Phase 2 : build the full‑size candidate from the best parameters ----------
    n_large = 5000
    alpha, beta, delta = best_params
    seq = _make_double_power_law(n_large, alpha, beta, delta)
    best_score = _eval(seq)
    print(f"[phase 2] initial large‑size score={best_score:.6f}")

    # -------- Phase 3 : intensive hill‑climbing on the full vector ----------
    seq, best_score = _hill_climb(seq, best_score, deadline, max_iters=120000)
    print(f"[phase 3] final score after hill‑climb={best_score:.6f}")

    # Convert to plain Python floats for the external evaluator
    final_sequence = [float(v) for v in seq.tolist()]
    print(f"Returning sequence length={len(final_sequence)}  final_score={best_score:.6f}")
    return final_sequence
