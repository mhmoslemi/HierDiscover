def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_score = inf
    best_seq = None

    # optional: load global best from height_sequence_1 if exists
    try:
        init_best = height_sequence_1
        # ensure in list of floats
        if isinstance(init_best, list):
            a_init = np.array(init_best, dtype=float)
            a_init = a_init / a_init.sum()
            score = evaluate_sequence(a_init.tolist())
            if score < best_score:
                best_score = score
                best_seq = a_init.copy()
    except NameError:
        pass

    # main loop
    while time.time() < deadline:
        # choose n
        n = np.random.randint(1200, 2500) 
        # shape generation
        shape_type = np.random.choice(['geom','power','mixed','rand','prev'])
        if shape_type == 'geom':
            r = np.exp(-np.random.uniform(0.0003,0.001))
            a = r**np.arange(n)
        elif shape_type == 'power':
            p = np.random.uniform(0.5,2.0)
            a = (np.arange(1,n+1, dtype=float)**-p)
        elif shape_type == 'mixed':
            r = np.exp(-np.random.uniform(0.0003,0.001))
            p = np.random.uniform(0.3,1.2)
            a = (r**np.arange(n))* (np.arange(1,n+1)**-p)
        elif shape_type == 'rand':
            a = np.random.gamma(1.0,1.0,size=n)
        else: # prev: try using best_seq if length matches else random
            if best_seq is not None and best_seq.size == n:
                a = best_seq.copy()
            else: # random from Dirichlet
                a = np.random.gamma(1.,1.,size=n)
        a = a / a.sum()

        # optimization
        a_opt, score_opt = _optimize_adam(a, n, deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"New global best: n={n}, shape={shape_type}, score={best_score:.6f}")

        # early break if below target
        if best_score < 1.503:
            break
    # if best_seq still None fallback uniform
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0/n)
    return best_seq.tolist()

"""
Propose a low‑score step‑function.

The algorithm mixes three ideas:
1. **Broad random sampling** – generate many candidate shapes
   (geometric, power‑law, mixed, pure random) for a randomly chosen length.
2. **Fast screening** – evaluate each candidate by the maximum of the
   self‑convolution (FFT) and keep the best few.
3. **Projected Adam optimisation** – starting from the best screened
   candidate, run a projected Adam optimiser on the simplex (non‑negative,
   sum = 1).  The sub‑gradient of the max‑convolution is computed from the
   active convolution indices; after each step the vector is projected back
   onto the probability simplex.

The search is time‑bounded (default 1000 s).  It repeatedly restarts with a
new length/shape, keeps the overall best sequence, and stops early if the
target score 1.503 is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution max (FFT based)
def _conv_max_fft(a: np.ndarray):
    """Return (max convolution value, indices where it is attained) of a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = float(np.max(conv))
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return max_val, idx.astype(int)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
def _gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] with respect to a."""
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # derivative of a_i * a_j w.r.t. a_i is a_j; both (i,j) and (j,i)
        grad[i] += 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (simple but effective)
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """
    Optimise a starting vector a0 (already normalised to sum=1) using
    Adam with projection onto the simplex.  Returns best vector seen and its
    score 2 * n * max(conv(a,a)).
    """
    a = a0.copy()
    best_a = a.copy()
    max_c, max_idx = _conv_max_fft(a)
    best_score = 2.0 * n * max_c

    # Adam hyper‑parameters
    lr0 = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0                     # iteration counter
    last_improve = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        # current convolution and gradient
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c
        grad = _gradient_for_max(a, max_idx)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)                # simple decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = _project_to_simplex(a - step)

        # evaluate candidate
        max_c2, _ = _conv_max_fft(a_candidate)
        cand_score = 2.0 * n * max_c2

        if cand_score < cur_score - 1e-12:
            a = a_candidate
            cur_score = cand_score
            last_improve = t
        else:
            # occasional random jitter to escape flat plateaus
            if t % 200 == 0:
                pert = a.copy()
                idxs = np.random.randint(n, size=10)
                pert[idxs] += np.random.randn(10) * 1e-3
                pert = np.maximum(pert, 0.0)
                pert = pert / np.sum(pert)
                max_c2, _ = _conv_max_fft(pert)
                pert_score = 2.0 * n * max_c2
                if pert_score < cur_score:
                    a = pert
                    cur_score = pert_score
                    last_improve = t

        # keep the best ever seen in this run
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            # early exit if we already beat the target
            if best_score < 1.503:
                break

        # if we have been stagnant for a while, reset moments
        if t - last_improve > 5000:
            m.fill(0.0)
            v.fill(0.0)
            last_improve = t

    return best_a, best_score


# ----------------------------------------------------------------------
# Shape generators
def _gen_geom(n: int) -> np.ndarray:
    """Geometric decay a_i = r**i with r close to 1."""
    t = np.random.uniform(0.0002, 0.0015)
    r = np.exp(-t)
    return r ** np.arange(n, dtype=float)


def _gen_power(n: int) -> np.ndarray:
    """Power‑law decay a_i = (i+1)^(-p)."""
    p = np.random.uniform(0.2, 2.5)
    return np.arange(1, n + 1, dtype=float) ** -p


def _gen_mixed(n: int) -> np.ndarray:
    """Product of a geometric and a power‑law decay."""
    t = np.random.uniform(0.0002, 0.001)
    r = np.exp(-t)
    p = np.random.uniform(0.2, 1.8)
    return (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)


def _gen_random(n: int) -> np.ndarray:
    """Dirichlet‑like random positive vector."""
    return np.random.gamma(1.0, 1.0, size=n)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) gives a low
    evaluation score.  The routine respects the time budget, uses random
    restarts, screens many candidates quickly, and refines the best ones
    with a projected Adam optimiser.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # If the user environment already supplied a good construction, start
    # from it (the variable height_sequence_1 is optional).
    try:
        hs = height_sequence_1      # type: ignore[name-defined]
        if isinstance(hs, (list, np.ndarray)):
            a0 = np.asarray(hs, dtype=float)
            if a0.size >= 2:
                a0 = np.maximum(a0, 0.0)
                a0 /= np.sum(a0)
                max_c, _ = _conv_max_fft(a0)
                score0 = 2.0 * a0.size * max_c
                best_score = score0
                best_seq = a0.copy()
                print(f"Seeded from height_sequence_1: n={a0.size}, score={best_score:.6f}")
    except NameError:
        # No pre‑existing sequence – nothing to do.
        pass

    # ------------------------------------------------------------------
    # Main search loop
    shape_fns = [_gen_geom, _gen_power, _gen_mixed, _gen_random]

    while time.time() < deadline:
        # ---- choose a problem size -------------------------------------------------
        n = np.random.randint(1200, 2500)          # length of the step function

        # ---- quick random sweep to find a promising seed ---------------------------
        best_local_a = None
        best_local_score = float("inf")

        sweep_iters = 25   # number of random shapes per restart (adjustable)
        for _ in range(sweep_iters):
            fn = np.random.choice(shape_fns)
            a = fn(n)

            # optional random permutation (does not affect statistics but adds variety)
            if np.random.rand() < 0.5:
                np.random.shuffle(a)

            # enforce non‑negativity & normalise to sum = 1
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            if s == 0:
                continue
            a /= s

            max_c, _ = _conv_max_fft(a)
            score = 2.0 * n * max_c
            if score < best_local_score:
                best_local_score = score
                best_local_a = a

        # If none survived the sweep (unlikely) skip this n
        if best_local_a is None:
            continue

        # ---- refine the best local candidate with Adam -----------------------------
        sub_deadline = min(deadline, time.time() + 8.0)   # spend at most ~8 s per refinement
        a_opt, score_opt = _adam_optimize(best_local_a, n, sub_deadline)

        # Global best update
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"Global improvement: n={n}, score={best_score:.6f}")

        # Early exit if we have already beaten the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: uniform distribution (should never be needed)
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # Return plain Python list (the external evaluator will clip for us)
    return best_seq.tolist()
