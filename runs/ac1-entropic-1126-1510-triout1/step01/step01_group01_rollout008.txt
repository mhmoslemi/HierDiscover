"""
propose_candidate
-----------------

Adam‑based projected gradient descent on a smooth approximation of the
maximum self‑convolution.  The objective is

    max_k (a*a)[k]

(the evaluation score is exactly this value when the ℓ¹‑norm is fixed to
√(2 n)).  The non‑smooth max is replaced by a log‑sum‑exp surrogate,
which yields a dense gradient that can be obtained cheaply with FFT‑based
convolutions.  After each step the vector is projected onto the simplex
{a ≥ 0, Σa = √(2 n)}.  Several random restarts with a “spike’’ initialisation
are performed until the time budget expires.  The best sequence found so
far is returned.

The algorithm works completely with NumPy, needs no external files and
fits easily into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project v onto the simplex of total mass z (non‑negative coordinates)."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full convolution (real inputs)
def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of real vectors x and y using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a (full convolution)."""
    conv = _convolve_fft(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float = 10.0):
    """
    Compute a gradient of the smooth max
        f(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
    and also return the exact max entry of the convolution.
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _convolve_fft(a, a)          # length = 2n‑1
    max_conv = float(conv.max())

    # 2) soft‑max weights
    #   w_k = exp(α·conv_k) / Σ_j exp(α·conv_j)
    #   we shift by max to avoid overflow.
    shift = max_conv
    exp_term = np.exp(alpha * (conv - shift))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp               # length 2n‑1, sums to 1

    # 3) gradient = Σ_k w_k·∂conv_k/∂a
    #    Using correlation: Σ_k w_k·a_{k‑i} = Σ_j a_j·w_{i+j}
    #    which is a convolution of a with reversed w.
    w_rev = w[::-1]
    corr_full = _convolve_fft(a, w_rev)  # length = n + (2n‑1) – 1 = 3n‑2

    L = conv.shape[0]                     # 2n‑1
    idx = (L - 1) - np.arange(n)          # indices that pick the right correlation entries
    correlation = corr_full[idx]          # shape (n,)

    # diagonal correction: Diagonal terms need an extra a_i·w_{2i}
    diag_extra = np.zeros_like(a)
    diag_pos = 2 * np.arange(n)
    diag_extra = a * w[diag_pos]          # w has length 2n‑1, so all positions are valid

    grad = correlation + diag_extra
    return grad, max_conv

    # ------------------------------------------------------------------
    # If something went terribly wrong we fall back to a uniform vector.
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    final_score = _max_convolution(best_seq)
    print(f"[final] best score = {final_score:.6f} (length {n})")
    return best_seq.tolist()
