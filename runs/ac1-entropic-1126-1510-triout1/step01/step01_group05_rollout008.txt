    curr = vec.copy()
    curr_score = fast_evaluate(curr)
    while time.time() < deadline:
        conv = np.convolve(curr, curr)
        k = np.argmax(conv)
        # compute contributions for each i that participates in k
        partner = k - np.arange(n)
        # mask valid indices:
        mask = (partner >=0) & (partner < n)
        contributions = np.zeros(n)
        contributions[mask] = curr[mask] * curr[partner[mask]]
        # choose source from top contributors
        src_candidates = np.argpartition(contributions, -3)[-3:]  # top 3
        src = np.random.choice(src_candidates)
        # choose target far from src and partner
        target = np.random.randint(0, n)
        if target == src:
            continue
        delta = min(curr[src] * 0.2, 1e-3)  # small

            continue

        cand[src] -= delta
        cand[target] += delta
        cand_score = fast_evaluate(cand)
        if cand_score < curr_score:
            curr = cand
            curr_score = cand_score

def fast_evaluate(vec):
    vec = np.asarray(vec, dtype=np.float64)
    # clip negative etc
    vec = np.maximum(vec, 0.0)
    vec = np.minimum(vec, 1000.0)
    s = vec.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(vec, vec)
    max_b = conv.max()
    n = vec.size
    return float(2.0 * n * max_b / (s*s))
```

"""
Propose a low‑value sequence for the provided `evaluate_sequence` function.

Algorithm
---------
1. **Length** – use the length of the previously discovered sequence
   (`height_sequence_1`) when it exists, otherwise fall back to 2000.

2. **Fast objective** – a tiny wrapper reproducing the behaviour of
   `evaluate_sequence` (clipping, non‑negativity, sum guard).  Using this
   wrapper avoids the Python‑list conversion overhead and makes it possible
   to evaluate millions of candidates inside the 1000 s budget.

3. **Exploratory random search** – generate many candidate families:
   * pure power‑law  a_i ∝ (i+1)^{-p}
   * shifted power‑law  a_i ∝ (i+δ)^{-p}
   * Beta‑shaped      a_i ∝ (i+1)^{α‑1}·(N‑i)^{β‑1}
   * piece‑wise power‑law with two exponents (p₁ on a prefix, p₂ on a suffix)

   Parameters are sampled uniformly (or log‑uniform for positive numbers).
   Every candidate is normalised to unit sum and scored with the fast
   evaluator.  The best vector found in the first ~30 % of the budget is kept.

4. **Guided hill‑climbing / simulated annealing** – starting from the best
   vector, repeatedly move a small amount of mass from an index that
   contributes heavily to the current maximal convolution entry to a random
   destination.  The move is accepted if it strictly improves the score;
   otherwise it may be accepted with a temperature‑scaled probability to
   escape local minima.  The temperature decays exponentially.

5. **Return** – the best vector (converted to a plain Python list) is
   returned.  All numbers are non‑negative and safely within the limits of
   `evaluate_sequence`.

The whole routine respects the 1000 s wall‑clock limit; if the time limit
is approached the current best solution is returned immediately.
"""

import time
import numpy as np
import math

# --------------------------------------------------------------------------- #
# Fast re‑implementation of `evaluate_sequence`
# --------------------------------------------------------------------------- #
def fast_score(vec: np.ndarray) -> float:
    """Return the same value as `evaluate_sequence` but without Python‑list
    conversion and with the same safety checks."""
    # enforce the same guards as the original function
    if not isinstance(vec, np.ndarray):
        vec = np.asarray(vec, dtype=np.float64)
    vec = np.maximum(vec, 0.0)          # no negatives
    vec = np.minimum(vec, 1000.0)       # clip huge values
    s = vec.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(vec, vec)
    max_b = conv.max()
    n = vec.size
    return float(2.0 * n * max_b / (s * s))


# --------------------------------------------------------------------------- #
# Family generators
# --------------------------------------------------------------------------- #
def power_law(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+shift)^{-p}, normalised to unit sum."""
    idx = np.arange(1, n + 1, dtype=np.float64) + shift
    a = idx ** (-p)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0))
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def piecewise_power_law(
    n: int, p1: float, p2: float, k: int
) -> np.ndarray:
    """
    First `k` entries follow exponent `p1`, remaining entries follow exponent
    `p2`.  The whole vector is normalised to unit sum.
    """
    if k <= 0:
        k = 1
    if k >= n:
        k = n - 1
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty_like(idx)
    a[:k] = idx[:k] ** (-p1)
    a[k:] = idx[k:] ** (-p2)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


# --------------------------------------------------------------------------- #
# Random search over families
# --------------------------------------------------------------------------- #
def random_search(n: int, deadline: float):
    """Explore many parametrised families and keep the best vector."""
    best_vec = np.full(n, 1.0 / n)   # uniform fallback
    best_sc = fast_score(best_vec)
    it = 0
    while time.time() < deadline:
        family = np.random.randint(4)   # choose one of the four families
        if family == 0:   # pure power‑law
            p = np.random.uniform(0.35, 0.90)
            v = power_law(n, p)
        elif family == 1:  # shifted power‑law
            p = np.random.uniform(0.35, 0.90)
            shift = np.random.exponential(scale=3.0)   # positive shift
            v = power_law(n, p, shift=shift)
        elif family == 2:  # beta‑shape
            # sample in log‑space to cover several orders of magnitude
            log_alpha = np.random.uniform(-1.0, 1.0)   # α ∈ [0.1,10]
            log_beta = np.random.uniform(-1.0, 1.0)    # β ∈ [0.1,10]
            alpha = 10 ** log_alpha
            beta = 10 ** log_beta
            v = beta_shape(n, alpha, beta)
        else:  # piece‑wise power‑law
            p1 = np.random.uniform(0.15, 0.85)
            p2 = np.random.uniform(0.15, 0.85)
            fraction = np.random.uniform(0.15, 0.85)
            k = int(fraction * n)
            v = piecewise_power_law(n, p1, p2, k)

        sc = fast_score(v)
        if sc < best_sc:
            best_sc = sc
            best_vec = v
            print(f"[random] iter {it:6d}  new best {best_sc:.6f}"
                  f"  family {family}")
        it += 1
    return best_vec, best_sc


# --------------------------------------------------------------------------- #
# Guided hill‑climbing with a simple annealing schedule
# --------------------------------------------------------------------------- #
def guided_hill_climb(start_vec: np.ndarray, deadline: float) -> np.ndarray:
    """Move a small amount of mass from a heavy contributor of the current
    maximal convolution entry to a random location.  Accept improving moves
    and sometimes accept worsening moves with a temperature that decays
    exponentially."""
    n = start_vec.size
    cur = start_vec.copy()
    cur_sc = fast_score(cur)

    # exponential cooling: T = T0 * (0.9999)^{steps}
    T0 = 1e-3
    decay = 0.9999
    T = T0
    steps = 0
    no_improve_streak = 0
    while time.time() < deadline:
        # compute the convolution and the index of the current maximum
        conv = np.convolve(cur, cur)
        k = int(np.argmax(conv))

        # contributions of each index to that maximal entry
        partner = k - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = cur[mask] * cur[partner[mask]]

        # pick a source index weighted towards large contributions
        # (take one of the three largest with a small random perturbation)
        top3 = np.argpartition(contrib, -3)[-3:]
        src = np.random.choice(top3)

        # pick a destination uniformly (different from source)
        dst = np.random.randint(0, n - 1)
        if dst >= src:
            dst += 1

        # move at most 5 % of the source mass
        delta = cur[src] * np.random.rand() * 0.05
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta

        # safety – keep non‑negativity
        if cand[src] < 0.0:
            continue

        cand_sc = fast_score(cand)

        # acceptance
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve_streak = 0
        else:
            # simulated‑annealing acceptance
            prob = math.exp(-(cand_sc - cur_sc) / max(T, 1e-12))
            if np.random.rand() < prob:
                cur = cand
                cur_sc = cand_sc
                no_improve_streak = 0
            else:
                no_improve_streak += 1

        # cooling
        steps += 1
        T = T0 * (decay ** steps)

        # occasional logging
        if steps % 5000 == 0:
            print(f"[hill] step {steps:6d}  cur score {cur_sc:.6f}")

        # if we haven't improved for a long while, inject a tiny random
        # perturbation to escape a flat region
        if no_improve_streak > 20000:
            rnd = np.random.rand(n)
            rnd /= rnd.sum()
            cur = 0.999 * cur + 0.001 * rnd
            cur_sc = fast_score(cur)
            no_improve_streak = 0

    return cur


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best bound.  The routine consists of a fast random search over
    several parametrised families followed by a guided hill‑climbing
    refinement.  It respects the supplied time budget.
    """
    np.random.seed(seed)

    # --------------------------------------------------- #
    # Determine sequence length
    # --------------------------------------------------- #
    try:
        n = int(len(height_sequence_1))   # use known construction if available
    except Exception:
        n = 2000                         # sensible fallback

    final_deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------- #
    # 1) Broad random search (≈30 % of the budget)
    # --------------------------------------------------- #
    search_deadline = time.time() + (budget_s * 0.30)
    best_vec, best_sc = random_search(n, search_deadline)
    print(f"[stage 1] best after random search: {best_sc:.6f}")

    # --------------------------------------------------- #
    # 2) Guided refinement (remaining time)
    # --------------------------------------------------- #
    refined_vec = guided_hill_climb(best_vec, final_deadline)
    refined_sc = fast_score(refined_vec)

    if refined_sc < best_sc:
        best_vec, best_sc = refined_vec, refined_sc
        print(f"[stage 2] improved after hill‑climb: {best_sc:.6f}")
    else:
        print(f"[stage 2] hill‑climb did not improve the score.")

    # --------------------------------------------------- #
    # Final report
    # --------------------------------------------------- #
    print(f"[final] best score obtained: {best_sc:.6f}  (length {n})")
    # The wrapper `evaluate_sequence` will apply the same clipping as we did.
    return best_vec.tolist()
