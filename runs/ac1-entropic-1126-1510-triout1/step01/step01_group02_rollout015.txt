    deadline = start + max(0.0, budget_s - 0.5)

"""
Hybrid smooth‑max gradient descent + simulated‑annealing for the
step‑function minimisation problem.

The evaluation metric is 2·n·max(conv(a,a)) / (sum(a)^2) and is
scale‑invariant.  We therefore normalise all candidates onto the
probability simplex (sum = 1, a_i ≥ 0).  The algorithm repeatedly
computes a smooth approximation to the max (soft‑max with an increasing
inverse‑temperature β) and takes a projected gradient step in the
direction that reduces this surrogate.  To avoid getting stuck in
local minima, random pairwise mass swaps are interleaved and a simple
Metropolis acceptance rule is used.  The best vector seen so far is
returned.

The implementation is pure NumPy, runs entirely in memory and respects
the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:           # all non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety – enforce exact sum 1
    w_sum = w.sum()
    if w_sum <= 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Score (exact, using the same formula as evaluate_sequence)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact value of the objective for a (assumes a ≥ 0)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(a, a)
    max_b = conv.max()
    n = a.shape[0]
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Random pairwise mass‑swap that preserves the total sum exactly
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small random amount of mass between two distinct coordinates."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # zero‑mean proposal, clipped to keep non‑negativity
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises the
    given evaluation function.  The routine runs for at most ``budget_s``
    seconds.  It starts from the globally supplied ``height_sequence_1`` if
    available, otherwise from a uniform distribution, and then performs a
    hybrid optimisation:
        * a smooth‑max surrogate of the convolution peak,
        * projected gradient descent on the simplex,
        * occasional random pairwise swaps,
        * a Metropolis acceptance rule,
        * simple restart logic when progress stalls.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, budget_s - 0.5)

    # ------------------------------------------------------------------
    # Initialise the search vector (normalised to sum = 1)
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        if n == 0:
            raise ValueError
        base = base / max(base.sum(), 1e-12)
    except Exception:
        n = 3000                      # a reasonable fallback size
        base = np.full(n, 1.0 / n, dtype=float)

    current = _project_simplex(base)
    best = current.copy()
    best_score = _score(best)
    current_score = best_score

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    beta = 5_000.0                # inverse temperature for the soft‑max
    beta_max = 50_000.0
    beta_inc = (beta_max - beta) / 2000.0   # linear increase
    step0 = 0.6                    # initial gradient step scale
    last_improve_iter = 0

    it = 0
    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # Choose move type
        # ------------------------------------------------------------------
        if np.random.rand() < 0.65:
            # ------- projected gradient step on the smooth‑max ----------
            conv = np.convolve(current, current)                # full convolution
            # soft‑max probabilities (stable version)
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()                       # shape (2n-1,)

            # gradient of smooth‑max w.r.t. a:
            # g_i = Σ_j a_j (p_{i+j} + p_{i-j})
            p_rev = p[::-1]
            conv_rev = np.convolve(current, p_rev)             # length 3n‑2
            # h_i = Σ_j a_j p_{i+j} = conv_rev[2n‑2‑i]
            h = conv_rev[2 * n - 2 - np.arange(n)]

            conv_fwd = np.convolve(current, p)                 # length 3n‑2
            g = h + conv_fwd[:n]                               # combine both parts

            # step size decays like 1/√it
            eta = step0 / np.sqrt(it)
            candidate = current - eta * g
            candidate = _project_simplex(candidate)
        else:
            # ------- random pairwise swap (exploration) ---------------
            sigma = 0.2 / np.sqrt(it) if it > 0 else 0.2
            candidate = _random_pair_move(current, sigma)

        # ------------------------------------------------------------------
        # Evaluate the candidate (exact objective)
        # ------------------------------------------------------------------
        cand_score = _score(candidate)

        # Keep the best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()
            last_improve_iter = it
            print(f"[Iter {it:6d}] New best score: {best_score:.6f}")

        # Metropolis acceptance rule – always accept improvements
        if cand_score < current_score:
            accept = True
        else:
            # simple temperature schedule for occasional uphill moves
            temp = max(1e-3, 1.0 - it / 5000.0)
            prob = np.exp(-(cand_score - current_score) / temp)
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score

        # ------------------------------------------------------------------
        # Annealing of β (smooth‑max becomes sharper over time)
        # ------------------------------------------------------------------
        beta = min(beta_max, beta + beta_inc)

        # ------------------------------------------------------------------
        # Simple restart if we have been stuck for a while
        # ------------------------------------------------------------------
        if it - last_improve_iter > 4000:
            # blend a tiny random perturbation into the current best
            perturb = np.random.rand(n)
            perturb /= perturb.sum()
            current = _project_simplex(0.92 * best + 0.08 * perturb)
            current_score = _score(current)
            last_improve_iter = it

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return best.tolist()
