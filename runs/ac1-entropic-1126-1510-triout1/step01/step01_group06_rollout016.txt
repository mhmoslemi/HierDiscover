def _conv_full(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Fa * Fa, n=size)[:2*n - 1]
    return conv

def _softmax_gradient(a, conv, beta):
    max_c = np.max(conv)
    # use diff = conv - max_c (<=0)
    w = np.exp(beta * (conv - max_c))
    w /= np.sum(w)
    # compute grad = 2 * np.convolve(w, a)[:n]
    # Use FFT for conv w*a.
    n = a.shape[0]
    size = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, n=size)
    fa = np.fft.rfft(a, n=size)
    grad_full = np.fft.irfft(fw * fa, n=size)[:len(w) + n - 1]
    grad = 2.0 * grad_full[:n]   # first n entries correspond to positions 0..n-1
    return grad

def _adam_optimize(a0, n, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    # Adam parameters
    alpha = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # schedule for beta (softmax) maybe start low and increase
    beta_val = 0.5
    while time.time() < deadline:
        t += 1
        conv = _conv_full(a)
        max_c = np.max(conv)
        score = 2.0 * n * max_c
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # early exit if reached target
        if best_score < 1.503:
            break
        # compute gradient
        # choose active indices tolerance maybe 1e-8
        # use softmax weighted gradient
        grad = _softmax_gradient(a, conv, beta_val)
        # update moving averages
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        # adjust step size scaling maybe by inv sqrt of t
        step = step / np.sqrt(t)
        a_candidate = _project_to_simplex(a - step)

            a = a_candidate
        else:
            # try smaller step factor
            for factor in (0.5, 0.2, 0.1):
                a_c = _project_to_simplex(a - step * factor)

"""
Candidate generator for the step-function optimization problem.

The algorithm combines several ideas:
- Lots of diverse random initializations (geometric, power‑law, sinusoid, linear,
  mixed, Dirichlet, and the best known previous construction if available).
- For each start a projected Adam optimizer works on the simplex.
  The gradient is obtained from a smooth “soft‑max” approximation of the
  max‑convolution, which yields a dense sub‑gradient that steers the search
  away from all large convolution peaks simultaneously.
- The smoothing parameter β is increased gradually, turning the soft‑max
  into the true max once the iterates are close to a local optimum.
- After each gradient step we project onto the probability simplex
  (non‑negative entries summing to one).  An occasional back‑tracking line
  search and random perturbations help to escape plateaus.
- The whole procedure is run repeatedly with a short time budget for each
  start (≈ 15 s) until the global budget (≈ 1000 s) expires.
  The best sequence encountered is returned.
"""

import numpy as np
import time
from math import log, sqrt

# ---------- Simplex projection -------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ---------- Convolution helpers -----------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Fa * Fa, n=size)[: 2 * n - 1]
    return conv

def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max functional.
    weight w_k ∝ exp(beta * (conv_k - max(conv))) and
    grad_i = 2 * Σ_k w_k a_{k-i}.
    """
    max_c = np.max(conv)
    # avoid overflow by subtracting max_c
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / np.sum(exp_term)                     # shape (2n-1,)

    n = a.shape[0]
    size = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, n=size)
    fa = np.fft.rfft(a, n=size)
    grad_full = np.fft.ifft(fw * fa).real[: len(w) + n - 1]
    # first n entries correspond to indices 0..n-1 of the formula Σ_k w_k a_{k-i}
    grad = 2.0 * grad_full[:n]
    return grad

# ---------- Initial‑shape generators ------------------------------------------
def _initial_vector(n: int, kind: str, rng: np.random.Generator) -> np.ndarray:
    """Return a random non‑negative vector of length n and sum 1."""
    if kind == "geom":
        t = rng.uniform(0.0003, 0.0015)
        r = np.exp(-t)
        a = r ** np.arange(n, dtype=float)
    elif kind == "power":
        p = rng.uniform(0.3, 2.0)
        a = (np.arange(1, n + 1, dtype=float) ** -p)
    elif kind == "mixed":
        t = rng.uniform(0.0003, 0.001)
        r = np.exp(-t)
        p = rng.uniform(0.5, 1.5)
        a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)
    elif kind == "linear_inc":
        a = np.arange(1, n + 1, dtype=float)
    elif kind == "linear_dec":
        a = np.arange(n, 0, -1, dtype=float)
    elif kind == "sin":
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)  # zero at boundaries, positive inside
    elif kind == "sin_pow":
        base = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        p = rng.uniform(0.5, 3.0)
        a = base ** p
    elif kind == "random":
        a = rng.gamma(1.0, 1.0, size=n)
    else:   # fallback uniform
        a = np.ones(n, dtype=float)

    # optional random permutation (does not change multiset)
    if rng.random() < 0.5:
        rng.shuffle(a)

    # normalise to sum = 1
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

# ---------- Adam optimiser on the simplex --------------------------------------
def _optimize_sequence(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Projected Adam optimisation of the max‑convolution objective."""
    a = a0.copy()
    best_a = a.copy()
    # current objective
    conv = _conv_full(a)
    max_c = np.max(conv)
    best_score = 2.0 * n * max_c

    # Adam hyper‑parameters
    alpha = 0.15                 # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t_iter = 0
    beta_smooth = 0.5           # start with a very soft approximation
    # schedule parameters
    beta_growth = 1.05          # multiplicative increase each 200 iterations
    target_beta = 80.0

    while time.time() < deadline:
        t_iter += 1

        # recompute convolution and score
        conv = _conv_full(a)
        max_c = np.max(conv)
        cur_score = 2.0 * n * max_c

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional log
            if t_iter % 200 == 0:
                print(f"    [iter {t_iter}] local best = {best_score:.6f}")

        # early stop if we reached the target bound
        if best_score < 1.503:
            break

        # gradient from smoothed max
        grad = _softmax_gradient(a, conv, beta_smooth)

        # Adam update (biased)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # step size (decreases like 1/sqrt(t) to guarantee stability)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        step = step / np.sqrt(t_iter)

        # line‑search like back‑tracking (project after each trial)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1):
            a_trial = _project_to_simplex(a - factor * step)
            trial_score = 2.0 * n * np.max(_conv_full(a_trial))
            if trial_score < cur_score - 1e-12:
                a = a_trial
                improved = True
                break

        if not improved:
            # occasional random jitter to escape shallow plateaus
            if rng.random() < 0.03:
                a_pert = a.copy()
                ix = rng.choice(n, size=min(5, n), replace=False)
                a_pert[ix] += 0.005 * rng.standard_normal(ix.shape)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                a = a_pert

        # increase smoothing parameter gradually
        if t_iter % 200 == 0 and beta_smooth < target_beta:
            beta_smooth = min(beta_smooth * beta_growth, target_beta)

    final_score = 2.0 * n * np.max(_conv_full(best_a))
    return best_a, final_score

            # adjust length if needed by padding/ truncating
