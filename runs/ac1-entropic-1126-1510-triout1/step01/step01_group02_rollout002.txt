"""
Optimising the infinity‑norm of a self‑convolution on the probability simplex.

The objective for a normalised step‑function a (∑a=1) is
    score(a) = 2·n·max_k (a * a)[k] .
We minimise this by a smooth‑approximation of the max (soft‑max) and an
exponentiated‑gradient (mirror‑descent) update on the simplex.
A small amount of random perturbation is injected periodically to break
symmetry and escape local minima.  The best sequence found before the
time budget expires is returned.

Key ideas
---------
* Use FFT‑based convolution / correlation – O(n log n) per iteration.
* Approximate max by soft‑max with a temperature α that grows during the run.
* Mirror‑descent with KL‑projection (multiplicative update + renormalisation).
* Occasional random “noise‑mix” steps and restart‑like perturbations.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities -----------------------------------------------------
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise (numerical safety)
    s = w.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / n)
    return w / s


def _next_power_of_two(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Convolution / correlation (FFT) ---------------------------------------
# ----------------------------------------------------------------------
def _conv_self_fft(seq: np.ndarray, size: int) -> np.ndarray:
    """Full convolution of seq with itself, using a pre‑computed FFT size."""
    fft_seq = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_seq * fft_seq, size)[: 2 * seq.size - 1]
    return conv


def _correlate_fft(w: np.ndarray, rev_seq: np.ndarray, size: int) -> np.ndarray:
    """
    Linear correlation of w (length 2n‑1) with rev_seq (length n).

    Returns the length‑n array g_i = Σ_j w_{i+j}·rev_seq_j
    (equivalently g_i = Σ_j w_{i+j}·seq_{n‑1‑j}).
    """
    fft_w = np.fft.rfft(w, size)
    fft_rev = np.fft.rfft(rev_seq, size)
    corr_full = np.fft.irfft(fft_w * fft_rev, size)[: w.size + rev_seq.size - 1]
    start = rev_seq.size - 1          # = n‑1
    return corr_full[start:start + rev_seq.size]   # length n


# ----------------------------------------------------------------------
# Scoring ---------------------------------------------------------------
# ----------------------------------------------------------------------
def _score(seq: np.ndarray, conv_size: int) -> float:
    """Fast evaluation of 2·n·max(conv(seq,seq)) – exact for sum(seq)=1."""
    conv = _conv_self_fft(seq, conv_size)
    return 2.0 * seq.size * float(conv.max())


# ----------------------------------------------------------------------
# Random perturbations ---------------------------------------------------
# ----------------------------------------------------------------------
def _random_dirichlet_like(n: int) -> np.ndarray:
    """Generate a random point on the simplex (Gamma(1,1) ⇒ Dirichlet(1))."""
    rnd = np.random.rand(n)
    rnd = np.maximum(rnd, 1e-12)          # avoid exact zeros
    return rnd / rnd.sum()


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a short safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 1.5)

    # --------------------------------------------------------------
    # Initialise the distribution
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` is expected to be a global variable.
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        base = base / max(base.sum(), 1e-12)   # normalise to sum=1
    except Exception:
        # fallback – a modestly large uniform vector
        n = 8000
        base = np.full(n, 1.0 / n, dtype=float)

    # break symmetry a little (helps the optimiser when only a uniform start)
    if np.allclose(base, 1.0 / n):
        base = base * 0.9 + _random_dirichlet_like(n) * 0.1
        base = base / base.sum()

    current = _project_simplex(base.copy())
    best = current.copy()
    # pre‑compute FFT sizes (they are constant throughout the run)
    conv_fft_size = _next_power_of_two(2 * n - 1)
    corr_fft_size = _next_power_of_two((2 * n - 1) + n - 1)

    best_score = _score(best, conv_fft_size)
    current_score = best_score

    # --------------------------------------------------------------
    # Hyper‑parameters (all of them decay / grow with the iteration count)
    # --------------------------------------------------------------
    max_iters_without_improve = 8000
    iter_last_improve = 0
    eps = 1e-12                     # floor to keep numbers strictly positive
    it = 0

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # Adaptive schedules
        # ------------------------------------------------------------------
        # step‑size for the exponentiated‑gradient update
        eta = 0.6 / math.sqrt(it)

        # soft‑max temperature – starts gentle and grows (sharper) over time
        alpha = min(180.0, 2.0 * math.sqrt(it))

        # linear cooling temperature for the Metropolis acceptance (optional)
        elapsed_frac = (time.time() - start_time) / (deadline - start_time)
        temperature = 1.0 * (1.0 - elapsed_frac) + 1e-4 * elapsed_frac

        # ------------------------------------------------------------------
        # 1) Compute self‑convolution (FFT)
        # ------------------------------------------------------------------
        conv = _conv_self_fft(current, conv_fft_size)
        max_conv = float(conv.max())

        # ------------------------------------------------------------------
        # 2) Soft‑max weights (smooth approximation of the max)
        # ------------------------------------------------------------------
        # shift by max_conv for numerical stability (max weight == 1)
        shifted = conv - max_conv
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0:

        else:
            w /= w_sum

        # ------------------------------------------------------------------
        # 3) Gradient of the smooth objective (correlation w * current)
        # ------------------------------------------------------------------
        rev_cur = current[::-1]
        grad = 2.0 * _correlate_fft(w, rev_cur, corr_fft_size)   # length n, non‑negative

        # ------------------------------------------------------------------
        # 4) Exponentiated‑gradient (mirror descent with KL‑projection)
        # ------------------------------------------------------------------
        mult = np.exp(-eta * grad)
        # keep values strictly positive before normalisation
        cand = np.maximum(current * mult, eps)
        cand /= cand.sum()

        # ------------------------------------------------------------------
        # 5) Small random mixing to avoid stagnation (decays with √it)
        # ------------------------------------------------------------------
        if it % 50 == 0:
            mix_factor = 0.01 / math.sqrt(it)          # 0.01 at it=1, quickly shrinks
            rnd = _random_dirichlet_like(n)
            cand = _project_simplex((1.0 - mix_factor) * cand + mix_factor * rnd)

        # ------------------------------------------------------------------
        # 6) Evaluate candidate (fast internal score)
        # ------------------------------------------------------------------
        cand_score = _score(cand, conv_fft_size)

        # ------------------------------------------------------------------
        # 7) Keep the globally best sequence
        # ------------------------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            iter_last_improve = it
            print(f"[Iter {it:6d}] New best score: {best_score:.6f}")

        # ------------------------------------------------------------------
        # 8) Acceptance rule (Metropolis style – allows occasional uphill moves)
        # ------------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # probability to accept a worse move
            prob = math.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score

        # ------------------------------------------------------------------
        # 9) Simple restart if we have been stuck for a while
        # ------------------------------------------------------------------
        if it - iter_last_improve > max_iters_without_improve:
            # re‑inject a fresh random component around the best point
            pert = _random_dirichlet_like(n)
            current = _project_simplex(
                0.97 * best + 0.03 * pert
            )
            current_score = _score(current, conv_fft_size)
            iter_last_improve = it
            print(f"[Iter {it:6d}] Restarted – stagnation detected")

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    return best.tolist()
