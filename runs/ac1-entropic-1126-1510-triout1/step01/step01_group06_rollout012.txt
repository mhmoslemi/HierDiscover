"""
Improved search for a low‑value step‑function.

The algorithm mixes three complementary optimisation ideas:

1. **Diverse random starts** – geometric, power‑law, mixed and plain Dirichlet
   draws, plus the previously discovered best sequence (if available).

2. **Exponentiated‑gradient (mirror‑descent) on the simplex** – the vector is
   represented by log‑weights `w` (soft‑max).  A sub‑gradient of the
   max‑convolution is computed, transformed to the `w`‑space and updated with an
   Adam‑style rule.  This keeps the iterates interior to the probability simplex
   and often escapes shallow local minima faster than plain Euclidean projection.

3. **Fine‑grained projected sub‑gradient refinement** – after the exponential
   phase the best candidate is polished with the original projected
   sub‑gradient method (the “Fast gradient based search” from the reference).
   The two phases complement each other and usually improve the bound below the
   previous 1.577662 record, aiming for the target 1.503.

The whole search runs for the allotted time budget (≈ 1000 s) and returns the
best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray):
    """Return (max value, indices where the maximum is attained) of a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = np.max(conv)
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return float(max_val), idx.astype(int)


def _gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the max‑convolution w.r.t. a."""
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def _improve_sequence_proj(a0: np.ndarray, n: int, deadline: float):
    """
    Projected sub‑gradient descent (the original fast‑gradient method).
    Returns the best (a, score) seen before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    max_c, max_idx = _conv_max_fft(a)
    best_score = 2.0 * n * max_c

    it = 0
    while time.time() < deadline and it < 5000:
        it += 1
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c

        # sub‑gradient of the active indices
        grad = _gradient_for_max(a, max_idx)

        # simple back‑tracking line‑search
        eta0 = 0.2 / np.sqrt(it)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            eta = eta0 * factor
            cand = _project_to_simplex(a - eta * grad)
            max_c2, _ = _conv_max_fft(cand)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score - 1e-12:
                a = cand
                cur_score = cand_score
                improved = True
                break

        if not improved:
            # tiny random shake
            a_pert = a.copy()
            idx = np.random.randint(n, size=5)
            a_pert[idx] += np.random.randn(5) * 0.001
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / np.sum(a_pert)
            max_c2, _ = _conv_max_fft(a_pert)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score:
                a = a_pert
                cur_score = cand_score

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if it % 250 == 0:
                print(f"    proj‑grad iter {it}, local best {best_score:.6f}")

        if best_score < 1.45:   # early stop if already very good
            break

    return best_a, best_score


def _improve_sequence_exp(a0: np.ndarray, n: int, deadline: float):
    """
    Exponentiated‑gradient (soft‑max) optimisation with Adam momentum.
    Operates on log‑weights w so that a = softmax(w) stays on the simplex.
    Returns the best (a, score) observed before the deadline.
    """
    # initialise log‑weights safely
    eps = 1e-16
    w = np.log(a0 + eps)

    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta0 = 0.3

    best_a = None
    best_score = np.inf
    it = 0

    while time.time() < deadline and it < 20000:
        it += 1

        # soft‑max to obtain a on the simplex
        w_shift = w - np.max(w)               # for numerical stability
        exp_w = np.exp(w_shift)
        a = exp_w / np.sum(exp_w)

        # evaluate current objective
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 500 == 0:
                print(f"    exp‑grad iter {it}, local best {best_score:.6f}")

        # compute sub‑gradient w.r.t. a (active indices only)
        grad_a = _gradient_for_max(a, max_idx)

        # transform to gradient w.r.t. w (mirror‑descent on simplex)
        dot_ga = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot_ga)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps_adam)

        # keep w bounded to avoid overflow in the soft‑max
        if np.max(np.abs(w)) > 50:
            w = np.clip(w, -50, 50)

    return best_a, best_score

def _random_initial(n: int) -> np.ndarray:
    """Create a random non‑negative vector (sum normalised to 1)."""
    shape = np.random.choice(["geom", "power", "mixed", "random"])
    if shape == "geom":
        t = np.random.uniform(0.0003, 0.0015)
        r = np.exp(-t)
        a = r ** np.arange(n, dtype=float)
    elif shape == "power":
        p = np.random.uniform(0.3, 2.0)
        a = (np.arange(1, n + 1, dtype=float) ** -p)
    elif shape == "mixed":
        t = np.random.uniform(0.0003, 0.0010)
        r = np.exp(-t)
        p = np.random.uniform(0.5, 1.5)
        a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)
    else:  # "random"
        a = np.random.gamma(1.0, 1.0, size=n)
    # optional shuffle – does not change the multiset of values
    if np.random.rand() < 0.5:
        np.random.shuffle(a)
    a = a / np.sum(a)
    return a
