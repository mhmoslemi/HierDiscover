def _exp_gradient_step(seq, eta, beta):
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # Compute softmax weights over conv entries scaled by beta
    # Avoid overflow: subtract max.
    scaled = beta * conv
    scaled -= scaled.max()
    # Compute exponentials safely
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    # Compute gradient g_i = 2 * sum_k w_k * seq_{k-i}
    # That's cross-correlation: compute via convolution.

    # Use correlation: g = 2 * (np.convolve(seq, w[::-1]))[len(w)-1 - np.arange(len(seq))]
    # Since w length L = 2n-1 same as conv length
    rev_w = w[::-1]
    conv2 = np.convolve(seq, rev_w)  # full length n+L-1
    L = len(w)
    n = len(seq)
    # Get slice indices:
    start = L - 1 - (n - 1)  # L-1 - (n-1) = L-n
    # Actually we need indices t = L-1 - i for i=0..n-1 => slice conv2[t] for t from L-1 down to L-n.
    # That's conv2[L-1 - np.arange(n)]
    g = 2.0 * conv2[L-1 - np.arange(n)]

    # Exponential update:
    new_seq = seq * np.exp(-eta * g)
    # Normalize onto simplex:
    new_seq = new_seq / max(new_seq.sum(), 1e-12)  # ensure sum to 1
    return new_seq

"""
Hybrid stochastic optimizer for the step‑function bound.

The algorithm works on the probability simplex (weights ≥0, sum = 1) so the
scaling freedom of ``evaluate_sequence`` disappears.  It mixes three
move families:

*  Projected gradient descent on the (hard) max‑convolution entry.
   This is the same step used by the reference code.

*  Exponentiated‑gradient descent on a *soft* maximum.
   The convolution values are turned into a soft‑max distribution
   (temperature β) and the gradient of the corresponding smooth objective
   is applied multiplicatively to the current vector.  Increasing β during
   the run makes the objective approach the true ∞‑norm while keeping the
   updates well‑behaved.

*  Simple random pairwise mass swaps that preserve the simplex constraint.

Moves are accepted with a Metropolis rule using a linear temperature
schedule, and the best vector seen so far is kept.  Every few thousand
iterations a short deterministic gradient sweep is performed to clean up
any remaining “spikes”.  The routine runs until the supplied time budget
expires and returns the best list of heights found.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional – not used in the final loop

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Projected gradient step (hard max of the convolution)
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """One projected gradient descent step on the maximal convolution entry."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)

    g = np.zeros_like(seq)
    g[mask] = 2.0 * seq[j[mask]]

    new_seq = seq - eta * g
    return _project_simplex(new_seq)


# ----------------------------------------------------------------------
# Random pairwise swap (keeps the sum exactly 1)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two random coordinates."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])      # take from j
    else:
        delta = -min(-delta, seq[i])    # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient step based on a soft‑max of the convolution
# ----------------------------------------------------------------------
def _exp_gradient_step(seq: np.ndarray, eta: float, beta: float) -> np.ndarray:
    """
    Soft‑max (temperature beta) version of the max‑convolution gradient.
    The convolution is turned into a distribution w ∝ exp(beta * conv),
    the gradient g_i = 2 Σ_k w_k * a_{k-i} is computed via a cross‑correlation,
    and the update a_i ← a_i·exp(-η g_i) is performed followed by simplex projection.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)                # full convolution, length 2n‑1
    # soft‑max over convolution entries
    scaled = beta * conv
    scaled -= scaled.max()                      # shift for numerical stability
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation of seq with w (g_i = 2 Σ_k w_k * a_{k-i})
    # Using convolution with a reversed copy of w.
    rev_w = w[::-1]
    corr = np.convolve(seq, rev_w)              # length 3n‑2
    L = len(w)                                   # = 2n‑1
    g = 2.0 * corr[L - 1 - np.arange(n)]        # gather the required slice

    # multiplicative (exponentiated) update, then re‑normalise onto the simplex
    new_seq = seq * np.exp(-eta * g)
    s = new_seq.sum()
    if s == 0:
        new_seq = np.full_like(seq, 1.0 / n)
    else:
        new_seq /= s
    return new_seq


# ----------------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 1.0)   # leave a 1‑second safety margin

    # ------------------------------------------------------------------
