def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # define lengths: try bigger lengths gradually
    base_n = 2000
    try: base_n = len(height_sequence_1)
    except: pass
    lengths = [base_n, int(base_n*1.5), int(base_n*2), int(base_n*2.5), int(base_n*3)]
    lengths = [n for n in lengths if n >= 200]

    best_seq = None
    best_score = float('inf')
    for n in lengths:
        # allocate time slice about 20% of remaining
        time_slice = max(0.2*(deadline - time.time()), 0.1)
        slice_deadline = time.time() + time_slice

        # generate multiple restarts within slice time
        start_time = time.time()
        n_restarts = 0
        while time.time() < slice_deadline:
            n_restarts += 1
            # init: combine arcsine + random perturbations
            a = _arcsine_distribution(n)
            # add small random noise (0 to 1% total mass) and project to simplex
            noise = np.random.rand(n) * 0.001
            a = a + noise
            a = _project_onto_simplex(a, s=1.0)

            # choose hyperparameters
            eta0 = 0.2
            beta = 0.9   # momentum
            momentum = np.zeros_like(a)
            max_iters = 5000   # may break early due to time

            # start smoothed max schedule
            alpha_start = 2.0
            alpha_end = 30.0
            # linear increase over iterations
            for it in range(1, max_iters+1):
                if time.time() >= slice_deadline:
                    break

                # compute convolution
                b = np.convolve(a, a)
                max_b = b.max()
                cur_score = 2.0 * n * max_b
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] New best score {best_score:.6f} length={n} it={it}")

                # smoothing parameter
                alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / max_iters)

                # compute weights w = exp(alpha * b)
                # might cause overflow; subtract max before exponent
                b_max = b.max()
                exp_arg = alpha * (b - b_max)
                w = np.exp(exp_arg)
                w_sum = w.sum()
                w = w / w_sum   # softmax, sum to 1

                # compute gradient: g = 2 * conv(w, a) slice
                # use FFT for speed
                # Determine size for FFT: next power of 2 >= len(w) + n -1
                conv_len = len(w) + n - 1
                N = 1 << (conv_len - 1).bit_length()
                # compute FFT of w and a
                W = np.fft.rfft(w, n=N)
                A = np.fft.rfft(a, n=N)
                conv_full = np.fft.irfft(W * A, n=N)  # real
                # slice indices n-1 : n-1 + n
                g = 2.0 * conv_full[n-1 : n-1 + n]

                # momentum update
                momentum = beta * momentum + (1 - beta) * g

                # step size
                eta = eta0 / np.sqrt(it)
                a = a - eta * momentum

                # project onto simplex (ensuring nonnegative sum=1)
                a = _project_onto_simplex(a, s=1.0)

                # occasionally random jitter
                if it % 300 == 0 and np.random.rand() < 0.1:
                    a += np.random.randn(n) * 1e-5
                    a = _project_onto_simplex(a, s=1.0)

        print(f"Length {n} done, restarts {n_restarts}")
        if time.time() >= deadline:
            break

    if best_seq is None:
        # fallback uniform
        n_fallback = 500
        best_seq = np.full(n_fallback, 1.0/n_fallback)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()

"""
Propose a step‑function (list of non‑negative heights) that minimizes
`evaluate_sequence`.

Algorithm
---------
* Use the discrete arcsine law as a sensible starting shape.
* Perform a two‑phase projected gradient descent on the *squared‑error*
  of the autocorrelation:  f(a)=∑_k ( (a∗a)_k – 1/(2n‑1) )² .
  This objective is smooth and drives the whole autocorrelation towards a
  constant value, which is the theoretical optimum.
* In the first phase (≈70 % of the iterations) we minimise the
  squared‑error using the exact gradient obtained via FFT.
* In the second phase we switch to a sub‑gradient update that directly
  reduces the current maximal convolution entry (the quantity that appears
  in `evaluate_sequence`).  The switch removes the remaining “peak’’ after
  the smoothing phase.
* After each gradient step we project back onto the probability simplex
  (non‑negative entries with unit sum) using the Euclidean projection.
* The optimisation is repeated for several lengths (the historic length,
  1.5× and 2× that size) and with a few random restarts.  The best
  sequence found before the deadline is returned.
* All heavy work (convolutions, gradients) is performed with FFTs, so
  even a few thousand iterations on a length of several thousands run
  comfortably inside the 1000 s budget.

The code respects the time limit, uses only NumPy and the standard
library, and prints occasional progress reports.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=s}
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Discrete arcsine distribution (probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights with a low
    `evaluate_sequence` value.
    """
    np.random.seed(seed)
    # keep a few seconds for the wrapper that calls us
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Determine a base length (use the historic length if it exists)
    try:
        base_n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        base_n = 2000

    # Lengths to explore (ensure they are at least 200)
    length_factors = (1.0, 1.5, 2.0)
    lengths = [max(200, int(base_n * f)) for f in length_factors]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        # each length receives roughly 30 % of the remaining time
        slice_deadline = time.time() + max(0.5,
                                          (deadline - time.time()) * 0.3)

        restarts = 0
        while time.time() < slice_deadline:
            restarts += 1

            # ---- initialise -------------------------------------------------
            a = _arcsine_distribution(n)
            # add a tiny random jitter to escape possible symmetric traps
            a += 0.001 * np.random.rand(n)
            a = _project_onto_simplex(a, s=1.0)

            # optimisation hyper‑parameters
            max_iters = 5000
            phase_cut = int(0.7 * max_iters)   # switch point between phases
            eta0 = 2.0                         # initial step‑size factor
            beta = 0.9                         # momentum coefficient
            momentum = np.zeros_like(a)
            avg = 1.0 / (2 * n - 1)            # target autocorrelation value

            # size of the FFT (next power‑of‑two covering the full convolution)
            conv_len = 3 * n - 2
            N_fft = 1 << ((conv_len - 1).bit_length())

            for it in range(1, max_iters + 1):
                # abort if we run out of time
                if time.time() >= slice_deadline or time.time() >= deadline:
                    break

                # ----- FFT of the current sequence -------------------------
                F_a = np.fft.rfft(a, n=N_fft)

                # ----- autocorrelation a * a --------------------------------
                b_full = np.fft.irfft(F_a * F_a, n=N_fft)
                b = b_full[: 2 * n - 1]               # linear convolution part

                max_b = b.max()
                cur_score = 2.0 * n * max_b

                # keep the globally best candidate
                if cur_score < overall_best_score:
                    overall_best_score = cur_score
                    overall_best_seq = a.copy()
                    print(f"[{time.time():.1f}] NEW BEST score={overall_best_score:.6f} "
                          f"len={n} iter={it}")

                # ------------------------------------------------------------
                # Phase‑1 : minimise the squared error of the autocorrelation
                # Phase‑2 : explicit sub‑gradient of the maximal entry
                # ------------------------------------------------------------
                if it <= phase_cut:
                    # squared‑error gradient:  g = 2 * conv_rev(R_rev, a)
                    R = b - avg                         # residual
                    R_rev = R[::-1]
                    F_R = np.fft.rfft(R_rev, n=N_fft)
                    conv_rev_full = np.fft.irfft(F_R * F_a, n=N_fft)
                    g = 2.0 * conv_rev_full[n - 1: n - 1 + n]
                else:
                    # sub‑gradient coming from the active (max) convolution entry
                    k_max = int(np.argmax(b))
                    idx = k_max - np.arange(n)
                    mask = (idx >= 0) & (idx < n)
                    g = np.zeros_like(a)
                    g[mask] = 2.0 * a[idx[mask]]

                # ----- momentum update ---------------------------------------
                momentum = beta * momentum + (1.0 - beta) * g

                # ----- diminishing step size ---------------------------------
                eta = eta0 / np.sqrt(it)

                # ----- gradient step + projection onto the simplex ------------
                a = a - eta * momentum
                a = _project_onto_simplex(a, s=1.0)

                # ----- occasional tiny random perturbation to avoid plateaus
                if it % 500 == 0 and np.random.rand() < 0.1:
                    a += np.random.randn(n) * 1e-4
                    a = _project_onto_simplex(a, s=1.0)

            # end of one optimisation run (a restart)

            if time.time() >= deadline:
                break

        print(f"Finished length={n} after {restarts} restart(s). "
              f"Best score so far: {overall_best_score:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # If nothing was produced (unlikely) fall back to a simple uniform vector
    if overall_best_seq is None:
        n_fallback = 500
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"FINAL BEST score={overall_best_score:.6f} "
          f"(length={len(overall_best_seq)})")
    # Convert to plain Python list as required by the evaluator
    return overall_best_seq.tolist()
