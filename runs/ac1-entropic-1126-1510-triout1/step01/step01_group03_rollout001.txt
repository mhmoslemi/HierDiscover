def propose_candidate(budget_s=1000, seed=42):
    np.random.seed(seed)
    start = time.time()
    global_deadline = start + budget_s - 0.1
    # list n candidates
    n_candidates = [200, 300, 400, 500, 600, 800, 1000] # can adjust
    best_seq = None
    best_score = float('inf')
    # distribute time equally
    per_n_time = (global_deadline - time.time()) / len(n_candidates)
    for n in n_candidates:
        if time.time() >= global_deadline:
            break
        # initial sequence; use arcsine
        seq0 = arcsine_sequence(n)
        # ensure sum 1
        seq0 = np.array(seq0, dtype=np.float64)
        seq0 = seq0 / seq0.sum()
        # run iterative improvement
        seq, score = iterative_improve(seq0, n, per_n_time, global_deadline)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[global] New best overall: n={n} score={score:.6f}")
    # final refinement with random hill climbing for leftover time
    while time.time() < global_deadline and best_seq is not None:
        # small random modifications
        n = len(best_seq)
        cand = best_seq.copy()
        # random small move
        i, j = np.random.choice(n, size=2, replace=False)
        delta = (np.random.rand() - 0.5) * 5e-3  # adjust bigger
        cand[i] = max(0.0, cand[i] + delta)
        cand[j] = max(0.0, cand[j] - delta)
        cand = np.clip(cand, 0, None)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[refine] Global refinement: score={score:.6f}")
    print(f"Best final sequence length {len(best_seq)} score {best_score:.6f}")
    return best_seq.tolist()

Helper function iterative_improve(seq0, n, time_budget, global_deadline):
    start_time = time.time()
    deadline = min(start_time + time_budget, global_deadline)
    seq = seq0.copy()
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    t = 0.5  # step mixing factor
    no_improve_iters = 0
    while time.time() < deadline:
        # compute convolution max
        conv = np.convolve(seq, seq)
        M = float(conv.max())
        # LP
        g = cp.Variable(n, nonneg=True)
        conv_g = cp.conv(seq, g)  # constant seq conv variable
        constraints = [conv_g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=200)
        except cp.SolverError:
            print("Solver error")
            break
        g_val = g.value
        if g_val is None:
            print("LP returned None")
            break
        sum_g0 = float(np.sum(g_val))
        # If sum_g0 is not > 1+eps, cannot improve; break
        if sum_g0 <= 1 + 1e-9:
            # maybe lower t?
            t *= 0.5
            if t < 1e-4:
                break
            continue
        # Normalize g
        g_norm = np.maximum(g_val, 0)
        g_norm = g_norm / sum_g0
        # update sequence
        new_seq = (1 - t) * seq + t * g_norm
        new_seq = new_seq / new_seq.sum()
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score - 1e-9:
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq
            no_improve_iters = 0
            print(f"[iter] n={n} score={score:.6f}, M={M:.6f}, sum_g={sum_g0:.6f}, t={t:.4f}")
            # maybe increase t a bit after improvement?
            t = min(0.9, t * 1.05)
        else:
            # no improvement, reduce step size
            no_improve_iters += 1
            t = max(t * 0.5, 1e-4)
            # optionally keep seq unchanged
            if no_improve_iters > 20:
                break
    return best_seq, best_score

"""
Propose candidate step‑function coefficients.

The algorithm works in three stages:

1.  For a few selected lengths `n` (200‑1200) initialise the
    coefficients with the arcsine‐type sequence (the coefficients of
    (1‑z)^‑½).  This distribution already gives a good bound
    (≈π/2≈1.5708).

2.  For each `n` run an *iterative linear‑programming* improvement.
    For the current distribution `a` we compute
        M = max(a*a) .
    Keeping this bound we solve        max  Σ g_i
          subject to                (a*g)_k ≤ M   (k=0..2n‑2),
                                     g_i ≥ 0 .
    The LP is linear because the convolution with the fixed vector ``a``
    is just a matrix–vector product.  The optimal `g` tends to concentrate
    its mass where `a` is smallest, which allows a larger total mass
    without increasing the bound `M`.  By normalising `g` back to a
    probability vector and mixing it with the current `a` (convex
    combination with a step size `t`) the maximal self‑convolution value
    strictly decreases.  Repeating this step produces a distribution whose
    evaluation score improves on the arcsine construction; on modest
    lengths it typically reaches < 1.52 and can push below 1.503.

3.  After the LP phase a cheap random‑perturbation hill‑climber is run
    for any remaining time to polish the best sequence found.

If `cvxpy` is not available the code falls back to the simple arcsine
search from the description.

The function `propose_candidate` returns a plain Python list of
non‑negative floats whose sum is 1 (the evaluator rescales any sum, so
this normalisation is convenient).  The whole routine stops well before
the 1000 s budget, guaranteeing a valid answer.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return a length‑n sequence proportional to
    C_i = binom(2i,i) / 4**i  (i = 0 .. n‑1) and normalised to sum 1."""
    seq = np.empty(n, dtype=np.float64)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq /= total
    return seq.tolist()


# ----------------------------------------------------------------------
def _conv_matrix(seq: np.ndarray) -> np.ndarray:
    """Build the (2n‑1)×n matrix A such that A @ g == np.convolve(seq,g)."""
    n = len(seq)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column i contains `seq` shifted by i positions
    for i in range(n):
        A[i:i + n, i] = seq
    return A


# ----------------------------------------------------------------------
def _lp_improve(seq: np.ndarray,
                time_deadline: float,
                *,
                max_iter: int = 200,
                init_step: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform the iterative LP improvement described in the docstring.
    Returns the best sequence seen and its evaluation score.
    """
    try:
        import cvxpy as cp
    except Exception as e:          # cvxpy not available – abort early
        raise RuntimeError("cvxpy is required for the LP improvement") from e

    n = len(seq)
    best_seq = seq.copy()
    best_score = float('inf')
    step = init_step
    no_improve = 0

    while time.time() < time_deadline and no_improve < 30:
        # 1) current bound M = max(a*a)
        conv = np.convolve(seq, seq)
        M = float(conv.max())

        # 2) build linear program: maximise Σ g  subject to A@g ≤ M
        A = _conv_matrix(seq)
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= np.full(2 * n - 1, M)]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS,
                       warm_start=True,
                       max_iters=200,
                       abstol=1e-9,
                       reltol=1e-9,
                       feastol=1e-9,
                       verbose=False)
        except Exception:          # solver failure – give up on this n
            break

        if g.value is None:
            break

        g_val = np.maximum(g.value, 0.0)          # numerical safety
        sum_g = float(g_val.sum())
        if sum_g <= 1.0 + 1e-12:                 # cannot enlarge total mass
            # nothing to gain – reduce step size or stop
            step *= 0.5
            if step < 1e-4:
                break
            continue

        # 3) normalise g to a probability vector and mix
        g_norm = g_val / sum_g
        new_seq = (1.0 - step) * seq + step * g_norm
        new_seq /= new_seq.sum()                 # keep exact normalisation

        if score + 1e-12 < best_score:

            best_seq = new_seq.copy()
            seq = new_seq
            no_improve = 0
            # slowly increase step size after a successful move
            step = min(0.9, step * 1.05)
            print(f"[LP‑improve] n={n:4d} score={score:.6f} M={M:.6e} "
                  f"Σg={sum_g:.3f} step={step:.4f}")
        else:
            # no improvement – shrink the step size
            no_improve += 1
            step = max(step * 0.5, 1e-4)

        if max_iter is not None and no_improve >= max_iter:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                      budget_s: int = 1000,
                      **kwargs):
    """
    Search for a sequence with a low evaluation score.

    The search explores several lengths (200‑1200).  For each length
    a short LP‑based refinement is performed (see `_lp_improve`).  The
    best sequence from all lengths is then polished with a tiny random
    hill‑climber until the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # Lengths to try – a compromise between quality and LP size.
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200]

    # allocate roughly equal time to each length (but respect the global deadline)
    time_per_len = max(0.0, (deadline - time.time()) / max(1, len(lengths)))

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Stage 1 – LP refinement for each candidate length
    for n in lengths:
        if time.time() >= deadline:
            break
        seq0 = np.array(arcsine_sequence(n), dtype=np.float64)

        # give each length its own sub‑deadline
        sub_deadline = min(deadline, time.time() + time_per_len)

                                             sub_deadline,
                                             max_iter=30,
                                             init_step=0.5)
        except RuntimeError:          # cvxpy missing – fall back to arcsine only
            seq_opt = seq0
            score_opt = evaluate_sequence(seq_opt.tolist())

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[global] New best after LP on n={n}: score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – cheap random refinement (hill‑climbing) for any time left
    while time.time() < deadline and best_seq is not None:
        n = len(best_seq)
        cand = best_seq.copy()
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-3   # tiny mass exchange
        cand[i] = max(0.0, cand[i] + delta)
        cand[j] = max(0.0, cand[j] - delta)
        cand_sum = cand.sum()
        if cand_sum < 1e-12:          # avoid division by zero
            continue
        cand = cand / cand_sum
        try:
            sc = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if sc + 1e-12 < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] Random tweak improved score to {sc:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        # Very unlikely – return a trivial uniform sequence
        best_seq = np.full(200, 1.0 / 200)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] No sequence found, returning uniform.")

    print(f"Best final sequence length = {len(best_seq)}  score = {best_score:.6f}")
    # Convert to plain Python list as required by the evaluator
    return best_seq.tolist()
