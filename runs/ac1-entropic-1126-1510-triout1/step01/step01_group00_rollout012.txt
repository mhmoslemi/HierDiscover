```
def convolve_fft(a,b):
    n = len(a) + len(b) - 1
    N = 1 << (n-1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    c = np.fft.irfft(C, N)
    return c[:n]
```

```
def gradient_max_smooth(a, alpha):
    n = len(a)
    b = convolve_fft(a, a)
    max_b = b.max()
    # compute softmax weights w
    # To avoid overflow:
    shift = b.max()
    exp_vals = np.exp(alpha * (b - shift))
    w = exp_vals / exp_vals.sum()
    # compute conv of w with reversed a
    a_rev = a[::-1]
    conv = convolve_fft(w, a_rev)
    # slice n elements from index n-1
    g = 2.0 * conv[n-1:n-1+n]
    return b, max_b, g, w
```

```
def optimise_sequence(a0, deadline, alpha0=5000, alpha_inc_factor=1.01, max_iters=20000, lr=0.01):
    a = a0.copy()
    n = len(a)
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    alpha=alpha0
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute gradient etc.
        b, max_b, g, w = gradient_max_smooth(a, alpha)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={t} score={best_score:.6f}")
        # Adam updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # update alpha schedule
        alpha = alpha * alpha_inc_factor
        # maybe clamp alpha not too high to avoid numerical underflow.
        if alpha > 1e7:
            alpha = 1e7
    return best_seq, best_score
```

```
    a = a0.copy()
    n = len(a)
    best_seq = a.copy()
    b = np.convolve(a, a)
    best_score = float(2 * n * b.max())
    # Adam states
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta2 = 0.999
    eps = 1e-8
    t = 0
    alpha = alpha0
    while time.time() < time_deadline and t < max_iters:
        t += 1
        # compute conv and gradient
        b = np.convolve(a, a)  # we could use FFT for speed: but for moderate n maybe ok

        shift = max_b

        # compute gradient conv(w, a_rev)
        a_rev = a[::-1]

        # update best score
```

"""
Propose a step‑function (list of non‑negative heights) with a low
`evaluate_sequence` score.

The algorithm builds a discrete arcsine distribution (the known
continuous‑time optimum) and refines it with a *projected Adam* optimiser.
We minimise a smooth approximation of the objective
    f(a) = max_i (a * a)[i]
by replacing the max with a soft‑max (log‑sum‑exp) using a temperature
parameter `alpha`.  The gradient of this smooth objective is the weighted
sum of the gradients of the individual convolution entries; this can be
computed efficiently with two FFT‑based convolutions per iteration.
After each Adam step the vector is projected onto the probability simplex
(`a_i ≥ 0`, `∑ a_i = 1`).  The temperature `alpha` is increased
gradually so that the optimisation finally targets the true
non‑smooth objective.

Multiple starts (arcsine and a random vector) are tried within the
available time budget, and the best sequence found is returned as a plain
Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper: project onto the probability simplex (non‑negative, sum = s)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: discrete arcsine distribution, normalised to unit ℓ₁‑mass
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# FFT‑based full convolution (real inputs)
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full convolution a * b using real FFT."""
    n = len(a) + len(b) - 1
    # next power of two for speed
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    c = np.fft.irfft(C, N)
    return c[:n]


# ----------------------------------------------------------------------
# Compute smooth max objective and its gradient
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    For a probability vector a, returns:
        max_b   – the (unsmoothed) maximum entry of a*a,
        grad    – gradient of the smoothed max (log‑sum‑exp) w.r.t. a,
        b_full  – the full convolution a*a (needed only for diagnostics).
    """
    n = len(a)
    # full self‑convolution
    b_full = _fft_convolve(a, a)               # length 2n‑1
    max_b = b_full.max()

    # soft‑max weights (stable formulation)
    shift = max_b
    exp_vals = np.exp(alpha * (b_full - shift))
    w = exp_vals / exp_vals.sum()              # length 2n‑1, sum to 1

    # gradient = 2 * (w * a_rev) sliced to length n
    a_rev = a[::-1]
    conv = _fft_convolve(w, a_rev)             # length 3n‑2
    grad = 2.0 * conv[n - 1 : n - 1 + n]        # slice the central n entries
    return max_b, grad, b_full


# ----------------------------------------------------------------------
# Projected Adam optimiser for the smoothed max‑convolution objective
def _optimise_adam(
    a0: np.ndarray,
    deadline: float,
    alpha0: float = 500.0,
    lr: float = 0.04,
    alpha_factor: float = 1.001,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Refine an initial probability vector a0 using Adam with projection onto
    the simplex.  Returns the best vector encountered and its (unsmoothed)
    score = 2·n·max(a*a).
    """
    a = a0.copy()
    n = len(a)

    # bookkeeping of the best solution seen so far
    b0 = _fft_convolve(a, a)
    best_score = float(2.0 * n * b0.max())
    best_seq = a.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    t = 0
    alpha = alpha0

    while time.time() < deadline and t < max_iters:
        t += 1

        max_b, grad, _ = _smooth_max_gradient(a, alpha)
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n} iter={t} "
                f"score={best_score:.6f} alpha={alpha:.1f}"
            )

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # project back onto the simplex (keeps sum = 1 and non‑negativity)
        a = _project_onto_simplex(a, s=1.0)

        # increase the temperature – the objective becomes less smooth
        alpha = min(alpha * alpha_factor, 1e7)

    return best_seq, best_score
