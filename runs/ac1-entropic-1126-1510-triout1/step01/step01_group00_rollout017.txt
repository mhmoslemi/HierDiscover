def _constant_magnitude_projection(a, N):
    # a: 1D array of length n
    n = a.shape[0]
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    # real FFT
    A = np.fft.rfft(a_pad)
    # compute energy (sum of squares)
    sum_sq = np.dot(a, a)  # = np.sum(a*a)
    # compute target magnitude for non‑DC components
    # number of unique frequencies = len(A)
    # first component is DC with magnitude |A[0]| = sum a = 1
    # other components count = M = len(A) - 1
    M = A.shape[0] - 1
    if M > 0:
        # total energy from Parseval: sum_k |A_k|^2 = N * sum_sq
        # => sum_{k>0} |A_k|^2 = N * sum_sq - |A0|^2
        residual = N * sum_sq - np.abs(A[0])**2
        # due to float errors residual could be negative small:
        if residual < 0:
            residual = 0.0
        target_sq = residual / M
        # target magnitude
        target_mag = np.sqrt(target_sq)
        # replace magnitudes for k>0
        phases = np.angle(A[1:])
        A_new = np.empty_like(A)
        A_new[0] = A[0]  # preserve DC
        A_new[1:] = target_mag * np.exp(1j * phases)
    else:
        A_new = A.copy()
    # inverse FFT
    a_new_pad = np.fft.irfft(A_new, n=N)
    a_new = a_new_pad[:n]
    # enforce non‑negativity and simplex constraint
    a_new = np.maximum(a_new, 0.0)

"""
Propose a step‑function (list of non‑negative heights) that minimizes the
`evaluate_sequence` score.

Algorithm
---------
1. **Initialization** – for a chosen length `n` we start from a
   discrete arcsine distribution (the continuous solution of the
   “flat autocorrelation” problem).  If a previously found sequence
   (`height_sequence_1`) of the same length exists we reuse it.

2. **Flat‑spectrum projection** – a sequence with a flat (constant)
   magnitude spectrum has an almost constant self‑convolution.
   In each iteration we:
        * zero‑pad to a power‑of‑two length `N ≥ 2·n‑1`,
        * compute the real FFT,
        * keep the DC component (which equals the ℓ₁‑mass = 1) and replace
          all other magnitudes by the same value that preserves the total
          energy (Parseval’s identity),
        * invert the FFT, truncate to length `n`, clamp negatives to zero,
        * project back onto the probability simplex (non‑negative, sum = 1).

   Repeating this step drives the autocorrelation towards a flat shape.

3. **Fine‑grained sub‑gradient polishing** – after the spectral phase we
   run a short projected sub‑gradient descent on the active
   self‑convolution entry (the maximal one).  This refines the sequence
   locally and often lowers the maximum a little further.

4. **Search over lengths** – several lengths around the historic one are
   tried (½ × n₀, ¾ × n₀, n₀, 5⁄4 × n₀, 3⁄2 × n₀).  Each gets a share of
   the remaining time budget; the best sequence overall is returned.

All helper functions are top‑level, no closures or lambdas are used.
The routine respects the global time budget (default 1000 s) and prints
intermediate scores for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=s}
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: next power of two (at least x)
def _next_power_of_two(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Helper: discrete arcsine distribution of length n (probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Helper: flat‑spectrum projection (keep DC, equalise other magnitudes)
def _constant_magnitude_projection(a: np.ndarray, N: int) -> np.ndarray:
    """Project `a` onto the set of vectors whose non‑DC FFT magnitudes are equal.
    Returns a new non‑negative vector of the same length with ℓ₁‑mass = 1."""
    n = a.shape[0]
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    # real FFT (half‑spectrum)
    A = np.fft.rfft(a_pad)
    # energy of a (ℓ₂‑norm squared)
    sum_sq = np.dot(a, a)
    # number of non‑DC frequency bins in the half‑spectrum
    M = A.shape[0] - 1
    if M > 0:
        # Parseval: Σ|A_k|² = N·Σ a_i²
        residual = N * sum_sq - np.abs(A[0]) ** 2
        residual = max(residual, 0.0)  # numerical safety
        target_sq = residual / M
        target_mag = np.sqrt(target_sq)
        # Preserve DC, set all other magnitudes to target_mag
        phases = np.angle(A[1:])
        A_new = np.empty_like(A)
        A_new[0] = A[0]               # DC = Σ a_i = 1
        A_new[1:] = target_mag * np.exp(1j * phases)
    else:
        A_new = A.copy()
    # inverse FFT
    a_new_pad = np.fft.irfft(A_new, n=N)
    a_new = a_new_pad[:n]
    # enforce non‑negativity and probability simplex
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_onto_simplex(a_new, s=1.0)
    return a_new

# Helper: short sub‑gradient descent on the maximal autocorrelation entry
def _subgradient_descent(
    a: np.ndarray,
    n_iters: int,
    time_deadline: float,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that minimises max(a*a)."""
    n = a.shape[0]
    best_seq = a.copy()
    b = np.convolve(a, a)
    best_score = 2.0 * n * b.max()
    momentum = np.zeros_like(a)
    beta = 0.9  # momentum factor

    for t in range(1, n_iters + 1):
        if time.time() >= time_deadline:
            break
        b = np.convolve(a, a)
        k_max = int(np.argmax(b))
        max_b = b[k_max]
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={t} length={n} score={cur_score:.6f}")

        # sub‑gradient of the active entry (2·a_{k_max-i})
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # step size schedule
        eta = eta0 / np.sqrt(t)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # gradient step + projection
        a = a - eta * momentum
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny jitter to escape plateaus
        if t % 250 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    # return best found in this phase
    return best_seq, best_score

# Core optimisation for a single length
def _optimise_one_length(
    n: int,
    time_deadline: float,
    init_seq: np.ndarray | None = None,
    max_proj_iters: int = 500,
    max_subgrad_iters: int = 1500,
) -> tuple[np.ndarray, float]:
    """Optimise a sequence of length `n` within the given time budget."""
    # initialise
    if init_seq is not None:
        a = init_seq.copy()
        a = _project_onto_simplex(a, s=1.0)
    else:
        a = _arcsine_distribution(n)

    best_seq = a.copy()
    b = np.convolve(a, a)
    best_score = 2.0 * n * b.max()

    # pre‑compute FFT length (power of two ≥ 2n‑1)
    N = _next_power_of_two(2 * n - 1)

    # ---------- Phase 1: flat‑spectrum projection ----------
    for it in range(1, max_proj_iters + 1):
        if time.time() >= time_deadline:
            break
        a = _constant_magnitude_projection(a, N)
        b = np.convolve(a, a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] length={n} proj iter={it} score={cur_score:.6f}")
        # small random perturbation every few hundred steps
        if it % 100 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    # ---------- Phase 2: sub‑gradient polishing ----------
    a, sub_score = _subgradient_descent(
        a,
        n_iters=max_subgrad_iters,
        time_deadline=time_deadline,
        eta0=0.5,
    )
    if sub_score < best_score:
        best_score = sub_score
        best_seq = a.copy()

    return best_seq, best_score
