"""
Search for a step‑function (non‑negative heights) that minimises
evaluate_sequence.  The algorithm is a stochastic projected gradient descent
on a smooth approximation of the max‑convolution objective, combined with
occasional random pairwise swaps to escape local minima.

Key ideas
---------
* The objective 2·n·max_k (a*a)_k / (sum a)^2 is scale‑invariant.
  We therefore work on the probability simplex (a_i ≥ 0, sum a = 1).
* Instead of the non‑smooth max we minimise the soft‑max
  L(a)= (1/α)·log Σ_i exp(α·conv_i)  (conv = a*a).
  Its gradient is  ∇L = 2·corr(a, w)  where
  w_i = exp(α·conv_i)/Σ_j exp(α·conv_j)  (a probability vector).
* A decreasing step‑size η and an increasing α drive the iterates towards a
  flatter convolution while still respecting the simplex via Euclidean
  projection.
* Random pairwise mass‑swap moves (keeping the sum exactly 1) are mixed in
  with the gradient step and accepted with a Metropolis rule whose
  temperature cools linearly.
* The best sequence seen so far (according to the true objective) is saved.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:   # all non‑positive -> uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Random pairwise mass‑swap move (keeps the simplex constraint exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Choose i≠j and transfer a small amount of mass (Gaussian‑like)."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma  # uniform in [-σ/2, σ/2]

    if delta > 0:
        delta = min(delta, seq[j])      # take from j
    else:
        delta = -min(-delta, seq[i])    # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that strives to minimise
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 2.0)   # tiny safety margin

    # --------------------------------------------------------------
    # Initialise a probability vector a (sum = 1)
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)
        if base.size == 0:
            raise ValueError
        base = base / max(base.sum(), 1e-12)
    except Exception:
        # Fallback: a modest‑size uniform start
        base_len = 7500
        base = np.full(base_len, 1.0 / base_len, dtype=float)

    a = _project_simplex(base)          # current iterate
    n = a.shape[0]

    # --------------------------------------------------------------
    # Track the best sequence seen so far (according to the true objective)
    # --------------------------------------------------------------
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_seq = a.copy()
    current_score = best_score

    # --------------------------------------------------------------
    # Hyper‑parameters (all schedules are time‑dependent)
    # --------------------------------------------------------------
    # temperature for Metropolis acceptance
    T_start, T_end = 1.0, 1e-5
    # smoothing parameter α (soft‑max temperature)
    alpha_start, alpha_end = 0.5, 200.0
    # gradient step size η
    eta_start, eta_end = 0.5, 0.001
    # probability of performing a random swap move
    prob_swap = 0.35

    # bookkeeping for simple stagnation detection
    last_improve_iter = 0
    iter_no = 0

    while time.time() < deadline:
        iter_no += 1

        # ----------------------------------------------------------
        # Time‑dependent schedule values
        # ----------------------------------------------------------
        elapsed = (time.time() - start_time) / (deadline - start_time)
        temperature = T_start * (1.0 - elapsed) + T_end * elapsed
        alpha = alpha_start * (alpha_end / alpha_start) ** elapsed
        eta = eta_start * (eta_end / eta_start) ** elapsed

        # ----------------------------------------------------------
        # Propose a candidate (gradient step or random swap)
        # ----------------------------------------------------------
        if np.random.rand() < prob_swap:
            # ---- random pairwise move --------------------------------
            sigma = 0.2 / np.sqrt(iter_no + 1.0)
            cand = _random_pair_move(a, sigma)
        else:
            # ---- smoothed gradient descent ---------------------------
            conv = np.convolve(a, a)                     # full convolution
            # soft‑max weights w_i ∝ exp(α·conv_i)
            scaled = alpha * conv
            # numerical stability: shift by max
            scaled -= np.max(scaled)
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient g = 2·corr(a, w) (only the n entries that correspond to a)
            corr = np.correlate(a, w, mode='full')
            start_idx = w.size - 1
            g = 2.0 * corr[start_idx:start_idx + n]

            # projected gradient step
            cand = a - eta * g
            cand = _project_simplex(cand)

        # ----------------------------------------------------------
        # Evaluate the candidate (exact objective)
        # ----------------------------------------------------------
        conv_cand = np.convolve(cand, cand)
        cand_score = 2.0 * n * conv_cand.max() / (cand.sum() ** 2)

        # ----------------------------------------------------------
        # Keep the best ever seen
        # ----------------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            last_improve_iter = iter_no
            print(f"[Iter {iter_no:6d}] New best score: {best_score:.6f}")

        # ----------------------------------------------------------
        # Metropolis acceptance rule (allows occasional uphill moves)
        # ----------------------------------------------------------
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = cand
            current_score = cand_score

        # ----------------------------------------------------------
        # Simple restart if we have been stuck for a while
        # ----------------------------------------------------------
        if iter_no - last_improve_iter > 20000:
            # Blend a tiny random perturbation into the current best
            pert = np.random.rand(n)
            pert /= pert.sum()
            a = _project_simplex(0.97 * best_seq + 0.03 * pert)
            current_score = 2.0 * n * np.convolve(a, a).max() / (a.sum() ** 2)
            last_improve_iter = iter_no

    # --------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # --------------------------------------------------------------
    return best_seq.tolist()
