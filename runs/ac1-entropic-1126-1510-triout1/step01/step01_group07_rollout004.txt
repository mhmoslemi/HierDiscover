"""
Propose a low‑evaluation step‑function by searching a one‑parameter family
a_i = (i+1)^p · exp(-γ·i/(n‑1)).
The exponent p (negative) controls how fast the heights decay; the
exponential factor γ fine‑tunes the tail.  For a given length n the
objective 2 n·max(b)/sum(a)^2 is (asymptotically) 2·a²·B(a,a) with
a = p+1 and B the beta‑function, which is minimised for a≈0.3 (p≈‑0.7).
We therefore perform a cheap grid‑search over:
    • length n  (a few thousand, fast to convolve)
    • exponent p (≈‑1 … 0)
    • taper γ (≈0 … 2)
The best sequence found is returned as a plain Python list.
All timing checks respect the overall 1000 s budget.
"""

import numpy as np
import time
import math

def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of non‑negative floats) found before
    the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer framework
    deadline = start_time + budget_s - 5.0

    best_score = float("inf")
    best_seq = None
    best_cfg = None   # (n, p, gamma)

    # ------------------------------------------------------------------
    # 1. Choose a set of candidate lengths.
    #    Larger n gives a better approximation of the asymptotic optimum
    #    but each evaluation grows ~n² (np.convolve is O(n²)).
    # ------------------------------------------------------------------
    lengths = [1500, 3000, 4000, 6000, 8000]
    # ------------------------------------------------------------------
    # 2. Coarse search over exponent p (negative) for each n.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        idx = np.arange(1, n + 1, dtype=np.float64)   # i = 1 … n

        # coarse grid for the exponent
        for p in np.linspace(-0.95, -0.15, 41):   # step ≈0.02
            seq = idx ** p                         # pure power‑law
            # evaluation (evaluate_sequence is supplied in the environment)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_cfg = (n, p, 0.0)            # gamma = 0 for now

            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # 3. Refine exponent around the best found so far (still no taper).
    # ------------------------------------------------------------------
    if best_cfg is not None and time.time() < deadline:
        n_best, p_best, _ = best_cfg
        idx = np.arange(1, n_best + 1, dtype=np.float64)

        lo = max(-0.95, p_best - 0.10)
        hi = min(-0.15, p_best + 0.10)
        for p in np.linspace(lo, hi, 61):       # finer step ≈0.003
            seq = idx ** p
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_cfg = (n_best, p, 0.0)

            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # 4. Add an exponential taper γ and sweep it.
    # ------------------------------------------------------------------
    if best_cfg is not None and time.time() < deadline:
        n_best, p_best, _ = best_cfg
        idx = np.arange(1, n_best + 1, dtype=np.float64)
        base = idx ** p_best

        # coarse sweep for γ
        for gamma in np.linspace(0.0, 2.0, 41):
            seq = base * np.exp(-gamma * (idx - 1) / (n_best - 1))
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_cfg = (n_best, p_best, gamma)

            if time.time() > deadline:
                break

        # fine sweep around the best γ (if time permits)
        if time.time() < deadline:
            best_gamma = best_cfg[2]
            lo = max(0.0, best_gamma - 0.25)
            hi = best_gamma + 0.25
            for gamma in np.linspace(lo, hi, 81):   # step ≈0.006
                seq = base * np.exp(-gamma * (idx - 1) / (n_best - 1))
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_cfg = (n_best, p_best, gamma)

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # 5. (Optional) try a second length if we still have plenty of time.
    # ------------------------------------------------------------------
    # The above loops already explore several lengths; if time permits we
    # repeat the whole process for the next longest n in the list.
    # This guard keeps us safe under the 1000 s limit.
    while time.time() < deadline:
        # pick the next longer n that we did not try yet
        remaining = [l for l in lengths if l > best_cfg[0]]
        if not remaining:
            break
        n = max(remaining)               # try the largest remaining size
        idx = np.arange(1, n + 1, dtype=np.float64)

        # reuse the best exponent found so far (close to optimal)
        p_center = best_cfg[1]

        # a tiny local search around that exponent for the new n
        lo = max(-0.95, p_center - 0.05)
        hi = min(-0.15, p_center + 0.05)
        for p in np.linspace(lo, hi, 21):
            base = idx ** p
            # no taper first
            score = evaluate_sequence(base.tolist())
            if score < best_score:
                best_score = score
                best_seq = base.copy()
                best_cfg = (n, p, 0.0)

            # try a few γ values
            for gamma in (0.0, 0.1, 0.5, 1.0):
                seq = base * np.exp(-gamma * (idx - 1) / (n - 1))
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_cfg = (n, p, gamma)

            if time.time() > deadline:
                break
        # after one extra length we break – we already spent most of the budget
        break

    # ------------------------------------------------------------------
    # 6. Report the final best configuration.
    # ------------------------------------------------------------------
    if best_cfg is not None:
        n, p, gamma = best_cfg
        print(
            f"Best configuration -> n={n}, p={p:.5f}, gamma={gamma:.5f}, "
            f"objective={best_score:.6f}"
        )
    else:
        print("No valid sequence found within the time budget.")
        return []   # evaluate_sequence would return inf for an empty list

    # convert to plain list as required by the outer harness
    return best_seq.tolist()
