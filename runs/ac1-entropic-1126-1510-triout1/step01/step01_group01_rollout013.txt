"""
Simulated‑annealing search for a step‑function with a low
max‑self‑convolution (the value returned by `evaluate_sequence`).

The algorithm works on a fixed length `n` (normally the length of the
known best construction `height_sequence_1`).  It keeps the ℓ¹‑norm
fixed to `sqrt(2 n)`.  The main loop performs many tiny mass‑transfer
moves `a[i] → a[j]`.  The convolution `b = a*a` is updated **incrementally**
in O(n) using the exact formula

    b[i]   -= 2·δ·a[p]   for p = 0 … n‑1  (source index i)
    b[j]   += 2·δ·a[p]   for p = 0 … n‑1  (target index j)
    b[2·i] += δ²
    b[2·j] += δ²
    b[i+j] -= 2·δ²

so the expensive O(n log n) FFT convolutions are avoided.

Two types of moves are used:

* **Targeted moves** – the current maximal convolution entry `k*` is
  identified; a source index participating most strongly in that entry
  (i.e. maximising `a[i]·a[k*‑i]`) is chosen and a small amount of mass
  is moved away from it.
* **Random moves** – a uniformly random source/target pair.

A simple Metropolis criterion with a linearly cooling temperature
accepts occasional worsening moves, allowing the search to escape
local minima.  After the annealing phase a short projected‑gradient
polish (the same sub‑gradient that the reference code uses) is applied
to squeeze out a little extra improvement.

The whole routine respects the 1000 s time budget and returns the best
sequence found as a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _gradient_of_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """Sub‑gradient of the convolution entry `a*a` at index `max_idx`."""
    n = a.shape[0]
    i = np.arange(n)
    j = max_idx - i
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    diag = mask & (i == j)
    grad[diag] -= a[i[diag]]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed.
    budget_s : float
        Approximate time budget (seconds).

    Returns
    -------
    list[float]
        The best sequence discovered (sum is √(2·n)).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # --------------------------------------------------------------
    # Length and target ℓ¹‑norm
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a few diverse starting points
    starts = []

    if have_ref:
        starts.append(np.asarray(height_sequence_1, dtype=float))  # type: ignore[name-defined]

    # random simplex point
    rnd = np.random.rand(n)
    starts.append(_project_to_simplex(rnd, target_sum))

    # block‑wise random start (2‑6 blocks)
    nb = np.random.randint(2, 7)
    cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
    blocks = np.split(np.arange(n), cuts)
    vals = np.random.rand(nb)
    block_vec = np.zeros(n, dtype=float)
    for blk, v in zip(blocks, vals):
        block_vec[blk] = v
    starts.append(_project_to_simplex(block_vec, target_sum))

    # limit the number of starts (time‑budget management)
    starts = starts[:3]

    # --------------------------------------------------------------
    # Time allocation per start (rough, may shrink on the fly)
    time_per_start = (deadline - time.time()) / max(len(starts), 1)

    best_global = None
    best_score = float("inf")

    for start_id, a0 in enumerate(starts):
        if time.time() > deadline:
            break
        t_start = time.time()

        # ensure feasibility
        a = _project_to_simplex(a0, target_sum)

        # initial convolution and score
        conv = np.convolve(a, a)
        cur_max = conv.max()
        best_local_a = a.copy()
        best_local_score = cur_max

        # SA parameters
        avg = target_sum / n                     # typical entry size
        step0 = avg * 0.8                        # maximal transfer per move
        T0 = cur_max * 0.02                      # initial temperature
        max_iters = 10 ** 7                      # safety cap
        it = 0
        no_improve = 0

        while time.time() - t_start < time_per_start and it < max_iters:
            # --------------------------------------------------
            # cooling schedules (linear)
            progress = it / max_iters
            step = step0 * (1.0 - progress) + 1e-9
            T = T0 * (1.0 - progress) + 1e-12

            # --------------------------------------------------
            # Choose move type
            if np.random.rand() < 0.7:                     # targeted move
                k_star = int(np.argmax(conv))              # index of current max
                i = np.arange(n)
                j = k_star - i
                mask = (j >= 0) & (j < n)
                if np.any(mask):
                    prod = a[i[mask]] * a[j[mask]]
                    src = i[mask][np.argmax(prod)]
                else:                                       # fallback to random
                    src = np.random.randint(0, n)
                # choose a distinct target uniformly
                tgt = np.random.randint(0, n - 1)
                if tgt >= src:
                    tgt += 1
            else:                                           # pure random move
                src = np.random.randint(0, n)
                if a[src] < 1e-12:                          # avoid empty source
                    it += 1
                    continue
                tgt = np.random.randint(0, n - 1)
                if tgt >= src:
                    tgt += 1

            # --------------------------------------------------
            # maximal transferable mass from the source
            max_move = min(step, a[src])
            if max_move <= 0:
                it += 1
                continue
            delta = max_move * np.random.rand()            # 0 ≤ δ ≤ max_move

            # ------------------- incremental convolution update ----------
            # use the *pre‑move* array a for the linear part
            conv[src: src + n] -= 2.0 * delta * a
            conv[tgt: tgt + n] += 2.0 * delta * a
            conv[2 * src] += delta * delta
            conv[2 * tgt] += delta * delta
            conv[src + tgt] -= 2.0 * delta * delta

            new_max = conv.max()

            # ------------------- acceptance test -------------------------
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                if np.random.rand() < math.exp(-(new_max - cur_max) / T):
                    accept = True

            if accept:
                # commit the move
                a[src] -= delta
                a[tgt] += delta
                cur_max = new_max
                no_improve = 0
                if new_max < best_local_score:
                    best_local_score = new_max
                    best_local_a = a.copy()
                    print(f"[start{start_id} it{it:07d}] new best = {best_local_score:.6f}")
            else:
                # revert convolution (a stays unchanged)
                conv[src: src + n] += 2.0 * delta * a
                conv[tgt: tgt + n] -= 2.0 * delta * a
                conv[2 * src] -= delta * delta
                conv[2 * tgt] -= delta * delta
                conv[src + tgt] += 2.0 * delta * delta
                no_improve += 1

            # occasional large random perturbation to escape plateaus
            if (it % 5000 == 0 and it > 0) or no_improve > 8000:
                noise = np.random.randn(n) * avg * 2e-3
                cand = a + noise
                cand = _project_to_simplex(cand, target_sum)
                conv = np.convolve(cand, cand)
                cur_max = conv.max()
                a = cand
                no_improve = 0
                print(f"[start{start_id} it{it:07d}] perturbation")

            it += 1

        # --------------------------------------------------------------
        # Short projected‑gradient polish (few hundred iterations)
        a_polish = best_local_a.copy()
        conv_polish = np.convolve(a_polish, a_polish)
        cur_max_polish = conv_polish.max()
        best_polish_score = cur_max_polish
        best_polish_a = a_polish.copy()

        gd_iters = 400
        eta0 = avg * 0.3
        decay = 5e-4

        for t in range(gd_iters):
            max_idx = int(np.argmax(conv_polish))
            grad = _gradient_of_max(a_polish, max_idx)
            eta = eta0 / (1.0 + decay * t)
            a_new = a_polish - eta * grad
            a_new = _project_to_simplex(a_new, target_sum)
            conv_new = np.convolve(a_new, a_new)
            cur_score = conv_new.max()
            if cur_score < best_polish_score:
                best_polish_score = cur_score
                best_polish_a = a_new.copy()
                print(f"[gd start{start_id} it{t:03d}] improved = {best_polish_score:.6f}")
            a_polish = a_new
            conv_polish = conv_new

        # --------------------------------------------------------------
        # Update global best
        if best_polish_score < best_score:
            best_score = best_polish_score
            best_global = best_polish_a.copy()
            print(f"[global] improved to {best_score:.6f} (start {start_id})")

    # ------------------------------------------------------------------
    if best_global is None:
        # fall back to a trivial feasible vector
        best_global = np.zeros(n, dtype=float)
        best_global[0] = target_sum

    # final safety projection (tiny numerical drift may appear)
    best_global = np.maximum(best_global, 0.0)
    s = best_global.sum()
    if s > 0:
        best_global *= target_sum / s
    else:
        best_global[0] = target_sum

    print(f"[final] best score = {best_score:.6f}")
    return best_global.tolist()
