import cvxpy as cp

def improve_f(f, t, max_iters, deadline):
    n = len(f)
    M = np.max(np.convolve(f, f))
    b = cp.Variable(n, nonneg=True)
    # constraints: A @ b <= M
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.GLPK, glpk={'msg_lev': 'GLP_MSG_OFF'})
    b_opt = b.value
    if b_opt is None:
        return f
    sum_b = np.sum(b_opt)
    if sum_b <= 0:
        return f
    g = np.sqrt(2*n) * b_opt / sum_b
    new_f = (1-t)*f + t*g
    return new_f
```

def project_to_simplex(v):
    # v is 1D numpy array
    # algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import cvxpy as cp

def solve_lp_max_sum(f, M):
    n = len(f)
    rows = 2*n-1
    # Build matrix A
    k = np.arange(rows)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((rows, n))
    A[mask] = f[idx[mask]]
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.GLPK, warm_start=True)
    if b.value is None:
        return None
    return np.array(b.value)
```

"""
Propose a high‑quality step‑function sequence.

The algorithm starts from a few simple families (arcsine, uniform, Dirichlet)
and improves them by an iterative linear‑programming scheme described in the
statement.  For a sequence f (length n) with ‖f‖₁ = √(2n) we compute
M = max(f * f).  Solving the LP

    max   ∑ b_i
    s.t.  b_i ≥ 0
          (f * b)_k ≤ M   for all k

produces a vector b that has a larger ℓ¹‑norm while still respecting the
same convolution bound.  After rescaling b to the same ℓ¹‑norm as f we
mix f←(1‑t)f + t·b (t≈0.2).  This mixing never increases the bound M, and
repeating the step drives the maximal convolution down.  A few LP steps are
performed for each candidate length, the best sequence (according to the
provided `evaluate_sequence`) is kept and finally returned.

The implementation respects the 1000 s time budget, works on the CPU only,
and uses only top‑level functions (no closures or lambdas).  It falls back
to the classic arcsine construction if cvxpy is unavailable.
"""

import time
import math
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)

def arcsine_sequence(n: int) -> np.ndarray:
    """Length‑n coefficients proportional to C_i = binom(2i,i)/4**i.
    Returned array is *not* normalised (sum = 1)."""
    cur = 1.0
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = cur
        # recurrence C_{i+1} = C_i * (2*i+1)/(2*(i+1))
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    return seq


def uniform_sequence(n: int) -> np.ndarray:
    """Uniform non‑negative sequence of length n (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=float)


def dirichlet_sequence(n: int, alpha: float = 0.5, rng: np.random.Generator = None) -> np.ndarray:
    """Random Dirichlet sample (sum = 1)."""
    if rng is None:
        rng = np.random.default_rng()
    seq = rng.dirichlet([alpha] * n)
    return seq.astype(float)


def scale_to_norm(seq: np.ndarray, target_norm: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ¹‑norm equals target_norm."""
    current = seq.sum()
    if current == 0:
        return np.full_like(seq, target_norm / len(seq))
    return seq * (target_norm / current)


def max_convolution(seq: np.ndarray) -> float:
    """Return max value of the discrete self‑convolution of seq."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def evaluate(seq: np.ndarray) -> float:
    """Wrapper around the supplied evaluate_sequence (expects a list)."""
    return evaluate_sequence(seq.tolist())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x≥0, Σx=1}. """
    if v.ndim != 1:
        raise ValueError("simplex projection only defined for 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be projected to zero -> return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1) × n matrix A such that A @ b = f * b."""
    n = f.size
    rows = 2 * n - 1
    # broadcasting to build index matrix k‑j
    k = np.arange(rows)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j                # shape (rows, n)
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((rows, n), dtype=float)
    A[mask] = f[idx[mask]]
    return A


def lp_step(f: np.ndarray, M: float, t: float = 0.2) -> np.ndarray:
    """
    Perform a single linear‑programming improvement step.
    f must satisfy ‖f‖₁ = sqrt(2·len(f)).  Returns the updated sequence.
    If cvxpy is unavailable or the LP fails, the original f is returned.
    """
    try:
        import cvxpy as cp
    except Exception:               # pragma: no cover – cvxpy not installed
        return f

    n = f.size
    A = build_conv_matrix(f)

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # Use a fast open‑source solver; ECOS works for pure LPs.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:               # pragma: no cover – solver failure
        return f

    if b.value is None:             # LP infeasible or numerical failure
        return f

    b_opt = np.asarray(b.value, dtype=float)
    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        return f

    # Rescale b to have the same ℓ¹‑norm as f (which is sqrt(2·n))
    target_norm = math.sqrt(2.0 * n)
    g = b_opt * (target_norm / sum_b)

    # Convex mixing step – preserves the ℓ¹‑norm
    new_f = (1.0 - t) * f + t * g
    return new_f


def improve_sequence(seq: np.ndarray, max_iter: int, deadline: float) -> np.ndarray:
    """
    Repeatedly apply the LP improvement step to the supplied sequence.
    Stops early if the time budget expires or if no progress is made.
    """
    n = seq.size
    # Normalise to the required ℓ¹‑norm once (the LP routine expects it)
    seq = scale_to_norm(seq, math.sqrt(2.0 * n))

    best_seq = seq.copy()
    best_score = evaluate(best_seq)

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        M = max_convolution(seq)
        new_seq = lp_step(seq, M, t=0.2)

        # If the LP step was a no‑op we stop.
        if np.allclose(new_seq, seq, atol=1e-12):
            break

        score = evaluate(new_seq)
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[LP] n={n:<4d} iter={it+1:<3d}  score={score:.6f}")

        # Stop if the bound stopped decreasing noticeably
        if score >= best_score * (1.0 + 1e-6):
            # no clear improvement – keep the current sequence but continue a few
            # more iterations just in case.
            pass

        seq = new_seq

    return best_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function giving a low evaluation score.
    The search consists of:
      1. Trying a handful of sequence lengths (300‑1800).
      2. For each length, initialising several starter families.
      3. Running a few LP‑based improvement cycles.
      4. Keeping the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.05   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Sweep over a modest set of lengths (fast enough for the time budget)
    lengths = list(range(300, 1901, 200))   # 300, 500, …, 1800
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----------------------------------------------------------------
        # 2) Initial families for this length
        starters = [
            arcsine_sequence(n),                 # arcsine coefficients (unnorm.)
            uniform_sequence(n),                 # flat distribution
            dirichlet_sequence(n, alpha=0.5, rng=rng)  # random Dirichlet sample
        ]

        for idx, init_seq in enumerate(starters):
            if time.time() >= deadline:
                break

            # normalise the ℓ¹‑norm to sqrt(2·n) – required by the LP step
            init_seq = scale_to_norm(init_seq, math.sqrt(2.0 * n))

            # Evaluate the raw candidate (no LP yet) for a quick bound
            try:
                init_score = evaluate(init_seq)
            except Exception:   # pragma: no cover – safety
                continue

            if init_score < best_score:
                best_score = init_score
                best_seq = init_seq.copy()
                print(f"[init] n={n:<4d} type={idx}  score={init_score:.6f}")

            # ----------------------------------------------------------------
            # 3) Apply a few LP refinement iterations (typically 4‑6)
            refined = improve_sequence(init_seq, max_iter=6, deadline=deadline)

            # Evaluate the refined sequence
            try:
                refined_score = evaluate(refined)
            except Exception:   # pragma: no cover
                continue

            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined.copy()
                print(f"[best] n={n:<4d} type={idx}  new best score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Tiny random local search around the current best (if time permits)
    if best_seq is not None:
        n = best_seq.size
        while time.time() < deadline:
            # pick two distinct indices and transfer a small random amount
            i, j = rng.choice(n, size=2, replace=False)
            delta = (rng.random() - 0.5) * 2e-3   # ±0.001
            cand = best_seq.copy()
            cand[i] = max(0.0, cand[i] + delta)
            cand[j] = max(0.0, cand[j] - delta)
            # re‑normalise to keep the ℓ¹‑norm unchanged (no scaling effect)
            s = cand.sum()
            if s == 0:
                continue
            cand = cand / s * best_seq.sum()      # keep same norm as best_seq
            try:
                cand_score = evaluate(cand)
            except Exception:
                continue
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand
                print(f"[refine] tiny tweak improved score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list (the submission format expects a list)
    if best_seq is None:
        # As a safety net return a very simple feasible sequence
        fallback = uniform_sequence(500)
        best_seq = scale_to_norm(fallback, math.sqrt(2.0 * 500))

    print(f"Final best length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq.tolist()
