def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:n]
```

"""
Propose a step‑function (list of non‑negative heights) with a low
`evaluate_sequence` score.

Algorithm
---------
* Initialise a distribution of length *n* (arcsine, uniform, random or the
  previously found sequence `height_sequence_1`).
* Repeatedly minimise the maximal value of the self‑convolution
  `b = a * a` using a projected sub‑gradient method.
  The sub‑gradient for the active constraint `max(b)` is
  `g_i = 2·a_{k−i}` where `k` is the index of the maximal convolution entry.
* Compute the convolution with an FFT‑based routine (O(n log n)), which
  makes it possible to perform many thousands of iterations within the
  1000 s budget.
* Use a diminishing step size `η_t = η₀ / sqrt(t)` and a simple momentum term.
* After every improvement project back onto the probability simplex
  (non‑negative entries summing to 1) and keep the best sequence seen.
* Try a few different lengths (derived from the historic construction)
  and initialisations; return the overall best sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=s}
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generate a discrete arcsine distribution of length n (probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Initialise a sequence of length n according to the requested scheme
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        # use the globally supplied sequence if it exists

            a = np.asarray(height_sequence_1, dtype=np.float64)

        except Exception:

        else:
            a = a / a.sum()
    else:  # fallback
        a = _arcsine_distribution(n)
    # safety: ensure exact simplex feasibility
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
# Fast convolution of a real vector with itself using FFT
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a using FFT (real input). Result length = 2·len(a)‑1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    # element‑wise multiplication (complex numbers)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]
