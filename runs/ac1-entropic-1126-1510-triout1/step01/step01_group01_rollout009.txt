"""
propose_candidate
-----------------
A simulated‑annealing / Adam style optimiser that minimises the
maximum entry of the self‑convolution of a non‑negative step function.
The objective `max_k (a*a)[k]` is approximated by the smooth
log‑sum‑exp (a “soft‑max”) with a temperature that is slowly annealed.
The gradient of the smooth objective can be written as a single
convolution, so each iteration costs only O(n) elementary operations
(up to the FFT‑free `np.convolve` used by NumPy).  After every gradient
step the vector is projected back onto the simplex
{ a ≥ 0, Σa = √(2 n) } to keep the ℓ¹‑norm fixed – the evaluation
function is homogeneous, so fixing the sum does not affect the
objective value.  Multiple random restarts and occasional tiny
perturbations help to escape shallow local minima.  The best sequence
found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex of total mass z."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)                         # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):                  # all entries become zero
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1        # +1 for zero‑based indexing
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max) of the self‑convolution.
def _softmax_grad(a: np.ndarray, tau: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where ``grad`` is the gradient of the
    smooth approximation of ``max_k (a*a)[k]`` with temperature ``tau``.
    Also returns the exact maximum of the convolution (used for the
    true score).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n‑1
    max_conv = conv.max()

    # soft‑max weights: w_k ∝ exp(conv_k / tau)
    # to avoid overflow, shift by the maximum value.
    if tau <= 0:
        # degenerate case – behave like exact max (sub‑gradient of max)
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        shifted = (conv - max_conv) / tau
        w = np.exp(shifted, dtype=np.float64)
        w /= w.sum()

    # gradient = 2 * ∑_k w_k * a_{k-i}
    # This is a convolution of a[::-1] with w, then extraction of a window.
    rev_a = a[::-1]
    tmp = np.convolve(rev_a, w)                 # length n + (2n‑1) - 1 = 3n‑2
    grad = 2.0 * tmp[(n - 1):(n - 1) + n]       # slice of length n
    return grad, max_conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises the evaluation
    score.  The function runs for at most ``budget_s`` seconds and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a small margin

    # --------------------------------------------------------------
    # Choose a problem size.
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)                # the normalising ℓ¹‑norm

    # --------------------------------------------------------------
    # Initial candidates for restarts.
    def _make_initial():
        """Return a feasible starting vector."""
        if have_ref and np.random.rand() < 0.5:
            a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        else:
            # Dirichlet (random) start
            a0 = np.random.rand(n)
        # project onto the simplex
        a0 = _project_to_simplex(a0, target_sum)
        return a0

    # --------------------------------------------------------------
    # Adam hyper‑parameters (shared across restarts)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.5 * target_sum / n          # ≈ 0.5·average entry

    # Temperature schedule – start relatively hot, anneal to a low value.
    tau_start = 0.5
    tau_end = 0.01
    tau_decay = (tau_end / tau_start) ** (1.0 / 200000)   # decay over ~200k iters

    # Book‑keeping of the global best.
    best_a = None
    best_score = float('inf')
    total_iters = 0

    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---- start a new restart -------------------------------------------------
        a = _make_initial()
        m = np.zeros_like(a)      # first moment
        v = np.zeros_like(a)      # second moment
        tau = tau_start
        it_since_best = 0
        restart_iters = 0

        # small inner loop – break earlier if the global deadline is close
        while time.time() < deadline and restart_iters < 50000:
            grad, cur_max = _softmax_grad(a, tau)

            # Adam update
            restart_iters += 1
            total_iters += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** restart_iters)
            v_hat = v / (1.0 - beta2 ** restart_iters)
            step = base_lr * (m_hat / (np.sqrt(v_hat) + eps))

            a_new = a - step
            a_new = _project_to_simplex(a_new, target_sum)

            # true score (exact max convolution)
            sum_a = a_new.sum()

                # safety – re‑project and continue
                a_new = _project_to_simplex(a_new, target_sum)
                sum_a = a_new.sum()
            # compute actual maximum convolution entry
            conv_full = np.convolve(a_new, a_new)
            max_conv = conv_full.max()
            cur_score = 2.0 * n * max_conv / (sum_a * sum_a)

            # keep the best globally

                best_score = cur_score
                best_a = a_new.copy()
                it_since_best = 0
                print(f"[{total_iters:08d}] new global best = {best_score:.6f}")
            else:
                it_since_best += 1

            a = a_new

            # anneal temperature
            tau *= tau_decay
            if tau < tau_end:
                tau = tau_end

            # occasional random jitter to jump out of plateaus
            if (restart_iters % 3000 == 0 and restart_iters > 0) or it_since_best > 4000:
                jitter = np.random.randn(n) * target_sum * 5e-4
                a = _project_to_simplex(a + jitter, target_sum)
                m[:] = 0.0
                v[:] = 0.0
                it_since_best = 0
                print(f"[{total_iters:08d}] jitter injected")

        # after the inner loop we try a quick “block” perturbation
        if best_a is not None:
            # tiny block‑pattern trial (cheap, can sometimes improve)
            nb = np.random.randint(2, 7)
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)
            cand = np.zeros(n, dtype=float)
            for blk, vblk in zip(blocks, vals):
                cand[blk] = vblk
            cand = _project_to_simplex(cand, target_sum)
            conv_c = np.convolve(cand, cand)
            score_c = 2.0 * n * conv_c.max() / (cand.sum() ** 2)
            if score_c < best_score:
                best_score = score_c
                best_a = cand.copy()
                print(f"[{total_iters:08d}] block‑candidate improved = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list).
    if best_a is None:
        # fallback – uniform distribution (should never happen)
        best_a = np.full(n, target_sum / n, dtype=float)

    print(f"[final] best score = {best_score:.6f} after {total_iters} iterations")
    return best_a.tolist()
