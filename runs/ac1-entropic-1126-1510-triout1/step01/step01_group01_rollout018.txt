"""
Propose candidate sequence using a hybrid of parametric shape search and
soft‑max‑guided projected gradient descent.

The algorithm works in two phases:

1. **Exploratory shape search** – a collection of simple families
   (inverse‑sqrt, power‑law, exponential, cosine, mixed, random blocks,
   uniform) are sampled.  Each candidate is projected onto the simplex
   {a≥0, Σa = √(2 n)} and its evaluation score is computed
   (2·n·max(a*a) / Σa²).  The best candidate from this stage is kept.

2. **Refinement** – starting from the best candidate, we minimise the
   smoothed objective  \sum_k w_k·(a*a)_k  where the weights w_k are a soft‑max
   of the convolution entries.  The temperature of the soft‑max is
   gradually decreased, turning the smooth objective into the true
   max‑convolution.  Gradients are obtained analytically, implemented with
   FFT‑based convolutions for speed, and a Nesterov‑type momentum step is
   applied followed by Euclidean projection onto the simplex.  Occasional
   random perturbations and a final round of random block candidates help
   to escape plateaus.

All operations respect the security checks performed by `evaluate_sequence`
(the optimiser works with non‑negative numbers and the sum is fixed, so
scaling does not affect the final score).

The function returns the best sequence found before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return a*a (full discrete convolution)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray,
                       size_fft: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k  w.r.t. a (weights w are assumed fixed).
    Implementation uses FFT for speed.
    """
    n = a.shape[0]
    # reverse a for the appropriate index shift
    a_rev = a[::-1]
    # convolution w * a_rev
    conv = _conv_fft(w, a_rev, size_fft)          # length = 2n-1 + n-1 = 3n-2
    start = n - 1                                 # slice that contains Σ_k w_k a_{k-i}
    grad = 2.0 * conv[start:start + n]           # 2* Σ_k w_k a_{k-i}
    # subtract the diagonal over‑counting (when i = k-i)
    diag_w = w[0:2 * n - 1:2]                     # w_{2i}, length n
    grad -= a * diag_w
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # length of the problem (use the reference length if available)

    # ------------------------------------------------------------------
