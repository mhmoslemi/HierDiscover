"""
Simulated‑annealing search for a low‑value step‑function.
--------------------------------------------------------------------
The objective is

    score(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

which is invariant under scaling of the vector.  We keep the L1‑norm
exactly 1 (by moving mass between two entries) and try to reduce the
maximum entry of the convolution.

The algorithm starts from a power‑law vector (the best simple
construction known) and then repeatedly:
  * evaluates the current convolution,
  * finds the index k where the convolution is maximal,
  * builds a weight vector `w_i = a_i·a_{k-i}` (zero outside the
    overlap) – this is proportional to the gradient of the max term,
  * chooses a source index i with probability ∝ w_i (large contribution)
    and a target index j with probability ∝ 1/(w_j+ε) (small contribution),
  * transfers a random fraction of a_i to a_j,
  * accepts the move with a Metropolis rule (temperature slowly cools).

Occasional purely random moves are mixed in to increase exploration.
All arithmetic is done with NumPy; the expensive convolution is performed
with `np.convolve`.  The routine stops after the allotted budget (default
1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper utilities (all top‑level, no closures)
# --------------------------------------------------------------------------- #

def compute_score(vec: np.ndarray) -> float:
    """
    Direct implementation of the evaluation metric.
    The function assumes a non‑negative vector; the surrounding framework
    will clamp values if needed.
    """
    n = len(vec)
    # full convolution (cost O(n²) but NumPy C code is fast)
    b = np.convolve(vec, vec)
    max_b = np.max(b)
    s = np.sum(vec)
    # the original evaluate_sequence returns np.inf for tiny sums;
    # we keep the same guard.
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    Decreasing power‑law a_i ∝ (i+1)^{-p}, normalised to L1‑norm 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def initialize_sequence(n: int) -> np.ndarray:
    """
    Initialise a candidate vector.  We start from the known good
    power‑law (p≈0.5) but add a tiny random perturbation to break symmetry.
    """
    a = power_law_sequence(n, 0.5)
    # a very small jitter to avoid exact symmetry which can stall the annealer
    jitter = np.random.rand(n) * 1e-6
    a += jitter
    a /= a.sum()
    return a


def weighted_choice(weights: np.ndarray) -> int:
    """
    Return an index sampled proportionally to the (non‑negative) weight
    array `weights`.  If the total weight is zero a uniform index is chosen.
    """
    total = weights.sum()
    if total == 0.0:
        return np.random.randint(len(weights))
    probs = weights / total
    # cumulative method is faster than np.random.choice for a single draw
    cum = np.cumsum(probs)
    r = np.random.rand()
    return int(np.searchsorted(cum, r, side="right"))

    deadline = start_time + budget_s - 5.0   # tiny safety margin

    T0 = 0.12       # initial temperature (controls acceptance of uphill moves)

    cur_a = a.copy()
    cur_conv = np.convolve(cur_a, cur_a)   # cache for the first step

    # ------------------------------------------------------------------- #

    # ------------------------------------------------------------------- #
    while time.time() < deadline:

        elapsed = time.time() - start_time

        T = max(T_end, T0 * (1.0 - frac))
