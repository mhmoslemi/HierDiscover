"""
Propose a candidate step‑function by minimizing the maximal entry of its
self‑convolution.  The objective  f(a)=max_k (a*a)[k]  is approximated with a
smooth “log‑sum‑exp” surrogate.  Its gradient can be written in closed form and
computed by a single correlation.  The algorithm performs a projected (onto the
simplex Σa=S) gradient descent with a slowly increasing smoothing parameter and
a simple momentum scheme.  Several restarts, random perturbations and a final
“block” search are used to escape local minima.  The code respects the time
budget (≈1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx = z} (Euclidean projection)."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all will be zero – put the whole mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: self‑convolution via FFT (real input → real output)
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution) using an FFT for speed."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    nfft = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=nfft)
    conv_full = np.fft.irfft(A * A, n=nfft)[:size]
    # Numerical errors may lead to tiny negative values – clip for safety
    return np.clip(conv_full, 0.0, None)

                w = exp_vals / exp_vals.sum()               # probabilities, sum=1

                # correlation w ⋆ a (valid mode gives exactly Σ_j w_{i+j} a_j)
                corr = np.correlate(w, a, mode='valid')     # length n
                grad = 2.0 * corr - w[2 * np.arange(n)] * a

                # ----- momentum update and projected step -----
                vel = momentum * vel - (1.0 - momentum) * grad
                eta = base_eta / (1.0 + 1e-5 * it)          # mild decay
                a_new = a + eta * vel                       # descend direction
                a_new = _project_to_simplex(a_new, target_sum)

                # ----- evaluate true objective (max convolution) -----
                conv_new = _conv_self(a_new)
                new_score = conv_new.max()

                # keep the best ever seen (both locally and globally)
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local_a = a_new.copy()
                    print(f"[n={n} r={rst} it={it:07d}] new local best = {new_score:.6f}")

                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"[n={n} r={rst} it={it:07d}] NEW GLOBAL BEST = {new_score:.6f}")

                # accept the new point (always, since we project)
                a = a_new
                it += 1

                # ----- occasional random jitter to escape flat plateaus -----
                if it % 4500 == 0:
                    noise = np.random.randn(n) * target_sum * 5e-4
                    a = _project_to_simplex(a + noise, target_sum)

                # ----- early exit if we already beat the target -----
                if best_score <= 1.503:
                    print(f"Target reached (score {best_score:.6f}); exiting.")
                    return best_seq.tolist()

    # ------------------------------------------------------------------
    # Final cheap “block” search – random piecewise‑constant candidates.
    if best_seq is None:
        # fall back to a trivial uniform vector
        n = lengths[0]
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _conv_self(best_seq).max()

    block_trials = 300
    n = best_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)
    for _ in range(block_trials):
        if time.time() >= deadline:
            break
        nb = np.random.randint(2, 8)                     # number of blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        cur = _conv_self(cand).max()
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            print(f"[block] improved global best = {best_score:.6f}")
            if best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} (length {best_seq.shape[0]})")
    return best_seq.tolist()
