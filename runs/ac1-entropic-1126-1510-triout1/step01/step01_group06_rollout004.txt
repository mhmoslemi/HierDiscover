"""
Fast smooth‑max based optimisation for the autocorrelation bound.

The score to be minimised (after fixing sum(a)=1) is

    score = 2 * n * max_k (a * a)[k] ,

where * denotes the discrete linear convolution.
The algorithm works as follows:

1.  Pick a random length n (small to medium values work well) and an
    random starting point a (drawn from a Dirichlet, geometric or power‑law
    distribution).  Apply the soft‑max transformation to keep the vector
    on the probability simplex automatically.

2.  Replace the non‑smooth max by a smooth approximation

        smooth_max_λ(x) = (1/λ) * log Σ_k exp(λ * x_k) ,

    with a moderate λ (≈ 8).  The gradient of this quantity can be written
    in closed form and evaluated with two FFT‑based convolutions,
    i.e. in O(n log n).

3.  Perform Adam‑style updates on the unconstrained variables
    (the logits of the soft‑max).  After each step the probabilities are
    recovered by the soft‑max, the current score is evaluated and the best
    vector seen so far is stored.

4.  The whole procedure is restarted many times with different n and
    initialisations while keeping the global best.  When the global time
    budget (≈ 1000 s) is almost exhausted a short projected sub‑gradient
    refinement (identical to the code that achieved 1.577…) is applied to
    the current best vector in order to squeeze out the last bits of
    improvement.

The returned list consists of non‑negative heights; its sum is 1, so the
evaluation function will see the same score that we have been
optimising.
"""

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (always returns a probability vector)."""
    mx = np.max(x)
    ex = np.exp(x - mx)
    return ex / np.sum(ex)


def _smooth_max_and_grad(a: np.ndarray, lam: float):
    """
    Compute (smooth_max, gradient) of max_k (a*a)[k] with smoothing factor lam.

    Returns
        mmax : float        – the ordinary max (needed for the final score)
        grad : np.ndarray   – ∂/∂a of the smooth max (same shape as a)
    """
    n = a.size
    # convolution a*a (real, length 2n-1)
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

    # ordinary max (used for the final score)
    mmax = np.max(conv)

    # smooth weights w_k = exp(lam * (conv_k - max))
    # the subtraction keeps numbers in a safe range.
    w = np.exp(lam * (conv - mmax))

    # weighted gradient:  grad_i = 2 * Σ_k w_k * a_{k-i}
    # This is a convolution of a with the reversed weight vector.
    w_rev = w[::-1]                       # length = 2n-1
    size2 = 1 << ((3 * n - 2).bit_length())   # enough for a (len n) * w_rev (len 2n-1)
    A_fft = np.fft.rfft(a, n=size2)
    W_fft = np.fft.rfft(w_rev, n=size2)
    conv_w = np.fft.irfft(A_fft * W_fft, n=size2)[: 3 * n - 2]

    # Extract the needed slice: index = (2n-2) - i   for i=0..n-1
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv_w[idx]

    return mmax, grad


def _conv_max(a: np.ndarray):
    """Plain max of the self‑convolution (no smoothing)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return np.max(conv)


def _adam_update(x, grad, m, v, t, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """One step of Adam on the logits x."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x_new, m, v

# Optional short sub‑gradient refinement (taken from the previous solution)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray):
    """Maximum value and indices of the self‑convolution (FFT version)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = np.max(conv)
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return float(max_val), idx.astype(int)


def _gradient_for_max(a: np.ndarray, max_idx):
    """Sub‑gradient of the max‑convolution w.r.t. a."""
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


def _refine_subgradient(a0: np.ndarray, time_limit: float):
    """A few seconds of the classic projected sub‑gradient method."""
    a = a0.copy()
    best_a = a.copy()
    n = a.size
    max_c, max_idx = _conv_max_fft(a)
    best_score = 2.0 * n * max_c

    it = 0
    while time.time() < time_limit and it < 3000:
        it += 1
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c
        grad = _gradient_for_max(a, max_idx)

        # simple back‑tracking line search
        eta0 = 0.2 / np.sqrt(it)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            eta = eta0 * factor
            a_candidate = _project_to_simplex(a - eta * grad)
            max_c2, _ = _conv_max_fft(a_candidate)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                break

        if not improved:
            # tiny random jitter
            a_pert = a.copy()
            idx = np.random.randint(n, size=5)
            a_pert[idx] += np.random.randn(5) * 0.001
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / np.sum(a_pert)
            max_c2, _ = _conv_max_fft(a_pert)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score:
                a = a_pert
                cur_score = cand_score

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 200 == 0:
                print(f"    [refine] iter {it}, new best {best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a probability vector) that
    minimises 2 * n * max(conv(a, a)).  The algorithm mixes a smooth‑max
    Adam optimiser with a brief sub‑gradient polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    # Try to initialise from the previously discovered sequence if it exists
    try:
        global height_sequence_1
        init_seq = np.array(height_sequence_1, dtype=float)
        if init_seq.size > 1:
            init_seq = init_seq / np.sum(init_seq)
            best_seq = init_seq.copy()
            best_score = 2.0 * init_seq.size * _conv_max(init_seq)
            print(f"Initialising from height_sequence_1, score {best_score:.6f}")
        else:
            best_seq = None
            best_score = float("inf")
    except Exception:
        best_seq = None
        best_score = float("inf")

    # Hyper‑parameters
    lambda_smooth = 8.0          # smoothing factor for the smooth‑max
    adam_lr = 0.05               # base learning rate for Adam
    max_iters_per_run = 2500    # generous but still safe

    # Candidate lengths – we explore a range, favouring medium sizes
    lengths = [200, 300, 400, 500, 600, 800, 1200, 1600, 2000]

    # Main optimisation loop
    while time.time() < deadline:
        n = int(np.random.choice(lengths))
        # ------------------- random start -------------------
        shape = np.random.choice(["geom", "power", "mixed", "random"])
        if shape == "geom":
            t = np.random.uniform(0.0004, 0.0016)
            r = np.exp(-t)
            a_raw = r ** np.arange(n, dtype=float)
        elif shape == "power":
            p = np.random.uniform(0.2, 1.8)
            a_raw = (np.arange(1, n + 1, dtype=float) ** -p)
        elif shape == "mixed":
            t = np.random.uniform(0.0004, 0.0010)
            r = np.exp(-t)
            p = np.random.uniform(0.4, 1.2)
            a_raw = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)
        else:  # random Dirichlet‑like
            a_raw = np.random.gamma(0.7, 1.0, size=n)   # slightly sparse

        a_raw = a_raw / np.sum(a_raw)   # probability simplex

        # Use soft‑max parametrisation: initialise logits from log‑probabilities
        x = np.log(a_raw + 1e-12)       # avoid log(0)

        # Adam state
        m = np.zeros_like(x)
        v = np.zeros_like(x)

        # Run Adam optimisation for this restart
        local_best_a = None
        local_best_score = float("inf")

        for t_iter in range(1, max_iters_per_run + 1):
            if time.time() >= deadline:
                break

            a = _softmax(x)                     # current probability vector
            max_conv, grad_smooth = _smooth_max_and_grad(a, lambda_smooth)

            # Convert gradient w.r.t. a into gradient w.r.t. logits x
            s = np.dot(a, grad_smooth)          # a·∇_a f
            grad_x = grad_smooth - s * a        # chain rule through soft‑max

            # Adam step
            x, m, v = _adam_update(x, grad_x, m, v, t_iter, lr=adam_lr)

            # Evaluate the true (non‑smooth) score
            cur_max = _conv_max(a)
            cur_score = 2.0 * n * cur_max

            if cur_score < local_best_score:
                local_best_score = cur_score
                local_best_a = a.copy()
                # Update global best as soon as we see an improvement
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:6.1f}s] Global best: n={n}, shape={shape}, score={best_score:.6f}")

            # Early stopping for this run if we are already very good
            if local_best_score < 1.45:   # a safe margin below the target
                break

        # End of one restart – optionally run a quick sub‑gradient polish
        if best_seq is not None and time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            # Give at most 5 seconds to this polishing step
            polish_time = min(5.0, remaining)
            polished, polish_score = _refine_subgradient(best_seq, time.time() + polish_time)
            if polish_score < best_score:
                best_score = polish_score
                best_seq = polished.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.1f}s] Polished best: n={len(best_seq)}, score={best_score:.6f}")

        # Early exit if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n)

    # Convert to plain Python list for the external evaluator
    return best_seq.tolist()
