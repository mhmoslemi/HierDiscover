def arcsine_raw(n):
    seq = np.empty(n, dtype=float)
    cur = 1.0
    for i in range(n):
        seq[i] = cur
        cur = cur * (2 * i + 1) / (2.0 * (i + 1)) if i+1 < n else cur
    return seq

    rows = np.empty(n*n, dtype=np.int32)

Optimise a non‑negative step‑function (vector a) to minimise
2·n·max(a*a) / (∑a)^2 .  The algorithm starts from the
well‑known arcsine (C_i = binom(2i,i)/4^i) sequence, normalises it so
that ∑a = √(2n) (so the evaluation metric equals max(a*a)).  It then
repeatedly solves a linear programme

    max Σ g_j   s.t.   g ≥ 0,   (a * g)_k ≤ M   ∀k

where M = max(a*a).  The LP finds a vector g with a larger ℓ₁‑norm but
still respecting the same convolution bound.  After scaling g to the
same ℓ₁‑norm (√(2n)) we convex‑combine a and g (mixing factor ½).  The
new vector has a strictly smaller max‑self‑convolution, thus a lower
score.  The process is repeated while time permits and the best
sequence seen is kept.  For each candidate length n a few iterations
are performed; a short random refinement is applied at the end.
The final best sequence (list of floats) is returned.
"""

import time
import numpy as np

# Try importing scipy; if unavailable we fall back to a simple arcsine search.
try:
    from scipy import sparse
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False


def arcsine_raw(n: int) -> np.ndarray:
    """Return the first n coefficients C_i = binom(2i,i)/4**i."""
    seq = np.empty(n, dtype=float)
    cur = 1.0
    for i in range(n):
        seq[i] = cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    return seq


def prepare_lp_indices(n: int):
    """Pre‑compute row/column index arrays for the matrix representing
    the constraints (a * g)_k ≤ M.  Returns (rows, cols, shape)."""
    nn = n * n
    rows = np.empty(nn, dtype=np.int32)
    cols = np.empty(nn, dtype=np.int32)
    # constant block of column indices 0..n-1
    col_block = np.arange(n, dtype=np.int32)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = i + col_block
        cols[pos:pos + n] = col_block
        pos += n
    shape = (2 * n - 1, n)
    return rows, cols, shape


def max_self_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)_k."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def solve_lp_given_f(f: np.ndarray, M: float, rows, cols, shape):
    """Linear programme:
       maximise Σ g_j
       s.t.   g_j ≥ 0
              (f * g)_k ≤ M   for every k.
    Returns the optimal g as a 1‑d ndarray, or None if LP failed."""
    n = f.shape[0]
    # data for the sparse constraint matrix A (size (2n-1) × n)
    data = np.repeat(f, n)
    A = sparse.csr_matrix((data, (rows, cols)), shape=shape)
    b = np.full(shape[0], M, dtype=float)
    c = -np.ones(n, dtype=float)          # linprog minimises, so use -Σg
    bounds = [(0.0, None)] * n
    # Use the HiGHS interior‑point solver – fast and robust.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    return np.array(res.x, dtype=float)


def improve_by_lp(f: np.ndarray, deadline: float, max_iter: int = 20):
    """Perform a few LP‑based improvement steps on f.
    The vector f is assumed to have ℓ₁‑norm = sqrt(2·len(f)).
    Returns a tuple (best_seq, best_score)."""
    n = f.shape[0]
    target_sum = np.sqrt(2.0 * n)
    rows, cols, shape = prepare_lp_indices(n)

    # Initial score (equal to max self‑convolution because sum = √(2n))
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    best_M = max_self_convolution(best_seq)
    # safety in case evaluate_sequence disagrees a bit
    # (they are mathematically equivalent)
    # Start iteration loop
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        M = max_self_convolution(f)
        # Solve LP to find a vector g with larger ℓ₁‑norm but respecting f*g ≤ M
        g = solve_lp_given_f(f, M, rows, cols, shape)
        if g is None:
            break
        # Rescale g so that its ℓ₁‑norm equals target_sum (same as f)
        g_sum = g.sum()
        if g_sum <= 0:
            break
        g = g * (target_sum / g_sum)

        # Convex combination – blending factor 0.5 is a good trade‑off
        t = 0.5
        f_new = (1.0 - t) * f + t * g

        # Compute the new score via the provided evaluator
        try:
            score = evaluate_sequence(f_new.tolist())
        except Exception:
            break

        # Accept improvement only if the score strictly improves
        if score < best_score - 1e-9:  # small epsilon to avoid noise
            best_seq = f_new.copy()
            best_score = float(score)
            best_M = max_self_convolution(best_seq)
            f = f_new
            # Print progress (visible to the harness)
            print(f"[LP‑improve] n={n} iter={it} score={best_score:.6f}")
        else:
            # No improvement – we stop early for this length
            break

    return best_seq, best_score


def random_refine(seq: np.ndarray, deadline: float, steps: int = 10000):
    """Tiny random perturbations that keep ℓ₁‑norm fixed."""
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    while time.time() < deadline and steps > 0:
        steps -= 1
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-3   # move at most ±5e‑4
        cand = best_seq.copy()
        cand[i] = max(0.0, cand[i] + delta)
        cand[j] = max(0.0, cand[j] - delta)
        s = cand.sum()
        if s <= 0:
            continue
        cand = cand / s * best_seq.sum()   # renormalise to the original ℓ₁‑norm
        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if score < best_score:
            best_seq = cand
            best_score = float(score)
            print(f"[refine] length={n} new best={best_score:.6f}")
    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    Strategy:
        1. Try a few moderate lengths (800, 1200, 1600, 2000).
        2. For each length initialise with the arcsine sequence,
           normalise to ℓ₁ = √(2n) and run a few LP‑based improvement steps.
        3. Apply a quick random‑refinement pass.
        4. Keep the best sequence seen over all lengths.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # tiny safety margin
    best_seq = None
    best_score = float('inf')

    # If SciPy is unavailable fall back to the simple arcsine sweep.
    if not SCIPY_AVAILABLE:
        print("SciPy not available – falling back to simple arcsine sweep.")
        # Use the previous arcsine‑only routine (coarse+fine) – omitted for brevity.
        # We'll just return a relatively long arcsine sequence.
        n = 3200
        seq = arcsine_raw(n)
        seq = (seq / seq.sum()).tolist()
        return seq

    # List of candidate lengths (moderate sizes keep LP cheap)
    candidate_lengths = [800, 1200, 1600, 2000]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        # ---- initialise arcsine raw and scale ---------------------------------
        raw = arcsine_raw(n)               # sum ≈ (2/√π)·√n  (≈ 1.13·√n)
        target = np.sqrt(2.0 * n)           # ℓ₁‑norm we enforce
        f = raw * (target / raw.sum())     # scaled vector, ℓ₁ = √(2n)

        # ---- LP‑driven improvement -----------------------------------------
        try:
            seq_lp, score_lp = improve_by_lp(f, deadline, max_iter=15)
        except Exception as e:
            print(f"[error] LP step failed for n={n}: {e}")
            continue

        # ---- tiny stochastic refinement --------------------------------------
        try:
            seq_ref, score_ref = random_refine(seq_lp, deadline, steps=2000)
        except Exception as e:
            print(f"[error] refinement failed for n={n}: {e}")
            seq_ref, score_ref = seq_lp, score_lp

        # --------------------------------------------------------------------
        # Record best across lengths
        if score_ref < best_score:
            best_score = score_ref
            best_seq = seq_ref
            print(f"[update] new overall best length={n} score={best_score:.6f}")

    # If we never found any sequence (unlikely), fall back to a short arcsine.
    if best_seq is None:
        n = 800
        raw = arcsine_raw(n)
        seq = (raw / raw.sum()).tolist()
        return seq

    # Ensure we return a plain Python list as required by the evaluation harness.
    return best_seq.tolist()

"""
Optimise a non‑negative step‑function (vector a) to minimise

    2·n·max(a*a) / (∑a)²

where “*” is the discrete convolution.  The algorithm works as follows:

1.  Choose a moderate length n (800‑2000 works well with the time budget).
2.  Initialise a with the classic arcsine coefficients
    C_i = binom(2i,i)/4**i  (i = 0…n‑1) and scale it so that
    ∑a = √(2·n).  With this normalisation the evaluation metric
    returned by ``evaluate_sequence`` equals the maximal self‑convolution
    M = max(a*a).
3.  Repeatedly solve the linear programme

        maximise   Σ g_j

    a larger ℓ₁‑norm than a while still respecting the same convolution

4.  Scale g back to the same ℓ₁‑norm (√(2·n)) and replace a by the

    therefore the score improves.
5.  Perform a few LP iterations (while there is time left) and finish with
    a cheap random‑refinement that shuffles a tiny amount of mass between
    two coordinates.
6.  Repeat the whole procedure for several candidate lengths and keep the
    best sequence overall.

procedure described in the problem statement but implemented here from
