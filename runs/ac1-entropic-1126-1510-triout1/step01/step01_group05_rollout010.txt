def project_simplex(v):
    # v is np.ndarray (float)
    # we follow algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    A = np.fft.rfft(a, nfft)

def piecewise_powerlaw_sequence(n, p1, p2, frac):
    L = int(max(1, min(n-1, round(frac * n))))  # ensure at least 1, at most n-1
    idx = np.arange(1, n+1, dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:L] = idx[:L] ** (-p1)
    a[L:] = idx[L:] ** (-p2)
    a /= a.sum()

def subgradient_optimize(a, deadline):
    a = a.copy()
    best_a = a.copy()
    best_score = fast_score(a)
    it = 0
    while time.time() < deadline:
        n = len(a)
        # compute conv via FFT
        nfft = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(a, nfft)
        conv = np.fft.irfft(A*A, nfft)[:2*n-1]
        max_b = conv.max()
        k_max = int(conv.argmax())
        # gradient
        i = np.arange(n)
        j = k_max - i
        mask = (j >=0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # step size schedule
        # Use diminishing learning rate: lr = 0.5 / sqrt(it+1)
        lr = 0.5 / np.sqrt(it + 1)
        # scale step by norm of g for stable step magnitude
        normg = np.linalg.norm(g)
        if normg > 0:
            step_vec = lr * g / normg

        if s == 0:

            s = a_new.sum()

"""
Propose a low‑value step function.

The algorithm has three stages:

1. **Randomised piece‑wise power‑law search** –   a fast FFT based
   evaluation is used to explore a three‑parameter family
   a_i ∝ (i+1)^{-p1} for i < β·n and a_i ∝ (i+1)^{-p2} for i ≥ β·n.
   This family is much richer than a single power law and can quickly
   discover a shape with a lower ∥a∗a∥_∞.

2. **Projected sub‑gradient descent** –   starting from the best vector
   of step 1 we iteratively reduce the dominant convolution entry.
   The gradient of the active shift is g_i = 2·a_{k_max-i} (zero when out of
   range).  After a step we clip to non‑negativity and re‑normalise
   (projection onto the probability simplex).  A diminishing learning
   rate drives the process to a local minimum of the evaluation function.

3. **Fine‑grained stochastic hill‑climbing** –   small random mass swaps
   are accepted only when they improve the score, polishing the result.

All heavy work is done with a fast FFT based “fast_score” that mirrors
the public `evaluate_sequence` (which is only called for the final
verification).  The routine respects the 1000 s wall‑time limit and
prints intermediate progress.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper utilities – all top level, no closures
# --------------------------------------------------------------------------- #

def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:                     # guard against degenerate vectors
        return float("inf")
    # length of FFT (next power of two ≥ 2n‑1)
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def piecewise_powerlaw_sequence(n: int,
                                p1: float,
                                p2: float,
                                frac: float) -> np.ndarray:
    """
    Build a decreasing piece‑wise power‑law vector of length n.
    The first `frac·n` entries follow exponent p1, the remainder exponent p2.
    The result is normalised to sum 1.
    """
    L = max(1, min(n - 1, int(round(frac * n))))   # breakpoint index
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:L] = idx[:L] ** (-p1)
    a[L:] = idx[L:] ** (-p2)
    a /= a.sum()
    return a


def random_piecewise_search(n: int,
                            time_deadline: float,
                            rng: np.random.Generator):
    """
    Randomly explore the three‑parameter family (p1, p2, frac) within given
    time limits.  Returns the best vector and its fast_score.
    """
    best_vec = None
    best_sc = float("inf")
    # a few deterministic seeds for reproducibility
    p1_low, p1_high = 0.1, 0.9
    p2_low, p2_high = 0.9, 2.5
    frac_low, frac_high = 0.05, 0.95

    while time.time() < time_deadline:
        p1 = rng.uniform(p1_low, p1_high)
        # ensure p2 > p1 (steeper tail)
        p2 = rng.uniform(max(p1 + 0.05, p2_low), p2_high)
        frac = rng.uniform(frac_low, frac_high)

        vec = piecewise_powerlaw_sequence(n, p1, p2, frac)
        sc = fast_score(vec)

        if sc < best_sc:
            best_sc = sc
            best_vec = vec
            print(f"[rnd] new best: score={best_sc:.6f} p1={p1:.3f} "
                  f"p2={p2:.3f} frac={frac:.3f}")

    # fallback – a simple power law if nothing was found
    if best_vec is None:
        vec = piecewise_powerlaw_sequence(n, 0.5, 0.5, 1.0)
        best_vec = vec
        best_sc = fast_score(vec)

    return best_vec, best_sc


def subgradient_optimize(init_vec: np.ndarray,
                         time_deadline: float) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent on the non‑negative simplex.
    The step direction is the gradient of the active convolution entry.
    """
    a = init_vec.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = fast_score(a)

    it = 0
    no_improve = 0
    while time.time() < time_deadline:
        # FFT‑based convolution
        nfft = 1 << ((2 * n - 1).bit_length())
        A = np.fft.rfft(a, nfft)
        conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]

        k_max = int(conv.argmax())               # shift with maximal value

        # gradient of that shift: g_i = 2·a_{k_max-i} (if index in range)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # diminishing learning rate
        lr = 0.7 / np.sqrt(it + 1)

        # take a step (no normalisation of the gradient – magnitude is already
        # appropriate because a_i are tiny, lr keeps the step moderate)
        a_candidate = a - lr * g

        # projection: non‑negative + re‑normalisation to sum 1
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0.0:                     # should virtually never happen
            a_candidate = a.copy()
            s = a_candidate.sum()
        a_candidate /= s

        sc = fast_score(a_candidate)

        if sc < best_sc:
            best_sc = sc
            best_a = a_candidate.copy()
            a = a_candidate
            it = 0
            no_improve = 0
        else:
            # keep current a (no acceptance) – a simple descent scheme
            it += 1
            no_improve += 1

        # occasional reset to the best found to avoid drifting too far
        if no_improve > 3000:
            a = best_a.copy()
            no_improve = 0

    return best_a, best_sc


def hill_climb_refine(start_vec: np.ndarray,
                      time_deadline: float,
                      max_no_improve: int = 4000) -> (np.ndarray, float):
    """
    Stochastic hill climbing that moves a small fraction of mass from one
    index to another; only strictly improving moves are kept.
    """
    n = start_vec.shape[0]
    rng = np.random.default_rng(123456)
    cur = start_vec.copy()
    cur_sc = fast_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    no_improve = 0

    while time.time() < time_deadline and no_improve < max_no_improve:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 20 % of the source entry
        delta = rng.random() * 0.20 * cur[i]
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = fast_score(cand)

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
        else:
            no_improve += 1

    return best, best_sc
