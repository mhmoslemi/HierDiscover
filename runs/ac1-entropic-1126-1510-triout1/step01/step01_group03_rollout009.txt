def project_simplex(v):
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v))+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_step(a, step):
    n = len(a)
    conv = np.convolve(a, a)
    max_val = conv.max()
    # identify indices where conv is near max
    # use tolerance = max_val * 1e-12
    tol = max_val * 1e-12 + 1e-14
    mask = conv >= (max_val - tol)
    # build indicator array b of same length as conv
    b = np.where(mask, 1.0, 0.0)
    # compute correlation of a with b
    corr = np.correlate(a, b, mode='full')
    start = len(b) - 1
    grad = 2 * corr[start: start + n]  # length n
    # update
    new_a = a - step * grad
    # project onto simplex
    new_a = project_simplex(new_a)
    return new_a, max_val

def project_simplex(v):
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho_idx = rho[-1]
        theta = cssv[rho_idx] / (rho_idx + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence with a low evaluation score.

This version starts from the classic arcsine (C_i) sequence and then
performs a simple sub‑gradient descent on the “max‑convolution” objective.
The gradient of the current maximal convolution entries is computed via
FFT‑based correlation, the update is projected onto the probability simplex
(non‑negative, sum = 1) and an adaptive step‑size is used.
A short exhaustive sweep over a few lengths is performed, the best length
is further refined, and a tiny random local search finishes the run.
The whole procedure respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Generate the (1‑z)^(-1/2) coefficients C_i for i=0..n‑1,
    normalised to sum‑to‑one."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=np.float64) / total
    return seq.tolist()


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {x | x_i >= 0, sum x_i = 1} (Duchi et al. 2008)."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, n: int, Nfft: int,
                      step: float) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Perform one sub‑gradient step that tries to reduce the maximal entry of
    a*a.  Returns (a_new, max_new, full_conv) where max_new is the new
    maximal convolution value.
    """
    # FFT of current a
    Af = np.fft.rfft(a, Nfft)

    # linear convolution a*a (only the first 2*n‑1 entries are needed)
    conv_full = np.fft.irfft(Af * Af, Nfft)[: 2 * n - 1]

    max_val = conv_full.max()
    # mask of all indices attaining (approximately) the maximal value
    eps = max_val * 1e-12 + 1e-15
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # FFT of mask
    Bf = np.fft.rfft(mask, Nfft)

    # linear correlation a * mask  (full length = n + (2n‑1)‑1 = 3n‑2)
    corr_full = np.fft.irfft(Af * np.conj(Bf), Nfft)

    # slice corresponding to non‑negative lags (0 … n‑1)
    start = 2 * n - 2               # = len(mask) - 1
    grad = 2.0 * corr_full[start : start + n]

    # take a step opposite to the gradient and project onto the simplex
    a_candidate = a - step * grad
    a_candidate = _project_simplex(a_candidate)

    # evaluate the candidate
    Af2 = np.fft.rfft(a_candidate, Nfft)
    conv2 = np.fft.irfft(Af2 * Af2, Nfft)[: 2 * n - 1]
    max2 = conv2.max()

    return a_candidate, max2, conv2


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑score coefficient sequence within ``budget_s`` seconds."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # ------------------------------------------------------------------
    # 1) Coarse sweep over a handful of lengths
    lengths = [800, 1200, 1600, 2000, 2400]
    best_seq = None
    best_score = float('inf')
    best_n = None

    for n in lengths:
        if time.time() >= deadline:

        # initialise with the arcsine construction

        # parameters for FFT‑based operations
        Nfft = 1 << ((3 * n - 2).bit_length())   # ≥ 3n‑2

        # a few quick sub‑gradient iterations (max 300 or time‑limited)
        step = 0.05
        cur_score = 2.0 * n * np.max(np.fft.irfft(np.fft.rfft(a, Nfft) *
                                                   np.fft.rfft(a, Nfft), Nfft)[: 2 * n - 1])
        iter_deadline = time.time() + 3.0          # 3 s per length (very cheap)

        while time.time() < iter_deadline:
            a_new, max_new, _ = _subgradient_step(a, n, Nfft, step)
            score_new = 2.0 * n * max_new

    # ------------------------------------------------------------------

        Nfft = 1 << ((3 * n - 2).bit_length())

            if score_new < cur_score:               # accept improvement

                cur_score = score_new

            else:

    # ------------------------------------------------------------------

            cand[i] = max(0.0, cand[i] + delta)

            cand = _project_simplex(cand)             # ensure sum‑to‑1
            # evaluate quickly (sum is 1, so metric = 2·n·max)
            conv = np.convolve(cand, cand)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                a = cand
                print(f"[tiny] improved score = {best_score:.6f}")

    print(f"\nBest sequence length = {len(best_seq)} with score = {best_score:.6f}")
    return best_seq.tolist()
