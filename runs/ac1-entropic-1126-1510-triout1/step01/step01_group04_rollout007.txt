"""
Generate a simple linear decreasing (triangular) step function.
The shape a_i = 2*(n-i)/(n*(n+1)) is known to minimise the maximum
self‑convolution among all non‑negative sequences of a given length,
giving the theoretical bound
    score = 4/3 * (n+2)/(n+1) → 4/3 ≈ 1.333…
The implementation builds this sequence, evaluates it, and then
performs a very cheap stochastic hill‑climb for the remaining time
in case a tiny improvement exists.
"""

import time
import numpy as np

def _triangular_sequence(n: int) -> np.ndarray:
    """Return a normalised decreasing linear (triangular) distribution of length n."""
    # a_i = 2 * (n - i) / (n * (n + 1))
    i = np.arange(n, dtype=np.float64)
    seq = 2.0 * (n - i) / (n * (n + 1))
    # Normalise – the formula already sums to 1, but floating‑point noise may appear.
    seq /= seq.sum()
    return seq

def _hill_climb(seq: np.ndarray, score: float, deadline: float) -> tuple[np.ndarray, float]:
    """Very cheap stochastic hill‑climb on the simplex."""
    n = seq.shape[0]
    cur = seq.copy()
    best_score = score
    rng = np.random.default_rng()
    it = 0
    # Try at most a few ten‑thousands of proposals – stop when out of time.
    while time.time() < deadline and it < 20000:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        # move a random fraction (≤3 %) of mass from i to j
        delta = cur[i] * rng.random() * 0.03
        cur[i] -= delta
        cur[j] += delta
        new_score = evaluate_sequence(cur.tolist())
        if new_score < best_score:
            best_score = new_score
            # keep the change
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
    if it:
        print(f"Hill‑climb performed {it} iterations, best_score={best_score:.6f}")
    return cur, best_score
