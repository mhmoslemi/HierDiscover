def _improve_lp(a, n, deadline):
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    cur_a = a.copy()
    t_prev = None
    b_var = None
    t_var = None
    while time.time() < deadline:
        # compute M
        conv = np.convolve(cur_a, cur_a)
        M = np.max(conv)
        # build matrix A
        rows = 2 * n - 1
        # Use dense A
        A = np.zeros((rows, n), dtype=np.float64)
        # fill columns
        for i in range(n):
            A[i:i+n, i] = cur_a[i]
        # Setup LP
        b = cp.Variable(n, nonneg=True)
        t = cp.Variable()
        constraints = [cp.sum(b) == 1, A @ b <= t]
        prob = cp.Problem(cp.Minimize(t), constraints)
        # Warm start: provide initial values
        if t_prev is not None:
            t.value = t_prev
        if b_var is not None:
            b.value = b_var
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-06, reltol=1e-06)
        if b.value is None:
            # fallback: break
            break
        b_opt = np.array(b.value).flatten()
        # ensure nonnegative
        b_opt = np.maximum(b_opt, 0)
        # normalize sum (should be ~1). Might deviate due to tolerance.
        sum_b = b_opt.sum()
        if sum_b == 0:
            break
        b_opt = b_opt / sum_b
        # compute new score
        new_conv = np.convolve(b_opt, b_opt)
        new_M = np.max(new_conv)
        new_score = 2.0 * n * new_M

            break

        b_var = b_opt
        t_prev = new_M
        # optionally also break if best_score < target ~1.5
        if best_score < 1.5:
            break
    return best_a, best_score

"""
Propose a low‑value step function using an LP‑based flattening of the
self‑convolution.

The score to minimise is   score = 2·n·max(conv(a,a))   (the sum of a is
normalised to 1).  For a fixed length n we repeatedly solve a linear
program

    minimise   t
    subject to A(a)·b ≤ t ,   Σ b = 1 ,   b ≥ 0

where A(a) is the convolution matrix of the current vector a:
(A·b)[k] = Σ_i a_i·b_{k-i}.  The optimum b is the distribution that,
while keeping the same total mass, yields the smallest possible peak of
the cross‑convolution with a.  Replacing a by this b never increases the
peak of the self‑convolution, and repeating the step converges to a
flattened autocorrelation.

The algorithm:
  * pick a random length n (1500–2500) and random initial a (Dirichlet,
    geometric or power‑law);
  * run a few LP improvement iterations (≤ 15 s each restart);
  * keep the best sequence seen across many restarts (including an
    optional warm start from the global variable ``height_sequence_1``);
  * return the best sequence as a plain Python list.

The implementation uses NumPy, SciPy (for FFT) and CVXPY (ECOS solver).
"""

import time
import numpy as np
import cvxpy as cp
from numpy.fft import rfft, irfft


def _conv_max_fft(a: np.ndarray) -> float:
    """Return max value of the self‑convolution a∗a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n convolution matrix A such that A @ b = a∗b.
    Row k holds a[i] for i where 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contains a[i] on rows i … i+n‑1
    for i in range(n):
        A[i : i + n, i] = a[i]
    return A


def _lp_improve(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform iterative LP improvement on a0 until the deadline is reached.
    Returns the best vector found and its score 2·n·max(conv).
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max_fft(a)

    # warm‑start values for the LP variables
    b_prev = None
    t_prev = None

    it = 0
    while time.time() < deadline:
        it += 1
        # Build the convolution matrix for the current a
        A = _build_conv_matrix(a)

        # LP variables: b (new distribution) and t (peak bound)
        b = cp.Variable(n, nonneg=True)
        t = cp.Variable(nonneg=True)

        constraints = [
            cp.sum(b) == 1,
            A @ b <= t,
        ]

        # warm‑starts (if we have a previous solution)
        if b_prev is not None:
            b.value = b_prev
        if t_prev is not None:
            t.value = t_prev

        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(
                solver=cp.ECOS,
                warm_start=True,
                max_iters=2000,
                abstol=1e-07,
                reltol=1e-07,
                feastol=1e-07,
            )
        except Exception:      # if the solver fails, abort this run
            break

        if b.value is None:
            break

        b_opt = np.maximum(np.asarray(b.value).flatten(), 0.0)
        sum_b = b_opt.sum()
        if sum_b == 0.0:
            break
        b_opt /= sum_b                # enforce Σb = 1 exactly (numerical safety)

        # evaluate the new peak
        new_max = _conv_max_fft(b_opt)
        new_score = 2.0 * n * new_max

        # keep the best ever seen in this run
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = b_opt.copy()

        # stop if improvement becomes negligible
        if new_score >= best_score - 1e-9:
            # no further progress
            break

        # prepare for the next iteration
        a = b_opt
        b_prev = b_opt
        t_prev = new_max

        # early exit if we already beat the target bound
        if best_score < 1.503:
            break

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The function respects the required time budget and returns a plain
    list of non‑negative heights (already normalised to sum = 1).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 1) Try to improve the previously found construction if it exists.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be a very long list; crop to a manageable size.
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.size > 2500:
            # simple down‑sampling by taking an equally spaced subsequence
            idx = np.linspace(0, init.size - 1, 2500, dtype=int)
            init = init[idx]
        init = np.maximum(init, 0.0)
        init = init / init.sum()
        sub_deadline = min(deadline, time.time() + 30.0)
        a_opt, score_opt = _lp_improve(init, sub_deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"Global improvement from warm start: n={a_opt.size}, score={best_score:.6f}")
    except NameError:
        # No warm start available – simply continue.
        pass

    # ------------------------------------------------------------------
    # 2) Random restarts with LP‑based refinement.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a problem size
        n = np.random.randint(1500, 2500)

        # random initial shape
        shape = np.random.choice(["geom", "power", "mixed", "random"])
        if shape == "geom":
            t = np.random.uniform(0.0003, 0.0015)
            r = np.exp(-t)
            a = r ** np.arange(n, dtype=np.float64)
        elif shape == "power":
            p = np.random.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=np.float64) ** -p)
        elif shape == "mixed":
            t = np.random.uniform(0.0003, 0.0010)
            r = np.exp(-t)
            p = np.random.uniform(0.5, 1.5)
            a = (r ** np.arange(n, dtype=np.float64)) * (np.arange(1, n + 1, dtype=np.float64) ** -p)
        else:  # random Dirichlet
            a = np.random.gamma(1.0, 1.0, size=n)

        # ensure positivity and normalise
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        # give each start ~15 s of optimisation (or less if time runs out)
        sub_deadline = min(deadline, time.time() + 15.0)
        a_opt, score_opt = _lp_improve(a, sub_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"Global improvement: n={n}, shape={shape}, score={best_score:.6f}")

        if best_score < 1.503:
            break

    # Fallback – uniform distribution (should never be used)
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    return best_seq.tolist()
