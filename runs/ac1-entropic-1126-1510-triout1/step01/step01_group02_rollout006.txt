def generate_sidon_set(n, target_k=None):
    S = []
    existing_sums = set()
    for x in range(n):
        # compute potential new sums: x+x and x + s for s in S
        conflict = False
        if (x + x) in existing_sums:
            conflict = True
        else:
            for s in S:
                if (x + s) in existing_sums:
                    conflict = True
                    break
        if not conflict:
            S.append(x)
            # add new sums
            existing_sums.add(x + x)
            for s in S[:-1]:
                existing_sums.add(x + s)
        if target_k and len(S) >= target_k:
            break
    return S

"""
Hybrid optimization for the step‑function bound.

Stage 1 – Linear‑programming “mass‑reallocation’’:
    * Normalise a vector to the probability simplex (sum = 1).
    * Let M = max((f * f)).  Solve the LP
          maximise   Σ g_i
          subject to (f * g)_k ≤ M   for all k,
                     g_i ≥ 0.
      The solution g pushes mass onto indices where f is small,
      thus reducing the peak of the autocorrelation.
    * Mix the current vector with the normalised LP‑solution:
          f ← (1−t)·f + t·g ,   t slowly decays.
    * Repeat while a fraction of the time budget remains.

Stage 2 – Fine‑tuning (projected gradient + random pair swaps):
    * Perform projected gradient steps on the location of the current
      maximal convolution entry.
    * Interleave with small random “pair‑moves’’ that keep the sum
      exactly constant.
    * Classic Metropolis acceptance with a linear cooling schedule.

Both stages keep the best vector seen so far and return it as a plain
list.  The algorithm works for any length ≤ ~2500 (the LP matrix is
dense with size (2n‑1)×n) and respects the 1000‑second wall‑time limit.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex using the algorithm of
    Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: re‑normalise exactly
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Convolution matrix (dense) needed for the linear program
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1)×n matrix A with A_{k,i}=f_{k‑i} (0 when out of range)."""
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    # each column i receives the vector f starting at row i
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
# One LP step: maximise Σg_i under the convolution bound M.
# Returns a normalised direction g (sum = 1).  If the LP fails,
# falls back to a uniform vector.
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    n = f.size
    A = _conv_matrix(f)                     # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=float)  # (f * g)_k ≤ M
    c = -np.ones(n, dtype=float)            # maximise Σg_i  → minimise −Σg_i
    bounds = [(0.0, None)] * n

    # Highs is fast and supports dense matrices directly.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success and res.x is not None:
        g = res.x
        s = g.sum()
        if s > 1e-12:
            return g / s  # normalise to the simplex
    # fallback – uniform distribution (guaranteed feasible)
    return np.full(n, 1.0 / n, dtype=float)


# ----------------------------------------------------------------------
# Gradient step that tries to reduce the current maximal convolution entry
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """
    One projected gradient step on the location of the maximal
    convolution entry.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    # gradient of (seq*seq)_k w.r.t. seq is 2*seq reversed around k
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)

    g = np.zeros_like(seq)
    g[mask] = 2.0 * seq[j[mask]]

    new_seq = seq - eta * g
    return _project_simplex(new_seq)


# ----------------------------------------------------------------------
# Random pairwise mass‑swap move (preserves the sum exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    delta = (np.random.rand() - 0.5) * sigma  # zero‑mean proposal

    if delta > 0:
        delta = min(delta, seq[j])  # take from j
    else:
        delta = -min(-delta, seq[i])  # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that
    strives to minimise ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds, first applying
    a linear‑programming based mass‑reallocation phase and then a
    hybrid projected‑gradient / simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start = time.time()
    # tiny safety margin to avoid overrunning the limit
    deadline = start + max(0.0, budget_s - 0.5)

    # --------------------------------------------------------------
    # Choose a working length (keep it modest for the LP phase)
    # --------------------------------------------------------------
    MAX_N = 2000                     # safe upper bound for the dense LP matrix
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = min(base.size, MAX_N)
        base = base[:n]              # truncate if necessary
    except Exception:
        n = MAX_N
        base = np.full(n, 1.0 / n, dtype=float)

    # Normalise onto the simplex (sum = 1)
    base = np.maximum(base, 0.0)
    if base.sum() == 0:
        base = np.full_like(base, 1.0 / n)
    else:
        base = base / base.sum()
    current = _project_simplex(base)

    # --------------------------------------------------------------
    # Initial evaluation
    # --------------------------------------------------------------
    try:
        best_score = evaluate_sequence(current.tolist())
    except Exception:
        best_score = float("inf")
    best = current.copy()
    current_score = best_score

    # --------------------------------------------------------------
    # Phase 1 – LP guided mass redistribution (≈30 % of total time)
    # --------------------------------------------------------------
    lp_time_fraction = 0.30
    lp_deadline = start + lp_time_fraction * budget_s

    it = 0
    while time.time() < lp_deadline:
        it += 1

        # current bound M
        conv = np.convolve(current, current)
        M = conv.max()

        # LP direction
        g = _lp_step(current, M)

        # mixing coefficient – gradually shrink to keep steps stable
        t = 0.3 / np.sqrt(it)          # start relatively aggressive
        candidate = (1.0 - t) * current + t * g
        candidate = _project_simplex(candidate)

        # evaluate
        try:
            cand_score = evaluate_sequence(candidate.tolist())
        except Exception:
            continue

        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()
            print(f"[LP  It {it:5d}] New best score: {best_score:.6f}")

        # simple Metropolis acceptance (temperature decays linearly)
        elapsed = (time.time() - start) / (deadline - start)
        temperature = 0.5 * (1.0 - elapsed) + 0.001 * elapsed
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score

    # --------------------------------------------------------------
    # Phase 2 – Fine‑grained stochastic optimisation (remainder of time)
    # --------------------------------------------------------------
    it = 0
    last_improve = 0
    while time.time() < deadline:
        it += 1

        # cooling schedule for simulated‑annealing part
        elapsed_frac = (time.time() - start) / (deadline - start)
        temperature = 0.9 * (1.0 - elapsed_frac) + 0.001 * elapsed_frac

        # decide move type
        if np.random.rand() < 0.55:
            # projected gradient step – step size decays like 1/√it
            eta = 0.6 / np.sqrt(it)
            candidate = _gradient_step(current, eta)
        else:
            # random pairwise swap – sigma also decays
            sigma = 0.2 / np.sqrt(it)
            candidate = _random_pair_move(current, sigma)

        # evaluate candidate
        try:
            cand_score = evaluate_sequence(candidate.tolist())
        except Exception:
            continue

        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()
            last_improve = it
            print(f"[Grad It {it:6d}] New best score: {best_score:.6f}")

        # Metropolis acceptance
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score

        # occasional restart if stuck for a while
        if it - last_improve > 5000:
            perturb = np.random.rand(n)
            perturb = perturb / perturb.sum()
            current = _project_simplex(0.95 * best + 0.05 * perturb)
            current_score = evaluate_sequence(current.tolist())
            last_improve = it
            print(f"[Restart] Injected random perturbation at iteration {it}")

    # --------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # --------------------------------------------------------------
    return best.tolist()
