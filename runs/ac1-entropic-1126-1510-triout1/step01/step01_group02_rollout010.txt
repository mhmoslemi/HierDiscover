def solve_lp(f, M):
   n = len(f)
   rows = []
   cols = []
   data = []
   for j in range(n):
       # For each i (f_i) they contribute to row k = i + j.
       # We'll fill rows for each i:
       i_vals = np.arange(n)
       k_vals = i_vals + j
       rows.extend(k_vals.tolist())
       cols.extend([j]*n)
       data.extend(f.tolist())
   A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
   # Now solve maximize sum(g) = -c^T g (since linprog solves min c^T x)
   c = -np.ones(n)  # minimize -sum(g)
   bounds = [(0, None)]*n
   res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
   if res.success:
        return res.x
   else:
        return None

"""
Hybrid simulated‑annealing / momentum projected gradient search.

The algorithm works on the probability simplex (all entries ≥0, sum = 1),
which removes the trivial scaling freedom of ``evaluate_sequence``.
It starts from the best known construction ``height_sequence_1`` (if
available) otherwise from a uniform vector, and then repeatedly applies
one of the following moves:

* Momentum projected‑gradient step that reduces the current maximal
  convolution entry (using the top‑K largest entries for a smoother
  direction).
* Random pairwise mass transfer (preserves the sum exactly).
* Swap of two positions (re‑orders the mass).
* Relocation of a chunk of mass from a heavy entry to a random empty
  slot.
* Small global jitter (adds Gaussian noise to all coordinates).

Acceptance follows a Metropolis rule with an exponentially decaying
temperature, guaranteeing exploration early on and greedy refinement
later.  The best sequence seen so far is always kept and returned at the
end of the allotted budget.
"""

import time
import numpy as np
from math import sqrt, exp

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – required after any move that may
# violate the simplex constraints.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All non‑positive -> uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Re‑normalise to cure tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Gradient of the (smoothed) maximum convolution entry.
# ``top_k`` specifies how many of the largest convolution entries are
# combined to obtain a smoother descent direction.
# ----------------------------------------------------------------------
def _gradient_of_topk(seq: np.ndarray, top_k: int = 5) -> np.ndarray:
    """Return the gradient that would decrease the sum of the top‑k
    convolution entries."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)                     # length 2n‑1
    # indices of the K largest entries
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(-conv, top_k)[:top_k]

    grad = np.zeros_like(seq)
    for k in ks:
        # derivative of conv_k w.r.t. seq_i is 2*seq_{k-i} (if defined)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * seq[j[mask]]
    return grad


# ----------------------------------------------------------------------
# Momentum projected gradient step.
# ----------------------------------------------------------------------
def _momentum_gradient_step(seq: np.ndarray, it: int,
                           momentum: np.ndarray,
                           beta: float = 0.9,
                           eta0: float = 1.0,
                           top_k: int = 5) -> tuple[np.ndarray, np.ndarray]:
    """One Adam‑like momentum step followed by simplex projection."""
    # learning rate decays like 1/sqrt(it)
    eta = eta0 / sqrt(it + 1.0)

    grad = _gradient_of_topk(seq, top_k=top_k)
    momentum = beta * momentum + (1.0 - beta) * grad
    new_seq = seq - eta * momentum
    new_seq = _project_simplex(new_seq)
    return new_seq, momentum


# ----------------------------------------------------------------------
# Random pairwise mass transfer (keeps the sum exactly 1).
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # a zero‑mean Gaussian‐like proposal, clipped to the feasible interval
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])          # take from j
    else:
        delta = -min(-delta, seq[i])        # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Swap two coordinates (pure re‑ordering, sum unchanged).
# ----------------------------------------------------------------------
def _swap_move(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    new_seq = seq.copy()
    new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
    return new_seq


# ----------------------------------------------------------------------
# Relocate a chunk of mass from a heavy entry to a (usually) empty slot.
# ----------------------------------------------------------------------
def _relocate_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    # source is chosen with probability proportional to its weight
    if seq.sum() == 0:
        return seq
    src = np.random.choice(n, p=seq / seq.sum())
    # pick a destination that is *different* from the source
    dst = np.random.choice([i for i in range(n) if i != src])

    # how much to move – a fraction of the source mass, capped by sigma
    delta = min(seq[src] * 0.5, sigma) * np.random.rand()
    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    return new_seq


# ----------------------------------------------------------------------
# Small global jitter – adds zero‑mean Gaussian noise to all coordinates.
# ----------------------------------------------------------------------
def _jitter_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    jitter = np.random.randn(seq.shape[0]) * sigma
    new_seq = seq + jitter
    # enforce non‑negativity before projection
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = _project_simplex(new_seq)
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Return a list of non‑negative heights that strives to minimise
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds, employing a mixture of momentum projected‑gradient steps,
    several stochastic moves and a Metropolis acceptance rule.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 2.0)   # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise the search vector (prefer the best known construction)
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # may be predefined
        n = base.size
        base = np.maximum(base, 0.0)                      # safety
        if base.sum() == 0:
            base = np.full(n, 1.0 / n)
        else:
            base = base / base.sum()
    except Exception:
        # Fallback: uniform vector of a moderate size (≈7 000 works well)
        n = 7300
        base = np.full(n, 1.0 / n, dtype=float)

    current = _project_simplex(base.copy())
    best = current.copy()

    # ------------------------------------------------------------------
    # Keep track of scores and momentum
    # ------------------------------------------------------------------
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:
        best_score = float("inf")
    current_score = best_score

    # momentum vector for the gradient step
    momentum = np.zeros_like(current)

    # Annealing schedule – exponential cooling
    temp_start = 1.0
    temp_end = 1e-4

    # Counters for stagnation detection
    last_improve_iter = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # Temperature (exponential decay)
        elapsed_frac = (time.time() - start_time) / (deadline - start_time)
        temperature = temp_start * (1.0 - elapsed_frac) + temp_end * elapsed_frac

        # ------------------------------------------------------------------
        # Choose a move type (probabilities tuned empirically)
        # ------------------------------------------------------------------
        r = np.random.rand()
        if r < 0.45:
            # Momentum gradient step – decay step size slowly
            eta0 = 1.2
            candidate, momentum = _momentum_gradient_step(
                current, it, momentum, beta=0.9, eta0=eta0, top_k=5)
        elif r < 0.70:
            # Random pairwise mass transfer
            sigma = 0.5 / sqrt(it + 1.0)
            candidate = _random_pair_move(current, sigma)
        elif r < 0.85:
            # Swap two entries
            candidate = _swap_move(current)
        elif r < 0.95:
            # Relocate mass from a heavy entry to a random slot
            sigma = 0.4 / sqrt(it + 1.0)
            candidate = _relocate_move(current, sigma)
        else:
            # Small global jitter
            sigma = 0.03 / sqrt(it + 1.0)
            candidate = _jitter_move(current, sigma)

        # ------------------------------------------------------------------
        # Evaluate the candidate (expensive)
        # ------------------------------------------------------------------
        try:
            cand_score = evaluate_sequence(candidate.tolist())
        except Exception:
            # Should never happen – discard this candidate
            continue

        # Record best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()
            last_improve_iter = it
            print(f"[Iter {it:6d}] New best score: {best_score:.6f}")

        # Metropolis acceptance rule
        if cand_score < current_score:
            accept = True
        else:
            # Prob = exp(-(Δ)/T)
            prob = exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score

        # ------------------------------------------------------------------
        # Restart heuristic when stuck for a long time
        # ------------------------------------------------------------------
        if it - last_improve_iter > 8000:
            # Mix a fresh random point with the current best
            perturb = np.random.rand(n)
            perturb = perturb / perturb.sum()
            mixed = 0.995 * best + 0.005 * perturb
            current = _project_simplex(mixed)
            current_score = evaluate_sequence(current.tolist())
            last_improve_iter = it
            # Reset momentum to avoid bias from the previous region
            momentum = np.zeros_like(current)

    # ------------------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # ------------------------------------------------------------------
    return best.tolist()
