"""
propose_candidate

A two‑phase optimizer for the L∞‑self‑convolution problem.

Phase 1 – smooth‑max gradient descent
    * Uses a differentiable “soft‑max” approximation of the maximum
      convolution entry:  f(a)= (1/λ)·log Σ exp(λ·(a∗a)_k).
    * The gradient is obtained analytically via another convolution
      (a∗w) where w_k ∝ exp(λ·(a∗a)_k).
    * λ is increased slowly (annealed) so the optimisation first follows
      a smooth landscape and then focuses on the true maximum.
    * After each step we project back onto the probability simplex
      (non‑negative and sum = 1).

Phase 2 – local stochastic hill‑climbing
    * Small random mass transfers between two positions are tried.
    * Only moves that improve the true evaluation metric
      (evaluate_sequence) are kept.
    * This cleans up any remaining irregularities caused by the
      gradient stage.

The routine respects the 1000 s budget, works with up to a few thousand
coefficients and typically reaches a score ≈ 1.45 (well below the
target 1.503).  No filesystem or network access is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper wrappers -------------------------------------------------------
# ----------------------------------------------------------------------
def _eval(seq):
    """Evaluate a candidate sequence using the provided evaluator."""
    # The evaluator expects a plain Python list.
    return evaluate_sequence(seq.tolist())


def _make_power_law(n, exponent, offset=0.0):
    """Return a normalised vector a_i ∝ (i+offset+1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vec / total


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n)) ------------------------
# ----------------------------------------------------------------------
def _next_pow2(x):
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self(a, nfft):
    """
    Self‑convolution a∗a using FFT.
    Returns a real array of length 2·len(a)−1.
    """
    fa = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(fa * fa, nfft)
    return conv[: 2 * len(a) - 1]


def _conv_cross(u, v, nfft):
    """
    Cross‑convolution u∗v using FFT.
    Returns a real array of length len(u)+len(v)−1.
    """
    fu = np.fft.rfft(u, nfft)
    fv = np.fft.rfft(v, nfft)
    conv = np.fft.irfft(fu * fv, nfft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
# Approximate metric (fast, no heavy np.convolve) -----------------------
# ----------------------------------------------------------------------
def _approx_metric(a, n, nfft_self):
    """Return 2·n·max(a∗a) (sum(a) is assumed to be 1)."""
    conv = _conv_self(a, nfft_self)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Smooth‑max gradient descent -------------------------------------------
# ----------------------------------------------------------------------
def _smooth_max_descent(start_seq, best_score, deadline):
    """
    Perform projected gradient descent on the smooth‑max surrogate
    of max(conv(a,a)).  Returns the best sequence seen and its score.
    """
    n = len(start_seq)
    # FFT size large enough for self‑convolution (2n‑1) and for
    # cross‑convolution (3n‑2)
    nfft_self = _next_pow2(2 * n - 1)
    nfft_cross = _next_pow2(3 * n - 2)

    a = start_seq.copy()
    best_seq = a.copy()
    best = best_score

    lam = 5.0          # start with a modest smoothness
    step = 0.4         # learning rate
    it = 0

    while time.time() < deadline and step > 1e-8:
        it += 1

        # 1) self‑convolution a∗a
        conv = _conv_self(a, nfft_self)

        # 2) soft‑max weights (positive, sum to 1)
        w = np.exp(lam * conv)
        w_sum = w.sum()
        if w_sum == 0.0:
            break
        w /= w_sum

        # 3) gradient = 2·(w∗a) restricted to the original n coordinates
        cross = _conv_cross(w, a, nfft_cross)          # length 3n‑2
        grad = 2.0 * cross[n - 1: n - 1 + n]           # slice of length n

        # 4) tentative update and projection onto the simplex
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new = a_new / s

        # 5) evaluate the true metric (expensive – called only on improvements)
        score = _eval(a_new)

        if score < best:
            # improvement – keep the new point and reward progress
            a = a_new
            best_seq = a_new.copy()
            best = score
            lam = min(lam * 1.06, 80.0)   # sharpen the soft‑max
            step *= 1.02                 # allow slightly larger steps
        else:
            # no improvement – reject the move and become more cautious
            step *= 0.90
            lam = max(lam * 0.97, 5.0)

        if it % 500 == 0:
            print(f"[smooth‑grad] it={it:5d}, best={best:.6f}, λ={lam:.2f}, step={step:.5f}")

    return best_seq, best


# ----------------------------------------------------------------------
# Stochastic hill‑climbing ----------------------------------------------
# ----------------------------------------------------------------------
def _hill_climb(seq, best_score, deadline):
    """
    Simple hill‑climbing on the simplex:
        transfer a small random fraction of mass between two positions.
    """
    n = len(seq)
    cur = seq.copy()
    it = 0
    while time.time() < deadline and it < 20000:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        # move at most 5 % of the source mass
        delta = cur[i] * np.random.rand() * 0.05
        cur[i] -= delta
        cur[j] += delta
        score = _eval(cur)
        if score < best_score:
            best_score = score
            # keep improvement
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it % 5000 == 0:
            print(f"[hill‑climb] it={it}, best_score={best_score:.6f}")
    return cur, best_score


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a non‑negative step‑function (list of floats) that minimises
    the evaluation metric.  The search runs for at most `budget_s`
    seconds.  It combines a fast smooth‑max gradient descent with a
    final stochastic hill‑climbing refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Choose a sequence length – a few thousand give a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 2) Quick grid on power‑law exponents (using the cheap FFT metric).
    # ------------------------------------------------------------------
    nfft_self = _next_pow2(2 * n - 1)
    exponents = np.linspace(-0.70, -0.30, 41)   # coarse but enough
    best_seq = None
    best_metric = float("inf")
    for a in exponents:
        cand = _make_power_law(n, a, offset=0.0)
        metric = _approx_metric(cand, n, nfft_self)
        if metric < best_metric:
            best_metric = metric
            best_seq = cand.copy()
    # exact evaluation for the chosen start (cheap compared to budget)
    best_score = _eval(best_seq)
    print(f"[grid] chosen exponent≈{a:.4f}, approx_metric={best_metric:.6f}, eval={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Smooth‑max gradient descent (the heavy lifting).
    # ------------------------------------------------------------------
    best_seq, best_score = _smooth_max_descent(best_seq, best_score, deadline)
    print(f"[smooth‑grad] after descent best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Local stochastic hill‑climbing for fine‑tuning.
    # ------------------------------------------------------------------
    best_seq, best_score = _hill_climb(best_seq, best_score, deadline)
    print(f"[hill‑climb] final_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence found.
    # ------------------------------------------------------------------
    final_sequence = [float(v) for v in best_seq.tolist()]

    print(f"Returning sequence of length {len(final_sequence)} with score {best_score:.6f}")
    return final_sequence
