def project_to_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑value step‑function using a fast projected‑gradient / directed hill‑climb.
The algorithm

1. builds an initial decreasing power‑law (≈optimal analytically) of length n,
   optionally warm‑started from the global variable `height_sequence_1`;
2. repeatedly finds the convolution index k with the current maximum;
3. moves a small fraction of the mass from the index that contributes most
   to that peak towards the index that contributes least;
4. accepts the move only if the score improves (the score = 2·n·max(conv));
5. the step size adapts automatically and the search stops when the time budget
   (≈1000 s) is exhausted or no improvement has been seen for many iterations.

All constraints of `evaluate_sequence` (non‑negative, sum ≥ 0.01, values ≪ 1000)
are satisfied because the vector is kept normalised to sum = 1 and we never
subtract more than the current entry.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper functions (top‑level, no closures)
# --------------------------------------------------------------------------- #

def power_law_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    return a / a_sum


def convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (fast for n≈few k)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2 * n - 1]
    return conv


def score_of(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))  (the denominator (sum a)^2 is 1 after normalisation)."""
    n = a.shape[0]
    conv = convolution(a)
    return 2.0 * n * conv.max()


def directed_hill_climb(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly flatten the current maximal convolution peak.
    Moves a fraction of the mass from the index that contributes most to the
    peak towards the index that contributes least.  The step size adapts
    automatically; only improving moves are kept.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = score_of(best_a)

    # termination control
    no_improve = 0
    max_no_improve = 3500          # stop after many failed attempts

    # start with a moderately aggressive step fraction
    step_frac = 0.25

    while time.time() < deadline and no_improve < max_no_improve:
        conv = convolution(a)
        k = int(np.argmax(conv))           # shift with the current worst peak

        # indices that participate in this shift
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=np.int64)

        # contribution of each index to the peak (a_i * a_{k-i})
        prod = a[idx] * a[k - idx]

        # source = index with the largest contribution, target = smallest
        src = idx[np.argmax(prod)]
        tgt = idx[np.argmin(prod)]

        # occasionally fall back to a random pair (helps escape tiny plateaus)
        if src == tgt or np.random.rand() < 0.001:
            src, tgt = np.random.choice(idx, size=2, replace=False)

        # amount to relocate (a fraction of the source entry)
        delta = step_frac * a[src]
        if delta <= 0.0:
            # source empty – try a different source next round
            no_improve += 1
            step_frac *= 0.98
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta          # total sum stays unchanged

        cand_score = score_of(cand)

        if cand_score < best_score - 1e-12:      # strict improvement
            a = cand
            best_a = cand.copy()
            best_score = cand_score
            no_improve = 0
            # when we improve we can afford a slightly larger step
            step_frac = min(0.5, step_frac * 1.03)
        else:
            # no improvement – shrink the step and count a failure
            no_improve += 1
            step_frac = max(1e-5, step_frac * 0.97)

    return best_a


def initialise_sequence(n: int) -> np.ndarray:
    """
    Return a starting sequence of length n.
    If the global variable `height_sequence_1` exists and is long enough we
    reuse its prefix, otherwise we fall back to a power‑law with exponent 0.5.
    The result is normalised to sum = 1.
    """
    try:
        seq = height_sequence_1               # type: ignore[name-defined]
        if isinstance(seq, (list, tuple, np.ndarray)) and len(seq) >= n:
            a = np.array(seq[:n], dtype=np.float64)
        else:
            raise ValueError
    except Exception:
        a = power_law_sequence(n, 0.5)

    # normalise – guarantees sum ≥ 0.01 and stays well below the 1000 cap
    total = a.sum()
    if total <= 0:
        a = power_law_sequence(n, 0.5)
        total = a.sum()
    return a / total


# --------------------------------------------------------------------------- #
# Main entry point required by the competition framework
# --------------------------------------------------------------------------- #

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Produce a list of non‑negative heights whose evaluation score is as low as
    possible within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # Length choice – a few thousand entries give a good trade‑off
    n = 2500

    print(f"[setup] chosen length = {n}")

    # ------------------------------------------------------------------- #
    # 1) initialise
    # ------------------------------------------------------------------- #
    a = initialise_sequence(n)
    init_score = score_of(a)
    print(f"[init] initial score = {init_score:.6f}")

    # ------------------------------------------------------------------- #
    # 2) directed hill‑climb (the heavy lifting)
    # ------------------------------------------------------------------- #
    a_opt = directed_hill_climb(a, deadline)

    final_score = score_of(a_opt)
    print(f"[final] best score = {final_score:.6f}  length = {len(a_opt)}")

    # Convert to plain Python list – the framework will call evaluate_sequence
    # which clamps values, but all entries are already safe.
    return a_opt.tolist()
