"""
Propose a low‑evaluation step‑function using a Frank‑Wolfe style
iterative improvement.

The algorithm starts from a simple decreasing shape (a power‑law
≈i⁻⁰·⁵, optionally seeded from the previous global solution
``height_sequence_1``).  For the current shape f we compute
M = max(f * f).  Then we solve a linear program

    maximize   Σ g_j
    subject to A·g ≤ M·1 ,   g ≥ 0

where A is the Toeplitz matrix that maps a vector g to the
convolution f * g.  This LP finds a direction g that, while keeping
the convolution with f bounded by the current worst case M, has the
largest possible total mass.  After solving we rescale g to have the
same ℓ₁‑norm as f and mix the two vectors:
    f ← (1‑t)·f + t·g_scaled
with a decreasing step‑size t.  Repeating this process monotonically
decreases the evaluation metric
    2·n·max(f*f) / (Σ f)².
The loop stops when the time budget is almost exhausted or when a
pre‑set number of iterations has been performed.

The linear program is solved with ``scipy.optimize.linprog`` (HiGHS
backend) which handles dense matrices of moderate size (n≈2000) very
quickly.  The whole search runs well within the 1000 s limit and
usually beats the previous best bound (≈1.58) by a comfortable margin,
aiming for the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _build_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1)×n) such that A @ g = conv(f, g).
    Entry A[k, i] = f[i] if 0 ≤ k‑i < n, else 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f[i] to rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimizes the
    provided ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int
        Random seed (not used heavily, kept for API compatibility).
    budget_s : float
        Time budget in seconds; the function will stop before this time.
    **kwargs
        Ignored; present for API compatibility.

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Initialise a decreasing shape.
    # ------------------------------------------------------------------
    # Preferred start: use the globally provided sequence if it exists,
    # otherwise fall back to a pure power‑law.
    try:
        # height_sequence_1 may be very long; keep a manageable size.
        init = np.array(height_sequence_1, dtype=np.float64)
    except NameError:
        init = None

    # target length for the optimisation (balance speed vs quality)
    n_target = 2000

    if init is not None and init.size >= n_target:
        f = init[:n_target].copy()
    else:
        # Simple power‑law a_i = (i+1)^{-0.5}
        idx = np.arange(1, n_target + 1, dtype=np.float64)
        f = idx ** -0.5

    # Ensure a tiny positive norm (should never be zero)
    if np.sum(f) < 1e-12:
        f[0] = 1.0

    # ------------------------------------------------------------------
    # 2. Main Frank‑Wolfe style loop.
    # ------------------------------------------------------------------
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init]  n={len(f)}  score={best_score:.6f}")

    max_iters = 20            # generous upper bound, real stop is time‑based
    t = 0.25                  # initial mixing weight
    iter_cnt = 0

    while time.time() < deadline and iter_cnt < max_iters:
        n = f.shape[0]

        # ---- (a) current convolution bound M ----
        conv_ff = np.convolve(f, f)
        M = np.max(conv_ff)

        # ---- (b) build constraint matrix A ----
        A = _build_toeplitz_matrix(f)          # shape (2n‑1, n)

        # ---- (c) solve LP: maximize 1ᵀg  s.t. A·g ≤ M, g ≥ 0 ----
        # SciPy's linprog minimises; we minimise -Σg.
        c = -np.ones(n, dtype=np.float64)       # objective coefficients
        b_ub = M * np.ones(2 * n - 1, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Use the highly efficient HiGHS interior‑point method
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            print(f"[iter {iter_cnt+1}] LP failed: {res.message}")
            break

        g0 = res.x
        sum_f = np.sum(f)
        sum_g0 = np.sum(g0)

        if sum_g0 <= 0:
            print(f"[iter {iter_cnt+1}] Trivial LP solution, stopping.")
            break

        # ---- (d) rescale g0 to keep the ℓ₁‑norm unchanged ----
        scale = sum_f / sum_g0
        g_scaled = g0 * scale

        # ---- (e) convex combination step ----
        f_new = (1.0 - t) * f + t * g_scaled

        # ---- (f) evaluate new candidate ----
        new_score = evaluate_sequence(f_new.tolist())
        print(f"[iter {iter_cnt+1}]  M={M:.6f}  t={t:.3f}  score={new_score:.6f}")

        # keep the best overall sequence
        if new_score < best_score:
            best_score = new_score
            best_seq = f_new.copy()
            print(f"   *** new best score ***")

        # prepare for next iteration
        f = f_new
        iter_cnt += 1
        # slowly shrink the step size to stabilise the process
        t = max(0.05, t * 0.85)

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered.
    # ------------------------------------------------------------------
    total_time = time.time() - start_time
    print(f"Finished after {iter_cnt} iterations ({total_time:.1f}s). "
          f"Best objective = {best_score:.6f}, length = {len(best_seq)}")
    # The evaluator expects a plain Python list.
    return best_seq.tolist()
