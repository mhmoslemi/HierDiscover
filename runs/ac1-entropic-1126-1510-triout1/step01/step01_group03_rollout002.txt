def solve_lp_maximizing_sum(f, M):
    n = len(f)
    L = f.sum()  # should be sqrt(2*n)
    # Build sparse matrix as described.
    # Use scipy.sparse.csr_matrix.
    rows = []
    cols = []
    data = []
    for j in range(n):
        # for each shift d where f[d] nonzero
        for d in range(n):
            rows.append(j + d)
            cols.append(j)
            data.append(f[d])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Upper bound vector
    b_ub = np.full(2*n-1, M)
    # Linear programming
    c = -np.ones(n)  # maximize sum -> minimize -sum

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    return res.x

def arcsine_raw(n):
    # generating C_i
    seq = np.empty(n, dtype=float)
    cur = 1.0
    for i in range(n):
        seq[i] = cur
        cur = cur * (2*i + 1) / (2.0 * (i+1)) if i+1 < n else 0.0
    return seq

def arcsine_scaled(n):
    raw = arcsine_raw(n)
    # sum of raw
    S = raw.sum()
    L = math.sqrt(2*n)
    scale = L / S
    scaled = raw * scale
    return scaled

def build_conv_matrix(f):

    # create list

    cols = []

    # vectorized construction to avoid Python loops overhead: use loops over j and compute rows range
    for j in range(n):

        start = j
        end = j + n  # exclusive

"""
High‑quality step‑function search via an iterative LP‑based improvement.

The algorithm:
1. Initialise a sequence with the truncated (1‑z)^‑1/2 coefficients
   (the “arcsine” construction) and rescale it so that its ℓ¹‑norm is
   √(2 n).  With this normalisation the evaluation metric returned by
   `evaluate_sequence` is exactly the maximal self‑convolution value M.

2. Repeatedly solve a linear programme:
      maximise   Σ b_j
      subject to  b_j ≥ 0
                  (f * b)_k ≤ M   for all convolution indices k
   where `f` is the current sequence and `M` its current max‑convolution.
   The LP finds a direction `b` that can increase the total mass without
   violating the current upper bound.

3. Rescale `b` to the required ℓ¹‑norm (√(2 n)) obtaining a new candidate
   `g`.  Mix the old and new sequences:
         f ← (1‑t) f + t g   with a step size `t` (starting at 0.5).
   The mix keeps the norm fixed and, for a sufficiently small `t`,
   strictly reduces the maximal convolution.

4. Iterate steps 2‑3 until no improvement is found or the time budget
   expires.  The best sequence over a variety of lengths is kept.

The whole search runs inside the supplied time budget (default 1000 s)
and returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
def arcsine_raw(n: int) -> np.ndarray:
    """Generate the unnormalised coefficients C_i = binom(2i,i)/4**i."""
    seq = np.empty(n, dtype=float)
    cur = 1.0  # C_0
    for i in range(n):
        seq[i] = cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    return seq


def arcsine_scaled(n: int) -> np.ndarray:
    """
    Return the arcsine coefficients scaled to have ℓ¹‑norm √(2 n).
    This is the conventional starting point for the LP‑based optimisation.
    """
    raw = arcsine_raw(n)
    target_norm = math.sqrt(2.0 * n)
    scale = target_norm / raw.sum()
    return raw * scale


# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n sparse matrix A such that
        (A @ b)_k = (f * b)[k]  (discrete convolution)
    for every k = 0 … 2n‑2.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # Column j contributes the vector f shifted by +j.
    for j in range(n):
        # rows j, j+1, …, j+n‑1  (n entries)
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        data.extend(f.tolist())

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_max_sum_under_conv(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme:
        maximise Σ b_j
        subject to b_j ≥ 0
                   (f * b)_k ≤ M   for all k
    Returns the optimal vector b (as a NumPy array) or None on failure.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)                     # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=float)   # upper bound for each convolution entry
    c = -np.ones(n, dtype=float)                 # minimise –sum ⇒ maximise sum

    # Using HiGHS (fast, sparse‑aware) solver
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
def improve_sequence(f: np.ndarray,
                     max_iters: int,
                     deadline: float,
                     t0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based improvement on a sequence `f`.
    Returns the improved sequence and its maximal convolution value.
    """
    n = f.shape[0]
    L = math.sqrt(2.0 * n)          # required ℓ¹‑norm
    t = t0
    best_f = f.copy()
    best_M = np.convolve(best_f, best_f).max()

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        b = lp_max_sum_under_conv(best_f, best_M)
        if b is None:
            # LP failed – stop the refinement
            break

        sum_b = b.sum()
        if sum_b <= L + 1e-12:
            # No possibility to increase the total mass; convergence
            break

        # Rescale to keep the ℓ¹‑norm fixed
        g = b * (L / sum_b)

        # Convex combination; ensures the norm stays exactly L
        f_new = (1.0 - t) * best_f + t * g
        f_new = np.maximum(f_new, 0.0)                 # protect against tiny negatives
        f_new = f_new * (L / f_new.sum())             # renormalise precisely

        M_new = np.convolve(f_new, f_new).max()

        if M_new < best_M - 1e-9:
            # Successful improvement
            print(f"[improve] iter {it:02d}: M {best_M:.6f} → {M_new:.6f}")
            best_f, best_M = f_new, M_new
            t = min(0.9, t * 1.05)                     # try a slightly larger step
        else:
            # Step was too aggressive – shrink it
            t *= 0.5
            if t < 1e-4:
                break

    return best_f, best_M

    """
