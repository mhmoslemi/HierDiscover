propose_candidate
-----------------

A two‑phase annealed projected gradient method.

Phase 1 (high temperature) uses a smooth soft‑max surrogate of the
convolution‑maximum.  The weights are
    w_k ∝ exp(( (a*a)[k] – max ) / μ )
with a temperature μ that is geometrically cooled.  The gradient of the
surrogate can be computed in O(n) by a single convolution, keeping the
iteration cheap.

Phase 2 (low temperature) automatically becomes a hard‑max descent
once μ is very small; the same formula then reproduces the sub‑gradient
used by the reference implementation but with a momentum term
(Adam‑style) and an adaptive step size.

Every iteration the iterate is Euclidean‑projected onto the simplex
{a≥0, Σa = √(2 n)}.  Random Gaussian “shakes’’ and occasional full
re‑starts (different random seeds) avoid being trapped in poor local
minima.  After the time budget a cheap block‑search (as in the

The returned list is a feasible step‑function; the external harness will

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int = 5) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] based on the `top_k` largest entries."""
    n = a.shape[0]
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    grad = np.zeros_like(a)
    i = np.arange(n)
    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad

    beta = 0.9                           # momentum decay
    m = np.zeros_like(a)                 # momentum vector

    # soft‑max temperature schedule
    mu = best_score                      # start with current max
    mu_decay = 0.9995                    # geometric cooling
    mu_min = 1e-6

    # restart handling
    max_restarts = 3
    restart_interval = (budget_s - 5) / max_restarts   # evenly spaced
    next_restart = time.time() + restart_interval
    restarts_done = 0

    it = 0
    no_improve = 0
    top_k_hard = 5                       # used when mu becomes tiny

    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- convolution and current score -----------------------
        conv = np.convolve(a, a)
        cur_max = conv.max()

        # keep global best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
            print(f"[{it:08d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # ----- gradient computation --------------------------------
        if mu > 0.01 * cur_max:          # high temperature -> smoothmax
            diff = conv - cur_max        # ≤ 0
            w = np.exp(diff / mu)        # stable: max entry gets weight 1
            w /= w.sum()                 # normalise to a probability vector
            # efficient gradient via a single convolution
            conv_wr = np.convolve(w, a[::-1])          # length 3n‑2
            sum_term = conv_wr[n - 1:n - 1 + n]       # ∑_k w_k a_{k-i}
            grad = 2.0 * sum_term
            # diagonal correction: subtract w_{2i}·a_i where defined
            double_idx = 2 * np.arange(n)
            mask = double_idx < len(conv)
            grad[mask] -= w[double_idx[mask]] * a[mask]
        else:                            # low temperature -> hard max (top‑k)
            grad = _gradient_topk(a, conv, top_k=top_k_hard)

        # ----- momentum (Adam‑style) -------------------------------
        m = beta * m + (1.0 - beta) * grad

        # ----- step size (decaying) -------------------------------
        eta = eta0 / (1.0 + decay_eta * it)

        # ----- gradient step + projection --------------------------
        a_new = a - eta * m
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- occasional random shake -----------------------------
        if it % 4000 == 0 and it > 0:
            a_new += np.random.randn(n) * target_sum * 1e-4
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0

        a = a_new

        # ----- temperature cooling ---------------------------------
        mu = max(mu * mu_decay, mu_min)

        # ----- handle restarts ------------------------------------
        if time.time() > next_restart and restarts_done < max_restarts:
            restarts_done += 1
            print(f"[{it:08d}] restart #{restarts_done}")
            a = np.random.rand(n)
            a = _project_to_simplex(a, target_sum)
            m.fill(0.0)
            mu = np.max(np.convolve(a, a))
            next_restart = time.time() + restart_interval

        it += 1

    # --------------------------------------------------------------
    # Cheap block‑search after the main annealing loop (as in the
    # reference implementation).  Very inexpensive but sometimes
    # discovers surprising patterns.
    block_trials = 300
    for _ in range(block_trials):
        nb = np.random.randint(2, 7)                       # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = np.max(np.convolve(cand, cand))
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved block score = {best_score:.6f}")

    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
