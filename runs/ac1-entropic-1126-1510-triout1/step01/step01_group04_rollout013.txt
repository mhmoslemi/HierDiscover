"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation proceeds in several fast stages:

1️⃣  **Parameter grid** – explore power‑law, exponential and a two‑segment
    power‑law family.  The grid is deliberately fine (≈30 k candidates) but
    uses a modest length (n ≈ 2000) so each evaluation is cheap.

2️⃣  **Random Dirichlet restarts** – draw many random probability vectors
    (different concentration parameters) and keep the best one.

3️⃣  **Piecewise random shapes** – generate random “break‑point’’ power‑law
    vectors (two exponents, a cut point and an offset).  This adds extra
    flexibility beyond a single power‑law.

4️⃣  **Simulated‑annealing hill climbing** – starting from the current best
    vector we repeatedly move a small amount of mass from one index to another.
    Moves that improve the score are always accepted; worsening moves are
    accepted with probability `exp(-Δ/T)` where the temperature `T` is
    linearly cooled to encourage escape from local minima.

All helpers are declared at top level (no closures) and the search respects
the 1000‑second budget.  The final vector is returned as a plain list of
Python floats, ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – they are pure and have no closures
# ----------------------------------------------------------------------


def _eval(seq: np.ndarray) -> float:
    """Wrapper around the provided `evaluate_sequence`."""
    # `evaluate_sequence` expects a list of Python floats.
    return evaluate_sequence(seq.tolist())


def _make_power_law(n: int, exponent: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+1+offset)^exponent, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vec = np.exp(-lam * i)
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


def _make_piecewise_power(
    n: int, exp1: float, exp2: float, cut: int, offset: float
) -> np.ndarray:
    """Two‑segment power‑law:
       a_i ∝ (i+1+offset)^exp1   for i < cut,
       a_i ∝ (i+1+offset)^exp2   for i ≥ cut.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = np.empty(n, dtype=np.float64)
    vec[:cut] = idx[:cut] ** exp1
    vec[cut:] = idx[cut:] ** exp2
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


def _anneal(seq: np.ndarray, score: float, deadline: float) -> (np.ndarray, float):
    """Simple simulated‑annealing on the simplex.
    Moves consist of transferring a random fraction of mass from i→j.
    """
    cur = seq.copy()
    cur_score = score
    best = cur.copy()
    best_score = cur_score
    n = len(cur)

    start = time.time()
    it = 0
    # Temperature schedule: linear from 1.0 down to 1e‑4
    T0 = 1.0
    Tf = 1e-4

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0.0:
            break
        frac = elapsed / total
        T = T0 * (1.0 - frac) + Tf * frac

        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] <= 0.0:
            continue

        # Transfer a random fraction (up to 10 % when T≈Tf, up to 30 % when T≈T0)
        max_frac = 0.10 + 0.20 * (T / T0)
        delta = cur[i] * np.random.rand() * max_frac
        nxt = cur.copy()
        nxt[i] -= delta
        nxt[j] += delta

        nxt_score = _eval(nxt)
        delta_score = nxt_score - cur_score
        accept = False
        if delta_score < 0.0:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < math.exp(-delta_score / max(T, 1e-12)):
                accept = True

        if accept:
            cur = nxt
            cur_score = nxt_score
            if nxt_score < best_score:
                best = nxt.copy()
                best_score = nxt_score

        it += 1
        if it % 5000 == 0:
            print(
                f"[anneal] it={it}  time={elapsed:.1f}s  T={T:.3e}  best={best_score:.6f}"
            )

    return best, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a low‑score step function.
    The function runs for at most ``budget_s`` seconds and returns a list of
    non‑negative heights (Python floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # Length – small enough for many evaluations, large enough for flexibility
    n = 2000

    # ------------------------------------------------------------------
    # 1) Exhaustive grid over families of simple analytic shapes
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ----- power‑law (single exponent + offset) -----------------------
    exponents = np.linspace(-0.90, -0.20, 141)   # step ≈0.005
    offsets = np.linspace(0.0, 10.0, 41)        # step 0.25
    for off in offsets:
        if time.time() >= deadline:
            break
        for a in exponents:
            cur = _make_power_law(n, a, off)
            sc = _eval(cur)
            if sc < best_score:
                best_score = sc
                best_seq = cur.copy()
        # occasional progress report
        print(
            f"[grid‑pl] offset={off:5.2f}  best_score={best_score:.6f}"
        )

    # ----- exponential decay -------------------------------------------
    lambdas = np.linspace(1e-4, 0.02, 200)      # 200 points
    for lam in lambdas:
        if time.time() >= deadline:
            break
        cur = _make_exponential(n, lam)
        sc = _eval(cur)
        if sc < best_score:
            best_score = sc
            best_seq = cur.copy()
    print(f"[grid‑exp] after exponentials best_score={best_score:.6f}")

    # ----- piecewise power‑law (two exponents, a cut point, offset) ----
    # Randomised search – we generate a few hundred candidates.
    pw_trials = 400
    for _ in range(pw_trials):
        if time.time() >= deadline:
            break
        cut = np.random.randint(int(0.1 * n), int(0.9 * n))
        exp1 = np.random.uniform(-0.90, -0.20)
        exp2 = np.random.uniform(-0.90, -0.20)
        off = np.random.uniform(0.0, 8.0)
        cur = _make_piecewise_power(n, exp1, exp2, cut, off)
        sc = _eval(cur)
        if sc < best_score:
            best_score = sc
            best_seq = cur.copy()
    print(f"[grid‑pw] after piecewise best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Random Dirichlet restarts (different concentration parameters)
    # ------------------------------------------------------------------
    dir_params = [0.1, 0.5, 1.0, 2.0, 5.0]  # very sparse → very dense
    dir_trials = 1200
    for _ in range(dir_trials):
        if time.time() >= deadline:
            break
        alpha = np.random.choice(dir_params)
        vec = np.random.gamma(alpha, 1.0, size=n)
        cur = vec / vec.sum()
        sc = _eval(cur)
        if sc < best_score:
            best_score = sc
            best_seq = cur.copy()
        if _ % 200 == 0 and _ > 0:
            print(f"[dirichlet] trial {_} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing refinement from the best candidate so far
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (never happens, but keep sanity)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _eval(best_seq)

    print(f"[start‑anneal] best_score before annealing = {best_score:.6f}")
    best_seq, best_score = _anneal(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # 4) Final sanity – ensure we return a plain Python list of floats
    # ------------------------------------------------------------------
    final_sequence = [float(v) for v in best_seq.tolist()]

    print(
        f"Returning sequence length {len(final_sequence)} with evaluation score {best_score:.6f}"
    )
    return final_sequence
