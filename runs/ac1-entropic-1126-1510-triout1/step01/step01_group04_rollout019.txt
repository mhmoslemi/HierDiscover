def _project_simplex(v):
    # ensure nonnegative after projection onto L1=1 simplex
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import time, math, numpy as np

def _eval(seq):
    return evaluate_sequence(seq.tolist())

def _make_power_law(n, exponent, offset=0.0):
    idx = np.arange(1, n+1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0:
        return np.full(n, 1.0/n)
    return vec / total

def _project_simplex(v):
    # Projects onto the probability simplex {x >=0, sum x =1}
    # Implements the O(n) algorithm from Duchi et al. (2008)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries zero or negative: return uniform distribution
        return np.full(n, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # ensure sum exactly 1 (may be off by floating error)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0/n)
    w /= w_sum
    return w

def propose_candidate(seed=0, budget_s=1000):
    """
    Optimises a non‑negative step function to minimise
    2*n*max(conv(a,a))/(sum(a)**2) using a smooth‑max surrogate and
    Adam‑style projected gradient descent on the simplex.
    A power‑law seed is selected by a brief grid search.
    The routine respects the 1000 s time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Choose the length of the candidate. 15 k–20 k gives a good trade‑off.
    n = 15000

    # ------------------------------------------------------------------
    # 1) Quick grid search for a decent power‑law seed.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    exponents = np.linspace(-0.65, -0.40, 41)   # 41 candidates
    offsets = [0.0, 0.2, 0.5]
    for off in offsets:
        for exp in exponents:
            cand = _make_power_law(n, exp, offset=off)
            # use cheap evaluation based on our own convolution (same scaling)
            A_fft = np.fft.rfft(cand, n=1 << ((2*n-1).bit_length()))
            conv = np.fft.irfft(A_fft * A_fft, n=A_fft.shape[0]*2-2)[:2*n-1]
            score = 2 * n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
    print(f"[grid] initial best score {best_score:.6f}")

    a = best_seq

    # ------------------------------------------------------------------
    # 2) Smooth‑max Adam optimisation on the simplex.
    # ------------------------------------------------------------------
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    lr = 0.3          # base learning rate for Adam
    # schedule for the smooth‑max temperature β
    beta = 1.0
    beta_target = 200.0
    beta_factor = 1.07   # multiplicative increase per outer loop

    # pre‑compute FFT length (next power of two >= 2n‑1)
    fft_len = 1 << ((2*n-1).bit_length())

    while time.time() < deadline:
        # ----- convolution of current a with itself (FFT) -----
        A_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A_fft * A_fft, n=fft_len)[:2*n-1]

        maxc = conv.max()
        # actual metric (sum(a)=1)
        actual_score = 2 * n * maxc
        if actual_score < best_score:
            best_score = actual_score
            best_seq = a.copy()
            print(f"[improve] new best {best_score:.6f} (beta={beta:.2f})")

        # ----- smooth‑max (log‑sum‑exp) -----
        # stable computation
        shift = maxc
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        # float smooth_max = shift + log(sum_exp)/beta  # not needed for gradient
        # ----- weight vector (softmax) -----
        w = exp_beta / sum_exp    # length 2n‑1, sums to 1

        # ----- gradient of smooth‑max w.r.t. a -----
        # gradient_i = 2 * sum_j a_j * w_{i+j}
        # we compute via convolution of reversed a with w.
        a_rev = a[::-1]
        A_rev_fft = np.fft.rfft(a_rev, n=fft_len)
        W_fft = np.fft.rfft(w, n=fft_len)
        cross = np.fft.irfft(A_rev_fft * W_fft, n=fft_len)   # length fft_len
        grad = 2.0 * cross[(n-1):(n-1+n)]

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # take a gradient‑step and project back to simplex
        a_new = a - step
        a = _project_simplex(a_new)

        # ----- increase the temperature β slowly -----
        if beta < beta_target:
            beta = min(beta_target, beta * beta_factor)

        # optional random jitter to escape plateaus (tiny)
        if np.random.rand() < 0.001:
            jitter = np.random.randn(n) * 1e-4
            a = _project_simplex(a + jitter)

        # early break if time about to expire
        if time.time() > deadline - 0.1:
            break

    # final evaluation with the official function (for safety)
    final_score = _eval(best_seq)
    print(f"Returning sequence length {len(best_seq)} with final official score {final_score:.6f}")
    return [float(v) for v in best_seq.tolist()]

def compute_metric(a):
    n = len(a)
    # Compute convolution via FFT
    m = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:2*n-1]
    max_conv = conv.max()
    return 2.0 * n * max_conv
```

    m = fft_len
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:2*n-1]
    maxc = conv.max()
    # stable softmax
    shift = maxc
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=m)
    W = np.fft.rfft(w, n=m)
    cross = np.fft.irfft(A_rev * W, n=m)
    grad = 2.0 * cross[(n-1):(n-1+n)]
    return conv, maxc, grad

"""
Optimise a non‑negative step function for the evaluation

    2 * n * max(conv(a, a)) / (sum(a)**2)

The score is scale‑invariant, therefore we keep the vector on the
probability simplex (non‑negative, sum = 1).  The algorithm has three
stages:

1.  **Coarse grid search** over a power‑law exponent α (a_i ∝ (i+off)^α)
    and an offset `off`.  The metric is computed with an FFT‑based
    convolution (O(n log n)) – this allows a dense scan of a few thousand
    candidates quickly.

2.  **Smooth‑max Adam optimisation**.  The non‑smooth max is replaced by
    the smooth approximation
        L_β(a) = (1/β) * log Σ_k exp(β·(a*a)_k)
    whose gradient is a weighted sum of the gradients of the individual
    convolution entries.  We use Adam with a projection onto the simplex
    after each step.  The temperature β is increased gradually so that
    the optimisation first sees a smooth landscape and finally sharpens
    to the true max.

3.  **Optional random jitter** to escape tiny plateaus.

All work respects the time‑budget (default 1000 s) and never calls the
slow `evaluate_sequence` except for the final verification.
"""


import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _evaluate(seq):
    """Wrapper around the provided evaluator (expects a list)."""
    return evaluate_sequence(seq.tolist())


def _power_law_sequence(n, exponent, offset=0.0):
    """Generate a normalised power‑law vector: (i+offset+1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vec / total


def _project_onto_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex:
        {x ≥ 0 , Σ x = 1}
    Implements the O(n) algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    # enforce exact sum = 1 (numeric safety)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w_sum
    return w


def _conv_fft(a):
    """Self‑convolution via real FFT (fast, O(n log n))."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


def _smooth_max_and_grad(a, beta, fft_len):
    """
    Compute (max conv, smooth‑max value, gradient of the smooth‑max)
    where the smooth‑max is the log‑sum‑exp surrogate.
    """
    n = a.shape[0]

    # convolution a*a
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]

    # smooth‑max via log‑sum‑exp (stable)
    max_c = conv.max()
    shifted = conv - max_c                     # for numerical stability
    exp_beta = np.exp(beta * shifted)
    sum_exp = exp_beta.sum()
    # weights = softmax(beta * conv)
    w = exp_beta / sum_exp

    # gradient of smooth‑max:   ∇ = 2 * (a_rev  *  w)   (restricted to n entries)
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    W = np.fft.rfft(w, n=fft_len)
    cross = np.fft.irfft(A_rev * W, n=fft_len)
    grad = 2.0 * cross[(n - 1) : (n - 1 + n)]   # slice the relevant part
    return conv, max_c, grad, max_c * 2 * n  # last term is the actual metric


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed=0, budget_s=1000):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    target score 1.503.  The routine works within the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ----- problem size -------------------------------------------------
    n = 8000                     # a few thousand entries give a good trade‑off

    # ----- 1) coarse grid over power‑law parameters --------------------
    best_seq = None
    best_score = float("inf")

    # ranges are deliberately wide – we explore exponents close to –1 as well
    exponents = np.linspace(-0.90, -0.20, 141)          # step ≈ 0.005
    offsets   = np.linspace(0.0, 5.0, 51)              # step ≈ 0.1

    # pre‑computed FFT length for the chosen n
    fft_len = 1 << ((2 * n - 1).bit_length())

    print("[grid] start scanning {} × {} candidates".format(len(exponents),
                                                             len(offsets)))
    for off in offsets:
        for exp in exponents:
            cand = _power_law_sequence(n, exp, offset=off)
            # fast metric via FFT (no scaling – the vector already sums to 1)
            conv = _conv_fft(cand)
            metric = 2.0 * n * conv.max()
            if metric < best_score:
                best_score = metric
                best_seq = cand.copy()
        # occasional early‑exit if time is thin
        if time.time() > deadline - 5.0:
            break

    print("[grid] best metric so far = {:.6f}".format(best_score))

    # ----- 2) smooth‑max Adam optimisation on the simplex ---------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.4                     # base learning rate
    m = np.zeros_like(best_seq)
    v = np.zeros_like(best_seq)
    t = 0

    # temperature for the smooth‑max (start low → smooth landscape)
    beta = 5.0
    beta_target = 200.0
    beta_factor = 1.07           # geometric increase per outer loop

    cur = best_seq.copy()
    cur_score = best_score

    # keep a copy of the best vector seen during optimisation
    best_vec = cur.copy()
    best_metric = cur_score

    it = 0
    max_iters = 50000
    while time.time() < deadline and it < max_iters:
        conv, max_c, grad, metric = _smooth_max_and_grad(cur, beta, fft_len)

        # keep track of the best objective (true max‑convolution metric)
        if metric < best_metric:
            best_metric = metric
            best_vec = cur.copy()
            print("[adam] new best metric = {:.6f}  (iter {})".format(
                best_metric, it))

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        nxt = cur - step
        nxt = _project_onto_simplex(nxt)

        # accept the step (always – Adam is deterministic)
        cur = nxt

        # increase temperature slowly so that we focus more on the true max
        if beta < beta_target:
            beta = min(beta_target, beta * beta_factor)

        # tiny random jitter every few hundred iterations (helps escape flat spots)
        if (it & 511) == 0 and np.random.rand() < 0.25:
            jitter = np.random.randn(n) * 1e-4
            cur = _project_onto_simplex(cur + jitter)

        it += 1
        if it % 5000 == 0:
            print("[adam] iter {} – metric {:.6f}, beta {:.2f}".format(
                it, metric, beta))

    print("[adam] finished after {} iterations, best metric = {:.6f}"
          .format(it, best_metric))

    # ----- 3) final verification with the official evaluator -----------
    final_seq = best_vec
    final_score = _evaluate(final_seq.tolist())
    print("Returning sequence length {} with official score {:.6f}"
          .format(len(final_seq), final_score))

    # convert to plain Python floats as required by the evaluator
    return [float(v) for v in final_seq.tolist()]
