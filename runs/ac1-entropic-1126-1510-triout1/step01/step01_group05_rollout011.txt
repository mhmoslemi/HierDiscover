Pseudo-code for projection:

def project_onto_simplex(v):
    """Project onto simplex {x>=0, sum x = 1}."""
    # from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def gradient_descent(a0, n, deadline):

    a = a0.copy()
    best_local_score = float('inf')
    best_local_vec = a.copy()
    it = 0
    beta = 30.0  # smoothing parameter
    # We'll use decreasing learning rate.
    while time.time() < deadline and it < 50000:  # limit iterations
        conv = np.convolve(a, a)  # length 2n-1
        maxB = conv.max()
        score = 2.0 * n * maxB
        if score < best_local_score:
            best_local_score = score
            best_local_vec = a.copy()
        # gradient via exponential weighting
        shifted = conv - maxB
        # compute exp weights
        # to avoid overflow, clip
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        grad = compute_subgradient(w, a)  # using weighted mask (nonbinary)
        # step size: tune
        lr = 0.5 / (1 + it * 0.001)  # start 0.5, decay
        a = a - lr * grad
        # project onto simplex
        a = project_onto_simplex(a)
        # occasional random perturbation
        if it % 1000 == 0:
            # add small noise to escape
            noise = np.random.randn(n) * 1e-4
            a = a + noise
            a = project_onto_simplex(a)
        it += 1

    return best_local_vec, best_local_score

"""
Propose a low‑value sequence for the evaluation function.

Algorithm
---------
1. Choose a length n (use the length of the known solution if available).
2. Initialise a candidate by a power‑law distribution a_i ∝ (i+1)^{-p}
   with the exponent p found by a very cheap golden‑section search.
3. Refine the candidate with a projected (simplex) gradient descent that
   minimises a smooth approximation of  max_k (a*a)[k].
   * The smooth approximation uses a soft‑max (exponential) weighting of the
     convolution entries.
   * The gradient of the weighted sum can be obtained by a single
     correlation (`np.correlate`).
   * After each step we clip to non‑negative values and renormalise to keep
     Σ a_i = 1 (projection onto the probability simplex).
   * The learning rate decays over the iterations.
4. Several random restarts (including the previous best solution) are run
   until the time budget (≈1000 s) is exhausted.  The best sequence seen is
   returned.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper utilities
# --------------------------------------------------------------------------- #

def power_law_sequence(n: int, p: float) -> np.ndarray:
    """Decreasing power‑law a_i ∝ (i+1)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def evaluate_vec(vec: np.ndarray) -> float:
    """Wrapper that calls the external evaluate_sequence on a NumPy array."""
    # the competition framework guarantees that evaluate_sequence exists
    return evaluate_sequence(vec.tolist())

def weighted_gradient(mask: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k mask_k·(a*a)[k] w.r.t. a (sub‑gradient of the max when mask
    is a soft‑max weight vector).  The factor 2 accounts for the two symmetric
    terms in the convolution derivative.
    """
    # correlation a ⋆ mask (full) gives Σ_i a_i·mask_{i+j}
    corr = np.correlate(a, mask, mode="full")
    offset = mask.size - 1          # index that corresponds to shift j=0
    grad = 2.0 * corr[offset: offset + a.size]
    return grad

def gradient_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    beta: float = 20.0,
    lr0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent on the smoothed max‑convolution.
    Returns the best vector found (still normalised to sum 1) and its score.
    """
    a = a0.copy()
    n = a.size
    best_vec = a.copy()
    # internal objective (sum is 1, so score = 2·n·max(conv))
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max()

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # convolution and soft‑max weights
        conv = np.convolve(a, a)               # length 2n‑1
        maxB = conv.max()
        # exponential weighting (more mass on the current maximum)
        w = np.exp(beta * (conv - maxB))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient and step
        grad = weighted_gradient(w, a)
        # decaying learning rate
        lr = lr0 / (1.0 + it * 0.001)
        a = a - lr * grad

        # projection onto the simplex (clip+renorm is enough for tiny steps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # keep the best local solution
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_vec = a.copy()
            # occasional progress output
            print(f"[gd] iter {it:5d} – score {best_score:.6f}")

        # small random perturbation every few thousand steps to escape plateaus
        if (it + 1) % 2000 == 0:
            noise = np.random.randn(n) * 1e-4
            a = a + noise
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum

    return best_vec, best_score

def golden_section_search(
    n: int,
    left: float,
    right: float,
    max_iters: int,
    deadline: float,
) -> tuple[float, float, np.ndarray]:
    """
    1‑D optimisation of the exponent p in the power‑law family.
    Returns (best_p, best_score, best_vector).
    """
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    a = left
    b = right
    # interior points
    p1 = a + (1 - phi) * (b - a)
    p2 = a + phi * (b - a)

    v1 = evaluate_vec(power_law_sequence(n, p1))
    v2 = evaluate_vec(power_law_sequence(n, p2))

    best_score = min(v1, v2)
    best_p = p1 if v1 <= v2 else p2
    best_vec = power_law_sequence(n, best_p)

    it = 0
    while it < max_iters and time.time() < deadline:
        if v1 < v2:
            b = p2
            p2 = p1
            v2 = v1
            p1 = a + (1 - phi) * (b - a)
            v1 = evaluate_vec(power_law_sequence(n, p1))
            if v1 < best_score:
                best_score = v1
                best_p = p1
                best_vec = power_law_sequence(n, best_p)
        else:
            a = p1
            p1 = p2
            v1 = v2
            p2 = a + phi * (b - a)
            v2 = evaluate_vec(power_law_sequence(n, p2))
            if v2 < best_score:
                best_score = v2
                best_p = p2
                best_vec = power_law_sequence(n, best_p)
        it += 1
    return best_p, best_score, best_vec

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)

    # --------------------------------------------------------------- #

    try:

    print(f"[info] working length n = {n}")

    # --------------------------------------------------------------- #

    # --------------------------------------------------------------- #
    best_overall_vec = best_vec.copy()
    best_overall_score = best_score

    # a few starting points: power‑law, uniform, and the previous best if any
    starters = [

    # include the existing global solution if it exists and has the same length
    try:

    except Exception:
        pass

    # number of gradient steps per restart (adjusted to the remaining time)
    max_iters_per_run = 5000

    while time.time() < deadline:
        # pick a starter; add a small random perturbation to diversify
        init = starters[np.random.randint(len(starters))].copy()
        init += np.random.randn(n) * 1e-3
        init = np.clip(init, 0.0, None)
        if init.sum() == 0.0:
            init = np.full(n, 1.0 / n)
        else:

        # run gradient descent for the allocated time slice
        run_deadline = min(deadline, time.time() + (budget_s * 0.15))
        vec, score = gradient_descent(
            init,
            max_iters=max_iters_per_run,
            deadline=run_deadline,
            beta=30.0,
            lr0=0.5,
        )

        if score < best_overall_score:
            best_overall_score = score
            best_overall_vec = vec.copy()
            print(f"[update] new best overall score = {best_overall_score:.6f}")

        # if we are very close to the time limit, break
        if time.time() + 1.0 >= deadline:
            break

    # --------------------------------------------------------------- #
    # Final evaluation via the official evaluate_sequence (just for logging)
    # --------------------------------------------------------------- #
    final_score = evaluate_vec(best_overall_vec)
    print(f"[final] best score after full optimisation = {final_score:.6f}")

    # The evaluate_sequence function will clamp negatives / large values,
    # but we already guarantee the vector respects the constraints.
    return best_overall_vec.tolist()
