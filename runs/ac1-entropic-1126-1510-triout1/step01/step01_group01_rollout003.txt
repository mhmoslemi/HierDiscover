def _gradient_of_max_correct(a, top_k=5):
    n = len(a)
    conv = np.convolve(a, a)
    max_val = conv.max()
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:

        i_end = min(k, n-1)

        i_vals = np.arange(i_start, i_end+1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
        # No diagonal correction needed
    return grad, max_val

"""
Improved search for a low‑evaluation step‑function.

The algorithm builds on the projected‑gradient descent from the previous
submission but fixes the gradient of the max‑convolution (the old code
undercounted the diagonal contribution), enlarges the active set of
binding convolution entries, and adds several diversification tricks:

*   a correct sub‑gradient: for each active convolution index k the
    derivative is 2·a_{k‑i} for all i with a valid partner.
*   adaptive active‑set: all indices whose convolution value lies
    within a small tolerance of the current maximum are used (instead of
    a fixed small ``top_k``).
*   multiple random restarts with block‑wise initialisation.
*   occasional random perturbations and a light simulated‑annealing
    phase to escape plateaus.
*   Euclidean projection onto the simplex {x ≥ 0, Σx = Σ_target}
    after every update (the objective is scale‑invariant, therefore a
    fixed ℓ¹‑norm is imposed).

The search runs for the allotted time (≈ 1000 s) and returns the best
sequence found, which is a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project vector v onto the simplex of total mass z."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries become zero
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _gradient_of_max(a: np.ndarray, rel_tol: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of   max_k (a*a)[k]   and the corresponding maximum value.

    All convolution indices whose value is within ``rel_tol`` of the
    current maximum are treated as active constraints.  The gradient
    contribution of an active index k is
        grad_i += 2 * a_{k-i}   for every i with a valid partner.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()

    # active set: all indices whose convolution value is close to max
    if rel_tol <= 0:
        active = np.array([np.argmax(conv)], dtype=int)
    else:
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

    grad = np.zeros_like(a)

    # for each active k add the shifted copy of a (scaled by 2)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += 2.0 * a[j]

    return grad, max_val

    Parameters

    budget_s : float
        Approximate time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered (non‑negative, arbitrary length).
    """

    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose the length of the vector.
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]

    except Exception:
        n = 3000
        have_ref = False

    # The sum is irrelevant for the objective (scale‑invariant);
    # we fix it to the conventional √(2 n).
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Projection onto the simplex is used after every update.
    # ------------------------------------------------------------------
    # Keep the globally best solution.
    best_a = None
    best_score = float("inf")

    # Helper: compute the (scale‑invariant) score from a vector a.
    def _score(a_vec: np.ndarray) -> float:
        conv_max = np.convolve(a_vec, a_vec).max()
        return 2.0 * n * conv_max / (target_sum ** 2)

    # ------------------------------------------------------------------
    # 1) Initialise a few starting points (reference + random block vectors)
    start_vectors = []

    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, target_sum)
        start_vectors.append(a_ref)

    # generate a handful of block‑wise random vectors
    for _ in range(6):
        nb = np.random.randint(2, 12)            # number of blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)

    # ------------------------------------------------------------------

    #    It is called repeatedly on different initial vectors.

        a = initial_a.copy()

        for it in range(max_iters):

            new_max = np.convolve(a_new, a_new).max()
            new_score = 2.0 * n * new_max / (target_sum ** 2)

            # accept the step only if it does not increase the max‑conv.
            # (this simple line‑search stabilises the descent)
            if new_max <= cur_max:

                cur_max = new_max
            else:

            # record best local solution
            if new_score < best_local_score:

                best_local_a = a.copy()

            # occasional random perturbation to escape plateaus

                a = _project_to_simplex(a + noise, target_sum)

            if report_every and it % report_every == 0:
                pass   # silent – could print if desired

        return best_local_a, best_local_score

    # ------------------------------------------------------------------
    # 3) Main search loop – multiple restarts with varying hyper‑parameters.
    restart_counter = 0
    while time.time() < deadline:
        # pick an initial vector: either a fresh random block or a perturbed

            init = _project_to_simplex(init, target_sum)

        # hyper‑parameters for this run (slightly jittered)
        eta0 = target_sum * (0.3 + 0.3 * np.random.rand()) / n   # base step size
        decay = 5e-5 + 1e-4 * np.random.rand()                  # decay rate
        rel_tol = 0.02 + 0.03 * np.random.rand()               # active‑set tolerance
        max_iters = 6000 + int(4000 * np.random.rand())

        # run the PGD optimisation
        cand_a, cand_score = _run_pg(init,
                                     max_iters=max_iters,
                                     eta0=eta0,
                                     decay=decay,
                                     rel_tol=rel_tol,
                                     report_every=0)

        # update global best
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[restart {restart_counter:03d}] new global best = {best_score:.6f}")

        restart_counter += 1

        # early exit if we already beat the target
        if best_score <= 1.503:
            print("Target achieved – stopping early.")
            break

    # ------------------------------------------------------------------
    # 4) Final local refinement (light simulated‑annealing) if time permits.
    remaining = deadline - time.time()
    if best_a is not None and remaining > 5.0:
        a = best_a.copy()
        cur_score = best_score
        temperature = 0.01
        while time.time() < deadline:
            # propose moving a small amount of mass between two random positions
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue

            if a[i] - delta < 0:

            a_new = a.copy()
            a_new[i] -= delta

                cur_score = new_score
                if new_score < best_score:

                    print(f"[SA] improved to {best_score:.6f}")

            # cooling schedule
            temperature *= 0.99997
            if temperature < 1e-5:

    # 5) Return the best sequence found.
    print(f"[final] best score = {best_score:.6f}")

        best_a = np.full(n, 1.0 / n)
        best_a = _project_to_simplex(best_a, target_sum)

    return best_a.tolist()
