"""Propose a low‑evaluation non‑negative step‑function.

The algorithm explores a simple three‑parameter family

    a_i = (i + 1 + shift)^{-p} · exp(-γ·i/(n‑1))

The exponent `p` controls the power‑law decay, `shift` moves the
singularity away from the origin (reducing the large early convolution
terms) and `γ` adds a mild exponential taper.  For every combination the
objective `evaluate_sequence` is computed, the best sequence is kept,
and a quick local random‐perturbation phase refines it.
The search respects the 1000 s time budget and prints progress.
"""

import time
import numpy as np

# evaluate_sequence is assumed to be already imported in the execution environment.


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimizes
    ``evaluate_sequence``.  The search explores exponents p∈[0.55,0.85],
    integer shifts shift∈[0,15] and, for the best (p,shift) pair,
    a fine sweep of the exponential‑taper parameter γ.
    A final stochastic refinement tries small random perturbations.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # keep a safety margin

    # ------------------------------------------------------------------
    # Settings
    # ------------------------------------------------------------------
    n = 4000                                    # length of the step function
    idx = np.arange(1, n + 1, dtype=np.float64) # 1 … n  (used for the basic power‑law)

    # grid for the exponent p
    p_grid = np.linspace(0.55, 0.85, 31)        # step ≈0.01
    shift_grid = np.arange(0, 16, dtype=int)   # shift = 0 … 15
    best_score = float("inf")
    best_seq = None
    best_params = None

    # ------------------------------------------------------------------
    # Coarse search over (p, shift) – no exponential taper yet
    # ------------------------------------------------------------------
    for p in p_grid:
        # pre‑compute the power‑law part for the current exponent
        # (i + shift) is handled inside the inner loop
        for shift in shift_grid:
            # build the candidate
            seq = (idx + shift) ** (-p)

            # evaluate
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_params = (p, shift, 0.0)

            # time check
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Fine sweep of the exponential taper γ around the best (p, shift)
    # ------------------------------------------------------------------
    if best_params is not None and time.time() < deadline:
        p_best, shift_best, _ = best_params
        base_seq = (idx + shift_best) ** (-p_best)

        # first a coarse γ sweep
        for gamma in np.linspace(0.0, 2.0, 41):
            seq = base_seq * np.exp(-gamma * (idx - 1) / (n - 1))
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_params = (p_best, shift_best, gamma)

            if time.time() > deadline:
                break

        # then a finer γ sweep centred on the current best γ
        if time.time() < deadline:
            lo = max(0.0, best_params[2] - 0.25)
            hi = best_params[2] + 0.25
            for gamma in np.linspace(lo, hi, 41):
                seq = base_seq * np.exp(-gamma * (idx - 1) / (n - 1))
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (p_best, shift_best, gamma)

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Stochastic refinement: small random perturbations of the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # use a relative perturbation of up to ±2 %
        for _ in range(200):
            if time.time() > deadline:
                break
            noise = 0.02 * np.random.randn(n)
            pert = best_seq * (1.0 + noise)
            pert = np.maximum(pert, 0.0)               # keep non‑negative
            # evaluation
            score = evaluate_sequence(pert.tolist())
            if score < best_score:
                best_score = score
                best_seq = pert.copy()
                # keep track of the parameters (they are no longer a simple
                # (p,shift,gamma) tuple after random perturbation)
                best_params = ("random‑refine",)

    # ------------------------------------------------------------------
    # Reporting
    # ------------------------------------------------------------------
    if best_params is not None:
        if isinstance(best_params, tuple) and len(best_params) == 3:
            p, shift, gamma = best_params
            print(
                f"Best parameters → p={p:.5f}, shift={shift}, gamma={gamma:.5f}"
            )
        else:
            print(f"Best parameters after refinement: {best_params}")

    print(f"Best objective = {best_score:.6f}, length = {len(best_seq)}")
    # Convert to plain Python list before returning (evaluate_sequence expects a list)
    return best_seq.tolist()
