def solve_lp_for_direction(f, M, upper_bound=1000):
    n = f.shape[0]
    from scipy.optimize import linprog
    from scipy.sparse import diags

    # Build sparse A_ub
    offsets = np.arange(n)  # [0,1,...,n-1]
    A_ub = diags(f, offsets, shape=(2*n-1, n), format='csc')
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, upper_bound) for _ in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x
```

"""
Propose a low‑score step‑function.

The evaluation score for a sequence a with ℓ¹‑norm √(2 n) is exactly the
maximum entry of its self‑convolution:

    score = max_k (a * a)[k] .

Thus the problem reduces to minimising the ℓ∞‑norm of the auto‑convolution
subject to a ≥ 0 and Σ a = √(2 n).

The algorithm combines two complementary ideas:

1. **Projected sub‑gradient descent** on the max‑convolution.
   The sub‑gradient is formed from *all* (within a tiny tolerance) active
   convolution peaks, not just a single index.  After a gradient step the
   vector is projected back onto the simplex {a ≥ 0, Σa = S}
   using a Euclidean projection.  A decaying step‑size and occasional
   random jitter help to escape plateaus.

2. **Linear‑programming refinement**.
   For the current vector f we solve the LP

        maximise   Σ g_i
        subject to (f * g)[k] ≤ M_f   (∀k)
                   g_i ≥ 0

   where M_f is the current maximal convolution value.
   The resulting feasible g is scaled to have the correct ℓ¹‑norm and,
   if its own convolution maximum is smaller, it replaces f.
   Otherwise a short line‑search over convex combinations (1‑t)·f + t·g
   is performed.  LP solves are performed with a sparse Toeplitz
   constraint matrix and the Highs solver (SciPy).

The two phases are interleaved: a quick sub‑gradient phase is run for a
fraction of the total time budget, after which LP refinements are applied
until the deadline.  The best sequence seen is returned.

All numbers are kept non‑negative; the outer harness will clamp values to
[0, 1000] and compute the final score.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Sub‑gradient of the max‑convolution (averaged over all active peaks)
def _subgradient_maxconv(a: np.ndarray, eps: float = 1e-8):
    """Return (sub‑gradient, current max) for max_k (a*a)[k]."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()
    # all indices whose value is within eps of the maximum are regarded as active
    active = np.where(conv >= M - eps)[0]
    grad = np.zeros_like(a)
    i = np.arange(n)

    for k in active:
        j = k - i                      # partner index for each i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]

        # diagonal correction (the term a_i*a_i appears only once)
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    if active.size > 0:
        grad /= float(active.size)     # average over the active set
    return grad, M


# --------------------------------------------------------------
# Solve the LP: maximise Σ g_i   s.t. (f * g)[k] ≤ M ,  g_i ≥ 0 ,  g_i ≤ ub
def _solve_lp_max_sum(f: np.ndarray, M: float, ub: float = 1000.0):
    """
    Return a feasible vector g (not normalised) that maximises its ℓ¹‑norm
    while respecting the convolution bound (f * g)[k] ≤ M.
    If the LP fails, return None.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import diags
    except Exception:      # SciPy not available – signal failure
        return None

    n = f.shape[0]
    # Toeplitz convolution constraint matrix (sparse)
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A_ub = diags(f, offsets, shape=(2 * n - 1, n), format='csc')
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_i   → minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, ub) for _ in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return np.asarray(res.x, dtype=np.float64)

    """

    deadline = time.time() + budget_s - 0.5   # tiny safety margin

        have_ref = False

    # Target ℓ¹‑norm (the value used by the evaluation function)
    S = math.sqrt(2.0 * n)

    # ----------------------------------------------------------

    if have_ref and np.random.rand() < 0.5:

    else:

    conv = np.convolve(a, a)

    # ----------------------------------------------------------
    # PHASE 1 – projected sub‑gradient descent (≈ 30 % of the time budget)
    phase1_end = time.time() + 0.30 * budget_s
    it = 0
    no_improve = 0
    eta0 = S * 0.4 / n          # base step size (≈ 0.4·average entry)

    # ----------------------------------------------------------
    # PHASE 2 – linear‑programming refinement (uses the remaining budget)
    # We iterate until the time budget expires or the LP ceases to improve.
    while time.time() < deadline:
        cur_M = np.convolve(a, a).max()

        g_raw = _solve_lp_max_sum(a, cur_M, ub=1000.0)

            a = a + np.random.randn(n) * S * 2e-4

        # Scale to the correct ℓ¹‑norm
        sum_g = g_raw.sum()
        if sum_g <= 1e-12:

            break
        g = g_raw * (S / sum_g)
        # Clip just in case the scaling pushes values beyond the hard cap of evaluate_sequence
        np.clip(g, 0.0, 1000.0, out=g)

        M_g = np.convolve(g, g).max()

        improved = False

        # 1) direct replacement if g is strictly better
        if M_g < cur_M - 1e-7:
            a = g
            cur_M = M_g
            improved = True
            print(f"[lp] direct improvement: max‑conv = {cur_M:.6f}")

        # 2) try a short line‑search over convex combinations
        if not improved:
            for t in (0.02, 0.05, 0.1, 0.15, 0.2):
                a_mix = (1.0 - t) * a + t * g
                M_mix = np.convolve(a_mix, a_mix).max()
                if M_mix < cur_M - 1e-7:
                    a = a_mix
                    cur_M = M_mix
                    improved = True
                    print(f"[lp] mix t={t:.2f} → max‑conv = {cur_M:.6f}")

        # 3) fallback: tiny random perturbation if LP gave nothing

            a = _project_to_simplex(a, S)

        if cur_M < best_M - 1e-9:

            best_a = a.copy()
