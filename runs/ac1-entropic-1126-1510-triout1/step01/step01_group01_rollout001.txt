"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1. **Initialisation** – Use the provided reference `height_sequence_1` if
   available, otherwise start from a random vector.  Project onto the
   simplex with ℓ¹‑norm `√(2·n)` (the evaluation then equals the maximal
   entry of the self‑convolution).

2. **Momentum projected gradient descent** – The sub‑gradient of the
   maximum convolution entry (or a few top entries) is computed and a
   Nesterov‑style momentum step is taken.  After each step the vector is
   projected back onto the simplex.  Small low‑frequency perturbations
   are injected occasionally to escape plateaus.

3. **Simulated annealing** – Starting from the best vector found,
   repeatedly move a small fraction of the mass from one coordinate to
   another.  Moves that increase the objective are accepted with the
   Metropolis probability `exp(-Δ / T)`.  The temperature decays slowly.

4. **Random block constructions** – A cheap exhaustive search over
   piece‑wise‑constant patterns (2‑6 blocks) is performed; any improvement
   is kept.

All steps respect the time budget (≈ 1000 s).  The best sequence found
is returned as a plain Python list; the external harness will evaluate it
with `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _gradient_of_max(a: np.ndarray, top_k: int = 3):
    """
    Sub‑gradient of the largest (or a few largest) entries of the
    self‑convolution a*a.

    Returns
    -------
    grad : np.ndarray
        Sub‑gradient (same shape as a).
    max_val : float
        The (largest) convolution value.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()

    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        # indices of the top_k entries (unordered)
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(a)
    i = np.arange(n)

    for k in idxs:
        j = k - i                         # partner index for each i
        mask = (j >= 0) & (j < n)         # valid pairs
        grad[mask] += 2.0 * a[j[mask]]    # derivative 2*a_j
        diag = mask & (i == j)            # diagonal terms i = j = k/2
        grad[diag] -= a[i[diag]]          # correction when i=j

    return grad, max_val
