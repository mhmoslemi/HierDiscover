"""
Propose a low‑score step‑function.

The evaluation metric 2·n·max(conv(a,a)) / (sum(a)**2) is scale‑invariant,
so we only need a shape of a probability distribution.
Empirically the optimal shape is close to a decreasing power‑law
      a_i ∝ (i+offset)^α  with α≈−0.55.
The algorithm consists of three stages

1.  A dense grid search over (α, offset) to obtain a good initial shape.
2.  A simple derivative‑free local optimisation that perturbs (α, offset)
    while staying inside the valid domain.
3.  A stochastic hill‑climbing on the actual entries of the vector
    (mass transfers and swaps) that respects the simplex constraint.

All stages respect the 1000 s wall‑clock budget and use only
numpy (and the supplied ``evaluate_sequence``).  The final sequence is
returned as a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (the evaluator is assumed to be already imported)
# ----------------------------------------------------------------------
def _eval(seq: np.ndarray) -> float:
    """Wrap the external ``evaluate_sequence`` – it expects a plain list."""
    return evaluate_sequence(seq.tolist())


def _make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """
    Return a normalised power‑law vector:
        a_i ∝ (i + 1 + offset) ** exponent ,   i = 0 … n‑1 .
    The returned vector sums to 1.0 (L1‑normalised).
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = np.power(idx, exponent)
    total = vec.sum()
    if total == 0.0:
        # degenerate – fall back to the uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vec / total


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative step function that minimises the provided
    ``evaluate_sequence`` metric.  The function runs for at most ``budget_s``
    seconds and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Grid search over exponent and offset (coarse but exhaustive)
    # --------------------------------------------------------------
    n = 5000                     # length of the sequence used for the search
    best_seq = None
    best_score = float("inf")
    best_alpha = None
    best_offset = None

    # exponent grid – the known good region is roughly [-0.70, -0.35]
    exponents = np.linspace(-0.70, -0.35, 71)          # step ≈0.005
    # a few offsets; offset =0 corresponds to the classic power‑law
    offsets = np.arange(0.0, 3.01, 0.25)               # 0 … 3 in steps of 0.25

    print("[grid] starting exhaustive scan (n={})".format(n))
    for off in offsets:
        for a in exponents:
            cur = _make_power_law(n, a, offset=off)
            score = _eval(cur)
            if score < best_score:
                best_score = score
                best_seq = cur.copy()
                best_alpha = a
                best_offset = off
        # occasional time‑check
        if time.time() >= deadline:
            break
    print("[grid] best α={:.5f}, offset={:.5f}, score={:.6f}".format(
        best_alpha, best_offset, best_score))

    # --------------------------------------------------------------
    # 2) Derivative‑free local optimisation of (α, offset)
    # --------------------------------------------------------------
    # We perform a simple stochastic hill‑climb in the 2‑D parameter space.
    sigma_alpha = 0.02          # initial standard deviation for α
    sigma_off   = 0.30          # initial standard deviation for offset
    it = 0
    max_local_iters = 20000
    while time.time() < deadline and it < max_local_iters:
        # propose a new point by Gaussian perturbation
        cand_alpha = best_alpha + np.random.randn() * sigma_alpha
        cand_off   = best_offset + np.random.randn() * sigma_off

        # keep parameters inside a sensible domain
        if not (-0.90 < cand_alpha < -0.10) or cand_off < 0.0:
            # reject out‑of‑bounds proposals
            it += 1
            continue

        cand_seq = _make_power_law(n, cand_alpha, offset=cand_off)
        cand_score = _eval(cand_seq)

        if cand_score < best_score:
            best_alpha, best_offset = cand_alpha, cand_off
            best_seq = cand_seq.copy()
            best_score = cand_score
            print("[local] iter {}: improved to α={:.5f}, off={:.5f}, score={:.6f}"
                  .format(it, best_alpha, best_offset, best_score))

        # slowly shrink the proposal radii (simulated‑annealing style)
        sigma_alpha *= 0.9995
        sigma_off   *= 0.9995
        it += 1

        # early exit if we have already passed the target bound
        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 3) Fine‑grained hill‑climbing on the actual vector entries
    # --------------------------------------------------------------
    #   – move a small random fraction of mass from one index to another,
    #   – occasionally swap two entries.
    #   All moves preserve the L1‑norm (sum = 1) up to floating‑point noise.
    # --------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    n = cur.shape[0]

    # fraction of mass that may be moved in a single proposal
    max_transfer_frac = 0.08   # up to 8 % of the source entry

    hill_iter = 0
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        if np.random.rand() < 0.7:
            # ----- transfer a random fraction from i to j -----
            if cur[i] == 0.0:
                continue
            delta = cur[i] * np.random.rand() * max_transfer_frac
            cur[i] -= delta
            cur[j] += delta
        else:
            # ----- swap the two values (no transfer) -----
            cur[i], cur[j] = cur[j], cur[i]

        new_score = _eval(cur)

        if new_score < cur_score:
            cur_score = new_score
            # keep the improvement
            if hill_iter % 5000 == 0:
                print("[hill‑climb] iter {}: improved score={:.6f}"
                      .format(hill_iter, cur_score))
        else:
            # revert the move
            # (swap case: swapping twice restores original order)
            if np.random.rand() < 0.7:
                cur[i] += delta
                cur[j] -= delta
            else:
                cur[i], cur[j] = cur[j], cur[i]

        hill_iter += 1

        # optionally stop if target already reached
        if cur_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 4) Return the final sequence as a plain Python list
    # --------------------------------------------------------------
    final_sequence = [float(v) for v in cur.tolist()]
    print("Returning sequence of length {} with final score {:.6f}"
          .format(len(final_sequence), cur_score))
    return final_sequence
