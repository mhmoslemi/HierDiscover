def beta_distribution(n, p):
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (p - 1) * (n - i - 0.5) ** (p - 1)
    vals /= vals.sum()

def triangular_distribution(n):
    i = np.arange(n)
    center = (n-1)/2.0
    vals = 1 - np.abs(i - center) / (center + 1)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals
```

def refined_descent(a0, time_deadline, max_iters=20000, eta0=0.2, momentum=0.9):
    a = a0.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for it in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        b = np.convolve(a, a)
        k = np.argmax(b)
        # compute subgradient
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        # step
        eta = eta0 / np.sqrt(it)
        v = momentum * v - eta * g
        a = a + v
        # projection to simplex
        a = _project_onto_simplex(a, s=1.0)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # optionally print
            print(f"[{time.time():.2f}] iter={it}, score={best_score:.6f}")
        # occasional random perturb
        if it % 1000 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(len(a))*1e-5
            a = _project_onto_simplex(a, s=1.0)
    return best_a, best_score
```

def _fft_convolve(a):
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[:2*n-1]
```

def _gradient_softmax(a, beta):
    n = len(a)
    b = _fft_convolve(a)
    max_b = b.max()
    shifted = b - max_b
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # compute conv of w and reversed a
    m = 2*n - 1
    size = 1 << ((m + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a[::-1], n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    # slice
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g
```

"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Shape search** – For a chosen length `n` we sweep a simple
   one‑parameter family of symmetric Beta‑like densities
   `a_i ∝ (i+½)^{p‑1}·(n‑i‑½)^{p‑1}` (p = 0.2 … 2.5).  The score
   `2·n·max(a*a)` is evaluated with an FFT‑based convolution, which is
   fast enough to test dozens of shapes.

2. **Refinement** – Starting from the best shape we run a projected
   (soft‑max) gradient descent.  The loss is the `L∞`‑norm of the
   autocorrelation; a smoothed version (soft‑max) gives a usable
   gradient that moves all large peaks simultaneously.  The update is
   momentum‑accelerated and each iterate is projected onto the probability
   simplex (non‑negative entries summing to 1).

3. **Finalisation** – The best sequence seen during the whole run is
   returned as a plain Python list.  The code respects the 1000 s budget,
   never touches the filesystem and uses only NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector *v* onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Fast convolution via FFT (real‑valued, symmetric)
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a using an FFT of length a power‑of‑two."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
# Score (the evaluation function without the scaling‑invariant checks)
def _score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) assuming sum(a)=1 and a≥0."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        return np.inf
    a /= total
    b = _fft_convolve(a)
    return float(2.0 * len(a) * b.max())

# ----------------------------------------------------------------------
# One‑parameter symmetric “Beta” distribution
def _beta_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) density on a discrete grid of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (p - 1) * (n - i - 0.5) ** (p - 1)
    vals /= vals.sum()
    return vals

# ----------------------------------------------------------------------
# Gaussian‑shaped distribution (symmetric)
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals

# ----------------------------------------------------------------------
# Gradient of a *soft‑max* approximation of max(b) w.r.t. a
def _softmax_grad(a: np.ndarray, beta: float) -> np.ndarray:
    """Return ∇_a  softmax_β(max(b)) where b = a*a."""
    n = a.shape[0]
    b = _fft_convolve(a)
    m = b.max()
    # soft‑max weights (stable)
    shifted = b - m               # max becomes 0, others ≤ 0
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                 # now Σ w = 1
    # gradient = 2 * cross‑correlation(w, a)
    # Compute conv(w, a[::-1]) and slice the central n entries
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a[::-1], n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    g = 2.0 * conv[start : start + n]
    return g

# ----------------------------------------------------------------------
# Projected (soft‑max) gradient descent
def _refine_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iters: int = 25000,
    eta0: float = 0.6,
    momentum: float = 0.9,
    beta_start: float = 2.0,
    beta_end: float = 15.0,
) -> tuple[np.ndarray, float]:
    """Refine a using a soft‑max gradient with momentum and simplex projection."""
    a = a0.copy()
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _score(a)

    n = len(a)
    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # gradually increase the soft‑max sharpness
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)

        g = _softmax_grad(a, beta)

        eta = eta0 / np.sqrt(it)
        v = momentum * v - eta * g
        a = a + v
        a = _project_onto_simplex(a, s=1.0)

        cur = _score(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            print(
                f"[{time.time():.2f}] iter={it:5d} β={beta:.2f} score={best_score:.6f}"
            )

        # small random jitter to escape tiny plateaus
        if it % 2000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (within the time budget)
    attains a low `evaluate_sequence` score.
    """
    np.random.seed(seed)
    # give a small safety margin
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a length that balances quality and speed.
    # 8000 is a sweet spot for the 1000 s budget.
    n = 8000

    # ------------------------------------------------------------------
    # 1️⃣  Scan a simple Beta family to obtain a good starting shape.
    best_seq = None
    best_score = np.inf

    # a fine grid of the shape parameter p
    p_grid = np.concatenate(
        [np.linspace(0.20, 0.80, 13), np.linspace(0.85, 2.00, 24)]
    )

    for p in p_grid:
        if time.time() >= deadline:
            break
        a = _beta_distribution(n, p)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"p={p:5.3f} → score={sc:.6f}")

    # also try a few Gaussian widths (just in case)
    sigma_grid = np.linspace(0.05 * n, 0.45 * n, 12)
    for sigma in sigma_grid:
        if time.time() >= deadline:
            break
        a = _gaussian_distribution(n, sigma)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"sigma={sigma:.0f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Refine the best candidate with projected soft‑max descent.
    # Allocate the remaining time (≈80 % of the budget) to this phase.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    refine_deadline = deadline
    refined_seq, refined_score = _refine_descent(
        best_seq,
        time_deadline=refine_deadline,
        max_iters=20000,
        eta0=0.6,
        momentum=0.9,
        beta_start=2.0,
        beta_end=15.0,
    )
    if refined_score < best_score:
        best_score, best_seq = refined_score, refined_seq.copy()

    # ------------------------------------------------------------------
    # 3️⃣  A quick post‑processing jitter to possibly shave a bit more.
    # (only if a little time is still left)
    if time.time() < deadline:
        post_deadline = deadline
        a = best_seq.copy()
        v = np.zeros_like(a)
        for it in range(1, 8001):
            if time.time() >= post_deadline:
                break
            # plain sub‑gradient of the active max‑entry (cheaper)
            b = _fft_convolve(a)
            k = int(np.argmax(b))
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            g = np.zeros_like(a)
            g[mask] = 2.0 * a[idx[mask]]
            eta = 0.3 / np.sqrt(it)
            v = 0.85 * v - eta * g
            a = a + v
            a = _project_onto_simplex(a, s=1.0)
            cur = _score(a)
            if cur < best_score:
                best_score, best_seq = cur, a.copy()
                print(

            # occasional tiny noise

                a += np.random.randn(n) * 1e-5
                a = _project_onto_simplex(a, s=1.0)

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list as required by the outer harness.
    return best_seq.tolist()
