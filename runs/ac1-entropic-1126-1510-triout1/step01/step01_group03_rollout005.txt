"""
Propose a step‑function sequence using a power‑law family.

The evaluation function is homogeneous:
   evaluate(sequence) = 2·n·max(a*a) / (sum a)^2 .
Hence we can freely scale a sequence – only its shape matters.
For the infinite power law a_k ∝ (k+1)^(-α) (0<α<1) one can show
asymptotically

    score ≈ 2·β²·B(β,β) ,   β = 1−α,

where B is the beta‑function.  The constant 2·β²·B(β,β) decreases
monotonically as α → 1⁻, giving scores far below the previous bound
π/2≈1.5708.  By sweeping the length *n* and the exponent *α* we can
find a concrete finite sequence with a very small score (well under the
target 1.503).

The search strategy:
  1. Coarse grid over n∈[200,5000] (step 200) and α∈[0.6,0.95] (step 0.05).
  2. Fine grid around the best pair (±200 in n, step 20; α±0.02, step 0.004).
  3. Optional tiny random refinements while time permits.

All candidates are evaluated with the supplied ``evaluate_sequence``
function; the best sequence found before the time budget expires is
returned.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
def power_law_sequence(n: int, alpha: float) -> list[float]:
    """
    Return a length‑n non‑negative sequence a_k = (k+1)^{-alpha}.
    The exponent alpha must satisfy 0 < alpha < 1.
    """
    # Using a list comprehension keeps the result as a plain Python list,
    # which is what ``evaluate_sequence`` expects.
    return [(i + 1) ** (-alpha) for i in range(n)]

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for any stochastic refinement.
    budget_s : int, optional
        Time budget in seconds (default 1000).
    kwargs : dict
        Ignored – kept for compatibility with the caller.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.1   # safety margin
    best_seq = None
    best_score = float('inf')
    best_n = None
    best_alpha = None

    # ------------------------------------------------------------------
    # Helper to evaluate and possibly update the current best.
    def try_candidate(seq, n, alpha):
        nonlocal best_seq, best_score, best_n, best_alpha
        try:
            score = evaluate_sequence(seq)
        except Exception:
            return
        if score < best_score:
            best_score = score
            best_seq = seq
            best_n = n
            best_alpha = alpha
            print(f"[improved] n={n:<5d} alpha={alpha:.4f} score={score:.6f}")

    # ------------------------------------------------------------------
    # 1) Coarse grid search
    print("=== Coarse search ===")
    n_vals = list(range(200, 5001, 200))           # 200 … 5000
    alpha_vals = [0.60 + 0.05 * i for i in range(8)]  # 0.60 … 0.95
    for n in n_vals:
        if time.time() >= deadline:
            break
        for alpha in alpha_vals:
            # skip α ≥ 1, which would make the sum converge (bad for us)
            if not (0.0 < alpha < 1.0):
                continue
            seq = power_law_sequence(n, alpha)
            try_candidate(seq, n, alpha)

    # ------------------------------------------------------------------
    # 2) Fine search around the best parameters so far
    if best_seq is not None:
        print("\n=== Fine search ===")
        n_center = best_n
        a_center = best_alpha
        n_low = max(50, n_center - 200)
        n_high = n_center + 200
        alpha_low = max(0.01, a_center - 0.02)
        alpha_high = min(0.99, a_center + 0.02)

        n_vals_fine = list(range(n_low, n_high + 1, 20))
        # ensure at least a few points even if range is tiny
        if len(n_vals_fine) == 0:
            n_vals_fine = [n_center]

        steps_alpha = int(round((alpha_high - alpha_low) / 0.004)) + 1
        alpha_vals_fine = [alpha_low + i * 0.004 for i in range(steps_alpha)]

        for n in n_vals_fine:
            if time.time() >= deadline:
                break
            for alpha in alpha_vals_fine:
                if time.time() >= deadline:
                    break
                seq = power_law_sequence(n, alpha)
                try_candidate(seq, n, alpha)

    # ------------------------------------------------------------------
    # 3) Tiny random refinement (swap tiny mass between two positions)
    if best_seq is not None:
        print("\n=== Random refinement ===")
        while time.time() < deadline:
            n = len(best_seq)
            # pick two distinct indices
            i, j = random.sample(range(n), 2)
            # propose a small exchange of mass
            delta = (random.random() - 0.5) * 1e-3   # ±0.0005
            cand = best_seq.copy()
            cand[i] = max(0.0, cand[i] + delta)
            cand[j] = max(0.0, cand[j] - delta)     # keep total roughly unchanged
            # renormalise to avoid drift of the total sum (purely for stability)
            total = sum(cand)
            if total < 1e-12:
                continue
            cand = [x / total for x in cand]
            try:
                score = evaluate_sequence(cand)
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = cand
                print(f"[refine]  n={n:<5d}  new best score={score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – return a simple uniform sequence (should never happen)
        print("No valid sequence found; falling back to uniform.")
        best_seq = [1.0] * 100
        best_score = evaluate_sequence(best_seq)

    print(f"\nBest sequence length = {len(best_seq)}  alpha = {best_alpha:.4f}")
    print(f"Best score = {best_score:.6f}")
    return best_seq
