"""
Propose a step‑function whose evaluation value is < 1.503.

The algorithm uses the iterative linear‑programming scheme described in the
statement (the “f–g‑update” method).  Starting from the classic
a_i = (i+1)^{-0.5} shape (normalised to sum = 1) we repeatedly

1. compute the current autocorrelation peak M = max(a*a),
2. solve a LP that maximises the total mass ∑b_j under the linear
   constraints (a*b)_k ≤ M (non‑negativity of b),
3. normalise the LP solution b to sum = 1,
4. blend the old vector a with the new one (a←(1‑t)a+t b) with a small
   mixing factor t.

Each LP strictly increases the total mass that can be placed under the
same peak‑constraint, therefore the subsequent blend reduces the peak
and the objective 2 n·max/(∑a)².  After a few LP iterations we finish with
a short local hill‑climb that moves a tiny amount of weight from indices
contributing heavily to the current peak to indices that do not, which
polishes the result.

The whole search stops after the allocated time budget and returns the
best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that (A @ b)[k] = (f * b)[k].
    Each column j contains a copy of f shifted by j positions.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)_k (linear convolution)."""
    return np.max(np.convolve(a, a))


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the linear programme:
        maximise   sum(b)
        subject to (f*b)_k ≤ M   for all k
                   b ≥ 0
    Returns the normalised solution b (sum(b) == 1).
    """
    n = f.shape[0]
    A = _conv_matrix(f)                     # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(b)  → minimise  -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # HiGHS is the default solver for recent SciPy versions and is fast.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    b_raw = res.x
    total = b_raw.sum()
    if total <= 0.0:
        # fallback – should never happen because b=0 is always feasible.
        return f.copy()
    return b_raw / total                         # normalise to sum = 1


def _blend(old: np.ndarray, new: np.ndarray, t: float) -> np.ndarray:
    """Return (1‑t)·old + t·new (both vectors have the same L¹‑norm)."""
    return (1.0 - t) * old + t * new


def _local_refine(seq: np.ndarray,
                  time_deadline: float,
                  max_iters: int = 5000) -> np.ndarray:
    """
    Small hill‑climbing refinement.  In each step a tiny amount of mass is moved
    from an index heavily involved in the current peak to a random index that
    contributes little.  The move is kept only if it lowers the peak.
    """
    n = seq.shape[0]
    best = seq.copy()
    best_score = 2.0 * n * _max_convolution(best)          # sum(best)=1 → denominator =1

    # pre‑allocate a buffer for the convolution; size = 2n‑1
    conv = np.convolve(best, best)

    for _ in range(max_iters):
        if time.time() > time_deadline:
            break

        # locate the current peak
        k_peak = int(np.argmax(conv))
        # contribution of each index to that peak (zero if out of range)
        contrib = np.zeros(n, dtype=np.float64)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contrib[i] = best[i] * best[j]

        # choose a source index weighted by its contribution
        if contrib.sum() == 0.0:
            break
        prob_src = contrib / contrib.sum()
        i_src = np.random.choice(n, p=prob_src)

        # choose a destination index that is *not* the partner of i_src in the peak
        # (simple uniform choice; avoid the exact partner to keep the peak down)
        while True:
            i_dst = np.random.randint(0, n)
            if i_dst != i_src:
                break

        # amount to transfer – a modest fraction of the source value
        eps = best[i_src] * np.random.uniform(0.01, 0.07)
        if eps <= 0.0:
            continue

        # apply the tentative move on copies
        cand = best.copy()
        cand[i_src] -= eps
        cand[i_dst] += eps

        # recompute convolution only for the candidate (full recompute – cheap for n≈800)
        new_conv = np.convolve(cand, cand)
        new_score = 2.0 * n * new_conv.max()

        if new_score < best_score:
            best = cand
            best_score = new_score
            conv = new_conv

    return best


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (empirically) minimise
    ``evaluate_sequence``.  The routine implements the LP‑based iteration
    described above, followed by a brief stochastic refinement.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5   # safety margin for final processing

    # --------------------------------------------------------------
    # 1.  Initial shape – inverse‑square‑root, normalised to sum = 1.
    # --------------------------------------------------------------
    n = 800                             # moderate length – good speed/quality trade‑off
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** -0.5
    a /= a.sum()                       # L¹‑normalised (sum = 1)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Main LP‑iteration loop.
    # --------------------------------------------------------------
    t = 0.25          # initial blend factor (will decay)
    it = 0
    while time.time() < deadline and it < 30:   # at most 30 LP steps
        it += 1
        M = _max_convolution(a)                 # current peak
        # solve LP to obtain a new direction g
        try:
            g = _lp_step(a, M)
        except Exception as e:
            print(f"[lp] iteration {it} failed: {e}")
            break

        # blend a and g
        a = _blend(a, g, t)

        # evaluate the new candidate
        cur_score = evaluate_sequence(a.tolist())
        print(f"[lp {it:02d}] score={cur_score:.6f}, t={t:.3f}")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print("   ** new best **")

        # slowly decay the blending factor – later steps become finer
        t = max(0.03, t * 0.85)

    # --------------------------------------------------------------
    # 3.  Optional local hill‑climb for the remaining time budget.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:                     # need a few seconds for the refinement
        refined = _local_refine(best_seq, deadline)
        refined_score = evaluate_sequence(refined.tolist())
        print(f"[refine] score={refined_score:.6f}")
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()

    print(f"[final] best score={best_score:.6f}, length={n}")
    # Evaluate once more just to be safe (evaluate_sequence applies clipping etc.)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] verify score={final_score:.6f}")

    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
