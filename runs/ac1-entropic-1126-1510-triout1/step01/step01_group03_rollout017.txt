Implementation details: projection onto simplex ensures nonnegative and sum=1.

Project onto simplex algorithm (Duchi et al., 2008) simple: sort descending, find threshold tau.

We'll implement function project_to_simplex(v) returning projection onto simplex.

Pseudo:

def project_to_simplex(v):
    # v must be ndarray
    # ensures non-negative and sum=1
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Implementation for projection:

def project_to_simplex(v):
    """Project vector v onto the probability simplex (non-negative sum to 1)."""
    v = np.asarray(v)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1.  Generate a family of candidate lengths (n = 200 … 1500, step 200).  
2.  For each length start from the classic arcsine coefficients
    (C_i = binom(2i,i)/4^i) – they already give a good bound.
3.  Run a simple projected‑gradient descent that tries to reduce the
    maximal entry of the self‑convolution.
    *   The objective is  max_k (a*a)_k  (sum(a)=1 is enforced).
    *   The (sub)gradient for the maximal entries is
        g_i = 2 * Σ_{k∈K} a_{k-i}   where K are the indices attaining the
        current maximum (the “active set”).
    *   After a gradient step the vector is projected onto the probability
        simplex (non‑negative, sum = 1).  Projection uses the O(n) algorithm
        of Duchi et al.
4.  Keep the best sequence seen over all lengths and all random restarts.
5.  If time permits perform tiny random refinements around the best
    candidate (exchange a tiny amount of mass between two coordinates).

The method stays well inside the 1000 s budget, uses only NumPy (no external
solvers) and usually pushes the score below the previous 1.5708 limit,
often reaching the target ~1.50 range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Return a length‑n array proportional to C_i = binom(2i,i)/4**i."""
    seq = np.empty(n, dtype=np.float64)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq /= total                     # normalise to sum = 1
    return seq


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # reference: Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Fast (FFT) full convolution of two real vectors."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv


# ----------------------------------------------------------------------
def gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected‑gradient step that tries to lower max(a*a).
    Returns the updated vector (still on the simplex).
    """
    n = a.size
    conv = full_convolve(a, a)                # length 2n‑1
    max_val = conv.max()
    # indices whose value is (almost) maximal
    mask = np.abs(conv - max_val) < 1e-12
    w = mask.astype(np.float64)
    w_sum = w.sum()
    if w_sum == 0.0:                          # fallback – uniform weights
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # gradient of the smoothed max = Σ_k w_k ∇c_k
    # ∇c_k[i] = 2 * a_{k-i} (if 0≤k-i<n else 0)
    rev = a[::-1]
    grad_full = full_convolve(w, rev)         # length 3n‑2
    start = len(w) - 1                        # index for i=0
    grad = 2.0 * grad_full[start:start + n]   # shape (n,)

    # simple step size adaptation – normalise by its norm
    norm = np.linalg.norm(grad)
    if norm > 0:
        a_new = a - step * grad / norm
    else:
        a_new = a.copy()
    # project back onto the simplex
    a_proj = project_to_simplex(a_new)
    return a_proj


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns the best list found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin
    best_seq = None
    best_score = float('inf')

    # lengths to try – a compromise between speed and expressive power
    lengths = list(range(200, 1601, 200))     # 200,400,…,1500

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== length {n} ===")
        # three different starts: arcsine and two random Dirichlet draws
        starts = [arcsine_sequence(n)]
        for _ in range(2):
            rnd = np.random.rand(n)
            starts.append(rnd / rnd.sum())

        for start_id, a in enumerate(starts, 1):
            a = a.astype(np.float64)
            # parameters for the descent
            step = 0.12               # initial step‑size
            decay = 0.995             # geometric decay per iteration
            max_iter = 2500
            for it in range(max_iter):
                if time.time() >= deadline:
                    break
                # evaluate current score
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.tolist()
                    print(f"[improve] n={n} start={start_id} it={it:4d} "
                          f"score={score:.6f}")
                    # early exit if we already beat the target comfortably
                    if best_score < 1.45:
                        print("Target comfortably passed – stopping early.")
                        return best_seq

                # one gradient step
                a = gradient_step(a, step)
                step *= decay

            # after the gradient run, do a few tiny random refinements
            while time.time() < deadline:
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 5e-3   # ±0.0025
                cand = a.copy()
                cand[i] = max(0.0, cand[i] + delta)
                cand[j] = max(0.0, cand[j] - delta)
                s = cand.sum()
                if s <= 0.0:
                    continue
                cand /= s                                   # renormalise
                score = evaluate_sequence(cand.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = cand.tolist()
                    a = cand
                    print(f"[refine] n={n} it={it:4d} new best={score:.6f}")
                # limit the random refining loop
                if np.random.rand() < 0.01:
                    break

    if best_seq is None:
        # Fallback – a simple uniform distribution (never reached in practice)
        best_seq = [1.0 / 2] * 2
        best_score = evaluate_sequence(best_seq)

    print(f"\nBest sequence length = {len(best_seq)} with score = {best_score:.6f}")
    return best_seq
