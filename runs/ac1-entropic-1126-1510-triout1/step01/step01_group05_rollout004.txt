"""
Propose a candidate step‑function sequence by:
1. Trying a few simple parametric families (power‑law with several exponents,
   a random simplex point and the existing best).
2. Picking the best of those as a starting point.
3. Running a simulated‑annealing / stochastic hill‑climbing phase that moves
   a small amount of probability mass between two positions while preserving
   the total sum.  The temperature schedule starts warm (allowing uphill
   steps) and cools down to zero, giving the algorithm a chance to escape
   local minima.
4. Finishing with a pure greedy hill‑climb for a final polish.

The code respects the 1000 s budget, prints progress and always returns a
non‑negative list of heights that (hopefully) beats the previous bound.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper utilities that wrap the provided evaluate_sequence function
# --------------------------------------------------------------------------- #

def evaluate_vec(vec: np.ndarray) -> float:
    """
    Evaluate a numpy vector using the supplied evaluate_sequence() function.
    The function converts the vector to a plain Python list.
    """
    return evaluate_sequence(vec.tolist())


def power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    Decreasing power‑law vector of length n, normalized to sum 1.
    a_i ∝ (i+1)^{-p}.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


def random_simplex(n: int) -> np.ndarray:
    """
    Generate a random non‑negative vector of length n and normalise it to sum 1.
    """
    a = np.random.rand(n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# --------------------------------------------------------------------------- #
# Core optimisation routines
# --------------------------------------------------------------------------- #

def anneal_optimize(start_vec: np.ndarray,
                    deadline: float,
                    init_temp: float = 1e-3,
                    decay_power: float = 3.0,
                    max_factor_start: float = 0.30,
                    max_factor_end: float = 0.01) -> np.ndarray:
    """
    Stochastic simulated‑annealing optimiser.

    Parameters
    ----------
    start_vec : np.ndarray
        Starting non‑negative vector (sum ≈ 1).
    deadline : float
        Absolute time (as returned by time.time()) at which the routine must stop.
    init_temp : float
        Temperature at the beginning of the run.
    decay_power : float
        Controls how quickly the temperature decays (higher → faster cooling).
    max_factor_start / max_factor_end : float
        Fraction of the source entry that may be moved at the start / at the end
        of the run.  The factor is interpolated linearly with elapsed time.

    Returns
    -------
    np.ndarray
        The best vector discovered (not necessarily the final one).
    """
    n = len(start_vec)
    cur = start_vec.copy()
    cur_score = evaluate_vec(cur)
    best_vec = cur.copy()
    best_score = cur_score

    start_time = time.time()
    total_budget = max(1e-6, deadline - start_time)   # avoid division by zero

    # main loop
    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = (now - start_time) / total_budget   # 0 → 1

        # temperature schedule (cooling to zero)
        temp = init_temp * (1.0 - elapsed) ** decay_power

        # gradually shrink the allowed move size
        max_factor = max_factor_start * (1.0 - elapsed) + max_factor_end * elapsed

        # pick two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1      # ensure j != i

        src_val = cur[i]
        if src_val <= 0.0:
            continue

        # amount to transfer – a random fraction of the source entry
        delta = max_factor * src_val * np.random.rand()
        if delta <= 0.0:
            continue

        # provisional move (preserve sum)
        cur[i] = src_val - delta
        cur[j] += delta

        # safety: numerical drift may create a tiny negative due to rounding
        if cur[i] < 0.0:
            # reject and revert immediately
            cur[i] = src_val
            cur[j] -= delta
            continue

        cand_score = evaluate_vec(cur)

        # acceptance test
        improve = cand_score < cur_score
        if improve:
            accept = True
        else:
            # metropolis criterion
            prob = np.exp(-(cand_score - cur_score) / (temp + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cur.copy()
                # occasional progress report
                if int(elapsed * 100) % 5 == 0:
                    print(f"[anneal] t={now-start_time:5.1f}s, "
                          f"best_score={best_score:.6f}")
        else:
            # revert the move
            cur[i] = src_val
            cur[j] -= delta

    return best_vec


def hill_climb(vec: np.ndarray,
               deadline: float,
               max_no_improve: int = 5000) -> np.ndarray:
    """
    Simple greedy hill‑climber that repeatedly moves a tiny amount of mass
    between two positions, accepting only strictly improving moves.
    """
    n = len(vec)
    cur = vec.copy()
    cur_score = evaluate_vec(cur)
    best_vec = cur.copy()
    best_score = cur_score
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        src = cur[i]
        if src <= 0.0:
            continue

        # move at most 2 % of the source entry (tiny local refinement)
        delta = 0.02 * src * np.random.rand()
        if delta <= 0.0:
            continue

        # apply move
        cur[i] = src - delta
        cur[j] += delta

        cand_score = evaluate_vec(cur)
        if cand_score < cur_score:
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cur.copy()
        else:
            # revert
            cur[i] = src
            cur[j] -= delta
            no_improve += 1

    return best_vec

# Main entry point required by the competition framework
# --------------------------------------------------------------------------- #

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a sequence (list of non‑negative heights) with a low evaluation score.

    The routine:
      * builds a few simple initial candidates (power‑laws with different exponents,
        a random simplex point and the legacy `height_sequence_1`);
      * selects the best of them;
      * runs a simulated‑annealing optimiser for the bulk of the time budget;
      * finishes with a short greedy hill‑climb;
      * returns the best vector found.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------- #
    # Determine the working length.
    # ------------------------------------------------------------------- #
    try:
        n = len(height_sequence_1)   # use the known construction length if present
    except Exception:
        n = 2000                     # reasonable fallback

    overall_deadline = time.time() + budget_s - 5.0   # 5 s safety margin

    # ------------------------------------------------------------------- #
    # 1) Build a small portfolio of starter vectors.
    # ------------------------------------------------------------------- #
    starters = []

    # power‑law seeds with a spread of exponents
    for p in np.linspace(0.30, 0.90, 13):
        starters.append(power_law_sequence(n, float(p)))

    # one random simplex point
    starters.append(random_simplex(n))

    # optionally use the previous best known construction
    try:
        legacy = np.array(height_sequence_1, dtype=np.float64)
        legacy_sum = legacy.sum()
        if legacy_sum > 0:
            starters.append(legacy / legacy_sum)   # normalise to sum 1
    except Exception:
        pass

    # ------------------------------------------------------------------- #
    # Evaluate starters and keep the best one.
    # ------------------------------------------------------------------- #
    best_vec = None
    best_score = float('inf')
    for idx, vec in enumerate(starters):
        sc = evaluate_vec(vec)
        print(f"[init] candidate {idx:02d} score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()

    print(f"[init] best start score = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # 2) Simulated‑annealing optimisation (majority of the budget).
    # ------------------------------------------------------------------- #
    anneal_deadline = overall_deadline          # use everything left
    best_vec = anneal_optimize(best_vec, anneal_deadline)
    best_score = evaluate_vec(best_vec)
    print(f"[anneal] finished, best score = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # 3) Final greedy refinement (if a little time remains).
    # ------------------------------------------------------------------- #
    hill_deadline = min(overall_deadline, time.time() + max(0.0, overall_deadline - time.time()))
    refined_vec = hill_climb(best_vec, hill_deadline)
    refined_score = evaluate_vec(refined_vec)
    if refined_score < best_score:
        best_vec = refined_vec
        best_score = refined_score
        print(f"[hill] improvement after final climb: score = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # Return result
    # ------------------------------------------------------------------- #
    print(f"[final] best score obtained: {best_score:.6f} (length {n})")
    # The evaluate_sequence will clamp negatives / huge numbers, but we keep the vector safe.
    return best_vec.tolist()
