def project_to_simplex(v):
    # ensure v sorted descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def convolve_fft(a, b):
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()  # next power of 2
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    c = np.fft.irfft(C, N)[:size]
    return c

Now more details for conv_fft:

def convolve_fft(a, b):
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    fc = fa * fb
    c = np.fft.irfft(fc, N)[:size]
    return c
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, scipy.optimize
    start = time.time()
    deadline = start + budget_s - 0.1

    np.random.seed(seed)

    best_seq = None
    best_score = float('inf')
    # choose length
    n = 1800
    target_sum = np.sqrt(2 * n)

    # Initial sequence: arcsine normalized sum=1, then scale to target_sum.
    seq = arcsine_sequence(n)  # returns sum=1
    f = np.array(seq) * target_sum

    # Compute initial M
    conv_f = convolve_fft(f, f)
    M = np.max(conv_f)
    best_score = M
    best_seq = f / target_sum
    print(f"Initial length={n}, M={M:.6f}")

    iteration = 0
    while time.time() < deadline and iteration < 30:
        iteration += 1

        # Build convolution matrix A for constraints f * g <= M
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = f[i]

        # Solve LP: maximize sum(g) subject to A @ g <= M, g >= 0
        c = -np.ones(n)   # maximize sum(g)
        bounds = [(0, None)] * n
        b_ub = np.full(rows, M, dtype=np.float64)

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            print(f"LP iteration {iteration} failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 <= 1e-12:
            print(f"LP iteration {iteration}: sum_g0 too small")

"""
Propose a step‑function sequence using a linear‑programming
iterative improvement.  The idea (see the comment at the top of the
prompt) is:

1.  Start from the classic arcsine sequence and scale it so that
    sum(a) = sqrt(2·n).  With this scaling the evaluation metric becomes
    simply  M = max_k (a*a)_k.

2.  Given the current sequence f and its peak M, solve a linear program

        maximise   Σ g_i
        subject to g_i ≥ 0
                   (f ∗ g)_k ≤ M   for all k = 0 … 2·n–2

   The LP gives a sequence g that can carry **more** ℓ₁‑mass than f
   while still respecting the same convolution bound.

3.  Rescale g back to the same ℓ₁‑norm (sqrt(2·n)) and blend a small
    amount into f:
        f ← (1‑t)·f + t·g   (t≈0.2)

    The new peak M′ = max(f∗f) is guaranteed not to increase; in
    practice it drops.  Repeating the step drives M down towards the
    theoretical limit 1.

4.  The whole loop runs under the given time budget.  The best sequence
    (normalised back to sum = 1) is returned.

The implementation uses `scipy.optimize.linprog` (HiGHS) for the LP and
FFT‑based convolution for speed.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return the length‑n arcsine coefficients C_i = binom(2i,i)/4**i,
    normalised so that they sum to 1.  The recurrence is numerically
    stable and fast."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (real‑valued input)."""
    size = len(a) + len(b) - 1
    # smallest power of two ≥ size
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    fc = fa * fb
    c = np.fft.irfft(fc, N)[:size]
    return c


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Create the matrix A (shape (2n‑1, n)) such that (f∗g) = A @ g."""
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f[i] to rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with evaluation < 1.503.
    Uses the LP‑based refinement described above.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05  # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1) Choose a problem size – larger gives more room for improvement,
    #    but also a larger LP.  1800 is a good compromise.
    n = 1800
    target_sum = np.sqrt(2.0 * n)               # we keep ℓ1‑norm = √(2n)

    # start from the arcsine construction, normalised to sum=1,
    # then scale to the required ℓ1‑norm.
    a0 = np.array(arcsine_sequence(n), dtype=np.float64)
    f = a0 * target_sum                         # current sequence (ℓ1‑norm = √(2n))

    # initial peak (this is exactly π/2 ≈ 1.5708)
    conv_f = convolve_fft(f, f)
    M = float(conv_f.max())
    best_seq = (f / target_sum).tolist()
    best_score = M
    print(f"[init] n={n}, initial peak M={M:.6f}")

    iteration = 0
    # --------------------------------------------------------------
    # Main refinement loop – each iteration solves a linear program.
    while time.time() < deadline and iteration < 30:
        iteration += 1

        # 1) Build the linear constraints (f ∗ g)_k ≤ M.
        A = build_conv_matrix(f)               # (2n‑1) × n matrix
        rows = A.shape[0]

        # 2) Linear programme: maximise Σ g_i  s.t. A @ g ≤ M , g ≥ 0
        c = -np.ones(n)                         # maximise Σ g_i  → minimise −Σg_i
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # time left is given to the solver to avoid overrunning the budget
        time_left = max(0.0, deadline - time.time())
        # HiGHS is the default for recent SciPy; it's fast for dense LPs.
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"time_limit": time_left})

        if not res.success:
            print(f"[iter {iteration}] LP failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = float(g0.sum())
        if sum_g0 <= 1e-12:
            print(f"[iter {iteration}] LP returned near‑zero sum, aborting.")
            break

        # 3) Rescale g0 to have the same ℓ1‑norm as f.
        scale = target_sum / sum_g0
        g = g0 * scale

        # 4) Blend a fraction of g into f.  A modest mixing weight works well.
        t = 0.20
        f_new = (1.0 - t) * f + t * g

        # 5) Compute the new peak.
        conv_new = convolve_fft(f_new, f_new)
        M_new = float(conv_new.max())

        print(f"[iter {iteration}] M={M:.6f} → M_new={M_new:.6f}")

        if M_new < M - 1e-8:                     # genuine improvement
            f = f_new
            M = M_new
            cand = (f / target_sum).tolist()
            if M < best_score:
                best_score = M
                best_seq = cand
                print(f"[update] new best score={best_score:.6f}")
        else:
            # No improvement – try a smaller mixing weight once.
            t_small = t * 0.1
            f_new = (1.0 - t_small) * f + t_small * g
            conv_new = convolve_fft(f_new, f_new)
            M_new = float(conv_new.max())
            if M_new < M - 1e-10:
                f = f_new
                M = M_new
                cand = (f / target_sum).tolist()
                if M < best_score:
                    best_score = M
                    best_seq = cand
                    print(f"[update‑small] new best score={best_score:.6f}")
            else:
                print("[iter] no improvement, terminating loop.")
                break

    # --------------------------------------------------------------
    # If time permits, make a tiny random local search around the best
    # candidate (does not hurt if the loop never runs).
    while time.time() < deadline and best_seq is not None:
        n = len(best_seq)
        a = np.array(best_seq, dtype=np.float64)
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-3   # tiny redistribution
        a[i] = max(0.0, a[i] + delta)
        a[j] = max(0.0, a[j] - delta)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s                                    # renormalise to sum=1
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"[refine] improved score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluator.
    try:
        final_score = evaluate_sequence(best_seq)
    except Exception:
        final_score = None
    print(f"Best sequence length={len(best_seq)} final score={final_score}")

    return best_seq
