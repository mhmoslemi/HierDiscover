def _conv_max_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_val = np.max(conv)
    return max_val
```

"""
A two‑stage optimizer for the step‑function score

    score = 2 n·max(conv(a,a)) / (sum(a)²)

The algorithm works as follows

1. **Random start** – pick a length `n` (few hundred is cheap for the
   linear program) and an initial shape (geometric, power‑law,
   mixed or purely random).  The vector is scaled to the convenient
   norm `sum(a)=√(2 n)`, because with this normalization the score
   equals the maximum of the self‑convolution.

2. **Projected sub‑gradient descent** – repeatedly move against the
   sub‑gradient of the active convolution entries while projecting
   back onto the probability simplex.  This quickly lowers the
   maximum convolution.

3. **Linear‑programming “mass‑inflation’’ step** – given the current
   sequence `f`, solve

        maximise   Σ b_j
        subject to A·b ≤ M ,   b ≥ 0

   where `A` is the Toeplitz matrix representing `f∗b` and `M` is the
   current maximal convolution.  The solution `b` adds as much total
   mass as possible without raising any convolution entry above `M`.
   After scaling `b` to the fixed sum `√(2 n)` we mix it with `f`
   (`f←(1‑t)f + t·scaled(b)`).  This convex combination provably
   reduces the maximal convolution.

4. **Iterate** the LP step a few times, then keep the best sequence
   ever seen.  The whole search is repeated with new random starts
   until the time budget (≈ 1000 s) expires.

The returned list can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return the uniform point.
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray):
    """Maximum value of the (auto)‑convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = float(conv.max())
    # indices where the maximum is attained (used for the sub‑gradient)
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return max_val, idx.astype(int)


def _gradient_for_max(a: np.ndarray, max_idx):
    """Sub‑gradient of max conv(a,a) w.r.t. a."""
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]          # d / d a_i Σ a_i a_{k‑i}
    return grad


def _subgradient_optimize(a0: np.ndarray, n: int, deadline: float):
    """
    Projected sub‑gradient descent on the simplex.
    Returns the best vector (as np.ndarray) and its max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_m, _ = _conv_max_fft(a)
    it = 0

    while time.time() < deadline and it < 3000:
        it += 1
        cur_m, max_idx = _conv_max_fft(a)

        # sub‑gradient of the active convolution entries
        grad = _gradient_for_max(a, max_idx)

        # decreasing step size, simple back‑tracking
        eta0 = 0.2 / np.sqrt(it)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1):
            eta = eta0 * factor
            a_cand = _project_to_simplex(a - eta * grad)
            m_cand, _ = _conv_max_fft(a_cand)
            if m_cand < cur_m - 1e-12:
                a = a_cand
                cur_m = m_cand
                improved = True
                break

        if not improved:
            # tiny random jitter to escape flat spots
            a_pert = a.copy()
            idx = np.random.randint(n, size=5)
            a_pert[idx] += np.random.randn(5) * 0.001
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            m_cand, _ = _conv_max_fft(a_pert)
            if m_cand < cur_m:
                a = a_pert
                cur_m = m_cand

        if cur_m < best_m:
            best_m = cur_m
            best_a = a.copy()
            if it % 200 == 0:
                print(f"    sub‑grad iter {it}, new local best max‑conv {best_m:.6f}")

        # premature stop once we are already ≤ 1.5
        if best_m < 1.5:
            break

    return best_a, best_m


def _lp_refine(f: np.ndarray, n: int, target_sum: float, max_iters: int = 5):
    """
    One or more LP‐based “mass‑inflation’’ steps.
    Returns refined vector and its max‑convolution.
    """
    M = _conv_max_fft(f)[0]          # current bound
    best_f = f.copy()
    best_M = M
    t = 0.5                          # mixing parameter

    for it in range(max_iters):
        # Build Toeplitz matrix A where (A·b)_k = (f∗b)_k
        A = np.zeros((2 * n - 1, n), dtype=float)
        for j in range(n):
            A[j:j + n, j] = f

        # Linear program: maximise Σ b_j  (i.e. minimise –Σ b_j)
        c = -np.ones(n)
        # Upper bounds are generous – the final scaling will enforce the sum.
        ub = target_sum * 10.0
        bounds = [(0.0, ub) for _ in range(n)]

        res = linprog(c,
                      A_ub=A,
                      b_ub=np.full(2 * n - 1, M),
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30.0})

        if not res.success:
            # LP failed – give up further refinements.
            break

        b = res.x
        s = b.sum()
        if s <= target_sum * (1.0 - 1e-12):
            # No useful mass increase.
            break

        # Scale back to the required sum.
        scale = target_sum / s
        g = b * scale

        # Convex combination.
        f = (1.0 - t) * f + t * g
        M_new = _conv_max_fft(f)[0]

        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = f.copy()
            print(f"    LP refinement {it+1}/{max_iters}, new max‑conv {best_M:.6f}")

        # Stop if the LP step did not improve the bound.
        if M_new >= M - 1e-12:
            break
        M = M_new

    return best_f, best_M


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) yields a low score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main search loop – random restarts with refinement
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a modest length (large n makes the LP heavy and the
        # factor 2 n in the score hurts)
        n = np.random.randint(300, 800)

        # ---------- generate a random start ----------
        shape = np.random.choice(["geom", "power", "mixed", "random"])

        if shape == "geom":
            # geometric decay a_i = r^i, r∈(0.9,0.9995)
            r = np.random.uniform(0.90, 0.9995)
            a = r ** np.arange(n, dtype=float)

        elif shape == "power":
            # power‑law a_i = (i+1)^(-p)
            p = np.random.uniform(0.3, 2.5)
            a = (np.arange(1, n + 1, dtype=float) ** -p)

        elif shape == "mixed":
            # product of geometric and power‑law
            r = np.random.uniform(0.90, 0.9995)
            p = np.random.uniform(0.4, 2.0)
            a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)

        else:   # "random"
            a = np.random.gamma(1.0, 1.0, size=n)

        # avoid exact zeros (would make the LP unbounded)
        a = np.maximum(a, 1e-12)

        # Normalise to sum = sqrt(2 n) – with this normalisation the score equals the max‑conv.
        target_sum = np.sqrt(2.0 * n)
        a = a / a.sum() * target_sum

        # ---------- local improvement by projected sub‑gradient ----------
        sub_deadline = min(deadline, time.time() + 8.0)   # a few seconds per start
        a_opt, max_conv = _subgradient_optimize(a, n, sub_deadline)

        # ---------- LP‑based refinement ----------
        a_opt2, max_conv2 = _lp_refine(a_opt, n, target_sum, max_iters=5)

        # Keep the best overall candidate.
        if max_conv2 < best_score:
            best_score = max_conv2
            best_seq = a_opt2.copy()
            print(f"Global improvement: n={n}, shape={shape}, max‑conv={best_score:.6f}")

        # Stop early if we already beat the target.
        if best_score < 1.51:
            break

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Return a plain Python list – the evaluator will clip automatically.
    return best_seq.tolist()
