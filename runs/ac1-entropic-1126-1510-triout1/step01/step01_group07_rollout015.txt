def compute_objective(seq):
    arr = np.array(seq, dtype=np.float64)
    # convolution via FFT:
    n = arr.size
    m = 2*n - 1
    # Next power of 2 for FFT length.
    L = 1 << ((2*n-1).bit_length())
    # Compute FFT with zero padding to L.
    A = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(A*A, n=L)[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a**2)

"""Propose a low‑evaluation step‑function.

The search space is a one‑parameter family of power‑law sequences
    a_i = (i+1)^{-p}
(and optionally an exponential taper).  For fixed length n the
objective

    2·n·max(conv(a,a)) / (sum(a))²

is scale‑invariant, so we only need to optimise the shape.
For p>0.5 the interior peak of the self‑convolution decays while the
first term a₀²=1 stays constant; the optimal trade‑off occurs for a
moderate length (≈ 80–200) and a exponent a little larger than 0.5.
We perform a cheap grid search on (n,p) (using an FFT‑based convolution
to obtain max(conv) quickly) and keep the best sequence.
If time permits we refine the best parameters with a finer grid and an
optional exponential taper.
The returned list satisfies all security checks of ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Fast objective computed exactly like ``evaluate_sequence`` (but without
# the input‑sanitising steps, which are unnecessary for our generated
# candidates).  This speeds up the inner loop dramatically.
# ----------------------------------------------------------------------
def _fast_objective(seq: np.ndarray) -> float:
    """Return the same value as ``evaluate_sequence`` for a valid sequence."""
    n = seq.shape[0]
    if n == 0:
        return np.inf
    # convolution via FFT (real–valued input)
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed (unused, kept for compatibility).
    budget_s : int
        Time budget in seconds.  The function returns the best sequence
        found before the deadline.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1) Coarse grid over length (n) and power‑law exponent (p).
    # ------------------------------------------------------------------
    # Lengths that are cheap to convolve but still give a good sum.
    n_candidates = [50, 70, 80, 100, 120, 150, 180,
                    200, 250, 300, 400, 500, 800, 1200, 2000, 4000]

    p_low, p_high = 0.55, 0.78

            score = _fast_objective(seq)

            if time.time() > deadline:

    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

        p_fine_low = max(p_low, best_p - 0.04)
