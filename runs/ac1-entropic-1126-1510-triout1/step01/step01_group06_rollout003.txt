    score = 2 * n * max(conv(a, a))

   is performed and the result is renormalised.  The step size η decays
   as 1/√t.  This converges quickly on the convex problem
   min max_k (a∗a)_k.

        subject to g ≥ 0,

   The LP pushes mass into indices where f is tiny while keeping the
   mixed convolution bounded.  The solution g is mixed back:
        f ← (1‑t) f + t g  (t≈0.2).

The whole search runs for the allotted time (≈ 1000 s), keeping the best
sequence seen.  A final long exponentiated‑gradient refinement is applied
to the global best before returning the list of floats.
"""

import time
import numpy as np

# ----- Helper functions -------------------------------------------------

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray):
    """Return (max value, array of indices achieving the max) of a∗a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = np.max(conv)
    idx = np.where(np.abs(conv - max_val) < 1e-14)[0]
    return float(max_val), idx.astype(int)


def _gradient_for_max(a: np.ndarray, max_idx):
    """Sub‑gradient of the max‑convolution w.r.t. a."""
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # derivative of a_i*a_j (or 2*a_i*a_j when i≠j)
        grad[i] += 2.0 * a[j]
    return grad


def _score(a: np.ndarray) -> float:
    """Score = 2 * n * max(conv(a,a)) (sum(a) is assumed to be 1)."""
    max_c, _ = _conv_max_fft(a)
    return 2.0 * a.size * max_c


def _exp_gradient_descent(a0: np.ndarray, iters: int, deadline: float):
    """Exponentiated‑gradient descent on the simplex."""
    a = a0.copy()
    best = a.copy()
    best_score = _score(a)

    for t in range(1, iters + 1):
        if time.time() >= deadline:
            break
        max_c, max_idx = _conv_max_fft(a)
        grad = _gradient_for_max(a, max_idx)

        # step size η_t = η0 / sqrt(t)
        eta = 0.5 / np.sqrt(t)

        # multiplicative (exponentiated) update
        update = np.exp(-eta * grad)
        a = a * update
        a_sum = np.sum(a)
        if a_sum == 0.0:
            a = _project_to_simplex(a)   # safety fallback
        else:
            a = a / a_sum

        cur = _score(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            # early exit if already very good
            if best_score < 1.51:
                break

    return best, best_score


def _lp_spread_step(f: np.ndarray, bound: float):
    """One LP step that maximises Σ g_i under (f∗g)_k ≤ bound."""
    import cvxpy as cp

    n = f.size
    # Build the convolution matrix K where (K@g)[k] = (f∗g)[k]
    K = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        K[i:i + n, i] = f

    g = cp.Variable(n)
    constraints = [K @ g <= bound, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=2000, abstol=1e-9,
                   reltol=1e-9, feastol=1e-9, verbose=False)
    except Exception:
        return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if g_val.sum() == 0.0:
        return None
    g_val = g_val / g_val.sum()
    return g_val


# ----- Main search routine ----------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function.
    Returns a plain list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    best_score = float("inf")
    best_seq = None
    restart = 0

    # main restart loop -------------------------------------------------
    while time.time() < deadline:
        restart += 1
        # give each restart a bounded wall‑clock (≈30 s)
        restart_deadline = min(deadline, time.time() + 30.0)

        # ---- choose length and shape ---------------------------------
        n = np.random.randint(800, 2000)          # moderate length
        shape = np.random.choice(["geom", "power", "mixed", "random"])

        if shape == "geom":
            # geometric decay a_i = r**i
            t = np.random.uniform(0.0001, 0.003)
            r = np.exp(-t)
            a = r ** np.arange(n, dtype=float)

        elif shape == "power":
            # power law a_i = (i+1)^{-p}
            p = np.random.uniform(0.4, 1.8)
            a = (np.arange(1, n + 1, dtype=float) ** -p)

        elif shape == "mixed":
            # product of geometric & power‑law
            t = np.random.uniform(0.0001, 0.002)
            r = np.exp(-t)
            p = np.random.uniform(0.5, 1.5)
            a = (r ** np.arange(n, dtype=float)) * (
                np.arange(1, n + 1, dtype=float) ** -p
            )

        else:      # "random"
            a = np.random.gamma(1.0, 1.0, size=n)

        if np.random.rand() < 0.3:
            np.random.shuffle(a)

        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            continue
        a = a / a.sum()

        # ---- initial evaluation ---------------------------------------
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Restart {restart}] init  n={n:<4} shape={shape:<6}  score={cur_score:.6f}")

        # ---- exponentiated‑gradient refinement ------------------------
        a_ref, score_ref = _exp_gradient_descent(a, iters=2000,
                                                deadline=restart_deadline)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[Restart {restart}] EG    n={n:<4} score={score_ref:.6f}")

        # ---- optional LP “spread” phase ------------------------------
        f_curr = a_ref.copy()
        for lp_iter in range(5):
            if time.time() >= deadline or time.time() >= restart_deadline:
                break

            # bound = max(conv(f_curr, f_curr))
            bound, _ = _conv_max_fft(f_curr)

            # solve LP to obtain a more spread distribution g
            g = _lp_spread_step(f_curr, bound)
            if g is None:
                break

            # mix a little of g into f_curr
            t_mix = 0.2
            f_new = (1.0 - t_mix) * f_curr + t_mix * g
            f_new = f_new / f_new.sum()

            new_score = _score(f_new)
            if new_score < best_score:
                best_score = new_score
                best_seq = f_new.copy()
                print(f"[Restart {restart}] LP {lp_iter} n={n:<4} score={new_score:.6f}")

            # accept the new point only if it improves the local score
            if new_score < score_ref:
                f_curr = f_new
                score_ref = new_score
            else:
                break  # stop LP loop if no further improvement

        # early exit if we already reached a very good bound
        if best_score < 1.51:
            print("Target reached – terminating early.")
            break

    # ----- final long refinement on the global best --------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # use the remaining time for a final EG run (up to ~5000 iters)
        final_iters = min(5000, int(remaining * 30))  # rough budget
        final_seq, final_score = _exp_gradient_descent(best_seq,
                                                       iters=final_iters,
                                                       deadline=deadline)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq
            print(f"[Final] refined score={final_score:.6f}")

    # fallback – uniform distribution (should never be used)
    if best_seq is None:
        n = 1500
        best_seq = np.full(n, 1.0 / n)

    # Return as plain Python list
    return best_seq.tolist()
