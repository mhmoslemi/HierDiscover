"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Initial shape** –  try two simple parametric families:
   * Power‑law  a_i ∝ (i+1)^(-p)            (p∈(0,1))
   * Exponential a_i ∝ r^i                  (r∈(0,1))
   For the chosen length *n* we perform a cheap golden‑section search
   on the single parameter of each family and keep the best vector.

2. **Simulated‑annealing refinement** – starting from the best vector,
   repeatedly move a small amount of mass from one index to another.
   Moves are a mixture of
      * *targeted* moves: take mass from the index that contributes most
        to the current maximal convolution entry and place it at an index
        that contributes least.
      * *random* moves: pure exploration.
   A classic SA schedule (temperature decreasing linearly) allows
   occasional uphill steps, helping the search escape local minima.

3. **Final hill‑climb** – a short greedy pass that only accepts strictly
   improving random moves.  This cleans up the remaining tiny oscillations.

All helper functions are top‑level (no closures or lambdas) and the total
runtime respects the supplied *budget_s* (default 1000 s).  The returned
list consists of non‑negative numbers; the evaluation score is printed for
reference.
"""

import time
import math
import random
import numpy as np

# --------------------------------------------------------------------------- #
# 1.  Simple parametric families
# --------------------------------------------------------------------------- #

def power_law_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


def exponential_sequence(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i, normalised to sum 1 (0<r<1)."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


# --------------------------------------------------------------------------- #
# 2.  Scoring function (exactly the same as evaluate_sequence)
# --------------------------------------------------------------------------- #

def sequence_score(vec: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative vector."""
    s = vec.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(vec, vec)
    max_conv = conv.max()
    return float(2.0 * len(vec) * max_conv / (s * s))

# 3.  Golden‑section searches for the two families
# --------------------------------------------------------------------------- #

def golden_section_powerlaw(n: int, left: float, right: float,
                           max_iters: int, time_limit: float):
    """One‑dimensional optimisation of the exponent `p`."""
    phi = (np.sqrt(5) - 1) / 2          # 0.618...
    a = left
    b = right
    p1 = a + (1 - phi) * (b - a)
    p2 = a + phi * (b - a)
    v1 = sequence_score(power_law_sequence(n, p1))
    v2 = sequence_score(power_law_sequence(n, p2))

    best_p, best_score, best_vec = (p1, v1, power_law_sequence(n, p1)) \
        if v1 <= v2 else (p2, v2, power_law_sequence(n, p2))

    it = 0
    while it < max_iters and time.time() < time_limit:
        if v1 < v2:
            b = p2
            p2 = p1
            v2 = v1
            p1 = a + (1 - phi) * (b - a)
            v1 = sequence_score(power_law_sequence(n, p1))
            if v1 < best_score:
                best_score, best_p = v1, p1
                best_vec = power_law_sequence(n, p1)
        else:
            a = p1
            p1 = p2
            v1 = v2
            p2 = a + phi * (b - a)
            v2 = sequence_score(power_law_sequence(n, p2))
            if v2 < best_score:
                best_score, best_p = v2, p2
                best_vec = power_law_sequence(n, p2)
        it += 1
    return best_p, best_score, best_vec


def golden_section_exponential(n: int, left: float, right: float,
                               max_iters: int, time_limit: float):
    """Optimise the decay factor `r` (0<r<1)."""
    phi = (np.sqrt(5) - 1) / 2
    a = left
    b = right
    r1 = a + (1 - phi) * (b - a)
    r2 = a + phi * (b - a)
    v1 = sequence_score(exponential_sequence(n, r1))
    v2 = sequence_score(exponential_sequence(n, r2))

    best_r, best_score, best_vec = (r1, v1, exponential_sequence(n, r1)) \
        if v1 <= v2 else (r2, v2, exponential_sequence(n, r2))

    it = 0
    while it < max_iters and time.time() < time_limit:
        if v1 < v2:
            b = r2
            r2 = r1
            v2 = v1
            r1 = a + (1 - phi) * (b - a)
            v1 = sequence_score(exponential_sequence(n, r1))
            if v1 < best_score:
                best_score, best_r = v1, r1
                best_vec = exponential_sequence(n, r1)
        else:
            a = r1
            r1 = r2
            v1 = v2
            r2 = a + phi * (b - a)
            v2 = sequence_score(exponential_sequence(n, r2))
            if v2 < best_score:
                best_score, best_r = v2, r2
                best_vec = exponential_sequence(n, r2)
        it += 1
    return best_r, best_score, best_vec

def random_move(vec: np.ndarray, move_frac: float) -> np.ndarray:
    """Transfer a random fraction of mass from i to j (preserve total sum)."""
    n = len(vec)
    i = np.random.randint(n)
    # avoid source with zero mass
    while vec[i] == 0.0:
        i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    eps = np.random.rand() * move_frac * vec[i]
    cand = vec.copy()
    cand[i] -= eps
    cand[j] += eps
    # tiny renormalisation to keep the sum exactly 1
    s = cand.sum()
    if s != 1.0:
        cand /= s
    return cand


def targeted_move(vec: np.ndarray, move_frac: float) -> np.ndarray:
    """
    Move a small amount of probability mass from the index that contributes
    most to the current maximal convolution entry to an index that contributes
    least.
    """
    n = len(vec)
    conv = np.convolve(vec, vec)
    kmax = int(np.argmax(conv))

    idx = np.arange(n)
    partner = kmax - idx
    mask = (partner >= 0) & (partner < n)

    contrib = np.zeros(n, dtype=np.float64)
    contrib[mask] = vec[idx[mask]] * vec[partner[mask]]

    # source = index with maximal contribution (and positive mass)
    src_candidates = np.where(contrib == contrib.max())[0]
    src = np.random.choice(src_candidates)
    if vec[src] == 0.0:
        # fallback to a random move if source is empty
        return random_move(vec, move_frac)

    # target = index with minimal contribution (avoid picking the same)
    tgt_candidates = np.where(contrib == contrib.min())[0]
    # remove src from possible targets
    tgt_candidates = tgt_candidates[tgt_candidates != src]
    if tgt_candidates.size == 0:                     # very unlikely
        tgt = (src + n // 2) % n
    else:
        tgt = np.random.choice(tgt_candidates)

    eps = np.random.rand() * move_frac * vec[src]
    cand = vec.copy()
    cand[src] -= eps
    cand[tgt] += eps

    # renormalise (tiny drift from floating point)
    s = cand.sum()
    if s != 1.0:
        cand /= s
    return cand


# --------------------------------------------------------------------------- #
# 5.  Simulated‑annealing optimisation
# --------------------------------------------------------------------------- #

def simulated_annealing(start_vec: np.ndarray,
                        time_deadline: float,
                        move_frac_start: float = 0.10,
                        move_frac_end: float = 0.001,
                        accept_worse_ratio: float = 0.05) -> np.ndarray:
    """
    SA that mixes random and targeted moves.
    - move_frac decays linearly from start to end.
    - Temperature starts at `accept_worse_ratio * start_score` and
      decays linearly to 0.
    Returns the best vector seen.
    """
    cur = start_vec.copy()
    cur_score = sequence_score(cur)
    best = cur.copy()
    best_score = cur_score

    n = len(cur)
    start_time = time.time()
    total_time = time_deadline - start_time
    if total_time <= 0:
        return best

    # benchmark a single evaluation to get a sensible iteration budget
    t0 = time.time()
    _ = sequence_score(cur)
    eval_time = max(time.time() - t0, 1e-6)

    # allow roughly (remaining / 2*eval_time) iterations – keep some slack
    max_iters = int((time_deadline - time.time()) / (eval_time * 2.0))
    max_iters = max(1, max_iters)

    init_temperature = accept_worse_ratio * cur_score
    for it in range(max_iters):
        now = time.time()
        if now >= time_deadline:
            break
        # linear schedule
        progress = (now - start_time) / total_time
        temperature = init_temperature * (1.0 - progress)
        move_frac = move_frac_start + (move_frac_end - move_frac_start) * progress

        # 70 % targeted moves, 30 % pure random
        if random.random() < 0.7:
            cand = targeted_move(cur, move_frac)
        else:
            cand = random_move(cur, move_frac)

        cand_score = sequence_score(cand)

        # acceptance test
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand, cand_score
        else:
            # occasional uphill moves
            if temperature > 0.0:
                prob = math.exp(-(cand_score - cur_score) / temperature)
                if random.random() < prob:
                    cur, cur_score = cand, cand_score
        # optional early‑stop if we already beat the target
        if best_score <= 1.503 * 2.0:   # (score = 2*n*max) target ≈ 1.503 → product ≈0.7515
            break
    return best


# --------------------------------------------------------------------------- #
# 6.  Final greedy hill‑climb (strictly improving moves only)
# --------------------------------------------------------------------------- #

def hill_climb(vec: np.ndarray, time_deadline: float,
               move_frac: float = 0.02, max_no_improve: int = 3000) -> np.ndarray:
    """Very small greedy local search."""
    n = len(vec)
    cur = vec.copy()
    cur_score = sequence_score(cur)
    no_improve = 0
    while time.time() < time_deadline and no_improve < max_no_improve:
        cand = random_move(cur, move_frac)
        cand_score = sequence_score(cand)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:
            no_improve += 1
    return cur

        * builds a good initial shape via a short GP search,
        * refines it with simulated annealing + a final hill‑climb,
        * prints the best score found.
    """
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------- #
    # 7.1 Choose a length
    # --------------------------------------------------------------- #
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000  # fallback length if the global is missing

    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin

    # --------------------------------------------------------------- #
    # 7.2 Base families – power‑law and exponential
    # --------------------------------------------------------------- #
    # power‑law search
    left_p, right_p = 0.05, 0.95
    p_best, p_score, p_vec = golden_section_powerlaw(
        n, left_p, right_p, max_iters=30, time_limit=deadline)

    # exponential search (r in (0.90, 0.9999))
    r_best, r_score, r_vec = golden_section_exponential(
        n, 0.90, 0.9999, max_iters=30, time_limit=deadline)

    # also keep the historic global construction if it exists
    try:
        hist_vec = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        hist_score = sequence_score(hist_vec)
    except Exception:
        hist_vec = None
        hist_score = float('inf')

    # pick the best of the three
    candidates = [(p_score, p_vec), (r_score, r_vec)]
    if hist_vec is not None:
        candidates.append((hist_score, hist_vec))

    best_score, best_vec = min(candidates, key=lambda t: t[0])
    print(f"[init] best start score = {best_score:.6f} (n={n})")

    # --------------------------------------------------------------- #
    # 7.3 Simulated annealing refinement (major part of the budget)
    # --------------------------------------------------------------- #
    # allocate ~70 % of the remaining time to SA
    remaining = deadline - time.time()
    sa_deadline = time.time() + remaining * 0.70
    refined_vec = simulated_annealing(best_vec, sa_deadline)
    refined_score = sequence_score(refined_vec)
    if refined_score < best_score:
        best_vec, best_score = refined_vec, refined_score
        print(f"[SA ] improved score = {best_score:.6f}")

    # --------------------------------------------------------------- #
    # 7.4 Final greedy hill‑climb (the last ~20 % of the time)
    # --------------------------------------------------------------- #
    hc_deadline = min(deadline, time.time() + remaining * 0.20)
    final_vec = hill_climb(best_vec, hc_deadline)
    final_score = sequence_score(final_vec)
    if final_score < best_score:
        best_vec, best_score = final_vec, final_score
        print(f"[HC ] final improvement = {best_score:.6f}")

    # --------------------------------------------------------------- #
    # 7.5 Output
    # --------------------------------------------------------------- #
    print(f"[final] best score obtained: {best_score:.6f} with length {n}")
    # ensure plain Python list output
    return best_vec.tolist()
