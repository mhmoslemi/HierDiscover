def _conv_max_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_val = np.max(conv)
    # Indices where near max
    eps = 1e-12
    idx = np.where(np.abs(conv - max_val) < eps)[0]
    return float(max_val), idx

def subgradient_opt(a0, n, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = objective_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c
        # Compute gradient
        grad = np.zeros_like(a)
        # for each k in max_idx, add contributions
        # we can vectorize: for each k compute contributions directly.
        # Approach: for each k, compute i range as described.
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi+1)
            j = k - i
            grad[i] += 2.0 * a[j]
        # step size
        eta0 = 0.2 / np.sqrt(it)
        improved = False
        # try decreasing multiples of eta0
        for factor in (1.0, 0.5, 0.2, 0.1):
            eta = eta0 * factor
            a_candidate = _project_to_simplex(a - eta * grad)
            cand_score = objective_score(a_candidate)

                a = a_candidate

        if not improved:

"""
Hybrid sub‑gradient + linear‑programming refinement for minimizing
    2·n·max(conv(a,a)) / (sum(a)^2).

The algorithm works in two phases.
1.  Many short restarts run a fast projected sub‑gradient descent
    (gradient of the max‑convolution) on random initial shapes.
    This quickly discovers good low‑peak structures.
2.  The best sequence found is then refined by a small number of
    linear‑programming steps.  For a fixed f the LP
        max   Σ g_j
        s.t.  f * g ≤ M   (pointwise),   g ≥ 0
    pushes mass to indices where f is small while keeping the
    self‑convolution bound M.  The solution g is mixed with f
    (convex combination) and the process repeats a few times.
Both phases respect the simplex constraint (non‑negative entries,
sum = 1).  The whole search stops before the given time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # all entries are zero or negative
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray):
    """Return (max(conv(a,a)), indices where the max is attained)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = float(np.max(conv))
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return max_val, idx.astype(int)


def _objective_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – the quantity to minimise (sum(a)=1)."""
    n = a.size
    max_c, _ = _conv_max_fft(a)
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
# Sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_opt(a0: np.ndarray, n: int, deadline: float):
    """Projected sub‑gradient descent with a simple back‑tracking line search."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c

        # gradient of the max‑convolution (active set only)
        grad = np.zeros_like(a)
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            grad[i] += 2.0 * a[j]

        # decreasing step size
        eta0 = 0.2 / np.sqrt(it)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            eta = eta0 * factor
            cand = _project_to_simplex(a - eta * grad)
            cand_score = _objective_score(cand)
            if cand_score < cur_score - 1e-12:
                a = cand
                cur_score = cand_score
                improved = True
                break

        # tiny random nudge if no improvement
        if not improved:
            a_pert = a.copy()
            idx = np.random.randint(n, size=5)
            a_pert[idx] += np.random.randn(5) * 0.001
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / np.sum(a_pert)
            cand_score = _objective_score(a_pert)
            if cand_score < cur_score:
                a = a_pert
                cur_score = cand_score

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # early stop if we already beat the target
        if best_score < 1.5:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   max Σ g_j   s.t.   conv(f,g) ≤ M  (pointwise),   g ≥ 0.
    Returns the optimal g (may be None on failure).
    """
    n = f.size
    # avoid exact zeros – they would make the LP unbounded
    f_adj = np.maximum(f, 1e-12)

    # Build the (2n‑1)×n constraint matrix A where (A g)_k = (f*g)_k
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(f_adj, n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                     # maximise sum(g)  → minimise -sum(g)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True})
    except Exception as e:
        print(f"Linprog exception: {e}")
        return None

    if not res.success:
        print(f"LP not successful: {res.message}")
        return None
    return np.maximum(res.x, 0.0)


def _lp_refine(a0: np.ndarray, n: int, deadline: float, max_iters: int = 5):
    """
    Repeatedly improve a by solving the LP described above and mixing.
    Stops early if no gain or time runs out.
    """
    a = a0.copy()
    best_score = _objective_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M, _ = _conv_max_fft(a)

        g = _solve_lp_max_sum(a, M)
        if g is None:
            break
        sum_g = np.sum(g)
        if sum_g <= 0:
            break
        g_norm = g / sum_g

        # try mixing factors (larger steps first)
        improved = False
        for t in (0.5, 0.25, 0.125, 0.0625):
            a_new = (1 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / np.sum(a_new)
            new_score = _objective_score(a_new)
            if new_score < best_score - 1e-8:
                a = a_new
                best_score = new_score
                improved = True
                break
        if not improved:
            # no mixing factor helped – stop the refinement
            break
        # optional tiny sub‑gradient polish after each LP step
        if time.time() < deadline:
            sub_dead = min(deadline, time.time() + 2.0)
            a, best_score = _subgradient_opt(a, n, sub_dead)
    return a, best_score
