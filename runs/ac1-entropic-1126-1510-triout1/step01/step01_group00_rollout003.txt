def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2
    try: n0 = len(height_sequence_1)
    except: n0 = 2000

    # Candidate lengths
    lengths = sorted(set([max(500, int(n0 * factor)) for factor in (0.5, 1.0, 1.5, 2.0) if int(n0*factor) <= 30000]))
    # maybe also include 8000 and 12000.

    best_seq = None
    best_score = float('inf')
    # main loop over lengths
    for n in lengths:
        # compute time slice
        remaining = deadline - time.time()
        if remaining <= 0: break
        # allocate maybe 0.2*remaining min.
        slice_deadline = time.time() + max(30.0, remaining * 0.25)  # allocate at least 30 sec
        seq, score = _optimise_with_softmax(n, slice_deadline)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"New global best after length {n}: {best_score}")
        # if score < target try break
        if best_score <= 1.503:
            break

    if best_seq is None:
        # fallback uniform
        n = 500
        best_seq = np.full(n, 1./n)

"""
Propose a low‑score step‑function.

Algorithm
---------
1.  Initialise a non‑negative vector `a` with the discrete arcsine law
    (the known near‑optimal shape).

2.  Optimise the vector by minimising a smooth approximation of the
    objective  `2·n·max(a*a)`.  The max is replaced by a
    log‑sum‑exp (soft‑max) with a temperature parameter `γ`.  For small
    `γ` the function is almost the mean, for large `γ` it approaches the
    true max.  We start with a small `γ` and increase it during the run.

3.  The gradient of the smooth objective is
        ∇ = 4·n·(a ⋆ w)      with   w_i ∝ exp(γ·b_i) ,  b = a*a
    where `⋆` denotes the (cross‑)correlation.  The correlation is
    obtained cheaply by a single convolution of the reversed `a` with `w`.

4.  After each gradient step we project back onto the probability simplex
    (non‑negative entries with ℓ₁‑mass 1).  Projection is done with the
    classic O(n log n) Euclidean‑projection algorithm.

5.  The optimisation is performed for several candidate lengths
    (derived from the previously known construction).  Each length gets a
    time slice; the best sequence over all lengths is returned.

The whole routine respects the 1000 s budget, uses only NumPy, and never
accesses the file system.  It returns a plain Python list as required
by the evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean projection)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path – already feasible
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Discrete arcsine distribution (probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    """Return length‑n probability vector approximating the arcsine law."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))   # avoid the exact singularities
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Core optimiser for a single length using a smooth max (log‑sum‑exp)
def _optimise_one_length_softmax(
    n: int,
    deadline: float,
    init: str = "arcsine",
    max_iters: int = 10000,
    eta0: float = 0.8,
    gamma0: float = 0.5,
    gamma_max: float = 30.0,
    gamma_inc: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Optimise a length‑n sequence (ℓ₁‑mass 1) by gradient descent on a
    soft‑max approximation of the max‑convolution.
    Returns the best sequence found and its true evaluate_sequence score.
    """
    # ---- initialise ----------------------------------------------------
    if init == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init == "random":
        a = np.random.rand(n)
        a /= a.sum()
    else:                     # arcsine
        a = _arcsine_distribution(n)

    # bookkeeping
    best_seq = a.copy()
    # true evaluation (sum==1, so denominator = 1)
    b_full = np.convolve(a, a)
    best_score = float(2.0 * n * b_full.max())

    gamma = gamma0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and true max
        b_full = np.convolve(a, a)          # length 2n‑1
        k_max = int(np.argmax(b_full))
        cur_score = 2.0 * n * b_full[k_max]

        # keep the best true score seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n} it={it} score={best_score:.6f} γ={gamma:.3f}"
            )

        # ----- soft‑max weights -----------------------------------------
        # shift by max to avoid overflow
        b_shift = b_full - b_full.max()
        w = np.exp(gamma * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical safeguard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                         # now Σ w = 1

        # ----- gradient of smoothed objective ---------------------------
        # correlation a ⋆ w  (cross‑correlation)
        a_rev = a[::-1]
        corr_full = np.convolve(a_rev, w)   # length 3n‑2
        # extract the n entries needed (indices n‑1 … n‑1+n‑1)
        grad = 4.0 * n * corr_full[n - 1 : n - 1 + n]

        # ----- step size ------------------------------------------------
        eta = eta0 / np.sqrt(it)

        # ----- gradient step + projection --------------------------------
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        # ----- slowly raise the temperature (γ) -------------------------
        if gamma < gamma_max:
            gamma = min(gamma_max, gamma + gamma_inc)

        # ----- occasional jitter to escape plateaus --------------------
        if it % 400 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    # final true evaluation for the best sequence (in case it changed)
    final_score = float(2.0 * n * np.max(np.convolve(best_seq, best_seq)))
    return best_seq, final_score

        n0 = 2000

    # candidate lengths – modest range around the historic size
    factors = (0.5, 0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted({max(200, int(n0 * f)) for f in factors if int(n0 * f) > 0})
    # also add a couple of absolute values in case n0 is very small
    if not lengths:
        lengths = [1000, 2000, 4000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # allocate a slice of the remaining time to each length (≈20 %)
    for n in lengths:
        if time.time() >= deadline:
            break
        # give each length at least 30 s or a fifth of the remaining budget
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(30.0, remaining * 0.2)

        # run optimisation from a few different starts (arcsine, uniform)
        for init in ("arcsine", "uniform"):
            seq, sc = _optimise_one_length_softmax(
                n=n,
                deadline=slice_deadline,
                init=init,
                max_iters=8000,
                eta0=0.8,
                gamma0=0.5,
                gamma_max=30.0,
                gamma_inc=0.02,
            )
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = seq.copy()
                print(
                    f"*** New global best – length={n}, init={init}, score={overall_best_score:.6f}"
                )
            # early exit if we have already beaten the target
            if overall_best_score <= 1.503:
                break
        if overall_best_score <= 1.503:
            break

    # fallback – should never happen, but keep a safe default
    if overall_best_seq is None:
        n_fallback = 500
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = float(2.0 * n_fallback * np.max(np.convolve(overall_best_seq, overall_best_seq)))

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
