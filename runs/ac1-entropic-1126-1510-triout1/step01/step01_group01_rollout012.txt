import cvxpy as cp

def lp_step(f, M):
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    constraints = []
    # create convolution constraints
    for k in range(2*n-1):
        # compute dot product of part of f and b:
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # indices i where f[i] multiplies b[k-i]
        # create expression sum_i f[i] * b[k-i]
        expr = 0
        for i in range(i_start, i_end+1):
            expr += f[i] * b[k - i]
        constraints.append(expr <= M)
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    if b.value is None:
        return None
    return np.array(b.value).flatten()
```

"""
Hybrid smooth‑gradient descent with momentum & random restarts.

The objective is 2·n·max(a∗a)/(∑a)².
Because the denominator equals 2·n when ∑a = sqrt(2n), the score is simply
max(a∗a).  We therefore keep the ℓ¹‑norm fixed (project onto the simplex) and
minimise the maximal coefficient of the self‑convolution.

The algorithm combines three ideas:

1. **Smooth max** – replace max by a log‑sum‑exp with a temperature β.
   The gradient of this smoothed objective is cheap (two convolutions) and
   provides information from *all* convolution entries, not only the peak.

2. **Momentum & adaptive step size** – a heavy‑ball update
   v ← μ·v – η·∇ improves convergence compared with plain projected GD.

3. **Multi‑restart + local swaps** – several independent runs (different
   random seeds, different lengths) are performed while time permits.
   After the descent a quick pair‑wise mass‑swap search is applied to escape
   tiny plateaux.

The code respects the 1000 s budget, prints occasional progress messages,
and finally returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex Σx = z , x≥0."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)                # 1‑based
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all zero
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smooth max  f_β(a)= (1/β)·log Σ_k exp(β·(a∗a)[k]).
    Returns (grad, max_conv) where max_conv = max_k (a∗a)[k].
    """
    conv = np.convolve(a, a)                 # length 2n‑1
    max_c = conv.max()
    # Stable soft‑max weights
    exp_vals = np.exp(beta * (conv - max_c))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:                       # should never happen
        p = np.full_like(conv, 1.0 / conv.size)
    else:
        p = exp_vals / sum_exp               # probabilities Σp = 1

    # grad_i = 2·Σ_j p_{i+j}·a_j  –  a_i·p_{2i}
    a_rev = a[::-1]
    conv2 = np.convolve(p, a_rev)            # length 3n‑2
    n = a.shape[0]
    # slice positions n‑1 … n‑1+n‑1  (inclusive)
    grad_base = conv2[n - 1 : n - 1 + n]
    diag = p[::2][:n]                        # p_{2i}
    grad = 2.0 * grad_base - a * diag
    return grad, max_c


# --------------------------------------------------------------
def _grad_topk(a: np.ndarray, top_k: int = 5):
    """
    Sub‑gradient based only on the `top_k` largest convolution entries.
    Returns (grad, max_conv).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(a)
    i = np.arange(n)

    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad, max_c


# --------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Return max_k (a∗a)[k]  (fast wrapper)."""
    return np.convolve(a, a).max()


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The routine performs several independent
    gradient‑descent runs (smooth‑max + momentum) on different lengths,
    finishes each run with a cheap pair‑wise swap local search and
    returns the best vector found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.15   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Choose lengths to explore.  Use the reference length if available.
    try:
        ref_len = len(height_sequence_1)      # type: ignore[name-defined]
        lengths = [ref_len,
                   max(20, int(ref_len * 0.75)),
                   max(20, int(ref_len * 1.25))]
    except Exception:
        # No reference – fall back to a few generic sizes
        lengths = [1000, 2000, 3000, 4000]

    # Remove duplicates, shuffle order, keep everything >= 10
    lengths = sorted({max(10, int(L)) for L in lengths})
    np.random.shuffle(lengths)

    # ------------------------------------------------------------------
    # Main multi‑restart loop
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Number of independent attempts for this length.
        # We keep the total time budget in mind – at most 3 attempts per size.
        attempts = 3
        for attempt in range(attempts):
            if time.time() >= deadline:
                break

            # ----------------------------------------------------------
            # Initialise a candidate.
            # Use the reference vector when it matches the current length,
            # otherwise start from a fresh random vector.
            if (
                attempt == 0
                and "height_sequence_1" in globals()
                and len(height_sequence_1) == n
            ):
                a = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            else:
                a = np.random.rand(n)

            a = _project_to_simplex(a, target_sum)

            # Best found *inside* this run.
            best_a_local = a.copy()
            best_score_local = _max_conv(best_a_local)

            # ----------------------------------------------------------
            # Parameters for the smooth‑gradient phase
            beta = 1.0
            beta_max = 25.0
            beta_inc = 0.0002          # per iteration – gradual tightening
            topk_fallback_iter = 200000  # after this switch to top‑k exact grad
            eta0 = target_sum * 0.6 / n   # base step length
            decay = 6e-5
            mu = 0.85                 # momentum coefficient
            v = np.zeros_like(a)      # momentum buffer
            it = 0
            no_improve = 0
            max_iters = 3_000_000    # hard cap; actual stop is time based

            while time.time() < deadline and it < max_iters:
                # ------------------------------------------------------
                # Choose gradient computation method
                if it < topk_fallback_iter:
                    grad, cur_max = _smooth_gradient(a, beta)
                else:
                    # near the end focus on the true max (more precise)
                    grad, cur_max = _grad_topk(a, top_k=12)

                # Step size with simple 1/(1+decay·t) decay
                eta = eta0 / (1.0 + decay * it)

                # Heavy‑ball update
                v = mu * v - eta * grad
                a_new = a + v
                a_new = _project_to_simplex(a_new, target_sum)

                # Evaluate the *real* max‑convolution (the true score)
                cur_score = _max_conv(a_new)

                # Record improvement
                if cur_score + 1e-9 < best_score_local:
                    best_score_local = cur_score
                    best_a_local = a_new.copy()
                    no_improve = 0
                    print(
                        f"[len{n} it{it:07d}] new best local score {best_score_local:.6f}"
                    )
                else:
                    no_improve += 1

                # Occasionally increase the temperature β (sharper approximation)
                if beta < beta_max and it % 2500 == 0:
                    beta = min(beta_max, beta * (1.0 + beta_inc * 2500))

                # Random perturbations to escape stagnation
                if (it % 5000 == 0 and it > 0) or (no_improve > 4000):
                    noise = np.random.randn(n) * target_sum * 6e-4
                    a_new = a_new + noise
                    a_new = _project_to_simplex(a_new, target_sum)
                    no_improve = 0
                    print(f"[len{n} it{it}] random perturbation")

                a = a_new
                it += 1

            # ----------------------------------------------------------
            # Local pair‑wise swap search (cheap hill‑climbing)
            a_local = best_a_local.copy()
            cur_local = best_score_local
            swap_iters = 2500
            for _ in range(swap_iters):
                i, j = np.random.randint(0, n, size=2)
                if i == j:
                    continue
                # Move a small fraction of mass from i to j
                delta = a_local[i] * np.random.rand() * 0.04
                if delta <= 0.0:
                    continue
                cand = a_local.copy()
                cand[i] -= delta
                cand[j] += delta
                # (projection not required – sum unchanged, non‑neg holds)
                cand_score = _max_conv(cand)
                if cand_score + 1e-10 < cur_local:
                    a_local = cand
                    cur_local = cand_score
                    print(
                        f"[len{n}] swap improvement → {cur_local:.6f}"
                    )

            # ----------------------------------------------------------
            # Update the global best
            if cur_local < best_score:
                best_score = cur_local
                best_seq = a_local.tolist()
                print(
                    f"[global] improved best score {best_score:.6f} (length {n})"
                )

    # ------------------------------------------------------------------
    # Fallback – if everything failed, return a simple uniform vector.
    if best_seq is None:
        n = 1500
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a.tolist()
        best_score = _max_conv(a)
        print("[fallback] uniform vector used, score =", best_score)

    print(f"[final] best score = {best_score:.6f}")
    return best_seq
