"""
Propose a low‑score step function.

The algorithm consists of two complementary stochastic phases.

1. Simulated‑Annealing (SA) phase
   – Starts from the best known construction (or a random one).
   – Randomly moves a small amount of mass between two indices while
     preserving the ℓ¹‑norm √(2 n).  The objective is the maximum entry of
     the self‑convolution, i.e. ``max(conv(a,a))``.
   – An exponential temperature schedule allows uphill moves early on,
     helping the search escape local minima.

2. Projected‑gradient descent (PGD) polishing phase
   – Uses the sub‑gradient of the current maximal convolution entries,
     takes a step opposite to it and projects back onto the simplex
     {a ≥ 0, Σa = √(2 n)}.
   – Guarantees monotone improvement (when a better point is found).

Both phases keep the ℓ¹‑norm fixed, so ``evaluate_sequence`` simply
returns the maximal convolution value.  The code respects the 1000 s
budget, prints progress, and finally returns the best sequence found
as a plain Python list.

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of total mass z."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)            # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # all entries will be zero
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1      # +1 because zero‑based indexing
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _gradient_of_max(a: np.ndarray, top_k: int = 3):
    """Sub‑gradient of max_k (a*a)[k] and the corresponding maximum value."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()

    # indices of the `top_k` largest entries (including ties)
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(a)
    i = np.arange(n)

    for k in idxs:
        j = k - i                # partner index for each i
        mask = (j >= 0) & (j < n)

        # derivative = 2 * a_j for all valid pairs
        grad[mask] += 2.0 * a[j[mask]]

        # diagonal correction: a_i*a_i appears only once
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad, max_val


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """
    Return max_k (a*a)[k] using an FFT‑based convolution (O(n log n)).
    The result is a Python float.
    """
    n = a.shape[0]
    # length of zero‑padded FFT: next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    C = A * A
    conv = np.fft.irfft(C, n=L)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values
    conv = np.maximum(conv, 0.0)
    return float(conv.max())

        a0 = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        # Scale the reference to the exact target sum and project to the simplex
        a0 = a0 * (target_sum / np.sum(a0))
    else:
        a0 = np.random.rand(n)

    a0 = _project_to_simplex(a0, target_sum)

    # --------------------------------------------------------------
    # Book‑keeping of the best solution.
    best_a = a0.copy()
    best_score = _max_convolution(best_a)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # --------------------  Phase 1: Simulated Annealing  --------------------
    # Allocate about 60 % of the budget to SA.
    phase1_deadline = time.time() + 0.60 * (deadline - time.time())
    cur_a = best_a.copy()
    cur_score = best_score

    # Temperature schedule (exponential decay)
    T0 = max(0.1 * cur_score, 1e-3)    # start temperature
    T_end = 1e-9                       # final temperature (near zero)

    # Move scale: initial fraction of the donor entry that can be transferred
    move_scale_initial = 0.25
    move_scale_final = 0.001

    it = 0
    while time.time() < phase1_deadline:
        # ---- pick a donor with positive mass ----
        donor = np.random.randint(0, n)
        if cur_a[donor] <= 1e-12:
            continue

        # ---- pick a distinct receiver ----
        receiver = np.random.randint(0, n - 1)
        if receiver >= donor:
            receiver += 1

        # ---- annealing parameters ----
        elapsed = time.time() - start_time
        total_phase = phase1_deadline - start_time
        tau = elapsed / total_phase                 # in [0,1)
        T = T0 * (T_end / T0) ** tau                # exponential cooling
        move_scale = move_scale_initial * (1 - tau) + move_scale_final * tau

        # ---- random amount of mass to move ----
        # delta is at most move_scale·donor_mass
        delta = np.random.rand() * move_scale * cur_a[donor]

        # ---- propose a new candidate ----
        cand = cur_a.copy()
        cand[donor] -= delta
        cand[receiver] += delta

        cand_score = _max_convolution(cand)

        # ---- Metropolis acceptance ----
        improve = cand_score < cur_score
        if improve:
            accept = True
        else:
            # upward move acceptance probability
            # Guard against division by zero (T may become extremely small)
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[SA  ] iter={it:07d}  score={best_score:.6f}  time={elapsed:.1f}s")

        it += 1

    # --------------------------------------------------------------
    # ------------------- Phase 2: Projected Gradient -------------------
    # Allocate remaining time to PGD.
    remaining_time = deadline - time.time()
    if remaining_time > 5.0:
        # Hyper‑parameters
        top_k = 5                        # use a few active convolution peaks
        eta0 = target_sum * 0.3 / n      # base step size (≈0.3·average entry)
        decay = 5e-4                     # step‑size decay factor
        max_iters = 10 ** 7
        pgd_it = 0
        no_improve = 0

        a = best_a.copy()                # start PGD from the best SA point

        while time.time() < deadline and pgd_it < max_iters:
            grad, cur_max = _gradient_of_max(a, top_k=top_k)

            eta = eta0 / (1.0 + decay * pgd_it)

            a_new = a - eta * grad
            a_new = _project_to_simplex(a_new, target_sum)

            # evaluate new max convolution (fast, reuse gradient routine)
            _, new_max = _gradient_of_max(a_new, top_k=1)
            new_score = new_max                       # because sum is fixed

            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                no_improve = 0
                print(f"[PGD ] iter={pgd_it:07d}  improved score={best_score:.6f}")
            else:
                no_improve += 1

            # occasional random perturbation to jump out of plateaus
            if (pgd_it % 3000 == 0 and pgd_it > 0) or no_improve > 6000:
                noise = np.random.randn(n) * target_sum * 5e-4
                a_new = a_new + noise
                a_new = _project_to_simplex(a_new, target_sum)
                no_improve = 0
                print(f"[PGD ] iter={pgd_it:07d}  random perturbation")

            a = a_new
            pgd_it += 1

    # --------------------------------------------------------------
    # ----------- Final cheap random “block” exploration (optional) ----------
    block_trials = 150
    for _ in range(block_trials):
        nb = np.random.randint(2, 7)                # number of blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        cand_score = _max_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[block] improved score={best_score:.6f}")

    # --------------------------------------------------------------
    total_elapsed = time.time() - start_time
    print(f"[final] best score = {best_score:.6f} after {total_elapsed:.1f}s")
    return best_a.tolist()
