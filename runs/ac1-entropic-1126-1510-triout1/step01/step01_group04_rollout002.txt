"""
Propose a candidate step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Initialisation** – use the previously discovered best sequence if it
   exists, otherwise start from a power‑law distribution (exponent ≈‑0.55).

2. **Exponentiated‑gradient optimisation** – minimise a smooth surrogate of
   the max‑convolution using a soft‑max (parameter λ) and an exponentiated
   update that automatically respects the non‑negativity constraint.
   λ is increased gradually (annealing) so the optimisation first smooths
   the objective and then focusses on the true maximum.

3. **Fine‑grained stochastic hill‑climb** – after the gradient phase a short
   random local search moves a tiny fraction of mass between two positions
   and keeps any improvement.

All operations keep the vector on the probability simplex (sum = 1),
which makes the evaluation equal to ``2 * n * max(conv)``.
The routine respects the 1000 s time budget and prints progress reports.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+offset+1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = np.power(idx, exponent, where=idx > 0, out=np.zeros_like(idx))
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


def _convolve_self(v: np.ndarray) -> np.ndarray:
    """
    Fast self‑convolution using FFT.
    Returns a real array of length 2*len(v)-1.
    """
    n = v.shape[0]
    size = 2 * n - 1
    # real FFT is sufficient because the input is real.
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fv * fv, n=size)
    return conv


def _softmax_weights(x: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i ∝ exp(lam * (x_i - max(x))).
    Always returns a probability vector that sums to one.
    """
    max_val = x.max()
    # shifting by the maximum prevents overflow
    w = np.exp(lam * (x - max_val))
    s = w.sum()
    if s == 0.0:
        # fallback – all entries are equal → uniform weights
        return np.full_like(w, 1.0 / w.size)
    return w / s

        init = np.maximum(init, 1e-15)          # avoid exact zeros
        init /= init.sum()
        print(f"[init] Loaded height_sequence_1 (length {n})")
    else:
        # Fall back to a power‑law start.
        n = 8000                                 # a comfortable size
        init = _make_power_law(n, exponent=-0.55, offset=0.5)
        print(f"[init] Power‑law start (length {n})")

    cur = init.copy()
    best_seq = cur.copy()
    best_score = 2.0 * n * _convolve_self(cur).max()
    print(f"[init] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Exponentiated‑gradient phase (soft‑max annealing)
    # ------------------------------------------------------------------
    lambda_schedule = [5, 10, 20, 40, 80]   # increasing smoothness
    eta_initial = 0.8                       # base step size for exp‑gradient

    for lam in lambda_schedule:
        eta = eta_initial
        it = 0
        # at most a few thousand updates per λ – stop when time runs out.
        while time.time() < deadline and it < 3000:
            # self‑convolution and current objective value
            conv = _convolve_self(cur)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep the best seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"[grad] λ={lam:3d} iter={it:4d} new best = {best_score:.6f}")

            # soft‑max distribution over convolution entries
            w = _softmax_weights(conv, lam)

            # gradient of the weighted sum (2 * reversed‑convolution)
            a_rev = cur[::-1]
            grad_full = np.convolve(w, a_rev)           # length = 2n-1 + n -1
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # slice length n

            # limit gradient magnitude to keep the exponent stable
            grad = np.clip(grad, -1e3, 1e3)

            # exponentiated (multiplicative) update → stays non‑negative
            cur = cur * np.exp(-eta * grad)

            # project back onto the simplex (non‑negative and sum = 1)
            cur = np.maximum(cur, 1e-15)
            cur /= cur.sum()

            # simple step‑size adaptation
            if it % 200 == 0 and it > 0:
                # if no improvement in the last block, shrink η
                recent_score = 2.0 * n * _convolve_self(cur).max()
                if recent_score >= cur_score:
                    eta *= 0.9
                else:
                    eta *= 1.02
            it += 1

        # after finishing a λ level, inject a tiny random jitter to help
        # escape possible plateaus (kept very small so we stay near a good point)
        jitter = np.random.rand(n) * 1e-4
        cur = np.maximum(cur + jitter, 0.0)
        cur /= cur.sum()
        # decay the base step size for the next λ
        eta_initial *= 0.7

        # safety check
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3) Stochastic hill‑climb for fine‑grained local optimisation
    # ------------------------------------------------------------------
    cur_hill = best_seq.copy()
    hill_score = best_score
    it = 0
    while time.time() < deadline and it < 5000:
        i, j = np.random.randint(n, size=2)
        if i == j or cur_hill[i] < 1e-15:
            it += 1
            continue
        # move a tiny random fraction (≤3 %) of mass from i to j
        delta = cur_hill[i] * np.random.rand() * 0.03
        cur_hill[i] -= delta
        cur_hill[j] += delta

        # evaluate quickly: reuse convolution routine
        conv = _convolve_self(cur_hill)
        new_score = 2.0 * n * conv.max()

        if new_score < hill_score:
            hill_score = new_score
            # keep the move
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_hill.copy()
                print(f"[hill] iter={it:4d} new best = {best_score:.6f}")
        else:
            # reject – revert the move
            cur_hill[i] += delta
            cur_hill[j] -= delta

        it += 1
        if it % 1000 == 0:
            print(f"[hill] iteration {it}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence discovered
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = float(best_score)
    print(f"[done] length={len(final_seq)} final_score={final_score:.6f}")
    return final_seq
