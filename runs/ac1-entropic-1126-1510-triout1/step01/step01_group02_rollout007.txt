def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2

    # Choose length n adaptively based on time
    n = 12000  # maybe 15000

    # Initialize sequence
    seq = np.full(n, 1.0/n)  # uniform
    # Add small random noise
    seq += np.random.rand(n) * 1e-4
    seq = np.maximum(seq, 0)
    seq /= seq.sum()

    # Adam parameters
    alpha = 0.1  # initial step size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros(n)
    v = np.zeros(n)

    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())

    it = 0
    while time.time() < deadline:
        # compute conv via FFT
        conv = autocorr_fft(seq)
        k = np.argmax(conv)
        max_val = conv[k]

        # compute gradient (subgradient)
        g = np.zeros_like(seq)
        # index mapping
        # for i in range(n):
        #    j = k - i if 0 <= k - i < n else None
        #    if valid: g[i] = 2*seq[j]
        # Efficient vectorized:
        # create array of i indices
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        g[mask] = 2*seq[j_arr[mask]]

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        # update and project onto simplex
        seq = seq - step
        seq = _project_simplex(seq)

        # evaluate score occasionally (every few iterations) to reduce compute cost?
        # we need to compute conv each iteration anyway for gradient, which yields max.
        # So we already have max conv; but evaluate_sequence also uses full conv, but we can compute score directly: score = 2*n*max_val/(seq.sum()**2)

        score = 2 * n * max_val / (seq.sum()**2)  # seq sum=1
        if score < best_score:
            best_score = score

Implementation:

```
def _autocorr_fft(seq):
    n = seq.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    # conv length size, take first 2*n-1 entries (real)
    return conv[:2*n-1]
```

def propose_candidate(...):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    best_seq = None
    best_score = float('inf')
    n = 12000  # base length
    # Determine n based on budget and time? Use maybe 15000 if time left > 200 seconds else 8000.

    # compute fast conv using fft
    L = 1 << ((2*n - 1).bit_length())

    # Precompute length for FFT for repeated use
    while time.time() < deadline:
        # Initialize random seq (Dirichlet)
        seq = np.random.dirichlet(np.ones(n))

        # Adam optimizer parameters
        m = np.zeros(n)
        v = np.zeros(n)
        alpha = 0.5
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        it_local = 0
        last_improve_iter = 0
        # run for limited iter (e.g., 5000) or until time runs out
        while it_local < 5000 and time.time() < deadline:
            # compute conv via fft
            fft_seq = np.fft.rfft(seq, n=L)
            conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
            max_val = conv.max()
            k = conv.argmax()

            # compute gradient
            g = np.zeros_like(seq)
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            # indices i_range
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range
            g[i_range] = 2 * seq[j_range]

            it_local += 1
            # Adam updates
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1**it_local)
            v_hat = v / (1 - beta2**it_local)
            step = alpha * m_hat / (np.sqrt(v_hat) + eps)

            seq = seq - step
            # projection onto simplex
            seq = _project_simplex(seq)

            # compute score quickly
            score = 2 * n * max_val / (seq.sum() ** 2)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                last_improve_iter = it_local
                print(f"[Restart seq {len(best_seq)}] New best score {best_score:.6f} at iter {it_local}")

            # occasional random perturb for escaping
            if it_local % 1000 == 0:
                # small random jitter
                perturb = np.random.rand(n)
                perturb = perturb / perturb.sum()
                seq = _project_simplex(seq + 0.001 * perturb)

        # after iterating this restart, maybe if stagnated, break and continue to new random start
```

"""
Propose a step‑function that minimises ``evaluate_sequence``.

The optimiser works on the probability simplex (all entries ≥0, sum=1) and
combines two complementary ideas:

1. **Exponentiated gradient descent** – the sub‑gradient of the maximal
   convolution entry is  ``g_i = 2·a_{k-i}`` (zero when the index is out of
   range).  An exponentiated‑gradient (mirror descent with KL divergence)
   update ``a ← a·exp(−ηg)`` automatically stays on the simplex and
   respects non‑negativity.

2. **Stochastic pairwise swaps** – occasional random mass transfers
   between two coordinates help the search escape plateaux and improve
   exploration.

All heavy work (the convolution and the maximal entry) is performed by
FFT‑based autocorrelation, which runs in ``O(n log n)`` time.  The routine
runs many short optimisation runs (random restarts) until the time budget
expires and returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast autocorrelation via FFT (real input)
# ----------------------------------------------------------------------
def _autocorr_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full (linear) autocorrelation of ``seq`` using FFT."""
    # rfft works on real input and returns the non‑negative frequencies
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    # Only the first 2·len(seq)−1 entries are the linear convolution
    n = seq.shape[0]
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exponentiated‑gradient step that targets the currently maximal convolution entry
# ----------------------------------------------------------------------
def _exp_gradient_step(seq: np.ndarray, eta: float, fft_len: int) -> np.ndarray:
    """
    Perform one mirror‑descent step on the simplex.

    Parameters
    ----------
    seq : np.ndarray
        Current probability vector (sum = 1).
    eta : float
        Step‑size (positive).  Smaller values give finer updates.
    fft_len : int
        Length of the zero‑padded FFT (power of two ≥ 2·len(seq)−1).

    Returns
    -------
    np.ndarray
        Updated vector, automatically normalised to sum = 1.
    """
    n = seq.shape[0]

    # autocorrelation and the index of its maximal entry
    conv = _autocorr_fft(seq, fft_len)
    k = int(np.argmax(conv))

    # build sub‑gradient g_i = 2·seq_{k−i} for the valid indices
    g = np.zeros_like(seq)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * seq[k - idx]

    # exponentiated‑gradient update (mirror descent w.r.t. KL)
    factor = np.exp(-eta * g)
    seq = seq * factor
    total = seq.sum()
    if total <= 0.0:                 # numerical safeguard
        seq = np.full_like(seq, 1.0 / n)
    else:
        seq = seq / total
    return seq


# ----------------------------------------------------------------------
# Random pairwise mass‑swap (keeps sum exactly 1, stays on simplex)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass between two randomly chosen indices.
    The proposal is symmetric and preserves the total sum.
    """
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # Zero‑mean Gaussian‑like proposal (bounded to keep non‑negativity)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0.0:
        delta = min(delta, seq[j])   # take from j
    else:
        delta = -min(-delta, seq[i]) # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that strives
    to minimise ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds, using multiple random restarts, exponentiated
    gradient descent and occasional random pairwise swaps.
    """
    np.random.seed(seed)

    start = time.time()
    # Keep a tiny safety margin for the harness
    deadline = start + max(0.0, budget_s - 1.0)

    # --------------------------------------------------------------
    # Problem size – a few ten‑thousands works well within the budget.
    # --------------------------------------------------------------
    n = 12000                                   # length of the sequence
    # FFT length: next power of two ≥ 2·n‑1 (optimal for np.fft)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Global best holder
    best_seq = None
    best_score = float("inf")
    global_iter = 0

    # --------------------------------------------------------------
    # Main optimisation loop – repeatedly start from a fresh random point.
    # --------------------------------------------------------------
    while time.time() < deadline:
        # --- initialise a random point on the simplex (Dirichlet(1,…,1)) ---
        seq = np.random.dirichlet(np.ones(n))

        # Local loop parameters
        local_iter = 0
        # we run a few thousand inner iterations or until the deadline
        max_local_iters = 5000

        while time.time() < deadline and local_iter < max_local_iters:
            # Choose move type
            if np.random.rand() < 0.7:
                # Exponentiated‑gradient step; step size slowly decays
                eta = 0.8 / np.sqrt(local_iter + 1)
                seq = _exp_gradient_step(seq, eta, fft_len)
            else:
                # Random pairwise swap; sigma slowly decays
                sigma = 0.12 / np.sqrt(local_iter + 1)
                seq = _random_pair_move(seq, sigma)

            # Evaluate current quality (fast – we reuse the autocorrelation)
            conv = _autocorr_fft(seq, fft_len)
            max_val = conv.max()
            # Because we normalise seq to sum = 1, the score simplifies to:
            score = 2.0 * n * max_val

            # Keep the globally best sequence seen so far
            if score < best_score:
                best_score = float(score)
                best_seq = seq.copy()
                print(f"[Iter {global_iter:6d}] New best score: {best_score:.6f}")

            # Simple stagnation handling – every 800 steps inject a tiny jitter
            if local_iter % 800 == 0 and local_iter > 0:
                jitter = np.random.rand(n)
                jitter = jitter / jitter.sum()
                # blend 1 % random jitter
                seq = 0.99 * seq + 0.01 * jitter
                seq = seq / seq.sum()  # re‑normalise (still on simplex)

            local_iter += 1
            global_iter += 1

        # End of one random‑restart run – continue with a fresh start
        # (no explicit restart condition needed; the outer ``while`` does it.)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).  The harness
    # will feed it to ``evaluate_sequence`` for the final score.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (never happens in practice)
        best_seq = np.full(n, 1.0 / n, dtype=float)

    return best_seq.tolist()
