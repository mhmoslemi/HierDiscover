    n = len(a)
    best_seq = a.copy()
    b = _conv_fft(a, a)
    best_score = 2.0 * n * b.max()
    a_rev = a[::-1]
    v = np.zeros_like(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute b and max
        b = _conv_fft(a, a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(...)
        # compute gradient
        if it <= 3000:
            avg = 1.0 / (2 * n - 1)
            d = b - avg
            conv = _conv_fft(d, a_rev)
            g = 4.0 * conv[n-1:n-1+n]
        else:
            alpha = min(50.0, 2.0 + (it - 3000) * (48.0 / 3000))
            # compute softmax weights
            w = np.exp(alpha * (b - max_b))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / len(b))
            else:
                w = w / w_sum

            g = 2.0 * conv[n-1:n-1+n]

        v = beta * v + (1 - beta) * g
        # step size
        eta = eta0 / np.sqrt(it)
        a = a - eta * v
        # ensure positivity, simplex projection
        a = _project_onto_simplex(a, s=1.0)

"""
Propose a step‑function (non‑negative heights) with a low evaluate_sequence score.

Algorithm
---------
1. Choose a few candidate lengths (≈4‑16 k).  
2. For each length search a one‑parameter family that mimics the continuous
   arcsine extremal:
        a_i(ε) ∝ 1/√((i+ε)*(n‑1‑i+ε)).
   A tiny grid on ε is scanned and the best (lowest max‑convolution) init is kept.
3. Starting from the best init, run a two‑phase projected gradient descent:
   * Phase 1 (first few thousand iterations) minimise the L²‑error
     ∑(a*a−avg)², whose gradient is 4·conv(d, a_rev).
   * Phase 2 switches to a smooth approximation of the ‑∞‑norm
     L = (1/α)·log∑exp(α·(a*a)), gradient = 2·conv(w, a_rev) where
     w is a soft‑max over the convolution entries.  α is increased
     gradually so the optimisation first smooths, then concentrates on the
     actual maximum.
   Momentum (β≈0.9) and a decreasing step‑size η₀/√t are used.
   After each step the vector is projected onto the probability simplex
   (non‑negative, sum = 1) by simple clipping + renormalisation.
4. Keep the best sequence seen during the whole run and return it.

All heavy work uses FFT‑based convolutions (O(n log n)) and fits comfortably
into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (clip‑negative and renormalise)
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x ≥ 0, sum(x)=1} by clipping negatives and renormalising."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total <= 0.0:
        # fall back to uniform distribution if everything vanished
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total


# ----------------------------------------------------------------------
# FFT‑based full convolution (real inputs)
def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of real vectors x and y using FFT."""
    size = len(x) + len(y) - 1
    # next power of two for efficient FFT
    nfft = 1 << (size - 1).bit_length()
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(fx * fy, n=nfft)[:size]
    # numerical noise may produce tiny negatives – clip them
    return np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Score for a probability vector (sum = 1)
def _score(a: np.ndarray) -> float:
    """Compute 2·n·max(a*a) (the evaluate_sequence value when sum(a)=1)."""
    n = a.shape[0]
    b = _conv_full(a, a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# One‑parameter arcsine family
def _arcsine_eps_distribution(n: int, eps: float) -> np.ndarray:
    """Discrete arcsine‑type distribution with shift ε."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _best_arcsine_eps(n: int, grid: int = 12):
    """Search ε on a small grid; return the best vector and its score."""
    best_seq = None
    best_score = float("inf")
    best_eps = None
    # search ε∈[0.1, 2.0]; the exact interval is not critical
    for eps in np.linspace(0.1, 2.0, grid):
        a = _arcsine_eps_distribution(n, eps)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq, best_eps = sc, a, eps
    return best_seq, best_score, best_eps


# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray, deadline: float,
                     max_iters: int = 30000, eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Two‑phase projected gradient descent starting from a0.

    Phase 1 (≤3000 iters)   : minimise L2‑error of the autocorrelation.
    Phase 2 (rest)          : minimise a smooth approximation of the L∞‑norm.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _score(a)

    # momentum buffer
    v = np.zeros_like(a)

    # pre‑compute constant for phase‑1 (average of the convolution)
    avg = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- evaluate current objective -----
        b = _conv_full(a, a)
        cur_max = b.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} score={cur_score:.6f}")

        # ----- compute gradient -----
        if it <= 3000:
            # phase 1: L2‑error gradient  g = 4·conv(d, a_rev)
            d = b - avg
            a_rev = a[::-1]                         # reversed view
            conv = _conv_full(d, a_rev)
            g = 4.0 * conv[n - 1: n - 1 + n]
        else:
            # phase 2: smooth L∞‑norm gradient
            # increase α linearly from 2 to 40
            alpha = 2.0 + min(38.0, (it - 3000) * (38.0 / 2000.0))
            # soft‑max weights over the convolution entries
            w = np.exp(alpha * (b - cur_max))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(b, 1.0 / b.shape[0])
            else:
                w /= w_sum
            a_rev = a[::-1]
            conv = _conv_full(w, a_rev)
            g = 2.0 * conv[n - 1: n - 1 + n]

        # ----- momentum & step -----
        beta = 0.9
        v = beta * v + (1.0 - beta) * g
        eta = eta0 / np.sqrt(it)          # diminishing step size
        a = a - eta * v

        # ----- simplex projection -----
        a = _proj_simplex(a)

        # ----- occasional jitter to escape plateaus -----
        if it % 500 == 0 and np.random.rand() < 0.3:
            a += np.random.randn(n) * 1e-5
            a = _proj_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluate_sequence value.
    The routine searches a few lengths, picks the best arcsine‑type start,
    refines it with the gradient method described above and finally returns
    the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # keep a small safety margin

    # ------------------------------------------------------------------
    # Determine a set of candidate lengths
    try:
        n0 = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n0 = 6500
    # pick a few lengths around n0 and a couple of powers of two
    lengths = sorted(
        {
            int(n0),
            int(n0 * 0.8),
            int(n0 * 1.2),
            4096,
            8192,
            12288,
            16384,
        }
    )
    # keep only sensible sizes
    lengths = [n for n in lengths if 1000 <= n <= 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Quick scan of the arcsine‑family for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        a_init, sc_init, eps = _best_arcsine_eps(n, grid=12)
        print(f"Arcsine init – length={n}, eps={eps:.3f}, score={sc_init:.6f}")
        if sc_init < overall_best_score:
            overall_best_seq = a_init
            overall_best_score = sc_init
            best_n = n

    # ------------------------------------------------------------------
    # Refine the currently best candidate (if any) with gradient descent
    if overall_best_seq is not None:
        refine_deadline = deadline
        refined_seq, refined_score = _refine_sequence(
            overall_best_seq,
            refine_deadline,
            max_iters=40000,
            eta0=0.6,
        )
        if refined_score < overall_best_score:
            overall_best_seq, overall_best_score = refined_seq, refined_score
            print(f"*** Refined best – length={best_n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never trigger)
    if overall_best_seq is None:
        n_fallback = 2000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback)

    print(f"Final best – length={len(overall_best_seq)} score={overall_best_score:.6f}")
    return overall_best_seq.tolist()
