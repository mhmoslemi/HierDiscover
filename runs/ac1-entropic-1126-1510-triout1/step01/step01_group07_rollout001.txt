def compute_obj(seq):
    arr = np.asarray(seq, dtype=np.float64)
    n = arr.shape[0]
    # zero pad to length 2*n
    N = 2*n
    fft_a = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)
    # conv length N, but linear conv length = 2n-1
    max_b = np.max(conv[:2*n-1])
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
```

def compute_obj(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    N = 2*n
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa*fa, n=N)
    max_b = np.max(conv[:2*n-1])
    sum_a = np.sum(a)
    if sum_a < 1e-9: return np.inf
    return 2.0*n*max_b/(sum_a*sum_a)
```

"""
Propose a low‑evaluation step‑function.

The search explores a three‑parameter family

    a_i = (i + Δ)^{-p} · exp(-γ·i/(n‑1)),   i = 1,…,n

where
    p     – power‑law exponent,
    Δ     – offset that damps the very first coefficients,
    γ     – exponential taper (often near zero).

The objective is evaluated efficiently with an FFT‑based convolution
(`compute_objective`).  A coarse grid over (p,Δ,γ) is scanned, then a fine
grid around the best point, followed by a short random‑search phase until the
time budget expires.  The best candidate is finally checked with the
authoritative `evaluate_sequence` function and returned as a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation – identical to evaluate_sequence up to clipping (which we
# avoid by construction).  It is defined at top level to satisfy the “no
# closures” rule.
# ----------------------------------------------------------------------
def compute_objective(seq):
    """
    Return the value of the objective function for a given non‑negative
    sequence `seq`.  The convolution is performed via FFT for speed.
    """
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    if n == 0:
        return float("inf")
    # zero‑pad to length 2·n for a linear convolution
    N = 2 * n
    # real FFT is faster for real input
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)
    max_b = np.max(conv[: 2 * n - 1])
    sum_a = np.sum(a)
    if sum_a < 1e-9:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)
