def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    if np.any(a < 0):
        a = np.maximum(a, 0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    # compute convolution using FFT
    n = a.size
    # zero-pad to length 2n-1
    m = 1 << (2*n -1).bit_length()
    # compute FFT
    fft_a = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]
    max_b = conv.max()
    return float(2 * n * max_b / (sum_a * sum_a))
```

def _hill_climb_fast(seq, best_score, deadline):
    n = len(seq)
    cur = seq.copy()
    it = 0
    while time.time() < deadline and it < 50000:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if cur[i] == 0:
            continue
        # propose moving some fraction of cur[i] to j
        delta = cur[i] * np.random.rand() * 0.2   # up to 20%
        cur[i] -= delta
        cur[j] += delta
        score = fast_score(cur)
        if score < best_score:
            best_score = score
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        # occasionally renormalize to avoid drift (should preserve sum)
        it += 1
        if it % 5000 == 0:
            print(f"HillClimb iter {it} best_score {best_score:.6f}")
            if time.time() >= deadline:
                break
    return cur, best_score

def _gradient_step_fast(seq, best_score, deadline):
    n = len(seq)
    cur = seq.copy()
    step = 0.1
    it = 0
    while time.time() < deadline and it < 5000:
        conv = np.fft.irfft(np.fft.rfft(cur, n=1 << (2*n-1).bit_length())**2)[:2*n-1]  # maybe compute once
        max_val = conv.max()
        # compute k index of max
        k = int(conv.argmax())
        # compute gradient: g_i = 2 * cur[k-i] if 0<=k-i < n else 0
        # This is effectively shift of cur reversed within appropriate bounds.
        grad = np.zeros_like(cur)
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        grad[idx] = 2.0 * cur[k - idx]
        # update
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        total = nxt.sum()
        if total == 0:

"""
Propose a low‑score step‑function.

The evaluation function is
    S(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The objective is homogeneous, therefore we may normalise the vector to sum=1.
For a power‑law a_i ∝ (i+offset+1)^α the asymptotic optimum is near
α≈−0.55, giving a score ≈1.44.  A fast FFT‑based score calculator is used
for the massive grid search; the final candidate is then polished with a
few thousand moves evaluated by the official ``evaluate_sequence`` routine
to guarantee a valid answer.

The algorithm proceeds in three stages

1. **Coarse & refined search over (α,offset,n)** – ternary search on α for a
   handful of offsets and a few different lengths (n≈1500…8000).  The fast
   score avoids the O(n²) convolution cost.

2. **Simple stochastic hill‑climbing** on the simplex (mass transfer
   between random indices) – a few ten‑thousand proposals.

3. **Projected sub‑gradient descent** on the maximal convolution entry.
   The sub‑gradient of max(conv) w.r.t. a is known analytically, so a
   cheap explicit step is possible.

All the heavy work is done with the fast score; only the final polishing
calls the official evaluator (which also applies the security checks).
The routine respects the 1000 s time budget and prints intermediate progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the official one for the non‑negative,
# <1000 values we generate).  Using FFT makes a single evaluation O(n log n).
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the same value as ``evaluate_sequence`` but using FFT.
    The input is assumed to be a 1‑D numpy array of non‑negative floats."""
    a = np.maximum(seq, 0.0)                # protect against accidental negatives
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    n = a.shape[0]
    # convolution length 2*n-1 → pad to next power of two for fast FFT
    size = 1 << ((2 * n - 2).bit_length())
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Power‑law generator
# ----------------------------------------------------------------------
def _make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """Return a normalised non‑negative vector of length *n* with
    a_i ∝ (i + offset + 1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


# ----------------------------------------------------------------------
# Wrapper around the official evaluator (used only for the final polishing)
# ----------------------------------------------------------------------
def _eval(seq: np.ndarray) -> float:
    """Call the supplied ``evaluate_sequence`` (expects a plain list)."""
    return evaluate_sequence(seq.tolist())


# ----------------------------------------------------------------------
# Stochastic hill‑climbing on the simplex (fast version – uses _fast_score)
# ----------------------------------------------------------------------
def _hill_climb_fast(seq: np.ndarray, best_score: float, deadline: float):
    """Try to improve *seq* by repeatedly moving a small fraction of mass."""
    n = seq.shape[0]
    cur = seq.copy()
    it = 0
    while time.time() < deadline and it < 60000:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        # move up to 8 % of the source mass
        delta = cur[i] * np.random.rand() * 0.08
        cur[i] -= delta
        cur[j] += delta
        score = _fast_score(cur)
        if score < best_score:
            best_score = score
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it % 10000 == 0:
            print(f"[hill‑climb] it={it} best_score={best_score:.6f}")
    return cur, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the maximal convolution entry
# (fast version – uses the analytical sub‑gradient and _fast_score)
# ----------------------------------------------------------------------
def _gradient_step_fast(seq: np.ndarray, best_score: float, deadline: float):
    n = seq.shape[0]
    cur = seq.copy()
    step = 0.12
    it = 0
    while time.time() < deadline and it < 3000:
        # compute convolution via FFT (only needed for max index)
        size = 1 << ((2 * n - 2).bit_length())
        fft_a = np.fft.rfft(cur, n=size)
        conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]

        max_val = conv.max()
        # early stop if we already beat the current best score
        if 2.0 * n * max_val < best_score:
            best_score = 2.0 * n * max_val

        k = int(conv.argmax())
        # sub‑gradient of max(conv) w.r.t. cur:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur)
        grad[idx] = 2.0 * cur[k - idx]

        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s

        new_score = _fast_score(nxt)
        if new_score < best_score:
            cur = nxt
            best_score = new_score
            step *= 0.97
        else:
            step *= 0.9
        it += 1
        if it % 500 == 0:
            print(f"[gradient] it={it} best_score={best_score:.6f}")
    return cur, best_score


# ----------------------------------------------------------------------
# Ternary (actually golden‑section) search for the optimal exponent
# ----------------------------------------------------------------------
def _search_exponent(n: int, offset: float, deadline: float):
    """Return (seq, score, exponent) giving the best exponent for a fixed
    offset and length.  Uses the fast score and a simple golden‑section search."""
    lo, hi = -0.70, -0.30
    phi = (1 + 5 ** 0.5) / 2
    invphi = 1 / phi
    # first internal points
    x1 = hi - (hi - lo) * invphi
    x2 = lo + (hi - lo) * invphi
    f1 = _fast_score(_make_power_law(n, x1, offset))
    f2 = _fast_score(_make_power_law(n, x2, offset))

    while time.time() < deadline and (hi - lo) > 1e-5:
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - (hi - lo) * invphi
            f1 = _fast_score(_make_power_law(n, x1, offset))
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + (hi - lo) * invphi
            f2 = _fast_score(_make_power_law(n, x2, offset))

    best_alpha = (lo + hi) / 2.0
    best_seq = _make_power_law(n, best_alpha, offset)
    best_score = _fast_score(best_seq)
    return best_seq, best_score, best_alpha

def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The search is time‑limited (default 1000 s) and returns the best
    sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.3   # tiny safety margin

    best_seq = None
    best_score = float('inf')
    best_alpha = None
    best_offset = None
    best_n = None

    # ------------------------------------------------------------------
    # 1) Scan several lengths – a trade‑off between speed and solution
    #    quality.  For each length we perform an exponent optimisation.
    # ------------------------------------------------------------------
    candidate_lengths = [1500, 3000, 5000, 8000]  # all comfortably fit in memory
    offsets_to_try = [0.0, 0.2, 0.5, 1.0]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Length n={n} ===")
        for off in offsets_to_try:
            if time.time() >= deadline:
                break
            seq, score, alpha = _search_exponent(n, off, deadline)
            print(f"[grid] offset={off:.2f} α={alpha:.5f} score={score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = seq
                best_alpha = alpha
                best_offset = off
                best_n = n

        # ------------------------------------------------------------------
        # 2) Random perturbations around the current best for this n.
        #    (simple Monte‑Carlo refinement)
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            for _ in range(2000):
                if time.time() >= deadline:
                    break

                off = best_offset + (np.random.rand() - 0.5) * 0.4
                if not (-0.70 <= alpha <= -0.30):
                    continue

                score = _fast_score(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    best_alpha = alpha
                    best_offset = off
                    best_n = n

        # ------------------------------------------------------------------
        # 3) Hill‑climbing (fast) on the current best sequence of this length.
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            best_seq, best_score = _hill_climb_fast(best_seq, best_score, deadline)
            print(f"[hill‑climb] best_score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 4) Gradient‑step refinement (fast).
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            best_seq, best_score = _gradient_step_fast(best_seq, best_score, deadline)
            print(f"[gradient] best_score={best_score:.6f}")

    # 5) Final polishing with the **official** evaluator (few thousand steps)
    # ------------------------------------------------------------------
    if best_seq is None:
        raise RuntimeError("No sequence generated – time budget too small.")

    # bring the best sequence back to the official evaluation domain
    cur = best_seq.copy()
    cur_score = _eval(cur)
    print(f"\n[final] pre‑polish score={cur_score:.6f}")

    # a small number of exact evaluations (slow) to fine‑tune the answer
    it = 0
    while time.time() < deadline and it < 8000:
        n = cur.shape[0]
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            continue

        delta = cur[i] * np.random.rand() * 0.05
        cur[i] -= delta
        cur[j] += delta
        score = _eval(cur)
        if score < cur_score:
            cur_score = score
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1

            print(f"[exact hill‑climb] it={it} cur_score={cur_score:.6f}")

    # ------------------------------------------------------------------

    final_score = evaluate_sequence(final_seq)  # final check with the official routine
    print(f"\nReturning sequence length {len(final_seq)} with final score {final_score:.6f}")
    return final_seq
