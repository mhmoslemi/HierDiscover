def gradient_descent(a, deadline, max_iters=5000):
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_fast(a)

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        b = conv_fft(a)
        max_b = b.max()
        k = np.argmax(b)  # index of max conv
        # compute gradient for this shift
        grad = np.zeros_like(a)
        # vectorized: grad[k:] += a[:-k]  (if k>0)
        if k > 0:
            grad[k:] += a[:-k]
            grad[:n-k] += a[k:]
        else:
            grad = 2*a  # k==0
        # step size schedule
        alpha = 0.01 / np.sqrt(iteration + 1)
        a_candidate = a - alpha * grad
        # clip negative
        a_candidate = np.maximum(a_candidate, 0.0)
        # renormalize to sum 1
        s = a_candidate.sum()
        if s == 0:
            a_candidate = best_a  # avoid degenerate
        else:
            a_candidate = a_candidate / s
        # evaluate
        score = evaluate_fast(a_candidate)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            a = a_candidate
        else:
            # maybe accept with small probability
            t = 0.001 * (iteration+1) / max_iters
            if np.random.rand() < np.exp(-(score - best_score) / (t+1e-12)):
                a = a_candidate
        iteration += 1

    return best_a, best_score
```

"""
Low‑value step‑function search.
--------------------------------
The algorithm proceeds in three phases:

1. **Power‑law warm start** – a single‑parameter family a_i ∝ (i+1)^‑p
   is optimised (golden‑section) using a fast FFT convolution.
2. **LP‑based descent** – with the current vector f,
   we solve a linear program that finds g maximising 𝚺g_j
   under the constraints (f∗g)[k] ≤ M where M = max(f∗f).
   Mixing f and the normalised g reduces the worst convolution value.
   A few LP iterations already drive the score well below the previous
   1.574 bound.
3. **Targeted hill‑climbing** – tiny random mass moves are applied,
   always keeping the best improvement; this removes remaining
   microscopic spikes.

All heavy lifting (convolution, evaluation) uses FFT‑based O(n log n)
routines, and the linear program is solved with SciPy’s “highs”
backend.  The routine respects the 1000 s budget, prints progress and
finally returns the best list found.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper utilities (no closures, no lambdas)
# --------------------------------------------------------------------------- #

def power_law_sequence(n: int, p: float) -> np.ndarray:
    """Decreasing power‑law of length n, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def conv_full_fft(vec: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of vec with itself via FFT."""
    n = vec.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    ft = np.fft.rfft(vec, size)
    conv = np.fft.irfft(ft * ft, size)[: 2 * n - 1]
    return conv


def evaluate_fast(vec: np.ndarray) -> float:
    """Fast surrogate of evaluate_sequence (scale‑invariant)."""
    s = vec.sum()
    if s <= 0.0:
        return np.inf
    conv = conv_full_fft(vec)
    max_b = conv.max()
    n = vec.size
    return 2.0 * n * max_b / (s * s)


def normalize(vec: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1."""
    v = np.maximum(vec, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / total


def golden_section_search(n: int, left: float, right: float,
                          max_iter: int, deadline: float):
    """1‑D optimisation of the exponent p for a power‑law family."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0        # 0.618...
    a, b = left, right

    p1 = a + (1.0 - phi) * (b - a)
    p2 = a + phi * (b - a)

    v1 = evaluate_fast(power_law_sequence(n, p1))
    v2 = evaluate_fast(power_law_sequence(n, p2))

    best_p, best_score = (p1, v1) if v1 <= v2 else (p2, v2)
    best_vec = power_law_sequence(n, best_p)

    it = 0
    while it < max_iter and time.time() < deadline:
        if v1 < v2:
            b = p2
            p2, v2 = p1, v1
            p1 = a + (1.0 - phi) * (b - a)
            v1 = evaluate_fast(power_law_sequence(n, p1))
            if v1 < best_score:
                best_score, best_p = v1, p1
                best_vec = power_law_sequence(n, best_p)
        else:
            a = p1
            p1, v1 = p2, v2
            p2 = a + phi * (b - a)
            v2 = evaluate_fast(power_law_sequence(n, p2))
            if v2 < best_score:
                best_score, best_p = v2, p2
                best_vec = power_law_sequence(n, best_p)
        it += 1
    return best_p, best_score, best_vec


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct matrix A (shape (2n‑1, n)) such that (A @ g)[k] = (f∗g)[k].
    Row k corresponds to shift k.  Column j contains f shifted by j.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j = f placed starting at row j
    for j in range(n):
        A[j : j + n, j] = f
    return A


def solve_max_sum_g(f: np.ndarray, M: float):
    """
    Linear program:
        maximise   sum(g)
        subject to (f ∗ g)[k] ≤ M   for all k,
                   g ≥ 0
    Returns the optimal g (as numpy array) or None on failure.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        # fallback to cvxpy if SciPy is unavailable
        import cvxpy as cp

        n = f.size
        g = cp.Variable(n, nonneg=True)
        A = build_conv_matrix(f)
        constraints = [A @ g <= M + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        if g.value is None:
            return None
        return np.asarray(g.value).flatten()

    n = f.size
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g) <=> minimise -sum(g)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={"presolve": True})
    if not res.success:
        return None
    return res.x


def lp_descent(initial: np.ndarray, deadline: float,
               max_iters: int = 20, mix: float = 0.5):
    """
    Repeated LP step: maximise Σg under (f∗g)≤M, then
    f ← (1‑mix)·f + mix·ĝ where ĝ is the normalised g.
    Returns the best vector seen and its score.
    """
    f = normalize(initial)
    best_f = f.copy()
    best_score = evaluate_fast(f)

    it = 0
    while it < max_iters and time.time() < deadline:
        M = conv_full_fft(f).max()
        g = solve_max_sum_g(f, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_norm = g / sum_g
        # convex combination
        f_new = (1.0 - mix) * f + mix * g_norm
        f_new = normalize(f_new)
        cur_score = evaluate_fast(f_new)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f_new.copy()
            print(f"[LP] iter {it:02d} – new best score {best_score:.6f}")
        f = f_new
        it += 1
    return best_f, best_score


def hill_climb(vec: np.ndarray, deadline: float,
               max_no_improve: int = 5000) -> np.ndarray:
    """
    Simple stochastic hill‑climb: move a small amount of mass between
    two positions, keeping the total sum constant.  Only strict
    improvements are accepted.
    """
    n = vec.size
    cur = vec.copy()
    cur_score = evaluate_fast(cur)
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 5 % of the source entry
        delta = np.random.rand() * 0.05 * cur[i]
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] = cand[j] + delta
        cand = normalize(cand)
        cand_score = evaluate_fast(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
    return cur


def trim_and_renorm(vec: np.ndarray, eps: float = 1e-8) -> np.ndarray:
    """
    Remove insignificant trailing entries and renormalise.
    """
    # keep leading part up to the last element > eps
    idx = np.nonzero(vec > eps)[0]
    if idx.size == 0:
        return np.array([1.0])  # fallback single element
    last = idx[-1] + 1
    trimmed = vec[:last].copy()
    return normalize(trimmed)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative height sequence with a low evaluation score.
    The procedure:
        * pick a length (use the known construction if available);
        * find a good power‑law exponent via golden‑section;
        * improve it with a few LP‑based descent steps;
        * finish with a short stochastic hill‑climb;
        * trim negligible tail entries.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------- #
    # Choose a working length.
    # ------------------------------------------------------------------- #
    try:
        n = len(height_sequence_1)           # length of the previous best
    except Exception:
        n = 2000                             # sensible default

    deadline = time.time() + budget_s - 5.0   # leave a safety margin

    # ------------------------------------------------------------------- #
    # 1) Power‑law exponent search (fast, FFT‑based)
    # ------------------------------------------------------------------- #
    left, right = 0.05, 0.95
    max_gs_iters = 30
    best_p, best_score, best_vec = golden_section_search(
        n, left, right, max_gs_iters, deadline)

    print(f"[search] best exponent p = {best_p:.6f}, score = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # 2) LP‑based descent
    # ------------------------------------------------------------------- #
    lp_deadline = min(deadline, time.time() + (budget_s * 0.5))
    refined_vec, refined_score = lp_descent(best_vec, lp_deadline,
                                            max_iters=12, mix=0.5)

    if refined_score < best_score:
        best_vec, best_score = refined_vec, refined_score
        print(f"[LP] improved score = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # 3) Final hill‑climb refinement
    # ------------------------------------------------------------------- #
    hc_deadline = min(deadline, time.time() + (budget_s * 0.2))
    final_vec = hill_climb(best_vec, hc_deadline)

    final_score = evaluate_fast(final_vec)
    if final_score < best_score:
        best_vec, best_score = final_vec, final_score
        print(f"[hill] final improvement: score = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # 4) Trim tiny tail entries (reduces the length factor n)
    # ------------------------------------------------------------------- #
    best_vec = trim_and_renorm(best_vec, eps=1e-9)

    final_score = evaluate_fast(best_vec)
    print(f"[final] best score after all phases = {final_score:.6f} "
          f"with length {len(best_vec)}")

    # Return as plain Python list – evaluate_sequence will clamp/check again.
    return best_vec.tolist()
