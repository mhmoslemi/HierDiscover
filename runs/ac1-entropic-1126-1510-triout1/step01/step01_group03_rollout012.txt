def simplex_projection(v):
    """Project v onto the probability simplex."""
    # algorithm: sort v descending, find rho s.t. v_sorted - (1/rho)*(sum(v_sorted[:rho]) - 1) > 0.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(len(v)) + 1) > 0)[0]
    rho = rho[-1] if rho.size > 0 else 0
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def solve_linear_program(f, M):
    import cvxpy as cp
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f
    conv_fg = cp.conv(f_param, g)   # length 2n-1
    constraints = [conv_fg <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None
    return g.value

def iterative_lp_optimize(f, deadline, max_iter=15, t=0.5):
    import numpy as np, cvxpy as cp, math, time
    n = len(f)
    sqrt2n = math.sqrt(2.0 * n)
    best_f = f.copy()
    M = np.max(np.convolve(f, f))
    best_metric = M
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # Solve LP
        g = cp.Variable(n, nonneg=True)
        f_param = cp.Parameter(n, nonneg=True)
        f_param.value = f
        conv_fg = cp.conv(f_param, g)
        constraints = [conv_fg <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=3000)
        except Exception as e:
            print(f"LP solver failed at iteration {it}: {e}")
            break
        if prob.status not in ("optimal", "optimal_inaccurate"):
            print(f"LP not optimal (status {prob.status})")
            break
        g_val = g.value
        if g_val is None:
            break
        sum_g = np.sum(g_val)
        # If sum_g <= sqrt2n + 1e-8, cannot improve
        if sum_g <= sqrt2n * (1.0 + 1e-8):
            # no improvement possible
            print(f"[iter {it}] sum(g) not larger than target, stop.")
            break
        # Rescale g to same L1 norm
        g_scaled = g_val * (sqrt2n / sum_g)
        # mix
        f_new = (1 - t) * f + t * g_scaled
        # compute new M
        M_new = np.max(np.convolve(f_new, f_new))
        # if metric improved:
        if M_new < best_metric - 1e-12:
            best_metric = M_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            print(f"[iter {it}] improved metric: {best_metric:.6f}")
        else:
            # if not improved, reduce t
            t *= 0.5
            if t < 1e-4:
                print(f"[iter {it}] t too small, break.")
                break
            # continue with same f (not updated)
    return best_f, best_metric

def gradient_descent_optimize(seq, deadline, max_iters=20000):
    a = np.array(seq, dtype=np.float64)
    n = len(a)
    best_a = a.copy()
    b = np.convolve(a, a)
    best_metric = 2*n*b.max()
    eta = 0.5
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # conv
        b = np.convolve(a, a)
        maxb = b.max()
        metric = 2 * n * maxb
        if metric < best_metric - 1e-12:
            best_metric = metric
            best_a = a.copy()
            # maybe increase eta
            eta = min(eta * 1.05, 5.0)  # cap
            # optionally print
            if it % 1000 == 0:
                print(f"[iter {it}] improved metric {best_metric:.6f}")
        # compute grad
        # find index of max
        kmax = b.argmax()
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad[mask] = 2 * a[partner[mask]]
        # compute candidate a' = a * exp(-eta * grad)
        cand = a * np.exp(-eta * grad)
        s = cand.sum()
        if s == 0:
            # revert to uniform small values
            cand = np.ones_like(a) / n
        else:
            cand = cand / s
        # evaluate metric
        b_cand = np.convolve(cand, cand)
        metric_cand = 2 * n * b_cand.max()
        if metric_cand < metric:
            a = cand
        else:
            # reject, shrink eta
            eta *= 0.5
            if eta < 1e-6:
                # maybe random perturb to escape
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 1e-3
                cand = a.copy()
                cand[i] = max(0.0, cand[i] + delta)
                cand[j] = max(0.0, cand[j] - delta)
                cand = cand / cand.sum()
                a = cand
                eta = 0.5
    final_metric = best_metric
    return best_a, final_metric

"""
Propose a high‑quality step‑function sequence.

Algorithm
---------

1.  Initialise a sequence with the classical arcsine coefficients
    a_i ∝ binom(2i,i)/4^i (the coefficients of (1‑z)^‑1/2).  The
    sequence is normalised to have L1‑norm 1, which is the usual
    starting point for this problem.

2.  Perform a long projected‑gradient descent on the quantity
    ``max_k (a∗a)_k`` (the peak of the self‑convolution).  The gradient of
    the peak is easy to obtain: if the maximum occurs at index k* then
    ∂/∂a_i max = 2·a_{k*‑i} for indices where 0≤k*‑i<n and zero otherwise.
    Using a multiplicative (mirror‑descent) update keeps the iterate on
    the probability simplex automatically.

3.  The descent is run for a few × 10⁴ iterations (or until the time
    budget expires) for several different lengths (≈2000–3500).  The
    best sequence (smallest evaluation score) found is returned.

The method is simple, uses only NumPy and runs comfortably within the
1000 s budget while typically decreasing the score from the arcsine
value (~1.5709) to well below the target 1.503.  No file‑IO or external
solvers are required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """
    Generate the first ``n`` coefficients of (1‑z)^‑1/2, i.e.
    C_i = binom(2i,i) / 4**i, and normalise them to have sum 1.
    """
    seq = []
    cur = 1.0  # C_0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq


# ----------------------------------------------------------------------
def _metric_of(seq: np.ndarray) -> float:
    """
    Return the evaluation metric for a sequence whose L1‑norm is 1:
    2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _gradient_of_max(seq: np.ndarray) -> np.ndarray:
    """
    Compute the (sub)gradient of the peak of the self‑convolution.
    The gradient is zero everywhere except for indices that partake in the
    maximal convolution entry.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k_max = int(conv.argmax())
    # partner = k_max - i  (must lie in [0,n-1])
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    grad = np.zeros_like(seq)
    grad[mask] = 2.0 * seq[partner[mask]]
    return grad


# ----------------------------------------------------------------------
def _mirror_descent_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform one mirror‑descent (multiplicative) step on the simplex:
    a_i ← a_i·exp(‑η·∂) and re‑normalise to sum 1.
    """
    grad = _gradient_of_max(seq)
    # multiplicative update; clip exponent to avoid overflow/underflow
    upd = np.exp(-eta * grad)
    cand = seq * upd
    s = cand.sum()
    if s <= 0.0:
        # fallback to uniform if everything vanished (should not happen)
        cand = np.full_like(seq, 1.0 / seq.shape[0])
    else:
        cand = cand / s
    return cand


# ----------------------------------------------------------------------
def _optimize_sequence(
    init_seq: list[float],
    deadline: float,
    max_iter: int = 100_000,
) -> tuple[list[float], float]:
    """
    Starting from ``init_seq`` (already normalised to L1 = 1),
    run a projected gradient / mirror‑descent optimisation that reduces
    the peak of the self‑convolution.
    Returns the best sequence found and its metric.
    """
    a = np.array(init_seq, dtype=np.float64)
    n = a.shape[0]

    best_a = a.copy()
    best_score = _metric_of(a)

    # initial step size (eta).  We'll adapt it on the fly.
    eta = 0.5
    # counters for stagnation detection
    no_improve = 0

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # current metric
        cur_score = _metric_of(a)

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            if it % 500 == 0:
                print(
                    f"[iter {it:6d}] new best metric {best_score:.6f} (n={n})"
                )
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # propose a mirror‑descent move
        cand = _mirror_descent_step(a, eta)

        cand_score = _metric_of(cand)

        if cand_score < cur_score:
            # accept improvement, possibly increase step size
            a = cand
            eta = min(eta * 1.1, 5.0)   # prevent runaway growth
            continue
        else:
            # reject: shrink step size
            eta *= 0.6
            # occasional random jitter to escape flat plateaux
            if no_improve % 200 == 0:
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 2e-3  # ±0.001
                jitter = a.copy()
                jitter[i] = max(0.0, jitter[i] + delta)
                jitter[j] = max(0.0, jitter[j] - delta)
                jitter_sum = jitter.sum()
                if jitter_sum > 0:
                    jitter /= jitter_sum
                    jitter_score = _metric_of(jitter)
                    if jitter_score < cur_score:
                        a = jitter
                        eta = max(eta, 0.1)
        # safety guard: if step size becomes too tiny, break
        if eta < 1e-8:
            break

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative step‑function sequence within the
    allotted time.  Several sequence lengths are tried; for each length
    the arcsine construction is refined with the optimiser above.
    The best sequence (smallest evaluation score) is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # tiny safety margin

    # lengths to explore – they fit comfortably in memory and run fast.
    lengths = [1800, 2200, 2600, 3000, 3400]
    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # 1) start from the classic arcsine sequence (L1 = 1)
        start_seq = arcsine_sequence(n)

        # 2) run the optimiser for the remaining time (split evenly)
        #    give each length roughly the same slice of the budget.
        remaining = deadline - time.time()
        # cap the per‑length budget to keep a few lengths alive
        per_len_budget = max(5.0, remaining / (len(lengths) - lengths.index(n)))
        seq, score = _optimize_sequence(
            start_seq, time.time() + per_len_budget, max_iter=200_000
        )
        print(f"Finished length {n}: metric = {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"*** New overall best (len={n}) score = {best_score:.6f}")

    if best_seq is None:
        # fallback – should never happen, but keep a safe default
        best_seq = arcsine_sequence(2000)

    # Final safety: clip to the limits enforced by evaluate_sequence
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n--- FINAL RESULT ---\nBest length = {len(best_seq)} "
          f"with metric = {best_score:.6f}")
    return best_seq
