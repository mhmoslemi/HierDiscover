---------
* Work with a probability vector `a` (non‑negative, sum = 1).  
* The evaluation metric equals `2·n·max(conv(a,a))`, so minimizing the
  maximum entry of the autocorrelation is the goal.
* Instead of a hard max we minimise a smooth approximation
  `log‑sum‑exp(α·b)` where `b = a*a`.  For growing `α` this approaches the
  true maximum while providing a well‑behaved gradient.
* The gradient of the smooth objective is
      g = 2·(w ⊛ a_rev)  (⊛ = convolution, a_rev = a reversed)
  where `w` are the soft‑max weights of `b`.  Convolutions are performed
  with FFTs (O(n log n)).
* An Adam optimiser updates `a`; after each step we project back onto the

* A short sub‑gradient “fine‑tune” phase (identical to the previous

* The routine tries a few problem sizes (derived from the historic

"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    rho = rho_arr[-1] if rho_arr.size else -1
    theta = cssv[rho] / (rho + 1) if rho >= 0 else 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: a_i ∝ 1/√((i+½)(n‑i‑½)), normalised to 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()
