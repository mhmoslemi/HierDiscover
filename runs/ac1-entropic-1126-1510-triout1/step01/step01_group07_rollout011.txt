"""propose_candidate
Search for a low‑evaluation step‑function by optimising the power‑law exponent.

The evaluation objective is invariant under scaling, so we may fix the L1‑norm
(e.g. sum = 1) and only minimise the peak of the self‑convolution.  For a
family

    a_i = (i+1)^{-p} · exp(-γ·i/(n−1))

the asymptotic objective can be expressed analytically and is minimised for
p≈0.57 with a very small (or zero) exponential taper.  The function below
performs a quick grid search over p and γ (first coarse, then a fine
refinement) using a fast NumPy implementation of the objective.  It then
does a brief stochastic local‑search to polish the best candidate found.
The whole procedure respects the 1000 s time‑budget and returns a Python
list that can be fed directly into the provided ``evaluate_sequence``
function.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    import numpy as np
    import time

    # --- helpers ---------------------------------------------------------

    def fast_objective(seq: np.ndarray) -> float:
        """Same formula as evaluate_sequence but without the safety checks."""
        s = seq.sum()
        if s < 0.01:                     # protects against division by zero
            return np.inf
        conv = np.convolve(seq, seq)
        m = conv.max()
        n = seq.size
        return 2.0 * n * m / (s * s)

    # --- initialisation --------------------------------------------------

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # length of the step function – large enough to reach the asymptotic regime
    n = 4000
    idx = np.arange(1, n + 1, dtype=np.float64)   # i = 1 … n

    best_score = float("inf")
    best_seq = None
    best_params = (None, None)

    # --------------------------------------------------------------------
    # 1. Coarse sweep over the exponent p (γ = 0 at first)
    # --------------------------------------------------------------------
    p_vals = np.linspace(0.45, 0.65, 41)   # step ≈0.005
    for p in p_vals:
        seq = idx ** (-p)                  # pure power‑law, no taper
        score = fast_objective(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_params = (p, 0.0)

        if time.time() > deadline:
            break

    # --------------------------------------------------------------------
    # 2. Fine sweep around the best p, now also exploring a tiny exponential taper γ
    # --------------------------------------------------------------------
    if time.time() < deadline:
        p_center = best_params[0] if best_params[0] is not None else 0.5
        p_fine = np.linspace(max(0.40, p_center - 0.03),
                             min(0.75, p_center + 0.03), 61)   # step ≈0.001
        gamma_vals = np.linspace(0.0, 0.8, 17)                 # step ≈0.05
        for p in p_fine:
            base = idx ** (-p)
            for gamma in gamma_vals:
                seq = base * np.exp(-gamma * (idx - 1) / (n - 1))
                score = fast_objective(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (p, gamma)

                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

    # --------------------------------------------------------------------
    # 3. Very short stochastic local refinement (random mass transfers)
    # --------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        seq = best_seq.copy()
        # a few thousand cheap moves – mix‑ratio t is implicitly 1 (we keep sum fixed)
        for it in range(2000):
            i, j = np.random.randint(0, n, size=2)
            if i == j:

            # take a tiny fraction of a_i and move it to a_j

    # --------------------------------------------------------------------

    )

    return best_seq.tolist()
