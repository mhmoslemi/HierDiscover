"""
Smooth‑max projected gradient descent for the autocorrelation minimisation problem.

The evaluation function is scaling‑invariant, therefore we normalise the
step‑function to sum = 1 and minimise

    score = 2 * n * max(conv(a, a))

The algorithm works as follows:

* Choose a random length n (1500‑3000) and a random non‑negative start vector.
* Repeatedly apply a *smooth‑max* surrogate:
        J_ε(a) = ε·log( Σ_k exp(conv_k(a)/ε) )
  whose gradient is a convex combination of the gradients of the individual
  convolution entries.  For a given weight vector w (soft‑max of the
  convolution values) the gradient can be computed by a single FFT‑based
  convolution.
* Take a projected (onto the probability simplex) gradient step with a
  back‑tracking line‑search.  The step size is gradually reduced and the
  smoothing parameter ε is annealed from 1e‑1 down to 1e‑3.
* The best vector seen over many random restarts (or from the supplied
  `height_sequence_1`) is returned.

The method is much more stable than the pure *max‑only* sub‑gradient used
previously and typically achieves a score below 1.503 within the 1000 s
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:  # all values non‑positive – project onto uniform distribution
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft_self(a: np.ndarray, size_self: int) -> np.ndarray:
    """Full convolution a*a using FFT (size_self is a power of two)."""
    fa = np.fft.rfft(a, n=size_self)
    conv = np.fft.irfft(fa * fa, n=size_self)[: 2 * a.size - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Full convolution x*y using FFT (size must be ≥ len(x)+len(y)-1)."""
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: len(x) + len(y) - 1]
    return conv


def _smooth_max_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Soft‑max weights proportional to exp(conv/eps).  Returned array sums to 1."""
    max_c = conv.max()
    # subtraction avoids overflow; no under‑flow because all terms ≤ 0
    raw = np.exp((conv - max_c) / eps)
    return raw / raw.sum()


def _gradient_smooth_max(a: np.ndarray, w: np.ndarray,
                         size_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective.
    a : current vector (length n, sum = 1)
    w : soft‑max weight vector (length 2n‑1, sums to 1)
    size_grad : FFT size (power of two) for the convolution a_rev * w
    Returns a vector of length n.
    """
    a_rev = a[::-1]
    # convolution a_rev (len n) with w (len 2n‑1) → length 3n‑2
    conv_full = _conv_fft_two(a_rev, w, size_grad)
    # the required slice is indices n‑1 … n‑1 + n‑1 (inclusive) → n entries
    start = a.size - 1
    grad = 2.0 * conv_full[start:start + a.size]
    return grad


def _score_from_conv(n: int, conv: np.ndarray) -> float:
    """Exact score used by evaluate_sequence when sum(a)=1."""
    return 2.0 * n * conv.max()


def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     max_iters_per_eps: int = 2000) -> tuple[np.ndarray, float]:
    """
    Perform the annealed smooth‑max projected gradient descent.
    Returns the best vector found and its exact score.
    """
    a = a0.copy()
    n = a.size

    # pre‑compute FFT sizes (next power of two)
    size_self = 1 << ((2 * n - 1).bit_length())
    size_grad = 1 << ((3 * n - 2).bit_length())

    # initial score
    conv = _conv_fft_self(a, size_self)
    best_score = _score_from_conv(n, conv)
    best_a = a.copy()
    cur_score = best_score

    eps_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
    it_global = 0

    for eps in eps_schedule:
        it_local = 0
        while it_local < max_iters_per_eps and time.time() < deadline:
            it_local += 1
            it_global += 1

            # -------- compute weights and gradient ----------
            conv = _conv_fft_self(a, size_self)
            w = _smooth_max_weights(conv, eps)          # length 2n‑1, sum=1
            grad = _gradient_smooth_max(a, w, size_grad)

            # -------- back‑tracking line‑search ----------
            step0 = 0.4 / np.sqrt(it_global)            # decreasing base step
            improved = False
            for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
                step = step0 * factor
                a_cand = _project_to_simplex(a - step * grad)
                conv_cand = _conv_fft_self(a_cand, size_self)
                cand_score = _score_from_conv(n, conv_cand)
                if cand_score < cur_score - 1e-13:
                    a = a_cand
                    cur_score = cand_score
                    improved = True
                    break

            # -------- if no improvement, small random jitter ----------
            if not improved:
                a_jit = a.copy()
                idx = np.random.randint(n, size=5)
                a_jit[idx] += np.random.randn(5) * 0.002
                a_jit = np.maximum(a_jit, 0.0)
                a_jit /= a_jit.sum()
                conv_jit = _conv_fft_self(a_jit, size_self)
                score_jit = _score_from_conv(n, conv_jit)
                if score_jit < cur_score:
                    a = a_jit
                    cur_score = score_jit

            # -------- keep global best ----------
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # occasional progress report
                if it_global % 300 == 0:
                    print(f"    [iter {it_global}] new best score = {best_score:.6f}")

            # early exit if already under the target
            if best_score < 1.503:
                return best_a, best_score

    return best_a, best_score

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function within the time budget.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # small safety margin

    # start from the previously discovered construction if available
    if 'height_sequence_1' in globals():
        init_vec = np.array(height_sequence_1, dtype=float)
        if init_vec.size > 0:
            init_vec = np.maximum(init_vec, 0.0)
            init_vec /= init_vec.sum()
            best_seq = init_vec.copy()
            conv = _conv_fft_self(best_seq, 1 << ((2 * best_seq.size - 1).bit_length()))
            best_score = _score_from_conv(best_seq.size, conv)
            print(f"Starting from provided sequence, score = {best_score:.6f}")
        else:
            best_seq = None
            best_score = np.inf
    else:
        best_seq = None
        best_score = np.inf

    # ------------------------------------------------------------------
    # Main multi‑restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # choose length
        n = np.random.randint(1500, 3000)

        # random initial shape
        shape = np.random.choice(["geom", "power", "mixed", "random"])
        if shape == "geom":
            # geometric decay a_i = r^i
            t = np.random.uniform(0.0003, 0.0015)
            r = np.exp(-t)
            a = r ** np.arange(n, dtype=float)
        elif shape == "power":
            p = np.random.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** -p)
        elif shape == "mixed":
            t = np.random.uniform(0.0003, 0.001)
            r = np.exp(-t)
            p = np.random.uniform(0.5, 1.5)
            a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)
        else:  # "random"
            a = np.random.gamma(1.0, 1.0, size=n)

        # optional shuffle (does not change the multiset)
        if np.random.rand() < 0.5:
            np.random.shuffle(a)

        a = np.maximum(a, 0.0)
        a /= a.sum()               # normalise to sum = 1

        # Give each start at most 30 s (or whatever remains)
        per_run_deadline = min(deadline, time.time() + 30.0)

        a_opt, score_opt = _smooth_optimize(a, per_run_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"Global improvement: n={n}, shape={shape}, score={best_score:.6f}")

        if best_score < 1.503:
            break

    # fallback – uniform distribution
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # final normalisation (the evaluator will clip anyway)
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    else:
        best_seq = best_seq / best_seq.sum()

    return best_seq.tolist()
