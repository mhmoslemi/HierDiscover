def generate_sidon_set(n, target_m=None, max_tries=10):
    if target_m is None:
        target_m = int((-1 + int(np.sqrt(16*n - 7))) // 2)  # bound

    best_S = []
    for attempt in range(max_tries):
        S = []
        sums = set()
        # random order of positions
        indices = np.random.permutation(n)
        for idx in indices:
            # compute new sums with existing
            conflict = False
            if idx + idx in sums:
                conflict = True
            else:
                for i in S:
                    if idx + i in sums:
                        conflict = True
                        break
            if not conflict:
                # add
                for i in S:
                    sums.add(idx + i)
                sums.add(idx + idx)
                S.append(idx)
                if target_m and len(S) >= target_m:
                    break
        if len(S) > len(best_S):
            best_S = S
            if len(best_S) >= target_m:
                break
    return best_S

    # a normalized sum=1
    cur_a = a.copy()
    M = _conv_max_fft(cur_a)[0]  # max conv value
    best_score = 2.0 * n * M
    for it in range(max_it):

        # build conv matrix

        # solve LP
        c = -np.ones(n)  # maximize sum
        b_ub = np.full(2*n - 1, M)

def _build_conv_matrix(a):
    n = len(a)
    # create sparse matrix with diag contributions
    diagonals = [a[d] * np.ones(n) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(diagonals, offsets, shape=(2*n - 1, n), format='csc')
    return A

"""
Hybrid optimizer for the step‑function attack.

The score for a non‑negative sequence a (with sum(a)>0) is

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

Because the score is invariant under positive scaling we keep sum(a)=1.
The algorithm consists of two complementary phases:

1. **Projected sub‑gradient descent** – a fast gradient method that
   directly minimizes the max‑convolution while staying on the probability
   simplex (non‑negative entries summing to one).  This gives a decent
   starting point (the previous implementation reached ≈1.58).

2. **Linear‑programming refinement** – given the current distribution f,
   we solve the linear program

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M  for all k,
                    g_j ≥ 0

   where M = max(conv(f,f)).  The optimum g tries to add as much extra
   mass as possible without increasing the current maximum convolution.
   The new distribution is a convex combination
        f ← (1‑t)·f + t·(g / Σg)
   with a back‑tracking line‑search on `t`.  Each successful LP step
   reduces `M`, and a few such steps bring the score well below 1.5.

Both phases are executed many times with random restarts until the 1000 s
budget runs out.  The best sequence seen is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _conv_max_fft(a: np.ndarray):
    """Return (maximum value of a*a, indices where it is attained)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = np.max(conv)
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return float(max_val), idx.astype(int)


def _gradient_for_max(a: np.ndarray, max_idx):
    """Sub‑gradient of max‑convolution w.r.t. a."""
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


def _improve_sequence(a0: np.ndarray, n: int, deadline: float):
    """Projected sub‑gradient descent (the same routine as the reference)."""
    a = a0.copy()
    best_a = a.copy()
    max_c, max_idx = _conv_max_fft(a)
    best_score = 2.0 * n * max_c

    it = 0
    while time.time() < deadline and it < 3000:
        it += 1
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c

        grad = _gradient_for_max(a, max_idx)
        eta0 = 0.2 / np.sqrt(it)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            eta = eta0 * factor
            a_candidate = _project_to_simplex(a - eta * grad)
            max_c2, _ = _conv_max_fft(a_candidate)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                break

        if not improved:
            # tiny random jiggle
            a_pert = a.copy()
            idx = np.random.randint(n, size=5)
            a_pert[idx] += np.random.randn(5) * 0.001
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= np.sum(a_pert)
            max_c2, _ = _conv_max_fft(a_pert)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score:
                a = a_pert
                cur_score = cand_score

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 200 == 0:
                print(f"    sub‑grad iter {it}, new local best {best_score:.6f}")

        if best_score < 1.4:      # early exit inside this local run
            break
    return best_a, best_score


def _build_conv_matrix(a: np.ndarray):
    """Return the (2n‑1) × n sparse matrix representing conv(a,·)."""
    n = a.size
    # each diagonal d (0 ≤ d < n) carries the constant coefficient a[d]
    diags = [a[d] * np.ones(n, dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)
    return sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csc')


def _lp_iterative_improve(a: np.ndarray, n: int, deadline: float, max_it: int = 6):
    """Linear‑programming refinement of a (keeps sum(a)=1)."""
    cur_a = a.copy()
    max_c, _ = _conv_max_fft(cur_a)
    cur_M = max_c
    best_score = 2.0 * n * cur_M

    for it in range(max_it):
        if time.time() >= deadline:
            break

        # --- solve  maximise Σ g  subject to conv(cur_a,g) ≤ cur_M  ---
        A = _build_conv_matrix(cur_a)
        c = -np.ones(n)                     # minimise -Σg  ⇔ maximise Σg
        b_ub = np.full(2 * n - 1, cur_M)
        bounds = [(0, None)] * n

        # SciPy's HiGHS solver is fast for these moderate sizes
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # if the LP fails, abort the refinement loop
            break

        g_raw = res.x
        sum_g = np.sum(g_raw)
        if sum_g <= 1e-12:   # cannot add anything useful
            break

        # normalise to keep the total mass = 1
        g = g_raw / sum_g

        # --- line‑search on the mixing factor t ---
        improved = False
        for t in (0.5, 0.25, 0.125, 0.0625, 0.03125):
            candidate = (1.0 - t) * cur_a + t * g
            max_c2, _ = _conv_max_fft(candidate)
            if max_c2 <= cur_M * (1.0 + 1e-12):
                cur_a = candidate
                cur_M = max_c2
                improved = True
                break
        if not improved:
            # could not improve without increasing the bound – stop
            break

        cur_score = 2.0 * n * cur_M
        if cur_score < best_score:
            best_score = cur_score
            # continue trying to improve further
        # optional early stopping
        if best_score < 1.3:
            break

    return cur_a, best_score


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Return a list of non‑negative step heights with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    best_score = float("inf")
    best_seq = None

    # helper to generate an initial shape
    def random_initial(n):
        shape = np.random.choice(["geom", "power", "mixed", "random"])
        if shape == "geom":
            t = np.random.uniform(0.0005, 0.0015)
            r = np.exp(-t)
            a = r ** np.arange(n, dtype=float)
        elif shape == "power":
            p = np.random.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** -p)
        elif shape == "mixed":
            t = np.random.uniform(0.0005, 0.0010)
            r = np.exp(-t)
            p = np.random.uniform(0.5, 1.5)
            a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)
        else:   # random Dirichlet draw
            a = np.random.gamma(1.0, 1.0, size=n)
        a = a / np.sum(a)
        return a

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) pick a problem size and an initial vector
        # --------------------------------------------------------------
        n = int(np.random.randint(1500, 3000))
        a = random_initial(n)

        # --------------------------------------------------------------
        # 2) fast sub‑gradient optimisation (a few seconds per start)
        # --------------------------------------------------------------
        sub_deadline = min(deadline, time.time() + 12.0)
        a_opt, score_opt = _improve_sequence(a, n, sub_deadline)

        # --------------------------------------------------------------
        # 3) LP refinement – often drops the score by a noticeable amount
        # --------------------------------------------------------------
        if time.time() < deadline:
            a_opt, score_opt = _lp_iterative_improve(a_opt, n, deadline)

        # --------------------------------------------------------------
        # 4) keep the best overall sequence
        # --------------------------------------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"Global improvement: n={n}, score={best_score:.6f}")

        # If we have already reached a very good bound, stop early
        if best_score < 1.3:
            break

    # fallback – uniform sequence (should never be used)
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # Convert to plain Python list (the evaluator does the clipping)
    return best_seq.tolist()
