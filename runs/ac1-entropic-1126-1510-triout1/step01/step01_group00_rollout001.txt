def _conv_fft_real(x, y):
    n = len(x) + len(y) - 1
    m = 1 << (n-1).bit_length() # next power of two
    X = np.fft.rfft(x, m)
    Y = np.fft.rfft(y, m)
    Z = X * Y

"""
Propose a low‑score step‑function using a smooth approximation of the
max‑convolution objective.

The objective to minimise (for a sequence a with Σa=1) is
    M(a) = max_k (a * a)[k] .
A smooth surrogate is the log‑sum‑exp:
    Fλ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k]) .
∇Fλ can be written in closed form using the soft‑max weights
    w_k = exp(λ·(b_k−max(b))) / Σ_j exp(λ·(b_j−max(b))) ,
where b = a*a.  The gradient is
    ∇Fλ = 2·conv(w, a_rev)[n−1:n−1+n] / Σw .
We perform projected (or mirror) gradient descent on the simplex,
gradually increasing λ so the surrogate tightens around the true
maximum.  The sequence is periodically projected onto the probability
simplex (non‑negative, sum = 1).  Several lengths and starting points are
tried within the 1000 s budget; the best sequence found is returned.
"""

import time
import numpy as np


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law on 0…n‑1, normalised to unit ℓ¹‑mass."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _conv_fft_real(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full discrete convolution using real‑FFT (fast for moderately large size)."""
    n = len(x) + len(y) - 1
    m = 1 << ((n - 1).bit_length())        # next power of two
    X = np.fft.rfft(x, m)
    Y = np.fft.rfft(y, m)
    Z = X * Y
    full = np.fft.irfft(Z, m)[:n]
    return full


def _optimise_one_length_smooth(
    n: int,
    time_deadline: float,
    init: str = "arcsine",
    max_iters: int = 20000,
    eta0: float = 0.5,
    lambda0: float = 5.0,
    lambda_factor: float = 1.02,
    lambda_max: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth max‑convolution surrogate.
    Returns the best sequence (ℓ¹‑norm = 1) seen and its score
        score = 2·n·max(a*a)  (= evaluate_sequence output because Σa=1).
    """
    # ----- initialise ----------------------------------------------------
    if init == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init == "random":
        a = np.random.rand(n)
        a /= a.sum()
    else:  # arcsine
        a = _arcsine_distribution(n)

    best_seq = a.copy()
    b = _conv_fft_real(a, a)
    best_score = 2.0 * n * b.max()

    # ----- main optimisation loop ----------------------------------------
    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # convolution and current objective value
        b = _conv_fft_real(a, a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] length={n} iter={it} "
                f"score={best_score:.6f}"
            )

        # ---- smooth max (log‑sum‑exp) gradient ---------------------------
        # soft‑max weights, stabilised by shifting with max_b
        w = np.exp(lambda0 * (b - max_b))
        Z = w.sum()
        if Z == 0.0:               # should not happen, but guard
            Z = 1.0
        a_rev = a[::-1]
        conv = _conv_fft_real(w, a_rev)          # length = 3n‑2
        grad = 2.0 * conv[n - 1 : n - 1 + n] / Z   # slice of length n

        # ---- step on the simplex ----------------------------------------
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        # ---- schedule for λ ---------------------------------------------
        lambda0 = min(lambda_max, lambda0 * lambda_factor)

        # ---- occasional jitter to escape plateaus ------------------------
        if it % 250 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score
