def improve_step(f, t=0.1):
    n = len(f)
    M = max(np.convolve(f, f))
    # define variable g
    g = cp.Variable(n, nonneg=True)
    constraints = []
    # Build constraints
    for k in range(2*n-1):
        coeff = np.zeros(n)
        j_min = max(0, k-(n-1))
        j_max = min(k, n-1)
        # for j in [j_min, j_max], coeff[j] = f[k-j]
        coeff[j_min:j_max+1] = f[k - np.arange(j_min, j_max+1)]
        constraints.append(coeff @ g <= M)
    # objective: maximize sum(g)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000) # etc
    if g.value is None:
        return f
    g_opt = np.maximum(g.value, 0) # ensure non-negative
    # normalize g_opt to sum 1
    if g_opt.sum() == 0:
        return f
    g_norm = g_opt / g_opt.sum()
    # convex combination
    f_new = (1 - t) * f + t * g_norm
    f_new = f_new / f_new.sum()
    return f_new

"""
Propose a high‑quality step‑function sequence.

The algorithm implements the “LP‑update” scheme mentioned in the
statement.  Starting from the classic arcsine (C_i = binom(2i,i)/4^i)
distribution we repeatedly:

1.  Scale the current vector `f` so that its ℓ1‑norm equals √(2 n)
    (this makes the self‑convolution bound `M` equal to π/2 for the
    arcsine start).

2.  Solve a linear program:
        maximise   Σ g_i
        subject to g_i ≥ 0
                   (f * g)_k ≤ M   for all convolution indices k
   where `(f * g)_k = Σ_i f_i·g_{k‑i}`.
   The LP is solved with `scipy.optimize.linprog`.

3.  Rescale the LP optimum `g` back to ℓ1‑norm √(2 n) and blend it with
    `f` using a small step size `t`.  The blend preserves the ℓ1‑norm
    and never increases the maximal convolution value, so the
    evaluation score `2·n·max(conv)/(Σa)^2` is non‑increasing.

4.  After a few LP updates we evaluate the normalised sequence
    (ℓ1‑norm = 1) with the provided `evaluate_sequence` function and
    keep the best result seen across several candidate lengths.

The whole search runs for at most the allotted budget (default 1000 s)
and returns the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def arcsine_raw(n: int) -> np.ndarray:
    """Raw (unnormalised) arcsine coefficients C_i = binom(2i,i)/4^i."""
    cur = 1.0
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = cur
        # recurrence C_{i+1} = C_i * (2*i+1) / (2*(i+1))
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    return seq


# ----------------------------------------------------------------------
def build_conv_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that   A @ g ≤ M   encodes (f * g)_k ≤ M
    for every convolution index k = 0 … 2·n‑2.
    Row k contains the coefficients f_{k-j} for all admissible j.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        # admissible g‑indices j satisfy 0 ≤ j ≤ n‑1 and 0 ≤ k‑j ≤ n‑1
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        j = np.arange(j_min, j_max + 1, dtype=int)
        A[k, j] = f[k - j]
    return A


# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.
    The search explores several lengths and performs a few LP‑updates
    for each length.  The best normalised sequence found within the
    time budget is returned.
    """
    deadline = time.time() + budget_s - 0.1  # safety margin
    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Lengths to try – a modest range keeps the LP solves fast.
    lengths = [300, 500, 800, 1200]  # feel free to adjust

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- initialise with the arcsine distribution -----------------
        raw = arcsine_raw(n)
        raw_sum = raw.sum()
        # scale to L1‑norm = √(2·n)  (this is the internal representation)
        scale = np.sqrt(2.0 * n) / raw_sum
        f = raw * scale               # shape (n,)
        sqrt2n = np.sqrt(2.0 * n)

        # ---- a few LP‑updates -----------------------------------------
        max_iters = 30                # enough to see progress for modest n
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # current maximal self‑convolution (M = max(f*f))
            conv_f = np.convolve(f, f)
            M = conv_f.max()

            # linear programme: maximise Σ g_i  subject to (f*g)_k ≤ M
            A = build_conv_constraint_matrix(f)
            b = np.full(A.shape[0], M, dtype=float)
            c = -np.ones(n, dtype=float)            # maximise → minimise –Σg_i
            bounds = [(0, None)] * n                # g_i ≥ 0

            # solve – the problem is modestly sized, so the default HiGHS
            # solver works well.
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success:
                # LP failed – abort further updates for this length.
                print(f"[{n}] LP failed at iteration {it}: {res.message}")
                break

            g = res.x
            g_sum = g.sum()
            if g_sum <= 0:
                break

            # rescale g back to the same ℓ1‑norm (sqrt(2·n))
            g = g * (sqrt2n / g_sum)

            # convex combination – a small step keeps M non‑increasing.
            # start with a larger step, then shrink it.
            if it < 5:
                t = 0.5
            elif it < 15:
                t = 0.2
            else:
                t = 0.1
            f = (1.0 - t) * f + t * g
            # ℓ1‑norm stays (numerically) √(2·n); no need to renormalise.

            # ---- evaluate the normalised candidate --------------------
            norm_seq = (f / f.sum()).tolist()   # now sum = 1
            try:
                score = evaluate_sequence(norm_seq)
            except Exception:
                continue

            if score < best_score:
                best_score = score
                best_seq = norm_seq
                print(f"[len={n:4d}] iter={it:2d}  new best score={score:.6f}")

        # ---- optional tiny random refinements ------------------------
        # once the LP-trajectory saturates we try a few random tweaks.
        if best_seq is not None:
            n = len(best_seq)
            while time.time() < deadline:
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 1e-4   # tiny mass shift
                cand = best_seq.copy()
                cand[i] = max(0.0, cand[i] + delta)
                cand[j] = max(0.0, cand[j] - delta)
                s = sum(cand)
                if s < 1e-12:
                    continue
                cand = [x / s for x in cand]
                try:
                    score = evaluate_sequence(cand)
                except Exception:
                    continue
                if score < best_score:
                    best_score = score
                    best_seq = cand
                    print(f"[refine] length={n:<4d}  improved score={score:.6f}")
                # limit the refinement loop to stay within the budget
                if time.time() >= deadline:
                    break

    # ------------------------------------------------------------------
    if best_seq is None:                              # fallback
        best_seq = arcsine_raw(200).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"Best sequence length = {len(best_seq)} with score = {best_score:.6f}")
    return best_seq
