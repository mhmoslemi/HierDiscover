def generate_initial(n, kind, rng):
    if kind == 'geom':
        r = rng.uniform(0.001, 0.999)
        a = r ** np.arange(n)
    elif kind == 'power':
        p = rng.uniform(0.1, 4.0)
        a = (np.arange(1, n+1, dtype=float) ** -p)
    elif kind == 'mixed':
        r = rng.uniform(0.001, 0.999)
        p = rng.uniform(0.1, 3.0)
        a = (r ** np.arange(n)) * (np.arange(1, n+1, dtype=float) ** -p)
    elif kind == 'u_shape':
        # a_i = ((i+1)*(n-i))** -p
        p = rng.uniform(0.1, 3.0)
        i = np.arange(n)
        a = ((i+1)*(n-i)) ** -p
    elif kind == 'random':
        a = rng.gamma(1.0, 1.0, size=n)
    elif kind == 'symmetric_random':
        # generate half and mirror.
        half = rng.gamma(1.0,1.0,size=(n+1)//2)
        a = np.concatenate([half, half[-2-(n%2)::-1] if n%2==0 else half[-2::-1]]) # careful
    else: # default uniform
        a = np.ones(n)
    # optionally shuffle
    if kind != 'symmetric_random':
        if rng.rand() < 0.5:
            rng.shuffle(a)
    # normalize to sum 1
    a /= a.sum()
    return a

def gradient_max(a, max_idx):
    n = len(a)
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad

"""
Propose a low‑score step function.

The algorithm works in three stages:

1. **Initialisation** – generate many random candidates of various
   shapes (geometric, power‑law, mixed, U‑shaped, bimodal, pure
   Dirichlet).  Lengths are sampled between 2000 and 8000.
2. **Local optimisation** – each candidate is refined by a projected
   (Adam) sub‑gradient descent on the objective  
        score = 2·n·max(conv(a,a)) / (sum(a)²) .  
   The gradient of the “max‑convolution’’ is computed analytically;
   after every step the vector is projected back onto the probability
   simplex.
3. **Global polishing** – the current best sequence is repeatedly
   perturbed (small Gaussian noise on a few coordinates) and re‑optimised
   for a short budget; any improvement is kept.

The code respects the 1000 s time limit, prints intermediate scores and
returns the best list of non‑negative heights found.
"""

import numpy as np
import time
from math import sqrt

# ----------------------------------------------------------------------
# Helper building blocks (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto {x >= 0, sum(x)=1}. """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:                         # all entries are <=0
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray):
    """Return (max value, indices achieving the max) of a*a (full conv)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = np.max(conv)
    idx = np.where(np.abs(conv - max_val) <= 1e-12)[0]
    return float(max_val), idx.astype(int)


def _gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


def _score_from_array(a: np.ndarray) -> float:
    """Exact score used by evaluate_sequence (sum(a) is assumed > 0)."""
    n = a.size
    max_c, _ = _conv_max_fft(a)
    return 2.0 * n * max_c / (np.sum(a) ** 2)


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------


def _generate_initial(n: int, kind: str, rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n according to *kind*."""
    i = np.arange(n, dtype=float)

    if kind == "geom":
        r = rng.uniform(0.80, 0.999)            # moderate decay
        a = r ** i

    elif kind == "geom_small":
        r = rng.uniform(0.01, 0.50)            # fast decay
        a = r ** i

    elif kind == "power":
        p = rng.uniform(0.1, 3.0)
        a = (i + 1) ** (-p)

    elif kind == "mixed":
        r = rng.uniform(0.70, 0.99)
        p = rng.uniform(0.1, 2.0)
        a = (r ** i) * ((i + 1) ** (-p))

    elif kind == "u_shape":
        # large values at the two ends, small in the middle
        p = rng.uniform(0.1, 3.0)
        a = ((i + 1) * (n - i)) ** (-p)

    elif kind == "bimodal":
        r1 = rng.uniform(0.70, 0.99)
        r2 = rng.uniform(0.70, 0.99)
        a = (r1 ** i) + (r2 ** (n - 1 - i))

    else:  # pure Dirichlet / random
        a = rng.gamma(1.0, 1.0, size=n)

    # avoid exact zeros (they are harmless but may cause numerical issues)
    a = np.maximum(a, 1e-12)

    # random permutation – does NOT change the multiset of values,
    # but changes the convolution pattern.  For some families a permutation
    # is useful, for others (U‑shape, bimodal) we keep the natural order.
    if kind not in ("u_shape", "bimodal") and rng.random() < 0.5:
        rng.shuffle(a)

    # normalise to sum = 1  (scale does not affect the evaluation)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Local optimiser (projected Adam)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   max_iters: int,
                   time_deadline: float,
                   early_stop: float = 1.503) -> (np.ndarray, float):
    """
    Projected Adam optimisation of the score.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _score_from_array(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    while time.time() < time_deadline and it < max_iters:
        it += 1

        # current objective & active indices
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c  # sum(a) == 1 → denominator = 1

        # sub‑gradient of the max‑convolution
        grad = _gradient_for_max(a, max_idx)

        # Adam update (un‑projected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate schedule – larger at the beginning,
        # slowly decreasing (the 0.2 factor was experimentally good)
        lr = 0.2 / (np.sqrt(it) + 1.0)

        delta = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = _project_to_simplex(a - delta)

        # evaluate candidate
        cand_score = _score_from_array(a_candidate)

        if cand_score < cur_score - 1e-12:
            a = a_candidate
            cur_score = cand_score
        else:
            # no improvement – try a tiny random perturbation
            pert = a.copy()
            idx = np.random.choice(n, size=min(7, n), replace=False)
            pert[idx] += np.random.randn(len(idx)) * 0.001
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            pert_score = _score_from_array(pert)
            if pert_score < cur_score:
                a = pert
                cur_score = pert_score
                # reset Adam statistics after a successful jump
                m.fill(0.0)
                v.fill(0.0)
                it = 0

        # keep the best ever seen during this run
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 200 == 0:
                print(f"      [local] iter {it:04d} – new best {best_score:.6f}")

        # early stop if we already beat the target
        if best_score < early_stop:
            break

    return best_a, best_score

# Global search driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence(...).  The function respects a ~1000 s time budget
    and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 0.  Start from a previously discovered sequence if available.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # try to grab the global variable that may contain a known good construction
    known = globals().get("height_sequence_1")
    if isinstance(known, (list, tuple, np.ndarray)):
        try:
            known_arr = np.asarray(known, dtype=float)
            if known_arr.size > 0:
                known_arr = np.maximum(known_arr, 0.0)
                known_arr /= known_arr.sum()
                # quick local refinement
                sub_dead = min(deadline, time.time() + 12.0)
                opt_seq, opt_score = _adam_optimize(known_arr,
                                                    max_iters=5000,
                                                    time_deadline=sub_dead)
                if opt_score < best_score:
                    best_score = opt_score
                    best_seq = opt_seq.copy()
                    print(f"[seed] known sequence improved to {best_score:.6f}")
        except Exception:
            pass   # ignore any problem with the external variable

    # ------------------------------------------------------------------
    # 1.  Random restarts with diverse initial families.
    # ------------------------------------------------------------------
    shape_kinds = ["geom", "geom_small", "power", "mixed",
                   "u_shape", "bimodal", "random"]
    while time.time() < deadline:
        # pick a problem size – larger sizes give a finer asymptotic bound
        n = rng.integers(2000, 8000)   # 2 k … 8 k elements
        kind = rng.choice(shape_kinds)

        a0 = _generate_initial(n, kind, rng)

        # local optimisation – give each start at most ~12 s
        sub_deadline = min(deadline, time.time() + rng.uniform(8.0, 14.0))
        a_opt, score_opt = _adam_optimize(a0,
                                          max_iters=8000,
                                          time_deadline=sub_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global] n={n:04d}, kind={kind:<12} → score={best_score:.6f}")

        # stop early once we have beaten the advertised target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 2.  Fine‑grained polishing around the current best solution.
    # ------------------------------------------------------------------
    if best_seq is not None:
        polishing_dead = min(deadline, time.time() + 30.0)
        a_cur = best_seq.copy()
        while time.time() < polishing_dead:
            # jitter a few coordinates
            pert = a_cur.copy()
            n = pert.size
            idx = np.random.choice(n, size=min(10, n), replace=False)
            pert[idx] += np.random.randn(len(idx)) * 0.002
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            # short Adam run
            sub_d = min(polishing_dead, time.time() + 5.0)
            a_new, s_new = _adam_optimize(pert, max_iters=3000,
                                          time_deadline=sub_d)
            if s_new < best_score:
                best_score = s_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"[polish] improved to {best_score:.6f}")
            else:
                # keep current best as starting point for next jitter
                a_cur = best_seq.copy()
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # 3.  Fallback – uniform sequence (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n)

    # return plain Python list (the evaluator will clip automatically)
    return best_seq.tolist()
