   # iterative LP improvement
   iteration = 0
   max_lp_iters = 10
   while has_scipy and time.time() < deadline and iteration < max_lp_iters:
       iteration += 1
       cur_M = np.max(np.convolve(cur_seq, cur_seq))
       # Build constraint matrix

           A[k, start:] = cur_seq[start:][::-1]
       ub = np.full(2*n-1, cur_M, dtype=np.float64)
       c = -np.ones(n, dtype=np.float64)   # maximize sum

       # Solve LP
       res = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)]*n, method='highs')
       if not res.success:
           print("[LP] iteration", iteration, "failed, message:", res.message)
           break
       bvec = res.x
       sum_b = np.sum(bvec)
       if sum_b <= 0:

           break
       # Normalize to sum 1
       g = bvec / sum_b
       # Compute new max for g
       M_g = np.max(np.convolve(g, g))
       print(f"[LP] iter {iteration}: sum_b={sum_b:.4f}, M_g={M_g:.8f}, cur_M={cur_M:.8f}")
       if M_g < cur_M * 0.999:   # improvement

"""
Propose a low‑value step‑function using a linear‑programming refinement.

Algorithm
---------

1.  Initialise a normalized (sum = 1) decreasing sequence.
    The power‑law a_i ∝ (i+1)^(-p) with p≈0.55 is a good start.

2.  Iteratively improve the sequence by solving a linear program:

        maximise    Σ_j b_j
        subject to  (a * b)[k] ≤ M      for every convolution shift k
                     b_j ≥ 0

    where M = max_k (a*a)[k] is the current worst convolution value.
    The LP finds a vector b that can be added to a without violating the
    bound M.  After solving, the vector is renormalised to have sum = 1:

        g = b / Σ b .

    If the self‑convolution of g is strictly smaller than M we simply
    replace a ← g (this already lowers the score).  Otherwise we try a
    small convex combination a ← (1‑t)·a + t·g for t∈{0.5,0.2,0.1}.

    Only a handful of LP iterations (≈ 5) are needed; each uses a dense
    constraint matrix of size (2n‑1) × n, which is fast enough for n≈2000.

3.  Fine‑tune the final sequence with a cheap stochastic hill‑climber that
    moves a tiny amount of mass between two random indices while keeping
    the total sum unchanged.  Only strictly improving moves are kept.

The whole routine respects the 1000 s budget, prints progress messages,
and finally returns the best list of heights found.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper utilities
# --------------------------------------------------------------------------- #

def power_law_sequence(n: int, p: float) -> np.ndarray:
    """Decreasing power‑law a_i ∝ (i+1)^(-p) normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def evaluate_vec(vec: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(vec.tolist())


def build_conv_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (a * b)[k] = (A @ b)[k]
    for every shift k.  All entries are non‑negative.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)

    # rows 0 … n‑1
    for k in range(n):
        A[k, :k + 1] = a[:k + 1][::-1]

    # rows n … 2n‑2
    for k in range(n, m):
        start = k - (n - 1)                # first column that participates
        A[k, start:] = a[start:][::-1]

    return A


def hill_climb(vec: np.ndarray, deadline: float,
               max_no_improve: int = 2000) -> np.ndarray:
    """
    Very small stochastic hill‑climber.
    Moves at most 5 % of a source entry to a random target while preserving
    the total sum.  Only strictly improving moves are kept.
    """
    n = vec.size
    cur = vec.copy()
    cur_score = evaluate_vec(cur)
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and no_improve < max_no_improve:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.random() * 0.05 * cur[i]
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] += delta
        cand_score = evaluate_vec(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
    return cur

            cur_M = np.max(np.convolve(cur_seq, cur_seq))
            A = build_conv_constraint_matrix(cur_seq)
            ub = np.full(2 * n - 1, cur_M, dtype=np.float64)
            c = -np.ones(n, dtype=np.float64)   # maximise Σ b_j
            bounds = [(0.0, None)] * n

            # Solve the LP
            res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success:
                print(f"[LP] iteration {lp_iter} failed ({res.message})")
                break
            bvec = res.x
            sum_b = bvec.sum()
            if sum_b <= 0.0 or np.isnan(sum_b):
                print(f"[LP] iteration {lp_iter} produced zero/invalid sum")
                break

            # Normalise b to a probability vector g
            g = bvec / sum_b
            M_g = np.max(np.convolve(g, g))
            print(f"[LP] iter {lp_iter}: sum_b={sum_b:.4f}, "
                  f"M_g={M_g:.8f}, cur_M={cur_M:.8f}")

            # If the new self‑convolution is smaller, replace outright.
            if M_g < cur_M * 0.999:
                cur_seq = g
                cur_score = evaluate_vec(cur_seq)
                cur_M = M_g
                print(f"[LP]  accepted replacement, new score={cur_score:.6f}")
                continue

            # Otherwise try a few convex combinations.
            improved = False
            for t in (0.5, 0.2, 0.1):
                cand = (1.0 - t) * cur_seq + t * g
                cand_M = np.max(np.convolve(cand, cand))
                if cand_M < cur_M:
                    cur_seq = cand
                    cur_score = evaluate_vec(cur_seq)
                    cur_M = cand_M
                    print(f"[LP]  blended with t={t:.2f}, new score={cur_score:.6f}")
                    improved = True
                    break
            if not improved:
                print("[LP]  no improvement – terminating LP phase")
                break
    else:
        print("[LP] SciPy not available – skipping LP refinement")

    # --------------------------------------------------------------- #
    # 2) Final hill‑climb (always executed, cheap and safe).
    # --------------------------------------------------------------- #
    hill_deadline = min(deadline, time.time() + budget_s * 0.25)
    refined_seq = hill_climb(cur_seq, hill_deadline)
    refined_score = evaluate_vec(refined_seq)
    if refined_score < cur_score:
        cur_seq = refined_seq
        cur_score = refined_score
        print(f"[hill] improved after hill‑climb, final score={cur_score:.6f}")

    # --------------------------------------------------------------- #
    # 3) Return result
    # --------------------------------------------------------------- #
    print(f"[final] best score obtained: {cur_score:.6f} (length {n})")
    # The evaluation routine will clamp any tiny numerical drift.
    return cur_seq.tolist()
