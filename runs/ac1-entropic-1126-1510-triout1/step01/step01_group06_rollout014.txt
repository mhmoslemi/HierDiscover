    S = []
    sums = set()
    for i in range(n):

    return S

    w = exp_shift / np.sum(exp_shift)
    # compute gradient via correlation
    size_corr = 1 << ((n + 2 * n - 1 - 1).bit_length())
    # ensure w length = 2n-1, a length = n
    Fa = np.fft.rfft(a, n=size_corr)
    Fw = np.fft.rfft(w, n=size_corr)
    corr = np.fft.irfft(np.conj(Fw) * Fa, n=size_corr)
    grad = 2.0 * corr[n-1:n-1+n]
    return conv, max_c, w, grad

"""
Gradient‑based optimisation of a step function.

We minimise the score
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

Because the evaluation rescales the vector to sum≈1,
the problem is equivalent to minimising the ℓ∞‑norm of the
auto‑convolution of a probability vector (a_i ≥ 0, Σa_i = 1).

The algorithm:
* Initialise a with a random Dirichlet / geometric / power‑law shape.
* Optimise the smooth approximation of the max‑operator:
    w_k = exp(α·c_k) / Σ exp(α·c_k) ,  c = a*a .
  The gradient of Σ w_k·c_k w.r.t. a is
      g = 2 * correlation(w, a) .
* An Adam optimiser (projected onto the simplex) is used.
* The temperature α is increased during training so the approximation
  becomes tighter.
* Many short restarts are performed within the time budget;
  the globally best vector is returned.
"""

import time
import math
import numpy as np

# ---------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ---------------------------------------------------------------
# Core: one optimisation step – convolution, soft‑max weights & gradient
def _conv_and_grad(a: np.ndarray, alpha: float, fft_size: int):
    """
    For a given probability vector a compute
        c = a * a               (full convolution, length 2n‑1)
        w = soft‑max(α·c)       (weights that approximate the max)
        g = 2 * correlation(w, a)
    Returns (c, max(c), w, g).
    All transforms are performed with the same FFT size (a power of two).
    """
    n = a.size
    # forward FFT of a (used for both convolution and correlation)
    Fa = np.fft.rfft(a, n=fft_size)

    # ---------- self‑convolution ----------
    conv_full = np.fft.irfft(Fa * Fa, n=fft_size)[: 2 * n - 1]
    max_c = float(conv_full.max())

    # ---------- soft‑max weights ----------
    # exp(α·(c‑max_c)) is stable because the exponent is ≤ 0
    exp_vals = np.exp(alpha * (conv_full - max_c))
    w = exp_vals / exp_vals.sum()

    # ---------- gradient ----------
    Fw = np.fft.rfft(w, n=fft_size)
    # correlation = irfft(conj(Fw) * Fa)
    corr_full = np.fft.irfft(np.conj(Fw) * Fa, n=fft_size)
    # indices n‑1 … n‑1+n‑1 correspond to Σ a_j w_{i+j}
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]

    return conv_full, max_c, w, grad


# ---------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) gives a low
    evaluation score.  The routine runs a series of Adam‑optimised
    gradient descents on the smooth max‑convolution objective,
    each time starting from a different random shape.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a few seconds margin

    best_score = float("inf")
    best_seq = None

    # ----------------------------------------------------------------
    # hyper‑parameters (tuned to finish well within the 1000 s budget)
    n_min, n_max = 1500, 3000          # length of the step function
    max_iters_per_restart = 1800      # gradient steps per restart
    lr = 0.5                          # base learning‑rate for Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha_start, alpha_target = 5.0, 180.0   # temperature schedule
    alpha_mul = 1.025                     # geometric increase per iteration

    # ----------------------------------------------------------------
    while time.time() < deadline:
        # ---- pick a length and a random starting shape ----------------
        n = np.random.randint(n_min, n_max + 1)

        choice = np.random.choice(["geom", "power", "dirichlet"])
        if choice == "geom":
            # geometric decay a_i = r^i   (r close to 1)
            r = np.random.uniform(0.90, 0.999)
            a = r ** np.arange(n, dtype=float)
        elif choice == "power":
            p = np.random.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** -p)
        else:  # dirichlet‑like
            a = np.random.gamma(0.7, 1.0, size=n)

        a = a / a.sum()                     # normalise to a probability vector

        # ---- pre‑compute FFT size (same for conv & correlation) ------
        fft_size = 1 << ((3 * n - 2).bit_length())   # ≥ 3n‑2 for correlation

        # ---- Adam state ------------------------------------------------
        m = np.zeros(n)
        v = np.zeros(n)

        alpha = alpha_start

        # ---- optimisation loop ----------------------------------------
        for it in range(1, max_iters_per_restart + 1):
            if time.time() >= deadline:
                break

            conv, max_c, w, grad = _conv_and_grad(a, alpha, fft_size)

            # Adam update (projected)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # back onto the simplex (ensures non‑negativity & Σ=1)
            a = _project_to_simplex(a)

            # current upper bound (exact because Σa=1)
            cur_score = 2.0 * n * max_c

            # store the best seen so far (occasionally reconfirm with the
            # official evaluate_sequence to avoid tiny numerical drift)
            if cur_score + 1e-9 < best_score:
                # evaluate with the exact routine (fast enough for a few hits)
                exact = evaluate_sequence(a.tolist())
                if exact < best_score:
                    best_score = exact
                    best_seq = a.copy()
                    print(
                        f"Global improvement: n={n}, shape={choice}, it={it}, score={best_score:.6f}"
                    )
                    # early exit if we already beat the target
                    if best_score < 1.51:
                        break

            # increase the temperature – the approximation becomes sharper
            if alpha < alpha_target:
                alpha = min(alpha_target, alpha * alpha_mul)

        # stop completely once the target is reached
        if best_score < 1.51:
            break

    # ----------------------------------------------------------------
    # fallback: uniform distribution (should never be used)
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # return a plain Python list – the evaluator does the final clipping
    return best_seq.tolist()
