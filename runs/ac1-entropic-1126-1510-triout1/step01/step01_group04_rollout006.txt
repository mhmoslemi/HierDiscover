def simplex_projection(v):
    # v is np.array, maybe after clipping
    # Ensure sum = 1, values >=0.
    # Using algorithm: sort v descending, find rho, lambda, etc.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _hill_climb(seq, best_score, deadline):
    cur = seq.copy()
    n = len(seq)
    max_iter = 50000
    it =0
    while time.time() < deadline and it < max_iter:
        # choose i with cur[i] > 0 to remove
        i = np.random.choice(np.where(cur > 1e-12)[0])
        j = np.random.randint(n)
        if i == j:
            continue
        # amount to transfer
        delta = cur[i] * (np.random.rand() * 0.1) # up to 10%
        cur[i] -= delta
        cur[j] += delta
        score = _fast_score(cur)
        if score < best_score:
            best_score = score
            # keep change
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it+=1
        if it % 5000 ==0:
            print(f"Hill‑climb iteration {it}: best_score={best_score:.6f}")
    return cur, best_score

def _smooth_descent(seq, best_score, deadline):
    cur = seq.copy()
    n = len(seq)
    # parameters
    lr = 0.5
    gamma_vals = [5, 10, 20, 40, 80, 160, 320]
    for gamma in gamma_vals:
        if time.time() >= deadline:
            break
        it =0
        max_iter = 2000
        while time.time() < deadline and it < max_iter:
            conv = np.convolve(cur, cur)
            # compute softmax weights
            maxc = conv.max()
            # avoid overflow
            exp_term = np.exp(gamma * (conv - maxc))
            w = exp_term / exp_term.sum()
            # gradient via correlation
            grad = 2.0 * np.correlate(w, cur, mode='valid')
            nxt = cur - lr * grad
            nxt = np.maximum(nxt, 0.0)
            # project onto simplex
            nxt = _project_simplex(nxt)
            new_score = _fast_score(nxt)
            if new_score < best_score:
                cur = nxt
                best_score = new_score
                lr *= 1.02
            else:
                lr *= 0.7
                # optionally revert not needed because we didn't modify cur
            it += 1
            if it % 500 ==0:
                print(f"[smooth] gamma={gamma}, iter {it}, best_score={best_score:.6f}")
        # after gamma reduce lr
        lr = max(lr * 0.5, 1e-6)
    return cur, best_score

def propose_candidate(seed=0, budget_s=1000):
    """
    Search for a non‑negative step‑function minimizing
        2 * n * max(conv(a,a)) / (sum(a)**2).

    The algorithm proceeds in stages:
    1. Initialise a distribution (power‑law or previous best).
    2. Smooth soft‑max gradient descent with increasing temperature
       (gradients of a smoothed max of the convolution).
    3. Random perturbation to escape shallow minima.
    4. Stochastic hill‑climbing on the simplex.
    5. Fine‑grained max‑gradient descent (same as earlier code).
    """
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2
    n = 5000

    # initialise
    init = globals().get('height_sequence_1')
    if init is not None:
        a0 = np.array(init, dtype=np.float64)
        if len(a0) > n:
            a0 = a0[:n]
        elif len(a0) < n:
            a0 = np.concatenate([a0, np.zeros(n - len(a0))])
        if a0.sum() == 0:
            a0 = _make_power_law(n, -0.55)
        else:
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
    else:
        a0 = _make_power_law(n, -0.55)

    best_seq = a0.copy()
    best_score = _fast_score(best_seq)
    print("[init] fast_score", best_score)

    # Stage 1: smooth soft‑max gradient descent
    best_seq, best_score = _smooth_descent(best_seq, best_score, deadline)
    print("[smooth] best_score", best_score)

    # Stage 2: a little random perturbation
    for _ in range(5):
        if time.time() >= deadline:
            break
        perturb = np.random.normal(scale=0.005, size=n)
        cand = best_seq + perturb
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score

    # Stage 3: hill‑climbing
    best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The evaluation function is homogeneous, therefore we keep the vector on the
probability simplex (sum = 1).  The search proceeds in several stages:

1. Initialise – use a power‑law shape (a_i ∝ (i+1)^α) or the previously
   discovered best sequence (global ``height_sequence_1``).

2. Smooth‑max gradient descent – minimise a soft‑max approximation of the
   convolution peak.  The temperature (γ) is increased progressively,
   while a projected gradient step (learning‑rate η) keeps the vector on the
   simplex.

3. Small random perturbations – occasionally jump out of shallow minima.

4. Stochastic hill‑climbing – transfer a random fraction of mass between
   coordinates, accepting only improving moves.

5. Fine‑grained max‑gradient descent – use the exact sub‑gradient of the
   maximal convolution entry (the same idea as the reference implementation).

All steps respect the security checks inside ``evaluate_sequence`` and stop
before the supplied time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+offset+1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays.")
    # shift by max to improve numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # due to floating‑point errors the sum can be slightly off:
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return w / w_sum


def _fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective when the vector is already normalised.
    Returns 2 * n * max(conv(a, a)).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _smooth_descent(
    seq: np.ndarray, best_score: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv) using projected gradient descent.
    A sequence of temperatures (γ) is used; the learning rate is adapted
    depending on improvement.
    """
    cur = seq.copy()
    n = cur.shape[0]

    # schedule of “inverse temperatures”
    gammas = [5, 10, 20, 40, 80, 160, 320]
    lr = 0.5  # initial learning rate

    for gamma in gammas:
        if time.time() >= deadline:
            break
        it = 0
        max_iter = 2000
        while time.time() < deadline and it < max_iter:
            # 1) convolution of the current vector with itself
            conv = np.convolve(cur, cur)

            # 2) soft‑max weights approximating the arg‑max
            m = conv.max()
            # subtract the max for numerical stability
            exp_term = np.exp(gamma * (conv - m))
            w = exp_term / exp_term.sum()

            # 3) gradient of the smooth objective:
            #    ∂/∂a_i  Σ_k w_k (a * a)_k = 2 * Σ_k w_k a_{k-i}
            grad = 2.0 * np.correlate(w, cur, mode="valid")

            # 4) projected gradient step
            nxt = cur - lr * grad
            nxt = np.maximum(nxt, 0.0)
            nxt = _project_simplex(nxt)

            new_score = _fast_score(nxt)
            if new_score < best_score:
                cur = nxt
                best_score = new_score
                lr *= 1.02  # gently increase step when we are improving
            else:
                lr *= 0.70  # otherwise shrink
                # keep cur unchanged
            it += 1
            if it % 500 == 0:
                print(
                    f"[smooth] γ={gamma:4d}, it={it:4d}, best_score={best_score:.6f}"
                )
        # after each temperature halve the learning rate to stabilise
        lr = max(lr * 0.5, 1e-6)
    return cur, best_score


def _hill_climb(
    seq: np.ndarray, best_score: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex.
    Randomly transfers a small fraction of mass from a non‑zero entry i to a
    random entry j, keeping the sum invariant.  Only improving moves are kept.
    """
    cur = seq.copy()
    n = cur.shape[0]
    max_iter = 50000
    it = 0
    while time.time() < deadline and it < max_iter:
        # pick a donor with positive mass
        donors = np.where(cur > 1e-12)[0]
        if donors.size == 0:
            break
        i = np.random.choice(donors)
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # amount to move (up to 5 % of the donor's mass)
        delta = cur[i] * (np.random.rand() * 0.05)
        cur[i] -= delta
        cur[j] += delta
        score = _fast_score(cur)
        if score < best_score:
            best_score = score
            # keep the changes
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it % 5000 == 0:
            print(f"[hill‑climb] iteration {it:5d}, best_score={best_score:.6f}")
    return cur, best_score


def _max_gradient_step(
    seq: np.ndarray, best_score: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Fine‑grained descent on the exact max‑convolution entry.
    Uses the sub‑gradient of the active (binding) convolution constraints.
    """
    cur = seq.copy()
    n = cur.shape[0]
    step = 0.12
    it = 0
    while time.time() < deadline and it < 2000:
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        k = int(conv.argmax())  # index where the maximum occurs

        # sub‑gradient: for all i with 0 ≤ k‑i < n, ∂/∂a_i = 2 * a_{k-i}
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur)
        grad[idx] = 2.0 * cur[k - idx]

        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s   # re‑project onto the simplex
        new_score = _fast_score(nxt)
        if new_score < best_score:
            cur = nxt
            best_score = new_score
            step *= 0.99
        else:
            step *= 0.90
        it += 1
        if it % 500 == 0:
            print(f"[max‑grad] it={it:4d}, best_score={best_score:.6f}")
    return cur, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    ``evaluate_sequence`` score.  The function respects the 1000‑second time
    budget, prints intermediate progress and employs the multi‑stage
    optimisation pipeline described in the module docstring.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin
    n = 5000

    # --------------------------------------------------------------
    # 1) Initialise – use the previously discovered best if available
    # --------------------------------------------------------------
    init = globals().get("height_sequence_1")
    if init is not None:
        a0 = np.asarray(init, dtype=np.float64)
        if a0.size > n:
            a0 = a0[:n]
        elif a0.size < n:
            a0 = np.concatenate([a0, np.zeros(n - a0.size, dtype=np.float64)])
        if a0.sum() == 0.0:
            a0 = _make_power_law(n, -0.55)
        else:
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
    else:
        a0 = _make_power_law(n, -0.55)

    best_seq = a0.copy()
    best_score = _fast_score(best_seq)
    print(f"[init] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Smoothed soft‑max gradient descent
    # --------------------------------------------------------------
    best_seq, best_score = _smooth_descent(best_seq, best_score, deadline)
    print(f"[smooth] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Small random perturbations to escape local minima
    # --------------------------------------------------------------
    for rep in range(5):
        if time.time() >= deadline:
            break
        perturb = np.random.normal(scale=0.005, size=n)
        cand = best_seq + perturb
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
    print(f"[perturb] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Stochastic hill‑climbing
    # --------------------------------------------------------------
    best_seq, best_score = _hill_climb(best_seq, best_score, deadline)
    print(f"[hill‑climb] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Fine‑grained max‑gradient descent (exact max of convolution)
    # --------------------------------------------------------------
    best_seq, best_score = _max_gradient_step(best_seq, best_score, deadline)
    print(f"[final] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the result as plain Python floats
    # --------------------------------------------------------------
    final_sequence = [float(v) for v in best_seq.tolist()]
    print(
        f"Returning sequence of length {len(final_sequence)} "
        f"with evaluation score {best_score:.6f}"
    )
    return final_sequence
