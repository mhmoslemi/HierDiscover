"""
Propose a low‑score step‑function for the evaluation metric

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages.

1. **Parametric random search** – generate candidates from several
   families (power‑law, exponential, product of both and a truncated
   power‑law).  All candidates are normalised to sum‑to‑one; the best
   candidate found so far is kept.

2. **Simulated‑annealing (SA)** – starting from the best candidate, move a
   random amount of mass from one index to another while keeping the
   L1‑norm fixed.  Improvements are always accepted; worsening moves are
   accepted according to a temperature schedule (Metropolis rule).  The
   SA stage explores the neighbourhood of the current best point and
   usually finds a substantially lower max‑convolution.

3. **Projected sub‑gradient descent** – at the current point we identify
   *all* convolution indices that attain (or are numerically very close to)
   the maximum.  The sub‑gradient of the max‑convolution is the sum of the
   individual gradients of these active constraints.  A projected step
   (onto the probability simplex) is taken and the step size is
   adapted.  This phase refines the SA result to a (locally) optimal
   solution.

All three phases respect the supplied time budget (default 1000 s) and
print intermediate progress.  The returned object is a plain Python list
of non‑negative floats ready for `evaluate_sequence`.
"""

import time

def _eval_vec(vec: np.ndarray) -> float:
    """Wrapper that calls the external `evaluate_sequence`."""
    # `evaluate_sequence` expects a plain list of floats.
    return evaluate_sequence(vec.tolist())


def _normalise(vec: np.ndarray) -> np.ndarray:
    """Make the vector non‑negative and rescale it to sum to one."""
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        # Degenerate – return a uniform tiny vector to keep the optimiser alive
        vec = np.full_like(vec, 1.0 / vec.size)
        return vec
    return vec / s


def make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset+1)^exponent, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    return _normalise(vec)


def make_exponential(n: int, lam: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ exp(-lam·(i+offset+1)), normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = np.exp(-lam * idx)
    return _normalise(vec)


def make_power_exp(
    n: int, exponent: float, lam: float, offset: float = 0.0
) -> np.ndarray:
    """a_i ∝ (i+offset+1)^exponent * exp(-lam·(i+offset+1))."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = (idx ** exponent) * np.exp(-lam * idx)
    return _normalise(vec)


def make_truncated_power(
    n: int, exponent: float, offset: float = 0.0, cutoff: float = None
) -> np.ndarray:
    """a_i ∝ (i+offset+1)^exponent for i ≤ cutoff, otherwise 0."""
    if cutoff is None:
        cutoff = n
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    mask = idx <= cutoff
    vec = np.where(mask, idx ** exponent, 0.0)
    return _normalise(vec)

def _parametric_random_search(
    n: int,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Explore several parametric families under a time budget."""
    families = ("power", "exp", "power_exp", "trunc")
    start = time.time()
    iter_cnt = 0

    while time.time() - start < time_limit:
        fam = np.random.choice(families)
        if fam == "power":
            exp = -np.random.uniform(0.1, 2.0)
            off = np.random.uniform(0.0, 3.0)
            seq = make_power_law(n, exp, off)
        elif fam == "exp":
            lam = np.random.uniform(0.0001, 0.02)
            off = np.random.uniform(0.0, 3.0)
            seq = make_exponential(n, lam, off)
        elif fam == "power_exp":
            exp = -np.random.uniform(0.1, 2.0)
            lam = np.random.uniform(0.0001, 0.02)
            off = np.random.uniform(0.0, 3.0)
            seq = make_power_exp(n, exp, lam, off)
        else:  # "trunc"
            exp = -np.random.uniform(0.1, 2.0)
            off = np.random.uniform(0.0, 3.0)
            prop = np.random.uniform(0.5, 1.0)
            cutoff_val = prop * n
            seq = make_truncated_power(n, exp, off, cutoff_val)

        # quick sanity check – all‑zero vectors are useless
        if seq.sum() == 0.0:
            continue

        score = _eval_vec(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(
                f"[random] iter={iter_cnt:06d} new best score={best_score:.6f}"
            )
        iter_cnt += 1

    return best_seq, best_score


def _simulated_annealing(
    n: int,
    time_limit: float,
    cur: np.ndarray,
    cur_score: float,
) -> tuple[np.ndarray, float]:
    """Classic SA on the simplex – move mass between two positions."""
    best_seq = cur.copy()
    best_score = cur_score

    start = time.time()
    # linear cooling from T0 to ~0
    T0 = 0.02
    max_iters = 2_000_000  # generous upper bound; real stop is time

    for it in range(max_iters):
        now = time.time()
        if now - start > time_limit:
            break

        # temperature decays linearly with elapsed SA time
        frac = (now - start) / time_limit
        T = T0 * (1.0 - frac)

        # propose a transfer of mass
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or cur[i] == 0.0:
            continue

        # early SA uses larger moves, later moves shrink
        max_move_frac = 0.2 if it < max_iters * 0.1 else 0.04
        move_frac = np.random.rand() * max_move_frac
        delta = cur[i] * move_frac

        nxt = cur.copy()
        nxt[i] -= delta
        nxt[j] += delta

        nxt_score = _eval_vec(nxt)
        improve = nxt_score < cur_score
        if improve:
            accept = True
        else:
            # Metropolis acceptance
            accept = T > 0 and np.random.rand() < np.exp(-(nxt_score - cur_score) / T)

        if accept:
            cur = nxt
            cur_score = nxt_score
            if nxt_score < best_score:
                best_score = nxt_score
                best_seq = nxt.copy()

        if it % 25_000 == 0 and it > 0:
            print(
                f"[SA] iter={it:07d} cur_score={cur_score:.6f} best_score={best_score:.6f}"
            )

    return best_seq, best_score


def _projected_subgradient_descent(
    n: int,
    time_limit: float,
    cur: np.ndarray,
    cur_score: float,
) -> tuple[np.ndarray, float]:
    """Reduce the maximal convolution entry using the full active set."""
    start = time.time()
    step = 0.22  # initial step size (empirically chosen)
    it = 0

    while time.time() - start < time_limit and it < 5000:
        conv = np.convolve(cur, cur)
        max_val = conv.max()

        # recompute exact score for safety (small overhead)
        cur_score = _eval_vec(cur)

        # active constraints – all indices that are numerically at the max
        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # aggregate sub‑gradient over the active set
        grad = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * cur[k - idx]

        # projected step onto the simplex
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s

        nxt_score = _eval_vec(nxt)

        if nxt_score < cur_score:
            cur = nxt
            cur_score = nxt_score
            step *= 0.96  # gentle reduction on improvement
        else:
            step *= 0.70  # stronger reduction on failure

        it += 1
        if it % 500 == 0:
            print(
                f"[grad] iter={it:04d} step={step:.5f} cur_score={cur_score:.6f}"
            )
        if step < 1e-7:
            break

    return cur, cur_score

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation metric.
    The function runs for at most ``budget_s`` seconds and returns a list
    of non‑negative heights (the best one found).
    """
    np.random.seed(seed)

    # total time reserved for the whole routine (a small safety margin is kept)
    total_deadline = time.time() + budget_s - 0.5
    total_budget = budget_s - 0.5

    # Length of the sequence – 5 000 gives a good trade‑off between quality
    # and evaluation speed.
    n = 5000

    # --------------------------------------------------------------
    # 0) A cheap deterministic grid (power‑law) to obtain a starter.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    exponents = np.linspace(-0.70, -0.30, 81)  # finer than the previous run
    offsets = [0.0, 0.3, 1.0, 2.0]
    for off in offsets:
        for e in exponents:
            cand = make_power_law(n, e, off)
            sc = _eval_vec(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    print(f"[grid] best_score after exhaustive grid = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Parametric random search (≈30 % of the total budget)
    # --------------------------------------------------------------
    rand_budget = total_budget * 0.30
    best_seq, best_score = _parametric_random_search(
        n, rand_budget, best_seq, best_score
    )
    print(f"[random-search] best_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Simulated annealing (≈55 % of the total budget)
    # --------------------------------------------------------------
    # Reserve a few seconds for the final gradient phase.
    remaining = total_deadline - time.time() - 5.0
    if remaining > 1.0:
        sa_budget = remaining
        best_seq, best_score = _simulated_annealing(
            n, sa_budget, best_seq, best_score
        )
        print(f"[SA] best_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Projected sub‑gradient descent (use whatever time is left)
    # --------------------------------------------------------------
    grad_budget = max(0.0, total_deadline - time.time())
    if grad_budget > 0.5:
        best_seq, best_score = _projected_subgradient_descent(
            n, grad_budget, best_seq, best_score
        )
        print(f"[gradient] final_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list.
    # ------------------------------------------------------------------
    final_list = [float(v) for v in best_seq.tolist()]
    final_evaluation = _eval_vec(best_seq)
    print(
        f"Returning sequence of length {len(final_list)} with evaluation {final_evaluation:.6f}"
    )
    return final_list
