def propose_candidate(seed=123, budget_s=1000):
    # Determine length n
    try: n = len(height_sequence_1)
    except: n = 3000

    target_sum = math.sqrt(2.0 * n)

    # compute c_i base shape
    c = np.empty(n, dtype=float)
    c[0] = 1.0
    for i in range(1,n):
        c[i] = c[i-1] * (2*i-1) / (2*i)
    # normalize by projection onto simplex to satisfy sum.
    a = _project_to_simplex(c, target_sum)

    # if previous best available, maybe start from it also and keep best.

    # optimizer parameters
    beta = 5.0  # smoothing for max
    eta = 0.1   # base learning rate for Adam (step size)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    start = time.time()
    iter = 0
    # compute FFT length
    M = 2*n - 1
    L = 1 << ((M + n - 1).bit_length())   # power of two >= M + n - 1

    while time.time() < start + budget_s - 0.5 and iter < 2000000:
        # FFT of a
        Fa = np.fft.rfft(a, n=L)

        # convolution a*a in frequency domain
        conv = np.fft.irfft(Fa * Fa, n=L)[:M]   # linear conv length M
        # smooth max
        max_conv = conv.max()
        # compute softmax weights
        # subtract max for numerical stability
        scaled = beta * (conv - max_conv)
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()

        # FFT of w
        Fw = np.fft.rfft(w, n=L)

        # correlation gradient: r = ifft(Fa * conj(Fw))
        r = np.fft.irfft(Fa * np.conj(Fw), n=L)
        # grad_i = 2 * r[i + (M-1)]
        shift = M - 1
        grad = 2.0 * r[shift:shift + n]

        # Adam step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (iter + 1))
        v_hat = v / (1 - beta2 ** (iter + 1))
        step = - eta * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a + step
        # Project onto simplex
        a_new = _project_to_simplex(a_new, target_sum)

        # evaluate
        cur_score = evaluate_sequence(a_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{iter}] new best score = {best_score:.6f}")

        # occasional random perturbation

            noise = np.random.randn(n) * target_sum * 1e-3

        a = a_new
        iter += 1

"""
propose_candidate
-----------------

A synthesis‑based optimizer that starts from the analytical “square‑root
kernel’’ (the coefficients of (1‑z)^‑½) – a sequence whose self‑convolution
is perfectly flat on the first n entries.  The sequence is projected onto
the feasible simplex (non‑negative entries, fixed ℓ¹‑norm √(2 n)).  From
there a smooth approximation of the max‑convolution is minimized by a
projected Adam optimiser.  The smooth max is realised with a soft‑max
(weight w_k ∝ exp(β·conv_k)); the resulting gradient is a cross‑correlation
which can be evaluated efficiently with FFTs.  The algorithm runs for the
allocated time budget, occasionally injects small random perturbations
and finally performs cheap random “block’’ trials.  The best sequence found
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum z."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)            # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
def _c_series(n: int) -> np.ndarray:
    """
    Return the first n coefficients of (1 - z)^(-1/2),
    i.e. c_i = binom(2i,i) / 4**i  (positive, decreasing ~1/sqrt(i)).
    """
    c = np.empty(n, dtype=float)
    c[0] = 1.0
    for i in range(1, n):
        c[i] = c[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return c


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence.  The optimiser runs for up to `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # reserve a little margin

    # ------------------------------------------------------------------
    # Length of the sequence (use the reference if it exists)
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # the ℓ¹‑norm used by the benchmark

    # ------------------------------------------------------------------
    # Initial vector: analytical c‑series (flat autocorrelation) + projection.
    a0 = _c_series(n)
    a0 = _project_to_simplex(a0, target_sum)

    # Occasionally start from a random point (50 % chance)
    if have_ref and np.random.rand() < 0.5:
        a0 = np.random.rand(n)
        a0 = _project_to_simplex(a0, target_sum)

    a = a0.astype(float)

    # ------------------------------------------------------------------
    # Pre‑compute FFT size (next power of two ≥ 3·n)
    M = 2 * n - 1                     # length of linear convolution a*a
    L = 1 << ((3 * n - 2).bit_length())  # power‑of‑two >= 3n‑2

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base = target_sum * 0.2 / n           # base learning rate (≈0.5·average entry)

    m = np.zeros_like(a)                     # first moment
    v = np.zeros_like(a)                     # second moment

    # Soft‑max temperature for the smoothed max (larger => sharper)
    beta_smooth = 5.0

    # ------------------------------------------------------------------
    # Book‑keeping of the best solution
    conv_initial = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:M]
    best_score = 2.0 * n * conv_initial.max() / (target_sum ** 2)
    best_a = a.copy()
    print(f"[init] length={n}, initial score={best_score:.6f}")

    it = 0
    # Main optimisation loop
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) FFT of the current vector
        Fa = np.fft.rfft(a, n=L)

        # 2) Linear convolution a*a (needed for the objective)
        conv = np.fft.irfft(Fa * Fa, n=L)[:M]

        # 3) Smoothed max via soft‑max
        max_c = conv.max()
        # shift to improve numerical stability
        scaled = beta_smooth * (conv - max_c)
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()          # w_k ≥0, Σw_k = 1

        # 4) Gradient of the smoothed max:
        #    grad_i = 2 * Σ_k w_k * a_{k-i}  (cross‑correlation a ⋆ w)
        Fw = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(Fa * np.conj(Fw), n=L)   # linear correlation a ⋆ w
        shift = M - 1                                 # index offset for lag = 0
        grad = 2.0 * corr[shift:shift + n]

        # 5) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = - lr_base * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a + step
        a_new = _project_to_simplex(a_new, target_sum)

        # 6) Evaluate the new sequence (max of linear convolution)
        Fa_new = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(Fa_new * Fa_new, n=L)[:M]
        cur_score = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{it:08d}] improved score = {best_score:.6f}")

        # 7) Occasionally perturb to escape plateaus
        if (it % 5000 == 0 and it > 0) or (np.random.rand() < 1e-4):
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # After the main run, try a few cheap random “block’’ candidates.
    block_trials = 300
    for _ in range(block_trials):
        nb = np.random.randint(2, 7)                       # number of blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        # fast score evaluation (no extra FFTs needed – reuse the formula)
        Fa_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(Fa_c * Fa_c, n=L)[:M]
        sc = 2.0 * n * conv_c.max() / (target_sum ** 2)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
