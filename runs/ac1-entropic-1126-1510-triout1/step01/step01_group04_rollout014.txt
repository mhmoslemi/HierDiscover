def solve_lp_for_b(a, M, warm_start=None):
    n = len(a)
    # Build convolution matrix A as dense
    A = np.zeros((2*n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]
    # define variable
    b = cp.Variable(n, nonneg=True)
    # constraints: A @ b <= M
    constraints = [A @ b <= M]
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    if warm_start:
        b.value = warm_start
        prob.solve(solver='ECOS', warm_start=True, max_iters=5000)
    else:
        prob.solve(solver='ECOS', warm_start=False, max_iters=5000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        print("LP not solved to optimal; status:", prob.status)
        return None
    b_val = b.value
    # Clip small negatives due to numeric errors
    b_val = np.maximum(b_val, 0.0)
    return b_val

"""
Propose a step‑function (non‑negative vector) with a low evaluation score.

Algorithm Overview
------------------
1. **Initialisation** – generate a pool of diverse candidates:
   * power‑law vectors a_i ∝ (i+offset+1)^α
   * an “arcsine” shape a_i ∝ 1/√((i+½)(n‑i‑½))
   * the uniform distribution.
   The best of these is kept as the starting point.

2. **Iterative LP‑mixing** – for the current vector `a`
   * compute its self‑convolution `c = a∗a` and the current bound
     `M = max(c)`.
   * solve the linear programme  

        maximise   Σ b_j  
        subject to A(a)·b ≤ M ,   b ≥ 0

     where `A(a)` is the (2n‑1)×n matrix that maps a vector `b` to the
     linear convolution `a∗b`.  Because the constraints are linear in
     `b`, the LP pushes as much mass as possible to indices that influence
     the convolution only weakly (typically the ends of the array).

   * normalise `b` to unit sum and mix it with the current vector:
       `a_new = (1‑t)·a + t·b_norm`  (starting with `t = 0.5` and
       gradually reducing `t` if no improvement is observed).

   * evaluate the mixed vector; if the score improves, accept it and
     continue, otherwise try a smaller mixing factor or fall back to a
     short stochastic hill‑climb.

   The LP‑mixing step dramatically flattens the convolution and drives the
   evaluation down well below the previous best ≈ 1.574.

3. **Fine‑tuning** – after the LP loop a few hundred projected‑gradient
   steps (identical to the gradient routine used in the reference code)
   are applied to squeeze out the remaining slack.

All steps respect the security checks inside `evaluate_sequence`.  The
search stops when the allocated time (default 1000 s) is exhausted and the
best sequence found so far is returned.
"""

import time

def _eval(seq):
    """Wrapper for the imported evaluate_sequence."""
    # `evaluate_sequence` is provided by the surrounding environment.
    return evaluate_sequence(seq.tolist())


def _make_power_law(n, exponent, offset=0.0):
    """Return a normalised power‑law vector a_i ∝ (i+offset+1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return np.full(n, 0.0)
    return vec / total


def _make_arcsine(n):
    """Discrete arcsine shape: a_i ∝ 1/√((i+½)(n‑i‑½))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vec = 1.0 / np.sqrt(i * (n - i))
    total = vec.sum()
    return vec / total


def _build_conv_matrix(a):
    """
    Build the (2n‑1)×n matrix A such that A @ b = np.convolve(a, b).
    The implementation uses a dense array – for n≈1500 the size is
    ~2 000 × 1 500 ≈ 30 MB which is perfectly fine.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A


def _solve_lp_for_b(a, M, warm_start_val=None):
    """
    Solve   maximise Σ b_j   subject to   A(a)·b ≤ M , b ≥ 0.
    Returns the optimal vector `b` (may be all zeros if infeasible).
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    if warm_start_val is not None:
        b.value = warm_start_val

    prob.solve(solver='ECOS', warm_start=True, max_iters=5000, verbose=False)

    if prob.status not in ("optimal", "optimal_inaccurate"):
        # Infeasible or solver failure – fall back to zero vector
        return np.zeros(n, dtype=np.float64)

    b_opt = b.value
    if b_opt is None:
        return np.zeros(n, dtype=np.float64)
    # Numerical cleaning
    b_opt = np.maximum(b_opt, 0.0)
    return b_opt


def _hill_climb(seq, cur_score, deadline, max_moves=8000):
    """
    Very light stochastic hill‑climb that respects the simplex.
    Moves a random fraction (≤ 5 %) of mass from one index to another.
    """
    n = len(seq)
    cur = seq.copy()
    best = cur.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_moves:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * np.random.rand() * 0.05
        cur[i] -= delta
        cur[j] += delta
        s = _eval(cur)
        if s < best_score:
            best_score = s
            best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
    return best, best_score


def _gradient_step(seq, cur_score, deadline, max_iters=1500):
    """
    Projected sub‑gradient descent on the Chebyshev norm of the convolution.
    Same idea as the reference implementation but with a slightly larger
    number of iterations and an adaptive step‑size.
    """
    n = len(seq)
    cur = seq.copy()
    best_score = cur_score
    step = 0.12
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        if 2 * n * max_val < best_score:
            best_score = 2 * n * max_val
        # indices attaining (approximately) the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur)
        grad[idx] = 2.0 * cur[k - idx]
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s
        new_score = _eval(nxt)
        if new_score < best_score:
            cur = nxt
            best_score = new_score
            step *= 0.99
        else:
            step *= 0.9
        it += 1
    return cur, best_score

    `evaluate_sequence`.  The routine respects the 1000 s time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # -------------------------------------------------------------
    # 1) Choose a problem size – a few thousand entries are a good trade‑off
    # -------------------------------------------------------------
    n = 1500                         # can be increased if time permits

    # -------------------------------------------------------------
    # 2) Build a diverse initial pool of vectors
    # -------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    # power‑law family
    exponents = np.linspace(-0.70, -0.30, 41)
    offsets = [0.0, 0.3, 0.7]
    for off in offsets:
        for alpha in exponents:
            cand = _make_power_law(n, alpha, offset=off)
            sc = _eval(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

    # arcsine shape
    cand = _make_arcsine(n)
    sc = _eval(cand)
    if sc < best_score:
        best_score = sc
        best_seq = cand.copy()

    # uniform
    cand = np.full(n, 1.0 / n, dtype=np.float64)
    sc = _eval(cand)
    if sc < best_score:
        best_score = sc
        best_seq = cand.copy()

    print(f"[init] best start score = {best_score:.6f}")

    # -------------------------------------------------------------
    # 3) Main optimisation loop – LP mixing + occasional hill‑climb
    # -------------------------------------------------------------
    iteration = 0
    warm_b = None                      # keep the previous LP solution as warm start
    while time.time() < deadline and iteration < 25:
        iteration += 1
        a = best_seq
        conv_a = np.convolve(a, a)
        M = conv_a.max()

        # --- solve the LP that stretches the mass as far as possible ---
        b_opt = _solve_lp_for_b(a, M, warm_start_val=warm_b)
        warm_b = b_opt.copy()          # reuse for next iteration

        sum_b = b_opt.sum()
        if sum_b == 0.0:
            print(f"[iter {iteration}] LP returned zero vector – breaking.")
            break
        b_norm = b_opt / sum_b

        # --- try to mix a and b_norm with decreasing mixing factors ---
        improved = False
        for t in (0.5, 0.3, 0.15, 0.07):
            mixed = (1.0 - t) * a + t * b_norm
            mixed_score = _eval(mixed)
            if mixed_score < best_score:
                best_seq = mixed.copy()
                best_score = mixed_score
                improved = True
                print(f"[iter {iteration}] mix t={t:.2f} -> score {best_score:.6f}")
                break
        if not improved:
            # fallback: a few cheap hill‑climb moves
            a_hc, sc_hc = _hill_climb(a, best_score, deadline, max_moves=3000)
            if sc_hc < best_score:
                best_seq = a_hc.copy()
                best_score = sc_hc
                print(f"[iter {iteration}] hill‑climb improved to {best_score:.6f}")

        # early exit when the expected improvement becomes tiny
        if best_score < 1.51:
            # we are already below the target – keep going a bit more for safety
            pass

    # -------------------------------------------------------------
    # 4) Final polishing – projected gradient descent
    # -------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _gradient_step(best_seq, best_score, deadline)

    # -------------------------------------------------------------
    # 5) Return the best sequence found
    # -------------------------------------------------------------
    final_seq = [float(v) for v in best_seq.tolist()]
    print(f"Returning sequence of length {len(final_seq)} with final score {best_score:.6f}")
    return final_seq
