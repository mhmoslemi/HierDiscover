import time, numpy as np

def project_simplex(v):
    # ensure sum(v) = 1, v >= 0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def arcsine_sequence(n):
    seq = np.empty(n, dtype=np.float64)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i+1 < n:
            cur = cur * (2*i + 1.0) / (2.0 * (i+1))
    seq /= total
    return seq

def gradient_step(a, step):
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()
    # create mask for indices achieving max (within tolerance)
    tol = 1e-12
    mask = (conv >= M - tol).astype(np.float64)
    # compute gradient via convolution a * mask
    grad_full = np.convolve(a, mask, mode='full')
    grad = 2.0 * grad_full[n-1:n-1+n]
    # descent
    a_new = a - step * grad
    # project to simplex
    a_new = project_simplex(a_new)
    return a_new, M

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    best_seq = None
    best_score = float('inf')
    # Define lengths to try
    lengths = list(range(400, 1801, 200))  # up to 1800
    for n in lengths:
        if time.time() >= deadline:
            break
        # start from arcsine
        a = arcsine_sequence(n)
        # compute initial score
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max()  # sum=1
        # keep current best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[init] n={n} score={score:.6f}")
        # gradient descent loop
        step = 0.5 / n  # initial step size
        iteration = 0
        while time.time() < deadline:
            a, M = gradient_step(a, step)
            iteration += 1
            conv = np.convolve(a, a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[improve] n={n} iter={iteration} score={score:.6f}")
                # shrink step to fine-tune
                step = max(step * 0.9, 1e-8)
            # occasionally decay step
            if iteration % 500 == 0:
                step *= 0.9
            # if no improvement for long time
            if iteration >= 5000:
                break
        # after gradient loop, optional random refinement
        # small random perturbations
        for _ in range(200):
            if time.time() >= deadline:
                break
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * step
            cand = a.copy()
            cand[i] = max(0.0, cand[i] + delta)
            cand[j] = max(0.0, cand[j] - delta)
            cand = project_simplex(cand)
            conv = np.convolve(cand, cand)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                a = cand
                print(f"[refine] n={n} score={score:.6f}")
    print(f"Best overall score {best_score:.6f} with length {len(best_seq)}")
    return best_seq.tolist()

"""
A simulated‑annealing optimiser for the “minimise max self‑convolution”
problem.

The score for a normalised (∑a_i = 1) non‑negative sequence a of length n
is   score = 2·n·max( a * a ) .
The algorithm

1. Starts from the classic arcsine ( (1‑z)^‑½ ) construction – a very
   strong baseline.
2. For several problem sizes (n ≃ 800 … 2000) it performs a
   temperature‑controlled random walk:
      * pick two positions i≠j,
      * transfer a small amount of mass from one to the other while
        keeping the total sum equal to one,
      * accept the move if the score improves or with a probability
        exp(−Δ/T) (standard Metropolis rule).
   The temperature T and the transfer magnitude slowly decay,
   allowing a coarse exploration early and fine tuning later.
3. Whenever a new best score is found it is printed and kept.
4. After the annealing for a given length a short “local‑refinement”
   phase makes tiny random exchanges that are always kept if they improve
   the score.

The whole search respects the 1000 s time budget, never uses closures,
and returns the best sequence found (as a plain list of floats).  The
sequence is already normalised, non‑negative and well within the bounds
checked by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    # find the largest rho where u_rho > cssv_rho / rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]               # last true index
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Return the normalised (sum = 1) arcsine sequence of length n.

    Coefficients are C_i = binom(2i,i)/4**i generated by the stable recurrence
        C_{i+1} = C_i * (2*i + 1) / (2*(i+1)).
    """
    seq = np.empty(n, dtype=np.float64)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1.0) / (2.0 * (i + 1))
    seq /= total
    return seq


# ----------------------------------------------------------------------
def score_of(seq: np.ndarray) -> float:
    """2·n·max(a*a)  (the sequence is already normalised to sum = 1)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Search for a low‑score step‑function sequence under the given time budget."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1          # safety margin

    best_seq: np.ndarray | None = None
    best_score = float('inf')

    # a few lengths – larger lengths give a better asymptotic bound,
    # smaller lengths are cheaper to optimise.
    lengths = [800, 1200, 1600, 2000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # 1) start from the arcsine construction for this length
        a = arcsine_sequence(n)          # normalised, non‑negative
        cur_score = score_of(a)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] length={n:<4d}  score={cur_score:.6f}")

        # --------------------------------------------------------------
        # 2) simulated annealing
        avg = 1.0 / n                     # typical magnitude of an entry
        max_delta = avg * 4.0            # start with a fairly aggressive move size
        temperature = cur_score * 0.05   # initial temperature (5 % of the score)
        it = 0

        while time.time() < deadline:
            # pick two distinct indices
            i, j = np.random.choice(n, 2, replace=False)

            # random signed transfer – limited by the available mass at the donor
            delta = (np.random.rand() - 0.5) * max_delta   # in [‑max_delta/2, +max_delta/2]

            cand = a.copy()
            if delta >= 0:
                # move from j to i
                transfer = min(delta, cand[j])
                cand[i] += transfer
                cand[j] -= transfer
            else:
                # move from i to j
                transfer = min(-delta, cand[i])
                cand[i] -= transfer
                cand[j] += transfer

            # (the move keeps the sum exactly 1, only numerical noise may appear)
            cand_score = score_of(cand)
            dscore = cand_score - cur_score

            # Metropolis acceptance
            if dscore < 0 or np.random.rand() < np.exp(-dscore / max(temperature, 1e-12)):
                a = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[improve] length={n:<4d}  iter={it:<6d}  score={cur_score:.6f}")

            # temperature and step‑size schedules
            it += 1
            if it % 500 == 0:
                temperature *= 0.9
                max_delta *= 0.95
            if it >= 200_000:          # stop the inner loop after a generous number of steps
                break

        # --------------------------------------------------------------
        # 3) tiny deterministic refinement – always keep improvements
        if best_seq is None:
            continue
        a = best_seq.copy()
        for _ in range(200):
            if time.time() >= deadline:
                break
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * avg * 0.5   # very small tweaks
            cand = a.copy()
            if delta >= 0:
                transfer = min(delta, cand[j])
                cand[i] += transfer
                cand[j] -= transfer
            else:
                transfer = min(-delta, cand[i])
                cand[i] -= transfer
                cand[j] += transfer
            # re‑project onto the simplex to kill any accumulated rounding error
            cand = project_simplex(cand)
            cand_score = score_of(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                print(f"[refine] length={n:<4d}  score={cand_score:.6f}")

    # ----------------------------------------------------------------------
    if best_seq is None:
        # (should never happen) fall back to a tiny uniform vector
        best_seq = np.full(1, 1.0)

    print(f"Best overall sequence length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq.tolist()
