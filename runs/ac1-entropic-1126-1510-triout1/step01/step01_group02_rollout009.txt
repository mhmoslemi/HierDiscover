The algorithm works on the probability simplex (all entries ≥0, sum = 1)

It starts from the best known construction (global ``height_sequence_1`` if
available, otherwise a uniform vector) and performs two phases:

1. **Linear‑programming flattening** – given the current sequence *f* we
   maximise the total mass of a new sequence *g* under the linear
   constraints  (f * g)[k] ≤ M, where M = max(f * f).  This is a linear
   program (LP) because the convolution is linear in *g*.
   The LP pushes mass into positions that are not “seen’’ by the current
   large convolution entry, thereby flattening the autocorrelation.
   The feasible optimum g is normalised back to sum = 1 and taken as the
   next candidate (or blended with *f* if the direct replacement does not
   improve the score).

2. **Fine‑grained projected‑gradient / annealing** – after the LP phase
   (or if the LP is unavailable) a short simulated‑annealing loop with
   projected gradient steps and random pairwise swaps refines the best
   solution.

Both phases respect the non‑negativity constraints, the sum‑normalisation
and the clipping performed inside ``evaluate_sequence``.  The routine runs
until the time budget is exhausted (leaving a small safety margin) and

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Projected gradient step that tries to reduce the maximal convolution entry
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)

    g = np.zeros_like(seq)
    g[mask] = 2.0 * seq[j[mask]]

    new_seq = seq - eta * g
    return _project_simplex(new_seq)


# ----------------------------------------------------------------------
# Random pairwise mass‑swap (preserves the sum exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Linear‑programming step: maximise sum(g) subject to f * g ≤ M
# ----------------------------------------------------------------------
def _lp_step(seq: np.ndarray):
    """
    Returns a new sequence g (normalised to sum = 1) that maximises total
    mass under the convolution bound of the current sequence.
    If the LP is infeasible or unbounded ``None`` is returned.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import diags
    except Exception:  # SciPy not available
        return None, None

    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    M = float(conv.max())

    # Build Toeplitz matrix A where A_{k,j} = seq[k-j] (0 if out of range).
    # Each diagonal d (offset = d) is constant = seq[d].
    data = [np.full(n, seq[d]) for d in range(n)]
    offsets = np.arange(0, n)
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    b = np.full(2 * n - 1, M, dtype=float)

    # Bounds: keep the variables that correspond to zero entries of seq fixed at 0
    eps = 1e-14
    bounds = [(0, None) if seq[i] > eps else (0, 0) for i in range(n)]
    c = -np.ones(n)  # minimise -sum(g)  <=> maximise sum(g)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "disp": False},
    )
    if not res.success:
        return None, None

    g0 = res.x
    total = float(g0.sum())
    if total <= 0:
        return None, None
    g = g0 / total
    return g, M


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Returns a list of non‑negative heights (a step function) that strives
    to minimise ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds, using an LP‑based flattening phase followed by a
    short gradient‑annealing refinement.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 2.0)  # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise the search vector (down‑sample if it is too long)
    # ------------------------------------------------------------------
    max_n = 3000                     # keep LP matrix manageable
    try:
        base = np.array(height_sequence_1, dtype=float)
        # Down‑sample to at most max_n entries by simple averaging
        if base.size > max_n:
            step = base.size // max_n
            trimmed = []
            for i in range(max_n - 1):
                s = i * step
                e = s + step
                trimmed.append(base[s:e].mean())
            # last block takes the remainder
            trimmed.append(base[(max_n - 1) * step :].mean())
            base = np.array(trimmed, dtype=float)
    except Exception:
        # fallback: uniform vector of size max_n
        base = np.full(max_n, 1.0 / max_n, dtype=float)

    # Normalise onto the simplex (sum = 1) and protect against all‑zero case
    if base.sum() == 0:
        n = base.size
        base = np.full(n, 1.0 / n, dtype=float)
    else:
        base = base / base.sum()

    current = _project_simplex(base.copy())
    best = current.copy()
    best_score = evaluate_sequence(best.tolist())
    current_score = best_score

    it = 0
    last_improve_iter = 0
    # ------------------------------------------------------------------
    # Phase 1 – LP flattening
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # Solve the LP based on the present sequence
        # ------------------------------------------------------------------
        g, M = _lp_step(current)

        if g is None:   # fall back to a cheap random move
            sigma = 0.02 / np.sqrt(it)
            candidate = _random_pair_move(current, sigma)
        else:
            # Candidate is the LP solution (already normalised)
            candidate = g

        # ------------------------------------------------------------------
        # Evaluate the candidate
        # ------------------------------------------------------------------
        cand_score = evaluate_sequence(candidate.tolist())
        if cand_score < best_score:
            best_score, best = cand_score, candidate.copy()
            last_improve_iter = it
            print(f"[Iter {it:6d}] New best score (LP/RA) : {best_score:.6f}")

        # ------------------------------------------------------------------
        # Acceptance rule (Metropolis style)
        # ------------------------------------------------------------------
        if cand_score < current_score:
            accept = True
        else:
            # Linear cooling schedule for the annealing temperature
            elapsed_frac = (time.time() - start) / (deadline - start)
            temperature = (1.0 - elapsed_frac) + 1e-4 * elapsed_frac
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score

        # ------------------------------------------------------------------
        # Simple stagnation detector – random restart if stuck too long
        # ------------------------------------------------------------------
        if it - last_improve_iter > 4000:
            perturb = np.random.rand(current.size)
            perturb = perturb / perturb.sum()
            # blend a tiny random direction into the best point
            current = _project_simplex(0.99 * best + 0.01 * perturb)
            current_score = evaluate_sequence(current.tolist())
            last_improve_iter = it

        # Early exit if we already beat the target
        if best_score < 1.503:
            print(f"Target reached (score={best_score:.6f}) at iteration {it}")
            break

    # ------------------------------------------------------------------
    # Phase 2 – fine‑grained gradient / annealing refinement (uses the
    #           same schedule as the reference implementation)
    # ------------------------------------------------------------------
    temp_start = 1.0
    temp_end = 1e-4
    while time.time() < deadline:
        it += 1
        elapsed = (time.time() - start) / (deadline - start)
        temperature = temp_start * (1.0 - elapsed) + temp_end * elapsed

        # Choose move type
        if np.random.rand() < 0.6:
            eta = 0.5 / np.sqrt(it)
            candidate = _gradient_step(current, eta)
        else:
            sigma = 0.2 / np.sqrt(it)
            candidate = _random_pair_move(current, sigma)

        cand_score = evaluate_sequence(candidate.tolist())
        if cand_score < best_score:
            best_score, best = cand_score, candidate.copy()
            print(f"[Iter {it:6d}] New best after refinement: {best_score:.6f}")

        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score

        if best_score < 1.503:
            print(f"Target reached (score={best_score:.6f}) at iteration {it}")
            break

    # ------------------------------------------------------------------
    # Return the best sequence encountered (as a plain Python list)
    # ------------------------------------------------------------------
    return best.tolist()
