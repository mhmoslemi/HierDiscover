def _optimise_one_length_fft(n, time_deadline, init="arcsine", max_iters=50000, eta0=0.5):
    if init=="arcsine":
        a = _arcsine_distribution(n)
    elif init=="random":
        a = np.random.rand(n)
        a /= a.sum()
    else:
        a = np.full(n, 1.0/n)
    # ensure symmetry
    a = (a + a[::-1]) / 2

    best_seq = a.copy()
    conv = _conv_fft(a)
    best_score = 2.0 * n * conv.max()

    for it in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break

        conv = _conv_fft(a)
        # find maximum index
        k_max = int(np.argmax(conv))
        max_b = conv[k_max]
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={it} score={cur_score:.6f}")

        # gradient
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        # subgradient step
        eta = eta0 / np.sqrt(it)
        a = a - eta * g

        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # symmetrization
        a = (a + a[::-1]) / 2.0

        # random jitter
        if it % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)
            a = (a + a[::-1]) / 2.0

    return best_seq, best_score

    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n, endpoint=False) + 0.5/old_n
    x_new = np.linspace(0,1,new_n, endpoint=False) + 0.5/new_n
    new_seq = np.interp(x_new, x_old, seq, left=0, right=0) # ensures positivity
    # enforce symmetry
    new_seq = (new_seq + new_seq[::-1]) / 2
    new_seq /= new_seq.sum()
    return new_seq

def _conv_fft(a):
    n = a.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n - 1]
    return conv

def _upsample(seq, new_n):
    old_n = len(seq)
    # positions for original: (i+0.5)/old_n
    old_pos = (np.arange(old_n) + 0.5) / old_n
    new_pos = (np.arange(new_n) + 0.5) / new_n
    new_seq = np.interp(new_pos, old_pos, seq, left=seq[0], right=seq[-1])
    # Ensure nonnegative
    new_seq = np.maximum(new_seq, 0.0)
    # Symmetrize
    new_seq = (new_seq + new_seq[::-1]) / 2.0
    # Normalize
    total = new_seq.sum()
    if total == 0:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq /= total
    return new_seq

"""
Optimise a non‑negative step‑function for the `evaluate_sequence` score.

The algorithm follows a fast projected sub‑gradient descent on the
maximum entry of the self‑convolution.  Key ideas:

* Work on the probability simplex (∑a = 1) – the score is scale‑invariant.
* Use an FFT for the convolution (O(n log n)) so that large n (≈10 k) is cheap.
* Enforce symmetry a_i = a_{n‑1‑i} after every projection – symmetric shapes
  give a flatter autocorrelation.
* Initialise with the discrete arcsine law, then up‑sample the current best
  when moving to a larger length.
* The sub‑gradient is summed over all almost‑maximal convolution entries
  (within a tiny relative tolerance) to flatten the whole peak region.
* A simple Euclidean projection onto the simplex keeps the iterate feasible.
* Random jitter is added occasionally to escape plateaus.
* The search is time‑bounded (default 1000 s) and keeps the best sequence
  discovered overall.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: a_i ∝ 1/√((i+½)(n‑i‑½)). Returns a probability vector."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length, then symmetrise."""
    old_n = seq.shape[0]
    old_pos = (np.arange(old_n) + 0.5) / old_n
    new_pos = (np.arange(new_n) + 0.5) / new_n
    new_seq = np.interp(new_pos, old_pos, seq, left=seq[0], right=seq[-1])
    new_seq = np.maximum(new_seq, 0.0)
    # enforce symmetry
    new_seq = (new_seq + new_seq[::-1]) / 2.0
    total = new_seq.sum()
    if total == 0:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq /= total
    return new_seq


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the discrete linear convolution a * a using an FFT.
    Returns a real array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray, rel_eps: float = 1e-9) -> np.ndarray:
    """
    Compute the sub‑gradient of the maximal convolution entry.
    All entries k for which conv[k] is within rel_eps of the maximum contribute.
    """
    n = a.shape[0]
    max_b = conv.max()
    # indices that are (almost) active
    active = np.where(conv >= max_b * (1 - rel_eps))[0]
    g = np.zeros_like(a)
    rng = np.arange(n)
    for k in active:
        idx = k - rng
        mask = (idx >= 0) & (idx < n)
        g[mask] += 2.0 * a[idx[mask]]
    return g


# ----------------------------------------------------------------------
def _optimise_one_length_fft(
    n: int,
    time_deadline: float,
    init,
    max_iters: int = 50000,
    eta0: float = 0.5,
    rel_eps: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for a fixed length `n`.
    Returns the best sequence encountered and its score (2·n·max(conv)).
    """
    # ---- initialise ----------------------------------------------------
    if isinstance(init, str):
        if init == "arcsine":
            a = _arcsine_distribution(n)
        elif init == "random":
            a = np.random.rand(n)
            a /= a.sum()
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(init, dtype=np.float64, copy=True)

    a = _project_onto_simplex(a, s=1.0)
    a = (a + a[::-1]) / 2.0                          # enforce symmetry

    # ---- keep the best seen -------------------------------------------
    conv = _conv_fft(a)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()

    # ---- main loop ----------------------------------------------------
    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        conv = _conv_fft(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n} iter={it}  score={best_score:.6f}"
            )

        # sub‑gradient (sum over all nearly maximal positions)
        g = _subgradient(a, conv, rel_eps=rel_eps)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection
        a = a - eta * g
        a = _project_onto_simplex(a, s=1.0)

        # enforce symmetry
        a = (a + a[::-1]) / 2.0

        # occasional tiny random shake to escape flat regions
        if it % 400 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)
            a = (a + a[::-1]) / 2.0

    return best_seq, best_score
