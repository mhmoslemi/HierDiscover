"""
Propose a low‑evaluation step function.

The search works on a compact 4‑parameter family

    a_i = (i + 1 + α)^{-p} · (1 - i/(n‑1))^{β} · exp(-γ·i/(n‑1))

where

* p  – power‑law exponent (around 0.5),
* α  – an additive shift that reduces the very first values
       (preventing the edge‐convolution a_0² from dominating),
* β  – a polynomial taper that suppresses the tail (the factor (1‑x)^β),
* γ  – an optional mild exponential taper for extra fine‑tuning.

All parameters are non‑negative and the sequence length n is fixed
(4000 by default – large enough for the continuous asymptotics while
still cheap to convolve).  A coarse grid is explored first, then a
fine local grid around the best point, and finally a short stochastic
hill‑climb that perturbs the best sequence directly.  The best sequence
found within the time budget (≈1000 s) is returned.

The objective being minimised is exactly the value returned by the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a sequence of non‑negative heights that (empirically) minimises
    ``evaluate_sequence``.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000 s).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # keep a safety margin

    # ------------------------------------------------------------------
    # Fixed problem size – 4000 gives a good trade‑off between quality
    # and speed (convolution ~ O(n log n) ≈ 5·10⁴ operations).
    # ------------------------------------------------------------------
    n = 4000
    idx = np.arange(n, dtype=np.float64)                # i = 0 … n‑1
    idx1 = idx + 1.0                                     # i+1
    one_minus = 1.0 - idx / (n - 1)                      # 1 - i/(n‑1)

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate given its four parameters.
    # ------------------------------------------------------------------
    def eval_params(p, alpha, beta, gamma):
        """Build the sequence from the four parameters and evaluate it."""
        # power‑law with additive shift
        seq = (idx1 + alpha) ** (-p)

        # polynomial tail (β) – the tail vanishes at the right edge
        if beta != 0.0:
            seq *= np.power(one_minus, beta)

        # optional exponential taper
        if gamma != 0.0:
            seq *= np.exp(-gamma * idx / (n - 1))

        # convert to plain Python list for evaluate_sequence
        return evaluate_sequence(seq.tolist()), seq

    best_score = float("inf")
    best_seq = None
    best_params = None

    # ------------------------------------------------------------------
    # 1️⃣ Coarse grid search
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.45, 0.55, 11)               # around the theoretical 0.5
    alpha_factors = np.linspace(0.0, 0.5, 11)          # α = factor * n
    beta_vals = np.linspace(0.0, 1.0, 11)              # tail exponent
    gamma_vals = [0.0]                                 # start without exponential taper

    for p in p_vals:
        # pre‑compute the power‑law part for each possible α to avoid repeated pow()
        for af in alpha_factors:
            alpha = int(round(af * n))
            # (idx1 + α)^{-p} is the same for all β, γ – compute once
            base = (idx1 + alpha) ** (-p)

            for beta in beta_vals:
                # combine the tail; if β == 0 skip the power to save time
                tail = one_minus ** beta if beta != 0.0 else np.ones_like(one_minus)

                for gamma in gamma_vals:
                    seq = base * tail
                    if gamma != 0.0:
                        seq = seq * np.exp(-gamma * idx / (n - 1))

                    score = evaluate_sequence(seq.tolist())
                    if score < best_score:
                        best_score = score
                        best_seq = seq.copy()
                        best_params = (p, alpha, beta, gamma)

                    if time.time() > deadline:
                        break
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Fine local grid around the best coarse point
    # ------------------------------------------------------------------
    if best_params is not None and time.time() < deadline:
        p0, alpha0, beta0, gamma0 = best_params
        # refine around the best values
        p_fine = np.linspace(max(0.3, p0 - 0.02), min(0.7, p0 + 0.02), 9)
        alpha_fine = np.linspace(max(0, alpha0 - int(0.05 * n)),
                                 min(n, alpha0 + int(0.05 * n)),
                                 9, dtype=int)
        beta_fine = np.linspace(max(0.0, beta0 - 0.2), beta0 + 0.2, 9)
        gamma_fine = np.linspace(0.0, 0.2, 5)          # small exponential tweak

        for p in p_fine:
            base_common = (idx1) ** (-p)               # the part without shift
            for alpha in alpha_fine:
                base = (idx1 + alpha) ** (-p)           # recompute with shift
                for beta in beta_fine:
                    tail = one_minus ** beta if beta != 0.0 else np.ones_like(one_minus)
                    for gamma in gamma_fine:
                        seq = base * tail
                        if gamma != 0.0:
                            seq = seq * np.exp(-gamma * idx / (n - 1))

                        score = evaluate_sequence(seq.tolist())
                        if score < best_score:
                            best_score = score
                            best_seq = seq.copy()
                            best_params = (p, alpha, beta, gamma)

                        if time.time() > deadline:
                            break
                    if time.time() > deadline:
                        break
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # 3️⃣ Stochastic hill‑climb on the best sequence (few seconds)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # work directly on the best sequence; scaling does not matter for the score
        cur_seq = best_seq.copy()
        cur_score = best_score

        # pre‑compute a mask that tells which convolution index gives the maximum
        # – we recompute it each iteration because the max may move
        while time.time() < deadline:
            # pick a random index and perturb it multiplicatively
            i = np.random.randint(0, n)
            perturb = 1.0 + np.random.normal(scale=0.15)  # 15 % change typical
            new_seq = cur_seq.copy()
            new_seq[i] = max(0.0, new_seq[i] * perturb)   # keep non‑negative

            # evaluate – keep the full list conversion inside evaluate_sequence
            new_score = evaluate_sequence(new_seq.tolist())

            # accept only improvements (simple hill‑climb)
            if new_score < cur_score:
                cur_seq = new_seq
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                    # no need to store parameters any more

    # ------------------------------------------------------------------
    # Report the final result
    # ------------------------------------------------------------------
    if best_params is not None:
        p, alpha, beta, gamma = best_params
        print(
            f"Best parameters → p={p:.5f}, α={alpha} (α/n={alpha/n:.4f}), "
            f"β={beta:.5f}, γ={gamma:.5f}"
        )
    print(f"Best objective = {best_score:.6f} with length = {n}")

    # ``evaluate_sequence`` works with plain Python lists; convert before returning.
    return best_seq.tolist()
