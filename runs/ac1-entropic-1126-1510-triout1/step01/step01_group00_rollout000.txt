def conv_fft(x, y):
    n = len(x) + len(y) - 1
    m = 1 << (n - 1).bit_length()  # next power of two
    X = np.fft.rfft(x, n=m)

    Z = X * Y
    conv = np.fft.irfft(Z, n=m)
    return conv[:n]

        a = a - eta * grad

        # enforce nonnegativity and symmetry
        a = np.maximum(a, 0.0)
        a = (a + a[::-1]) / 2.0

        # renormalize sum to 1
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0/n)

"""
Propose a low‑score step function.

Algorithm
---------
* Use a symmetric probability distribution on the simplex (L¹‑mass 1).
* Optimize the smoothed max of the auto‑convolution:
      f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)               (β≥0)
  Its gradient is a weighted sum of the gradients of the individual
  convolution entries.  The weight of entry k is proportional to
      exp(β·((a*a)_k – max_k (a*a)_k)).
* Perform Euclidean gradient descent with a diminishing step size
  ηₜ = η₀ / √t, followed by:
    – projection onto the non‑negative orthant,
    – symmetrisation a←(a+rev(a))/2,
    – renormalisation to sum‑1 (simple simplex projection).
* β starts small (≈0.5) and is multiplied by a factor ≈1.01 each iteration
  until a modest cap (β_max≈5).  This makes the objective initially smooth
  (average autocorrelation) and later focus on the true ∞‑norm.
* The convolutions are performed by FFT (≈O(n log n) per iteration), so
  thousands of iterations are feasible for n≈8 000.
* Several lengths (≈0.5 · n₀ … 1.5 · n₀) are tried; for each length the
  optimisation is run for a time slice of the total budget.  The best
  sequence over all lengths is returned.

The routine respects the 1000 s limit, uses only NumPy and the provided
`evaluate_sequence`, and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple helper: FFT‑based convolution (real input → real output)
def _fft_convolve(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Return the linear convolution of real vectors x and y using FFT.
    The result has length len(x)+len(y)-1 (truncated to that length)."""
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    Z = X * Y
    conv = np.fft.irfft(Z, n=size)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: a_i ∝ 1/√((i+½)(n‑i‑½)), normalised to sum‑1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    deadline: float,
    init: str = "arcsine",
    max_iters: int = 20000,
    eta0: float = 0.1,
    beta0: float = 0.5,
    beta_factor: float = 1.01,
    beta_max: float = 5.0,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max_k (a*a)_k  (with Σ a = 1, a_i ≥ 0) by gradient descent
    on a smoothed max.

    Returns the best sequence found and its corresponding score.
    """
    # ----- initialise ----------------------------------------------------
    if init == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init == "random":
        a = np.random.rand(n)
        a /= a.sum()
    else:  # arcsine (default)
        a = _arcsine_distribution(n)

    best_a = a.copy()
    # initial score (exact, no smoothing needed)
    size_fft = 1 << ((3 * n - 2).bit_length())  # power of two ≥ convolution length
    a_fft = np.fft.rfft(a, n=size_fft)
    b = np.fft.irfft(a_fft * a_fft, n=size_fft)[: 2 * n - 1]
    best_score = float(2.0 * n * b.max())

    beta = beta0

    for it in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break

        # ---- convolution a*a -------------------------------------------------
        a_fft = np.fft.rfft(a, n=size_fft)
        b = np.fft.irfft(a_fft * a_fft, n=size_fft)[: 2 * n - 1]
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # ---- keep track of best ---------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{now:.1f}] length={n} iter={it} score={best_score:.6f} beta={beta:.3f}"
            )

        # ---- soft‑max weights for the smoothed max ---------------------------
        # stability: shift by the maximum, so the largest weight is 1
        shifted = b - max_b
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ---- gradient of Σ w_k·(a*a)_k --------------------------------------
        # gradient of (a*a)_k w.r.t. a_i is 2·a_{k-i} (zero outside)
        # The weighted sum can be written as a convolution with the reversed weight vector.
        w_rev = w_norm[::-1]                     # length = 2n‑1
        conv_grad = np.fft.irfft(
            a_fft * np.fft.rfft(w_rev, n=size_fft), n=size_fft
        )[: 3 * n - 2]                           # length = n + (2n‑1) – 1
        idx = 2 * n - 2 - np.arange(n)           # positions L‑1‑i, L=2n‑1
        grad = 2.0 * conv_grad[idx]

        # ---- step size -------------------------------------------------------
        eta = eta0 / np.sqrt(it)

        # ---- gradient descent step -------------------------------------------
        a = a - eta * grad

        # ---- projection onto the feasible set ---------------------------------
        a = np.maximum(a, 0.0)                   # non‑negative
        a = (a + a[::-1]) / 2.0                 # enforce symmetry
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # ---- anneal beta -----------------------------------------------------
        beta = min(beta_max, beta * beta_factor)

        # ---- occasional jitter to escape plateaus -----------------------------
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) / 2.0
            a /= a.sum()

    return best_a, best_score
