"""
Propose a step‑function (list of non‑negative heights) with a low
evaluate_sequence value.

Algorithm
---------
* Initialise a probability vector using the discrete arcsine law
  (the known near‑optimal shape).
* Refine the vector by a small Frank‑Wolfe / LP step:
    – given the current f, compute M = max(f*f).
    – solve the linear program  maximise Σ g   subject to
        g ≥ 0  and   f*g ≤ M   (convolution is linear in g).
    – normalise g to unit ℓ₁‑mass and move a fraction t towards it.
    – repeat a few LP iterations.
* Finish with a short projected sub‑gradient descent on the same
  objective (minimise the ℓ∞‑norm of f*f) to polish the result.
* Keep the best sequence seen during the whole run and return it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keep probabilities non‑negative and sum to 1.
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Discrete arcsine distribution (probability vector, sum = 1).
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Fast self‑convolution using FFT (real–valued vectors).
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, n=size)
    b = np.fft.irfft(fa * fa, n=size)
    return b[: 2 * n - 1]


# ----------------------------------------------------------------------
# Build the dense convolution matrix A such that (A @ g) = f * g .
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i, fi in enumerate(f):
        A[i : i + n, i] = fi
    return A


# ----------------------------------------------------------------------
# Solve the LP: maximise Σ g  subject to g ≥ 0 and f * g ≤ M.
def _solve_lp_max_sum(f: np.ndarray, M: float):
    """Return the optimal g (as a dense ndarray) or None on failure."""
    try:
        import cvxpy as cp
    except Exception:  # cvxpy not available
        return None

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    # linear constraints  f * g ≤ M
    A = _build_conv_matrix(f)
    constraints = [A @ g <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=False, max_iters=2000, verbose=False)
    except Exception:
        return None

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)
