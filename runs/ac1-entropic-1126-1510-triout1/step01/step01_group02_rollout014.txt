def lp_step(f, M, n):
    import scipy.sparse as sp
    import scipy.optimize as opt

    # Build convolution matrix A:
    # diag_data list of arrays each length n, constant f[i]
    diag_data = [np.full(n, fi) for fi in f]
    offsets = np.arange(n)
    A = sp.diags(diag_data, offsets, shape=(2*n-1, n), format='csc')
    # Bound vector
    b_ub = np.full(2*n-1, M, dtype=float)
    # Solve LP: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # fallback or return None
        return None
    b = res.x
    return b

""" 
Implementation of an iterative improvement algorithm for the squared-convolution 
maximization problem. The method combines a linear‑programming step (maximising
the ℓ₁‑norm of a candidate while respecting the current maximum convolution bound)
with a simple random pairwise‑mass‑swap exploration. Starting from a known
construction (or a uniform distribution) the algorithm repeatedly reduces the
objective 2·n·max(f∗f) (equivalently the value returned by ``evaluate_sequence``).
"""
import time, math, random
import numpy as np

# Try to import SciPy, fallback if unavailable
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:
    _SCIPY_AVAILABLE = False

def _objective(seq: np.ndarray) -> float:
    """Return the un‑scaled objective 2·n·max(conv) for a sequence that sums to 1."""
    n = seq.size
    max_conv = np.convolve(seq, seq).max()
    return float(2.0 * n * max_conv)

def _mix_vectors(a: np.ndarray, b: np.ndarray, t: float) -> np.ndarray:
    """Convex combination a←(1−t)a + t b followed by renormalisation onto the simplex."""
    c = (1.0 - t) * a + t * b
    c = np.clip(c, 0.0, None)
    total = c.sum()
    if total == 0:
        # fallback to uniform
        return np.full_like(c, 1.0 / c.size)
    return c / total

def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   max Σ b_i   s.t.   conv(f,b) ≤ M ,   b ≥ 0.
    Returns the optimal b (length equal to f) or None on failure."""
    n = f.size
    # Build sparse Toeplitz‑type matrix A where A[k,i] = f[i] if 0≤k‑i<n else 0
    # Each diagonal offset i contains the constant value f[i].
    diag_data = [np.full(n, f[i], dtype=float) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csc')
    b_ub = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n)  # maximise sum(b) → minimise -sum(b)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception:
        return None
    if not res.success:
        return None
    return res.x

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine mixes a linear‑programming refinement (when SciPy is available)
    with a cheap random pairwise swap.  It runs for at most ``budget_s`` seconds
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Choose a working length.  A length around 2500 offers a good
    # trade‑off between optimisation quality and LP size.
    # ------------------------------------------------------------------
    MAX_N = 2500
    try:
        base = np.array(height_sequence_1, dtype=float)
        if base.size > MAX_N:
            base = base[:MAX_N]               # truncate if necessary
        elif base.size < MAX_N:
            # pad with tiny values to reach MAX_N without breaking the shape
            pad = np.full(MAX_N - base.size, 0.0)
            base = np.concatenate([base, pad])
        n = base.size
    except Exception:
        n = MAX_N
        base = np.full(n, 1.0 / n, dtype=float)

    # Normalise onto the probability simplex (sum = 1)
    f = np.maximum(base, 0.0)
    f_sum = f.sum()
    if f_sum == 0:
        f = np.full(n, 1.0 / n)
    else:
        f = f / f_sum

    # ------------------------------------------------------------------
    # State variables
    # ------------------------------------------------------------------
    best = f.copy()
    best_score = _objective(best)
    cur_score = best_score

    # Simple annealing schedule for the random swaps
    temp_start = 1.0
    temp_end = 1e-4

    improvement_counter = 0
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temperature = temp_start * (1.0 - frac) + temp_end * frac

        # ------------------------------------------------------------------
        # 1) LP refinement (if SciPy is available)
        # ------------------------------------------------------------------
        if _SCIPY_AVAILABLE:
            # Current bound M
            conv_f = np.convolve(f, f)
            M = conv_f.max()

            # Solve the LP
            b = _solve_lp(f, M)

            if b is not None:
                sum_b = b.sum()
                # We only proceed when the LP can increase the ℓ₁‑norm appreciably
                if sum_b > 1.0 + 1e-6:
                    # Normalised partner (same ℓ₁‑norm as f)
                    g = b / sum_b

                    # Mixing factor – start rather aggressive and shrink over time
                    t = 0.5
                    f_candidate = _mix_vectors(f, g, t)

                    cand_score = _objective(f_candidate)

                    if cand_score < best_score:
                        best_score = cand_score
                        best = f_candidate.copy()
                        improvement_counter = 0
                        print(f"[Iter {iteration:6d}] LP mix improved score: {best_score:.6f}")

                    # Metropolis‑style acceptance
                    if cand_score < cur_score:
                        accept = True
                    else:
                        prob = math.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
                        accept = np.random.rand() < prob

                    if accept:
                        f = f_candidate
                        cur_score = cand_score
                        continue   # go to next iteration (skip the random move)

        # ------------------------------------------------------------------
        # 2) Simple random pairwise mass swap (keeps the sum exactly 1)
        # ------------------------------------------------------------------
        i, j = np.random.choice(n, size=2, replace=False)
        sigma = 0.2 / max(1.0, math.sqrt(iteration))
        delta = (np.random.rand() - 0.5) * sigma

        # Clip to keep non‑negative entries
        if delta > 0:
            delta = min(delta, f[j])
        else:
            delta = -min(-delta, f[i])

        f_candidate = f.copy()
        f_candidate[i] += delta
        f_candidate[j] -= delta

        # Renormalise (numerical drift)
        f_candidate = np.clip(f_candidate, 0.0, None)
        f_candidate_sum = f_candidate.sum()
        if f_candidate_sum == 0:
            continue
        f_candidate /= f_candidate_sum

        cand_score = _objective(f_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best = f_candidate.copy()
            improvement_counter = 0
            print(f"[Iter {iteration:6d}] Random swap improved score: {best_score:.6f}")

        # Acceptance rule (same as above)
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            f = f_candidate
            cur_score = cand_score

        # ------------------------------------------------------------------
        # Optional restart if we have been stuck for a while
        # ------------------------------------------------------------------
        improvement_counter += 1
        if improvement_counter > 10000:
            # Perturb a little around the best solution
            perturb = np.random.rand(n)
            perturb = perturb / perturb.sum()
            f = _mix_vectors(best, perturb, 0.05)
            cur_score = _objective(f)
            improvement_counter = 0
            print(f"[Iter {iteration:6d}] Restarted after stagnation, new score: {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    # Scale back to the original magnitude (scaling does not affect the
    # evaluation because the objective is homogeneous).  We simply return
    # the probabilities as heights.
    return best.tolist()

"""
Iterative LP‑refinement + random pairwise swaps.

The algorithm works with a probability‑simplex vector (non‑negative,
sum = 1).  For a given vector f we compute the current bound
M = max(conv(f,f)).  A linear programme

    maximise   Σ b_i
    subject to conv(f,b) ≤ M ,  b ≥ 0

returns a vector b with a larger ℓ₁‑norm while keeping the same
convolution bound.  Scaling b back to sum = 1 gives a partner g.
Mixing f←(1‑t)f+t g (t≈0.5) strictly reduces the quantity
2·n·max(conv) = evaluate_sequence(f) provided the LP made progress.
Occasionally the LP has no effect (e.g. when f is uniform); in that
case a cheap random pairwise mass‑swap is performed.  The loop runs
until the time budget expires and the best sequence seen is returned.

If SciPy is unavailable the routine falls back to the pure random‑swap
strategy (the same as the previous baseline).
"""

import time
import math
import random

# ----------------------------------------------------------------------
# Optional SciPy import – used for the linear‑programming step
# ----------------------------------------------------------------------
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:                 # pragma: no cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _objective(x: np.ndarray) -> float:
    """
    Objective 2·n·max(conv(x,x)) for a vector that sums to 1.
    The scaling of the vector does not affect the value, therefore we
    keep the representation on the simplex.
    """
    n = x.size
    max_conv = np.convolve(x, x).max()
    return float(2.0 * n * max_conv)


def _mix(a: np.ndarray, b: np.ndarray, t: float) -> np.ndarray:
    """
    Convex combination (1‑t)·a + t·b followed by projection onto the
    simplex (non‑negative, sum = 1).
    """
    c = (1.0 - t) * a + t * b
    c = np.clip(c, 0.0, None)
    s = c.sum()
    if s == 0.0:                     # extremely unlikely, but safe guard
        return np.full_like(c, 1.0 / c.size)
    return c / s


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ b_i
        subject to   conv(f,b) ≤ M ,   b ≥ 0
    Returns the optimal b (same length as f) or None on failure.
    """
    n = f.size
    # Build the convolution matrix A where A[k,i] = f[i] if 0≤k‑i<n else 0.
    # Each diagonal offset i contains the constant value f[i].
    diag_data = [np.full(n, f[i], dtype=float) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csc')
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # maximise Σb_i  → minimise −Σb_i
    c = -np.ones(n, dtype=float)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception:               # pragma: no cover
        return None
    if not res.success:
        return None
    return res.x
