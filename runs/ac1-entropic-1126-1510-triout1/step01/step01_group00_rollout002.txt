"""
Propose a low‑score step‑function.

Algorithm
---------
1. Initialise a probability vector on a chosen length `n` using the
   discrete arcsine distribution (the known near‑optimal shape).
2. Optimise the vector with a *smoothed* Chebyshev objective.
   - Let `b = a * a` (full convolution) and `c = max(b)`.
   - Approximate `max(b)` by the log‑sum‑exp:
        f(a) = (1/λ)·log Σ_j exp(λ·b_j) .
     The gradient of `f` w.r.t. `a` is a weighted sum of the gradients of
     each convolution entry, weights being the normalised exponentials.
   - Compute the gradient efficiently with FFTs:
        g_i = 2·Σ_k a_k·w_{i+k},
     where `w` are the normalised exponentials.
   - Update `a` with Adam‑style adaptive steps and project back onto the
     probability simplex (non‑negative entries that sum to 1).
3. Run the optimisation for several lengths (≈2000, 4000, 8000) within the
   time budget, keeping the best sequence seen.
4. Return the best sequence as a plain Python list.

The method improves on the simple active‑set sub‑gradient approach by
using information from *all* convolution entries (soft max) and an
adaptive optimiser, which typically reduces the peak of the autocorrelation
and hence the evaluation score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean projection)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Return the Euclidean projection of `v` onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path: already feasible
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Discrete arcsine distribution (probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    """Return a length‑n probability vector approximating the arcsine law."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Core optimiser for a single length using a smoothed max objective
def _optimise_one_length_smooth(
    n: int,
    deadline: float,
    max_iters: int = 5000,
    eta0: float = 0.5,
    lambda_max: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector of length `n` to minimise
    2·n·max(conv(a,a)).

    Returns the best vector found and its score.
    """
    # initialise from arcsine (good starting point)
    a = _arcsine_distribution(n)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = a.copy()
    # initial score (full convolution via FFT for speed)
    L_full = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, L_full)
    b_full = np.fft.irfft(A_fft * A_fft, L_full)[: 2 * n - 1]
    best_score = float(2.0 * n * b_full.max())

    # pre‑compute FFT length for the a–w_rev convolution (max needed length = 3n‑2)
    L2 = 1 << ((3 * n - 2).bit_length())

    # main loop
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- convolution a*a -------------------------------------------------
        A_fft = np.fft.rfft(a, L_full)
        b_full = np.fft.irfft(A_fft * A_fft, L_full)[: 2 * n - 1]
        # protect against tiny negative rounds
        b_full = np.maximum(b_full, 0.0)
        max_b = b_full.max()
        cur_score = 2.0 * n * max_b

        # keep the best seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n} it={it} score={best_score:.6f}"
            )

        # ---- smoothed max (log‑sum‑exp) ------------------------------------
        # λ grows with iteration, capped at lambda_max
        lam = min(lambda_max, 0.2 * it)  # linear growth, smooth at the start
        # compute weights w_j = exp(λ·(b_j‑max_b)) / Σ
        # subtract max_b to avoid overflow (max exponent = 0)
        exps = np.exp(lam * (b_full - max_b))
        w = exps / exps.sum()          # shape (2n‑1,)

        # ---- gradient of f(a) = (1/λ)·log Σ exp(λ·b_j) ----------------------
        # gradient = Σ_j w_j ∇b_j, and ∇b_j_i = 2·a_{j‑i}
        w_rev = w[::-1]                # reversed weights
        # convolution a * w_rev (full) via FFT
        A_fft = np.fft.rfft(a, L2)
        W_fft = np.fft.rfft(w_rev, L2)
        conv_full = np.fft.irfft(A_fft * W_fft, L2)
        # indices needed: g_i = 2·conv_full[(2n‑2)‑i]  for i=0..n‑1
        offset = 2 * n - 2
        g = 2.0 * conv_full[offset - np.arange(n)]

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing base step size
        eta = eta0 / np.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto the simplex (sum = 1, non‑negative) -----------
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny jitter to escape plateaux
        if it % 400 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Return a plain Python list as required by the caller
    return overall_best_seq.tolist()
