def _conv_full(a):
    n = len(a)
    m = 1 << ((2*n-1).bit_length()) # next power of two
    A = np.fft.rfft(a, m)
    B = A * A
    b = np.fft.irfft(B, m)
    return b[:2*n-1]

"""
Propose a low‑score step‑function sequence.

The score is 2·n·max(conv(a,a)) because the evaluation normalises the
ℓ₁‑mass.  Hence we minimise the peak of the self‑convolution under the
simplex constraint a≥0, Σa=1.

The programme works in three stages

1.  **Shape sweep** –  for a chosen length n we generate symmetric
    Beta‑type densities  a_i ∝ (x_i)^{α‑1}(1‑x_i)^{α‑1}  (x_i = (i+½)/n)
    for many values of the shape parameter α (including the arcsine case
    α=½).  The best of these crude candidates is kept.

2.  **Smooth optimisation** –  starting from the best candidate we
    minimise a smooth approximation of the max‑function:
        F(a)= (1/λ)·log Σ_k exp(λ·(a*a)_k) .
    Its gradient can be written in closed form
        ∇F(a)_i = 2· Σ_k w_k·a_{k‑i},
    where w_k = soft‑max(λ·(a*a)_k).  The gradient is evaluated with a
    single FFT‑based correlation, and the update is performed by an
    exponentiated‑gradient step (mirror descent on the simplex).
    λ is increased gradually so that the optimisation first smooths
    the objective and finally concentrates on the true peak.

3.  **Multiple lengths** –  short time slices are allocated to several
    lengths (2000 … 20000).  The best sequence among all runs is
    returned.  If a historic sequence ``height_sequence_1`` is present it
    is used as an additional starter.

All heavy work (convolution, correlation) uses NumPy FFTs, making the
search fast enough to finish comfortably inside the 1000 s budget.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, Σa = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the grid x_i = (i+½)/n.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid 0^neg and 1^neg by clipping
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a power‑of‑two FFT.
    Returned length = 2·len(a)‑1.
    """
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    B = A * A
    c = np.fft.irfft(B, m)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns the slice g[0…n‑1].
    """
    # FFT of both operands (real → rfft)
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    # correlation = ifft(conj(W) * A)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1           # position where lag 0 appears
    return 2.0 * corr[start : start + len(a)]


# ----------------------------------------------------------------------
def _optimise_smooth(
    n: int,
    deadline: float,
    init_seq: np.ndarray,
    max_iters: int = 2000,
    eta0: float = 0.2,
    lambda0: float = 0.5,
    lambda_max: float = 20.0,
) -> tuple[np.ndarray, float]:
    """
    Refine `init_seq` (already normalised to sum=1) by minimising a
    smooth approximation of the max‑convolution peak.
    Returns the best sequence seen and its (unsmoothed) score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()                       # guard against rounding errors
    best_seq = a.copy()
    best_score = float(2.0 * n * _conv_full_fft(a).max())

    # FFT size sufficient for correlation w (len=2n‑1) with a (len=n)
    m = 1 << ((3 * n - 2).bit_length())   # >= (2n‑1)+(n‑1)

    lamb = lambda0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- current score -------------------------------------------------
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n:5d} iter={it:5d} score={best_score:.6f}"
            )

        # --- smooth max (soft‑max) -----------------------------------------
        maxb = b.max()
        # exponentials are safe because (b‑maxb) ≤ 0
        exp_arg = lamb * (b - maxb)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # --- gradient (correlation) ----------------------------------------
        g = _gradient_via_correlation(w, a, m)

        # --- exponentiated‑gradient step (mirror descent) -------------------
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)           # stays non‑negative
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # --- occasional jitter (helps escape tiny plateaus) ---------------
        if it % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # --- increase the temperature parameter λ ---------------------------
        lamb = min(lambda_max, lamb * 1.005)

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a very low
    `evaluate_sequence` score.
    The routine tries several lengths, a sweep over Beta shape
    parameters and a smooth projected optimisation.  The best sequence
    found before the time budget expires is returned as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Use the historic construction if it exists (might already be good)
    overall_best_seq = None
    overall_best_score = float("inf")
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() == 0.0:
            raise ValueError
        hist = hist / hist.sum()
        hist_score = 2.0 * len(hist) * _conv_full_fft(hist).max()
        overall_best_seq = hist.copy()
        overall_best_score = hist_score
        print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Choose a set of lengths to explore (larger ⇒ lower theoretical bound)
    candidate_lengths = [2000, 4000, 8000, 12000, 20000]
    # keep only lengths that fit into the remaining budget (very generous)
    candidate_lengths = [n for n in candidate_lengths if n >= 100]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        # ---- a) sweep over Beta shape parameters -------------------------
        alphas = [0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
        alphas += list(np.random.uniform(0.2, 1.4, size=6))
        best_local_seq = None
        best_local_score = float("inf")

        for alpha in alphas:
            a = _beta_distribution(n, alpha)
            score = 2.0 * n * _conv_full_fft(a).max()
            if score < best_local_score:
                best_local_score = score
                best_local_seq = a.copy()

        print(f"Length {n:5d}: best score after β‑sweep = {best_local_score:.6f}")

        # ---- b) a short smooth optimisation starting from the best β -----
        slice_deadline = time.time() + max(0.5, (deadline - time.time()) * 0.2)

        refined_seq, refined_score = _optimise_smooth(
            n=n,
            deadline=slice_deadline,
            init_seq=best_local_seq,
            max_iters=4000,
            eta0=0.2,
            lambda0=0.5,
            lambda_max=15.0,
        )

        if refined_score < overall_best_score:
            overall_best_score = refined_score
            overall_best_seq = refined_seq.copy()
            print(
                f"*** New global best → length={n:5d}  score={overall_best_score:.6f}"
            )

        # early exit if we have already reached the target
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Fallback – ensure something is returned
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _conv_full_fft(overall_best_seq).max()

    print(
        f"Final best (approx) score: {overall_best_score:.6f}  length={len(overall_best_seq)}"
    )
    # convert to plain Python list for the runner
    return overall_best_seq.tolist()
