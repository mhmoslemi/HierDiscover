```
def geometric_score(r, n):
    if r <= 0 or r >= 1:
        return float('inf')
    # compute log_r = np.log(r)
    log_r = np.log(r)
    r_pow_n = np.exp(log_r * n) # r**n
    c = (1 - r) / (1 - r_pow_n)
    # Determine max term: compute (k+1)*r**k for k up to n-1.
    # This can be computed iteratively to avoid large powers.
    max_term = 0.0
    cur = 1.0 # r**0
    for k in range(n):
        term = (k+1) * cur
        if term > max_term:
            max_term = term
        cur *= r
    M = c*c * max_term
    return 2 * n * M
```

Propose a candidate step‑function sequence using a simple parametric family.
The family a_i = (n-i)^p (i = 0,…,n‑1) produces a strongly decreasing
distribution.  By choosing the exponent p we can equalise the dominant
convolution entries (the edge term a_0^2 and the central term
∑ a_i a_{n‑1‑i}) and obtain a much smaller value of the
objective ``evaluate_sequence`` than the previously recorded 1.55.
A small grid‑search over (n,p) is performed (the objective is cheap to
evaluate because n is kept modest, ~3000).  The best sequence found is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: evaluate a candidate shape (n, p) quickly using the provided
# ``evaluate_sequence`` function.  The function is cheap – we keep n
# moderate (≤ 4000) so a direct convolution is fast enough.
# ----------------------------------------------------------------------
def _score_shape(n: int, p: float) -> float:
    """Return evaluate_sequence for the shape a_i = (n-i)^p (normalised)."""
    # Build the decreasing power law
    seq = np.arange(n, 0, -1, dtype=np.float64) ** p
    # Normalise to sum = 1 (any positive scaling leaves the score unchanged)
    seq_sum = seq.sum()
    if seq_sum == 0:
        return float("inf")
    seq = seq / seq_sum
    # ``evaluate_sequence`` is assumed to be available in the global scope
    return evaluate_sequence(seq.tolist())

# Main entry point required by the harness
# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    start = time.time()

    n_candidates = [2000, 3000, 4000, 5000]
