"""
A fast, multi‑move simulated‑annealing optimizer for the step‑function
objective defined by ``evaluate_sequence``.

The algorithm works on the probability simplex (non‑negative entries,
sum = 1).  It mixes four kinds of moves:

1. Plain projected‑gradient descent on the currently maximal convolution
   entry.
2. A soft‑max (temperature‑controlled) gradient step that spreads the
   descent direction over the top convolution entries.
3. Random pairwise mass swaps that preserve the simplex exactly.
4. A targeted “reduce‑max” move that explicitly transfers mass away from
   the pair of indices that contribute most to the current worst
   convolution entry.

The temperature for Metropolis acceptance decays exponentially,
while the soft‑max sharpness (β) grows linearly, letting the search
focus on the worst peak later on.  Stagnation triggers a modest random
perturbation of the best solution.  After the time budget a short
refinement using the soft‑max gradient is performed.

The returned list is a valid step‑function (all non‑negative) and aims
to minimise the evaluation score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Score used internally (identical to evaluate_sequence for valid inputs)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1) Projected gradient step using the single maximal convolution entry
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(seq)
    g[mask] = 2.0 * seq[j[mask]]
    new_seq = seq - eta * g
    return _project_simplex(new_seq)


# ----------------------------------------------------------------------
# 2) Soft‑max weighted gradient step (beta controls sharpness)
# ----------------------------------------------------------------------
def _softmax_gradient_step(seq: np.ndarray, eta: float, beta: float) -> np.ndarray:
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_c = conv.max()
    # weights that concentrate on the largest entries
    diff = conv - max_c          # ≤ 0
    # avoid overflow/underflow
    w = np.exp(beta * diff)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # gradient = 2 * cross‑correlation(w, seq)
    seq_rev = seq[::-1]
    grad_full = np.convolve(w, seq_rev)
    grad = grad_full[n - 1 : n - 1 + n]   # slice of length n
    new_seq = seq - eta * (2.0 * grad)
    return _project_simplex(new_seq)


# ----------------------------------------------------------------------
# 3) Random pairwise transfer (keeps the sum exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    # zero‑mean proposal (actually a signed delta)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:                     # take from j, give to i
        delta = min(delta, seq[j])
    else:                             # take from i, give to j
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# 4) Targeted reduction of the worst convolution entry
# ----------------------------------------------------------------------
def _reduce_max_move(seq: np.ndarray, alpha: float) -> np.ndarray:
    """
    Move a small fraction (alpha) of mass away from the pair of
    indices that contribute most to the current maximal convolution entry.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    idx = np.arange(n)
    partner = k - idx
    mask = (partner >= 0) & (partner < n)

    # contribution of each i to the maximal entry
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[idx[mask]] * seq[partner[mask]]
    total = contrib.sum()
    if total == 0.0:
        # fallback to a simple random move
        return _random_pair_move(seq, sigma=0.2)

    probs = contrib / total
    i = np.random.choice(n, p=probs)
    j = k - i
    # transfer a balanced amount from both i and j (or just i if i==j)
    if i == j:
        delta = seq[i] * alpha
        new_seq = seq.copy()
        new_seq[i] -= delta
        # give to a random position
        r = np.random.choice(n)
        new_seq[r] += delta
    else:
        # amount taken from both sides (same amount ensures sum preservation)
        delta = min(seq[i], seq[j]) * alpha
        if delta <= 0.0:
            return seq
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] -= delta
        # distribute to two (possibly identical) random spots
        r1 = np.random.choice(n)
        r2 = np.random.choice(n)
        new_seq[r1] += delta
        new_seq[r2] += delta
    # Projection removes any tiny negative numerical artifacts
    return _project_simplex(new_seq)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that strives
    to minimise ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds, mixing several stochastic moves and a short
    deterministic refinement at the end.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a tiny safety margin for the harness
    deadline = start + max(0.0, budget_s - 0.5)

    # --------------------------------------------------------------
    # Initialise the search vector
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)
        if base.size == 0:
            raise ValueError
        base = base / max(base.sum(), 1e-12)   # normalise to sum = 1
    except Exception:
        # fallback: a random Dirichlet vector (sum = 1)
        n = 8000
        base = np.random.dirichlet(np.ones(n))
    n = base.shape[0]

    current = _project_simplex(base)
    best = current.copy()

    best_score = _score(best)
    current_score = best_score

    # annealing schedule parameters
    temp_start = 1.0
    temp_end = 1e-5          # final temperature
    beta_start = 0.5
    beta_end = 30.0          # final soft‑max sharpness

    # stagnation detection
    last_improve_iter = 0
    it = 0

    # main optimisation loop
    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # schedule values
        # ----------------------------------------------------------
        elapsed = (time.time() - start) / (deadline - start)
        # exponential cooling
        temperature = temp_start * (temp_end / temp_start) ** elapsed
        beta = beta_start + (beta_end - beta_start) * elapsed
        # decreasing step‑size / perturbation magnitudes
        eta = 0.6 / np.sqrt(it)
        sigma = 0.3 / np.sqrt(it)
        alpha = 0.30 * (1 - elapsed) + 0.05 * elapsed   # from 0.30 → 0.05

        # ----------------------------------------------------------
        # pick a move
        # ----------------------------------------------------------
        r = np.random.rand()
        if r < 0.35:
            # plain gradient on the single worst entry
            candidate = _gradient_step(current, eta)
        elif r < 0.60:
            # random pairwise swap
            candidate = _random_pair_move(current, sigma)
        elif r < 0.90:
            # soft‑max weighted gradient
            candidate = _softmax_gradient_step(current, eta, beta)
        else:
            # targeted reduction of the maximal convolution entry
            candidate = _reduce_max_move(current, alpha)

        # ----------------------------------------------------------
        # evaluate the candidate (fast internal score)
        # ----------------------------------------------------------
        cand_score = _score(candidate)

        # keep the best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()
            last_improve_iter = it
            print(f"[Iter {it:6d}] New best score: {best_score:.6f}")

        # ----------------------------------------------------------
        # Metropolis acceptance
        # ----------------------------------------------------------
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score

        # ----------------------------------------------------------
        # restart if stuck for a while
        # ----------------------------------------------------------
        if it - last_improve_iter > 8000:
            # tiny random perturbation of the best solution
            perturb = np.random.rand(n)
            perturb = perturb / perturb.sum()
            current = _project_simplex(0.98 * best + 0.02 * perturb)
            current_score = _score(current)
            last_improve_iter = it

    # --------------------------------------------------------------
    # Optional short deterministic refinement (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        ref = best.copy()
        ref_score = best_score

                ref_score = cand_score
                best = cand
                best_score = cand_score
                # occasional log
                if j % 100 == 0:
                    print(f"[Refine {j:4d}] score: {best_score:.6f}")

    # final log
    print(f"Finished after {it} iterations, best score = {best_score:.6f}")
    # return as plain Python list
    return best.tolist()
