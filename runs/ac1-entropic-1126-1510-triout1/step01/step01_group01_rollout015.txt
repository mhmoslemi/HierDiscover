    bestS = None
    for t in range(trials):
        indices = np.random.permutation(n).tolist()
        S = []
        sums = set()
        for idx in indices:
            conflict = False
            # compute new sums with existing S
            for s in S:
                # i+j and j+i (same) but we treat unordered
                val = s + idx
                if val in sums:
                    conflict=True
                    break
                # also check diagonal for idx+idx

            if conflict:
                continue
            # also need to ensure idx+idx not in sums (no other pair has same sum)
            if (2*idx) in sums:
                continue
            # If passes, add idx:
            for s in S:
                sums.add(s+idx)
            sums.add(2*idx)
            S.append(idx)
            if len(S) >= target_m:
                break
        if len(S) > (best_len):
            best_len = len(S); bestS = S.copy()
        if best_len >= target_m: break

def gradient_of_max_fft(a, top_k):
    n = len(a)
    conv = conv_fft(a)
    max_val = conv.max()
    idxs = np.argpartition(-conv, top_k-1)[:top_k]
    grad = np.zeros_like(a)
    # For each k, compute contributions
    for k in idxs:
        # compute j = k - i for all i
        # vector j = k - np.arange(n)
        # mask where 0 <= j < n
        j = k - np.arange(n)
        mask = (j >= 0) & (j < n)
        # contributions: 2*a[j] for mask, then subtract a[i] when i==j
        grad[mask] += 2.0 * a[j[mask]]
        # diagonal
        diag = mask & (np.arange(n) == j)
        grad[diag] -= a[diag]
    return grad, max_val

"""
Propose candidate using a hybrid simulated‑annealing + projected Adam descent.
The algorithm works on a simplex (∑a = √(2 n), a_i ≥ 0) and minimises the
maximum entry of the self‑convolution, which is exactly the score returned by
`evaluate_sequence` when the ℓ¹‑norm is normalised.  The search proceeds in three
stages:

1. **Exploratory Monte‑Carlo** – many random Dirichlet vectors are generated;
   the best one is kept.
2. **Stochastic descent** – a two‑point mass‑exchange move is performed
   repeatedly (simulated‑annealing).  The move is accepted either if it
   improves the current maximum or with a probability that depends on a
   temperature that slowly cools.
3. **Fine‑grained Adam‑style projected gradient descent** – the best vector
   obtained from stage 2 is refined using a sub‑gradient of the active
   convolution peaks and an Adam optimiser, always projected back onto the
   simplex.

When the time budget is exhausted the best sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries become zero – put all mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Return the maximum entry of a*a using a real FFT."""
    n = a.shape[0]
    L = 2 * n - 1
    # real FFT of length L (zero‑padded automatically)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return float(conv.max())

# ----------------------------------------------------------------------
def _gradient_of_max(a: np.ndarray, top_k: int = 5):
    """
    Sub‑gradient of the maximum entry of the self‑convolution.
    Returns (grad, max_val).  The gradient is computed using the `top_k`
    largest convolution entries (including ties).
    """
    n = a.shape[0]
    # compute full convolution once (FFT is cheap)
    L = 2 * n - 1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)

    max_val = float(conv.max())
    # indices of the `top_k` largest entries
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(a)
    i = np.arange(n)

    for k in idxs:
        j = k - i                     # partner index for each i
        mask = (j >= 0) & (j < n)     # valid pairs
        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)        # diagonal terms appear only once
        grad[diag] -= a[i[diag]]
    return grad, max_val

# ----------------------------------------------------------------------
def _random_simplex_vector(n: int, total: float) -> np.ndarray:
    """Draw a Dirichlet(1,…,1) vector and scale it to the given total."""
    x = np.random.rand(n)
    x = x / x.sum() * total
    return x

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (a list of heights) that minimises
    ``evaluate_sequence``.  The search respects the internal security checks of
    ``evaluate_sequence`` (non‑negative numbers, no NaNs, no infinities).  The
    function returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a little margin for clean‑up

    # ------------------------------------------------------------------
    # Choose a length.  If a reference sequence exists we keep its length,
    # otherwise we work with a moderately large size that still fits easily
    # into memory.
    try:
        n_ref = len(height_sequence_1)   # type: ignore[name-defined]
        n = max(3500, n_ref)            # at least 3500 for a healthy search
    except Exception:
        n = 5000

    target_sum = math.sqrt(2.0 * n)        # the ℓ¹‑norm that makes the score == max

    # ------------------------------------------------------------------
    # --------------------------------------------------------------
    # 1️⃣  Exploratory Monte‑Carlo – keep the best random Dirichlet point.
    best_a = None
    best_score = float("inf")
    mc_trials = 2000
    print("[stage 0] Monte‑Carlo start – length =", n)
    for _ in range(mc_trials):
        a = _random_simplex_vector(n, target_sum)
        sc = _max_convolution_fft(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[stage 0] best score after {mc_trials} trials = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Simulated annealing with two‑point mass exchange.
    a = best_a.copy()
    cur_score = best_score
    temperature = best_score * 0.5 + 1e-6   # start a bit below the current value
    min_temp = 1e-6
    decay = 0.9995                         # exponential cooling
    step_scale = max(target_sum / n, 1e-3)  # initial step size (≈ average entry)
    min_step = step_scale * 1e-4

    it = 0
    no_improve = 0
    print("[stage 1] simulated annealing – start")
    while time.time() < deadline and it < 5_000_000:
        # pick a donor (prefer larger coefficients) and a receiver
        donor = np.random.choice(n, p=a / target_sum)           # biased towards big entries
        recip = np.random.randint(n)
        while recip == donor:
            recip = np.random.randint(n)

        # amount to transfer – limited by the donor and the current step scale
        max_transfer = min(a[donor], step_scale)
        if max_transfer <= 0.0:
            it += 1
            continue
        delta = max_transfer * np.random.rand()

        a[donor] -= delta
        a[recip] += delta

        new_score = _max_convolution_fft(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - new_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            # record best if we improved globally
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
                print(f"[SA {it:08d}] new global best = {best_score:.6f}")
            else:
                no_improve += 1
        else:
            # revert the move
            a[donor] += delta
            a[recip] -= delta
            no_improve += 1

        # temperature and step cooling
        temperature = max(temperature * decay, min_temp)
        step_scale = max(step_scale * 0.9999, min_step)

        # occasional full‑restart if we have been stuck for a while
        if no_improve > 20000:
            a = _random_simplex_vector(n, target_sum)
            cur_score = _max_convolution_fft(a)
            temperature = best_score * 0.5 + 1e-6
            step_scale = max(target_sum / n, 1e-3)
            no_improve = 0
            print(f"[SA {it:08d}] restart – score {cur_score:.6f}")

        it += 1

    print(f"[stage 1] finished – best score = {best_score:.6f} after {it} moves")

    # ------------------------------------------------------------------
    # 3️⃣  Adam‑style projected sub‑gradient descent for fine‑tuning.
    a = best_a.copy()

Hybrid simulated‑annealing + Adam‑style projected sub‑gradient descent.

The score returned by ``evaluate_sequence`` (with the ℓ¹‑norm forced to
√(2 n)) equals the maximum entry of the self‑convolution a∗a.
Therefore the optimisation problem is

    minimise   max_k (a∗a)[k]   subject to   a_i ≥ 0 , Σ a_i = √(2 n).

The algorithm works in three stages:

1️⃣  Monte‑Carlo – generate many random Dirichlet vectors, keep the best.
2️⃣  Simulated annealing – repeatedly move a tiny amount of mass from one
    index to another (mass‑exchange move).  The move is accepted either if
    it improves the current maximum or with a Metropolis probability that
    slowly cools.
3️⃣  Adam‑style projected sub‑gradient descent – starting from the best
    vector obtained in stage 2, a sub‑gradient of the active convolution
    peaks is computed and an Adam optimiser is applied; after each step the
    vector is projected back onto the simplex {x ≥ 0, Σx = √(2 n)}.

If time permits a cheap “block” search (random piece‑wise constant
vectors) is performed afterwards.  The function returns the best sequence
found before the 1000 s budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a∗a using a real FFT."""
    n = a.shape[0]
    L = 2 * n - 1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return float(conv.max())


# ----------------------------------------------------------------------
def _gradient_of_max(a: np.ndarray, top_k: int = 5):
    """
    Sub‑gradient of the maximum entry of a∗a.
    Returns (grad, max_val).  Only the `top_k` largest convolution entries are
    considered (ties are kept).
    """
    n = a.shape[0]
    L = 2 * n - 1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)

    max_val = float(conv.max())
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(a)
    i = np.arange(n)

    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)

        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad, max_val

def _random_simplex_vector(n: int, total: float) -> np.ndarray:
    """Draw a Dirichlet(1,…,1) vector and scale it to the given total."""
    x = np.random.rand(n)
    x = x / x.sum() * total
    return x
