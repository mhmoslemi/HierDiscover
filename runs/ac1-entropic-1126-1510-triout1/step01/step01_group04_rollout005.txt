```python
def convolve_max(seq):
    # compute convolution using FFT
    n = len(seq)
    m = 1 << ((2*n - 1).bit_length())
    # Use real FFT
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    # conv length is m, but relevant part is up to 2n-1
    conv = conv[: 2*n-1]
    return conv.max()
```

```python
def solve_lp(a, M):
    n = len(a)
    # Build matrix using scipy.sparse.diags
    from scipy.sparse import diags
    data = [a[d] * np.ones(n) for d in range(n)]
    offsets = list(range(n))
    C = diags(data, offsets, shape=(2*n-1, n), format='csr')
    # Use linprog
    from scipy.optimize import linprog
    c = -np.ones(n)  # minimize -sum(b) -> maximize sum(b)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=C, b_ub=np.full(2*n-1, M), bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        # fallback: return current a (no change)
        return a.copy()
    return res.x
```

```python
def hill_climb(seq, best_score, deadline, max_iters=20000):
    n = len(seq)
    cur = seq.copy()
    cur_score = best_score
    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0:
            continue
        frac = np.random.rand() * 0.1
        delta = cur[i] * frac
        cur[i] -= delta
        cur[j] += delta
        # Normalize sum to 1
        # Actually transfer keeps sum unchanged, so sum stays 1.
        new_score = fast_score(cur)  # using fast evaluation
        if new_score < cur_score:
            cur_score = new_score
            # keep improvement
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
    return cur, cur_score
```

A two‑stage optimizer for the step‑function evaluation problem.

Stage 1 – a cheap global search:
    * Scan a family of power‑law sequences a_i ∝ (i+offset)^α
      (α∈[‑0.8,‑0.3], offset∈[0,3]).
    * The score of a normalised vector is 2·n·max(conv(a,a)).
    * The best (α,offset) is taken as a warm start.

Stage 2 – local refinement on a modest size (n≈2000):
    * The objective  max_k (a*a)_k is linear in a second vector b,
      therefore for a fixed maximum M we can **maximise** Σb_j
      under the linear constraints (a*b)_k ≤ M.
      This is a linear programme (LP) solved with SciPy’s “highs”.
    * The optimiser iterates:
          – compute M = max(conv(a,a))
          – solve the LP → b
          – rescale b to a probability vector
          – mix a ← (1‑t)·a + t·b   (t≈0.5, slowly reduced)
      Each iteration never raises the score and typically lowers it.

After the LP phase we upscale the distribution to a longer vector
(e.g. n=5000) by linear interpolation and polish it with cheap
hill‑climbing and a projected gradient step that use a fast FFT based
convolution.  The final sequence is handed to the provided
`evaluate_sequence` for the official score.

All heavy‑weight operations are based on FFT (O(n log n)) or sparse
linear programming, keeping the whole search comfortably below the
1000 s budget.
"""

import time
import numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility functions – all are top‑level (no closures) as required.
# ----------------------------------------------------------------------


def make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+offset+1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0:
        return vec
    return vec / total


def convolve_max(seq: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of `seq` (FFT based)."""
    n = seq.shape[0]
    # next power of two ≥ 2n‑1 for an efficient FFT
    m = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    conv = conv[: 2 * n - 1]
    return float(conv.max())


def fast_score(seq: np.ndarray) -> float:
    """Score 2·n·max(conv) / (sum(seq)^2) – fast version (no clipping)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * seq.shape[0] * convolve_max(seq) / (s * s)


def grid_search(n: int, deadline: float):
    """Exhaustive scan over exponent & offset; returns best normalised vector."""
    best_score = float("inf")
    best_vec = None
    best_params = None

    exponents = np.linspace(-0.80, -0.30, 71)   # step ≈0.007
    offsets = np.linspace(0.0, 3.0, 31)        # step 0.1

    for exp in exponents:
        for off in offsets:
            if time.time() > deadline:
                break
            vec = make_power_law(n, exp, offset=off)
            score = fast_score(vec)
            if score < best_score:
                best_score = score
                best_vec = vec.copy()
                best_params = (exp, off)
    if best_params is None:
        raise RuntimeError("grid_search ran out of time")
    print(f"[grid] best exponent={best_params[0]:.5f}, offset={best_params[1]:.3f}, "
          f"score={best_score:.6f}")
    return best_vec, best_score


def build_conv_matrix(a: np.ndarray):
    """
    Build the (2n‑1) × n sparse matrix C with
        C_{k,j} = a_{k‑j}   if   0 ≤ k‑j < n   else 0.
    This encodes (a * b)_k = (C b)_k.
    """
    n = a.shape[0]
    # each diagonal offset d≥0 contains the constant a[d]
    data = [a[d] * np.ones(n, dtype=np.float64) for d in range(n)]
    offsets = list(range(n))
    C = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return C


def solve_lp(a: np.ndarray, M: float):
    """
    Solve   maximise Σ b_j   subject to   (a * b)_k ≤ M   and   b ≥ 0.
    Returns the optimal b (not normalised).
    """
    n = a.shape[0]
    C = build_conv_matrix(a)
    # linprog solves  min c·x  s.t.  A_ub·x ≤ b_ub
    c = -np.ones(n, dtype=np.float64)          # –Σb  → maximise Σb
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    # use the high‑performance HiGHS interior‑point solver
    res = linprog(
        c,
        A_ub=C,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        # fallback: return the current vector (no progress)
        return a.copy()
    return res.x


def iterative_lp(seq: np.ndarray, max_iters: int, deadline: float):
    """Repeated LP‑mixing (t slowly decays). Returns refined vector."""
    cur = seq.copy()
    t = 0.5
    best_score = fast_score(cur)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = convolve_max(cur)
        b = solve_lp(cur, M)
        s = b.sum()
        if s == 0:
            break
        b = b / s                     # normalise to a probability vector
        # convex combination
        cur = (1.0 - t) * cur + t * b
        cur = cur / cur.sum()
        cur_score = fast_score(cur)
        print(f"[lp iter {it}] M={M:.6e}, score={cur_score:.6f}, t={t:.4f}")
        if cur_score < best_score:
            best_score = cur_score
        else:
            # if no improvement, become more cautious
            t *= 0.9
        # slowly shrink the mixing weight
        t = max(t * 0.99, 0.05)
    return cur, best_score


def hill_climb(seq: np.ndarray, best_score: float, deadline: float,
               max_iters: int = 40000, max_frac: float = 0.08):
    """Simple stochastic hill climbing on the simplex (fast score)."""
    n = seq.shape[0]
    cur = seq.copy()
    score = best_score
    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] <= 0.0:
            continue
        frac = np.random.rand() * max_frac
        delta = cur[i] * frac
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur)
        if new_score < score:
            score = new_score
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it % 5000 == 0:
            print(f"[hill] iter {it}, best_score={score:.6f}")
    return cur, score


def gradient_step(seq: np.ndarray, best_score: float, deadline: float,
                  max_iters: int = 2500, init_step: float = 0.15):
    """
    Projected sub‑gradient descent on the maximal convolution entry.
    Works on the fast score; the step size decays automatically.
    """
    n = seq.shape[0]
    cur = seq.copy()
    step = init_step
    score = best_score
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(cur, cur)          # length 2n‑1, still cheap for n≈5k
        k = int(conv.argmax())
        max_val = conv[k]

        # gradient of (a*a)_k w.r.t. a
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur)
        grad[idx] = 2.0 * cur[k - idx]

        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s
        new_score = fast_score(nxt)
        if new_score < score:
            cur = nxt
            score = new_score
            step *= 0.99
        else:
            step *= 0.9
        it += 1
        if it % 200 == 0:
            print(f"[grad] iter {it}, best_score={score:.6f}, step={step:.5f}")
    return cur, score


def upscale(seq_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation of a small probability vector to a larger size."""
    x_small = np.linspace(0.0, 1.0, seq_small.shape[0])
    x_large = np.linspace(0.0, 1.0, n_large)
    large = np.interp(x_large, x_small, seq_small)
    large = np.maximum(large, 0.0)
    total = large.sum()
    if total == 0.0:
        return large
    return large / total


# ----------------------------------------------------------------------
# Main entry point required by the judging harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a low‑score step‑function.

    The algorithm follows the outline described at the top of the file.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ---- 1. Coarse global search on a modest size --------------------
    n_grid = 2000
    best_vec, best_score = grid_search(n_grid, deadline)

    # ---- 2. LP‑based refinement (still on the modest size) ----------
    best_vec, best_score = iterative_lp(best_vec, max_iters=6, deadline=deadline)

    # ---- 3. Upscale to a longer vector for final polishing ----------
    n_final = 5000
    cur = upscale(best_vec, n_final)

    # ---- 4. Local stochastic hill‑climbing ---------------------------
    cur, best_score = hill_climb(cur, fast_score(cur), deadline, max_iters=50000)

    # ---- 5. Projected gradient descent (fine tuning) ------------------
    cur, best_score = gradient_step(cur, best_score, deadline)

    # ---- 6. Final official evaluation --------------------------------
    final_seq = [float(v) for v in cur.tolist()]   # evaluate_sequence expects list
    final_score = evaluate_sequence(final_seq)
    print(f"Returning sequence of length {len(final_seq)} with official score "
          f"{final_score:.6f}")
    return final_seq
