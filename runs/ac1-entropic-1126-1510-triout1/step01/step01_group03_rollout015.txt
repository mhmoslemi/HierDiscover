    n = len(f)

    A = np.zeros((2*n-1, n), dtype=np.float64)

    g = cp.Variable(n, nonneg=True)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

"""
Propose a step‐function sequence with a low evaluation score.

The algorithm starts from the classic arcsine (1‑z)^‑½ coefficients,
which already give a good bound (≈1.57).  It then applies a few
iterations of a Frank–Wolfe–style update:

*   Given the current sequence f (∑f = 1, f ≥ 0) compute
    M = max(f * f) (self‑convolution).
*   Solve the linear program
        maximise   1ᵀg
        subject to f * g ≤ M ,   g ≥ 0
  (the convolution f * g is linear in g).  The optimum g adds as much
  mass as possible without violating the current peak M.
*   Normalise g to unit sum and mix it with f:
        f ← (1−t)·f + t·g_norm
  with a few trial values of t.  The mix keeps the sum equal to one
  and usually reduces the peak.
*   Repeat the LP–mix step a handful of times, then perform a cheap
  random refinement.

If CVXPY is unavailable the code falls back to a pure arcsine sweep.
All helper functions are top‑level (no closures) and respect the
time budget (default 1000 s).  The returned list is ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """
    Return n coefficients proportional to C_i = binom(2i,i)/4**i,
    normalised so that the sum equals 1 (∑a_i = 1).
    """
    seq = []
    cur = 1.0          # C_0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    # normalise
    scale = 1.0 / total
    seq = [x * scale for x in seq]
    return seq


# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2·n‑1, n)) such that (A @ g)[k]
    equals (f * g)[k] for full discrete convolution.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   maximise 1ᵀg   subject to   f * g ≤ M , g ≥ 0.
    Returns the optimal g as a NumPy array, or None on failure.
    """
    # Import lazily – the environment may not have cvxpy.
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = f.shape[0]
    A = build_conv_matrix(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=False, max_iters=3000, verbose=False)
    except Exception:
        return None
    if g.value is None:
        return None
    return np.array(g.value, dtype=np.float64).ravel()

    # ------------------------------------------------------------------

        return

    # ------------------------------------------------------------------
    # Try to import cvxpy – if unavailable we fall back to a pure arcsine sweep.
    # ------------------------------------------------------------------
    try:
        import cvxpy  # noqa: F401
        have_cvx = True
    except Exception:
        have_cvx = False

    # ------------------------------------------------------------------
    # 1) If we have a linear‑programming solver, run a few Frank–Wolfe iterations.
    # ------------------------------------------------------------------
    if have_cvx:
        # Try a handful of lengths; they are cheap enough for the budget.
        for n in (600, 800, 1000, 1200):
            if time.time() >= deadline:
                break
            print(f"--- length = {n} ---")
            f = np.array(arcsine_sequence(n), dtype=np.float64)   # sum = 1
            update_best(f)

            # current peak of the self‑convolution
            M = float(np.convolve(f, f).max())
            iteration = 0
            # limit the number of LP steps per length – usually a few are enough.
            while time.time() < deadline and iteration < 12:
                iteration += 1
                g_raw = solve_lp_max_sum(f, M)
                if g_raw is None:
                    print("[lp] solve failed, aborting this length")
                    break
                sum_g = g_raw.sum()
                if sum_g <= 1e-12:
                    print("[lp] zero sum returned, stop")
                    break
                # Normalise the direction to unit L1‑norm
                g = g_raw / sum_g

                improved = False
                # Try a few mixing parameters; the first improvement is kept.
                for t in (0.2, 0.4, 0.6, 0.8):
                    f_new = (1.0 - t) * f + t * g
                    # Guard against numerical drift
                    total = f_new.sum()
                    if total <= 0:
                        continue
                    f_new = f_new / total
                    # Evaluate the mixed sequence
                    try:
                        new_score = evaluate_sequence(f_new.tolist())
                    except Exception:
                        continue
                    if new_score < best_score:
                        best_seq = f_new.copy()
                        best_score = new_score
                        f = f_new
                        M = float(np.convolve(f, f).max())
                        print(f"[iter n={n} it={iteration} t={t:.1f}] improved score = {new_score:.6f}")
                        improved = True
                        break
                if not improved:
                    # No mixing factor gave improvement – stop iterating on this n.
                    print("[iter] no improvement, stop inner loop")
                    break
        # ------------------------------------------------------------------
        # 2) Random local refinement (tiny perturbations) if time permits.
        # ------------------------------------------------------------------
        while time.time() < deadline and best_seq is not None:
            n = len(best_seq)
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 1e-4   # tiny shuffle
            cand = best_seq.copy()
            cand[i] = max(0.0, cand[i] + delta)
            cand[j] = max(0.0, cand[j] - delta)
            s = cand.sum()
            if s < 1e-12:
                continue
            cand = cand / s
            update_best(cand)

    # ------------------------------------------------------------------
    # 3) Fallback: pure arcsine sweep (works even without cvxpy).
    # ------------------------------------------------------------------
    else:
        # Coarse sweep over lengths
        for n in range(200, 3001, 100):
            if time.time() >= deadline:
                break
            seq = np.array(arcsine_sequence(n), dtype=np.float64)
            update_best(seq)

        # Fine sweep around the best length found so far
        if best_seq is not None:
            best_n = len(best_seq)
            low = max(50, best_n - 200)
            high = best_n + 200
            for n in range(low, high + 1, 10):
                if time.time() >= deadline:
                    break
                seq = np.array(arcsine_sequence(n), dtype=np.float64)
                update_best(seq)

        # Tiny random tweaks
        while time.time() < deadline and best_seq is not None:
            n = len(best_seq)
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 1e-4
            cand = best_seq.copy()
            cand[i] = max(0.0, cand[i] + delta)
            cand[j] = max(0.0, cand[j] - delta)
            s = cand.sum()
            if s < 1e-12:
                continue
            cand = cand / s
            update_best(cand)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    if best_seq is None:
        # As a last resort, return a simple uniform sequence.
        n = 100
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] uniform sequence score = {best_score:.6f}")

    print(f"Best sequence length = {len(best_seq)} with score = {best_score:.6f}")
    return best_seq.tolist()
