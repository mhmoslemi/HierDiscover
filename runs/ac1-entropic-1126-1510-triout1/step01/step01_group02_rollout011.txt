"""
Hybrid search for the minimal value of ``evaluate_sequence``.

The algorithm works on the probability simplex (all entries ≥ 0, sum = 1).
It combines three move types:

* **Multi‑gradient step** –  a sub‑gradient of the max‑convolution
  is built from all convolution entries whose value is within a
  relative tolerance of the current maximum.  An Adam‑style momentum
  update is applied and the result is projected back onto the simplex.

* **Random pairwise mass swap** – two indices exchange a small amount
  of mass while keeping the total sum exactly 1.

* **Random permutation** – the whole vector is shuffled.  The multiset
  of values stays the same, but the convolution pattern changes
  abruptly, helping the search escape local minima.

A simple simulated‑annealing acceptance rule is used so that occasional
up‑hill moves are allowed early on.  The best sequence seen so far is
kept and returned after the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # v is all non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical correction (possible round‑off)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Convolution helpers (full linear convolution)
# ----------------------------------------------------------------------
def _full_conv(x: np.ndarray) -> np.ndarray:
    """Return np.convolve(x, x) – the full linear self‑convolution."""
    return np.convolve(x, x)


# ----------------------------------------------------------------------
# Gradient derived from *all* near‑maximal convolution entries
# ----------------------------------------------------------------------
def _multi_gradient(seq: np.ndarray, eps: float) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv) using all entries within
    (1‑eps) of the current maximum.
    """
    n = seq.shape[0]
    conv = _full_conv(seq)
    mx = conv.max()
    # indices that are close to the max
    idx = np.where(conv >= mx * (1.0 - eps))[0]

    # accumulate contributions from each relevant index
    g = np.zeros_like(seq)
    for k in idx:
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * seq[j[mask]]
    return g


# ----------------------------------------------------------------------
# Random pairwise mass‑swap (preserves the sum exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    # propose a zero‑mean transfer
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0.0:
        delta = min(delta, seq[j])   # take from j
    else:
        delta = -min(-delta, seq[i]) # take from i
    out = seq.copy()
    out[i] += delta
    out[j] -= delta
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds, mixing projected Adam‑style gradient steps, pairwise swaps
    and occasional full shuffles.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a tiny safety margin for the harness
    deadline = start + max(0.0, budget_s - 2.0)

    # ------------------------------------------------------------------
    # Initial vector (try to reuse the previously discovered construction)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=float).copy()
        if base.ndim != 1:
            raise ValueError
        n = base.size
        # normalise onto the simplex (sum = 1)
        base = base / max(base.sum(), 1e-12)
    except Exception:
        # fallback – a uniform vector of a moderate size
        n = 8000
        base = np.full(n, 1.0 / n, dtype=float)

    current = _project_simplex(base)
    best = current.copy()

    # ------------------------------------------------------------------
    # Track the best score seen so far
    # ------------------------------------------------------------------
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:
        best_score = float("inf")
    current_score = best_score

    # ------------------------------------------------------------------
    # Adam‑style moment estimates (initialised to zero)
    # ------------------------------------------------------------------
    m = np.zeros_like(current)   # first moment
    v = np.zeros_like(current)   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # ------------------------------------------------------------------
    # Annealing schedule (linear in log‑time)
    # ------------------------------------------------------------------
    temp_start = 1.0
    temp_end = 1e-4

    # Stagnation detector – restart if no improvement for a while
    last_improve_it = 0
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = (time.time() - start) / (deadline - start)
        temperature = temp_start * (1.0 - elapsed) + temp_end * elapsed

        # ----------------------------------------------------------------
        # Choose a move type
        # ----------------------------------------------------------------
        move_type = np.random.rand()
        if move_type < 0.55:
            # ------------------- multi‑gradient step --------------------
            # step‑size decays like 1/√it
            eta = 0.6 / np.sqrt(it)
            # relative tolerance for “near‑max’’ indices – starts larger,
            # then shrinks slowly so that the gradient becomes more focussed.
            eps = 0.06 * np.exp(-it / 15000.0) + 0.005
            g = _multi_gradient(current, eps)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)

            candidate = _project_simplex(current - step)

        elif move_type < 0.85:

            sigma = 0.22 / np.sqrt(it)
            candidate = _random_pair_move(current, sigma)

        else:
            # ---------------------- full shuffle -----------------------
            perm = np.random.permutation(n)
            candidate = current[perm]   # sum unchanged, only ordering varies

        # ----------------------------------------------------------------
        # Evaluate the candidate (the expensive part)
        # ----------------------------------------------------------------
        try:
            cand_score = evaluate_sequence(candidate.tolist())
        except Exception:
            # Defensive – should never happen because we keep the vector legal.
            continue

        # ----------------------------------------------------------------
        # Record the best ever seen
        # ----------------------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()
            last_improve_it = it
            print(f"[Iter {it:6d}] New best score: {best_score:.6f}")

        # ----------------------------------------------------------------
        # Metropolis‑style acceptance
        # ----------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # probability = exp( - Δ / T )
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score

        # ----------------------------------------------------------------
        # Occasionally restart from a perturbed version of the best to
        # avoid long stagnation periods.
        # ----------------------------------------------------------------
        if it - last_improve_it > 6000:
            # generate a tiny random simplex perturbation and blend it
            perturb = np.random.rand(n)
            perturb = perturb / perturb.sum()
            current = _project_simplex(0.97 * best + 0.03 * perturb)
            current_score = evaluate_sequence(current.tolist())
            last_improve_it = it
            # reset Adam moments (they no longer correspond to the new point)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return best.tolist()
