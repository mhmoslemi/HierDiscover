propose_candidate
-----------------
A two‑phase optimiser for the max‑self‑convolution objective.

Phase 1 – smoothed projected gradient descent.
    * The nonsmooth “max” is approximated by a soft‑max
      F_α(a)= (1/α)·log Σ exp(α·(a*a)[k]).

      the mass globally and later focuses on the true
      maximum.
    * After each gradient step the vector is projected

Phase 2 – cheap local refinements.
    * Pairwise mass transfers between two coordinates

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Fast convolution using real FFT (length = len(x)+len(y)-1)."""
    n = len(x) + len(y) - 1
    nfft = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    out = np.fft.irfft(X * Y, n=nfft)
    return out[:n]

def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Compute soft‑max weights for a given convolution array and smoothing α."""
    # shift by the maximum for numerical stability
    m = conv.max()
    exps = np.exp(alpha * (conv - m))
    w = exps / exps.sum()
    return w
