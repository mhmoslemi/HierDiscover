def solve_lp_for_g(f, M, max_val=1.0):
    n = len(f)
    # construct matrix A_ub
    rows = []
    data = [] # but use dense
    # Build dense matrix with shape (2n-1,n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        # j valid range: max(0, k-(n-1)) <= j <= min(k, n-1)
        low = max(0, k-(n-1))
        high = min(k, n-1)
        # for j in [low, high], set A[k, j] = f[k - j]
        # vectorized assignment:
        idxs = np.arange(low, high+1, dtype=int)
        A[k, idxs] = f[k - idxs]   # k - idxs yields appropriate f indices
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0.0, max_val) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Optimise a non‑negative step function for the provided evaluate_sequence.

Algorithm
---------
1. **Initial guess** – a truncated power‑law a_i ∝ (i+1)^(-p) with a
   golden‑section search on the exponent p for several lengths n.
2. **Linear‑programming refinement** – following the idea from the
   reference: for a fixed vector f compute M = max(f*f).  Solve the LP

        maximise   sum(g_j)
        subject to conv(f, g)_k ≤ M   (k = 0 … 2n‑2)
                  g_j ≥ 0 , g_j ≤ 1

   (conv(f,g) is linear in the unknown g).  The solution g spreads the
   mass to positions where f is small, therefore reducing the peak of
   the autocorrelation when we mix the two vectors.
3. **Iterative mixing** – repeatedly replace  f ← (1‑t)·f + t·g   (t≈0.2)
   and recompute M.  Each iteration can only lower the evaluation
   score, and we stop when the time budget expires or no improvement
   occurs.
4. **Fine‑grained hill‑climbing** – stochastic local moves that transfer
   a tiny amount of mass between two indices, accepting only improving
   moves.
5. Return the best vector found (as a plain Python list).  All vectors
   are normalised to sum‑to‑one, which makes the score scale‑invariant.

The routine respects the 1000 s budget, prints progress messages and
finally yields the best sequence discovered.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog

# --------------------------------------------------------------------------- #
# Helper functions
# --------------------------------------------------------------------------- #

def power_law_sequence(n: int, p: float) -> np.ndarray:
    """Decreasing power‑law of length n, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def evaluate_vec(v: np.ndarray) -> float:
    """Thin wrapper around the globally provided evaluate_sequence."""
    return evaluate_sequence(v.tolist())


def golden_section_search(n: int, left: float, right: float,
                         max_iters: int, deadline: float):
    """1‑D optimisation of the exponent p for a power‑law shape."""
    phi = (np.sqrt(5) - 1) / 2          # 0.618...
    a, b = left, right
    p1 = a + (1 - phi) * (b - a)
    p2 = a + phi * (b - a)

    v1 = evaluate_vec(power_law_sequence(n, p1))
    v2 = evaluate_vec(power_law_sequence(n, p2))

    best_score = min(v1, v2)
    best_p = p1 if v1 <= v2 else p2
    best_vec = power_law_sequence(n, best_p)

    it = 0
    while it < max_iters and time.time() < deadline:
        if v1 < v2:
            b = p2
            p2 = p1
            v2 = v1
            p1 = a + (1 - phi) * (b - a)
            v1 = evaluate_vec(power_law_sequence(n, p1))
            if v1 < best_score:
                best_score, best_p, best_vec = v1, p1, power_law_sequence(n, p1)
        else:
            a = p1
            p1 = p2
            v1 = v2
            p2 = a + phi * (b - a)
            v2 = evaluate_vec(power_law_sequence(n, p2))
            if v2 < best_score:
                best_score, best_p, best_vec = v2, p2, power_law_sequence(n, p2)
        it += 1
    return best_p, best_score, best_vec


def hill_climb(vec: np.ndarray, deadline: float,
               max_no_improve: int = 2000) -> np.ndarray:
    """Stochastic hill‑climbing that redistributes a tiny fraction of mass."""
    n = len(vec)
    curr = vec.copy()
    curr_score = evaluate_vec(curr)
    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = np.random.rand() * 0.05 * curr[i]   # move ≤5 % of source
        if delta <= 0.0:
            continue
        cand = curr.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] = cand[j] + delta                  # total sum unchanged
        cand_score = evaluate_vec(cand)
        if cand_score < curr_score:
            curr = cand
            curr_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
    return curr


def solve_lp_for_g(f: np.ndarray, M: float, upper_bound: float = 1.0) -> np.ndarray:
    """
    Solve:
        maximise   sum(g)
        subject to conv(f, g) ≤ M   (for all k)
                  0 ≤ g_j ≤ upper_bound

    Returns the optimal g as a 1‑D NumPy array, or None on failure.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # Build the constraint matrix A such that A @ g ≤ M
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # A[k, idx] = f[k - idx]
        A[k, idx] = f[k - idx]
    b = np.full(rows, M, dtype=np.float64)

    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, upper_bound) for _ in range(n)]

    # Use HiGHS solver (fast and robust)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


def lp_refine(seq: np.ndarray, deadline: float,
              max_iters: int = 15, mix: float = 0.2) -> np.ndarray:
    """
    Iteratively apply the LP refinement described in the prompt.
    Returns the best sequence obtained within the given time budget.
    """
    f = seq.copy()
    best = f.copy()
    best_score = evaluate_vec(best)
    t = mix
    it = 0
    while it < max_iters and time.time() < deadline:
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        g = solve_lp_for_g(f, M, upper_bound=1.0)
        if g is None:
            # LP failed – abort refinement
            break
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # Rescale g to have the same ℓ₁‑norm as f (normally 1.0)
        g = g * (f.sum() / sum_g)

        # Mix and renormalise
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, None)
        f_new_sum = f_new.sum()
        if f_new_sum == 0.0:
            break
        f_new = f_new / f_new_sum

        new_score = evaluate_vec(f_new)
        if new_score < best_score:
            best_score = new_score
            best = f_new.copy()
            f = f_new
            print(f"[lp] iter {it:02d} improved score {new_score:.6f}")
        else:
            # No improvement – shrink the mixing factor to be more conservative
            t *= 0.5
            if t < 1e-4:
                # mixing too tiny, stop iterating
                break
            print(f"[lp] iter {it:02d} no improvement, reducing t → {t:.5f}")
        it += 1
    return best


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative step sequence within the time budget.
    Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------- #
    # 1) Scan several lengths and pick the best power‑law start vector
    # ------------------------------------------------------------------- #
    lengths = [1200, 1500, 1800, 2100, 2400]
    best_vec = None
    best_score = np.inf

    for n in lengths:
        if time.time() >= deadline:
            break
        # golden‑section search on the exponent p
        left, right = 0.05, 0.95
        p_opt, sc, vec = golden_section_search(
            n, left, right, max_iters=30, deadline=deadline)
        print(f"[search] n={n:4d} best p={p_opt:.5f} → score {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()

    # ------------------------------------------------------------------- #
    # 2) Small random restarts – may find a better normalisation
    # ------------------------------------------------------------------- #
    if best_vec is not None:
        n_best = len(best_vec)
        for _ in range(3):
            if time.time() >= deadline:
                break
            rand_vec = np.random.rand(n_best)
            rand_vec /= rand_vec.sum()
            sc = evaluate_vec(rand_vec)
            if sc < best_score:
                best_score = sc
                best_vec = rand_vec.copy()
                print(f"[rand] new best score {sc:.6f}")

    # ------------------------------------------------------------------- #
    # 3) LP‑based refinement (core of the reference algorithm)
    # ------------------------------------------------------------------- #
    if best_vec is not None and time.time() < deadline:
        refined = lp_refine(best_vec, deadline, max_iters=20, mix=0.2)
        sc_ref = evaluate_vec(refined)
        if sc_ref < best_score:
            best_score = sc_ref
            best_vec = refined.copy()
            print(f"[lp‑final] improved to {sc_ref:.6f}")

    # ------------------------------------------------------------------- #
    # 4) Final stochastic hill‑climbing for fine tuning
    # ------------------------------------------------------------------- #
    if best_vec is not None and time.time() < deadline:
        final_vec = hill_climb(best_vec, deadline, max_no_improve=3000)
        sc_final = evaluate_vec(final_vec)
        if sc_final < best_score:
            best_score = sc_final
            best_vec = final_vec.copy()
            print(f"[hill‑final] improved to {sc_final:.6f}")

    # ------------------------------------------------------------------- #
    # 5) Return result
    # ------------------------------------------------------------------- #
    if best_vec is None:
        # fallback – uniform vector of modest size
        best_vec = np.full(1000, 1.0 / 1000)
        print("[fallback] using uniform vector")

    print(f"[final] best score {best_score:.6f} with length {len(best_vec)}")
    # The evaluation function will clamp negatives / large values,
    # but we already guarantee non‑negativity and reasonable magnitude.
    return best_vec.tolist()
