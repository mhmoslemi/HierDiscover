def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a low‑evaluation step function.
    The search explores a family a_i = (i+1)^{-p} * exp(-γ i/(n-1))
    with p and γ tuned by a three‑stage grid + simple stochastic refinement.
    """
    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety

    n = 4000
    idx = np.arange(1, n+1, dtype=np.float64)

    best_score = float('inf')
    best_seq = None
    best_params = None

    def evaluate(arr):
        # arr is numpy array
        return evaluate_sequence(arr.tolist())

    # ----------------------------------------------------------------
    # Stage 1: coarse grid over p and gamma (including pure power, gamma=0)
    # ----------------------------------------------------------------
    p_grid = np.linspace(0.3, 1.0, 15)   # 0.3 to 1.0
    gamma_grid = np.linspace(0.0, 2.0, 21)  # 0..2

    for p in p_grid:
        base = idx ** -p
        for gamma in gamma_grid:
            seq = base * np.exp(-gamma * (idx - 1) / (n - 1))
            score = evaluate(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_params = (p, gamma)
                print(f"New best (coarse) p={p:.4f} gamma={gamma:.4f} score={score:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ----------------------------------------------------------------
    # Stage 2: fine grid around best parameters
    # ----------------------------------------------------------------
    if best_params is not None:
        p_best, gamma_best = best_params
        p_fine = np.linspace(max(0.05, p_best - 0.2), min(2.0, p_best + 0.2), 41)
        gamma_fine = np.linspace(max(0.0, gamma_best - 0.5), gamma_best + 0.5, 41)
        for p in p_fine:
            base = idx ** -p
            for gamma in gamma_fine:
                seq = base * np.exp(-gamma * (idx - 1) / (n - 1))
                score = evaluate(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (p, gamma)
                    print(f"New best (fine) p={p:.4f} gamma={gamma:.4f} score={score:.6f}")
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

    # ----------------------------------------------------------------
    # Stage 3: pure power‑law refinement (gamma=0)
    # ----------------------------------------------------------------
    if time.time() < deadline:
        p_grid2 = np.linspace(0.05, 1.2, 61)
        for p in p_grid2:
            seq = idx ** -p
            score = evaluate(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_params = (p, 0.0)
                print(f"New best (pure power) p={p:.4f} score={score:.6f}")
            if time.time() > deadline:
                break

    # ----------------------------------------------------------------
    # Stage 4: stochastic local refinement on the best sequence
    # ----------------------------------------------------------------
    sigma = 0.2   # relative noise level
    while time.time() < deadline:
        # multiplicative perturbation on log‑scale
        noise = np.random.randn(n) * sigma
        cand = best_seq * np.exp(noise)   # guarantees positivity
        # clip to allowed range
        cand = np.clip(cand, 0.0, 1000.0)
        score = evaluate(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"New best (local) score={score:.6f} sigma={sigma:.5f}")
            # reset sigma a bit to continue fine exploration
            sigma = max(0.01, sigma * 0.9)
        else:
            # gradually reduce noise

    # ----------------------------------------------------------------
    # Report
    # ----------------------------------------------------------------
    if best_params is not None:

    else:
        print(f"Best objective {best_score:.6f}")

    return best_seq.tolist()

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step function.

    The search explores the two‑parameter family
        a_i = (i+1)^{-p} * exp(-γ·i/(n‑1))

    where the exponent p and the exponential decay γ are tuned by a
    three‑stage grid search (coarse → fine → pure‑power refinement) and
    a short stochastic local refinement.  The objective is
        2·n·max(conv(a,a)) / (sum(a)^2)
    which is scale‑invariant, so all sequences are generated with a
    maximum value of 1.0 (well below the 1000‑clip).  The algorithm
    respects the supplied time budget (default 1000 s) and prints
    intermediate progress.  The best sequence found is returned as a
    plain Python list.
    """
    import time
    import numpy as np

    # seed the RNG – the search is deterministic for a given seed.
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Fixed length – 4000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000
    idx = np.arange(1, n + 1, dtype=np.float64)  # i = 1 … n

    best_score = float("inf")
    best_seq = None
    best_params = None  # (p, gamma)

    # ------------------------------------------------------------------
    # Helper that forwards a NumPy array to the given evaluate_sequence.
    # ------------------------------------------------------------------
    def _eval(arr):
        # evaluate_sequence expects a Python list of floats
        return evaluate_sequence(arr.tolist())

    # ------------------------------------------------------------------
    # Stage 1 – coarse grid over (p, γ).  The grid is intentionally generous
    #           but cheap enough to finish well before the time limit.
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.30, 1.00, 15)          # 0.30 … 1.00
    gamma_grid = np.linspace(0.0, 2.0, 21)       # 0 … 2.0

    for p in p_grid:
        base = idx ** -p
        for gamma in gamma_grid:
            seq = base * np.exp(-gamma * (idx - 1) / (n - 1))
            score = _eval(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_params = (p, gamma)
                print(f"[coarse]  p={p:.4f} γ={gamma:.4f} → {score:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Stage 2 – fine grid around the best (p, γ) found so far.
    # ------------------------------------------------------------------
    if best_params is not None and time.time() < deadline:
        p_best, gamma_best = best_params
        p_fine = np.linspace(max(0.05, p_best - 0.20),
                             min(2.0, p_best + 0.20), 41)
        gamma_fine = np.linspace(max(0.0, gamma_best - 0.50),
                                 gamma_best + 0.50, 41)

        for p in p_fine:
            base = idx ** -p
            for gamma in gamma_fine:
                seq = base * np.exp(-gamma * (idx - 1) / (n - 1))
                score = _eval(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (p, gamma)
                    print(f"[fine]    p={p:.4f} γ={gamma:.4f} → {score:.6f}")
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Stage 3 – pure power‑law (γ = 0) refinement.  Sometimes the optimum
    #           lies exactly at γ = 0, so we scan a finer p‑grid.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        p_grid2 = np.linspace(0.05, 1.20, 61)
        for p in p_grid2:
            seq = idx ** -p                     # γ = 0
            score = _eval(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_params = (p, 0.0)
                print(f"[pure‑p]  p={p:.4f} → {score:.6f}")
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Stage 4 – stochastic local refinement around the best sequence.
    #           Multiplicative log‑normal perturbations keep the vector
    #           non‑negative and respect scale‑invariance.
    # ------------------------------------------------------------------
    sigma = 0.20           # starting relative noise amplitude
    while time.time() < deadline:
        # generate a candidate by perturbing every coefficient multiplicatively
        noise = np.random.randn(n) * sigma
        cand = best_seq * np.exp(noise)      # always positive
        cand = np.clip(cand, 0.0, 1000.0)     # respect the hard caps
        score = _eval(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[local]   improved to {score:.6f} (σ={sigma:.5f})")
            # when we improve, allow a slightly larger step again
            sigma = max(0.01, sigma * 0.9)
        else:
            # slowly shrink the perturbation magnitude
            sigma = max(0.01, sigma * 0.995)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    if best_params is not None:
        print(f"Best parameters → p={best_params[0]:.5f}, γ={best_params[1]:.5f}")
    print(f"Best objective   → {best_score:.6f}  (length = {n})")

    # evaluate_sequence expects a plain list, so convert before returning
    return best_seq.tolist()
