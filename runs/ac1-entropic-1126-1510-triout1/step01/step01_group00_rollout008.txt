def _optimize_length(n, time_deadline, max_iters=5000, base_eta=0.5):
    a = _arcsine_distribution(n)

    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # Might also compute score via formula: 2*n*max_b (since sum=1). But evaluate_sequence includes clipping etc; better compute actual to ensure same (but same).
    # Actually evaluate_sequence also clips values to [0,1000] and sum>0. Since sum=1 fine.

    for it in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break

        # compute self convolution b
        b = full_convolve_fft(a,a) # length 2n-1

        # compute smoothweights
        # use adaptive tau:
        if it < 500:
            tau = 0.5
        elif it < 2000:
            tau = 2.0
        else:
            tau = 8.0

        # compute stable weights
        maxb = np.max(b)
        exp_term = np.exp(tau * (b - maxb))
        w = exp_term / np.sum(exp_term)

        # compute gradient: 2 * correlation(w, a)
        # correlation via conv: conv = full_convolve_fft(a, w[::-1])
        w_rev = w[::-1]
        conv = full_convolve_fft(a, w_rev) # length n+len(w)-1 = n+2n-2 = 3n-2
        # reversed slice:
        grad = 2 * conv[(len(w)-1) - np.arange(n)]

        # step size
        eta = base_eta / np.sqrt(it)

        # gradient descent step + projection
        a = a - eta * grad
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter
        if it % 250 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

        # evaluate periodically, say every 100 iterations
        if it % 100 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} it={it} score={best_score:.6f}")

    # final evaluation
    final_score = evaluate_sequence(a.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = a.copy()

    return best_seq, best_score

def _full_convolve_fft(a,b):
    size = len(a) + len(b) - 1
    nfft = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)

    conv = np.fft.irfft(C, n=nfft)[:size]

"""
Propose a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  The algorithm uses a smooth approximation of the
L∞‑norm of the self‑convolution and performs projected gradient descent on
the probability simplex.  Convolutions are carried out with FFTs, which
allows us to handle sequences of several‑tens of thousands of elements
within the 1000 s time budget.

Key ideas
---------
* Initialise the sequence with the discrete arcsine law (known to give a
  near‑flat autocorrelation).
* At each iteration compute the full self‑convolution `b = a*a`.
* Replace the hard max by a softmax:
        w_i ∝ exp(τ·b_i)      (τ grows during the run)
  The gradient of the soft‑maxed objective w.r.t. `a` is
        g = 2·corr(w, a)      where corr denotes the (non‑reversed) correlation.
  Using FFTs we obtain `g` in O(n log n) time.
* Take a decreasing step (`η = η0/√t`) and project back onto the simplex
  (non‑negative entries summing to one).
* Run several independent restarts with different lengths (including the
  historic length) and keep the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full 1‑D convolution of two real vectors using FFT.
    Returns length len(a)+len(b)-1 array.
    """
    size = a.size + b.size - 1
    nfft = 1 << ((size - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
def _correlate_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute c_j = Σ_i a_i * w_{i+j}   for j = 0 … len(a)-1
    using FFTs (equivalent to a cross‑correlation with w, *not* reversed).
    """
    # conv = a * (w reversed)
    w_rev = w[::-1]
    conv = _full_convolve_fft(a, w_rev)          # length = len(a) + len(w) - 1
    L = w.size
    # c_j = conv[L-1 - j]   (see derivation in the docstring)
    c = conv[(L - 1) - np.arange(a.size)]
    return c


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _optimise_length(
    n: int,
    deadline: float,
    max_iters: int = 5000,
    eta0: float = 0.4,
    tau_initial: float = 0.5,
    tau_final: float = 12.0,
    tau_ramp: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex for a fixed length `n`.

    Returns the best sequence (as a numpy array) and its (lower‑bound) score
    2·n·max(b) where b = a*a (sum(a) is ≈1, so this equals the true score).
    """
    # start from the discrete arcsine law (good baseline)
    a = _arcsine_distribution(n)

    best_seq = a.copy()
    b = _full_convolve_fft(a, a)
    best_score = 2.0 * n * b.max() / (a.sum() ** 2)

    # main optimisation loop
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # self‑convolution
        b = _full_convolve_fft(a, a)               # length 2n‑1

        # soft‑max temperature schedule (smooth → sharp)
        if it < tau_ramp // 3:
            tau = tau_initial
        elif it < 2 * tau_ramp // 3:
            tau = tau_initial * 4.0
        elif it < tau_ramp:
            tau = tau_initial * 10.0
        else:
            # after ramp keep a relatively high temperature to focus on the peak
            tau = tau_final

        # compute softmax weights for the convolution entries
        maxb = np.max(b)
        exp_term = np.exp(tau * (b - maxb))        # stabilised
        w = exp_term / np.sum(exp_term)            # ∑ w_i = 1

        # gradient of the smooth objective: 2·corr(w, a)
        g = 2.0 * _correlate_fft(w, a)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection onto the probability simplex
        a = a - eta * g
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny random perturbation to escape shallow plateaus
        if it % 250 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

        # evaluate the hard objective every 100 iterations
        if it % 100 == 0:
            cur_score = 2.0 * n * np.max(_full_convolve_fft(a, a)) / (a.sum() ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] length={n} iter={it} "
                    f"score={best_score:.6f}"
                )

    # final evaluation
    cur_score = 2.0 * n * np.max(_full_convolve_fft(a, a)) / (a.sum() ** 2)
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()

    return best_seq, best_score

    """
