"""
Propose a low‑value step‑function.

The algorithm is a two‑phase search:

1.  **Momentum projected gradient descent** on a smoothed version of the
    maximum self‑convolution.  The gradient of a convolution entry
    `k` is `∂(a*a)[k]/∂a_i = 2·a_{k‑i}` for all admissible `i`; this
    corrects the diagonal term used in the previous baseline.  A soft‑
    max (exponential weighting) is applied to several of the largest
    convolution entries so the descent does not get stuck on a single
    peak.  After each step the iterate is projected onto the simplex
    `{a≥0, Σa = √(2 n)}`.

2.  **Local random swaps** for the remaining time budget.  Small amounts
    of mass are moved from one index to another; a move is kept if it
    reduces the current maximum convolution value.

Both phases keep the best vector seen and finally perform a handful of
cheap “block‑pattern” trials (similar to the reference code).  The
returned list is guaranteed to be non‑negative and to respect the
normalisation, so `evaluate_sequence` will simply return the achieved
upper bound (the maximum convolution entry).

The whole routine respects the 1000 s time budget and prints occasional
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    # condition for the “active” set
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1       # +1 because of zero‑based indexing
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray, top_k: int = 5, weight_exp: float = 0.0):
    """
    Sub‑gradient of the smoothed max of the self‑convolution.

    Parameters
    ----------
    a : np.ndarray
        Current step‑function (non‑negative, length n).
    top_k : int
        Number of largest convolution entries to consider.
    weight_exp : float
        Exponential smoothing parameter.  If 0 a uniform weight is used;
        otherwise weights ∝ exp(weight_exp·(conv_k−max_conv)).

    Returns
    -------
    grad : np.ndarray
        Gradient vector (same shape as `a`).
    max_val : float
        The current maximum convolution entry (used for bookkeeping).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()

    # indices of the `top_k` largest entries (including ties)
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    # compute (possibly smoothed) weights for the selected entries
    if weight_exp == 0.0:
        w = np.full_like(idxs, 1.0 / idxs.size, dtype=float)
    else:
        vals = conv[idxs]
        w_raw = np.exp(weight_exp * (vals - max_val))
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum > 0 else np.full_like(idxs, 1.0 / idxs.size)

    grad = np.zeros_like(a)
    i = np.arange(n)

    for wk, k in zip(w, idxs):
        j = k - i                     # partner index for each i
        mask = (j >= 0) & (j < n)     # valid pairs
        # exact gradient is 2 * a_j for all valid pairs
        grad[mask] += 2.0 * wk * a[j[mask]]
    return grad, max_val

    """

    target_sum = math.sqrt(2.0 * n)

    a = _project_to_simplex(a_start, target_sum)

    top_k = 12                     # how many peaks to treat each iteration
    eta0 = target_sum * 0.4 / n    # initial step size (≈ 0.4·average entry)

        if score_cand < best_score:

            best_a = cand.copy()
            a = cand  # continue refinement from the improved point

        it += 1

    # Return as a plain Python list for the external harness.
