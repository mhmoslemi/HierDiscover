def lp_refinement(a: np.ndarray, t: float, time_deadline: float) -> np.ndarray:
    # returns refined a after one iteration.
    # ...
    import scipy.optimize as opt
    import scipy.sparse as sp

    n = len(a)
    L = 2*n - 1
    M = np.max(np.convolve(a, a))

    # build sparse convolution matrix A (L x n)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows indices where a_i participates: from i to i+n-1
        start = i
        end = i + n
        # column indices: 0..n-1
        rows.extend(range(start, end))
        cols.extend(range(n))
        data.extend([a[i]] * n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(L, n))

    # Setup LP
    c = -np.ones(n)  # maximize sum(b)
    # inequality A @ b <= M * np.ones(L)
    b_ub = np.full(L, M)
    bounds = [(0, None)] * n

    # solve
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'disp': False, 'time_limit': max(0, time_deadline - time.time())})
    if not res.success:
        # if fails return original a
        return a

    b = res.x
    sum_b = np.sum(b)
    if sum_b <= 0:
        return a

    # Normalize b to sum=1
    b_norm = b / sum_b

    # Now try mixing candidates
    candidates = []
    candidates.append(b_norm)  # direct substitution
    candidates.append((a + b_norm) / 2.0)  # average

    # maybe also weighted combos with t values
    for beta in [0.3, 0.5, 0.7]:
        candidates.append((1-beta) * a + beta * b_norm)

    best_vec = a
    best_score = evaluate_vec(a)
    for cand in candidates:
        score = evaluate_vec(cand)
        if score < best_score:
            best_score = score
            best_vec = cand

    return best_vec

def lp_refine_once(a, time_deadline):

    n = len(a)
    M = np.max(np.convolve(a, a))  # compute current max conv
    # build conv matrix
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    idx = 0
    for i in range(n):
        rows[idx:idx+n] = np.arange(i, i + n)
        cols[idx:idx+n] = np.arange(n)
        data[idx:idx+n] = a[i]
        idx += n
    L = 2*n - 1
    A = sp.csr_matrix((data, (rows, cols)), shape=(L, n))
    # Solve LP
    import scipy.optimize as opt
    c = -np.ones(n)
    b_ub = np.full(L, M)
    bounds = [(0, None)] * n
    # Option: set options time limit: perhaps not needed but we can pass via 'maxiter'.
    # Actually SciPy's highs doesn't support explicit time limit; we rely on external.
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # If LP fails, return original a
        return a
    b_sol = res.x
    sum_b = b_sol.sum()
    if sum_b <= 0:
        return a
    # Normalize b to sum 1
    b_norm = b_sol / sum_b
    # Generate candidate mixes
    candidates = [b_norm,
                  (a + b_norm) / 2,
                  (0.7 * a + 0.3 * b_norm),
                  (0.3 * a + 0.7 * b_norm)]

    best_score = evaluate_vec(a)
    for cand in candidates:
        score = evaluate_vec(cand)
        if score < best_score:
            best_candidate = cand
            best_score = score
    return best_candidate

"""
Propose a low‑value step‑function sequence.

Strategy
--------
1. **One‑dimensional family** – Search a decreasing power‑law
   a_i ∝ (i+1)^(-p) (p∈[0.05,0.95]) with a golden‑section line search.
   This yields a good baseline (≈1.57 on the current benchmark).

2. **Linear‑programming refinement** – For the current vector `a`
   solve the LP  

        maximise   Σ b_j
        subject to  a ∗ b ≤ M          (element‑wise)
                    b_j ≥ 0

   where `M = max(a∗a)`.  The LP finds a distribution `b` that can
   carry *more* total mass without violating the current worst
   convolution bound `M`.  Mixing `a` and the normalised `b`
   (simple averaging or a short line‑search) strictly reduces the
   evaluation score.  The LP is built via a sparse Toeplitz matrix
   (size (2n‑1) × n) and solved with SciPy’s HiGHS interior‑point
   method.  The refinement is repeated while it yields improvement.

3. **Local stochastic hill‑climb** – A final short hill‑climbing
   phase moves a small amount of probability mass between random
   indices, accepting only strictly improving moves.  This removes
   remaining local irregularities.

The whole routine respects the 1000 s budget, prints progress and
returns the best sequence found (as a plain Python list).  It falls
back to the pure power‑law + hill‑climb method if SciPy is not
available.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper utilities (no closures, no lambdas)
# --------------------------------------------------------------------------- #

def power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a decreasing power‑law vector of length n, normalised to sum = 1.
    a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a /= total
    return a


def evaluate_vec(vec: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence (expects a list)."""
    return evaluate_sequence(vec.tolist())


def golden_section_search(
    n: int,
    left: float,
    right: float,
    max_iters: int,
    time_limit: float,
) -> tuple[float, float, np.ndarray]:
    """1‑D optimisation of the exponent `p` in the power‑law family.
    Returns (best_p, best_score, best_vector)."""
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    a, b = left, right
    p1 = a + (1 - phi) * (b - a)
    p2 = a + phi * (b - a)

    v1 = evaluate_vec(power_law_sequence(n, p1))
    v2 = evaluate_vec(power_law_sequence(n, p2))

    best_score = min(v1, v2)
    best_p = p1 if v1 <= v2 else p2
    best_vec = power_law_sequence(n, best_p)

    it = 0
    while it < max_iters and time.time() < time_limit:
        if v1 < v2:
            b = p2
            p2 = p1
            v2 = v1
            p1 = a + (1 - phi) * (b - a)
            v1 = evaluate_vec(power_law_sequence(n, p1))
            if v1 < best_score:
                best_score = v1
                best_p = p1
                best_vec = power_law_sequence(n, best_p)
        else:
            a = p1
            p1 = p2
            v1 = v2
            p2 = a + phi * (b - a)
            v2 = evaluate_vec(power_law_sequence(n, p2))
            if v2 < best_score:
                best_score = v2
                best_p = p2
                best_vec = power_law_sequence(n, best_p)
        it += 1
    return best_p, best_score, best_vec


def hill_climb(
    vec: np.ndarray,
    time_deadline: float,
    max_no_improve: int = 2000,
) -> np.ndarray:
    """Simple stochastic hill‑climbing that moves a tiny amount of mass.
    Only strictly improving moves are kept."""
    n = len(vec)
    cur = vec.copy()
    cur_score = evaluate_vec(cur)
    no_improve = 0

    while time.time() < time_deadline and no_improve < max_no_improve:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # Transfer up to 5 % of the source entry
        delta = np.random.rand() * 0.05 * cur[i]
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] = cand[j] + delta
        cand_score = evaluate_vec(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
    return cur


# --------------------------------------------------------------------------- #
# LP‑based refinement (requires SciPy)
# --------------------------------------------------------------------------- #

try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


def _build_conv_matrix(a: np.ndarray) -> _sp.csr_matrix:
    """Return the sparse Toeplitz matrix T (size (2n‑1)×n) such that
    T @ b == np.convolve(a, b) (linear convolution)."""
    n = a.shape[0]
    L = 2 * n - 1

    # i repeated n times, j tiled n times
    i_vals = np.repeat(np.arange(n, dtype=np.int32), n)
    j_vals = np.tile(np.arange(n, dtype=np.int32), n)

    rows = i_vals + j_vals          # shape (n*n,)
    cols = j_vals
    data = a[i_vals]                # a_i for each row block

    return _sp.csr_matrix((data, (rows, cols)), shape=(L, n))


def lp_refine_once(a: np.ndarray, time_deadline: float) -> np.ndarray:
    """Perform a single LP refinement step.
    Returns a vector that is at least as good as `a` (may be unchanged)."""
    if not _SCIPY_OK:
        return a  # fallback – SciPy missing

    n = a.shape[0]
    L = 2 * n - 1
    M = np.max(np.convolve(a, a))

    # Build the convolution matrix (sparse)
    A = _build_conv_matrix(a)

    # Linear programme:
    # maximise   sum(b)    <=> minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(L, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # HiGHS will ignore a supplied time limit; we rely on the outer budget.
    res = _opt.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return a

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return a

    # Normalise to unit sum – the evaluation function is scale invariant.
    b_norm = b / sum_b

    # Some simple mixtures of a and b_norm are tried; the best is kept.
    candidates = [
        b_norm,
        (a + b_norm) / 2.0,
        0.7 * a + 0.3 * b_norm,
        0.3 * a + 0.7 * b_norm,
    ]

    best_vec = a
    best_score = evaluate_vec(a)

    for cand in candidates:
        score = evaluate_vec(cand)
        if score < best_score:
            best_score = score
            best_vec = cand

    return best_vec


def iterative_lp_refine(
    start_vec: np.ndarray,
    overall_deadline: float,
    max_iters: int = 10,
) -> tuple[np.ndarray, float]:
    """Repeatedly apply `lp_refine_once` while it yields improvement.
    Returns (best_vector, best_score)."""
    a = start_vec.copy()
    best_score = evaluate_vec(a)
    it = 0

    while it < max_iters and time.time() < overall_deadline:
        iter_deadline = overall_deadline  # inner LP sees the global deadline
        a_new = lp_refine_once(a, iter_deadline)
        new_score = evaluate_vec(a_new)

        if new_score < best_score - 1e-12:
            print(f"[LP] iteration {it}: improved score {new_score:.6f}")
            a = a_new
            best_score = new_score
            it += 1
        else:
            print(f"[LP] iteration {it}: no further improvement – stopping.")
            break

    return a, best_score
