def generate_seq(p, gamma, n, taper='exp'):
    idx = np.arange(1, n+1)
    base = idx ** -p
    if taper == 'exp':
        taper_factor = np.exp(-gamma * (idx-1) / (n-1))
    elif taper == 'gauss':
        taper_factor = np.exp(-gamma * ((idx-1)/(n-1))**2)
    else:
        taper_factor = np.ones_like(idx)
    return base * taper_factor

"""
Propose a low‑evaluation step‑function.

The search works in three phases:
1. Baseline – a pure power‑law a_i = (i+1)^-0.5 (the previous best).
2. Coarse grid – we vary three knobs:
      * p    – exponent of the power‑law (0.45, 0.5, 0.55)
      * k    – index at which the early part is “capped’’:
                a_i = max(i+1, k+1)^-p  (so the first k entries all have the
                height of the (k+1)-th entry).
      * γ    – strength of an exponential taper  exp(-γ·i/(n‑1)).
      * type – two taper shapes: plain exponential and Gaussian‑like
               (exp(-γ·(i/(n‑1))²)).
   The objective `evaluate_sequence` is computed for each candidate and the
   best is kept.
3. Fine local search – a finer grid around the current best and a short
   random‑hill‑climbing phase to polish the parameters.

All parameters are kept small enough that the convolution (np.convolve) is
fast; with n=4000 a single evaluation costs only a few milliseconds,
allowing several thousand evaluations well within the 1000 s budget.
The algorithm returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

def _make_sequence(p: float, k: int, gamma: float, n: int,
                   taper_type: str) -> np.ndarray:
    """
    Build a candidate sequence of length n.

    Parameters
    ----------
    p : float
        Power‑law exponent (a_i ≈ (i+1)^-p).
    k : int
        Number of leading entries that are capped to the value at index k.
    gamma : float
        Exponential‑taper strength.
    n : int
        Length of the step function.
    taper_type : str
        Either 'exp' (exp(-γ·i/(n‑1))) or 'sqexp' (exp(-γ·(i/(n‑1))²)).

    Returns
    -------
    np.ndarray
        The generated sequence (all elements ≥0).
    """
    idx = np.arange(1, n + 1, dtype=np.float64)      # i = 1 … n
    # power‑law with a flat cap for the first k entries
    if k > 0:
        # max(i, k+1) gives a plateau of height (k+1)^-p for i≤k
        base = np.maximum(idx, k + 1) ** -p
    else:
        base = idx ** -p

    # optional exponential taper
    if gamma > 0.0:
        if taper_type == "exp":
            factor = np.exp(-gamma * (idx - 1) / (n - 1))
        else:  # 'sqexp'
            factor = np.exp(-gamma * ((idx - 1) / (n - 1)) ** 2)
        base = base * factor
    return base


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Return a sequence of non‑negative heights that (empirically) minimizes
    the supplied `evaluate_sequence` function.

    The routine respects the 1000 s time limit, performs a multi‑stage
    grid + random search and finally returns the best candidate as a list.
    """
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a small safety margin

    # length of the step function – a few thousand gives a good trade‑off
    n = 4000

    # ------------------------------------------------------------------
    # 0.) Baseline (pure power‑law) – keep it as a reference.
    # ------------------------------------------------------------------
    baseline_seq = _make_sequence(p=0.5, k=0, gamma=0.0, n=n, taper_type="exp")
    best_score = float("inf")
    best_seq = baseline_seq.copy()
    best_params = {"p": 0.5, "k": 0, "gamma": 0.0, "type": "exp"}

    try:
        best_score = evaluate_sequence(baseline_seq.tolist())
        print(f"Baseline score (p=0.5, k=0, γ=0): {best_score:.6f}")
    except Exception as exc:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    p_vals = [0.45, 0.5, 0.55]                       # exponent candidates
