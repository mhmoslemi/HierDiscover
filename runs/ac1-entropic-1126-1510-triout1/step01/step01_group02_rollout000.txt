"""
LP‑based refinement for minimizing the step‑function evaluation score.

The algorithm works on the probability simplex (all entries ≥0, sum = 1)
so that the scale invariance of ``evaluate_sequence`` is removed.
Starting from a good known construction (``height_sequence_1`` when
available) we repeatedly

1. compute the current maximal autocorrelation value M = max(f*f);
2. solve the linear program

        maximise   Σ g_j
        subject to ∀k   (f * g)_k ≤ M ,   g_j ≥ 0 ,

   which is linear in the unknown vector g because f is fixed.
3. scale g back to unit ‑L1 norm and convex‑combine it with the current
   f (mixing coefficient t that adapts automatically).

If the new candidate improves the evaluation score it is kept; otherwise
the mixing coefficient is reduced.  After a few LP steps a short
simulated‑annealing / random‑pair local search is performed to polish the
result.  The best sequence seen before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional import of cvxpy – the core of the LP refinement.
# ----------------------------------------------------------------------
try:
    import cvxpy as cp      # type: ignore
    _HAS_CVX = True
except Exception:           # pragma: no cover
    _HAS_CVX = False

# ----------------------------------------------------------------------
# Helper: a cheap random pairwise mass‑swap (keeps sum exactly 1).
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Pick two distinct indices i, j and move a small amount of mass
    (drawn from a symmetric uniform interval of width ``sigma``) from one
    to the other while preserving non‑negativity."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # Zero‑mean proposal (uniform rather than Gaussian – cheap and bounded)
    delta = (np.random.rand() - 0.5) * sigma

    # Clip to keep both entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])      # take from j
    else:
        delta = -min(-delta, seq[i])    # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Linear‑programming step: given a current f and its max autocorrelation M,
# find a new g that maximises total mass while respecting (f * g) ≤ M.
# ----------------------------------------------------------------------
def _lp_max_mass(f: np.ndarray, M: float):
    """Solve   maximise Σ g   subject to   (f * g) ≤ M , g ≥ 0.
    Returns the optimal g as a NumPy array or ``None`` on failure."""
    if not _HAS_CVX:
        return None

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    # Convolution of a constant (f) with the variable g.
    # ``cp.Constant`` creates a CVXPY constant expression.
    conv_fg = cp.conv(cp.Constant(f), g)            # length 2n‑1 vector

    constraints = [conv_fg <= M]                    # component‑wise
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is a fast interior‑point solver for LPs; warm_start speeds up
    # successive solves with the same size.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=500, verbose=False)
    except Exception:                               # pragma: no cover
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    g_val = g.value
    if g_val is None:
        return None
    return np.asarray(g_val, dtype=float)

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that strives
    to minimise ``evaluate_sequence`` within the allotted time.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a tiny safety margin for the harness
    deadline = start + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # Initialise a starting sequence.
    # --------------------------------------------------------------
    try:
        # Use the best known construction if the global variable exists.
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        if base.size == 0:
            raise ValueError
        base = np.maximum(base, 0.0)                     # safety
        base /= max(base.sum(), 1e-12)                    # normalise to sum=1
    except Exception:                                    # pragma: no cover
        # Fallback: a modest‑size uniform vector.
        base = np.full(2000, 1.0 / 2000.0, dtype=float)

    # Keep track of the best sequence seen so far.
    best_seq = base.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float("inf")

    # --------------------------------------------------------------
    # Primary optimisation loop – LP refinement + short local search.
    # --------------------------------------------------------------
    # Number of outer LP iterations (tuned empirically).  We stop early if
    # time runs out or if the LP ceases to make progress.
    max_lp_iters = 30
    # Mixing coefficient (convex combination weight).  It is adapted
    # dynamically: increased after an improvement, decreased otherwise.
    mix_coef = 0.5

    f = base.copy()
    current_score = best_score

    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1

        # ----------------------------------------------------------
        # 1. Compute the current maximal autocorrelation value M.
        # ----------------------------------------------------------
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # ----------------------------------------------------------
        # 2. Solve the LP that tries to push as much mass as possible
        #    under the same bound M.
        # ----------------------------------------------------------
        g_raw = _lp_max_mass(f, M)
        if g_raw is None:
            # Either cvxpy is unavailable or the problem became infeasible.
            break

        sum_g = g_raw.sum()
        # If the LP cannot increase the total mass beyond the current sum
        # (which is 1 after normalisation), we have reached a stationary point.
        if sum_g <= 1.0 + 1e-12:
            break

        # ----------------------------------------------------------
        # 3. Scale g to unit L1 norm and perform a convex update.
        # ----------------------------------------------------------
        g_norm = g_raw / sum_g                       # now Σ g_norm = 1
        f_candidate = (1.0 - mix_coef) * f + mix_coef * g_norm

        # ----------------------------------------------------------
        # 4. Evaluate and possibly accept the new candidate.
        # ----------------------------------------------------------
        try:
            cand_score = evaluate_sequence(f_candidate.tolist())
        except Exception:                           # pragma: no cover
            cand_score = float("inf")

        # Keep the best ever seen.
        if cand_score < best_score:
            best_score = cand_score
            best_seq = f_candidate.copy()
            print(f"[LP iter {lp_iter:02d}] New best score: {best_score:.6f}")

        # Accept if we improve; otherwise shrink the mixing coefficient.
        if cand_score < current_score:
            f = f_candidate
            current_score = cand_score
            mix_coef = min(0.9, mix_coef * 1.05)   # slowly accelerate
        else:
            mix_coef = max(0.1, mix_coef * 0.5)    # back‑off

    # --------------------------------------------------------------
    # A short simulated‑annealing / random‑pair polishing phase.
    # --------------------------------------------------------------
    polish_iters = 4000
    it = 0
    while time.time() < deadline and it < polish_iters:
        it += 1
        # Temperature schedule – linear cooling.
        temp_frac = it / polish_iters
        sigma = 0.3 * (1.0 - temp_frac) + 0.02 * temp_frac

        candidate = _random_pair_move(best_seq, sigma)

        try:
            cand_score = evaluate_sequence(candidate.tolist())
        except Exception:                           # pragma: no cover
            continue

        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate
            print(f"[Polish iter {it:04d}] New best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # --------------------------------------------------------------
    # Safety: guarantee non‑negative numbers and a reasonable length.
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
