---------

def power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    Decreasing power‑law a_i ∝ (i+1)^{-p}, normalised to sum 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    total = a.sum()
    if total == 0:
        return np.full(n, 1.0 / n)
    return a / total


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, ∑x = 1}.
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def evaluate_vec(v: np.ndarray) -> float:
    """Wrapper that forwards a numpy array to the provided evaluate_sequence."""
    return evaluate_sequence(v.tolist())


def hill_climb(vec: np.ndarray, deadline: float,
               max_no_improve: int = 3000) -> np.ndarray:
    """
    Tiny stochastic moves that preserve the total sum.
    Only strictly improving moves are kept.
    """
    n = len(vec)
    cur = vec.copy()
    cur_score = evaluate_vec(cur)
    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 2 % of the source entry
        delta = np.random.rand() * 0.02 * cur[i]
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = evaluate_vec(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
    return cur


# --------------------------------------------------------------------------- #
# Main optimisation routine
# --------------------------------------------------------------------------- #

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) obtains a score
    below the current best bound (≈1.503).  The function runs for at most
    *budget_s* seconds and prints intermediate scores.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------- #
    # Determine problem size
    # ------------------------------------------------------------------- #
    try:
        n = len(height_sequence_1)           # use the length from the previous run
    except Exception:
        n = 2000

    # ------------------------------------------------------------------- #
    # 1) Initialise with a power‑law (p≈0.5) – good baseline
    # ------------------------------------------------------------------- #
    a = power_law_sequence(n, p=0.5)
    best_score = evaluate_vec(a)
    best_vec = a.copy()
    print(f"[init] length {n}, initial score {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # 2) Smooth‑max projected gradient descent
    # ------------------------------------------------------------------- #
    # Parameters – they adapt during the run
    lam = 5.0          # smooth‑max temperature (will be increased)
    eta = 0.6          # step size
    it = 0
    while time.time() < deadline and it < 800:
        conv = np.convolve(a, a)                     # full convolution, length 2n‑1
        maxc = conv.max()

        # soft‑max weights (stable numerics)
        w = np.exp(lam * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            break
        w /= w_sum

        # gradient of the smooth‑max term:
        #   grad_i = 2 * Σ_k w_k * a_{k-i}   (indices out of range give 0)
        # This is exactly a convolution of w with the reversed a.
        a_rev = a[::-1]
        grad_full = np.convolve(w, a_rev)            # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]      # keep the n relevant entries

        # projected gradient step
        a_next = a - eta * grad
        a_next = simplex_projection(a_next)

        # evaluate new point
        new_score = evaluate_vec(a_next)

        # keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_vec = a_next.copy()
            print(f"[gd] iter {it:04d}, λ={lam:.2f}, η={eta:.3f}, score={new_score:.6f}")

        # simple adaptive scheme:
        if new_score > best_score:               # step was too aggressive
            eta *= 0.85
        else:                                    # successful – modestly increase step
            eta = min(eta * 1.05, 2.0)

        # slowly tighten the smooth‑max (approach the true max)
        lam = min(lam * 1.02, 40.0)

        a = a_next
        it += 1

    # ------------------------------------------------------------------- #
    # 3) Final random hill‑climbing polish
    # ------------------------------------------------------------------- #
    hill_deadline = min(deadline, time.time() + 0.25 * budget_s)
    polished = hill_climb(best_vec, hill_deadline)

    polished_score = evaluate_vec(polished)
    if polished_score < best_score:
        best_score = polished_score
        best_vec = polished
        print(f"[hill] after polishing score={best_score:.6f}")

    print(f"[final] best score obtained: {best_score:.6f} (length {n})")
    return best_vec.tolist()
