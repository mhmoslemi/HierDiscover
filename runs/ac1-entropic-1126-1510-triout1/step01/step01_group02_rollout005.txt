"""
Evolution‑Strategy search for a step‑function with a small
evaluate_sequence() value.

The algorithm works on the probability simplex (non‑negative entries,
sum = 1).  It maintains a small population, mutates it with
Gaussian noise, projects back onto the simplex by simple clipping &
renormalisation, and keeps the μ best individuals (μ/λ‑ES).  In
addition, every generation a *targeted* move is created that tries to
reduce the currently maximal convolution entry – this mimics the
active‑set idea from the published approach.  Periodically a few
projected‑gradient steps (using the top‑k convolution peaks) are
applied to the current best solution.

The search runs until the time budget (≈ 1000 s) expires and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: clip to the simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negative entries to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:                     # fallback to uniform point
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Fast scoring (same formula as evaluate_sequence but without the
# safety wrapper – all our vectors are already valid).
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """
    Return the evaluation score for a probability vector `seq`.
    Equivalent to evaluate_sequence(seq.tolist()).
    """
    n = seq.size
    conv = np.convolve(seq, seq)          # linear convolution
    max_conv = conv.max()
    # denominator = (sum(seq))**2  == 1 because we stay on the simplex
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Gradient step that reduces several large convolution peaks at once.
# ----------------------------------------------------------------------
def _multi_lag_gradient_step(seq: np.ndarray, eta: float, top_k: int = 3) -> np.ndarray:
    """
    One projected (actually clip‑and‑renormalise) gradient step.
    The gradient is the sum of the sub‑gradients of the `top_k`
    largest convolution entries.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    # indices of the k largest entries (ties broken arbitrarily)
    top_idxs = np.argpartition(-conv, range(top_k))[:top_k]

    g = np.zeros_like(seq)
    idx = np.arange(n)
    for k in top_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * seq[j[mask]]

    # step opposite to the gradient
    new_seq = seq - eta * g
    return _clip_and_normalise(new_seq)


# ----------------------------------------------------------------------
# Targeted move: try to reduce the currently maximal convolution entry.
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, frac: float = 0.05) -> np.ndarray:
    """
    Identify the lag with maximal convolution value, pick the pair
    (i, k‑i) that contributes most to it and move a small fraction
    of mass from the larger index to a random location.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())

    # contribution = a_i * a_{k-i} for all feasible i
    i_arr = np.arange(n)
    j = k - i_arr
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i_arr[mask]] * seq[j[mask]]

    if not np.any(mask):
        # fallback – just return a small random perturbation
        cand = seq + frac * (np.random.rand(n) - 0.5)
        return _clip_and_normalise(cand)

    # pick the index with the largest contribution
    i_max = int(np.argmax(contrib))
    j_max = k - i_max

    # decide from which of the two we take mass (the larger one)
    src = i_max if seq[i_max] >= seq[j_max] else j_max
    dst = np.random.randint(n)          # destination (can be any index)
    while dst == src:
        dst = np.random.randint(n)

    # amount to transfer (up to `frac` of the source mass)
    amount = frac * seq[src] * np.random.rand()
    cand = seq.copy()
    cand[src] -= amount
    cand[dst] += amount
    return _clip_and_normalise(cand)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Evolution‑Strategy optimiser for the step‑function problem.
    Returns a list of non‑negative heights that minimise `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a very small safety margin
    deadline = start_time + max(0.0, budget_s - 0.5)

    # --------------------------------------------------------------
    # Initialise population size and dimension
    # --------------------------------------------------------------
    try:
        # use the previously discovered construction if it exists
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        base = _clip_and_normalise(base)          # put on simplex
    except Exception:
        # fallback – choose a reasonable length
        n = 8000
        base = np.full(n, 1.0 / n, dtype=float)

    # Evolution‑Strategy parameters
    mu = 5                         # number of parents kept
    lam = 30                       # number of offspring per generation
    sigma = 0.08                   # mutation step size (std. dev. in absolute terms)

    # initialise the (μ) parent population – start with the base solution
    parents = np.tile(base, (mu, 1)).copy()
    # add a few random simplex points to increase diversity
    for i in range(mu - 1):
        rand = np.random.rand(n)
        parents[i] = _clip_and_normalise(rand)

    # evaluate parents once
    parent_scores = np.array([_score(p) for p in parents])
    best_idx = int(parent_scores.argmin())
    best = parents[best_idx].copy()
    best_score = float(parent_scores[best_idx])

    # keep track of progress
    generation = 0
    last_improve_gen = 0
    print(f"[Init] n = {n}, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Evolution loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        generation += 1

        # ----------------------------------------------------------
        # Generate λ offspring via Gaussian mutation of the current mean
        # ----------------------------------------------------------
        mean = parents.mean(axis=0)                # (μ,λ)-ES recombination
        offspring = np.empty((lam, n), dtype=float)

        for i in range(lam):
            cand = mean + sigma * np.random.randn(n)
            cand = _clip_and_normalise(cand)
            offspring[i] = cand

        # ----------------------------------------------------------
        # Add a couple of deterministic, problem‑specific candidates
        # ----------------------------------------------------------
        # 1) targeted move based on current best
        offspring[0] = _targeted_move(best, frac=0.03)
        # 2) symmetrised version (average with its reverse)
        rev = best[::-1]
        offspring[1] = _clip_and_normalise(0.5 * (best + rev))

        # ----------------------------------------------------------
        # Evaluate all offspring
        # ----------------------------------------------------------
        scores = np.empty(lam, dtype=float)
        for i, cand in enumerate(offspring):

        # ----------------------------------------------------------
        # Update best solution if any offspring is better
        # ----------------------------------------------------------
        min_off_idx = int(scores.argmin())
        if scores[min_off_idx] < best_score:
            best_score = float(scores[min_off_idx])
            best = offspring[min_off_idx].copy()
            last_improve_gen = generation
            print(f"[Gen {generation:5d}] New best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Survival selection: keep the μ best among parents+offspring
        # ----------------------------------------------------------
        combined = np.vstack([parents, offspring])
        combined_scores = np.concatenate([parent_scores, scores])
        keep_idx = combined_scores.argsort()[:mu]
        parents = combined[keep_idx]
        parent_scores = combined_scores[keep_idx]

        # ----------------------------------------------------------
        # Adapt the mutation step size (simple 1/5‑th rule)
        # ----------------------------------------------------------
        # success = fraction of offspring that are strictly better than
        # the current mean (approximated by the best parent)
        success = np.mean(scores < best_score + 1e-12)
        if success > 0.2:
            sigma = min(sigma * 1.2, 0.5)
        else:
            sigma = max(sigma * 0.85, 1e-5)

        # ----------------------------------------------------------
        # Occasionally refine the best individual with multi‑lag gradients
        # ----------------------------------------------------------
        if generation % 30 == 0:
            tmp = best.copy()
            # a few decreasing‑step gradient steps
            for step in range(5):
                eta = 0.5 / (generation + step + 1.0)
                tmp = _multi_lag_gradient_step(tmp, eta=eta, top_k=3)
            tmp_score = _score(tmp)
            if tmp_score < best_score:
                best_score = float(tmp_score)
                best = tmp
                print(f"[Gen {generation:5d}] Gradient‑refined best = {best_score:.6f}")

        # ----------------------------------------------------------
        # Simple stagnation restart: if no improvement for many generations,
        # inject a fresh random individual as the new mean.
        # ----------------------------------------------------------
        if generation - last_improve_gen > 500:
            # replace the worst parent with a fresh random point
            new_rand = _clip_and_normalise(np.random.rand(n))
            parents[-1] = new_rand
            parent_scores[-1] = _score(new_rand)
            last_improve_gen = generation
            sigma = 0.08
            print(f"[Gen {generation:5d}] Restart due to stagnation")

    # --------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # --------------------------------------------------------------
    print(f"[Done] Best score after {generation} generations = {best_score:.6f}")
    return best.tolist()
