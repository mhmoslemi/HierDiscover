    n = len(seq)

    m = 1 << ((2*n - 1).bit_length())

    fft_seq = np.fft.rfft(seq, n=m)

    conv = conv[:2*n-1]
    return np.max(conv.real) # conv should be real

def _max_conv_gradient(seq):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    left = max(0, k - (n - 1))
    right = min(k, n - 1)
    # indices i from left to right inclusive
    idx = np.arange(left, right + 1)
    g = np.zeros_like(seq)
    g[idx] = 2.0 * seq[k - idx]
    return g

def _pairwise_move(seq: np.ndarray, sigma: float):
    n = len(seq)
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq

def _exp_seq(n: int, c: float) -> np.ndarray:
    s = c / n
    r = 1.0 - s
    if r <= 0.0:
        # fallback to uniform
        return np.full(n, 1.0 / n)
    a = r ** np.arange(n)
    total = a.sum()
    return a / total

def _power_seq(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    total = a.sum()
    if total == 0:
        return np.full(n, 1.0 / n)
    return a / total

"""
Hybrid shape‑search + mirror‑descent optimizer for the step‑function
evaluation problem.

* 1.  Fast “offline” scan over a few parametric families (exponential,
   power‑law, mixtures) using an FFT‑based max‑convolution that runs in
   O(n log n).  The best candidate of this scan seeds the local search.

* 2.  Local optimisation mixes two moves:
      – Mirror‑descent (multiplicative update on the simplex) based on the
        sub‑gradient of the current maximal convolution entry.
      – Small random pairwise mass swaps that preserve the simplex exactly.
   The step‑size / swap‑scale decays like 1/√iteration and a simple
   simulated‑annealing acceptance rule lets the algorithm escape shallow
   plateaus.

* 3.  The whole routine runs for at most ``budget_s`` seconds (default
   1000).  The final sequence is evaluated with the supplied
   ``evaluate_sequence`` function and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Fast convolution via FFT (full length = 2n‑1)
# ----------------------------------------------------------------------
def _fast_convolution(seq: np.ndarray) -> np.ndarray:
    """Full convolution of seq with itself using FFT (real inputs)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv_full = np.fft.irfft(fft_seq * fft_seq, n=size)
    return conv_full[: 2 * n - 1]


def _fast_max_conv(seq: np.ndarray) -> float:
    """Maximum entry of seq * seq."""
    conv = _fast_convolution(seq)
    return float(np.max(conv))


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast proxy for ``evaluate_sequence`` when the sequence is already
    normalised to sum = 1.  Returns 2·n·max(conv) (the exact value of the
    evaluation function under the sum‑normalisation).
    """
    seq = np.asarray(seq, dtype=float)
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    seq = seq / total
    n = seq.shape[0]
    max_conv = _fast_max_conv(seq)
    # sum == 1, so denominator is 1
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _max_conv_gradient(seq: np.ndarray):
    """Return the sub‑gradient of the maximal convolution entry."""
    n = seq.shape[0]
    conv = _fast_convolution(seq)
    k = int(np.argmax(conv))
    left = max(0, k - (n - 1))
    right = min(k, n - 1)
    idx = np.arange(left, right + 1, dtype=int)
    grad = np.zeros_like(seq)
    grad[idx] = 2.0 * seq[k - idx]
    return grad, k


# ----------------------------------------------------------------------
# Mirror‑descent (multiplicative) step on the simplex
# ----------------------------------------------------------------------
def _mirror_descent_step(seq: np.ndarray, grad: np.ndarray, eta: float):
    """Multiplicative update seq_i ← seq_i·exp(−η·grad_i) followed by projection."""
    new_seq = seq * np.exp(-eta * grad)
    s = new_seq.sum()
    if s <= 0:
        # degenerate – fall back to the old point
        return seq.copy()
    new_seq = new_seq / s
    return new_seq


# ----------------------------------------------------------------------
# Random pairwise mass‑swap (keeps the sum exactly 1)
# ----------------------------------------------------------------------
def _pairwise_move(seq: np.ndarray, sigma: float):
    """Transfer a small amount of mass between two random coordinates."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------
def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=float)


def _exp_seq(n: int, c: float) -> np.ndarray:
    """Geometric (exponential) shape a_i ∝ r^i with r = 1−c/n."""
    s = c / n
    r = 1.0 - s
    if r <= 0.0:
        return _uniform_seq(n)
    a = r ** np.arange(n, dtype=float)
    a_sum = a.sum()
    if a_sum == 0:
        return _uniform_seq(n)
    return a / a_sum


def _power_seq(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^{−p}."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        return _uniform_seq(n)
    return a / a_sum


def _mixed_seq(n: int, base_seq, mix_seq, alpha: float) -> np.ndarray:
    """Convex combination α·base + (1−α)·mix (both already normalised)."""
    return alpha * base_seq + (1.0 - alpha) * mix_seq
