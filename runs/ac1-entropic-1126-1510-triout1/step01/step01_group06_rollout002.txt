def build_constraints(f, M):
    n = len(f)
    m = 2*n - 1
    A = np.zeros((m, n))
    for k in range(m):
        i_min = max(0, k - (n-1))
        i_max = min(k, n-1)
        # i runs from i_min to i_max
        # j = k - i runs from k-i_max to k-i_min (i.e., valid)

    b_ub = np.full(m, M)
    return A, b_ub

"""
Propose a low‑score step‑function.

The algorithm combines several ideas that differ from the previous
gradient‑only approach:

* 1️⃣  Exponentiated‑gradient (mirror‑descent) on the probability simplex.
  The update a_i ← a_i·exp(‑η·g_i) keeps all entries non‑negative and
  automatically respects the sum‑to‑one constraint after renormalisation.
  The sub‑gradient g is taken only at the active (maximal) convolution
  indices.

* 2️⃣  A simple back‑tracking line‑search for the step size η and occasional
  random perturbations to escape shallow plateaus.

* 3️⃣  After the multiplicative descent a few cheap linear‑programming
  refinements are performed (if SciPy is available).  The LP maximises the
  ℓ₁‑norm of a new vector b under the current convolution bound
  ‖a∗b‖∞ ≤ M.  Mixing a with the scaled optimum b provably cannot increase
  the bound and often yields a further drop.

* 4️⃣  Multiple random restarts with diverse initial shapes (geometric,
  power‑law, mixed, Dirichlet) are tried until the time budget (~1000 s)
  expires.  The best sequence seen globally is returned.

All numbers are non‑negative; the final list is normalised to sum = 1, which
does not affect the evaluation score because the evaluator rescales the
sequence internally.
"""

def _conv_max_fft(a: np.ndarray):
    """Return (max convolution value, indices where it is attained) for a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = np.max(conv)
    # Use a tiny tolerance to collect all indices that are effectively maximal.
    idx = np.where(np.abs(conv - max_val) < 1e-14)[0]
    return float(max_val), idx.astype(int)


def _gradient_for_max(a: np.ndarray, max_idx: np.ndarray):
    """
    Sub‑gradient of the maximal self‑convolution w.r.t. a.
    Implements the same formula as the reference code (factor 2).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # each pair (i,j) contributes 2·a_j to ∂/∂a_i
        grad[i] += 2.0 * a[j]
    return grad


def _exp_gradient_optimize(a0: np.ndarray, n: int, deadline: float):
    """
    Mirror‑descent on the simplex with exponential updates.
    Returns the best sequence found (as a numpy array) and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    max_c, max_idx = _conv_max_fft(a)
    best_score = 2.0 * n * max_c
    it = 0
    # initial step‑size (will be decayed like 1/sqrt(it))
    eta0 = 0.4

    while time.time() < deadline and it < 3000:
        it += 1
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c

        # gradient at the current active set
        grad = _gradient_for_max(a, max_idx)

        improved = False
        # back‑tracking over a few scaled step sizes
        for scale in (1.0, 0.5, 0.2, 0.1, 0.05):
            eta = eta0 / np.sqrt(it) * scale
            # exponential (multiplicative) update, then re‑normalise
            cand = a * np.exp(-eta * grad)
            s = np.sum(cand)
            if s == 0.0:                     # numerical safety – fall back to uniform
                cand = np.full_like(a, 1.0 / n)
            else:
                cand = cand / s
            max_c2, _ = _conv_max_fft(cand)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score - 1e-12:
                a = cand
                cur_score = cand_score
                improved = True
                break

        if not improved:
            pert = a.copy()
            # pick up to 6 random coordinates (more when n is large)
            idx = np.random.randint(n, size=min(6, n))
            # add a small Gaussian bump
            pert[idx] += np.random.randn(len(idx)) * 0.001
            pert = np.maximum(pert, 0.0)
            s = np.sum(pert)
            if s > 0.0:
                pert = pert / s
                max_c2, _ = _conv_max_fft(pert)
                pert_score = 2.0 * n * max_c2
                if pert_score < cur_score:
                    a = pert
                    cur_score = pert_score
                    improved = True

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if it % 200 == 0:
                print(f"    exp‑grad it {it}, new local best {best_score:.6f}")

        # early exit if we already reached a very good bound
        if best_score < 1.50:
            break

    return best_a, best_score


def _build_lp_constraints(a: np.ndarray, M: float):
    """
    Build the LP matrices for:
        maximize Σ b_j
        s.t.   (a * b)[k] ≤ M   for all k = 0 … 2n‑2
               b_j ≥ 0
    Returns (A_ub, b_ub) suitable for scipy.optimize.linprog.
    """
    n = a.size
    m = 2 * n - 1
    # Pre‑allocate dense matrix (m × n)
    A = np.zeros((m, n), dtype=np.float64)
    # Each row k corresponds to convolution index k
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # i runs from i_min … i_max, j = k - i
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        A[k, j_vals] = a[i_vals]          # coefficient of b_j in (a*b)[k]
    b_ub = np.full(m, M, dtype=np.float64)
    return A, b_ub


def _lp_refine(a: np.ndarray, n: int, max_iter: int = 3):
    """
    Perform a few LP‑based refinement steps.
    Each step solves:
        maximise Σ b_j   subject to   a∗b ≤ M   and   b ≥ 0,
    where M is the current maximal convolution value.
    The resulting b is mixed with a (t = 0.5) and re‑normalised.
    """
    if not _LP_AVAILABLE:
        return a   # fall back silently

    for _ in range(max_iter):
        max_c, _ = _conv_max_fft(a)
        M = max_c
        A, b_ub = _build_lp_constraints(a, M)

        # LP: minimize -Σ b_j  (i.e. maximise Σ b_j)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:                     # pragma: no cover
            break
        if not res.success:
            break

        b = res.x
        sum_a = np.sum(a)
        sum_b = np.sum(b)
        if sum_b <= sum_a * (1.0 + 1e-9):
            # No improvement in the ℓ₁‑norm, stop early
            break

        # Scale b back to the same total mass as a
        b_scaled = b * (sum_a / sum_b)

        # Mix a and the scaled b (simple average)
        a = 0.5 * a + 0.5 * b_scaled
        # Renormalise to guard against numerical drift
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine respects the 1000 s time limit, performs
    many random restarts, runs an exponentiated‑gradient local optimiser and
    optionally refines the result with a small linear‑programming routine.
    """
    np.random.seed(seed)
    end_time = time.time() + budget_s - 5.0      # small safety margin

    best_score = np.inf
    best_seq = None

    # ------------------------------------------------------------------
    # 1️⃣  Try to improve the previously discovered sequence, if it exists.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)   # global variable from the runner
        if init.size > 0:
            init = np.maximum(init, 0.0)
            init = init / np.sum(init)
            n0 = init.size
            # give the gradient optimiser a decent slice of the time budget
            sub_dead = min(end_time, time.time() + 12.0)
            a_opt, sc_opt = _exp_gradient_optimize(init, n0, sub_dead)
            a_opt = _lp_refine(a_opt, n0, max_iter=2)
            max_c, _ = _conv_max_fft(a_opt)
            sc_opt = 2.0 * n0 * max_c
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"Improved from global start: n={n0}, score={best_score:.6f}")
    except NameError:
        # height_sequence_1 is not defined – ignore
        pass

    # -------------------------------------------------
    # 2️⃣  Main stochastic search with many restarts.
    # -------------------------------------------------
    while time.time() < end_time:
        # ----- choose problem size and an initial shape -----
        n = np.random.randint(1400, 2600)   # moderate size, good trade‑off
        shape = np.random.choice(["geom", "power", "mixed", "random"])

        if shape == "geom":
            # geometric decay a_i = r^i   (r ≈ 0.997‑0.999)
            t = np.random.uniform(0.0002, 0.0012)
            r = np.exp(-t)
            a = r ** np.arange(n, dtype=np.float64)

        elif shape == "power":
            # power‑law a_i = (i+1)^{-p}
            p = np.random.uniform(0.1, 2.2)
            a = (np.arange(1, n + 1, dtype=np.float64) ** -p)

        elif shape == "mixed":
            # product of geometric and power‑law
            t = np.random.uniform(0.0002, 0.0008)
            r = np.exp(-t)
            p = np.random.uniform(0.3, 1.4)
            a = (r ** np.arange(n, dtype=np.float64)) * (
                np.arange(1, n + 1, dtype=np.float64) ** -p)

        else:  # "random"
            a = np.random.gamma(shape=1.0, scale=1.0, size=n)

        # Random permutation does not change the multiset of values but
        # often yields a different convolution shape.
        if np.random.rand() < 0.5:
            np.random.shuffle(a)

        # Normalise to the probability simplex (sum = 1)
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # ----- local optimisation (exponential mirror descent) -----
        # Allocate a modest time slice (≈8 s) per restart.
        sub_dead = min(end_time, time.time() + 8.0)
        a_opt, sc_opt = _exp_gradient_optimize(a, n, sub_dead)

        # ----- optional LP refinement (quick, 1‑2 steps) -----
        if _LP_AVAILABLE and time.time() < end_time:
            a_opt = _lp_refine(a_opt, n, max_iter=2)
            max_c, _ = _conv_max_fft(a_opt)
            sc_opt = 2.0 * n * max_c

        # ----- keep the best overall solution -----
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"Global improvement: n={n}, shape={shape}, score={best_score:.6f}")

        # stop early if the target is already met
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Fallback – a uniform distribution (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Return a plain Python list; the evaluator will clip automatically.
    return best_seq.tolist()
