"""
Optimisation of a non‑negative step function for the score

    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

The scale of ``a`` does not matter, therefore we normalise to
``sum(a)=1`` and simply minimise ``2*n*max(conv)``.
The algorithm combines two ideas:

1.  **Exploration of parametrised families** – geometric,
    power‑law, mixed geometric·power, and a “U‑shaped’’ arcsine‑like
    distribution.  These cheap constructions already give scores well
    below the uniform baseline (2.0) and sometimes beat the previous
    best (≈ 1.58).

2.  **Smooth‑max gradient descent** – the hard max is replaced by a
    softmax (temperature based) which yields a dense sub‑gradient.
    The gradient is computed by a single FFT‑based convolution.
    An Adam‑style update together with Euclidean projection onto the
    probability simplex drives the sequence towards a flatter
    self‑convolution.  The optimiser is run for a few seconds on each
    restart; the best sequence seen globally is returned.

The whole search respects the 1000 s time‑budget and always returns a
valid list of non‑negative heights (the evaluator clips values to
[0,1000] automatically).
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:                                   # all entries are non‑positive
        theta = cssv[-1] / n
    return np.maximum(v - theta, 0.0)


def _full_conv(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Smooth approximation of the max using a softmax."""
    # shift by the maximum to avoid overflow
    max_c = np.max(conv)
    shifted = (conv - max_c) / temperature
    # stability: clip very negative values
    shifted = np.clip(shifted, -50, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max of the self‑convolution.
    For each i :  grad_i = 2 * Σ_k w_k * a_{k‑i}   (indices out of range → 0)
    """
    n = a.size
    # reverse w (needed for the cross‑correlation)
    w_rev = w[::-1]
    size = 1 << ((n + w_rev.size - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(fa * fw, n=size)[: n + w_rev.size - 1]   # length 3n‑2
    # grad_i = 2 * conv[(len(w)-1) - i]
    idx = w.size - 1 - np.arange(n)          # = 2n‑2‑i
    grad = 2.0 * conv[idx]
    return grad


def _score_from_conv(n: int, conv: np.ndarray) -> float:
    """Return the evaluation score (sum is 1 → 2*n*max)."""
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Parametrised families of starting points
# ----------------------------------------------------------------------


def _geom_shape(n: int, t: float) -> np.ndarray:
    """Geometric decay a_i = exp(-t*i)."""
    r = np.exp(-t)
    a = r ** np.arange(n, dtype=float)
    return a / np.sum(a)


def _power_shape(n: int, p: float) -> np.ndarray:
    """Power law a_i = (i+1)^(-p)."""
    a = (np.arange(1, n + 1, dtype=float) ** -p)
    return a / np.sum(a)


def _mixed_shape(n: int, t: float, p: float) -> np.ndarray:
    """Product of a geometric and a power‑law."""
    r = np.exp(-t)
    a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)
    return a / np.sum(a)


def _arcsine_shape(n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine density:
    w_i = 1/sqrt((i+1)*(n-i))  (finite at the ends).
    """
    i = np.arange(n, dtype=float)
    w = 1.0 / np.sqrt((i + 1.0) * (n - i))
    return w / np.sum(w)


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric linear increase then decrease (peak in the centre)."""
    half = (n + 1) // 2
    inc = np.arange(1, half + 1, dtype=float)
    dec = inc[::-1] if n % 2 == 1 else inc[:-1][::-1]
    a = np.concatenate([inc, dec])
    return a / np.sum(a)


# ----------------------------------------------------------------------
# Local optimiser – smooth‑max Adam on the simplex
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float, target: float) -> tuple:
    """
    Adam‑style projected optimisation starting from a0.
    Returns (best_a, best_score) seen before *deadline*.
    """
    a = a0.copy()
    n = a.size
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.8
    best_a = a.copy()
    conv = _full_conv(a)
    best_score = _score_from_conv(n, conv)

    it = 0
    while time.time() < deadline and best_score > target:
        it += 1
        # ----- compute convolution and soft‑max weights -----
        conv = _full_conv(a)
        max_c = np.max(conv)
        cur_score = 2.0 * n * max_c
        # exponential temperature (a small fraction of the current max)
        temperature = max_c * 0.04 + 1e-12
        w = _softmax_weights(conv, temperature)

        # ----- gradient from the weighted sum -----
        grad = _gradient_from_weights(a, w)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        a_candidate = _project_to_simplex(a - step * m_hat)
        # ----- evaluate new candidate -----
        conv_new = _full_conv(a_candidate)
        new_score = _score_from_conv(n, conv_new)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_candidate.copy()
            a = a_candidate
            # occasional progress report
            if it % 200 == 0:
                print(f"    it {it:4d}, new local best {best_score:.6f}")
        else:
            # keep the current a (the Adam step already moved it)
            a = a_candidate

        # early exit if we have already reached the target
        if best_score <= target:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function.

    The function explores a handful of analytic families, keeps the best
    one, and then refines it using a smoothed‑max Adam optimiser.
    The whole search respects the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a small safety margin

    best_score = float("inf")
    best_seq = None
    target_score = 1.503                      # stop when we reach the target

    # ------------------------------------------------------------------
    # 1) quick scan of parametrised families
    # ------------------------------------------------------------------
    family_functions = [
        ("geom", _geom_shape, {"t": (0.0004, 0.002)}),
        ("power", _power_shape, {"p": (0.4, 2.0)}),
        ("mixed", _mixed_shape, {"t": (0.0003, 0.001), "p": (0.5, 1.5)}),
        ("arcsine", _arcsine_shape, {}),
        ("triangular", _triangular_shape, {}),
        ("random", None, {}),  # plain Dirichlet draw
    ]

    while time.time() < deadline:
        # length sampled uniformly from a moderately wide interval
        n = np.random.randint(1800, 3500)

        # try each analytic family once per outer loop
        for name, func, params in family_functions:
            if func is None:   # pure random Dirichlet
                a = np.random.gamma(1.0, 1.0, size=n)
                a = a / a.sum()
            else:
                # draw parameters uniformly in the given ranges
                kwargs_f = {}
                for k, rng in params.items():
                    if isinstance(rng, tuple):
                        kwargs_f[k] = np.random.uniform(*rng)
                a = func(n, **kwargs_f)

            # small random permutation – does not change the multiset
            if np.random.rand() < 0.5:
                np.random.shuffle(a)

            # evaluate (sum is 1 → score = 2*n*max(conv))
            conv = _full_conv(a)
            score = _score_from_conv(n, conv)

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Initial candidate: n={n}, shape={name}, score={best_score:.6f}")

            if best_score <= target_score:
                break
        if best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # 2) local refinement of the best candidates seen so far
    # ------------------------------------------------------------------
    while time.time() < deadline and best_score > target_score:
        # a new restart – pick a length near the current best length
        n = best_seq.size if best_seq is not None else np.random.randint(1800, 3500)

        # randomise a start – either perturb the current best or fresh draw
        if np.random.rand() < 0.6 and best_seq is not None:
            a0 = best_seq.copy()
            # inject a modest random perturbation before optimisation
            pert = np.random.randn(n) * 0.002
            a0 = np.maximum(a0 + pert, 0.0)
            a0 = a0 / a0.sum()
        else:
            # fresh random Dirichlet draw
            a0 = np.random.gamma(1.0, 1.0, size=n)
            a0 = a0 / a0.sum()

        sub_deadline = min(deadline, time.time() + 12.0)   # ~12 s per refinement
        a_opt, score_opt = _adam_optimize(a0, sub_deadline, target_score)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"Global improvement: n={n}, refined score={best_score:.6f}")

        if best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing worked (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # return a plain Python list – evaluate_sequence will clip automatically
    return best_seq.tolist()
