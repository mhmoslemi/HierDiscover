def compute_score(a):
    n = len(a)
    m = 2*n - 1
    L = 1 << (int(np.ceil(np.log2(m))))
    # use real FFT, more efficient: A = np.fft.rfft(a, n=L)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:m]
    max_val = conv.max()
    return 2 * n * max_val

def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    lengths = list(range(2000, 5001, 500))  # e.g., [2000,2500,...]
    for n in lengths:
        if time.time() >= deadline: break
        a = arcsine_sequence(n)  # sum to 1
        cur_score = compute_score(a)
        print(f"Start length {n}, baseline score {cur_score:.6f}")
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # Local stochastic improvement
        # Parameters
        iters = 0
        max_iters = 200000  # will be limited by time
        # step size schedule maybe decreases exponentially
        while time.time() < deadline and iters < max_iters:
            iters += 1
            # choose step size
            # temperature = max(0.01, (deadline - time.time()) / budget_s)
            # step_scale = 0.1 * temperature  # relative to typical magnitude
            # Instead use decreasing schedule:
            step_scale = 0.5 / (np.sqrt(iters) + 1)  # gradually decreasing
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # Determine delta limited by a_i and step_scale
            max_delta = min(a[i], step_scale)
            delta = (np.random.rand() - 0.5) * 2 * max_delta
            if delta == 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # Ensure nonnegativity (should be)
            if a_new[i] < 0 or a_new[j] < 0:
                continue
            new_score = compute_score(a_new)
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"[improve] length={n} iter={iters} score={new_score:.6f}")
            else:
                # Accept with small probability to escape local minima
                # temperature schedule decays
                temp = max(0.001, (deadline - time.time()) / (budget_s))
                if np.random.rand() < np.exp(-(new_score - cur_score)/temp):
                    a = a_new
                    cur_score = new_score
        print(f"Finished length {n}, best local score {cur_score:.6f}")
    print(f"Overall best length={len(best_seq)} score={best_score:.6f}")
    return best_seq.tolist()

def simplex_proj(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step-function sequence using a hill‑climbing search anchored on the
classical arcsine ( (1−z)^{−½} ) construction.  The algorithm:
1.  Try several lengths (default 2000–5000) and keep the best arcsine baseline.
2.  For each length run a fast projected‑gradient / stochastic hill‑climb that
    transfers a small amount of mass from a relatively large coefficient to a
    smaller one, always keeping the vector non‑negative and ℓ₁‑normalised.
    The objective is 2·n·max(conv) (the same value returned by
    ``evaluate_sequence`` after the ℓ₁‑normalisation is fixed).
3.  The search is stopped when the time budget expires and the best sequence
    seen so far is returned.
"""

import time
import numpy as np

def arcsine_sequence(n: int) -> list[float]:
    """Return the normalised truncation of the series
    (1‑z)^{‑½} = Σ C_i z^i,  C_i = binom(2i,i)/4ⁱ,  i=0..n‑1.
    The sequence sums to 1.0 (up to floating‑point error)."""
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    # normalise
    seq = [x / total for x in seq]
    return seq

def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def score_fast(a: np.ndarray, L: int = None) -> float:
    """Return 2·n·max(a*a) for a vector a with ℓ₁‑sum = 1.
    The convolution is computed by an FFT of length L (chosen automatically
    if not supplied)."""
    n = a.shape[0]
    m = 2 * n - 1
    if L is None:
        L = _next_pow_two(m)
    # real FFT (only the positive frequencies are needed)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:m]
    return 2.0 * n * conv.max()

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    # Duchi et al., “Efficient Projections onto the L1‑Ball for
    # Learning in High Dimensions”, ICML 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.maximum(v, 0)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def hill_climb(a0: np.ndarray, deadline: float, max_iter: int = 200000) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climbing that preserves the ℓ₁‑norm.
    Small amounts of mass are moved from a large coefficient to a small one.
    The function returns the best vector found and its score."""
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow_two(2 * n - 1)
    cur_score = score_fast(a, L)
    best = a.copy()
    best_score = cur_score

    it = 0
    # initial step size as a fraction of the current average coefficient (≈1/n)
    step_frac = 0.5   # we may move up to 50 % of the source coefficient
    while time.time() < deadline and it < max_iter:
        it += 1
        # Choose a source index proportionally to the current magnitude,
        # so we tend to move mass away from large entries.
        src = np.random.choice(n, p=a / a.sum())
        # Choose a distinct destination uniformly at random.
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1
        # Amount to transfer – a random fraction of the source entry.
        max_delta = step_frac * a[src]
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        # Perform the trial move in‑place.
        a[src] -= delta
        a[dst] += delta

        new_score = score_fast(a, L)

        if new_score < cur_score:
            # Accept the improvement.
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[improve] n={n:5d} iter={it:6d} score={new_score:.6f}")
        else:
            # Simple simulated‑annealing acceptance based on a linearly decaying temperature.
            remaining = max(deadline - time.time(), 0.0)
            temperature = remaining / (deadline - (deadline - (max_iter * 0.0)))  # proportional to time left
            if temperature > 0 and np.random.rand() < np.exp(-(new_score - cur_score) / (temperature + 1e-12)):
                cur_score = new_score
            else:
                # Revert the move.
                a[src] += delta
                a[dst] -= delta
        # Occasionally shrink the step fraction to allow fine‑tuning.
        if it % 5000 == 0:
            step_frac *= 0.9
            step_frac = max(step_frac, 0.01)
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises ``evaluate_sequence``.
    The procedure explores several lengths, starts from the classic arcsine
    construction and then refines each candidate with a fast stochastic hill‑climb.
    The best sequence discovered before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # leave a tiny safety margin
    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1) Scan a few lengths and keep the best arcsine baseline.
    lengths = list(range(1800, 3401, 400))   # 1800, 2200, …, 3200
    for n in lengths:
        if time.time() >= deadline:
            break
        seq = np.array(arcsine_sequence(n), dtype=float)
        sc = score_fast(seq)                # fast proxy of evaluate_sequence
        print(f"[baseline] n={n:<4d} score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # 2) Refine each length with hill climbing as long as time permits.
    for n in lengths:
        if time.time() >= deadline:
            break
        # initialise from the arcsine sequence of this length
        a0 = np.array(arcsine_sequence(n), dtype=float)
        # run a local search; each call respects the global deadline.
        a_opt, sc_opt = hill_climb(a0, deadline)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[global‑improve] n={n} new best score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) If there is still time, perform a cheap random perturbation on the
    #    overall best vector (might help when the optimal length is not in
    #    the list above).
    while time.time() < deadline and best_seq is not None:
        n = len(best_seq)
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # a random transfer respecting non‑negativity
        delta = (np.random.rand() - 0.5) * 0.02 * best_seq[i]   # up to ±1 % of a_i
        cand = best_seq.copy()
        cand[i] = max(0.0, cand[i] + delta)
        cand[j] = max(0.0, cand[j] - delta)
        # renormalise to unit ℓ₁
        s = cand.sum()
        if s == 0:
            continue
        cand /= s
        sc = score_fast(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
            print(f"[refine] length={n:<4d} improved score={sc:.6f}")

    # ------------------------------------------------------------------
    print(f"Best sequence length = {len(best_seq)} with score = {best_score:.6f}")
    # Return a plain Python list as required by the evaluation harness.
    return best_seq.tolist()

"""
Propose a step‑function sequence that minimises the supplied
``evaluate_sequence`` function.

The algorithm works in three stages:

1. **Baseline scan** – a few lengths (1800, 2200, …, 3200) are tested
   using the classic arcsine construction (coefficients of (1‑z)^(-1/2)).
   The best of those baselines is kept.

2. **Local stochastic hill‑climbing** – for each length we start from the
   arcsine sequence and repeatedly move a small amount of mass from a
   relatively large coefficient to a smaller one.  After every move the
   convolution is evaluated (via an FFT, O(n log n)).  Moves that improve
   the objective are kept, otherwise they are rejected.  The step size is
   gradually reduced, allowing a coarse‑to‑fine optimisation.  The best
   sequence found across all lengths is retained.

3. **Final cheap refinement** – if time remains a tiny random perturbation
   is applied to the globally best vector.

The score that is optimised is
    2·n·max( a * a )
which equals the value returned by ``evaluate_sequence`` after the
ℓ₁‑normalisation (the normalisation cancels out).  The entire search respects
the 1000 s time budget and returns a plain Python list of non‑negative
heights.
"""

import time
import numpy as np


def arcsine_sequence(n: int) -> list[float]:
    """
    Generate the length‑n truncation of the power series of
    (1‑z)^(-1/2) = Σ C_i z^i,   C_i = binom(2i,i) / 4**i .

    The list is normalised to have ℓ₁‑norm 1.0.
    """

    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = [x / total for x in seq]
    return seq
