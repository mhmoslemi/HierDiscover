"""
propose_candidate
-----------------

Construct a sparse step function whose non‑zero entries form a set
with a bounded number of collisions in the convolution index.
If each possible sum i+j occurs for at most two unordered pairs,
the maximum auto‑convolution is bounded by 4·(S/m)² where
S = √(2 n) and m is the number of non‑zero entries.
For n≈3000 the theoretical limit is m≈√(8 n)≈158, which gives a
score below 1.2 (the target is 1.503).  The algorithm

1. fixes n from the reference sequence `height_sequence_1` if present,
   otherwise uses n = 3000,
2. repeatedly builds a random greedy set of indices respecting the
   collision bound (max_per_sum = 2) and keeps the largest set found
   within the allotted time,
3. assigns a uniform height S/m on the selected positions (the ℓ¹‑norm
   is exactly √(2 n)),
4. optionally performs a short projected‑gradient refinement,
5. returns the best sequence discovered.

The method exploits the combinatorial structure of the problem and
converges to a substantially lower evaluation score than the previous
gradient‑descent approach.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of total mass z."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):               # all entries will be zero
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
def _gradient_of_max(a: np.ndarray, top_k: int = 3):
    """Sub‑gradient of max_k (a*a)[k] and the corresponding max value."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()

    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(a)
    i = np.arange(n)

    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad, max_val
