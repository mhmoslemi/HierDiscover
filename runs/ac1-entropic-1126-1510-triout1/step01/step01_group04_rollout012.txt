def fast_score(seq):
    n = len(seq)
    a = np.asarray(seq, dtype=np.float64)
    if a.sum() == 0: return float('inf')
    # compute conv using FFT
    m = 1 << ((2*n-1).bit_length())
    # Actually use next power of 2.
    m = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]
    max_conv = conv.max()
    # ensure nonnegative
    if max_conv < 0: max_conv = 0.0
    score = 2 * n * max_conv / (a.sum() ** 2)
    return float(score)

        max_idx = int(conv.argmax())

"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Fast evaluation** – a lightweight implementation of the objective using
   FFT‑based convolution (exactly the same formula as `evaluate_sequence`
   but without the input validation/clipping).  This makes millions of
   score evaluations cheap.

2. **Parameter sweep** – generate a variety of candidate shapes
   * power‑law  a[i] ∝ (i+offset+1)^α
   * Beta‑like  a[i] ∝ (i+1)^{p‑1}·(n‑i)^{q‑1}
   * a few random Dirichlet vectors (including the provided `height_sequence_1`).

   The sweep uses the fast evaluator and keeps the best vector.

3. **Simulated‑annealing hill‑climb** – starting from the best candidate,
   repeatedly move a random fraction of mass from one index to another.
   Moves that improve the fast score are always accepted; worsening moves
   are accepted with probability exp(‑Δ/T) where the temperature T decays.
   This escapes local minima and explores a large neighbourhood.

4. **Projected gradient descent** – after annealing we perform a small
   number of exact sub‑gradient steps on the *max‑convolution* entry,
   projected onto the probability simplex.  This refines the solution.

5. **Final evaluation** – the candidate is handed to the official
   `evaluate_sequence` once (the only expensive call allowed) and returned.

All numbers are non‑negative, sum to 1, and the routine respects the
1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast (approximate) objective – identical to evaluate_sequence but
# without the security checks.  It is used for the vast majority of
# candidate evaluations.
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq))/sum(seq)**2 using FFT."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    # length of the full convolution = 2*n‑1 → next power‑of‑2 for FFT
    m = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(ft * ft, n=m)[: 2 * n - 1]
    max_conv = conv.max()
    if max_conv < 0.0:      # numerical noise
        max_conv = 0.0
    return float(2.0 * n * max_conv / (s * s))

# Shape factories
# ----------------------------------------------------------------------
def _make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """a[i] ∝ (i+offset+1)^exponent, normalised to sum‑1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


def _make_beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """
    Beta‑like shape: a[i] ∝ (i+1)^{p‑1}·(n‑i)^{q‑1}, normalised.
    p,q>0.  Equivalent to a discretised Beta(p,q) density on {1,…,n}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (p - 1.0)
    right = (n - i + 1.0) ** (q - 1.0)
    vec = left * right
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total

# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            cur_score: float,
            deadline: float,
            max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass around; accept worsening moves with decreasing temperature."""
    n = len(seq)
    cur = seq.copy()
    best = cur.copy()
    best_score = cur_score

    T = 0.1                     # initial temperature
    it = 0
    while time.time() < deadline and it < max_iter:

            print(f"[anneal] iter {it:,} – best≈{best_score:.6f} – T={T:.5f}")

        it += 1

    return best, best_score

# ----------------------------------------------------------------------
def _gradient_descent(seq: np.ndarray,
                     cur_score: float,
                     deadline: float,
                     max_iter: int = 5_000) -> tuple[np.ndarray, float]:
    """Refine the vector by following the sub‑gradient of the max‑convolution."""
    a = seq.copy()
    n = len(a)
    step = 0.12
    it = 0
    while time.time() < deadline and it < max_iter:
        # FFT‑based convolution
        m = 1 << ((2 * n - 1).bit_length())
