        max_conv = conv.max()

        alpha = 5.0 + 0.00005 * it  # gradually increase

"""
Optimiser for the step‑function rating.

The algorithm combines three ideas:

1️⃣  A **soft‑max weighted gradient** of the self‑convolution.
    For a vector a the convolution c = a*a is computed by FFT.
    A weight vector w (soft‑max of c) focuses the gradient on the
    largest convolution entries while keeping the direction smooth.
    The gradient of Σ w_k·c_k equals 2·(a * w_rev) where w_rev is w
    reversed; this is obtained efficiently with a second FFT.

2️⃣  An **Adam‑style projected gradient descent** on the simplex
    {a≥0, Σ a = √(2 n)}.  After each Adam step the vector is projected
    back onto the simplex using the classic O(n log n) Euclidean
    projection.

3️⃣  After the main optimisation a cheap **random‑block search** is
    performed (as in the previous baseline) to discover occasional
    better structured patterns.

The whole routine respects the 1000 s time budget, prints intermediate
scores and finally returns the best sequence found as a plain Python
list (non‑negative, ready for `evaluate_sequence`)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)          # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):               # all entries will be zero
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1    # +1 because of 0‑based indexing
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using FFT (real result, length 2·n‑1)."""
    n = a.shape[0]
    # size for FFT: next power of two >= 2·n‑1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.fft(a, N)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _gradient_weighted(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the gradient of the soft‑max weighted sum Σ w_k·(a*a)[k]
    where w_k ∝ exp(alpha·conv_k).  The analytical gradient is

        grad = 2 * (a * w_rev)

    with w_rev the reversed weight vector.
    """
    n = a.shape[0]
    # soft‑max (stable)
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))
    w /= w.sum()                     # normalise to a probability vector

    # reversed weight vector
    w_rev = w[::-1]

    # convolution a * w_rev (length 3·n‑2)
    rev_len = len(w_rev)
    N = 1 << ((n + rev_len - 1).bit_length())
    A = np.fft.fft(a, N)
    B = np.fft.fft(w_rev, N)
    conv_rev = np.fft.ifft(A * B).real

    # index that maps to the required shift
    L2 = 2 * n - 2
    grad = 2.0 * conv_rev[L2 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The method uses Adam‑projected gradient descent on a
    soft‑max‑weighted convolution objective plus a final random‑block
    refinement.  The best vector encountered is returned as a Python
    list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Length of the sequence ------------------------------------------------
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)           # fixed ℓ¹‑norm (scale‑invariant)

    # ------------------------------------------------------------------
    # Initialise a few simple candidates -----------------------------------
    def _normalise(v):
        return _project_to_simplex(v, target_sum)

    candidates = []

    # uniform
    candidates.append(np.full(n, target_sum / n))

    # triangular (piecewise linear, peak in the centre)
    t = np.arange(1, n // 2 + 2, dtype=float)
    if n % 2 == 0:
        tri = np.concatenate([t, t[-2::-1]])
    else:
        tri = np.concatenate([t, t[-1::-1]])
    candidates.append(_normalise(tri[:n]))

    # cosine‑shaped (non‑negative)
    x = np.arange(n, dtype=float)
    cos = np.cos(np.pi * (x - (n - 1) / 2) / (n - 1)) + 1.0   # lies in [0,2]
    candidates.append(_normalise(cos))

    # random
    candidates.append(_normalise(np.random.rand(n)))

    # if we have the previous construction, also try it
    if have_ref:

    best_a = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):

        score = conv.max()          # because Σa = √(2n) ⇒ score = max(conv)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
        print(f"[init {idx:02d}] score = {score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    # Base learning rate – roughly the size of one coordinate
    lr = 2.0 * target_sum / n          # about twice the average entry

    # Moment estimates
    m = np.zeros(n)

    # Gradient‑weight schedule
    alpha_start = 5.0

    # We’ll increase alpha slowly as the optimisation proceeds
    max_iters = 2_000_000      # hard cap; real stop is time‑based
    it = 0

    while time.time() < deadline and it < max_iters:
        # --------------------------------------------------------------
        # 1) current convolution and weighted gradient
        conv = _conv_self_fft(a)
        # schedule alpha (increase linearly, clamp)
        alpha = alpha_start + (alpha_max - alpha_start) * (it / max_iters)
        if alpha > alpha_max:

        grad = _gradient_weighted(a, conv, alpha)

        # --------------------------------------------------------------
        # 2) Adam update (minimise ⇒ move opposite to gradient)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 3) evaluate and keep the best
        conv_new = _conv_self_fft(a_new)
        score_new = conv_new.max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            print(f"[{it:08d}] improved score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 4) occasional random perturbation to escape plateaus
        if it % 4000 == 0:
            noise = np.random.randn(n) * target_sum * 1e-3
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new

    # ------------------------------------------------------------------
    # Final cheap random‑block search (often discovers sparse patterns)
    block_trials = 300
    for _ in range(block_trials):
        if time.time() >= deadline:
            break
        nb = np.random.randint(2, 8)                 # 2 … 7 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)

        cand = np.zeros(n, dtype=float)
        for blk, vblk in zip(blocks, vals):
            cand[blk] = vblk
        cand = _project_to_simplex(cand, target_sum)
        sc = _conv_self_fft(cand).max()
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved score = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
