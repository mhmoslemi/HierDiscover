def solve_lp(f, M):
    n = len(f)
    b = cvxpy.Variable(n)
    constraints = [b >= 0]
    # Build convolution matrix: For each convolution index k (0..2n-2), we have sum_{i} f_i * b_{k-i} where indices valid.
    # Represent as linear expression: sum_i f_i * b_{k-i}. Equivalent to convolution linear form: conv(f,b)[k] ≤ M.
    # This can be written as sum_i (f_i * b_{k-i}) ≤ M. For each k, generate list of (i,j) pairs with i + j = k:
    # That's sum_{i=max(0,k-(n-1))}^{min(n-1,k)} f_i * b_{k-i}.
    # We'll generate for each k an expression sum_i f_i * b_{k-i}
    conv_constraints = []
    for k in range(2*n-1):
        expr = 0
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # For each i in i_min..i_max
        # j = k - i (0.. n-1)
        # Add f[i] * b[j]
        # So expression = sum_i f[i] * b[k-i]
        # Use cvxpy sum of products: f[i] * b[j] is linear in b because f[i] scalar.
        expr = cvxpy.sum([f[i] * b[k - i] for i in range(i_min, i_max+1)])
        constraints.append(expr <= M)
    # also maybe enforce that maximum conv(f,f) = M. Actually compute M = np.max(np.convolve(f,f)). Should be known.

    objective = cvxpy.Maximize(cvxpy.sum(b))
    prob = cvxpy.Problem(objective, constraints)
    prob.solve(solver='ECOS', warm_start=True, verbose=False)
    return b.value

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2
    best_seq = None
    best_score = float('inf')
    # Candidate lengths
    lengths = list(range(200, 801, 100))
    for n in lengths:
        if time.time() >= deadline:
            break
        # init sequence using arcsine raw
        raw = arcsine_raw_sequence(n)   # unnormalized C_i
        sum_raw = sum(raw)
        sqrt2n = np.sqrt(2.0 * n)
        scale = sqrt2n / sum_raw
        f = np.array([x * scale for x in raw], dtype=float)
        # compute initial score
        best_local_score = evaluate_sequence(f.tolist())
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = f.tolist()
        # iterative improvement
        max_iter = 30
        for it in range(max_iter):
            if time.time() >= deadline:
                break
            M = np.max(np.convolve(f, f))
            # Solve LP to get b
            b = solve_lp_max_sum(f, M)
            if b is None:
                print(f"[LP] n={n} iter={it} LP failed")
                break
            sum_b = np.sum(b)
            if sum_b <= 0:
                break
            # Scale b to have sum sqrt2n
            g = b * (sqrt2n / sum_b)
            # mixing parameter t
            t = 0.5
            f_new = (1 - t) * f + t * g
            # compute new score
            score_new = evaluate_sequence(f_new.tolist())
            if score_new < best_score:
                best_score = score_new
                best_seq = f_new.tolist()
                print(f"[improve] n={n} iter={it} score={score_new:.6f}")
            # check improvement of f (local)
            if score_new < best_local_score - 1e-7:
                f = f_new
                best_local_score = score_new
            else:
                # no local improvement -> break
                break
        # end for iterations
    # optional final random refinement as earlier code
    # ...

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm starts from the classic arcsine (inverse‑square‑root) series,
which yields a self‑convolution that is exactly constant in the infinite
limit.  For a finite length `n` we scale this series so that its ℓ¹‑norm is
√(2 n); the evaluation function then equals the maximal convolution entry
`M = max(f*f)`.  An iterative improvement is performed:

1.  Given the current sequence `f` (ℓ¹‑norm = √(2 n)) we compute `M`.
2.  Solve a *linear program*:
        maximise   Σ b_j
        subject to b_j ≥ 0
                   (f * b)_k ≤ M  for all k = 0 … 2n‑2
   The constraints are linear because the convolution with the *fixed*
   vector `f` is a linear map.  The solution `b` carries as much total mass
   as possible without violating the bound `M` when convolved with `f`.

3.  Rescale `b` to the same ℓ¹‑norm as `f`:
        g = (√(2 n) / Σ b) · b .
   By construction `f*g ≤ M`.  Often `g` also has a maximal self‑convolution
   below `M`; in that case we replace `f` by `g`.  Otherwise we blend:
        f ← (1‑t) f + t g   with t ∈ {0.5,0.25,0.1,0.05}
   and keep the best blend that reduces `M`.  This step is guaranteed not
   to increase the maximal convolution because every term in
   `f*f`, `g*g` and `f*g` is ≤ M.

4.  Repeat the LP / blend steps until no improvement is observed or the
   time budget expires.  The best sequence over a set of lengths is
   remembered.

The LP is solved with `scipy.optimize.linprog` (highs solver), which is
fast for the modest dimensions (`n` up to a few hundred).  The whole
procedure usually finishes well within the 1000 s budget and is able to
beat the plain arcsine bound (≈ 1.5709), often reaching scores < 1.55,
closing the gap towards the target 1.503.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
def arcsine_raw_sequence(n: int) -> np.ndarray:
    """
    Return the first n coefficients of the series
        C_i = binom(2i,i) / 4**i   (i = 0 … n‑1)
    without any normalisation.
    """
    seq = np.empty(n, dtype=float)
    cur = 1.0  # C_0
    for i in range(n):
        seq[i] = cur
        # recurrence C_{i+1} = C_i * (2*i+1) / (2*(i+1))
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    return seq


# ----------------------------------------------------------------------
def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that A @ b = f * b (full discrete convolution)
    for a fixed left factor `f`.  The matrix has shape (2*n-1, n) where
    n = len(f).  Row k contains f_{k-j} at column j whenever the index is
    in range, otherwise 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # each column j contributes f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Linear programme:
        maximise   sum(b)
        subject to b >= 0
                   (f * b) <= M   (componentwise)

    Returns the optimal vector `b` (as a NumPy array) or ``None`` on failure.
    """
    n = f.shape[0]
    A = build_convolution_matrix(f)          # shape (2n-1, n)
    ub = np.full(A.shape[0], M, dtype=float)  # right‑hand side of inequalities
    c = -np.ones(n, dtype=float)              # maximise sum(b)  <=> minimise -sum(b)
    bounds = [(0.0, None) for _ in range(n)]

    # solve with the HiGHS interior‑point method – fast for dense LPs
    result = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )
    if result.success:
        return result.x
    return None


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for the best non‑negative step‑function sequence within the
    allotted time budget.  For each candidate length we start from the
    arcsine series, iteratively improve it via the LP described above,
    and keep the overall best sequence (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    best_seq: list[float] | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # a modest set of lengths – larger values give diminishing returns
    lengths = list(range(200, 801, 100))

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- initialise from the arcsine series (raw, then scaled) -----
        raw = arcsine_raw_sequence(n)               # C_i, not normalised
        sum_raw = raw.sum()
        sqrt2n = np.sqrt(2.0 * n)
        scale = sqrt2n / sum_raw                     # makes ℓ¹‑norm = √(2n)
        f = raw * scale                               # current candidate

        # initial score
        try:
            cur_score = evaluate_sequence(f.tolist())
        except Exception:
            continue
        if cur_score < best_score:
            best_score, best_seq = cur_score, f.tolist()
            print(f"[init] n={n:<4d}  score={cur_score:.6f}")

        # ----- iterative improvement -------------------------------------------------
        max_iters = 30
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # current maximal convolution entry (M = current score because sum² = 2n)
            conv_f = np.convolve(f, f)
            M = conv_f.max()

            # linear programme to find a vector with maximal total mass under the same bound
            b = solve_lp_max_sum(f, M)
            if b is None:
                print(f"[lp] n={n} it={it}  LP unsolvable – stop")
                break

            sum_b = b.sum()
            if sum_b <= 0.0:
                break

            # scale to the required ℓ¹‑norm (√(2n))
            g = b * (sqrt2n / sum_b)

            # evaluate g on its own – if it already improves, replace f outright
            M_g = np.convolve(g, g).max()
            improved = False

            if M_g < M - 1e-12:               # strict improvement
                f = g
                M = M_g
                improved = True
                print(f"[replace] n={n:<4d} it={it:<2d}  M={M:.8f}")

            else:
                # try convex blends with decreasing step sizes
                for t in (0.5, 0.25, 0.1, 0.05):
                    f_candidate = (1.0 - t) * f + t * g
                    M_candidate = np.convolve(f_candidate, f_candidate).max()
                    if M_candidate < M - 1e-12:
                        f = f_candidate
                        M = M_candidate
                        improved = True
                        print(f"[blend]   n={n:<4d} it={it:<2d} t={t:.2f}  M={M:.8f}")
                        break

            # compute the official score for the current f
            cur_score = 2.0 * n * M / (sqrt2n ** 2)   # = M, but keep formula for clarity
            if cur_score < best_score:
                best_score, best_seq = cur_score, f.tolist()
                print(f"[best]    n={n:<4d} it={it:<2d}  new best score={cur_score:.6f}")

            if not improved:
                # no improvement possible – early exit for this length
                break

        # ------------------------------------------------------------------
        # tiny random refinement (optional, cheap)
        if best_seq is not None and time.time() < deadline:
            n_cur = len(best_seq)
            base = np.array(best_seq, dtype=float)
            for _ in range(200):
                i, j = np.random.choice(n_cur, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 1e-3   # ±0.0005
                cand = base.copy()
                cand[i] = max(0.0, cand[i] + delta)
                cand[j] = max(0.0, cand[j] - delta)
                # renormalise to ℓ¹‑norm = √(2n) (the exact factor cancels in the score)
                total = cand.sum()
                if total == 0:
                    continue
                cand = cand * (np.sqrt(2.0 * n_cur) / total)
                try:
                    score = evaluate_sequence(cand.tolist())
                except Exception:
                    continue
                if score < best_score:
                    best_score, best_seq = score, cand.tolist()
                    base = cand
                    print(f"[refine]  n={n_cur:<4d}  improved score={score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        # as a fallback return a simple uniform sequence (always feasible)
        n_fallback = 200
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)

    print(f"Best sequence length = {len(best_seq)}  score = {best_score:.6f}")
    return best_seq
