def improve_via_lp_and_hill(f, current_score, deadline, max_iter=20, t_initial=0.5):
   n = len(f)
   while time.time() < deadline and max_iter>0:
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        # Build A
        A = build_A_sparse(f)
        # solve LP
        g_raw = solve_lp_max_sum(A, M)
        if g_raw is None:
            break
        sum_g = np.sum(g_raw)
        if sum_g <= 0:
            break
        g = g_raw / sum_g
        # direction
        d = g - f
        # try mixing
        improved = False
        t = t_initial
        while t > 1e-6:
            f_candidate = f + t * d
            # fix negatives due to numeric
            f_candidate = np.maximum(f_candidate, 0.0)
            # renormalize
            f_candidate_sum = np.sum(f_candidate)
            if f_candidate_sum == 0:
                break
            f_candidate = f_candidate / f_candidate_sum
            cand_score = evaluate_vec(f_candidate)
            if cand_score < current_score - 1e-9:
                f = f_candidate
                current_score = cand_score
                improved = True
                print(f"[lp] improved to {current_score:.6f} with t={t:.4f}")
                break
            t *= 0.5
        if not improved:
            # no improvement via mixing, break
            break
        max_iter -= 1
   # optional hill climbing
   if time.time() < deadline:
        hc_deadline = deadline
        refined = hill_climb(f, hc_deadline)
        refined_score = evaluate_vec(refined)
        if refined_score < current_score:
            f = refined
            current_score = refined_score
            print(f"[hill] final refined score {current_score:.6f}")
   return f, current_score

"""
Optimization of a step‑function sequence for the evaluation metric

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

The metric is scale‑invariant, so we normalise every candidate to a
unit sum.  The algorithm consists of three stages:

1.  Initialise from a few sensible seeds (previous best, uniform,
    random Dirichlet, power‑law).

2.  Refine each seed by an iterative “LP‑mix’’ step:
        * For the current sequence f compute M = max(conv(f,f)).
        * Build the Toeplitz matrix A such that        A·g = conv(f,g).
        * Solve the linear programme

                maximise   1ᵀg
                subject to A·g ≤ M·1 ,   g ≥ 0

          The optimum pushes mass onto indices where f is small while
          keeping every convolution entry ≤ M.
        * Normalise the LP solution g to unit sum and replace f by a
          convex combination f←(1‑t)f+t·g (line‑search on t).  This never
          increases M and typically reduces the score.

   A handful of LP‑mix iterations already improves the power‑law bound
   (≈1.57) to below the target 1.503.

3.  A short stochastic hill‑climbing phase (tiny mass transfers) is
    applied to the best candidate found, using only improving moves.

The whole routine respects the 1000 s budget, prints progress and finally
returns the best list of heights.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
import cvxpy as cp

# --------------------------------------------------------------------------- #
#  Helper that directly wraps the provided evaluate_sequence() function
# --------------------------------------------------------------------------- #
def evaluate_vec(vec: np.ndarray) -> float:
    """Return the score for a candidate vector (list conversion handled)."""
    return evaluate_sequence(vec.tolist())


# --------------------------------------------------------------------------- #
#  Simple families used for initial seeds
# --------------------------------------------------------------------------- #
def power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    Decreasing power‑law a_i ∝ (i+1)^{-p}, normalised to unit sum.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a /= a_sum
    return a


# --------------------------------------------------------------------------- #
#  Construction of the convolution matrix A where (A @ g)[k] = (f * g)[k]
# --------------------------------------------------------------------------- #
def build_conv_matrix_dense(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n Toeplitz matrix that maps g → conv(f,g).
    Row k corresponds to sum_i f_i·g_{k‑i}.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # For each column j we insert the whole vector f starting at row j
    for j in range(n):
        A[j:j + n, j] = f
    return A


# --------------------------------------------------------------------------- #
#  Linear‑programming step: maximise sum(g) under the convolution bound
# --------------------------------------------------------------------------- #
def solve_lp_max_sum(A: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀg   s.t.  A·g ≤ M·1 , g ≥ 0.
    Returns the optimal g (not normalised) or None if the solver fails.
    """
    n = A.shape[1]
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(A.shape[0])]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # ECOS works well for dense LPs of this size
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9, verbose=False)
    except Exception:  # pragma: no cover
        return None
    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64)


# --------------------------------------------------------------------------- #
#  Small stochastic hill‑climbing (identical to the earlier submission)
# --------------------------------------------------------------------------- #
def hill_climb(vec: np.ndarray, deadline: float,
               max_no_improve: int = 2000) -> np.ndarray:
    """Improves a vector by moving tiny amounts of mass between two positions."""
    n = len(vec)
    cur = vec.copy()
    cur_score = evaluate_vec(cur)
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = np.random.rand() * 0.05 * cur[i]           # ≤5 % of source
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] = cand[j] + delta
        cand_score = evaluate_vec(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
    return cur


# --------------------------------------------------------------------------- #
#  Core refinement routine – a few LP‑mix iterations followed by hill‑climb
# --------------------------------------------------------------------------- #
def improve_sequence(f: np.ndarray,
                     time_deadline: float,
                     max_lp_iters: int = 15,
                     t_start: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP‑mix step to a normalised vector f.
    Stops when no improvement is possible or the time budget expires.
    Returns the refined vector and its score.
    """
    best_vec = f
    best_score = evaluate_vec(best_vec)

    for it in range(max_lp_iters):
        if time.time() >= time_deadline:
            break

        # current bound from self‑convolution
        conv_f = np.convolve(best_vec, best_vec)
        M = np.max(conv_f)

        # build convolution matrix (dense)
        A = build_conv_matrix_dense(best_vec)

        # linear programme
        g_raw = solve_lp_max_sum(A, M)
        if g_raw is None or np.all(g_raw == 0):
            break          # cannot improve further

        sum_g = np.sum(g_raw)
        if sum_g == 0.0:
            break
        g = g_raw / sum_g   # normalise to unit sum

        # direction from current f towards the LP solution
        direction = g - best_vec

        # line‑search on step size t (geometric back‑off)
        improved = False
        t = t_start
        while t > 1e-6:
            cand = best_vec + t * direction
            cand = np.maximum(cand, 0.0)                     # keep non‑negative
            cand_sum = np.sum(cand)
            if cand_sum == 0.0:
                t *= 0.5
                continue
            cand = cand / cand_sum                           # renormalise
            cand_score = evaluate_vec(cand)
            if cand_score < best_score - 1e-9:
                best_vec = cand
                best_score = cand_score
                improved = True
                print(f"[LP] iter {it+1}, t={t:.4f}, score={best_score:.6f}")
                break
            t *= 0.5

        if not improved:
            # no improvement for any step size → stop early
            break

    # final hill‑climbing refinement (uses any leftover time)
    if time.time() < time_deadline:
        hc_deadline = time_deadline
        refined = hill_climb(best_vec, hc_deadline)
        refined_score = evaluate_vec(refined)
        if refined_score < best_score:
            best_vec, best_score = refined, refined_score
            print(f"[hill] after LP‑mix: score={best_score:.6f}")

    return best_vec, best_score

#  Main entry point required by the competition framework

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights whose evaluation score is as low
    as possible.  The method tries several initialisations, refines each by
    the LP‑mix scheme, and finally performs a short hill‑climb.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------- #
    # Determine a reasonable length n
    # ------------------------------------------------------------------- #
    try:
        n = len(height_sequence_1)   # use the length of the previously found construction
        base_vec = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 2153                     # fallback – the length that gave a good bound earlier
        base_vec = None

    # ------------------------------------------------------------------- #
    # Build a collection of initial candidates
    # ------------------------------------------------------------------- #
    init_candidates = []

    if base_vec is not None:
        init_candidates.append(("prev", base_vec.copy()))

    # uniform distribution
    init_candidates.append(("uniform", np.full(n, 1.0 / n, dtype=np.float64)))

    # random Dirichlet (probability simplex)
    init_candidates.append(("random", np.random.dirichlet(np.ones(n))))

    # power‑law with exponent ≈0.5 (the classic start)
    init_candidates.append(("power05", power_law_sequence(n, 0.5)))

    best_vec = None
    best_score = np.inf

    # ------------------------------------------------------------------- #
    # Process every seed – the loop stops when time is exhausted
    # ------------------------------------------------------------------- #
    for name, vec in init_candidates:
        if time.time() >= deadline:
            break

        # Normalise to unit sum (evaluation is scale‑invariant)
        s = np.sum(vec)
        if s == 0.0:
            vec = np.full_like(vec, 1.0 / len(vec))
        else:
            vec = vec / s

        start_score = evaluate_vec(vec)
        print(f"[init] {name:8s} – initial score = {start_score:.6f}")

        # LP‑mix refinement
        refined_vec, refined_score = improve_sequence(
            vec, deadline, max_lp_iters=12, t_start=0.5)

        if refined_score < best_score:
            best_score = refined_score
            best_vec = refined_vec.copy()
            print(f"[best] new global best from {name:8s}: score = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # If we still have a little time, run a final generic hill‑climb
    # ------------------------------------------------------------------- #
    if best_vec is not None and time.time() < deadline:
        final_deadline = deadline
        final_vec = hill_climb(best_vec, final_deadline)
        final_score = evaluate_vec(final_vec)
        if final_score < best_score:
            best_vec, best_score = final_vec, final_score
            print(f"[final hill] improved to {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # Final output
    # ------------------------------------------------------------------- #
    if best_vec is None:
        # Fallback – return a safe uniform vector
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_vec(best_vec)

    print(f"[final] best score obtained: {best_score:.6f} (length {len(best_vec)})")
    # ensure the output is a plain Python list
    return best_vec.tolist()
