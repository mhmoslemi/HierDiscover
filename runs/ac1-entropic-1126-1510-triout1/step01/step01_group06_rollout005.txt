Gradient‐descent search for a low‑value non‑negative step function.

Overview
--------
* The evaluation function is invariant under scaling, so we keep the
  sequence on the probability simplex (non‑negative, sum = 1).
* We minimise   score = 2·n·max(conv(a,a)).
* Instead of the raw max we use a smooth approximation
  log‑sum‑exp (soft‑max) with a growing temperature α.
  This yields a dense sub‑gradient that steers the whole
  convolution landscape towards flatness.
* An Adam optimiser (momentum + adaptive step) is applied,
  followed by Euclidean projection onto the simplex.
* After the smooth phase we switch to the exact max‑sub‑gradient
  (the same gradient used in the reference code) for fine tuning.
* Many random restarts with different lengths (2000–8000) and
  initial shapes (uniform, geometric, power‑law, mixed) are tried
  until the time budget (≈ 1000 s) expires.
* If a variable `height_sequence_1` is present (the best construction
  from previous runs) it is used as an additional warm‑start.

The code is pure NumPy, uses only FFTs for O(n log n) convolutions and

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) of the self‑convolution.
    Returns ∂/∂a  (log‑sum‑exp[α·conv]) .
    """
    n = a.size
    conv = _conv_full_fft(a)                     # length 2n‑1
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)              # avoid overflow
    w_sum = np.sum(w)
    w /= w_sum                                   # now Σ w = 1

    # gradient = 2 * (w ⋆ a)   where ⋆ is cross‑correlation.
    # Cross‑correlation = convolution of w with reversed a.
    a_rev = a[::-1]
    len_w = conv.shape[0]                        # = 2n‑1
    size = 1 << ((len_w + n - 1).bit_length())   # ≥ 3n‑2
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    corr_full = np.fft.irfft(fw * fa_rev, n=size)[: len_w + n - 1]  # 3n‑2
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]      # extract n entries
    return grad


def _max_gradient(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)).
    max_idx are the indices where the convolution attains its current maximum.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


def _conv_max_and_idx(a: np.ndarray):
    """Return (max value, array of indices attaining the max) of a*a."""
    conv = _conv_full_fft(a)
    max_val = np.max(conv)
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return float(max_val), idx.astype(int)


# ----------------------------------------------------------------------
# Local optimisation routine
# ----------------------------------------------------------------------


def _local_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam + soft‑max gradient + final exact max‑sub‑gradient.
    Returns the best sequence seen and its evaluation score.
    """
    a = a0.copy()
    best_a = a.copy()
    # initial score
    cur_max, _ = _conv_max_and_idx(a)
    best_score = 2.0 * n * cur_max

    # Adam parameters
    lr = 0.12
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline and it < 3500:
        it += 1

        # current convolution & score
        conv = _conv_full_fft(a)
        max_val = np.max(conv)
        cur_score = 2.0 * n * max_val

        # record best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 250 == 0:
                print(f"      iter {it}, new best {best_score:.6f}")

        # --- phase 1: smooth gradient (first 70% of iterations) ----------
        if it < 2500:
            alpha = 8.0 + 42.0 * (it / 2500.0)      # 8 → 50
            grad = _smooth_gradient(a, alpha)
        else:
            # --- phase 2: exact max sub‑gradient -------------------------
            _, max_idx = _conv_max_and_idx(a)
            grad = _max_gradient(a, max_idx)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = _project_to_simplex(a - step)

        # evaluate candidate
        cand_max = np.max(_conv_full_fft(a_candidate))
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            a = a_candidate
        else:
            # occasional tiny random shake to escape plateaus
            if np.random.rand() < 0.03:
                a_pert = a.copy()
                ids = np.random.randint(n, size=4)
                a_pert[ids] += np.random.randn(4) * 0.001
                a_pert = np.maximum(a_pert, 0.0)
                a_pert /= np.sum(a_pert)
                pert_max = np.max(_conv_full_fft(a_pert))
                pert_score = 2.0 * n * pert_max
                if pert_score < cur_score:
                    a = a_pert

    return best_a, best_score


# ----------------------------------------------------------------------
# Candidate generator (entry point)
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑evaluation step‑function.
    The function respects the 1000 s time budget and returns a plain
    Python list of non‑negative heights (sum is not required to be 1;
    the evaluator will normalise it automatically).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0          # a small safety margin

    best_score = float("inf")
    best_seq = None

    # optionally warm‑start from a previously found construction
    try:
        init_from_prev = np.array(height_sequence_1, dtype=float)
        init_from_prev = np.maximum(init_from_prev, 0.0)
        init_from_prev /= np.sum(init_from_prev)
        # try a few optimisation runs starting from this sequence
        for _ in range(2):
            n_prev = init_from_prev.size
            sub_dead = min(deadline, time.time() + 12.0)
            a_opt, s_opt = _local_optimize(init_from_prev, n_prev, sub_dead)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"Warm‑start improvement: n={n_prev}, score={best_score:.6f}")
            # jitter a little before the next attempt
            jitter = np.random.randn(n_prev) * 0.001
            init_from_prev = np.maximum(init_from_prev + jitter, 0.0)
            init_from_prev /= np.sum(init_from_prev)
    except NameError:
        # nothing to warm‑start from
        pass

    # ------------------------------------------------------------------
    # Main random‑restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # random length – larger n gives a lower theoretical bound
        n = np.random.randint(2000, 8000)

        # pick a random seed shape
        shape = np.random.choice(["uniform", "geom", "power", "mixed", "random"])

        if shape == "uniform":
            a = np.full(n, 1.0 / n)
        elif shape == "geom":
            # geometric decay a_i = (1‑r) r**i
            t = np.random.uniform(0.0005, 0.0025)   # t = -log r
            r = np.exp(-t)
            a = (1 - r) * (r ** np.arange(n, dtype=float))
        elif shape == "power":
            # power‑law decay a_i = (i+1)^‑p
            p = np.random.uniform(0.3, 2.2)
            a = (np.arange(1, n + 1, dtype=float) ** -p)
        elif shape == "mixed":
            t = np.random.uniform(0.0005, 0.0015)
            r = np.exp(-t)
            p = np.random.uniform(0.5, 1.7)
            a = (1 - r) * (r ** np.arange(n, dtype=float)) * ((np.arange(1, n + 1) ** -p))
        else:  # "random"
            a = np.random.gamma(1.0, 1.0, size=n)

        # small random permutation – does not affect the multiset
        if np.random.rand() < 0.5:
            np.random.shuffle(a)

        # normalise to the simplex (sum = 1)
        a = np.maximum(a, 0.0)
        if np.sum(a) == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= np.sum(a)

        # allocate at most 12 s for this start (or less if we are near deadline)
        sub_deadline = min(deadline, time.time() + 12.0)
        a_opt, s_opt = _local_optimize(a, n, sub_deadline)

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"Global improvement: n={n}, shape={shape}, score={best_score:.6f}")

        # early termination if we already beat the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * (1.0 / n)  # =2

    # Return a plain Python list – the external evaluator will clip/scale.
    return best_seq.tolist()
