def project_simplex(v):
    # returns projection onto simplex {x>=0, sum=1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    """Project v onto simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if len(rho) == 0:
        # all zero?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def compute_score_and_grad(a):
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf, np.zeros_like(a), None
    B = np.convolve(a, a)
    max_b = B.max()
    k = B.argmax()
    # compute a_rev
    a_rev = np.zeros_like(a)
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    # i from low to high inclusive
    i = np.arange(low, high + 1)
    a_rev[i] = a[k - i]
    grad_dir = a_rev - (max_b / s) * a
    score = 2.0 * n * max_b / (s * s)
    return score, grad_dir, (max_b, s, k)

def gradient_descent(a0, deadline, max_iter=100000):
    a = a0.copy()
    n = a.shape[0]
    best_score, grad_dir, _ = compute_score_and_grad(a)
    best_vec = a.copy()
    # step size parameters
    alpha = 0.1  # initial step magnitude
    min_alpha = 1e-8
    iter_count = 0
    while time.time() < deadline and iter_count < max_iter:
        # compute gradient and current score already available from previous iteration?
        # re-calc each loop
        score, grad_dir, _ = compute_score_and_grad(a)
        # compute direction
        norm = np.linalg.norm(grad_dir)
        if norm < 1e-12:
            break
        # attempt step with backtracking
        step = alpha
        improved = False
        while step > min_alpha:
            dir_normed = grad_dir / norm
            cand = a - step * dir_normed
            cand = simplex_projection(cand)
            # compute cand score
            cand_score, _, _ = compute_score_and_grad(cand)
            if cand_score < score:
                a = cand
                best_score = cand_score
                best_vec = cand.copy()
                improved = True
                # optionally increase alpha for next iteration
                alpha = step * 1.2
                break
            else:
                step *= 0.5
        if not improved:
            # reduce alpha
            alpha *= 0.5
            # if alpha too small, maybe break
            if alpha < min_alpha:
                break
        iter_count += 1
    return best_vec, best_score, iter_count

Now implement function solve_lp(f, M):

    n = len(f)
    # Build A matrix
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f
    # RHS vector b = M * ones
    b = np.full(2*n-1, M)
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # Bounds: (0, None)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g = res.x
    # Normalize sum to 1 (to keep scaling uniform)
    if g.sum() > 0:
        g = g / g.sum()
    return g

def gradient_descent_with_projection(a0, deadline, max_iter=100000, verbose=False):

    n = len(a0)
    a = a0.copy()
    best_score, _, _ = compute_score_and_grad(a)  # compute score directly.
    best_vec = a.copy()
    iter_counter = 0
    alpha = 0.1
    # ensure we have sum = 1, else project.
    a = simplex_projection(a)
    while time.time() < deadline and iter_counter < max_iter:
        score, grad_dir, _ = compute_score_and_grad(a)
        # skip if grad_dir norm zero
        norm = np.linalg.norm(grad_dir)
        if norm < 1e-12:
            break
        # try step with backtracking
        step = alpha
        improved = False
        # unit direction
        direction = grad_dir / norm
        while step > 1e-8:
            cand = a - step * direction
            cand = simplex_projection(cand)
            cand_score, _, _ = compute_score_and_grad(cand)
            if cand_score < score:
                a = cand
                score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_vec = cand.copy()
                # increase step for next iter
                alpha = step * 1.5
                break
            step *= 0.5
        if not improved:
            alpha *= 0.5
        iter_counter += 1
        # optionally break if alpha small
        if alpha < 1e-9:
            break
    return best_vec, best_score

"""
Propose a low‑value step sequence.
-------------------------------------------------
The evaluation  score = 2·n·max(conv(a,a))/sum(a)^2  is scale‑invariant.
We therefore work with probability vectors (sum == 1) and try to make the
auto‑convolution as flat as possible.

Algorithm
---------
1.  Initialise several candidate families:
    * power‑law   a_i ∝ (i+1)^(-p)            (1‑D golden‑section)
    * exponential a_i ∝ exp(-t·i)            (1‑D golden‑section)
   Take the best of the two families.

2.  Projected gradient descent on the simplex:
    * The gradient of the score w.r.t. a (with sum(a)=1) is
          g_i = a_rev_i – max_b·a_i,
      where a_rev_i = a_{k_max‑i} and k_max is the index of the
      maximal convolution entry.
    * Take a step opposite to the gradient, project onto the probability
      simplex and backtrack on the step size.
    * Repeat while time permits.

3.  Linear‑programming refinement (the “LP‐step” of the paper):
    * For the current sequence f compute M = max(conv(f,f)).
    * Solve   maximise   sum(g)   subject to   f ∗ g ≤ M   and   g ≥ 0.
      This is a standard LP (variables = g_i, constraints = 2·n‑1).
    * Normalise g to sum 1 and mix it with f:
          f_new = (1‑t)·f + t·g   (try t = 0.2,0.5,0.8).
      Keep the best mixture that reduces the score.
    * Iterate a few times (usually 1‑3 iterations are enough).

4.  A short stochastic hill‑climb (tiny mass moves) for final polishing.

All steps respect the 1000 s budget; the routine prints progress and finally
returns the best list it discovered.
"""

import time
import numpy as np
from scipy.optimize import linprog

# --------------------------------------------------------------------------- #
# Utility: projection onto the probability simplex {x≥0, sum(x)=1}
# --------------------------------------------------------------------------- #
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Families of candidates
# --------------------------------------------------------------------------- #
def power_law_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def exponential_sequence(n: int, t: float) -> np.ndarray:
    """a_i ∝ exp(-t·i), normalised to sum 1."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-t * idx)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


# --------------------------------------------------------------------------- #
# Score / gradient helpers (direct, no security checks)
# --------------------------------------------------------------------------- #
def score_and_gradient(a: np.ndarray):
    """Return (score, grad_dir, (max_b, k_max)).  a is assumed non‑negative."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf, np.zeros_like(a), (0.0, None)
    # convolution
    conv = np.convolve(a, a)
    max_b = conv.max()
    k = conv.argmax()
    # a_rev[i] = a[k-i] when the index is inside the array, else 0
    a_rev = np.zeros_like(a)
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    i = np.arange(low, high + 1)
    a_rev[i] = a[k - i]
    # gradient of the *score* (ignoring the constant 4n/s² factor)
    grad_dir = a_rev - max_b * a          # because s == 1 after normalisation
    score = 2.0 * n * max_b / (s * s)
    return score, grad_dir, (max_b, k)


# --------------------------------------------------------------------------- #
# 1‑D golden‑section search (generic)
# --------------------------------------------------------------------------- #
def golden_section_search(func, a: float, b: float, max_iters: int,
                         deadline: float):
    """Return (best_x, best_val, best_vec) for a scalar‑parameter search."""
    phi = (np.sqrt(5) - 1) / 2            # 0.618...
    x1 = a + (1 - phi) * (b - a)
    x2 = a + phi * (b - a)
    v1, vec1 = func(x1)
    v2, vec2 = func(x2)

    best_val = min(v1, v2)
    best_x = x1 if v1 <= v2 else x2
    best_vec = vec1 if v1 <= v2 else vec2

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        if v1 < v2:
            b = x2
            x2 = x1
            v2 = v1
            x1 = a + (1 - phi) * (b - a)
            v1, vec1 = func(x1)
            if v1 < best_val:
                best_val, best_x, best_vec = v1, x1, vec1
        else:
            a = x1
            x1 = x2
            v1 = v2
            x2 = a + phi * (b - a)
            v2, vec2 = func(x2)
            if v2 < best_val:
                best_val, best_x, best_vec = v2, x2, vec2
    return best_x, best_val, best_vec


# --------------------------------------------------------------------------- #
# Gradient descent on the simplex
# --------------------------------------------------------------------------- #
def gradient_descent(start_vec: np.ndarray, deadline: float,
                     max_iter: int = 200000) -> tuple[np.ndarray, float]:
    """Projected gradient descent (with backtracking) on the probability simplex."""
    a = simplex_projection(start_vec.copy())
    n = a.shape[0]
    score, _, _ = score_and_gradient(a)
    best_score, best_vec = score, a.copy()

    # initial step magnitude (in Euclidean norm of the *direction* vector)
    step_size = 0.1
    min_step = 1e-9
    it = 0

    while time.time() < deadline and it < max_iter:
        cur_score, grad_dir, _ = score_and_gradient(a)
        norm = np.linalg.norm(grad_dir)
        if norm < 1e-12:
            break

        direction = grad_dir / norm
        # backtrack line search (only decreasing steps are accepted)
        step = step_size
        improved = False
        while step > min_step:
            cand = a - step * direction
            cand = simplex_projection(cand)
            cand_score, _, _ = score_and_gradient(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                # allow a slightly larger step next iteration
                step_size = step * 1.3
                break
            step *= 0.5

        if not improved:
            step_size *= 0.5
            if step_size < min_step:
                break

        if cur_score < best_score:
            best_score, best_vec = cur_score, a.copy()

        it += 1

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Linear‑programming refinement (the “LP step”)
# --------------------------------------------------------------------------- #
def lp_refine(start_vec: np.ndarray, deadline: float,
              max_iters: int = 5) -> tuple[np.ndarray, float]:
    """Iteratively solve LP max∑g s.t. f∗g ≤ M and mix the result."""
    f = simplex_projection(start_vec.copy())
    best_score, _, _ = score_and_gradient(f)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # current bound M
        conv = np.convolve(f, f)
        M = conv.max()
        n = f.shape[0]

        # Build constraint matrix A (size (2n‑1) × n)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f

        # Linear program: maximise sum(g)  ⇔ minimise –sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs')
        except Exception as e:
            print("[lp] linprog exception:", e)
            break

        if not res.success:
            print("[lp] linprog failed:", res.message)
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()                       # renormalise to sum 1

        # Try a few mixing parameters and keep the best improvement
        improved = False
        for t in (0.2, 0.5, 0.8):
            cand = (1.0 - t) * f + t * g
            cand = simplex_projection(cand)    # safety
            cand_score, _, _ = score_and_gradient(cand)
            if cand_score < best_score - 1e-12:
                f = cand
                best_score = cand_score
                improved = True
                print(f"[lp] iter {it}, mix t={t:.2f} → score {best_score:.6f}")

        if not improved:
            # No improvement → stop early
            break

    return f, best_score


# --------------------------------------------------------------------------- #
# Simple stochastic hill‑climb (tiny mass moves)
# --------------------------------------------------------------------------- #
def hill_climb(start_vec: np.ndarray, deadline: float,
               max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """Randomly relocate a small fraction of mass; keep only improving moves."""
    a = simplex_projection(start_vec.copy())
    cur_score, _, _ = score_and_gradient(a)
    no_improve = 0
    n = a.shape[0]

    while time.time() < deadline and no_improve < max_no_improve:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 5 % of the source entry
        delta = np.random.rand() * 0.05 * a[i]
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] = cand[j] + delta
        cand = simplex_projection(cand)    # keep exact sum 1
        cand_score, _, _ = score_and_gradient(cand)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

    return a, cur_score

    """
    np.random.seed(seed)

    # --------------------------------------------------- #

    try:

    # --------------------------------------------------- #

    # --------------------------------------------------- #
    def eval_power(p):

    p_left, p_right = 0.05, 0.95

    # Convert to plain Python list (evaluate_sequence will clamp any stray values)
