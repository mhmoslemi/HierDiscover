def lp_optimize(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    conv = cp.conv(cp.Constant(f), g) # returns length 2n-1 vector
    constraints = [conv <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    return g.value

def _subgradient_refine(a, deadline, n, best_score, best_seq):
    a_curr = a.copy()
    best_seq_local = best_seq.copy()
    best_score_local = best_score
    eta0 = 0.5
    max_iters = 5000
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        b = np.convolve(a_curr, a_curr)
        k_max = int(np.argmax(b))
        cur_score = 2.0 * n * b[k_max]  # same as evaluate_sequence
        if cur_score < best_score_local:
            best_score_local = cur_score
            best_seq_local = a_curr.copy()
            print(f"[grad] iter={it} score={best_score_local:.6f}")
        # gradient
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a_curr)
        g[mask] = 2 * a_curr[idx[mask]]
        eta = eta0 / np.sqrt(it)
        a_curr = a_curr - eta*g
        # projection onto simplex
        a_curr = _project_onto_simplex(a_curr, s=1.0)
        # jitter
        if it % 200 == 0 and np.random.rand() < 0.2:
            a_curr += np.random.randn(n)*1e-4
            a_curr = _project_onto_simplex(a_curr, s=1.0)
    return best_seq_local, best_score_local

"""
Propose a low‑score step‑function.

The algorithm has three stages:

1️⃣  Initialise with the discrete arcsine law (the known near‑optimal shape).

2️⃣  Smooth‑max projected gradient descent.
    Instead of moving only on the single active convolution entry,
    we minimise a smooth approximation of the max:
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k) .
    Its gradient is a weighted sum of all shifted copies of ``a``.
    Using FFT‑based convolutions we take a step opposite the gradient
    and project back onto the probability simplex.  This quickly
    flattens the autocorrelation and reduces the max‑ratio.

3️⃣  Linear‑programming refinement.
    For the current vector ``f`` we compute M = max(f*f).
    The LP   max Σ g_i   s.t.   g_i ≥ 0  and  (f*g)_k ≤ M  (∀k)
    finds a direction that can increase the ℓ¹‑mass without breaking
    the current bound.  Mixing the solution with ``f`` (small step
    size) strictly reduces the max‑convolution.  A few LP iterations
    give a noticeable extra improvement.

Finally a short projected sub‑gradient descent (as in the previous
baseline) polishes the result.

All steps respect the total 1000 s budget and stay within the
allowed libraries (numpy, cvxpy, time, math).  The returned list
contains non‑negative heights and is ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=s}
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine law on {0,…,n‑1} (ℓ¹‑normalised)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# LP sub‑routine: maximise Σ g_i subject to (f*g)_k ≤ M, g ≥ 0.
def _lp_max_mass(f: np.ndarray, M: float):
    """Return a vector g (ℓ¹‑normalised) solving the LP, or None on failure."""
    try:
        import cvxpy as cp
    except Exception as exc:
        print("cvxpy not available – skipping LP stage.")
        return None

    n = len(f)
    g = cp.Variable(n, nonneg=True)

    # Convolution f * g (full length 2n‑1) – linear because f is constant.
    conv_fg = cp.conv(cp.Constant(f), g)

    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception as exc:
        print("LP solve error:", exc)
        return None

    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64)


# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(a: np.ndarray,
                                 deadline: float,
                                 beta: float = 20000.0,
                                 step0: float = 0.3,
                                 max_iters: int = 5000):
    """Smooth‑max descent on the autocorrelation max‑ratio."""
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = np.convolve(a, a)
        max_conv = conv.max()

        # smooth weights w_k = exp(beta·(conv_k – max_conv))   (max weight = 1)
        w = np.exp(beta * (conv - max_conv))

        # gradient = 2 * (w * a_rev)_{n-1 … n-1+n-1}
        a_rev = a[::-1]
        grad_full = np.convolve(w, a_rev)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        # occasional evaluation
        if it % 100 == 0:
            cur_score = 2.0 * n * np.convolve(a, a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[smooth‑GD] it={it} score={best_score:.6f}")

        # small random jitter to avoid stagnation
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_score


# ----------------------------------------------------------------------
def _lp_refinement(a: np.ndarray,
                   deadline: float,
                   max_lp_iters: int = 12,
                   mix_factor: float = 0.2):
    """Iteratively improve ``a`` using the LP described in the prompt."""
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()

    for lp_it in range(max_lp_iters):
        if time.time() >= deadline:
            break

        conv = np.convolve(best_a, best_a)
        M = conv.max()

        g = _lp_max_mass(best_a, M)
        if g is None:
            print("LP gave no solution – aborting LP stage.")
            break

        # normalise to unit ℓ¹‑mass
        g = np.maximum(g, 0.0)
        s = g.sum()
        if s == 0:
            print("LP result is all zero – aborting LP stage.")
            break
        g /= s

        # try a few mixing coefficients, keep the best reduction
        improved = False
        t = mix_factor
        while t > 1e-4:
            cand = (1 - t) * best_a + t * g
            cand_score = 2.0 * n * np.convolve(cand, cand).max()
            if cand_score < best_score:
                best_a = cand
                best_score = cand_score
                improved = True
                print(f"[LP] iter={lp_it} t={t:.4f} score={best_score:.6f}")
                break
            t *= 0.5
        if not improved:
            # No progress – stop early
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_refine(a: np.ndarray,
                        deadline: float,
                        eta0: float = 0.5,
                        max_iters: int = 3000):
    """Classic projected sub‑gradient descent on the max‑convolution entry."""
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[k_max]

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[subgrad] it={it} score={best_score:.6f}")

        # sub‑gradient of the active constraint (kernel centred at k_max)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_onto_simplex(a, s=1.0)

        # occasional random jitter
        if it % 250 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_score
