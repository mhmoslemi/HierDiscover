def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time

    start_time = time.time()
    deadline = start_time + budget_s - 2

    # parameters to try
    best_obj = float('inf')
    best_seq = None
    # Choose a set of lengths and blocks
    lengths = [200, 400, 600, 800, 1000]
    exponents = [0.5, 0.55]   # may try a bit higher exponent
    B_range = range(2, 11)   # number of blocks
    for L in lengths:
        idx = np.arange(1, L+1, dtype=np.float64)
        for p in exponents:
            block = idx ** -p   # power law
            # compute max convolution of block
            conv_block = np.convolve(block, block)
            M_block = float(conv_block.max())
            S_block = float(np.sum(block))
            for B in B_range:
                # spacing gap = max(2*L, int(2*L)) ensures separation
                gap = 2 * L
                n_total = B * L + (B - 1) * gap
                # total sum
                S_total = B * S_block
                # total max
                M_total = M_block   # due to separation
                obj = 2 * n_total * M_total / (S_total ** 2)
                if obj < best_obj:
                    best_obj = obj
                    # construct the full sequence
                    seq = []
                    for b in range(B):
                        seq.extend(block.tolist())
                        if b != B - 1:
                            seq.extend([0.0] * gap)
                    best_seq = seq
                # break early if time near deadline
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    print(f"Best objective = {best_obj:.6f}, length = {len(best_seq)}")
    return best_seq

"""
Propose a low‑evaluation step‑function using the classic “LP‑improvement”
procedure.

The algorithm starts from a simple power‑law (a_i = (i+1)^‑0.5) which already
gives the asymptotic value π/2 ≈ 1.571.  At each iteration we solve a linear
program

    maximise   Σ_j g_j
    subject to   g_j ≥ 0  and   (f * g)_k ≤ M   for all k

where * denotes convolution, f is the current sequence and M = max(f*f).
The LP finds a feasible sequence g whose ℓ₁‑norm can be larger than that of f
while keeping the same convolution bound.  By mixing f and the rescaled g we
decrease the maximum convolution value without sacrificing the total sum.
Repeating the step several times monotonically reduces the evaluation
objective

    2·n·max(f*f) / (sum(f)²).

The method uses only NumPy and SciPy (the ``linprog`` routine with the
high‑performance “highs” backend).  A modest size (n≈2000) is chosen so that
the LP stays fast while still allowing a significant improvement.  The search
stops when the time budget is exhausted or when no further improvement is
observed.

The returned sequence is a Python list of non‑negative floats ready to be
passed to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ g) = np.convolve(a, g)
    for any non‑negative vector g of length n.

    Parameters
    ----------
    a : np.ndarray
        Current sequence (length n).

    Returns
    """
    n = a.shape[0]
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)

    # column j contributes a[i] to rows i+j, i = 0..n‑1
    for j in range(n):
        A[j : j + n, j] = a
    return A


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Return a step‑function (list of non‑negative heights) whose
    ``evaluate_sequence`` value is as low as possible.

    Parameters
    ----------
    seed : int, optional
        Random seed (kept for reproducibility, not used by the deterministic part).
    budget_s : int, optional
        Time budget in seconds.  The function must return before the budget expires.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5  # a small safety margin

    # --------------------------------------------------------------
    # Initial sequence – a simple power‑law with exponent 0.5.
    # --------------------------------------------------------------
    n = 2000                     # length of the optimisation vector
    idx = np.arange(1, n + 1)    # i = 1 … n
    a = idx ** -0.5              # a_i = (i+1)^‑0.5

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, objective={best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop (LP + mixing)
    # --------------------------------------------------------------
    max_iters = 30               # safety upper bound
    t = 0.2                      # initial mixing weight
    no_improve = 0               # counter for stagnation

    for it in range(max_iters):
        # stop if we are about to exceed the time budget
        if time.time() > deadline:
            print("[time] budget exhausted, terminating early.")
            break

        # Current max self‑convolution value M
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()

        # Build the linear‑program matrix for the constraint (a * g) ≤ M
        A = _build_convolution_matrix(a)

        # Right‑hand side = M for every convolution entry
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        # Linear program: maximise Σ g_j  <=> minimise –Σ g_j
        c = -np.ones(n, dtype=np.float64)   # minimisation of -sum(g)
        bounds = [(0.0, None)] * n           # g_j ≥ 0

        # Solve the LP (highs is fast and robust for dense problems)
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0, deadline - time.time())},
        )

        if not res.success:
            print(f"[iter {it}] LP failed (status: {res.message}).")
            break

        g = res.x
        sum_g = g.sum()
        sum_a = a.sum()

        # Rescale g so that its ℓ₁‑norm matches that of a.
        # This keeps the mixing step simple and prevents the total sum from
        # drifting uncontrollably.
        if sum_g == 0:
            print(f"[iter {it}] LP returned the zero vector, aborting.")
            break
        scale = sum_a / sum_g
        g_scaled = g * scale

        # Convex combination of the old and the new sequence.
        a_new = (1.0 - t) * a + t * g_scaled

        # Evaluate the new candidate.
        score = evaluate_sequence(a_new.tolist())
        print(f"[iter {it}] t={t:.2f}, M={M:.6f}, sum={sum_a:.6f}, "
              f"objective={score:.6f}")

        # Accept the new sequence if it improves the objective.
        if score < best_score - 1e-9:          # a tiny tolerance to avoid noise
            best_score = score
            best_seq = a_new.copy()
            a = a_new
            no_improve = 0
            # gradually reduce the mixing step to fine‑tune the solution
            t = max(0.05, t * 0.9)
        else:
            # No improvement – shrink the step size; after several failures we stop.
            t = max(0.05, t * 0.5)
            no_improve += 1
            if no_improve >= 5:
                print("[stagnation] no improvement for several iterations, stopping.")
                break

    # -----------------------------------------------------------------
    # Final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best objective = {final_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
