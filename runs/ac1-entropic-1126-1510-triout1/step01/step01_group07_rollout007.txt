   exponential taper is applied.  The parameters p, α, γ, δ are
   explored on a modest grid (≈500 candidates) on a short vector
   (n≈2000) using a fast FFT‑based convolution to evaluate the
   objective.

2. **Fine local refinement** –  
   Around the best parameters we perform a denser grid search to
   improve the result.

3. **Gradient‑descent polishing** –  
   Starting from the best sequence (scaled to length 4000) we run a
   simple projected gradient descent that repeatedly reduces the
   maximum entry of the self‑convolution while re‑normalising the
   sum (the objective is homogeneous, so the scale does not matter).
   The gradient for the active convolution index k* is
   `∂(f*f)[k*]/∂a_i = 2·a_{k*‑i}` (or `a_i` on the diagonal).  After a
   step we clip to [0,1000] and renormalise.

All computation respects the security checks performed by
`evaluate_sequence`.  The function returns a Python list of floats
ready to be handed to the evaluator.  The total running time never
exceeds the supplied budget (default 1000 s).
"""

import time
import numpy as np

def _fast_objective(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    n = seq.shape[0]
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    L = 2 * n - 1
    # Use real FFT for speed
    fa = np.fft.rfft(seq, L)
    conv = np.fft.irfft(fa * fa, L)
    m = np.max(conv)
    return 2.0 * n * m / (s * s)


def _make_sequence(n: int, p: float, alpha: float,
                   gamma: float, delta: float) -> np.ndarray:
    """Build a candidate sequence of length n from the four parameters."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1.0) ** (-p)

    if alpha <= 0.0:                      # pure exponential taper
        shift = idx / (n - 1.0)
    else:
        cut = int(alpha * n)
        if cut >= n - 1:
            shift = np.zeros_like(idx)
        else:
            shift = np.zeros_like(idx)
            mask = idx >= cut
            denom = n - 1.0 - cut
            shift[mask] = (idx[mask] - cut) / denom

    taper = np.exp(-gamma * np.power(shift, delta))
    return base * taper


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # ------------------------------------------------------------------
    # 1) Coarse grid search on a small vector (fast evaluation)
    # ------------------------------------------------------------------
    n_search = 2000
    idx = np.arange(n_search, dtype=np.float64)

    best_score = np.inf
    best_params = None
    best_seq   = None

    # parameter grids
    p_vals      = np.linspace(0.35, 0.65, 13)      # 0.025 step
    alpha_vals  = np.linspace(0.0, 0.8, 9)        # fraction of length for cut‑off
    gamma_vals  = np.linspace(0.0, 6.0, 13)       # 0.5 step
    delta_vals  = [1.0, 1.5, 2.0, 2.5, 3.0]       # a few exponents

    for p in p_vals:
        base = (idx + 1.0) ** (-p)                 # reuse for speed
        for alpha in alpha_vals:
            for gamma in gamma_vals:
                for delta in delta_vals:
                    # build taper
                    if alpha <= 0.0:
                        shift = idx / (n_search - 1.0)
                    else:
                        cut = int(alpha * n_search)
                        if cut >= n_search - 1:
                            shift = np.zeros_like(idx)
                        else:
                            shift = np.zeros_like(idx)
                            mask = idx >= cut
                            denom = n_search - 1.0 - cut
                            shift[mask] = (idx[mask] - cut) / denom
                    taper = np.exp(-gamma * np.power(shift, delta))
                    seq = base * taper
                    obj = _fast_objective(seq)
                    if obj < best_score:
                        best_score = obj
                        best_params = (p, alpha, gamma, delta)
                        best_seq = seq.copy()
                    if time.time() > deadline:
                        break
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    if best_params is None:
        # fallback to a simple power‑law if something went wrong
        best_params = (0.5, 0.0, 0.0, 1.0)
        best_seq = (np.arange(n_search) + 1) ** -0.5

    print(f"[Coarse] best obj={best_score:.6f}  params={best_params}")

    # ------------------------------------------------------------------
    # 2) Fine local grid around the best parameters (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        p0, a0, g0, d0 = best_params
        p_range = np.linspace(max(0.2, p0 - 0.05), min(0.9, p0 + 0.05), 11)
        a_range = np.linspace(max(0.0, a0 - 0.2), min(0.9, a0 + 0.2), 11)
        g_range = np.linspace(max(0.0, g0 - 1.0), g0 + 1.0, 11)
        d_range = np.linspace(max(0.2, d0 - 0.5), d0 + 0.5, 11)

        for p in p_range:
            base = (idx + 1.0) ** (-p)
            for alpha in a_range:
                for gamma in g_range:
                    for delta in d_range:
                        # build taper (same logic as above)
                        if alpha <= 0.0:
                            shift = idx / (n_search - 1.0)
                        else:
                            cut = int(alpha * n_search)
                            if cut >= n_search - 1:
                                shift = np.zeros_like(idx)
                            else:
                                shift = np.zeros_like(idx)
                                mask = idx >= cut
                                denom = n_search - 1.0 - cut
                                shift[mask] = (idx[mask] - cut) / denom
                        taper = np.exp(-gamma * np.power(shift, delta))
                        seq = base * taper
                        obj = _fast_objective(seq)
                        if obj < best_score:
                            best_score = obj
                            best_params = (p, alpha, gamma, delta)
                            best_seq = seq.copy()
                        if time.time() > deadline:
                            break
                    if time.time() > deadline:
                        break
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

        print(f"[Fine] best obj={best_score:.6f}  params={best_params}")

    # ------------------------------------------------------------------
    # 3) Gradient‑descent polishing on the final length (≈4000)
    # ------------------------------------------------------------------
    final_n = 4000
    p, alpha, gamma, delta = best_params

    # build the final‑length start vector
    a = _make_sequence(final_n, p, alpha, gamma, delta)

    # normalise (scale does not affect the objective)
    a = a / np.sum(a)

    # initial objective for logging
    best_obj = _fast_objective(a)
    print(f"[GD start] obj={best_obj:.6f}")

    step = 0.02          # learning rate
    it = 0
    max_iters = 2000

    while time.time() < deadline - 0.5 and it < max_iters:
        n = final_n
        L = 2 * n - 1
        fa = np.fft.rfft(a, L)
        conv = np.fft.irfft(fa * fa, L)
        k_star = int(np.argmax(conv))

        # gradient of the active convolution entry
        i = np.arange(n)
        j = k_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        diag = np.where(mask & (i == j))[0]
        if diag.size:
            grad[diag] = a[i[diag]]

        # gradient step with projection
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        s = np.sum(a_new)
        if s < 1e-12:
            break
        a_new = a_new / s

        obj_new = _fast_objective(a_new)

        if obj_new < best_obj:
            a = a_new
            best_obj = obj_new
            step = min(step * 1.1, 0.2)   # slowly increase step
            print(f"[GD {it:04d}] obj={best_obj:.6f} step={step:.4f}")
        else:
            step *= 0.5                    # shrink step on non‑improvement
        it += 1

    # ------------------------------------------------------------------
    # Convert to plain Python list and run the official evaluator once
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    # ensure clean numeric values
    final_seq = [float(x) for x in final_seq if np.isfinite(x)]

    # final official score (for our own information)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] official score = {final_score:.6f}")
    except Exception as exc:
        print("[Result] evaluation failed:", exc)

    return final_seq
