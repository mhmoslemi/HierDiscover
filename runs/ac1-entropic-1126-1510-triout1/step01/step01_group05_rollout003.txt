def project_simplex(v):
    # ensures non-negative and sum to 1
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose candidate for the step‑function challenge.

Algorithm
---------
1. Pick a few useful lengths (usually the length of the known
   construction `height_sequence_1` and a couple of larger lengths).
2. Build a small portfolio of initial non‑negative sequences for each
   length:
      – power‑law families a_i ∝ (i+1)^(‑p) with several p,
      – exponential decays a_i ∝ r^i,
      – a linear decreasing profile,
      – the uniform vector,
      – a few random Dirichlet draws,
      – the previous best construction (if its length matches).
3. For every start vector run a *smooth‑max* projected gradient descent.
   We minimise the soft‑max of the convolution values
        fβ(a)= (1/β) log Σ_k exp(β·(a*a)[k])
   whose gradient can be written as a single cross‑correlation,
   so each iteration costs only a couple of O(n) convolutions.
   After each gradient step the iterate is projected onto the simplex
   (non‑negative entries with total mass = 1).
   The temperature β is a modest constant (10–30) which makes the
   optimisation focus on the current maximal convolution entry.
4. When the gradient phase terminates, perform a short *targeted*
   hill‑climbing pass that moves a tiny amount of mass from indices
   that contribute most to the current maximal convolution towards
   indices that do not affect it.
5. Keep the best sequence found over the whole budget (1000 s) and
   return it as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper utilities
# --------------------------------------------------------------------------- #

def project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where u_i > (cumulative sum - 1)/i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def power_law_sequence(n, p):
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s


def exponential_sequence(n, r):
    """a_i ∝ r**i, normalised to sum = 1 (0<r<1)."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s


def linear_decreasing_sequence(n):
    """a_i ∝ (n-i), i=0..n-1, normalised to sum = 1."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def objective_score(a):
    """Exact evaluation of the competition metric for a vector a."""
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    return float(2.0 * len(a) * np.max(conv) / (sum_a ** 2))


def fast_objective(a):
    """Same as objective_score but assumes sum(a)==1 (used internally)."""
    conv = np.convolve(a, a)
    return float(2.0 * len(a) * np.max(conv))


# --------------------------------------------------------------------------- #
# Smooth‑max projected gradient descent
# --------------------------------------------------------------------------- #

def smooth_max_gradient_descent(a0, beta, step, max_iter, deadline):
    """
    Minimise the soft‑max of the autocorrelation of a non‑negative vector.
    a0 must already be normalised (sum==1).  Returns (best_vec, best_score).
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_objective(a)

    # Pre‑compute an array for the index range used in the gradient slice
    arange_n = np.arange(n)

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # ----- convolution and soft‑max weights -----
        conv = np.convolve(a, a)
        max_c = conv.max()
        # stabilise the exponential
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient = 2 * cross‑correlation(w, a) -----
        # cross‑correlation can be obtained via a convolution with the reversed w
        rev_w = w[::-1]
        full = np.convolve(rev_w, a)          # length = 3n‑2
        L = len(w) - 1                        # = 2n‑2
        g = 2.0 * full[L - arange_n]          # shape (n,)

        # ----- gradient step + projection onto simplex -----
        a_next = a - step * g
        a_next = project_simplex(a_next)

        # ----- keep the best -----
        cur_score = fast_objective(a_next)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_next.copy()

        a = a_next
        # gradually shrink the step size
        step *= 0.995

    return best_a, best_score


# --------------------------------------------------------------------------- #
# Targeted hill‑climbing refinement
# --------------------------------------------------------------------------- #

def hill_climb_targeted(a, deadline, max_no_improve=2000):
    """
    Small stochastic hill‑climb that preferentially moves probability mass
    away from the indices that dominate the current maximal convolution entry.
    """
    n = len(a)
    a_cur = a.copy()
    best_score = fast_objective(a_cur)
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = np.convolve(a_cur, a_cur)
        K = int(np.argmax(conv))          # index of current maximum
        # compute contributions a_i * a_{K-i} for all i
        idx = np.arange(n)
        j_idx = K - idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a_cur[idx[mask]] * a_cur[j_idx[mask]]

        if contrib.sum() == 0:
            # fallback to random source index
            i_src = np.random.randint(n)
        else:
            # sample source index proportionally to its contribution
            prob = contrib / contrib.sum()
            i_src = np.random.choice(n, p=prob)

        # choose a destination index that does NOT affect the maximal entry
        if K < n - 1:
            # any index > K will have K‑j negative → no contribution
            dest_candidates = np.arange(K + 1, n)
        else:
            # K is at the far right, pick a small index
            dest_candidates = np.arange(0, K) if K > 0 else np.arange(n)
        if dest_candidates.size == 0:
            j_dst = np.random.randint(n)
        else:
            j_dst = np.random.choice(dest_candidates)

        # move at most 5 % of the source mass
        delta = 0.05 * a_cur[i_src]
        if delta <= 0:
            no_improve += 1
            continue

        cand = a_cur.copy()
        cand[i_src] -= delta
        cand[j_dst] += delta

        # numerical safeguard – keep non‑negative
        if cand[i_src] < 0:
            # transfer the remainder that would go negative
            delta_adj = a_cur[i_src]
            cand[i_src] = 0.0
            cand[j_dst] = a_cur[j_dst] + delta_adj

        # (the sum is preserved exactly)
        cand_score = fast_objective(cand)
        if cand_score < best_score - 1e-12:
            a_cur = cand
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

    return a_cur
