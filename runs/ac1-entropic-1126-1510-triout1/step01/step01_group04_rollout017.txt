def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 5000

    # initial exponent and offset
    cur_exp = -0.55
    cur_off = 0.0

    cur_seq = _make_power_law(n, cur_exp, cur_off)
    best_score = _eval(cur_seq)
    best_seq = cur_seq.copy()
    best_exp, best_off = cur_exp, cur_off

    # Simulated annealing (SA) over (exp, off)
    iteration = 0
    while time.time() < deadline and iteration < 200000:
        # propose new parameters
        # temperature schedule
        t = max(0.001, (deadline - time.time()) / (budget_s - 0.5))
        # propose step sizes decreasing with iteration
        exp_step = 0.01 * t  # adapt
        off_step = 0.5 * t

        new_exp = cur_exp + np.random.randn() * exp_step
        new_off = max(0.0, cur_off + np.random.randn() * off_step)

        # clamp exponents
        if new_exp < -0.8 or new_exp > -0.3:
            continue

        new_seq = _make_power_law(n, new_exp, new_off)
        new_score = _eval(new_seq)

        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            best_exp, best_off = new_exp, new_off
            cur_exp, cur_off = new_exp, new_off
            print(f"[SA] iter {iteration:05d} improved score {best_score:.6f} exp {best_exp:.5f} off {best_off:.2f}")
        else:
            # accept with probability
            prob = np.exp(-(new_score - cur_score) / (t+1e-12))
            if np.random.rand() < prob:
                cur_exp, cur_off = new_exp, new_off
                cur_seq = new_seq
                cur_score = new_score

        iteration += 1

    # after param search, fine-tune with hill climbing on actual seq
    best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # gradient descent refinement
    best_seq, best_score = _gradient_step(best_seq, best_score, deadline)

    final_sequence = [float(v) for v in best_seq.tolist()]
    print(f"Returning sequence length {len(final_sequence)} final_score {best_score:.6f}")
    return final_sequence

def _gradient_step(seq, best_score, deadline):
    n = len(seq)
    cur = seq.copy()
    step = 0.1
    it = 0
    eps = 1e-12
    while time.time() < deadline and it < 5000:
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        # compute active indices: near max within eps fraction
        active = np.where(conv >= max_val - 1e-12)[0]  # use exact equality maybe.
        # Build gradient average over active indices
        grad = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * cur[k - idx]
        grad /= max(1, len(active))
        # step
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s
        new_score = _eval(nxt)
        if new_score < best_score:
            cur = nxt
            best_score = new_score
            step = step * 0.9  # maybe reduce step gradually
        else:
            step = step * 0.5
        it += 1
        if it % 500 == 0:
            print(f"[gradient] iter {it} best_score={best_score:.6f}")
    return cur, best_score

"""
Propose a step‑function (non‑negative height vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The evaluation metric is homogeneous, so we can (and do) normalise the
vector to sum to one.  The algorithm consists of three stages:

1. **Coarse grid search** on a small problem (n≈2000) over a two‑parameter
   family  a_i ∝ (i+offset)^{exponent}.  The grid is fine enough to locate
   the region of the best constant (exponent ≈ –0.55, offset a few units).

2. **Local random refinement** of the exponent/offset on the same small
   problem.  The best parameters are then used to build a full‑size
   candidate (n≈5000).

3. **Simple optimisation on the simplex** for the full‑size vector:
   * stochastic hill‑climbing (move a small amount of mass from one index
     to another) and
   * projected sub‑gradient descent on the maximal convolution entry,
     averaging the sub‑gradients of all active (maximal) constraints.
   Both steps respect the security checks of `evaluate_sequence` (non‑negative,
   clamped to [0, 1000]).

The whole procedure stops when the time budget (default 1000 s) is exhausted.
The returned list of floats can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (wrappers around the supplied evaluator)
# ----------------------------------------------------------------------
def _eval(seq: np.ndarray) -> float:
    """Wrap evaluate_sequence – expects a plain Python list."""
    return evaluate_sequence(seq.tolist())

def _make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """
    Return a normalised power‑law vector
        a_i ∝ (i + offset + 1)^{exponent},   i = 0 … n‑1 .
    The vector is scaled so that its sum equals 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


# ----------------------------------------------------------------------
# Simple optimisation on the simplex
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray, best_score: float, deadline: float):
    """Stochastic hill‑climbing that moves a small fraction of mass."""
    n = len(seq)
    cur = seq.copy()
    it = 0
    while time.time() < deadline and it < 30000:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * np.random.rand() * 0.05  # ≤5 % of the source mass
        cur[i] -= delta
        cur[j] += delta
        score = _eval(cur)
        if score < best_score:
            best_score = score
        else:
            # reject the move
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it % 5000 == 0:
            print(f"[hill‑climb] iteration {it:,d}, best_score={best_score:.6f}")
    return cur, best_score


def _gradient_step(seq: np.ndarray, best_score: float, deadline: float):
    """Projected sub‑gradient descent on the maximal convolution entry."""
    n = len(seq)
    cur = seq.copy()
    step = 0.12
    it = 0
    while time.time() < deadline and it < 3000:
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        # early exit if we already beat the score we are tracking
        if 2 * n * max_val <= best_score:
            best_score = 2 * n * max_val

        # indices where the convolution attains (or is numerically close to) the max
        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # build the (averaged) sub‑gradient of max(conv) w.r.t. cur
        grad = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * cur[k - idx]
        grad /= max(1, len(active))

        # take a step opposite to the gradient, then project onto the simplex
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s

        new_score = _eval(nxt)
        if new_score < best_score:
            cur = nxt
            best_score = new_score
            step *= 0.95               # a bit more aggressive after success
        else:
            step *= 0.7                # shrink step when no improvement
        it += 1
        if it % 500 == 0:
            print(f"[gradient] iteration {it:,d}, best_score={best_score:.6f}")
    return cur, best_score


# ----------------------------------------------------------------------
# Parameter search on the two‑parameter power‑law family
# ----------------------------------------------------------------------
def _grid_search_small(n: int, deadline: float):
    """Coarse grid search (exponent, offset) on a small problem."""
    exponents = np.linspace(-0.70, -0.30, 81)   # step ≈0.005
    offsets   = np.linspace(0.0, 5.0, 21)      # step 0.25
    best_score = float("inf")
    best_exp = None
    best_off = None
    for a in exponents:
        for off in offsets:
            if time.time() >= deadline:
                break
            seq = _make_power_law(n, a, offset=off)
            score = _eval(seq)
            if score < best_score:
                best_score, best_exp, best_off = score, a, off
    print(f"[grid] best exponent={best_exp:.5f}, offset={best_off:.2f}, score={best_score:.6f}")
    return best_exp, best_off, best_score


def _local_refine(exp: float, off: float, score: float,
                  n: int, deadline: float):
    """
    Random walk around the current best (exp, off) on the small problem.
    """
    best_exp, best_off, best_score = exp, off, score
    cur_exp, cur_off = exp, off
    while time.time() < deadline:
        # propose a new point in the neighbourhood
        new_exp = cur_exp + np.random.uniform(-0.01, 0.01)
        new_off = max(0.0, cur_off + np.random.uniform(-0.5, 0.5))
        if not (-0.80 <= new_exp <= -0.25):
            continue
        seq = _make_power_law(n, new_exp, offset=new_off)
        new_score = _eval(seq)
        if new_score < best_score:
            best_score, best_exp, best_off = new_score, new_exp, new_off
            cur_exp, cur_off = new_exp, new_off
            print(f"[refine] improved to {best_score:.6f} "
                  f"(exp={best_exp:.5f}, off={best_off:.2f})")
        else:
            # occasional uphill move to escape plateaus
            if np.random.rand() < 0.02:
                cur_exp, cur_off = new_exp, new_off
    return best_exp, best_off, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging system
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """Return a list of non‑negative heights that achieves a low score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Small problem for parameter search (fast)
    n_small = 2000
    exp, off, score = _grid_search_small(n_small, deadline)
    exp, off, score = _local_refine(exp, off, score, n_small, deadline)

    # Build the full‑size candidate with the best parameters found
    n_full = 5000
    seq = _make_power_law(n_full, exp, offset=off)
    best_score = _eval(seq)
    print(f"[init] full size score={best_score:.6f} (exp={exp:.5f}, off={off:.2f})")

    # Stage 2 – stochastic hill‑climbing on the full vector
    seq, best_score = _hill_climb(seq, best_score, deadline)
    print(f"[hill‑climb] best_score={best_score:.6f}")

    # Stage 3 – projected gradient descent refinement
    seq, best_score = _gradient_step(seq, best_score, deadline)
    print(f"[gradient] final_score={best_score:.6f}")

    final_sequence = [float(v) for v in seq.tolist()]
    print(f"Returning sequence of length {len(final_sequence)} "
          f"with evaluation score {best_score:.6f}")
    return final_sequence
