"""
Propose a step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines three ideas:

1. **Power‑law grid search** – a quick scan over exponents (and a tiny offset)
   to obtain a good starting distribution.

2. **Linear‑programming boost** – given the current vector `f`,
   solve the linear program

        maximise   Σ g_j
        subject to g_j ≥ 0,
                   (f * g)[k] ≤ M   for all k,
                   g_j ≤ 1000

   where `M = max(f * f)`.  The solution pushes mass to indices where the
   current convolution constraints are slack, thereby flattening the peak.
   The solution is scaled to the same L1‑norm as `f` and mixed with `f`
   (mixing factor `t`).  Several LP iterations are performed while time permits.

3. **Local refinement** – a fast stochastic hill‑climber followed by a
   projected sub‑gradient descent (as used in the reference solution).

The whole search respects the security checks inside `evaluate_sequence`,
runs within the supplied time budget (default 1000 s) and returns a plain
Python list of non‑negative heights.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper wrappers -------------------------------------------------------
# ----------------------------------------------------------------------
def _eval(seq: np.ndarray) -> float:
    """Call the imported `evaluate_sequence` on a plain list."""
    return evaluate_sequence(seq.tolist())


def _make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+offset+1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the autocorrelation of `seq`."""
    return np.convolve(seq, seq).max()


# ----------------------------------------------------------------------
# Hill‑climbing ---------------------------------------------------------
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray, best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing on the probability simplex."""
    n = seq.shape[0]
    cur = seq.copy()
    it = 0
    max_iters = 20000
    while time.time() < deadline and it < max_iters:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            it += 1
            continue
        # move at most 5 % of the mass at i toward j
        delta = cur[i] * np.random.rand() * 0.05
        cur[i] -= delta
        cur[j] += delta
        score = _eval(cur)
        if score < best_score:
            best_score = score
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it % 5000 == 0:
            print(f"[hill‑climb] iteration {it}, best_score={best_score:.6f}")
    return cur, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent ----------------------------------------
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the maximal convolution entry."""
    n = seq.shape[0]
    cur = seq.copy()
    step = 0.12
    it = 0
    max_iters = 2000
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        # current objective for comparison
        cur_obj = 2.0 * n * max_val
        if cur_obj < best_score:
            best_score = cur_obj
        # index attaining the max (any one is fine)
        k = int(conv.argmax())
        # gradient of max(conv) w.r.t. cur:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur)
        grad[idx] = 2.0 * cur[k - idx]
        # take a step opposite to the gradient
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s                     # re‑project onto the simplex (L1‑norm = 1)
        new_score = _eval(nxt)
        if new_score < best_score:
            cur = nxt
            best_score = new_score
            step *= 0.99
        else:
            step *= 0.90
        it += 1
        if it % 500 == 0:
            print(f"[gradient] iteration {it}, best_score={best_score:.6f}")
    return cur, best_score


# ----------------------------------------------------------------------
# Linear‑programming boost -----------------------------------------------
# ----------------------------------------------------------------------
def _lp_improve(seq: np.ndarray, best_score: float, deadline: float,
                mix_frac: float = 0.3, max_iters: int = 10) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP

        maximise Σ g_j
        s.t.   g_j ≥ 0, g_j ≤ 1000
               (seq * g)[k] ≤ M  ∀k,   where M = max(seq*seq)

    The solution `g` is scaled to the same L1‑norm as `seq` and mixed
    (convex combination) with `seq`.  Several such refinements are performed
    while time permits.
    """
    cur = seq.copy()
    n = cur.shape[0]
    rows = 2 * n - 1
    # pre‑allocate the constraint matrix (dense – still modest for n≈1500)
    A_template = np.zeros((rows, n), dtype=np.float64)

    for outer in range(max_iters):
        if time.time() >= deadline:
            break
        M = _conv_max(cur)
        # build convolution matrix for the *current* `cur`
        # column j contains `cur` shifted by j rows
        A_template.fill(0.0)
        for j in range(n):
            A_template[j:j + n, j] = cur
        b = np.full(rows, M, dtype=np.float64)

        # objective: maximise sum(g)  → minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        # variable bounds: 0 ≤ g_j ≤ 1000
        bounds = [(0.0, 1000.0) for _ in range(n)]

        # solve the LP
        res = linprog(c, A_ub=A_template, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP] iteration {outer}: linprog failed – {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            print(f"[LP] iteration {outer}: zero solution")
            break

        # scale g to have the same L1‑norm as `cur` (which is 1.0 after normalisation)
        scale = cur.sum() / sum_g
        g *= scale

        # convex mixture
        t = mix_frac
        nxt = (1.0 - t) * cur + t * g
        # safety clipping (the evaluator caps values at 1000 anyway)
        nxt = np.clip(nxt, 0.0, 1000.0)

        new_score = _eval(nxt)
        if new_score < best_score:
            cur = nxt
            best_score = new_score
            print(f"[LP] iteration {outer}: improved score to {best_score:.6f}")
            # keep the mixing fraction, maybe even increase a little
        else:
            # no improvement – shrink the mixing factor
            mix_frac *= 0.5
            if mix_frac < 0.01:
                print("[LP] mixing fraction too small – stop")
                break
        # continue with the updated vector
    return cur, best_score

                best_score = score
                best_seq = cand.copy()
    print(f"[grid] best exponent≈{a:.4f}, offset={off:.2f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Linear‑programming boost – flatten the peak
    # ------------------------------------------------------------------
    best_seq, best_score = _lp_improve(best_seq, best_score, deadline,
                                       mix_frac=0.3, max_iters=12)
    print(f"[LP] after boost best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Stochastic hill‑climbing for fine‑grained adjustments
    # ------------------------------------------------------------------
    best_seq, best_score = _hill_climb(best_seq, best_score, deadline)
    print(f"[hill‑climb] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Projected gradient descent for final polishing
    # ------------------------------------------------------------------
    best_seq, best_score = _gradient_step(best_seq, best_score, deadline)
    print(f"[gradient] final_score={best_score:.6f}")

    # Convert to plain Python floats for the evaluator
    final_sequence = [float(v) for v in best_seq.tolist()]

    print(f"Returning sequence of length {len(final_sequence)} "
          f"with evaluation score {best_score:.6f}")
    return final_sequence
