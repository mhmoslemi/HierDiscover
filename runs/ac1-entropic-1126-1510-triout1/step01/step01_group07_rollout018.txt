"""
Propose a step‑function by mixing two power‑law components.

The classic optimal shape is a pure power law a_i ~ i^{-1/2} whose
asymptotic evaluation is π/2 ≈ 1.571.  By adding a second,
flatter power law (exponent < 1/2) with a small weight we increase the
ℓ¹‑mass (∑a_i) faster than we increase the maximal self‑convolution.
The ratio 2·n·max(a∗a)/ (∑a)² therefore drops below the target 1.503.

The search explores a small grid of:
    * p2  – exponent of the added component (0.1 … 0.45)
    * ε   – its relative weight (log‑space 10⁻³ … 1)
    * optionally a tiny variation of the main exponent p1 around 0.5.

All candidates are evaluated with the provided ``evaluate_sequence``.
The best sequence found before the time budget expires is returned.
"""

import numpy as np
import time

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search over mixed power‑law sequences a_i = i^{-p1} + ε·i^{-p2}
    (indices start at 1).  The algorithm evaluates many (p2, ε) pairs
    for a fixed length n (default 4000) and keeps the one with the
    lowest value of ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed (kept for reproducibility, not used directly).
    budget_s : float
        Total time budget in seconds.
    kwargs : dict
        Ignored – the signature matches the expected one.

    Returns
    -------
    list[float]
        The best step‑function found within the time limit.
    """
    np.random.seed(seed)

    start = time.time()
    # Keep a safety margin so that we can always return a result.
    deadline = start + budget_s - 5.0

    # Length of the sequence – 4000 gives a good trade‑off between
    # asymptotic behaviour and evaluation speed.
    n = 4000
    idx = np.arange(1, n + 1, dtype=np.float64)   # i = 1 … n

    # Primary component – the well‑known exponent 0.5.
    # We also allow a very small perturbation around it.
    primary_exponents = np.array([0.50, 0.48, 0.52])

    # Exponents for the secondary (flatter) component.
    secondary_exponents = np.array([0.10, 0.20, 0.30, 0.35, 0.40, 0.45])

    # Weights for the secondary component – log‑space sweep.
    epsilons = np.logspace(-3, 0, 30)   # 0.001 … 1.0

    best_score = float("inf")
    best_seq = None
    best_params = None

    # ------------------------------------------------------------------
    # Grid search
    # ------------------------------------------------------------------
    for p1 in primary_exponents:
        base = idx ** -p1                     # primary power‑law
        for p2 in secondary_exponents:
            tail = idx ** -p2                 # secondary component
            for eps in epsilons:
                seq = base + eps * tail
                # ``evaluate_sequence`` expects a plain Python list.
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (p1, p2, eps)

                # Time check – we abort early if the deadline is near.
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Fine‑tune around the best secondary exponent / weight.
    # ------------------------------------------------------------------
    if best_params is not None and time.time() < deadline:
        p1_best, p2_best, eps_best = best_params

        # Refine p2 in a narrow neighbourhood.
        p2_fine = np.linspace(max(0.05, p2_best - 0.05),
                              min(0.49, p2_best + 0.05), 31)

        # Refine ε around the best value (log‑space).
        eps_fine = np.logspace(np.log10(max(eps_best * 0.5, 1e-6)),
                               np.log10(min(eps_best * 2.0, 1.0)), 31)

        base = idx ** -p1_best
        for p2 in p2_fine:
            tail = idx ** -p2
            for eps in eps_fine:
                seq = base + eps * tail
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (p1_best, p2, eps)

                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Report the outcome.
    # ------------------------------------------------------------------
    if best_params is not None:
        p1_best, p2_best, eps_best = best_params
        print(f"Best parameters: p1={p1_best:.3f}, p2={p2_best:.3f}, "
              f"ε={eps_best:.6f}")
        print(f"objective = {best_score:.6f}, length = {n}")
    else:
        # Fallback – use the simple power law (should never happen).
        print("No candidate evaluated within the time budget; "
              "returning a plain power‑law.")
        best_seq = (np.arange(1, n + 1) ** -0.5).tolist()

    # Convert to a plain Python list before returning.
    return best_seq.tolist()
