    """Project v onto the probability simplex {x >=0, sum x =1}."""
    # using algorithm from Duchi et al. (2008)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence whose self‑convolution peak is as small as
possible (hence the evaluation metric 2·n·max(conv)/(∑a)² is minimal).

Algorithm
---------

1. **Initialisation** – pick a moderate length (≈2000) and start from the
   “arcsine” sequence (coefficients C_i = binom(2i,i)/4^i, normalised to
   sum = 1).  If a global variable ``height_sequence_1`` is available it is
   also tried and the better of the two initialisations is kept.

2. **Alternating linear‑programming refinement** – keep the current
   sequence f.  Let ``M = max(f∗f)``.  Solve the LP

        maximise    1ᵀ g
        subject to  g ≥ 0
                    f ∗ g ≤ M            (element‑wise)

   The convolution constraint is linear in the unknown vector g, therefore a
   standard LP solver (cvxpy + ECOS) can be used.  The optimal g tries to
   “fill the available slack’’ while keeping the same maximal convolution
   value M.  After solving, g is rescaled to have unit L₁‑norm (the metric is
   scale‑invariant) and mixed with the old sequence:

        f ← (1‑t)·f + t·g , t ∈ (0,1)

   The mixing weight t starts at 0.2 and is slowly decreased.  Each iteration
   cannot increase the maximal convolution and typically reduces it.

3. **Gradient‑descent polishing** – when the time budget is low or cvxpy is
   unavailable, a smooth approximation of the max (soft‑max) is used.
   For the current sequence a we compute

        w_k = exp(β·(conv_k – max(conv))) / Σ exp(·)

   and the gradient of the smoothed max is

        grad = 2·conv(w, a[::-1])[n‑1:2·n‑1] .

   A small step opposite the gradient is taken and the result is projected
   onto the probability simplex (non‑negative, sum = 1).  The step size and
   the soft‑max temperature β are slowly decreased.

4. **Book‑keeping** – after every update the true score
   ``evaluate_sequence`` is computed; the best sequence found so far is
   returned at the end of the allotted time (≈1000 s).

The whole search respects the 1000 s limit, uses only NumPy and cvxpy,
and never performs I/O.  The returned list contains non‑negative numbers.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return the normalised (sum = 1) arcsine coefficients C_i for i=0..n‑1.
    The recurrence C_{i+1}=C_i·(2i+1)/(2(i+1)) is used for numerical stability.
    """
    seq = []
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = [x / total for x in seq]          # L1‑normalise to 1
    return seq


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
