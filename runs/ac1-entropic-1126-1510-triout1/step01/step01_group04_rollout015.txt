def _best_response_lp(f, deadline):
    n = len(f)
    rows = 2*n-1
    # Build A_ub
    A_ub = np.zeros((rows, n+1), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        # j indexes
        # i = k - j
        # fill slice
        j_range = slice(lo, hi+1)
        i_vals = k - np.arange(lo, hi+1)
        A_ub[k, j_range] = f[i_vals]
        A_ub[k, -1] = -1.0
    b_ub = np.zeros(rows, dtype=np.float64)
    A_eq = np.zeros((1,n+1), dtype=np.float64)
    A_eq[0,:n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n+1, dtype=np.float64)
    c[-1] = 1.0  # minimize M
    bounds = [(0, None)]*n + [(0, None)]
    # Use linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0,deadline-time.time())})
    if not res.success:

"""
Optimization of non‑negative step function minimizing
2 * n * max(conv(a,a)) / (sum(a)**2).

Strategy:
1. Build several deterministic candidates (power‑law, exponential,
   arcsine, triangular) and keep the best.
2. For the current best vector f solve a linear program:
       minimise M
       s.t.   (f * b)_k ≤ M   for all k,
              ∑ b_i = 1,
              b_i ≥ 0.
   The solution b is the best response to f.  Mixing f and b with a
   small convex coefficient usually reduces the self‑convolution peak.
   Repeat the LP+mixing step while improvement is observed.
3. Refine the result with a cheap stochastic hill‑climb and a projected
   sub‑gradient descent on the maximal convolution entry.
All steps respect the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _score(seq):
    """Fast evaluation of the objective (sum is assumed >0 and values are safe)."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    return 2 * len(seq) * conv.max() / (s * s)

def _make_power_law(n, exponent, offset=0.0):
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total

def _make_exponential(n, rate):
    idx = np.arange(n, dtype=np.float64)
    vec = rate ** idx
    total = vec.sum()
    return vec / total

def _make_arcsine(n):
    # positions at the centre of each unit interval: i+0.5
    i = np.arange(n, dtype=np.float64) + 0.5
    vec = 1.0 / np.sqrt(i * (n - i))
    return vec / vec.sum()

def _make_triangular(n):
    if n % 2 == 0:
        half = n // 2
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        half = n // 2
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    vec = np.concatenate([left, right]).astype(np.float64)
    return vec / vec.sum()

def _best_response_lp(f, time_limit):
    """Return b (probability vector) that minimises max(f*b) and the associated max value."""
    n = len(f)
    rows = 2 * n - 1
    # Build the inequality matrix A_ub * x <= 0  (x = [b, M])
    A_ub = np.zeros((rows, n + 1), dtype=np.float64)
    # Fill the convolution part.
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # j runs from lo..hi, i = k - j
        js = np.arange(lo, hi + 1, dtype=np.int64)
        A_ub[k, lo:hi + 1] = f[k - js]
        A_ub[k, -1] = -1.0
    b_ub = np.zeros(rows, dtype=np.float64)
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Objective: minimise M
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0
    bounds = [(0.0, None)] * n + [(0.0, None)]
    # HiGHS accepts a time limit in seconds
    options = {"disp": False, "presolve": True, "time_limit": max(0.0, time_limit)}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method="highs", options=options)
    if not res.success:
        # fallback: return the original vector
        return None, None
    b = res.x[:n]
    M = res.x[-1]
    # Numerical cleanup – enforce sum(b)=1
    b_sum = b.sum()
    if b_sum == 0:
        return None, None
    b = b / b_sum
    return b, M

def _hill_climb(seq, best_score, deadline, max_iter=50000):
    n = len(seq)
    cur = seq.copy()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] <= 0.0:
            it += 1
            continue
        # move up to 10 % of the mass at i to j
        delta = cur[i] * np.random.rand() * 0.10
        cur[i] -= delta
        cur[j] += delta
        score = _score(cur)
        if score < best_score:
            best_score = score
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
    return cur, best_score

def _gradient_step(seq, best_score, deadline, max_iter=2000):
    n = len(seq)
    cur = seq.copy()
    step = 0.15
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        if 2 * n * max_val < best_score - 1e-12:
            best_score = 2 * n * max_val
        # Identify a single index attaining the max (first found)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur)
        grad[idx] = 2.0 * cur[k - idx]
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s
        score = _score(nxt)
        if score < best_score:
            cur = nxt
            best_score = score
            step *= 0.99
        else:
            step *= 0.9
    return cur, best_score

def propose_candidate(seed=0, budget_s=1000):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    reference bound 1.503.  The routine combines a short grid search,
    a linear‑programming “best‑response’’ step, a stochastic hill‑climb
    and a projected sub‑gradient refinement.  The whole search respects
    the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Generate a handful of sensible starting points
    # ------------------------------------------------------------------
    n = 2000   # moderate size – enough for a good constant, cheap for LPs
    best_seq = None
    best_score = float('inf')

    # power‑law family
    exponents = np.linspace(-0.70, -0.30, 81)   # fine grid
    for a in exponents:
        seq = _make_power_law(n, a, offset=0.0)
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
    print(f"[grid] best power‑law score = {best_score:.6f}")

    # exponential family
    for rate in np.linspace(0.95, 0.60, 8):
        seq = _make_exponential(n, rate)
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
    print(f"[grid] after exponentials best score = {best_score:.6f}")

    # arcsine‑type shape
    seq = _make_arcsine(n)
    sc = _score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
    print(f"[grid] after arcsine best score = {best_score:.6f}")

    # triangular shape
    seq = _make_triangular(n)
    sc = _score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
    print(f"[grid] after triangular best score = {best_score:.6f}")

    cur = best_seq.copy()
    # ------------------------------------------------------------------
    # 2) Repeated LP “best‑response’’ + mixing
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        time_left = deadline - time.time()
        # Solve the linear programme that minimises max(f*b)
        b, M_lp = _best_response_lp(cur, time_limit=time_left)
        if b is None:
            break
        # Try a convex combination of the current vector and the LP solution.
        # Small mixing coefficients are usually enough to push the peak down.
        improved = False
        for t in (0.5, 0.3, 0.1, 0.05):
            cand = (1.0 - t) * cur + t * b
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            cand_score = _score(cand)
            if cand_score < best_score - 1e-8:
                cur = cand
                best_score = cand_score
                improved = True
                print(f"[LP‑mix] iter {iteration}, t={t:.2f}, score={best_score:.6f}")
                break
        if not improved:
            # no mixing coefficient improved the score – stop the LP phase
            break
        iteration += 1
        if iteration >= 15:   # safety guard
            break

    print(f"[LP] finished after {iteration} iterations, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Stochastic hill‑climbing (fine‑grained local search)
    # ------------------------------------------------------------------
    cur, best_score = _hill_climb(cur, best_score, deadline, max_iter=120000)
    print(f"[hill‑climb] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Projected sub‑gradient descent on the max‑convolution entry
    # ------------------------------------------------------------------
    cur, best_score = _gradient_step(cur, best_score, deadline, max_iter=2500)
    print(f"[gradient] final_score={best_score:.6f}")

    final_sequence = [float(v) for v in cur.tolist()]
    print(f"Returning sequence of length {len(final_sequence)} with evaluation {best_score:.6f}")
    return final_sequence

"""
Optimization of non‑negative step‑function heights minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

Strategy
---------
1. Build a small portfolio of analytically motivated candidates
   (power‑law, exponential, arcsine‑type and triangular shapes) and keep
   the best one.
2. Repeatedly solve a linear programme

        minimise   M
        s.t.       (f * b)_k ≤ M   for every k,
                   ∑ b_i = 1,
                   b_i ≥ 0.

   The LP gives the "best response" `b` to the current vector `f`.
   Mixing a fraction of `b` into `f` almost always lowers the self‑
   convolution peak.  The mixing step is repeated while improvements are
   observed (or the time budget is exhausted).
3. A cheap stochastic hill‑climb is then applied to fine‑tune the
   distribution.
4. Finally a projected sub‑gradient descent on the maximal convolution
   entry is performed.

All steps respect the 1000 s time budget.  The routine returns the
list of heights; the evaluator will clamp and normalise them automatically.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _score(seq: np.ndarray) -> float:
    """Fast internal evaluation (identical to `evaluate_sequence` for safe inputs)."""
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    return 2.0 * len(seq) * conv.max() / (s * s)


def _make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^exponent, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


def _make_exponential(n: int, rate: float) -> np.ndarray:
    """a_i ∝ rate**i, normalised."""
    vec = rate ** np.arange(n, dtype=np.float64)
    return vec / vec.sum()


def _make_arcsine(n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine density:
          a_i ∝ 1 / sqrt( (i+0.5) (n-i-0.5) )
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    vec = 1.0 / np.sqrt(i * (n - i))
    return vec / vec.sum()


def _make_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape (increasing then decreasing)."""
    if n % 2 == 0:
        half = n // 2
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        half = n // 2
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    vec = np.concatenate([left, right]).astype(np.float64)
    return vec / vec.sum()


def _best_response_lp(f: np.ndarray, time_limit: float):
    """
    Solve the LP

        minimise   M
        s.t.       (f * b)_k ≤ M   ∀k,
                   ∑ b_i = 1,
                   b_i ≥ 0.

    Returns the optimal probability vector b and the optimal value M.
    """
    n = len(f)
    rows = 2 * n - 1

    # Inequality matrix  A_ub @ [b; M] ≤ 0
    A_ub = np.zeros((rows, n + 1), dtype=np.float64)

    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1, dtype=np.int64)          # column indices
        A_ub[k, lo:hi + 1] = f[k - js]                     # f_{k-j}
        A_ub[k, -1] = -1.0                                 # –M

    b_ub = np.zeros(rows, dtype=np.float64)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimise M
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]

    # HiGHS accepts a time limit (seconds) via the options dict
    opts = {"presolve": True, "time_limit": max(0.0, time_limit)}

    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options=opts,
    )

    if not res.success:
        # In rare cases the LP may fail; fall back to the current vector.
        return None, None

    b = res.x[:n]
    M = res.x[-1]

    # Numerical clean‑up: enforce the probability simplex exactly.
    total = b.sum()
    if total == 0.0:
        return None, None
    b /= total
    return b, M


def _hill_climb(seq: np.ndarray, best_score: float, deadline: float,
                max_iter: int = 50000) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing on the simplex."""
    n = len(seq)
    cur = seq.copy()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] <= 0.0:
            it += 1
            continue
        # Transfer up to 10 % of the mass at i to j
        delta = cur[i] * np.random.rand() * 0.10
        cur[i] -= delta
        cur[j] += delta
        score = _score(cur)
        if score < best_score:
            best_score = score
        else:
            # revert the move
            cur[i] += delta
            cur[j] -= delta
        it += 1
    return cur, best_score


def _gradient_step(seq: np.ndarray, best_score: float, deadline: float,
                   max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the maximal convolution entry."""
    n = len(seq)
    cur = seq.copy()
    step = 0.15
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        if 2.0 * n * max_val < best_score - 1e-12:
            best_score = 2.0 * n * max_val
        # pick a single index attaining the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur)
        grad[idx] = 2.0 * cur[k - idx]          # sub‑gradient of max(conv)
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s
        score = _score(nxt)
        if score < best_score:
            cur = nxt
            best_score = score
            step *= 0.99
        else:
            step *= 0.9
    return cur, best_score

    """
