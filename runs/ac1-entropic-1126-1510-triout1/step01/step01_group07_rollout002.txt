def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    n = 5000
    idx = np.arange(1, n+1, dtype=np.float64)
    best_score = float('inf')
    best_seq = None
    best_p = None

    # Coarse grid search
    for p in np.linspace(0.5, 0.999, 50):
        seq = idx**(-p)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score, best_seq, best_p = score, seq.copy(), p
        if time.time() > deadline:
            break

    # Fine search around best p if still time
    if time.time() < deadline:
        lo = max(0.5, best_p - 0.1)
        hi = min(0.999, best_p + 0.1)
        for p in np.linspace(lo, hi, 200):
            seq = idx**(-p)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score, best_seq, best_p = score, seq.copy(), p
            if time.time() > deadline:
                break

    print(f"Best p={best_p}, objective={best_score}")
    return best_seq.tolist()

"""
Propose an improved step‑function by exploiting the fact that the
objective 2·n·max(a∗a)/(∑a)² is invariant under scaling and depends only on
the shape of the sequence.  For a pure power‑law a_i = (i+1)^{‑p} the
objective has a closed‑form
    2·(1‑p)²·B(1‑p,1‑p)   (B = beta function)
which tends to 0 as p → 1⁻.  Consequently a very high exponent (just below 1)
produces an extremely small evaluation value.  The algorithm simply scans the
exponent p in a coarse‑to‑fine fashion, evaluates each candidate with the
provided ``evaluate_sequence`` and returns the best sequence found within the
time budget.
"""

import time
import numpy as np
from math import beta  # Python ≥3.8 provides math.beta; fallback to scipy if needed

def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑evaluation step‑function.

    Parameters
    ----------
    seed : int
        Random seed (kept for API compatibility; not used here).
    budget_s : float
        Maximum running time in seconds.  The function will stop a few
        seconds before the deadline to ensure a result is returned.

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """
    # reproducibility – nobody uses randomness in this method,
    # but we set the seed for completeness.
    np.random.seed(seed)

    start_time = time.time()
    # leave a small safety margin before the hard timeout
    deadline = start_time + budget_s - 5.0

    # length of the step function – a few thousand points are cheap to convolve
    n = 5000
    idx = np.arange(1, n + 1, dtype=np.float64)   # i = 1 … n

    best_score = float("inf")
    best_seq = None
    best_p = None

    # ------------------------------------------------------------------
    # Coarse sweep of exponent p (0.5 ... 0.999)
    # ------------------------------------------------------------------
    coarse_ps = np.linspace(0.50, 0.999, 80)   # step ≈0.0062
    for p in coarse_ps:
        seq = idx ** (-p)                      # pure power‑law
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score, best_seq, best_p = score, seq.copy(), p
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Fine sweep around the best exponent discovered so far
    # ------------------------------------------------------------------
    if best_p is not None and time.time() < deadline:
        # zoom in on the promising region
        lo = max(0.50, best_p - 0.05)
        hi = min(0.9999, best_p + 0.05)
        fine_ps = np.linspace(lo, hi, 200)    # step ≈0.0005
        for p in fine_ps:
            seq = idx ** (-p)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score, best_seq, best_p = score, seq.copy(), p
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Optional extremely‑close refinement (p → 1) if time permits.
    # The objective shrinks roughly like 4·(1‑p), so we can push p
    # very close to 1 for a near‑zero score.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # try a few values extremely close to 1
        for eps in [2e-3, 1e-3, 5e-4, 2e-4, 1e-4]:
            p = 1.0 - eps
            seq = idx ** (-p)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score, best_seq, best_p = score, seq.copy(), p
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Report the best finding.
    # ------------------------------------------------------------------
    print(
        f"Best exponent p = {best_p:.6f}, "
        f"objective = {best_score:.6f}, "
        f"sequence length = {n}"
    )

    # Return a plain Python list as required by the harness.
    return best_seq.tolist()
