            t = np.random.uniform(0.0002, 0.0015)

            a = r ** np.arange(n, dtype=float)
        elif shape == 'power':
            p = np.random.uniform(0.2, 2.5)
            a = (np.arange(1, n+1, dtype=float) ** -p)

            t = np.random.uniform(0.0002, 0.001)
            r = np.exp(-t)
            p = np.random.uniform(0.5, 1.5)

            sigma = np.random.uniform(n*0.1, n*0.3)

            x = np.arange(n, dtype=float)
            a = np.exp(-0.5*((x - mu)/sigma)**2)
        else:
            a = np.random.gamma(1.0,1.0,size=n)

        # optionally shuffle but we keep symmetric for gauss

            np.random.shuffle(a)

"""
Low‑max‑convolution step‑function search.

The code repeatedly draws a random length `n` (2000‑5000) and a random
initial shape (geometric, power‑law, mixed, Gaussian or Dirichlet).
Each candidate is refined by a projected gradient descent that minimizes
a *smooth* approximation of the maximum self‑convolution.
The smooth max is the log‑sum‑exp (softmax) of the full convolution,
controlled by a temperature parameter `beta` that is increased
gradually during the run.  The gradient of this smoothed objective is
computed with two FFT‑based convolutions, giving an O(n log n) per‑
iteration cost even for several thousand dimensions.

The iterates are projected onto the probability simplex
(`a_i ≥ 0, Σ a_i = 1`) after each step.  A simple back‑tracking line
search accepts only steps that strictly improve the *true* score
`2·n·max(conv(a,a))`.  If no improvement is found a tiny random
perturbation is tried.  The best sequence seen across all restarts is
returned; any remaining time is spent further refining that best
sequence.

This method extends the earlier sub‑gradient approach by using a
smooth surrogate for the max, an adaptive temperature schedule and
full‑gradient information, which typically drives the search past the
previous best bound (≈ 1.58) toward the target ≈ 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a*a via FFT, length 2·n‑1."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _gradient_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) of a*a w.r.t. a.

    p is the softmax probability vector over the full convolution
    (same length as the convolution, i.e. 2·n‑1).  The formula is:

        g = 2 * Σ_i p_i * a_{i‑k}  –  p_{2k} * a_k

    which is efficiently computed by an FFT‑based convolution.
    """
    n = a.shape[0]
    L = 2 * n - 1               # length of p
    # reverse p for the convolution trick
    p_rev = p[::-1]

    # size sufficient for the convolution a (len n) * p_rev (len L)
    size = 1 << ((n + L - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fp = np.fft.rfft(p_rev, n=size)
    conv_rev = np.fft.irfft(fa * fp, n=size)   # length = n+L-1 = 3n-2

    # extract the part needed for Σ_i p_i a_{i‑k}
    # conv_rev[t] with t = (2n‑2)‑k ; slice n‑1 … 2n‑2 and reverse
    sub = conv_rev[n - 1 : 2 * n - 1]          # length n, forward order
    part = sub[::-1]                           # Σ_i p_i a_{i‑k}

    # diagonal correction for i = 2k (the term counted only once)
    diag_weights = p[2 * np.arange(n)]

    grad = 2.0 * part - diag_weights * a
    return grad


def _improve_sequence_softmax(a0: np.ndarray, n: int, deadline: float):
    """
    Refine a candidate probability vector `a0` (sum=1, non‑negative)
    by projected gradient descent on a smoothed max-convolution objective.

    Returns the best vector found and its true evaluation score
    (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()

    conv = _conv_full_fft(a)
    max_conv = conv.max()
    best_score = 2.0 * n * max_conv

    it = 0
    beta = 1.0          # temperature for the softmax (log‑sum‑exp)
    beta_max = 64.0
    beta_growth = 1.25

    while time.time() < deadline and it < 5000:
        it += 1

        # current convolution and true score
        conv = _conv_full_fft(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ----- smoothed max (log‑sum‑exp) -----
        max_c = max_conv
        # compute exp(beta * (conv - max_c)) safely
        exp_terms = np.exp(beta * (conv - max_c))
        sum_exp = exp_terms.sum()
        # smoothed max (not directly used for step acceptance, but useful
        # for the gradient magnitude)
        _ = max_c + np.log(sum_exp) / beta

        # softmax probabilities over the full convolution
        p = exp_terms / sum_exp

        # gradient of the smoothed objective
        grad = _gradient_softmax(a, p)

        # step size (decreasing)
        eta0 = 0.5 / np.sqrt(it)

        improved = False
        # back‑tracking line search
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            eta = eta0 * factor
            a_candidate = _project_to_simplex(a - eta * grad)
            conv_cand = _conv_full_fft(a_candidate)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                break

        # if no descent, try a tiny random nudge
        if not improved:
            a_pert = a.copy()
            idx = np.random.randint(n, size=5)
            a_pert[idx] += np.random.randn(5) * 0.001
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= a_pert.sum()
            conv_pert = _conv_full_fft(a_pert)
            cand_score = 2.0 * n * conv_pert.max()
            if cand_score < cur_score:
                a = a_pert
                cur_score = cand_score

        # keep global best for this start
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 200 == 0:
                print(f"    iter {it}, new local best {best_score:.6f}")

        # increase temperature gradually (approaches true max)
        if beta < beta_max:
            beta = min(beta_max, beta * beta_growth)

        # early stop if near the target bound
        if best_score < 1.51:
            break

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.

    The function respects the 1000‑second budget, performs many random
    restarts with different lengths and shapes, refines each via the
    smoothed‑max gradient method, and finally returns the best sequence
    found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_score = float("inf")
    best_seq = None

    # bounds for the length of the step function
    MIN_N = 2000
    MAX_N = 5000

    while time.time() < deadline:
        n = np.random.randint(MIN_N, MAX_N + 1)
        shape = np.random.choice(["geom", "power", "mixed", "gauss", "random"])

        if shape == "geom":
            # geometric decay a_i = r**i
            t = np.random.uniform(0.0002, 0.0015)
            r = np.exp(-t)
            a = r ** np.arange(n, dtype=float)

        elif shape == "power":
            p = np.random.uniform(0.2, 2.5)
            a = (np.arange(1, n + 1, dtype=float) ** -p)

        elif shape == "mixed":
            t = np.random.uniform(0.0002, 0.001)
            r = np.exp(-t)
            p = np.random.uniform(0.5, 1.5)
            a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)

        elif shape == "gauss":
            sigma = np.random.uniform(0.1 * n, 0.3 * n)
            mu = np.random.uniform(0.0, n - 1.0)
            x = np.arange(n, dtype=float)
            a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)

        else:   # "random"
            a = np.random.gamma(1.0, 1.0, size=n)

        # keep symmetry for Gaussian; otherwise random permutation is fine
        if shape != "gauss" and np.random.rand() < 0.5:
            np.random.shuffle(a)

        # normalise to a probability vector (sum = 1)
        a = a / a.sum()

        # allocate at most 12 s to improve this start (or less if time runs out)
        sub_deadline = min(deadline, time.time() + 12.0)

        a_opt, score_opt = _improve_sequence_softmax(a, n, sub_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"Global improvement: n={n}, shape={shape}, score={best_score:.6f}")

        # stop early if we have already passed the target
        if best_score < 1.503:
            break

    # final polishing of the best candidate with any remaining time
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            best_seq, best_score = _improve_sequence_softmax(
                np.array(best_seq), len(best_seq), time.time() + remaining
            )
            print(f"Final polishing: score={best_score:.6f}")

    # fallback – uniform distribution (should never be needed)
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)

    # return a plain Python list for the external evaluator
    return best_seq.tolist()
