Implementation for real arrays:

def convolve_fft(x, y):
    L = len(x) + len(y) - 1
    N = 1 << (L-1).bit_length() # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    result = np.fft.irfft(X * Y, n=N)[:L]
    return result

    start = time.time()
    deadline = start + max(0., budget_s - 2.)

    # Determine n
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
    except Exception:
        n = 5000 # try 5000 for initial
    # Create arcsine shape
    idx = np.arange(n)
    eps = 1e-12
    a = np.zeros(n)
    # avoid i=0,n-1
    # For i from 0..n-1 use safe formula
    with np.errstate(divide='ignore'):

    a[0] = a[-1] = a[1]  # just set boundary to same as near border maybe
    a = a / a.sum()

    # Or fallback to uniform
    if np.any(np.isnan(a)) or np.isinf(a).any():
        a = np.full(n, 1.0/n)

    current = a.copy()
    best = current.copy()
    best_score = compute_objective(best)
    current_score = best_score

    iteration = 0
    last_improve_iter = 0
    # constants
    max_iter = 10000000

    while time.time() < deadline:
        iteration += 1
        # Linear interpolation of alpha from start to end
        elapsed = (time.time() - start) / (deadline - start)
        alpha = 0.1 + 0.9 * elapsed * 50   # e.g., go from 0.1 to 50 over time
        # Or implement as
        alpha = 0.5 + 49.5 * elapsed   # from 0.5 to 50

        eta = 0.5 / np.sqrt(iteration)

        # compute gradient step
        conv = _convolve_fft(current, current)  # length 2n-1
        max_conv = conv.max()
        # softmax weights (normalized)
        w = np.exp(alpha * (conv - max_conv))
        w = w / w.sum()

        grad = 2.0 * _convolve_fft(w, current[::-1])
        # slice of length n
        grad = grad[n-1:n-1+n]

        # occasional random perturbation
        if np.random.rand() < 0.02:

        if cand_score < best_score:

"""
Propose a candidate step‑function sequence.

The evaluation metric is

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

where `conv` is the full (linear) convolution.
Because the metric is homogeneous, we can restrict the sequence to the
probability simplex (sum = 1, a_i ≥ 0) – the score then becomes
`2·n·max(conv)`.

For a uniform sequence `a_i = 1/n` the score is 2.  The optimum can be
lowered by making the convolution as flat as possible; the theoretical
limit is 1 (for very large n) because the convolution values must sum to
1² = 1, so the best possible maximum is the average value `1/(2n−1)`.

We use a two‑phase optimisation:

1. **Flattening phase** – minimise the L² distance between the convolution
   and the uniform vector `C = 1/(2n−1)`.  The gradient of
   `∥conv(a,a) – C∥²` has a simple closed form and pushes the
   distribution toward the (continuous) arcsine law, which is known to be
   the “self‑convolution root’’ of the uniform distribution.

2. **Peak‑reduction phase** – minimise a smooth approximation of the
   `max` via a soft‑max (log‑sum‑exp) with an increasing temperature
   parameter `α`.  This refines the solution and drives the maximum
   convolution value down.

Both phases employ projected gradient steps on the simplex and occasional
random pair‑wise mass swaps (simulated‑annealing style) to escape shallow
local minima.  All heavy linear algebra is performed with FFT‑based
convolutions (`O(n log n)`), allowing thousands of iterations within the
1000 s budget.

The routine returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Random pairwise “mass‑swap’’ move (keeps the sum exactly 1)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # zero‑mean proposal, scaled to sigma
    delta = (np.random.rand() - 0.5) * sigma

    if delta > 0:
        delta = min(delta, seq[j])        # take from j
    else:
        delta = -min(-delta, seq[i])      # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# FFT‑based convolution (real input)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors using FFT (fast for n≈10⁴)."""
    n = x.shape[0] + y.shape[0] - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:n]
    return conv


# ----------------------------------------------------------------------
# Objective (fast, no security checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return the exact value of evaluate_sequence for non‑negative seq."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)          # tiny overhead compared to FFT version
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:                       # keep the same guard as evaluate_sequence
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Gradient of L2 flattening loss   L = ||conv(a,a) - C||²
#   where C = (sum(a)²)/(2n‑1) is the uniform target.
# ----------------------------------------------------------------------
def _gradient_flatten(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    conv = _fft_convolve(seq, seq)            # length 2n‑1
    C = (seq.sum() ** 2) / (2 * n - 1)        # ideal uniform value
    e = conv - C                               # error vector (length 2n‑1)

    # gradient = 4 * (e * rev(seq))   (slice the middle n entries)
    grad_full = 4.0 * _fft_convolve(e, seq[::-1])
    grad = grad_full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of soft‑max approximation of max(conv)
#   L = (1/α)·log Σ_k exp(α·conv_k)
# ----------------------------------------------------------------------
def _gradient_softmax(seq: np.ndarray, alpha: float) -> np.ndarray:
    n = seq.shape[0]
    conv = _fft_convolve(seq, seq)            # length 2n‑1
    max_c = conv.max()
    # numerically stable soft‑max weights
    w = np.exp(alpha * (conv - max_c))
    w = w / w.sum()
    # gradient = 2·(w * rev(seq))   (again slice central n entries)
    grad_full = 2.0 * _fft_convolve(w, seq[::-1])
    grad = grad_full[n - 1:n - 1 + n]
    return grad, conv, max_c

# Main entry point required by the harness
# ----------------------------------------------------------------------

    start = time.time()
    deadline = start + max(0.0, budget_s - 2.0)   # safety margin
