    import numpy as np, time, math

    best_score = float('inf')

    best_params = None
    lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000]
    p_vals = np.linspace(0.2, 1.0, 17)  # inclusive
    gamma_vals = np.linspace(0.0, 2.0, 21)
    beta_vals = np.linspace(0.0, 5.0, 11)
    # base families
    # Family 1: power law
    for n in lengths:
        idx = np.arange(1, n+1, dtype=np.float64)
        for p in p_vals:
            seq = idx ** -p
            score = objective(seq)
            if score < best_score:
                best_score = score; best_seq = seq.copy(); best_params=("power", n, p)
            if time.time() > deadline:
                return best_seq.tolist()
    # Family 2: power*exp
    for n in lengths:
        idx = np.arange(1, n+1, dtype=np.float64)
        for p in p_vals:

                seq = base * np.exp(-gamma * (idx-1) / (n-1))
                score = objective(seq)
                if score < best_score:
                    best_score = score; best_seq = seq.copy(); best_params=('exp', n, p, gamma)
                if time.time() > deadline:
                    return best_seq.tolist()
    # Family 3: power*poly
    for n in lengths:
        idx = np.arange(1, n+1, dtype=np.float64)
        for p in p_vals:
            base = idx ** -p
            for beta in beta_vals:
                taper = (1 - (idx-1)/(n-1)) ** beta

                if score < best_score:

    if best_params is not None and time.time() < deadline:

1. Pure power‑law                a_i = (i+1)^{-p}
2. Power‑law with exponential   a_i = (i+1)^{-p}·exp[-γ·i/(n‑1)]
3. Power‑law with polynomial    a_i = (i+1)^{-p}·(1‑i/(n‑1))^{β}
4. Power‑law with squared‑exponential taper
                                 a_i = (i+1)^{-p}·exp[-γ·(i/(n‑1))^{2}]

All parameters (exponent p, decay γ, polynomial β) are swept on a coarse grid.
The best configuration is refined on a finer grid and, finally, a short
random‑local‑search around the best point is performed.  A few different
lengths (n) are examined; the objective is computed directly (identical to
``evaluate_sequence``) to keep the whole search fast.  The function stops
well before the 1000 s budget and returns the best sequence found as a plain
list of floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – exact duplicate of the scoring rule used by ``evaluate_sequence``.
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / (sum(seq)**2) after applying the same
    clipping that evaluate_sequence performs.
    """
    # enforce the same safety checks as evaluate_sequence (no need for type checks)
    seq = np.maximum(0.0, np.minimum(1000.0, seq))
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # linear full convolution (same as evaluate_sequence)
    max_b = conv.max()
    n = seq.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main entry point – the platform will call this function.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises the
    evaluation function.  The routine respects the time budget and returns the
    best sequence found so far.
    """
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    best_score = float("inf")
    best_seq = None
    best_cfg = None

    # ------------------------------------------------------------------
    # Parameter grids
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000]   # moderate sizes for the coarse phase
    p_vals   = np.linspace(0.30, 0.80, 11)                # exponent of the power‑law
    gamma_vals = np.linspace(0.0, 2.5, 26)                # exponential decay
    beta_vals = np.linspace(0.0, 6.0, 13)                 # polynomial taper

    # ------------------------------------------------------------------
    # 1) pure power‑law
    # ------------------------------------------------------------------
    for n in lengths:
        idx = np.arange(1, n + 1, dtype=np.float64)      # i = 1 … n
        for p in p_vals:
            seq = idx ** (-p)
            score = _objective(seq)
            if score < best_score:
                best_score, best_seq, best_cfg = score, seq.copy(), ("pow", n, p)
            if time.time() > deadline:
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 2) power‑law × exponential taper
    # ------------------------------------------------------------------
    for n in lengths:
        idx = np.arange(1, n + 1, dtype=np.float64)
        scaled = (idx - 1) / (n - 1)                     # values in [0,1]
        for p in p_vals:
            base = idx ** (-p)
            for gamma in gamma_vals:
                seq = base * np.exp(-gamma * scaled)
                score = _objective(seq)
                if score < best_score:
                    best_score, best_seq, best_cfg = score, seq.copy(), ("exp", n, p, gamma)
                if time.time() > deadline:
                    return best_seq.tolist()

    # ------------------------------------------------------------------
    # 3) power‑law × polynomial taper  ( (1‑x)^{beta} )
    # ------------------------------------------------------------------
    for n in lengths:
        idx = np.arange(1, n + 1, dtype=np.float64)
        scaled = (idx - 1) / (n - 1)
        for p in p_vals:
            base = idx ** (-p)
            for beta in beta_vals:
                taper = (1.0 - scaled) ** beta
                seq = base * taper
                score = _objective(seq)
                if score < best_score:
                    best_score, best_seq, best_cfg = score, seq.copy(), ("poly", n, p, beta)
                if time.time() > deadline:
                    return best_seq.tolist()

    # ------------------------------------------------------------------
    # 4) power‑law × squared‑exponential taper
    # ------------------------------------------------------------------
    for n in lengths:
        idx = np.arange(1, n + 1, dtype=np.float64)
        scaled = (idx - 1) / (n - 1)
        for p in p_vals:
            base = idx ** (-p)
            for gamma in gamma_vals:
                seq = base * np.exp(-gamma * (scaled ** 2))
                score = _objective(seq)
                if score < best_score:
                    best_score, best_seq, best_cfg = score, seq.copy(), ("exp2", n, p, gamma)
                if time.time() > deadline:
                    return best_seq.tolist()

    # ------------------------------------------------------------------
    # Fine‑grained refinement around the best configuration (if any)
    # ------------------------------------------------------------------
    if best_cfg is not None and time.time() < deadline:
        kind = best_cfg[0]

        if kind in ("exp", "exp2"):
            _, n, p, gamma = best_cfg
            idx = np.arange(1, n + 1, dtype=np.float64)
            scaled = (idx - 1) / (n - 1)
            base = idx ** (-p)

            lo = max(0.0, gamma - 0.3)
            hi = gamma + 0.3
            for gamma2 in np.linspace(lo, hi, 61):
                if kind == "exp":
                    seq = base * np.exp(-gamma2 * scaled)
                else:   # squared‑exponential
                    seq = base * np.exp(-gamma2 * (scaled ** 2))
                score = _objective(seq)
                if score < best_score:
                    best_score, best_seq, best_cfg = score, seq.copy(), (kind, n, p, gamma2)
                if time.time() > deadline:
                    break

        elif kind == "poly":
            _, n, p, beta = best_cfg
            idx = np.arange(1, n + 1, dtype=np.float64)
            scaled = (idx - 1) / (n - 1)
            base = idx ** (-p)

            lo = max(0.0, beta - 2.0)
            hi = beta + 2.0
            for beta2 in np.linspace(lo, hi, 81):
                taper = (1.0 - scaled) ** beta2
                seq = base * taper
                score = _objective(seq)
                if score < best_score:
                    best_score, best_seq, best_cfg = score, seq.copy(), ("poly", n, p, beta2)
                if time.time() > deadline:
                    break

        elif kind == "pow":
            _, n, p = best_cfg
            idx = np.arange(1, n + 1, dtype=np.float64)
            lo = max(0.05, p - 0.15)
            hi = p + 0.15
            for p2 in np.linspace(lo, hi, 61):
                seq = idx ** (-p2)
                score = _objective(seq)
                if score < best_score:
                    best_score, best_seq, best_cfg = score, seq.copy(), ("pow", n, p2)
                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Try the same best shape on a few larger lengths (optional, cheap).
    # ------------------------------------------------------------------
    if best_cfg is not None and time.time() < deadline:
        kind = best_cfg[0]
        # keep the other parameters, just vary n
        larger_ns = [6000, 8000, 10000, 12000]
        for n in larger_ns:
            idx = np.arange(1, n + 1, dtype=np.float64)
            scaled = (idx - 1) / (n - 1)
            if kind == "pow":
                _, _, p = best_cfg
                seq = idx ** (-p)
            elif kind == "exp":
                _, _, p, gamma = best_cfg
                base = idx ** (-p)
                seq = base * np.exp(-gamma * scaled)
            elif kind == "exp2":
                _, _, p, gamma = best_cfg
                base = idx ** (-p)
                seq = base * np.exp(-gamma * (scaled ** 2))
            elif kind == "poly":
                _, _, p, beta = best_cfg
                base = idx ** (-p)
                taper = (1.0 - scaled) ** beta
                seq = base * taper
            else:
                continue

            score = _objective(seq)
            if score < best_score:
                best_score, best_seq, best_cfg = score, seq.copy(), (kind, n) + best_cfg[2:]
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Small random local‑search around the current best point.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        for _ in range(200):                # limited number of tries
            # multiplicative Gaussian noise (≈5 % typical)
            noise = np.random.normal(loc=1.0, scale=0.05, size=n)
            cand = best_seq * noise
            # enforce the hard caps used by evaluate_sequence
            cand = np.clip(cand, 0.0, 1000.0)
            score = _objective(cand)
            if score < best_score:
                best_score, best_seq = score, cand.copy()
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    print(
        f"Best configuration: {best_cfg}, "
        f"objective = {best_score:.6f}, "
        f"length = {len(best_seq)}"
    )
    # Return a plain Python list – the evaluation harness expects this.
    return best_seq.tolist()
