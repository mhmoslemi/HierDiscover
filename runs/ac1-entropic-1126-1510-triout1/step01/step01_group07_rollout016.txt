def simplex_projection(v):
    # v: array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑evaluation step‑function by direct minimisation of the
self‑convolution peak.

The evaluation function depends only on the maximal entry of the
autoconvolution of the normalised coefficient vector:
    evaluate_sequence(a) = 2·n·max_k (a*a)_k     (with sum(a)=1).

Hence the problem is: minimise   max_k (a*a)_k   over the simplex
{a_i≥0, Σa_i=1}.  This is a convex‐piecewise‑quadratic problem.
We use an inexpensive projected sub‑gradient descent:

* start from the classic power‑law a_i ∝ (i+1)^-0.5 (the known near‑optimal
  shape);
* at each iteration locate a peak index k where the convolution attains its
  maximum;
* the contribution of coefficient a_i to this peak is a_{k‑i} (zero outside
  the support).  The coefficient with the largest contribution is the best
  candidate to *decrease*, while a coefficient with the smallest (or zero)
  contribution is the best to *increase*.
* a small amount of mass is transferred from the “down” index to the “up”
  index; the vector is re‑normalised (projection onto the probability
  simplex) and the new peak is recomputed.
* step size adapts automatically: it is increased after a successful move
  and decreased after a failure.

The loop runs until the allotted time runs out (leaving a safety margin)
and the best sequence seen is returned.  All numbers stay non‑negative,
so the security checks in ``evaluate_sequence`` are satisfied.
"""

import numpy as np
import time
import math


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
