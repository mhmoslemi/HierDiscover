def fft_convolve(a,b):
    n = len(a) + len(b) -1
    N = 1 << (n - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    c = np.fft.irfft(C, n=N)[:n]
    return c

"""
Propose a step‑function (list of non‑negative heights) that minimises
`evaluate_sequence`.

Algorithm
---------
1. Start from the discrete arcsine distribution – the known near‑optimal
   shape for the “flat convolution’’ problem.
2. Refine it with a fast projected gradient method:
   * At each iteration compute the full autocorrelation `b = a * a`
     using an FFT (cost O(n log n)).
   * Approximate the non‑smooth objective `max(b)` by a soft‑max:
       f(a) = (1/β)·log Σ exp(β·b_k)
     with a temperature β that grows during the run.
     The gradient of `f` w.r.t. `a` is
        g = 2·conv_rev_a,w
     where `w` are the soft‑max weights and `conv_rev_a,w` is the
     convolution of the reversed `a` with `w`.  This convolution is also
     performed with an FFT.
   * Use a simple Adam‑style update (momentum + adaptive step) and
     project the result back onto the probability simplex
     (non‑negative entries summing to 1) by clipping and renormalising.
3. Keep the best sequence (lowest score) seen so far.
4. When the time budget is almost exhausted, perform a short
   sub‑gradient “max‑only’’ refinement to squeeze out a final gain.
5. Return the best sequence as a plain Python list.

The whole routine respects a 1000 s wall‑clock limit, needs only NumPy,
and avoids any filesystem or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector `v` onto the simplex {x ≥ 0, sum(x)=1}.

    The implementation simply clips negatives and rescales – this is
    fast (O(n)) and sufficient for the optimisation loop.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine density of length `n`, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights with a low `evaluate_sequence`
    score.  The function respects the given time budget (seconds).
    """
    np.random.seed(seed)
    # leave a safety margin for the runner itself
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a working length.  If a historic construction is available,
    # use a multiple of its size; otherwise fall back to a reasonable
    # default.  Cap the length to keep FFTs fast.
    try:
        n0 = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n0 = 2000

    # a factor > 1 gives a better asymptotic bound; 2 × works well.
    n = int(max(5000, min(20000, n0 * 2)))
    print(f"[init] chosen length = {n}")

    # ------------------------------------------------------------------
    # Initialise the sequence with the (scaled) arcsine law and add a tiny
    # random jitter to break symmetry.
    a = _arcsine_distribution(n)
    a += np.random.randn(n) * 1e-5
    a = _project_onto_simplex(a)

    # Pre‑compute the FFT size (next power of two larger than 3·n)
    m_len = 3 * n - 2
    N_fft = 1 << (m_len - 1).bit_length()  # power‑of‑two ≥ 3·n‑2

    # Adam‑style state
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2                     # base learning rate

    # Soft‑max temperature schedule
    beta_start = 0.01
    beta_max = 50.0
    beta = beta_start

    # bookkeeping
    best_seq = a.copy()
    # compute initial score (sum(a)=1 → denominator=1)
    A = np.fft.rfft(a, n=N_fft)
    b = np.fft.irfft(A * A, n=N_fft)[: 2 * n - 1]
    best_score = float(2.0 * n * b.max())
    print(f"[init] initial score = {best_score:.6f}")

    it = 0
    # ------------------------------------------------------------------
    # Main optimisation loop (FFT‑based Adam + simplex projection)
    while True:
        now = time.time()
        if now >= deadline:
            break
        it += 1

        # ---- autocorrelation ------------------------------------------------
        A = np.fft.rfft(a, n=N_fft)
        b = np.fft.irfft(A * A, n=N_fft)[: 2 * n - 1]

        # ---- soft‑max weighting ---------------------------------------------
        beta = min(beta_max, beta_start + it * (beta_max - beta_start) / 300000.0)
        w = _softmax(beta * b)          # shape (2n‑1,)

        # ---- gradient of the smoothed max -----------------------------------
        rev_a = a[::-1]
        rev_fft = np.fft.rfft(rev_a, n=N_fft)
        w_fft = np.fft.rfft(w, n=N_fft)
        conv_full = np.fft.irfft(rev_fft * w_fft, n=N_fft)[: m_len]
        # slice the central n entries (indices n‑1 … n‑1+n‑1)
        g = 2.0 * conv_full[n - 1 : n - 1 + n]

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        # bias‑corrected estimates (avoid pow‑overhead for very large it)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / np.sqrt(it)           # diminishing learning rate
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto the simplex ------------------------------------
        a = _project_onto_simplex(a)

        # ---- keep the best ---------------------------------------------------

            best_seq = a.copy()

            if best_score <= 1.503:
                # target reached – we can stop early
                break

        if it % 25000 == 0:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    # Final short “max‑only’’ fine‑tuning (sub‑gradient on the true max)
    # This section runs only if a little time is left.
    while time.time() < deadline - 0.1:
        # compute exact autocorrelation (full)
        A = np.fft.rfft(a, n=N_fft)
        b = np.fft.irfft(A * A, n=N_fft)[: 2 * n - 1]
        k = int(np.argmax(b))               # index of the active constraint
        # gradient for the active maximal entry: g_i = 2·a_{k‑i}
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # simple diminishing step size
        eta = 0.05 / np.sqrt(it + 1.0)
        a = a - eta * g
        a = _project_onto_simplex(a)

        cur_score = float(2.0 * n * b.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] fine‑tune score={best_score:.6f}")

        it += 1
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Safety net – if something went wrong, fall back to a uniform vector.
    if best_seq is None or best_seq.size == 0:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
