def simplex_projection(v):
    # ensure v is a numpy array.
    if np.sum(v) == 0:
        return np.full_like(v, 1.0 / len(v))
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑score step‑function sequence.

The algorithm works in three stages:

1. **Global random search** (small vector, n≈800) over a power‑law
   a_i ∝ (i+offset+1)^α.  The exponent α∈(‑0.7,‑0.35) and the offset are
   sampled uniformly.  Every candidate is evaluated with the supplied
   `evaluate_sequence`.  The best (α, offset) pair is retained.

2. **Upscaling & smooth‑max gradient descent**.
   The best α/offset is used to build a longer vector (n≈4000).  The
   objective is 2·n·max(conv(a,a)).  To minimise the max we optimise a
   smooth approximation  (1/λ)·log Σ exp(λ·conv) – a standard
   “soft‑max” surrogate.  Its gradient can be written analytically as
   a correlation of the weight vector with the current sequence.
   After each gradient step we project onto the probability simplex
   (non‑negative entries summing to one).  The smoothing parameter λ
   is slowly increased so that the optimiser first reduces the overall
   bulk of the convolution and finally concentrates on the peak.

3. **Fine‑grained stochastic hill‑climbing** that moves a tiny
   fraction of mass from one coordinate to another while keeping the
   sum invariant.  Accepted moves strictly improve the true objective.

All stages respect the security checks performed inside
`evaluate_sequence`.  The total runtime is limited by the `budget_s`
parameter (default 1000 s).  Intermediate scores and timings are printed
to aid debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _eval(seq: np.ndarray) -> float:
    """Wrapper for the externally provided evaluate_sequence."""
    # evaluate_sequence expects a plain list; conversion is cheap.
    return evaluate_sequence(seq.tolist())


def _make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+offset+1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}
    (Duchi et al., 2008)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Convolution a*a using FFT (real‑valued, returns full length)."""
    n = a.shape[0]
    # size = next power‑of‑two ≥ 2n‑1 for speed
    size = 1 << ((2 * n - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)[: 2 * n - 1]
    return conv.real


def _smoothmax_grad(a: np.ndarray, lam: float):
    """
    Gradient of the smooth‑max surrogate
          f(a) = (1/lam)·log Σ exp(lam·(a*a))
    with respect to a.
    Returns (grad, max_conv, full_convolution).
    """
    n = a.shape[0]
    conv = _conv_fft(a)                     # length 2n‑1
    max_val = conv.max()
    # soft‑max weights (stable by subtracting max)
    exp_vals = np.exp(lam * (conv - max_val))
    weights = exp_vals / exp_vals.sum()     # sum = 1

    # correlation = Σ_j weights[i+j] * a_j   (i = 0..n‑1)
    # This is a convolution of weights with the reversed a.
    size = 1 << ((len(weights) + n - 1).bit_length())
    W = np.fft.rfft(weights, size)
    A_rev = np.fft.rfft(a[::-1], size)
    corr_full = np.fft.irfft(W * A_rev, size)[: len(weights) + n - 1]
    corr = corr_full[n - 1 : n - 1 + n]      # length n

    grad = 4.0 * n * corr                    # ∂(2n·smoothmax)/∂a_i
    return grad, max_val, conv


def _hill_climb(seq: np.ndarray, best_score: float, deadline: float):
    """Simple stochastic hill‑climbing on the simplex."""
    n = len(seq)
    cur = seq.copy()
    it = 0
    # limit the total number of proposals to keep within the time budget
    while time.time() < deadline and it < 20000:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            it += 1
            continue
        # move up to 5 % of the mass from i to j
        delta = cur[i] * np.random.rand() * 0.05
        cur[i] -= delta
        cur[j] += delta
        score = _eval(cur)
        if score < best_score:
            best_score = score
            # keep improvement
        else:
            # reject move
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it % 5000 == 0:
            print(f"[hill‑climb] it={it}, best_score={best_score:.6f}")
    return cur, best_score


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights that (hopefully) gives a low
    value of `evaluate_sequence`.  The function runs for at most
    `budget_s` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Coarse random search on a small vector (fast evaluation)
    # ------------------------------------------------------------------
    n_small = 800
    best_seq_small = None
    best_score = float("inf")
    best_alpha = -0.5
    best_offset = 0.0

    it = 0
    while time.time() < deadline and it < 3000:
        alpha = np.random.uniform(-0.70, -0.35)
        offset = np.random.uniform(0.0, 5.0)
        seq = _make_power_law(n_small, alpha, offset)
        score = _eval(seq)
        if score < best_score:
            best_score = score
            best_seq_small = seq.copy()
            best_alpha, best_offset = alpha, offset
        it += 1
        if it % 500 == 0:
            print(f"[rand‑search] it={it}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Fine‑grid refinement around the current best parameters
    # ------------------------------------------------------------------
    for a in np.linspace(best_alpha - 0.05, best_alpha + 0.05, 21):
        for off in np.linspace(best_offset - 1.0, best_offset + 1.0, 21):
            seq = _make_power_law(n_small, a, off)
            score = _eval(seq)
            if score < best_score:
                best_score = score
                best_seq_small = seq.copy()
                best_alpha, best_offset = a, off
    print(f"[grid‑refine] α={best_alpha:.4f}, offset={best_offset:.4f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Upscale to a larger problem size and run smooth‑max descent
    # ------------------------------------------------------------------
    n_large = 4000
    cur = _make_power_law(n_large, best_alpha, best_offset)
    cur = cur / cur.sum()          # ensure exact normalisation
    best_score = _eval(cur)
    print(f"[upscale] initial score (n={n_large}) = {best_score:.6f}")

    lam = 5.0                # start with a modest smoothing factor
    step = 0.12              # gradient step size
    it = 0
    while time.time() < deadline and it < 2500:
        grad, max_conv, _ = _smoothmax_grad(cur, lam)
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        nxt = _project_simplex(nxt)
        new_score = _eval(nxt)
        if new_score < best_score:
            cur = nxt
            best_score = new_score
            # when we improve we increase λ a little – we now focus more on the peak
            lam = min(lam * 1.07, 100.0)
            step *= 0.98
        else:
            # no improvement – become more cautious
            step *= 0.90
            lam *= 1.02
        it += 1
        if it % 250 == 0:
            print(f"[smooth‑grad] it={it}, λ={lam:.3f}, step={step:.5f}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final stochastic hill‑climbing for extra polishing
    # ------------------------------------------------------------------
    cur, best_score = _hill_climb(cur, best_score, deadline)
    print(f"[final‑hill] best_score={best_score:.6f}")

    # Convert to plain Python list for the evaluator / downstream consumer
    final_sequence = [float(v) for v in cur.tolist()]
    print(f"Returning sequence of length {len(final_sequence)} with score {best_score:.6f}")
    return final_sequence
