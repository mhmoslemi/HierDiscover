def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # Get base
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
    except Exception:
        n = 8000
        base = np.full(n, 1.0/n)
    # Normalize sum to 1
    base = base / max(base.sum(), 1e-12)
    # Ensure nonnegative and proper scaling
    current = _project_simplex(base)
    best = current.copy()
    # compute objectives
    current_score = compute_obj(current)
    best_score = current_score
    # parameters
    total_iter = 0
    last_improve_iter = 0
    beta_start = 0.5
    beta_end = 0.01
    temp_start = 1.0
    temp_end = 1e-4
    while time.time() < deadline:
        total_iter += 1
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temperature = temp_start * (1.0 - frac) + temp_end * frac

        # choose move
        r = np.random.rand()
        if r < 0.5:
            # softmax gradient step
            beta_factor = beta_start * (1.0 - frac) + beta_end * frac
            eta = 0.5 / np.sqrt(total_iter)
            candidate = _gradient_step_soft(current, eta, beta_factor)
        elif r < 0.8:
            # max gradient step
            eta = 0.5 / np.sqrt(total_iter)
            candidate = _gradient_step_max(current, eta)
        else:
            # random pair move
            sigma = 0.2 / np.sqrt(total_iter)
            candidate = _random_pair_move(current, sigma)

        # evaluate
        cand_score = compute_obj(candidate)

        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()
            last_improve_iter = total_iter
            print(f"[Iter {total_iter:6d}] New best score: {best_score:.6f}")

        # acceptance
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob
        if accept:
            current = candidate
            current_score = cand_score

        # restart if stuck
        if total_iter - last_improve_iter > 8000:
            # perturb best and project
            perturb = np.random.rand(n)
            perturb = perturb / perturb.sum()
            current = _project_simplex(0.99 * best + 0.01 * perturb)
            current_score = compute_obj(current)
            last_improve_iter = total_iter

    # final best list
    return best.tolist()

def _gradient_step_soft(seq, eta, beta_factor):
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_conv = conv.max()
    # softmax weights
    # avoid overflow: shift by max
    beta = max_conv * beta_factor
    if beta <= 0:
        beta = 1e-12
    w = np.exp((conv - max_conv) / beta)
    # normalize weights, optional? gradient scaling does not need normalized, but step size will adapt
    # optional: w = w / w.sum()

    g = 2.0 * slice_
    new_seq = seq - eta * g
    new_seq = _project_simplex(new_seq)

"""
Propose a candidate step‑function sequence.

The optimiser works on the probability simplex (all entries ≥0, sum = 1) –
the scaling freedom of ``evaluate_sequence`` is removed.  It mixes three
move types:

1. **Smoothed‑gradient step** – a softmax weighting of all convolution
   entries produces a gradient that reduces the whole convolution profile,
   not only the peak.  The softening parameter β is annealed from a
   relatively large value (≈50 % of the current maximal autocorrelation)
   down to almost zero, at which point the step reduces to the classic
   max‑gradient step.

2. **Exact max‑gradient step** – the original gradient direction that
   directly attacks the current maximal convolution entry.

3. **Random pairwise mass swap** – preserves the simplex constraint while
   providing stochastic exploration.

The algorithm keeps the best solution seen, accepts worsening moves with
a Metropolis rule (temperature linearly cooled) and periodically injects
a small random perturbation if progress stalls.  The objective is
computed analytically (2·n·max(conv)/sum²) to avoid the heavy call to
``evaluate_sequence`` during the search; the final sequence is returned
as a plain list which the harness will evaluate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive – fall back to uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Objective (exact, no call to evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Exact max‑gradient step (uses the index of the maximal convolution entry)
# ----------------------------------------------------------------------
def _max_gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)

    g = np.zeros_like(seq)
    g[mask] = 2.0 * seq[j[mask]]
    new_seq = seq - eta * g
    return _project_simplex(new_seq)


# ----------------------------------------------------------------------
# Smoothed‑gradient step (softmax weighting of all convolution entries)
# ----------------------------------------------------------------------
def _softmax_gradient_step(seq: np.ndarray, eta: float, beta_factor: float) -> np.ndarray:
    """
    Perform a gradient step on the softmax‑weighted sum
        F(seq) = Σ_k w_k·(seq*seq)_k
    where w_k ∝ exp((conv_k−max(conv))/β) and β = beta_factor·max(conv).
    """
    n = seq.shape[0]

    # full autocorrelation of the current sequence
    conv = np.convolve(seq, seq)
    max_conv = conv.max()
    beta = max_conv * beta_factor
    if beta <= 0.0:
        beta = 1e-12

    # softmax weights (largest entry has weight 1, the rest are ≤1)
    w = np.exp((conv - max_conv) / beta)      # shape = 2n‑1

    # gradient g_i = 2·Σ_j seq_j·w_{i+j}
    # This is a correlation of seq and w.
    # Using the identity:
    #   corr(seq, w)[i] = Σ_j seq_j·w_{i+j}
    # can be obtained as conv(seq_rev, w)[n‑1+i] where seq_rev = seq[::-1].
    seq_rev = seq[::-1]
    corr_full = np.convolve(seq_rev, w)       # length = 3n‑2
    g = 2.0 * corr_full[n - 1 : n - 1 + n]    # slice indices n‑1 … 2n‑2 → length n
    new_seq = seq - eta * g
    return _project_simplex(new_seq)


# ----------------------------------------------------------------------
# Random pairwise mass‑swap move (preserves the sum exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # zero‑mean proposal, uniformly drawn from (‑σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma

    # clip to keep both entries non‑negative
    if delta > 0.0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that tries to minimise
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, budget_s - 0.5)

    # ------------------------------------------------------------------
    # Initialise the search vector (use the previously discovered construction
    # if it exists; otherwise fall back to a uniform vector of moderate length)
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # provided globally
        n = base.size
        base = base / max(base.sum(), 1e-12)               # normalise to sum=1
    except Exception:
        n = 7200                     # a size comparable to the historic best
        base = np.full(n, 1.0 / n, dtype=float)

    current = _project_simplex(base)
    best = current.copy()

    current_score = _objective(current)
    best_score = current_score

    # ------------------------------------------------------------------
    # Annealing / schedule parameters
    # ------------------------------------------------------------------
    temp_start = 1.0
    temp_end = 1e-4
    beta_start = 0.5     # β = beta_start·max_conv  (very smooth)
    beta_end = 0.01      # gradually tighten to the max‑gradient behaviour
    last_improve_iter = 0
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)                # [0,1]
        temperature = temp_start * (1.0 - frac) + temp_end * frac

        # ------------------------------------------------------------------
        # Choose a move type
        # ------------------------------------------------------------------
        r = np.random.rand()
        if r < 0.5:
            # Smoothed‑gradient step
            beta_factor = beta_start * (1.0 - frac) + beta_end * frac
            eta = 0.6 / np.sqrt(it)                       # step size decays
            candidate = _softmax_gradient_step(current, eta, beta_factor)
        elif r < 0.8:
            # Exact max‑gradient step
            eta = 0.5 / np.sqrt(it)
            candidate = _max_gradient_step(current, eta)
        else:
            # Random pairwise swap
            sigma = 0.25 / np.sqrt(it)
            candidate = _random_pair_move(current, sigma)

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

        # Metropolis acceptance
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:

            current_score = cand_score

        # ------------------------------------------------------------------
        # Restart if we have been stuck for a while
        # ------------------------------------------------------------------

            perturb = np.random.rand(n)
            perturb /= perturb.sum()
            current = _project_simplex(0.99 * best + 0.01 * perturb)
            current_score = _objective(current)
            last_improve_iter = it

    # ------------------------------------------------------------------

                best = cand.copy()
                print(f"[Polish {p:4d}] Improved best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return best.tolist()
