"""
Propose a low‑evaluation step function using a two‑phase optimisation:

1. **LP‑driven improvement** – we repeatedly solve a linear programme:
   maximise Σ g_j subject to g ≥ 0 and (f * g)[k] ≤ M (the current
   self‑convolution maximum).  The solution g is scaled to have the same
   ℓ₁‑norm as f and mixed with f (convex combination).  This iteration
   provably never increases the maximal convolution value and quickly
   drives it down.

2. **Fine‑grained projected sub‑gradient descent** – after the LP phase
   we polish the best sequence with a short run of the classic
   projected sub‑gradient optimiser (the routine from the previous
   state‑of‑the‑art).

The search explores several lengths (≈ 500–3000) and a handful of random
initialisations for each length.  The whole process respects the 1000 s
budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (simple utilities)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2*n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    # Clip tiny negatives caused by rounding.
    conv = np.maximum(conv, 0.0)
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact score 2 * n * max(conv(a, a))."""
    n = a.size
    max_c = np.max(conv_full(a))
    return float(2.0 * n * max_c)


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (fine‑tuning phase)
# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return (max value, indices achieving the max) of a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = np.max(conv)
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0].astype(int)
    return float(max_val), idx


def _gradient_for_max(a: np.ndarray, max_idx):
    """Sub‑gradient of max‑convolution w.r.t. a."""
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


def _improve_sequence(a0: np.ndarray, n, deadline):
    """Projected sub‑gradient descent (small‑scale polishing)."""
    a = a0.copy()
    best_a = a.copy()
    max_c, max_idx = _conv_max_fft(a)
    best_score = 2.0 * n * max_c

    it = 0
    while time.time() < deadline and it < 3000:
        it += 1
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c

        grad = _gradient_for_max(a, max_idx)

        # simple decreasing step size
        eta0 = 0.2 / np.sqrt(it)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            eta = eta0 * factor
            a_candidate = _project_to_simplex(a - eta * grad)
            max_c2, _ = _conv_max_fft(a_candidate)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                break

        if not improved:
            a_pert = a.copy()
            idx = np.random.randint(n, size=5)
            a_pert[idx] += np.random.randn(5) * 0.001
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / np.sum(a_pert)
            max_c2, _ = _conv_max_fft(a_pert)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score:
                a = a_pert
                cur_score = cand_score

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 200 == 0:
                print(f"    sub‑grad iter {it}, new local best {best_score:.6f}")

        if best_score < 1.51:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming based improvement (core phase)
# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float):
    """Solve  max Σ g_j   s.t. g ≥ 0 , conv(f,g) ≤ M."""
    import cvxpy as cp

    n = f.size
    g = cp.Variable(n, nonneg=True)
    # cvxpy can convolve a variable with a constant array directly.
    conv_fg = cp.conv(g, f)  # length 2n‑1
    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver='ECOS', warm_start=True, verbose=False, max_iters=5000)
    except cp.error.SolverError:
        return None

    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    return g0


def _improve_via_lp(f_init: np.ndarray, n: int, deadline: float,
                    t: float = 0.5, max_iter: int = 20):
    """Iteratively mix the current sequence with LP solutions."""
    f = f_init.copy()
    best_score = compute_score(f)
    best_f = f.copy()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv_f = conv_full(f)
        M = conv_f.max()

        g0 = _solve_lp(f, M)
        if g0 is None:
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0

        # convex combination
        f_new = (1 - t) * f + t * g
        score_new = compute_score(f_new)

        if score_new + 1e-12 < best_score:
            best_f = f_new.copy()
            best_score = score_new
            f = f_new
            print(f"    LP iter {it}, improved score {best_score:.6f}")
        else:
            # no improvement – stop early
            break

        if best_score < 1.503:
            break

    return best_f, best_score

# Main entry point
# ----------------------------------------------------------------------

    """

    deadline = time.time() + budget_s - 5.0  # safety margin

    best_score = float("inf")
    best_seq = None

    # Candidate lengths – we explore a range of sizes.
    lengths = [500, 800, 1200, 1500, 2000, 2500, 3000]

    for n in lengths:
        if time.time() >= deadline:
            break
        # Up to a few restarts per length.
        restarts = 5
        for rs in range(restarts):
            if time.time() >= deadline:
                break

            # ---- initialise a random shape ---------------------------------
            shape = np.random.choice(["geom", "power", "mixed", "random"])
            if shape == "geom":
                t_val = np.random.uniform(0.0003, 0.0015)
                r = np.exp(-t_val)
                a = r ** np.arange(n, dtype=float)
            elif shape == "power":
                p = np.random.uniform(0.3, 2.0)
                a = (np.arange(1, n + 1, dtype=float) ** -p)
            elif shape == "mixed":
                t_val = np.random.uniform(0.0003, 0.001)
                r = np.exp(-t_val)
                p = np.random.uniform(0.5, 1.5)
                a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)
            else:  # random Dirichlet‑like draw
                a = np.random.gamma(1.0, 1.0, size=n)

            a = np.maximum(a, 0.0)
            a = a / a.sum()  # now a is on the simplex

            # ---- LP‑driven improvement ------------------------------------
            sub_deadline = min(deadline, time.time() + 30.0)  # cap per start
            f_lp, score_lp = _improve_via_lp(a, n, sub_deadline,
                                            t=0.5, max_iter=15)

            # ---- optional fine polishing -----------------------------------
            remaining = deadline - time.time()
            if remaining > 5.0:
                # give the sub‑gradient method a few seconds
                polish_deadline = time.time() + min(remaining, 8.0)
                f_ref, score_ref = _improve_sequence(f_lp, n, polish_deadline)
            else:
                f_ref, score_ref = f_lp, score_lp

            # ---- update global best -----------------------------------------
            if score_ref < best_score:
                best_score = score_ref
                best_seq = f_ref.copy()
                print(f"Global improvement: n={n}, shape={shape}, "

                if best_score < 1.503:
                    break
        if best_score < 1.503:
            break

    # Fallback – uniform distribution (should never be used)
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # Return plain Python list for the external evaluator.
    return best_seq.tolist()
