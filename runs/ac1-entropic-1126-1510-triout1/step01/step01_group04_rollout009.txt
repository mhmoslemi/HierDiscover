"""
propose_candidate – optimisation of a non‑negative step function.

The algorithm proceeds in four stages:

1️⃣  **Parameter sweep** – a dense grid search over power‑law exponents
    (a_i ∝ (i+offset+1)^α) and a few offsets gives a good starting shape.
2️⃣  **Random seeds** – a handful of Dirichlet (Gamma) samples are examined;
    the best of all candidates becomes the current best.
3️⃣  **Smoothed‑max gradient descent** – we minimise the smooth approximation
    L = (1/λ) log Σ exp(λ · (a*a)_k) (λ large makes L≈max (a*a)).
    Its gradient can be obtained by a single convolution,
    allowing an efficient projected additive update on the probability simplex.
    The step‑size and the smoothing parameter λ are adapted on‑the‑fly.
4️⃣  **Fine‑grained hill‑climbing** – stochastic moves of a few per‑cent of mass
    between random indices finish the optimisation.

All steps respect the security checks of `evaluate_sequence`.
The total running time is limited by `budget_s` (default 1000 s).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper wrappers -------------------------------------------------------
# ----------------------------------------------------------------------
def _eval(seq):
    """Call the provided evaluate_sequence (expects a Python list)."""
    return evaluate_sequence(seq.tolist())


def _make_power_law(n, exponent, offset=0.0):
    """
    Return a normalised power‑law vector a_i ∝ (i+offset+1)^exponent.
    The result sums to 1 and contains only non‑negative numbers.
    """
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return vec
    return vec / total


# ----------------------------------------------------------------------
# Smoothed max gradient utilities ---------------------------------------
# ----------------------------------------------------------------------
def _smooth_max_grad(a, lam):
    """
    Compute the gradient of the smooth‑max loss
        L(a)=max_{k}(a*a)_k ≈ (1/lam) log Σ exp(lam·(a*a)_k)
    together with the exact maximal convolution entry.
    Returns (grad, max_b, L) where

        grad_i = 2 * Σ_j w_{i+j} a_j,
        w are the soft‑max weights on the convolution vector,
        max_b = max_k (a*a)_k,
        L    = max_b + (1/lam)·log Σ exp(lam·(b_k-max_b)).
    """
    n = a.shape[0]
    # self‑convolution
    b = np.convolve(a, a)                     # length 2n‑1
    max_b = b.max()

    # soft‑max weights (stable)
    diff = b - max_b                          # ≤0
    exp_vals = np.exp(lam * diff)             # max entry → 1
    w = exp_vals / exp_vals.sum()             # Σ w = 1, length 2n‑1

    # gradient via a single convolution
    conv = np.convolve(a[::-1], w)            # length 3n‑2
    grad = 2.0 * conv[n - 1 : n - 1 + n]      # slice of length n

    # smooth‑max loss value
    L = max_b + np.log(exp_vals.sum()) / lam
    return grad, max_b, L


def _smooth_gradient_descent(seq, best_score, deadline):
    """
    Projected additive gradient descent on the smooth‑max loss.
    Keeps the vector on the probability simplex after every step.
    """
    cur = seq.copy()
    n = cur.shape[0]
    lam = 2000.0               # start smoothing parameter
    step = 0.08                # additive step size
    it = 0
    stagnant = 0               # counts iterations without improvement

    while time.time() < deadline and it < 6000:
        grad, max_b, L = _smooth_max_grad(cur, lam)

        # opposite direction of the gradient (we minimise max_b)
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s

        # evaluate cheaply via max_b of the new vector
        new_max_b = np.convolve(nxt, nxt).max()
        new_score = 2.0 * n * new_max_b

        if new_score < best_score:
            cur = nxt
            best_score = new_score
            step *= 1.03                     # modestly increase step after a win
            stagnant = 0
        else:
            # reject – shrink step and possibly increase smoothing
            step *= 0.70
            stagnant += 1

        # every 1500 iterations make the objective sharper
        if it % 1500 == 0 and it > 0:
            lam = min(lam * 1.5, 20000.0)

        # stop early if we are stuck for many iterations
        if stagnant > 2000:
            break

        it += 1
        if it % 1000 == 0:
            print(f"[smooth‑grad] it={it}, best={best_score:.6f}, step={step:.6f}, λ={lam:.1f}")

    return cur, best_score


# ----------------------------------------------------------------------
# Simple hill‑climbing --------------------------------------------------
# ----------------------------------------------------------------------
def _hill_climb(seq, best_score, deadline):
    """
    Stochastic hill‑climbing on the simplex:
    move a random fraction (≤5 %) of mass from i to j,
    keep the change only if the evaluation improves.
    """
    n = len(seq)
    cur = seq.copy()
    it = 0
    while time.time() < deadline and it < 25000:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * np.random.rand() * 0.05
        cur[i] -= delta
        cur[j] += delta
        # cheap evaluation using convolution (sum stays 1)
        new_score = 2.0 * n * np.convolve(cur, cur).max()
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it % 5000 == 0:
            print(f"[hill‑climb] it={it}, best={best_score:.6f}")
    return cur, best_score
