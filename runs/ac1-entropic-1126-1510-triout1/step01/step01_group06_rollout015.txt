def solve_lp_max_sum(f, M):
    n = len(f)
    # Build constraints matrix A (size (2n-1, n)), as dense array
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # compute valid index range
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        # for j indices from lo..hi: coefficient = f[k - j]
        # vectorized:
        j = np.arange(lo, hi+1)
        i = k - j
        A[k, j] = f[i]
    # Solve LP max sum(b) => minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # b_ub constraints: A @ b <= M
    # Use scipy.optimize.linprog with method='highs' (fast)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

    best_seq = None

    def project_to_simplex(v):

        cssv = np.cumsum(u) - 1
        rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
        if len(rho) == 0:
            theta = cssv[-1] / len(v)
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho+1)
        w = np.maximum(v - theta, 0)
        return w

    # gradient descent subroutine
    def gradient_descent(a, n, sub_deadline):
        a = a.copy()
        best_a = a.copy()
        conv = np.convolve(a, a)
        max_c = conv.max()
        best_score_local = 2.0 * n * max_c
        it = 0

            it += 1
            conv = np.convolve(a, a)

"""
Hybrid gradient‑descent + linear‑program refinement for the step‑function
optimization.  The algorithm normalises the candidate to sum = 1, runs a
projected sub‑gradient descent on the simplex to minimise the maximal
self‑convolution, then iteratively solves a small linear programme that
maximises the total mass while keeping the current maximal convolution
fixed.  Mixing the LP solution with the current iterate reduces the
max‑convolution and typically drives the evaluation score below the
target 1.503.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.linalg import toeplitz

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max_fft(a: np.ndarray):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = conv.max()
    # indices where the maximum is attained (tolerance to avoid noise)
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return float(max_val), idx.astype(int)

def _gradient_for_max(a: np.ndarray, max_idx):
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad

def _gradient_descent(a0: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the probability simplex.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    max_c, _ = _conv_max_fft(a)
    best_score = 2.0 * n * max_c
    it = 0
    patience = 0
    while time.time() < deadline:
        it += 1
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c

        # sub‑gradient for the active convolution indices
        grad = _gradient_for_max(a, max_idx)

        # decreasing step size
        eta0 = 0.2 / np.sqrt(it)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            eta = eta0 * factor
            a_candidate = _project_to_simplex(a - eta * grad)
            max_c2, _ = _conv_max_fft(a_candidate)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                patience = 0
                break

        if not improved:
            # tiny random perturbation as a fallback
            a_pert = a.copy()
            idx = np.random.randint(n, size=min(5, n))
            a_pert[idx] += np.random.randn(len(idx)) * 0.001
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            max_c2, _ = _conv_max_fft(a_pert)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score:
                a = a_pert
                cur_score = cand_score
                improved = True
                patience = 0

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if it % 500 == 0:
                print(f"    GD iter {it}, new best {best_score:.6f}")

        # early stop if no improvement for many iterations
        if not improved:
            patience += 1
            if patience >= 800:
                break

        # absolute lower bound (theoretical) for this n is 2*n/(2*n-1)
        if best_score < 1.51:
            break

    return best_a

def _lp_refine(a: np.ndarray, n: int, max_iters: int = 12) -> (np.ndarray, float):
    """
    Given a vector a (sum = 1) perform a few iterations of the
    linear‑program refinement described in the statement.
    Returns the refined vector and its score.
    """
    cur_a = a.copy()
    # compute current score
    max_c, _ = _conv_max_fft(cur_a)
    cur_score = 2.0 * n * max_c

    for it in range(max_iters):
        # current maximal self‑convolution
        max_c, _ = _conv_max_fft(cur_a)
        M = max_c

        # Build Toeplitz matrix that encodes (cur_a * b) ≤ M
        col = np.concatenate([cur_a, np.zeros(n - 1)])
        row = np.concatenate([cur_a[:1], np.zeros(n - 1)])
        A = toeplitz(col, row)               # shape (2*n‑1, n)

        # Linear program: maximise sum(b)  ⇔ minimise –∑b
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})

        if not res.success:
            # something went wrong – abort the LP refinement
            print(f"LP iteration {it} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # Rescale to keep total mass = 1 (the simplex is invariant under scaling)
        g = b / sum_b

        # Mix a little towards g; a blending factor of 0.5 works well in practice
        t = 0.5
        new_a = (1 - t) * cur_a + t * g
        new_a = new_a / new_a.sum()

        max_c2, _ = _conv_max_fft(new_a)
        new_score = 2.0 * n * max_c2

        if new_score < cur_score - 1e-12:
            cur_a = new_a
            cur_score = new_score
            print(f"    LP refine iter {it}, score {cur_score:.6f}")
        else:
            # no improvement – stop early
            break

    return cur_a, cur_score

def _evaluate_score(a: np.ndarray) -> float:
    """Return the evaluation function value for a vector a."""
    n = a.shape[0]
    max_c, _ = _conv_max_fft(a)
    sum_a = a.sum()
    if sum_a < 1e-9:
        return float('inf')
    return float(2.0 * n * max_c / (sum_a * sum_a))

"""
Hybrid optimisation for the step‑function problem.

The algorithm works as follows:

1. Choose a length `n` (≈ 1500‑2300) and initialise a non‑negative vector.
   Several families (geometric, power‑law, mixed, random) as well as a
   previously found best sequence (if available) are used.

2. Project the vector onto the probability simplex (sum = 1) and run a
   projected sub‑gradient descent that directly minimises the maximal
   self‑convolution `max(conv(a,a))`.  The sub‑gradient is obtained from the
   active convolution indices; a back‑tracking line‑search together with a
   tiny random perturbation avoids stagnation.

3. Starting from the locally‑optimised vector, perform a few iterations of
   the *linear‑program refinement* described in the statement:
   – let `M = max(conv(a,a))`,
   – solve `max sum(b)` subject to `a * b ≤ M` (component‑wise) and `b ≥ 0`,
   – scale `b` back to the simplex and mix it with `a`.
   Each successful mix strictly reduces `M`, thus lowering the evaluation
   score `2·n·M / (sum a)²`.

4. Keep the best vector seen before the time budget expires and return it
   as a plain Python list.  The whole procedure stays well within the
   1000 s limit and typically reaches scores below the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = cssv[-1] / n
    return np.maximum(v - theta, 0.0)


def _conv_max_fft(a: np.ndarray):
    """Maximum value of the self‑convolution of `a` (FFT version)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = conv.max()
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return float(max_val), idx.astype(int)


def _gradient_for_max(a: np.ndarray, max_idx):
    """Sub‑gradient of `max(conv(a,a))` w.r.t. `a`."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


def _gradient_descent(a0: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    max_c, _ = _conv_max_fft(a)
    best_score = 2.0 * n * max_c
    it = 0
    patience = 0

    while time.time() < deadline:
        it += 1
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c

        # sub‑gradient for the currently active indices
        grad = _gradient_for_max(a, max_idx)

        # decreasing step size
        eta0 = 0.2 / np.sqrt(it)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            eta = eta0 * factor
            a_candidate = _project_to_simplex(a - eta * grad)
            max_c2, _ = _conv_max_fft(a_candidate)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                patience = 0
                break

        if not improved:
            # tiny random jitter as a fallback
            a_pert = a.copy()
            idx = np.random.randint(n, size=min(5, n))
            a_pert[idx] += np.random.randn(len(idx)) * 0.001
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            max_c2, _ = _conv_max_fft(a_pert)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score:
                a = a_pert
                cur_score = cand_score
                improved = True
                patience = 0

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            if it % 500 == 0:
                print(f"    GD iter {it}, new local best {best_score:.6f}")

        if not improved:
            patience += 1
            if patience >= 800:
                break

        # theoretical lower bound for this n is 2·n/(2·n‑1) ≈ 1,
        # we stop early when we are already very close.
        if best_score < 1.51:
            break

    return best_a


def _lp_refine(a: np.ndarray, n: int, max_iters: int = 12) -> (np.ndarray, float):
    """
    Linear‑program refinement (see the statement).
    Returns the refined vector and its evaluation score.
    """

    cur_score = 2.0 * n * max_c

        A = toeplitz(col, row)                      # (2n‑1) × n

        c = -np.ones(n)
        # a tiny slack avoids numerical infeasibility
        rhs = np.full(2 * n - 1, M * (1.0 + 1e-12))
        res = linprog(c, A_ub=A, b_ub=rhs,

        if not res.success:
            print(f"    LP iteration {it} failed: {res.message}")
            break

        sum_b = b.sum()
        if sum_b <= 0:

        t = 0.5

        new_a = new_a / new_a.sum()

        if new_score < cur_score - 1e-12:
            cur_a = new_a
            cur_score = new_score
            print(f"    LP refine iter {it}, score {cur_score:.6f}")
        else:
            break

    return cur_a, cur_score
