"""
Propose a step‑function sequence that beats the arcsine construction.

Algorithm
---------
1.  Initialise with the known arcsine sequence (coefficients of (1‑z)^‑1/2)
    which gives a metric ≈ π/2 ≈ 1.5708.

2.  Perform a cheap global search over a *parametric family*  

        a_i ∝ (i + s)^‑p ,   i = 0 … n‑1

    where
        n   – length of the sequence,
        p   – power‑law exponent   (0.3 … 0.9),
        s   – positive shift (0.05 … 5).

    For each triple (n,p,s) the vector is normalised to sum 1 and the
    metric  2·n·max(conv)  is evaluated using an FFT‑based convolution.
    The best triple is kept.

3.  Refine the best parameters by stochastic local search:
    – small random perturbations of (n,p,s) drawn around the current best,
    – occasional random “mass‑swap’’ perturbations directly on the
      coefficient vector.

4.  Finally run a simple sub‑gradient descent on the *max‑convolution*
    objective.  At every iteration the index k where the convolution attains
    its maximum is identified; the gradient of that entry w.r.t. the
    coefficients is `2·a_{k‑i}` (zero outside the support).  After a step
    the vector is projected onto the probability simplex (non‑negative,
    sum = 1).  The best vector encountered during this phase is stored.

The whole procedure respects the 1000 s budget, prints intermediate
progress, and returns the best sequence found (a plain list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """
    Return the length‑n arcsine sequence (coefficients of (1‑z)^‑1/2)
    normalised so that the sum equals 1.
    """
    seq = []
    cur = 1.0  # C_0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        if i + 1 < n:
            cur = cur * (2.0 * i + 1.0) / (2.0 * (i + 1))
    seq = np.array(seq, dtype=np.float64) / total
    return seq.tolist()


def power_law_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    Return a normalised length‑n sequence proportional to (i+shift)^-p,
    i = 0 … n‑1.  The result is a numpy array with sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)
    seq = (idx + shift) ** (-p)
    s = seq.sum()
    if s == 0.0:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq /= s
    return seq


def max_convolution(a: np.ndarray) -> float:
    """
    Compute max_{k} (a*a)_k using an FFT of size a power of two.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv.max()


def metric_from_vector(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective 2·n·max(conv) assuming sum(a)=1.
    """
    n = a.shape[0]
    m = max_convolution(a)
    return float(2.0 * n * m)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a probability vector that minimises
        2·n·max(a*a) / (sum a)^2
    under the non‑negativity constraint.
    The function returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ---- 1) start from the classic arcsine construction -----------------
    best_n = 3200
    best_seq = np.array(arcsine_sequence(best_n), dtype=np.float64)
    best_score = metric_from_vector(best_seq)
    best_p = None      # not applicable for arcsine
    best_shift = None
    print(f"[init] arcsine length={best_n}  score={best_score:.6f}")

    # ---- 2) coarse grid search over power‑law families -----------------
    length_grid = list(range(500, 4001, 250))          # 500 … 4000
    p_grid = np.linspace(0.35, 0.85, 11)              # 0.35 … 0.85
    shift_grid = [0.05, 0.2, 0.5, 1.0, 2.0]            # a few shifts

    for n in length_grid:
        if time.time() >= deadline:
            break
        for p in p_grid:
            for shift in shift_grid:
                seq = power_law_sequence(n, p, shift)
                score = metric_from_vector(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    best_n = n
                    best_p = p
                    best_shift = shift
                    print(f"[coarse] n={n:<4d} p={p:.3f} s={shift:.2f} "
                          f"score={score:.6f}")

    # ---- 3) stochastic refinement of (n,p,shift) ----------------------
    # narrow search around current best parameters
    while time.time() < deadline:
        # propose a new length (±200) but stay >= 200

            if cand_sum > 0:

                if score < best_score:
                    best_score = score

                    print(f"[swap]  improved score={score:.6f}")

    # ---- 4) sub‑gradient descent on the max‑convolution objective -----
    a = best_seq.copy()
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    step0 = 0.6   # initial learning rate

    for it in range(1, 20001):
        if time.time() >= deadline:
            break

        # convolution via FFT (fast)
        A = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
        k = int(conv.argmax())
        max_val = conv[k]

        # gradient of the active entry (sub‑gradient of max)
        idx = np.arange(n, dtype=np.int64)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = a[j[mask]]
        grad *= 2.0

        step = step0 / np.sqrt(it)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:

        a_new /= s

        score = metric_from_vector(a_new)
        if score < best_score:

            best_seq = a_new.copy()
            print(f"[sgd]  iter={it:5d}  score={score:.6f}")

        a = a_new

    # ------------------------------------------------------------------
    print(f"\nBest sequence length = {len(best_seq)}  best metric = "
          f"{best_score:.6f}")
    # convert to plain Python list before returning
    return best_seq.tolist()
