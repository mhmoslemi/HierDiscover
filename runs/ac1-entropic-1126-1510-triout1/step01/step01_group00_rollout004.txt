def compute_conv_and_grad(a, beta):
    n = len(a)
    N = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=N)
    # compute auto-conv
    b_fft = a_fft * a_fft
    b = np.fft.irfft(b_fft, n=N)[:2*n-1]

    # smoothing
    max_b = b.max()
    # compute probabilities
    exp_beta = np.exp(beta * (b - max_b))
    sumexp = exp_beta.sum()
    p = exp_beta / sumexp

    # compute gradient: g_i = 2 * sum_j a_j * p_{i+j}

    conv_full = np.fft.irfft(g_fft, n=N)[:3*n-2]  # length 3n-2
    idx = (2*n - 2) - np.arange(n)
    g = 2.0 * conv_full[idx]
    return max_b, g

    # compute N
    N = 1 << ((3*n - 2).bit_length())
    # FFT of a (zero-padded)
    a_fft = np.fft.rfft(a, n=N)

    exp_beta = np.exp(beta * (b - max_b))

    idx = (2*n - 2) - np.arange(n)

"""
Propose a low‑score step‑function using a smoothed max‑convolution objective.

Algorithm
---------
1.  Initialise a probability vector with the discrete arcsine law
    a_i ∝ 1/√((i+½)(n‑i‑½)).  This distribution makes the
    self‑convolution nearly flat (its max is close to the average).

2.  Optimise the vector on the probability simplex by projected
    gradient descent on a smooth approximation of the quantity
        max_k (a * a)[k]
    The smooth approximation is the log‑sum‑exp (soft‑max) with a
    temperature β that is slowly increased (β‑annealing).  The gradient
    of the soft‑max can be written as a convolution, which we evaluate
    with a single FFT of length at least 3 n (linear convolution).

3.  After each gradient step the vector is projected back onto the
    simplex {x ≥ 0, Σx = 1} using the classic O(n) Euclidean‑projection
    algorithm.

4.  The loop is run for as many iterations as the time budget allows.
    The best sequence (according to the true score
    2·n·max(a*a)) seen so far is kept.  Several lengths are tried;
    the overall best is returned as a plain Python list.

The whole routine respects the 1000 s limit, uses only NumPy, and avoids
any filesystem or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law of length n (probability vector, sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _max_conv_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    For a probability vector a (Σa = 1) compute
        max_b = max_k (a * a)[k]
    and the gradient of the smooth objective
        G(a) = (1/β) log Σ_k exp(β·(a*a)[k]).

    Returns (max_b, grad) where grad has shape a.shape.
    """
    n = a.shape[0]
    # Linear convolution length needed for a * a is 2n‑1.
    # For the gradient we need linear convolution of a (len n) with
    # p_rev (len 2n‑1) → length 3n‑2.
    N = 1 << ((3 * n - 2).bit_length())  # next power of two ≥ 3n‑2

    # FFT of a (zero‑padded to length N)
    a_fft = np.fft.rfft(a, n=N)

    # ----- auto‑convolution a*a -----
    b_fft = a_fft * a_fft
    b_full = np.fft.irfft(b_fft, n=N)
    b = b_full[: 2 * n - 1]               # exact linear convolution
    max_b = b.max()

    # ----- soft‑max probabilities -----
    # Stabilise by subtracting max_b
    exp_beta = np.exp(beta * (b - max_b))
    sumexp = exp_beta.sum()
    if sumexp == 0.0:                     # should not happen, but guard anyway
        p = np.full_like(b, 1.0 / b.size)
    else:
        p = exp_beta / sumexp             # length 2n‑1, Σp = 1

    # ----- gradient via convolution a * p_rev -----
    p_rev = p[::-1]                       # reversed probabilities
    p_rev_fft = np.fft.rfft(p_rev, n=N)
    g_fft = a_fft * p_rev_fft
    conv_full = np.fft.irfft(g_fft, n=N)
    conv_full = conv_full[: 3 * n - 2]    # linear convolution a * p_rev
    idx = (2 * n - 2) - np.arange(n)      # pick the slice that yields Σ_j a_j·p_{i+j}
    grad = 2.0 * conv_full[idx]           # factor 2 from ∂(a*a)/∂a

    return max_b, grad


# ----------------------------------------------------------------------
def _optimise_one_length_smooth(
    n: int,
    deadline: float,
    beta_start: float = 0.5,
    beta_end: float = 30.0,
    eta0: float = 0.5,
    max_iters: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Projected (smoothed‑max) gradient descent for a fixed length n.
    Returns the best sequence seen and its true score
        score = 2·n·max(a*a)   (Σa = 1).
    """
    a = _arcsine_distribution(n)          # start from arcsine
    best_seq = a.copy()
    # initial true score (exact max of linear convolution)
    b0 = np.convolve(a, a)                # small n for this one call is fine
    best_score = 2.0 * n * b0.max()

    beta = beta_start
    beta_inc = (beta_end - beta_start) / max_iters
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- gradient of smoothed objective -----
        max_b, grad = _max_conv_and_grad(a, beta)

        # ----- true score for bookkeeping -----
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] len={n} iter={it} score={best_score:.6f}"
            )

        # ----- step size (diminishing) -----
        eta = eta0 / np.sqrt(it)

        # ----- gradient step + projection onto simplex -----
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        # ----- anneal β -----
        beta = min(beta_end, beta + beta_inc)

        # ----- occasional jitter to avoid stagnation -----
        if it % 500 == 0:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a list of non‑negative heights with a low evaluation score.
    The routine tries several lengths (preferring larger ones) and runs a
    projected gradient descent with a smoothed max objective within the
    allocated time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # Determine a set of lengths to explore.
    # If a previous construction exists, start from its length.
    try:
        n0 = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n0 = 5000

    # Build a list ranging from a bit smaller than n0 to considerably larger.
    factors = (0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0)
    lengths = sorted({max(2000, min(50000, int(n0 * f))) for f in factors}, reverse=True)

    # Ensure we also try a very large size (if time permits)
    if 20000 not in lengths:
        lengths.insert(0, 20000)   # try 20000 first

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # Allocate a time slice for this length (roughly proportional to remaining length count)
        remaining = len(lengths) - lengths.index(n)
        remaining_time = deadline - time.time()
        slice_deadline = time.time() + max(0.1, remaining_time * 0.7 / remaining)

        print(f"--- Optimising length {n} (time slice until {slice_deadline - start:.1f}s) ---")
        seq, score = _optimise_one_length_smooth(
            n=n,
            deadline=slice_deadline,
            beta_start=0.5,
            beta_end=30.0,
            eta0=0.6,
            max_iters=20_000,
        )

        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(f"*** New global best -> length={n} score={overall_best_score:.6f}")

        # Early exit if we already beat the target bound
        if overall_best_score <= 1.503:
            break

    # Fallback in the unlikely event that nothing was produced
    if overall_best_seq is None:
        n_fallback = 500
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Final verification with the provided evaluate_sequence (may be expensive for huge n)
    try:
        final_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"Final verified score: {final_score:.6f} (length={len(overall_best_seq)})")
    except Exception as exc:
        print(f"Could not run final verification: {exc}")

    return overall_best_seq.tolist()
