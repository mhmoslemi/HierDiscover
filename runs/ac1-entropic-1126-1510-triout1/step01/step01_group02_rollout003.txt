"""
step_optimize.py

A fast simulated‑annealing / Adam hybrid for the non‑negative step‑function
minimisation problem defined by ``evaluate_sequence``.

Main ideas
-----------

* Work on the probability simplex (∑a = 1, a_i ≥ 0).  The evaluation
  function is homogeneous, therefore fixing the sum does not affect the
  score but removes the trivial scaling freedom.
* Most of the objective is controlled by the *largest* entry of the
  self‑convolution.  We therefore use the sub‑gradient of that entry
  (g_i = 2·a_{k‑i}) and feed it to an Adam optimiser – this gives an
  adaptive stepsize without manual tuning.
* To escape local minima we interleave Adam updates with cheap
  random “pair‑swap’’ moves (mass is transferred between two coordinates
  while keeping the sum exactly 1).  Worse candidates are accepted with a
  Metropolis probability that slowly cools down.
* If the search stagnates for a while we perturb the current best point
  by mixing in a fresh random simplex point – this is a cheap restart
  that retains most of the accumulated knowledge.
* The routine respects the time budget (default 1000 s).  The best
  sequence found so far is returned as a plain Python list.

The algorithm is deliberately simple, fast and stateless apart from the
global ``height_sequence_1`` (used as a strong starting point when
available).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Random pairwise mass move (preserves the sum exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two distinct coordinates."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # zero‑mean proposal drawn from a symmetric interval
    delta = (np.random.rand() - 0.5) * sigma

    # enforce non‑negativity
    if delta > 0.0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Adam optimiser for the sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _adam_step(seq: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               it: int,
               beta1: float,
               beta2: float,
               eps: float,
               lr: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One Adam step on the sub‑gradient of the current maximal convolution entry.
    Returns (new_seq, new_m, new_v).
    """
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    # gradient of the active convolution entry
    idx = np.arange(seq.shape[0])
    j = k - idx
    mask = (j >= 0) & (j < seq.shape[0])
    g = np.zeros_like(seq)
    g[mask] = 2.0 * seq[j[mask]]

    # Adam moments
    m_new = beta1 * m + (1.0 - beta1) * g
    v_new = beta2 * v + (1.0 - beta2) * (g * g)

    # bias‑corrected moments
    m_hat = m_new / (1.0 - beta1 ** it)
    v_hat = v_new / (1.0 - beta2 ** it)

    step = lr / (np.sqrt(v_hat) + eps)
    new_seq = seq - step * m_hat
    new_seq = _project_simplex(new_seq)

    return new_seq, m_new, v_new


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (a step function) that tries
    to minimise ``evaluate_sequence``.

    The optimiser mixes Adam gradient steps on the worst convolution entry
    with random pairwise mass swaps, using a simple simulated‑annealing
    acceptance rule and occasional restarts.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, budget_s - 1.0)

    # --------------------------------------------------------------
    # Initialisation – try to reuse the best known construction if present
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore[name‑defined]
        n = base.size
    except Exception:
        # fallback – a modestly large uniform vector
        n = 8000
        base = np.full(n, 1.0 / n, dtype=float)

    # Normalise onto the simplex (sum = 1)
    base_sum = base.sum()
    if base_sum <= 0.0:
        base = np.full_like(base, 1.0 / n)
    else:
        base = base / base_sum

    # Ensure we are on the simplex (projecting also removes tiny negatives)
    current = _project_simplex(base.copy())
    best = current.copy()

    # ------------------------------------------------------------------
    # Initial scores
    # ------------------------------------------------------------------
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:
        best_score = float("inf")
    current_score = best_score

    # Adam state
    m = np.zeros_like(current)
    v = np.zeros_like(current)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    # Stagnation tracking
    last_improve_iter = 0
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # Linear cooling schedule for the Metropolis acceptance rule
        # ------------------------------------------------------------------
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temperature = max(1e-6, 1.0 - frac)          # goes from 1 → 0

        # ------------------------------------------------------------------
        # Choose a move type
        # ------------------------------------------------------------------
        if np.random.rand() < 0.65:
            # Adam gradient step on the active convolution entry
            candidate, m, v = _adam_step(
                current, m, v, it, beta1, beta2, eps, lr
            )
        else:
            # Random pairwise mass transfer (exploration)
            sigma = 0.4 / np.sqrt(it)               # slowly shrinking
            candidate = _random_pair_move(current, sigma)

        # ------------------------------------------------------------------
        # Evaluate the proposal (expensive part)
        # ------------------------------------------------------------------
        try:
            cand_score = evaluate_sequence(candidate.tolist())
        except Exception:
            # Defensive – should never happen because we keep everything valid
            continue

        # Keep the overall best
        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()
            last_improve_iter = it
            print(f"[Iter {it:6d}] New best score: {best_score:.6f}")

        # ------------------------------------------------------------------
        # Metropolis acceptance (allows occasional uphill moves)
        # ------------------------------------------------------------------
        if cand_score < current_score:
            accept = True
        else:
            # probability = exp(-(Δ)/T)
            prob = np.exp(-(cand_score - current_score) / temperature)
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score
        # ------------------------------------------------------------------
        # Restart / perturbation if we have been idle for a while
        # ------------------------------------------------------------------
        if it - last_improve_iter > 6000:
            # Tiny random walk around the best point
            perturb = np.random.rand(n)
            perturb = perturb / perturb.sum()
            current = _project_simplex(0.98 * best + 0.02 * perturb)
            current_score = evaluate_sequence(current.tolist())
            # reset the Adam moments – otherwise old momentum may drive us back
            m[:] = 0.0
            v[:] = 0.0
            last_improve_iter = it

    # ----------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ----------------------------------------------------------------------
    return best.tolist()
