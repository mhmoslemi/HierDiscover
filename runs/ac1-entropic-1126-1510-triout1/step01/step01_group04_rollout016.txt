    cur_score = best_score
    T0 = 0.1
    T_decay = 0.9995
    T = T0
    it = 0

    while time.time() < deadline and it < 200000:
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # amount to move = random fraction up to maybe 0.2
        delta = seq[i] * np.random.rand() * 0.2
        if delta == 0:
            continue
        seq[i] -= delta
        seq[j] += delta
        # renormalize to avoid drift (computationally cheap)
        # after each move, because we only moved mass, sum remains unchanged; but due to rounding we keep.
        # Evaluate
        new_score = evaluate_sequence(seq.tolist())
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score)/max(T,1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = seq.copy()
        else:
            # reject, revert
            seq[i] += delta
            seq[j] -= delta
        # temperature decay
        T *= T_decay
        it += 1
        if it % 5000 == 0:
            print(f"[anneal] iter {it}, T={T:.6f}, cur_score={cur_score:.6f}, best_score={best_score:.6f}")

    # Stage 3: Exponentiated gradient descent to flatten the peak
    seq = best_seq.copy()
    cur_score = best_score
    eta = 0.1

    while time.time() < deadline and grad_iters < 5000:
        conv = np.convolve(seq, seq)
        max_idx = int(conv.argmax())
        # compute subgradient
        grad = np.zeros_like(seq)

def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 7000   # moderate size

    # Norm: The evaluate_sequence returns 2 * n * max(conv) / (sum^2)
    # Since sum large? Actually our sequences sum to 1.

    # compute uniform vector

    offsets = np.linspace(0.0, 2.0, 41)       # step 0.05
    mix_ratios = [0.5, 0.7, 0.85, 0.95, 1.0]  # 1.0 = pure power law

    best_score = float('inf')
    best_seq = None
    best_params = None

    for off in offsets:
        for a in exponents:
            # compute power-law base vector
            idx = np.arange(1, n+1, dtype=np.float64) + off
            base = idx ** a
            base_sum = base.sum()
            if base_sum == 0:
                continue
            base = base / base_sum
            # mixture
            for lam in mix_ratios:
                seq = lam * base + (1 - lam) * uniform
                # ensure non-negative
                # conv evaluation
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (a, off, lam)
                # check time
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break
    print(f"[grid] best exponent {best_params[0]:.4f}, offset {best_params[1]:.4f}, mix {best_params[2]:.2f}, score {best_score:.6f}")

    # Hill climbing with simulated annealing style
    seq = best_seq.copy()
    cur_score = best_score

    T = 0.2
    decay = 0.9993
    iter_cnt = 0
    while time.time() < deadline and iter_cnt < 200000:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or seq[i] == 0:
            continue
        # move fraction up to 30%
        delta = seq[i] * np.random.rand() * 0.3
        seq[i] -= delta
        seq[j] += delta
        # evaluate
        new_score = evaluate_sequence(seq.tolist())
        # acceptance
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / max(T, 1e-12)):

            if new_score < best_score:
                best_score = new_score

            seq[j] -= delta

        T *= decay
        iter_cnt += 1

"""
Propose a step‑function (non‑negative vector) with a low value of
`evaluate_sequence`.  The algorithm consists of four stages:

1. **Extensive grid search** over a power‑law shape a_i ∝ (i+offset)^α
   (α∈[−0.80,−0.45]) and a convex combination with the uniform
   distribution.  The best candidate is kept.
2. **Greedy hill‑climbing** – repeatedly move a random fraction of mass
   from one coordinate to another, keeping the total sum constant.
   Only improving moves are accepted.
3. **Exponentiated‑gradient refinement** – using a soft‑max of the
   largest convolution entries we compute a smooth sub‑gradient of the
   max‑convolution, take a multiplicative (mirror‑descent) step and
   project back onto the probability simplex.
4. **Final polishing** – a short round of tiny random mass transfers.

All stages respect the 1000 s budget and the security checks of the
provided `evaluate_sequence`.  The returned list contains plain Python
floats (the evaluator expects a list, not a NumPy array).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _eval(seq: np.ndarray) -> float:
    """
    Wrapper around the external `evaluate_sequence` that accepts a NumPy
    array and returns the float score.
    """
    # `evaluate_sequence` expects a list of float
    return evaluate_sequence(seq.tolist())


def _power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised power‑law vector: a_i ∝ (i + 1 + offset)^exponent,
    i = 0,…,n−1.  The vector is scaled to sum to 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        # degenerate exponent – return uniform as a fallback
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vec / total


def _initial_grid_search(n: int,
                         deadline: float,
                         exponents: np.ndarray,
                         offsets: np.ndarray,
                         mix_ratios: np.ndarray) -> tuple[np.ndarray, float, tuple]:
    """
    Exhaustive search over (exponent, offset, mixture) and return the
    best vector, its score and the parameters that produced it.
    """
    best_score = float('inf')
    best_vec = None
    best_params = None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    for off in offsets:
        for exp in exponents:
            base = _power_law(n, exp, offset=off)
            for lam in mix_ratios:
                vec = lam * base + (1.0 - lam) * uniform
                score = _eval(vec)
                if score < best_score:
                    best_score = score
                    best_vec = vec.copy()
                    best_params = (exp, off, lam)
            if time.time() > deadline:
                return best_vec, best_score, best_params
        if time.time() > deadline:
            break
    return best_vec, best_score, best_params


def _greedy_hill_climb(vec: np.ndarray,
                       cur_score: float,
                       deadline: float,
                       max_iters: int = 200_000,
                       max_frac: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfer on the simplex.  Only strictly improving moves
    are kept; the total mass (sum) stays constant.
    """
    n = len(vec)
    best_vec = vec.copy()
    best_score = cur_score
    cur_vec = vec.copy()
    cur_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = np.random.randint(n, size=2)
        if i == j or cur_vec[i] <= 0.0:
            continue
        # move up to `max_frac` of the source mass
        delta = cur_vec[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue
        cur_vec[i] -= delta
        cur_vec[j] += delta

        new_score = _eval(cur_vec)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = cur_vec.copy()
        else:
            # revert
            cur_vec[i] += delta
            cur_vec[j] -= delta

        it += 1
        if it % 5_000 == 0:
            print(f"[hill‑climb] it={it:,}, cur_score={cur_score:.6f}, best_score={best_score:.6f}")
    return best_vec, best_score


def _exp_gradient_refine(vec: np.ndarray,
                         cur_score: float,
                         deadline: float,
                         max_iters: int = 3_000,
                         top_frac: float = 0.99,
                         beta: float = 20.0,
                         eta0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) step that uses a soft‑max over
    the largest convolution entries.  The gradient is formed from the
    top‑K entries (those within `top_frac` of the maximum).  After each
    multiplicative update the vector is re‑normalised onto the simplex.
    """
    n = len(vec)
    cur_vec = vec.copy()
    cur_score = cur_score
    best_vec = vec.copy()
    best_score = cur_score
    eta = eta0

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(cur_vec, cur_vec)
        max_val = conv.max()
        # indices whose conv value is close to the maximum
        thresh = max_val * top_frac
        top_idx = np.where(conv >= thresh)[0]
        if top_idx.size == 0:
            break

        # soft‑max weights over the selected indices
        w = np.exp(beta * (conv[top_idx] - max_val))
        w /= w.sum()

        # build the (sub)gradient
        grad = np.zeros_like(cur_vec)
        for k, wk in zip(top_idx, w):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * wk * cur_vec[k - idx]

        # exponentiated (mirror) update
        factor = np.exp(-eta * grad)
        cur_vec = cur_vec * factor

        # projection onto the probability simplex (non‑negative + sum=1)
        total = cur_vec.sum()
        if total <= 0.0:
            break
        cur_vec /= total

        new_score = _eval(cur_vec)
        if new_score < cur_score:
            cur_score = new_score
            best_vec = cur_vec.copy()
            best_score = new_score
            eta *= 1.02               # a slight acceleration when we improve
        else:
            eta *= 0.95                # otherwise shrink the step size

        it += 1
        if it % 200 == 0:
            print(f"[exp‑grad] it={it:,}, eta={eta:.5f}, cur_score={cur_score:.6f}, best_score={best_score:.6f}")

    return best_vec, best_score


def _final_polish(vec: np.ndarray,
                  cur_score: float,
                  deadline: float,
                  max_iters: int = 5_000,
                  max_frac: float = 0.10) -> tuple[np.ndarray, float]:
    """
    Very small random mass moves to possibly escape a final tiny plateau.
    """
    n = len(vec)
    cur_vec = vec.copy()
    cur_score = cur_score
    best_vec = vec.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = np.random.randint(n, size=2)
        if i == j or cur_vec[i] <= 0.0:
            continue
        delta = cur_vec[i] * np.random.rand() * max_frac
        cur_vec[i] -= delta
        cur_vec[j] += delta

        new_score = _eval(cur_vec)
        if new_score < cur_score:
            cur_score = new_score
            best_vec = cur_vec.copy()
            best_score = new_score
        else:
            # revert the move
            cur_vec[i] += delta
            cur_vec[j] -= delta

        it += 1
        if it % 1_000 == 0:
            print(f"[polish] it={it:,}, cur_score={cur_score:.6f}, best_score={best_score:.6f}")

    return best_vec, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluation metric.  The function respects the time budget (`budget_s`),
    using a blend of grid search, hill‑climbing, exponentiated‑gradient
    refinement and a final polishing phase.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    # Size of the vector – a few thousand gives a good trade‑off.
    n = 6000

    # ------------------------------------------------------------------
    # 1) Grid search over exponent, offset and mixture with the uniform
    # ------------------------------------------------------------------
    exponents = np.linspace(-0.80, -0.45, 71)          # step ≈0.005
    offsets   = np.linspace(0.0, 2.0, 41)            # step 0.05
    mix_ratios = np.array([0.5, 0.7, 0.85, 0.95, 1.0])  # λ for λ·power + (1‑λ)·uniform

    best_vec, best_score, best_params = _initial_grid_search(
        n=n,
        deadline=deadline,
        exponents=exponents,
        offsets=offsets,
        mix_ratios=mix_ratios,
    )
    if best_vec is None:
        # Should never happen, but fallback to uniform
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _eval(best_vec)
        best_params = (None, None, None)

    print(f"[grid] best exponent={best_params[0]:.4f}, "
          f"offset={best_params[1]:.4f}, mix={best_params[2]:.2f}, "
          f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Greedy hill‑climbing (large moves)
    # ------------------------------------------------------------------
    best_vec, best_score = _greedy_hill_climb(
        vec=best_vec,
        cur_score=best_score,
        deadline=deadline,
        max_iters=150_000,
        max_frac=0.30,
    )
    print(f"[hill‑climb] score after climbing={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Exponentiated‑gradient refinement (soft‑max on top conv entries)
    # ------------------------------------------------------------------
    best_vec, best_score = _exp_gradient_refine(
        vec=best_vec,
        cur_score=best_score,
        deadline=deadline,
        max_iters=2_500,
        top_frac=0.99,
        beta=20.0,
        eta0=0.05,
    )
    print(f"[exp‑grad] score after gradient refinement={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final tiny‑step polishing
    # ------------------------------------------------------------------
    best_vec, best_score = _final_polish(
        vec=best_vec,
        cur_score=best_score,
        deadline=deadline,
        max_iters=5_000,
        max_frac=0.10,
    )
    print(f"[polish] final score={best_score:.6f}")

    # Convert to plain Python list of floats (the evaluator expects this)
    final_sequence = [float(v) for v in best_vec.tolist()]

    print(f"Returning sequence of length {len(final_sequence)} with evaluation score {best_score:.6f}")
    return final_sequence
