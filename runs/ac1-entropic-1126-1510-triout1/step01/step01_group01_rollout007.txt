"""
Propose a low‑score step‑function using a projected Adam‑style sub‑gradient
method with occasional random perturbations and block‑search refinements.

The objective (after fixing the ℓ¹–norm Σa = √(2 n)) is simply
    max_k (a * a)[k] .
We minimise this by iteratively
  1. computing the self‑convolution via an FFT,
  2. building a sub‑gradient from all (or almost all) active constraints
     (indices where the convolution is near its current maximum),
  3. performing an Adam update,
  4. projecting the result back onto the simplex {a ≥ 0, Σa = √(2 n)}.
The Adam optimiser adapts the step‑size per coordinate, which helps to
escape plateaus that slowed the earlier simple projected gradient descent.
Every few thousand iterations a tiny isotropic Gaussian perturbation is added
to provoke exploration, and after the main optimisation a cheap “block”
search is performed to catch simple structured improvements.

The routine respects the 1000 s budget, prints progress, and finally returns
the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):               # all entries would be zero
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using a real FFT (fast O(n log n))."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two ≥ 2n‑1
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray,
                          rel_eps: float = 1e-3) -> np.ndarray:
    """
    Compute a sub‑gradient of the functional max_k (a*a)[k].
    Active set = indices where conv >= max - rel_eps*max.
    """
    n = a.shape[0]
    max_val = conv.max()
    # active indices (allow a few of the top entries)
    active = np.where(conv >= max_val - rel_eps * max_val)[0]

    grad = np.zeros_like(a)

    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        partner = k - i_range
        grad[i_range] += 2.0 * a[partner]

        # diagonal correction (i == partner) – counted only once, not twice
        if (k % 2) == 0:
            mid = k // 2
            if i_start <= mid <= i_end:
                grad[mid] -= a[mid]

    return grad, max_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
        evaluate_sequence(a)
    using an Adam‑based projected sub‑gradient optimiser.
    """
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Length choice – prefer the length of the known construction.
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    # Initial vector (reference or fresh random)
    if have_ref and np.random.rand() < 0.5:
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    else:
        a = np.random.rand(n)

    target_sum = math.sqrt(2.0 * n)               # Σa = √(2 n)
    a = _project_to_simplex(a, target_sum)

    # Adam state
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # Base learning rate – about the size of a typical entry
    base_lr = target_sum * 0.2 / n

    # Book‑keeping
    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_score = conv.max()                      # because denominator = 2n
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    it = 0
    last_improve_it = 0
    # Relative tolerance for the active set – tighter → more constraints
    rel_eps = 1e-3

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) convolution and sub‑gradient
        conv = _conv_self_fft(a)
        grad, cur_max = _subgradient_max_conv(a, conv, rel_eps=rel_eps)

        # 2) Adam moment updates
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # 3) step (learning‑rate schedules a bit over time)
        #    smaller step as we approach the time limit
        time_frac = (deadline - time.time()) / budget_s
        lr = base_lr * (0.5 + 0.5 * time_frac)   # slowly decay to half size

        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # 4) evaluate
        conv_new = _conv_self_fft(a_new)
        new_max = conv_new.max()

        # 5) keep the best ever seen
        if new_max < best_score:
            best_score = new_max
            best_a = a_new.copy()
            last_improve_it = it
            print(f"[{it:08d}] improved score = {best_score:.6f}")

        # 6) accept the new point (always, since projection forces feasibility)
        a = a_new

        # 7) occasional exploration
        if (it % 4000 == 0) or (it - last_improve_it > 20000):
            # add a tiny isotropic perturbation
            noise = np.random.randn(n) * target_sum * 5e-4
            a = _project_to_simplex(a + noise, target_sum)
            # reset Adam moments to avoid stale directions after a jump
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{it:08d}] random perturbation")
            last_improve_it = it

        # 8) adapt the active‑set tolerance: tighten as we improve
        if it % 5000 == 0:
            rel_eps = max(5e-4, rel_eps * 0.9)   # never go below 5e‑4

    # ------------------------------------------------------------------
    # After the main AdaGrad run we perform a cheap block‑search.
    block_trials = 250
    for _ in range(block_trials):
        nb = np.random.randint(2, 7)                       # number of blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n, dtype=float)
        for blk, vval in zip(blocks, vals):
            cand[blk] = vval
        cand = _project_to_simplex(cand, target_sum)
        conv_c = _conv_self_fft(cand)
        sc = conv_c.max()
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved score = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
