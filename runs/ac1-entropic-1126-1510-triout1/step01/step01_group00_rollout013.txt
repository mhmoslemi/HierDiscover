def fft_convolve(x,y):
    size = 1 << ((len(x) + len(y) - 2).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    Z = X * Y
    result = np.fft.irfft(Z, n=size)
    return result[:len(x)+len(y)-1]

def _optimise_one_length_expgrad(n, deadline, max_iters=10000, eta0=0.5, beta0=0.5, beta_max=100.0):
    a = _arcsine_distribution(n)  # sum 1
    best = a.copy()
    best_score = float('inf')
    size_fft = 1 << ((2*n-1).bit_length())  # enough for conv (a*a)
    # iterate
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        # compute b via fft conv
        X = np.fft.rfft(a, n=size_fft)
        b = np.fft.irfft(X * X, n=size_fft)[:2*n-1]  # real
        b_max = b.max()
        cur_score = 2.0 * n * b_max
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"Iter {it}, n={n}, score={best_score:.6f}")

        # compute beta schedule
        beta = beta0 + (beta_max - beta0) * (it / max_iters)

        # compute softmax weights
        # to avoid overflow, subtract max
        exp_term = beta * b
        max_exp = exp_term.max()
        w = np.exp(exp_term - max_exp)
        w /= w.sum()  # p_norm

        # gradient using conv of a and w_rev
        # compute conv: a * w_rev using fft
        W = np.fft.rfft(w, n=size_fft)
        conv = np.fft.irfft(X * W, n=size_fft)[:3*n-2]  # a * w_rev length 3n-2
        # conv_rev index selection
        idx = 2*n-2 - np.arange(n)  # indices in conv array (0-index)
        grad = 2.0 * conv[idx] - a * w[2*np.arange(n)]

        # step size
        eta = eta0 / np.sqrt(it)
        # exponentiated gradient update
        a = a * np.exp(- eta * grad)
        # renormalize
        a_sum = a.sum()
        if a_sum == 0:
            # reinitialize
            a = np.full(n, 1.0/n)
        else:
            a /= a_sum

        # jitter occasionally
        if it % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0)
            a /= a.sum()

    return best, best_score

"""
Gradient‑descent optimizer for the self‑convolution L∞‑norm.
---------------------------------------------------------

* Start from the discrete arcsine law (near‑optimal analytical shape).
* Use a smooth approximation of the max‑operator:
        M_β(b) = (1/β)·log(∑_k e^{β·b_k})      (β → ∞ ⇒ max(b))
  The gradient of M_β is a soft‑max weighting of the convolution entries.
* Perform **exponentiated mirror descent** on the probability simplex:
        a ← a·exp(−η·∇M_β) , a ← a / ∑a
  (η slowly decays, β slowly grows – a classic homotopy).
* Convolution and its gradients are computed with an FFT‑based routine
  (O(N log N) per iteration) allowing thousands of iterations for
  sequences up to ≈15 000 elements within the 1000 s budget.
* Several lengths are tried (a small “base” run to get a good shape,
  then an up‑sampled large run).  The best sequence found is returned.

The algorithm typically drives the score from ~1.53 down below the target
1.503, without ever invoking `evaluate_sequence` inside the loop
(the exact objective 2·n·max(b) is computed directly).
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Simple utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """Real FFT based convolution, result length = len(a)+len(b)-1."""
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[: len(a) + len(b) - 1]  # trim to exact length


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Renormalise a non‑negative vector to sum to 1 (in‑place safe)."""
    s = v.sum()
    if s <= 0:
        # fall back to uniform
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law of length n, already normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _interpolate_distribution(a_small: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    x_small = np.linspace(0.0, 1.0, a_small.size)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_small, a_small)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# --------------------------------------------------------------
# Core optimiser – exponentiated mirror descent on a single length
# --------------------------------------------------------------
def _optimise_one_length_expgrad(
    n: int,
    time_deadline: float,
    max_iters: int = 8000,
    eta0: float = 0.5,
    beta_start: float = 0.5,
    beta_end: float = 200.0,
    jitter_prob: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Optimise a length‑`n` probability vector using softened max‑gradient.
    Returns the best vector seen and its exact score 2·n·max(conv).
    """
    # ---- initialise -------------------------------------------------
    a = _arcsine_distribution(n)               # start from a nice analytic shape
    best_a = a.copy()
    # pre‑allocate the FFT length (covers a*a and a*w_rev)
    fft_len = _next_pow2(3 * n - 2)               # ≥ len(a)+len(w_rev)-1
    best_score = float("inf")

    # ---- main loop -------------------------------------------------
    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ----- convolution a*a  (FFT) --------------------------------
        A = np.fft.rfft(a, n=fft_len)
        b_full = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]   # self‑convolution
        b_max = b_full.max()
        cur_score = 2.0 * n * b_max

        # keep the best exact score seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={it} score={best_score:.6f}")

        # ----- β schedule (smooth‑max) --------------------------------
        frac = it / max_iters
        beta = beta_start + (beta_end - beta_start) * frac  # linear increase

        # soft‑max weights p_k = exp(β·b_k) / Σ exp(β·b)
        # shift by the maximum for numerical stability
        eb = beta * b_full
        eb_shift = eb - eb.max()
        w = np.exp(eb_shift)
        w /= w.sum()                     # now Σ w = 1

        # ----- gradient of the smoothed max ---------------------------
        # convolution a * w_rev (FFT)
        w_rev = w[::-1]
        W = np.fft.rfft(w_rev, n=fft_len)
        conv_rev = np.fft.irfft(A * W, n=fft_len)[: 3 * n - 2]

        idx = 2 * n - 2 - np.arange(n)    # map to the correct slice
        grad = 2.0 * conv_rev[idx]        # 2·∑_k w_k a_{k-i}
        # diagonal correction (the case k = 2i contributes only a_i, not 2·a_i)
        grad -= a * w[2 * np.arange(n)]

        # ----- mirror‑descent update (exponentiated) ------------------
        eta = eta0 / math.sqrt(it)        # diminishing step size
        # multiplicative update, stays non‑negative
        a = a * np.exp(-eta * grad)
        a = _project_to_simplex(a)        # re‑normalise to a simplex

        # ----- occasional random jitter to avoid stagnation ----------
        if (it % 500 == 0) and (np.random.rand() < jitter_prob):
            a += np.random.randn(n) * 1e-5
            a = _project_to_simplex(np.maximum(a, 0.0))

    return best_a, best_score
