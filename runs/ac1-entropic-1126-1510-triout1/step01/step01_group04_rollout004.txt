"""
Propose a low‑score step‑function.

The algorithm proceeds in three stages, all running inside the given time
budget (default 1000 s).

1. **Coarse grid search** – a dense scan over power‑law exponents and a
   small offset (a_i ∝ (i+offset+1)^α) using a fast FFT‑based convolution.
   The best candidate from the grid is kept as a warm start.

2. **Stochastic hill‑climbing on the simplex** – repeatedly move a random
   fraction of mass from one index to another while keeping the sum equal
   to 1.  The move is kept if it improves the fast score
   `2·n·max(conv(a,a))`.  The step size starts relatively large and is
   gradually reduced; the loop stops when the time budget expires.

3. **Projected sub‑gradient descent** – at the current maximal convolution
   index k the sub‑gradient of the max‑entry is known analytically:
   g_i = 2·a_{k‑i} (when the index is valid).  A projected gradient step
   onto the probability simplex is taken with an adaptive step size.
   This stage finishes the refinement.

All internal evaluations use an FFT‑based convolution (`conv_max_fast`)
which is orders of magnitude faster than `np.convolve`.  The final
sequence is finally checked with the provided `evaluate_sequence`
function and returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def conv_max_fast(seq: np.ndarray) -> float:
    """Return max(conv(seq, seq)) using a power‑of‑two FFT size."""
    n = seq.shape[0]
    size = 1 << int(np.ceil(np.log2(2 * n - 1)))      # smallest power of two ≥2n‑1
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[: 2 * n - 1]
    return float(conv.max())


def fast_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv)  (the denominator is 1 because we keep sum(seq)=1)."""
    n = seq.shape[0]
    return 2.0 * n * conv_max_fast(seq)


def make_power_law(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """Create a normalised decreasing power‑law vector a_i ∝ (i+offset+1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vec / total


def make_exponential(n: int, base: float, offset: float = 0.0) -> np.ndarray:
    """Create a normalised exponential decay a_i ∝ base^{i+offset} (0<base<1)."""
    idx = np.arange(n, dtype=np.float64) + offset
    vec = base ** idx
    total = vec.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vec / total


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def hill_climb(
    seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 400_000,
    start_frac: float = 0.20,
    end_frac: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing on the probability simplex.

    A random fraction of the mass at a source index is moved to a destination
    index.  The move is kept only if it strictly improves the fast score.
    The allowed fraction shrinks linearly from ``start_frac`` to ``end_frac``
    over the course of the run.
    """
    cur = seq.copy()
    n = cur.shape[0]
    it = 0
    while time.time() < deadline and it < max_iter:
        # linear annealing of the transferred fraction
        frac = start_frac - (start_frac - end_frac) * (it / max_iter)
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * np.random.rand() * frac
        cur[i] -= delta
        cur[j] += delta

        score = fast_score(cur)
        if score < best_score:
            best_score = score
            # keep the improvement
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1
        if it % 50_000 == 0:
            print(f"[hill‑climb] iter {it:,} – best_score = {best_score:.6f}")

    return cur, best_score


def gradient_descent(
    seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 5_000,
    step0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the maximal convolution entry.
    The sub‑gradient of max(conv) w.r.t. a_i is 2·a_{k‑i} where k is the
    maximising index.
    """
    cur = seq.copy()
    n = cur.shape[0]
    step = step0
    it = 0

    while time.time() < deadline and it < max_iter:
        # full convolution for locating the active index
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft = np.fft.rfft(cur, n=size)
        conv = np.fft.irfft(fft * fft, n=size)[: 2 * n - 1]

        max_val = conv.max()
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score

        # index of the current maximum
        k = int(conv.argmax())

        # sub‑gradient for the max‑entry
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur)
        grad[idx] = 2.0 * cur[k - idx]

        # Gradient step + simplex projection
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:
            break
        nxt /= s

        new_score = fast_score(nxt)
        if new_score < best_score:
            cur = nxt
            best_score = new_score
            step *= 0.99            # slowly shrink step after success
        else:
            step *= 0.90            # be more cautious after failure

        it += 1
        if it % 200 == 0:
            print(f"[gradient] iter {it:,} – best_score = {best_score:.6f}")

    return cur, best_score


def random_search(
    n: int,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    attempts: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap random Dirichlet draws – useful as a last‑ditch exploration
    when a tiny amount of time is left.
    """
    while time.time() < deadline and attempts > 0:
        candidate = np.random.exponential(scale=1.0, size=n)
        candidate /= candidate.sum()
        sc = fast_score(candidate)
        if sc < best_score:
            best_score = sc
            best_seq = candidate
        attempts -= 1
        if attempts % 2_000 == 0:
            print(f"[random] {attempts:,} attempts left – best_score = {best_score:.6f}")
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (a step function) with a low
    `evaluate_sequence` score.  The search runs for at most ``budget_s``
    seconds and prints periodic progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  8000 points is fast enough for FFT‑based
    # evaluation and gives a finer discretisation than the reference 5000.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 1) Coarse grid over families of decaying sequences
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # Power‑law family
    exponents = np.linspace(-0.9, -0.1, 81)          # 81 points
    offsets = np.linspace(0.0, 2.0, 21)              # 21 points
    print("[grid] exploring power‑law family …")
    for off in offsets:
        for a in exponents:
            cur = make_power_law(n, a, offset=off)
            sc = fast_score(cur)
            if sc < best_score:
                best_score = sc
                best_seq = cur.copy()
    print(f"[grid] best power‑law – score = {best_score:.6f}")

    # Exponential decay family (optional, cheap)
    bases = np.linspace(0.5, 0.999, 81)
    print("[grid] exploring exponential family …")
    for base in bases:
        cur = make_exponential(n, base, offset=0.0)
        sc = fast_score(cur)
        if sc < best_score:
            best_score = sc
            best_seq = cur.copy()
    print(f"[grid] after exponential – best_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Stochastic hill‑climbing on the simplex
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[hill‑climb] start")
        best_seq, best_score = hill_climb(best_seq, best_score, deadline)
        print(f"[hill‑climb] finished – best_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected sub‑gradient descent (fine polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[gradient] start")
        best_seq, best_score = gradient_descent(best_seq, best_score, deadline)
        print(f"[gradient] finished – best_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) A quick random‑Dirichlet sweep as a safety net
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[random] start")
        best_seq, best_score = random_search(
            n, best_seq, best_score, deadline, attempts=30_000
        )
        print(f"[random] finished – best_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final validation with the official evaluator
    # ------------------------------------------------------------------
    final_seq = [float(v) for v in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(
        f"Returning sequence of length {len(final_seq)} with "
        f"official evaluation score {final_score:.6f}"
    )
    return final_seq
