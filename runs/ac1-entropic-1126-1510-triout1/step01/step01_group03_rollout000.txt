"""
Propose a step‑function sequence whose self‑convolution peak is as small as possible.
The algorithm starts from the classic “arcsine” (binomial) coefficients, then
iteratively improves them using a linear‑programming based “mix‑with‑better‑g”
procedure described in the problem statement:

    • f is the current sequence (∑f = 1).
    • M = max(f∗f)  (peak of the self‑convolution).
    • Solve   max   Σ g_i
        s.t.   (f∗g)_k ≤ M   (k = 0 … 2n‑2) ,   g_i ≥ 0 .
      This LP finds the “most massive” non‑negative sequence that does not
      increase the convolution peak.
    • Normalise g to have the same ℓ₁‑norm as f (here 1) and replace
        f ← (1‑t)·f + t·g   with a modest mixing factor t (≈0.5).
      The new f has a strictly smaller peak, so the process can be repeated.
    • The loop stops after a few iterations or when no improvement is seen.
A short random‑perturbation phase finishes the search.

The whole search explores a few lengths (≈200…1300) and keeps the best
sequence (lowest evaluation score) found before the time budget expires.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse
import scipy.linalg


def arcsine_sequence(n: int) -> list[float]:
    """
    Return a length‑n sequence proportional to
        C_i = binom(2i,i) / 4**i  (i = 0 .. n‑1)
    Normalised so that the ℓ₁‑norm equals 1.
    """
    seq = np.empty(n, dtype=float)
    cur = 1.0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    seq /= total
    return seq.tolist()


def _conv_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that A @ g = conv(f, g).
    Each row k contains the coefficients f_{k‑j} (or 0 when out of range).
    """
    n = f.shape[0]
    # first column of the Toeplitz matrix: f followed by zeros
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # first row: f[0] followed by zeros
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    A_dense = scipy.linalg.toeplitz(col, row)
    return scipy.sparse.csr_matrix(A_dense)


def _improve_sequence(
    init_seq: np.ndarray,
    mix_factor: float = 0.5,
    max_iter: int = 20,
    deadline: float = 1e9,
) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based improvement on a single length.
    Returns the best sequence found and its self‑convolution peak.
    """
    n = init_seq.size
    f = init_seq.copy()
    best_M = np.max(np.convolve(f, f))
    # short‑circuit: if the peak is already tiny we skip the LP
    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # Build the convolution matrix for the current f
        A = _conv_matrix(f)

        # Upper‑bound for each convolution entry: the current peak M
        ub = np.full(2 * n - 1, best_M, dtype=float)

        # Linear programme: maximise Σ g_i  ⇔ minimise –Σ g_i
        c_obj = -np.ones(n, dtype=float)          # minimise –sum(g)
        bounds = [(0.0, None)] * n                # g_i ≥ 0

        # solve (HiGHS is the default high‑performance method)
        res = scipy.optimize.linprog(
            c=c_obj,
            A_ub=A,
            b_ub=ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )
        if not res.success:
            # LP failed – stop improving this length
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # Normalise g so that it also sums to 1 (the same ℓ₁‑norm as f)
        g = g0 / sum_g0

        # Convex mixture
        new_f = (1.0 - mix_factor) * f + mix_factor * g
        new_f /= new_f.sum()                     # re‑normalise to avoid drift

        new_M = np.max(np.convolve(new_f, new_f))

        # Accept the mixture only if the peak strictly decreases
        if new_M < best_M - 1e-12:
            f = new_f
            best_M = new_M
            # optional: shrink the mixing factor as we approach the optimum
            # (not strictly needed for our modest iteration count)
        else:
            # no improvement → stop the inner loop
            break

    return f, best_M


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The method tries several lengths, each time
    starting from the arcsine construction and applying the LP‑based
    refinement described above.  A final tiny random‑perturbation step
    attempts to squeeze out any remaining improvement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # tiny safety margin

    # ------------------------------------------------------------------
    # Candidate lengths (we keep them modest so the LP stays fast)
    lengths = list(range(200, 1301, 200))          # 200,400,…,1200
    # also try a few intermediate lengths
    lengths += [250, 350, 450, 550, 650, 750, 850, 950, 1050, 1150, 1250]

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        # --- start from the arcsine sequence -------------------------------------------------
        seq = np.array(arcsine_sequence(n), dtype=float)   # ℓ₁‑norm = 1
        base_score = evaluate_sequence(seq.tolist())
        print(f"[init] n={n:<4d}  arcsine score={base_score:.6f}")

        # --- LP‑based refinement ---------------------------------------------------------------
        # we keep a small mixing factor; 20 iterations are usually enough
        refined_seq, refined_M = _improve_sequence(
            seq,
            mix_factor=0.5,
            max_iter=20,
            deadline=deadline,
        )
        refined_score = evaluate_sequence(refined_seq.tolist())
        print(f"[refined] n={n:<4d}  score={refined_score:.6f}  peak={refined_M:.6e}")

        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.tolist()
            print(f"   ** new global best (n={n})  score={best_score:.6f}")

        # ------------------------------------------------------------------
        # tiny random fine‑tuning while we still have time (optional)
        while time.time() < deadline:
            # pick two indices and move a tiny amount of mass
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 1e-4   # ±5e‑5
            cand = best_seq.copy()
            cand[i] = max(0.0, cand[i] + delta)
            cand[j] = max(0.0, cand[j] - delta)
            s = sum(cand)
            if s < 1e-12:
                break
            cand = [x / s for x in cand]
            try:
                sc = evaluate_sequence(cand)
            except Exception:
                break
            if sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"[local] n={n:<4d}  improved score={best_score:.6f}")
            # one tiny tweak per length is enough – break to keep the loop cheap
            break

    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen, but keep the runtime safe
        best_seq = arcsine_sequence(400)

    print(f"Best sequence length = {len(best_seq)}  final score = {best_score:.6f}")
    return best_seq
