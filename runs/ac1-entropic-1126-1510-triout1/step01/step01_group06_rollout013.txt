The evaluation function computes:

- n = len(sequence)
- b_sequence = np.convolve(sequence, sequence)
- max_b = max(b_sequence)
- sum_a = sum(sequence)

We compute result = 2 * n * max_b / (sum_a**2). Note also that before it clips negative values to 0 and caps at max 1000, but we will output non-negative anyway.

Goal: minimize this expression.

def _conv_max(a):
    # compute convolution using FFT for speed
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def _score(a):
    # assume a non-negative
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    n = a.size
    conv = _conv_max(a)
    max_c = np.max(conv)
    return float(2.0 * n * max_c / (sum_a**2))

def _smoothed_gradient_descent(a0, n, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iter_cnt = 0
    # schedule for α: start low and increase
    while time.time() < deadline and iter_cnt < 4000:
        iter_cnt += 1
        # compute convolution via FFT
        conv = _conv_max(a)
        max_conv = np.max(conv)
        # smoothed max with parameter alpha
        # gradually increase alpha up to 30
        alpha = 5 + 25.0 * iter_cnt / 4000  # linear scaling
        # compute weights
        w = np.exp(alpha * (conv - max_conv))
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum
        # gradient via convolution with reversed weights
        w_rev = w[::-1]
        # convolution a * w_rev
        size = 1 << ((n + (2*n-1) - 1).bit_length())
        fa = np.fft.rfft(a, n=size)
        fw = np.fft.rfft(w_rev, n=size)
        grad_full = np.fft.irfft(fa * fw, n=size)[:(n + (2*n-1) - 1)]  # length 3n-2
        # extract relevant entries
        idx = (2*n - 2) - np.arange(n)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

                a_j = a + jitter
                a_j = np.maximum(a_j, 0)
                a_j = a_j / np.sum(a_j)
                a = a_j
    return best_a, best_score

def _lp_refinement(a, n, steps, t):
    # iteratively solve LP and mix
    f = a.copy()
    for s in range(steps):
        conv_f_f = np.convolve(f, f)
        M = np.max(conv_f_f)
        # Build A matrix
        # shape (2n-1, n)
        A = np.empty((2*n-1, n), dtype=np.float64)
        # fill by broadcasting: each column j yields f values placed from row j to j+n-1
        # using slicing
        for j in range(n):
            A[j:j+n, j] = f
        # Linear program: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            break
        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0  # normalize to sum 1
        # convex combination
        f = (1 - t) * f + t * g
    return f

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0
    best_score = float('inf')
    best_seq = None

    while time.time() < deadline:
        # choose length
        n = np.random.randint(1500, 2500)  # moderate size
        # pick initial shape
        shape = np.random.choice(["uniform","hann","gaussian","geom","power","mixed","random"])
        if shape == "uniform":
            a = np.full(n, 1.0 / n)
        elif shape == "hann":
            idx = np.arange(n)
            a = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
            a = a / np.sum(a)
        elif shape == "gaussian":
            sigma = n / 6.0
            idx = np.arange(n) - (n-1)/2.0
            a = np.exp(-0.5 * (idx / sigma) ** 2)
            a = a / np.sum(a)
        elif shape == "geom":
            t = np.random.uniform(0.0005, 0.0015)
            r = np.exp(-t)
            a = r ** np.arange(n, dtype=float)
            a = a / np.sum(a)
        elif shape == "power":
            p = np.random.uniform(0.3, 2.0)
            a = (np.arange(1, n+1, dtype=float) ** -p)
            a = a / np.sum(a)
        elif shape == "mixed":
            t = np.random.uniform(0.0003, 0.001)
            r = np.exp(-t)
            p = np.random.uniform(0.5, 1.5)
            a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n+1, dtype=float) ** -p)
            a = a / np.sum(a)
        else:  # random
            a = np.random.gamma(1.0, 1.0, size=n)
            a = a / np.sum(a)

        # optional random permutation for variety
        if np.random.rand() < 0.5:
            np.random.shuffle(a)

        # Local smoothed gradient descent
        sub_deadline = min(deadline, time.time() + 12.0)  # up to 12 sec per start
        a_opt, score_opt = _smoothed_gradient_descent(a, n, sub_deadline)

        # LP refinement (few steps)
        if score_opt < best_score:
            # attempt refinement only if promising
            a_ref = _lp_refinement(a_opt, n, steps=5, t=0.2)
            score_ref = _score(a_ref)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"New global best (LP) n={n}, shape={shape}, score={best_score:.6f}")
            else:
                if score_opt < best_score:
                    best_score = score_opt
                    best_seq = a_opt.copy()
                    print(f"New global best (GD) n={n}, shape={shape}, score={best_score:.6f}")

        # early exit if below target
        if best_score < 1.503:
            break

    # fallback
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    return best_seq.tolist()

"""
Hybrid optimisation for the step‑function problem.

The score (up to a scale‑invariant factor) is 2·n·max(conv(a,a)) with
∑a_i = 1 and a_i ≥ 0.  We use a two‑stage optimiser:

1.  Smoothed‑max gradient descent (Adam + Euclidean projection onto the
    probability simplex).  The “soft‑max” of the convolution provides a
    smooth approximation of the ℓ∞‑norm, giving a useful gradient even
    when the true maximum is attained at several indices.

2.  A short linear‑program refinement based on the classic alternating
    LP scheme:  for the current f we maximise ∑g_j under the linear
    constraints (f∗g)_k ≤ M (where M = max(f∗f)).  The obtained g is
    scaled to the same normalisation and mixed with f.
    A few mixing steps (t≈0.2) usually lower the peak of the self‑convolution.

Various initial shapes (uniform, Hann, Gaussian, geometric, power‑law,
mixed, random) are tried, each restart runs for at most ~12 s.
The best sequence found before the budget expires is returned.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a using FFT; returns the full vector."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) / (sum(a)^2).  Sum is expected to be 1."""
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    n = a.size
    max_c = np.max(_conv_max(a))
    return float(2.0 * n * max_c / (s * s))


def _smoothed_gradient_descent(a0: np.ndarray, n: int, deadline: float):
    """Adam‑style descent on the soft‑max of the convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = _score(a)
    # Adam accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline and it < 5000:
        it += 1
        cur_sc = _score(a)

        # full convolution
        conv = _conv_max(a)
        max_conv = np.max(conv)

        # annealing of the soft‑max temperature
        alpha = 5.0 + 30.0 * it / 5000.0  # grows from 5 to ~35

        # soft‑max weights (numerically stable)
        w = np.exp(alpha * (conv - max_conv))
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient = 2·(a * w_rev) restricted to the n central positions
        w_rev = w[::-1]
        size = 1 << ((n + (2 * n - 1) - 1).bit_length())  # length for a*w_rev
        fa = np.fft.rfft(a, n=size)
        fw = np.fft.rfft(w_rev, n=size)
        grad_full = np.fft.irfft(fa * fw, n=size)[: (n + (2 * n - 1) - 1)]
        idx = (2 * n - 2) - np.arange(n)               # indices n‑1 … 2n‑2 reversed
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = 0.2 / np.sqrt(it)                         # decreasing learning rate
        a_cand = _project_to_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))
        cand_sc = _score(a_cand)

        # keep best ever seen
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_a = a_cand.copy()

        # accept only improving moves; otherwise try a tiny random jitter occasionally
        if cand_sc < cur_sc:
            a = a_cand
        elif it % 100 == 0:
            jitter = np.random.randn(n) * 1e-4
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j /= np.sum(a_j)
            a = a_j

        if best_sc < 1.51:          # early cut‑off
            break

    return best_a, best_sc


def _lp_refinement(f: np.ndarray, steps: int = 5, mix: float = 0.2):
    """Iteratively improve f by solving the LP described in the statement."""
    n = f.size
    for _ in range(steps):
        M = np.max(np.convolve(f, f))
        # build the (2n‑1)×n matrix A where A[k, j] = f[k‑j] (if inside range)
        A = np.empty((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f
        # maximise sum(g)  ⇔ minimise –∑g
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break
        g0 = res.x
        s = np.sum(g0)
        if s <= 0:
            break
        g = g0 / s                # scale to sum = 1 (feasible because scaling down keeps constraints)
        f = (1.0 - mix) * f + mix * g
    return f


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Entry point – returns a list of non‑negative heights with a low score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin
    print(f"Search started – time budget {budget_s}s (deadline {deadline:.2f})")

    best_score = float("inf")
    best_seq = None

    # main restart loop
    while time.time() < deadline:
        # ----- problem size & initial shape -----
        n = np.random.randint(1500, 2500)                     # moderate size
        shape = np.random.choice(
            ["uniform", "hann", "gaussian", "geom", "power", "mixed", "random"]
        )
        if shape == "uniform":
            a = np.full(n, 1.0 / n)
        elif shape == "hann":
            i = np.arange(n)
            a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
            a = a / np.sum(a)
        elif shape == "gaussian":
            sigma = n / 6.0
            i = np.arange(n) - (n - 1) / 2.0
            a = np.exp(-0.5 * (i / sigma) ** 2)
            a = a / np.sum(a)
        elif shape == "geom":
            t = np.random.uniform(0.0005, 0.0015)
            r = np.exp(-t)
            a = r ** np.arange(n, dtype=float)
            a = a / np.sum(a)
        elif shape == "power":
            p = np.random.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** -p)
            a = a / np.sum(a)
        elif shape == "mixed":
            t = np.random.uniform(0.0003, 0.001)
            r = np.exp(-t)
            p = np.random.uniform(0.5, 1.5)
            a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)
            a = a / np.sum(a)
        else:                                            # "random"
            a = np.random.gamma(1.0, 1.0, size=n)
            a = a / np.sum(a)

        # optional permutation (does not change the multiset of values)
        if np.random.rand() < 0.5:
            np.random.shuffle(a)

        # ----- smoothed gradient descent -----
        sub_deadline = min(deadline, time.time() + 12.0)      # ≈12 s per start
        a_opt, sc_opt = _smoothed_gradient_descent(a, n, sub_deadline)

        # ----- LP refinement (only if promising) -----
        if sc_opt < best_score:
            a_ref = _lp_refinement(a_opt, steps=5, mix=0.2)
            sc_ref = _score(a_ref)
        else:
            a_ref, sc_ref = a_opt, sc_opt   # keep what we have

        # ----- keep the best overall -----
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(
                f"GLOBAL IMPROVEMENT: n={n}, shape={shape}, score={best_score:.6f}"
            )
            # early stop if we cross the target
            if best_score < 1.503:
                break

    # fallback (should never happen)
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    print(f"Search finished – best score {best_score:.6f}")
    return best_seq.tolist()
