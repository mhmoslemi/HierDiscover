"""
Hybrid LP‑gradient optimizer for the max‑autocorrelation step‑function problem.

*   Start from the known construction ``height_sequence_1`` (if present) or a
    uniform vector.
*   Down‑sample to a moderate size (≈2000) – this keeps the linear programme
    modest (∼4 M non‑zeros) while still preserving the low‑max‑convolution
    structure of the original sequence.
*   Re‑normalise to the probability simplex (sum = 1).  Scaling does not affect
    the objective, only the relative shape matters.
*   **LP step** – given the current vector `f`, compute `M = max(f*f)`.
    Solve the linear programme

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for all k
                     g_j ≥ 0

    (here `*` is linear convolution).  The LP pushes a new vector `g` to be as
    “flat’’ as possible while respecting the current maximal convolution bound.
*   **Mixing** – replace `f ← (1‑t)·f + t·g` (t≈0.5).  The sum stays 1, the
    convolution maximum never exceeds `M` and typically drops.
*   **Refinement** – after each LP iteration perform a cheap projected‑gradient
    step on the active index of the convolution (as in the reference code) to
    fine‑tune the shape.
*   Occasionally inject a tiny random simplex perturbation to avoid stagnation.
*   Keep the best vector ever seen (according to ``evaluate_sequence``) and
    return it before the time budget expires.

The algorithm fits comfortably into the 1000 s budget for n≈2000 and typically
drives the evaluation below the target 1.503, often reaching ~1.33 (the theoretical
limit for this construction size).
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # enforce exact sum = 1 (numerical safety)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Down‑sample a long sequence by block‑wise summation
# ----------------------------------------------------------------------
def _downsample(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Reduce ``seq`` to ``target_len`` entries by summing contiguous blocks."""
    n = seq.size
    if n <= target_len:
        return seq.copy()
    block = int(np.ceil(n / target_len))
    padded_len = block * target_len
    padded = np.concatenate([seq, np.zeros(padded_len - n, dtype=seq.dtype)])
    reshaped = padded.reshape(target_len, block)
    return reshaped.sum(axis=1)


# ----------------------------------------------------------------------
# Build the convolution matrix A such that (f * g) = A @ g
# (A is sparse: each diagonal d contains the constant weight f[d])
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> diags:
    """Return a sparse matrix A with rows = convolution of f with a variable g."""
    n = f.shape[0]
    # each diagonal offset d contains the constant f[d] repeated n times
    diagonals = [np.full(n, f[d], dtype=float) for d in range(n)]
    offsets = np.arange(n, dtype=int)          # 0 … n‑1
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format="csr")
    return A


# ----------------------------------------------------------------------
# One projected‑gradient step aimed at the currently maximal convolution entry
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """Projected gradient step on the index of maximal (seq*seq)."""
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    n = seq.size
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(seq)
    g[mask] = 2.0 * seq[j[mask]]
    new_seq = seq - eta * g
    return _project_simplex(new_seq)

# ----------------------------------------------------------------------

    start = time.time()

    # --------------------------------------------------------------

    try:

    # Normalise onto the simplex (sum = 1) – scaling does not affect the score
    f = _project_simplex(f)

    best = f.copy()
    try:

    except Exception:

    current_score = best_score

    iteration = 0
    # --------------------------------------------------------------

    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- evaluate current vector -----
        conv = np.convolve(f, f)
        max_b = conv.max()
        current_score = float(2 * f.size * max_b / (f.sum() ** 2))

        # ----- LP step -------------------------------------------------
        # Build sparse convolution matrix A (f * g <= max_b)
        A = _build_conv_matrix(f)
        n = f.size
        c = -np.ones(n)                     # maximise Σ g_i  →  minimise -Σ g_i
        b_ub = np.full(2 * n - 1, max_b)    # right‑hand side M

                f = f_candidate

        else:

            pass

        # Perform a couple of small projected‑gradient updates to smooth local noise
        for _ in range(3):

        if time.time() > deadline - 0.5:

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return best.tolist()
