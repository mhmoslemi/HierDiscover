from scipy.optimize import linprog

def lp_step(f, M, t):
    n = len(f)
    # Build constraint matrix A_ub shape (2n-1, n)
    # For each k in 0..2n-2:
    #   row = [ f_i where i from max(0, k-(n-1)) to min(n-1, k) ; mapping index j = k-i
    # Actually conv: (f * g)[k] = sum_i f[i] * g[k-i]
    # So coefficient for g[j] in row k is f[k-j] if 0 <= k-j < n else 0.
    # Build using loops.
    rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # i runs over possible f index; then j = k-i.
        # So set row[j] = f[i] for each i in i_min..i_max
        # j = k - i
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
    A_ub = np.array(rows)
    b_ub = np.full(2*n-1, M)
    # Solve maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b_opt = res.x
    sum_b = np.sum(b_opt)
    if sum_b <= 0:
        return None
    g = b_opt / sum_b
    # mixture
    f_new = (1 - t) * f + t * g
    # Ensure non-negative (should be)
    # Normalize to sum 1 if needed (tiny rounding):
    s = np.sum(f_new)
    if s > 0:
        f_new = f_new / s
    else:
        return f
    return f_new
```

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    lengths = [200, 300, 400, 500, 600, 800, 1000] # maybe more up to 1500

    for n in lengths:
        if time.time() >= deadline:
            break
        # start with arcsine
        f = arcsine_sequence(n)
        # normalize to sum=1
        f = np.array(f) / np.sum(f)
        # compute M
        conv = np.convolve(f, f)
        M = np.max(conv)
        score = evaluate_sequence(f.tolist())
        print(f"[init] n={n} score={score:.6f}")
        if score < best_score:
            best_score = score; best_seq = f.tolist()
        # iterative improvement
        t = 0.2
        for it in range(30):
            if time.time() >= deadline:
                break
            f_new = lp_step(f, M, t)
            if f_new is None:
                break
            conv_new = np.convolve(f_new, f_new)
            M_new = np.max(conv_new)
            # compute new evaluation
            new_score = evaluate_sequence(f_new.tolist())
            if new_score < best_score:
                best_score = new_score; best_seq = f_new.tolist()
                print(f"[improve] n={n} iter={it} score={new_score:.6f}")
            # accept new if M_new <= M + small tolerance
            if M_new <= M * 1.0001:  # allow slight increase
                f = f_new
                M = M_new
            else:
                # if not improved, reduce t
                t *= 0.5
                if t < 1e-4:
                    break
        # final local random refinement as before
        while time.time() < deadline:
            # small random perturbation
            nlen = len(f)
            i, j = np.random.choice(nlen, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 1e-3
            cand = f.copy()
            cand[i] = max(0.0, cand[i] + delta)
            cand[j] = max(0.0, cand[j] - delta)
            s = np.sum(cand)
            if s <= 0:
                continue
            cand = cand / s
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = cand.tolist()
                print(f"[refine] n={n} random score={sc:.6f}")
                f = cand
                # update M
                M = np.max(np.convolve(f, f))
                # continue
            else:
                # small chance to accept? Could also break
                pass
    print(f"Best final score={best_score:.6f} length={len(best_seq)}")
    return best_seq

"""
Propose a step‑function sequence with a very low evaluation score.

The algorithm mixes three ideas:
1. Initialise with a good analytic family (the arcsine series) and also with a
   random Dirichlet point.
2. Perform a fast, vectorised simulated‑annealing / hill‑climbing on the
   convolution max.  A move transfers a tiny amount of mass from one index to
   another; the convolution is updated in O(n) time by a simple slice update,
   avoiding the O(n²) cost of a full recompute.
3. Keep the best sequence seen over several lengths (200…1000).  The search
   stops as soon as the allotted time (≈1000 s) is exhausted.

The code is self‑contained, uses only numpy and the already‑imported
evaluate_sequence, and returns a Python list suitable for the scorer.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> np.ndarray:
    """Return the normalised arcsine (C_i) sequence of length n."""
    seq = np.empty(n, dtype=np.float64)
    cur = 1.0  # C_0
    total = 0.0
    for i in range(n):
        seq[i] = cur
        total += cur
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    # normalise to unit L1‑norm
    seq /= total
    return seq


def random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex of dimension n."""
    x = np.random.rand(n).astype(np.float64)
    x /= x.sum()
    return x


def conv_max(a: np.ndarray, conv: np.ndarray = None) -> tuple:
    """
    Return (max_convolution, full_convolution) for a vector a.
    If conv is supplied it is assumed to be the current convolution and is
    returned unchanged to avoid recomputation.
    """
    if conv is None:
        conv = np.convolve(a, a)
    return conv.max(), conv


def move_and_update(a: np.ndarray, conv: np.ndarray, i: int, j: int,
                    delta: float) -> tuple:
    """
    Transfer `delta` mass from position i to position j (delta>0, a[i]>=delta).
    Return the new (a, conv) pair after an O(n) update of the convolution.
    """
    n = a.shape[0]
    # keep copies of the old vectors
    a_old = a
    conv_new = conv.copy()

    # signed changes for the two slots
    di = -delta
    dj = +delta

    # slice updates: (a * Δ) appears twice in a*a, hence factor 2
    conv_new[i:i + n] += 2.0 * di * a_old
    conv_new[j:j + n] += 2.0 * dj * a_old

    # self‑interaction terms of the transferred mass
    k = 2 * i
    if k < conv_new.size:
        conv_new[k] += di * di
    k = 2 * j
    if k < conv_new.size:
        conv_new[k] += dj * dj
    k = i + j
    if k < conv_new.size:
        conv_new[k] += 2.0 * di * dj

    # new a vector
    a_new = a_old.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new, conv_new


def simulated_annealing(a0: np.ndarray,
                        max_time: float,
                        n_iter: int = 200_000,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-6) -> tuple:
    """
    Perform a fast annealing walk starting from a0.
    Returns (best_a, best_score, best_conv).
    """
    n = a0.shape[0]
    a = a0.copy()
    conv = np.convolve(a, a)
    best_a = a.copy()
    best_conv = conv.copy()
    best_score = 2.0 * n * conv.max()

    # pre‑select a list of donor candidates to avoid picking zeros
    donors = np.arange(n)

    for it in range(n_iter):
        # time check – quit early if we run out of budget
        if time.time() >= max_time:
            break

        # temperature schedule (exponential decay)
        temp = start_temp * (end_temp / start_temp) ** (it / n_iter)

        # pick donor (must have some mass) and receiver
        i = np.random.choice(donors)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(0, n)
        if j == i:
            continue

        # propose a transfer; cap it to a modest fraction of the donor mass
        max_transfer = min(a[i], 0.05)          # never move more than 5 %
        delta = np.random.rand() * max_transfer

        a_new, conv_new = move_and_update(a, conv, i, j, delta)
        new_score = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            best_conv = conv_new.copy()
        if new_score <= 2.0 * n * conv.max() or np.random.rand() < np.exp(-(new_score - 2.0 * n * conv.max()) / max(temp, 1e-12)):
            a = a_new
            conv = conv_new

    return best_a, best_score, best_conv

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative step‑function sequence under the given
    time budget.  Several lengths are tried; each length is optimised with a
    quick simulated‑annealing walk starting from both the arcsine construction
    and a random simplex point.  The best sequence found is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Candidate lengths – a compromise between quality and speed.
    lengths = list(range(200, 1001, 200))      # 200, 400, …, 1000

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Exploring length n={n} ===")

        # two initial points per length: arcsine and a random simplex
        initial_vectors = [arcsine_sequence(n), random_simplex(n)]

        for start_id, a0 in enumerate(initial_vectors):
            if time.time() >= deadline:
                break
            init_type = "arcsine" if start_id == 0 else "random"
            print(f"[{init_type}] initial score → ", end="")

            # initial evaluation
            init_score = evaluate_sequence(a0.tolist())
            print(f"{init_score:.6f}")

            # keep a simple fix: if the evaluate call returns inf (should not),
            # fall back to the raw score.
            if init_score == float('inf'):
                init_score = 2.0 * n * np.convolve(a0, a0).max()

            # run a limited annealing stage – each stage gets a share of the
            # remaining time budget.
            remaining = deadline - time.time()
            stage_time = remaining * 0.25   # ≤ 25 % of what’s left for this start
            a_opt, score_opt, _ = simulated_annealing(
                a0,
                max_time=time.time() + stage_time,
                n_iter=200_000,
                start_temp=5e-2,
                end_temp=1e-6,
            )
            print(f"[{init_type}] post‑anneal score = {score_opt:.6f}")

            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.tolist()
                print(f" ** New global best! ({best_score:.6f})")

        # ------------------------------------------------------------------
        # Tiny random refinement on the current global best for this length
        if best_seq is not None and len(best_seq) == n:
            a_curr = np.array(best_seq, dtype=np.float64)
            conv_curr = np.convolve(a_curr, a_curr)

            # short refinement loop (few thousand moves)
            while time.time() < deadline and np.random.rand() < 0.99:
                i, j = np.random.choice(n, 2, replace=False)
                if a_curr[i] < 1e-12:
                    continue
                delta = min(a_curr[i], 1e-3) * np.random.rand()
                a_new, conv_new = move_and_update(a_curr, conv_curr, i, j, delta)
                new_score = 2.0 * n * conv_new.max()
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.tolist()
                    a_curr = a_new
                    conv_curr = conv_new
                    print(f"[refine] improved to {best_score:.6f}")
                # accept only if strictly better – this is a greedy tweak
                # (no temperature needed for these tiny moves).

    print("\n=== Search finished ===")
    if best_seq is None:
        # fallback – a simple uniform distribution (never happens in practice)
        best_seq = [1.0 / 1]  # length‑1 sequence
        best_score = evaluate_sequence(best_seq)
    print(f"Best sequence length = {len(best_seq)}   best score = {best_score:.6f}")
    return best_seq
