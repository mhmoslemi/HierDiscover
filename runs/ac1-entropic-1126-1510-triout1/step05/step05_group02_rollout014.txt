"""
Propose a low‑score step‑function.

The search proceeds in three stages:
  1. Generate many candidate probability vectors on a fixed length (5000)
     using Beta distributions (including mixtures) with a very wide range of
     shape parameters (down to 0.01).  The best raw score 2·n·max(conv) is kept.
  2. Refine the best candidate with an Adam‑type projected optimizer.
     The objective is a smoothed max of the self‑convolution:
       f(a) = (1/γ)·log Σ exp(γ·(a*a)_k) .
     γ is increased gradually so the smooth objective approaches the true max.
  3. Return the refined vector as a plain Python list.
All vectors are kept on the simplex Σa_i = 1, a_i ≥ 0, which makes the
evaluation function equal to 2·n·max(conv).
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalized to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow/underflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    # Guard against log(0)
    with np.errstate(divide="ignore"):
        logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf, out=np.full_like(logpdf, 0.0))
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum value of a * a (self‑convolution)."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray):
    """Sub‑gradient of max_k (a*a)_k and the current max value."""
    n = a.shape[0]
    b = _fft_self_convolve_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights with low evaluate_sequence score.
    The routine:
      * samples a very rich family of (possibly mixed) Beta‑distributions,
      * keeps the best raw score (2·n·max(conv)),
      * refines it with an Adam optimiser on a smoothly‑approximated max,
      * finally returns the sequence as a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # final length – large enough to give a good “smoothness’’ budget
    n = 5000

    # ------------------------------------------------------------------
    # Helper: raw objective (identical to evaluate_sequence when Σa_i = 1)
    def _raw_score(vec: np.ndarray) -> float:
        return 2.0 * n * _fft_self_convolve_max(vec)

    # ------------------------------------------------------------------
    # 1) collect a pool of initial candidates
    candidates = []

    # a) previous construction if available
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base = _interpolate_to_length(base, n)
        candidates.append(base)
    except Exception:
        pass

    # b) simple baselines
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))               # uniform
    candidates.append(_beta_distribution(n, 0.5, 0.5))                     # arcsine

    # c) systematic sweep of symmetric Beta(α,α) – include very small α
    alphas = np.concatenate((
        np.logspace(-2.3, -0.3, 12),               # 0.005 … 0.5 (log‑scale)
        np.linspace(0.5, 2.0, 10, endpoint=False),
        np.linspace(2.0, 5.0, 7)
    ))
    for a in alphas:
        if time.time() >= deadline:
            break
        vec = _beta_distribution(n, a, a)
        candidates.append(vec)

    # d) random asymmetric Beta draws (α,β in [0.01, 2])
    for _ in range(600):
        if time.time() >= deadline:
            break
        a = 10 ** np.random.uniform(-2, 0.3)   # 0.01 … ~2
        b = 10 ** np.random.uniform(-2, 0.3)
        vec = _beta_distribution(n, a, b)
        candidates.append(vec)

    # e) mixtures of two Beta components
    for _ in range(800):
        if time.time() >= deadline:
            break
        a1, b1 = 10 ** np.random.uniform(-2, 0.3), 10 ** np.random.uniform(-2, 0.3)
        a2, b2 = 10 ** np.random.uniform(-2, 0.3), 10 ** np.random.uniform(-2, 0.3)
        lam = np.random.rand()
        p1 = _beta_distribution(n, a1, b1)
        p2 = _beta_distribution(n, a2, b2)
        vec = lam * p1 + (1.0 - lam) * p2
        vec = _project_onto_simplex(vec)
        candidates.append(vec)

    # f) mixtures of three Beta components (fewer trials)
    for _ in range(200):
        if time.time() >= deadline:
            break
        weights = np.random.rand(3)
        weights /= weights.sum()
        vec = np.zeros(n)
        for w in weights:
            a, b = 10 ** np.random.uniform(-2, 0.3), 10 ** np.random.uniform(-2, 0.3)
            vec += w * _beta_distribution(n, a, b)
        vec = _project_onto_simplex(vec)
        candidates.append(vec)

    # ------------------------------------------------------------------
    # Pick the best raw score among all generated candidates
    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        if cand.shape[0] != n:
            cand = _interpolate_to_length(cand, n)
        cand = _project_onto_simplex(cand)
        sc = _raw_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.1f}] new best raw score {best_score:.6f}")

    if best_seq is None:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_seq)
        print("fallback uniform used.")

    print(f"Best raw score after sampling: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Local refinement with Adam on a smoothed max
    a = best_seq.copy()
    best_a = a.copy()
    best_val = best_score

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    gamma = 1.0           # start smoothing factor
    gamma_inc = 0.02
    gamma_max = 200.0

    while time.time() < deadline:
        it += 1

        # Full self‑convolution
        conv_full = _fft_self_convolve_full(a)
        cur_max = conv_full.max()

        # Smooth‑max weights (soft‑max over the convolution values)
        shifted = conv_full - cur_max                     # for numerical stability
        if gamma < gamma_max:
            gamma = min(gamma_max, gamma + gamma_inc)
        exp_vals = np.exp(gamma * shifted)
        w = exp_vals / exp_vals.sum()

        # Gradient of the smooth‑max objective
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.12 * m_hat / (np.sqrt(v_hat) + eps)    # learning‑rate ≈0.12

        a = a - step
        a = _project_onto_simplex(a)

        # Evaluate the true (unsmoothed) objective
        cur_val = 2.0 * n * _fft_self_convolve_max(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] refinement it={it}  score={best_val:.6f}")

        # occasional random jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 4e-7
            a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    # 3) Return the best found sequence as a Python list
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq
