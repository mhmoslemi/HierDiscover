"""
Improved non‑negative step‑function optimizer.

Main ideas
-----------
*  Explore a richer set of parametric families (Beta with very small shape,
   power‑law tails up to the integrability limit, simple two‑spike mixtures).
*  Run a fast “soft‑max” surrogate optimisation using the Adam optimiser
   (exponentiated‑gradient with momentum).  The temperature α of the
   log‑sum‑exp surrogate is slowly increased so that the optimisation first
   sees a smooth landscape and then concentrates on the true maximum.
*  Keep a small pool of promising candidates, refine the best few with the
   Adam optimiser and finish with a cheap random‑move hill‑climber.
*  All heavy work (convolution, gradients) is done with real FFTs; the code
   respects the 1000 s budget and returns a plain Python list.

The returned sequence is already projected onto the simplex (sum = 1), so
`evaluate_sequence` can be called directly on the result.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (symmetric, β=½)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """
    Symmetric weight ∝ (i·(n‑i))^{±p}.
    invert=True gives the heavy‑tail version (p>0).   invert=False gives a
    centre‑heavy version (not used here).
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) distribution on the integer grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """Convex combination (1‑lam)·arcsine + lam·uniform."""
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _two_spike_mixture(n: int, edge_mass: float) -> np.ndarray:
    """
    Allocate `edge_mass` of the total probability to the two extreme points,
    the rest uniformly.
    edge_mass ∈ [0,1].
    """
    if edge_mass < 0.0:
        edge_mass = 0.0
    if edge_mass > 1.0:
        edge_mass = 1.0
    interior = (1.0 - edge_mass) / n
    vals = np.full(n, interior, dtype=np.float64)
    vals[0] = vals[-1] = edge_mass / 2.0
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return float(_fft_self_convolve(a).max())


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) / (sum a)^2.
    The vector is projected onto the simplex (sum = 1) before evaluation
    so the denominator is always 1.
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·b_k) w.r.t. a.
    Implements  g = 2·conv_rev(w, a_rev) where
    w_k ∝ exp(α·b_k) are the soft‑max weights.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:               # numerical safeguard
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g

def _optimize_adam(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient / Adam optimiser on the log‑sum‑exp surrogate.
    Returns the best vector seen up to `deadline`.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.8                     # base learning‑rate (applied after exp‑step)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_max = 20000.0
    alpha_mul = 1.02

    # FFT size for the surrogate gradient (a bit larger than needed)
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution (full) – needed for the surrogate gradient
        b = _fft_self_convolve(a)

        # surrogate gradient
        grad = _gradient_softmax(a, b, alpha, size_fft)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # exponentiated‑gradient update (adds positivity automatically)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # protect against tiny negativities, renormalise onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # objective check
        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] Adam improved to {best_score:.6f}")

        # temperature annealing (increase α → closer to the true max)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny random jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best, best_score

def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap random‑move hill‑climber.
    Transfers a small fraction of probability mass from one coordinate
    to another whenever it yields a strictly lower objective.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.1
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur = _objective(cand)
        if cur < best_score:
            best_score = cur
            best = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best, best_score

            seq = _mixture_arcsine_uniform(n, lam)
        elif fam == "two_spike":
            edge = np.random.uniform(0.0, 0.4)     # up to 40 % on the two ends
            seq = _two_spike_mixture(n, edge)
        else:
            seq = _arcsine_distribution(n)

        seq = _project_onto_simplex(seq, s=1.0)
        score = _objective(seq)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New global best (rand) n={n} fam={fam} score={best_score:.6f}")

        # keep a small pool of promising candidates (max 12)
        if len(best_candidates) < 12:
            best_candidates.append((score, seq))
        else:
            worst_idx = max(range(len(best_candidates)),
                            key=lambda i: best_candidates[i][0])
            if score < best_candidates[worst_idx][0]:
                best_candidates[worst_idx] = (score, seq)

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑based refinement of the top candidates
    best_candidates.sort(key=lambda x: x[0])
    top_candidates = [c[1] for c in best_candidates[:4]]

    for cand in top_candidates:
        if time.time() >= deadline - 130.0:   # keep some time for the next phases
            break
        n = cand.shape[0]
        # allocate up to 60 s (or the remaining time) for each run
        remaining = deadline - time.time()
        run_deadline = time.time() + min(60.0, remaining * 0.5)
        refined, _ = _optimize_adam(cand, n, run_deadline)
        refined_score = _objective(refined)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()
            print(f"Adam refinement improved to {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap hill‑climbing
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        refined, _ = _local_refinement(best_seq, n, deadline)
        refined_score = _objective(refined)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()
            print(f"After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback (should never happen)
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length = {best_seq.shape[0]})")
    return best_seq.tolist()
