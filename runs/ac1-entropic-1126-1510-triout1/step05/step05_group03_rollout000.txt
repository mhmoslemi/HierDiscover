"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Candidate generation** –  exhaustive grid over a two‑parameter family
   `a_i ∝ ((i+δ)*(n‑1‑i+δ))^{-p}` (a smoothed “arcsine” shape).  We also mix
   in a small amount of the uniform distribution.  For each length `n`
   (several values around the size of the previously known construction)
   we evaluate the cheap surrogate  `2·n·max(a*a)` and keep the best vector.

2. **Smooth‑max Adam** –  starting from the best candidate we minimise the
   smooth surrogate
        L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
   using Adam with an exponentially increasing temperature `α`.  This
   quickly drives the peak of the convolution down.

3. **Exact sub‑gradient polishing** –  we switch to the true
   `max(a*a)` objective.  The sub‑gradient is obtained from the active
   set of indices attaining the current maximum and a projected sub‑gradient
   descent with a diminishing step size is performed.

4. **Simulated‑annealing polishing** –  a final stochastic phase moves a
   small amount of probability mass between random positions while
   accepting uphill moves according to an exponential temperature schedule.

5. **Black‑box evaluation** –  the resulting vector is finally evaluated
   with the supplied `evaluate_sequence`.  If a previously discovered
   construction (`height_sequence_1`) exists it is also considered as a
   starting point.

The whole routine respects the time budget (default 1000 s) and always
projects onto the probability simplex `Σa_i = 1, a_i ≥ 0`.
"""

import math
import time
import numpy as np
import random

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 0.5,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 4000.0,
                   lr0: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered and its peak convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule
        lr = lr0 / math.sqrt(t)

        # update & project back onto the simplex
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # anneal α
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape plateaus
        if t % 400 == 0 and random.random() < 0.07:
            a += np.random.randn(n) * 7e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses a uniform weight on the active set (indices attaining the current maximum).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short exact‑max refinement using sub‑gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        # classic diminishing step size
        step = 0.015 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()
        # occasional random jitter
        if it % 500 == 0 and random.random() < 0.06:
            a += np.random.randn(n) * 6e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a

def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Simple simulated‑annealing polishing.
    Moves a small amount of mass between two random positions,
    accepting uphill moves with a temperature schedule.
    Returns the best vector found and its peak convolution.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    # temperature schedule (exponential decay)
    start_temp = max(cur_peak, 1e-3) * 0.45
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = random.randint(0, n - 1)
        if cur_a[src] <= 1e-13:
            continue
        dst = random.randint(0, n - 1)
        if src == dst:
            continue

        # move up to 5 % of the source mass
        max_delta = min(cur_a[src], 0.05)
        delta = random.random() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and random.random() < 0.04:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _power_arcsine_distribution(n: int, offset: float, exponent: float) -> np.ndarray:
    """
    Generalised arcsine‑type distribution:
        a_i ∝ ((i+offset)·(n‑1‑i+offset))^{-exponent}
    Returns a probability vector (sum = 1, all entries ≥ 0).
    """
    i = np.arange(n, dtype=np.float64)
    term = (i + offset) * (n - 1 - i + offset)
    # guard against non‑positive terms (should not happen if offset>0)
    term = np.where(term <= 0, 1e-12, term)
    vals = np.power(term, -exponent)
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= s
    return vals


# ----------------------------------------------------------------------
def _candidate_distribution(n: int,
                           offset: float,
                           exponent: float,
                           mix: float) -> np.ndarray:
    """
    Returns a mixture of a power‑arcsine distribution and the uniform
    distribution.  `mix`∈[0,1] is the weight of the uniform part.
    """
    base = _power_arcsine_distribution(n, offset, exponent)
    if mix > 0.0:
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        base = (1.0 - mix) * base + mix * uniform
    base = np.maximum(base, 0.0)
    base /= base.sum()
    return base


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function.
    Returns a Python list suitable for `evaluate_sequence`.
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Load a previously discovered sequence if it exists.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # 2.  Choose a handful of lengths to explore.
    #    We centre them around the length of the previous best (if any).
    base_len = len(base_seq) if base_seq is not None else 3500
    lengths = [max(500, int(base_len * f))
               for f in (0.6, 0.8, 1.0, 1.2, 1.4)
               if int(base_len * f) >= 500]
    if not lengths:
        lengths = [max(500, base_len)]

    # ------------------------------------------------------------------
    # 3.  Broad grid search over the two‑parameter family.
    best_seq = None
    best_score = float("inf")
    print("[phase 1] grid search over parametric families")

    # Parameter grids (chosen to concentrate around the known good arcsine)
    offset_grid = np.linspace(0.2, 2.0, 10)          # δ
    exponent_grid = np.linspace(0.30, 0.70, 9)      # p
    mix_grid = (0.0, 0.1, 0.2, 0.3)                # uniform mixing weight

    for n in lengths:
        if time.time() >= deadline:
            break
        for offset in offset_grid:
            for exponent in exponent_grid:
                for mix in mix_grid:
                    vec = _candidate_distribution(n, offset, exponent, mix)
                    # fast surrogate: 2·n·max(conv)
                    fast_score = 2.0 * n * _fft_self_convolve(vec).max()
                    if fast_score < best_score:
                        best_score = fast_score
                        best_seq = vec.copy()
                        print(f"[grid] n={n:4d} δ={offset:4.2f} p={exponent:4.2f} mix={mix:.1f}"
                              f"  fast_score={fast_score:.6f}")

    # Include the previously known construction as a fallback candidate.
    if base_seq is not None:
        vec = base_seq
        fast_score = 2.0 * len(vec) * _fft_self_convolve(vec).max()
        if fast_score < best_score:
            best_score = fast_score
            best_seq = vec.copy()
            print("[fallback] using previous best construction"
                  f"  fast_score={fast_score:.6f}")

    # Unexpected case: ensure we have something.
    if best_seq is None:
        n = max(500, base_len)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
        print("[fallback] defaulted to uniform distribution")

    # ------------------------------------------------------------------
    # 4.  Adam optimisation on the smooth max surrogate.
    time_left = deadline - time.time()
    if time_left > 4.0:
        n = len(best_seq)
        print(f"[phase 2] Adam optimisation (n={n}) – time left {time_left:.1f}s")
        phase_deadline = time.time() + time_left * 0.55
        best_seq, _ = _adam_optimize(best_seq, phase_deadline)
        best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
        print(f"[phase 2] post‑Adam fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Exact ℓ∞ sub‑gradient polishing.
    time_left = deadline - time.time()
    if time_left > 3.0:
        n = len(best_seq)
        print(f"[phase 3] Sub‑gradient refinement (n={n}) – time left {time_left:.1f}s")
        phase_deadline = time.time() + time_left * 0.35
        best_seq = _refine_max(best_seq, phase_deadline)
        best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
        print(f"[phase 3] after sub‑grad fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Simulated‑annealing polishing (if any budget remains).
    time_left = deadline - time.time()
    if time_left > 2.0:
        n = len(best_seq)
        print(f"[phase 4] Simulated annealing (n={n}) – time left {time_left:.1f}s")
        best_seq, final_peak = _simulated_annealing(best_seq, deadline)
        best_score = 2.0 * n * final_peak
        print(f"[phase 4] SA fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final black‑box evaluation.
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < best_score:
            best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best reported score (approx.): {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
