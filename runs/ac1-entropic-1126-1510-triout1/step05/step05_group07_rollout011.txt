"""
Hybrid search for a low‑value step‑function.
The algorithm mixes three ideas:

1.  Hand‑crafted parametric families (uniform, arcsine, Beta, mixtures,
    endpoint spikes) are scored quickly with an FFT based convolution.
    The best few are kept as starting points.

2.  A projected sub‑gradient descent that directly optimises the
    nonsmooth objective 2·n·max(a*a)/‖a‖₁².  The active set of indices
    attaining the maximum is used to form an exact sub‑gradient
    (averaged over the active set).  A simple 1/√t step‑size schedule,
    Euclidean projection onto the simplex and a time‑budget guard give a
    fast, robust descent.

3.  A short Adam phase (soft‑max surrogate for the max) followed by a
    stochastic hill‑climber (mass swaps + jitter) for final polishing.

All work is bounded by the supplied time budget (default 1000 s).  The
returned list is a probability vector (non‑negative, sums to 1) that
minimises ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a⋆a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(conv)/sum(a)² (identical to evaluate_sequence)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n bins."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float, b: float | None = None) -> np.ndarray:
    """Symmetric Beta(a,a) or asymmetric Beta(a,b) pdf on mid‑points."""
    if b is None:
        b = a
    a = max(a, 0.01)
    b = max(b, 0.01)
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0 or not np.isfinite(pdf_sum):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= pdf_sum
    return pdf


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest collection of hand‑crafted probability vectors."""
    rng = np.random.default_rng()
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) arcsine
    candidates.append(_arcsine_distribution(n))

    # 3) symmetric Beta(a,a) on a grid
    for a in np.linspace(0.15, 0.95, 17):
        candidates.append(_beta_distribution(n, a))

    # 4) asymmetric Beta(a,b) coarse grid
    for a in (0.2, 0.4, 0.6, 0.8):
        for b in (0.2, 0.4, 0.6, 0.8):
            if a == b:
                continue
            candidates.append(_beta_distribution(n, a, b))

    # 5) mixtures of arcsine and uniform
    arcsine = _arcsine_distribution(n)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for w in np.linspace(0.0, 0.5, 11):
        mix = (1.0 - w) * arcsine + w * uniform
        candidates.append(mix)

    # 6) spikes at the two ends on top of arcsine
    for spike_frac in (0.05, 0.10, 0.15):
        mix = arcsine.copy()
        mix[0] += spike_frac / 2.0
        mix[-1] += spike_frac / 2.0
        mix = _project_onto_simplex(mix, 1.0)
        candidates.append(mix)

    # 7) a few Dirichlet draws
    for _ in range(6):
        candidates.append(rng.dirichlet(np.ones(n)))

    # 8) previous best, if available and length matches
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())

    # Normalise all candidates onto the simplex (numerical safety)
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final


# ----------------------------------------------------------------------
def _subgradient_descent(
    init_a: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(a*a)/‖a‖₁².
    The sub‑gradient is the average of 2·a reversed over the active
    indices attaining the maximum of the self‑convolution.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = _score(a)

    size_fft = _next_pow_two(3 * n - 2)          # length needed for w⋆a_rev
    it = 0
    max_iters = 200_000

    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _fft_self_convolve(a)
        M = conv.max()

        # active set (allow a tiny relative tolerance)
        tol = max(1e-14, M * 1e-9)
        mask = conv >= M - tol
        active_cnt = mask.sum()
        if active_cnt == 0:
            mask[np.argmax(conv)] = True
            active_cnt = 1

        w = mask.astype(float) / active_cnt               # probability on active indices

        # gradient = 2 * (w * a_rev) via FFT
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        conv_w = np.fft.irfft(W * A_rev, n=size_fft)
        grad = 2.0 * conv_w[n - 1 : n - 1 + n]

        # diminishing step size
        eta = 0.5 / math.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, 1.0)

        # occasional score check
        if it % 50 == 0:
            cur = _score(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best = a.copy()
                print(f"[subgrad] it={it} best={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
def _gradient_via_softmax(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    size_fft: int,
) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv) w.r.t. a.
    Returns 2·(softmax_weights ⋆ a_rev).
    """
    n = a.shape[0]

    # soft‑max weights over convolution entries
    if alpha > 2_000:
        max_b = conv.max()
        mask = np.abs(conv - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(conv)] = 1.0
    else:
        shift = conv - conv.max()
        w = np.exp(alpha * shift)
        ws = w.sum()
        if ws == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_w = np.fft.irfft(W * A_rev, n=size_fft)
    g = 2.0 * conv_w[n - 1 : n - 1 + n]
    return g


# ----------------------------------------------------------------------
def _adam_optimize(
    init_a: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimizer using the soft‑max surrogate.
    Returns the best vector observed and its exact (max) score.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = _score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.07
    alpha_max = 50_000.0

    size_fft = _next_pow_two(3 * n - 2)
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[adam] it={it} best={best_score:.6f}")

        # gradient of soft‑max surrogate
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_onto_simplex(a, 1.0)

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    # final exact score for the best found
    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    start_a: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search: relocate a random fraction of mass
    from one index to another; keep the move if it improves the exact
    objective.  Periodic tiny jitter helps escaping shallow plateaus.
    """
    best = start_a.copy()
    best_score = _score(best)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue

        # transfer up to 30 % of the mass at i to j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = _score(jitter)
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point required by the grader.
    Returns a list of non‑negative heights (probability vector) that
    minimises ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    # safety margin for the final conversion / printing
    deadline = time.time() + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose a sensible length.
    try:
        base_n = int(len(height_sequence_1))  # type: ignore[name-defined]
    except Exception:
        base_n = 10000

    n = max(4000, min(base_n, 15000))

    # --------------------------------------------------------------
    # Phase 0 – quick exploration of hand‑crafted families.
    prev_seq = None
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        pass

    shape_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.15)
    candidates = _candidate_shapes(n, prev_seq)

    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        if time.time() >= shape_deadline:
            break
        scr = _score(cand)
        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
            print(f"[shape] new best score {best_score:.6f}")

    # Safeguard – should never happen
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent (exact max).
    subgrad_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.45)
    seq_subgrad, score_subgrad = _subgradient_descent(best_seq, n, subgrad_deadline)
    if score_subgrad < best_score:
        best_seq, best_score = seq_subgrad, score_subgrad
        print(f"[subgrad] improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation with a soft‑max surrogate.
    adam_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.30)
    seq_adam, score_adam = _adam_optimize(best_seq, n, adam_deadline)
    if score_adam < best_score:
        best_seq, best_score = seq_adam, score_adam
        print(f"[adam] improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – stochastic hill‑climbing polishing.
    hill_deadline = deadline
    seq_hill, score_hill = _hill_climb(best_seq, n, hill_deadline)
    if score_hill < best_score:
        best_seq, best_score = seq_hill, score_hill
        print(f"[hill] final best score {best_score:.6f}")

    # Final verification with the provided evaluate_sequence (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())  # type: ignore[name-defined]
        print(f"Final evaluate_sequence score: {final_score:.6f}")
    except Exception as exc:
        print("Warning: could not evaluate final sequence –", exc)

    return best_seq.tolist()
