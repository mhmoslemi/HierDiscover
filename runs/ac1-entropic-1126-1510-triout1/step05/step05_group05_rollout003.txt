def _optimize_gradient(a0, n, deadline):
    a = a0.copy()
    best_score = _objective(a) # uses max conv (not smoothed)
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.9
    eta = 0.5 # initial step size
    alpha = 1.0
    alpha_mult = 1.05
    min_alpha = 1.0
    max_alpha = 5000.0
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # compute smoothed gradient
        conv = _fft_self_convolve(a)  # length 2n-1
        # compute gradient
        grad = _gradient_softmax(a, conv, alpha, size_fft) # but we need size_fft chosen each iteration uncon: compute size = next pow2 >= 2n-1? Actually convolution length for gradient is also same.

        # Nesterov momentum: v = beta*v + (1-beta)*grad
        momentum = beta*momentum + (1-beta)*grad

        # step
        step = eta / np.sqrt(iteration) # decreasing step
        a = a - step * momentum

        # projection
        a = _project_onto_simplex(a, s=1.0)

        # compute exact objective occasionally for best tracking.
        if iteration % 100 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_a

def _optimize_smooth_gradient(a0, n, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    grad_mom = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.5
    alpha = 1.0
    alpha_mult = 1.03
    max_alpha = 5000.0
    size_fft = 1 << ((3 * n - 2).bit_length())
    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution:
        b = _fft_self_convolve(a) # length 2n-1
        # compute gradient with softmax
        grad = _gradient_softmax(a, b, alpha, size_fft)
        # momentum
        grad_mom = beta * grad_mom + (1 - beta) * grad
        # step size
        eta = eta0 / ( np.sqrt(it) )
        a = a - eta * grad_mom
        # projection
        a = _project_onto_simplex(a, s=1.0)
        if it % 200 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] n={n} it={it} best={best_score:.6f}")
        # increase alpha gradually
        if alpha < max_alpha:
            alpha = min(max_alpha, alpha * alpha_mult)
        # occasional jitter
        if it % 5000 == 0 and np.random.rand() < 0.1:
            a = a + np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)
    return best_a, best_score

"""
Search for a low‑max self‑convolution step‑function.

The algorithm proceeds in three stages:

1. **Random parametric sampling** – many candidates are drawn from a few
   simple families (arcsine, power‑law, beta, Gaussian, uniform and several
   convex mixtures).  Each candidate is projected onto the probability
   simplex and evaluated with a fast FFT based maximum‑convolution proxy.

2. **Smooth‑gradient descent** – the best few candidates are refined on the
   simplex using the gradient of a soft‑max surrogate for the maximum of the
   self‑convolution.  The gradient is computed via FFT, a Nesterov‑type
   momentum term is employed, and a temperature parameter α is annealed so
   that the surrogate gradually approaches the true max.  This stage
   substantially reduces the objective while remaining cheap.

3. **Local hill‑climbing** – a final stochastic refinement moves a small
   fraction of probability mass between random coordinates and accepts moves
   that improve the exact objective.

All work is limited to the supplied `budget_s` seconds.  The returned
sequence is a plain Python list of non‑negative numbers (it is automatically
scaled to sum to 1, which leaves the true `evaluate_sequence` objective
unchanged).  The code has no external dependencies other than NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, sum(x) = s}.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """Symmetric 1/(i·(n‑i))^p  (invert=True) or (i·(n‑i))^p  (invert=False)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) distribution on the index set."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
def _mixture_two(v1: np.ndarray, v2: np.ndarray, lam: float) -> np.ndarray:
    """
    Convex combination (1‑lam)*v1 + lam*v2.  Both vectors must already sum to 1.
    """
    w = (1.0 - lam) * v1 + lam * v2
    return _project_onto_simplex(w, s=1.0)


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a)  under the simplex constraint (sum(a)=1).
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the smoothed max  (1/α)·log‑sum‑exp(α·b_k)  w.r.t. a.
    Implements   g = 2 * (w ◦ a)_rev  where w is the soft‑max weights.
    """
    # compute soft‑max weights
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_smooth_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Smooth‑gradient descent on the simplex with momentum.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    # hyper‑parameters (tuned empirically)
    beta = 0.9                # momentum coefficient
    eta0 = 0.8                # initial step‑size factor
    alpha = 1.0               # temperature for soft‑max
    alpha_mult = 1.02
    max_alpha = 5000.0
    size_fft = 1 << ((3 * n - 2).bit_length())
    grad_mom = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # convolution and gradient
        b = _fft_self_convolve(a)
        grad = _gradient_softmax(a, b, alpha, size_fft)

        # momentum update
        grad_mom = beta * grad_mom + (1.0 - beta) * grad

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection
        a = a - eta * grad_mom
        a = _project_onto_simplex(a, s=1.0)

        # occasional exact evaluation
        if it % 200 == 0:
            cur_score = _objective(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] n={n} it={it} best={best_score:.6f}")

        # anneal temperature
        if alpha < max_alpha:
            alpha = min(max_alpha, alpha * alpha_mult)

        # tiny random jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: move a random fraction of mass from one index to another.
    Accept moves that strictly improve the exact objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        # transfer up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        cur_score = _objective(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_a, best_score

        if base_n < 500:
            base_n = 8000
    except Exception:
        base_n = 8000

    # admissible lengths (keep them ≤ 20000 for speed)
    length_factors = (0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    })
    # add a few hand‑picked values
    lengths.extend([5000, 8000, 12000, 16000, 20000])
    lengths = sorted({l for l in lengths if l <= 20000})

    # ------------------------------------------------------------------
    # bookkeeping for the global best
    best_seq = None
    best_score = float("inf")
    best_candidates = []   # list of (score, sequence)

    # ------------------------------------------------------------------
    # seed pool with some deterministic constructions
    def _register(seq: np.ndarray):
        nonlocal best_seq, best_score, best_candidates
        score = _objective(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        best_candidates.append((score, seq))

    # previous solution, if any
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        seq0 = np.maximum(seq0, 0.0)
        seq0 = _project_onto_simplex(seq0, s=1.0)
        _register(seq0)
        print(f"Prev‑solution score={best_score:.6f}")
    except Exception:
        pass

    # plain arcsine of the base length
    arcs = _arcsine_distribution(base_n)
    _register(arcs)
    print(f"Arcsine (n={base_n}) score={best_score:.6f}")

    # uniform baseline
    uni = _uniform_distribution(base_n)
    _register(uni)

    # ------------------------------------------------------------------
    # Phase 1: random parametric search
    families = (
        "arcsine", "power_inv", "power_dir", "beta",
        "gauss", "mix_arc_uni", "mix_arc_gauss", "mix_arc_beta"
    )
    search_deadline = deadline - 250.0   # reserve about 250 s for refinements
    while time.time() < search_deadline:
        n = int(np.random.choice(lengths))
        fam = np.random.choice(families)

        if fam == "arcsine":
            seq = _arcsine_distribution(n)

        elif fam == "power_inv":
            p = np.random.uniform(0.2, 1.2)
            seq = _power_law_distribution(n, p, invert=True)

        elif fam == "power_dir":
            p = np.random.uniform(0.2, 1.2)
            seq = _power_law_distribution(n, p, invert=False)

        elif fam == "beta":
            a = np.random.uniform(0.2, 5.0)
            seq = _beta_distribution(n, a)

        elif fam == "gauss":
            sigma = np.random.uniform(n / 20.0, n / 2.0)
            seq = _gaussian_distribution(n, sigma)

        elif fam == "mix_arc_uni":
            lam = np.random.uniform(0.0, 1.0)
            seq = _mixture_two(_arcsine_distribution(n),
                                _uniform_distribution(n), lam)

        elif fam == "mix_arc_gauss":
            lam = np.random.uniform(0.0, 1.0)
            sigma = np.random.uniform(n / 20.0, n / 2.0)
            seq = _mixture_two(_arcsine_distribution(n),
                                _gaussian_distribution(n, sigma), lam)

        elif fam == "mix_arc_beta":
            lam = np.random.uniform(0.0, 1.0)
            a = np.random.uniform(0.2, 5.0)
            seq = _mixture_two(_arcsine_distribution(n),
                                _beta_distribution(n, a), lam)

        else:
            seq = _uniform_distribution(n)

        seq = _project_onto_simplex(seq, s=1.0)
        score = _objective(seq)

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New global best (rand) len={n} fam={fam} score={best_score:.6f}")

        # maintain a small pool of promising candidates (size ≤ 12)
        if len(best_candidates) < 12:
            best_candidates.append((score, seq))
        else:
            # replace the worst entry if the new one is better
            worst_idx = max(range(len(best_candidates)),
                            key=lambda i: best_candidates[i][0])
            if score < best_candidates[worst_idx][0]:
                best_candidates[worst_idx] = (score, seq)

    # ------------------------------------------------------------------
    # Phase 2: smooth‑gradient refinement of the top candidates
    best_candidates.sort(key=lambda x: x[0])
    top_candidates = [c[1] for c in best_candidates[:4]]

    for cand in top_candidates:
        if time.time() >= deadline - 30.0:
            break
        n = cand.shape[0]
        # Allocate a slice of the remaining budget (max 60 s)
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(60.0, remaining * 0.6)

        refined_seq, refined_score = _optimize_smooth_gradient(cand, n, slice_deadline)
        refined_score = _objective(refined_seq)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"Refinement improved to {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Phase 3: final local hill‑climbing
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        refined_seq, refined_score = _local_refinement(best_seq, n, deadline)
        refined_score = _objective(refined_seq)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback (should never be needed)
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length = {best_seq.shape[0]})")
    return best_seq.tolist()
