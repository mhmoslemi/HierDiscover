"""
Propose a non‑negative step function with a low score for
``evaluate_sequence``.

Algorithm
---------
1. **Coarse family scan** – generate a few analytic candidates
   (arcsine, symmetric beta, random Dirichlet) for several lengths
   (500 … 12000).  Keep the best (lowest objective) as a seed.

2. **Smooth‐max Adam optimisation** – minimise a smooth approximation of
   ``max(a*a)``:
        L_γ(a) = (1/γ)·log∑_k exp(γ·(a*a)_k)
   The gradient of L_γ is 2·conv_rev(a, w) where w is the soft‑max
   over the convolution entries.  An Adam optimizer (with simplex
   projection after each step) is run for a schedule of increasing
   γ (0.1 → 80).  This gives a fine‑grained descent that can escape the
   plateaus faced by a pure sub‑gradient method.

3. **Local random‑transfer hill‑climbing** – small amounts of mass are
   moved between two random positions; improvements are kept.  Occasionally
   a tiny random jitter is added to jump out of flat regions.

The whole search stops after ``budget_s`` seconds (leaving a 2 s safety
margin).  The returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x) = s}.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p, p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))  # shift for stability
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) – a is assumed to sum to 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _smooth_softmax_weights(b: np.ndarray, gamma: float) -> np.ndarray:
    """
    Stable soft‑max weights  w_k ∝ exp(gamma·b_k) .
    When gamma=0 the weights are uniform.
    """
    if gamma == 0.0:
        return np.full_like(b, 1.0 / b.size)
    b_max = b.max()
    # shift by the maximum to avoid overflow
    exp_vals = np.exp(gamma * (b - b_max))
    w = exp_vals / exp_vals.sum()
    return w


# ----------------------------------------------------------------------
def _gradient_smooth_max(a: np.ndarray, gamma: float):
    """
    Returns (grad, b) where
        grad = ∂/∂a (smooth ≈ max(a*a))   (size n)
        b    = a*a (full convolution, length 2·n‑1)
    """
    b = _fft_self_convolve(a)
    w = _smooth_softmax_weights(b, gamma)
    # conv_rev = a_rev * w
    conv_rev = _fft_convolve(a[::-1], w)
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   gamma_schedule: list[float],
                   lr: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best sequence seen and its (exact) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0

    for gamma in gamma_schedule:
        # a modest cap per gamma – we will exit earlier if time runs out
        while time.time() < deadline:
            t += 1
            grad, _ = _gradient_smooth_max(a, gamma)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            # bias‑correction is optional; the raw Adam update works fine here
            step = lr * m / (np.sqrt(v) + eps)
            a -= step
            a = _project_onto_simplex(a, s=1.0)

            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam γ={gamma:.2f} iter={t} new best {cur_score:.6f}")

            # stop the inner loop after a few thousand steps to switch γ
            if t % 3000 == 0:
                break
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
def _local_hill_climb(a_start: np.ndarray,
                     deadline: float,
                     max_transfer: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Simple random‑transfer hill climbing.
    Returns the best sequence encountered and its (exact) score.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    it = 0

    while time.time() < deadline:
        it += 1
        # exponentially decreasing transfer magnitude
        transfer = max_transfer * (0.5 ** (it / 15000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={it} improved best {best_score:.6f}")

        # occasional jitter to escape flat plateaux
        if it % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_a + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_score
