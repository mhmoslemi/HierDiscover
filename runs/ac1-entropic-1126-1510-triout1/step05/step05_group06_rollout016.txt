def _beta_distribution_shape(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (alpha - 1) * (n - i - 0.5) ** (beta - 1)
    # for alpha or beta less than 1, we might get division by 0 at edges; ensure no zero param.
    # Use max to avoid nan.
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0:
        vals = np.full(n, 1.0/n)
        total = 1.0
    return vals / total

def _adam_optimize(a0, n, deadline):
    a = a0.copy()
    # avoid zeros: add small epsilon before log
    eps = 1e-12
    y = np.log(a + eps)

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta1 = 0.9
    beta2 = 0.999
    epsilon_adam = 1e-8

    lr0 = 0.05
    t = 0

    # softmax surrogate temperature schedule
    alpha = 0.5
    alpha_mul = 1.02
    alpha_max = 400.0
    size_fft = 1 << ((3 * n - 2).bit_length())

    best_seq = a.copy()
    best_score_internal = 2.0 * n * _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        a = np.exp(y - np.max(y))  # compute softmax stable
        a /= a.sum()
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        score_internal = 2.0 * n * max_conv
        if score_internal < best_score_internal - 1e-14:
            best_score_internal = score_internal
            best_seq = a.copy()

        # gradient
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # compute gradient wrt logits
        # grad_y = a * (grad - np.dot(a, grad))
        dot = np.dot(a, grad)
        grad_y = a * (grad - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)  # diminishing step

        y -= lr * m_hat / (np.sqrt(v_hat) + epsilon_adam)

        # occasional jitter
        if t % 2000 == 0 and np.random.rand() < 0.1:
            # random noise to logits
            y += np.random.randn(n) * 1e-5

    # After loop, return best_seq as list and internal score
    return best_seq, best_score_internal

def _annealing_refine(a_start, deadline):
    a = a_start.copy()
    n = len(a)
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    # initial temperature based on current best score; say T0 = 0.1*best_score
    T = best_score * 0.2
    # cooling schedule: T *= 0.995 each iteration
    while time.time() < deadline:
        # choose random i,j, epsilon
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

The evaluation function is  

\[
E(a)=\frac{2n\cdot\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 .
\]

All candidates are normalised to `sum(a)=1`; therefore  

\[
E(a)=2n\;\max (a*a).
\]

The task is to minimise the maximum entry of the *linear* convolution
`a*a`.  
The optimum is reached when the convolution is as flat as possible –
the ratio “peak / average’’ (the score) is close to 1.  

The code below explores the search space in three stages  

1. **Diverse initial shapes** – symmetric and asymmetric Beta‑distributions,
   mixtures with the uniform law and some pure random vectors.
2. **Adam‑optimisation on a smooth surrogate** – the
   `log‑sum‑exp` (soft‑max) of the convolution is used as a differentiable
   approximation of the `max`.  The variables are the *logits* of the
   probability vector, i.e. `a = softmax(y)`.  Adam’s adaptive steps,
   a diminishing learning‑rate and a slowly increasing temperature give a
   fast, stable descent.
3. **Simulated‑annealing hill‑climbing** – after the gradient phase a
   cheap stochastic move (transfer a tiny mass from one index to another)
   is accepted if it improves the true score or with a probability that
   decreases with a temperature schedule.  This refines the solution
   beyond the smooth surrogate.

The whole routine respects the 1000 s budget, continually updates the
best sequence found and finally returns it as a plain Python list.

```python
"""
Propose a non‑negative step function (list of heights) with a low
evaluate_sequence score.

Algorithm
---------
1.  Build a set of diverse initial vectors of a given length n.
    • symmetric / asymmetric Beta‑shapes  a_i ∝ (i+½)^{α‑1}(n‑i‑½)^{β‑1}
    • mixtures of the above with the uniform law
    • a few random vectors
2.  Optimise each candidate with Adam on the smooth surrogate
    f_α(a) = (1/α)·log∑exp(α·(a*a)).
    The parameters are logits y (a = softmax(y)), which guarantees
    non‑negativity and the simplex constraint automatically.
    • temperature α slowly grows (soft‑max → true max)
    • learning‑rate η_t = η₀/√t, Adam moments (β₁,β₂) = (0.9,0.999)
3.  Finish with a simulated‑annealing hill‑climber that moves a tiny
    amount of mass between two positions whenever the true score
    (evaluate_sequence) improves, otherwise accepts with probability
    exp(‑Δ/T).  Temperature decays geometrically.
4.  Keep the best sequence ever seen (global best) and return it.

All helper functions are top‑level, no closures or lambdas are used.
The total running time is limited to the supplied budget (≈1000 s).
"""

import time
import numpy as np
from math import log, exp, sqrt

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, very fast)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Exact sub‑gradient of max(conv) when the max occurs at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def _gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·conv) w.r.t. a.
    Implements the same trick as in the reference code.
    """
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv2 = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv2[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
# Families of initial shapes (Beta‑type distributions)
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric / asymmetric Beta‑type shape.
    a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1},   α,β>0.
    Returned vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = n - i - 0.5
    # Prevent overflow / underflow
    vals = np.power(left, alpha - 1.0, where=left > 0, out=np.zeros_like(left))
    vals *= np.power(right, beta - 1.0, where=right > 0, out=np.ones_like(right))
    # Replace NaNs/Infs by zero and renormalise
    vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total


def _beta_mixture(n: int,
                  alpha: float,
                  beta: float,
                  mix_uniform: float) -> np.ndarray:
    """
    Mixture (1‑mix_uniform)·Beta(α,β)  +  mix_uniform·Uniform.
    mix_uniform∈[0,1].
    """
    beta_part = _beta_shape(n, alpha, beta)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = (1.0 - mix_uniform) * beta_part + mix_uniform * uniform
    # Project onto the simplex (tiny numerical errors may appear)
    return _project_onto_simplex(mixed, s=1.0)


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth surrogate of max(a*a).
    Returns the best vector encountered (as a numpy array) and its
    internal (2·n·max) score.
    """
    # logits representation guarantees positivity and sum‑1 automatically
    eps = 1e-12
    y = np.log(a0 + eps)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    eta0 = 0.04                # initial learning‑rate factor
    t = 0

    # soft‑max temperature schedule
    alpha = 0.5                # start with a very smooth surrogate
    alpha_mul = 1.02
    alpha_max = 300.0

    size_fft = 1 << ((3 * n - 2).bit_length())

    best_seq = None
    best_score = np.inf

    while time.time() < deadline:
        t += 1

        # softmax → a (guaranteed to lie on the simplex)
        max_y = np.max(y)
        a = np.exp(y - max_y)
        a /= a.sum()

        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()

        # gradient of the surrogate (soft‑max) or true sub‑gradient
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad)
        grad_y = a * (grad - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = eta0 / sqrt(t)

        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional tiny random jitter to escape plateaus
        if t % 2500 == 0 and np.random.rand() < 0.12:
            y += np.random.randn(n) * 1e-4

    # final best vector as a probability distribution
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass transfer moves)
# ----------------------------------------------------------------------
def _annealing_refine(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with a simple annealing schedule.
    Returns the refined vector and the true evaluate_sequence score.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_seq = a.copy()
    best_int_score = 2.0 * n * np.max(_fft_self_convolve(a))

    # temperature proportional to the current score
    T = best_int_score * 0.2
    cooling = 0.9999

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 5 % of the source mass
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue

        cur_score = 2.0 * n * np.max(_fft_self_convolve(cand))
        delta = cur_score - best_int_score

        if delta < 0 or np.random.rand() < exp(-delta / max(T, 1e-12)):
            a = cand
            if cur_score < best_int_score:
                best_int_score = cur_score
                best_seq = cand.copy()

        T = max(T * cooling, 1e-8)

    # evaluate with the black‑box function
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        final_score = best_int_score
    return best_seq, final_score
```
