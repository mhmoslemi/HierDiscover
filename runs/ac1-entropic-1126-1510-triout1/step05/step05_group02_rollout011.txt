def _smooth_max_and_grad(a, gamma):
    n = a.shape[0]
    b = _fft_self_convolve_full(a)
    max_b = b.max()
    shift = b - max_b
    # compute w = exp(gamma* shift) to have values in (0,1]
    # Avoid overflow: shift <=0, gamma positive. This yields values <=1.
    w = np.exp(gamma * shift)
    w_sum = w.sum()
    if w_sum == 0:
        # all zero due to underflow; fallback to uniform weight on active max indices.
        active = np.where(b >= max_b - 1e-12)[0]
        w = np.zeros_like(b)
        w[active] = 1.0 / active.size
    else:
        w /= w_sum

    # gradient wrt a of sum_{k} w_k b_k = 2 * conv_rev(w)
    a_rev = a[::-1]

"""
Propose a low‑score step‑function.

The algorithm consists of three stages:

1️⃣  **Broad parametric search** – a collection of Beta‑distributions
    (including asymmetric ones and random mixtures) is sampled on several
    sequence lengths.  The best raw candidate (according to the
    scale‑invariant score 2·n·max(a*a)) is retained.

2️⃣  **Smooth‑max Adam optimisation** – the retained candidate is refined
    by minimising the log‑sum‑exp of the self‑convolution (a smooth
    approximation of the ℓ∞ norm).  Parameters are kept in the log‑soft‑max
    domain to satisfy the simplex constraint automatically.
    An Adam optimiser with a slowly increasing “sharpness’’ parameter
    (γ) provides fast, stable descent.

3️⃣  **Final evaluation** – the best sequence found is passed to the
    supplied ``evaluate_sequence`` function and returned as a plain list.

The whole routine respects the 1000‑second budget, works with any
previously stored solution ``height_sequence_1`` and never produces
negative or overly large entries.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility returning the next power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors (real FFT)."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    return e / e.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


def _smooth_max_and_grad(a: np.ndarray, gamma: float):
    """
    Compute a smooth approximation of max_k (a*a)_k together with its gradient.
    The smooth max is L = (1/γ)·log Σ exp(γ·b_k) with b = a*a.
    Returns (L, grad_a) where grad_a has the same shape as a.
    """
    n = a.shape[0]
    b = _fft_self_convolve_full(a)                 # length 2n‑1
    max_b = b.max()

    # stable log‑sum‑exp weights (softmax on b)
    shift = b - max_b                              # ≤ 0
    w = np.exp(gamma * shift)                      # in (0,1]
    w_sum = w.sum()
    if w_sum == 0.0:                               # numerical safety
        active = np.where(b >= max_b - 1e-12)[0]
        w[:] = 0.0
        w[active] = 1.0 / active.size
    else:
        w /= w_sum                                 # now Σ w = 1

    # gradient of Σ w_k·b_k  (∂b_k/∂a = 2 * conv_rev(e_k))
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1: n - 1 + n]
    # smooth max value (log‑sum‑exp)
    smooth_val = max_b + np.log(w_sum) / gamma
    return smooth_val, grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence score.
    The routine performs a parametric Beta search followed by a smooth‑max
    Adam refinement.  Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric search (Beta families & mixtures)
    # --------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        base_seq = None

    # lengths to explore – larger lengths give more freedom but cost more
    lengths = [2000, 4000, 6000, 8000]

    best_seq = None
    best_score = float("inf")
    best_len = None

    # Helper to compute the scale‑invariant raw score (2·n·maxconv)
    def _raw_score(vec: np.ndarray, n: int) -> float:
        return 2.0 * n * _fft_self_convolve_max(vec)

    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"[{time.time():.1f}] Exploring length {n}")

        # ---- a) Interpolate previous solution if available ----
        if base_seq is not None:
            cand = _interpolate_to_length(base_seq, n)
            cand = _project_onto_simplex(cand)
            sc = _raw_score(cand, n)
            if sc < best_score:
                best_score, best_seq, best_len = sc, cand.copy(), n
                print(f"   init=prev  score={best_score:.6f}")

        # ---- b) Simple baselines ----
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform, n)
        if sc < best_score:
            best_score, best_seq, best_len = sc, uniform.copy(), n
            print(f"   init=uniform score={best_score:.6f}")

        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine, n)
        if sc < best_score:
            best_score, best_seq, best_len = sc, arcsine.copy(), n
            print(f"   init=arcsine score={best_score:.6f}")

        # ---- c) Grid over symmetric Beta(α,α) (very U‑shaped allowed) ----
        alphas = np.concatenate((
            np.linspace(0.02, 0.12, 12, endpoint=False),
            np.linspace(0.12, 0.6, 12, endpoint=False),
            np.linspace(0.6, 2.0, 15),
            np.linspace(2.0, 5.0, 8)
        ))
        for a in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a, a)
            sc = _raw_score(vec, n)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   symBeta a={a:.3f} score={best_score:.6f}")

        # ---- d) Random asymmetric Beta distributions ----
        for _ in range(150):
            if time.time() >= deadline:
                break
            a = np.random.rand() * 5.0 + 0.02
            b = np.random.rand() * 5.0 + 0.02
            vec = _beta_distribution(n, a, b)
            sc = _raw_score(vec, n)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   beta({a:.2f},{b:.2f}) score={best_score:.6f}")

        # ---- e) Random mixtures of up to three Beta components ----
        for _ in range(200):
            if time.time() >= deadline:
                break
            comps = np.random.randint(2, 4)            # 2 or 3 components
            weights = np.random.rand(comps)
            weights /= weights.sum()
            mix = np.zeros(n, dtype=np.float64)
            for w in weights:
                a = np.random.rand() * 5.0 + 0.02
                b = np.random.rand() * 5.0 + 0.02
                mix += w * _beta_distribution(n, a, b)
            mix = _project_onto_simplex(mix)
            sc = _raw_score(mix, n)
            if sc < best_score:
                best_score, best_seq, best_len = sc, mix.copy(), n
                print(f"   mix{comps} score={best_score:.6f}")

    if best_seq is None:
        # Very unlikely, but keep a safe fallback
        n = 3000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_len = n
        best_score = _raw_score(best_seq, n)
        print("Fallback uniform used.")

    print(f"Best raw score after param search: {best_score:.6f} (len={best_len})")

    # --------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam refinement on the best candidate
    # --------------------------------------------------------------
    n = best_len
    a = best_seq.copy()
    a = _project_onto_simplex(a)               # enforce exact simplex
    eps = 1e-12
    z = np.log(a + eps)                        # log‑soft‑max parameters

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    epsadam = 1e-12
    lr0 = 0.08                                 # base learning rate

    best_a = a.copy()
    best_val = 2.0 * n * _fft_self_convolve_max(a)
    it = 0

    # Main optimisation loop
    while time.time() < deadline:
        it += 1
        a = _softmax(z)                         # always on the simplex

        # gradually increase the “sharpness’’ of the smooth max
        gamma = 5.0 + min(200.0, it * 0.001)     # 5 → ~200 over the run

        # smooth max value and its gradient w.r.t. a
        smooth_val, grad_a = _smooth_max_and_grad(a, gamma)

        # chain rule through the soft‑max
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + it * 1e-5)            # tiny decay
        step = lr * m_hat / (np.sqrt(v_hat) + epsadam)
        z -= step

        # occasional tiny random jitter to help escape plateaus
        if (it % 2500 == 0) and (np.random.rand() < 0.12):
            z += np.random.randn(n) * 5e-7

        # periodic exact evaluation (every few hundred iterations)
        if it % 400 == 0:
            cur_val = 2.0 * n * _fft_self_convolve_max(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] iter {it:8d}  score={best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final evaluation and return
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_a)})")
    return best_a.tolist()
