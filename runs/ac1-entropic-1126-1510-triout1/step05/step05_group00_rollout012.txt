"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. Choose a few candidate lengths (based on the previous best length or a
   default of 4000).  For each length we enumerate several analytic families
   (arcsine, symmetric beta, symmetric power‑law, triangular, exponential,
   sinusoid) with a wide range of shape parameters and keep the best
   distribution (the one with smallest 2·n·max(conv) – the objective when the
   sequence is normalised to sum = 1).

2. Starting from that best distribution we run a smooth‑approximation
   optimisation: replace the nonsmooth max by a soft‑max (log‑sum‑exp) with a
   temperature that is slowly increased.  The gradient of the smoothed
   objective is obtained analytically via FFT‑based convolutions.  An Adam
   optimiser (momentum + adaptive steps) updates the coefficients, after
   each step projecting back onto the probability simplex.

3. A cheap combinatorial hill‑climber performs random mass transfers between
   two coordinates (and occasional jitter) to squeeze out the remaining
   marginal improvement.

All work is bounded by the supplied time budget; the best sequence found
across all lengths is returned.  The returned list is already non‑negative
and sums to 1, so ``evaluate_sequence`` will compute the same objective
value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective value for a normalised (sum=1) sequence: 2·n·max(a*a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p, p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # compute in log‑space to avoid overflow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law (x·(1‑x))^p on a grid, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64) + 0.5
    left = i
    right = n - i
    if p >= 0:
        vals = (left ** p) * (right ** p)
    else:
        # use log‐space for p<0 (heavy tails)
        logv = p * (np.log(left) + np.log(right))
        vals = np.exp(logv)
    vals[vals < 0] = 0.0
    s = vals.sum()
    if s == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= s
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _sinusoid_distribution(n: int) -> np.ndarray:
    """Half‑period sinusoid (sin(π·x)) on [0,1], normalised to sum 1."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _adam_softmax_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max approximation of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1

        # full self‑convolution
        conv = _fft_self_convolve(a)

        # temperature schedule – slowly drive alpha up to ~200
        t = min(1.0, iter_cnt / 25000.0)
        alpha = 5.0 + 195.0 * t

        # soft‑max weights (stable)
        z = alpha * conv
        z_max = np.max(z)
        expz = np.exp(z - z_max)
        w = expz / expz.sum()

        # gradient of the smoothed objective
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam iter={iter_cnt} new best {best_score:.6f}")

        # occasional early‑stop check
        if iter_cnt % 5000 == 0 and time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Purely combinatorial local search: random mass transfers."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    it = 0
    max_transfer = 0.07  # fraction of the total mass

    while time.time() < deadline:
        it += 1
        # geometric decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (it / 20000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hillclimb iter={it} improved best {best_score:.6f}")

        # tiny jitter every few thousand steps
        if it % 4000 == 0 and np.random.rand() < 0.08:
            a = best_a + np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_score


# ----------------------------------------------------------------------

    """Search for a non‑negative step function with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # spare a couple of seconds for cleanup

    # ------------------------------------------------------------------
    # Base length (use the previous construction if available)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    # Lengths to explore

    for f in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    for _ in range(5):
        n = np.random.randint(500, min(20000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------

    beta_ps = [0.01, 0.02, 0.05, 0.10, 0.20, 0.40, 0.60, 0.80, 1.0, 1.5, 2.0]
    lambdas = [0.001, 0.0025, 0.005, 0.01, 0.02]

    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # 1️⃣  Coarse analytic sweep (keep the best raw candidate)
        local_best_seq = None
        local_best_score = float("inf")

        def consider(candidate: np.ndarray):
            nonlocal local_best_seq, local_best_score
            sc = _max_convolution_score(candidate)
