def compute_gradient(a, conv, alpha):
    # softmax surrogate gradient
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad_a = 2.0 * conv_rev[n-1:n-1+n]
    return grad_a

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1
    overall_best_seq = None
    overall_best_score = float('inf')
    # define lengths list
    lengths = [2000, 3000, 5000, 8000, 12000, 20000, 30000, 40000]
    lengths = [l for l in lengths if l <= 50000]
    # shape list
    alphas = [0.2, 0.3, 0.4, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0]
    betas = alphas # start symmetric
    skew_factors = [0.2, 0.5, 2.0, 5.0] # for asymmetric combos
    candidate_cache = []
    # generate candidates
    for n in lengths:
        # generate symmetric Beta
        for a in alphas:
            b = a
            vec = _beta_distribution(n, a, b)
            candidate_cache.append((n, vec))
        # generate asymmetric
        for a in alphas:
            for factor in skew_factors:
                b = a * factor
                vec = _beta_distribution(n, a, b)
                candidate_cache.append((n, vec))
        # add sin_power
        for power in [1, 2, 3, 4]:
            x = np.arange(n, dtype=float) + 0.5
            v = np.sin(np.pi * x / n) ** power
            v = v / v.sum()
            candidate_cache.append((n, v))
        # add triangular
        tri = np.concatenate([np.arange(1, n//2+1), np.arange(n//2, 0, -1)])
        tri = tri[:n]  # handle odd n
        tri = tri / tri.sum()
        candidate_cache.append((n, tri))
    # Evaluate cheap scores

def _refine(a0, n, deadline):
    # use log representation
    eps = 1e-12
    x = np.log(a0 + eps)  # shape n
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0
    best_a = a0.copy()
    best_score = cheap_score(a0)
    # temperature schedule: start small alpha (0.05) and increase multiplicatively
    alpha = 0.05
    alpha_factor = 1.0015
    alpha_max = 5000
    while time.time() < deadline:
        t += 1
        a = np.exp(x)
        a = a / a.sum()
        conv = _fft_self_convolve(a)
        # compute gradient w.r.t a
        if alpha < 1000.0:
            grad_a = _softmax_gradient(a, conv, alpha)
            alpha = min(alpha_max, alpha * alpha_factor)
        else:
            grad_a = _exact_gradient(a, conv)
        # gradient w.r.t x (softmax)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.1 / (np.sqrt(t) + 1)  # decreasing step
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # compute cheap score
        cur_score = cheap_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # early stop if score already below target (<1.51 maybe)
        if best_score <= 1.45:
            break
    return best_a, best_score

"""
Improved step‑function search.

The algorithm proceeds in three stages:

1. **Broad candidate sweep** – for a set of lengths (≈2 k … 40 k) a
   library of analytically‑motivated probability vectors is built:
   symmetric and skewed Beta densities, power‑law “edge‑heavy” densities,
   sine‑power shapes, a triangular shape and (if present) the previous
   construction.  All candidates are normalised to sum = 1 and are
   cheap‑scored via an FFT self‑convolution (score = 2·n·max(conv)).
   The ten best candidates (by this surrogate) are kept for refinement.

2. **Adam‑based surrogate optimisation** – each kept candidate is
   optimised on the simplex using an exponentiated‑gradient (softmax)
   parametrisation `a = softmax(x)`.  The loss is a smoothed version of
   `max(conv)` obtained by a temperature‑controlled soft‑max over the
   convolution entries.  The temperature (parameter `alpha`) starts
   small (very smooth) and is multiplied by a factor each iteration;
   once it exceeds a threshold a true sub‑gradient (`exact_gradient`)
   is used.  Adam with bias‑correction supplies an adaptive step size.
   The best iterate (lowest cheap score) is remembered.

3. **Final verification** – the best refined vector is evaluated with
   the official `evaluate_sequence`.  If time permits additional
   jitter‑perturbations of the global best are tried.

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list of non‑negative
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, gamma: float) -> np.ndarray:
    """Edge‑heavy distribution ~ i^{-γ} + (n−i)^{-γ}."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    vals = (idx ** -gamma) + ((n + 0.5 - idx) ** -gamma)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _sin_power_distribution(n: int, p: float) -> np.ndarray:
    """Discrete sin^{p} shape, normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = np.sin(np.pi * x) ** p
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Simple triangular shape, symmetric, normalised."""
    half = (n + 1) // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[:-1][::-1] if n % 2 == 0 else inc[::-1]
    vals = np.concatenate([inc, dec])
    vals = vals[:n]
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def cheap_score(a: np.ndarray) -> float:
    """Fast surrogate score = 2·n·max(a*a) (assumes sum(a)=1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    Returns ∂max/∂a (vector length n).
    """
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max, using a uniform weight over all
    (approximately) active indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    # diagonal correction for indices i where k = 2*i (self‑product)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _refine_candidate(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector with Adam on the soft‑max parametrisation.
    Returns the best vector seen (according to cheap_score) and its score.
    """
    n = a0.shape[0]
    eps = 1e-12
    # initialise log‑parameters (softmax)
    x = np.log(a0 + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # temperature schedule (alpha = 1/temperature)
    alpha = 0.05          # very smooth at start
    alpha_factor = 1.004
    alpha_switch = 4000.0

    best_a = a0.copy()
    best_sc = cheap_score(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- reconstruct a from x (softmax) -----
        a = np.exp(x)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
            a_sum = 1.0
        a = a / a_sum

        # ----- convolution & gradient -----
        conv = _fft_self_convolve(a)
        if alpha < alpha_switch:
            grad_a = _softmax_gradient(a, conv, alpha)
            alpha = min(alpha_switch, alpha * alpha_factor)
        else:
            grad_a = _exact_gradient(a, conv)

        # ----- back‑prop to log‑space -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (≈ η₀ / sqrt(it))
        eta0 = 0.8
        lr = eta0 / (np.sqrt(it) + 1.0)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- cheap evaluation -----
        cur_sc = cheap_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            # occasional progress report
            if it % 2000 == 0:
                print(f"[{time.time():.1f}] refinement iter={it} n={n} score={cur_sc:.6f}")

        # early exit if we are already very good
        if best_sc <= 1.45:
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
def _initial_vector(prev_len: int, target_len: int) -> np.ndarray:
    """
    Interpolate the globally available `height_sequence_1` (if any) to a
    new length.  Falls back to a uniform vector.
    """
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.size != target_len:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, target_len)
            a = np.interp(x_new, x_old, prev)
        else:
            a = prev.copy()
        a = _project_onto_simplex(a, s=1.0)
        return a
    except Exception:
        return np.full(target_len, 1.0 / target_len, dtype=np.float64)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Searches for a non‑negative step‑function that
    minimises `evaluate_sequence`.  Respects the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # tiny guard for finalisation

    # ------------------------------------------------------------------
    # 1. Choose lengths to explore.
    #    If a previous construction exists we centre around its length.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    length_factors = (0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0, 4.0)
    lengths = sorted(
        max(500, int(base_n * f)) for f in length_factors if int(base_n * f) >= 500
    )
    # cap the longest length to keep FFT sizes reasonable
    lengths = [min(l, 50000) for l in lengths]
    lengths = list(dict.fromkeys(lengths))   # deduplicate while preserving order

    # ------------------------------------------------------------------
    # 2. Build a sizeable library of analytic candidates.
    candidates = []   # list of (n, vector)

    # shape parameter grids
    beta_alphas = [0.2, 0.3, 0.4, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 5.0]
    skew_factors = [0.2, 0.5, 2.0, 5.0]
    power_gammas = [0.5, 1.0, 1.5, 2.0, 2.5]
    sin_powers = [1, 2, 3, 4, 5]

    for n in lengths:
        # symmetric Beta
        for a in beta_alphas:
            candidates.append((n, _beta_distribution(n, a, a)))
        # skewed Beta
        for a in beta_alphas:
            for f in skew_factors:
                candidates.append((n, _beta_distribution(n, a, a * f)))
        # power‑law edge‑heavy
        for g in power_gammas:
            candidates.append((n, _power_law_distribution(n, g)))
        # sine‑power shapes
        for p in sin_powers:
            candidates.append((n, _sin_power_distribution(n, p)))
        # triangular
        candidates.append((n, _triangular_distribution(n)))
        # previous construction (interpolated) and uniform
        candidates.append((n, _initial_vector(None, n)))               # uniform fallback
        candidates.append((n, _initial_vector(n, n)))                 # interpolated previous if any

        # a few random Dirichlet draws (cheap but give diversity)
        for _ in range(2):
            vec = np.random.dirichlet(np.ones(n))
            candidates.append((n, vec))

    # ------------------------------------------------------------------
    # 3. Cheap scoring & keep the best few.
    print(f"[{time.time():.1f}] Scoring {len(candidates)} analytic candidates...")
    scored = []
    for n, vec in candidates:
        # safety: ensure non‑negativity and normalisation
        vec = np.maximum(vec, 0.0)
        vec_sum = vec.sum()
        if vec_sum == 0.0:
            continue
        vec = vec / vec_sum
        sc = cheap_score(vec)
        scored.append((sc, n, vec))

    scored.sort(key=lambda t: t[0])
    top_k = 10
    selected = []
    seen_lengths = set()
    for sc, n, vec in scored:
        # keep at most one candidate per length (the best for that length)
        if n not in seen_lengths:
            selected.append((n, vec, sc))
            seen_lengths.add(n)
        if len(selected) >= top_k:
            break

    print(f"[{time.time():.1f}] Selected {len(selected)} seeds for refinement.")

    # ------------------------------------------------------------------
    # 4. Refine each seed with Adam optimisation.
    overall_best_seq = None
    overall_best_score = float("inf")

    for idx, (n, a0, cheap_sc) in enumerate(selected):
        if time.time() >= deadline - 5.0:
            break   # not enough time left

        # give each seed a slice of the remaining budget
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(60.0, remaining * 0.4)   # ≤60 s per seed

        print(f"[{time.time():.1f}] Refining seed {idx} length={n} cheap_score={cheap_sc:.6f}")

        refined_a, refined_sc = _refine_candidate(a0, sub_deadline)

        # evaluate with the official black‑box (may be a bit slower)
        try:
            final_score = evaluate_sequence(refined_a.tolist())
        except Exception:
            final_score = refined_sc  # fallback to surrogate

        print(f"[{time.time():.1f}] seed {idx} final_score={final_score:.6f}")

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = refined_a.copy()
            print(f"*** New global best: {overall_best_score:.6f}")

        if overall_best_score <= 1.503:
            print("Target achieved – exiting early.")
            return overall_best_seq.tolist()

    # ------------------------------------------------------------------
    # 5. Optional jitter‑based local search around the current best.
    if overall_best_seq is not None and time.time() < deadline - 5.0:
        n_best = overall_best_seq.shape[0]
        for it in range(5):
            if time.time() >= deadline - 5.0:
                break
            jitter = np.random.randn(n_best) * 1e-4
            a_pert = overall_best_seq + jitter
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= a_pert.sum()
            sub_deadline = time.time() + min(20.0, deadline - time.time())
            refined_a, _ = _refine_candidate(a_pert, sub_deadline)
            try:
                final_score = evaluate_sequence(refined_a.tolist())
            except Exception:
                final_score = cheap_score(refined_a)
            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = refined_a.copy()
                print(f"*** Improved via jitter: {overall_best_score:.6f}")
                if overall_best_score <= 1.503:
                    break

    # ------------------------------------------------------------------
    # 6. Fallback – return something reasonable if nothing succeeded.
    if overall_best_seq is None:
        n_fallback = 4000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("[fallback] Used uniform vector.")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
