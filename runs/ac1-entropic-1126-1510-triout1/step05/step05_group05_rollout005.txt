"""
Propose a high‑quality non‑negative step‑function.

The algorithm consists of three phases within the 1000 s budget:

1. **Broad random sampling** – many candidate probability vectors are
   generated from a variety of parametric families (arcsine, power‑law,
   beta, Gaussian, mixtures, Dirichlet‑like, exponentiated‑Gaussian).
   The vectors are forced to be symmetric (a_i = a_{n‑1‑i}) and projected
   onto the simplex.  The best few candidates are kept in a small elite
   pool.

2. **Adaptive exponentiated‑gradient refinement** – each elite candidate
   is refined by mirror‑descent on the simplex.  The update uses an
   AdaGrad‑style per‑coordinate step size and a soft‑max surrogate for
   the ``max`` in the objective.  When the temperature becomes large a
   true sub‑gradient (averaged over all near‑max indices) is used.
   After every update the iterate is symmetrised and projected back onto
   the simplex.

3. **Local stochastic hill‑climbing** – the current best sequence is
   further improved by repeatedly moving a small random fraction of mass
   from one coordinate to another, accepting only improvements.

The objective used internally is the scale‑invariant
``2·n·max(a∗a)`` (the denominator of ``evaluate_sequence`` is 1 because
the iterates are normalised to sum = 1).  The function finally returns
the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector (sum=1)."""
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _subgradient_max_multi(a: np.ndarray,
                           b: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """
    Average sub‑gradient of max(a*a) over all indices whose value is within
    ``tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = b.max()
    active = np.where(b >= max_val - tol)[0]
    if active.size == 0:          # should not happen
        active = np.array([int(np.argmax(b))])
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    grad /= float(active.size)
    return grad


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the (1/α)·log‑sum‑exp(α·b) surrogate w.r.t. a.
    Implements the formula g = 2·conv_rev(w, a_rev).
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_exponential_adaptive(a0: np.ndarray,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adaptive exponentiated‑gradient minimisation on the simplex.
    Uses AdaGrad step sizes, a temperature schedule for the soft‑max
    surrogate and averages sub‑gradients over an active set.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_seq = a.copy()
    best_score = _objective(a)

    # hyper‑parameters
    eta0 = 1.0                # base learning‑rate for AdaGrad
    alpha = 1.0               # start temperature
    alpha_mul = 1.02
    alpha_max = 15000.0
    alpha_thr = 2000.0        # switch to sub‑gradient after this
    size_fft = 1 << ((3 * n - 2).bit_length())

    # AdaGrad accumulator
    G = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # convolution and current score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}]  n={n}  it={it}  best={best_score:.6f}")

        # gradient computation
        if alpha > alpha_thr:
            grad = _subgradient_max_multi(a, b, tol=1e-12)
        else:
            grad = _gradient_softmax(a, b, alpha, size_fft)

        # AdaGrad accumulator and per‑coordinate step
        G += grad * grad
        step = eta0 / (np.sqrt(G) + 1e-12)

        # exponentiated‑gradient update (mirror descent on KL)
        exponent = -step * grad
        # clip to avoid overflow in exp
        exponent = np.clip(exponent, -50.0, 50.0)
        a = a * np.exp(exponent)

        # enforce symmetry and non‑negativity
        a = np.maximum(a, 0.0)
        a = (a + a[::-1]) * 0.5

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()

    return best_seq, best_score

def _local_refinement(a_start: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random‑move hill‑climbing that respects symmetry.
    A random fraction (≤10 %) of mass is transferred from i to j; the
    move is kept only if it improves the objective.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # because of symmetry we move the same amount from the mirror of i
        # (unless i is the centre of an odd‑length vector)
        eps = a[i] * np.random.rand() * 0.1
        if eps <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps

        # also mirror the move
        im = n - 1 - i
        jm = n - 1 - j
        if im != i:
            a_candidate[im] -= eps
        if jm != j:
            a_candidate[jm] += eps

        if np.min(a_candidate) < 0.0:
            continue
        a_candidate = a_candidate / a_candidate.sum()
        cur_score = _objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _symmetrise(v: np.ndarray) -> np.ndarray:
    """Force v to be symmetric (v[i] = v[n‑1‑i])."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
def _candidate_from_family(n: int, family: str) -> np.ndarray:
    """
    Produce a random candidate probability vector belonging to ``family``.
    The result is symmetrised and projected onto the simplex.
    """
    i = np.arange(n, dtype=np.float64)

    if family == "arcsine":
        vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    elif family == "power_invert":
        p = np.random.uniform(0.2, 1.2)
        prod = (i + 0.5) * (n - i - 0.5)
        vals = prod ** (-p)
    elif family == "power_direct":
        p = np.random.uniform(0.2, 1.2)
        prod = (i + 0.5) * (n - i - 0.5)
        vals = prod ** p
    elif family == "beta":
        a = np.random.uniform(0.2, 5.0)
        x = (i + 0.5) / n
        vals = (x ** (a - 1.0)) * ((1.0 - x) ** (a - 1.0))
    elif family == "gauss":
        sigma = np.random.uniform(n / 20.0, n / 2.0)
        center = (n - 1) / 2.0
        vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    elif family == "mix_arcsine_uniform":
        lam = np.random.uniform(0.0, 1.0)
        arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        uni = np.full_like(i, 1.0 / n)
        vals = (1.0 - lam) * arcs + lam * uni
    elif family == "dirichlet":
        # random concentration between 0.05 and 2.0, symmetric Dirichlet
        conc = np.random.uniform(0.05, 2.0)
        half = (n + 1) // 2
        gamma = np.random.gamma(shape=conc, scale=1.0, size=half)
        vals = np.concatenate([gamma,
                               gamma[-2::-1] if n % 2 == 0 else gamma[-1::-1]])
    elif family == "exp_gauss":
        half = (n + 1) // 2
        sigma = np.random.uniform(0.5, 2.0)
        v = np.random.normal(scale=sigma, size=half)
        vals_half = np.exp(v)
        vals = np.concatenate([vals_half,
                               vals_half[-2::-1] if n % 2 == 0 else vals_half[-1::-1]])
    else:
        # fallback uniform
        vals = np.full_like(i, 1.0 / n)

    # ensure positivity
    vals = np.maximum(vals, 0.0)
    vals = _symmetrise(vals)
    vals = _project_onto_simplex(vals, s=1.0)
    return vals

def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Main entry point for the competition harness.
    Returns a list of non‑negative heights (a step function) that
    minimises ``evaluate_sequence`` under a 1000 s time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Determine a sensible base length
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # Build a modest set of lengths we are willing to explore
    length_factors = (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    })
    # keep a few hand‑picked lengths
    lengths.extend([5000, 8000, 12000, 16000, 20000])
    lengths = sorted({l for l in lengths if l <= 20000})

    # ------------------------------------------------------------------
    # Global bookkeeping
    best_seq = None
    best_score = float("inf")
    pool: list[tuple[float, np.ndarray]] = []   # (score, sequence)

    # ------------------------------------------------------------------
    # Seed the pool with a few deterministic candidates
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        seq0 = np.maximum(seq0, 0.0)
        seq0 = _symmetrise(seq0)
        seq0 = _project_onto_simplex(seq0, s=1.0)
        score0 = _objective(seq0)
        print(f"[seed] previous solution score={score0:.6f}")
        best_seq, best_score = seq0.copy(), score0
        pool.append((score0, seq0))
    except Exception:
        pass

    # always keep a pure arcsine as a baseline
    arcs = _candidate_from_family(base_n, "arcsine")
    arcs_score = _objective(arcs)
    print(f"[seed] arcsine (n={base_n}) score={arcs_score:.6f}")
    if arcs_score < best_score:
        best_seq, best_score = arcs.copy(), arcs_score
    pool.append((arcs_score, arcs))

    # ------------------------------------------------------------------
    # Phase 1 – massive random sampling
    families = ("arcsine", "power_invert", "power_direct", "beta",
                "gauss", "mix_arcsine_uniform", "dirichlet", "exp_gauss")
    pool_size = 12
    phase1_deadline = deadline - 300.0          # reserve ~5 min for refinements
    while time.time() < phase1_deadline:
        n = int(np.random.choice(lengths))
        family = np.random.choice(families)
        cand = _candidate_from_family(n, family)

        # evaluate
        score = _objective(cand)

        # update global best
        if score < best_score:
            best_score, best_seq = score, cand.copy()
            print(f"[rand] new global best len={n} fam={family} score={best_score:.6f}")

        # maintain elite pool
        if len(pool) < pool_size:
            pool.append((score, cand))
        else:
            worst_idx = max(range(len(pool)), key=lambda i: pool[i][0])
            if score < pool[worst_idx][0]:
                pool[worst_idx] = (score, cand)

    # ------------------------------------------------------------------
    # Phase 2 – adaptive exponentiated‑gradient refinement
    pool.sort(key=lambda x: x[0])
    top_k = min(6, len(pool))
    for idx in range(top_k):
        if time.time() >= deadline - 30.0:
            break
        start_seq = pool[idx][1]
        n = start_seq.shape[0]
        # give each refinement roughly equal remaining time
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(45.0, remaining * 0.6)
        refined_seq, _ = _optimize_exponential_adaptive(start_seq, slice_deadline)
        refined_score = _objective(refined_seq)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[refine] improved to {best_score:.6f} (n={n})")
        # keep refined version back into the pool for possible further use
        pool.append((refined_score, refined_seq))
        pool.sort(key=lambda x: x[0])
        pool = pool[:pool_size]

    # ------------------------------------------------------------------
    # Phase 3 – final local stochastic hill‑climbing
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        refined_seq, _ = _local_refinement(best_seq, deadline)
        refined_score = _objective(refined_seq)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[final] after local search best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – uniform distribution if everything else failed
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
