"""
Propose a high‑quality non‑negative step‑function.

The search runs in three stages under the 1000 s budget:

1. **Exploratory random sampling** – a wide collection of parametric families
   (arcsine, power‑law, beta, Gaussian, mixture, exponential‑decay,
   linear‑decay, Dirichlet) is sampled for many lengths.  Each candidate is
   projected onto the simplex and evaluated with a fast FFT‑based exact
   objective 2·n·max(a∗a) (the denominator is 1 because the vector is
   normalised).  The best few candidates are kept.

2. **Adam‑accelerated exponentiated‑gradient refinement** – the kept candidates
   are further optimised on the simplex using an Adam‑style mirror‑descent
   scheme.  The gradient is the sub‑gradient of the max‑convolution; a
   soft‑max surrogate (temperature α) is used initially to obtain a smooth
   direction and later switched to the exact sub‑gradient.  Step‑sizes
   decay with √t.

3. **Local stochastic hill‑climbing** – the best refined vector is polished by
   repeatedly moving a random fraction of mass from one index to another
   (several fractions are tried).  Moves are kept only if they improve the
   objective.

The routine respects the time budget, prints progress information and
returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----- families of probability vectors ---------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


def _beta_distribution(n: int, a: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


def _exponential_decay_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i)
    vals /= vals.sum()
    return vals


def _linear_decrease_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (n - i).astype(np.float64)
    vals /= vals.sum()
    return vals


def _dirichlet_distribution(n: int, concentration: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, concentration))


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a."""
    conv = _fft_self_convolve(a)
    return float(conv.max())


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) (the sum is normalised to 1)."""
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv


def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g


def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of (1/α)·log‑sum‑exp(α·b) w.r.t. a."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑accelerated exponentiated‑gradient optimiser on the simplex.
    Returns the best vector seen and its objective.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 1.0                # start with smooth surrogate
    alpha_mul = 1.03
    alpha_max = 15000.0
    alpha_thr = 2000.0         # after this use exact sub‑gradient

    size_fft = 1 << ((3 * n - 2).bit_length())
    eta0 = 0.8                 # base step size
    t = 0

    while time.time() < deadline:
        t += 1
        # current objective (fast)
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Adam it={t} n={n} score={best_score:.6f}")

        # gradient
        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, b, alpha, size_fft)

        # Adam state update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step‑size schedule
        eta = eta0 / np.sqrt(t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-step)
        a = np.maximum(a, 0.0)

        # re‑normalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing: move a random fraction of mass from one
    coordinate to another if the move improves the exact objective.
    Several fractions are tried per move.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    # fractions of source mass to try
    eps_fractions = np.array([0.01, 0.03, 0.07, 0.12, 0.20], dtype=np.float64)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # pick a random fraction
        frac = eps_fractions[np.random.randint(len(eps_fractions))]
        eps = a[i] * frac
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        # enforce simplex (sum stays 1) and non‑negativity
        a_candidate = _project_onto_simplex(a_candidate, s=1.0)
        cur_score = _objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine runs for at most
    `budget_s` seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Length grid – use the length of a possible previous solution if
    # available, otherwise fall back to a sensible default.
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    length_factors = (0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    })
    # add a few hand‑picked lengths
    lengths.extend([5000, 8000, 12000, 16000, 20000])
    lengths = sorted({l for l in lengths if l <= 20000})

    # ------------------------------------------------------------------
    # bookkeeping
    best_seq = None
    best_score = float("inf")
    top_candidates = []        # list of (score, vector) tuples

    # ------------------------------------------------------------------
    # initialise from the previous solution if we have one
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if seq0.size:
            seq0 = np.maximum(seq0, 0.0)
            seq0 = _project_onto_simplex(seq0, s=1.0)
            score0 = _objective(seq0)
            print(f"Seeded from previous solution: score={score0:.6f}")
            best_seq = seq0.copy()
            best_score = score0
            top_candidates.append((score0, seq0))
    except Exception:
        pass

    # seed with a plain arcsine distribution (baseline)
    arcs_seq = _arcsine_distribution(base_n)
    arcs_score = _objective(arcs_seq)
    print(f"Arcsine baseline (n={base_n}) score={arcs_score:.6f}")
    if arcs_score < best_score:
        best_seq = arcs_seq.copy()
        best_score = arcs_score
    top_candidates.append((arcs_score, arcs_seq))

    # ------------------------------------------------------------------
    # ---------- Phase 1 : random parametric search --------------------
    families = (
        "arcsine", "power_invert", "power_direct", "beta",
        "gauss", "mix", "exp", "lin", "dirichlet"
    )
    search_deadline = deadline - 300.0   # keep ~5 min for refinements
    while time.time() < search_deadline:
        n = int(np.random.choice(lengths))
        fam = np.random.choice(families)

        if fam == "arcsine":
            seq = _arcsine_distribution(n)
        elif fam == "power_invert":
            p = np.random.uniform(0.2, 1.2)
            seq = _power_law_distribution(n, p, invert=True)
        elif fam == "power_direct":
            p = np.random.uniform(0.2, 1.2)
            seq = _power_law_distribution(n, p, invert=False)
        elif fam == "beta":
            a = np.random.uniform(0.2, 5.0)
            seq = _beta_distribution(n, a)
        elif fam == "gauss":
            sigma = np.random.uniform(n / 20.0, n / 2.0)
            seq = _gaussian_distribution(n, sigma)
        elif fam == "mix":
            lam = np.random.uniform(0.0, 1.0)
            seq = _mixture_arcsine_uniform(n, lam)
        elif fam == "exp":
            lam = np.random.uniform(0.0, 5.0) / n  # decay rate
            seq = _exponential_decay_distribution(n, lam)
        elif fam == "lin":
            seq = _linear_decrease_distribution(n)
        elif fam == "dirichlet":
            conc = np.random.uniform(0.1, 5.0)
            seq = _dirichlet_distribution(n, conc)
        else:
            seq = _arcsine_distribution(n)

        seq = _project_onto_simplex(seq, s=1.0)
        score = _objective(seq)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New global best (random) n={n} fam={fam} score={best_score:.6f}")

        # keep a small pool of promising candidates (size ≤ 12)
        if len(top_candidates) < 12:
            top_candidates.append((score, seq))
        else:
            # replace the worst entry if the new one is better
            worst_idx = max(range(len(top_candidates)),
                            key=lambda i: top_candidates[i][0])
            if score < top_candidates[worst_idx][0]:
                top_candidates[worst_idx] = (score, seq)

    # ------------------------------------------------------------------
    # ---------- Phase 2 : Adam‑accelerated refinement -----------------
    # keep the 5 best candidates for deep optimisation
    top_candidates.sort(key=lambda x: x[0])
    refine_list = [cand[1] for cand in top_candidates[:5]]

    for idx, cand in enumerate(refine_list):
        if time.time() >= deadline - 30.0:
            break
        n = cand.shape[0]
        remaining = deadline - time.time()
        # allocate at most 60 s per refinement run (or the remaining share)
        slice_deadline = time.time() + min(60.0, remaining * 0.5)
        refined_seq, refined_score = _optimize_adam(cand, n, slice_deadline)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"Refinement {idx+1} improved best to {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # ---------- Phase 3 : final local hill‑climbing ------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        refined_seq, refined_score = _local_refinement(best_seq, n, deadline)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"After local hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback (should never be needed)
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length = {best_seq.shape[0]})")
    return best_seq.tolist()
