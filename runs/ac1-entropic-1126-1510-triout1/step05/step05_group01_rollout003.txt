def _optimize(a_init, deadline, global_best_score):

   a = a_init.copy()

   n = len(a)

   best_a = a.copy()

   best_score = _score(a)

   # Adam parameters

   m = np.zeros(n)

   v = np.zeros(n)

   beta1 = 0.9

   beta2 = 0.999

   eps = 1e-8

   eta0 = 0.2  # base learning rate

   alpha = 0.5

   alpha_factor = 1.005

   max_alpha = 5000.0

   t = 0

   while time.time() < deadline:

       t += 1

       # compute conv

       conv = _fft_self_convolve(a)

       cur_score = 2 * n * conv.max()

       if cur_score < best_score:

           best_score = cur_score

           best_a = a.copy()

       # compute gradient

       if alpha < max_alpha:

           # softmax gradient

           w = np.exp(alpha * (conv - conv.max()))

           w_sum = w.sum()

           if w_sum == 0.0:

               w = np.zeros_like(w)

               w[np.argmax(conv)] = 1.0

           else:

               w /= w_sum

       else:

           # exact gradient: indicator of max

           max_val = conv.max()

           active = np.flatnonzero(conv >= max_val - 1e-12)

           w = np.zeros_like(conv)

           w[active] = 1.0 / active.size

       # compute gradient of max: grad = 2 * convolve(reversed a, w)[n-1:n-1+n]

       conv_rev = _fft_convolve(a[::-1], w)

       grad = 2.0 * conv_rev[n-1:n-1+n]

       # Adam updates

       m = beta1 * m + (1 - beta1) * grad

       v = beta2 * v + (1 - beta2) * (grad ** 2)

       m_hat = m / (1 - beta1 ** t)

       v_hat = v / (1 - beta2 ** t)

       step = eta0 / (np.sqrt(v_hat) + eps) * m_hat

       # descent

       a = a - step

       # project onto simplex

       a = _project_onto_simplex(a)

       # increase temperature

       alpha = min(max_alpha, alpha * alpha_factor)

       # early stopping

       if best_score <= 1.5:

           break

       # optional jitter

       if t % 2000 == 0:

           jitter = np.random.randn(n) * 1e-7

           a = a + jitter

           a = _project_onto_simplex(a)

   return best_a, best_score

def _fft_self_convolve(a):

    size = 1 << ((2*n-1) - 1).bit_length()

    conv = np.fft.irfft(A*A, n=size)

    return conv[:2*n-1]

"""
Projected‑Adam optimizer for the self‑convolution L∞‑minimisation problem.

The algorithm works as follows

1.  Choose a handful of promising lengths (≈ 4 k – 12 k) and generate a
    small library of analytic candidates (Beta / arcsine / Dirichlet /
    the previous best).  Each candidate is a probability vector (sum = 1).

2.  Keep the best few (by the cheap score 2·n·max(conv)) and improve them
    with a projected Adam scheme.  The loss is a smoothed max
    (log‑sum‑exp) of the self‑convolution; the temperature (α) is increased
    gradually so optimisation first sees a smooth landscape and then
    focuses on the true maximum.

    Gradient of the smoothed max is
        ∇L(a) = 2·conv_rev(a, w)   with  w_i ∝ exp(α·(conv_i‑max(conv))).

    After each Adam step the iterate is projected onto the simplex
    {x≥0, Σx=1}.  Occasional jitter helps escape plateaus.

3.  The best refined sequence of each length is evaluated with the
    official ``evaluate_sequence`` function; the globally best sequence
    is returned.  If the target 1.503 is reached early, the search stops.

All heavy work is done with real FFTs for speed, and the whole search
honours the supplied time budget (default 1000 s).  No file or network
access is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1.0 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1.0 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Score assuming Σa = 1 (i.e. 2·n·max(conv))."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Weights w_i ∝ exp(α·(conv_i‑max(conv))) used in the smoothed max."""
    cur_max = conv.max()
    # subtract cur_max for numerical stability
    w = np.exp(alpha * (conv - cur_max))
    s = w.sum()
    if s == 0.0 or not np.isfinite(s):
        # fallback to a single max index
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / s
    return w


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the (soft‑max) loss w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    conv_rev = _fft_convolve(a[::-1], w)                # length 3·n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]            # central n entries
    return grad


# ----------------------------------------------------------------------
def _optimize_sequence(a0: np.ndarray,
                       deadline: float,
                       target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of a single starter vector.

    Returns (best_vector, best_cheap_score) found before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _cheap_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # learning‑rate schedule
    eta0 = 0.2

    # temperature (α) schedule for the soft‑max surrogate
    alpha = 0.5
    alpha_factor = 1.005
    alpha_max = 5000.0

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.1f}] n={n} iter={iteration} "
                f"score={cur_score:.6f}"
            )
            if best_score <= target_score:
                break

        # choose surrogate gradient
        if alpha < alpha_max:
            w = _softmax_weights(conv, alpha)
            alpha = min(alpha_max, alpha * alpha_factor)
        else:
            # exact sub‑gradient (uniform weight over all nearly‑maximal indices)
            cur_max = conv.max()
            eps_act = 1e-12 * (abs(cur_max) + 1.0)
            act = np.flatnonzero(conv >= cur_max - eps_act)
            if act.size == 0:
                act = np.array([np.argmax(conv)], dtype=np.int64)
            w = np.zeros_like(conv)
            w[act] = 1.0 / act.size

        grad = _gradient_from_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                     # descent step
        a = _project_onto_simplex(a)     # keep feasibility

        # occasional jitter
        if iteration % 2500 == 0:
            jitter = np.random.randn(n) * 1e-7
            a += jitter
            a = _project_onto_simplex(a)

    return best_a, best_score
