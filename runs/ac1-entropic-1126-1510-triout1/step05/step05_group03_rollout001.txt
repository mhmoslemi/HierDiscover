"""
Propose a low‑score step‑function.

Algorithm Overview
------------------
1. **Initialisation** – Gather a few well‑known families (uniform,
   arcsine, symmetric Beta, random Dirichlet) and, if present,
   the previously discovered best sequence (`height_sequence_1`).
   All candidates are projected onto the simplex Σaᵢ=1.

2. **Population‑based search** – A modest population (≈30 vectors) of the
   chosen length is evolved for the bulk of the time budget.
   Mutations are simple mass‑transfer moves (±Δ) and small Gaussian
   perturbations.  Each offspring is projected back onto the simplex and
   evaluated by the cheap surrogate `fast_score = 2·n·max(a*a)`.
   The worst individual is replaced when a better offspring appears.
   The global best surrogate score is tracked.

3. **Continuous optimisation** – The best surrogate solution is refined
   with three stages:
   * Adam optimisation of a smooth‐max surrogate of the convolution.
   * A short exact‑max sub‑gradient descent.
   * Simulated‑annealing polishing.

4. **Final evaluation** – The refined candidate is handed to the
   black‑box `evaluate_sequence`.  The returned Python list satisfies all
   security checks of the evaluator.

The routine respects the 1000 s budget, uses only NumPy (no external
solvers) and works for any available `height_sequence_1`.  All helper
functions are defined at top level for clarity.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) distribution of length n, normalised to 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalised to 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space to avoid overflow/underflow when α,β<1
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # stable soft‑max
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns (best_a, best_peak) where best_peak = max((best_a)*(best_a)).
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # decreasing learning rate
        lr = lr0 / np.sqrt(t)

        # update & project
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep best
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # anneal smoothing parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ convolution objective.
    Uniform weight on the active set (indices attaining the current maximum).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Simple simulated‑annealing polishing.
    Returns (best_vector, best_peak).
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    # exponential cooling
    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # pick a source with non‑zero mass
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        # move a random fraction (up to 5 %) of the source mass
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        if delta_obj < 0.0 or np.random.rand() < math.exp(-delta_obj / max(temperature, 1e-12)):
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # every so often reset to the current best (helps later stage)
        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Compute the cheap surrogate score = 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine mixes a population‑based evolutionary phase with
    Adam, sub‑gradient, and simulated‑annealing refinements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # keep 1 s safety margin

    # ------------------------------------------------------------------
    # Length selection – reuse previous construction if present,
    # otherwise fall back to a round number around 4000.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        n_target = len(base_seq)
        print(f"[info] using previous best length n={n_target}")
    except Exception:
        n_target = 4000
        print(f"[info] no previous sequence – using default length n={n_target}")

    # ------------------------------------------------------------------
    # Build an initial population of diverse candidates.
    pop_size = 30
    population = []
    scores = []

    # deterministic starters
    starters = {
        "uniform": np.full(n_target, 1.0 / n_target, dtype=np.float64),
        "arcsine": _arcsine_distribution(n_target),
    }

    # symmetric Beta families (alpha = beta)
    for a in (0.35, 0.45, 0.55, 0.65, 0.75, 0.9, 1.2, 1.5):
        starters[f"beta_{a:.2f}"] = _beta_distribution(n_target, a, a)

    # random Dirichlet draws
    for i in range(8):
        rand = np.random.rand(n_target)
        rand /= rand.sum()
        starters[f"rand{i}"] = rand

    # include previous best if available
    if 'base_seq' in locals():
        starters["prev"] = base_seq

    # keep only up to pop_size distinct vectors
    for name, vec in starters.items():
        vec = _project_onto_simplex(vec, s=1.0)
        population.append(vec)
        scores.append(_fast_score(vec))
        if len(population) >= pop_size:
            break

    # If population still too small, fill with extra random draws
    while len(population) < pop_size:
        rand = np.random.rand(n_target)
        rand /= rand.sum()
        population.append(_project_onto_simplex(rand, s=1.0))
        scores.append(_fast_score(population[-1]))

    # keep only the best pop_size individuals
    order = np.argsort(scores)[:pop_size]
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]

    best_seq = population[0].copy()
    best_score = scores[0]
    print(f"[init] best initial fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Evolutionary phase – occupies the majority of the time budget.
    evol_deadline = time.time() + (deadline - time.time()) * 0.55
    n = n_target
    moves_per_iter = 2   # number of elementary mass moves per offspring

    while time.time() < evol_deadline:
        # tournament selection: pick a parent among the top half
        parent_idx = np.random.randint(pop_size // 2)
        parent = population[parent_idx]
        child = parent.copy()

        # ----- mutation -----
        for _ in range(moves_per_iter):
            src = np.random.randint(n)
            if child[src] <= 1e-14:
                continue
            dst = np.random.randint(n)
            if src == dst:
                continue
            # move up to 7 % of src mass
            max_delta = min(child[src], 0.07)
            delta = np.random.rand() * max_delta
            child[src] -= delta
            child[dst] += delta

        # occasional Gaussian perturbation
        if np.random.rand() < 0.15:
            child += np.random.normal(scale=5e-4, size=n)
            child = np.maximum(child, 0.0)

        # project back onto simplex (preserves sum)
        child = _project_onto_simplex(child, s=1.0)

        # evaluate surrogate
        child_score = _fast_score(child)

        # replace worst if improvement
        worst_idx = np.argmax(scores)
        if child_score < scores[worst_idx] - 1e-14:
            population[worst_idx] = child
            scores[worst_idx] = child_score

        # update global best
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_score - 1e-12:
            best_score = scores[best_idx]
            best_seq = population[best_idx].copy()
            print(f"[evo] new best fast_score={best_score:.6f} (time {time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Continuous refinement using the three classic phases.
    time_left = deadline - time.time()
    if time_left > 12.0:
        # 1) Adam on smooth‑max surrogate
        adam_deadline = time.time() + time_left * 0.45
        a_opt, _ = _adam_optimize(best_seq, adam_deadline)

        # 2) Sub‑gradient polishing
        time_left = deadline - time.time()
        if time_left > 6.0:
            sg_deadline = time.time() + time_left * 0.30
            a_opt = _refine_max(a_opt, sg_deadline)

        # 3) Simulated annealing for final polishing
        time_left = deadline - time.time()
        if time_left > 4.0:
            a_opt, final_peak = _simulated_annealing(a_opt, deadline)
        else:
            a_opt = best_seq  # fallback if not enough time

        # Update best sequence with refined version
        refined_score = _fast_score(a_opt)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = a_opt.copy()
            print(f"[refine] post‑refinement fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final black‑box evaluation.
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        # keep true value for reporting
        best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best approximate score {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
