"""
Propose a low‑score step function.

Algorithm outline
-----------------
1. **Problem size** – use the length of the reference sequence (if
   available) and enlarge it a little (capped at 9000).  The ℓ¹‑norm
   target is S = √(2·n).

2. **Smooth‑max L‑BFGS** – minimise a smooth approximation of the
   hard‑max objective  max(conv(a,a)).  A continuation on the
   smoothing parameter μ (geometric cooling) is performed for a number
   of restarts with diverse initial points (uniform, exponential,
   Dirichlet, random Gaussian, reference).

3. **Projected Adam‑style sub‑gradient descent** – after the L‑BFGS
   phase we minimise the exact max‑convolution on the simplex using an
   Adam‑like update with a back‑tracking line‑search for safety.

4. **Large‑scale random block perturbations** – cheap random
   “piece‑wise constant” candidates are generated, projected onto the
   simplex and evaluated.

5. **Rotation & monotone‑sorting heuristics** – the current best
   sequence is rotated at random offsets and both sorted variants are
   tried; any improvement is kept.

All helper functions are defined at module level (no closures).  The
routine respects the supplied time budget and prints progress messages.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0).
    """
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max; Σp = 1, p ≥ 0."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate and its gradient.
    f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Parameters
    ----------
    y : (n,) ndarray   – unconstrained parameters (log‑odds of simplex)
    mu : float          – smoothing temperature (>0)
    n  : int            – length of the step function
    S  : float          – ℓ¹‑norm target (√(2·n))

    Returns
    -------
    f_val : float               – objective value
    grad_y : (n,) ndarray       – gradient ∂f/∂y
    """
    # map to the simplex
    p = _softmax(y)                     # Σp = 1, p ≥ 0
    a = S * p                           # Σa = S

    # convolution a*a (length 2·n‑1)
    conv = np.convolve(a, a)

    # smooth‑max weighting
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)     # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – uniform weight among all maximal entries
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient wrt the heights a (see derivation in the description)
    conv_wr = np.convolve(w, a[::-1])                # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]             # (n,)

    # chain rule through the soft‑max
    inner = np.dot(g_a, p)                           # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """
    Callable used by scipy.optimize.minimize; returns (value, gradient).
    """
    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    """Maximum coefficient of the auto‑convolution a*a."""
    return np.convolve(a, a).max()


def _objective_score(a: np.ndarray, n: int) -> float:
    """
    Exact score returned by evaluate_sequence for a non‑negative a.
    Assumes all entries lie in [0,1000] and sum(a) > 0.01.
    """
    s = a.sum()
    if s < 0.01:
        return math.inf
    max_c = _conv_max(a)
    return 2.0 * n * max_c / (s * s)


def _maybe_update(best_a: np.ndarray, best_score: float,
                  cand: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Return possibly updated (best_a, best_score)."""
    sc = _objective_score(cand, n)
    if sc < best_score:
        print(f"[improve] new best score = {sc:.6f}")
        return cand.copy(), sc
    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a
    low evaluate_sequence score.  The routine respects the given time
    budget (seconds) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size ------------------------------------------------
    try:
        ref_len = len(height_sequence_1)          # type: ignore[name-defined]
        n = min(max(ref_len * 2, 4000), 9000)
        have_ref = True
        print(f"[setup] using reference length {ref_len}, n = {n}")
    except Exception:
        n = 6000
        have_ref = False
        print(f"[setup] no reference, using n = {n}")

    S = math.sqrt(2.0 * n)                       # target ℓ¹‑norm
    best_a = np.zeros(n, dtype=float)
    best_score = math.inf

    # ------------------------------------------------------------------
    # Generate diverse initial points (log‑odds y) -----------------------
    init_ys = []

    # uniform (y = 0)
    init_ys.append(np.zeros(n, dtype=np.float64))

    # decreasing exponential
    alpha = rng.uniform(0.0005, 0.003) * (n / 2000.0)
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    # increasing exponential (big on the right)
    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    # random Gaussian
    init_ys.append(rng.standard_normal(n) * 0.5)

    # sparse Dirichlet (α < 1)
    dir_alpha = rng.uniform(0.2, 0.6)
    p_dir = rng.dirichlet([dir_alpha] * n)
    eps = 1e-12
    init_ys.append(np.log(p_dir + eps))

    # reference (if available)
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, S)
        init_ys.append(np.log(a_ref + eps))

    n_restart = len(init_ys)
    print(f"[init] {n_restart} restart points created")

    # ------------------------------------------------------------------
    # Smooth‑max L‑BFGS phase --------------------------------------------
    mu_factor = 0.80          # slower cooling -> finer optimisation
    mu_min = 1e-9
    maxiter_per_mu = 500
    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}] starting")
        y = y0.copy()

        # initial temperature based on current max‑convolution
        a_cur = S * _softmax(y)
        mu = max(_conv_max(a_cur), 1.0)

        # continuation over μ
        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False}
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            best_a, best_score = _maybe_update(best_a, best_score, a, n)

            mu *= mu_factor

    # ------------------------------------------------------------------
    # Projected Adam‑style sub‑gradient descent (hard max) ---------------
    if time.time() < deadline and best_score < math.inf:
        print("[subgrad] entering Adam‑style descent")
        a = best_a.copy()
        # Adam parameters
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        lr = 0.4 * S / n               # base learning rate, similar magnitude to earlier step0
        max_local_iters = 8000
        it = 0
        while time.time() < deadline and it < max_local_iters:
            # compute sub‑gradient of max‑convolution
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt
            grad = 2.0 * np.convolve(w, a[::-1])[n - 1:n - 1 + n]

            # Adam update (on the simplex, then project)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            direction = m_hat / (np.sqrt(v_hat) + eps)

            # try a back‑tracking line search with the base learning rate
            step = lr
            improved = False
            for _ in range(12):
                a_new = a - step * direction
                a_new = _project_to_simplex(a_new, S)
                sc = _objective_score(a_new, n)
                if sc < best_score - 1e-12:
                    best_a, best_score = a_new.copy(), sc
                    a = a_new
                    improved = True
                    print(f"[adam] iter {it:4d} step {step:.6f} => {best_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # occasional random jitter to escape plateaus
                i, j = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a[i] * 0.03
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, S)
            it += 1

    # ------------------------------------------------------------------
    # Random block perturbations (cheap large‑scale search) -------------
    remaining = deadline - time.time()
    if remaining > 2.0 and best_score < math.inf:
        max_trials = int(min(8000, remaining * 130))
        print(f"[block] {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(3, 17)                 # 3 … 16 blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            best_a, best_score = _maybe_update(best_a, best_score, cand, n)

    # ------------------------------------------------------------------
    # Rotation heuristic – try several cyclic shifts --------------------
    if time.time() < deadline and best_score < math.inf:
        print("[rotate] testing random rotations")
        for _ in range(30):
            shift = rng.integers(n)
            cand = np.roll(best_a, shift)
            # re‑project is not necessary because rotation preserves ℓ¹‑norm
            best_a, best_score = _maybe_update(best_a, best_score, cand, n)

    # ------------------------------------------------------------------
    # Monotone‑sorting heuristic -----------------------------------------
    if time.time() < deadline and best_score < math.inf:
        # descending sort
        cand_desc = np.sort(best_a)[::-1]
        cand_desc = _project_to_simplex(cand_desc, S)
        best_a, best_score = _maybe_update(best_a, best_score, cand_desc, n)

        # ascending sort
        cand_asc = np.sort(best_a)
        cand_asc = _project_to_simplex(cand_asc, S)
        best_a, best_score = _maybe_update(best_a, best_score, cand_asc, n)

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}  (length {n})")
    return best_a.tolist()
