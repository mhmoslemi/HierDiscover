def _arcsine_distribution(n):
    # returns vector of length n, sum=1.
    # Using integral over intervals.
    i = np.arange(n)
    # left boundaries: (i)/n, right: (i+1)/n
    left = i / n
    right = (i+1) / n
    # CDF = (2/π) arcsin(sqrt(x))
    left_cdf = 2.0/np.pi * np.arcsin(np.sqrt(left))
    right_cdf = 2.0/np.pi * np.arcsin(np.sqrt(right))
    a = right_cdf - left_cdf
    # due to numeric at boundaries left=0 sqrt(0)=0 arcsin(0)=0, right=n/n=1, arcsin(1)=π/2 => right_cdf=1.
    # extremely small left intervals for i=0 ensure positive values.
    return a

"""
Propose a low‑score step‑function.

Algorithm
---------
1. Choose a few problem sizes (lengths of the step vector).  
2. Build a small library of analytic candidates:
   * Integrated arcsine distribution (the continuous optimal shape).
   * Integrated symmetric Beta(α,α) for several α∈(0.2,0.8).
   * Uniform, a random Dirichlet draw and the previous best construction
     (if available).
3. For each candidate run a sequence of limited‑memory BFGS optimisations.
   * The variables are the log‑weights `x`; a = softmax(x) guarantees
     positivity and Σa = 1.
   * The objective is a smooth surrogate of  max(a∗a) :
         f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)  .
     Its gradient can be written in closed form using a single
     convolution, see `_obj_and_grad`.
   * α is increased geometrically (0.5 → … → 1024) so the optimisation
     starts smooth and finishes close to the true max.
4. After the BFGS schedule a few exponentiated‑gradient steps with the
   exact sub‑gradient (active‑set) are performed – this removes the
   remaining smoothing bias.
5. The best sequence seen (according to the cheap FFT score) is finally
   evaluated with the official `evaluate_sequence`.  The best overall
   vector is returned.

The whole search respects the supplied time budget (default 1000 s).
"""

import time
import numpy as np
import functools

# ----------------------------------------------------------------------
# Basic geometry helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: a_i = F((i+1)/n) - F(i/n) with
    F(x) = 2/π·arcsin(√x).  The vector sums to 1."""
    # CDF values at the grid points 0,1/n,...,1
    x = np.arange(n + 1, dtype=np.float64) / n
    cdf = 2.0 / np.pi * np.arcsin(np.sqrt(x))
    a = cdf[1:] - cdf[:-1]
    # numerical normalisation (tiny rounding errors)
    return a / a.sum()


def _beta_distribution_integrated(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) obtained by integrating the continuous density
    over each unit interval.
    """
    from scipy.special import betainc  # regularised incomplete beta
    x = np.arange(n + 1, dtype=np.float64) / n
    cdf = betainc(alpha, beta, x)      # regularised (∈[0,1])
    a = cdf[1:] - cdf[:-1]
    # avoid pathological zero‑sum cases
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _obj_and_grad(x: np.ndarray, alpha: float, n: int):
    """
    Returns (objective, gradient) for the smooth surrogate
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    where a = softmax(x).
    Gradient is w.r.t. x (the log‑variables).
    """
    a = _softmax(x)

    # self‑convolution
    conv = _fft_self_convolve(a)                 # length 2n‑1
    cur_max = conv.max()
    # soft‑max weights over the convolution entries
    shifted = conv - cur_max                     # ≤ 0
    exp_term = np.exp(alpha * shifted)
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp                       # length 2n‑1

    # gradient w.r.t. a :  ∂f/∂a = 2· (a_rev ∗ w)  (central n entries)
    conv_rev = _fft_convolve(a[::-1], w)        # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # convert to gradient w.r.t. x using the Jacobian of softmax
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)

    # objective value (smooth approximation of max)
    obj = cur_max + np.log(sum_exp) / alpha
    return obj, grad_x


# ----------------------------------------------------------------------
# Exact sub‑gradient of the true max (used for a short fine‑tuning phase)
# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Uniform sub‑gradient over all approximately active indices."""
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms where k = 2·i (self‑product)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad


def _exponential_update(a: np.ndarray, grad: np.ndarray, eta: float,
                       momentum: np.ndarray, beta: float) -> tuple[np.ndarray, np.ndarray]:
    """Multiplicative (exponentiated) update with simple momentum."""
    momentum = beta * momentum + (1.0 - beta) * grad
    shift = np.max(momentum)
    factor = np.exp(-eta * (momentum - shift))
    a = a * factor
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s
    return a, momentum


# ----------------------------------------------------------------------
# Cheap scoring (exact for the evaluation function because the score is scale‑invariant)
# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """2·n·max(a∗a) – the same quantity returned by ``evaluate_sequence``."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()
