"""
Propose a low‑score step‑function.

The algorithm works in two stages.

1. **Global search** – we explore a rich family of Beta‑distributions
   (including mixtures) on a moderately large length (a power of two).
   A cheap golden‑section search refines the symmetric‐Beta shape.
   The best candidate among these is kept.

2. **Local refinement** – starting from the best candidate we minimise a
   smooth approximation of the objective
       maxₖ (a*a)ₖ
   using the log‑sum‑exp (soft‑max) trick.
   Its gradient is computed by FFT‑based convolutions, then we perform
   projected gradient descent with a simple momentum scheme.
   The smoothing parameter γ is increased (continuation) so that the
   optimisation slowly approaches the true ℓ∞ objective.
   Random jolts and occasional restarts help avoid stagnation.

All iterates are projected onto the simplex {a_i≥0, Σa_i=1} to respect
the problem constraints.  The routine respects the 1000 s time budget
and returns a plain list of floats suitable for `evaluate_sequence`.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    conv_rev = _fft_convolve(rev_a, w_norm)
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, b_max

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height vector with a low `evaluate_sequence` score.
    The routine performs a global Beta‑family search followed by a
    continuation‑based projected gradient descent.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------
    # Choose a power‑of‑two length – FFTs are cheapest on such sizes.
    n = 8192

    # ------------------------------------------------------------
    def raw_score(vec: np.ndarray) -> float:
        """2·n·max(a*a) – the exact numerator before the denominator (which is 1)."""
        return 2.0 * n * _fft_self_convolve_full(vec).max()

    best_seq = None
    best_raw = float("inf")

    # ------------------------------------------------------------
    # 1) start from a previously discovered construction, if any.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        cand = _interpolate_to_length(base_seq, n)
        cand = _project_onto_simplex(cand)
        sc = raw_score(cand)
        best_seq, best_raw = cand, sc
        print(f"[init] from height_sequence_1 → raw={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------
    # 2) trivial baselines
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    sc = raw_score(uniform)
    if sc < best_raw:
        best_seq, best_raw = uniform.copy(), sc
        print(f"[init] uniform → raw={sc:.6f}")

    arcsine = _beta_distribution(n, 0.5, 0.5)
    sc = raw_score(arcsine)
    if sc < best_raw:
        best_seq, best_raw = arcsine.copy(), sc
        print(f"[init] arcsine → raw={sc:.6f}")

    # ------------------------------------------------------------
    # 3) fine tune a symmetric Beta(α,α) via golden‑section search.
    def beta_sym_score(alpha: float) -> float:
        vec = _beta_distribution(n, alpha, alpha)
        return raw_score(vec)

    lo, hi = 0.03, 1.5                     # plausible α‑range
    phi = (1.0 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi

    a1 = hi - (hi - lo) * invphi
    a2 = lo + (hi - lo) * invphi
    f1 = beta_sym_score(a1)
    f2 = beta_sym_score(a2)

    while time.time() < deadline and (hi - lo) > 1e-5:
        if f1 < f2:
            hi, a2, f2 = a2, a1, f1
            a1 = hi - (hi - lo) * invphi
            f1 = beta_sym_score(a1)
        else:
            lo, a1, f1 = a1, a2, f2
            a2 = lo + (hi - lo) * invphi
            f2 = beta_sym_score(a2)

        if f1 < best_raw:
            best_raw = f1
            best_seq = _beta_distribution(n, a1, a1)
            print(f"[search] symBeta α={a1:.5f} → raw={best_raw:.6f}")
        if f2 < best_raw:
            best_raw = f2
            best_seq = _beta_distribution(n, a2, a2)
            print(f"[search] symBeta α={a2:.5f} → raw={best_raw:.6f}")

    # ------------------------------------------------------------
    # 4) random mixtures of two independent Betas (time‑bounded).
    while time.time() < deadline:
        a1, b1 = np.random.rand() * 4.5 + 0.05, np.random.rand() * 4.5 + 0.05
        a2, b2 = np.random.rand() * 4.5 + 0.05, np.random.rand() * 4.5 + 0.05
        lam = np.random.rand()
        p1 = _beta_distribution(n, a1, b1)
        p2 = _beta_distribution(n, a2, b2)
        cand = lam * p1 + (1.0 - lam) * p2
        cand = _project_onto_simplex(cand)
        sc = raw_score(cand)
        if sc < best_raw:
            best_raw, best_seq = sc, cand.copy()
            print(f"[mix] ({a1:.2f},{b1:.2f})+({a2:.2f},{b2:.2f}) λ={lam:.2f} → raw={best_raw:.6f}")

    # ------------------------------------------------------------
    print(f"[summary] best raw after global search: {best_raw:.6f}")

    # ------------------------------------------------------------
    # 5) Continuation‑based projected gradient descent on the smooth max.
    a = best_seq.copy()
    best_a = a.copy()
    best_val = best_raw
    it = 0

    gamma = 1.0          # start with a fairly smooth approximation
    gamma_max = 300.0
    gamma_factor = 1.4
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    step0 = 0.5
    last_improve = 0

    while time.time() < deadline:
        it += 1
        grad, _ = _softmax_gradient(a, gamma)

        # simple heavy‑ball momentum and a diminishing step size
        momentum = beta_mom * momentum - (step0 / math.sqrt(it)) * grad
        a_new = a + momentum
        a_new = _project_onto_simplex(a_new)

        # evaluate exact objective after projection
        cur_max = _fft_self_convolve_full(a_new).max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a_new.copy()
            last_improve = it
            print(f"[{time.time():.1f}] it={it} γ={gamma:.1f} best={best_val:.6f}")

        a = a_new

        # increase smoothing parameter gradually
        if it % 3000 == 0 and gamma < gamma_max:
            gamma = min(gamma * gamma_factor, gamma_max)

        # occasional random jitter to escape shallow plateaus
        if it % 5000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a)

        # restart if nothing improves for a long stretch
        if it - last_improve > 20000:
            a = best_a.copy()
            momentum = np.zeros_like(a)
            last_improve = it
            print(f"[{time.time():.1f}] restart after stagnation at it={it}")

    # ------------------------------------------------------------
    final_seq = best_a
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq.tolist()
