def run_optimisation(a, n, deadline):
    best_seq = a.copy()
    best_score = _max_convolution_score(a)   # remember it's 2*n*max(b)
    # Smoothing schedule
    alphas = [(1.0, 2000), (5.0, 2000), (20.0, 2000), (100.0, 2000), (500.0, 2000)]
    # Adam parameters
    beta1=0.9; beta2=0.999; eps=1e-8
    lr = 0.1
    m = np.zeros_like(a); v = np.zeros_like(a); t=0
    for alpha, steps in alphas:
        for s in range(steps):
            if time.time()>deadline: break
            t+=1
            # compute b, w, mu, grad
            b = _fft_self_convolve(a)
            # stable softmax
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            sum_exp = exp_terms.sum()
            w = exp_terms / sum_exp
            # gradient
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n-1:n-1+n]   # shape n
            # Adam update
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad*grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)
            # compute current exact score every few iterations
            if t % 100 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    # after smoothing, run exact subgradient refinement (like earlier code)
    a = best_seq.copy()
    best_score = _max_convolution_score(a)
    # subgrad parameters:
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    while time.time()<deadline:

        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)

def hill_climb(a, deadline):
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score
    # parameters
    max_transfer = 0.1
    it = 0
    while time.time()<deadline:
        it+=1
        # gradually shrink transfer
        transfer = max_transfer * (0.5 ** (it/5000.0))
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j>=i: j+=1
        delta = np.random.rand()*min(transfer, a[i])
        if delta<=0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f'[hill] iter={it} improved best {best_score}')
        if it%5000==0 and np.random.rand()<0.2:
            perturb = np.random.randn(n)*1e-4
            a = best_seq + perturb
            a = _project_onto_simplex(a,1.0)
            cur_score = _max_convolution_score(a)
    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Initial families** – generate a few analytic shapes (uniform,
   arcsine, symmetric beta, triangular, exponential) for several
   lengths.  Keep the best one as a seed.

2. **Smoothed gradient descent** – minimise a smooth approximation of
   ``max(a*a)``:
        μ_α(a) = (1/α)·log Σ_i exp(α·(a*a)_i)

   The gradient of μ_α is a weighted convolution of ``a`` with the
   soft‑max weights.  We run an Adam optimiser on the simplex
   (Euclidean projection after each step) while gradually increasing
   the smoothing parameter α (1 → 5 → 20 → 100 → 500).  This helps
   escape local minima that trap a pure sub‑gradient method.

3. **Exact max sub‑gradient refinement** – after the continuation phase
   we switch to the exact sub‑gradient of the true max (active set)
   with momentum and a diminishing stepsize, exactly as in the
   previous state‑of‑the‑art code.

4. **Local random‑transfer hill climbing** – finally a cheap
   combinatorial hill‑climber moves a small amount of mass between two
   random positions, accepting only improving moves.  Periodic tiny
   random jitteres aid escaping shallow plateaux.

The routine runs for at most ``budget_s`` seconds (default 1000) and
keeps the best vector ever seen.  It tries several lengths (≈ 800 … 8000)
and several random starts to avoid being stuck in a single basin.

All helpers are defined at top level; no closures or file I/O are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a handful of plausible starts for a given length."""
    cand = []
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.01))
    # a few symmetric beta shapes
    for p in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_beta_sym(n, p))
    # a random sparse start (few non‑zeros)
    rnd = np.zeros(n, dtype=np.float64)
    k = max(2, n // 200)                # about 0.5 % non‑zeros
    idx = np.random.choice(n, k, replace=False)
    rnd[idx] = np.random.rand(k)
    rnd /= rnd.sum()
    cand.append(rnd)
    return cand

def _smooth_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max objective.
    Returns (best_vector, best_score).
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing schedule (alpha, steps)
    schedule = [(1.0, 1200), (5.0, 1200), (20.0, 1200),
                (100.0, 1200), (500.0, 1200)]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            # self‑convolution (full)
            b = _fft_self_convolve(a)

            # stable softmax over b
            maxb = b.max()
            exp_terms = np.exp(alpha * (b - maxb))
            sum_exp = exp_terms.sum()
            w = exp_terms / sum_exp                     # weights sum to 1

            # gradient = 2 * conv_rev(a[::-1], w) (first n entries)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update on the simplex (Euclidean projection afterwards)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasionally evaluate the exact objective
            if t % 100 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _exact_subgradient_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact max‑subgradient descent with momentum, as in the reference
    implementation. Returns the best seen vector and its score.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        w = np.zeros_like(b)
        w[active] = 1.0

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum / diminishing step
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={iteration} new best {best_score:.6f}")

        if best_score <= 1.503:
            break
        if time.time() + 0.5 > deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random–transfer hill climbing on the simplex.
    Returns the best vector found and its score.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_score = best_score

    max_transfer = 0.07          # at most 7 % of total mass per trial
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # Shrink transfer exponentially
        transfer = max_transfer * (0.5 ** (iteration / 4000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] hill iter={iteration} improved best {best_score:.6f}")

        # occasional jitter to escape plateaux
        if iteration % 5000 == 0 and np.random.rand() < 0.2:
            pert = np.random.randn(n) * 1e-4
            a = best_seq + pert
            a = _project_onto_simplex(a, 1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.
    Returns the best sequence found (as a Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Determine a set of lengths to explore (500 … 8000, preferring
    # multiples of a few hundred).  The exact values are not critical.
    lengths = [800, 1200, 2000, 3000, 5000, 8000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Coarse sampling from analytic families.
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates(n):
            if time.time() >= deadline:
                break
            # ensure normalisation
            cand = cand / cand.sum()
            score = _max_convolution_score(cand)
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] n={n} init score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. If a previous construction exists (height_sequence_1), consider it.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # Fallback (should never be needed)
    if overall_best_seq is None:
        n = max(800, lengths[0])
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _max_convolution_score(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed Adam descent on the best length.
    n = overall_best_seq.shape[0]
    a = overall_best_seq.copy()
    a, score = _smooth_descent(a, deadline)

    # ------------------------------------------------------------------
    # 4. Exact max sub‑gradient refinement (as in the previous SOTA).
    a, score = _exact_subgradient_refine(a, deadline)

    # ------------------------------------------------------------------
    # 5. Local random‑transfer hill climbing.
    a, score = _local_hill_climb(a, deadline)

    # ------------------------------------------------------------------
    print(f"Final best score: {score:.6f} (length={len(a)})")
    return a.tolist()
