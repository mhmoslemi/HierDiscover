"""
Propose a near‑optimal non‑negative step‑function (probability vector).

The search consists of three stages:

1. **Coarse parametric scan** – a set of symmetric families
   (arcsine, Beta, power‑law, Gaussian, mixtures, plateau‑exponential)
   are instantiated for several lengths.  The objective
   `2·n·max(a*a)` (sum(a)=1) is evaluated using an FFT based max‑convolution
   which is extremely fast (O(n log n)).  The best few candidates are kept.

2. **Exponential‑gradient refinement** – each of the top candidates is
   refined with an exponentiated‑gradient (mirror‑descent) algorithm.
   The gradient of a smooth soft‑max surrogate of the max‑convolution is
   computed via a single FFT, and the step‑size decays as 1/√t.
   A modest temperature schedule (soft‑max → hard max) keeps the updates
   stable while still focusing on the worst convolution coordinate.

3. **Local random‑move hill climbing** – a cheap stochastic move that
   transfers a small fraction of mass between two coordinates is applied
   until the time budget expires.

All helper functions are defined at the module level, the code works on a
single thread and respects the 1000 s budget.  The returned list contains
non‑negative heights; scaling does not affect the score because the
objective is homogeneous.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a simplex‑normalised vector."""
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
# ---- families of symmetric probability vectors -------------------------

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    v = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    v /= v.sum()
    return v


def _beta_distribution(n: int, a: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    v = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    v_sum = v.sum()
    if v_sum == 0:
        v = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        v = v / v_sum
    return v


def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        v = prod ** (-p)
    else:
        v = prod ** p
    s = v.sum()
    if s == 0:
        v = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        v = v / s
    return v


def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    v = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    v /= v.sum()
    return v


def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    v = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(v, s=1.0)


def _plateau_exp_distribution(n: int, plateau_frac: float,
                              lam: float) -> np.ndarray:
    """Symmetric distribution: flat plateau in the centre, exponential decay to the edges."""
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    d = np.abs(idx - centre)
    max_d = centre
    # half‑width of the plateau (in terms of index distance)
    plateau_width = max_d * plateau_frac
    v = np.where(d <= plateau_width,
                 1.0,
                 np.exp(-lam * (d - plateau_width)))
    return _project_onto_simplex(v, s=1.0)


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the (1/α)·log‑sum‑exp(α·b) surrogate w.r.t. a.
    Implements g = 2 * conv_rev(w, a_rev) where w are normalized soft‑max weights.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    return 2.0 * conv[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _refine_exponential(a0: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient refinement.
    Uses a smooth soft‑max surrogate throughout the run.
    Returns the best sequence encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)

    # hyper‑parameters
    eta0 = 1.2               # initial step‑size
    alpha = 5.0              # start temperature (soft)
    alpha_mul = 1.012
    alpha_max = 2500.0
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient of the smooth surrogate
        grad = _gradient_softmax(a, b, alpha, size_fft)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # increase temperature (focus more on the true max)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # tiny jitter occasionally to escape flat plateaus
        if it % 1500 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 1e-6
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random‑move hill‑climbing on the best sequence.
    Moves a small fraction of mass from one index to another if it improves.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.07   # move up to 7 % of source mass
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur = _objective(cand)
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score
