def refine_via_lp(a, steps=20, t=0.2):
    for iter in range(steps):
        n = len(a)
        b = np.convolve(a,a)
        M = b.max()
        # Solve LP: maximize sum(g) subject to A*g <= M
        # Build A matrix of shape (2n-1, n) where A[i,j] = a[i-j] if 0<=i-j<n else 0.
        # Use scipy.optimize.linprog.
        c = np.full(n, -1.0) # maximize sum -> minimize -sum
        # bounds for g: (0, None)
        bounds = [(0, None)] * n
        # constraints: A @ g <= M
        # Provide as linprog: A_ub, b_ub.
        # Use method='highs' for speed
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success: break
        g = res.x
        # Scale to sum=1
        if g.sum() == 0:
            break
        g = g / g.sum()
        # convex combine
        a = (1 - t) * a + t * g
        # optionally renorm to sum=1 exactly.
        a = a / a.sum()
    return a
```

        if time.time() > deadline: break
        n = len(a)
        b = np.convolve(a,a)
        M = b.max()
        # Build convolution matrix A for a.
        A = build_conv_matrix(a)  # shape (2n-1,n)
        # Solve LP: maximize sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()
        # Choose step size t that decays

def _conv_matrix(a):
    n = len(a)
    data = []
    offsets = []
    for d in range(n):
        data.append(np.full(n - d, a[d]))
        offsets.append(d)
    A = diags(data, offsets, shape=(2*n-1, n), format='csr')
    return A

def _lp_refine(a0, deadline, max_iter=10):
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    it = 0
    t = 0.5
    while time.time() < deadline and it < max_iter:
        it += 1
        n = len(a)
        b = np.convolve(a, a)
        M = b.max()
        # Build convolution matrix
        A = _conv_matrix(a)   # shape (2n-1, n)
        # Solve LP maximize sum(g)
        c = -np.ones(n)
        ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print(f"LP failed at iter {it}")
            break
        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()
        # Update a using convex combination with step size decreasing
        a = (1 - t) * a + t * g
        a = a / a.sum()
        # compute new score
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} score={best_score:.6f}")
        # decrease step size
        t = max(0.01, t * 0.7)
    return best_seq, best_score
```

"""
Propose a non‑negative step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in three stages:

1. **Coarse parametric sweep** – a short grid over a few
   parametric families (arcsine + uniform, power‑law, beta,
   Gaussian).  Each candidate is projected onto the simplex and
   evaluated with a fast FFT‑based exact objective.

2. **LP‑based refinement** – for the most promising candidates (and
   only for moderate lengths, ≤ 4000, to keep the LP tractable) we
   solve the linear programme

       maximise   Σ g_j
       subject to a * g ≤ M  (elementwise),
                  g ≥ 0 ,

   where * denotes convolution and M = max(a*a).  The optimal g
   (scaled to Σ g = 1) is mixed with the current a using a
   decreasing step‑size.  This is exactly the descent step described
   in the reference solution, but here we solve the LP with
   *SciPy* (high‑speed “highs” solver) instead of a black‑box
   optimiser.

3. **Local random‑move hill‑climbing** – a cheap stochastic
   perturbation that repeatedly moves a tiny amount of probability
   mass from one coordinate to another if it improves the objective.

The whole search respects the 1000 s budget and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    vals = prod ** (-p) if invert else prod ** p
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _beta_distribution(n: int, a: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a)  (the denominator is 1 because Σa = 1).
    The vector is first projected onto the simplex.
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray):
    """
    Build the (2n‑1) × n matrix A with A[i,j] = a[i‑j] for 0 ≤ i‑j < n,
    otherwise 0.  Each row corresponds to one convolution constraint:
        (a * g)_i = Σ_j A[i,j]·g_j .
    """
    n = a.shape[0]
    data = [np.full(n - d, a[d], dtype=np.float64) for d in range(n)]
    offsets = list(range(n))                 # super‑diagonal offsets
    return diags(data, offsets, shape=(2 * n - 1, n), format='csr')


# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray, deadline: float, max_iter: int = 10) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the LP‑based refinement described above.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    # initial step‑size for convex mixing
    mix = 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        n = a.shape[0]
        b = np.convolve(a, a)
        M = b.max()

        # Build the Toeplitz matrix for the constraints a * g ≤ M
        A = _conv_matrix(a)

        # Linear programme: maximise Σ g_j  (i.e. minimise -Σ g_j)
        c = -np.ones(n)                          # coefficients for maximise
        ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs', options={"presolve": True})
        if not res.success:
            # LP failed – fall back to a simple random perturbation
            print(f"[{time.time():.2f}] LP failed (iter {it}), aborting refinement.")
            break

        g = res.x
        if g.sum() == 0.0:
            break

        # Normalise to stay on the simplex
        g = g / g.sum()

        # Convex mixture with decreasing weight
        a = (1.0 - mix) * a + mix * g
        a = a / a.sum()          # protect against rounding drift

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine it={it} score={best_score:.6f}")

        # decay the mixing factor (small steps as we get close)
        mix = max(0.02, mix * 0.7)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing that moves a tiny mass between two indices."""
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.1   # move at most 10 % of source mass
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur_score = _objective(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs up to `budget_s` seconds and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # 2 s safety margin

    # ------------------------------------------------------------------
    # Lengths to explore.  We keep a moderate range; very large lengths are
    # costly for the LP stage, so we cap LP‑refinement to ≤ 4000.
    base_n = 8000
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        pass
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, base_n]
    lengths = [l for l in lengths if l >= 500]

    # ------------------------------------------------------------------
    # Global bookkeeping
    best_seq = None
    best_score = float("inf")
    candidates = []   # (score, seq)

    # seed with any previously‑found solution
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if seq0.size:
            seq0 = np.maximum(seq0, 0.0)
            seq0 = _project_onto_simplex(seq0, s=1.0)
            sc0 = _objective(seq0)
            best_seq, best_score = seq0, sc0
            candidates.append((sc0, seq0))
            print(f"Seeded with previous solution, score={sc0:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – systematic parametric grid search
    families = ("arc_mix", "pow_inv", "pow_dir", "beta", "gauss")
    for n in lengths:
        # arcsine + uniform mixture
        for lam in np.linspace(0.0, 1.0, 11):
            seq = _mixture_arcsine_uniform(n, lam)
            score = _objective(seq)
            candidates.append((score, seq.copy()))
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"New global best (arc_mix) n={n} lam={lam:.2f} score={score:.6f}")

        # power‑law (inverse)
        for p in np.linspace(0.2, 2.0, 11):
            seq = _power_law_distribution(n, p, invert=True)
            score = _objective(seq)
            candidates.append((score, seq.copy()))
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"New global best (pow_inv) n={n} p={p:.2f} score={score:.6f}")

        # power‑law (direct)
        for p in np.linspace(0.2, 2.0, 11):
            seq = _power_law_distribution(n, p, invert=False)
            score = _objective(seq)
            candidates.append((score, seq.copy()))
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"New global best (pow_dir) n={n} p={p:.2f} score={score:.6f}")

        # beta distribution
        for a in np.linspace(0.2, 5.0, 11):
            seq = _beta_distribution(n, a)
            score = _objective(seq)
            candidates.append((score, seq.copy()))
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"New global best (beta) n={n} a={a:.2f} score={score:.6f}")

        # Gaussian family
        for sigma in np.linspace(n / 20.0, n / 2.0, 11):
            seq = _gaussian_distribution(n, sigma)
            score = _objective(seq)
            candidates.append((score, seq.copy()))
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"New global best (gauss) n={n} sigma={sigma:.1f} score={score:.6f}")

        # early exit if time is low
        if time.time() > deadline - 30:
            break

    # ------------------------------------------------------------------
    # Keep a small pool of the best candidates (sorted by score)
    candidates.sort(key=lambda x: x[0])
    top_candidates = [c[1] for c in candidates[:6]]

    # ------------------------------------------------------------------
    # Phase 2 – LP based refinement (restricted to moderate lengths)
    for cand in top_candidates:
        if time.time() >= deadline - 30:
            break
        n = cand.shape[0]
        if n > 4000:                     # larger sizes are too heavy for LP
            continue
        # Allow at most 60 s per refinement run (or the remaining budget)
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(60.0, remaining * 0.6)
        refined, refined_score = _lp_refine(cand, slice_deadline, max_iter=12)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined.copy()
            print(f"LP refinement improved best to {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Phase 3 – cheap random‑move hill climbing on the current best
    if best_seq is not None and time.time() < deadline - 1.0:
        refined, refined_score = _local_refinement(best_seq,
                                                   best_seq.shape[0],
                                                   deadline)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined.copy()
            print(f"After local hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length = {best_seq.shape[0]})")
    # Convert to plain Python list for the competition harness
    return best_seq.tolist()
