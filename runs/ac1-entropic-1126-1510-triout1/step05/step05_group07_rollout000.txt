    conv = conv_self_convolve(best); max_val = conv.max(); best_score = 2*n*max_val
    while time.time() < deadline:
        conv = conv_self_convolve(best)
        max_val = conv.max()
        # find positions achieving max.
        active = np.where(np.abs(conv - max_val) < 1e-12)[0]  # list of k
        # pick random active position.
        p = np.random.choice(active)
        # compute contributions
        i_start = max(0, p - (n-1))
        i_end = min(p, n-1)
        i_range = np.arange(i_start, i_end+1)
        contributions = best[i_range] * best[p - i_range]
        # pick i that maximizes contribution.
        idx = np.argmax(contributions)
        i = i_range[idx]
        j = p - i
        # choose which of i,j to reduce: the larger value.
        if best[i] >= best[j]:
            src = i
            other = j
        else:
            src = j
            other = i
        # pick neighbor dest index randomly not equal src
        dest = np.random.choice(n-1)
        if dest >= src:
            dest += 1
        # amount to move
        delta = best[src] * 0.01 # 1% of src mass
        if delta <= 1e-12:

        cand = best.copy()
        cand[src] -= delta

        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # ...
    return best, best_score

Algorithm
---------

   Keep the best one.

   * After each projection we also enforce symmetry a = (a + a[::-1]) / 2,

   * Occasionally apply a “max‑focused” move that reduces the contribution

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         conv: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·conv_k) w.r.t. a.
    For huge α, the soft‑max becomes an (almost) exact sub‑gradient of max.
    """
    n = a.shape[0]

    # ---- soft‑max weights ------------------------------------------------
    if alpha > 2_000:
        # treat every entry that is numerically maximal as active
        max_c = conv.max()
        mask = np.abs(conv - max_c) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(conv)] = 1.0
    else:
        shift = conv - conv.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

    # ---- convolution w * reversed(a) ------------------------------------
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_w_rev = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    grad = 2.0 * conv_w_rev[start:start + n]   # factor 2 from d(a*a)/da
    return grad


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n bins."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) evaluated on mid‑points of n bins."""
    a = max(a, 0.05)                # avoid singularities at 0
    x = (np.arange(n) + 0.5) / n    # midpoints in (0,1)
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0 or not np.isfinite(pdf_sum):
        pdf[:] = 1.0 / n
    else:
        pdf /= pdf_sum
    return pdf


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest list of hand‑crafted probability vectors."""
    rng = np.random.default_rng()
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # arcsine (Beta 1/2,1/2)
    candidates.append(_arcsine_distribution(n))

    # symmetric Beta(a,a) for a grid
    for a in np.linspace(0.10, 0.90, 17):
        candidates.append(_beta_distribution(n, a))

    # mixture arcsine + uniform
    arcsine = _arcsine_distribution(n)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for w in np.linspace(0.0, 0.5, 6):
        candidates.append((1.0 - w) * arcsine + w * uniform)

    # mixture of two Betas (a small and a large)
    for a1, a2, w in [(0.20, 0.70, 0.3), (0.30, 0.80, 0.5),
                     (0.10, 0.50, 0.4)]:
        mix = w * _beta_distribution(n, a1) + (1 - w) * _beta_distribution(n, a2)
        candidates.append(mix)

    # previous solution (if any)
    if previous is not None and previous.size == n:
        candidates.append(previous.copy())

    # a few random Dirichlet draws
    for _ in range(5):
        candidates.append(rng.dirichlet(np.ones(n)))

    # final projection to simplex
    candidates = [_project_onto_simplex(c, 1.0) for c in candidates]
    return candidates


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex with enforced symmetry.
    Returns the best vector seen and its raw (2·n·max(conv)) score.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.07
    alpha_max = 500_000.0

    # FFT size for convolution of (2n‑1) with n
    size_fft = 1 << ((3 * n - 2).bit_length())  # >= (2n‑1)+(n)-1

    it = 0
    while time.time() < deadline:
        it += 1

        # current objective (raw, without denominator because sum=1)
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] it={it} score={best_score:.6f}")

        # gradient via soft‑max surrogate
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto simplex + symmetry enforcement
        a = _project_onto_simplex(a, 1.0)
        a = (a + a[::-1]) * 0.5   # enforce exact symmetry

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search that moves small mass packets,
    respects the simplex, and keeps the vector symmetric.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- generic random swap ----------
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue
        frac = rng.random() * 0.30          # up to 30 % of source mass
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand, 1.0)
        cand = (cand + cand[::-1]) * 0.5
        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best, best_score = cand, cand_score

        # ---------- max‑focused improvement ----------
        if it % 2000 == 0:
            conv = _fft_self_convolve(best)
            max_val = conv.max()
            active = np.where(conv >= max_val - 1e-12)[0]
            p = rng.choice(active)           # pick a maximal index
            # find pair (i, p-i) contributing most to this max entry
            lo = max(0, p - (n - 1))
            hi = min(p, n - 1)
            idxs = np.arange(lo, hi + 1)
            contrib = best[idxs] * best[p - idxs]
            if contrib.size > 0:
                k = np.argmax(contrib)
                i0 = idxs[k]
                j0 = p - i0
                # move a tiny packet from the larger of the two to a random neighbour
                src = i0 if best[i0] >= best[j0] else j0
                # pick a neighbour index (cyclic to stay inside bounds)
                neigh = src + (1 if rng.random() < 0.5 else -1)
                neigh = neigh % n
                if best[src] > 1e-12:
                    delta = best[src] * 0.01   # 1 % move
                    cand2 = best.copy()
                    cand2[src] -= delta
                    cand2[neigh] += delta
                    cand2 = _project_onto_simplex(cand2, 1.0)
                    cand2 = (cand2 + cand2[::-1]) * 0.5
                    cand2_score = 2.0 * n * _fft_self_convolve(cand2).max()
                    if cand2_score < best_score:
                        best, best_score = cand2, cand2_score

        # occasional jitter to escape shallow plateaus
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter = (jitter + jitter[::-1]) * 0.5
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best, best_score = jitter, jitter_score

    return best, best_score

    candidates = _candidate_shapes(n, prev_seq)

    best_seq = None
    best_score = float("inf")
    for cand in candidates:

        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
            print(f"[shape] new best {best_score:.6f}")

    if best_seq is None:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform")

    # --------------------------------------------------------------
    # Phase 1 – projected Adam optimisation (symmetry enforced).
    adam_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.55)
    seq_adam, _ = _adam_optimize(best_seq, n, adam_deadline)

    # --------------------------------------------------------------
    # Phase 2 – stochastic hill‑climbing polishing.
    hill_deadline = deadline
