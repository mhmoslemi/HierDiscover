"""
Multi‑start projected Adam optimiser with simplex projection and explicit
symmetrisation, followed by a refined stochastic hill‑climber.  The
algorithm explores a richer family of Beta‑type initial shapes,
adds random Dirichlet restarts, and optionally upsamples the solution
to a larger number of bins.  All steps respect the non‑negative simplex
constraints required by ``evaluate_sequence``.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT (returns length 2·n‑1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    # Trim to the exact length of the linear convolution
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         conv: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·conv_k) w.r.t. a.
    For large α this becomes a sub‑gradient of max(conv).
    """
    n = a.shape[0]

    # --- soft‑max weights -------------------------------------------------
    if alpha > 2_000:
        # treat every entry that is numerically maximal as active
        max_c = conv.max()
        mask = np.abs(conv - max_c) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(conv)] = 1.0
    else:
        shift = conv - conv.max()                # for numerical stability
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

    # --- convolution w * reversed(a) ------------------------------------
    a_rev = a[::-1]                       # reversed vector
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_w = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    grad = 2.0 * conv_w[start:start + n]  # factor 2 from derivative of a*a
    return grad


# ----------------------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    """
    Symmetric (if a==b) or asymmetric Beta(a,b) density evaluated on the
    mid‑points of n equal bins.  Returns a probability vector (sums to 1).
    """
    a = max(a, 0.01)
    b = max(b, 0.01)
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0 or not np.isfinite(pdf_sum):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= pdf_sum
    return pdf


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest collection of hand‑crafted probability vectors.
    All returned vectors lie on the simplex Σx = 1.
    """
    candidates = []

    # 1) uniform
    candidates.append(_uniform_distribution(n))

    # 2) arcsine (Beta(½,½))
    candidates.append(_arcsine_distribution(n))

    # 3) symmetric Beta(a,a) for a in a small grid
    for a in np.linspace(0.2, 0.9, 9):
        candidates.append(_beta_distribution(n, a, a))

    # 4) asymmetric Beta(a,b) grid (a <= b to keep shape roughly
    #    U‑shaped but not completely symmetric)
    for a, b in [(0.2, 0.7), (0.3, 0.8), (0.4, 1.0), (0.5, 1.2)]:
        candidates.append(_beta_distribution(n, a, b))

    # 5) mixtures uniform + arcsine
    arcs = _arcsine_distribution(n)
    uni = _uniform_distribution(n)
    for w in np.linspace(0.0, 0.4, 5):
        mix = (1.0 - w) * arcs + w * uni
        candidates.append(mix)

    # 6) arcsine with small spikes at the boundaries
    for spike_frac in (0.08, 0.15):
        mix = arcs.copy()
        mix[0] += spike_frac / 2
        mix[-1] += spike_frac / 2
        mix = _project_onto_simplex(mix, 1.0)
        candidates.append(mix)

    # 7) previous best (if supplied)
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())

    # 8) a few random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(6):
        candidates.append(rng.dirichlet(np.ones(n)))

    # Project everything onto the simplex (numeric safety)
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float,
                   enforce_symmetry: bool = True) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex.
    Returns the best vector seen and its internal 2·n·max(conv) score.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Soft‑max temperature schedule (starts low, ramps up very fast)
    alpha = 5.0
    alpha_mul = 1.12
    alpha_max = 2e5               # large enough to act as exact sub‑gradient

    # Size for the convolution w * reversed(a) – must accommodate
    # len(conv)=2n‑1 and len(a_rev)=n → at least 3n‑2.
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # current objective (exact, no soft‑max)
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} it={it} score={best_score:.6f}")

        # gradient via (soft‑max) surrogate; when alpha is huge it is exact
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.25 / (np.sqrt(v_hat) + eps)   # a bit more aggressive than 0.2
        a = a - step * m_hat

        # projection onto simplex
        a = _project_onto_simplex(a, 1.0)

        # optional symmetrisation
        if enforce_symmetry:
            a = (a + a[::-1]) * 0.5
            a = _project_onto_simplex(a, 1.0)

        # increase temperature quickly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float,
                enforce_symmetry: bool = True) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a random fraction of the mass
    from one index to another and keep the move if it improves the score.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)

        if best[i] < 1e-12:
            continue

        # move up to 25 % of the mass at i towards j
        frac = rng.random() * 0.25
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        if enforce_symmetry:
            # enforce the mirror move as well (if i and j are not mirrored)
            mi, mj = n - 1 - i, n - 1 - j
            cand[mi] -= delta
            cand[mj] += delta

        # keep non‑negative just in case numerical noise appears
        cand = np.maximum(cand, 0.0)

        # re‑project onto simplex (and symmetrise again)
        cand = _project_onto_simplex(cand, 1.0)
        if enforce_symmetry:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_onto_simplex(cand, 1.0)

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional tiny jitter to escape plateaus
        if it % 4000 == 0:
            jitter = best + rng.normal(scale=1e-7, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            if enforce_symmetry:
                jitter = (jitter + jitter[::-1]) * 0.5
                jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score


# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector a from its current length
    to new_n points, followed by a projection onto the simplex.
    """
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    # cumulative positions of bin mid‑points in [0,1)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_a = np.interp(new_x, old_x, a)
    new_a = _project_onto_simplex(new_a, 1.0)
    return new_a

            print(f"[shape] New best score {best_score:.6f} (length={n})")

    if best_seq is None:
        best_seq = _uniform_distribution(n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[shape] Fallback to uniform distribution")

    # Phase 1 – multi‑restart projected Adam.
    # Allocate roughly 60 % of the remaining budget to Adam,
    # the rest will be used for hill‑climbing and a possible up‑sample.
    remaining = deadline - time.time()

    rng = np.random.default_rng()

        init = rng.dirichlet(np.ones(n))
        # give each restart at most 2 s (or the remaining time if less)
        slice_deadline = min(time.time() + 2.0, adam_deadline)
        seq_tmp, _ = _adam_optimize(init, n, slice_deadline, enforce_symmetry=True)
        # compare using the cheap internal score
        tmp_score = 2.0 * n * _fft_self_convolve(seq_tmp).max()
        cur_score = 2.0 * n * _fft_self_convolve(seq_adam).max()
        if tmp_score < cur_score:
            seq_adam = seq_tmp
            print(f"[restart] Improved Adam solution: score={tmp_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – optional up‑sample to a larger grid for finer resolution.
    # This costs a little extra time but often tightens the bound.
    if upsample_budget > 5.0:   # only if we have a noticeable slice left
        new_n = min(15000, n * 2)
        up_seq = _upsample(seq_adam, new_n)
        up_deadline = time.time() + upsample_budget
        # a short Adam refinement on the larger grid
        up_seq_refined, _ = _adam_optimize(up_seq, new_n, up_deadline,

"""
Multi‑start projected Adam optimiser with simplex projection and explicit
symmetrisation, followed by a refined stochastic hill‑climber.  The
algorithm explores a richer family of Beta‑type initial shapes,
adds random Dirichlet restarts, and optionally upsamples the solution
to a larger number of bins.  All steps respect the non‑negative simplex
constraints required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT (returns length 2·n‑1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         conv: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·conv_k) w.r.t. a.
    For huge α this becomes a sub‑gradient of max(conv).
    """
    n = a.shape[0]

    # ----- soft‑max weights ------------------------------------------------
    if alpha > 2_000:
        # treat every entry that is numerically maximal as active
        max_c = conv.max()
        mask = np.abs(conv - max_c) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(conv)] = 1.0
    else:
        shift = conv - conv.max()                # stability shift
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

    # ----- convolution w * reversed(a) ------------------------------------
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_w = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    grad = 2.0 * conv_w[start:start + n]   # factor 2 from derivative of a*a
    return grad


# ----------------------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    """
    Beta(a,b) density on the mid‑points of n equal bins.
    Returns a probability vector (sums to 1).  a,b are forced to be >0.
    """
    a = max(a, 0.01)
    b = max(b, 0.01)
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    s = pdf.sum()
    if s == 0.0 or not np.isfinite(s):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= s
    return pdf


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest collection of hand‑crafted probability vectors."""
    candidates = []

    # 1) uniform
    candidates.append(_uniform_distribution(n))

    # 2) arcsine (Beta(½,½))
    candidates.append(_arcsine_distribution(n))

    # 3) symmetric Beta(a,a) grid
    for a in np.linspace(0.2, 0.9, 9):
        candidates.append(_beta_distribution(n, a, a))

    # 4) a few asymmetric Beta shapes (still roughly U‑shaped)
    for a, b in [(0.2, 0.7), (0.3, 0.8), (0.4, 1.0), (0.5, 1.2)]:
        candidates.append(_beta_distribution(n, a, b))

    # 5) mixtures uniform + arcsine
    arcs = _arcsine_distribution(n)
    uni = _uniform_distribution(n)
    for w in np.linspace(0.0, 0.4, 5):
        mix = (1.0 - w) * arcs + w * uni
        candidates.append(mix)

    # 6) arcsine with tiny spikes at the boundaries
    for spike_frac in (0.08, 0.15):
        mix = arcs.copy()
        mix[0] += spike_frac / 2
        mix[-1] += spike_frac / 2
        mix = _project_onto_simplex(mix, 1.0)
        candidates.append(mix)

    # 7) previous best (if supplied and same length)
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())

    # 8) a few random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(6):
        candidates.append(rng.dirichlet(np.ones(n)))

    # Ensure exact simplex feasibility
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float,
                   enforce_symmetry: bool = True) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex.
    Returns the best vector seen and its internal 2·n·max(conv) score.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Soft‑max temperature schedule (ramp quickly to a huge value)
    alpha = 5.0
    alpha_mul = 1.12
    alpha_max = 2e5

    # Size for w * reversed(a) – must be at least 3n‑2
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} it={it} score={best_score:.6f}")

        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.25 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto simplex
        a = _project_onto_simplex(a, 1.0)

        # optional symmetrisation
        if enforce_symmetry:
            a = (a + a[::-1]) * 0.5
            a = _project_onto_simplex(a, 1.0)

        # temperature increase
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float,
                enforce_symmetry: bool = True) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search: move a random fraction of mass from one
    index to another and keep the move if it improves the objective.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)

        if best[i] < 1e-12:
            continue

        # move up to 25 % of the mass at i towards j
        frac = rng.random() * 0.25
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        if enforce_symmetry:
            mi, mj = n - 1 - i, n - 1 - j
            cand[mi] -= delta
            cand[mj] += delta

        cand = np.maximum(cand, 0.0)
        cand = _project_onto_simplex(cand, 1.0)
        if enforce_symmetry:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_onto_simplex(cand, 1.0)

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter
        if it % 4000 == 0:
            jitter = best + rng.normal(scale=1e-7, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            if enforce_symmetry:
                jitter = (jitter + jitter[::-1]) * 0.5
                jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score


# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector a from length len(a)
    to new_n points, followed by projection onto the simplex.
    """
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_a = np.interp(new_x, old_x, a)
    new_a = _project_onto_simplex(new_a, 1.0)
    return new_a
