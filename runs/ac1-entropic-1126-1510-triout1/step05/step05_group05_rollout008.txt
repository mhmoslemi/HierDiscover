"""
A hybrid optimizer for the non‑negative step‑function problem.

The approach mixes three ideas:

1. **Diverse parametric seeds** – several families (arcsine, power‑law,
   beta, Gaussian, mixtures) are sampled on a handful of lengths and
   projected onto the simplex.  The best few seeds are kept.

2. **Exponentiated‑gradient refinement** – each seed is polished with a
   mirror‑descent / exponentiated‑gradient loop that starts with a
   smooth soft‑max surrogate of the maximum convolution and slowly
   anneals to the true sub‑gradient.

3. **Simulated‑annealing fine‑tuning** – the current best sequence is
   further improved by a simple random‑move SA that moves a small
   fraction of probability mass between two coordinates.  The temperature
   is exponentially cooled, and every improvement is kept.

The whole search respects the 1000 s budget and returns the best
sequence (as a plain Python list) that it discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """Symmetric 1/(i·(n‑i))^p (invert=True) or (i·(n‑i))^p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) distribution on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on the index set."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """Convex combination (1‑lam)*arcsine + lam*uniform."""
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) for a simplex‑normalised vector."""
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv  # denominator is 1 because sum(a)==1


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the (1/α)·log‑sum‑exp(α·b) surrogate w.r.t. a.
    Implements g = 2 * conv_rev(w, a_rev) via FFT.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser.
    Returns the best sequence seen and its (2·n·max) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    eta0 = 0.9               # a slightly larger initial step‑size
    alpha = 1.0              # temperature for soft‑max surrogate
    alpha_mul = 1.015
    alpha_max = 20000.0
    alpha_thr = 2500.0       # switch to pure sub‑gradient after this
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient
        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, b, alpha, size_fft)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)

        # renormalise onto simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.14:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random‑move hill‑climbing on the best sequence.
    Moves a fraction of mass from one index to another if it improves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.08  # at most 8 % of source
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cur_score = _objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _optimize_sa(initial_a: np.ndarray,
                 n: int,
                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.  Mass is repeatedly moved from a random
    source index to a random target index; moves are accepted either if
    they improve the objective or with a probability depending on a
    temperature that slowly cools.
    """
    a = initial_a.copy()
    best_seq = a.copy()
    cur_score = _objective(a)
    best_score = cur_score

    # SA hyper‑parameters
    move_factor = 0.25                # up‑to this fraction of source mass
    T = max(1e-4, cur_score * 0.05)   # initial temperature
    cooling = 0.9996
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1                     # ensure i ≠ j

        eps = a[i] * move_factor * np.random.rand()
        if eps <= 1e-15:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue

        cand_score = _objective(a_candidate)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] SA it={it} improved score={best_score:.6f}")

        # exponential cooling
        T *= cooling
        if T < 1e-7:
            T = 1e-7
        # slowly tighten the move magnitude
        if it % 5000 == 0:
            move_factor = max(0.02, move_factor * 0.95)

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Main entry point used by the competition harness.
    Runs a multi‑phase optimisation (random seeds → EG → SA → hill‑climb)
    within the allotted time and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin for final prints

    # ------------------------------------------------------------------
    # Determine a sensible set of lengths
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    length_factors = (0.5, 0.7, 1.0, 1.3, 1.6, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    })
    # ensure some hand‑picked values are present
    lengths.extend([5000, 8000, 12000, 16000, 20000])
    lengths = sorted({l for l in lengths if l <= 20000})

    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    candidates = []   # (score, sequence)

    # ------------------------------------------------------------------
    # Phase 1 – generate a diverse pool of seeds
    families = ("arcsine", "power_invert", "power_direct",
                "beta", "gauss", "mix")
    for n in lengths:
        for fam in families:
            if fam == "arcsine":
                seq = _arcsine_distribution(n)
            elif fam == "power_invert":
                p = np.random.uniform(0.2, 1.8)
                seq = _power_law_distribution(n, p, invert=True)
            elif fam == "power_direct":
                p = np.random.uniform(0.2, 1.8)
                seq = _power_law_distribution(n, p, invert=False)
            elif fam == "beta":
                a = np.random.uniform(0.2, 4.0)
                seq = _beta_distribution(n, a)
            elif fam == "gauss":
                sigma = np.random.uniform(n / 20.0, n / 2.0)
                seq = _gaussian_distribution(n, sigma)
            elif fam == "mix":
                lam = np.random.rand()
                seq = _mixture_arcsine_uniform(n, lam)
            else:
                seq = _arcsine_distribution(n)

            seq = _project_onto_simplex(seq, s=1.0)
            score = _objective(seq)
            candidates.append((score, seq))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] New global seed best: n={n} fam={fam} score={best_score:.6f}")

    # keep the top few seeds for refinement
    candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidates[:5]]

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient refinement of the seeds
    for seed_seq in top_seeds:
        if time.time() >= deadline - 30.0:
            break
        n = seed_seq.shape[0]
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(45.0, remaining * 0.6)
        refined_seq, _ = _optimize_exponential(seed_seq, n, slice_deadline)
        refined_score = _objective(refined_seq)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time():.2f}] EG refinement improved to {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing fine‑tuning on the current best
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        sa_seq, sa_score = _optimize_sa(best_seq, n, deadline)
        if sa_score < best_score:
            best_score = sa_score
            best_seq = sa_seq.copy()
            print(f"[{time.time():.2f}] SA improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climbing
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        lc_seq, lc_score = _local_refinement(best_seq, n, deadline)
        if lc_score < best_score:
            best_score = lc_score
            best_seq = lc_seq.copy()
            print(f"[{time.time():.2f}] Final hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # Return plain Python list as required by the harness
    print(f"[{time.time():.2f}] Returning final best score={best_score:.6f} "
          f"with length={best_seq.shape[0]}")
    return best_seq.tolist()
