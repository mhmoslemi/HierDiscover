"""
search for a low‑score non‑negative step function.

The algorithm proceeds in three stages:

1. **Broad generation** – a pool of candidate probability vectors
   (arcsine, Beta, power‑law, Gaussian, mixtures, random, interpolation of a
   previously discovered best) is built for several lengths.  The K best
   candidates according to the fast surrogate 2·n·max(conv) are kept.

2. **Refinement** – each of the kept candidates is refined by three
   optimisation phases:
      • Adam on a smooth‑max surrogate (log‑sum‑exp) with a slowly
        increasing temperature α.
      • Exact ℓ∞ sub‑gradient descent.
      • Simulated‑annealing polishing that moves small amounts of mass
        between locations while occasionally accepting uphill moves.

   The best result over all candidates is retained.

3. **Final evaluation** – the best vector is finally evaluated with the
   black‑box `evaluate_sequence`.  All operations respect the simplex
   Σaᵢ = 1, aᵢ ≥ 0.

The routine adapts its behaviour to the remaining time budget (default
1000 s) and prints progress information.  It returns a plain Python list
that can be fed directly to `evaluate_sequence`.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid overflow for α,β<1
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, exponent: float) -> np.ndarray:
    """Distribution proportional to ((i+½)(n‑i‑½))^{-exponent}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-exponent)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma_factor: float) -> np.ndarray:
    """
    Discrete centred Gaussian (σ = sigma_factor·n) normalised to sum 1.
    """
    sigma = sigma_factor * n
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lₐ(a) = (1/α)·log Σₖ exp(α·(a*a)ₖ)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # stable soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max

                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 3000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector seen (according to exact max) and its peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        # update and re‑project onto simplex
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep best according to the exact max
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # temperature annealing for the smooth‑max
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to shake the optimiser
        if t % 400 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One exact ℓ∞ sub‑gradient step.
    Uses uniform weights on the active set (indices attaining the current maximum).
    Returns (grad, cur_max).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short exact‑max refinement using sub‑gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.03 / np.sqrt(it)          # slightly larger initial step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if it % 300 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 4e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Simple simulated‑annealing polishing.
    Moves a small amount of mass between two random positions,
    accepting uphill moves with a temperature schedule.
    Returns the best vector found and its peak convolution.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    # temperature schedule (exponential decay)
    start_temp = max(cur_peak, 1e-3) * 0.6
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # pick source index with non‑zero mass
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        # move a random fraction (up to 10 %) of the source mass
        max_delta = min(cur_a[src], 0.10)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional reset to the best found so far
        if iter_cnt % 15000 == 0 and np.random.rand() < 0.07:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative sequence to a new length via linear interpolation."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights with a low evaluation score.
    Returns a plain Python list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    # ------------------------------------------------------------------
    # Try to load a previously discovered construction.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
        print(f"[init] loaded previous sequence (length={base_n})")
    except Exception:
        base_seq = None
        base_n = 3500
        print("[init] no previous sequence found – starting from scratch")

    # ------------------------------------------------------------------
    # Choose a handful of lengths to explore (both down‑ and up‑scaling).
    lengths = []
    for factor in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0):
        n = int(base_n * factor)
        n = max(500, min(n, 8000))
        lengths.append(n)
    # add a few fixed reference lengths
    for n in (500, 1024, 2048, 4096, 6000):
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Broad generation of candidates – keep the K best according to the fast surrogate.
    K = 8
    best_candidates = []   # list of tuples (fast_score, vector, tag)

    def _maybe_keep(vec: np.ndarray, tag: str, n: int):
        vec = _project_onto_simplex(vec, s=1.0)
        maxb = _fft_self_convolve(vec).max()
        fast = 2.0 * n * maxb
        if len(best_candidates) < K:
            best_candidates.append((fast, vec, tag))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if fast < best_candidates[-1][0]:
                best_candidates[-1] = (fast, vec, tag)
                best_candidates.sort(key=lambda x: x[0])

    # generate candidates for each length
    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families
        _maybe_keep(_arcsine_distribution(n), f"arcsine_n{n}", n)
        _maybe_keep(np.full(n, 1.0 / n), f"uniform_n{n}", n)

        # Gaussian families
        for sigma_factor in (0.12, 0.15, 0.20, 0.25):
            _maybe_keep(_gaussian_distribution(n, sigma_factor),
                        f"gauss_{sigma_factor:.2f}_n{n}", n)

        # Power‑law families
        for exp in (0.4, 0.5, 0.6, 0.8, 1.0, 1.2):
            _maybe_keep(_power_law_distribution(n, exp),
                        f"power_{exp}_n{n}", n)

        # Symmetric Beta families (including very spiky cases)
        for a in (0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.8):
            _maybe_keep(_beta_distribution(n, a, a),
                        f"beta_{a:.2f}_n{n}", n)

        # Asymmetric Beta families
        for (a, b) in ((0.15, 0.8), (0.8, 0.15), (0.25, 0.7), (0.7, 0.25)):
            _maybe_keep(_beta_distribution(n, a, b),
                        f"beta_{a:.1f}_{b:.1f}_n{n}", n)

        # Mixtures of arcsine and uniform
        arcs = _arcsine_distribution(n)
        uniform = np.full(n, 1.0 / n)
        for w in (0.1, 0.3, 0.5, 0.7):
            mix = (1.0 - w) * arcs + w * uniform
            _maybe_keep(mix, f"mix_{w:.1f}_n{n}", n)

        # Random starts
        for _ in range(2):
            rand = np.random.rand(n)
            _maybe_keep(rand, "rand", n)

        # Interpolation of previous best if available
        if base_seq is not None:
            _maybe_keep(_interpolate_to_length(base_seq, n),
                        "prev_interp", n)

    # ------------------------------------------------------------------
    print(f"[phase 0] collected {len(best_candidates)} top candidates after broad search")
    for i, (score, _, tag) in enumerate(best_candidates):
        print(f"  #{i+1:02d}: {tag:<30s} fast_score={score:.6f}")

    # ------------------------------------------------------------------
    # Refinement of the shortlisted candidates
    overall_best_vec = None
    overall_best_score = float("inf")

    for idx, (fast, a0, tag) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        n = a0.shape[0]

        # allocate a slice of remaining time (max 300 s per candidate)
        time_left = deadline - time.time()
        cand_deadline = time.time() + min(time_left * 0.25, 300.0)
        print(f"[candidate {idx+1}/{len(best_candidates)}] {tag} (n={n}) – time budget≈{cand_deadline - time.time():.1f}s")

        # ---- Phase A: Adam on smooth‑max surrogate ----
        adam_deadline = time.time() + (cand_deadline - time.time()) * 0.55
        a_opt, _ = _adam_optimize(a0, adam_deadline,
                                  init_alpha=1.0,
                                  alpha_factor=1.01,
                                  alpha_max=3000.0,
                                  lr0=0.3)

        # ---- Phase B: exact‑max sub‑gradient descent ----
        sub_deadline = time.time() + (cand_deadline - time.time()) * 0.30
        a_opt = _refine_max(a_opt, sub_deadline)

        # ---- Phase C: simulated‑annealing polishing ----
        sa_deadline = time.time() + (cand_deadline - time.time()) * 0.15
        a_opt, final_peak = _simulated_annealing(a_opt, sa_deadline)

        fast_score = 2.0 * n * final_peak
        print(f"    after refinement – fast_score={fast_score:.6f}")

        if fast_score < overall_best_score:
            overall_best_score = fast_score
            overall_best_vec = a_opt.copy()
            print(f"    *** new overall best (fast) score {overall_best_score:.6f}")

        # a quick true evaluation if time permits
        if time.time() + 0.5 < deadline:
            try:
                true_score = evaluate_sequence(a_opt.tolist())
                print(f"    true evaluate_score={true_score:.6f}")
                if true_score < overall_best_score:
                    overall_best_score = true_score
                    overall_best_vec = a_opt.copy()
                    print(f"    *** new overall best (true) score {overall_best_score:.6f}")
            except Exception as exc:
                print("    evaluate_sequence raised:", exc)

    # ------------------------------------------------------------------
    # Fallback in the improbable case that nothing was found
    if overall_best_vec is None:
        n = max(500, base_n)
        overall_best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_vec).max()
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Final evaluation with the real black‑box, if time remains
    if deadline - time.time() > 0.5:
        try:
            true_final = evaluate_sequence(overall_best_vec.tolist())
            print(f"[final] evaluate_sequence score={true_final:.6f}")
            if true_final < overall_best_score:
                overall_best_score = true_final
        except Exception as exc:
            print("[final] evaluate_sequence raised:", exc)

    print(f"Best approximate score {overall_best_score:.6f} (length={len(overall_best_vec)})")
    return overall_best_vec.tolist()
