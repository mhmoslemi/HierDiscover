def _exp_gradient_opt(a0,n,deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _fast_score(a)
    it = 0
    # step size param and schedule
    eta0 = 0.6
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        maxk = int(np.argmax(conv))
        cur_score = 2 * n * conv[maxk]
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute subgradient at max
        idx = np.arange(n)
        valid = (maxk - idx >= 0) & (maxk - idx < n)
        grad = np.zeros_like(a)
        grad[valid] = 2.0 * a[maxk - idx[valid]]
        # step size
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        # re-normalize onto simplex
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total <= 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / total
        # occasional jitter to escape plateau
        if it % 5000 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a,0.0)
            a = a / a.sum()
    return best_a, best_score

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm overview
------------------
1. **Select a few promising lengths** (500 … ≈30 000).  Larger n gives a
   lower theoretical limit (≈1.5) but also costs a little more per
   iteration, therefore we start with a moderate length and only move
   to larger ones when time permits.

2. **Generate a small pool of seed vectors** for each length:
   * the continuous arcsine density (the known near‑optimal shape),
   * the uniform vector,
   * a few random Dirichlet samples,
   * the best vector from a *fine* grid search over the one‑parameter
     power‑law family ``a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ`` and a small uniform
     mixture.

3. **Exponential‑gradient (mirror‑descent) refinement**  
   For each seed we run a long exponentiated‑gradient loop that uses the
   exact sub‑gradient of the max‑entry of the self‑convolution.  The
   iterates stay on the simplex via a projection step.
   This phase quickly drives the score down close to the optimum.

4. **Simulated‑annealing / stochastic hill climbing**  
   The best vector found so far is further improved by a temperature‑
   controlled random walk that moves a fraction of the mass from one
   index to another and accepts worsening moves with probability
   ``exp(-Δ/temperature)``.  The temperature decays linearly to a very
   small value, allowing the search to escape shallow local minima.

5. **Return the best sequence** discovered before the budget expires.
   All values are non‑negative, sum to one (exactly) and therefore the
   scaling invariance of the objective guarantees that the returned list
   can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho_arr.size == 0 else cssv[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) self‑convolution via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power‑of‑two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Internal cheap evaluation: 2·n·max(conv)  (sum(a)=1)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def _beta_power_shape(n: int, p: float) -> np.ndarray:
    """Shape a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ, sum‑normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _asymmetric_power_shape(n: int, pl: float, pr: float) -> np.ndarray:
    """Shape a_i ∝ (i+½)⁻ˡ·(n‑i‑½)⁻ʳ, sum‑normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-pl) * (n - i - 0.5) ** (-pr)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_shape(n: int) -> np.ndarray:
    """Discretised arcsine density (the classical near‑optimal shape)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _grid_search_best_fine(n: int) -> tuple[np.ndarray, float]:
    """
    Small but fine grid over the one‑parameter family (p,w) and a few
    asymmetric variants.  Returns the best vector and its *fast* score.
    """
    best_seq = None
    best_sc = np.inf

    # 1) symmetric power‑law + tiny uniform mix
    p_vals = np.linspace(0.40, 0.60, 21)          # step 0.01
    w_vals = np.linspace(0.0, 0.20, 11)           # only a small uniform part
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    for p in p_vals:
        shape = _beta_power_shape(n, p)
        for w in w_vals:
            cand = (1.0 - w) * shape + w * uniform
            cand = _project_onto_simplex(cand, 1.0)
            sc = _fast_score(cand)
            if sc < best_sc:
                best_sc, best_seq = sc, cand.copy()

    # 2) asymmetric two‑parameter family (a few points)
    exp_vals = np.linspace(0.30, 0.70, 9)          # coarse range around 0.5
    for pl in exp_vals:
        for pr in exp_vals:
            cand = _asymmetric_power_shape(n, pl, pr)
            sc = _fast_score(cand)
            if sc < best_sc:
                best_sc, best_seq = sc, cand.copy()

    # 3) fallback: pure arcsine
    cand = _arcsine_shape(n)
    sc = _fast_score(cand)
    if sc < best_sc:
        best_sc, best_seq = sc, cand.copy()

    return best_seq, best_sc

def _exp_gradient_opt(a0: np.ndarray,
                     n: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the simplex.
    Uses the exact sub‑gradient of the max‑entry of the self‑convolution.
    Returns the best vector seen and its fast score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = _fast_score(a)

    it = 0
    eta0 = 0.55                     # initial learning‑rate factor

    idx_all = np.arange(n, dtype=np.int64)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        cur_sc = 2.0 * n * conv[k]

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # sub‑gradient of the max entry (see _subgradient_max in the
        # reference code)
        valid = (k - idx_all >= 0) & (k - idx_all < n)
        grad = np.zeros_like(a)
        grad[valid] = 2.0 * a[k - idx_all[valid]]

        # diminishing step‑size
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # clip, renormalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # occasional tiny jitter to break possible plateaus
        if it % 5000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk on the simplex.
    Moves a random fraction of mass from one index to another;
    worsening moves are accepted with probability exp(-Δ/T).
    Returns the best vector encountered.
    """
    cur = a0.copy()
    cur_sc = _fast_score(cur)
    best = cur.copy()
    best_sc = cur_sc

    start = time.time()
    max_temp = 2.0e-2   # relatively high at the beginning
    min_temp = 5.0e-4   # very low at the end

    while time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start
        ratio = elapsed / (deadline - start)
        temp = max_temp * (1.0 - ratio) + min_temp * ratio

        # propose a move: transfer a fraction of mass from i -> j
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if cur[i] == 0.0:
            continue
        delta = cur[i] * np.random.rand() * 0.40   # up to 40 % of source mass
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # clean‑up rounding errors
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        cand_sc = _fast_score(cand)
        diff = cand_sc - cur_sc
        if diff < 0.0 or np.random.rand() < math.exp(-diff / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc

        # very occasional global jitter
        if np.random.rand() < 0.001:
            cur += np.random.randn(n) * 1e-5
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_sc = _fast_score(cur)

    return best, best_sc

            rnd /= rnd.sum()
            seeds.append(rnd)

        # d) best shape from a cheap fine grid search
        shape_best, _ = _grid_search_best_fine(n)
        seeds.append(shape_best)

        # keep only the 4 best according to the cheap score
        seeds = sorted(seeds, key=_fast_score)[:4]

        # --------------------------------------------------------------
        # exponentiated‑gradient refinement for each seed
        # --------------------------------------------------------------
        for seed_vec in seeds:
            if time.time() >= slice_deadline:
                break
            opt_vec, _ = _exp_gradient_opt(seed_vec, n, slice_deadline)

            # evaluate with the *real* black‑box (it does the same scaling)
            try:
                real_score = evaluate_sequence(opt_vec.tolist())
            except Exception:
                real_score = _fast_score(opt_vec)

            if real_score < overall_score:
                overall_score = real_score
                overall_best = opt_vec.copy()
                print(f"*** New global best → len={n} score={overall_score:.6f}")

            if overall_score <= target_score:
                break

        if overall_score <= target_score:
            break

    # ------------------------------------------------------------------
    # 3) final stochastic refinement on the globally best vector
    # ------------------------------------------------------------------
    if overall_best is not None and time.time() < deadline - 1.0:
        n_best = len(overall_best)
        refined, _ = _simulated_annealing(overall_best, n_best, deadline)

        try:
            final_score = evaluate_sequence(refined.tolist())
        except Exception:
            final_score = _fast_score(refined)

        if final_score < overall_score:
            overall_score = final_score
            overall_best = refined.copy()
            print(f"*** After annealing score={overall_score:.6f}")

    # ------------------------------------------------------------------
    # 4) safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if overall_best is None:
        n_fallback = 1000
        overall_best = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_score = evaluate_sequence(overall_best.tolist())
        print("Fallback sequence used.")

    print(f"Final best score: {overall_score:.6f} (length = {len(overall_best)})")
    return overall_best.tolist()
