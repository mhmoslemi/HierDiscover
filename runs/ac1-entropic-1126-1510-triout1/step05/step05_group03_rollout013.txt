"""
Propose a step‑function with a very low evaluation score.

The algorithm works in three stages:

1️⃣ Broad search – a large collection of simple distributions (uniform,
   arcsine, many symmetric/asymmetric Beta‑densities, random Dirichlet
   draws and simple mixtures) is generated for several lengths.  The
   best candidate according to the cheap surrogate 2·n·max(conv) is kept.

2️⃣ Long gradient optimisation – the best candidate is refined many
   times with Adam on a smooth‐max surrogate of the ℓ∞‑convolution.
   Multiple random restarts are performed inside the same time budget
   to escape local minima.

3️⃣ Polishing – a short exact‑max sub‑gradient descent followed by a
   simulated‑annealing phase transfers tiny amounts of mass between
   positions while occasionally accepting uphill moves.  The final
   sequence is evaluated with the true black‑box ``evaluate_sequence``.

All operations respect the simplex constraint Σaᵢ=1, aᵢ≥0.  The code
stays within the 1000 s budget and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # power of two >= 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow when α,β<1
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k).

    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # numerically stable soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)            # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 3000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered and the peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    last_improve = time.time()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (diminishing)
        lr = lr0 / np.sqrt(t)

        # parameter update + projection
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()
            last_improve = time.time()

        # increase smoothing parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 800 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

        # early stop if no improvement for a while (but keep trying until deadline)
        if time.time() - last_improve > 10.0 and alpha > alpha_max * 0.5:
            # push smoothing a little further to sharpen the landscape
            alpha = min(alpha_max, alpha * 1.01)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses a uniform weight on the active set (indices attaining the current maximum).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short exact‑max refinement using sub‑gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
            last_improve = time.time()

        # occasional tiny jitter
        if it % 600 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

        # if we are stuck for a while, inject a small random pulse
        if time.time() - last_improve > 15.0:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)
            last_improve = time.time()

    return best_a

def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Simple simulated‑annealing polishing.
    Moves a small amount of mass between two random positions,
    accepting uphill moves with an exponential temperature schedule.
    Returns the best vector found and its peak convolution.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    # temperature schedule (exponential decay)
    start_temp = max(cur_peak, 1e-3) * 0.6
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # pick a source index that currently holds some mass
        src = np.random.randint(n)
        if cur_a[src] <= 1e-14:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        # move a random fraction (up to 4 %) of the source mass
        max_delta = min(cur_a[src], 0.04)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional reset to best solution (helps when temperature is low)
        if iter_cnt % 25000 == 0 and np.random.rand() < 0.07:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights with a low score.
    Returns a plain Python list suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # 1 s safety margin

    # ------------------------------------------------------------------
    # Try to start from a previously discovered construction.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 1000

    # ------------------------------------------------------------------
    # Choose several lengths (the problem gets easier as n grows,
    # but we keep n ≤ 15000 to stay safe for the final O(n²) evaluation).
    lengths = [max(500, int(base_n * f))
               for f in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0)
               if int(base_n * f) <= 15000]
    if not lengths:
        lengths = [max(500, base_n)]

    overall_best_seq = None
    overall_best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # ---------- 1. BROAD SEARCH ---------------------------------------
    print("[phase 1] broad search for a promising start point")
    for n in lengths:
        if time.time() >= deadline:
            break

        starters = {}

        # simple families
        starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)
        starters["arcsine"] = _arcsine_distribution(n)

        # symmetric Beta grid (including extreme cases)
        for a in [0.12, 0.15, 0.2, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85]:
            starters[f"beta_sym_{a:.2f}"] = _beta_distribution(n, a, a)

        # asymmetric Beta families
        for a, b in [(0.12, 0.9), (0.2, 0.8), (0.3, 0.6), (0.6, 0.3), (0.8, 0.2)]:
            starters[f"beta_{a:.2f}_{b:.2f}"] = _beta_distribution(n, a, b)

        # random Dirichlet draws with small concentration (spiky)
        for _ in range(4):
            conc = np.random.rand() * 0.25 + 0.02
            v = np.random.gamma(conc, 1.0, size=n)
            v /= v.sum()
            starters[f"dir_{conc:.3f}"] = v

        # mixtures of arcsine and uniform
        for w in (0.2, 0.4, 0.6):
            starters[f"mix_{w:.1f}"] = (1.0 - w) * _arcsine_distribution(n) + w * np.full(n, 1.0 / n)

        # reuse the previous best construction if available
        if base_seq is not None:
            starters["prev"] = _interpolate_to_length(base_seq, n)

        # evaluate all starters with the cheap surrogate
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                best_n = n
                print(f"[init] n={n:4d} {tag:<16s} fast_score={fast_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        best_n = n
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # ---------- 2. LONG GRADIENT‑BASED REFINEMENT --------------------
    time_left = deadline - time.time()
    if time_left > 10.0:
        n = best_n
        print(f"[phase 2] Adam optimisation (n={n}) – time left {time_left:.1f}s")
        # allocate ~60 % of the remaining budget to Adam, the rest to polishing
        adam_deadline = time.time() + time_left * 0.6
        # run a first optimisation from the best starter
        a_opt, _ = _adam_optimize(overall_best_seq, adam_deadline)

        # perform a few additional random restarts inside the same window
        while time.time() < adam_deadline - 2.0:
            pert = a_opt + np.random.randn(n) * 0.02
            pert = _project_onto_simplex(pert, s=1.0)
            a_tmp, _ = _adam_optimize(pert, adam_deadline)
            if _fft_self_convolve(a_tmp).max() < _fft_self_convolve(a_opt).max():
                a_opt = a_tmp

        # short exact‑max polishing
        polish_deadline = time.time() + time_left * 0.3
        a_opt = _refine_max(a_opt, polish_deadline)

        # simulated‑annealing polishing
        sa_deadline = deadline - 0.5          # leave a tiny margin for final eval
        a_opt, final_peak = _simulated_annealing(a_opt, sa_deadline)

        overall_best_seq = a_opt.copy()
        overall_best_score = 2.0 * n * final_peak
        print(f"[phase 3] final fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---------- 3. TRUE EVALUATION -----------------------------------
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("evaluate_sequence raised:", exc)

    print(f"Final best score (approx.): {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
