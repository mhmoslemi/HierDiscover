"""
Search for a low‑score step‑function.

The algorithm consists of four phases:

1. **Length selection** – try a handful of moderate lengths (≈5‑20 k)
   because the asymptotic constant improves with n.

2. **Shape grid** – for each length a cheap grid over several
   parametric families (symmetric power‑law, asymmetric power‑law,
   mixture with uniform, random Dirichlet draws) is built and the few
   best candidates are kept.

3. **Momentum mirror‑descent** – each candidate is refined by a
   projected exponentiated‑gradient method with Nesterov‑style momentum.
   A soft‑max surrogate of the max‑convolution is used at the start,
   the temperature is slowly increased until the true sub‑gradient is
   applied.

4. **Stochastic local refinement** – finally the globally best vector
   is polished by tiny random mass transfers that are kept only when
   they improve the true evaluation score.

All work is bounded by ``budget_s`` seconds (default 1000).  The best
list of non‑negative heights (sum≈1, all values ≪ 1000) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho_arr.size == 0 else cssv[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (2·n‑1) convolution a∗a via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power‑of‑2
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max‑entry of a∗a at index *k* (exact max)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b) w.r.t. a (soft‑max surrogate for max).
    ``b`` is the full convolution a∗a.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]                     # convolution, not correlation
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _beta_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ   (p>0)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _asymmetric_power_shape(n: int, pl: float, pr: float) -> np.ndarray:
    """Asymmetric a_i ∝ (i+½)⁻ˡ·(n‑i‑½)⁻ʳ."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-pl) * (n - i - 0.5) ** (-pr)
    vals /= vals.sum()
    return vals

def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Simple seed generators (uniform, random, arcsine, beta, previous)."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a = a / a.sum()
    elif kind == "beta":
        a = np.random.beta(0.5, 0.5, size=n)
        a = a / a.sum()
    elif kind == "arcsine":
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a = a / a.sum()
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = a / a.sum()
        except Exception:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    return _project_onto_simplex(a, s=1.0)

def _optimize_with_momentum(a0: np.ndarray,
                            n: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient with momentum.
    Returns the best vector seen and its internal score 2·n·max(conv).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # hyper‑parameters
    eta0 = 0.8                     # initial stepsize factor
    mu = 0.9                       # momentum coefficient
    m = np.zeros_like(a)          # accumulated momentum
    alpha = 5.0                    # start temperature for soft‑max
    alpha_mul = 1.02
    alpha_max = 300.0
    size_fft = 1 << ((3 * n - 2).bit_length())   # safe length for gradient conv

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_seq = a.copy()

        # ---- gradient (soft‑max → true sub‑gradient) ----------
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # ---- momentum + exponentiated update -----------------
        m = mu * m + (1.0 - mu) * grad
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * m)

        # ---- simplex projection --------------------------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # ---- occasional jitter to escape plateaus ------------
        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score

def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Tiny random‑move hill‑climbing that keeps a move only when the true
    evaluation score improves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        # Occasionally try a larger move from the most massive index
        if np.random.rand() < 0.07:
            i = int(np.argmax(a))
            j = int(np.argmin(a))
            eps = a[i] * np.random.rand() * 0.25
        else:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps = a[i] * np.random.rand() * 0.10

        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue

        cur_score = 2.0 * n * np.max(_fft_self_convolve(cand))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand
            # optional: print improvement occasionally
            # if it % 1000 == 0:
            #     print(f"[local] iter={it} score={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Main entry point required by the competition harness.
    Runs for at most ``budget_s`` seconds and returns the best
    non‑negative step‑function (list of floats) discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0          # 1 s safety margin

    # --------------------------------------------------------------
    # Choose a base length – try to reuse a previous construction if
    # available; otherwise fall back to a moderate default.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # Explore a short list of lengths (larger n usually yields a smaller
    # constant, but very large n is expensive).
    length_factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted(
        {max(500, int(base_n * f)) for f in length_factors}
    )
    # Clamp to a safe upper bound
    lengths = [min(20000, n) for n in lengths]

    # --------------------------------------------------------------
    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503

    # Helper to compute the cheap internal score (2·n·max(conv)):
    def cheap_internal_score(vec: np.ndarray, n_len: int) -> float:
        return 2.0 * n_len * _fft_self_convolve(vec).max()

    # --------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(0.2, remaining * 0.33)

        # ------- build an initial pool of promising vectors ----------
        # (a) shape grid search – keep the top few candidates
        _, _, top_candidates = _grid_search_top(n, top_k=5)

        # (b) a few pure random Dirichlet seeds for extra diversity
        for _ in range(3):
            a_rand = np.random.rand(n)
            a_rand = _project_onto_simplex(a_rand, s=1.0)
            top_candidates.append(a_rand)

        # keep only the most promising according to the cheap score
        top_candidates = sorted(
            top_candidates,
            key=lambda v: cheap_internal_score(v, n)
        )[:4]

        # --------------------------------------------------------------
        # Optimise each candidate with the momentum mirror‑descent routine.
        for a0 in top_candidates:
            if time.time() >= slice_deadline:
                break

            seq_opt, internal_score = _optimize_with_momentum(a0, n, slice_deadline)

            # Evaluate with the official black‑box (handles clipping etc.)
            try:
                final_score = evaluate_sequence(seq_opt.tolist())
            except Exception:
                final_score = internal_score  # fallback to internal metric

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq_opt.copy()
                print(
                    f"*** New global best → len={n} "
                    f"score={overall_best_score:.6f}"
                )
                # early exit if we have already beaten the target
                if overall_best_score <= target_score:
                    break

        if overall_best_score <= target_score:
            break

    # --------------------------------------------------------------
    # Final stochastic local refinement on the best vector found so far.
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n_best = len(overall_best_seq)
        seq_refined, _ = _local_refinement(
            overall_best_seq, n_best, deadline
        )
        try:
            final_score = evaluate_sequence(seq_refined.tolist())
        except Exception:
            final_score = overall_best_score
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq_refined.copy()
            print(f"*** After local refinement score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – should never trigger.
    if overall_best_seq is None:
        n_fallback = 1000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback uniform sequence used.")

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
