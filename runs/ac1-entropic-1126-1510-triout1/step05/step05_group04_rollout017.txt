def propose_candidate(seed=12345, budget_s=1000):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    best_score = float('inf')
    best_a = None

    def update_best(a):
        nonlocal best_score, best_a
        # compute conv and score
        n = a.shape[0]
        # compute conv via FFT
        N = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(A * A, n=N)[:2*n-1]
        max_c = conv.max()
        score = 2 * n * max_c / (a.sum()**2)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[improve] n={n} score={best_score:.6f}")
        return

    # loop over n values
    n_values = [500, 1000, 1500, 2000, 3000, 4000, 5000, 6000]
    for n in n_values:
        if time.time() > deadline:
            break
        S = math.sqrt(2.0 * n)
        # Precompute N for FFT
        N = 1 << ((2*n-1).bit_length())
        # initial seeds
        init_seqs = []

        # uniform
        a_uniform = np.full(n, S / n)
        init_seqs.append(a_uniform)

        # Dirichlet sparse
        dir_alpha = rng.uniform(0.2, 0.6)
        p_dir = rng.dirichlet([dir_alpha] * n)
        a_dir = p_dir * S
        init_seqs.append(a_dir)

        # exponential decreasing
        alpha = rng.uniform(0.0005, 0.003) * (n / 2000.0)
        a_exp = np.exp(-alpha * np.arange(n)) * S
        a_exp = _project_to_simplex(a_exp, S)
        init_seqs.append(a_exp)

        # arcsine-like shape
        i = np.arange(n, dtype=float)
        w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
        a_arc = w / w.sum() * S
        init_seqs.append(a_arc)

        # random Gaussian
        a_gauss = rng.normal(size=n) * 0.5
        a_gauss = np.exp(a_gauss)  # positive
        a_gauss = _project_to_simplex(a_gauss, S)
        init_seqs.append(a_gauss)

        # multiple restarts for each n:
        for restart_idx, a0 in enumerate(init_seqs):
            if time.time() > deadline:
                break
            a = a0.copy()
            # evaluate initial
            update_best(a)

            # Mirror descent iterations
            max_iters = 4000
            for it in range(max_iters):
                if time.time() > deadline:
                    break
                # compute convolution and max
                A = np.fft.rfft(a, n=N)
                conv = np.fft.irfft(A * A, n=N)[:2*n-1]
                max_c = conv.max()
                # compute subgradient
                mask = conv == max_c
                cnt = np.count_nonzero(mask)
                if cnt == 0:
                    break
                w_vec = mask.astype(float) / cnt
                A_rev = np.fft.rfft(a[::-1], n=N)
                W = np.fft.rfft(w_vec, n=N)
                conv_wr = np.fft.irfft(W * A_rev, n=N)[:2*n-1]
                g = 2.0 * conv_wr[n-1:n-1+n]

                # step size
                eta = 0.5 / math.sqrt(it + 1)

                # exponentiated update
                a_new = a * np.exp(-eta * g)
                # project to simplex
                a_new = _project_to_simplex(a_new, S)

                # occasionally evaluate best
                if (it % 20) == 0:
                    update_best(a_new)

                a = a_new

                # occasionally random jitter to escape stagnation
                if (it % 200) == 0:
                    # random swap of small mass
                    i1, i2 = rng.choice(n, size=2, replace=False)
                    delta = a[i1] * rng.random()*0.05
                    a[i1] -= delta
                    a[i2] += delta
                    a = _project_to_simplex(a, S)

    # final best refinement with block perturbations
    if best_a is not None and time.time() < deadline:
        n = len(best_a)
        S = best_a.sum()
        N = 1 << ((2*n-1).bit_length())
        for _ in range(2000):
            if time.time() > deadline:
                break
            # choose random block
            nb = rng.integers(2, 10)
            cuts = np.sort(rng.choice(n-1, nb-1, replace=False)+1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(len(blocks))
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            # compute score quickly
            A = np.fft.rfft(cand, n=N)
            conv = np.fft.irfft(A * A, n=N)[:2*n-1]
            score = 2 * n * conv.max() / (S**2)
            if score < best_score:
                best_score = score
                best_a = cand.copy()
                print(f"[block_improve] score={best_score:.6f}")

    # final monotone sorting heuristics
    if best_a is not None and time.time() < deadline:
        sorted_desc = np.sort(best_a)[::-1]
        sorted_desc = _project_to_simplex(sorted_desc, best_a.sum())
        update_best(sorted_desc)

        sorted_asc = np.sort(best_a)
        sorted_asc = _project_to_simplex(sorted_asc, best_a.sum())
        update_best(sorted_asc)

    print(f"[final] best_score={best_score:.6f}")
    return best_a.tolist()

"""
Propose a low‑score step function.

The algorithm uses two complementary optimisation phases:

1. **Additive sub‑gradient descent** – for a fixed length `n` we initialise
   several diverse sequences (uniform, exponential, U‑shaped, Dirichlet‑sparse,
   a reference if available, etc.).  At each iteration the exact max‑convolution
   is obtained via an FFT, the sub‑gradient of the hard max is built from the
   active indices, a (decreasing) step is taken and the result is projected back
   onto the ℓ¹‑simplex `Σa = √(2·n)`.  This phase performs many cheap iterations
   (few × 10⁴) and quickly drives the max‑convolution value down.

2. **Exponentiated‑gradient (mirror) refinement** – the sequence from the best
   sub‑gradient run is further refined with a KL‑mirror step
   `a ← a·exp(‑η·g)` using the same sub‑gradient `g`.  The mirror step keeps the
   variables strictly positive and often escapes shallow plateaus.  The
   step‑size `η` decays like `1/√t`.

Both phases share the same fast FFT based convolution and a lightweight
objective `score = 2·n·max(conv) / (Σa)²`.  The routine respects a user supplied
time budget (default 1000 s) and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z <= 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, N: int) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full convolution (length 2·len(a)‑1)."""
    n = a.shape[0]
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray, N: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution with respect to the vector `a`.
    The active set is taken as all indices within a tiny tolerance of the
    maximum to avoid missing near‑tied constraints.
    """
    n = a.shape[0]
    max_c = conv.max()
    # treat values within 1e-12 relative tolerance as active
    tol = max(1e-12, 1e-12 * max_c)
    mask = conv >= max_c - tol
    cnt = np.count_nonzero(mask)
    if cnt == 0:
        # should not happen – fall back to a single active index
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
    w = mask.astype(float) / cnt                     # uniform weight on active shifts
    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=N)
    W = np.fft.rfft(w, n=N)
    conv_wr = np.fft.irfft(W * A_rev, n=N)[: 2 * n - 1]
    g = 2.0 * conv_wr[n - 1 : n - 1 + n]             # slice of length n
    return g


# ----------------------------------------------------------------------
def _score_from_max(n: int, max_c: float, sum_a: float) -> float:
    """Exact value returned by evaluate_sequence() for a given max conv."""
    return 2.0 * n * max_c / (sum_a * sum_a)

    # --------------------------------------------------------------
    # Simple monotone sorting heuristic (sometimes helps)
    if best_a is not None and time.time() < deadline:
        for kind, arr in (("desc", np.sort(best_a)[::-1]),
                          ("asc", np.sort(best_a))):
            arr = _project_to_simplex(arr, best_a.sum())
            _maybe_update(arr, f"sorted {kind}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
