def _adam_optimize(a, max_iters, deadline):
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    epsilon = 1e-8
    lr0 = 0.5  # base learning rate
    tau = 0.1  # temperature

    best_seq = a.copy()
    best_score = _objective(a)

    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute convolution and b and max
        b = _fft_self_convolve(a)
        # compute softmax weights
        w = np.exp(b / tau - np.max(b) / tau)  # subtract max for stability
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(b) / b.size
        else:
            w = w / w_sum

        # gradient
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[len(a)-1 : len(a)-1+len(a)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        # step size schedule
        lr = lr0 / (np.sqrt(t))

        a = a - lr * (m_hat / (np.sqrt(v_hat) + epsilon))

        # project onto simplex s=1
        a = _project_onto_simplex(a, s=1.0)

        # compute exact score
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam iter={t} new best {cur_score:.6f}")

        # anneal temperature
        tau *= 0.99995  # gradually reduce
        if tau < 1e-8:
            tau = 1e-8

        # optionally early stop if best_score <= target
        if best_score <= 1.503:
            break

    return best_seq, best_score

def _simulated_annealing(a, best_seq, best_score, deadline):
    cur_score = best_score
    cur_a = a.copy()
    n = len(a)
    iter_count = 0
    max_iter = 2000000  # big but break early
    # set initial temperature
    T0 = 1e-3
    T = T0
    # for annealing schedule: T = T0 * (0.99**(iter/1000))
    while time.time() < deadline and iter_count < max_iter:
        iter_count += 1
        # gradually reduce temperature
        if iter_count % 1000 == 0:
            T *= 0.95
            if T < 1e-8:
                T = 1e-8

"""
Propose a low‑score non‑negative step function.

The search consists of three stages:

1️⃣  Analytic seeding – a collection of parametric families (arcsine,
    symmetric Beta(p,p), triangular, exponential) is evaluated for several
    lengths (up to ~20000).  The best candidate (lowest exact objective) is
    kept as a starting point.

2️⃣  Adam‑based simplex optimisation – we minimise the exact objective
    2·n·max(a∗a)/(∑a)².  The gradient of the (soft‑max) approximation of the
    max is computed analytically using FFTs, and the parameters are updated
    with the Adam optimiser.  After each step the vector is projected onto
    the probability simplex (∑a = 1, a ≥ 0).

3️⃣  Simulated‑annealing hill‑climbing – small random transfers of mass
    between two positions are accepted either if they improve the exact
    objective or with a Metropolis probability that slowly cools.  Periodic
    jitter helps to escape plateaux.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found as a plain list of floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _objective(seq: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a non‑negative vector.
    The function is scale‑invariant, therefore we keep the vector normalised
    to sum = 1 (the denominator becomes 1).
    """
    n = len(seq)
    if n == 0:
        return float("inf")
    # convolution (full, length 2·n‑1)
    conv = _fft_self_convolve(seq)
    max_conv = conv.max()
    # denominator = (∑a)² = 1 because we keep the vector normalised.
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # compute in log‑space to avoid overflow/underflow
    log_w = (p - 1.0) * (np.log(x) + np.log(1.0 - x))
    # shift to avoid extreme numbers
    log_w -= log_w.max()
    w = np.exp(log_w)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    # pad if rounding produced a length‑mismatch
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle, normalised to sum 1."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max over ``b`` with temperature ``tau`` (tau>0)."""
    # for numerical stability subtract the max before exponentiation
    shift = b.max() / tau
    w = np.exp(b / tau - shift)
    total = w.sum()
    if total == 0.0:
        return np.full_like(b, 1.0 / b.size)
    return w / total
