"""
Propose a non‑negative step function that minimizes `evaluate_sequence`.

The algorithm works in four stages:

1. **Length selection & warm‑start** – use a length at least twice the
   previously discovered solution (capped at 15000) and resize the old
   best sequence if it exists.
2. **Broad family search** – evaluate a modest collection of hand‑crafted
   probability vectors (uniform, arcsine, symmetric Beta, triangular,
   mixtures with spikes and random Dirichlet draws) on the chosen length.
3. **Multi‑restart projected Adam** – run several independent Adam
   optimisations on the simplex.  Each run uses a soft‑max surrogate of
   the ∞‑norm (with an exponential temperature schedule) and projects
   back onto the simplex via Euclidean projection.  The best vector over
   all restarts is retained.
4. **Fine hill‑climbing** – a cheap stochastic local search that moves a
   random fraction of the mass between two bins (plus occasional jitter)
   to polish the final result.

All stages respect the global time budget (default 1000 s).  The returned
list sums to 1, consists of non‑negative floats and is passed directly
to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·b_k) w.r.t. a (soft‑max surrogate of max(b)).
    If α is very large the soft‑max becomes a uniform weight over the
    numerically maximal entries.
    """
    n = a.shape[0]

    # --- weights for the soft‑max of the convolution vector ----------------
    if alpha > 2_000:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # --- convolution w * reversed(a) ------------------------------------
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]        # factor 2 from derivative of a*a
    return g


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) pdf evaluated on bin mid‑points, normalised."""
    a = max(a, 0.05)                # keep away from the singular case a→0
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0 or not np.isfinite(pdf_sum):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= pdf_sum
    return pdf


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle on [0,1] – increases linearly to the centre then
    decreases – normalised to sum = 1.
    """
    i = np.arange(n)
    mid = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - mid) / (mid + 1.0)
    vals = np.clip(vals, 0, None)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _resize_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a probability vector by linear interpolation on the underlying
    cumulative distribution function.  The output is again a probability
    vector (sum = 1)."""
    if old.shape[0] == new_n:
        return old.copy()
    old_x = (np.arange(old.shape[0]) + 0.5) / old.shape[0]
    new_x = (np.arange(new_n) + 0.5) / new_n
    new = np.interp(new_x, old_x, old, left=0.0, right=0.0)
    # renormalise (tiny numerical drift can happen)
    new = _project_onto_simplex(new, 1.0)
    return new


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest collection of hand‑crafted probability vectors."""
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) arcsine (Beta(½,½))
    candidates.append(_arcsine_distribution(n))

    # 3) symmetric triangular
    candidates.append(_triangular_distribution(n))

    # 4) symmetric Beta(a,a) on a grid
    for a in np.linspace(0.10, 1.20, 12):
        candidates.append(_beta_distribution(n, a))

    # 5) mixtures of arcsine & uniform
    arcsine = _arcsine_distribution(n)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for w in np.linspace(0.0, 0.5, 7):
        mix = (1.0 - w) * arcsine + w * uniform
        candidates.append(mix)

    # 6) arcsine with small spikes at the ends
    for spike_frac in (0.05, 0.10, 0.15):
        mix = arcsine.copy()
        mix[0] += spike_frac / 2
        mix[-1] += spike_frac / 2
        mix = _project_onto_simplex(mix, 1.0)
        candidates.append(mix)

    # 7) previous best (if supplied)
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())

    # 8) a handful of random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(5):
        rand = rng.dirichlet(np.ones(n))
        candidates.append(rand)

    # Ensure every candidate lives on the simplex
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex.
    Returns (best_vector, best_raw_score = 2·n·max(conv)).
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.07
    alpha_max = 5.0e5

    # FFT size needed for the gradient step (w length = 2n‑1, a_rev length = n)
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] n={n} it={it} score={best_score:.6f}")

        # gradient of the smooth max
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto simplex
        a = _project_onto_simplex(a, 1.0)

        # temperature increase
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search with mass transfers and occasional jitter.
    Returns (best_vector, best_raw_score).
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-14:
            continue

        # move a random fraction (up to 30 %) of mass from i to j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter to escape flat plateaus
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point required by the grader.
    Returns a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – we want to finish a little before the hard limit
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1) Choose a working length.
    # Prefer twice the length of a previously found solution (if any),
    # but never exceed 15000 and never go below 2000.
    base_n = 8000
    try:
        base_n = int(len(height_sequence_1))          # type: ignore[name-defined]
    except Exception:
        pass
    n = max(2000, min(15000, base_n * 2))

    # ------------------------------------------------------------------
    # 2) Warm‑start: resize the previous solution if it exists.
    prev_seq = None
    try:
        old = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev_seq = _resize_distribution(old, n)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) Fast parametric sweep over a handful of families.
    shape_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.15)
    candidates = _candidate_shapes(n, prev_seq)
    best_seq = None
    best_score = float("inf")

    for cand in candidates:
        if time.time() >= shape_deadline:
            break
        scr = evaluate_sequence(cand.tolist())
        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
            print(f"[shape] New best score {best_score:.6f}")

    # Fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 4) Multi‑restart Adam optimisation.
    # Allocate the bulk of the remaining time to this stage.
    adam_time = (deadline - time.time()) * 0.70
    adam_deadline = time.time() + adam_time
    global_best_seq = best_seq.copy()
    global_best_score = best_score

    restart = 0
    while time.time() < adam_deadline - 5.0:
        restart += 1
        # give each restart a modest slice of the remaining budget
        run_deadline = time.time() + min(30.0, adam_deadline - time.time() - 5.0)

        if restart == 1:
            init = global_best_seq.copy()
        else:
            # random perturbation of the current global best
            rng = np.random.default_rng(seed + restart)
            perturb = rng.normal(scale=0.02, size=n)
            init = global_best_seq + perturb
            init = _project_onto_simplex(init, 1.0)

        seq, raw = _adam_optimize(init, n, run_deadline)

        # raw is exactly 2·n·max(conv) because sum(seq)=1
        if raw < global_best_score:
            global_best_score = raw
            global_best_seq = seq.copy()
            print(f"[restart {restart}] Improved score {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final hill‑climbing polishing.
    hill_deadline = deadline
    seq_final, final_raw = _hill_climb(global_best_seq, n, hill_deadline)

    # Convert raw score back to the format of evaluate_sequence (they are
    # identical when the sum of the vector is 1).
    final_score = final_raw
    print(f"Final best score: {final_score:.6f} (length = {len(seq_final)})")

    # Return a plain Python list – the grader will feed it to evaluate_sequence.
    return seq_final.tolist()
