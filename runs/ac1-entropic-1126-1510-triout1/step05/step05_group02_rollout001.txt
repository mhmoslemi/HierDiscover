def _run_adam(y, n, deadline, best_sofar):
   # initialize Adam
   m = np.zeros_like(y); v = np.zeros_like(y)
   beta1=0.9; beta2=0.999; eps=1e-8
   lr=0.1  # adapt scaling maybe
   tau = 100.0  # smoothing temperature.
   best_local_seq=None; best_local_score=inf
   t=0
   while time.time() < deadline:
        t+=1
        a = _softmax(y)   # shape (n,)
        b = _fft_self_convolve_full(a)   # length 2n-1
        # compute raw max
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = a.copy()
                # also if better than global we could print
                if cur_score < best_sofar:
                        print(f"[{time.time():.1f}] New global best n={n} score={cur_score:.6f}")
        # compute softened weights
        # compute w = softmax(tau * b)
        # to avoid overflow, shift by max
        max_tb = np.max(tau * b)
        exp_tb = np.exp(tau * b - max_tb)
        w = exp_tb / exp_tb.sum()
        # compute gradient wrt a
        a_rev = a[::-1]
        grad_a = 2.0 * _fft_convolve(w, a_rev)[n-1:n-1+n]
        # grad wrt y
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)
        # adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
        # optionally reduce learning rate over time: lr = lr0 / sqrt(t)
        # but we can keep constant.
        # break if no improvement for many iterations?
        if t % 5000 == 0 and time.time() > deadline - 5:
                break
   return best_local_seq, best_local_score
```

"""
Propose a low‑score step‑function.

Algorithm
---------
* Search a few lengths (2000‑5000).  For each length generate several
  diverse initial probability vectors (uniform, arcsine, symmetric beta,
  random mixtures of two beta PDFs and – if available – the previous best
  construction interpolated to the current length).

* Parameterise a vector `a` on the simplex by a soft‑max of an unrestricted
  vector `y`:   a = softmax(y).  This guarantees `a_i ≥ 0` and Σa_i = 1
  without an explicit projection.

* Optimise `y` with Adam using the true sub‑gradient of the
  ℓ∞‑self‑convolution objective.
  For a given `a` we compute the full self‑convolution `b = a*a`
  (FFT, O(n log n)).  Let `M = max(b)`.  The active set
  `I = {k : b_k ≥ M - tol}` (tol = 1e‑12·M) yields the sub‑gradient
  `w_k = 1/|I|` for k∈I, else 0.  The gradient wrt `a` is

        grad_a = 2 * conv(w, a_rev)[n‑1 : n‑1+n] ,

  where `a_rev` is `a` reversed.  Via the soft‑max Jacobian we obtain the
  gradient wrt `y`:

        grad_y = a * (grad_a - ⟨a, grad_a⟩) .

* Run Adam for as many iterations as the remaining time budget permits,
  keeping the best raw score (2·n·max(b)) encountered.

* After all lengths are processed evaluate the best sequence with the
  provided `evaluate_sequence` and return it as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution on n points, normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow/underflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (returns length |a|+|b|‑1)."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    shift = np.max(y)
    exps = np.exp(y - shift)
    return exps / exps.sum()


# ----------------------------------------------------------------------
def _active_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Return the sub‑gradient of max_k (a*a)_k w.r.t. a,
    using the uniform weight on the active set.
    """
    n = a.shape[0]
    b = _fft_self_convolve_full(a)
    cur_max = b.max()
    # tolerance to decide which indices are active
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    # gradient wrt a
    a_rev = a[::-1]
    grad_a = 2.0 * _fft_convolve(w, a_rev)[n - 1 : n - 1 + n]
    return grad_a, cur_max


# ----------------------------------------------------------------------
def _adam_optimize_from_y(y0: np.ndarray, n: int,
                          deadline: float, global_best: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max parameters `y`.
    Returns the best probability vector seen and its raw score.
    """
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2

    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        a = _softmax(y)                     # current probability vector
        grad_a, cur_max = _active_subgradient(a)

        # raw objective (what evaluate_sequence uses)
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            if cur_score < global_best:
                print(f"[{time.time():.1f}] New global best n={n} it={it} score={cur_score:.6f}")

        # gradient w.r.t y via Jacobian of softmax
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional stochastic jitter to escape plateaus
        if it % 4000 == 0 and np.random.rand() < 0.12:
            y += np.random.randn(n) * 5e-6

    return best_seq, best_score
