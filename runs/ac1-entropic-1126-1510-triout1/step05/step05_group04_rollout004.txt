"""
Propose a low‑score step function.

Algorithm
---------
1. Choose a problem size (n ≈ 12000) and fix the ℓ¹‑norm
   S = sqrt(2·n).  The evaluation score then equals
   max_k (a*a)_k, i.e. the max auto‑convolution.
2. Perform many restarts of a smooth‑max optimisation.
   The true max is approximated by
       f_μ(a) = max(conv) + μ·log Σ exp((conv‑max)/μ)
   with a decreasing temperature μ.  The optimisation is carried out on
   the log‑softmax parameters y (so that a = S·softmax(y) stays on the
   simplex).  Gradients are evaluated using fast FFT‑based convolutions,
   making each iteration cheap even for large n.
3. Refine the best candidate with hard‑max sub‑gradient descent
   (μ = 0) and an adaptive back‑tracking line search.
4. A short simulated‑annealing phase performs random mass transfers
   that sometimes accept uphill moves, helping to escape shallow minima.
5. Finally, try monotone‑sorted variants of the best vector.

All steps respect a global time budget (default 1000 s) and keep the
sequence non‑negative with the required ℓ¹‑norm.  The routine returns
the best height list found.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution via FFT (real inputs, returns real output)."""
    n = len(x) + len(y) - 1
    L = _next_pow2(n)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
def _fft_correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation w ⋆ a (i.e. convolution of w with reversed a) via FFT.
    Result length = len(w)+len(a)-1.
    """
    n = len(w) + len(a) - 1
    L = _next_pow2(n)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(W * np.conj(A), n=L)[:n]
    return corr


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)_k using FFT (fast)."""
    conv = _fft_convolve(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, mu: float,
                         n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Compute smooth‑max surrogate value and gradient w.r.t. logits y.

    Parameters
    ----------
    y : (n,) ndarray
        Unconstrained parameters; soft‑max yields the step heights.
    mu : float (>0)
        Smoothing temperature; μ=0 corresponds to the hard max.
    n : int
        Length of the sequence.
    S : float
        Target ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max objective value.
    grad_y : (n,) ndarray
        Gradient ∂f/∂y.
    """
    # probabilities on simplex and actual heights
    p = _softmax(y)
    a = S * p

    # auto‑convolution a*a
    conv = _fft_convolve(a, a)               # length 2n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        # clip to avoid overflow in exp
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max: uniform weight on all active indices
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a:
    #   g_a = 2·(w ⋆ a)_{centered}
    corr = _fft_correlation(w, a)                 # length 3n‑2
    g_a = 2.0 * corr[n - 1:n - 1 + n]             # slice of length n

    # chain rule to logits y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                        # scalar
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class SmoothMaxObjective:
    """Callable for scipy.optimize.minimize that returns (value, grad)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------

    Search for a non‑negative step‑function with a low evaluate_sequence()
    score.  The routine respects a time budget (seconds) and returns the
    best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # problem size ----------------------------------------------------
    # Aim for a fairly large n while staying within the time budget.
    n = 12000
    try:
        ref_len = len(height_sequence_1)  # type: ignore[name-defined]
        n = max(n, ref_len * 2)
    except Exception:
        pass
    n = min(n, 16000)                     # hard upper bound for safety
    S = math.sqrt(2.0 * n)                # target ℓ¹‑norm

    best_a = np.zeros(n, dtype=np.float64)
    best_score = float("inf")

    # ------------------------------------------------------------------
    def _maybe_update(a_vec: np.ndarray) -> None:
        """Evaluate a candidate (fast) and keep it if it improves the incumbent."""
        nonlocal best_a, best_score
        sc = _max_convolution(a_vec)
        if sc + 1e-12 < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # generate diverse initial logits (y)
    init_ys = []

    # 0 – uniform
    init_ys.append(np.zeros(n, dtype=np.float64))

    # 1 – Dirichlet (moderately sparse)
    p = rng.dirichlet([0.5] * n)
    init_ys.append(np.log(p + 1e-12))

    # 2 – Dirichlet (spiky)
    p = rng.dirichlet([0.15] * n)
    init_ys.append(np.log(p + 1e-12))

    # 3 – random Gaussian
    init_ys.append(rng.standard_normal(n) * 0.5)

    # 4 – decreasing exponential
    slope = rng.uniform(0.001, 0.008) * (n / 5000.0)
    init_ys.append(-slope * np.arange(n, dtype=np.float64))

    # 5 – increasing exponential (large values on the right)
    init_ys.append(-slope * np.arange(n - 1, -1, -1, dtype=np.float64))

    # 6 – random block‑constant shape
    nb = rng.integers(4, 12)
    cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
    blocks = np.split(np.arange(n), cuts)
    vals = rng.random(nb)
    a_block = np.empty(n, dtype=np.float64)
    for blk, v in zip(blocks, vals):
        a_block[blk] = v
    p = a_block / a_block.sum()
    init_ys.append(np.log(p + 1e-12))

    # 7 – reference (if available)
    try:
        a_ref = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, S)
        p = a_ref / S
        init_ys.append(np.log(p + 1e-12))
    except Exception:
        pass

    # ------------------------------------------------------------------
    mu_factor = 0.5          # faster cooling
    mu_min = 1e-8
    maxiter_per_mu = 300
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}]")
        y = y0.copy()

        # initialise temperature from current max‑convolution
        a_cur = S * _softmax(y)
        mu = max(_max_convolution(a_cur), 1.0)

        while mu > mu_min and time.time() < deadline:
            obj = SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False}
                )
            except Exception as exc:
                print("[error] L‑BFGS failed:", exc)
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    # ------------------------------------------------------------------
    # Hard‑max sub‑gradient descent ------------------------------------
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        step0 = S * 0.4 / n
        it = 0
        max_local_iters = 6000
        while time.time() < deadline and it < max_local_iters:
            conv = _fft_convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            w = np.zeros_like(conv)
            w[mask] = 1.0 / cnt
            corr = _fft_correlation(w, a)          # length 3n‑2
            g = 2.0 * corr[n - 1:n - 1 + n]

            step = step0 / (1.0 + 1e-6 * it)
            improved = False
            for _ in range(15):
                a_new = a - step * g
                a_new = _project_to_simplex(a_new, S)
                sc = _max_convolution(a_new)
                if sc + 1e-12 < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[subgrad] iter {it:04d} step {step:.2e} => {best_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # random jitter to possibly leave a flat region
                i, j = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a[i] * 0.05
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, S)
            it += 1

    # ------------------------------------------------------------------
    # Simulated annealing (lightweight polishing) --------------------
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        curr_score = best_score
        T0 = 0.1
        Tmin = 1e-5
        total_iters = 0
        max_iters = 20000
        while time.time() < deadline and total_iters < max_iters:
            T = max(T0 * (1.0 - total_iters / max_iters), Tmin)
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.random() * a[i] * 0.1
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            if a_new[i] < 0:
                continue  # reject infeasible move
            new_score = _max_convolution(a_new)
            dscore = new_score - curr_score
            if dscore < -1e-12 or rng.random() < math.exp(-dscore / T):
                a = a_new
                curr_score = new_score
                if new_score + 1e-12 < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[anneal] iter {total_iters:05d} T {T:.2e} new best {best_score:.6f}")
            total_iters += 1

    # ------------------------------------------------------------------
    # Monotone‑sorting heuristics --------------------------------------
    if time.time() < deadline and best_score < float("inf"):
        desc = np.sort(best_a)[::-1]
        desc = _project_to_simplex(desc, S)
        _maybe_update(desc)

        asc = np.sort(best_a)
        asc = _project_to_simplex(asc, S)
        _maybe_update(asc)

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
