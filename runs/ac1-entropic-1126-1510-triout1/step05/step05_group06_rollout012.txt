def _adam_optimize(a0, n, deadline):
   a = a0.copy()
   best = a.copy()
   best_score = compute_score(a)
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   t = 0
   size_fft = 1 << ((2*n - 1).bit_length())
   while time.time() < deadline:
        t += 1
        conv = fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # compute gradient of smooth max:
        # gradient of max surro w.r.t a is same as previous _gradient_softmax
        # we can gradually increase temperature: alpha = min(alpha_max, alpha0 * (1+0.01*t))
        # For simplicity, use large alpha (say 150) to approximate max.
        # We'll compute w = softmax(alpha * conv) vector.
        # Then gradient as w gradient: grad = 2 * (conv of w reversed with a reversed). Use same code as before.
        # using FFT size maybe size_fft.
        alpha = 200.0  # high
        # compute w:
        shift = conv - conv.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(w)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        # gradient in conv domain: g = 2 * (conv of w reversed with a reversed)
        # We can compute via FFT as earlier:
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=size_fft)
        a_rev_fft = np.fft.rfft(a_rev, n=size_fft)
        conv_grad = np.fft.irfft(w_fft * a_rev_fft, n=size_fft)
        start = n - 1
        grad = 2.0 * conv_grad[start:start+n]
        # Adam update:
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step  # gradient descent (since we want to minimize)
        # project onto simplex and keep non-negative
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / sum_a
   return best, best_score

"""
A fast two‑stage optimizer for the “step‑function” challenge.

Stage 1 – shape search:
    * For a given length *n* we explore the 1‑parameter family
      a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ (the classic „U‑shape“) and a tiny mixture
      with the uniform distribution.
    * A fine grid over p∈[0.4,1.2] and w∈[0,0.3] (weight of the uniform part)
      is scanned using an FFT‑based convolution to compute the cheap
      score 2·n·max(conv).  The best (p,w) is kept.

Stage 2 – local refinement:
    * Starting from the best shapes we run a mirror‑descent / exponentiated‑gradient
      loop (multiplicative updates a←a·exp(‑η·∇)) with a soft‑max surrogate
      for the ∞‑norm that gradually hardens (α grows until ≈300).
    * The iterates are projected onto the probability simplex after each step.
    * Finally a cheap stochastic hill‑climber moves a small amount of mass
      between two positions whenever the true evaluation improves.

The outer driver tries several lengths (up to 20 000) and keeps the best
sequence found before the time budget expires.  The code contains no
closures or lambda expressions and works with only NumPy and SciPy
(if available).  All numbers are non‑negative, sum to one and are returned
as a plain Python list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full (length 2·n‑1) convolution of a with itself via a real FFT.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two ≥ 2·n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """
    Internal evaluation score: 2·n·max(conv(a,a)).
    The input is assumed to be non‑negative and sum(a)=1,
    which matches the external ``evaluate_sequence``.
    """
    n = a.shape[0]
    max_b = _fft_self_convolve(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the index *k* where the maximum occurs.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the smooth soft‑max surrogate
        (1/α)·log Σ exp(α·conv)
    w.r.t. *a*.
    """
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:                     # fall back to pure max
        w = np.zeros_like(w)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_grad = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv_grad[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _beta_power_shape(n: int, p: float) -> np.ndarray:
    """
    Symmetric U‑shaped distribution a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ .
    Returns a vector of length n that already sums to 1.
    """
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _make_candidate(n: int, p: float, w: float) -> np.ndarray:
    """
    Mixture of the power‑law shape (parameter *p*) with the uniform law.
    *w*∈[0,1] is the weight of the uniform part (0 → pure power‑law).
    The result is projected onto the simplex and thus sums to 1.
    """
    shape = _beta_power_shape(n, p)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    cand = (1.0 - w) * shape + w * uniform
    return _project_onto_simplex(cand, s=1.0)


# ----------------------------------------------------------------------
def _grid_search_shape(n: int):
    """
    Exhaustive scan of a fine grid of (p,w) for a given length *n*.
    Returns the best candidate vector, its score, and the parameters (p,w).
    """
    best_a = None
    best_score = np.inf
    best_p = None
    best_w = None

    # parameters – chosen to give a dense exploration around the known good region
    p_vals = np.arange(0.40, 1.21, 0.02)      # 0.40 … 1.20
    w_vals = np.arange(0.0, 0.31, 0.02)       # 0.00 … 0.30

    for p in p_vals:
        shape = _beta_power_shape(n, p)
        for w in w_vals:
            cand = (1.0 - w) * shape + w * (1.0 / n)
            cand = _project_onto_simplex(cand, s=1.0)
            sc = _score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand
                best_p = p
                best_w = w

    return best_a, best_score, best_p, best_w


# ----------------------------------------------------------------------
def _refine_shape_single(n: int, p0: float, w0: float, deadline: float):
    """
    Refine a pair (p,w) using L‑BFGS‑B on the true score.
    Returns (p_opt,w_opt,a_opt,score_opt).
    """
    try:
        import scipy.optimize as opt
    except Exception:                     # SciPy not available – give up
        a0 = _make_candidate(n, p0, w0)
        return p0, w0, a0, _score(a0)

    def _obj(params):
        p, w = params
        if p <= 0 or w < 0 or w > 1:
            return np.inf
        a = _make_candidate(n, p, w)
        return _score(a)

    init = np.array([p0, w0], dtype=np.float64)
    bounds = [(0.10, 2.0), (0.0, 0.5)]       # wide but sensible
    # limit the number of function evaluations to keep within the time slice
    options = {"maxfun": 180, "ftol": 1e-12}

    res = opt.minimize(_obj, init, method="L-BFGS-B",
                       bounds=bounds, options=options)
    p_opt, w_opt = res.x
    a_opt = _make_candidate(n, p_opt, w_opt)
    s_opt = _score(a_opt)
    return p_opt, w_opt, a_opt, s_opt


# ----------------------------------------------------------------------
def _refine_shape_multiple(n: int, p0: float, w0: float, deadline: float):
    """
    Run a handful of L‑BFGS‑B refinements from different starting points.
    Returns the best vector found together with its (p,w) and score.
    """
    start_points = [
        (p0, w0),
        (0.55, 0.0),
        (0.70, 0.0),
        (0.80, 0.0),
        (0.60, 0.10),
        (0.90, 0.05),
    ]

    best_a = None
    best_score = np.inf
    best_params = None

    for p_start, w_start in start_points:
        if time.time() >= deadline:
            break
        p_opt, w_opt, a_opt, s_opt = _refine_shape_single(n, p_start, w_start, deadline)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt
            best_params = (p_opt, w_opt)

    return best_a, best_score, best_params


# ----------------------------------------------------------------------
def _exponential_gradient_optimize(a0: np.ndarray,
                                   n: int,
                                   deadline: float):
    """
    Mirror‑descent / exponentiated‑gradient refinement.
    Returns the best vector seen and its internal (2·n·max) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _score(a)

    # hyper‑parameters (tuned for a faster convergence)
    eta0 = 1.0               # larger initial step size
    alpha = 5.0              # start with a moderate temperature
    alpha_mul = 1.03
    alpha_max = 300.0
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()

        # gradient (smooth → hard)
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        eta = eta0 / np.sqrt(it)          # diminishing step size
        a = a * np.exp(-eta * grad)       # multiplicative update
        a = np.maximum(a, 0.0)            # stay non‑negative

        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # occasional tiny jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float):
    """
    Simple stochastic hill‑climbing: move a small fraction of mass
    from one index to another whenever this improves the true score.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        cur_score = _score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative list) that minimises
    ``evaluate_sequence``.  The routine respects the time budget, tries several
    lengths, refines a few promising shape families, then performs a
    gradient‑based polishing and a final stochastic local search.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a handful of lengths (moderate to large) – 5 k–20 k works well.
    lengths = [5000, 8000, 12000, 16000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503                     # stop as soon as we beat the target

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        # allocate a slice of remaining time to this length
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(5.0, remaining * 0.35)

        # ------------------------------------------------------------------
        # 1. Initial candidates (previous solution, random, uniform)
        init_candidates = []

        # a) previous solution if lengths match
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.size == n:
                prev = np.maximum(prev, 0.0)
                prev = prev / prev.sum()
                init_candidates.append(prev)
        except Exception:
            pass

        # b) pure uniform
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # c) random start
        rnd = np.random.rand(n)
        rnd = np.maximum(rnd, 0.0)
        rnd = rnd / rnd.sum()
        init_candidates.append(rnd)

        # ------------------------------------------------------------------
        # 2. Coarse shape search – fine grid over (p,w)
        shape_a, shape_score, p_best, w_best = _grid_search_shape(n)
        init_candidates.append(shape_a)

        # keep only the most promising few (based on cheap internal score)
        init_candidates = sorted(init_candidates,
                                 key=_score)[:4]

        # ------------------------------------------------------------------
        # 3. Refine the shape parameters using a small LP‑free optimisation
        refined_a, refined_score, refined_params = _refine_shape_multiple(
            n, p_best, w_best, slice_deadline
        )
        init_candidates.append(refined_a)

        # keep the top 3 (still cheap internal score)
        init_candidates = sorted(init_candidates,
                                 key=_score)[:3]

        # ------------------------------------------------------------------
        # 4. Exponentiated‑gradient polishing for each starter
        for a0 in init_candidates:
            if time.time() >= slice_deadline:
                break
            a_opt, internal_score = _exponential_gradient_optimize(a0, n, slice_deadline)

            # evaluate with the black‑box validator (exact same as internal score)
            try:
                final_score = evaluate_sequence(a_opt.tolist())
            except Exception:
                final_score = internal_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = a_opt.copy()
                print(f"*** New global best → len={n} score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break
        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Final stochastic refinement on the overall best vector (if any)
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n_best = len(overall_best_seq)
        refined_seq, _ = _local_refinement(overall_best_seq, n_best, deadline)
        try:
            final_score = evaluate_sequence(refined_seq.tolist())
        except Exception:
            final_score = overall_best_score
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = refined_seq.copy()
            print(f"*** After final local refinement score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – should never be needed.
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Safety fallback used.")

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
```

This implementation follows the two‑stage strategy described above, explores a finer family of “U‑shaped’’ distributions, refines the parameters with a gradient‑free L‑BFGS‑B optimiser, then polishing with an exponentiated‑gradient loop and a final random‑move hill‑climber.  It stays within the 1000‑second budget and returns the best sequence found.
