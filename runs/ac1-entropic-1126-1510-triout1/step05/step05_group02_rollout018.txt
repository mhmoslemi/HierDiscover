def build_conv_matrix(a):
    n = len(a)
    P = 2*n - 1
    A = np.zeros((P, n))
    for i in range(P):
        k_min = max(0, i - n + 1)
        k_max = min(i, n - 1)
        ks = np.arange(k_min, k_max+1)
        A[i, ks] = a[i - ks]
    return A
```

"""
Propose a low‑score step function.

Strategy
--------
1. **Wide search** over a rich family of Beta‑distributions.
   * Symmetric Beta(α,α) with α ranging from 1e‑3 to 5.
   * Skewed Beta(α,β) (α≪β and α≫β) – strong mass at one end.
   * Random mixtures of two Beta components (including opposite skews).
   * Random Dirichlet draws (low concentration).
   All candidates are projected onto the simplex ( Σa = 1 ).
2. **Select the best raw candidate** (the objective 2·n·max(a*a) is
   scale‑invariant, so we can keep Σa = 1 throughout).
3. **Refinement** with a projected Adam optimizer on the exact
   ℓ∞ convolution objective.  The max‑convolution is smoothed using a
   soft‑max (temperature increases during optimisation) to obtain a
   useful gradient.  The iterate is always projected back onto the
   probability simplex.
4. Return the final vector as a plain Python list.

The whole routine respects the provided time budget (default 1000 s)
and prints intermediate progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalized Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf for stability
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _dirichlet_distribution(n: int, concentration: float) -> np.ndarray:
    """Draw a Dirichlet vector of length n with identical concentration."""
    draws = np.random.gamma(concentration, 1.0, size=n)
    draws_sum = draws.sum()
    if draws_sum == 0.0:
        draws[:] = 1.0 / n
    else:
        draws /= draws_sum
    return draws


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a * a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _fft_self_convolve_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Search for a low‑score non‑negative height sequence."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # ------------------------------------------------------------------
    # 1. Base candidate (if a previous solution is available)
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # 2. Lengths to explore (moderate sizes keep FFT cheap)
    lengths = [1200, 2400, 4800, 9600]   # up to ~10k elements
    best_seq = None
    best_score = float("inf")
    best_len = None

    # ------------------------------------------------------------------
    # Helper: raw objective for a vector with Σ=1  (scale‑invariant)
    def _raw_score(vec: np.ndarray) -> float:
        n = vec.shape[0]
        return 2.0 * n * _fft_self_convolve_max(vec)

    # ------------------------------------------------------------------
    # 3. Exhaustive sweep over families
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"[{time.time():.1f}] Exploring length {n}")

        # a) previous solution interpolated
        if base_seq is not None:
            cand = _interpolate_to_length(base_seq, n)
            cand = _project_onto_simplex(cand)
            sc = _raw_score(cand)
            if sc < best_score:
                best_score, best_seq, best_len = sc, cand.copy(), n
                print(f"   init=prev  score={best_score:.6f}")

        # b) uniform baseline
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        uniform = _project_onto_simplex(uniform)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_score, best_seq, best_len = sc, uniform.copy(), n
            print(f"   init=uniform score={best_score:.6f}")

        # c) classic arcsine (Beta(0.5,0.5))
        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine)
        if sc < best_score:
            best_score, best_seq, best_len = sc, arcsine.copy(), n
            print(f"   init=arcsine score={best_score:.6f}")

        # d) symmetric Beta(α,α) – very fine grid, including extreme α
        sym_alphas = np.concatenate((
            np.logspace(-3, -1, 12, endpoint=False),    # 0.001 .. 0.1
            np.linspace(0.12, 0.6, 12, endpoint=False),
            np.linspace(0.6, 2.0, 15),
            np.linspace(2.0, 5.0, 8)
        ))
        for a in sym_alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a, a)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   symBeta a={a:.4f} score={best_score:.6f}")

        # e) skewed Beta families (α small, β large) and vice‑versa
        skew_params = [
            (0.05, 5.0), (0.05, 2.0), (0.05, 1.0),
            (0.2, 5.0), (0.2, 2.0),
            (5.0, 0.05), (2.0, 0.05), (1.0, 0.05),
            (2.0, 0.2), (5.0, 0.2)
        ]
        for a, b in skew_params:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a, b)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   skewBeta ({a:.2f},{b:.2f}) score={best_score:.6f}")
            # also mirror the distribution (swap ends)
            vec_rev = vec[::-1]
            sc = _raw_score(vec_rev)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec_rev.copy(), n
                print(f"   skewBeta rev ({a:.2f},{b:.2f}) score={best_score:.6f}")

        # f) random Dirichlet draws (low concentration)
        dirichlet_trials = 300
        for _ in range(dirichlet_trials):
            if time.time() >= deadline:
                break
            conc = np.random.rand() * 0.9 + 0.1   # ∈[0.1,1.0]
            vec = _dirichlet_distribution(n, conc)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   dirichlet conc={conc:.3f} score={best_score:.6f}")

        # g) random mixtures of two Beta components (including opposite skews)
        mixture_trials = 1500
        for _ in range(mixture_trials):
            if time.time() >= deadline:
                break
            a1, b1 = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
            a2, b2 = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
            lam = np.random.rand()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mix λ={lam:.2f} ({a1:.2f},{b1:.2f})+({a2:.2f},{b2:.2f}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – simple uniform of moderate size
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_len = n
        best_score = _raw_score(best_seq)
        print("Fallback uniform used.")

    print(f"[{time.time():.1f}] Best raw score after search: {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # 4. Refinement with projected Adam on the exact ℓ∞ self‑convolution.
    #    Remaining time budget is used here.
    sub_deadline = deadline
    a = best_seq.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t_iter = 0

    # temperature for soft‑max (starts low → smoother gradient)
    temp = 2.0
    temp_inc = 0.001  # increase each iteration

    best_a = a.copy()
    best_val = 2.0 * n * _fft_self_convolve_max(a)

    while time.time() < sub_deadline:
        t_iter += 1
        # compute full convolution once
        conv_full = _fft_self_convolve_full(a)

        # soft‑max weights to approximate sub‑gradient of the max
        scaled = conv_full / temp
        max_scaled = scaled.max()
        exp_shifted = np.exp(scaled - max_scaled)   # shift for stability
        w = exp_shifted / exp_shifted.sum()          # sums to 1

        # gradient of the smoothed objective
        a_rev = a[::-1]
        grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step_size = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step_size * m_hat
        a = _project_onto_simplex(a)

        # occasionally add tiny jitter to help escape plateaus
        if t_iter % 800 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a)

        # evaluate exact objective
        cur_val = 2.0 * n * _fft_self_convolve_max(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam it={t_iter}  score={best_val:.6f}")

        # increase temperature slowly – makes gradient less noisy over time
        temp = min(200.0, temp + temp_inc)

        # safety: stop after reasonable number of iterations (e.g. 3 M)
        if t_iter >= 3_000_000:
            break

    final_seq = best_a
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq.tolist()

def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals
