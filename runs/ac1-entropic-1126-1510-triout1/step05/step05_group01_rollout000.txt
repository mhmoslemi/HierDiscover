    # Determine base lengths

        base_n = 8000

    lengths = sorted(set(lengths))

    best_seq = None
    best_score = float('inf')
    iteration_counter = 0

    # For each n:
    for n in lengths:
        # generate candidate pool
        candidates = []
        # Beta grid
        al_vals = np.linspace(0.1, 4.0, 13)  # 13 values
        be_vals = np.linspace(0.1, 4.0, 13)
        for al in al_vals:
            for be in be_vals:
                a = _beta_distribution(n, al, be)
                candidates.append(a)
        # Skewed combos
        for al, be in [(0.2,0.8),(0.8,0.2), (0.4,2.0), (2.0,0.4)]:
            a = _beta_distribution(n, al, be)
            candidates.append(a)
        # random Dirichlet with concentration 0.5 or 1/n
        for conc in [0.01, 0.1, 0.5, 1.0]:
            a = np.random.dirichlet(np.full(n, conc))
            candidates.append(a)
        # mixture of arcsine and uniform:
        a1 = _beta_distribution(n, 0.5,0.5)
        a2 = np.full(n, 1.0/n)
        w = np.random.rand()
        a = w * a1 + (1-w) * a2
        candidates.append(a)
        # Possibly previous sequence scaled to n
        if 'height_sequence_1' in globals():
            try:
                prev = np.asarray(height_sequence_1, dtype=np.float64)
                if prev.size != n:
                    x_old = np.linspace(0, 1, prev.size)
                    x_new = np.linspace(0, 1, n)
                    a = np.interp(x_new, x_old, prev)
                else:
                    a = prev.copy()
                candidates.append(a)
            except Exception:
                pass

        # Evaluate cheap scores and keep top.
        scored = [(cheap_score(a), a) for a in candidates]
        scored.sort(key=lambda t: t[0])
        starters = [a.copy() for _, a in scored[:5]]
        # gradient refine each
        for idx, a0 in enumerate(starters):
            # allocate per starter time proportionally maybe.
            start_time = time.time()
            a = a0.copy()
            # ensure simplex
            a = _project_onto_simplex(a)
            m = np.zeros_like(a)  # first moment
            v = np.zeros_like(a)  # second moment
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            t = 0
            best_local_seq = a.copy()

"""
Propose a step‑function with a low evaluate_sequence score.

Algorithm
---------
1. **Length selection** – use the length of the previously discovered
   construction (if any) and explore a small neighbourhood around it.

2. **Broad candidate generation** – for each chosen length we create a
   large pool of cheap candidates:
   * a dense grid of Beta(α,β) densities (α,β in [0.05,3] ),
   * random Dirichlet/Exponential draws with several concentration levels,
   * a few hand‑crafted mixtures (arcsine, uniform, skewed Beta),
   * the previous construction interpolated to the current length.
   The cheap score `2·n·max(conv(a,a))` (sum(a)=1) is computed by an FFT
   and the best *k* candidates are kept.

3. **Refinement** – each kept candidate is refined with a projected
   Adam optimiser on the exact sub‑gradient of the max‑convolution
   (the gradient is obtained by convolving the current vector with a
   uniform weight on all active convolution indices).  After each
   step the vector is projected back onto the probability simplex and
   symmetrised (a←(a+a[::-1])/2) – the optimum is believed to be symmetric.

4. **Local random walk** – from the best refined vector we perform a
   short random walk that moves a tiny amount of mass between two
   coordinates; improvements are accepted immediately.

5. **Evaluation** – the best vector found (after the random walk) is
   handed to the official `evaluate_sequence` function and returned.

The whole routine respects the supplied time budget (default 1000 s)
and prints progress information.  All helper functions are top‑level
and no closures or lambda‑expressions are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1.0 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1.0 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Score assuming sum(a)=1: 2·n·max(conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for self‑product terms
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_step(a: np.ndarray,
               grad: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.05,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One Adam update on the simplex (additive). Returns new a, m, v.
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr / (np.sqrt(v_hat) + eps)
    a_new = a - step * m_hat
    # project onto simplex and enforce symmetry
    a_new = _project_onto_simplex(a_new, s=1.0)
    a_new = (a_new + a_new[::-1]) * 0.5
    return a_new, m, v


# ----------------------------------------------------------------------
def _keep_best(best_list: list[tuple[float, np.ndarray]],
               score: float,
               vec: np.ndarray,
               max_keep: int) -> None:
    """
    Insert (score,vec) into best_list keeping at most max_keep items.
    best_list is kept sorted in increasing order of score.
    """
    if len(best_list) < max_keep:
        best_list.append((score, vec.copy()))
        best_list.sort(key=lambda x: x[0])
    else:
        if score < best_list[-1][0]:
            best_list[-1] = (score, vec.copy())
            best_list.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
