def best_w_for_shape(conv_shape, conv_shape_unif, conv_uniform):
    # convex objective g(w) = max_k ((1-w)**2 * conv_shape[k] + 2*w*(1-w)*conv_shape_unif[k] + w**2 * conv_uniform[k])
    # Compute using ternary search over w in [0,1].
    low, high = 0.0, 1.0
    for _ in range(30):
        m1 = low + (high-low)/3
        m2 = high - (high-low)/3
        val1 = np.max((1-m1)**2 * conv_shape + 2*m1*(1-m1)*conv_shape_unif + m1**2 * conv_uniform)
        val2 = np.max((1-m2)**2 * conv_shape + 2*m2*(1-m2)*conv_shape_unif + m2**2 * conv_uniform)
        if val1 < val2:
            high = m2
        else:
            low = m1

    best_val = np.max((1-w_opt)**2 * conv_shape + 2*w_opt*(1-w_opt)*conv_shape_unif + w_opt**2 * conv_uniform)

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The algorithm works in three stages:

1. **Parametric shape sweep** – for a chosen support length *n* we
   generate a few families of U‑shaped distributions (symmetric
   power‑law, shifted power‑law and asymmetric power‑law).  For each
   shape the optimal mixture weight with the uniform distribution is
   found analytically (the objective is a convex function of the weight,
   so a short ternary search suffices).  The best few candidates (by the
   fast internal score = 2·n·max (conv)) are kept.

2. **Mirror‑descent refinement** – the selected seeds are polished with
   an exponentiated‑gradient (entropic mirror descent) optimiser that
   uses a smooth log‑sum‑exp surrogate for the ‑∞‑norm and switches to
   the true sub‑gradient once the temperature is large.  The routine
   stops when the per‑length time budget expires.

3. **Local stochastic hill‑climbing** – a final random mass‑transfer
   stage moves tiny amounts of probability between two indices whenever
   it reduces the true evaluation score.

The whole search respects a user‑provided budget (default 1000 s).  The
best sequence found within the time limit is returned as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho_arr.size == 0 else cssv[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) self‑convolution of *a* using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b via FFT (both arguments are real)."""
    n, m = a.shape[0], b.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
def _beta_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ (p>0)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_power_shifted_shape(n: int, p: float, delta: float) -> np.ndarray:
    """Shifted version of the symmetric power‑law with additive delta."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5 + delta) * (n - i - 0.5 + delta)) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _asymmetric_power_shape(n: int, pl: float, pr: float) -> np.ndarray:
    """Asymmetric power‑law a_i ∝ (i+½)⁻ˡ·(n‑i‑½)⁻ʳ."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-pl) * (n - i - 0.5) ** (-pr)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create a normalised non‑negative vector of length *n*."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "beta":
        a = np.random.beta(0.5, 0.5, size=n)
        s = a.sum()
        a = np.full(n, 1.0 / n) if s == 0 else a / s
    elif kind == "arcsine":
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a /= a.sum()
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = a / a.sum()
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)
    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the maximum entry of a*a at index *k*."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b) w.r.t. *a* (soft‑max surrogate).
    *b* is the self‑convolution a*a, *size_fft* is a power‑of‑two ≥2·n‑1.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:                         # fall back to pure max
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimal_mix_weight(conv_shape: np.ndarray,
                        conv_shape_unif: np.ndarray,
                        conv_unif: np.ndarray,
                        n: int) -> tuple[float, float]:
    """
    Finds the optimal mixture weight w∈[0,1] for a convex combination

        C(w) = (1‑w)²·conv_shape + 2w(1‑w)·conv_shape_unif + w²·conv_unif

    by ternary search.  Returns (w_opt, max(C(w_opt))).
    """
    low, high = 0.0, 1.0
    for _ in range(30):
        w1 = low + (high - low) / 3.0
        w2 = high - (high - low) / 3.0
        val1 = np.max((1 - w1) ** 2 * conv_shape +
                      2 * w1 * (1 - w1) * conv_shape_unif +
                      w1 ** 2 * conv_unif)
        val2 = np.max((1 - w2) ** 2 * conv_shape +
                      2 * w2 * (1 - w2) * conv_shape_unif +
                      w2 ** 2 * conv_unif)
        if val1 < val2:
            high = w2
        else:
            low = w1
    w_opt = (low + high) / 2.0
    best_val = np.max((1 - w_opt) ** 2 * conv_shape +
                      2 * w_opt * (1 - w_opt) * conv_shape_unif +
                      w_opt ** 2 * conv_unif)
    return w_opt, best_val


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Entropic mirror‑descent (exponentiated‑gradient) refinement.
    Returns the best vector seen and its internal score 2·n·max(conv).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    eta0 = 0.5          # initial learning‑rate factor
    alpha = 5.0         # start temperature for the soft‑max surrogate
    alpha_mul = 1.02
    alpha_max = 300.0
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()

        # gradient
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)

        # occasional jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing: move a tiny fraction of mass from one index
    to another whenever the true evaluation score improves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.10
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        cur_score = 2.0 * n * np.max(_fft_self_convolve(cand))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand
    return best_seq, best_score

    # ------------------------------------------------------------------
    # Determine a reasonable base length (use the previous construction if
    # available, otherwise fall back to a default of 8000).
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
        if base_n < 500:
            base_n = 8000
    except Exception:
        base_n = 8000

    # try a few nearby lengths – they are cheap to evaluate
    lengths = sorted({max(500, int(base_n * f))
                      for f in (0.85, 1.0, 1.15)
                      if max(500, int(base_n * f)) <= 20000})
    if not lengths:
        lengths = [base_n]

    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n={n} ===")

        # pre‑compute the uniform distribution and its self‑convolution
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        conv_uniform = _fft_self_convolve(uniform)

        # --------------------------------------------------------------
        # generate a pool of candidate vectors from several shape families
        candidate_pool = []

        # 1) previous construction if it matches the length
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size == n:
                candidate_pool.append(prev / prev.sum())
        except Exception:
            pass

        # 2) plain arcsine law (the classic starting point)
        candidate_pool.append(_initial_sequence(n, "arcsine"))

        # 3) symmetric power‑law shapes with optimal mixing weight
        p_grid = np.linspace(0.30, 0.80, 11)  # 0.30 … 0.80
        for p in p_grid:
            shape = _beta_power_shape(n, p)
            conv_shape = _fft_self_convolve(shape)
            conv_shape_unif = _fft_convolve(shape, uniform)
            w_opt, _ = _optimal_mix_weight(conv_shape, conv_shape_unif,
                                           conv_uniform, n)
            combined = (1 - w_opt) * shape + w_opt * uniform
            candidate_pool.append(combined)

            # shifted variants – small deltas smooth the singularities
            for delta in (0.5, 1.0, 2.0):
                shape_s = _beta_power_shifted_shape(n, p, delta)
                conv_s = _fft_self_convolve(shape_s)
                conv_s_unif = _fft_convolve(shape_s, uniform)
                w_opt_s, _ = _optimal_mix_weight(conv_s, conv_s_unif,
                                                 conv_uniform, n)
                combined_s = (1 - w_opt_s) * shape_s + w_opt_s * uniform
                candidate_pool.append(combined_s)

        # 4) asymmetric power‑law shapes (different left/right exponents)
        pl_vals = np.linspace(0.35, 0.75, 5)
        pr_vals = np.linspace(0.35, 0.75, 5)
        for pl in pl_vals:
            for pr in pr_vals:
                shape_a = _asymmetric_power_shape(n, pl, pr)
                conv_a = _fft_self_convolve(shape_a)
                conv_a_unif = _fft_convolve(shape_a, uniform)
                w_opt_a, _ = _optimal_mix_weight(conv_a, conv_a_unif,
                                                 conv_uniform, n)
                combined_a = (1 - w_opt_a) * shape_a + w_opt_a * uniform
                candidate_pool.append(combined_a)

        # 5) a handful of pure random seeds (helpful for escaping basins)
        for _ in range(4):
            rand = np.random.rand(n)
            candidate_pool.append(rand / rand.sum())

        # --------------------------------------------------------------
        # evaluate the cheap internal score for all pooled candidates
        scored = []
        for vec in candidate_pool:
            cheap = 2.0 * n * _fft_self_convolve(vec).max()
            scored.append((cheap, vec))
        scored.sort(key=lambda x: x[0])
        top_candidates = [vec for _, vec in scored[:5]]   # keep the 5 best

        # --------------------------------------------------------------
        # refine each top candidate with entropic mirror descent
        for idx, seed_vec in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            # allocate a slice of the remaining budget (max 25 s per seed)
            slice_deadline = time.time() + min(25.0, (deadline - time.time()) * 0.3)
            seq_opt, cheap_opt = _optimize_exponential(seed_vec, n, slice_deadline)

            # evaluate with the true black‑box function
            try:
                final_score = evaluate_sequence(seq_opt.tolist())
            except Exception:
                final_score = cheap_opt

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq_opt.copy()
                print(f"*** New global best: len={n} score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break
        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # final stochastic hill‑climbing on the globally best vector
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n = len(overall_best_seq)
        seq_refined, _ = _local_refinement(overall_best_seq, n, deadline)
        try:
            final_score = evaluate_sequence(seq_refined.tolist())
        except Exception:
            final_score = evaluate_sequence(overall_best_seq.tolist())
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq_refined.copy()
            print(f"*** After local refinement: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback – should never be needed but guarantees a return value
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback construction used.")

    print(f"\nFinal best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
