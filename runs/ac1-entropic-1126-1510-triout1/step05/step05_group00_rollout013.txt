"""
Propose a low‑score non‑negative step function.

The search proceeds in three stages:

1. **Exploration of analytic families** – for several lengths we generate
   candidates from arcsine, symmetric beta, triangular, exponential and a
   random Dirichlet.  The best of those candidates (by the exact objective
   2·n·max(a*a) with ∑a=1) becomes the start point for the next stage.

2. **Adam‑guided projected sub‑gradient descent** – we minimise the
   max‑convolution while keeping the vector on the simplex.  The gradient
   of the maximum is obtained from the active convolution indices;
   Adam’s adaptive moments together with a Euclidean projection onto the
   simplex give fast, stable convergence.  No artificial diagonal
   correction is applied (the exact sub‑gradient of the ordered convolution
   already accounts for diagonal terms).

3. **Local random‑transfer hill climbing** – after the continuous phase
   we perform cheap combinatorial moves that transfer a small amount of mass
   between two positions.  Improvements are kept; occasional jitter helps
   escape flat plateaux.

The whole routine respects the 1000 s budget, automatically stopping when the
deadline is reached or when the target score (≈ 1.503) is achieved.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p, p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid overflow for extreme p
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw - np.max(logw))  # stabilise
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:                               # pragma: no cover
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle, normalised."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """
    Return the evaluation score 2·n·max(a*a) under the assumption sum(a)=1.
    (Exactly the value returned by ``evaluate_sequence`` for unit‑sum vectors.)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgradient_max_convolution(a: np.ndarray,
                                 b: np.ndarray,
                                 tol: float = 1e-12) -> np.ndarray:
    """
    Compute a (sub)gradient of max(a*a) at the point ``a``.
    ``b`` must be the full self‑convolution of ``a``.
    The active set is defined by a tolerance ``tol``.
    """
    n = a.shape[0]
    cur_max = b.max()
    eps = tol * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)
    if active.size == 0:                         # pragma: no cover
        active = np.array([np.argmax(b)], dtype=int)

    # build a probability vector on the active indices (averaging)
    w = np.zeros_like(b)
    w[active] = 1.0 / active.size

    # gradient = 2 * (a_rev * w) restricted to the original length
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   max_iter: int,
                   deadline: float,
                   target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the max‑convolution objective.
    Returns the best seen vector and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.3                     # base learning rate

    best_seq = a.copy()
    best_score = _objective_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # self‑convolution and gradient of the max term
        b = _fft_self_convolve(a)
        grad = _subgradient_max_convolution(a, b)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # diminishing step size
        eta = step0 / np.sqrt(t)

        # Adam step followed by simplex projection
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _objective_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam iter={t} new best {best_score:.6f}")

            if best_score <= target:
                break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simple random‑transfer hill climbing on the simplex.
    Small amounts of mass are moved between two random positions.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective_score(a)
    cur_score = best_score

    iteration = 0
    max_transfer = 0.08   # at most 8 % of total mass per move (scaled later)

    while time.time() < deadline:
        iteration += 1
        # exponential decay of the transferred mass
        transfer = max_transfer * (0.5 ** (iteration / 25000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _objective_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Transfer iter={iteration} improved best {best_score:.6f}")
                if best_score <= target:
                    break

        # occasional jitter to escape flat plateaux
        if iteration % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _objective_score(a)

    return best_seq, best_score
