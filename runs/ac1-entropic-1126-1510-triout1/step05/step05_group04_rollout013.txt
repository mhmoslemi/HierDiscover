def propose_candidate(budget_s=1000, seed=0):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.1

    # Determine length n
    try:
        n0 = len(height_sequence_1)
        n = min(max(n0 * 2, 3000), 20000)
    except Exception:
        n = 12000

    S = math.sqrt(2.0 * n)

    best_score = float('inf')
    best_seq = None

    exponents = [0, 0.5, 1, 1.5, 2, 2.5, 3, 3.5, 4, 5, 6, 7, 8, 10]

    for p in exponents:
        if time.time() > deadline:
            break
        if p == 0:
            w = np.ones(n, dtype=np.float64)
        else:
            i = np.arange(n, dtype=np.float64)
            w = i ** p
        # Normalise to have sum S
        a = S * w / w.sum()
        score = evaluate_sequence(a.tolist())
        print(f"[exponent {p}] score = {score:.6f}")
        if math.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
    # Additional local random perturbation around best_seq
    if best_seq is not None:
        a = best_seq.copy()
        # try random jitter 2000 times until time limit
        it = 0
        while time.time() < deadline and it < 2000:
            # pick random indices to modify
            idx = rng.integers(0, n, size=2, replace=False)
            # small perturbation proportional to current value
            delta = (rng.random() - 0.5) * a[idx[0]] * 0.1
            a_new = a.copy()
            a_new[idx[0]] = max(0.0, a_new[idx[0]] + delta)
            a_new[idx[1]] = max(0.0, a_new[idx[1]] - delta)  # conserve sum roughly
            # re-normalize to same total S to keep scale (optional)
            total = a_new.sum()
            if total > 0:
                a_new *= S / total
            sc = evaluate_sequence(a_new.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a_new.copy()
                a = a_new  # continue from improved
                print(f"[jitter] new best score = {best_score:.6f}")
            it += 1

    if best_seq is None:
        # fallback: uniform
        best_seq = np.full(n, S / n, dtype=np.float64)

    print(f"[final] best score = {best_score:.6f}")
    return best_seq.tolist()
