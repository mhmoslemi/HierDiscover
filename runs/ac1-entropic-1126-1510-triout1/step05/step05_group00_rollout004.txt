"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1.  Sample a variety of analytic families (arcsine, symmetric beta,
    triangular, exponential, semicircle, Gaussian) at several lengths.
    The best of these is kept as a starting point.

2.  Iteratively improve the shape using a hybrid of
    * exact sub‑gradient descent on the convex objective
      `max(conv(a,a))` (projected onto the simplex), and
    * a tiny linear‑programming “boost” that maximises the total mass
      while keeping all active convolution entries ≤ the current maximum.
    The LP step often yields a large beneficial move that the simple
    sub‑gradient cannot find.

3.  Finish with a cheap random‑transfer hill‑climber to polish the
    result.

The whole search respects the provided time budget (≈ 1000 s) and
returns a list suitable for `evaluate_sequence`.
"""

import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a)) (a is assumed to have sum = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p, p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _semicircle_distribution(n: int) -> np.ndarray:
    """Wigner‑semicircle law on a grid, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = (i / n) * 2.0 - 1.0               # map to [-1, 1]
    vals = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.15) -> np.ndarray:
    """Discrete Gaussian centred in the middle, normalised to sum 1."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _build_active_conv_rows(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Construct rows of the convolution matrix that correspond to the
    given active indices.  Row k satisfies (a * g)[k] = row_k @ g.
    """
    n = a.shape[0]
    rows = np.zeros((active_idx.size, n), dtype=a.dtype)
    for r, k in enumerate(active_idx):
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # j runs from low … high, coefficient = a[k-j]
        src = k - np.arange(low, high + 1)
        rows[r, low:high + 1] = a[src]
    return rows

    best_score = float("inf")

    beta_ps = [0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50, 0.70, 1.00, 1.50, 2.00, 3.00]
    exp_lams = [0.001, 0.002, 0.005, 0.01, 0.02]

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine
        a = _arcsine_distribution(n)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] n={n} arcsine   score={sc:.6f}")

        # symmetric beta with various p
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] n={n} beta(p={p}) score={sc:.6f}")

        # triangular
        a = _triangular_distribution(n)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] n={n} triangular score={sc:.6f}")

        # exponential with several λ
        for lam in exp_lams:
            a = _exponential_distribution(n, lam=lam)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] n={n} exponential(lam={lam}) score={sc:.6f}")

        # semicircle
        a = _semicircle_distribution(n)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] n={n} semicircle score={sc:.6f}")

        # Gaussian with a couple of sigmas
        for sf in (0.10, 0.15, 0.20):
            a = _gaussian_distribution(n, sigma_factor=sf)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] n={n} gaussian(sf={sf}) score={sc:.6f}")

        # Random Dirichlet draws (few, different concentration)
        for conc in (0.3, 0.7, 2.0):
            a = np.random.dirichlet(np.full(n, conc))
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] n={n} dirichlet(c={conc}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Consider a previously supplied construction if it beats the current best.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # Fallback (should never happen)
    if best_seq is None:
        n = max(500, base_n)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine the best candidate with a hybrid sub‑gradient / LP scheme.
    a = best_seq.copy()          # always kept normalised to sum=1
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    # parameters for the sub‑gradient part
    velocity = np.zeros_like(a)
    beta = 0.9               # momentum
    step0 = 0.5              # initial step‑size scale
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1

        # current convolution and objective
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep global best
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={iter_cnt} new best {best_score:.6f}")

        # find all almost‑active indices
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps)

        # ------------------------------------------------------------------
        # 4a.  Linear‑programming boost (if scipy is available)
        lp_success = False
        if _HAVE_SCIPY and active.size > 0:
            rows = _build_active_conv_rows(a, active)
            try:
                res = scipy.optimize.linprog(
                    c=-np.ones(n),                # maximise sum(g)  <=> minimise -sum(g)
                    A_ub=rows,
                    b_ub=np.full(active.shape[0], cur_max),
                    bounds=[(0, None)] * n,
                    method='highs',
                    options={'presolve': True, 'disp': False},
                )
                if res.success:
                    g = res.x
                    if g.sum() > 0:
                        g_norm = g / g.sum()
                        # try a few mixing coefficients, starting with a full jump
                        for tau in (1.0, 0.5, 0.3, 0.1, 0.05):
                            cand = (1.0 - tau) * a + tau * g_norm
                            cand = _project_onto_simplex(cand, s=1.0)
                            cand_score = _max_convolution_score(cand)
                            if cand_score < cur_score - 1e-14:
                                a = cand
                                cur_score = cand_score
                                lp_success = True
                                break
            except Exception:   # pragma: no cover
                pass

        if lp_success:
            continue   # go to next iteration, sub‑gradient step is skipped

        # ------------------------------------------------------------------
        # 4b.  Sub‑gradient descent (exact max)
        w = np.zeros_like(conv)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # diagonal correction (indices k=2i that are active)
        even_mask = (active % 2 == 0)
        diag_idx = (active[even_mask] // 2)
        if diag_idx.size > 0:
            grad[diag_idx] -= a[diag_idx]

        # momentum update and diminishing step size
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / (np.sqrt(iter_cnt) + 1e-12)
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # occasional random perturbation to escape plateaux
        if iter_cnt % 3000 == 0 and np.random.rand() < 0.2:
            pert = np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a + pert, s=1.0)

        # early stop if target achieved
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 5.  Local random‑transfer hill climbing (fine‑tuning).
    a = best_seq.copy()
    cur_score = best_score
    iter_hc = 0
    max_transfer = 0.05   # up‑to 5 % of total mass per trial

    while time.time() < deadline:
        iter_hc += 1
        # gradually shrink the transferred mass
        transfer = max_transfer * (0.5 ** (iter_hc / 25000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iter_hc} best {best_score:.6f}")

        # occasional jitter to shake free from flat regions
        if iter_hc % 5000 == 0 and np.random.rand() < 0.1:
            jitter = np.random.randn(n) * 1e-4
            a = _project_onto_simplex(best_seq + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
