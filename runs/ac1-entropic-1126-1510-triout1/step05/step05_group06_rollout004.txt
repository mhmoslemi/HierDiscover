def generate_shape(n, pl, pr, mix, eps):
    i = np.arange(n, dtype=np.float64)
    left = (i + 0.5 + eps) ** (-pl)
    right = (n - i - 0.5 + eps) ** (-pr)
    shape = left * right
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape = shape / shape_sum
    else:
        shape = np.full(n, 0.0)
    if mix > 0:
        uniform = np.full(n, 1.0 / n)
        shape = (1 - mix) * shape + mix * uniform
    shape = project_onto_simplex(shape, 1.0)
    return shape
```

def grid_search(n, max_candidates=10):
    best = []
    pl_vals = np.linspace(0.2, 1.0, 9)  # 0.2 to 1.0
    pr_vals = np.linspace(0.2, 1.0, 9)
    w_vals = np.linspace(0.0, 0.4, 5)

                        if s < best[-1][0]:

def hill_climb_params(params, n, initial_score, time_deadline):
    pl, pr, w, eps = params
    best_score = initial_score
    best_params = list(params)
    # define step sizes
    step_pl = 0.1
    step_pr = 0.1
    step_w = 0.05
    step_eps = 0.5
    # bounds
    while time.time() < time_deadline:
        improved = False
        for i, step in enumerate([step_pl, step_pr, step_w, step_eps]):
            for sign in (-1, 1):
                new_params = best_params.copy()
                new_params[i] += sign * step
                # enforce bounds
                if i == 0: # pl
                    new_params[i] = max(0.01, min(2.0, new_params[i]))
                elif i == 1: # pr
                    new_params[i] = max(0.01, min(2.0, new_params[i]))
                elif i == 2: # w
                    new_params[i] = max(0.0, min(0.5, new_params[i]))
                else: # eps
                    new_params[i] = max(0.0, min(5.0, new_params[i]))
                a = generate_shape(n, *new_params)
                s = 2.0 * n * np.max(_fft_self_convolve(a))
                if s < best_score - 1e-9:
                    best_score = s
                    best_params = new_params
                    improved = True
        if not improved:
            # shrink steps
            step_pl *= 0.5
            step_pr *= 0.5
            step_w *= 0.5
            step_eps *= 0.5

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The algorithm follows four stages:

1. **Coarse grid search** over a 4‑parameter family
   a_i ∝ (i+½+ε)^(-p_l)·(n-i-½+ε)^(-p_r) optionally mixed with a uniform
   component.  The best few shapes are kept.

2. **Parameter hill‑climbing** – a simple coordinate‑descent that refines
   the four parameters (p_l, p_r, mix, ε) using the cheap FFT‑based score.

3. **Exponentiated‑gradient refinement** – mirror‑descent on the simplex
   with a soft‑max surrogate for the ‑∞‑norm, followed by the true
   sub‑gradient once the temperature is large.

4. **Stochastic local refinement** – random mass transfers between two
   indices (with simulated‑annealing acceptance) to escape possible
   plateaus.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho_arr.size == 0 else cssv[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) convolution of *a* with itself via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max entry of a*a at index *k*."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of (1/α)·log∑exp(α·b) w.r.t. a (soft‑max surrogate)."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _generate_shape(n: int,
                    pl: float,
                    pr: float,
                    mix: float,
                    eps: float) -> np.ndarray:
    """
    Return a normalised (sum‑1) non‑negative vector of length *n*.
    a_i ∝ (i+0.5+ε)^(-pl)·(n‑i‑0.5+ε)^(-pr) mixed with uniform.
    """
    i = np.arange(n, dtype=np.float64)
    left = (i + 0.5 + eps) ** (-pl)
    right = (n - i - 0.5 + eps) ** (-pr)
    shape = left * right
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    if mix > 0.0:
        shape = (1.0 - mix) * shape + mix * (1.0 / n)
    shape = _project_onto_simplex(shape, s=1.0)
    return shape


# ----------------------------------------------------------------------
def _grid_search_best(n: int,
                      max_candidates: int = 8) -> list[tuple[float, tuple[float, float, float, float], np.ndarray]]:
    """
    Exhaustively scan a modest grid of (pl, pr, mix, eps) and return the
    best *max_candidates* vectors together with their parameters and cheap
    score 2·n·max(conv).
    """
    best = []   # list of (score, params, vector)
    pl_vals = np.linspace(0.20, 1.00, 9)   # 0.20 … 1.00
    pr_vals = np.linspace(0.20, 1.00, 9)
    mix_vals = np.linspace(0.0, 0.40, 5)   # up to 40 % uniform
    eps_vals = np.linspace(0.0, 2.0, 5)    # smoothing offset

    for pl in pl_vals:
        for pr in pr_vals:
            for mix in mix_vals:
                for eps in eps_vals:
                    a = _generate_shape(n, pl, pr, mix, eps)
                    score = 2.0 * n * _fft_self_convolve(a).max()
                    if len(best) < max_candidates:
                        best.append((score, (pl, pr, mix, eps), a))
                        best.sort(key=lambda x: x[0])
                    else:
                        if score < best[-1][0]:
                            best[-1] = (score, (pl, pr, mix, eps), a)
                            best.sort(key=lambda x: x[0])
    return best


# ----------------------------------------------------------------------
def _hill_climb_params(params: tuple[float, float, float, float],
                       n: int,
                       init_score: float,
                       deadline: float) -> tuple[tuple[float, float, float, float], float]:
    """
    Simple coordinate‑descent on the four shape parameters.
    Returns the improved parameters and its cheap score.
    """
    pl, pr, mix, eps = params
    best_params = [pl, pr, mix, eps]
    best_score = init_score

    # initial step sizes
    step_pl = 0.10
    step_pr = 0.10
    step_mix = 0.05
    step_eps = 0.40

    while time.time() < deadline:
        improved = False
        # loop over the four coordinates
        for idx, step in enumerate([step_pl, step_pr, step_mix, step_eps]):
            for sign in (-1, 1):
                cand_params = best_params.copy()
                cand_params[idx] += sign * step
                # enforce bounds
                if idx == 0:   # pl
                    cand_params[idx] = max(0.01, min(2.0, cand_params[idx]))
                elif idx == 1:  # pr
                    cand_params[idx] = max(0.01, min(2.0, cand_params[idx]))
                elif idx == 2:  # mix
                    cand_params[idx] = max(0.0, min(0.5, cand_params[idx]))
                else:           # eps
                    cand_params[idx] = max(0.0, min(5.0, cand_params[idx]))

                a = _generate_shape(n, *cand_params)
                s = 2.0 * n * _fft_self_convolve(a).max()
                if s < best_score - 1e-12:
                    best_score = s
                    best_params = cand_params
                    improved = True
        if not improved:
            # shrink steps
            step_pl *= 0.5
            step_pr *= 0.5
            step_mix *= 0.5
            step_eps *= 0.5
            if max(step_pl, step_pr, step_mix, step_eps) < 1e-4:
                break
    return tuple(best_params), best_score


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient refinement of a starting vector.
    Returns the best vector seen and its cheap 2·n·max score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    eta0 = 0.6               # initial learning‑rate factor
    alpha = 6.0              # start temperature (soft‑max)
    alpha_mul = 1.015
    alpha_max = 300.0
    size_fft = 1 << ((3 * n - 2).bit_length())
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()

        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # occasional tiny random jitter to shake out plateaus
        if it % 3000 == 0 and np.random.rand() < 0.10:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with simulated‑annealing acceptance.
    Moves a fraction of mass from one index to another whenever the true
    evaluation score improves, and occasionally accepts worsening moves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move up to 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur_score = 2.0 * n * np.max(_fft_self_convolve(cand))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand
        else:
            # accept worse move with a decaying probability
            T = max(0.01, 0.1 / np.sqrt(it))
            if np.random.rand() < np.exp(-(cur_score - best_score) / T):
                a = cand
    return best_seq, best_score

def _candidate_from_previous(n: int) -> np.ndarray | None:
    """
    If a global variable ``height_sequence_1`` exists, interpolate it to length n
    and return a normalised vector.  Returns None on failure.
    """
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.size == n:
            a = prev.copy()
        else:
            # linear interpolation to the new length
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, prev)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        a = _project_onto_simplex(a, 1.0)
        return a
    except Exception:
        return None

    and returns the best sequence discovered.
    """

    deadline = time.time() + budget_s - 1.0   # small safety margin

    # --------------------------------------------------------------
    # Choose a work length.  Use the length of the previous solution if
    # available, otherwise pick a moderate size that balances speed and
    # quality.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
        base_n = max(2000, min(12000, base_n))
    except Exception:
        base_n = 8000
    n = base_n

    # --------------------------------------------------------------
    # Split the remaining time budget.
    total_left = deadline - time.time()

    t_param  = total_left * 0.18   # parameter hill‑climb
    t_exp    = total_left * 0.45   # exponentiated‑gradient runs

        score_prev = 2.0 * n * _fft_self_convolve(prev_candidate).max()
        grid_best.append((score_prev, ("prev",), prev_candidate))

        grid_best = grid_best[:8]

    top_candidates = grid_best[:4]

    refined = []
    param_deadline = time.time() + t_param

    for score0, params, vec in top_candidates:

        seq, cheap_score = _optimize_exponential(a0, n, slice_deadline)
        # Evaluate with the black‑box validator (exact objective)
        try:
            final_score = evaluate_sequence(seq.tolist())
        except Exception:
            final_score = cheap_score        # fallback if something goes wrong
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq.copy()
            print(f"*** New global best → len={n} score={overall_best_score:.6f}")
        if overall_best_score <= 1.503:

        seq, local_score = _local_refinement(overall_best_seq.copy(), n, deadline)
        try:

        except Exception:
            final_score = local_score
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq.copy()
            print(f"*** After final refinement score={overall_best_score:.6f}")

        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
