"""Utility to search for a low‑score step‑function.

The algorithm proceeds in three stages:

1. **Broad initialization** – generate many candidate probability vectors
   (arcsine, Beta, power‑law, random Dirichlet, mixtures).  Keep the best
   according to the cheap proxy 2·n·max(conv).

2. **Continuous refinement** – run a smooth‑max Adam optimisation, then a
   short exact ℓ∞ sub‑gradient descent.  A small linear‑programming
   “dual‑widening” step (if cvxpy is available) is applied afterwards:
   given the current vector *f*, we maximise ‖b‖₁ under the linear
   constraints `(f∗b)_k ≤ M`, where `M = max(f∗f)`.  The optimiser then mixes
   the solution back into *f*; a few repetitions of this LP step typically
   lower the ℓ∞ bound a little.

3. **Stochastic polishing** – a lightweight simulated‑annealing phase that
   moves a tiny amount of mass between two random positions while always
   accepting improvements and occasionally accepting uphill moves.

The whole procedure respects the simplex (`Σa_i = 1, a_i ≥ 0`), uses fast FFT
convolutions and respects the global time‑budget supplied by the caller.
"""

import math

from numpy.fft import rfft, irfft

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    """Distribution proportional to ((i+0.5)*(n-i-0.5))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered and its peak convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak

def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses a uniform weight on the active set (indices attaining the current maximum).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short exact‑max refinement using sub‑gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a

def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Simple simulated‑annealing polishing.
    Moves a small amount of mass between two random positions,
    accepting uphill moves with a temperature schedule.
    Returns the best vector found and its peak convolution.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak

def _lp_improve(a: np.ndarray, max_iters: int = 2, mix_t: float = 0.5,
                inner_deadline: float = None) -> np.ndarray:
    """
    Linear‑programming based improvement reminiscent of the iterative scheme
    described in the original paper.  For each iteration we maximise ‖b‖₁
    under the linear constraints (a∗b)_k ≤ M, where M = max(a∗a).
    The resulting vector b is normalised to a probability vector and mixed
    back into a using a convex combination with weight ``mix_t``.
    If cvxpy is not available the function simply returns ``a`` unchanged.
    """
    if not _cvxpy_available:
        return a

    n = a.shape[0]
    # Build the Toeplitz convolution matrix once; it does not change in the loop.
    # Columns are shifted copies of a.
    col = np.r_[a, np.zeros(n - 1)]
    row = np.r_[a[0], np.zeros(n - 1)]
    A_dense = scipy.linalg.toeplitz(col, row)   # shape (2n‑1, n)

    # Convert to a sparse matrix for cvxpy efficiency.
    A = cp.Parameter((2 * n - 1, n), value=A_dense)

    for _ in range(max_iters):
        M = _fft_self_convolve(a).max()
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            break
        if b.value is None:
            break
        b_vec = np.maximum(b.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)
        # update A matrix for the new a (required for the next iteration)
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A_dense = scipy.linalg.toeplitz(col, row)
        A.value = A_dense
        # optional early stop if time is exhausted
        if inner_deadline is not None and time.time() > inner_deadline:
            break
    return a


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

"""Utility to search for a low‑score step‑function.

The algorithm proceeds in four stages:

1. **Broad initialization** – generate many candidate probability vectors
   (arcsine, Beta, power‑law, Dirichlet, mixtures).  The best according to the
   cheap proxy ``2·n·max(conv)`` is kept.

2. **Continuous refinement** – Adam optimisation on a smooth‑max surrogate,
   followed by a short exact ℓ∞ sub‑gradient descent.

3. **Linear‑programming improvement** (optional, uses *cvxpy* if available):
   for the current vector *f* we maximise ‖b‖₁ under the linear constraints
   ``(f∗b)_k ≤ M`` with ``M = max(f∗f)``.  The optimiser is mixed back into *f*
   (convex combination) a few times.

4. **Simulated‑annealing polishing** – cheap stochastic moves that always
   accept improvements and occasionally accept uphill moves.

The whole procedure respects the simplex (Σ aᵢ = 1, aᵢ ≥ 0), uses fast FFT
convolutions and obeys the time budget supplied by the caller.
"""

import math
import time
import numpy as np
from numpy.fft import rfft, irfft
import scipy.linalg

# optional LP support
try:
    import cvxpy as cp
    _cvxpy_available = True
except Exception:  # pragma: no cover
    _cvxpy_available = False


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length `n`."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _powerlaw_boundary_distribution(n: int, p: float) -> np.ndarray:
    """Distribution ∝ ((i+0.5)*(n‑i‑0.5))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of the smooth‑max surrogate Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """One sub‑gradient step for the exact ℓ∞ objective."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Light simulated‑annealing polish."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                max_iters: int = 2,
                mix_t: float = 0.5,
                inner_deadline: float = None) -> np.ndarray:
    """
    LP‑based improvement (dual‑widening).  For each iteration we maximise
    ‖b‖₁ under (a∗b)_k ≤ M where M = max(a∗a).  The resulting b is normalised
    and mixed back into a.  If *cvxpy* is unavailable the function returns ``a``.
    """
    if not _cvxpy_available:
        return a

    n = a.shape[0]
    for _ in range(max_iters):
        if inner_deadline is not None and time.time() > inner_deadline:
            break
        M = _fft_self_convolve(a).max()
        # build convolution matrix A (2n‑1)×n: columns are shifted copies of a
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)        # dense NumPy array

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            break
        if b_var.value is None:
            break
        b_vec = np.maximum(b_var.value, 0.0)
        s = b_vec.sum()
        if s <= 1e-12:
            break
        g = b_vec / s
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low score.
    Returns a plain Python list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Try to seed from a previously discovered construction.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # ------------------------------------------------------------------
    # Lengths to explore (at least 500 entries).
    lengths = [
        max(500, int(base_n * f))
        for f in (0.6, 0.8, 1.0, 1.2, 1.4)
        if int(base_n * f) >= 500
    ]
    if not lengths:
        lengths = [max(500, base_n)]

    overall_best_seq = None
    overall_best_score = float("inf")

    print("[phase 1] broad search for a good start point")
    # ------------------------------------------------------------------
    # 1.  Build a diverse collection of starter vectors.
    for n in lengths:
        if time.time() >= deadline:
            break

        starters = {}

        starters["arcsine"] = _arcsine_distribution(n)
        starters["uniform"] = np.full(n, 1.0 / n, dtype=np.float64)

        # Dirichlet draws with several concentration parameters
        for conc in (0.2, 0.5, 1.0, 2.0):
            starters[f"dirichlet_{conc:.1f}"] = np.random.dirichlet(np.full(n, conc))

        # Symmetric Beta family
        for a in (0.35, 0.45, 0.55, 0.65, 0.75):
            starters[f"beta_{a:.2f}"] = _beta_distribution(n, a, a)

        # Power‑law boundary family
        for p in (0.4, 0.45, 0.5, 0.55, 0.6):
            starters[f"power_{p:.2f}"] = _powerlaw_boundary_distribution(n, p)

        # Mixtures of arcsine and uniform
        for w in (0.2, 0.4, 0.6, 0.8):
            starters[f"mix_{w:.1f}"] = (1.0 - w) * _arcsine_distribution(n) + w * np.full(n,
                                                                                           1.0 / n)

        if base_seq is not None:
            starters["prev"] = _interpolate_to_length(base_seq, n)

        # ------------------------------------------------------------------
        # Evaluate the cheap proxy 2·n·max(conv) and keep the best.
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<12s} fast_score={fast_score:.6f}")

    # Fallback in the unlikely case that nothing was generated.
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 2.  Adam + sub‑gradient refinement.
    time_left = deadline - time.time()
    if time_left > 4.0:
        n = len(overall_best_seq)
        print(f"[phase 2] Adam optimisation (n={n}) – {time_left:.1f}s left")
        phase_adam_deadline = time.time() + time_left * 0.55
        a_opt, _ = _adam_optimize(overall_best_seq, phase_adam_deadline)
        phase_subgrad_deadline = time.time() + time_left * 0.30
        a_opt = _refine_max(a_opt, phase_subgrad_deadline)
        overall_best_seq = a_opt.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 2] post‑refinement fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Optional LP improvement (skipped for very large n to keep memory low).
    time_left = deadline - time.time()
    if time_left > 2.5 and len(overall_best_seq) <= 3000 and _cvxpy_available:
        n = len(overall_best_seq)
        print(f"[phase 3] LP improvement (n={n}) – {time_left:.1f}s left")
        lp_deadline = time.time() + time_left * 0.6
        a_lp = _lp_improve(overall_best_seq, max_iters=2,
                           mix_t=0.5, inner_deadline=lp_deadline)
        overall_best_seq = a_lp.copy()
        overall_best_score = 2.0 * n * _fft_self_convolve(overall_best_seq).max()
        print(f"[phase 3] after LP fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Light simulated‑annealing polishing.
    time_left = deadline - time.time()
    if time_left > 1.0:
        n = len(overall_best_seq)
        print(f"[phase 4] simulated annealing (n={n}) – {time_left:.1f}s left")
        a_final, final_peak = _simulated_annealing(overall_best_seq, deadline)
        overall_best_seq = a_final.copy()
        overall_best_score = 2.0 * n * final_peak
        print(f"[phase 4] SA fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation.
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score (approx.): {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
