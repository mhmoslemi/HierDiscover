def _subgradient(a):
    conv = _fft_self_convolve(a)
    max_b = conv.max()
    act_idx = np.where(conv == max_b)[0]  # could be multiple
    # compute gradient
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in act_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        # i runs start_i..end_i
        # j = k - i
        # accumulate
        grad[start_i:end_i+1] += 2 * a[k - np.arange(start_i, end_i+1)]
    # average across active indices
    grad /= len(act_idx)
    return grad

def _proj_subgradient_opt(a_init, n, deadline):
    a = a_init.copy()
    best = a.copy()
    best_score = _objective(a)
    t0 = time.time()
    it = 0
    while time.time() < deadline:
        it+=1
        conv = _fft_self_convolve(a)
        t = conv.max()
        act_idx = np.where(conv == t)[0]
        # compute subgradient for t:
        grad_t = np.zeros_like(a)
        num_act = len(act_idx)
        for k in act_idx:
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            i_range = np.arange(start_i, end_i + 1)
            grad_t[i_range] += a[k - i_range]
        grad_t = grad_t * (2.0 / num_act)  # factor 2
        # update a: move opposite gradient
        # step size schedule: start at lr=0.2 maybe and decays
        # we can compute lr = 0.5 / (1+0.01*it) something
        lr = 0.5 / np.sqrt(it)  # sqrt decay
        a = a - lr * grad_t
        # projection onto simplex
        a = _project_onto_simplex(a, 1.0)
        # compute score
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[projGD] n={n} it={it} score={best_score:.6f}")
    return best, best_score

    if best_overall_seq is None:

        best_overall_seq = np.full(n, 1.0/n)
        best_len = n
        best_overall_score = _objective_from_array(best_overall_seq)

    # Phase 1: subgradient descent on best length

def _adam_optimize_exact(init_a, n, deadline):
    a = init_a.copy()
    best = a.copy()
    best_score = _objective_from_array(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b / (a.sum()**2)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] n={n} it={it} score={best_score:.6f}")
        # compute subgradient of max conv
        act_idx = np.where(conv == max_b)[0]
        grad_t = np.zeros_like(a)
        num_act = len(act_idx)
        for k in act_idx:
            start = max(0, k - (n-1))
            end = min(n-1, k)
            idx_range = np.arange(start, end+1)
            grad_t[idx_range] += a[k - idx_range]
        grad_t = (2.0 / num_act) * grad_t
        # full gradient of objective (ignoring constant term as projection)
        grad = grad_t  # because after projection constant term removed
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # projection onto simplex
        a = _project_onto_simplex(a, 1.0)
    return best, best_score

   onto the simplex.  This replaces the soft‑max Adam step used previously
   and converges faster to a lower objective.

3. **Stochastic hill‑climbing polish** –  a cheap local search that moves
   a random fraction of the mass from one index to another (and occasional
   tiny Gaussian jitter) is performed until the time budget expires.

All work respects the 1000 s budget; the final best sequence is returned as
a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points (mid‑point rule)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) pdf on n bins (mid‑points)."""
    a = max(a, 0.05)                # avoid too‑sharp singularities
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0 or not np.isfinite(pdf_sum):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= pdf_sum
    return pdf


# ----------------------------------------------------------------------
def _candidate_shapes_extended(n: int,
                               prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest collection of hand‑crafted probability vectors
    for a given length n.
    """
    cand = []

    # 0) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 1) arcsine (Beta(½,½))
    cand.append(_arcsine_distribution(n))

    # 2) symmetric Beta(a,a) on a fine grid (including a<0.5)
    for a in np.linspace(0.10, 0.90, 17):
        cand.append(_beta_distribution(n, a))

    # 3) mixtures uniform + arcsine
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    arcsine = _arcsine_distribution(n)
    for w in np.linspace(0.0, 0.5, 11):
        cand.append((1.0 - w) * arcsine + w * uniform)

    # 4) spikes added to arcsine (different total spike fractions)
    for spike_frac in (0.05, 0.10, 0.15, 0.20):
        mix = arcsine.copy()
        mix[0] += spike_frac / 2
        mix[-1] += spike_frac / 2
        mix = _project_onto_simplex(mix, 1.0)
        cand.append(mix)

    # 5) periodic spikes (mass concentrated on equally spaced points)
    for blocks in (5, 10):
        mix = np.full(n, 1.0 / n, dtype=np.float64)
        idx = np.arange(0, n, n // blocks)
        mass_spike = 0.4                     # 40 % of total mass to spikes
        spike_each = mass_spike / len(idx)
        mix[idx] = spike_each
        rest = 1.0 - mass_spike
        non_idx = np.setdiff1d(np.arange(n), idx)
        mix[non_idx] = rest / len(non_idx)
        mix = _project_onto_simplex(mix, 1.0)
        cand.append(mix)

    # 6) Dirichlet draws with various concentration parameters
    rng = np.random.default_rng()
    for conc in (0.3, 0.5, 0.8):
        for _ in range(2):
            cand.append(rng.dirichlet(np.full(n, conc)))

    # 7) previous best if its length matches
    if prev_seq is not None and prev_seq.size == n:
        cand.append(prev_seq.copy())

    # final sanity – all lie on the simplex
    return [_project_onto_simplex(c, 1.0) for c in cand]


# ----------------------------------------------------------------------
def _objective_from_array(a: np.ndarray) -> float:
    """
    Internal objective 2·n·max(conv) / (sum(a)^2).  For our projected vectors
    sum(a) is always 1, so the denominator is omitted.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxb = conv.max()
    return float(2.0 * n * maxb / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _subgradient_of_max_conv(a: np.ndarray,
                             conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution w.r.t. a (averaged over all
    active indices where conv attains its maximum).  Returns a vector of
    length len(a).
    """
    n = a.shape[0]
    max_val = conv.max()
    active = np.where(conv == max_val)[0]
    grad = np.zeros_like(a)
    m = len(active)

    for k in active:
        # indices i where a_i participates in conv[k]
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        idx = np.arange(start, end + 1, dtype=np.int64)
        # contribution a_{k-i}
        grad[idx] += a[k - idx]

    # factor 2 from derivative of a*a and average over active set
    if m > 0:
        grad = (2.0 / m) * grad
    return grad


# ----------------------------------------------------------------------
def _proj_subgradient_opt(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the simplex.
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _objective_from_array(a)

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[subgrad] n={n} it={it} score={best_score:.6f}")

        # exact sub‑gradient of the max‑convolution
        grad = _subgradient_of_max_conv(a, conv)

        # simple diminishing step size (√‑schedule)
        lr = 0.5 / math.sqrt(it + 1.0)

        a = a - lr * grad
        a = _project_onto_simplex(a, 1.0)

        # occasional tiny random jitter to escape plateaus
        if it % 3000 == 0:
            jitter = a + rng.normal(scale=1e-7, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = _objective_from_array(jitter)
            if jitter_score < best_score:
                a = jitter
                best_score = jitter_score
                best = jitter.copy()
                print(f"[subgrad] jitter improvement score={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a random fraction of the mass
    from one index to another and keep the move if it improves the score.
    """
    best = start_a.copy()
    best_score = _objective_from_array(best)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue

        # move up to 30 % of the mass at i to j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = _objective_from_array(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = _objective_from_array(jitter)
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point required by the grader.
    Returns a list of non‑negative heights that (hopefully) minimise
    ``evaluate_sequence`` within the allotted time.
    """
    np.random.seed(seed)
    # give ourselves a small safety margin for the final conversion / print
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1) decide which lengths to examine (2 k … 15 k, plus any known length)
    length_candidates = [2000, 4000, 6000, 8000, 10000, 12000, 15000]
    try:
        old_len = int(len(height_sequence_1))        # type: ignore[name-defined]
        length_candidates.append(old_len)
    except Exception:
        pass
    length_candidates = sorted(set(length_candidates))

    # ------------------------------------------------------------------
    # 2) quick exhaustive scan over hand‑crafted families
    best_overall_seq = None
    best_overall_score = float("inf")
    best_len = None

    for n in length_candidates:
        # use the previous solution as a warm‑start when lengths match
        prev = None
        try:
            tmp = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if tmp.shape[0] == n:
                prev = tmp
        except Exception:
            pass

        candidates = _candidate_shapes_extended(n, prev)

        for cand in candidates:
            scr = _objective_from_array(cand)
            if scr < best_overall_score:
                best_overall_score = scr
                best_overall_seq = cand.copy()
                best_len = n
                print(f"[scan] new best length={n} score={scr:.6f}")

        # early exit if we run out of time
        if time.time() >= deadline:
            break

    # fallback (should never happen)
    if best_overall_seq is None:
        n = 8000
        best_overall_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_len = n
        best_overall_score = _objective_from_array(best_overall_seq)

    # ------------------------------------------------------------------
    # 3) refined optimisation (projected sub‑gradient descent)
    remaining = max(0.0, deadline - time.time())
    subgrad_deadline = time.time() + remaining * 0.65  # 65 % of what’s left
    seq_opt, opt_score = _proj_subgradient_opt(best_overall_seq,
                                               best_len,
                                               subgrad_deadline)

    if opt_score < best_overall_score:
        best_overall_seq = seq_opt.copy()
        best_overall_score = opt_score
        print(f"[subgrad] improved best to {opt_score:.6f}")

    # ------------------------------------------------------------------
    # 4) stochastic hill‑climbing polishing (use any time left)
    hill_deadline = deadline
    seq_final, final_score = _hill_climb(best_overall_seq,
                                         best_len,
                                         hill_deadline)

    if final_score < best_overall_score:
        best_overall_seq = seq_final.copy()
        best_overall_score = final_score

    print(f"Final best score: {best_overall_score:.6f} (length={best_len})")
    return best_overall_seq.tolist()
