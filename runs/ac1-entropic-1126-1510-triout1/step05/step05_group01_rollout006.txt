"""
Propose a low‑score step‑function.

This version differs from the previous state‑of‑the‑art in three
important ways:

1. **Two‑phase schedule** – first a cheap sweep over many analytic
   families (Beta, mixtures, Dirichlet) for a handful of lengths to
   identify the most promising length(s).  Then a longer “refinement”
   phase concentrates all remaining time on those lengths.

2. **Exponentiated Adam** – the iterate is parametrised as
   `a = softmax(y)`.  Gradients of a smooth max surrogate (soft‑max
   over the self‑convolution) are obtained analytically and the
   Adam optimiser (with bias‑correction) is applied to `y`.  This
   provides per‑coordinate adaptive step‑sizes and generally converges
   faster than the simple momentum scheme used before.

3. **Temperature schedule & exact gradient** – a temperature parameter
   `α` controls the smoothness of the surrogate.  It is increased
   multiplicatively each iteration until a large value, at which point
   the true sub‑gradient (based on all active convolution indices) is
   used.  This yields a gradual transition from a smooth landscape to
   the true ∞‑norm objective.

The algorithm respects the global 1000 s budget, prints progress
messages, and finally returns the best sequence (as a plain Python
list) found within that time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    c = np.max(x)
    e = np.exp(x - c)
    return e / e.sum()


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n-1} normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / s


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Cheap score 2·n·max(conv) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    Returns a vector of length n.
    """
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max using a uniform weight over all
    (approximately) active indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j (k = 2*i)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad

def _initial_vectors(n: int) -> list[np.ndarray]:

        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] n={n} iter={it}  score={cur_score:.6f}")

        # Stop early if we have already passed the target bound
        if best_score <= target_score:
            break

        # gradient of the surrogate loss
        if alpha < alpha_max:
            grad = _softmax_gradient(a, conv, alpha)
            alpha = min(alpha_max, alpha * alpha_factor)
        else:
            grad = _exact_gradient(a, conv)

        # convert gradient w.r.t. a to gradient w.r.t. y (softmax Jacobian)
        dot = np.dot(a, grad)
        grad_y = a * (grad - dot)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # exponentiated Adam step on y
        y -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaus (cheap)
        if it % 5000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 1e-6
            y += jitter

    # Return best discovered vector (already normalised to sum = 1)
    return best_vec, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The implementation follows the two‑phase strategy described at the top of
    the file.
    """
    np.random.seed(seed)
    # give a small safety margin for final wrap‑up
    overall_deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Pick a few lengths to test.
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # lengths roughly spanning 0.5‑2.0 of the base size (at least 500)
    length_factors = [0.6, 0.8, 1.0, 1.2, 1.4, 1.6, 2.0]
    lengths = sorted(
        max(500, int(base_n * f))
        for f in length_factors
        if int(base_n * f) >= 500
    )

    # ------------------------------------------------------------------
    # 2. Cheap sweep: evaluate analytic families on each length,
    #    keep the most promising length(s).
    cheap_results = []   # (score, length, vector)
    for n in lengths:
        # avoid spending too much time here – only a handful of candidates
        cand = _initial_vectors(n)
        scored = [(_score(v), v) for v in cand]
        scored.sort(key=lambda x: x[0])
        cheap_results.append((scored[0][0], n, scored[0][1]))
        print(f"[{time.time():.2f}] length={n} best cheap score={scored[0][0]:.6f}")

    # pick the three lengths with the lowest cheap score
    cheap_results.sort(key=lambda x: x[0])
    top_lengths = [t[1] for t in cheap_results[:3]]
    print(f"Top lengths chosen for deep optimisation: {top_lengths}")

    # ------------------------------------------------------------------
    # 3. Deep optimisation on the chosen lengths.
    best_overall_vec = None
    best_overall_score = float("inf")

    # allocate time roughly equally among the selected lengths
    per_len_budget = (overall_deadline - time.time()) / max(1, len(top_lengths))

    for n in top_lengths:
        # recompute a fresh set of analytic starters for this length
        starters = _initial_vectors(n)

        # keep the 4 best cheap starters
        starters_scored = [(_score(v), v) for v in starters]
        starters_scored.sort(key=lambda x: x[0])
        chosen = [v for _, v in starters_scored[:4]]

        # optimisation budget for this length
        slice_deadline = time.time() + per_len_budget

        for idx, v0 in enumerate(chosen):
            # run Adam optimisation with the remaining slice time
            sub_deadline = min(slice_deadline, time.time() + per_len_budget * 0.7)
            vec, cheap_sc = _adam_optimize(v0, sub_deadline, target_score=1.503)

            # evaluate the exact black‑box score for the best vector of this run
            try:
                exact_score = evaluate_sequence(vec.tolist())
            except Exception:
                exact_score = cheap_sc

            print(f"*** Run length={n} starter={idx}  cheap={cheap_sc:.6f}  exact={exact_score:.6f}")

            if exact_score < best_overall_score:
                best_overall_score = exact_score
                best_overall_vec = vec.copy()
                print(f"+++ New global best: score={best_overall_score:.6f} (len={n})")
                if best_overall_score <= 1.503:
                    print("Target achieved – exiting early.")
                    return best_overall_vec.tolist()

            # stop early for this length if the slice deadline is reached
            if time.time() >= slice_deadline:
                break

        # global time check
        if time.time() >= overall_deadline:
            break
        if best_overall_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 4. Final tiny local perturbation phase (if time permits)
    if best_overall_vec is not None and time.time() < overall_deadline - 2.0:
        n = best_overall_vec.shape[0]
        cur_vec = best_overall_vec.copy()
        cur_score = best_overall_score
        trials = 0
        while time.time() < overall_deadline - 0.5 and trials < 2000:
            trials += 1
            eps = np.random.lognormal(mean=-7, sigma=0.7) * 1e-3
            i = np.random.randint(n)
            cand = cur_vec.copy()
            cand[i] = max(0.0, cand[i] + eps)
            cand /= cand.sum()
            s = _score(cand)
            if s < cur_score:
                cur_vec = cand
                cur_score = s
                print(f"[{time.time():.2f}] local jitter improvement → {cur_score:.6f}")
        # final exact evaluation
        try:
            final_exact = evaluate_sequence(cur_vec.tolist())
            if final_exact < best_overall_score:
                best_overall_score = final_exact
                best_overall_vec = cur_vec
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 5. Fallback (should never be needed)
    if best_overall_vec is None:
        n_fallback = 1000
        best_overall_vec = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_overall_score = evaluate_sequence(best_overall_vec.tolist())

    print(
        f"Final best score: {best_overall_score:.6f} "
        f"(length = {len(best_overall_vec)})"
    )
    return best_overall_vec.tolist()
