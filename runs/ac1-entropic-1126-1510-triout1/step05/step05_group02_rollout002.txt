def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0

    # lengths to try
    lengths = [2000, 4000, 5000, 7000]   # maybe
    best_seq = None
    best_score = float('inf')
    best_len = None

    # generate base candidates for each length
    for n in lengths:
        # gather candidates
        candidates = []
        # uniform
        candidates.append(np.full(n, 1.0/n))
        # arcsine
        candidates.append(_beta_distribution(n, 0.5, 0.5))
        # Beta low alpha
        for a in [0.02, 0.05, 0.1, 0.2]:
            candidates.append(_beta_distribution(n, a, a))
        # random mixtures (30)
        for _ in range(30):
            a1,b1,a2,b2 = np.random.rand(4)*4.5+0.05 # [0.05,4.55]
            lam = np.random.rand()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            mix = lam * p1 + (1-lam) * p2
            candidates.append(mix)
        # if height_sequence_1 available: interpolate
        if 'height_sequence_1' in globals():
            try:
                base = np.asarray(height_sequence_1, dtype=np.float64)
                cand = _interpolate_to_length(base, n)
                candidates.append(cand)
            except Exception:
                pass

        # evaluate raw scores
        for cand in candidates:
            cand = _project_onto_simplex(cand)
            raw = _raw_score(cand)
            if raw < best_score:
                best_score = raw
                best_seq = cand.copy()
                best_len = n

    print(f"Initial best raw score {best_score:.6f} at n={best_len}")

    # now refinement using softmax gradient descent
    a = best_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_obj = best_score
    # Adam params
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # schedule of beta (softmax temperature)
    betas = [2, 5, 10, 30, 80, 200, 500]
    it = 0
    for beta in betas:
        # number of iterations per beta maybe up to 100k but break if deadline
        # define steps length: maybe 20000 for early, less later.
        # Could use while time < subdeadline and iteration less than max per stage.
        max_iters = 20000
        for _ in range(max_iters):
            if time.time() >= deadline:
                break
            it += 1
            grad, cur_max = _softmax_gradient_step(a, beta)
            # Adam update
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad*grad)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            step = 0.1 * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = _project_onto_simplex(a)

            # occasional jitter
            if it % 2000 == 0 and np.random.rand() < 0.05:
                a += np.random.randn(n) * 1e-6
                a = _project_onto_simplex(a)

            cur_val = 2.0 * n * _fft_self_convolve_max(a)
            if cur_val < best_obj - 1e-12:
                best_obj = cur_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] iter {it}, beta {beta}, new best {best_obj:.6f}")

    final_seq = best_a.tolist()

"""
Propose a low‑score step‑function.

The algorithm works in three phases:

1. **Broad exploration** – sample many simple families (uniform,
   arcsine, symmetric Beta and random two‑component Beta mixtures) at
   several lengths and keep the best raw candidate (the objective
   without the final evaluate_sequence scaling).

2. **Smooth soft‑max gradient descent** – minimise a smooth surrogate
   of the ℓ∞ self‑convolution norm:
       F_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)
   for a sequence `a` on the probability simplex.
   The gradient of `F_β` is a weighted version of the sub‑gradient
   used in the reference code.  An Adam optimiser with a schedule of
   increasing β (2 → 5 → 10 → 30 → 80 → 200 → 500) drives the
   convolution values toward a flat profile.

3. **Final refinement** – a few iterations of the pure max‑sub‑gradient
   descent (the method from the reference) polish the solution.

All operations respect the simplex constraint Σ a_i = 1, a_i ≥ 0.
The routine stops automatically before the 1000 s budget expires and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    # Fast path for already feasible vectors
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow/underflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _next_pow_two(k: int) -> int:
    """Small helper to get the next power of two ≥ k."""
    return 1 << ((k - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum value of a * a (self‑convolution)."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = _next_pow_two(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Score used during the search (sum is forced to 1)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)


# ----------------------------------------------------------------------
def _softmax_gradient_step(a: np.ndarray, beta: float):
    """
    Gradient of the smooth surrogate
        F_β(a) = (1/β) log Σ_k exp(β·(a∗a)_k)
    Returns (grad, current_max_convolution).
    """
    n = a.shape[0]
    # self‑convolution (full length 2n‑1)
    conv = _fft_self_convolve_full(a)

    # soft‑max weights
    m = conv.max()
    # subtract m for numerical stability
    exp_vals = np.exp(beta * (conv - m))
    w = exp_vals / exp_vals.sum()           # length 2n‑1

    # gradient = 2 * (a_rev ∗ w) restricted to the central n entries
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w)     # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]

    return grad, conv.max()


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of the exact max objective (used as a final polishing step).
    Returns (grad, cur_max).
    """
    n = a.shape[0]
    b = _fft_self_convolve_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:

    The routine combines a wide random sampling of Beta families,

    Returns a plain Python list ready for the black‑box evaluator.
    """

        np.linspace(0.02, 0.08, 5, endpoint=False),
