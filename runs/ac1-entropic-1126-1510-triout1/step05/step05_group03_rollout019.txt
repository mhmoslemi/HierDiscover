def lp_max_sum(a, M):
    n = a.shape[0]
    A = build_conv_matrix(a)
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximize sum -> minimize -sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x
```

def linear_program_iterate(a0, max_iters=10, t=0.2, time_deadline=None):
    a = a0.copy()
    n = a.shape[0]
    current_max = _fft_self_convolve(a).max()
    best_a = a.copy()
    best_score = 2.0 * n * current_max
    for i in range(max_iters):
        if time_deadline is not None and time.time() > time_deadline:
            break
        g0 = lp_max_sum(a, current_max)
        if g0 is None:
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0  # rescale to sum 1
        # blend
        a = (1 - t) * a + t * g
        # ensure positivity and sum=1 (maybe numerical)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        # recompute max
        new_max = _fft_self_convolve(a).max()
        if new_max < current_max - 1e-12:
            current_max = new_max
            if 2.0 * n * current_max < best_score:
                best_score = 2.0 * n * current_max
                best_a = a.copy()
        else:
            # no improvement, maybe break
            break
    return best_a, best_score
```

"""
Low‑score step‑function generator.

The algorithm works in four stages:

1. **Initial seed** – a modest‑size (≈1500) distribution obtained from the
   arcsine law plus a small random perturbation.

2. **Linear‑programming refinement** – repeatedly solve a LP that maximises
   the ℓ1‑mass of a new candidate *g* under the constraint
   `a * g ≤ M` (where `M = max(a*a)`).  Scaling the optimizer back to a
   probability vector strictly reduces the peak of the self‑convolution.
   This is the “Frank‑Wolfe‑style” step described in the statement.

3. **Upsampling + smooth‑max gradient descent** – linearly interpolate the
   improved vector to a larger length (≈4000) and run Adam on a soft‑max
   surrogate of the peak, followed by a short exact‑max sub‑gradient polishing.

4. **Simulated‑annealing polishing** – final stochastic moves that explore
   the neighbourhood of the best vector found so far.

All intermediate vectors are projected onto the simplex
`{x ≥ 0, Σx = 1}`.  The routine respects the 1000 s budget and prints
progress information.  The returned Python list can be handed directly to
`evaluate_sequence`.
"""

import math
import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate:
        Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses a uniform weight on the active set (indices attaining the current maximum).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short exact‑max refinement using sub‑gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a

def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Simple simulated‑annealing polishing.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak
        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

def _dense_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Return the dense matrix A where (A @ g)_k = (a * g)_k."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        A[k, i_start:i_end + 1] = a[i_start:i_end + 1]
    return A


# ----------------------------------------------------------------------
def _lp_optimize(a: np.ndarray, M: float):
    """
    Solve
        maximise Σ g_j
        s.t.   a * g ≤ M     (component‑wise)
               g ≥ 0
    Returns the optimal g (not normalised).  If the LP fails, ``None`` is returned.
    """
    n = a.shape[0]
    A = _dense_conv_matrix(a)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise → minimise -Σg
    bounds = [(0.0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
def _lp_iterate(a0: np.ndarray, max_iters: int, deadline: float):
    """
    Frank‑Wolfe style iteration based on the LP described in the
    problem statement.  Returns the best vector found and its fast‑score.
    """
    a = a0.copy()
    n = len(a)
    cur_max = _fft_self_convolve(a).max()
    best_a = a.copy()
    best_score = 2.0 * n * cur_max

    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = _lp_optimize(a, cur_max)
        if g is None:
            break
        total = g.sum()
        if total <= 1.0 + 1e-12:          # cannot increase mass → stop
            break
        # Scale to a probability vector and project (tiny numerical errors may appear)
        a = g / total
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, s=1.0)
        cur_max = _fft_self_convolve(a).max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
        # optional early exit if improvement stalls
        if it % 5 == 4 and abs(cur_score - best_score) < 1e-12:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Combines a linear‑programming warm‑start, Adam optimisation,
    exact sub‑gradient polishing and a final simulated‑annealing pass.
    Returns a plain Python list suitable for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 0. A short helper: a small random perturbation of a known construction
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # 1. Start from a modest size (good for the LP) – 1500 for memory comfort
    n_lp = 1500
    if base_seq is not None:
        a = _interpolate_to_length(base_seq, n_lp)
    else:
        a = _arcsine_distribution(n_lp)
    # add a tiny random component to avoid pathological symmetry
    a += 0.01 * np.random.rand(n_lp)
    a = _project_onto_simplex(a, s=1.0)

    print("[phase 0] initial candidate (n = {})".format(n_lp))

    # ------------------------------------------------------------------
    # 2. Linear‑programming refinement (≈30 % of the budget)
    time_left = deadline - time.time()
    if time_left > 30:
        lp_deadline = time.time() + time_left * 0.30
        a, lp_score = _lp_iterate(a, max_iters=25, deadline=lp_deadline)
        print("[phase 1] LP refinement → fast_score={:.6f}".format(lp_score))
    else:
        print("[phase 1] skipped (insufficient time)")

    # ------------------------------------------------------------------
    # 3. Upsample to a larger length and perform smooth‑max gradient descent
    time_left = deadline - time.time()
    if time_left > 30:
        target_n = max(3500, int(n_lp * 2.4))
        a = _interpolate_to_length(a, target_n)
        a = _project_onto_simplex(a, s=1.0)
        print("[phase 2] upsampled to n = {}".format(target_n))

        # Adam optimisation on the smooth‑max surrogate
        phase2_deadline = time.time() + time_left * 0.45
        a_opt, _ = _adam_optimize(a, phase2_deadline)

        # Exact‑max sub‑gradient polishing (short)
        phase2b_deadline = time.time() + time_left * 0.15
        a_opt = _refine_max(a_opt, phase2b_deadline)

        a = a_opt
        fast_score = 2.0 * len(a) * _fft_self_convolve(a).max()
        print("[phase 2] after gradient/polish fast_score={:.6f}".format(fast_score))
    else:
        print("[phase 2] skipped (insufficient time)")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing polishing (use whatever time remains)
    time_left = deadline - time.time()
    if time_left > 5:
        print("[phase 3] simulated annealing – time left {:.1f}s".format(time_left))
        a_final, final_peak = _simulated_annealing(a, deadline)
        a = a_final
        final_fast = 2.0 * len(a) * final_peak
        print("[phase 3] SA fast_score={:.6f}".format(final_fast))
    else:
        print("[phase 3] skipped (insufficient time)")

    # ------------------------------------------------------------------
    # 5. Final true evaluation
    try:
        true_score = evaluate_sequence(a.tolist())
        print("[final] evaluate_sequence score={:.6f}".format(true_score))
    except Exception as exc:
        true_score = float("inf")
        print("[final] evaluate_sequence raised:", exc)

    print("Best approximate score = {:.6f} (length = {})".format(
        2.0 * len(a) * _fft_self_convolve(a).max(), len(a)))
    return a.tolist()
