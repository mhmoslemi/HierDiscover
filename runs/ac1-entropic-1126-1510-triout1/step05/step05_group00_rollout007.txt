Propose a low‑score non‑negative step function.

Algorithm
---------
1. Choose a length `n` (default: length of the previous best construction,
   otherwise 4000).  Generate a suite of simple analytic families
   (arcsine, symmetric beta, triangular, exponential) and keep the best
   one (according to the exact objective).

2. Refine the chosen distribution with a *soft‑max* approximation of the
   max‑convolution.  The smooth objective is  

       f(a) = (1/β)·log Σ_k exp(β·(a*a)_k) ,

   whose gradient can be written in closed form using a convolution.
   We minimise `f` on the simplex using the Adam optimiser with Euclidean
   projection after each step.  The smoothing parameter β is increased
   slowly from 0.5 to ≈200 so that the optimisation first follows a smooth
   landscape and then focuses on the true max‑operator.

3. After the Adam phase, a cheap hill‑climbing stage moves tiny amounts of
   mass between random positions, accepting only improving moves.
   Occasionally a small random jitter is added to escape plateaus.

The whole procedure respects the 1000 s budget, works for any length,
and returns a list of non‑negative floats that can be fed directly to
`evaluate_sequence`.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full autocorrelation a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle, normalised to sum 1."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Return the objective value 2·n·max(a*a) assuming Σa = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises the evaluation."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # 1.  Choose a length.
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 4000
    n = max(500, int(base_n))

    # --------------------------------------------------------------
    # 2.  Build a small library of analytic candidates.
    candidates = []

    # classic families
    candidates.append(_arcsine_distribution(n))
    candidates.append(_triangular_distribution(n))
    candidates.append(_exponential_distribution(n, lam=0.005))
    candidates.append(_exponential_distribution(n, lam=0.01))

    # symmetric beta with a wide range of shape parameters (including very
    # spiky distributions)
    for p in [0.01, 0.02, 0.05, 0.10, 0.20, 0.30, 0.40,
              0.50, 0.70, 1.00, 1.50, 2.00]:
        candidates.append(_beta_symmetric_distribution(n, p))

    # keep the best analytic shape
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.1f}] analytic #{idx} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Adam + soft‑max refinement on the best length.
    #    The variables are kept on the simplex (Σa = 1, a_i ≥ 0).
    a = best_seq.copy()
    cur_conv = _fft_self_convolve(a)
    cur_score = 2.0 * n * cur_conv.max()
    best_seq = a.copy()
    best_score = cur_score

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.05                     # base learning rate
    it = 0

    # schedule for the soft‑max temperature (larger → sharper)
    beta_start = 0.5
    beta_end = 200.0

    while time.time() < deadline:
        it += 1

        # ----  gradient of the smooth max (soft‑max) ----------------
        max_conv = cur_conv.max()
        # smoothness parameter grows linearly with the iteration count
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 200000.0)

        shifted = cur_conv - max_conv          # for numerical stability
        exp_vals = np.exp(beta * shifted)
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or np.isnan(sum_exp):
            # fallback to hard max (active set) if soft‑max collapses
            w = np.zeros_like(exp_vals)
            w[np.argmax(cur_conv)] = 1.0
        else:
            w = exp_vals / sum_exp

        # gradient = 2·(a_rev * w)_{central n entries}
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        # ----  Adam update + projection -----------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # ----  evaluate new point -----------------------------------
        cur_conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * cur_conv.max()

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam iter={it} new best {best_score:.6f}")

        # early exit if target reached
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 4.  Local random‑transfer hill climbing (purely combinatorial).
    a = best_seq.copy()
    cur_conv = _fft_self_convolve(a)
    cur_score = 2.0 * n * cur_conv.max()
    best_score = cur_score
    best_seq = a.copy()
    it = 0
    max_transfer = 0.05      # 5 % of total mass per trial (the sum is 1)

    while time.time() < deadline:
        it += 1
        # gradually shrink transfer amount
        transfer = max_transfer * (0.5 ** (it / 20000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_conv = _fft_self_convolve(a_new)
        new_score = 2.0 * n * new_conv.max()

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] transfer iter={it} improved best {best_score:.6f}")

        # occasional jitter to escape flat regions
        if it % 5000 == 0 and np.random.rand() < 0.1:
            jitter = np.random.randn(n) * 1e-4
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_conv = _fft_self_convolve(a)
            cur_score = 2.0 * n * cur_conv.max()

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
