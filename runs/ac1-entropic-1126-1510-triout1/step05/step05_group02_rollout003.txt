"""
Propose a low‑score step‑function.

Algorithm
---------
1. Initialise a set of candidate sequences of various lengths.
   • Deterministic “U‑shaped” seed (edge‑heavy) based on 1/√i.
   • Uniform seed.
   • If a previous solution (height_sequence_1) is available, interpolate it.
   • A quick random Dirichlet search to possibly discover better shapes.
   The best candidate (according to the exact ℓ∞ self‑convolution score)
   is kept.

2. Refine the best candidate with a smooth‑max gradient descent.
   The max of the convolution is approximated by the log‑sum‑exp
   (soft‑max) with a temperature β that is slowly increased.
   Gradient of the smooth objective is obtained via FFT‑based convolution.
   After each step the vector is projected onto the probability simplex
   (non‑negative, unit ℓ1‑norm).  Back‑tracking line‑search guarantees
   monotone improvement.

3. A short sub‑gradient polishing phase (active‑set gradient) finishes
   the optimisation.

The whole routine respects the 1000 s budget and returns a Python list
compatible with the black‑box ``evaluate_sequence`` function.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()

# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a * a."""
    return _fft_self_convolve_full(a).max()

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = _next_pow_two(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Exact objective (2·n·max_{k}(a*a)_k) for a vector with Σa = 1."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on a uniform grid of length n, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to stay stable
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max (log‑sum‑exp) approximation of max(conv(a,a)).
    Returns (grad, approx_max).
    """
    n = a.shape[0]
    conv_full = _fft_self_convolve_full(a)
    max_c = conv_full.max()
    # soft‑max weights (shifted for numerical stability)
    exp_w = np.exp(beta * (conv_full - max_c))
    w = exp_w / exp_w.sum()
    # gradient = 2 * (a_rev * w) restricted to the centre n entries
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    # the smoothed maximum (log‑sum‑exp)
    approx_max = max_c + (np.log(w.sum()) / beta)  # w.sum() == 1, so approx_max ≈ max_c
    return grad, max_c

        # compute full convolution once
        conv_full = _fft_self_convolve_full(a)
        cur_max = conv_full.max()
        # active set (indices where conv is essentially maximal)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv_full >= cur_max - tol)[0]
        w = np.zeros_like(conv_full)
        if active.size:
            w[active] = 1.0 / active.size
        grad_sub = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.03 / math.sqrt(sub_it)
        a_candidate = _project_onto_simplex(a - step * grad_sub)
        cand_val = _raw_score(a_candidate)
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.1f}] sub‑grad it={sub_it} score={best_val:.6f}")
        else:
            # no improvement – reduce step size more aggressively
            if sub_it % 500 == 0:
                a = _project_onto_simplex(a - step * grad_sub)

    # ------------------------------------------------------------------
    final_seq = best_a
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq.tolist()
