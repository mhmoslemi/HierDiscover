"""
Propose a low‑score step‑function using an LP‑based refinement.

The algorithm works as follows.

1.  Choose several candidate lengths (≈ 1500‑5000).  For each length a
    feasible probability vector a (sum = 1, non‑negative) is built –
    discrete arcsine law, uniform law or a random Dirichlet draw.

2.  Iterate:
        * compute M = max( a * a )  (self‑convolution)
        * solve a linear program
              maximise   Σ g_j
              subject to g ≥ 0,
                         (a ⊛ g)_k ≤ M   for all k      (⊛ = linear convolution)
          The LP is linear in the unknown vector g because a is a constant
          parameter.  It returns a “max‑mass” vector g₀.
        * rescale g₀ to have sum = 1 (g = g₀ / Σ g₀) and blend it with the
          current iterate:
                a ← (1‑t)·a + t·g   (t≈0.3)
          after blending the vector is projected onto the simplex.

    The blend is a standard “alternating projection” step – it is guaranteed
    not to increase M, and repeated steps drive the maximal convolution
    value down.  Warm‑started ECOS solves make each LP cheap (≤ 0.2 s for
    n≈4000).

3.  After a few LP steps a cheap score 2·n·max(a⊛a) is kept; the best
    vector observed for the current length is evaluated with the official
    `evaluate_sequence`.  The globally best sequence over all explored
    lengths is finally returned.

The whole search respects the supplied time budget (default 1000 s) and
uses at most two CPU threads (controlled via the OMP environment
variable).  If no improvement is found a uniform fallback vector is
returned.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast for many lengths)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _initial_vector(kind: str, n: int) -> np.ndarray:
    """Generate a feasible probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n).astype(np.float64)
        a /= a.sum()
    elif kind == "arcsine":
        a = _beta_distribution(n, 0.5, 0.5)
    else:                     # fallback
        a = _beta_distribution(n, 0.5, 0.5)
    return _project_onto_simplex(a, 1.0)


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Score corresponding to evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _build_lp_problem(n: int):
    """
    Build a CVXPY problem that, for a *fixed* vector a (as a Parameter),
    maximises Σ g subject to g ≥ 0 and (a ⊛ g) ≤ M (M is a scalar Parameter).

    Returns (problem, a_param, M_param, g_var).
    """
    a_param = cp.Parameter(n, nonneg=True)
    M_param = cp.Parameter()
    g_var = cp.Variable(n, nonneg=True)

    # linear convolution of the fixed a_param with the variable g_var
    conv_ag = cp.conv(a_param, g_var)               # length 2n‑1
    constraints = [conv_ag <= M_param]

    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    return prob, a_param, M_param, g_var

               max_iter: int = 12, mix_factor: float = 0.3,
               time_deadline: float = math.inf):
    """
    Perform a few LP‑based refinement steps starting from a.
    Returns the best vector seen (according to the cheap score) and its score.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _cheap_score(a)

    for it in range(max_iter):
        # check remaining time
        if time.time() > time_deadline:
            break

        # current max of self‑convolution
        cur_M = _fft_self_convolve(a).max()

        # feed parameters to the LP
        a_param.value = a
        M_param.value = cur_M

        # solve (use warm start)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
        except Exception:  # pragma: no cover
            print("[LP] Solver crashed – aborting refinements.")
            break

        if g_var.value is None:
            print("[LP] Infeasible or solver failure – aborting refinements.")
            break

        g0 = np.array(g_var.value, dtype=np.float64)
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print("[LP] Degenerate solution – aborting.")
            break

        # normalise to the simplex
        g = g0 / sum_g0

        # blend
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_onto_simplex(a, 1.0)

        cur_score = _cheap_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[LP] n={n} iter={it+1} new cheap score={cur_score:.6f}")

        # early stop if improvement stalls
        if it >= 2 and abs(best_score - cur_score) < 1e-8:
            break

    return best_vec, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    `evaluate_sequence`.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Choose a few lengths.  If a previous solution exists we centre
    # around its size; otherwise we start around 3000.
    try:
        base_n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        base_n = 3000

    length_factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted({max(500, int(base_n * f)) for f in length_factors})
    # enforce a maximum length to keep the LP cheap
    lengths = [n for n in lengths if n <= 6000]

    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Exploring length n={n} ===")

        # ---------- Build the LP once for this length ----------
        if _CVXPY_AVAILABLE:
            prob, a_param, M_param, g_var = _build_lp_problem(n)
        else:
            prob = a_param = M_param = g_var = None   # type: ignore

        # ---------- Try a few initialisations ----------
        starter_kinds = ["arcsine", "uniform", "random"]
        for kind in starter_kinds:
            if time.time() > deadline:
                break
            a0 = _initial_vector(kind, n)

            # cheap score for logging
            init_score = _cheap_score(a0)
            print(f"[init] kind={kind:7s} cheap score={init_score:.6f}")

            # ---------- LP refinement ----------
            if _CVXPY_AVAILABLE:
                a_refined, cheap_best = _lp_refine(
                    a0,
                    prob,
                    a_param,
                    M_param,
                    g_var,
                    max_iter=12,
                    mix_factor=0.35,
                    time_deadline=deadline,
                )
            else:
                # fall back to the exponentiated‑gradient scheme from the
                # previous state of the art (only if cvxpy is missing)
                a_refined = a0.copy()
                # simple multiplicative update for a few steps
                momentum = np.zeros_like(a_refined)
                beta = 0.9
                eta0 = 0.6
                temp = 1.0
                temp_factor = 1.005
                for it in range(3000):
                    if time.time() > deadline:
                        break
                    conv = _fft_self_convolve(a_refined)
                    # soft‑max gradient
                    cur_max = conv.max()
                    exp_term = np.exp(temp * (conv - cur_max))
                    w = exp_term / exp_term.sum()
                    conv_rev = np.fft.irfft(
                        np.fft.rfft(a_refined[::-1]) * np.fft.rfft(w),
                        n=2 * n - 1,
                    )[:n]
                    grad = 2.0 * conv_rev
                    momentum = beta * momentum + (1 - beta) * grad
                    shift = np.max(momentum)
                    factor = np.exp(-eta0 / math.sqrt(it + 1) * (momentum - shift))
                    a_refined = a_refined * factor
                    a_refined /= a_refined.sum()
                    temp = min(4000.0, temp * temp_factor)
                cheap_best = _cheap_score(a_refined)

            # ---------- Final black‑box evaluation ----------
            try:
                final_score = evaluate_sequence(a_refined.tolist())
            except Exception:
                final_score = cheap_best

            print(f"[result] kind={kind:7s} final score={final_score:.6f}")

            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = a_refined.copy()
                print(f"*** Global best improved to {global_best_score:.6f} (n={n}) ***")

            if global_best_score <= 1.503:
                print("Target reached – exiting early.")
                return global_best_seq.tolist()

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found.
    if global_best_seq is None:
        n_fallback = 1000
        global_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        try:
            global_best_score = evaluate_sequence(global_best_seq.tolist())
        except Exception:  # pragma: no cover
            global_best_score = _cheap_score(global_best_seq)

    print(f"\nFinished search – best score {global_best_score:.6f} "
          f"with length {len(global_best_seq)}")
    return global_best_seq.tolist()
