```
def max_self_convolution(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:2*n-1].max()
```

"""
Propose a low‑score step function.

The algorithm works in two phases.

1. **Broad random search** – for several sequence lengths we generate
   many candidates:
   * uniform / arcsine (Beta(0.5,0.5))
   * symmetric Beta(α,α) on a dense grid
   * random Beta(α,β) draws (α,β log‑uniform)
   * Dirichlet draws with random concentration
   * mixtures of two or three Beta components (random weights)
   * a mixture of a Beta component with a uniform component
   All candidates are projected onto the simplex (Σa_i = 1, a_i ≥ 0) and
   evaluated by the fast FFT‑based proxy  2·n·max(a*a).  The best candidate
   over the whole exploration is kept.

2. **Local refinement** – the best candidate is improved with a projected
   sub‑gradient method on the exact max‑convolution objective.
   A momentum term (exponential moving average of past gradients) and a
   decreasing step size (∝1/√iteration) are used.  Every few hundred
   iterations a tiny random jitter is added to help escape plateaus.
   The refinement runs until the time budget is exhausted.

The final sequence (sum≈1) is returned; ``evaluate_sequence`` rescales it
automatically, so the score printed at the end equals the true objective.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalized to Σ=1."""
    # mid‑points of the bins
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf (stable)
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # clip extreme values to avoid overflow in exp
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()

def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score assuming Σa = 1 :  2·n·max(a*a).
    This equals the true objective because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    # tolerance for the active set
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluate_sequence score.
    The routine performs a large random search over several families of
    distributions, then refines the best candidate with a projected
    sub‑gradient + momentum method.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered sequence
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = base_seq / base_seq.sum()
    except Exception:
        base_seq = None

    best_seq = None
    best_score = float("inf")
    best_len = None

    # ------------------------------------------------------------------
    # Lengths to explore – larger length gives a smoother shape but costs a bit more time
    lengths = [2000, 4000, 8000, 12000]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # ---- deterministic baselines -------------------------------------------------
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_score, best_seq, best_len = sc, uniform.copy(), n
            print(f"   uniform score={best_score:.6f}")

        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine)
        if sc < best_score:
            best_score, best_seq, best_len = sc, arcsine.copy(), n
            print(f"   arcsine score={best_score:.6f}")

        if base_seq is not None:
            cand = _interpolate_to_length(base_seq, n)
            sc = _raw_score(cand)
            if sc < best_score:
                best_score, best_seq, best_len = sc, cand.copy(), n
                print(f"   from base score={best_score:.6f}")

        # ---- symmetric Beta grid -----------------------------------------------------
        alphas = np.concatenate((
            np.linspace(0.12, 0.6, 12, endpoint=False),
            np.linspace(0.6, 2.0, 15),
            np.linspace(2.0, 5.0, 8)
        ))
        for a in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a, a)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   symBeta a={a:.3f} score={best_score:.6f}")

        # ---- random Beta draws -------------------------------------------------------
        for _ in range(500):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
            beta = 10 ** rng.uniform(-2, 1)
            vec = _beta_distribution(n, alpha, beta)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   randBeta a={alpha:.3f} b={beta:.3f} score={best_score:.6f}")

        # ---- Dirichlet draws ---------------------------------------------------------
        for _ in range(500):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-2, 1)   # concentration 0.01 … 10
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   dirichlet conc={conc:.3f} score={best_score:.6f}")

        # ---- mixture of two Betas ----------------------------------------------------
        for _ in range(300):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mix2 λ={lam:.3f} score={best_score:.6f}")

        # ---- mixture of three Betas --------------------------------------------------
        for _ in range(200):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a3, b3 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            weights = rng.dirichlet([1, 1, 1])
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            p3 = _beta_distribution(n, a3, b3)
            vec = weights[0] * p1 + weights[1] * p2 + weights[2] * p3
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mix3 score={best_score:.6f}")

        # ---- uniform + Beta mixture ---------------------------------------------------
        for _ in range(200):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)
            beta = 10 ** rng.uniform(-2, 1)
            beta_vec = _beta_distribution(n, alpha, beta)
            lam = rng.random()
            vec = lam * beta_vec + (1.0 - lam) * uniform
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mixU λ={lam:.3f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        # ultra‑fallback (should never happen)
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_len = n
        best_score = _raw_score(best_seq)
        print("Fallback uniform used.")

    print(f"Best raw score after exploration: {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # 5) Local refinement with projected sub‑gradient + momentum
    a = best_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = best_score
    it = 0
    momentum = np.zeros_like(a)
    mu = 0.9  # momentum coefficient

    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        # step size schedule – aggressive early, slowly diminishing
        step = 0.25 / math.sqrt(it)
        # momentum update
        momentum = mu * momentum + (1.0 - mu) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional tiny random jitter to escape flat regions
        if it % 500 == 0 and rng.random() < 0.07:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] refinement it={it}  score={best_val:.6f}")

    # ------------------------------------------------------------------
    final_seq = best_a
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq.tolist()
