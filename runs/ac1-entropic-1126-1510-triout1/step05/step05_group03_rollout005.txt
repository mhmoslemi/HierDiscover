   * symmetric Beta(α,α) densities,
   * mixtures of arcsine and uniform.
   Every candidate is projected onto the simplex (Σa_i=1) and its
   fast surrogate 2·n·max(a*a) is evaluated via an FFT‑based convolution.
   The best candidate is kept.

2. **Gradient‑Based Refinement** – Starting from the best candidate,
   run Adam on a smooth log‑sum‑exp surrogate of the ℓ∞‑convolution.
   The temperature (α) is increased slowly up to a large value
   (≈10 000) so that the surrogate approaches the true maximum.
   The best exact peak seen during Adam is retained.

3. **Exact ℓ∞ Sub‑gradient Polishing** – A short descent using the true
   ℓ∞‑sub‑gradient (uniform weight on the active set) further reduces
   the peak.

4. **Stochastic Annealing** – A simulated‑annealing phase moves a small
   fraction of probability mass between random positions, accepting
   uphill moves with an exponential temperature schedule.  The move
   size is allowed up to 10 % of the source mass to explore more widely.

5. **Final Evaluation** – The resulting vector is handed to the
   black‑box `evaluate_sequence`.  The returned list is guaranteed to be
   non‑negative and to sum to a positive number (scaling does not affect
   the score).

All heavy lifting (convolutions, gradients) is done with real FFTs for
speed, and every step respects the simplex constraint.  The routine
uses at most the allotted 1000 s and prints progress information.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalized to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    # Clip infinities that arise when α or β < 1
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _powerlaw_distribution(n: int, beta: float) -> np.ndarray:
    """Distribution a_i ∝ ((i+0.5)*(n-i-0.5))^{-beta}, normalized."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    # avoid division by zero – the denominator is never zero for interior points
    vals = denom ** (-beta)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # numerically stable soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)      # length = 3n‑2
    start = len(a) - 1
    grad = 2.0 * conv_rev[start:start + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.006,
                   alpha_max: float = 10000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector seen and the corresponding exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # update best exact peak
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # temperature (smoothness) scheduled increase
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # tiny random jitter every now and then to avoid stagnation
        if t % 400 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 8e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses a uniform weight on the active set (indices attaining the current maximum).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1:n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short exact‑max refinement using sub‑gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.015 / np.sqrt(it)          # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 6e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Simple simulated‑annealing polishing.
    Moves a small amount of mass between two random positions,
    accepting uphill moves with an exponential temperature schedule.
    Returns the best vector found and its peak convolution.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    # temperature schedule (exponential decay)
    start_temp = max(cur_peak, 1e-4) * 0.7
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # pick a source with non‑zero mass
        src = np.random.randint(n)
        if cur_a[src] <= 1e-14:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        # move up to 10 % of the source mass
        max_delta = min(cur_a[src], 0.10)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start_time
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional reset to the best seen so far
        if iter_cnt % 25000 == 0 and np.random.rand() < 0.07:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns a plain Python list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if present.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # ------------------------------------------------------------------
    # Choose a handful of lengths to explore.
    lengths = [
        max(500, int(base_n * f))
        for f in (0.6, 0.8, 1.0, 1.2, 1.4)
        if int(base_n * f) >= 500
    ]
    if not lengths:
        lengths = [max(500, base_n)]

    # ------------------------------------------------------------------
    # ---- 1. BROAD SEARCH ------------------------------------------------
    print("[phase 1] broad search for a promising start point")
    best_vec = None
    best_fast = np.inf

    # parameter grids
    beta_grid = np.arange(0.30, 0.81, 0.025)          # power‑law exponent
    alpha_grid = np.arange(0.20, 1.02, 0.04)         # symmetric Beta α=β
    mix_weights = np.linspace(0.0, 1.0, 21)          # arcsine↔uniform mix

    for n in lengths:
        if time.time() >= deadline:
            break

        # a small set of deterministic starters
        starters = {
            "arcsine": _arcsine_distribution(n),
            "uniform": np.full(n, 1.0 / n, dtype=np.float64),
        }

        # power‑law family
        for beta in beta_grid:
            starters[f"pow_{beta:.3f}"] = _powerlaw_distribution(n, beta)

        # symmetric Beta family
        for a in alpha_grid:
            starters[f"beta_{a:.2f}"] = _beta_distribution(n, a, a)

        # mixtures of arcsine and uniform
        arc = _arcsine_distribution(n)
        uni = np.full(n, 1.0 / n, dtype=np.float64)
        for w in mix_weights:
            starters[f"mix_{w:.2f}"] = (1 - w) * arc + w * uni

        # reuse previous solution if available
        if base_seq is not None:
            starters["prev"] = _interpolate_to_length(base_seq, n)

        # evaluate each starter (fast surrogate)
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)   # ensure simplex
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < best_fast - 1e-12:
                best_fast = fast_score
                best_vec = vec.copy()
                print(f"[init] n={n:4d} {tag:<12s} fast_score={fast_score:.6f}")

    # fallback
    if best_vec is None:
        n = max(500, base_n)
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_fast = 2.0 * n * _fft_self_convolve(best_vec).max()
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # ---- 2. GRADIENT‑BASED REFINEMENT ---------------------------------
    time_left = deadline - time.time()
    if time_left > 8.0:
        n = len(best_vec)
        print(f"[phase 2] Adam optimisation (n={n}) – time left {time_left:.1f}s")
        # allocate roughly 60 % of remaining time to Adam + short refinement
        phase_deadline = time.time() + time_left * 0.55
        a_opt, _ = _adam_optimize(best_vec, phase_deadline,
                                  init_alpha=0.5,
                                  alpha_max=10000.0,
                                  lr0=0.25)

        # short exact‑max polishing
        phase_deadline2 = time.time() + time_left * 0.25
        a_opt = _refine_max(a_opt, phase_deadline2)

        best_vec = a_opt.copy()
        best_fast = 2.0 * n * _fft_self_convolve(best_vec).max()
        print(f"[phase 2] post‑refine fast_score={best_fast:.6f}")

    # ------------------------------------------------------------------
    # ---- 3. SIMULATED‑ANNEALING POLISHING -----------------------------
    time_left = deadline - time.time()
    if time_left > 4.0:
        n = len(best_vec)
        print(f"[phase 3] simulated annealing (n={n}) – time left {time_left:.1f}s")
        a_final, final_peak = _simulated_annealing(best_vec, deadline)
        best_vec = a_final.copy()
        best_fast = 2.0 * n * final_peak
        print(f"[phase 3] SA fast_score={best_fast:.6f}")

    # ------------------------------------------------------------------
    # ---- 4. FINAL BLACK‑BOX EVALUATION --------------------------------
    try:
        true_score = evaluate_sequence(best_vec.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < best_fast:
            best_fast = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Final best score (approx.): {best_fast:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
