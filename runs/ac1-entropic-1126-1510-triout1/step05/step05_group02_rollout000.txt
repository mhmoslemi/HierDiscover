"""
Propose a step‑function with a low evaluate_sequence score.

Strategy
--------
1. **Broad parametric search** – sample many Beta (and mixtures of two Betas)
   probability vectors on several lengths.  Keep the best one w.r.t.
   the *raw* objective 2·n·max(a*a), which is exactly the evaluate_sequence
   value up to the irrelevant scaling factor.

2. **Smooth max optimisation** – minimise a smooth approximation of the
   ℓ∞ self‑convolution:
        F(a) = log Σ_k exp(α·(a*a)_k) .
   Its gradient is cheap via FFT:
        ∇F = 2·α·conv_rev(a, w)   with  w_k ∝ exp(α·c_k) .
   We use the Adam optimiser, project onto the simplex after every step,
   and increase α gradually (homotopy) so that the smooth problem
   converges to the true max‑problem.

3. **Final sub‑gradient polishing** – a few iterations of the classic
   sub‑gradient method (active‑set weights) to tighten the result.

All heavy work is done with FFT‑based convolutions (O(n log n)).  The whole
procedure fits comfortably into the 1000 s budget and works for moderate
lengths (n≈2000–5000).  The returned list is a plain Python list of floats,
ready for evaluate_sequence().
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """2·n·max(a*a); the quantity we minimise (sum of a is 1)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) on n points, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # using log‑pdf for stability
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of   log Σ_k exp(α·(a*a)_k)   w.r.t. a.
    Returned gradient has the same shape as a.
    """
    n = a.shape[0]
    # self‑convolution
    conv = _fft_self_convolve_full(a)            # length 2n‑1
    # stable log‑sum‑exp
    maxc = conv.max()
    exp_shifted = np.exp(alpha * (conv - maxc))
    Z = exp_shifted.sum()
    w = exp_shifted / Z                          # weight vector, length 2n‑1
    # gradient = 2·α·conv_rev(a, w)
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w)          # length 3n‑2
    grad = 2.0 * alpha * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluate_sequence score.
    Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    # --------------------------------------------------------------
    # 1) Broad parametric search (Beta families + mixtures)
    lengths = [1200, 2400, 4800]   # lengths large enough for fine shapes
    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # --- uniform baseline ---
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_score, best_seq = sc, uniform.copy()
            print(f"   uniform score={best_score:.6f}")

        # --- arcsine (Beta(0.5,0.5)) ---
        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine)
        if sc < best_score:
            best_score, best_seq = sc, arcsine.copy()
            print(f"   arcsine score={best_score:.6f}")

        # --- symmetric Beta grid ---
        alphas = np.concatenate((
            np.linspace(0.12, 0.6, 12, endpoint=False),
            np.linspace(0.6, 2.0, 15),
            np.linspace(2.0, 5.0, 8)
        ))
        for a in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a, a)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.copy()
                print(f"   symBeta a={a:.3f} score={best_score:.6f}")

        # --- random mixtures of two Betas ---
        for _ in range(200):
            if time.time() >= deadline:
                break
            a1, b1 = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
            a2, b2 = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
            lam = np.random.rand()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)   # enforce simplex
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.copy()
                print(f"   mix λ={lam:.2f} ({a1:.2f},{b1:.2f})+({a2:.2f},{b2:.2f}) score={best_score:.6f}")

    if best_seq is None:
        # fallback (should never happen)
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_seq)

    print(f"Best raw score after param search: {best_score:.6f} (len={best_seq.shape[0]})")

    # --------------------------------------------------------------
    # 2) Smooth‑max optimisation with Adam + homotopy on α
    a = best_seq.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for α (larger α → sharper approximation of max)
    alpha_schedule = [(30, 5000), (80, 5000), (200, 5000), (500, 5000), (1500, 5000)]
    schedule_idx = 0
    cur_alpha, cur_iters = alpha_schedule[schedule_idx]

    it = 0
    best_a = a.copy()
    best_val = _raw_score(a)

    while time.time() < deadline:
        it += 1
        # --- gradient of smooth max ---
        grad = _smooth_max_gradient(a, cur_alpha)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.12 / math.sqrt(it)   # decaying base learning rate
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex (sum = 1, non‑negative)
        a = _project_onto_simplex(a)

        # evaluate exact raw objective every few hundred steps
        if it % 400 == 0:
            cur_val = _raw_score(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth it={it} α={cur_alpha} score={best_val:.6f}")

        # switch to next α after allotted iterations
        if it >= cur_iters:
            schedule_idx += 1
            if schedule_idx < len(alpha_schedule):
                cur_alpha, cur_iters = alpha_schedule[schedule_idx]
                it = 0
                # re‑initialise Adam moments for a clean start at new α
                m.fill(0.0)
                v.fill(0.0)
                print(f"   → increase α to {cur_alpha}")
            else:
                # finished all stages
                break

    # --------------------------------------------------------------
    # 3) Final polishing with classic sub‑gradient method
    a = best_a.copy()
    n = a.shape[0]
    best_val = _raw_score(a)
    best_a = a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        # sub‑gradient of the true max (active‑set version)
        conv_full = _fft_self_convolve_full(a)
        cur_max = conv_full.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv_full >= cur_max - tol)[0]
        w = np.zeros_like(conv_full)
        if active.size:
            w[active] = 1.0 / active.size
        a_rev = a[::-1]
        grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]

        step = 0.05 / math.sqrt(it)      # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a)

        if it % 400 == 0:
            cur_val = _raw_score(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] subgrad it={it} score={best_val:.6f}")

    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)   # final black‑box check
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq
