def _lp_optimize_direction(a, M):
    n = a.shape[0]
    # Build matrix A_ub
    A = _build_conv_matrix(a) # shape (2n-1, n)
    # Right-hand side b_ub = M for all rows
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    # Objective: maximize sum(b) <=> minimize -sum(b)
    c = -np.ones(n, dtype=np.float64)
    # Bounds: (0, None)
    bounds = [(0, None)] * n
    # Solve using HiGHS
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        b = res.x
        # Clip tiny negatives
        b[b < 0] = 0.0
        return b
    else:
        # Infeasible or other issue, fall back to zero vector
        return np.zeros(n, dtype=np.float64)
```

"""
Search for a low‑score step function.

The algorithm works in three stages:

1. **Exploration** – generate a diverse family of candidates (uniform,
   arcsine, symmetric Beta, random mixtures of Beta, a triangular shape and,
   an interpolation of the previous best solution if available).  All candidates
   are projected onto the simplex ∑a_i = 1.

2. **Frank‑Wolfe style refinement** – for a selected candidate we solve the
   linear programme
        maximise   ∑ b_i
        subject to (a * b)_k ≤ M   (for all k) ,   b_i ≥ 0
   where M = max_k (a * a)_k.  The LP gives a direction b that keeps the
   self‑convolution bound unchanged while increasing the total mass.  The
   direction is scaled back to a simplex point g and mixed with the current a
   (a ← (1‑t)a + t g).  A small number of such steps already reduces the
   L_∞‑norm of the convolution.

3. **Projected sub‑gradient descent** – finally we minimise the non‑smooth
   objective 2·n·max_k (a*a)_k on the simplex using a simple sub‑gradient
   scheme with a diminishing step size and occasional random jitter.

All operations respect the constraints a_i ≥ 0,   ∑a_i = 1, and the routine
returns the best sequence found before the time budget expires.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b (real FFT)."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of  max_k (a*a)_k.
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _fft_self_convolve_full(a)
    cur_max = b.max()
    # tolerance for active set
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, cur_max


def _raw_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) assuming Σa_i = 1."""
    a = _project_onto_simplex(a)
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑pdf for stability
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


def _interpolate_seq(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


def _triangular_distribution(n: int) -> np.ndarray:
    """Simple symmetric triangular (tent) distribution."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=float)
    if n % 2 == 0:
        right = np.arange(half, 0, -1, dtype=float)
        seq = np.concatenate([left, right])
    else:
        right = np.arange(half, 0, -1, dtype=float)
        seq = np.concatenate([left, np.array([half + 1], dtype=float), right])
    return _project_onto_simplex(seq)


def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2·n‑1) × n sparse matrix that maps b → a * b
    (full linear convolution).  Rows correspond to convolution indices.
    """
    n = a.shape[0]
    # rows[i, j] = i + j  (i = index in a, j = index in b)
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    cols = np.tile(np.arange(n), n)            # each column index repeated n times
    data = np.repeat(a, n)                     # a_i repeated for each column
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_optimize_direction(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise   Σ b_i   subject to   (a * b)_k ≤ M,   b_i ≥ 0.
    Returns the optimal b (may be all zeros when M is extremely small).
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)                     # shape (2n‑1, n)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)             # maximise Σb_i  ↔  minimise -Σb_i
    bounds = [(0, None)] * n
    # HiGHS handles sparse matrices efficiently
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                 bounds=bounds, method='highs')
    if res.success:
        b = res.x
        b[b < 0] = 0.0
        return b
    # fallback (should rarely happen)
    return np.zeros(n, dtype=np.float64)


def _subgradient_refine(a: np.ndarray, max_iters: int, deadline: float):
    """
    Projected sub‑gradient descent on 2·n·max(a*a) with diminishing step size.
    Returns the best vector found and its raw score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        grad, _ = _max_subgradient_step(a)
        step = 0.2 / math.sqrt(it + 1)          # relatively aggressive early steps
        a = a - step * grad
        a = _project_onto_simplex(a)
        # occasional jitter to jump out of plateaus
        if it % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a)
        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it} score={best_val:.6f}")
        it += 1
    return best_a, best_val


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence score.
    The function respects the 1000‑second time budget and returns a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Try to reuse the previously discovered sequence if it exists.
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Problem size – a moderate length makes the LP tractable while still
    # providing a good constant factor.
    n = 2000

    # ------------------------------------------------------------------
    # Build a pool of diverse initial candidates.
    candidates = []

    # uniform and arcsine
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))
    candidates.append(_beta_distribution(n, 0.5, 0.5))

    # symmetric Beta with a wide range of α
    alphas = np.concatenate((
        np.linspace(0.02, 0.6, 12, endpoint=False),
        np.linspace(0.6, 2.0, 15),
        np.linspace(2.0, 5.0, 8)
    ))
    for a in alphas:
        candidates.append(_beta_distribution(n, a, a))

    # triangular shape
    candidates.append(_triangular_distribution(n))

    # random mixtures of two Betas
    for _ in range(50):
        a1, b1 = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
        a2, b2 = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
        lam = np.random.rand()
        mix = lam * _beta_distribution(n, a1, b1) + (1 - lam) * _beta_distribution(n, a2, b2)
        candidates.append(mix)

    # random mixtures of three Betas (more expressive)
    for _ in range(30):
        ks = np.random.rand(3)
        ks /= ks.sum()
        comp = np.zeros(n, dtype=np.float64)
        for weight in ks:
            a, b = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
            comp += weight * _beta_distribution(n, a, b)
        candidates.append(comp)

    # interpolation of the previous best solution, if available
    if base_seq is not None:
        candidates.append(_interpolate_seq(base_seq, n))

    # ------------------------------------------------------------------
    # Evaluate all candidates on the raw objective and keep the best few.
    evaluated = []
    for vec in candidates:
        vec = _project_onto_simplex(vec)
        score = _raw_score(vec)
        evaluated.append((score, vec))
    evaluated.sort(key=lambda x: x[0])

    # keep the top‑3 for deeper optimisation
    top_k = 3
    best_overall_vec = None
    best_overall_score = float('inf')
    best_len = n

    for raw_score, vec in evaluated[:top_k]:
        if time.time() >= deadline:
            break
        print(f"Starting refinement from raw score {raw_score:.6f}")

        # ------------------------------------------------------------------
        # Frank‑Wolfe style LP refinement (few iterations, cheap)
        a = vec.copy()
        for it in range(12):                     # at most 12 LP steps
            if time.time() >= deadline:
                break
            M = _fft_self_convolve_max(a)
            b_dir = _lp_optimize_direction(a, M)
            sum_b = b_dir.sum()
            if sum_b <= 1e-12:
                break
            g = b_dir / sum_b                     # normalise to simplex
            t = 0.5                               # mixing weight
            a = (1 - t) * a + t * g
            a = _project_onto_simplex(a)
            cur_raw = _raw_score(a)
            if cur_raw < best_overall_score:
                best_overall_score = cur_raw
                best_overall_vec = a.copy()
                print(f"  LP it={it} improved raw to {best_overall_score:.6f}")

        # ------------------------------------------------------------------
        # Sub‑gradient final polishing (time‑bounded)
        sub_deadline = deadline
        a_refined, refined_val = _subgradient_refine(a, max_iters=3000, deadline=sub_deadline)
        if refined_val < best_overall_score:
            best_overall_score = refined_val
            best_overall_vec = a_refined.copy()
            print(f"  Sub‑grad refined raw to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing succeeded.
    if best_overall_vec is None:
        best_overall_vec = np.full(1000, 1.0 / 1000, dtype=np.float64)
        best_len = 1000
        best_overall_score = _raw_score(best_overall_vec)

    # ------------------------------------------------------------------
    # Final evaluation with the provided black‑box.
    final_score = evaluate_sequence(best_overall_vec.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_overall_vec)})")

    return best_overall_vec.tolist()
