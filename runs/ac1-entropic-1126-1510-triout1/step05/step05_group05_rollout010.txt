    # Now run a few refinement cycles on best_seq using exponentiated gradient
    if best_seq is not None:
        n = best_seq.shape[0]
        # set refinement parameters
        a = best_seq.copy()
        best_obj = best_score
        eta0 = 1.0
        num_iters = 20000
        for it in range(1, num_iters+1):
            # compute convolution and max index
            b = _fft_self_convolve(a)
            cur_max = b.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_obj - 1e-12:
                best_obj = cur_score
                best_seq = a.copy()
                print(f"Refine it={it} score={best_obj:.6f}")

            # subgradient at max index
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = np.full(n, 1.0/n)
    # Finally evaluate using evaluate_sequence for final score
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        final_score = float('inf')
    print(f"Final evaluate_sequence score: {final_score:.6f}")

"""
Propose a low‑score step‑function using a fast exhaustive search over a few
hand‑crafted families (arcsine, mixtures, power‑law, beta).  The search is
performed on a small set of lengths up to the allowed maximum (20 000).  For
each candidate we evaluate the exact objective
    2·n·max(a∗a)          (the denominator is 1 because the vector is
    normalised to a simplex)
with an FFT‑based convolution, which is orders of magnitude faster than the
reference `np.convolve` used inside `evaluate_sequence`.

The best candidate is then refined by a short exponentiated‑gradient (mirror–
descent) run that directly minimises the max‑convolution.  All updates stay on
the probability simplex by renormalisation after each multiplicative step.

The routine respects the 1000 s budget, prints progress, and finally returns
the best sequence as a plain Python list (the harness will call
`evaluate_sequence` on the returned list).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                         # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution on {0,…,n‑1} (already normalised)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """Symmetric power‑law family  (i·(n‑i))^{±p}."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric beta(a,a) on the index set."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1.0 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """Convex combination (1‑lam)·arcsine + lam·uniform."""
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    # Numerical drift may appear; re‑project onto the simplex.
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum value of the full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of 2
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    # keep only the first 2·n‑1 entries (full linear convolution)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) under the simplex normalisation
    (i.e. sum(a) == 1).  Returns the same quantity that `evaluate_sequence`
    would produce for a normalised vector.
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _refine_exponential(a0: np.ndarray,
                        max_seconds: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient refinement.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _objective(a)

    # basic schedules (tuned empirically)
    eta0 = 0.9
    it = 0
    start = time.time()
    while time.time() - start < max_seconds:
        it += 1
        # current convolution
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best = a.copy()
            print(f"[{time.time():.2f}] refine it={it} score={best_val:.6f}")

        # sub‑gradient at the current maximal index
        k = int(np.argmax(b))
        grad = _subgradient_max(a, k)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

    return best, best_val
