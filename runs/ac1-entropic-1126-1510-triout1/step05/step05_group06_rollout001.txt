def _simulated_annealing(a, n, deadline, init_temp=0.01, min_temp=1e-6):
    best = a.copy()
    best_score = internal_score(a)
    cur = a.copy()
    cur_score = best_score
    it = 0
    while time.time() < deadline:
        it += 1
        # temperature schedule: exponential decay or linear
        t = max(min_temp, init_temp * (0.99 ** (it // 100)))  # decays slowly
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # magnitude epsilon: draw from uniform [0, cur[i]*alpha] where alpha small
        alpha = 0.3  # max fraction
        eps = cur[i] * np.random.rand() * alpha
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = _project_onto_simplex(cand, s=1.0)  # ensures sum remains 1 and nonnegative
        cand_score = internal_score(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / t):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
        # Occasionally apply small smoothing
        if it % 5000 == 0:
            # perturb all entries slightly
            noise = np.random.randn(n) * 1e-4
            cur = cur + noise
            cur = np.maximum(cur, 0)
            cur = cur / cur.sum()
            cur_score = internal_score(cur)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
    return best, best_score

def _exponential_gradient_optimize(a0, n, deadline):
    a = a0.copy()
    best = a.copy()
    best_score = _internal_score(a)

    eta0 = 0.7
    decay = 1e-4
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # compute subgradient at max index
        k = int(np.argmax(conv))
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k - idx]

        # compute step size
        eta = eta0 / (1.0 + decay * t)

        if s == 0:

        else:

        # occasional random jitter

            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()
    return best, best_score

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

The algorithm mixes three ideas:

1. **Rich initialisation** – a small set of shapes (uniform, arcsine,
   beta‑sample, random and an interpolation of the previously best
   sequence if it exists) are generated for several lengths.

2. **Exponential‑gradient (mirror descent)** – a fast sub‑gradient
   optimiser on the simplex that quickly drives the max‑convolution down.

3. **Simulated‑annealing hill‑climbing** – stochastic mass moves with a
   temperature schedule, able to escape local minima left by the gradient
   phase.  A final short local refinement step cleans up remaining tiny
   improvements.

The routine respects the given time budget, continuously records the
best sequence seen and finally returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex {x ≥ 0, sum(x) = s}.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full convolution a*a via a real FFT (length 2·n‑1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _internal_score(a: np.ndarray) -> float:
    """
    Fast surrogate for ``evaluate_sequence`` when sum(a)=1:
          2·n·max(a*a)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _interpolate_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence to a new length, followed by a
    projection onto the simplex (so the sum stays equal to 1).
    """
    old_n = seq.shape[0]
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = _project_onto_simplex(new_seq, s=1.0)
    return new_seq


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """
    Produce a simple normalised (sum=1) non‑negative vector of length *n*.
    Supported kinds: "uniform", "arcsine", "beta", "random".
    """
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "arcsine":
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a /= a.sum()
    elif kind == "beta":
        a = np.random.beta(0.5, 0.5, size=n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
    elif kind == "random":
        a = np.random.rand(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    else:
        a = np.full(n, 1.0 / n)

    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _exponential_gradient_optimize(a0: np.ndarray,
                                   n: int,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimisation on the simplex.
    Uses the sub‑gradient of the maximal convolution entry.
    Returns the best vector seen and its internal 2·n·max(conv) score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _internal_score(a)

    eta0 = 0.7          # initial step‑size factor
    decay = 1e-4        # decay of step‑size
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        k = int(np.argmax(conv))
        # sub‑gradient of the max‑entry (conv[k] = Σ a_i a_{k−i})
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k - idx]

        eta = eta0 / (1.0 + decay * it)

        # exponentiated‑gradient update (mirror descent w.r.t. KL)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional tiny jitter to shake out plateaus
        if it % 3000 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best, best_score


# ----------------------------------------------------------------------
def _simulated_annealing_optimize(a0: np.ndarray,
                                  n: int,
                                  deadline: float,
                                  init_temp: float = 0.02,
                                  min_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Stochastic annealing: move a random fraction of mass from one index to
    another, accepting worsening moves with probability exp(−Δ/ T).
    Temperature decays geometrically.
    """
    a = a0.copy()
    cur_score = _internal_score(a)
    best = a.copy()
    best_score = cur_score

    it = 0
    temp = init_temp

    while time.time() < deadline:
        it += 1

        # temperature schedule – decay every 1000 steps
        if it % 1000 == 0:
            temp = max(min_temp, temp * 0.90)

        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue

        # move at most 20 % of source mass
        eps = a[i] * np.random.rand() * 0.20
        if eps <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps

        # projection onto simplex to clean tiny rounding errors
        cand = _project_onto_simplex(cand, s=1.0)

        cand_score = _internal_score(cand)
        delta = cand_score - cur_score

        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score

        # every few thousand steps add a global tiny Gaussian noise
        if it % 5000 == 0:
            noise = np.random.randn(n) * 1e-4
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            cur_score = _internal_score(a)
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score

    return best, best_score


# ----------------------------------------------------------------------
def _local_hill_climb(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing: move a small fraction of mass between two
    indices whenever the true internal score improves.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = _internal_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # up to 10 % of source mass
        eps = a[i] * np.random.rand() * 0.10
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < -1e-12:          # safeguard
            continue
        cand = _project_onto_simplex(cand, s=1.0)

        cand_score = _internal_score(cand)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            a = cand

    return best, best_score
