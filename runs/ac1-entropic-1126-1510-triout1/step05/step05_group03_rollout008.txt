            'type': 'eq',
            'fun': lambda x: np.sum(x) - 1.0,
            'jac': lambda x: np.ones_like(x)
        },)
        bounds = [(0.0, None)]*len(a)
        res = scipy.optimize.minimize(fun_smooth, a, method='SLSQP', jac=grad_smooth,
               bounds=bounds, constraints=cons, options={'maxiter':200,'ftol':1e-12})
        a = res.x
        # projection may slight violate sum due to numeric errors; correct:
        a = _project_onto_simplex(a, s=1.0)
        val = _fft_self_convolve(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
        if time.time() >= deadline:
            break
    return best_a

"""
Propose a low‑score step‑function.

The search proceeds in four stages:

1️⃣  **Broad exploration** – try several lengths (500 … 15000) and a
    collection of simple families (uniform, triangular, exponential,
    Beta‑densities, random Dirichlet).  The best candidate according to
    the cheap surrogate   `fast_score = 2·n·max(a*a)` is kept.

2️⃣  **Adam optimisation** – a smooth‑max surrogate
    `L_α(a)= (1/α)·log Σ_k exp(α·(a*a)_k)` is minimised with Adam.
    The smoothing parameter α is slowly increased from 1 to ≈2000,
    which drives the solution towards the true ℓ∞‑objective.

3️⃣  **Deterministic polishing** – an adaptive sub‑gradient descent
    with a back‑tracking line‑search is applied to the exact peak.
    This step often reduces the peak by a noticeable amount.

4️⃣  **Stochastic polishing** – a simulated‑annealing phase moves
    small amounts of probability mass between random positions
    while accepting uphill moves according to an exponential schedule.
    The best vector found is finally evaluated with the black‑box
    `evaluate_sequence`.

All vectors are kept on the probability simplex (non‑negative,
sum = 1).  The algorithm respects the 1000 s budget; each stage uses
the remaining time adaptively.
"""

import math
import time
import random
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangular (lin‑ramp up then down) distribution."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        tri = np.concatenate([inc, inc[::-1]])
    else:
        tri = np.concatenate([inc, np.array([half + 1], dtype=np.float64), inc[::-1]])
    tri /= tri.sum()
    return tri


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, r: float = 0.99) -> np.ndarray:
    """Geometric (exponential decay) distribution, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = r ** i
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on a uniform grid, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _random_dirichlet(n: int, concentration: float = 1.0) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with equal alphas)."""
    x = np.random.gamma(concentration, 1.0, size=n)
    x /= x.sum()
    return x


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, alpha: float):
    """
    Smooth‑max of the convolution and its gradient.
    Returns (value, grad) where ``value = (1/α)·log Σ exp(α·b_k)``.
    """
    b = _fft_self_convolve(a)
    b_max = b.max()
    # stable log‑sum‑exp
    exp_term = np.exp(alpha * (b - b_max))
    w_sum = exp_term.sum()
    smooth = (np.log(w_sum) / alpha) + b_max
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_val = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        smooth, grad = _smooth_max_and_grad(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep exact peak
        cur_peak = _fft_self_convolve(a).max()
        if cur_peak < best_val - 1e-14:
            best_val = cur_peak
            best_a = a.copy()

        # raise smoothing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter
        if t % 400 == 0 and random.random() < 0.07:
            a += np.random.randn(a.shape[0]) * 2e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a


# ----------------------------------------------------------------------
def _adaptive_subgradient(a0: np.ndarray,
                         deadline: float,
                         init_step: float = 0.1,
                         min_step: float = 1e-6):
    """
    Sub‑gradient descent on the exact ℓ∞‑objective with an adaptive
    back‑tracking line‑search (no smoothing).  Returns the best vector.
    """
    a = a0.copy()
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    step = init_step

    while time.time() < deadline:
        # current peak and active set
        b = _fft_self_convolve(a)
        cur_max = b.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(b >= cur_max - tol)[0]

        # sub‑gradient (uniform weights on the active set)
        w = np.zeros_like(b)
        if active.size:
            w[active] = 1.0 / active.size
        a_rev = a[::-1]
        grad = 2.0 * _fft_convolve(a_rev, w)[len(a) - 1 : len(a) - 1 + len(a)]

        # try a few step reductions until improvement
        improved = False
        for _ in range(20):
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new, s=1.0)
            new_peak = _fft_self_convolve(a_new).max()
            if new_peak < cur_max - 1e-14:
                a = a_new
                cur_max = new_peak
                improved = True
                break
            step *= 0.5
        if not improved:
            # if no improvement even after many halvings, reduce step globally
            step = max(step * 0.7, min_step)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # occasional random jitter to escape plateaus
        if random.random() < 0.03:
            a += np.random.randn(a.shape[0]) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Simple simulated‑annealing polishing.
    Moves a random fraction of mass between two positions,
    accepts uphill moves with a temperature schedule.
    Returns the best vector encountered.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    # temperature schedule (exponential decay)
    start_temp = max(cur_peak, 1e-3) * 0.6
    final_temp = start_temp * 1e-6
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    while time.time() < deadline:
        src = random.randint(0, n - 1)
        if cur_a[src] <= 1e-12:
            continue
        dst = random.randint(0, n - 1)
        if src == dst:
            continue

        max_delta = min(cur_a[src], 0.04)           # up‑to 4 % moves
        delta = random.random() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional reset to the best solution (helps in late stage)
        if random.random() < 0.001:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative sequence to a new length by linear interpolation."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list:
    """Search for a sequence of non‑negative heights with a low score."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # keep 1 s safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Broad search over several lengths and families
    print("[phase 1] broad search")
    lengths = [800, 1600, 3200, 6400, 12000]    # a good spread, still tractable
    global height_sequence_1                     # try to reuse a known good construction
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        base_seq = _project_onto_simplex(base_seq, s=1.0)
    except Exception:
        base_seq = None

    overall_best_a = None
    overall_best_score = float("inf")
    chosen_n = None

    for n in lengths:
        if time.time() >= deadline:
            break
        starters = {}

        # a few deterministic families
        starters["uniform"] = _uniform_distribution(n)
        starters["triangular"] = _triangular_distribution(n)
        starters["exp_099"] = _exponential_distribution(n, r=0.99)
        starters["exp_095"] = _exponential_distribution(n, r=0.95)

        # a few Beta shapes
        for a in (0.7, 1.0, 1.3, 1.6):
            starters[f"beta_{a:.1f}"] = _beta_distribution(n, a, a)

        # a few random Dirichlet draws
        for conc in (0.3, 0.7, 1.0, 2.0):
            starters[f"rand_{conc:.1f}"] = _random_dirichlet(n, concentration=conc)

        # reuse previous construction if available
        if base_seq is not None:
            starters["prev"] = _interpolate_to_length(base_seq, n)

        # evaluate cheap fast_score = 2·n·max(b)
        for tag, vec in starters.items():
            vec = _project_onto_simplex(vec, s=1.0)
            peak = _fft_self_convolve(vec).max()
            fast_score = 2.0 * n * peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_a = vec.copy()
                chosen_n = n
                print(f"[init] n={n:5d} {tag:<12} fast_score={fast_score:.6f}")

    if overall_best_a is None:
        # fallback – plain uniform, length 2000
        chosen_n = 2000
        overall_best_a = _uniform_distribution(chosen_n)
        overall_best_score = 2.0 * chosen_n * _fft_self_convolve(overall_best_a).max()
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimisation (smooth‑max)
    time_left = deadline - time.time()
    if time_left > 5.0:
        print(f"[phase 2] Adam optimisation (n={chosen_n}) – time left {time_left:.1f}s")
        adam_deadline = time.time() + time_left * 0.55
        a_opt = _adam_optimize(overall_best_a, adam_deadline)
        # keep the best exact peak after Adam
        overall_best_a = a_opt.copy()
        overall_best_score = 2.0 * chosen_n * _fft_self_convolve(overall_best_a).max()
        print(f"[phase 2] post‑Adam fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Adaptive sub‑gradient descent on the exact max
    time_left = deadline - time.time()
    if time_left > 3.0:
        print(f"[phase 3] exact sub‑gradient descent (n={chosen_n}) – time left {time_left:.1f}s")
        sub_deadline = time.time() + time_left * 0.30
        a_opt = _adaptive_subgradient(overall_best_a, sub_deadline)
        overall_best_a = a_opt.copy()
        overall_best_score = 2.0 * chosen_n * _fft_self_convolve(overall_best_a).max()
        print(f"[phase 3] after sub‑grad fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing polishing
    time_left = deadline - time.time()
    if time_left > 2.0:
        print(f"[phase 4] simulated annealing (n={chosen_n}) – time left {time_left:.1f}s")
        a_final = _simulated_annealing(overall_best_a, deadline)
        overall_best_a = a_final.copy()
        overall_best_score = 2.0 * chosen_n * _fft_self_convolve(overall_best_a).max()
        print(f"[phase 4] final fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the true black‑box function
    try:
        true_score = evaluate_sequence(overall_best_a.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")

        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score (approx.) = {overall_best_score:.6f} (length={chosen_n})")
    return overall_best_a.tolist()
