"""step function optimizer

Searches for a non‑negative step function that minimises
`evaluate_sequence`.  The method combines:

1.  A quick scan of several analytic families (arcsine, beta,
    triangular, exponential, power‑law) plus a few random Dirichlet
    draws.  The best candidate from this pool is kept.

2.  A smooth‑max Adam optimisation on the simplex.
    The true max of the self‑convolution is approximated by a
    temperature‑scaled soft‑max (β·max).  The gradient of the smooth
    objective is   ∇ = a + conv_rev ,   where `conv_rev` is the
    convolution of the reversed sequence with the soft‑max weights.
    After each step the iterate is Euclidean‑projected onto the simplex
    (sum == 1, a_i ≥ 0).

3.  A cheap random‑transfer hill‑climber that moves a tiny amount of
    mass between two random positions and accepts strictly improving
    moves.  Occasional Gaussian jitter helps escape plateaus.

The whole routine respects the supplied time budget (≈ 1000 s) and
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0.0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p, p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space computation for stability
    logw = (p - 1.0) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns smooth max of the self‑convolution and its gradient.
    Uses f(a)= (1/β)·log Σ exp(β·b_k)   with b = a*a.
    Gradient: ∇f = a + conv_rev, where conv_rev = (a_rev * w)[n‑1 : n‑1+n].
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    # stable soft‑max weights
    exp_w = np.exp(beta * (b - max_b))
    w = exp_w / exp_w.sum()               # length 2·n‑1, sum = 1
    # conv of reversed a with the weight vector
    conv_rev = _fft_convolve(a[::-1], w)
    grad = a + conv_rev[n - 1 : n - 1 + n]
    smooth_val = max_b + np.log(np.sum(exp_w)) / beta
    return smooth_val, grad, b.max()


# ----------------------------------------------------------------------

    try:

    except Exception:
        base_n = 4000
    n = max(2000, base_n)        # we keep a moderate size for speed
