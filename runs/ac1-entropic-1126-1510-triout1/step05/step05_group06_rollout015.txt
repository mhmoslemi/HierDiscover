"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The search proceeds in three phases:

1. **Broad random sampling** – generate many candidate shapes from several
   parametric families (symmetric / asymmetric power laws, cosine‑modulated
   power laws, piece‑wise constant, Dirichlet draws, mixtures with uniform,
   etc.).  Candidates are projected on the simplex (sum = 1) and a cheap
   internal score   2·n·max(conv)   is computed via an FFT based self‑convolution.

2. **Refinement** – the best few candidates of each length are polished with
   two complementary first‑order methods:
   * exponentiated‑gradient (mirror‑descent) on a soft‑max surrogate of the
     max‑convolution value;
   * additive sub‑gradient descent on the true max‑convolution.
   Both steps keep the vector on the simplex.

3. **Local stochastic hill‑climbing** – from the globally best vector we
   repeatedly move a tiny amount of mass between two random positions whenever
   the true evaluation score improves.

The routine respects the given time budget (default 1000 s), prints progress,
and finally returns the best sequence found as a plain Python list.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max entry of a*a at index *k*."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of (1/α)·log∑exp(α·b) w.r.t. a (soft‑max surrogate)."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:                         # fallback – pure max
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]                          # convolution is not a correlation
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _internal_score(a: np.ndarray, n: int) -> float:
    """Cheap proxy for evaluate_sequence (sum=1 ⇒ 2·n·max(conv))."""
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _beta_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ (p>0)."""
    i = np.arange(n, dtype=np.float64)
    base = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    # avoid numerical overflow for very large exponents
    base = np.clip(base, 0.0, 1e20)
    base /= base.sum()
    return base


# ----------------------------------------------------------------------
def _asymmetric_power_shape(n: int, pl: float, pr: float) -> np.ndarray:
    """a_i ∝ (i+½)⁻ˡ·(n‑i‑½)⁻ʳ  (pl,pr>0)."""
    i = np.arange(n, dtype=np.float64)
    base = (i + 0.5) ** (-pl) * (n - i - 0.5) ** (-pr)
    base = np.clip(base, 0.0, 1e20)
    base /= base.sum()
    return base


# ----------------------------------------------------------------------
def _cosine_modulated_power_shape(n: int, p: float, gamma: float) -> np.ndarray:
    """Power‑law shape multiplied by (1+γ·cos(2πi/(n‑1))) , γ∈[0,1]."""
    i = np.arange(n, dtype=np.float64)
    power = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    power = np.clip(power, 0.0, 1e20)
    mod = 1.0 + gamma * np.cos(2.0 * np.pi * i / (n - 1))
    shape = power * mod
    shape = np.clip(shape, 0.0, None)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
def _piecewise_constant_shape(n: int, segments: int) -> np.ndarray:
    """Random step‑function with *segments* constant blocks."""
    # Choose segment borders
    cuts = np.sort(np.random.choice(n - 1, segments - 1, replace=False) + 1)
    cuts = np.concatenate(([0], cuts, [n]))
    vals = np.random.rand(segments) + 0.01   # strictly positive
    a = np.empty(n, dtype=np.float64)
    for k in range(segments):
        a[cuts[k] : cuts[k + 1]] = vals[k]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _mix_with_uniform(shape: np.ndarray, weight: float) -> np.ndarray:
    """Mixture (1‑w)·shape + w·uniform, then projected onto simplex."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = (1.0 - weight) * shape + weight * uniform
    return _project_onto_simplex(mixed, s=1.0)


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using a soft‑max surrogate.
    Returns the best vector seen and its internal 2·n·max(conv) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _internal_score(a, n)

    # hyper‑parameters
    eta0 = 0.8                 # larger initial step‐size
    alpha = 5.0                # start temperature for soft‑max
    alpha_mul = 1.05
    alpha_max = 350.0
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = _internal_score(a, n)

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_seq = a.copy()

        # gradient
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad)

        # re‑project onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter to escape plateaus
        if it % 3000 == 0 and np.random.rand() < 0.10:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent on the true max‑convolution.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _internal_score(a, n)

    eta0 = 0.6
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = _internal_score(a, n)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_seq = a.copy()

        # active set (indices attaining the max, up to a small tolerance)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-14)[0]

        # combined sub‑gradient (average over the active set)
        grad = np.zeros_like(a)
        for k in active:
            grad += _subgradient_max(a, k)
        grad /= max(1, active.size)

        eta = eta0 / math.sqrt(it)
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a tiny fraction of mass between two
    positions whenever the true evaluation score decreases.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _internal_score(a, n)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer at most 5 % of the source mass
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur_score = _internal_score(cand, n)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Choose a handful of lengths (moderate‑to‑large) and allocate time.
    base_n = 8000
    try:
        base_n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        pass

    lengths = sorted(
        {max(500, int(base_n * f))
         for f in (0.75, 1.0, 1.25, 1.5, 2.0)
         if max(500, int(base_n * f)) <= 20000}
        | {15000, 18000, 20000}
    )
    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(5.0, remaining * 0.30)

        # ------------------------------------------------------------------
        # 1) Random broad search for promising candidates
        candidates = []

        # we cap the total number of generated candidates per length
        max_samples = 3000
        sample_cnt = 0

        while time.time() < slice_deadline and sample_cnt < max_samples:
            sample_cnt += 1
            rnd = np.random.rand()

            if rnd < 0.18:
                # symmetric power‑law
                p = np.random.uniform(0.1, 2.0)
                seq = _beta_power_shape(n, p)
                # optionally mix a little uniform mass
                if np.random.rand() < 0.3:
                    seq = _mix_with_uniform(seq, np.random.rand() * 0.4)
            elif rnd < 0.33:
                # asymmetric power‑law
                pl = np.random.uniform(0.1, 2.0)
                pr = np.random.uniform(0.1, 2.0)
                seq = _asymmetric_power_shape(n, pl, pr)
            elif rnd < 0.48:
                # cosine‑modulated power‑law
                p = np.random.uniform(0.1, 2.0)
                gamma = np.random.rand()               # 0 … 1
                seq = _cosine_modulated_power_shape(n, p, gamma)
                if np.random.rand() < 0.2:
                    seq = _mix_with_uniform(seq, np.random.rand() * 0.3)
            elif rnd < 0.63:
                # piece‑wise constant
                segs = np.random.randint(2, 7)
                seq = _piecewise_constant_shape(n, segs)
            elif rnd < 0.78:
                # Dirichlet draw (concentration controls sparsity)
                alpha = np.random.uniform(0.01, 5.0)
                seq = np.random.dirichlet(alpha * np.ones(n))
            elif rnd < 0.90:
                # random uniform mixture of arcsine & uniform
                arcs = _beta_power_shape(n, 0.5)        # arcsine law
                w = np.random.rand() * 0.5
                seq = _mix_with_uniform(arcs, w)
            else:
                # pure random vector, then simplex‑project
                seq = np.random.rand(n)
                seq = _project_onto_simplex(seq, s=1.0)

            # safeguard – all entries non‑negative and sum = 1
            seq = _project_onto_simplex(seq, s=1.0)

            # cheap internal score
            sc = _internal_score(seq, n)
            candidates.append((sc, seq))

        # keep the best few candidates of this length
        candidates.sort(key=lambda x: x[0])
        top_candidates = [seq for (_, seq) in candidates[:4]]

        # also add deterministic seeds (previous best if same length)
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size == n:
                prev = _project_onto_simplex(np.maximum(prev, 0.0), s=1.0)
                top_candidates.append(prev)
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 2) Refine each promising candidate
        for a0 in top_candidates:
            if time.time() >= deadline - 5.0:
                break

            # Exponential‑gradient refinement
            seq_e, sc_e = _optimize_exponential(a0, n, deadline)

            # Sub‑gradient refinement (starting from the result of the previous step)
            seq_s, sc_s = _subgradient_descent(seq_e, n, deadline)

            # choose the better of the two refined vectors
            if sc_s < sc_e:
                best_local_seq = seq_s
                best_local_score = sc_s
            else:
                best_local_seq = seq_e
                best_local_score = sc_e

            # Full black‑box evaluation (identical to internal score under sum=1)
            try:
                external_score = evaluate_sequence(best_local_seq.tolist())
            except Exception:
                external_score = best_local_score

            if external_score < overall_best_score:
                overall_best_score = external_score
                overall_best_seq = best_local_seq.copy()
                print(f"*** New global best → len={n} score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break

        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # 3) Final stochastic local refinement on the globally best vector
    if overall_best_seq is not None and time.time() < deadline - 2.0:
        n = len(overall_best_seq)
        seq_ref, _ = _local_refinement(overall_best_seq, n, deadline)
        try:
            final_score = evaluate_sequence(seq_ref.tolist())
        except Exception:
            final_score = evaluate_sequence(overall_best_seq.tolist())
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq_ref.copy()
            print(f"*** After local refinement score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Safety fallback used.")

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
