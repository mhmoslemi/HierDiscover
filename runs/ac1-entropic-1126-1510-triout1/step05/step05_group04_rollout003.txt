"""
Propose a low‑score step function.

Algorithm
---------
1. **Diversified initialization** – generate a set of candidate
   sequences (uniform, exponential, reverse exponential, Dirichlet,
   random Gaussian, and the reference if available) and keep the
   best one.

2. **Smooth‑max L‑BFGS optimisation** – minimise a smooth surrogate of
   the hard max (max + μ·log‑sum‑exp) with a continuation on the
   temperature μ.  Gradients are computed via FFT‑based convolutions,
   which makes each iteration cheap even for n≈8000.  Multiple restarts
   and a slower cooling schedule (μ←0.5 μ) explore the landscape.

3. **Hard‑max sub‑gradient descent** – starting from the best incumbent,
   repeatedly move opposite to a sub‑gradient of the exact objective,
   project onto the ℓ¹‑simplex and accept only improving moves.
   A simple jitter helps escape plateaus.

4. **Large‑scale random‑block perturbations** – create many piecewise‑
   constant candidates with random block structure, project, and keep
   any improvement.

5. **Monotone‑sorting heuristics** – try sorting the best vector in
   both orders (descending / ascending) and keep the better one.

All steps respect the global time budget (default 1000 s).  The
function returns the best sequence found; its score is printed during
execution.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ---------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ---------------------------------------------------------------
def _next_pow_of_two(x: int) -> int:
    """Small utility returning the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ---------------------------------------------------------------
def _conv_full_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of two real vectors using FFT.
    Returns an array of length len(u)+len(v)-1.
    """
    n = u.shape[0]
    m = v.shape[0]
    size = _next_pow_of_two(n + m - 1)
    fu = np.fft.rfft(u, size)
    fv = np.fft.rfft(v, size)
    conv = np.fft.irfft(fu * fv, size)[: n + m - 1]
    return conv


# ---------------------------------------------------------------
def _compute_smooth_obj_and_grad(
    y: np.ndarray, mu: float, n: int, S: float
) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate:
        f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Parameters
    ----------
    y : ndarray (n,)
        Unconstrained log‑odds.
    mu : float (>0)
    n : int – length of the step function.
    S : float – target ℓ¹ norm (Σ a_i = S).

    Returns
    -------
    f_val : float
    grad_y : ndarray (n,)
    """
    # probabilities on the simplex, actual heights
    p = _softmax(y)      # Σ p_i = 1, p_i ≥ 0
    a = S * p            # Σ a_i = S

    # convolution a * a (full length 2n‑1) via FFT
    conv = _conv_full_fft(a, a)
    max_c = conv.max()

    # smooth‑max weights w_k ∝ exp((conv_k - max_c)/μ)
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – uniform weight on all active indices
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = float(max_c)

    # gradient w.r.t. heights a:
    #   g_a[i] = 2 * Σ_k w_k * a_{k-i}
    conv_wr = _conv_full_fft(w, a[::-1])          # length 3n‑2
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]        # slice length n

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                        # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ---------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable returning (value, gradient) for scipy.optimize."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)
