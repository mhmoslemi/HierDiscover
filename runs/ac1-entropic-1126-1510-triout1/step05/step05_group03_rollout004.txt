"""Propose a low‑score step‑function.

This implementation extends the previous three‑phase heuristic with a
linear‑programming (LP) refinement step that follows the classic
“f‑g” iteration: for a fixed non‑negative vector `f` we solve

    maximize    Σ b_j
    subject to  f * b ≤ M   (convolution bound)
                b_j ≥ 0

where `M = max(f*f)`.  The solution provides a vector `b` whose total
mass can be larger than 1.  After normalisation we mix `b` into `f`
and repeat.  The LP step quickly lowers the convolution supremum.
After a short LP phase we upscale the distribution, perform Adam
optimisation of a smooth max surrogate, a sub‑gradient polishing,
and a final simulated‑annealing polish.  All work under the 1000 s
budget and respect `evaluate_sequence`’s non‑negative and
normalisation constraints.
"""

import time

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of the smooth max surrogate Lα."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam optimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # adaptive step
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """Exact ℓ∞ sub‑gradient step (uniform weight on active set)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max sub‑gradient polishing."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Simple simulated‑annealing polishing."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        max_delta = min(cur_a[src], 0.07)            # allow up to 7 % moves
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 30000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the matrix A (size (2n‑1)×n) such that A @ b = f*b (full conv)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes f over rows j … j+n‑1
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> tuple[np.ndarray, float]:
    """Solve the LP that maximises Σb subject to f*b ≤ M; return b and Σb."""
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σb  → minimise −Σb
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0.0, None)] * n,
                  method='highs')
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)
    b = res.x
    S = b.sum()
    return b, S


# ----------------------------------------------------------------------
def _lp_iterative_improvement(f0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform a few LP‑based improvement steps.
    Stops when time runs out or no improvement is observed.
    """
    f = f0.copy()
    n = f.shape[0]
    best_f = f.copy()
    best_M = _fft_self_convolve(f).max()
    it = 0
    while time.time() < deadline:
        it += 1
        # current bound from self‑convolution
        M = best_M
        try:
            b, S = _lp_step(f, M)
        except Exception as exc:
            print("[LP] step failed:", exc)
            break
        if S <= 1.0 + 1e-8:
            # cannot increase total mass → nothing to gain
            print("[LP] no mass increase (S≈1) after", it, "iterations")
            break

        # normalise the LP solution
        b_norm = b / S

        # try a moderate mixing; if it does not improve, try a smaller step
        for t in (0.5, 0.2, 0.1):
            new_f = (1.0 - t) * f + t * b_norm
            new_f = _project_onto_simplex(new_f, s=1.0)
            new_M = _fft_self_convolve(new_f).max()
            if new_M < best_M - 1e-12:
                f = new_f
                best_f = new_f
                best_M = new_M
                print(f"[LP] iter {it:2d} mix {t:.2f} → new M={new_M:.8f}")
                break
        else:
            # none of the mixings improved the bound → stop
            print("[LP] mixing did not improve after", it, "iterations")
            break
    return best_f


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence() score.
    The routine combines:
      * a broad search of simple parametric families,
      * an LP‑based refinement on a moderate size,
      * up‑scaling,
      * Adam optimisation of a smooth ℓ∞ surrogate,
      * exact‑max sub‑gradient polishing,
      * and a simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if present.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
        base_n = len(base_seq)
        print("[init] reusing height_sequence_1 of length", base_n)
    except Exception:
        base_seq = None
        base_n = 3800   # default long length when nothing is stored

    # ------------------------------------------------------------------
    # Phase 1 – cheap broad search on a *moderate* length (≤2000)
    # ------------------------------------------------------------------
    small_n = min(2000, base_n) if base_n >= 500 else 1500
    print("[phase 1] broad search on length", small_n)

    starters = {}
    starters["arcsine"] = _arcsine_distribution(small_n)
    starters["uniform"] = np.full(small_n, 1.0 / small_n, dtype=np.float64)
    starters["random"] = np.random.rand(small_n)
    starters["random"] /= starters["random"].sum()
    # a few symmetric Beta shapes (α<1 concentrates near the edges)
    for a in (0.35, 0.45, 0.55):
        starters[f"beta_{a:.2f}"] = _beta_distribution(small_n, a, a)

    if base_seq is not None:
        starters["prev"] = _interpolate_to_length(base_seq, small_n)

    best_small_seq = None
    best_small_score = float("inf")
    for tag, vec in starters.items():
        vec = _project_onto_simplex(vec, s=1.0)
        peak = _fft_self_convolve(vec).max()
        fast_score = 2.0 * small_n * peak
        print(f"[init] {tag:<12s} fast_score={fast_score:.6f}")
        if fast_score < best_small_score:
            best_small_score = fast_score
            best_small_seq = vec.copy()

    if best_small_seq is None:
        # fallback (should never happen)
        best_small_seq = np.full(small_n, 1.0 / small_n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based improvement on the small sequence
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 10.0:
        phase2_deadline = time.time() + time_left * 0.35
        print(f"[phase 2] LP refinements (n={small_n}) – time left {time_left:.1f}s")
        best_small_seq = _lp_iterative_improvement(best_small_seq, phase2_deadline)
        best_small_score = 2.0 * small_n * _fft_self_convolve(best_small_seq).max()
        print(f"[phase 2] after LP fast_score={best_small_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – upscale to a larger length (≈ the original best length)
    # ------------------------------------------------------------------
    large_n = max(3500, int(base_n * 1.05))
    print(f"[phase 3] upscale to length {large_n}")
    large_seq = _interpolate_to_length(best_small_seq, large_n)
    large_seq = _project_onto_simplex(large_seq, s=1.0)

    # ------------------------------------------------------------------
    # Phase 4 – Adam optimisation of smooth max surrogate
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 8.0:
        phase4_deadline = time.time() + time_left * 0.40
        print(f"[phase 4] Adam optimisation (n={large_n}) – time left {time_left:.1f}s")
        a_opt, _ = _adam_optimize(large_seq, phase4_deadline,
                                   init_alpha=0.5,
                                   alpha_factor=1.004,
                                   alpha_max=1500.0,
                                   lr0=0.15)
        large_seq = a_opt.copy()
        large_score = 2.0 * large_n * _fft_self_convolve(large_seq).max()
        print(f"[phase 4] post‑Adam fast_score={large_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – exact max sub‑gradient polishing
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        phase5_deadline = time.time() + time_left * 0.30
        print(f"[phase 5] sub‑gradient polishing (n={large_n}) – time left {time_left:.1f}s")
        large_seq = _refine_max(large_seq, phase5_deadline)
        large_score = 2.0 * large_n * _fft_self_convolve(large_seq).max()
        print(f"[phase 5] after polishing fast_score={large_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – simulated‑annealing polish
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 2.0:
        print(f"[phase 6] simulated annealing (n={large_n}) – time left {time_left:.1f}s")
        a_final, final_peak = _simulated_annealing(large_seq, deadline)
        large_seq = a_final.copy()
        large_score = 2.0 * large_n * final_peak
        print(f"[phase 6] SA fast_score={large_score:.6f}")

    # ------------------------------------------------------------------
    # Final black‑box evaluation
    # ------------------------------------------------------------------
    try:
        true_score = evaluate_sequence(large_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)
        best_score = large_score

    print(f"Final best approx. score {best_score:.6f} (length={len(large_seq)})")
    return large_seq.tolist()
