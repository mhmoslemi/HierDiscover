def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0

    best_seq = None
    best_score = float('inf')
    # try series of lengths
    lengths = [2000,4000,6000,8000,12000,16000,20000]
    for n in lengths:
        # arcsine baseline
        seq = _arcsine_distribution(n)
        score = _objective(seq)
        if score < best_score:
            best_score=score; best_seq=seq.copy()
            print(f"Baseline arcsine n={n} score={score:.6f}")
        # power law inv around p=0.5
        # first coarse grid
        for p in np.linspace(0.4,0.6,21):
            seq = _power_law_distribution(n, p, invert=True)
            # seq already sums to 1
            score = _objective(seq)
            if score < best_score:
                best_score=score; best_seq=seq.copy()
                print(f"Coarse p={p:.3f} n={n} score={score:.6f}")
            # could break if time near end
            if time.time() > deadline - 100:
                break
        if time.time() > deadline - 100:

        # compute gradient via softmax
        grad = _gradient_softmax(a, b, alpha, size_fft)

"""
Propose a near‑optimal non‑negative step function.

Algorithm
---------
1. **Coarse grid search** over several sequence lengths (up to 20 000)
   and over the exponent `p` of the symmetric power‑law family  

   a_i ∝ ((i+½)(n‑i‑½))^{-p}   (the arcsine law corresponds to p=0.5).

   For each candidate we project onto the simplex (so the sum is 1) and
   evaluate the fast surrogate objective  

   `obj(a) = 2·n·max(a*a)`, which equals the official
   `evaluate_sequence` value for normalized sequences.

2. **Smooth exponential‑gradient refinement** (mirror descent on the
   simplex).  The gradient is the soft‑max approximation of the
   sub‑gradient of the max‑convolution, with a temperature that is
   slowly increased so that the optimisation first explores a smooth
   landscape and later focuses on the true maximum.  The update is

        a ← a ⊙ exp( -η·∇ )   (⊙ = element‑wise product)

   followed by a projection onto the simplex.

3. **Local random‑move hill climbing** – tiny mass transfers between
   two coordinates are accepted only if they improve the exact
   objective.

All steps respect the overall 1000 s budget and the best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum(a)=1)."""
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv  # denominator is 1 because sum == 1


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """Symmetric power‑law family: a_i ∝ ((i+½)(n‑i‑½))^{±p}."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (the p=0.5 case of the above)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, b: np.ndarray,
                     alpha: float, size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·b_k) w.r.t. a.
    Implements g = 2 * conv_rev(w, a_rev) efficiently.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Smooth mirror‑descent on the simplex using a soft‑max surrogate for max.
    Returns the best sequence seen and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)

    # parameters
    eta0 = 0.7                     # initial step‑size
    alpha = 1.0                    # soft‑max temperature
    alpha_mul = 1.015
    alpha_max = 4000.0
    size_fft = 1 << ((3 * n - 2).bit_length())   # for gradient conv

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and exact score
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] it={it} new best {best_score:.6f}")

        # soft‑max gradient
        grad = _gradient_softmax(a, b, alpha, size_fft)

        # diminishing learning rate
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaux
        if it % 4000 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Random‑move hill‑climbing: move a small fraction of mass from i → j
    if the exact objective improves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.07   # up to 7 % move
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cur_score = _objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] hill‑climb it={it} score={best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1) Coarse grid search over length & power‑law exponent
    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000]
    best_seq = None
    best_score = float("inf")

    for n in lengths:
        # arcsine baseline (p = 0.5)
        seq = _arcsine_distribution(n)
        score = _objective(seq)
        if score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"Arcsine n={n}  score={score:.6f}")

        # coarse sweep of p around 0.5
        for p in np.linspace(0.40, 0.60, 21):   # step ≈0.01
            seq = _power_law_distribution(n, p, invert=True)
            score = _objective(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"Power p={p:.3f} n={n}  score={score:.6f}")

            if time.time() > deadline - 120:   # keep ~2 min for refinements
                break
        if time.time() > deadline - 120:
            break

    # ------------------------------------------------------------------
    # 2) Refine the current best with smooth exponentiated‑gradient
    if best_seq is not None and time.time() < deadline - 40:
        rem = deadline - time.time()
        # give up to 35 s for the first refinement, the rest for a second if possible
        slice_deadline = time.time() + min(35.0, rem * 0.6)
        refined_seq, refined_score = _optimize_exponential(best_seq, slice_deadline)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"Refinement improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final local random‑move hill climbing
    if best_seq is not None and time.time() < deadline - 5:
        refined_seq, refined_score = _local_refinement(best_seq, deadline)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"After hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net
    if best_seq is None:
        best_seq = np.full(1000, 1.0 / 1000, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
