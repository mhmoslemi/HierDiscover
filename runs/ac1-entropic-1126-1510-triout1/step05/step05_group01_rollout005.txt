def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)   # tolerance
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for self-products: each active k that is even contributes extra a[i].
    # The original function subtracted; we keep same implementation to be safe.
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad

    lr0 = 0.2   # base learning rate

        t += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # print progress
            print(f"[{time.time():.1f}] n={n} iter={t} score={cur_score:.6f}")
            if best_score <= target_score:
                break
        # gradient
        grad = _exact_gradient(a, conv)
        # Adam update

        v = beta2 * v + (1 - beta2) * (grad ** 2)

"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Broad sampling** – For several candidate lengths (around the length of the
   previously known construction) we generate a modest library of analytic
   distributions (Beta / Dirichlet) and a few random draws.  Each candidate is
   normalised to sum = 1 and scored cheaply using an FFT‑based self‑convolution.
   The best *K* candidates (global across all lengths) are kept.

2. **Intensive refinement** – The retained candidates are optimised one‑by‑one
   with a projected Adam optimiser on the simplex.  In the first few hundred
   iterations a smooth “soft‑max’’ surrogate of the max‑convolution is used;
   afterwards the exact sub‑gradient of the max is applied.  After each update
   the iterate is symmetrised (a←½(a+aᵀ)) and projected onto the simplex.

3. **Final evaluation** – The best refined vector is evaluated with the official
   ``evaluate_sequence`` function.  If the target 1.503 is reached the search
   stops early.

The code respects the supplied time budget (default 1000 s) and prints
intermediate progress.  All helper functions are defined at top level so that
the evaluator can import ``propose_candidate`` directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
def _dirichlet_distribution(n: int, concentration: float) -> np.ndarray:
    """Draw from Dirichlet(α,…,α) with α = concentration."""
    if concentration <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    g = np.random.gamma(concentration, 1.0, size=n)
    s = g.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return g / s


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Score assuming sum(a)=1  →  2·n·max(conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of smoothed max (soft‑max) of the convolution."""
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max, uniformly over all (approximately) active
    convolution entries.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    max_time: float,
    target_score: float = 1.503,
    warmup_iters: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the simplex.
    Returns the best vector seen (still normalised to sum = 1) and its cheap score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.25

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _cheap_score(a)

    start = time.time()
    t = 0
    alpha = 1.0          # temperature for soft‑max warm‑up
    alpha_factor = 1.02

    while time.time() - start < max_time:
        t += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={t} score={cur_score:.6f}")

        # gradient choice
        if t <= warmup_iters:
            grad = _softmax_gradient(a, conv, alpha)
            alpha = min(alpha_factor * alpha, 5000.0)
        else:
            grad = _exact_gradient(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 / (np.sqrt(v_hat) + eps)

        # additive update on the simplex
        a = a - step * m_hat
        a = np.maximum(a, 0.0)               # keep non‑negative
        a = _project_onto_simplex(a, s=1.0)  # renormalise
        a = (a + a[::-1]) / 2.0              # enforce mirror symmetry

        # occasional tiny jitter to shake off plateaus
        if t % 3000 == 0:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)

        if best_score <= target_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # keep a tiny safety margin

    # ------------------------------------------------------------------
    # 0. Global bookkeeping
    best_seq = None                # type: np.ndarray | None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Determine a set of lengths to explore
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    factors = (0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0, 4.0)
    lengths = sorted(
        {max(500, int(base_n * f)) for f in factors if int(base_n * f) >= 500}
    )
    if not lengths:
        lengths = [8000]

    # ------------------------------------------------------------------
    # 2. Broad sampling – keep the K best candidates globally
    K_GLOBAL = 12
    global_candidates = []   # list of (cheap_score, vector)

    def _maybe_store(vec: np.ndarray) -> None:
        sc = _cheap_score(vec)
        if len(global_candidates) < K_GLOBAL:
            global_candidates.append((sc, vec))
            global_candidates.sort(key=lambda x: x[0])
        else:
            if sc < global_candidates[-1][0]:
                global_candidates[-1] = (sc, vec)
                global_candidates.sort(key=lambda x: x[0])

    phase1_deadline = time.time() + 0.30 * (deadline - time.time())
    for n in lengths:
        if time.time() > phase1_deadline:
            break
        print(f"[{time.time():.1f}] Sampling length n={n}")

        # ---- analytic Beta shapes -------------------------------------------------
        shape_params = [
            (0.2, 0.2), (0.3, 0.3), (0.4, 0.4), (0.5, 0.5),
            (0.75, 0.75), (1.0, 1.0),
            (0.2, 0.8), (0.8, 0.2),
        ]
        for al, be in shape_params:
            vec = _beta_distribution(n, al, be)
            _maybe_store(vec)

        # ---- random Dirichlet draws -----------------------------------------------
        for conc in (0.2, 0.5, 1.0):
            vec = _dirichlet_distribution(n, conc)
            _maybe_store(vec)

        # ---- uniform ---------------------------------------------------------------
        _maybe_store(np.full(n, 1.0 / n, dtype=np.float64))

        # ---- previous construction (if available) ---------------------------------
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                vec = np.interp(x_new, x_old, prev)
                vec = vec / vec.sum()
            else:
                vec = prev.copy()
                vec = vec / vec.sum()
        except Exception:
            vec = _beta_distribution(n, 0.5, 0.5)
        _maybe_store(vec)

    # ------------------------------------------------------------------
    # 3. Intensive refinement of the top candidates
    remaining = max(0.0, deadline - time.time())
    if remaining <= 0.0:
        remaining = 0.0
    if not global_candidates:
        # safety fallback
        uniform = np.full(1000, 1.0 / 1000, dtype=np.float64)
        return uniform.tolist()

    time_per_candidate = remaining / len(global_candidates)
    for idx, (init_score, a0) in enumerate(global_candidates):
        now = time.time()
        sub_time = min(time_per_candidate, deadline - now - 0.5)
        if sub_time < 5.0:
            break

        print(f"[{now:.1f}] Refining candidate {idx + 1}/{len(global_candidates)} "
              f"(len={a0.shape[0]}, init_score={init_score:.6f})")
        a_opt, cheap_opt = _adam_optimize(a0, sub_time, target_score=1.503)

        # Exact evaluation
        try:
            exact_score = evaluate_sequence(a_opt.tolist())
        except Exception:
            exact_score = cheap_opt  # fallback to cheap score

        print(f"[{time.time():.1f}] Candidate {idx + 1} exact score = {exact_score:.6f}")
        if exact_score < best_score:
            best_score = exact_score
            best_seq = a_opt.copy()
            print(f"*** New global best → score={best_score:.6f}")

        if best_score <= 1.503:
            print("Target reached – returning early.")
            return best_seq.tolist()

        # respect the overall deadline
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Fallback (should never be needed)
    if best_seq is None:
        n_fallback = 1000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Fallback used, score={best_score:.6f}")

    print(f"[{time.time():.1f}] Final best score: {best_score:.6f} "
          f"(length={len(best_seq)})")
    return best_seq.tolist()
