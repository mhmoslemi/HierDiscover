def _adam_optimize(a0, n, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2  # initial learning rate? Might be large.

    # temperature schedule
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 250.0

    size_fft = 1 << ((3 * n - 2).bit_length())
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
        # gradient
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # learning rate decays
        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # occasional random jitter
        if t % 4000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)
    return best_a, best_score

"""
Propose a step‑function (list of non‑negative heights) with a low
evaluation score.

The algorithm mixes several ideas:

1. **Rich family of initial shapes** – arcsine law, power‑law
   distributions, a mixture with the uniform distribution and random
   Dirichlet draws.  The best‑known construction (``height_sequence_1``)
   is also interpolated to the current length.

2. **Projected Adam optimisation** – minimise a smooth soft‑max
   surrogate of the ∞‑norm of the self‑convolution.
   The temperature (``alpha``) is increased gradually so the optimisation
   finally works on the true max.  After each Adam step the vector is
   projected onto the probability simplex (non‑negative, sum = 1).

3. **Final stochastic hill‑climbing** – tiny mass transfers between two
   positions are accepted only if they improve the true evaluation
   score.

The routine respects the time budget (default 1000 s) and constantly
stores the best sequence seen.  If the target score < 1.503 is reached it
terminates early.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) convolution of *a* with itself via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max entry of ``a*a`` at index ``k``."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate (1/α)·log∑exp(α·b) w.r.t. ``a``."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:                     # degenerate – fall back to hard max
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _beta_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ (p>0)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create a normalised (sum‑1) non‑negative vector of length *n*."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif kind == "beta":
        a = np.random.beta(0.5, 0.5, size=n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = np.full(n, 1.0 / n) if s == 0 else a / s
    elif kind == "arcsine":
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a /= a.sum()
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max surrogate of the L∞‐norm.
    Returns the best vector encountered and its internal (2·n·max) score.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.25                     # initial learning‑rate factor

    alpha = 1.0                    # temperature for soft‑max
    alpha_mul = 1.02
    alpha_max = 250.0

    size_fft = 1 << ((3 * n - 2).bit_length())
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()

        # ---- gradient (soft‑max → true max) ----
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # ---- Adam update ----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                     # ascend in negative gradient direction
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter to escape plateaus
        if it % 4000 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a small fraction of mass from one
    index to another whenever the true evaluation score improves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.05      # move up to 5 % of source mass
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        # re‑normalise (keeps sum = 1 and guarantees non‑negativity)
        cand = _project_onto_simplex(cand, s=1.0)
        cur_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------

    try:

    # A few candidate lengths (keep them ≤ 20000 for speed)
    lengths = sorted(
