def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0

    # Determine lengths to explore
    lengths = [4000, 6000, 8000, 10000, 12000, 15000]

    best_seq = None
    best_score = float('inf')
    best_length = None

    for n in lengths:
        if time.time() >= deadline: break
        # generate candidates for this n
        candidates = generate_candidates(n)
        # evaluate quick each candidate and keep best
        local_best = None
        local_best_score = float('inf')
        for cand in candidates:
            if time.time() >= deadline: break
            scr = evaluate_sequence(cand.tolist())
            if scr < local_best_score:
                local_best_score = scr
                local_best = cand.copy()
                print(f"[init] n={n} new best init score {scr:.6f}")
        # perform several independent Adam runs
        restarts = 5
        for r in range(restarts):
            if time.time() >= deadline: break
            # start from local_best plus jitter
            init = local_best.copy()
            # jitter a bit
            jitter_scale = 0.02 / (r+1)  # decreasing jitter
            init += np.random.normal(scale=jitter_scale, size=n)
            init = np.clip(init, 0, None)
            init = _project_onto_simplex(init, 1.0)

            # run Adam with time limit fraction (maybe 0.1* remaining)
            sub_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.15)
            seq_opt, _ = _adam_optimize(init, n, sub_deadline)

            scr_opt = evaluate_sequence(seq_opt.tolist())
            if scr_opt < best_score:
                best_score = scr_opt
                best_seq = seq_opt.copy()
                best_length = n
                print(f"[adam] n={n} restart={r} new global best {best_score:.6f}")

        # optional extra refinement for this n
        if best_seq is not None and n == best_length:
            sub_deadline = deadline
            seq_refined, scr_refined = _hill_climb(best_seq, n, sub_deadline)
            if scr_refined < best_score:
                best_score = scr_refined
                best_seq = seq_refined.copy()
                print(f"[hill] new global best after hill {best_score:.6f}")

    # fallback ensure a valid sequence
    if best_seq is None:
        best_seq = np.full(2000, 1.0 / 2000, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"Final best score {best_score:.6f} length {len(best_seq)}")
    return best_seq.tolist()

def _candidate_shapes(n, prev_seq):
    candidates = []
    # Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))
    # arcsine
    candidates.append(_arcsine_distribution(n))
    # symmetric beta a range
    for a in np.linspace(0.1, 0.9, 9):
        candidates.append(_beta_distribution(n, a))
    # exponential
    for lam in np.linspace(0.0005, 0.02, 8):
        cand = np.exp(-lam * np.arange(n, dtype=np.float64))
        cand /= cand.sum()
        candidates.append(cand)
    # power law
    for p in np.linspace(0.5, 2.0, 7):
        i = np.arange(1, n+1, dtype=np.float64)
        cand = 1.0 / np.power(i, p)
        cand /= cand.sum()
        candidates.append(cand)
    # mixture of uniform + arcsine
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    arcsine = _arcsine_distribution(n)
    for w in np.linspace(0.0, 0.5, 6):
        candidates.append((1-w)*arcsine + w*uniform)
    # previous best (if any)
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())
    # reversed previous best to explore asymmetry
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq[::-1].copy())
    # random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(5):
        cand = rng.dirichlet(np.ones(n))
        candidates.append(cand)
    # sorted Dirichlet (decreasing)
    for _ in range(3):
        cand = rng.dirichlet(np.ones(n))
        cand = np.sort(cand)[::-1]
        candidates.append(cand)
    # ensure simplex
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final

"""
Search for a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  The strategy extends the previous state‑of‑the‑art:

1️⃣  Generate a diverse set of parametric “seed” vectors on several
    lengths (uniform, arcsine, symmetric Beta, exponential, power‑law,
    mixtures and random Dirichlet draws).  Every seed is projected onto
    the probability simplex (∑a=1).

2️⃣  For each length run a number of *projected Adam* optimisations.
    The Adam step works on the simplex, the objective is the smooth
    soft‑max surrogate of `max(conv(a,a))`.  Temperature α starts at 5
    and exponentially grows (up to 5·10⁴).  Restarts are jittered copies
    of the best vector seen for that length.

3️⃣  After the Adam phase a cheap stochastic hill‑climber (mass swaps
    + occasional Gaussian jitter) polish the current global best.

All work is bounded by the 1000 s budget; `evaluate_sequence` is called
only for the final answer (intermediate scores are obtained with a fast
FFT‑based estimator).  The routine returns a plain list of floats ready
for the grader.
"""

import time
import math
import numpy as np
from scipy.stats import beta as beta_dist  # only for the name, not used directly


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    # next power‑of‑two for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·b_k) w.r.t. a.
    For huge α the soft‑max degenerates to a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # ---- soft‑max weights ------------------------------------------------
    if alpha > 5_000:
        max_b = b.max()
        mask = (np.abs(b - max_b) < 1e-12).astype(float)
        cnt = mask.sum()
        if cnt > 0:
            w = mask / cnt
        else:                      # fall‑back (numerically impossible)
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
    else:
        # classic log‑sum‑exp trick
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # ---- convolution w * reversed(a) ------------------------------------
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]            # factor 2 from derivative of a*a
    return g


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law on n points (Beta(½,½) up to normalisation)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) on n bins (mid‑point Riemann sum)."""
    a = max(a, 0.05)                      # avoid singularities
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0 or not np.isfinite(pdf_sum):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= pdf_sum
    return pdf


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    """Unnormalised e^{-lam·i} on n points, then normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """1/(i+1)^p on n points, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = np.power(i, -p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Generate a reasonably rich collection of seed vectors for length `n`.
    Every vector is projected onto the probability simplex.
    """
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) arcsine (Beta(½,½))
    candidates.append(_arcsine_distribution(n))

    # 3) symmetric Beta(a,a) for a grid
    for a in np.linspace(0.10, 0.90, 9):
        candidates.append(_beta_distribution(n, a))

    # 4) exponential families (different decay rates)
    for lam in np.linspace(0.0005, 0.020, 8):
        candidates.append(_exponential_distribution(n, lam))

    # 5) power‑law families
    for p in np.linspace(0.5, 2.0, 7):
        candidates.append(_power_law_distribution(n, p))

    # 6) mixtures uniform + arcsine
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    arcsine = _arcsine_distribution(n)
    for w in np.linspace(0.0, 0.5, 6):
        candidates.append((1.0 - w) * arcsine + w * uniform)

    # 7) mixtures uniform + exponential (adds a slight front‑heavy tilt)
    for w in np.linspace(0.0, 0.5, 5):
        exp_shape = _exponential_distribution(n, 0.005)
        candidates.append((1.0 - w) * exp_shape + w * uniform)

    # 8) previous best (if supplied) and its reversal – adds asymmetry
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())
        candidates.append(prev_seq[::-1].copy())

    # 9) random Dirichlet draws + sorted copies (descending)
    rng = np.random.default_rng()
    for _ in range(5):
        rand = rng.dirichlet(np.ones(n))
        candidates.append(rand)
        candidates.append(np.sort(rand)[::-1])          # decreasing version

    # final projection onto the simplex (ensures exact sum = 1)
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final


# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Fast internal proxy for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex.
    Returns the best vector seen during the run together with its
    internal (2·n·max(conv)) score.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = _fast_score(best)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.05
    alpha_max = 50_000.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # ≥ (2n‑1)+(n)‑1

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            # occasional progress print
            if it % 200 == 0:
                print(f"[Adam] n={n} it={it} best={best_score:.6f}")

        # gradient of the smooth surrogate
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate 0.3 works well for the early stage, we decay it
        step = 0.3 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection back onto the simplex
        a = _project_onto_simplex(a, 1.0)

        # temperature increase
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search:
      – move a random fraction of the mass from one index to another;
      – keep move if it improves the true objective;
      – every few thousand steps add a tiny Gaussian jitter.
    """
    best = start_a.copy()
    best_score = _fast_score(best)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue

        # move a random fraction (up to 30 %) of the source mass to the target
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter to escape shallow plateaus
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = _fast_score(jitter)
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score

    Returns a list of non‑negative heights that (approximately) minimise

    deadline = time.time() + budget_s - 2.0          # safety margin

    # but stay within the safe window [3000, 15000].
    try:

    except Exception:
        base_len = 8000

    lengths = [3000, 4000, 6000, 8000, 10000, 12000, 15000]
    # make sure we test lengths that are not absurdly larger than base_len
    lengths = [min(l, 15000) for l in lengths if l >= 2000]

    # ------------------------------------------------------------------
    # Remember the globally best solution.
    global_best_seq = None
    global_best_score = float("inf")
    global_best_n = None

    # ------------------------------------------------------------------
    # Main exploration loop over lengths.
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Exploring length n={n} ===")

        # Try to obtain a previous best of the same length (if any)
        prev_seq = None
        if global_best_seq is not None and global_best_n == n:
            prev_seq = global_best_seq
        else:
            try:
                prev_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
                if prev_seq.size != n:
                    # resample to required size by linear interpolation
                    x_old = np.linspace(0, 1, prev_seq.size)
                    x_new = np.linspace(0, 1, n)
                    prev_seq = np.interp(x_new, x_old, prev_seq)
                prev_seq = _project_onto_simplex(prev_seq, 1.0)
            except Exception:
                prev_seq = None

        # ----------------------------------------------------------------
        # Phase 1 – generate & evaluate cheap candidates
        shape_deadline = time.time() + min(20.0,
                                           (deadline - time.time()) * 0.10)
        candidates = _candidate_shapes(n, prev_seq)
        local_best = None
        local_best_score = float("inf")

        for cand in candidates:

            if sc < local_best_score:
                local_best_score = sc
                local_best = cand.copy()
                print(f"[shape] n={n} new local best {sc:.6f}")

        # Fallback – uniform (should never be needed)

            local_best_score = _fast_score(local_best)

        # ----------------------------------------------------------------

        # Allocate roughly 60 % of the remaining budget to Adam

                break
            # jitter the start point a little to diversify the restarts

            init = local_best + np.random.normal(scale=jitter_scale, size=n)
            init = np.clip(init, 0.0, None)
            init = _project_onto_simplex(init, 1.0)

            sub_deadline = time.time() + min(30.0, adam_time / n_restarts)
            seq_opt, _ = _adam_optimize(init, n, sub_deadline)

            sc_opt = _fast_score(seq_opt)
            if sc_opt < global_best_score:
                global_best_score = sc_opt
                global_best_seq = seq_opt.copy()
                global_best_n = n
                print(f"[Adam] n={n} restart={r} NEW GLOBAL BEST {sc_opt:.6f}")

        # ----------------------------------------------------------------

        # (use whatever time is left)

            seq_ref, sc_ref = _hill_climb(global_best_seq, n, hill_deadline)
            if sc_ref < global_best_score:
                global_best_score = sc_ref
                global_best_seq = seq_ref.copy()
                print(f"[Hill] n={n} refined GLOBAL BEST {sc_ref:.6f}")

    # fall back to a uniform distribution of a modest size.
