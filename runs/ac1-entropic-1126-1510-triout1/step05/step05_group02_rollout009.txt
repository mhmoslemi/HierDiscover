def _smooth_gradient(a, gamma):
    n = a.shape[0]
    conv = _fft_self_convolve_full(a) # shape 2n-1
    max_conv = conv.max()
    # compute weights
    w = np.exp(gamma * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0:
        # fallback to uniform weight on max indices
        active = np.where(conv >= max_conv - 1e-12)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)
    else:
        w = w / w_sum
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w) # returns length n + 2n-1 -1 = 3n-2
    start = n-1
    grad = 2.0 * grad_full[start: start + n]
    return grad, conv.max()

def adam_optimize(a_init, time_deadline, n_iter_max=None):
    a = a_init.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t = 0
    best_a = a.copy()
    best_score = raw_score(a)
    # gamma schedule
    gamma = 10.0
    gamma_target = 2000.0
    gamma_factor = (gamma_target/gamma)**(1/2000) # inc factor per iter (approx)
    while time.time() < time_deadline:
        t += 1
        grad, cur_max = _smooth_gradient(a, gamma)
        # Adam:
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.01 / (np.sqrt(t)) # decaying
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_onto_simplex(a)
        cur_score = raw_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam it={t} score={best_score:.6f}")
        # update gamma gradually
        if gamma < gamma_target:
            gamma *= gamma_factor
        if n_iter_max is not None and t >= n_iter_max:
            break
    return best_a

def _fft_convolve(a,b):

    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)

def _adam_refine(a, time_deadline, init_lr=0.02, gamma_start=5.0, gamma_end=500.0, max_iters=1000000):
   n = len(a)
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1,beta2=0.9,0.999
   eps=1e-8
   t=0
   best_a=a.copy()
   best_score=_raw_score(a)
   # schedule for gamma using simple exponential increase:
   gamma=gamma_start
   gamma_factor = (gamma_end/gamma_start) ** (1/2000)  # increase per iter upto 2000 step; after that keep max
   while time.time() < time_deadline:
        t += 1
        grad, cur_max = _smooth_gradient(a, gamma)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        # learning rate schedule: decays as lr = init_lr / sqrt(t)
        lr = init_lr / math.sqrt(t)
        step = lr * m_hat/(np.sqrt(v_hat)+eps)
        a = a - step
        a = _project_onto_simplex(a)
        cur_score=_raw_score(a)
        if cur_score < best_score - 1e-12:
            best_score=cur_score
            best_a=a.copy()
            print(f"[{time.time():.1f}] Adam it={t} gamma={gamma:.2f} score={best_score:.6f}")
        # gamma update
        if gamma < gamma_end:
            gamma = min(gamma_end, gamma*gamma_factor)
        # optional early stop if stagnation
        if max_iters and t>=max_iters:
            break
   return best_a

   final_score = evaluate_sequence(final_seq.tolist())
   print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
   return final_seq.tolist()

"""
Propose a low‑score step‑function.

Algorithm
---------
1. Global exploration:
   * Test a few deterministic shapes (uniform, arcsine, tent).
   * Scan symmetric Beta(α,α) across a wide range.
   * Randomly generate mixtures of up to three Beta densities (α,β ∈ [0.05,5.05]).
   * All candidates are projected onto the simplex (Σa_i = 1, a_i ≥ 0).
   * Keep the best candidate (lowest 2·n·max(a*a)) for each tested length.

2. Local refinement:
   * Use a smooth approximation of the ℓ∞‑norm:
         f_γ(a) = (1/γ)·log(∑_k exp(γ·(a*a)_k)),
     whose gradient is a weighted sum of the convolution gradients.
   * Optimise f_γ with Adam (adaptive moment estimation) while
     projecting onto the simplex after each update.
   * γ is increased slowly (γ: 5 → 500) so the optimisation starts
     with a smooth landscape and finishes with an accurate max‑approximation.
   * The refinement runs for the remaining time budget.

3. Return the best sequence (as a plain Python list) for the black‑box
   ``evaluate_sequence`` evaluator.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow/underflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf, out=np.zeros_like(logpdf))
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a * a."""
    return _fft_self_convolve_full(a).max()

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b (length a+b‑1) using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Objective used during the search: 2·n·max(a*a).  Σa = 1."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)

# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """
    Gradient of the smoothed max: f_γ(a) = (1/γ)·log Σ exp(γ·(a*a)_k).
    Returns (grad, current_max_value_of_convolution).
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)               # length 2n‑1
    max_conv = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(gamma * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0.0:
        # fall back to uniform weight on active set (identical to sub‑gradient)
        active = np.where(conv >= max_conv - 1e-12)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
    else:
        w = w / w_sum
    # gradient of each term (a*a)_k is 2 * (a_rev * e_k) where e_k is unit at k
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w)            # length 3n‑2
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad, max_conv

# ----------------------------------------------------------------------
def _adam_refine(a0: np.ndarray,
                 deadline: float,
                 init_lr: float = 0.02,
                 gamma_start: float = 5.0,
                 gamma_end: float = 500.0,
                 max_iters: int = 10**9):
    """
    Projected Adam optimisation of the smoothed max objective.
    Stops when the deadline is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = _raw_score(a)
    gamma = gamma_start
    # exponential schedule for γ (≈2000 steps to reach γ_end)
    gamma_factor = (gamma_end / gamma_start) ** (1.0 / 2000.0)

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, cur_max = _smooth_gradient(a, gamma)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # diminishing learning rate
        lr = init_lr / math.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam it={t} γ={gamma:.2f} score={best_val:.6f}")

        # increase smoothing parameter
        if gamma < gamma_end:
            gamma = min(gamma_end, gamma * gamma_factor)

    return best_a
