"""
propose_candidate
=================

A time‑bounded search for a non‑negative step function (a probability vector)
that minimises the given ``evaluate_sequence`` objective.

The algorithm works in three stages:

1. **Broad parametric sweep** (≈10 % of the total budget) – generate many
   hand‑crafted families (uniform, arcsine, symmetric Beta, a 2‑parameter
   “power‑law” family, mixtures and small spikes) on several candidate
   lengths.  The best few candidates are kept.

2. **Projected Adam optimisation** (≈80 % of the remaining budget) – run a
   projected Adam optimiser on the simplex using a soft‑max surrogate for
   the max‑convolution.  The temperature is increased aggressively so that
   the optimiser eventually follows the true sub‑gradient of the max.

3. **Stochastic hill‑climbing** (the rest of the time) – cheap local moves
   (mass transfer between two positions) plus occasional jitter are applied
   to polish the solution.

All helper routines are defined at top level, there are no closures or
lambda‑expressions, and the implementation respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         conv: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·conv_k) w.r.t. a.
    For very large α we fall back to the exact sub‑gradient of the max.
    """
    n = a.shape[0]

    # ---- soft‑max weights -------------------------------------------------
    if alpha > 2000.0:
        max_val = conv.max()
        mask = np.abs(conv - max_val) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(conv)] = 1.0
    else:
        shift = conv - conv.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

    # ---- convolution of weights with reversed a ---------------------------
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    tmp = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    grad = 2.0 * tmp[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) on the mid‑points of n equal bins."""
    a = max(a, 0.05)               # avoid singularities
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    s = pdf.sum()
    if s == 0.0 or not np.isfinite(s):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= s
    return pdf


# ----------------------------------------------------------------------
def _powerlaw_distribution(n: int, p: float = 0.5, shift: float = 0.5) -> np.ndarray:
    """
    Two‑parameter family: a_i ∝ ((i+shift)*(n-i-shift))^{-p}.
    The arcsine law corresponds to p=0.5, shift=0.5.
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + shift) * (n - i - shift)
    # avoid division by zero; shift∈(0,1) guarantees denom>0
    vals = np.power(denom, -p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _candidate_shapes(n: int,
                     prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse set of probability vectors for a given length."""
    cand = []

    # 1) Uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) Arcsine
    cand.append(_arcsine_distribution(n))

    # 3) Symmetric Beta on a grid
    for a in np.linspace(0.25, 1.0, 9):
        cand.append(_beta_distribution(n, a))

    # 4) Power‑law family (grid over exponent and shift)
    for p in [0.40, 0.45, 0.50, 0.55, 0.60]:
        for sh in [0.30, 0.40, 0.50, 0.60, 0.70]:
            cand.append(_powerlaw_distribution(n, p, sh))

    # 5) Mixture arcsine + uniform (weights up to 0.4)
    arcs = _arcsine_distribution(n)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for w in np.linspace(0.0, 0.4, 5):
        mix = (1.0 - w) * arcs + w * uniform
        cand.append(mix)

    # 6) Add spikes at the two ends (10 % and 15 % of total mass)
    base = _arcsine_distribution(n)
    for spike_frac in (0.10, 0.15):
        mix = base.copy()
        mix[0] += spike_frac / 2.0
        mix[-1] += spike_frac / 2.0
        mix = _project_onto_simplex(mix, 1.0)
        cand.append(mix)

    # 7) Previous best (if compatible)
    if prev_seq is not None and prev_seq.size == n:
        cand.append(prev_seq.copy())

    # 8) A handful of random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(4):
        cand.append(rng.dirichlet(np.ones(n)))

    # final sanity – project all onto the simplex
    cand = [_project_onto_simplex(c, 1.0) for c in cand]
    return cand


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex using a soft‑max surrogate.
    Returns the best vector observed (and its raw 2·n·max(conv) value).
    """
    a = init_a.copy()
    best = a.copy()
    best_val = 2.0 * n * _fft_self_convolve(a).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.06
    alpha_max = 60000.0

    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val - 1e-16:
            best_val = cur_val
            best = a.copy()
            # occasional progress report
            if it % 100 == 0:
                print(f"[Adam] n={n} it={it} best={best_val:.6f}")

        # gradient of the smooth approximation of max
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.3 / (np.sqrt(v_hat) + eps)   # a slightly larger base LR
        a = a - step * m_hat

        # projection onto simplex
        a = _project_onto_simplex(a, 1.0)

        # temperature increase
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_val


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local improvement: move a random fraction of the
    mass from one index to another; keep the move if it improves the score.
    """
    best = start_a.copy()
    best_val = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-15:
            continue
        # move a random fraction (0–35 %) of mass from i to j
        frac = rng.random() * 0.35
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_val < best_val:
            best, best_val = cand, cand_val

        # every few thousand steps add a tiny jitter
        if it % 6000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_val = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_val < best_val:
                best, best_val = jitter, jitter_val

    return best, best_val
