def lp_refinement(f, M, t, g_iters, sub_deadline):
    n = len(f)
    # Precompute sqrt2n = sqrt(2*n)
    sqrt2n = math.sqrt(2*n)

    for i in range(g_iters):
        if time.time() > sub_deadline:
            break

        # Build constraint matrix A_ub for current f
        A_ub = build_convolution_matrix(f)

        # RHS vector: M (scalar replicated)
        b_ub = np.full(2*n-1, M, dtype=np.float64)

        # Objective c = -1 for maximize sum(g)
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,

            break

"""
Propose a low‑score step‑function.

The search combines three ideas.

1️⃣  Analytic seeds: beta‑distribution (including the arcsine law),
    a random Dirichlet draw and the uniform vector give a diverse
    set of starting points.

2️⃣  Mirror‑descent on the probability simplex.
    We work with a probability vector a (sum = 1) and minimise
    the surrogate  Lα(a)= (1/α)·log∑exp(α·(a*a)) .
    The gradient of the soft‑max surrogate is computed by FFT‑based
    convolutions.  An Adam‑style multiplicative update (exponentiated
    gradient on the KL‑geometry) is applied, with a temperature α
    that is increased exponentially (soft‑max → true max).

3️⃣  Optional LP polishing.
    After the gradient phase (for moderate lengths ≤ 3000) we solve
    a small linear programme that maximises ∑gj under the linear
    constraints (a∗g)k ≤ M, where M = max(a∗a).  Mixing a small fraction
    of the LP solution back into a provably never increases the bound.
    A few such polishing steps can knock the score a little further.

The whole routine respects a global time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate Lα(a) = (1/α)·log∑exp(α·conv).
    Returns a vector of length n.
    """
    n = a.shape[0]
    cur_max = conv.max()
    # stable soft‑max weights
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    # convolution of reversed a with the weight vector
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of the true max (α→∞) using a uniform weight over all
    approximately active indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    active = np.flatnonzero(conv >= cur_max - eps * (abs(cur_max) + 1.0))
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for self‑product terms (i,i) that appear only once
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad


# ----------------------------------------------------------------------
def _adam_multiplicative_update(
    a: np.ndarray,
    grad: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    step: int,
    lr: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Adam‑style multiplicative (exponentiated) update on the simplex.
    The update keeps the vector non‑negative and re‑normalises to sum = 1.
    """
    # first moment
    m = beta1 * m + (1.0 - beta1) * grad
    # second moment
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    # bias‑correction
    m_hat = m / (1.0 - beta1 ** step)
    v_hat = v / (1.0 - beta2 ** step)

    # exponentiated step (log‑space update)
    log_factor = -lr * (m_hat / (np.sqrt(v_hat) + eps))
    shift = np.max(log_factor)  # for numerical stability
    a = a * np.exp(log_factor - shift)

    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        # recover uniform if something went wrong
        a = np.full_like(a, 1.0 / a.shape[0], dtype=np.float64)
    else:
        a = a / s
    return a, m, v

def _initial_vector(init_kind: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a feasible probability vector of length n.
    init_kind can be 'arcsine', 'uniform', 'random', 'prev' or any beta‑shape.
    """
    if init_kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init_kind == "random":
        a = rng.random(n)
        a = a / a.sum()
    elif init_kind == "arcsine":
        a = _beta_distribution(n, 0.5, 0.5)
    elif init_kind == "prev":
        try:
            # Use the globally supplied previous construction if it exists
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                # linear interpolation to required length
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _beta_distribution(n, 0.5, 0.5)
    else:
        # generic beta shape supplied as "beta:α,β"

            al, be = map(float, parts)
            a = _beta_distribution(n, al, be)
        except Exception:

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Score for a normalised (sum = 1) vector: 2·n·max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _build_convolution_matrix(f: np.ndarray) -> csr_matrix:  # pragma: no cover
    """
    Build the sparse matrix A with rows k (0 … 2·n‑2) and columns i (0 … n‑1)
    such that (A·g)_k = (f∗g)_k.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for i in range(n):
        coeff = f[i]
        if coeff == 0.0:
            continue
        # positions k where f_i participates:
        k_start = i
        k_end = i + n      # exclusive
        rows.extend(range(k_start, k_end))
        cols.extend([i] * n)
        data.extend([coeff] * n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)
    return A


# ----------------------------------------------------------------------
def _lp_polish(f: np.ndarray, M: float, t: float, max_iters: int,
               deadline: float) -> Tuple[np.ndarray, float]:
    """
    Perform a few LP polishing steps.
    Goal: maximise Σg_j subject to (f∗g)_k ≤ M and g≥0.
    The solution g is rescaled to have the same ℓ1‑norm as f, then mixed.
    """
    if not _SCIPY_AVAILABLE:
        return f, M

    n = f.shape[0]
    target_norm = math.sqrt(2 * n)               # the normalisation used by the theory
    f = f * target_norm / f.sum()                # ensure the invariant Σf = target_norm

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # Build constraint matrix with the current f
        A_ub = _build_convolution_matrix(f)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise Σg  →  minimise −Σg
        c = -np.ones(n, dtype=np.float64)

        # Solve – we ask HiGHS to stop after a modest time to stay safe
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': 20})

        if not res.success or res.x is None:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break

        # Rescale g back to the target ℓ1‑norm
        g = g * (target_norm / sum_g)

        # Convex combination with a small mixing parameter t
        f = (1.0 - t) * f + t * g

        # Update M for the next round
        conv_f = _fft_self_convolve(f)
        M = conv_f.max()
    # finally renormalise to 1 (the outside code works with sum = 1)
    f = f / f.sum()
    return f, M

def _optimize_length(
    n: int,
    init_vec: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> Tuple[np.ndarray, float]:
    """
    Gradient‑descent phase for a fixed length n.
    Returns the best vector seen (still normalised to sum = 1) and its cheap score.
    """
    a = init_vec.copy()
    n_a = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters
    eta0 = 0.8               # base learning‑rate for Adam
    beta1 = 0.9
    beta2 = 0.999
    temp = 1.0               # soft‑max temperature α
    temp_factor = 1.006      # grow α each iteration
    temp_max = 5000.0

    best_a = a.copy()
    best_score = _cheap_score(a)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # convolution & cheap score
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n_a * conv.max()

        # keep a copy if improvement
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n_a} iter={iteration}  score={cur_score:.6f}")

        # gradient (soft‑max → exact)
        if temp < 600.0:
            grad = _softmax_gradient(a, conv, temp)
            temp = min(temp_max, temp * temp_factor)
        else:
            grad = _exact_gradient(a, conv)

        # Adam‑style multiplicative update
        lr = eta0 / math.sqrt(iteration)
        a, m, v = _adam_multiplicative_update(a, grad, m, v,
                                              step=iteration,
                                              lr=lr,

        # occasional jitter to escape plateaus

            a = _project_onto_simplex(a, s=1.0)

        # early stop if already extremely good (cheap bound)
        if best_score <= 1.45:          # well below the target – no point continuing
