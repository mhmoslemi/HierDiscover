"""
propose_candidate
-----------------

Search for a low‑score non‑negative step function.

Strategy
~~~~~~~~
1. **Analytic family** – a power‑law shape
       a_i ∝ (i+1)^{-α}                     (i = 0 … n‑1)

   For a fixed length `n` and the required ℓ¹‑norm `S = sqrt(2·n)`,
   the only free parameter is the exponent `α`.  The objective
   `evaluate_sequence` is essentially 2·n·max(conv)/S², i.e. we want to
   minimise the maximum entry of the auto‑convolution while keeping the
   sum fixed.  For a power‑law the maximum is either a₀² or the centre
   of the convolution; the best `α` balances the two terms.

2. **Root finding** – for each candidate length we solve the equation

       n^{1‑2β} = B(β,β)   with β = 1‑α

   where `B` is the Euler beta function.
   The solution gives the exponent where a₀² ≈ conv_center,
   which is a good starting point.

3. **Grid refinement** – around the root we evaluate a small grid of
   exponents (±0.05) and also the reversed shape (high values at the
   right end) and keep the best sequence.

4. **Projected sub‑gradient descent** – starting from the best candidate
   we perform a modest number of hard‑max sub‑gradient steps
   (the exact non‑smooth objective) with a back‑tracking line search.
   After each step the vector is projected onto the ℓ¹‑simplex
   `{x ≥ 0, Σx = S}`.  This refines the shape without breaking the
   constraints.

5. **Time‑budget aware** – the whole procedure respects the supplied
   time budget (default 1000 s).  If the budget runs out the best
   sequence found so far is returned.

All helper functions are defined at module level (no closures, no lambdas)
and only `numpy`, `scipy` and the provided `evaluate_sequence` are used.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.special as sp

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _make_powerlaw(n: int, alpha: float, S: float) -> np.ndarray:
    """Return a length‑n power‑law vector a_i ∝ (i+1)^{-α}, scaled to sum S."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-alpha)
    total = a.sum()
    if total == 0.0:
        return np.zeros_like(a)
    a *= (S / total)
    # the scaling already guarantees Σa = S and a_i ≥ 0
    return a


# ----------------------------------------------------------------------
def _evaluate_and_update(a: np.ndarray,
                         best_score: float,
                         best_vec: np.ndarray) -> tuple[float, np.ndarray]:
    """Compute evaluate_sequence(a) and keep the better of (a, best_vec)."""
    sc = evaluate_sequence(a.tolist())
    if sc < best_score:
        return sc, a.copy()
    return best_score, best_vec


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights with a low evaluate_sequence() score.
    The routine respects the given time budget (seconds) and prints progress.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # 1) pick a few lengths to try – all comfortably below 10 000
    lengths = [3000, 5000, 7000, 9000]
    best_score = float("inf")
    best_vec = None

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        S = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1a) compute analytic α where a0² ≈ centre‑convolution
        # solve   log B(β,β) = (1‑2β)·log n   with β = 1‑α
        def _root_fun(beta):
            # beta stays strictly inside (0,1)
            return math.log(sp.beta(beta, beta)) - (1.0 - 2.0 * beta) * math.log(n)

        # bracket the root – for n≥2000 the solution is between 0.3 and 0.6
        try:
            beta_root = opt.brentq(_root_fun, 0.3, 0.6)
            alpha_root = 1.0 - beta_root
        except Exception:
            # fall back to a safe default if root finding fails
            alpha_root = 0.5

        # --------------------------------------------------------------
        # 1b) evaluate a small grid around the root (and also reversed)
        alphas = np.concatenate((
            np.linspace(max(0.01, alpha_root - 0.10), min(0.99, alpha_root + 0.10), 11),
            np.linspace(0.05, 0.95, 19)          # a broader sweep for safety
        ))
        alphas = np.unique(alphas)

        for alpha in alphas:
            if time.time() > deadline:
                break

            # decreasing power‑law (large values at the left)
            a = _make_powerlaw(n, alpha, S)
            best_score, best_vec = _evaluate_and_update(a, best_score, best_vec)

            # increasing power‑law (large values at the right)
            a_rev = _make_powerlaw(n, alpha, S)[::-1]
            best_score, best_vec = _evaluate_and_update(a_rev, best_score, best_vec)

    # ------------------------------------------------------------------
    # 2) if we have a candidate, polish it with hard‑max sub‑gradient steps
    if best_vec is not None and time.time() < deadline:
        a = best_vec.copy()
        n = a.shape[0]
        S = math.sqrt(2.0 * n)      # exact target ℓ¹‑norm

        # keep exact sum (tiny rounding error may have accumulated)
        a *= (S / a.sum())

        # sub‑gradient descent parameters
        it = 0
        max_iters = 6000
        step0 = S * 0.5 / n      # a heuristic initial step size

        while time.time() < deadline and it < max_iters:
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = int(mask.sum())
            if cnt == 0:
                break

            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt

            # gradient of max‑convolution w.r.t. a (hard max)
            g = 2.0 * np.convolve(w, a[::-1])[n - 1:n - 1 + n]

            # simple back‑tracking line search
            step = step0 / (1.0 + 1e-4 * it)
            improved = False
            for _ in range(20):
                a_new = a - step * g
                a_new = _project_to_simplex(a_new, S)   # keep feasibility
                sc = evaluate_sequence(a_new.tolist())
                if sc < best_score - 1e-9:
                    best_score = sc
                    best_vec = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[subgrad] iter {it:4d} step {step:.6f} → {sc:.6f}")
                    break
                step *= 0.5

            if not improved:
                # occasional random jitter to escape plateaus
                i, j = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a[i] * 0.05
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, S)

            it += 1

    # ------------------------------------------------------------------
    # 3) fallback – if nothing succeeded, return a uniform vector
    if best_vec is None:
        n = 4000
        S = math.sqrt(2.0 * n)
        best_vec = np.full(n, S / n, dtype=float)
        best_score = evaluate_sequence(best_vec.tolist())
        print("[fallback] using uniform vector")

    print(f"[final] best score = {best_score:.6f} (length {best_vec.shape[0]})")
    return best_vec.tolist()
