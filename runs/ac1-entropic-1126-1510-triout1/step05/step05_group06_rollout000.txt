   a_i ∝ (i+½)^{‑p_l}·(n‑i‑½)^{‑p_r} (p_l,p_r∈[0.2,1.0]).
   For each shape we also test a few mixtures with the uniform
   distribution.  The cheap score 2·n·max(conv) is used for ranking.
   The best 4 shapes of each length are kept.

2. **Exponentiated‑gradient refinement** – Each kept shape is polished
   with a mirror‑descent / Adam‑style update on the probability simplex.
   The gradient is obtained from a soft‑max surrogate of the max‑convolution
   (temperature slowly increased).  When the surrogate temperature is
   large we fall back to the true sub‑gradient of the max entry.
   The current best vector (lowest true 2·n·max(conv)) is recorded.

3. **Local stochastic hill‑climbing** – The globally best vector is
   further refined by repeatedly moving a tiny amount of mass between
   two random indices whenever the true evaluation score improves.
   Simple simulated‑annealing‑style up‑moves are also allowed with a
   small probability.

The routine prints progress information and returns the best list of
heights found before the time limit expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (return length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max entry of a*a at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of (1/α)·log∑exp(α·conv) w.r.t. a."""
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_grad = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv_grad[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _asymmetric_power_shape(n: int, pl: float, pr: float) -> np.ndarray:
    """a_i ∝ (i+½)^{-pl}·(n‑i‑½)^{-pr} (pl,pr>0)."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-pl) * (n - i - 0.5) ** (-pr)
    if vals.sum() == 0.0:
        return np.full(n, 1.0 / n)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a (already summed to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _refine_exponential(a0: np.ndarray,
                        n: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient refinement on the simplex.
    Returns the best vector seen and its cheap internal score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _cheap_score(a)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # surrogate temperature schedule
    alpha = 5.0
    alpha_mul = 1.02
    alpha_max = 300.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # for grad‑softmax
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()

        # gradient (soft‑max surrogate, then switch to real sub‑gradient)
        if alpha < alpha_max:
            g = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            g = _subgradient_max(a, k)

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # learning‑rate schedule (slow decay)
        eta = 0.5 / np.sqrt(it)

        # natural‑gradient step: exponentiated update
        step = -eta * m_hat / (np.sqrt(v_hat) + eps)
        # clip excessively large steps to avoid under‑flow/overflow
        step = np.clip(step, -20, 20)

        a = a * np.exp(step)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional jitter to escape plateaus
        if it % 3000 == 0 and np.random.rand() < 0.13:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best, _cheap_score(best)


# ----------------------------------------------------------------------
def _local_hill_climb(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the true max‑convolution.
    Occasionally accepts a slightly worse move (simulated annealing).
    """
    a = a_start.copy()
    best = a.copy()
    best_score = _cheap_score(a)

    # temperature schedule for uphill moves
    T0 = 0.02
    it = 0
    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a fraction of mass from i to j
        eps = a[i] * np.random.rand() * 0.12   # up to 12 % of source mass
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur_score = _cheap_score(cand)

        improve = cur_score < best_score - 1e-12
        if improve:
            best_score = cur_score
            best = cand.copy()
            a = cand
        else:
            # uphill acceptance with probability exp(-(Δ)/T)
            T = T0 / np.sqrt(it)
            if np.random.rand() < np.exp(-(cur_score - best_score) / max(T, 1e-8)):
                a = cand

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative list of heights that minimises
    ``evaluate_sequence``.  The routine respects a total runtime of
    ``budget_s`` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0    # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a small set of lengths to explore.
    # ------------------------------------------------------------------
    base_len = 8000
    try:
        base_len = len(height_sequence_1)    # type: ignore[name-defined]
    except Exception:
        pass
    lengths = sorted({max(800, int(base_len * f))
                      for f in (0.8, 1.0, 1.2, 1.5, 2.0)
                      if max(800, int(base_len * f)) <= 20000})
    if not lengths:
        lengths = [8000]

    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503      # stop early if we hit the target

    # ------------------------------------------------------------------
    # 2. Coarse grid search over asymmetric beta shapes for each length.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"--- Exploring length n={n} ---")

        # grid parameters
        pl_vals = np.linspace(0.30, 0.80, 9)          # left exponent
        pr_vals = np.linspace(0.30, 0.80, 9)          # right exponent
        w_vals = np.linspace(0.0, 0.5, 6)             # mixture weight with uniform

        candidates = []   # list of (seq, cheap_score)

        # (a) pure asymmetric power‑law shapes
        for pl in pl_vals:
            for pr in pr_vals:
                shape = _asymmetric_power_shape(n, pl, pr)
                score = _cheap_score(shape)
                candidates.append((shape, score))

        # (b) mixtures with uniform
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        for pl in pl_vals:
            for pr in pr_vals:
                shape = _asymmetric_power_shape(n, pl, pr)
                for w in w_vals:
                    cand = (1.0 - w) * shape + w * uniform
                    cand = _project_onto_simplex(cand, s=1.0)
                    score = _cheap_score(cand)
                    candidates.append((cand, score))

        # (c) some predefined seeds (use the previous best if present)
        predefined = []
        for kind in ("arcsine", "beta", "random", "uniform"):
            # generate a quick seed
            if kind == "arcsine":
                i = np.arange(n, dtype=np.float64)
                a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
                a = a / a.sum()
            elif kind == "beta":
                a = np.random.beta(0.5, 0.5, size=n)
                a = a / a.sum()
            elif kind == "random":
                a = np.random.rand(n)
                a = a / a.sum()
            else:   # uniform
                a = np.full(n, 1.0 / n)
            a = _project_onto_simplex(a, s=1.0)
            predefined.append((a, _cheap_score(a)))
        candidates.extend(predefined)

        # keep the best few candidates (by cheap score)
        candidates.sort(key=lambda t: t[1])
        top_candidates = [seq for seq, _ in candidates[:6]]

        # ------------------------------------------------------------------
        # 3. Refine each top candidate with exponentiated‑gradient + Adam.
        # ------------------------------------------------------------------
        for a0 in top_candidates:
            if time.time() >= deadline - 4.0:
                break
            # allocate a slice of time for this refinement
            slice_deadline = min(deadline, time.time() + 0.45 * (deadline - time.time()))
            seq_refined, cheap_score = _refine_exponential(a0, n, slice_deadline)

            # Evaluate with the black‑box (exact) evaluator
            try:
                exact_score = evaluate_sequence(seq_refined.tolist())
            except Exception:                           # fallback to cheap score
                exact_score = cheap_score

            if exact_score < overall_best_score:
                overall_best_score = exact_score
                overall_best_seq = seq_refined.copy()
                print(f"*** New global best (len={n}) → score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break

        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climbing on the best vector found.
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n_best = overall_best_seq.shape[0]
        print(f"--- Final local optimisation on length {n_best} ---")
        seq_final, cheap_score = _local_hill_climb(overall_best_seq, n_best, deadline)

        try:
            exact_score = evaluate_sequence(seq_final.tolist())
        except Exception:
            exact_score = cheap_score

        if exact_score < overall_best_score:
            overall_best_score = exact_score
            overall_best_seq = seq_final.copy()
            print(f"*** After hill‑climb → score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback safety (should never trigger).
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 1200
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Safety fallback used.")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
