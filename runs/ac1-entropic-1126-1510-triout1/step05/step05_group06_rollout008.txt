def _optimize_adam(a0, n, deadline):
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2*n*_fft_self_convolve(a).max()
    # hyperparams
    alpha = 5.0
    max_alpha = 300.0
    alpha_mul = 1.02
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    size_fft = 1 << ((3*n-2).bit_length())
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2*n*cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
        # gradient
        if alpha < max_alpha:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(max_alpha, alpha*alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = m_hat/(np.sqrt(v_hat)+eps_adam)
        # Since we need positivity, we can do a = a * np.exp(-eta * step) as before,
        # but with step size scaled: use eta_t = 0.5/np.sqrt(t) maybe.
        eta = 0.5/np.sqrt(t)
        a = a * np.exp(-eta*step)
        # Project onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # occasionally jitter
        if t % 3000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n)*1e-5
            a = _project_onto_simplex(a, s=1.0)
    return best_seq, best_score

"""
Propose a step‑function (list of non‑negative heights) with a low
evaluate_sequence() score.

Algorithm
---------
1. **Length selection** –  a few candidate lengths (≈5 k … 12 k) are
   examined using a very cheap internal score based on an FFT
   convolution of the classic arcsine shape. The two best lengths are
   kept.

2. **Shape exploration** – for each chosen length we first run a coarse
   grid over the two‑parameter family
        a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ   mixed with a uniform component.
   The best (p,w) pair is refined by a fine grid (step 0.005 in *p* and
   0.02 in *w*).

3. **Exponentiated‑gradient refinement** – the best few shapes are
   polished with a long mirror‑descent (multiplicative) update.
   A soft‑max surrogate of the L∞ norm is used while the temperature
   rises, then the true sub‑gradient of the max entry is applied.
   The learning rate decays as η₀/√t and after each step the vector is
   projected onto the probability simplex.

4. **Simulated‑annealing hill‑climb** – the globally best vector found
   so far is further improved by a stochastic annealing phase that
   moves a small fraction of mass between two random indices and accepts
   worsening moves with a linearly cooling temperature.

All internal evaluations use a real‑FFT based convolution, which is
O(n log n) and fast enough to allow thousands of iterations within the
1000 s time budget.  The final list is returned; its score is computed
with the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho_arr.size == 0 else cssv[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) convolution of *a* with itself via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max entry of a*a at index *k*."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of (1/α)·log∑exp(α·b) w.r.t. a (soft‑max surrogate)."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:                # fallback – pure max
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # convolution of a_rev with weight vector w
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _beta_power_shape(n: int, p: float) -> np.ndarray:
    """Distribution a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ (symmetric, p>0)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create a normalised (sum‑1) non‑negative vector of length *n*."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "beta":
        a = np.random.beta(0.5, 0.5, size=n)
        a /= a.sum()
    elif kind == "arcsine":
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a /= a.sum()
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    a = _project_onto_simplex(a, s=1.0)
    return a

def _cheap_score(a: np.ndarray) -> float:
    """Fast internal objective 2·n·max(a*a) (sum of a is 1)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()

def _grid_search_best(n: int) -> tuple[np.ndarray, float]:
    """
    Small coarse grid on (p,w) and on an asymmetric family.
    Returns the best vector (already normalised to sum‑1) and its cheap score.
    """
    best_seq = None
    best_score = np.inf

    # symmetric power‑law + uniform mixture
    p_vals = np.linspace(0.30, 0.70, 9)      # 0.30 … 0.70
    w_vals = np.linspace(0.0, 1.0, 11)      # 0 … 1
    for p in p_vals:
        shape = _beta_power_shape(n, p)
        for w in w_vals:
            cand = (1.0 - w) * shape + w * np.full(n, 1.0 / n)
            cand = _project_onto_simplex(cand, s=1.0)
            score = _cheap_score(cand)
            if score < best_score:
                best_score, best_seq = score, cand.copy()

    # asymmetric two‑parameter family
    pl_vals = np.linspace(0.20, 0.80, 4)
    pr_vals = np.linspace(0.20, 0.80, 4)

            score = _cheap_score(cand)

def _optimize_exponential(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) with a soft‑max surrogate.
    Returns the best vector seen and its internal cheap score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _cheap_score(a)

    # hyper‑parameters
    eta0 = 0.5               # initial learning‑rate factor
    alpha = 5.0              # start temperature for soft‑max
    alpha_mul = 1.02
    alpha_max = 300.0
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_seq = a.copy()

        # gradient (soft‑max while temperature is moderate)
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # step‑size – diminishing
        eta = eta0 / np.sqrt(it)

        # multiplicative update (exponentiated gradient)
        a = a * np.exp(-eta * grad)

        # projection onto simplex (guarantees non‑negativity and sum‑1)
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

def _simulated_annealing(a_start: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple annealing: move a small fraction of mass between two indices.
    Accepts worsening moves with a linearly cooling temperature.
    Returns the best vector and its cheap score.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _cheap_score(a)

    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best_seq, best_score

    T0 = 0.008                     # initial temperature (tuned empirically)
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 8 % of the source mass
        eps = a[i] * np.random.rand() * 0.08
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cand = _project_onto_simplex(cand, s=1.0)

        cur_score = _cheap_score(cand)
        delta = cur_score - best_score

        elapsed = time.time() - start_time
        # linear cooling, never negative
        T = max(0.0, T0 * (1.0 - elapsed / total_time))

        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            a = cand
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cand.copy()
    return best_seq, best_score

    base_n = 8000

        base_n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        pass

    length_factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted({max(500, int(base_n * f)) for f in length_factors
                      if max(500, int(base_n * f)) <= 20000})
    if not lengths:
        lengths = [8000]

    # ------------------------------------------------------------------
    # Phase 1 – cheap ranking of lengths (using pure arcsine shape)
    length_infos = []
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        seq = _initial_sequence(n, "arcsine")
        score = _cheap_score(seq)
        length_infos.append((score, n, seq))
    length_infos.sort(key=lambda t: t[0])
    top_lengths = [info[1] for info in length_infos[:2]]   # keep two best

    # ------------------------------------------------------------------
    best_global_seq = None
    best_global_score = float("inf")
    best_global_n = None

    # ------------------------------------------------------------------
    for n in top_lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ---- (a) coarse shape search (already cheap) -------------------
        shape_coarse, _ = _grid_search_best(n)

        # ---- (b) fine grid around the arcsine centre ------------------
        shape_fine = _fine_grid_search(n, p_center=0.5, w_center=0.0,
                                       p_range=0.12, w_range=0.30,
                                       p_steps=23, w_steps=23)

        # ---- (c) a few additional seeds -------------------------------
        init_candidates = [
            shape_coarse,
            shape_fine,
            _initial_sequence(n, "random"),
            _initial_sequence(n, "uniform")
        ]

        # keep the 3 most promising according to cheap score
        init_candidates = sorted(init_candidates,
                                 key=_cheap_score)[:3]

        # ---- (d) exponential‑gradient refinement -----------------------
        slice_deadline = time.time() + max(0.2, (deadline - time.time()) * 0.35)
        for a0 in init_candidates:
            if time.time() >= slice_deadline:
                break
            seq_opt, internal_score = _optimize_exponential(a0, n, slice_deadline)

            # Evaluate with the black‑box function (scale invariant, so
            # the exact normalisation does not matter).
            try:
                final_score = evaluate_sequence(seq_opt.tolist())
            except Exception:
                final_score = internal_score

            if final_score < best_global_score:
                best_global_score = final_score
                best_global_seq = seq_opt.copy()
                best_global_n = n
                print(f"*** New global best → len={n} score={best_global_score:.6f}")

            # stop early if we already beat the target
            if best_global_score <= 1.503:
                break
        if best_global_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Final stochastic refinement on the best vector found so far
    if best_global_seq is not None and time.time() < deadline - 1.0:
        print("\n--- Simulated annealing refinement ---")
        refined_seq, _ = _simulated_annealing(best_global_seq,
                                              best_global_n,
                                              deadline)
        try:
            final_score = evaluate_sequence(refined_seq.tolist())
        except Exception:
            final_score = evaluate_sequence(best_global_seq.tolist())
        if final_score < best_global_score:
            best_global_score = final_score
            best_global_seq = refined_seq.copy()
            print(f"*** After annealing score={best_global_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    if best_global_seq is None:
        n = 1000
        best_global_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_global_score = evaluate_sequence(best_global_seq.tolist())
        print("Safety fallback used.")

    print(f"\nFinal best score: {best_global_score:.6f} (length = {len(best_global_seq)})")
    return best_global_seq.tolist()
