    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = init_beta
    best_a = a.copy()
    best_obj = 2.0 * n * _fft_self_convolve_max(a) # raw objective
    start = time.time()
    while time.time() - start < time_allow:
        t += 1
        # compute gradient w.r.t smoothed max objective
        conv = _fft_self_convolve_full(a)
        # log-sum-exp
        maxc = conv.max()
        # compute w
        expb = np.exp(beta * (conv - maxc))
        sumexp = expb.sum()
        w = expb / sumexp
        # gradient
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

"""
Propose a low‑score step‑function.

The search works in three stages:

1)  A quick exploration of a rich family of candidate vectors:
    - the previous best sequence (if available),
    - uniform and arcsine (Beta(½,½)) vectors,
    - many random Dirichlet draws,
    - random mixtures of two Beta distributions.
    All candidates are projected onto the probability simplex (Σa_i = 1).

2)  A fast local refinement using a smoothed version of the
    max‑convolution objective.
    The smoothed objective is
        F_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k).
    Its gradient can be computed in O(n log n) with FFTs.
    We minimise F_β with projected Adam (Adam + Euclidean projection
    onto the simplex).  The smoothing parameter β is increased
    gradually (continuation) so that the algorithm first sees a
    smooth landscape and finally behaves like a sub‑gradient method.

3)  The best vector found is handed to the black‑box
    ``evaluate_sequence`` and returned as a plain Python list.

The whole routine respects the 1000 s time limit, works
with a single CPU and only uses ``numpy`` and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) of length n, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to keep numerical stability
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a * a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns (smooth_value, gradient, current_max) for the
    smoothed max objective F_β(a) = (1/β)·log Σ exp(β·(a*a)_k).

    gradient = 2·conv_rev(a, w) where w_k = exp(β·c_k)/Σ exp(β·c)
    and c = a*a (full convolution).
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)                # length 2n‑1
    maxc = conv.max()
    # stable log‑sum‑exp
    expb = np.exp(beta * (conv - maxc))
    sumexp = expb.sum()
    w = expb / sumexp                                 # same length as conv
    # gradient (see derivation in analysis)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    smooth_val = (maxc + (1.0 / beta) * math.log(sumexp))
    return smooth_val, grad, maxc


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   time_allow: float,
                   init_beta: float = 5.0,
                   beta_factor: float = 1.5,
                   beta_max: float = 5000.0,
                   lr: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Projected Adam minimising the smoothed max‑convolution.
    Returns the best vector found (still on the simplex) and its
    raw evaluation 2·n·max(a*a).
    """
    n = a0.shape[0]
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    cur_max = _fft_self_convolve_max(a)
    best_score = 2.0 * n * cur_max

    start = time.time()
    it = 0
    beta = init_beta

    while time.time() - start < time_allow:
        it += 1

        # smooth objective, gradient, and true max for monitoring
        _, grad, cur_max = _smooth_max_and_grad(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_onto_simplex(a)

        # occasionally record improvement
        if it % 100 == 0:
            cur_max = _fft_self_convolve_max(a)
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] iter={it} beta={beta:.1f} "
                      f"best_score={best_score:.6f}")

        # increase smoothing (continuation)
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function with low evaluate_sequence score.
    The routine:
      • builds a diverse set of initial vectors,
      • keeps the best one,
      • refines it with projected Adam on a smoothed max‑convolution,
      • returns the final sequence as a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Length of the sequence – we stay with 5000 (used by the previous best)
    n = 5000

    # ------------------------------------------------------------------
    # Gather a pool of candidate vectors
    candidates = []

    # 1) previous solution, if visible as global variable height_sequence_1
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _interpolate_to_length(base_seq, n)
        candidates.append(_project_onto_simplex(base_seq))
    except Exception:
        pass

    # 2) very cheap baselines
    candidates.append(np.full(n, 1.0 / n))                     # uniform
    candidates.append(_beta_distribution(n, 0.5, 0.5))          # arcsine

    # 3) a handful of random Dirichlet draws
    for _ in range(50):
        vec = np.random.exponential(scale=1.0, size=n)
        vec /= vec.sum()
        candidates.append(vec)

    # 4) random mixtures of two independent Beta components
    for _ in range(50):
        a1, b1 = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
        a2, b2 = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
        lam = np.random.rand()
        p1 = _beta_distribution(n, a1, b1)
        p2 = _beta_distribution(n, a2, b2)
        vec = lam * p1 + (1.0 - lam) * p2
        candidates.append(_project_onto_simplex(vec))

    # ------------------------------------------------------------------
    # Pick the best raw candidate (raw = 2·n·max(a*a), sum(a)=1)
    best_seq = None
    best_score = float('inf')
    for vec in candidates:
        cur_score = 2.0 * n * _fft_self_convolve_max(vec)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = vec.copy()
    print(f"Initial best raw score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Local refinement with projected Adam (continuation on β)
    remaining = deadline - time.time()
    if remaining > 20 and best_seq is not None:
        refined_seq, refined_score = _adam_optimize(
            best_seq,
            time_allow=remaining,
            init_beta=5.0,
            beta_factor=1.5,
            beta_max=5000.0,
            lr=0.2,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq
            print(f"After Adam refinement raw score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final call to the provided black‑box evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
