"""
Optimisation of a non‑negative step‑function to minimise
                2·n·max(a∗a)/(∑a)^2

The algorithm proceeds as follows:

1. Generate a diverse pool of initial candidates using several
   parametric families (arcsine, power‑law, beta, Gaussian, mixtures).
2. For each candidate run a smooth “soft‑max” Adam optimiser that
   minimises the log‑sum‑exp of the self‑convolution (a differentiable
   surrogate for the max).  The temperature parameter α is slowly
   increased so that the surrogate converges to the true max.
3. Keep the best sequence seen.  In a second phase the best four
   candidates are refined further with an exponentiated‑gradient mirror
   descent (the method from the previous solution) to exploit its
   fast convergence close to the optimum.
4. Finish with a cheap random‑move hill‑climber that moves a small
   fraction of probability mass between two coordinates whenever it
   improves the exact objective.
5. Return the best sequence found as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]          # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) / (∑a)^2, assuming a lives on the simplex."""
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _fft_self_convolve(a).max()
    return 2.0 * n * max_conv

# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total

# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total

# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total

# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)

# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float,
                      size_fft: int) -> np.ndarray:
    """
    Gradient of the smoothed objective
        (1/α) log Σ exp(α·conv_k)
    w.r.t. a. Implements 2 * conv_rev(w, a_rev) where w are the softmax
    weights.
    """
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        # degenerate case – fall back to the max index only
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_rev = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    grad = 2.0 * conv_rev[start:start + a.shape[0]]
    return grad

# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best sequence seen (projected onto the simplex) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    # smoothing temperature
    alpha = 1.0
    alpha_mul = 1.05
    alpha_max = 20000.0

    size_fft = 1 << ((3 * n - 2).bit_length())

    best = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)               # length 2n‑1
        # gradient of smoothed max
        grad = _softmax_gradient(a, conv, alpha, size_fft)

        # Adam update (in the original a space)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasibility
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n, dtype=np.float64)

        # exact score check
        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] Adam improve it={it} score={best_score:.6f}")

        # temperature schedule
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # step‑size decay
        if it % 1500 == 0:
            lr *= 0.7

    return best, best_score

# ----------------------------------------------------------------------
def _exponential_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser (identical to the
    reference solution, kept for its strong local convergence).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)

    eta0 = 0.8
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 15000.0
    alpha_thr = 2000.0
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Exp‑grad improve it={it} score={best_score:.6f}")

        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, b, alpha, size_fft)

        eta = eta0 / np.sqrt(it)
        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score

# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g

# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Soft‑max gradient used by the exponential optimiser."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g

# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random‑move hill‑climbing: transfer a tiny fraction of mass
    from one index to another if it strictly improves the exact objective.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 10 % of source mass
        eps = a[i] * np.random.rand() * 0.1
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cur_score = _objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score

"""
Optimisation of a non‑negative step‑function to minimise

        2·n·max(a∗a) / (∑a)²

The algorithm proceeds in four phases:

1. **Random parametric generation** – A pool of candidates is built from
   several simple families (arcsine, power‑law, beta, Gaussian,
   mixtures, edge‑spike).  All candidates are projected onto the
   probability simplex and evaluated with the exact objective.

2. **Adam optimisation on a soft‑max surrogate** – The best few candidates
   are refined with an Adam optimiser that minimises the
   log‑sum‑exp of the self‑convolution (a smooth approximation of the
   max).  The temperature parameter α is increased gradually so that the
   surrogate converges to the true max.

3. **Exponentiated‑gradient refinement** – The top candidates are further
   processed with the mirror‑descent / exponentiated‑gradient method
   from the reference solution, which is very effective close to a
   local optimum.

4. **Local random‑move hill‑climbing** – A cheap stochastic refinement
   moves a small fraction of probability mass between two coordinates
   whenever it strictly improves the exact objective.

The routine respects the 1000 s wall‑time budget, keeps the best sequence
found and returns it as a plain Python list.
"""

import time
import numpy as np
import operator

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) / (∑a)² (a is projected to the simplex)."""
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _fft_self_convolve(a).max()
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total


# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _edge_spike_distribution(n: int, edge_frac: float) -> np.ndarray:
    """Mass `edge_frac` on each endpoint, remainder uniform on the interior."""
    edge_frac = min(max(edge_frac, 0.0), 0.5)
    interior_mass = max(0.0, 1.0 - 2.0 * edge_frac)
    if n <= 2:
        return np.full(n, 1.0 / n, dtype=np.float64)
    interior = interior_mass / (n - 2)
    vals = np.full(n, interior, dtype=np.float64)
    vals[0] = edge_frac
    vals[-1] = edge_frac
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate used by the exponential optimiser."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      alpha: float,
                      size_fft: int) -> np.ndarray:
    """
    Gradient of the smoothed objective
        (1/α)·log Σ exp(α·conv_k)
    w.r.t. a.
    Implements 2·conv_rev(w, a_rev) where w are the soft‑max weights.
    """
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_rev = np.fft.irfft(W * A_rev, n=size_fft)

    start = a.shape[0] - 1
    grad = 2.0 * conv_rev[start:start + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best sequence (projected onto the simplex) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    # temperature schedule
    alpha = 1.0
    alpha_mul = 1.07
    alpha_max = 20000.0

    size_fft = 1 << ((3 * n - 2).bit_length())

    best = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)                     # full self‑convolution
        grad = _softmax_gradient(a, conv, alpha, size_fft)

        # Adam update in the original space
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # feasibility projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n, dtype=np.float64)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] Adam improve it={it} score={best_score:.6f}")

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # modest decay of the learning rate
        if it % 1500 == 0:
            lr *= 0.7

    return best, best_score


# ----------------------------------------------------------------------
def _exponential_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser (identical to the reference).
    """
    a = a0.copy()
    n = a.shape[0]

    best_seq = a.copy()
    best_score = _objective(a)

    eta0 = 0.8
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 15000.0
    alpha_thr = 2000.0
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Exp‑grad improve it={it} score={best_score:.6f}")

        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, b, alpha, size_fft)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random‑move hill‑climbing: transfer a small fraction of mass between two
    indices whenever it strictly improves the exact objective.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.1         # up to 10 % move
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cur_score = _objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score

            lam = np.random.uniform(0.0, 0.6)
            seq = _mixture_arcsine_uniform(n, lam)
        elif fam == "edge_spike":
            edge_frac = np.random.uniform(0.0, 0.4)
            seq = _edge_spike_distribution(n, edge_frac)
        else:
            seq = _arcsine_distribution(n)

        seq = _project_onto_simplex(seq, s=1.0)
        sc = _objective(seq)

        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New global best (rand) n={n} fam={fam} score={best_score:.6f}")

        if len(candidate_pool) < 12:
            candidate_pool.append((sc, seq))
        else:
            # find current worst
            worst_idx = 0
            worst_sc = candidate_pool[0][0]
            for idx in range(1, len(candidate_pool)):
                if candidate_pool[idx][0] > worst_sc:
                    worst_sc = candidate_pool[idx][0]
                    worst_idx = idx
            if sc < worst_sc:
                candidate_pool[worst_idx] = (sc, seq)

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation of the most promising candidates
    candidate_pool.sort(key=operator.itemgetter(0))
    top_for_adam = [cand[1] for cand in candidate_pool[:5]]

    for cand in top_for_adam:
        if time.time() >= deadline - 30.0:
            break
        n = cand.shape[0]
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(30.0, remaining * 0.5)
        refined, _ = _adam_optimize(cand, slice_deadline)
        refined = _project_onto_simplex(refined, s=1.0)
        sc_ref = _objective(refined)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = refined.copy()
            print(f"[{time.time():.2f}] Adam refinement improved to {best_score:.6f} (n={n})")
        # keep refined candidate in the pool
        if len(candidate_pool) < 12:
            candidate_pool.append((sc_ref, refined))
        else:
            # replace worst again if better
            worst_idx = 0
            worst_sc = candidate_pool[0][0]
            for idx in range(1, len(candidate_pool)):
                if candidate_pool[idx][0] > worst_sc:
                    worst_sc = candidate_pool[idx][0]
                    worst_idx = idx
            if sc_ref < worst_sc:
                candidate_pool[worst_idx] = (sc_ref, refined)

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient refinement (as in reference)
    candidate_pool.sort(key=operator.itemgetter(0))
    top_for_exp = [cand[1] for cand in candidate_pool[:3]]

    for cand in top_for_exp:
        if time.time() >= deadline - 30.0:
            break
        n = cand.shape[0]
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(45.0, remaining * 0.6)
        refined, _ = _exponential_optimize(cand, slice_deadline)
        refined = _project_onto_simplex(refined, s=1.0)
        sc_ref = _objective(refined)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = refined.copy()
            print(f"[{time.time():.2f}] Exp‑grad refinement improved to {best_score:.6f} (n={n})")
        # store refined version in pool
        if len(candidate_pool) < 12:
            candidate_pool.append((sc_ref, refined))
        else:
            worst_idx = 0
            worst_sc = candidate_pool[0][0]
            for idx in range(1, len(candidate_pool)):
                if candidate_pool[idx][0] > worst_sc:
                    worst_sc = candidate_pool[idx][0]
                    worst_idx = idx
            if sc_ref < worst_sc:
                candidate_pool[worst_idx] = (sc_ref, refined)

    # ------------------------------------------------------------------
    # Phase 4 – final local random‑move hill climbing
    if best_seq is not None and time.time() < deadline - 1.0:
        refined, _ = _local_refinement(best_seq, deadline)
        sc_ref = _objective(refined)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = refined.copy()
            print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length = {best_seq.shape[0]})")
    return best_seq.tolist()
