def _optimize_adam(a0, n, deadline, max_iter=None):
    a = a0.copy()
    # initialize x = log(a) - mean
    x = np.log(a + 1e-15)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    max_alpha = 300.0
    alpha = 5.0
    alpha_mul = 1.02
    size_fft = 1 << ((3 * n - 2).bit_length())
    best_seq = a.copy()
    best_score = 2 * n * _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        if max_iter is not None and it >= max_iter:
            break
        it += 1
        # compute a from x for numerical stability
        shift = x - np.max(x)
        exp_shift = np.exp(shift)
        a = exp_shift / exp_shift.sum()
        # compute conv and score
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2 * n * cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
        # compute gradient
        if alpha < max_alpha:
            grad_a = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(max_alpha, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad_a = _subgradient_max(a, k)
        # grad_x = a * (grad_a - np.dot(a, grad_a))
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
        # optional decorrelation: at intervals, add small noise
        if it % 2000 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 1e-5
    return best_seq, best_score

"""
Propose a non‑negative step function (list of heights) that minimizes
``evaluate_sequence``.  The algorithm consists of three phases:

1.  **Rich family search** –  for each candidate length we explore a
    dense grid of symmetric power‑law shapes
    a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ (p∈[0.2,1.8]) possibly mixed with the uniform
    distribution, and a simple “two‑tail” shape (mass only in the first
    and last k positions).  The best few candidates are kept.

2.  **Adam on the probability simplex** –  each kept candidate is refined
    by an Adam optimiser written in the log‑softmax (exponential‑map)
    coordinates.  The surrogate objective is a log‑sum‑exp (soft‑max)
    approximation of the L∞‑norm of the self‑convolution; the temperature
    α is increased slowly so the optimisation finally targets the true
    maximum.

3.  **Stochastic hill‑climbing** –  the globally best vector is finally
    polished by moving small amounts of mass between random indices while
    strictly improving the true evaluation score.

The routine respects the given time budget (default 1000 s) and prints
intermediate progress.  The returned list is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max_entry of a*a at index *k*."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of (1/α)·log∑exp(α·b) w.r.t. a (soft‑max surrogate)."""
    shift = b - b.max()               # for numerical stability
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create a normalised (sum‑1) non‑negative vector of length *n*."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a = a / a.sum()
    elif kind == "beta":
        a = np.random.beta(0.5, 0.5, size=n)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
    elif kind == "arcsine":
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a /= a.sum()
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = np.maximum(a, 0.0)
            if a.sum() == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / a.sum()
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _beta_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _two_tail_shape(n: int, k: int) -> np.ndarray:
    """Uniform mass on the first k and last k positions, zero elsewhere."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        a[:k] = 1.0
        a[-k:] = 1.0
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _grid_search_best_fine(n: int) -> tuple[np.ndarray, float]:
    """
    Fine grid search over symmetric power‑law shapes (p) and a uniform
    mixture weight (w).  Returns the best vector and its internal score
    (2·n·max(conv)).
    """
    best_seq = None
    best_score = np.inf

    p_vals = np.linspace(0.20, 1.80, 33)      # step ≈0.05
    w_vals = np.linspace(0.0, 0.50, 11)       # up to 50 % uniform

    for p in p_vals:
        shape = _beta_power_shape(n, p)
        for w in w_vals:
            cand = (1.0 - w) * shape + w * (1.0 / n)
            cand = _project_onto_simplex(cand, s=1.0)
            score = 2.0 * n * _fft_self_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
    return best_seq, float(best_score)


# ----------------------------------------------------------------------
def _search_tail_best(n: int) -> tuple[np.ndarray, float]:
    """
    Scan simple two‑tail shapes (mass only at the ends) for the best
    internal score.
    """
    best_seq = None
    best_score = np.inf
    fractions = np.linspace(0.01, 0.20, 20)   # 1 % … 20 % of the length
    for f in fractions:
        k = max(1, int(round(f * n)))
        if 2 * k >= n:
            continue
        cand = _two_tail_shape(n, k)
        score = 2.0 * n * _fft_self_convolve(cand).max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    return best_seq, float(best_score)


# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the probability simplex.
    Returns the best vector seen and its internal (2·n·max) score.
    """
    # initialise log‑coordinates
    x = np.log(a0 + 1e-15)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    max_alpha = 300.0
    alpha = 5.0
    alpha_mul = 1.02
    size_fft = 1 << ((3 * n - 2).bit_length())

    best_seq = a0.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_seq).max()

    it = 0
    while time.time() < deadline:
        it += 1
        # recompute a from x (soft‑max)
        shift = x - np.max(x)
        exp_shift = np.exp(shift)
        a = exp_shift / exp_shift.sum()

        # convolution and current score
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()

        # gradient of surrogate (soft‑max) or true max
        if alpha < max_alpha:
            grad_a = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(max_alpha, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad_a = _subgradient_max(a, k)

        # convert to gradient w.r.t. log‑coordinates
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # occasional tiny random jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.15:
            x += np.random.randn(n) * 1e-5

    return best_seq, float(best_score)


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: transfer a fraction of mass between
    two random indices whenever the true evaluation improves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # move at most 20 % of the source mass
        eps_move = a[i] * np.random.rand() * 0.20
        if eps_move <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps_move
        cand[j] += eps_move
        if cand[i] < 0.0:
            continue
        cur_score = 2.0 * n * np.max(_fft_self_convolve(cand))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand
    return best_seq, float(best_score)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine runs at most ``budget_s`` seconds and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    # Determine a set of candidate lengths (focus on medium‑large sizes)
    base_n = 8000
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        pass
    length_factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted(
        {max(500, int(base_n * f)) for f in length_factors if max(500, int(base_n * f)) <= 20000}
    )
    if not lengths:
        lengths = [8000]

    overall_best_seq = None
    overall_best_score = np.inf
    target_score = 1.503

    # --------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        # allocate a slice of the remaining budget to this length
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(0.2, remaining * 0.35)

        # --------------------------------------------------------------
        # 1) build a pool of initial candidates
        candidates = []

        # predefined seeds
        for kind in ("prev", "arcsine", "beta", "uniform", "random"):
            try:
                candidates.append(_initial_sequence(n, kind))
            except Exception:
                continue

        # fine grid over power‑law shapes
        shape_best, _ = _grid_search_best_fine(n)
        if shape_best is not None:
            candidates.append(shape_best)

        # simple two‑tail family
        tail_best, _ = _search_tail_best(n)
        if tail_best is not None:
            candidates.append(tail_best)

        # a few random beta draws
        for _ in range(12):
            a = np.random.beta(np.random.uniform(0.1, 1.0),
                               np.random.uniform(0.1, 1.0),
                               size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            candidates.append(a)

        # keep the few most promising according to the cheap internal score
        def cheap_score(v):
            return 2.0 * n * _fft_self_convolve(v).max()
        candidates = sorted(candidates, key=cheap_score)[:5]

        # --------------------------------------------------------------
        # 2) Adam optimisation for each promising candidate
        for a0 in candidates:
            if time.time() >= slice_deadline - 1.0:
                break
            seq, internal_score = _optimize_adam(a0, n, slice_deadline)

            # The internal score equals the public evaluation (no clamping needed)
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = internal_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq.copy()
                print(f"*** New global best → len={n} score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break
        if overall_best_score <= target_score:
            break

    # --------------------------------------------------------------
    # 3) Final stochastic refinement on the globally best vector
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n_best = len(overall_best_seq)
        refined_seq, _ = _local_refinement(overall_best_seq, n_best, deadline)
        try:
            final_score = evaluate_sequence(refined_seq.tolist())
        except Exception:
            final_score = None
        if final_score is not None and final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = refined_seq.copy()
            print(f"*** After final refinement score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # safety fallback (should never be needed)
    if overall_best_seq is None:
        n_fallback = 5000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Safety fallback used.")

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
