def _exact_gradient(a):
    n = len(a)
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # tolerance for numerics
    eps = 1e-12
    mask = np.abs(conv - max_val) <= eps
    if not mask.any():
        mask = conv == max_val
    w = mask.astype(float)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt
    # Compute convolution w * a_rev using FFT
    size_fft = 1 << ((len(w) + len(a) - 1).bit_length())
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a[::-1], n=size_fft)
    conv_w = np.fft.irfft(W * A_rev, n=size_fft)
    # conv_w indices correspond to length len(w)+len(a)-1.
    # A_rev's length = n, w's length = 2n-1, conv length = 3n-2
    # We need entries from start = n-1 to start + n - 1 (i.e., the same region as for gradient)
    start = n - 1
    g = 2.0 * conv_w[start:start + n]
    return g

def _adam_optimize(init_a, n, deadline):
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()
    m, v = np.zeros_like(a), np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it=0
    size_fft = 1 << ((3*n-2).bit_length())
    while time.time() < deadline:
        it+=1
        # compute conv and score
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] n={n} it={it} score={best_score:.6f}")
        # compute exact gradient
        max_val = conv.max()
        mask = np.abs(conv - max_val) <= 1e-12
        if not mask.any():
            mask = conv == max_val
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        # compute gradient via FFT
        W = np.fft.rfft(w, n=size_fft)
        A_rev = np.fft.rfft(a[::-1], n=size_fft)
        conv_w = np.fft.irfft(W * A_rev, n=size_fft)
        start = n-1
        grad = 2.0 * conv_w[start:start+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = 0.5 / (np.sqrt(v_hat) + eps)  # base step 0.5
        a = a - step * m_hat
        a = _project_onto_simplex(a, 1.0)

    return best, best_score

"""
Proposes a non‑negative step function (list of heights) that minimises
`evaluate_sequence`.  The algorithm consists of three stages:

1. **Broad shape search** – Enumerate a handful of lengths (2 k … 15 k)
   and several parametric families (uniform, arcsine, symmetric Beta,
   mixtures with uniform).  The best candidate (lowest score) across all
   lengths is kept as a warm‑start.

   run a few independent Adam restarts using the exact sub‑gradient of the
   max‑convolution (computed by a masked convolution via FFT).  The
   optimizer works directly on the simplex (Euclidean projection).

3. **Stochastic hill‑climbing polishing** – A cheap local search that
   repeatedly moves a random fraction of the mass between two coordinates
   and keeps improvements.  Tiny Gaussian jitter is added periodically.

All work respects the 1000 s budget; the final best sequence is returned
as a plain ``list[float]``.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = v.astype(np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points, then normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) pdf evaluated on mid‑points of n bins."""
    a = max(a, 0.05)                     # avoid blow‑up at zero
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0 or not np.isfinite(pdf_sum):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= pdf_sum
    return pdf


# ----------------------------------------------------------------------
def _candidate_shapes_for_n(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest list of hand‑crafted probability vectors for a given length."""
    cand = []

    # 1) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) arcsine
    cand.append(_arcsine_distribution(n))

    # 3) symmetric Beta for a grid of a‑values
    for a in np.linspace(0.10, 0.90, 9):
        cand.append(_beta_distribution(n, a))

    # 4) mixtures arcsine + uniform
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    for w in (0.0, 0.2, 0.4):
        mix = (1.0 - w) * arcs + w * uni
        cand.append(mix)

    # 5) a few random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(3):
        cand.append(rng.dirichlet(np.ones(n)))

    # 6) previous best (if it matches the length)
    if prev_seq is not None and prev_seq.size == n:
        cand.append(prev_seq.copy())

    # Ensure every candidate lies on the simplex
    final = [_project_onto_simplex(c, 1.0) for c in cand]
    return final


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex using the *exact* sub‑gradient of the
    max‑convolution (masked convolution via FFT).  Returns the best
    vector seen and its raw 2·n·max(conv) score.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # FFT size sufficient for a (length n) convolved with a mask of length 2n‑1
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # --- objective ----------------------------------------------------
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] n={n} it={it} score={best_score:.6f}")

        # --- exact sub‑gradient of max(conv) ------------------------------
        max_val = conv.max()
        # numerical tolerance for ties
        mask = np.abs(conv - max_val) <= 1e-12
        if not mask.any():
            mask = conv == max_val
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(conv)] = 1.0

        # convolution w * reversed(a)
        W = np.fft.rfft(w, n=size_fft)
        A_rev = np.fft.rfft(a[::-1], n=size_fft)
        conv_w = np.fft.irfft(W * A_rev, n=size_fft)

        start = n - 1
        grad = 2.0 * conv_w[start:start + n]

        # --- Adam update --------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / (np.sqrt(v_hat) + eps)   # base learning‑rate 0.5
        a = a - step * m_hat

        # projection onto simplex
        a = _project_onto_simplex(a, 1.0)

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: move a random fraction of the mass
    from one index to another, keep the move if it improves the score.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue

        # move a random fraction (up to 40 %) of the mass from i to j
        frac = rng.random() * 0.40
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Main entry point required by the grader.
    Returns a list of non‑negative heights (probability vector) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for final conversion / printing
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length candidates (respect the 2 k … 15 k interval)
    length_candidates = [2000, 4000, 6000, 8000, 10000, 12000, 15000]

    # Try to keep a previously found solution as a warm‑start
    prev_seq = None
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – fast parametric sweep over lengths and families
    shape_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.15)
    best_seq = None
    best_score = float("inf")
    best_n = None

    for n in length_candidates:
        if time.time() >= shape_deadline:
            break
        candidates = _candidate_shapes_for_n(n, prev_seq)
        for cand in candidates:
            if time.time() >= shape_deadline:
                break
            scr = evaluate_sequence(cand.tolist())
            if scr < best_score:
                best_score = scr
                best_seq = cand.copy()
                best_n = n
                print(f"[shape] n={n} new best score {best_score:.6f}")

    # Fallback in the unlikely event that nothing was generated
    if best_seq is None:
        best_n = 8000
        best_seq = np.full(best_n, 1.0 / best_n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Phase 1 – several Adam restarts (long optimisation)
    remaining = deadline - time.time()
    # reserve a slice for hill‑climbing
    hill_reserve = max(10.0, remaining * 0.15)
    adam_time = remaining - hill_reserve
    restart_count = max(1, int(adam_time // 20.0))  # at most ~20 s per restart
    per_restart = adam_time / restart_count if restart_count else 0.0

    rng = np.random.default_rng()
    best_adam_seq = best_seq.copy()
    best_adam_score = 2.0 * best_n * _fft_self_convolve(best_seq).max()

    for r in range(restart_count):
        # perturb the start point a little to diversify the search
        init = best_seq + rng.normal(scale=0.02, size=best_seq.shape)
        init = np.maximum(init, 0.0)
        init = _project_onto_simplex(init, 1.0)

        run_deadline = time.time() + per_restart
        seq_opt, score_opt = _adam_optimize(init, best_n, run_deadline)

        if score_opt < best_adam_score:
            best_adam_score = score_opt
            best_adam_seq = seq_opt.copy()
            print(f"[Adam restart {r}] improved score to {best_adam_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final hill‑climbing polishing
    hill_deadline = deadline
    seq_final, final_score = _hill_climb(best_adam_seq, best_n, hill_deadline)

    # ------------------------------------------------------------------
    # Safety fallback (should never trigger)
    if final_score is None or math.isinf(final_score):
        seq_final = np.full(2000, 1.0 / 2000, dtype=np.float64)
        final_score = evaluate_sequence(seq_final.tolist())

    print(f"Final best score: {final_score:.6f} (length = {len(seq_final)})")
    return seq_final.tolist()
