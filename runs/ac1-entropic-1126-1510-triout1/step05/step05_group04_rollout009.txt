def compute_score(a):

def subgradient_max(a):
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = mask.sum()
    w = np.zeros_like(conv)
    if cnt > 0:
        w[mask] = 1.0 / cnt
    # Compute gradient: g_a = 2 * conv(w, a[::-1]) slice as before.
    conv_wr = np.convolve(w, a[::-1])
    n = a.shape[0]
    g_a = 2.0 * conv_wr[n-1:n-1+n]
    return g_a, max_c

"""
Propose a low‑score step function.

The algorithm combines several complementary optimisation phases:

1.  A multi‑restart continuation of a smooth‑max surrogate using L‑BFGS‑B.
    Diverse initial points (uniform, exponential, Dirichlet‑sparse,
    Gaussian, a projected reference and random piecewise‑constant patterns)
    are explored.  The temperature μ is decreased geometrically.

2.  Projected sub‑gradient descent on the true hard‑max objective.
    The gradient of the max‑convolution is computed analytically,
    a back‑tracking line‑search is performed and the iterate is projected
    onto the ℓ¹‑simplex.

3.  Simulated‑annealing‑style local search.  At each step a small amount
    of mass is transferred from a source index to a destination index;
    moves that improve the score are always accepted, otherwise they are
    accepted with a probability that decays with a temperature schedule.

4.  A cheap large‑scale random‑block perturbation followed by a monotone
    sorting heuristic.

All phases respect a global time budget (default 1000 s).  The best
candidate found is printed and returned as a plain Python list.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    sum_e = np.sum(e)
    if sum_e == 0.0:
        # Should not happen, but protect against division by zero
        return np.full_like(y, 1.0 / y.size)
    return e / sum_e

# ----------------------------------------------------------------------
def _fast_max_convolution(a: np.ndarray) -> float:
    """
    Return max_{k} (a * a)[k] using an FFT‑based convolution.
    a is assumed to be real, non‑negative and already clipped to [0, 1000].
    """
    n = a.shape[0]
    # length of the full convolution is 2*n-1
    m = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    # Only the first 2*n-1 entries are the true linear convolution
    return float(conv[:2 * n - 1].max())

# ----------------------------------------------------------------------
def _compute_score(a: np.ndarray) -> float:
    """
    Exact score used by evaluate_sequence (but much faster):
        2·n·max(a*a) / (sum(a))²
    The heights are clipped to the interval [0, 1000] before the
    computation, exactly as evaluate_sequence does.
    """
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    n = a.shape[0]
    max_c = _fast_max_convolution(a)
    return float(2.0 * n * max_c / (sum_a * sum_a))

# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate:
        f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Parameters
    ----------
    y : ndarray, shape (n,)
        Unconstrained parameters (logits of the simplex).
    mu : float
        Smoothing temperature (>0).  μ = 0 corresponds to the hard max.
    n : int
        Length of the step function.
    S : float
        Target ℓ¹‑norm (√(2·n)).
    """
    # probabilities on the simplex, actual heights
    p = _softmax(y)                     # Σ p_i = 1, p_i ≥ 0
    a = S * p                           # Σ a_i = S

    # full (linear) convolution a*a (length 2·n‑1)
    conv = np.convolve(a, a)

    # smooth‑max weights w_k ∝ exp((conv_k - max)/μ)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        # protect against overflow/underflow
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – uniform weight on all active indices
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. heights a:
    #   g_a[i] = 2 * Σ_k w_k * a_{k-i}
    # The convolution w * a_rev (where a_rev = a[::-1]) gives the needed sums.
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable that returns (value, gradient) for scipy.optimize."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _subgradient_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Return a sub‑gradient of the hard‑max objective and the current max."""
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    w = np.zeros_like(conv, dtype=float)
    if cnt > 0:
        w[mask] = 1.0 / cnt
    # gradient w.r.t. a
    conv_wr = np.convolve(w, a[::-1])
    n = a.shape[0]
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g_a, max_c

    init_ys.append(np.log(pwc + 1e-12))

    mu_min = 1e-8

                    obj,
                    y,

            it += 1

            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)

            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # Stage 5 – Monotone sorting heuristics ---------------------------
    if time.time() < deadline and best_score < float("inf"):
        # descending order
        cand_desc = np.sort(best_a)[::-1]
        cand_desc = _project_to_simplex(cand_desc, S)

        # ascending order
        cand_asc = np.sort(best_a)
        cand_asc = _project_to_simplex(cand_asc, S)
        _maybe_update(cand_asc)

        print(f"[final] official score = {official_score:.6f}")
    except Exception:
        official_score = best_score
        print(f"[final] best internal score = {best_score:.6f}")

"""
Propose a low‑score step function.

The algorithm combines several complementary optimisation phases:

1.  Multi‑restart continuation of a smooth‑max surrogate using L‑BFGS‑B.
    Diverse initial points (uniform, exponential, Dirichlet‑sparse,
    Gaussian, a projected reference and random piece‑wise‑constant
    patterns) are explored.  The temperature μ is decreased geometrically.

2.  Projected sub‑gradient descent on the exact hard‑max objective.
    The gradient of the max‑convolution is obtained analytically,
    a back‑tracking line‑search is performed and the iterate is projected
    onto the ℓ¹‑simplex.

3.  Simulated‑annealing‑style local search.  At each step a small amount
    of mass is transferred from a source index to a destination index;
    moves that improve the score are always accepted, otherwise they are
    accepted with a Metropolis probability that decays with a temperature
    schedule.

4.  A cheap large‑scale random‑block perturbation followed by a monotone‑
    sorting heuristic.

All stages respect a global time budget (default 1000 s).  The best
candidate found is printed and returned as a plain Python list.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    sum_e = np.sum(e)
    if sum_e == 0.0:  # safeguard, should not happen
        return np.full_like(y, 1.0 / y.size)
    return e / sum_e

# ----------------------------------------------------------------------
def _fast_max_convolution(a: np.ndarray) -> float:
    """
    Return max_{k} (a * a)[k] using an FFT‑based convolution.
    Heights are assumed to be already clipped to [0, 1000].
    """
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return float(conv[: 2 * n - 1].max())

# ----------------------------------------------------------------------
def _compute_score(a: np.ndarray) -> float:
    """
    Exact score used by evaluate_sequence (but much faster):
        2·n·max(a*a) / (sum(a))²
    The heights are clipped to the interval [0, 1000] before the
    computation, exactly as evaluate_sequence does.
    """
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    n = a.shape[0]
    max_c = _fast_max_convolution(a)
    return float(2.0 * n * max_c / (sum_a * sum_a))

# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate:
        f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Parameters
    ----------
    y : ndarray, shape (n,)
        Unconstrained parameters (logits of the simplex).
    mu : float
        Smoothing temperature (>0).  μ = 0 corresponds to the hard max.
    n : int
        Length of the step function.
    S : float
        Target ℓ¹‑norm (√(2·n)).
    """
    p = _softmax(y)               # Σ p_i = 1, p_i ≥ 0
    a = S * p                     # Σ a_i = S

    conv = np.convolve(a, a)      # length 2·n‑1

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # guard overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable returning (value, gradient) for SciPy's L‑BFGS‑B."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _subgradient_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Return a sub‑gradient of the hard‑max objective and the current max."""
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    w = np.zeros_like(conv, dtype=float)
    if cnt > 0:
        w[mask] = 1.0 / cnt
    conv_wr = np.convolve(w, a[::-1])
    n = a.shape[0]
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g_a, max_c

    maxiter_per_mu = 250
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:

        print(f"[lbfgs] restart {rid + 1}/{n_restart}")

        a_cur = S * _softmax(y)
        mu = max(_fast_max_convolution(a_cur), 1.0)

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,

    if time.time() < deadline and best_score < float("inf"):

            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1

            max_move = cur_a[src] * 0.07    # move up to 7 % of source mass
            delta = rng.random() * max_move
            if delta <= 0.0:
                continue

            a_new = cur_a.copy()
            a_new[src] -= delta
            a_new[dst] += delta

            new_score = _compute_score(a_new)

            if new_score < cur_score - 1e-12:
                cur_a, cur_score = a_new, new_score
                _maybe_update(cur_a)
            else:
                if rng.random() < math.exp(-(new_score - cur_score) / T):
                    cur_a, cur_score = a_new, new_score
            it += 1
        print(f"[anneal] performed {it} iterations")

    # ------------------------------------------------------------------
    # Stage 4 – random block perturbations -----------------------------
    if time.time() < deadline and best_score < float("inf"):
        remaining = deadline - time.time()
        max_trials = int(min(7000, remaining * 130))
        print(f"[block] {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(3, 20)                     # number of blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # ------------------------------------------------------------------
    # Stage 5 – monotone‑sorting heuristics -----------------------------
    if time.time() < deadline and best_score < float("inf"):
        cand_desc = np.sort(best_a)[::-1]
        cand_desc = _project_to_simplex(cand_desc, S)
        _maybe_update(cand_desc)

        cand_asc = np.sort(best_a)
        cand_asc = _project_to_simplex(cand_asc, S)
        _maybe_update(cand_asc)

    # ------------------------------------------------------------------
    # final validation (optional) --------------------------------------
    try:
        official_score = evaluate_sequence(best_a.tolist())
        print(f"[final] official score = {official_score:.6f}")
    except Exception:
        print(f"[final] best internal score = {best_score:.6f}")

    return best_a.tolist()
