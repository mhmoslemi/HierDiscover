def _smooth_gradient(a, gamma):
    b = _fft_self_convolve(a)
    max_b = b.max()
    shifted = b - max_b
    # for numerical stability: use float64

"""
Propose a low‑score non‑negative step function.

The algorithm works in three stages:

1. **Diverse initialization** – a random Dirichlet draw, a deterministic
   arcsine law or a random permutation thereof are used to create many
   starting vectors of various lengths.

2. **Smooth Adam‑based sub‑gradient descent** – the objective
   ``2·n·max(conv(a,a))`` (with ``sum(a)=1``) is minimised.
   Instead of the plain sub‑gradient (active‐set indicator) a softmax
   weighting of all convolution entries is used (controlled by a
   temperature parameter that is gradually increased).  The update is
   performed with the Adam optimiser and the iterate is projected onto
   the probability simplex after each step.

3. **Stochastic hill climbing** – once the continuous optimisation has
   stabilised, cheap combinatorial moves (mass transfer between two
   positions, swapping two entries) are applied.  Improvements are kept
   and occasional jitter helps to escape plateaux.

The whole search runs until the time budget expires (≈ 1000 s).  The
best sequence found is returned as a plain list of floats suitable for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project ``v`` onto the simplex {x ≥ 0, sum(x) = s} (Euclidean)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two greater or equal to ``x``."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution ``a * a`` using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Return ``2·n·max(a*a)`` (assumes ``sum(a)=1``).  This is the same
    quantity returned by ``evaluate_sequence`` up to the normalisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _adam_smooth_subgrad_opt(
    a: np.ndarray,
    deadline: float,
    target_score: float = 1.503,
    max_gamma: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    ``a`` must already be on the simplex (sum=1).  Returns the best vector
    seen and its score.
    """
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    iteration = 0
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    # start with a gentle smoothing, increase slowly
    gamma = 10.0
    gamma_mul = 1.001

    while time.time() < deadline:
        iteration += 1

        # ----- convolution and smooth weights -----
        b = _fft_self_convolve(a)
        max_b = b.max()

        # softmax over the whole convolution (numerically stable)
        shifted = b - max_b
        w_raw = np.exp(gamma * shifted)
        w = w_raw / w_raw.sum()                     # shape (2n‑1,)

        # ----- gradient of the smoothed max -----
        conv_rev = _fft_convolve(a[::-1], w)        # length = n + (2n‑1) -1
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # diagonal correction for the i=j terms (k = 2i)
        even_idxs = np.flatnonzero(np.arange(len(w)) % 2 == 0)
        if even_idxs.size:
            diag_i = even_idxs // 2                 # positions i where 2i is active
            # subtract a[i] * weight_of_offset_2i
            grad[diag_i] -= a[diag_i] * w[even_idxs]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = (0.1 / (np.sqrt(v_hat) + eps)) * m_hat   # base learning rate = 0.1

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # ----- evaluation -----
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam iter={iteration} new best {best_score:.6f}")

        # increase smoothing parameter slowly (approaches the true max)
        gamma = min(max_gamma, gamma * gamma_mul)

        # early exit if already good enough
        if best_score <= target_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    cur_score: float,
    deadline: float,
    target_score: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search:
      * transfer a random amount of mass between two positions,
      * swap two entries.
    Accept only improving moves; occasional jitter helps to escape flat
    regions.  Returns the best vector found.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    iteration = 0
    max_transfer = 0.05  # start with up‑to 5 % of the total mass

    while time.time() < deadline:
        iteration += 1

        # gradually shrink the transferred mass
        transfer = max_transfer * (0.5 ** (iteration / 20000.0))

        if np.random.rand() < 0.5:
            # ----- mass transfer -----
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = np.random.rand() * min(transfer, a[i])
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            # ----- swap two entries -----
            i, j = np.random.choice(n, 2, replace=False)
            a_new = a.copy()
            a_new[i], a_new[j] = a[j], a[i]

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape plateaux
        if iteration % 5000 == 0 and np.random.rand() < 0.1:
            a = best_a + np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

        if best_score <= target_score:
            break

    return best_a, best_score

        else:
            # Dirichlet draw with a random concentration parameter
            alpha = 10 ** np.random.uniform(-1.0, 1.0)   # between 0.1 and 10
            a = np.random.dirichlet(np.full(n, alpha))

        # occasional random permutation (ordering matters for the convolution)
        if np.random.rand() < 0.2:
            np.random.shuffle(a)

        a = a / a.sum()   # ensure exact sum = 1

        # ----- phase 1: smooth Adam‑based sub‑gradient optimisation -----
        a_opt, score_opt = _adam_smooth_subgrad_opt(a, deadline)

        # ----- phase 2: cheap stochastic hill climbing -----
        a_opt, score_opt = _hill_climb(a_opt, score_opt, deadline)

        # ----- keep the global best -----
        if score_opt < best_score - 1e-14:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] GLOBAL new best {best_score:.6f} (len={n})")

        # early exit if we have already reached the target bound
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # fallback – should never happen, but guarantees a valid return value.
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _max_convolution_score(best_seq)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # ``evaluate_sequence`` will clip/normalise again, so we simply return the raw list.
    return best_seq.tolist()
