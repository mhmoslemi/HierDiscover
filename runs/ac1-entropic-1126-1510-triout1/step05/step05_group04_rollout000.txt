"""
Propose a low‑score step function.

Algorithm Overview
------------------
1. **Problem size** – Choose a length `n` (≈ 6 000‑10 000) and the target
   ℓ¹‑norm `S = sqrt(2·n)`.  The search works directly on the simplex
   `Σ a_i = S, a_i ≥ 0`.

2. **Smooth‑max L‑BFGS** – Re‑parameterise `a` by logits `y`
   (`a = S·softmax(y)`).  For a decreasing sequence of smoothing
   parameters `μ` we minimise the smooth surrogate
   `max(conv) + μ·log∑exp((conv‑max)/μ)` using L‑BFGS‑B.
   Several diverse initial logits are tried and the best result is kept.

3. **Projected sub‑gradient descent** – From the best smoothed solution
   we run a classic sub‑gradient method on the *hard* max objective,
   with an adaptive back‑tracking line search and a small momentum term.

4. **Simulated‑annealing fine‑tuning** – A large number of tiny

   block‑averaging, local smoothing) are attempted.  Moves are accepted
   according to a temperature schedule; every improvement updates the
   incumbent.

5. **Final polishing** – A short L‑BFGS run with a very small `μ`
   (`1e‑9`) finishes the optimisation, followed by monotone sorting
   heuristics.

All steps respect the time budget (default 1000 s) and the code prints
progress messages.  The returned list of floats is the best step‑function
found.

import time

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z <= 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / np.sum(e)


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate (value + gradient) for the auto‑convolution
    objective.  Returns (f(y), ∇f(y)).
    """
    p = _softmax(y)                     # Σ p = 1, p ≥ 0
    a = S * p                           # Σ a = S

    # linear convolution a*a (length 2·n‑1) via FFT (fast for large n)
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)      # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient wrt a:  g_a[i] = 2 * Σ_k w_k * a_{k-i}
    # which is a convolution of w with reversed a
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

    # chain rule through soft‑max
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable that returns (value, gradient) for scipy.optimize."""
    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear auto‑convolution of a (via FFT)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
    # numerical noise may create tiny negative values – clamp
    return float(np.maximum(conv, 0.0).max())


# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """
    Exact objective used in evaluate_sequence (without the safety clamps).
    Assumes a is non‑negative.
    """
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 1e-9:
        return float("inf")
    max_c = _max_conv_fft(a)
    return float(2.0 * n * max_c / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) with a
    low evaluate_sequence() score.  The routine respects the time budget
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size ------------------------------------------------
    try:
        ref_len = len(height_sequence_1)   # type: ignore[name-defined]
        n = min(max(ref_len * 2, 6000), 10000)
        have_ref = True
    except Exception:
        n = 8000
        have_ref = False

    S = math.sqrt(2.0 * n)          # target ℓ¹‑norm
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # ------------------------------------------------------------------
    def _maybe_update(a_vec: np.ndarray, note: str = "") -> None:
        """Evaluate a candidate and keep it if it improves the incumbent."""
        nonlocal best_score, best_a
        sc = _objective_score(a_vec)
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve{note}] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper: build initial simplex vectors (a) -------------------------
    init_as = []

    # 0 – uniform
    init_as.append(np.full(n, S / n, dtype=float))

    # 1 – left‑hand exponential decay
    r1 = rng.uniform(0.92, 0.99)
    exp_left = (r1 ** np.arange(n))[::-1]    # large on the left
    init_as.append(S * exp_left / exp_left.sum())

    # 2 – right‑hand exponential decay
    r2 = rng.uniform(0.92, 0.99)
    exp_right = r2 ** np.arange(n)          # large on the right
    init_as.append(S * exp_right / exp_right.sum())

    # 3 – centred Gaussian
    sigma = rng.uniform(0.08, 0.20) * n
    centre = (np.arange(n) - n // 2) / sigma
    gauss = np.exp(-0.5 * centre ** 2)
    init_as.append(S * gauss / gauss.sum())

    # 4 – sparse Dirichlet (α < 1)
    alpha_sparse = rng.uniform(0.15, 0.45)
    dir_sparse = rng.dirichlet([alpha_sparse] * n)
    init_as.append(S * dir_sparse)

    # 5 – dense Dirichlet (α > 1)
    alpha_dense = rng.uniform(1.5, 3.0)
    dir_dense = rng.dirichlet([alpha_dense] * n)
    init_as.append(S * dir_dense)

    # 6 – reference if available
    if have_ref:
        ref = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        ref = _project_to_simplex(ref, S)
        init_as.append(ref)

    # ------------------------------------------------------------------
    # Phase I – smooth‑max L‑BFGS with continuation --------------------
    mu_factor = 0.90
    mu_min = 1e-6
    maxiter_per_mu = 400
    n_restart = len(init_as)

    for rid, a0 in enumerate(init_as):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}] initialising")
        # logits y (unconstrained)
        eps = 1e-12
        p0 = a0 / S
        y = np.log(p0 + eps)

        # initialise μ from the current max‑convolution
        mu = max(_max_conv_fft(a0), 1.0)

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a, note="(L‑BFGS)")

            mu *= mu_factor

    # ------------------------------------------------------------------
    # Phase II – projected sub‑gradient descent -------------------------
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        it = 0
        max_iters = 8000
        step0 = S * 0.2 / n
        momentum = np.zeros_like(a)

        while time.time() < deadline and it < max_iters:
            # sub‑gradient of max(conv) at current a
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            w = np.zeros_like(conv)
            w[mask] = 1.0 / cnt
            conv_rev = np.convolve(w, a[::-1])
            g = 2.0 * conv_rev[n - 1 : n - 1 + n]

            # Nesterov‑style momentum
            beta = 0.7
            momentum = beta * momentum + (1.0 - beta) * g

            # back‑tracking line search
            step = step0 / (1.0 + 1e-5 * it)
            improved = False
            for _ in range(12):
                a_new = a - step * momentum
                a_new = _project_to_simplex(a_new, S)
                sc = _objective_score(a_new)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[subgrad] iter {it:4d} step {step:.6e} => {best_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # occasional random jitter to escape plateaus
                i, j = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a[i] * 0.07
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, S)
            it += 1

    # ------------------------------------------------------------------
    # Phase III – simulated‑annealing fine‑tuning ------------------------
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        score = best_score
        T = 1.0
        T_min = 1e-4
        cool = 0.9993
        # pre‑compute power‑of‑two length for FFT (used by _max_conv_fft)
        while time.time() < deadline and T > T_min:
            # ---- propose a neighbour ----
            if rng.random() < 0.6:
                # simple mass transfer between two coordinates
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                if a[i] <= 0.0:
                    continue
                eps_move = rng.random() * a[i] * 0.12
                cand = a.copy()
                cand[i] -= eps_move
                cand[j] += eps_move
            elif rng.random() < 0.85:
                # block‑averaging move
                nb = rng.integers(3, 9)
                cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
                blocks = np.split(np.arange(n), cuts)
                cand = a.copy()
                for blk in blocks:
                    if blk.size == 0:
                        continue
                    avg = cand[blk].mean()
                    cand[blk] = avg
                cand = _project_to_simplex(cand, S)
            else:
                # local smoothing by a small convolution kernel
                kernel = np.array([0.25, 0.5, 0.25])
                cand = np.convolve(a, kernel, mode="same")
                cand = _project_to_simplex(cand, S)

            sc_new = _objective_score(cand)
            delta = sc_new - score
            if delta < 0 or rng.random() < math.exp(-delta / T):
                a = cand
                score = sc_new
                if sc_new < best_score:
                    _maybe_update(a, note="(SA)")

            T *= cool

    # ------------------------------------------------------------------
    # Phase IV – final ultra‑smooth L‑BFGS polishing --------------------
    if time.time() < deadline and best_score < float("inf"):
        # Start from the best found logits
        eps = 1e-12
        y = np.log(best_a / S + eps)
        mu = 1e-9
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 300, "gtol": 1e-13, "disp": False},
            )
            a_fin = S * _softmax(res.x)
            _maybe_update(a_fin, note="(final L‑BFGS)")
        except Exception as exc:
            print(f"[error] final L‑BFGS failed: {exc}")

    # ------------------------------------------------------------------
    # Phase V – monotone sorting heuristics -----------------------------
    if time.time() < deadline and best_score < float("inf"):
        for direction in ("desc", "asc"):
            if direction == "desc":
                cand = np.sort(best_a)[::-1]
            else:
                cand = np.sort(best_a)
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand, note=f"({direction})")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
