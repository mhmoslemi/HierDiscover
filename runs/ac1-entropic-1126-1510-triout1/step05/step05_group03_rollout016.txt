"""
search for a low‑score non‑negative step function.

The algorithm uses three ideas that are not present in the previous
solution:

1.  **LP‑based “capacity filling”.**  Starting from a reasonable
    distribution (arcsine) we solve a linear program that finds the
    vector `g` of maximal ℓ₁‐mass subject to the constraint
    `f * g ≤ M` (where `M` is the current peak of `f*f`).  Mixing a
    small fraction of this `g` back into `f` strictly reduces the
    peak.  Repeating the LP step a few times dramatically flattens the
    convolution while staying on the simplex.

2.  **Fine‑grained Adam optimisation** of a smooth approximation of the
    ℓ∞‑norm (soft‑max surrogate).  This gives a fast descent into a
    low‑peak region.

3.  **Exact sub‑gradient polishing** of the true ℓ∞ objective.

All stages respect the simplex constraint Σaᵢ=1, aᵢ≥0.
The routine adapts its work‑load to the remaining time budget (default
1000 s) and finally returns the best sequence found.
"""

import math
import time
import numpy as np

# optional heavy imports – fall back to the previous method if unavailable
try:
    from scipy.optimize import linprog         # linear programming
    from scipy.sparse import csr_matrix        # sparse matrix for the LP
    SCIPY_AVAILABLE = True

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int, shift: float = 0.5) -> np.ndarray:
    """
    Discrete arcsine distribution on `n` points.
    The points are taken at x_i = (i + shift) / n with 0<shift<1.
    """
    i = np.arange(n, dtype=np.float64) + shift
    x = i / n
    # keep away from the exact 0 or 1 to avoid infinities
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    vals = 1.0 / np.sqrt(x * (1.0 - x))
    vals /= vals.sum()
    return vals

def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate

        L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights, numerically stable
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = np.fft.ifft(np.fft.rfft(a_rev) * np.fft.rfft(w)).real
    # the relevant slice is of length n, centred at index n‑1
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to avoid stagnation
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses a uniform weight on the active set (indices attaining the current maximum).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    conv_rev = np.fft.ifft(np.fft.rfft(a_rev) * np.fft.rfft(w)).real
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short exact‑max refinement using sub‑gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, max_iters: int = 5, mix: float = 0.2,
               deadline: float = 0.0) -> np.ndarray:
    """
    Perform a few linear‑programming steps.
    Each step solves

        max  Σ g_j
        s.t.  f * g ≤ M , g ≥ 0

    where M = max (f*f).  The optimum g is mixed back into f with factor `mix`.
    """
    if not SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    a = f.copy()
    best_a = a.copy()
    best_score = 2 * n * _fft_self_convolve(a).max()

    for it in range(max_iters):
        if deadline and time.time() >= deadline:
            break
        M = _fft_self_convolve(a).max()
        # build the linear operator  (f * g)_k = Σ_i f_i g_{k-i}
        # matrix with shape (2n‑1, n)
        m_rows = 2 * n - 1
        conv_mat = np.zeros((m_rows, n), dtype=np.float64)
        for i in range(n):
            conv_mat[i:i + n, i] = a
        A = csr_matrix(conv_mat)
        b = np.full(m_rows, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_j
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        except Exception:   # pragma: no cover
            break
        if not res.success:
            break
        g = res.x
        s = g.sum()
        if s <= 0:
            break
        g_norm = g / s
        a = (1.0 - mix) * a + mix * g_norm
        a = _project_onto_simplex(a, s=1.0)

        cur_score = 2 * n * _fft_self_convolve(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point – searches for a sequence with a low `evaluate_sequence`
    score.  The routine is deterministic given the seed and stops well before
    the budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # a safety margin

    # ------------------------------------------------------------------
    # try to reuse a known construction if it exists
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq, s=1.0)
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # length candidates – we explore a modest range around a few
    # “medium” values where the arcsine‐based construction was good.
    lengths = [800, 1000, 1200, 1500, 1800, 2000, 2400, 2800, 3200, 3800, 4400]
    lengths = [n for n in lengths if n >= 500]

    best_seq = None
    best_score = float('inf')

    print("[phase 1] broad exploration of lengths and simple arcsine shifts")
    for n in lengths:
        if time.time() >= deadline:
            break

        # a few shift values for the arcsine sampling (avoid exact 0/1)
        for shift in (0.5, 0.33, 0.66):
            a = _arcsine_distribution(n, shift=shift)
            a = _project_onto_simplex(a, s=1.0)

            peak = _fft_self_convolve(a).max()
            fast_score = 2.0 * n * peak
            if fast_score < best_score:
                best_score = fast_score
                best_seq = a.copy()
                print(f"  [init] n={n:4d} shift={shift:<4.2f} fast_score={fast_score:.6f}")

            # ------------------------------------------------------------------
            # if we are in the “reasonable” size regime, run a tiny LP‑refine
            if SCIPY_AVAILABLE and n <= 2000:
                lp_deadline = time.time() + min(8.0, (deadline - time.time()) / 4.0)
                a_lp = _lp_refine(a, max_iters=4, mix=0.2, deadline=lp_deadline)
                peak_lp = _fft_self_convolve(a_lp).max()
                fast_score_lp = 2.0 * n * peak_lp
                if fast_score_lp < best_score:
                    best_score = fast_score_lp
                    best_seq = a_lp.copy()
                    print(f"  [LP‑ref] n={n:4d} shift={shift:<4.2f} fast_score={fast_score_lp:.6f}")

    # ------------------------------------------------------------------
    # If nothing was produced (unlikely), fall back to a uniform vector.
    if best_seq is None:
        n = 1200
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # ----- 2. GRADIENT‑BASED REFINEMENT ---------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        n = len(best_seq)
        print(f"[phase 2] Adam optimisation (n={n}) – time left {time_left:.1f}s")
        phase1_deadline = time.time() + time_left * 0.55
        a_opt, _ = _adam_optimize(best_seq, phase1_deadline)

        # short exact‑max polishing
        phase2_deadline = time.time() + time_left * 0.30
        a_opt = _refine_max(a_opt, phase2_deadline)

        best_seq = a_opt.copy()
        best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
        print(f"[phase 2] post‑gradient fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # ----- 3. FINAL black‑box evaluation ---------------------------------
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < best_score:
            best_score = true_score
    except Exception as exc:                    # pragma: no cover
        print("evaluate_sequence raised:", exc)

    print(f"Best obtained score (approx.): {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
