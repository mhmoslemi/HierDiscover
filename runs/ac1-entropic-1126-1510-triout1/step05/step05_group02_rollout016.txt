```python
"""
Candidate proposal using random Dirichlet / Beta families and
an accelerated extreme–subgradient descent on the ℓ∞ self‑convolution.
All updates respect the probability simplex via Euclidean projection.
The algorithm first explores a rich family of parameterised step functions,
keeps the best few, then refines them for the remaining time budget.
"""

import time
import math
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, Σx=s}. """
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a * a."""
    return _fft_self_convolve_full(a).max()

def _raw_score(a: np.ndarray) -> float:
    """2·n·max(a*a) – the value that the black‑box returns up to scaling."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete mid‑point Beta(α,β) on n points, normalised to sum = 1."""
    # avoid singularities for very small params
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

def _dirichlet_random(n: int, concentration: float) -> np.ndarray:
    """Draw a random point on the simplex using a symmetric Dirichlet."""
    if concentration <= 0:
        concentration = 1e-6
    g = np.random.gamma(concentration, 1.0, size=n)
    s = g.sum()
    if s == 0.0:
        g[:] = 1.0 / n
    else:
        g /= s
    return g

def _extreme_subgradient_step(a: np.ndarray):
    """Sub‑gradient of max_k (a*a)_k using a single active index."""
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    k_max = conv.argmax()
    rev = a[::-1]
    shift = k_max - (n - 1)        # shift of the one‑hot filter
    grad = np.zeros_like(a)
    if shift >= 0:
        # grad[shift:] = rev[:n-shift]
        grad[shift:] = rev[: n - shift]
    else:
        # shift < 0
        grad[: n + shift] = rev[-shift : ]
    grad *= 2.0
    return grad, cur_max

def _refine_sequence(seq: np.ndarray, deadline: float):
    """Projected sub‑gradient descent with a decreasing step size."""
    a = seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    # parameters
    init_step = 0.45               # aggressive first step
    while time.time() < deadline:
        it += 1
        grad, cur_max = _extreme_subgradient_step(a)
        step = init_step / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)
        # occasional jitter to leave flat plateaus
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] refinement it={it} score={best_val:.6f}")
    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a probability vector (step‑function heights) with a low
    self‑convolution ℓ∞ norm. The routine
        * draws many candidates from Beta, asymmetric Beta and Dirichlet families,
        * keeps the best few,
        * then refines them with a fast extreme‑subgradient descent.
    The returned list can be fed directly to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin
    # ----------------------------------------------------------------------
    # 0)  Start from a previously discovered solution if it exists
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq)                # enforce exact simplex
    except Exception:
        base_seq = None

    # ----------------------------------------------------------------------
    # Parameters for the exploratory stage
    lengths = [1024, 2048, 4096, 5000]  # longer lengths give finer shapes
    top_k = 5                          # keep the best K candidates for refinement
    top_candidates = []                # list of (score, vector)

    def _maybe_push(vec: np.ndarray):
        sc = _raw_score(vec)
        # keep list sorted ascending by score, size limited to top_k
        if len(top_candidates) < top_k:
            top_candidates.append((sc, vec.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif sc < top_candidates[-1][0]:
            top_candidates[-1] = (sc, vec.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ----------------------------------------------------------------------
    # 1)  Exhaustive space of simple families (uniform, arcsine, symmetric Beta)
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # a) reuse previous solution (interpolated)
        if base_seq is not None:
            cand = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, base_seq.size), base_seq)
            cand = _project_onto_simplex(cand)
            _maybe_push(cand)

        # b) Uniform baseline
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        _maybe_push(uniform)

        # c) Arcsine (α=β=0.5)
        arcsine = _beta_distribution(n, 0.5, 0.5)
        _maybe_push(arcsine)

        # d) Symmetric Beta grid (α from 0.04 to 6)
        alphas = np.concatenate((
            np.logspace(-1.4, -0.2, 12),   # ≈0.04 … 0.63
            np.linspace(0.6, 2.0, 15),
            np.linspace(2.0, 6.0, 8)
        ))
        for a in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a, a)
            _maybe_push(vec)

        # e) Random asymmetric Beta draws
        for _ in range(80):
            if time.time() >= deadline:
                break
            a = 10 ** np.random.uniform(-2, 0.8)   # 0.01 … 6.3
            b = 10 ** np.random.uniform(-2, 0.8)
            vec = _beta_distribution(n, a, b)
            _maybe_push(vec)

        # f) Random Dirichlet draws (different concentration)
        for _ in range(80):
            if time.time() >= deadline:
                break
            conc = 10 ** np.random.uniform(-2, 0.5)   # 0.01 … 3.2
            vec = _dirichlet_random(n, conc)
            _maybe_push(vec)

        # g) Random mixtures of 2–3 Beta components
        for _ in range(120):
            if time.time() >= deadline:
                break
            k = np.random.choice([2, 3])
            weights = np.random.rand(k)
            weights /= weights.sum()
            mix = np.zeros(n, dtype=np.float64)
            for i in range(k):
                a = 10 ** np.random.uniform(-2, 0.8)
                b = 10 ** np.random.uniform(-2, 0.8)
                comp = _beta_distribution(n, a, b)
                mix += weights[i] * comp
            mix = _project_onto_simplex(mix)
            _maybe_push(mix)

    # ----------------------------------------------------------------------
    # If the exploratory stage failed to produce any candidate, fall back to uniform.
    if not top_candidates:
        n = 3000
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        top_candidates.append((_raw_score(uniform), uniform))
        print("Fallback uniform used as initial candidate.")

    # Report best from exploratory stage
    best_score, best_seq = top_candidates[0]
    print(f"Best exploratory score: {best_score:.6f} (len={best_seq.shape[0]})")

    # ----------------------------------------------------------------------
    # 2)  Local refinement – allocate remaining time proportionally
    #    to the kept candidates (most time for the current best).
    remaining = deadline - time.time()
    if remaining <= 0:
        final_seq = best_seq
        final_score = evaluate_sequence(final_seq.tolist())
        print(f"Time exhausted, returning score: {final_score:.6f}")
        return final_seq.tolist()

    # Divide time: 60 % for the current best, the rest spread over the others.
    alloc = [0.6] + [0.4 / (len(top_candidates) - 1)] * (len(top_candidates) - 1) if len(top_candidates) > 1 else [1.0]
    for (frac, (sc, seq)) in zip(alloc, top_candidates):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + remaining * frac
        print(f"[{time.time():.1f}] Refining candidate (len={len(seq)}) with {sub_deadline - time.time():.1f}s budget")
        refined_seq, refined_score = _refine_sequence(seq, sub_deadline)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq
            print(f"New global best after refinement: {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Final check with the black‑box evaluator (mostly for logging)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()

"""
Candidate proposal using a rich random‑parameter search followed by a fast
extreme‑subgradient descent that respects the simplex constraint.
The search explores uniform, arcsine, symmetric/asymmetric Beta families,
random Dirichlet draws and mixtures of Beta components.
The best few candidates are then refined for the remaining time budget
with a projected sub‑gradient method that uses a single active convolution
index (an “extreme” subgradient).  Updates are kept cheap by normalising
instead of a full Euclidean projection.
The returned list can be supplied directly to the black‑box evaluator.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _simplex_normalize(v: np.ndarray) -> np.ndarray:
    """Fast non‑negative normalisation (projection onto the probability simplex)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v[:] = 1.0 / len(v)
    else:
        v /= s
    return v


def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a * a."""
    return _fft_self_convolve_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """2·n·max(a*a) – the quantity that the playground evaluator returns (up to scaling)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution on n points, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    tot = pdf.sum()
    if tot == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= tot
    return pdf


def _dirichlet_random(n: int, concentration: float) -> np.ndarray:
    """Symmetric Dirichlet draw (all concentration parameters equal)."""
    if concentration <= 0.0:
        concentration = 1e-6
    g = np.random.gamma(concentration, 1.0, size=n)
    s = g.sum()
    if s == 0.0:
        g[:] = 1.0 / n
    else:
        g /= s
    return g


def _extreme_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of  max_k (a*a)_k  using a *single* active index.
    Returns (grad, cur_max).  grad is non‑negative.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    k_max = conv.argmax()
    rev = a[::-1]
    shift = k_max - (n - 1)          # shift of the one‑hot filter
    grad = np.zeros_like(a)
    if shift >= 0:
        # right shift
        grad[shift:] = rev[: n - shift]
    else:
        # left shift (shift is negative)
        d = -shift
        grad[: n - d] = rev[d:]
    grad *= 2.0
    return grad, cur_max


def _refine_sequence(seq: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent with a decreasing step size.
    Uses the cheap simplex normalisation for speed.
    """
    a = seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    init_step = 0.45                 # relatively aggressive first step

    while time.time() < deadline:
        it += 1
        grad, cur_max = _extreme_subgradient_step(a)
        step = init_step / math.sqrt(it)
        a = a - step * grad
        a = _simplex_normalize(a)

        # occasional tiny random jitter to escape plateaus
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _simplex_normalize(a)

        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] refinement it={it}  score={best_val:.6f}")

    return best_a, best_val
