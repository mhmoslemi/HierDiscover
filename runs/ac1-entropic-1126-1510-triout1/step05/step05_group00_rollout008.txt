def _adam_refine(a0, deadline):
    a = a0.copy()
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iteration = 0
    lam0 = 5.0
    lam_max = 400.0
    lam_inc_iter = 100000  # after which lam reaches lam_max
    # set initial step size
    base_lr = 0.1
    while time.time() < deadline:
        iteration += 1
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # lambda schedule
        lam = lam0 + (lam_max - lam0) * min(1.0, iteration / lam_inc_iter)
        # compute softmax weights
        exp_vals = np.exp(lam * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient of smoothed max
        grad = 2.0 * _fft_convolve(a[::-1], w)[len(a)-1:len(a)-1+len(a)]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = base_lr / (np.sqrt(iteration) + 1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_onto_simplex(a, 1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] adam iter={iteration} new best {cur_score:.6f}")
            # optional early stop if low enough
            if best_score <= 1.503:
                break
    return best_seq, best_score

def _lp_improve(a, deadline):
    # solve LP to get g
    conv = _fft_self_convolve(a)
    M = conv.max()
    eps = 1e-12 * (abs(M) + 1.0)
    active = np.flatnonzero(conv >= M - eps)
    if active.size == 0:
        return a, _max_convolution_score(a)
    n = a.size
    # Build A_ub
    A_rows = []
    for k in active:
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # indices i from i_min to i_max inclusive
        inds = np.arange(i_min, i_max + 1)
        row[inds] = a[k - inds]
        A_rows.append(row)
    A_ub = np.vstack(A_rows) if A_rows else np.empty((0, n))
    b_ub = np.full(active.shape, M)
    c = -np.ones(n)
    # Solve LP using Highs
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    if not res.success:
        return a, _max_convolution_score(a)
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a, _max_convolution_score(a)
    g_norm = g / sum_g
    # Mix with a: try several mixing fractions
    best_seq = a
    best_score = _max_convolution_score(a)
    for t in [0.2, 0.5, 0.8]:
        new_seq = (1 - t) * a + t * g_norm
        new_seq = _project_onto_simplex(new_seq, 1.0)
        score = _max_convolution_score(new_seq)
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
    return best_seq, best_score

def _hill_climb(a, deadline):
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_seq = a.copy()
    cur_score = best_score
    iteration = 0
    max_transfer = 0.05
    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 20000.0))
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(transfer, cur_seq[i])
        if delta <= 0:
            continue
        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand, 1.0)
        cand_score = _max_convolution_score(cand)
        if cand_score < cur_score - 1e-12:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-14:
                best_score = cand_score
                best_seq = cand.copy()

        if iteration % 5000 == 0 and np.random.rand() < 0.1:
            perturb = np.random.randn(n) * 1e-4
            cur_seq = best_seq + perturb
            cur_seq = _project_onto_simplex(cur_seq, 1.0)
            cur_score = _max_convolution_score(cur_seq)
    return best_seq, best_score

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. **Exploratory sampling** –  evaluate several analytic families
   (arcsine, symmetric beta, triangular, exponential) for a variety of
   lengths.  Keep the best candidate.

2. **Smooth Adam optimisation** –  minimise a smooth approximation of the
   objective 2·n·max(a*a) using Adam on the simplex.
   The smoothed max is the log‑sum‑exp of the self‑convolution with a
   gradually increasing temperature (parameter λ).  After each update the
   vector is projected onto the simplex.

3. **Linear‑programming boost** –  given the current distribution f,
   solve a small LP that maximises the total mass of a distribution g
   under the constraints (f * g) ≤ max(f*f).  Mixing f with the normalised
   optimiser g typically reduces the supremum of f*f.

4. **Local hill‑climbing** –  finish with cheap random transfers of mass
   between two positions and occasional jitter to escape plateaux.

The whole procedure respects the supplied time‑budget (≈ 1000 s) and
always returns a list of non‑negative floats that can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(0.5, 0.5)) law, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p, p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle, normalised."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _adam_refine(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max of the self‑convolution.
    Returns the best sequence found and its (exact) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    iteration = 0
    base_lr = 0.15
    lam0 = 5.0          # start temperature
    lam_max = 400.0     # final temperature
    lam_ramp = 80000    # iterations to reach lam_max

    while time.time() < deadline:
        iteration += 1

        # self‑convolution and smooth max
        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # temperature schedule
        lam = lam0 + (lam_max - lam0) * min(1.0, iteration / lam_ramp)

        # soft‑max weights (numerically stable)
        w_raw = np.exp(lam * (conv - maxc))
        w = w_raw / w_raw.sum()

        # gradient of the smooth max
        grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (np.sqrt(iteration) + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a, 1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] adam iter={iteration} new best {cur_score:.6f}")
            if best_score <= 1.503:
                break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Solve a small LP to find a distribution g with larger total mass
    while keeping (a * g) ≤ max(a*a).  Mix the normalised g back into a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    eps = 1e-12 * (abs(M) + 1.0)
    active = np.flatnonzero(conv >= M - eps)

    if active.size == 0:
        return a, _max_convolution_score(a)

    # Build constraint matrix A_ub * g <= M
    rows = []
    for k in active:
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        row[idx] = a[k - idx]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M)

    c = -np.ones(n)  # maximise sum(g)  <=> minimise -sum(g)
    # solve with HiGHS (fast for these small problems)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
    except Exception:
        return a, _max_convolution_score(a)

    if not res.success:
        return a, _max_convolution_score(a)

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a, _max_convolution_score(a)

    g_norm = g / sum_g

    # try a few mixing fractions
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    for t in (0.2, 0.5, 0.8):
        cand = (1.0 - t) * a + t * g_norm
        cand = _project_onto_simplex(cand, 1.0)
        s = _max_convolution_score(cand)
        if s < best_score:
            best_score = s
            best_seq = cand.copy()
            print(f"[{time.time():.1f}] LP mix t={t:.1f} improved best {s:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Simple random‑transfer hill climbing to polish the solution."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    cur_seq = a.copy()
    cur_score = best_score
    iteration = 0
    max_transfer = 0.05

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 20000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, cur_seq[i])
        if delta <= 0:
            continue

        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand, 1.0)

        cand_score = _max_convolution_score(cand)

        if cand_score < cur_score - 1e-12:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-14:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] hillclimb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape flat plateaux
        if iteration % 5000 == 0 and np.random.rand() < 0.1:
            perturb = np.random.randn(n) * 1e-4
            cur_seq = best_seq + perturb
            cur_seq = _project_onto_simplex(cur_seq, 1.0)
            cur_score = _max_convolution_score(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a list of floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # tiny safety margin

    # ------------------------------------------------------------------
    # 1.  Determine a set of lengths to explore.
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    lengths = set()
    for f in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    for _ in range(5):
        n = np.random.randint(500, min(20000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 2.  Coarse analytic sampling.
    overall_best_seq = None
    overall_best_score = float("inf")

    beta_ps = [0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40,
               0.45, 0.50, 0.60, 0.80, 1.00, 1.20, 1.50, 2.00]

    for n in lengths:
        if time.time() >= deadline:
            break

        a = _arcsine_distribution(n)
        sc = _max_convolution_score(a)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} arcsine   score={sc:.6f}")

        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            sc = _max_convolution_score(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} beta(p={p}) score={sc:.6f}")

        a = _triangular_distribution(n)
        sc = _max_convolution_score(a)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} triangular score={sc:.6f}")

        a = _exponential_distribution(n, lam=0.005)
        sc = _max_convolution_score(a)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} exponential score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Use a previously stored construction if it exists.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # Fallback – should never happen.
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _max_convolution_score(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    print(f"Initial best length={len(overall_best_seq)} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smooth Adam refinement (major part of the budget).
    now = time.time()
    time_left = deadline - now
    refine_deadline = now + time_left * 0.70
    best_seq, best_score = _adam_refine(overall_best_seq.copy(), refine_deadline)

    # ------------------------------------------------------------------
    # 5.  LP boost (if time permits).
    if time.time() < deadline:
        lp_deadline = time.time() + (deadline - time.time()) * 0.45
        best_seq, best_score = _lp_improve(best_seq, lp_deadline)

    # ------------------------------------------------------------------
    # 6.  Local random‑transfer hill climbing for the remaining budget.
    if time.time() < deadline:
        hill_deadline = deadline
        best_seq, best_score = _hill_climb(best_seq, hill_deadline)

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # Return as plain Python list (evaluate_sequence expects list)
    return best_seq.tolist()
