def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:2*n - 1]
```

def gradient_exact(a, conv):
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    grad_full = _fft_convolve(a[::-1], w)
    grad = 2.0 * grad_full[n-1:n-1+n]
    # diagonal correction
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, conv

def _adam_optimize(a_start, n, max_iters, phase, deadline, temp_start=1.0, temp_factor=1.005, lr0=0.1):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    # initial score
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    t = 0
    temp = temp_start
    while t < max_iters and time.time() < deadline:
        t += 1
        if phase == 'l2':
            grad, conv = gradient_l2(a)
        elif phase == 'softmax':
            grad, conv = gradient_softmax(a, temp)
            temp = min(1e6, temp * temp_factor)
        elif phase == 'exact':
            grad, conv = gradient_exact(a)
        else:
            raise ValueError
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # optional print
        # early stop if near theoretical bound? maybe 1.4
        if best_score < 1.4:
            break
    return best_a, best_score

"""
Propose a low‑score step‑function.

Algorithm
---------

* **Data layout** – all sequences are kept on the probability simplex
  (non‑negative, sum = 1).  The evaluation score for a normalized sequence
  is simply 2 · n · max(conv), where conv = a * a (full linear convolution).

* **Two‑phase optimisation** – each starter candidate is first smoothed
  with an L2 loss that encourages the whole convolution to be flat.
  Afterwards a temperature‑controlled soft‑max surrogate of the
  ``max`` is optimised, finally a few exact‐max sub‑gradient steps are
  taken.  All updates are performed with Adam (adaptive moment
  estimation) and the iterate is projected back onto the simplex.

* **Search strategy** – for several lengths (≈ ½‒2 × a base length)
  a handful of analytically‑motivated starters (arcsine Beta, uniform,
  a few other Beta shapes, Dirichlet draws and mixtures) are generated.
  The best few (by cheap score) are refined by the optimiser.
  The global best is returned.

* **Time management** – the total budget (default 1000 s) is split
  between lengths and between starters; the function always returns
  the best sequence found before the deadline.

The code is fully self‑contained, uses only NumPy and the provided
``evaluate_sequence`` function, and respects the sandbox restrictions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Score for a normalised vector (sum=1) – 2·n·max(conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _gradient_l2(a: np.ndarray):
    """Gradient of the L2 flattening loss ‖a*a – avg‖²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    avg = 1.0 / (2 * n - 1)
    r = conv - avg
    grad_full = _fft_convolve(a[::-1], r)
    grad = 4.0 * grad_full[n - 1:n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, temp: float):
    """Gradient of a temperature‑softmax surrogate of max(conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # stable soft‑max
    exp_term = np.exp(temp * (conv - cur_max))
    w = exp_term / exp_term.sum()
    grad_full = _fft_convolve(a[::-1], w)
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
def _gradient_exact(a: np.ndarray):
    """Exact sub‑gradient of max(conv) (uniform over active indices)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    grad_full = _fft_convolve(a[::-1], w)
    grad = 2.0 * grad_full[n - 1:n - 1 + n]

    # diagonal correction for k = 2*i (off‑by‑one)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, conv


# ----------------------------------------------------------------------
def _adam_optimize(a_start: np.ndarray,
                   n: int,
                   max_iters: int,
                   phase: str,
                   deadline: float,
                   *,
                   temp_start: float = 1.0,
                   temp_factor: float = 1.004,
                   lr0: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex.

    Parameters
    ----------
    a_start : np.ndarray
        Starting probability vector (already normalised).
    n : int
        Length of the vector.
    max_iters : int
        Upper bound on number of iterations.
    phase : {'l2', 'softmax', 'exact'}
        Which surrogate gradient to use.
    deadline : float
        Absolute time after which the optimisation must stop.
    temp_start, temp_factor : float
        Temperature schedule (only used for ``softmax`` phase).
    lr0 : float
        Base learning‑rate (effective step ≈ lr0 / sqrt(t)).

    Returns
    -------
    best_a : np.ndarray
        Best vector seen (still normalised).
    best_score : float
        Cheap score of the best vector.
    """
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    temp = temp_start

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        if phase == 'l2':
            grad, conv = _gradient_l2(a)
        elif phase == 'softmax':
            grad, conv = _gradient_softmax(a, temp)
            temp = min(1e6, temp * temp_factor)
        elif phase == 'exact':
            grad, conv = _gradient_exact(a)
        else:
            raise ValueError('unknown phase')

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_onto_simplex(a, s=1.0)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report (lightweight)
            if t % 1000 == 0:
                print(f"[{time.time():.1f}] phase={phase} iter={t} score={best_score:.6f}")

        # jitter to help escape plateaus
        if t % 2500 == 0:
            a += np.random.randn(n) * 1e-8
            a = _project_onto_simplex(a, s=1.0)

        # an empirical early‑stop
        if best_score < 1.4:
            break

    return best_a, best_score
