"""
propose_candidate – a multi‑phase stochastic optimizer for the step‑function
problem.

Phase 0 – initialise with the known best solution (if any) and with a plain
arcsine distribution.

Phase 1 – exhaustive / random search over a few compact parametric families:
    * power‑law (inverse)  a_i ∝ (i·(n‑i))^{-p}
    * symmetric Beta       a_i ∝ x^{α‑1}(1‑x)^{α‑1}
    * centred Gaussian     a_i ∝ exp(‑½((i‑c)/σ)^2)
    * convex mixtures of arcsine with a uniform or a Gaussian component.
All candidates are projected onto the simplex and evaluated with an exact
FFT‑based convolution objective
        obj(a) = 2·n·max(a*a)          (∑a_i = 1).

Phase 2 – mirror‑descent / exponentiated‑gradient refinement of the
​M​ best candidates.  The algorithm minimises the smooth surrogate
S_α(a)= (1/α)·log∑_k exp(α·(a*a)_k) and gradually raises the temperature
α (α starts at 10, then α←α·1.05 each iteration).  The step‑size follows
η_t = η₀/√t (η₀=1.0).  After each update the vector is renormalised onto the
simplex.

Phase 3 – a directed greedy move‑based local improvement: the index k that
realises the current max‑convolution is identified, the corresponding
sub‑gradient g_i = 2·a_{k‑i} is built and a small fraction of mass is moved
from the index with largest g_i to the index with smallest g_i whenever this
decreases the exact objective.

The routine stops after the allotted budget (default 1000 s) and returns the
best sequence (as a plain Python list).  All helper functions are defined at
module level – no closures or lambdas – and only NumPy is required.

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½))."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """Symmetric (i·(n‑i))^{±p} distribution."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    vals = prod ** (-p) if invert else prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Centered (non‑normalised) Gaussian on the index set."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _mix_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    arcs = _arcsine_distribution(n)
    uni = _uniform_distribution(n)
    return _project_onto_simplex((1.0 - lam) * arcs + lam * uni, s=1.0)


# ----------------------------------------------------------------------
def _mix_arcsine_gaussian(n: int, lam: float, sigma: float) -> np.ndarray:
    arcs = _arcsine_distribution(n)
    gauss = _gaussian_distribution(n, sigma)
    return _project_onto_simplex((1.0 - lam) * arcs + lam * gauss, s=1.0)


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) under the simplex constraint (∑a = 1).
    a is assumed non‑negative; the function projects onto the simplex.
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of the (1/α)·log‑sum‑exp(α·b_k) surrogate w.r.t. a."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    grad = 2.0 * conv[start:start + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a*a) at the active index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g

    alpha = alpha_start
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] it={it} improved score={best_score:.6f}")

        # soft‑max gradient
        grad = _gradient_softmax(a, b, alpha, size_fft)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional jitter to avoid plateaus
        if it % 3000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _greedy_local_improvement(a_start: np.ndarray,
                              n: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Directed greedy hill‑climbing: repeatedly move a small fraction of mass
    from the index with the largest sub‑gradient (i.e. largest contribution to
    the current peak) to the index with the smallest sub‑gradient, accepting
    only improving moves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    while time.time() < deadline:
        b = _fft_self_convolve(a)
        k = int(np.argmax(b))
        # sub‑gradient of the active max‑entry
        g = _subgradient_max(a, k)

        # candidate source / target
        src = int(np.argmax(g))
        tgt = int(np.argmin(g))

        if a[src] <= 0.0 or src == tgt:
            break

        eps = a[src] * 0.05        # move up to 5 % of the source mass
        if eps <= 0.0:
            break

        a_candidate = a.copy()
        a_candidate[src] -= eps
        a_candidate[tgt] += eps

        if a_candidate[src] < 0.0:
            continue

        cand_score = _objective(a_candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] greedy improve score={best_score:.6f}")
        else:
            # try a smaller step
            eps = a[src] * 0.01
            if eps <= 0.0:
                break
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose a working length (between 5 000 and 20 000).  If a previous
    # solution is available we start from a similar size, otherwise we pick
    # a medium‑size vector.
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 8000
    n = int(min(max(5000, int(base_n * 1.0)), 20000))

    # ------------------------------------------------------------------
    # Global bookkeeping
    best_seq = None
    best_score = float("inf")
    top_candidates = []           # list of (score, seq)

    # ------------------------------------------------------------------
    # Initialise with a previous solution (if any) and with a pure arcsine
    # distribution.
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if seq0.size:
            seq0 = np.maximum(seq0, 0.0)
            seq0 = _project_onto_simplex(seq0, s=1.0)
            score0 = _objective(seq0)
            print(f"[init] previous solution score={score0:.6f}")
            best_seq, best_score = seq0.copy(), score0
            top_candidates.append((score0, seq0))
    except Exception:
        pass

    arcs = _arcsine_distribution(n)
    arcs_score = _objective(arcs)
    print(f"[init] arcsine (n={n}) score={arcs_score:.6f}")
    if arcs_score < best_score:
        best_seq, best_score = arcs.copy(), arcs_score
    top_candidates.append((arcs_score, arcs))

    # ------------------------------------------------------------------
    # Phase 1 – focussed grid + random parametric search
    # ------------------------------------------------------------------
    families = ("power", "beta", "gauss", "mix_uni", "mix_gau")
    grid_iters = 0
    random_iters = 0
    while time.time() < deadline - 250.0:    # reserve ~250 s for refinements
        # --- occasional fine grid over λ (mixture with uniform) ----------
        if grid_iters < 120:
            lam_grid = np.linspace(0.0, 0.2, 41)   # 0 % … 20 % uniform
            for lam in lam_grid:
                cand = _mix_arcsine_uniform(n, lam)
                sc = _objective(cand)
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[grid] mix_uni λ={lam:.3f} score={sc:.6f}")
                if len(top_candidates) < 8:
                    top_candidates.append((sc, cand))
                else:
                    # keep only the best few
                    worst = max(range(len(top_candidates)),
                               key=lambda i: top_candidates[i][0])
                    if sc < top_candidates[worst][0]:
                        top_candidates[worst] = (sc, cand)
            grid_iters += 1
            continue

        # ----------------------------------------------------------------
        # Random exploration of families
        fam = np.random.choice(families)
        if fam == "power":
            p = np.random.uniform(0.05, 0.9)               # inverse power‑law
            cand = _power_law_distribution(n, p, invert=True)
        elif fam == "beta":
            a = np.random.uniform(0.55, 2.0)               # symmetric Beta
            cand = _beta_distribution(n, a)
        elif fam == "gauss":
            sigma = np.random.uniform(n / 30.0, n / 2.0)   # centred Gaussian
            cand = _gaussian_distribution(n, sigma)
        elif fam == "mix_uni":
            lam = np.random.uniform(0.0, 0.3)
            cand = _mix_arcsine_uniform(n, lam)
        elif fam == "mix_gau":
            lam = np.random.uniform(0.0, 0.2)
            sigma = np.random.uniform(n / 30.0, n / 2.0)
            cand = _mix_arcsine_gaussian(n, lam, sigma)
        else:
            continue

        cand = _project_onto_simplex(cand, s=1.0)
        sc = _objective(cand)
        random_iters += 1
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[rand] fam={fam} it={random_iters} score={sc:.6f}")

        if len(top_candidates) < 10:
            top_candidates.append((sc, cand))
        else:
            worst = max(range(len(top_candidates)),
                       key=lambda i: top_candidates[i][0])
            if sc < top_candidates[worst][0]:
                top_candidates[worst] = (sc, cand)

    # ------------------------------------------------------------------
    # Phase 2 – exponential‑gradient refinement of the most promising candidates
    # ------------------------------------------------------------------
    top_candidates.sort(key=lambda x: x[0])
    for rank, (score, cand) in enumerate(top_candidates[:4]):
        if time.time() >= deadline - 120.0:
            break
        print(f"[refine] start candidate #{rank+1} (score={score:.6f})")
        cand_len = cand.shape[0]
        slice_deadline = time.time() + min(40.0, (deadline - time.time()) * 0.5)
        refined, _ = _optimize_exponential(cand, cand_len, slice_deadline,
                                           eta0=1.0, alpha_start=10.0,
                                           alpha_mul=1.05, alpha_max=2000.0)
        refined_score = _objective(refined)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined.copy()
            print(f"[refine] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – directed greedy local improvement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[local] start greedy refinement")
        refined, refined_score = _greedy_local_improvement(best_seq, n,
                                                          deadline)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined.copy()
            print(f"[local] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback in the extremely unlikely case that nothing was found
    if best_seq is None:
        best_seq = np.full(1000, 1.0 / 1000, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Finished – best score: {best_score:.6f} (length={best_seq.shape[0]})")
    # Return as plain Python list for the external harness
    return best_seq.tolist()
