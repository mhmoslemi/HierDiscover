def adam_projected_update(a, grad, m, v, t, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    a_new = np.maximum(a_new, 0.0)
    a_new = _project_onto_simplex(a_new)  # sum to 1.
    return a_new, m, v

"""
Propose a low‑score step function.

Algorithm
---------
* Work on the probability simplex (non‑negative entries, sum = 1).
* For several candidate lengths `n` generate a diverse set of initial
  distributions (Beta families, Dirichlet draws, uniform, previous
  construction).
* Each start is refined by a projected Adam optimizer:
    – The objective is `2·n·max(conv(a))` where `conv(a)=a*a`.
    – Early iterations use a smooth soft‑max surrogate of the max;
      the temperature is increased slowly so that the gradient
      approaches a true sub‑gradient.
    – After each Adam step the vector is projected back onto the simplex.
* The best sequence seen over all lengths and starts is kept.
* When the global budget is nearly exhausted a short extra refinement
  of the current best is performed.
* The final sequence (as a plain ``list[float]``) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution `a*a` using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1} normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray,
                      temperature: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of `max(conv)`.
    Returns a vector of length n (same length as a).
    """
    n = a.shape[0]
    cur_max = conv.max()
    # stable soft‑max
    exp_term = np.exp(temperature * (conv - cur_max))
    w = exp_term / exp_term.sum()
    # convolution of reversed a with the weight vector w,
    # then pick the central n entries
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max (uniform weight over all active indices).
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for self‑products (k = 2*i)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Score assuming sum(a)=1, i.e. 2·n·max(conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _generate_initials(n: int) -> list[np.ndarray]:
    """Return a list of diverse feasible start vectors on the simplex."""
    seqs: list[np.ndarray] = []

    # a few Beta shapes (including extreme ends)
    for al, be in [(0.05, 0.05), (0.05, 0.95), (0.95, 0.05),
                   (0.2, 0.2), (0.2, 0.8), (0.8, 0.2),
                   (0.5, 0.5), (1.0, 1.0)]:
        seqs.append(_beta_distribution(n, al, be))

    # Dirichlet draws with different concentration parameters
    for alpha in [0.1, 0.5, 1.0, 5.0]:
        seqs.append(np.random.dirichlet(np.full(n, alpha)))

    # uniform
    seqs.append(np.full(n, 1.0 / n, dtype=np.float64))

    # previous construction if available (interpolated if lengths differ)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.size != n:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            p = np.interp(x_new, x_old, prev)
        else:
            p = prev.copy()
        seqs.append(_project_onto_simplex(p))
    except Exception:
        pass

    # ensure simplex feasibility
    seqs = [_project_onto_simplex(s) for s in seqs]
    return seqs
