    best = a.copy()
    best_score = obj(a)
    while time < deadline:
        conv = fft_self_convolve(a)
        max_idx = conv.argmax()
        max_val = conv[max_idx]
        # subgradient: g_i = 2*a[max_idx - i] if 0 <= max_idx-i < n else 0
        g = np.zeros_like(a)

            j = max_idx - i
            if 0 <= j < n:
                g[i] = 2*a[j]
        # step size schedule
        step_size = 0.1 / np.sqrt(iter)

""" 
Proposes a non‑negative step function (probability vector) that minimises 
evaluate_sequence. The algorithm works as follows:

1. Choose a sequence length (default 12000, bounded between 2000 and 15000).
2. Exhaustive parametric search over several hand‑crafted families:
   - uniform,
   - discrete arcsine (Beta(½,½)),
   - symmetric Beta(α,α) for a dense grid of α,
   - mixtures arcsine + uniform,
   - arcsine with extra spikes at the boundaries,
   - a few random Dirichlet draws,
   - the previously discovered solution if available.
   The best few candidates are retained.
3. Starting from each retained candidate a projected Adam optimiser is run.
   The objective 2·n·max(a*a) is approximated by a soft‑max; the temperature
   (α) is increased exponentially up to a very large value so that the
   optimisation eventually minimises the true max‑convolution.
4. The best Adam result is polished by a cheap stochastic hill‑climbing
   (mass‑transfer) procedure.
5. The final sequence is returned as a plain list of floats.
"""

import time
import math
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = v.astype(np.float64, copy=False)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) self‑convolution a*a via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Return softmax weights for vector b with temperature 1/α."""
    if alpha > 2000.0:
        # treat each entry that is numerically maximal as active
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
        return w
    shift = b - b.max()
    w = np.exp(alpha * shift)
    total = w.sum()
    if total == 0.0:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        return w
    return w / total

def _gradient_via_softmax(a: np.ndarray,
                          conv: np.ndarray,
                          alpha: float,
                          size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·conv_k) w.r.t a.
    For large α this becomes a sub‑gradient of max(conv).
    """
    n = a.shape[0]
    w = _softmax_weights(conv, alpha)

    # Convolution w * reversed(a)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_w = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv_w[start:start + n]
    return g

def _beta_symmetric(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) pdf evaluated on n bins (mid‑points)."""
    a = max(a, 0.05)                # avoid division by zero / infinite pdf
    x = (np.arange(n) + 0.5) / n    # mid‑point of each bin
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    # Normalise to sum 1
    s = pdf.sum()
    if s == 0.0 or not np.isfinite(s):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= s
    return pdf

def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _generate_candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Produce a diverse list of probability vectors of length n."""
    candidates = []

    # Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Arcsine
    arcsine = _arcsine_distribution(n)
    candidates.append(arcsine)

    # Symmetric Beta on a fine grid
    alphas = np.concatenate([np.arange(0.06, 0.5, 0.02), np.arange(0.5, 0.92, 0.04)])
    for a in alphas:
        candidates.append(_beta_symmetric(n, a))

    # Mixtures arcsine + uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for w in np.linspace(0.0, 0.6, 13):
        mix = (1.0 - w) * arcsine + w * uniform
        candidates.append(mix)

    # Edge spikes added to arcsine
    for spike_frac in np.linspace(0.05, 0.45, 9):
        shape = arcsine.copy()
        shape[0] += spike_frac / 2.0
        shape[-1] += spike_frac / 2.0
        shape = _project_onto_simplex(shape, 1.0)
        candidates.append(shape)

    # Random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(12):
        candidates.append(rng.dirichlet(np.ones(n)))

    # Previous solution if supplied
    if prev_seq is not None and prev_seq.shape[0] == n:
        candidates.append(prev_seq.copy())

    # Ensure all lie on the simplex
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    # Remove near‑duplicates (optional, but cheap)
    uniq = []
    for c in final:
        # compare with existing vectors by max absolute diff
        if not any(np.max(np.abs(c - u)) < 1e-12 for u in uniq):
            uniq.append(c)
    return uniq

def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex.
    Returns the best vector seen and its raw objective 2·n·max(conv).
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.07
    alpha_max = 200_000.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # ≥ (2n‑1)+(n)‑1

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] n={n} it={it} score={best_score:.6f}")

        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_onto_simplex(a, 1.0)

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score

def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a random fraction of the mass from
    one index to another and keep the move if it improves the objective.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue
        # transfer up to 30 % of the mass at i to j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Entry point required by the grader.
    Returns a list of non‑negative heights (probability vector) that
    minimises ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the final conversion / printing
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a length.  Prefer the length of a previously discovered
    # solution if available, otherwise pick a medium‑large value.
    try:
        base_n = int(len(height_sequence_1))  # type: ignore[name-defined]
    except Exception:
        base_n = 12000

    # Clamp to a sensible interval (2 k … 15 k)
    n = max(2000, min(base_n, 15000))

    # ------------------------------------------------------------------
    # Phase 0 – broad parametric sweep
    prev_seq = None
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        pass

    shape_deadline = time.time() + min(45.0, (deadline - time.time()) * 0.20)

    all_cands = _generate_candidate_shapes(n, prev_seq)
    best_seq = None
    best_score = float("inf")
    evaluated = 0

    for cand in all_cands:
        if time.time() >= shape_deadline:
            break
        scr = evaluate_sequence(cand.tolist())
        evaluated += 1
        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
            print(f"[shape] New best score {best_score:.6f} (evaluated={evaluated})")
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] Using uniform after shape stage")

    # Keep the top few candidates for Adam (including the best one)
    # Re‑evaluate all candidates (if not already) to get exact scores and
    # sort.  The list ``all_cands`` is short (<~100), so this is cheap.
    scored = []
    for cand in all_cands:
        scr = evaluate_sequence(cand.tolist())
        scored.append((scr, cand))
    scored.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [cand for (_, cand) in scored[:top_k]]

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on the retained candidates
    remaining = deadline - time.time()
    if remaining <= 0:
        remaining = 0.0
    per_cand = remaining * 0.55 / max(1, len(top_candidates))
    best_overall = best_seq.copy()
    best_overall_score = best_score

    for idx, cand in enumerate(top_candidates):
        cand_deadline = time.time() + per_cand
        seq_opt, raw_score = _adam_optimize(cand, n, cand_deadline)
        # Convert raw_score (2·n·max) to the same scale as evaluate_sequence
        scr = raw_score
        if scr < best_overall_score:
            best_overall_score = scr
            best_overall = seq_opt.copy()
            print(f"[adam] Candidate {idx} improved best to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final hill‑climbing polishing
    hill_deadline = deadline
    final_seq, final_raw = _hill_climb(best_overall, n, hill_deadline)

    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final best score (evaluate_sequence) = {final_score:.6f} (length={len(final_seq)})")
    return final_seq.tolist()

"""
Proposes a non‑negative step function (probability vector) that minimises 
``evaluate_sequence``.  The algorithm consists of three phases:

1. **Parametric sweep** – generate a diverse collection of hand‑crafted
   shapes (uniform, discrete arcsine, symmetric Beta(α,α) on a fine grid,
   mixtures of arcsine+uniform, arcsine with extra spikes at the
   boundaries, a few random Dirichlet draws and the previous best
   solution if available).  The best few candidates are retained.

2. **Projected Adam optimisation** – starting from each retained
   candidate run a projected Adam optimiser on the simplex.  The objective
   ``2·n·max(conv)`` is approximated by a soft‑max; the temperature is
   increased exponentially up to a very large value so that the optimiser
   eventually minimises the true max‑convolution.

3. **Stochastic hill‑climbing** – the best Adam result is polished by a
   cheap hill‑climb that repeatedly moves a random fraction of mass from
   one index to another and keeps the move if it improves the score.

All work is bounded by the supplied time budget (default 1000 s).  The
final sequence is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = v.astype(np.float64, copy=False)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution ``a*a`` (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights for a vector ``b`` with temperature 1/α.
    For huge α the function returns a uniform distribution over the
    numerically maximal entries (sub‑gradient of the true max).
    """
    if alpha > 2000.0:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
        return w
    shift = b - b.max()
    w = np.exp(alpha * shift)
    total = w.sum()
    if total == 0.0:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        return w
    return w / total


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                          conv: np.ndarray,
                          alpha: float,
                          size_fft: int) -> np.ndarray:
    """
    Gradient of the smoothed objective (1/α)·log Σ exp(α·conv_k) w.r.t. ``a``.
    """
    n = a.shape[0]
    w = _softmax_weights(conv, alpha)

    # convolution of w with the reversed ``a``
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_w = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv_w[start:start + n]      # factor 2 from derivative of a*a
    return g


# ----------------------------------------------------------------------
def _beta_symmetric(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(α,α) PDF on ``n`` bins (mid‑points)."""
    a = max(a, 0.05)                     # avoid singularities
    x = (np.arange(n) + 0.5) / n         # mid‑point of each bin
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    s = pdf.sum()
    if s == 0.0 or not np.isfinite(s):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= s
    return pdf


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on ``n`` points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _generate_candidate_shapes(n: int,
                               prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Return a modest list of diverse probability vectors of length ``n``."""
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) discrete arcsine (Beta(½,½))
    arcsine = _arcsine_distribution(n)
    candidates.append(arcsine)

    # 3) symmetric Beta(α,α) on a fine grid
    alphas = np.concatenate([np.arange(0.06, 0.5, 0.02),
                             np.arange(0.5, 0.92, 0.04)])
    for a in alphas:
        candidates.append(_beta_symmetric(n, a))

    # 4) mixtures arcsine + uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for w in np.linspace(0.0, 0.6, 13):
        mix = (1.0 - w) * arcsine + w * uniform
        candidates.append(mix)

    # 5) arcsine with extra spikes at the two ends
    for spike_frac in np.linspace(0.05, 0.45, 9):
        shape = arcsine.copy()
        shape[0] += spike_frac / 2.0
        shape[-1] += spike_frac / 2.0
        shape = _project_onto_simplex(shape, 1.0)
        candidates.append(shape)

    # 6) a few random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(12):
        candidates.append(rng.dirichlet(np.ones(n)))

    # 7) previous solution if we have one
    if prev_seq is not None and prev_seq.shape[0] == n:
        candidates.append(prev_seq.copy())

    # ensure exact simplex compliance and prune near‑duplicates
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    uniq = []
    for c in final:
        if not any(np.max(np.abs(c - u)) < 1e-12 for u in uniq):
            uniq.append(c)
    return uniq


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex.
    Returns the best vector observed and its raw objective ``2·n·max(conv)``.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.07
    alpha_max = 200_000.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # ≥ (2n‑1)+(n)‑1

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] n={n} it={it} score={best_score:.6f}")

        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_onto_simplex(a, 1.0)

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climb: move a random fraction of mass from one
    index to another and keep the move if it improves the raw objective.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue
        frac = rng.random() * 0.30                # up to 30 % transfer
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # tiny jitter every few thousand steps
        if it % 4000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score
