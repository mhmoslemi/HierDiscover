def adam_optimize_start(a0, n, max_iters, initial_alpha, alpha_factor, max_alpha):
    # a0: initial probability vector
    z = np.log(a0 + 1e-15)  # avoid log(0)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_a = a0.copy()
    best_score = cheap_score(a0)
    alpha = initial_alpha
    for t in range(1, max_iters+1):
        a = np.exp(z - np.max(z))  # to avoid overflow
        a = a / a.sum()
        conv = _fft_self_convolve(a)
        # compute softmax weights w
        if alpha < max_alpha:
            exp_term = np.exp(alpha * (conv - conv.max()))
            w = exp_term / exp_term.sum()
            alpha = min(max_alpha, alpha * alpha_factor)
        else:
            # use exact subgradient: uniform on active set
            cur_max = conv.max()
            eps = 1e-12*(abs(cur_max)+1.0)
            active = np.flatnonzero(conv >= cur_max - eps)
            w = np.zeros_like(conv)
            w[active] = 1.0/active.size
        # gradient wrt a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # gradient wrt z
        grad_z = a * (grad_a - np.dot(grad_a, a))
        # Adam updates
        m = beta1*m + (1-beta1) * grad_z
        v = beta2*v + (1-beta2) * (grad_z**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.01 / (np.sqrt(v_hat) + epsilon)
        z += - step * m_hat
        # evaluate every few steps
        cur_score = cheap_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score
```

def _adam_optimize(a0, n, time_budget):
    max_iters = 20000 
    start = time.time()
    # initial logits
    # Use log of a0 + small epsilon
    z = np.log(np.maximum(a0, 1e-16))
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1   # base learning rate
    best_a = a0.copy()
    best_score = cheap_score(best_a)
    # temperature for smooth max
    alpha = 1.0
    alpha_factor = 1.01
    alpha_max = 5000.0
    for t in range(1, max_iters+1):
        # time check
        if time.time() - start > time_budget:
            break
        # compute a from z (softmax)
        z_shift = z - np.max(z)  # for numeric stability
        a = np.exp(z_shift)
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum
        # compute convolution
        conv = _fft_self_convolve(a)
        # compute gradient using softmax weights
        if alpha < alpha_max:
            max_conv = conv.max()
            exp_term = np.exp(alpha * (conv - max_conv))
            w = exp_term / exp_term.sum()
            alpha = min(alpha_max, alpha * alpha_factor)
        else:
            # exact gradient via active set
            cur_max = conv.max()
            eps_thr = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(conv >= cur_max - eps_thr)
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size
        # gradient wrt a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
        # gradient wrt z
        grad_z = a * (grad_a - np.dot(grad_a, a))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        z = z - step
        # evaluate cheap score occasionally
        if t % 50 == 0:
            cur_score = cheap_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score

            continue

        # we ensure cur[i] > 0
        if cur[i] <= 0:
            continue
        delta = np.random.rand() * min(cur[i], 0.01)  # move up to 1% of i's mass
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # compute score
        new_score = cheap_score(new)
        # accept if better or with prob exp(-(new_score - cur_score)/T)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
        else:
            # temperature schedule; linear decay
            elapsed = time.time() - start
            fraction = elapsed / time_budget
            T = T0 * (1 - fraction)  # cooldown
            if T > 0 and np.random.rand() < np.exp(-(new_score - cur_score) / T):
                cur = new
                cur_score = new_score
        # update best
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
        # occasional reheat: small chance to increase T temporarily? maybe not needed

"""
Propose a low‑score step‑function.

Algorithm Overview
------------------
1. **Length selection** – we start from the length of the previously
   discovered construction (``height_sequence_1``) and explore a few
   scaled variants (0.75×, 1.0×, 1.5×).  Keeping the length moderate
   (≈ 5 k – 12 k) balances search speed and expressive power.

2. **Initial candidates** – for each length we generate a small library
   of analytically motivated probability vectors:
        * arcsine law (Beta(½,½)) and several symmetric / skewed Beta
          distributions,
        * uniform and Dirichlet‑random vectors,
        * the best construction found so far (interpolated to the new
          length).
   The library is cheap‑scored with a FFT based self‑convolution and the
   best few are kept.

3. **Mirror‑descent on the simplex (Adam on logits)** – every kept
   candidate is optimised by *mirror descent* on the probability simplex.
   We parameterise the vector as a soft‑max of free logits ``z``:
   ``a_i = exp(z_i) / sum_j exp(z_j)``.  An Adam optimiser is applied to
   the logits using the gradient of a *smooth* max of the convolution
   (soft‑max with a temperature that grows exponentially).  When the
   temperature reaches a large value the surrogate becomes the true max
   and the exact sub‑gradient is used.

   The optimiser runs until the per‑run time‑budget expires (or a
   predefined iteration limit).  The best iterate (according to the
   cheap score ``2·n·max(a*a)``) is retained.

4. **Local stochastic refinement (simulated annealing)** – the best
   vector for the current length is further refined by a simple
   simulated‑annealing scheme that moves a tiny amount of mass between
   two random coordinates.  The cheap score is used for acceptance.

5. **Evaluation** – the globally best vector discovered across all
   lengths is finally evaluated with the official ``evaluate_sequence``
   function.  If a score ≤ 1.503 is reached the search stops early.

All heavy operations (convolutions) are performed with real FFTs, and
the algorithm respects the overall time budget (default 1000 s).  The
implementation uses only NumPy and the standard library.
"""

import time
import numpy as np

# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / s
    return vals


def cheap_score(a: np.ndarray) -> float:
    """Score for a probability vector (sum = 1) – 2·n·max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Compute soft‑max weighting of `conv` with temperature α."""
    # temperature is the exponent multiplier (α).  Larger → sharper.
    cur_max = conv.max()
    exp_term = np.exp(temperature * (conv - cur_max))
    return exp_term / exp_term.sum()


def _exact_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights over all (approximately) active maximal entries."""
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    w = np.zeros_like(conv)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)
    w[active] = 1.0 / active.size
    return w


def _gradient_wrt_a(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k weights_k · (a*a)[k] with respect to `a`.
    Returns a vector of length n.
    """
    n = a.shape[0]
    conv_rev = _fft_convolve(a[::-1], weights)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    # diagonal correction for terms where k = 2*i (i.e. a_i·a_i)
    # Those terms have been counted only once in the convolution above,
    # but the true derivative of a_i*a_i is 2*a_i.  The convolution
    # supplies a_i once, so we subtract a_i to get the missing part.
    active = np.where(weights > 0)[0]
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad


def _grad_z_from_a(a: np.ndarray, grad_a: np.ndarray) -> np.ndarray:
    """
    Chain rule: gradient w.r.t. logits z (soft‑max parameters).
    a = softmax(z); grad_z = a * (grad_a – ⟨grad_a, a⟩).
    """
    dot = np.dot(grad_a, a)
    return a * (grad_a - dot)

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    time_budget: float,
    max_iters: int = 20000,
    lr: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using Adam on the logits.
    Returns the best probability vector found and its cheap score.
    """
    n = a0.shape[0]
    # initialise logits (z) from probabilities
    z = np.log(np.maximum(a0, 1e-16))
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = a0.copy()
    best_sc = cheap_score(best_a)

    # temperature schedule for the smooth‑max surrogate
    temp = 1.0
    temp_factor = 1.015
    temp_max = 5000.0

    start = time.time()
    for t in range(1, max_iters + 1):
        if time.time() - start > time_budget:
            break

        # ---- probability vector from logits (stable soft‑max) ----
        z_shift = z - np.max(z)
        a = np.exp(z_shift)
        a_sum = a.sum()
        if a_sum == 0.0 or np.isnan(a_sum):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # ---- convolution and weighting ----
        conv = _fft_self_convolve(a)
        if temp < temp_max:
            w = _softmax_weights(conv, temp)
            temp = min(temp_max, temp * temp_factor)
        else:
            w = _exact_weights(conv)

        # ---- gradient w.r.t. a and then w.r.t. logits ----
        grad_a = _gradient_wrt_a(a, w)
        grad_z = _grad_z_from_a(a, grad_a)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        z = z - step

        # ---- occasional jitter to escape plateaus ----
        if t % 2500 == 0:
            z += np.random.randn(n) * 1e-4

        # ---- track best cheap score ----
        if t % 30 == 0:
            cur_sc = cheap_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


def _simulated_annealing(
    a_start: np.ndarray,
    time_budget: float,
    max_moves: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Very simple SA that moves a small fraction of mass between two
    randomly chosen indices.  Works directly on the probability simplex.
    """
    n = a_start.shape[0]
    cur = a_start.copy()
    cur_sc = cheap_score(cur)
    best = cur.copy()
    best_sc = cur_sc

    start = time.time()
    # linear temperature schedule, starting a few percent of the score range
    T0 = 0.02 * cur_sc
    for step in range(1, max_moves + 1):
        if time.time() - start > time_budget:
            break
        # propose a move
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] <= 0.0:
            continue
        # move at most 1 % of the source mass
        delta = np.random.rand() * min(cur[i], 0.01)
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_sc = cheap_score(new)

        if new_sc < cur_sc:
            cur, cur_sc = new, new_sc
        else:
            # cooling
            elapsed = time.time() - start
            frac = elapsed / time_budget
            T = max(0.0, T0 * (1.0 - frac))
            if T > 0.0 and np.random.rand() < np.exp(-(new_sc - cur_sc) / T):
                cur, cur_sc = new, new_sc

        if cur_sc < best_sc:
            best, best_sc = cur.copy(), cur_sc

        # occasional large perturbation to shake out of deep valleys
        if step % 50000 == 0:
            idx = np.random.choice(n, size=2, replace=False)
            cur[idx] = np.maximum(cur[idx], 1e-12)
            cur = _project_onto_simplex(cur, s=1.0)

    return best, best_sc

# Candidate generation utilities
# ----------------------------------------------------------------------


def _initial_vector(init_kind: str, n: int) -> np.ndarray:
    """
    Produce an initial feasible probability vector.
    """
    if init_kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init_kind == "random":
        a = np.random.rand(n)
        a = a / a.sum()
    elif init_kind == "arcsine":
        a = _beta_distribution(n, 0.5, 0.5)
    elif init_kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                # linear interpolation to required length
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _beta_distribution(n, 0.5, 0.5)
    else:
        a = _beta_distribution(n, 0.5, 0.5)

    return _project_onto_simplex(a, s=1.0)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the time budget and
    returns the best sequence (as a Python list) found so far.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # tiny safety margin

    # ------------------------------------------------------------------
    # 0. Determine a base length (use the previously discovered one if any)
    # ------------------------------------------------------------------
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # explore a modest set of lengths; keep them odd/even doesn't matter
    lengths = sorted(
        set(
            max(500, int(base_n * f))
            for f in (0.75, 1.0, 1.5)
        )
    )
    if not lengths:
        lengths = [8000]

    # Global best across all lengths
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # Main loop over selected lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length n={n}")

        # --------------------------------------------------------------
        # 1. Build a small pool of analytically motivated candidates
        # --------------------------------------------------------------
        beta_shapes = [
            (0.5, 0.5),   # arcsine
            (0.4, 0.4),
            (0.3, 0.3),
            (0.2, 0.2),
            (0.75, 0.75),
            (0.2, 0.8),   # skewed
            (0.8, 0.2),
        ]
        cand_pool = [_beta_distribution(n, a, b) for a, b in beta_shapes]
        cand_pool.append(_initial_vector("uniform", n))
        cand_pool.append(_initial_vector("random", n))

        # add the up‑sampled global best if it already exists
        if overall_best_seq is not None:
            old = overall_best_seq
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, old)
            cand_pool.append(_project_onto_simplex(interp, s=1.0))

        # cheap evaluation & keep the top few starters
        scored = [(cheap_score(c), c) for c in cand_pool]
        scored.sort(key=lambda x: x[0])
        starters = [c.copy() for _, c in scored[:min(5, len(scored))]]

        # --------------------------------------------------------------
        # 2. Refine each starter with Adam‑mirror descent
        # --------------------------------------------------------------
        for idx, starter in enumerate(starters):
            if time.time() >= deadline:
                break
            # allocate a per‑starter slice of the remaining time
            remaining = deadline - time.time()
            # keep at most 120 s per starter (or less if we are low on time)
            slice_budget = min(remaining * 0.4, 120.0)
            if slice_budget < 5.0:
                continue

            best_local_seq, best_local_score = _adam_optimize(
                starter, time_budget=slice_budget
            )

            # evaluate with the official black‑box (once)
            try:
                exact = evaluate_sequence(best_local_seq.tolist())
            except Exception:
                exact = best_local_score

            if exact < overall_best_score:
                overall_best_score = exact
                overall_best_seq = best_local_seq.copy()
                print(
                    f"[{time.time():.1f}] New global best length={n} starter={idx} "
                    f"score={overall_best_score:.6f}"
                )

            # early termination if we have already reached the target
            if overall_best_score <= 1.503:
                print("Target achieved – terminating search.")
                return overall_best_seq.tolist()

        # --------------------------------------------------------------
        # 3. Optional stochastic refinement on the current best of this length
        # --------------------------------------------------------------
        if overall_best_seq is not None and len(overall_best_seq) == n:
            remaining = deadline - time.time()
            if remaining > 5.0:
                sa_seq, sa_score = _simulated_annealing(
                    overall_best_seq, time_budget=remaining * 0.5
                )
                try:
                    exact_sa = evaluate_sequence(sa_seq.tolist())
                except Exception:
                    exact_sa = sa_score
                if exact_sa < overall_best_score:
                    overall_best_score = exact_sa
                    overall_best_seq = sa_seq.copy()
                    print(
                        f"[{time.time():.1f}] Post‑SA improvement: score={overall_best_score:.6f}"
                    )
                if overall_best_score <= 1.503:
                    return overall_best_seq.tolist()

        # --------------------------------------------------------------
        # Global timeout check
        # --------------------------------------------------------------
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n_fallback = 1000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(
        f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
