def generate_candidates(n, prev_seq=None):
    candidates = []
    # uniform
    candidates.append(np.full(n, 1.0/n))
    # arcsine
    candidates.append(arcsine_distribution(n))
    # Beta a in grid
    for a in np.linspace(0.05, 1.4, 15):
        candidates.append(beta_distribution(n, a))
    # mixture arcsine+uniform
    arcsine = arcsine_distribution(n)
    uniform = np.full(n, 1.0/n)
    for w in np.linspace(0.0, 0.8, 9):
        mix = (1 - w) * arcsine + w * uniform
        candidates.append(mix)
    # power law shapes
    for alpha in [0.2, 0.3, 0.4, 0.5]:
        pl = powerlaw_distribution(n, alpha)
        candidates.append(pl)
    # spikes additions: start from arcsine add mass at edges
    for spike in [0.1, 0.2, 0.3]:
        mix = arcsine_distribution(n).copy()
        mix[0] += spike/2
        mix[-1] += spike/2
        mix = project_onto_simplex(mix, 1.0)
        candidates.append(mix)
    # previous best
    if prev_seq is not None and prev_seq.shape[0] == n:
        candidates.append(prev_seq.copy())
    # random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(5):
        rnd = rng.dirichlet(np.ones(n))
        candidates.append(rnd)
    # project all onto simplex to ensure sum=1
    final = [project_onto_simplex(c, 1.0) for c in candidates]
    return final

def hill_climb(start_a, n, deadline):
    best = start_a.copy()
    best_score = metric_from_vector(best)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = metric_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        it += 1
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = project_onto_simplex(jitter, 1.0)
            jitter_score = metric_from_vector(jitter)
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score
    return best, best_score

"""
propose_candidate
-----------------

Search for a non‑negative step function (probability vector) that minimizes the
provided ``evaluate_sequence`` metric.

Algorithm
~~~~~~~~~
1. **Parametric sweep** – generate a modest collection of hand‑crafted families
   (uniform, arcsine, symmetric Beta(a,a), power‑law 1/((i+½)(n‑i‑½))^α,
   mixtures with uniform or spikes).  Evaluate each candidate with a fast FFT
   based convolution and keep the best one.

2. **Projected sub‑gradient descent** – starting from the best candidate,
   repeatedly compute the exact sub‑gradient of the max‑convolution
   (weights are the active indices where the convolution attains its maximum),
   take a small step and project back onto the probability simplex.
   The step size decays geometrically.

3. **Stochastic hill‑climbing** – final polishing by random mass swaps and
   occasional jitter, always keeping improvements.

All work is time‑boxed by the ``budget_s`` argument (default 1000 s).  The
returned list of floats is guaranteed to be non‑negative; the metric is
scale‑invariant, so the vectors are normalised to sum to 1 during the search.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _metric_from_vector(a: np.ndarray) -> float:
    """Return the evaluate_sequence value for a vector with sum≈1."""
    n = a.shape[0]
    max_conv = _fft_self_convolve(a).max()
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) on n mid‑points, normalised to sum 1."""
    a = max(a, 1e-8)                     # avoid exact zero
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    s = pdf.sum()
    if s == 0 or not np.isfinite(s):
        pdf[:] = 1.0 / n
    else:
        pdf /= s
    return pdf


# ----------------------------------------------------------------------
def _powerlaw_distribution(n: int, alpha: float) -> np.ndarray:
    """Shape ∝ ((i+½)(n-i‑½))^{-α}, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest collection of probability vectors."""
    cand = []

    # 1) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) arcsine
    cand.append(_arcsine_distribution(n))

    # 3) symmetric Beta with a fine grid (including a<0.5)
    for a in np.linspace(0.05, 1.4, 15):
        cand.append(_beta_distribution(n, a))

    # 4) power‑law family
    for alph in (0.2, 0.3, 0.4, 0.5):
        cand.append(_powerlaw_distribution(n, alph))

    # 5) mixtures arcsine + uniform (heavy interior part)
    arcsine = _arcsine_distribution(n)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for w in np.linspace(0.0, 0.8, 9):
        mix = (1.0 - w) * arcsine + w * uniform
        cand.append(mix)

    # 6) arcsine with extra spikes at the borders
    for spike in (0.10, 0.20, 0.30):
        mix = _arcsine_distribution(n).copy()
        mix[0] += spike / 2
        mix[-1] += spike / 2
        cand.append(mix)

    # 7) previous best (if supplied and compatible length)
    if prev_seq is not None and prev_seq.shape[0] == n:
        cand.append(prev_seq.copy())

    # 8) a handful of random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n)))

    # final projection onto the simplex (guard against round‑off)
    final = [_project_onto_simplex(c, 1.0) for c in cand]
    return final


# ----------------------------------------------------------------------
def _subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200000,
    init_step: float = 0.5,
    step_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the simplex to minimise 2·n·max(a*a).
    Uses the exact active‑set sub‑gradient (mask of max‑convolution entries).
    """
    a = init_a.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _metric_from_vector(best)

    # pre‑compute FFT size for gradient convolutions
    size_fft = 1 << ((3 * n - 2).bit_length())          # ≥ (2n‑1)+(n)‑1

    step = init_step
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # current convolution and maximal value
        conv = _fft_self_convolve(a)
        max_val = conv.max()

        # active set mask (indices attaining the maximum up to small tolerance)
        tol = max_val * 1e-12
        mask = (conv >= max_val - tol).astype(np.float64)
        active = mask.sum()
        if active == 0:
            break
        mask /= active                     # normalise to obtain a convex weights vector

        # gradient = 2 * (mask * a_rev) restricted to the original coordinates
        a_rev = a[::-1]
        W = np.fft.rfft(mask, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        grad_full = np.fft.irfft(W * A_rev, n=size_fft)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # projected sub‑gradient step
        a = a - step * grad
        a = _project_onto_simplex(a, 1.0)

        # step size decay
        step *= step_decay

        # occasional evaluation of the objective
        if it % 50 == 0:
            cur_score = _metric_from_vector(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                # Debug output
                print(f"[{time.time():.2f}] subgrad it={it} best={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    start_a: np.ndarray, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: relocate a random fraction of mass
    from one index to another, keeping moves that improve the score.
    """
    best = start_a.copy()
    best_score = _metric_from_vector(best)
    n = best.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _metric_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # tiny jitter every few thousand iterations to escape plateaus
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = _metric_from_vector(jitter)
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score
