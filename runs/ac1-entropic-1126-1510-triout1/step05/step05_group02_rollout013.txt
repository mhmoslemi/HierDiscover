Proposes a step‑function by (i) exhaustive Beta‑family search, (ii) many
random Dirichlet restarts, and (iii) a smoothed‑max projected Adam optimiser.
The optimiser works directly on the simplex and uses a soft‑max
approximation of the ``max`` operator, gradually sharpening the
approximation (temperature annealing).  This typically finds a
distribution with a smaller ``evaluate_sequence`` value than the previous
baseline.
"""

# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1} using midpoint rule, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow/underflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()

    B = np.fft.rfft(b, n=size)

def _raw_score(a: np.ndarray) -> float:
    """2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    conv = _fft_convolve(a, a)
    return 2.0 * n * conv.max()

# --------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max approximation   g(a) = (1/lam) log Σ exp(lam·c_k)
    where c = a*a (full convolution). Returns (grad, g_value).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    # convolution a*a
    A = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

    # soft‑max weights
    scaled = lam * conv_full
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)          # stable
    w = exp_vals / exp_vals.sum()                  # shape (2n‑1,)

    # gradient = 2 * (a_rev ⊛ w)  (central part of length n)
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    grad_full = np.fft.irfft(A_rev * W, n=size)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]

    # smooth objective value (optional, not used for selection)
    g_val = (max_scaled + math.log(exp_vals.sum())) / lam
    return grad, g_val, conv_full.max()

# --------------------------------------------------------------
def _adam_step(a: np.ndarray, grad: np.ndarray, m: np.ndarray, v: np.ndarray,
               t: int, lr: float = 0.05, beta1: float = 0.9,
               beta2: float = 0.999, eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """One Adam update followed by projection onto the simplex."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = _project_onto_simplex(a_new)
    return a_new, m, v

    Returns the final sequence as a plain Python list.

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0    # safety margin

    # ------------------------------------------------------------------
    # 1.  Exhaustive Beta‑family sweep
    # ------------------------------------------------------------------
    lengths = [3000, 4000, 5000]   # a compromise between resolution and speed
    best_seq = None
    best_score = float("inf")
    best_n = None

    # reuse a previous solution if available
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
    except Exception:
        base_seq = None

    def _maybe_update(vec: np.ndarray, tag: str, n: int):
        """Update the global best if ``vec`` is better."""
        nonlocal best_seq, best_score, best_n
        sc = _raw_score(vec)
        if sc < best_score:
            best_seq, best_score, best_n = vec.copy(), sc, n
            print(f"   new best ({tag})   score={best_score:.6f}")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # (a) previous solution interpolated
        if base_seq is not None:
            interp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(base_seq)),
                               base_seq)
            _maybe_update(_project_onto_simplex(interp), "prev", n)

        # (b) uniform
        _maybe_update(_project_onto_simplex(np.full(n, 1.0 / n)), "uniform", n)

        # (c) symmetric Betas – we also look at very small alphas (0.02)
        alphas = np.concatenate((
            np.linspace(0.02, 0.30, 12, endpoint=False),
            np.linspace(0.30, 1.00, 15),
            np.linspace(1.00, 4.00, 12)
        ))
        for a in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a, a)
            _maybe_update(vec, f"symBeta a={a:.3f}", n)

        # (d) random 2‑component Beta mixtures
        for _ in range(200):
            if time.time() >= deadline:
                break
            a1, b1 = np.random.rand() * 4.5 + 0.05, np.random.rand() * 4.5 + 0.05
            a2, b2 = np.random.rand() * 4.5 + 0.05, np.random.rand() * 4.5 + 0.05
            lam_mix = np.random.rand()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            mix = lam_mix * p1 + (1.0 - lam_mix) * p2
            _maybe_update(_project_onto_simplex(mix), "mix2", n)

    # ------------------------------------------------------------------
    # Fallback (very unlikely) – uniform at the largest length considered
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = _project_onto_simplex(np.full(n, 1.0 / n))
        best_score = _raw_score(best_seq)
        best_n = n
        print("Fallback to uniform.")

    print(f"Best raw score after Beta sweep: {best_score:.6f} (len={best_n})")

    # ------------------------------------------------------------------
    # 2. Random Dirichlet restarts with short Adam refinements
    # ------------------------------------------------------------------
    n = best_n
    restarts = 30
    for i in range(restarts):
        if time.time() >= deadline:
            break
        # Dirichlet(1) gives a uniform random point on the simplex
        a0 = np.random.dirichlet(np.ones(n))
        # give each restart roughly 5 s (or less if we are close to deadline)
        sub_deadline = min(deadline, time.time() + 5.0)
        a_refined = _optimize(a0, sub_deadline)
        sc = _raw_score(a_refined)
        if sc < best_score - 1e-12:
            best_seq, best_score = a_refined.copy(), sc
            print(f"[{time.time():.1f}] restart {i+1}/{restarts} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Long final optimisation of the current best candidate
    # ------------------------------------------------------------------
    best_seq = _optimize(best_seq, deadline)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
