def _refine_exponential(a0, deadline):
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        n = a.shape[0]
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # subgradient
        k = int(np.argmax(conv))
        grad = _subgradient_max(a, k)
        # try several etas
        best_local = (best_score, a)
        for eta in (0.5, 0.2, 0.1, 0.05, 0.02):
            a_new = a * np.exp(-eta * grad)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 0:
                a_new = a_new / s
                score = _objective(a_new)
                if score < best_local[0]:
                    best_local = (score, a_new)
        if best_local[0] < best_score - 1e-12:
            best_score, a = best_local
            best_seq = a.copy()

            # jitter
            if iteration % 100 ==0:
                a += np.random.randn(n)*1e-5
                a = np.maximum(a,0)
                a = a / a.sum()
    return best_seq, best_score

"""
Propose a low‑objective step‑function using a rich parametric
initialisation and an aggressive mirror‑descent / line‑search refinement.
The algorithm works on the simplex (∑ a_i = 1) because the official
objective 2·n·max(a∗a)/(∑a)² is scale‑invariant.  The search proceeds in
three stages:

1.  Random families (arcsine, power‑law, (asymmetric) Beta, Gaussian,
    mixtures) on many lengths up to 20000.  The best few candidates are
    kept.

2.  For each of the top candidates a fast exponentiated‑gradient loop is
    run.  At every step we compute the true sub‑gradient of the
    ∞‑norm of the self‑convolution (or a soft‑max surrogate once a large
    temperature is reached) and try several step‑sizes; the best
    improvement is kept.  Jitter is added periodically to escape plates.

3.  Final cheap hill‑climbing that moves a random fraction of mass
    between two coordinates (simulated‑annealing style).

All heavy work uses FFT‑based convolution (≈ O(n log n)).  The routine
returns the best sequence found before the 1000 s budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) distribution of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """Symmetric 1/(i·(n‑i))^p (invert=True) or (i·(n‑i))^p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    """Beta(a,b) on {0,…,n‑1} (continuous density sampled at mid‑points)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow for extreme parameters
    with np.errstate(divide='ignore', invalid='ignore'):
        vals = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    vals[np.isnan(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on the index set."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """Convex combination (1‑lam)·arcsine + lam·uniform."""
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a using FFT."""
    return float(_fft_self_convolve(a).max())


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) for a vector that already lies on the simplex.
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·conv_k) w.r.t. a.
    Implements the formula g = 2 * conv_rev(w, a_rev).
    """
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv2 = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv2[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient refinement with line‑search.
    Tries several learning‑rates each step and uses a soft‑max surrogate
    after the temperature grows large enough.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    size_fft = 1 << ((3 * n - 2).bit_length())

    # schedule parameters
    eta0 = 0.5
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 8000.0
    softmax_start = 500.0           # switch to softmax after alpha exceeds this

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] refine it={it} score={best_score:.6f}")

        # decide which gradient to use
        if alpha > softmax_start:
            # use the smooth surrogate gradient
            grad = _softmax_gradient(a, conv, alpha, size_fft)
        else:
            # exact sub‑gradient
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # try a small set of step sizes
        best_local_score = cur_score
        best_local_a = a
        for eta in (0.5, 0.2, 0.1, 0.05, 0.02):
            a_candidate = a * np.exp(-eta * grad)
            a_candidate = np.maximum(a_candidate, 0.0)
            s = a_candidate.sum()
            if s > 0.0:
                a_candidate = a_candidate / s
            else:
                continue
            cand_score = _objective(a_candidate)
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local_a = a_candidate

        # accept the best candidate among the tried step sizes
        if best_local_score < cur_score - 1e-12:
            a = best_local_a
            # best_seq is updated at top of loop on next iteration
        else:
            # occasional jitter to escape plateaus
            if it % 3000 == 0 and np.random.rand() < 0.25:
                a += np.random.randn(n) * 1e-5
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # diminishing base step‑size
        eta0 = max(1e-4, eta0 / np.sqrt(it))

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_hill_climb(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic hill‑climbing that moves a fraction of probability mass.
    Accepts a move if it strictly reduces the objective.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 15 % of source mass
        eps = a[i] * np.random.rand() * 0.15
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cand_score = _objective(a_candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local it={it} score={best_score:.6f}")

    return best_seq, best_score
