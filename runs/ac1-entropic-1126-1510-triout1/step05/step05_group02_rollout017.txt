def _fft_self_convolve_full(a):
    n = a.shape[0]; size = 1 << ((2*n-1)-1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A*A, n=size)
    return conv[:2*n-1]

def _fft_convolve(a,b):
    size = 1 << ((a.size + b.size -1)-1).bit_length()
    A = np.fft.rfft(a, n=size); B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A*B, n=size)
    return conv[:a.size+b.size-1]

    import time, math, numpy as np
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine length n
    # We'll start with n=8000 if possible else fallback to 5000.
    # maybe we can compute size_t = 8000; if memory insufficient maybe n=5000
    n = 8000
    # to ensure feasible within time, we can adapt if time already low.

    # base initial candidate - arcsine
    a = _beta_distribution(n, 0.5, 0.5)
    a = _project_onto_simplex(a)

    # Keep best
    best_a = a.copy()
    best_val = 2 * n * _fft_self_convolve_max(a)   # raw score

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    t = 0

    beta_temp = 0.5
    beta_max = 64.0

    while time.time() < deadline:
        t += 1
        # compute conv and gradient
        conv = _fft_self_convolve_full(a)
        # compute smooth max objective
        max_conv = conv.max()
        # compute softmax weights
        # for numerical stability: subtract max
        exp_term = np.exp(beta_temp * (conv - max_conv))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp  # shape 2n-1

        # compute gradient using w
        a_rev = a[::-1]
        grad_full = _fft_convolve(a_rev, w)
        grad = 2.0 * grad_full[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_onto_simplex(a)

        # enforce symmetry: average with reversed
        a = (a + a[::-1]) / 2.0

        # compute raw score
        cur_val = 2 * n * _fft_self_convolve_max(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] new best raw={best_val:.6f}")

        # update beta (increase gradually)
        if t % 2000 == 0 and beta_temp < beta_max:
            beta_temp = min(beta_max, beta_temp * 2.0)

        # occasional random noise
        if t % 5000 == 0:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a)
            a = (a + a[::-1]) / 2.0

    # final best sequence
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Final raw best={best_val:.6f}, evaluated={final_score:.6f}, len={len(final_seq)}")
    return final_seq

"""
propose_candidate
-----------------

Search for a low‑score step‑function (non‑negative vector) using two phases:

1. **Coarse sampling** – explore a few families of Beta‑distributed probability
   vectors (uniform, arcsine and random two‑Beta mixtures).  The best candidate
   found (in terms of the raw objective 2·n·max(conv)) is taken as the start
   point.

2. **Refinement** – minimise the smoothed ℓ∞‑convolution objective
   ``f(a)=log‑sum‑exp(β·(a*a)) / β`` by projected (Adam) gradient descent.
   The soft‑max weight concentrates on the current maximal convolution entry,
   so the method drives the peak down.  The temperature β is increased
   geometrically, turning the smooth objective into the true maximum.
   All updates are projected onto the simplex (Σa_i = 1, a_i≥0).

The routine respects the 1000 s time budget, prints progress, and returns a
plain Python list ready for ``evaluate_sequence``.
"""

from __future__ import annotations
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete β(α,β) distribution on {0,…,n‑1} (mid‑point rule, normalised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow
    logc = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logc + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf, dtype=np.float64)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex vector onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _raw_score(vec: np.ndarray) -> float:
    """Raw objective 2·n·max(a*a) for a vector with Σa = 1."""
    n = vec.shape[0]
    return 2.0 * n * _fft_self_convolve_max(vec)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current best bound (≈ 1.503).  The routine performs a quick
    parametric search followed by a smooth‑max gradient descent.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    # Choose a problem size – larger n gives a smoother shape but costs
    # a bit more per iteration.  8000 works comfortably within the budget.
    n = 8000
    # Use the previous construction if it exists (it may be better).
    base = globals().get("height_sequence_1", None)
    if base is not None:

            base = np.asarray(base, dtype=np.float64)
            base = _interpolate_to_length(base, n)
            base = _project_onto_simplex(base)
        except Exception:
            base = None

    # --------------------------------------------------------------
    # 1️⃣  Coarse search over a few families (uniform, arcsine, symmetric Beta,
    #     random two‑Beta mixtures).  Keep the best vector found.
    best_vec = None
    best_val = float("inf")

    def _update_best(v):
        nonlocal best_vec, best_val
        val = _raw_score(v)
        if val < best_val:
            best_val, best_vec = val, v.copy()
            print(f"[{time.time():.1f}] new coarse best = {best_val:.6f}")

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _update_best(uniform)

    # arcsine (Beta(0.5,0.5))
    arcsine = _beta_distribution(n, 0.5, 0.5)
    _update_best(arcsine)

    # symmetric Beta(α,α) on a modest grid
    alphas = np.concatenate((
        np.linspace(0.12, 0.6, 10, endpoint=False),
        np.linspace(0.6, 2.0, 12),
        np.linspace(2.0, 5.0, 8)
    ))
    for a in alphas:
        if time.time() >= deadline:
            break
        vb = _beta_distribution(n, a, a)
        _update_best(vb)

    # random mixtures of two independent Beta components
    for _ in range(250):
        if time.time() >= deadline:
            break
        a1, b1 = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
        a2, b2 = np.random.rand() * 4.9 + 0.1, np.random.rand() * 4.9 + 0.1
        lam = np.random.rand()
        p1 = _beta_distribution(n, a1, b1)
        p2 = _beta_distribution(n, a2, b2)
        mix = lam * p1 + (1.0 - lam) * p2
        mix = _project_onto_simplex(mix)

        best_val = _raw_score(best_vec)

    a = best_vec.copy()
    best_a = a.copy()
    best_raw = best_val
    it = 0

    # Adam hyper‑parameters (works well for the smoothed objective)
    lr = 0.12
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature (starts low → smooth, ends high → true max)
    beta_temp = 0.5
    beta_max = 64.0

    while time.time() < deadline:
        it += 1

        # ---------- convolution and soft‑max weights ----------
        conv_full = _fft_self_convolve_full(a)            # length 2n‑1
        c_max = conv_full.max()

        diff = conv_full - c_max
        exp_beta = np.exp(beta_temp * diff)               # ≤ 1
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp                            # soft‑max weights

        a_rev = a[::-1]
        grad_full = _fft_convolve(a_rev, w)

        norm = np.linalg.norm(grad)
        if norm > 0:
            direction = grad / norm
        else:
            direction = grad

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * direction
        v = beta2 * v + (1.0 - beta2) * (direction * direction)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ---------- feasibility (simplex) ----------
        a = _project_onto_simplex(a)

        if cur_raw + 1e-12 < best_raw:
            best_raw, best_a = cur_raw, a.copy()
            print(f"[{time.time():.1f}] it={it}  improved raw={best_raw:.6f}")

        # ---------- temperature schedule ----------
        if beta_temp < beta_max:
            beta_temp = min(beta_max, beta_temp * 1.018)   # slow exponential growth

    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(
        f"Finished: raw best={best_raw:.6f}  evaluated={final_score:.6f}  len={len(final_seq)}"
    )
    return final_seq
