    a = seq.copy()
    best_seq = a.copy()
    best_score = _objective(a)
    while time.time() < deadline:
        # decide move type
        if np.random.rand() < 0.3:
            # swap
            i,j = np.random.randint(0,n,size=2)
            if i==j: continue
            a_candidate = a.copy()
            tmp = a_candidate[i]
            a_candidate[i] = a_candidate[j]
            a_candidate[j] = tmp
        else:
            # epsilon move
            i,j = np.random.randint(0,n,size=2)
            if i==j: continue
            eps = a[i] * np.random.rand() * 0.1
            a_candidate = a.copy()
            a_candidate[i] -= eps
            a_candidate[j] += eps
        # reproject onto simplex if needed (should preserve sum)
        # but possible negative due to rounding, guard
        if a_candidate[i] < -1e-12: continue
        a_candidate = np.maximum(a_candidate,0)
        s = a_candidate.sum()
        if s>0: a_candidate/=s
        cur_score = _objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
    return best_seq,best_score

"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(a*a) / (sum a)^2

where * denotes discrete convolution.  The code explores a few
parameteric families (power‑law, symmetric Beta, mixtures) on several
lengths, keeps the best few candidates and then refines them with an
exponentiated‑gradient mirror‑descent optimiser followed by a simple
stochastic hill‑climber that also swaps indices.  All work is done on
the probability simplex (sum = 1), because the objective is scale‑invariant.
The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector (sum = 1)."""
    a = _project_onto_simplex(seq, s=1.0)
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1.0 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """(1‑lam)·arcsine + lam·uniform."""
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a*a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the (1/α)·log‑sum‑exp(α·b_k) surrogate w.r.t. a.
    Implements g = 2 * conv_rev(w, a_rev) where w is the soft‑max over b.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser on the probability simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    eta0 = 0.7               # initial step‑size constant
    alpha = 20.0             # start with a modest temperature
    alpha_mul = 1.02
    alpha_max = 3000.0
    size_fft = 1 << ((3 * a.shape[0] - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * a.shape[0] * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] EG it={it} n={a.shape[0]} score={best_score:.6f}")

        # use the smooth surrogate while alpha is moderate, switch to hard max
        # when alpha grows large (the surrogate becomes very sharp)
        if alpha < 2000.0:
            grad = _gradient_softmax(a, b, alpha, size_fft)
        else:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)

        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / s

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional jitter to escape plateaus
        if it % 3000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(a.shape[0]) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Random‑move hill‑climbing (epsilon transfers + swaps) on the simplex.
    Accepts only improving moves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        if np.random.rand() < 0.4:
            # epsilon transfer
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps = a[i] * np.random.rand() * 0.08
            a_candidate = a.copy()
            a_candidate[i] -= eps
            a_candidate[j] += eps
        else:
            # swap two entries
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            a_candidate = a.copy()
            a_candidate[i], a_candidate[j] = a_candidate[j], a_candidate[i]

        # re‑project (tiny drift may appear)
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s > 0.0:
            a_candidate /= s
        else:
            continue

        cur_score = _objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local it={it} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000,
                      seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine stops after ``budget_s`` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # Length candidates -------------------------------------------------
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    length_factors = (0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    })
    # ensure a few hand‑picked lengths are present
    lengths.extend([5000, 8000, 12000, 16000, 20000])
    lengths = sorted({l for l in lengths if l <= 20000})

    # ------------------------------------------------------------------
    # Parameter grids ---------------------------------------------------
    beta_vals = np.linspace(0.20, 0.90, 15)          # symmetric Beta(a,a), a<1 (U‑shaped)
    power_vals = np.linspace(0.30, 0.80, 11)         # (i·(n‑i))^{-p}
    mix_lam_vals = np.linspace(0.0, 0.45, 10)       # mixture arcsine–uniform

    # ------------------------------------------------------------------
    # Global bookkeeping ------------------------------------------------
    best_seq = None
    best_score = float("inf")
    top_candidates = []         # list of (score, vector) – we keep the 10 best

    # ------------------------------------------------------------------
    # Seed with any previously‑found solution -------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if seq0.size:
            seq0 = np.maximum(seq0, 0.0)
            seq0 = _project_onto_simplex(seq0, s=1.0)
            score0 = _objective(seq0)
            print(f"Seeded with previous solution: score={score0:.6f}")
            best_seq = seq0.copy()
            best_score = score0
            top_candidates.append((score0, seq0))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Exhaustive sweep over the simple families -----------------------
    for n in lengths:
        # uniform (pure) – useful baseline
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        score = _objective(uniform)
        if score < best_score:
            best_score = score
            best_seq = uniform.copy()
            print(f"Uniform length={n} improved best to {best_score:.6f}")
        top_candidates.append((score, uniform))

        # arcsine (Beta 0.5) – strong baseline
        arcs = _arcsine_distribution(n)
        score = _objective(arcs)
        if score < best_score:
            best_score = score
            best_seq = arcs.copy()
            print(f"Arcsine length={n} improved best to {best_score:.6f}")
        top_candidates.append((score, arcs))

        # power‑law family
        for p in power_vals:
            seq = _power_law_distribution(n, p, invert=True)
            score = _objective(seq)
            top_candidates.append((score, seq))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"Power p={p:.3f} n={n} new best {best_score:.6f}")

        # Beta family
        for a in beta_vals:
            seq = _beta_distribution(n, a)
            score = _objective(seq)
            top_candidates.append((score, seq))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"Beta a={a:.3f} n={n} new best {best_score:.6f}")

        # mixture arcsine + uniform
        for lam in mix_lam_vals:
            seq = _mixture_arcsine_uniform(n, lam)
            score = _objective(seq)
            top_candidates.append((score, seq))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"Mix λ={lam:.3f} n={n} new best {best_score:.6f}")

        # early stop if we are out of time
        if time.time() >= deadline - 30.0:
            break

    # keep only the ten best candidates for the refinement phase
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:10]

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient refinement of the elite ----------
    for idx, (sc, cand) in enumerate(top_candidates):
        if time.time() >= deadline - 30.0:
            break
        n = cand.shape[0]
        remaining = deadline - time.time()
        # give each refinement run up to 45 s (or whatever is left)
        run_deadline = time.time() + min(45.0, remaining * 0.6)
        refined_seq, refined_score = _optimize_exponential(cand, run_deadline)
        refined_score = _objective(refined_seq)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"Refinement #{idx} improved best to {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Stage 3 – local random‑move hill climbing ------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        refined_seq, refined_score = _local_refinement(best_seq, deadline)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"Local refinement gave final best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed) --------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Finished. Best objective = {best_score:.6f} (length = {best_seq.shape[0]})")
    return best_seq.tolist()
