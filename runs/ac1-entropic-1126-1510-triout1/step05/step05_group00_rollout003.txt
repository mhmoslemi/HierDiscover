def _adam_optimize(a, deadline):
    n = len(a)
    # hyperparameters
    lr0 = 0.3
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    # schedule for temperature: start T = 0.02? Actually alpha = 1/T.
    # We'll start with alpha = 10, increase to alpha = 2000
    # Use exponential schedule: alpha = alpha0 * (alpha_max/alpha0) ** (t / max_iter)
    # But we don't know max_iter, maybe use time-based: increase alpha every 50 seconds?
    # Simpler: each iteration compute alpha = min(alpha_max, alpha0 + alpha_inc * t)
    alpha0 = 5.0
    alpha_inc = 0.02      # increase slowly
    alpha_max = 3000.0

    while time.time() < deadline:
        t += 1
        # compute convolution
        b = _fft_self_convolve(a)
        # compute smooth weights
        alpha = min(alpha_max, alpha0 + alpha_inc * t)
        # stable softmax
        # as b may be small, we can do w = np.exp(alpha * (b - b.max())) and normalize
        maxb = b.max()
        w_unnorm = np.exp(alpha * (b - maxb))
        w = w_unnorm / w_unnorm.sum()
        # gradient
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 / (np.sqrt(t))  # optional decreasing step size
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_onto_simplex(a_new, s=1.0)
        # evaluate actual objective
        cur_score = _max_convolution_score(a_new)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[{time.time():.1f}] Adam iter={t} alpha={alpha:.1f} new best {cur_score:.6f}")
        a = a_new

        # early exit if best_score below target (1.503)
        if best_score <= 1.503:
            break
    return best_seq, best_score

def _hill_climb(a, deadline, best_score):
    n = len(a)
    cur_seq = a.copy()
    cur_score = best_score
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # choose a number of positions and mass
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(0.05, cur_seq[i])  # up to 5%
        # optionally random direction sign? we always move from i to j.
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_score = _max_convolution_score(new_seq)
        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.1f}] Hill climb iter={iteration} improved to {best_score:.6f}")
        # Occasionally random jitter
        if iteration % 5000 == 0:
            perturb = np.random.randn(n) * 1e-5
            cur_seq = best_seq + perturb
            cur_seq = _project_onto_simplex(cur_seq, s=1.0)
            cur_score = _max_convolution_score(cur_seq)
    return best_seq, best_score

def _optimize_adam(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5
    t = 0
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    alpha = 5.0
    alpha_max = 3000.0
    alpha_inc = 0.02

    while time.time() < deadline:
        t += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        b = _fft_self_convolve(a)
        cur_alpha = min(alpha_max, alpha + alpha_inc * t)
        # compute softmax
        maxb = b.max()
        w_unnorm = np.exp(cur_alpha * (b - maxb))
        w = w_unnorm / w_unnorm.sum()

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n-1:n-1+n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. **Exploratory sampling** – try several analytic families (arcsine,
   symmetric beta, triangle, exponential) and a few random Dirichlet draws
   for a handful of lengths.  Keep the best few candidates.

2. **Adam optimisation with a smooth max** – minimise
   `max(a*a)` using a soft‑max approximation.
   The “temperature’’ (inverse of a soft‑max parameter) is slowly raised,
   turning the smooth objective into the true max.
   An Adam step is taken and the iterate is projected onto the probability
   simplex (sum = 1, a_i ≥ 0).  The best sequence found for each start is
   retained.

3. **Local hill‑climbing** – small random transfers of mass between two
   positions (and occasional jitter) are applied to squeeze out any
   remaining improvement.

The whole search respects the time budget (≈ 1000 s).  The returned list
can be fed directly to ``evaluate_sequence``; the code never writes to
disk or uses external services.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x) = s}.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a probability vector (sum = 1).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p, p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _random_dirichlet_distribution(n: int, alpha: float = 0.5) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(alpha)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        x = np.full_like(x, 1.0 / n)
    else:
        x /= x.sum()
    return x

def _adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with a gradually sharpening soft‑max of the convolution.
    Returns the best sequence and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.3

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # soft‑max temperature schedule (alpha = 1/T)
    alpha_start = 5.0
    alpha_inc = 0.025          # per iteration
    alpha_max = 3000.0

    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # full self‑convolution
        b = _fft_self_convolve(a)

        # soft‑max weights
        cur_alpha = min(alpha_max, alpha_start + alpha_inc * it)
        maxb = b.max()
        w_unnorm = np.exp(cur_alpha * (b - maxb))
        w = w_unnorm / w_unnorm.sum()

        # gradient of weighted sum  (2 * a convolved with w)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr0 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam iter={it} alpha={cur_alpha:.1f} best={cur_score:.6f}")

        # Early stop if we already beat the target
        if best_score <= 1.503:
            break

        # occasional jitter to escape plateaux
        if it % 1000 == 0 and np.random.rand() < 0.2:
            a = best_seq + np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

def _hill_climb(seq: np.ndarray, cur_best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random‑transfer hill climbing, starting from ``seq``.
    Returns the improved sequence and its exact score.
    """
    a = seq.copy()
    best_seq = a.copy()
    best_score = cur_best_score
    cur_score = cur_best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.05  # up to 5 % of total mass per trial

    while time.time() < deadline:
        it += 1
        # decay the transferred mass slowly
        transfer = max_transfer * (0.5 ** (it / 20000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] Hill climb iter={it} new best={best_score:.6f}")

        # occasional global jitter
        if it % 5000 == 0 and np.random.rand() < 0.1:
            a = best_seq + np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score

    def _maybe_push(score, seq, n):

            best_candidates.insert(i, (score, seq.copy(), n))

        if time.time() >= deadline:
            break

        # arcsine
        a = _arcsine_distribution(n)
        _maybe_push(_max_convolution_score(a), a, n)

        # triangle
        a = _triangular_distribution(n)
        _maybe_push(_max_convolution_score(a), a, n)

        # exponential
        a = _exponential_distribution(n, lam=0.005)
        _maybe_push(_max_convolution_score(a), a, n)

        # symmetric beta
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            _maybe_push(_max_convolution_score(a), a, n)

        # few random Dirichlet draws
        for _ in range(2):
            a = _random_dirichlet_distribution(n, alpha=0.5)
            _maybe_push(_max_convolution_score(a), a, n)

    # --------------------------------------------------------------
    # 3.  Include the previous best construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            _maybe_push(_max_convolution_score(prev), prev, len(prev))
    except Exception:
        pass

    # Fallback – should never happen
    if not best_candidates:
        n = max(500, base_n)
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        best_candidates.append((_max_convolution_score(uniform), uniform, n))

    # --------------------------------------------------------------
    # 4.  Adam optimisation (soft‑max) from each promising start
    best_seq_overall = None
    best_score_overall = np.inf

    for init_score, init_seq, n in best_candidates:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Starting Adam on length={n} init_score={init_score:.6f}")
        seq_opt, score_opt = _adam_optimize(init_seq, deadline)
        if score_opt < best_score_overall:
            best_score_overall = score_opt
            best_seq_overall = seq_opt.copy()
            print(f"[{time.time():.1f}] New overall best after Adam: {score_opt:.6f}")

        # early exit if we already beat the target
        if best_score_overall <= 1.503:
            break

    # --------------------------------------------------------------
    # 5.  Local random‑transfer hill climbing
    if best_seq_overall is not None:
        print(f"[{time.time():.1f}] Starting hill climbing (length={len(best_seq_overall)})")
        final_seq, final_score = _hill_climb(best_seq_overall, best_score_overall, deadline)
    else:
        # should not happen, but keep something safe
        final_seq = np.full(base_n, 1.0 / base_n, dtype=np.float64)
        final_score = _max_convolution_score(final_seq)

    # --------------------------------------------------------------
    print(f"Final best score: {final_score:.6f} (length={len(final_seq)})")
    return final_seq.tolist()
