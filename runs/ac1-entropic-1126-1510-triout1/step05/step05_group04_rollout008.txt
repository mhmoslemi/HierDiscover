"""
Propose a low‑score step function.

The search proceeds in four phases:

1. **Diverse warm‑starts** – several families of probability vectors
   (uniform, exponential, power‑law, U‑shaped, Dirichlet, Gaussian,
   and the reference if present) are transformed to logits and optimised
   with a smooth‑max surrogate (log‑sum‑exp) using L‑BFGS‑B.  The
   temperature μ is reduced gradually (μ←μ·0.85) to sharpen the
   surrogate.

2. **Hard‑max sub‑gradient descent** – starting from the best candidate
   the exact max‑convolution is differentiated (sub‑gradient) and a
   line‑search update on the probability simplex is performed.  A small
   random jitter is added when progress stalls.

3. **Large‑scale random‑block perturbations** – many cheap candidates are
   generated by assigning a constant value to a random partition of the
   index set; each is projected onto the simplex and evaluated.

4. **Monotone‑sorting post‑processing** – the best sequence is sorted in
   descending, ascending and a “U‑shaped’’ order (large values at the
   ends) to exploit the rearrangement inequality.

All helpers are top‑level (no closures) and the routine respects the
user‑supplied time budget (default 1000 s).  The returned list is a
non‑negative step‑function; its quality is measured by
``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = z} (z≥0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(
    y: np.ndarray, mu: float, n: int, S: float
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max (log‑sum‑exp) surrogate of max(conv) and its gradient.

    Parameters
    ----------
    y : (n,) ndarray
        Unconstrained logits (log‑odds of the simplex).
    mu : float > 0
        Smoothing temperature.
    n : int
        Length of the step function.
    S : float
        Target ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : (n,) ndarray
        Gradient w.r.t. y.
    """
    # probabilities on the simplex and actual heights
    p = _softmax(y)            # Σ p = 1, p ≥ 0
    a = S * p                  # Σ a = S

    # full linear convolution a*a (length 2·n−1)
    conv = np.convolve(a, a)

    max_c = conv.max()
    # smooth‑max weights
    diff = (conv - max_c) / mu
    diff = np.clip(diff, -700.0, 700.0)
    exp_diff = np.exp(diff)
    Z = exp_diff.sum()
    w = exp_diff / Z
    f_val = max_c + mu * math.log(Z)

    # gradient w.r.t. heights a:
    #   g_a[i] = 2 * Σ_k w_k * a_{k-i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n−2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule for logits (soft‑max Jacobian)
    inner = np.dot(g_a, p)                     # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize that returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective (identical to evaluate_sequence
    but without the security clamping).  a must be non‑negative.
    """
    a = np.asarray(a, dtype=float)
    a = np.maximum(a, 0.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    max_b = np.convolve(a, a).max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
class _SearchState:
    """Holds the incumbent best sequence and its score."""

    __slots__ = ("best_score", "best_a")

    def __init__(self):
        self.best_score = float("inf")
        self.best_a = None


def _maybe_update(state: _SearchState, a_vec: np.ndarray) -> None:
    """Evaluate a candidate and keep it if it improves the incumbent."""
    sc = _compute_score(a_vec)
    if sc < state.best_score:
        state.best_score = sc
        state.best_a = a_vec.copy()
        print(f"[improve] new best score = {state.best_score:.6f}")


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluate_sequence() score.  The routine respects the given time budget
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Determine problem size
    try:
        ref_len = len(height_sequence_1)      # type: ignore[name-defined]
        n = min(max(ref_len * 2, 5000), 9000)  # stay below 10 000 for speed
        have_ref = True
    except Exception:
        n = 6000
        have_ref = False

    S = math.sqrt(2.0 * n)   # target ℓ¹‑norm
    state = _SearchState()

    # --------------------------------------------------------------
    # Build diverse initial logits (unconstrained y)
    eps = 1e-12
    init_ys = []

    # 0 – uniform
    init_ys.append(np.zeros(n, dtype=np.float64))

    # 1 – decreasing exponential
    alpha = rng.uniform(0.0005, 0.003) * (n / 2000.0)
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    # 2 – increasing exponential
    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    # 3 – power‑law decay (β in [0.3, 1.5])
    beta = rng.uniform(0.3, 1.5)
    p = (np.arange(1, n + 1, dtype=np.float64) ** -beta)
    p /= p.sum()
    init_ys.append(np.log(p + eps))

    # 4 – linear decay (max at left, zero at right)
    p = np.linspace(1.0, 0.0, n)
    p /= p.sum()
    init_ys.append(np.log(p + eps))

    # 5 – symmetric U‑shape (large at both ends)
    alpha_u = rng.uniform(0.0005, 0.003) * (n / 2000.0)
    left = np.exp(-alpha_u * np.arange(n, dtype=np.float64))
    right = np.exp(-alpha_u * np.arange(n - 1, -1, -1, dtype=np.float64))
    p = left + right
    p /= p.sum()
    init_ys.append(np.log(p + eps))

    # 6 – sparse Dirichlet (α < 1)
    dir_alpha = rng.uniform(0.2, 0.6)
    p = rng.dirichlet([dir_alpha] * n)
    init_ys.append(np.log(p + eps))

    # 7 – dense Dirichlet (α > 1)
    dir_alpha2 = rng.uniform(1.5, 3.0)
    p = rng.dirichlet([dir_alpha2] * n)
    init_ys.append(np.log(p + eps))

    # 8 – random Gaussian logits
    init_ys.append(rng.standard_normal(n) * 0.5)

    # 9 – reference if available
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, S)
        init_ys.append(np.log(a_ref + eps))

    # --------------------------------------------------------------
    # Phase 1 – smooth‑max optimisation (L‑BFGS‑B)
    MU_FACTOR = 0.85
    MU_MIN = 1e-5
    MAX_ITER_PER_MU = 250
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}]")
        y = y0.copy()

        # initialise temperature from current max‑convolution
        a_cur = S * _softmax(y)
        mu = float(np.convolve(a_cur, a_cur).max())
        mu = max(mu, 1.0)   # avoid zero temperature

        while mu > MU_MIN and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": MAX_ITER_PER_MU, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(state, a)

            mu *= MU_FACTOR

    # --------------------------------------------------------------
    # Phase 2 – hard‑max sub‑gradient descent
    if state.best_a is None:
        # fallback – should not happen, but protect against it
        state.best_a = np.full(n, S / n, dtype=float)
        state.best_score = _compute_score(state.best_a)

    a = state.best_a.copy()
    max_local_iters = 4000
    step0 = S * 0.4 / n
    no_improve_counter = 0

    while time.time() < deadline and max_local_iters > 0:
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break

        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        g = 2.0 * np.convolve(w, a[::-1])[n - 1 : n - 1 + n]

        step = step0 / (1.0 + 0.0005 * (4000 - max_local_iters))
        improved = False
        for _ in range(12):
            a_new = a - step * g
            a_new = _project_to_simplex(a_new, S)
            sc = _compute_score(a_new)
            if sc < state.best_score - 1e-10:
                state.best_score = sc
                state.best_a = a_new.copy()
                a = a_new
                improved = True
                no_improve_counter = 0
                print(f"[subgrad] iter {4000 - max_local_iters:4d} step {step:.6e} => {state.best_score:.6f}")
                break
            step *= 0.5

        if not improved:
            no_improve_counter += 1
            # occasional random jitter to escape flat regions
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.random() * a[i] * 0.03
            a[i] -= delta
            a[j] += delta
            a = _project_to_simplex(a, S)

        max_local_iters -= 1
        if no_improve_counter >= 500:
            # stop early if progress has stalled
            break

    # --------------------------------------------------------------
    # Phase 3 – random block perturbations (cheap large‑scale search)
    remaining = deadline - time.time()
    if remaining > 2.0:
        max_trials = int(min(8000, remaining * 150))
        print(f"[block] {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(3, 15)               # number of blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(state, cand)

    # --------------------------------------------------------------
    # Phase 4 – monotone‑sorting heuristics
    if time.time() < deadline and state.best_a is not None:
        # descending order (large values left)
        desc = np.sort(state.best_a)[::-1]
        desc = _project_to_simplex(desc, S)
        _maybe_update(state, desc)

        # ascending order (large values right)
        asc = np.sort(state.best_a)
        asc = _project_to_simplex(asc, S)
        _maybe_update(state, asc)

        # U‑shaped order: biggest values at the ends, decreasing towards centre
        vals_sorted = np.sort(state.best_a)[::-1]
        u_seq = np.empty_like(vals_sorted)
        left = 0
        right = n - 1
        for k, v in enumerate(vals_sorted):
            if k % 2 == 0:
                u_seq[left] = v
                left += 1
            else:
                u_seq[right] = v
                right -= 1
        u_seq = _project_to_simplex(u_seq, S)
        _maybe_update(state, u_seq)

    # --------------------------------------------------------------
    print(f"[final] best score = {state.best_score:.6f}")
    return state.best_a.tolist()
