"""
Optimisation of non‑negative step‑function heights.

Approach
--------
1. *Candidate generation* – for each length we build a modest library of
   analytic shapes (Beta/Arcsine), a few Dirichlet draws with various
   concentration parameters and (if available) the previously best
   construction.  All candidates are normalised to sum = 1 and cheaply
   scored by a fast FFT self‑convolution.

2. *Refinement* – the top few candidates are improved by an Adam‑style
   projected gradient method.  The objective is the infinity‑norm of the
   self‑convolution; we minimise a smooth “soft‑max’’ surrogate whose
   temperature is increased during optimisation and finally switch to
   the exact sub‑gradient of the max‑operator.  After each gradient step
   the iterate is projected back onto the probability simplex.

3. *Result* – the best sequence found (according to the true
   ``evaluate_sequence``) is returned.  The routine respects the total
   time budget (default 1000 s) and stops early once the target score
   1.503 is reached.

The implementation is deliberately lightweight (no external solvers)
and works for vectors up to several 10 000 entries within the allotted
time.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Cheap evaluation (sum = 1) – returns 2·n·max(conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    Returns a vector of length n.
    """
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max, using a uniform weight over all
    (approximately) active indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where k = 2*i (i.e. self‑product)

Optimisation of non‑negative step‑function heights.

Approach
--------
1. **Candidate generation** – for each examined length we build a modest
   library of analytic shapes (Beta / arcsine), a few Dirichlet draws
   with different concentration parameters and, if available, the
   previously discovered best construction (interpolated to the needed
   size).  All candidates are normalised to sum = 1 and quickly scored
   via a fast FFT self‑convolution.

2. **Refinement** – the most promising candidates are improved by a
   projected Adam optimiser.  The objective is ``max(a*a)``; we minimise
   a smooth soft‑max surrogate whose temperature is annealed, switching
   to the exact sub‑gradient of the max‑operator once the temperature
   is large enough.  Each Adam step is followed by Euclidean projection
   onto the probability simplex (non‑negative, sum = 1).

3. **Result** – the best sequence seen (according to the true
   ``evaluate_sequence``) is returned.  The routine respects a wall‑clock
   budget (default 1000 s) and stops early once the target score 1.503
   is reached.

The implementation is lightweight (no external solvers) and works for
vectors up to several 10 000 entries within the allotted time.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / s
    return vals

def _score(a: np.ndarray) -> float:
    """Cheap evaluation (sum = 1) – returns 2·n·max(conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the soft‑max surrogate of max(conv)."""
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad
