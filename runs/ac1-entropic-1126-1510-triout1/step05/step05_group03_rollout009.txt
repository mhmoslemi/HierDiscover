def local_search(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # find max indices?
        max_indices = np.where(b >= cur_max - 1e-12)[0]
        k = np.random.choice(max_indices)
        # get i range for this k
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # choose i by weighted sampling
        i_vals = np.arange(i_low, i_high+1)
        prod = a[i_vals] * a[k - i_vals]
        if prod.sum() == 0:
            # choose random source
            src = np.random.randint(n)
        else:
            probabilities = prod / prod.sum()
            src = np.random.choice(i_vals, p=probabilities)
        # source index src; now pick destination dst random not same and with non-zero capacity:
        dst = np.random.randint(n)
        if src == dst:
            continue
        # move small fraction delta from src to dst
        max_delta = min(a[src], 0.1)  # up to 10%
        delta = np.random.rand() * max_delta
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()
        if new_max < cur_max - 1e-14:
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        # else maybe small acceptance? We skip acceptance for now.
        # limit iteration loops maybe with jitter.
        if np.random.rand() < 0.01:
            # random jitter
            a += np.random.randn(n)*1e-7
            a = _project_onto_simplex(a)
            b = _fft_self_convolve(a)
            cur_max = b.max()
    return best_a

"""
Propose a low‑score step‑function.

The search works in four stages:

1.  Broad sampling of simple families (Arcsine, uniform, random,
    Beta‑densities, exponential) for several lengths and a cheap
    “two‑level’’ parametric family (a front block of one height and a
    back block of another).  The best fast‑score (2·n·max(conv))
    candidate is kept.

2.  Gradient optimisation of a smooth‐max surrogate of the
    ℓ∞‑convolution using Adam (continuation on the smoothness
    parameter).  The best vector obtained by Adam is retained.

3.  Exact‑max sub‑gradient descent combined with a directed local
    mass‑transfer search that focuses on the active peak of the
    convolution.  This stage receives the majority of the remaining
    time.

4.  Final evaluation with the true black‑box `evaluate_sequence`
    and return the best sequence found.

All vectors are projected onto the simplex Σa_i = 1 after every
update; they remain non‑negative, and the implementation respects the
time budget (default 1000 s).  The routine is completely self‑contained
and uses only NumPy.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered and its exact peak.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # continuation on the smoothness parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional small jitter
        if t % 400 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses an equal weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)  # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter to avoid stagnation
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _directed_local_search(a0: np.ndarray, deadline: float):
    """
    Mass‑transfer local search that always targets the current peak.
    A small fraction of mass is moved from a source index that
    participates heavily in the peak to a random destination.
    Accepts only improving moves (greedy descent).
    """
    a = a0.copy()
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        # locate one index belonging to a maximal convolution term
        max_idx = np.random.choice(np.where(b >= cur_max - 1e-12)[0])
        # possible i contributing to (a*a)[max_idx] : i in [low, high]
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[i_vals] * a[max_idx - i_vals]
        if prod.sum() == 0.0:
            src = np.random.randint(n)
        else:
            prob = prod / prod.sum()
            src = np.random.choice(i_vals, p=prob)

        # pick a destination different from src
        dst = np.random.randint(n)
        if dst == src:
            continue

        # move a modest fraction of the source mass
        max_delta = min(a[src], 0.05)          # up to 5 % of src
        delta = np.random.rand() * max_delta

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-15:         # accept only if improvement
            a = a_new
            b = b_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # occasional random jitter to explore neighbourhood
        if np.random.rand() < 0.005:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a, s=1.0)
            b = _fft_self_convolve(a)
            cur_max = b.max()

    return best_a


# ----------------------------------------------------------------------
def _two_level_family(n: int, deadline: float):
    """
    Search a simple two‑level family:
        a_i = p   for i < L   (front block)
        a_i = q   otherwise   (back block)
    with p = r·q, r≥1.  Returns the best vector found and its fast score.
    """
    best_seq = None
    best_score = float('inf')
    step_L = max(1, n // 100)          # step ≈ 1 % of the length
    # a modest set of ratios to explore (log‑space)
    r_grid = np.logspace(0.0, 2.3, num=25)   # from 1 to ≈200

    for L in range(step_L, n, step_L):
        # early stop if we are out of time
        if time.time() > deadline:
            break
        denom_base = n - L
        for r in r_grid:
            denom = L * r + denom_base
            q = 1.0 / denom
            p = r * q
            seq = np.empty(n, dtype=np.float64)
            seq[:L] = p
            seq[L:] = q
            # quick evaluate fast score
            peak = _fft_self_convolve(seq).max()
            score = 2.0 * n * peak
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score
