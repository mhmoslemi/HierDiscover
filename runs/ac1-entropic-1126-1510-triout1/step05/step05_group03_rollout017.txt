"""
Propose a low‑score step‑function.

The algorithm works in four phases:

1. **Broad exploratory search** – try many parametric families (uniform,
   arcsine, Beta, power‑law, mixtures) for several lengths and keep the best
   vector according to the fast surrogate 2·n·max(a*a).

2. **Adam optimisation of a smooth ℓ∞ surrogate** – minimise a
   log‑sum‑exp approximation of the peak of the self‑convolution while staying
   on the simplex.  The temperature (soft‑max parameter) is slowly increased up
   to 5000.

3. **Exact ℓ∞ sub‑gradient polishing** – use the true max‑convolution
   gradient (uniform weight on the active set) with a diminishing step size.

4. **Short simulated‑annealing polishing** – move tiny amounts of mass
   between random positions, accepting uphill moves according to an exponential
   cooling schedule.

All phases respect the constraints a_i ≥ 0, Σ a_i = 1.
The function returns a plain Python list suitable for `evaluate_sequence`.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(0.5,0.5)) distribution."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _power_distribution(n: int, p: float = 0.5, gamma: float = 1.0) -> np.ndarray:
    """
    Generalised arcsine‑like distribution:
    a_i ∝ ((i + p)·(n - i - 1 + p))^{-γ/2}.
    """
    i = np.arange(n, dtype=np.float64)
    denom = np.power((i + p) * (n - i - 1 + p), gamma / 2.0)
    vals = 1.0 / denom
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _mix_with_uniform(vec: np.ndarray, w: float) -> np.ndarray:
    """Convex combination (1‑w)*vec + w*uniform, w∈[0,1]."""
    n = vec.shape[0]
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    return (1.0 - w) * vec + w * uni


# ----------------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Fast proxy 2·n·max(self‑convolution); assumes Σvec = 1."""
    n = vec.shape[0]
    peak = _fft_self_convolve(vec).max()
    return 2.0 * n * peak


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """Gradient of L_α(a) = (1/α)·log Σ exp(α·(a*a)_k)."""
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """One sub‑gradient step for the exact ℓ∞ objective."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Exact ℓ∞ refinement using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """Simple SA polishing moving a tiny amount of mass between two positions."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    # temperature schedule (exponential decay)
    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative sequence to a new length with linear interpolation."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights with a low score.
    Returns a plain Python list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # reuse a previous construction if available
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = _project_onto_simplex(base_seq)
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 2000

    # ------------------------------------------------------------------
    # a handful of lengths to probe (avoid too small, allow a bit larger)
    lengths = [
        max(500, int(base_n * f))
        for f in (0.6, 0.8, 1.0, 1.2, 1.5)
        if int(base_n * f) >= 500
    ]
    if not lengths:
        lengths = [max(500, base_n)]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # ----- 1. BROAD EXPLORATORY SEARCH --------------------------------
    print("[phase 1] broad exploratory search")
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # simple deterministic starters
        candidates.append(("arcsine", _arcsine_distribution(n)))
        candidates.append(("uniform", np.full(n, 1.0 / n, dtype=np.float64)))
        # a few random points
        for k in range(3):
            r = np.random.rand(n)
            r /= r.sum()
            candidates.append((f"random{k}", r))

        # power‑law families (p offsets, γ exponents)
        for p in (0.3, 0.5, 0.8, 1.0, 2.0):
            for gamma in (0.7, 0.9, 1.0, 1.2):
                cand = _power_distribution(n, p, gamma)
                candidates.append((f"pow_p{p}_g{gamma}", cand))

        # symmetric Beta grid
        for a in (0.35, 0.45, 0.55, 0.65, 0.75):
            cand = _beta_distribution(n, a, a)
            candidates.append((f"beta_{a:.2f}", cand))

        # mixtures of arcsine with uniform
        for w in (0.1, 0.2, 0.3, 0.4):
            cand = _mix_with_uniform(_arcsine_distribution(n), w)
            candidates.append((f"arc_mix{w:.1f}", cand))

        # reuse previously found solution if we have one
        if base_seq is not None:
            cand = _interpolate_to_length(base_seq, n)
            candidates.append(("prev", cand))

        # evaluate each candidate (fast proxy)
        for tag, vec in candidates:
            vec = _project_onto_simplex(vec)
            score = _fast_score(vec)
            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = vec.copy()
                print(f"[init] n={n:4d} {tag:<15} fast_score={score:.6f}")

        # a few random perturbations around the current best for this length
        if overall_best_seq is not None:
            for _ in range(30):
                if time.time() >= deadline:
                    break
                pert = overall_best_seq + 0.01 * np.random.randn(n)
                pert = np.maximum(pert, 0.0)
                pert /= pert.sum()
                score = _fast_score(pert)
                if score < overall_best_score - 1e-6:
                    overall_best_score = score
                    overall_best_seq = pert.copy()

    # ------------------------------------------------------------------
    # Fallback if nothing has been produced (unlikely)
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _fast_score(overall_best_seq)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # ----- 2. ADAM OPTIMISATION ---------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        n = len(overall_best_seq)
        print(f"[phase 2] Adam optimisation – n={n} – time left {time_left:.1f}s")
        phase_deadline = time.time() + time_left * 0.55
        a_opt, _ = _adam_optimize(overall_best_seq, phase_deadline)

        # a short second Adam finetune (higher start temperature)
        phase_deadline2 = time.time() + time_left * 0.10
        a_opt, _ = _adam_optimize(a_opt, phase_deadline2,
                                  init_alpha=10.0,
                                  alpha_factor=1.005,
                                  alpha_max=5000.0,
                                  lr0=0.15)

        overall_best_seq = a_opt
        overall_best_score = _fast_score(overall_best_seq)
        print(f"[phase 2] post‑Adam fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # ----- 3. EXACT ℓ∞ SUB‑GRADIENT REFINEMENT ------------------------
    time_left = deadline - time.time()
    if time_left > 2.0:
        n = len(overall_best_seq)
        print(f"[phase 3] sub‑gradient refinement – n={n} – time left {time_left:.1f}s")
        phase_deadline = time.time() + time_left * 0.30
        overall_best_seq = _refine_max(overall_best_seq, phase_deadline)
        overall_best_score = _fast_score(overall_best_seq)
        print(f"[phase 3] after sub‑grad fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # ----- 4. SIMULATED ANNEALING POLISHING ---------------------------
    time_left = deadline - time.time()
    if time_left > 1.0:
        n = len(overall_best_seq)
        print(f"[phase 4] simulated annealing – n={n} – time left {time_left:.1f}s")
        a_final, final_peak = _simulated_annealing(overall_best_seq, deadline)
        overall_best_seq = a_final
        overall_best_score = 2.0 * n * final_peak
        print(f"[phase 4] SA fast_score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # ----- FINAL TRUE EVALUATION ---------------------------------------
    try:
        true_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] evaluate_sequence score={true_score:.6f}")
        if true_score < overall_best_score:
            overall_best_score = true_score
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    print(f"Best score (approx.): {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
