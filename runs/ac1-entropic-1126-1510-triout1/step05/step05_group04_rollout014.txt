"""
Propose a low‑score step‑function.

The algorithm is a hybrid of three ideas:

1️⃣ **Broad parametric sweep** – we generate many families of
   deterministic shapes (exponential, power‑law, classic windows …),
   normalise them to the required ℓ¹‑norm  √(2·n) and keep the best
   candidate.

2️⃣ **Smooth‑max L‑BFGS optimisation** – starting from a handful of
   diverse initial points we minimise a smooth approximation of the
   hard “max‑convolution’’ objective.
   The temperature μ is reduced slowly (μ←0.85·μ) and each temperature
   is solved with up to 500 L‑BFGS‑B iterations, giving the optimiser
   time to refine the shape.

3️⃣ **Hard‑max sub‑gradient refinement & cheap random perturbations** –
   from the best smooth‑max solution we run a projected sub‑gradient
   descent on the true max‑convolution objective (exact gradient
   computed from the active indices), followed by many random
   block‑perturbations.  All moves are projected onto the ℓ¹‑simplex
   (sum = √(2·n), non‑negative) using an Euclidean projection.

The whole routine respects a user supplied time budget (default
1000 s) and prints intermediate scores.  The returned list is a
non‑negative step‑function ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float):
    """
    Value and gradient of the smooth‑max surrogate:

        f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Parameters
    ----------
    y : ndarray, shape (n,)
        Unconstrained parameters (log‑odds of the simplex).
    mu : float
        Smoothing temperature (>0).  μ = 0 corresponds to the hard max.
    n : int
        Length of the step function.
    S : float
        Target ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float      – smooth‑max value.
    grad_y : ndarray   – gradient w.r.t. y.
    """
    # probabilities on the simplex (Σp=1, p≥0)
    p = _softmax(y)
    a = S * p                     # Σa = S

    # full convolution a*a  (length 2·n‑1)
    conv = np.convolve(a, a)

    # smooth‑max weights w_k ∝ exp((conv_k - max)/μ)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        # avoid overflow / underflow
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                         # hard max (rarely used)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a:
    #   g_a[i] = 2 * Σ_k w_k * a_{k-i}
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule back to y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                 # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable that returns (value, gradient) for scipy.optimize."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    ``evaluate_sequence`` score.  The routine respects the given time
    budget (seconds) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size --------------------------------------------------
    try:
        # if a reference from a previous run exists, use its length as a guide
        ref_len = len(height_sequence_1)   # type: ignore[name-defined]
        n = min(max(ref_len * 2, 4000), 9000)
        have_ref = True
    except Exception:
        n = 6000
        have_ref = False

    S = math.sqrt(2.0 * n)          # target ℓ¹‑norm (scale‑invariant)

    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # --------------------------------------------------------------
    def _maybe_update(a_vec: np.ndarray) -> None:
        """Evaluate a candidate and keep it if it improves the incumbent."""
        nonlocal best_score, best_a
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  PARAMETRIC SWEEP
    # --------------------------------------------------------------
    # Exponential decays (r very close to 1)
    for r in np.linspace(0.995, 0.9999, 210):
        if time.time() > deadline:
            break
        a = np.exp(np.log(r) * np.arange(n))
        a = a * (S / a.sum())
        _maybe_update(a)

    # Power‑law families
    for p in np.linspace(0.1, 3.0, 180):
        if time.time() > deadline:
            break
        i = np.arange(1, n + 1)
        a = i.astype(float) ** (-p)
        a = a * (S / a.sum())
        _maybe_update(a)

    # Classic windows (always non‑negative)
    i = np.arange(n, dtype=float)
    # Hann
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a = a * (S / a.sum())
    _maybe_update(a)
    # Hamming
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    a = a * (S / a.sum())
    _maybe_update(a)
    # Blackman
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = a * (S / a.sum())
    _maybe_update(a)

    # Simple linear decay (monotone)
    a = (n - i) / n
    a = a * (S / a.sum())
    _maybe_update(a)

    # Triangular shape (peak in the middle, then decreasing)
    a = np.minimum(i, n - 1 - i) + 1.0
    a = a[::-1]                     # flip to make it decreasing
    a = a * (S / a.sum())
    _maybe_update(a)

    # If a reference is available, seed it (after projection)
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        _maybe_update(a_ref)

    # --------------------------------------------------------------
    # 2️⃣  SMOOTH‑MAX L‑BFGS (multiple restarts)
    # --------------------------------------------------------------
    # Prepare a set of diverse starting points for y (log‑odds)
    init_ys = []

    # 0 – uniform (y = 0)
    init_ys.append(np.zeros(n, dtype=np.float64))

    # 1 – decreasing exponential (log‑space)
    alpha = rng.uniform(0.0004, 0.0025) * (n / 2000.0)
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    # 2 – increasing exponential
    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    # 3 – random Gaussian
    init_ys.append(rng.standard_normal(n) * 0.5)

    # 4 – sparse Dirichlet (α < 1)
    dir_alpha = rng.uniform(0.15, 0.55)
    p_dir = rng.dirichlet([dir_alpha] * n)
    eps = 1e-12
    init_ys.append(np.log(p_dir + eps))

    # 5 – best candidate so far (if any)
    if best_score < float("inf"):
        init_ys.append(np.log(best_a + eps))

    mu_factor = 0.85               # slower cooling than baseline
    mu_min = 1e-9
    maxiter_per_mu = 500           # more L‑BFGS iterations per temperature
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}] initialising")
        y = y0.copy()

        # initialise μ from a quick evaluation of the current max‑conv
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        mu = max(mu, 1.0)          # never start below 1

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False}
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor          # cool
        # end while mu

    # --------------------------------------------------------------
    # 3️⃣  HARD‑MAX SUB‑GRADIENT REFINEMENT
    # --------------------------------------------------------------
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        step0 = S * 0.45 / n          # heuristic initial step
        it = 0
        max_local_iters = 4000
        while it < max_local_iters and time.time() < deadline:
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = int(mask.sum())
            if cnt == 0:
                break
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt
            # sub‑gradient w.r.t. a
            grad_a = 2.0 * np.convolve(w, a[::-1])[n - 1:n - 1 + n]

            # simple diminishing step size
            step = step0 / (1.0 + 1e-5 * it)

            a_new = a - step * grad_a
            a_new = _project_to_simplex(a_new, S)
            sc = evaluate_sequence(a_new.tolist())
            if sc < best_score - 1e-9:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                print(f"[subgrad] iter {it:4d} step {step:.6e} => {best_score:.6f}")
            else:
                # occasional random jitter to escape flats
                i1, i2 = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a[i1] * 0.03
                a[i1] -= delta
                a[i2] += delta
                a = _project_to_simplex(a, S)
            it += 1

    # --------------------------------------------------------------
    # 4️⃣  RANDOM BLOCK PERTURBATIONS (cheap large‑scale search)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and best_score < float("inf"):
        max_trials = int(min(12000, remaining * 200))
        print(f"[block] performing {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(3, 12)                # number of blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = cand * (S / cand.sum())
            _maybe_update(cand)

    # --------------------------------------------------------------
    # 5️⃣  MONOTONE‑SORTING HEURISTIC (often gives a tidy improvement)
    # --------------------------------------------------------------
    if time.time() < deadline and best_score < float("inf"):
        desc = np.sort(best_a)[::-1]
        desc = _project_to_simplex(desc, S)
        _maybe_update(desc)

        asc = np.sort(best_a)
        asc = _project_to_simplex(asc, S)
        _maybe_update(asc)

    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
