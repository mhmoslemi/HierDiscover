"""
Propose a low‑score step‑function.

The search proceeds in two phases.

1. **Exploration** – a collection of analytic candidates (Beta /
   arcsine / skewed Beta, random Dirichlet, uniform, previous best) is
   generated for several lengths.  Each candidate is scored cheaply
   (the convolution is computed with an FFT, the objective is taken as
   2·n·max(conv) because the vectors are normalised to sum = 1).  The
   few best candidates of each length are kept for the second phase.

2. **Refinement** – the retained candidates are optimised with an
   Adam‑style multiplicative update that stays on the probability simplex.
   The surrogate loss is a soft‑max of the convolution, with a temperature
   that grows from a smooth regime to a hard max.  The update uses
   exponential re‑weighting (guaranteeing non‑negativity) and momentum
   adapts the effective step size.  Early‑stop and jitter are added to
   escape plateaus.  The best sequence seen (cheap score) is finally
   evaluated with the official ``evaluate_sequence`` call.

The algorithm respects a global time budget (default 1000 s) and returns
the best sequence found.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals

def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv); returns a vector of
    length n (the same size as a).  ``alpha`` is the temperature (larger → sharper).
    """
    n = a.shape[0]
    cur_max = conv.max()
    # stable soft‑max
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    # convolution of reversed a with the weight vector w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max (uniform weight on all active indices).
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (where i pairs with itself)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_update(a: np.ndarray,
                 grad: np.ndarray,
                 t: int,
                 m: np.ndarray,
                 v: np.ndarray,
                 lr: float,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Multiplicative Adam‑style update keeping a on the simplex.
    Returns (new_a, new_m, new_v).
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    factor = np.exp(-lr * m_hat / (np.sqrt(v_hat) + eps))
    a = a * factor
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        # fallback to uniform if we collapse
        a = np.full_like(a, 1.0 / a.size, dtype=np.float64)
    else:
        a = a / s
    return a, m, v


# ----------------------------------------------------------------------
def _initial_vector(kind: str, n: int) -> np.ndarray:
    """
    Generate a feasible starting vector of length n.
    ``kind`` can be "uniform", "random", "arcsine", "prev".
    """
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _beta_distribution(n, 0.5, 0.5)
    elif kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _beta_distribution(n, 0.5, 0.5)
    else:
        a = _beta_distribution(n, 0.5, 0.5)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """
    Fast surrogate of `evaluate_sequence` for a vector that already
    satisfies sum(a)=1 (the denominator becomes 1).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    global_best_seq: np.ndarray | None = None
    global_best_score = float("inf")
    global_best_cheap = float("inf")

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths to explore.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # exploration factors (cover a modest range around the base length)
    factors = (0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 2.5)
    lengths = sorted(
        set(
            max(500, int(base_n * f))
            for f in factors
            if int(base_n * f) >= 500
        )
    )
    if not lengths:
        lengths = [8000]

    # ------------------------------------------------------------------
    # Store promising candidates for a deep run later
    deep_candidates: list[tuple[float, np.ndarray, int]] = []

    # ------------------------------------------------------------------
    # Exploration phase (≈30 % of the total budget)
    explore_deadline = start_time + 0.30 * budget_s

    for n in lengths:
        if time.time() >= explore_deadline:
            break
        print(f"[{time.time():.1f}] Exploration – length {n}")

        # ----- generate a diverse pool of starters -----
        pool: list[np.ndarray] = []

        # symmetric Beta shapes
        shape_params = [
            (0.2, 0.2), (0.3, 0.3), (0.4, 0.4), (0.5, 0.5),
            (0.75, 0.75), (1.0, 1.0),
            (0.2, 0.8), (0.8, 0.2),
        ]
        for a, b in shape_params:
            pool.append(_beta_distribution(n, a, b))

        # random mixtures of two mirrored Betas
        for _ in range(3):
            a = np.random.uniform(0.1, 2.0)
            b = np.random.uniform(0.1, 2.0)
            w = np.random.rand()
            beta = _beta_distribution(n, a, b)
            mix = w * beta + (1.0 - w) * beta[::-1]
            mix /= mix.sum()
            pool.append(mix)

        # uniform / random / previous
        pool.append(_initial_vector("uniform", n))
        pool.append(_initial_vector("random", n))
        pool.append(_initial_vector("prev", n))

        # ----- cheap evaluation, keep the top‑k -----
        scored = [( _cheap_score(vec), vec) for vec in pool]
        scored.sort(key=lambda x: x[0])
        top_k = 4
        for cheap, vec in scored[:top_k]:
            if cheap < global_best_cheap:
                global_best_cheap = cheap
            deep_candidates.append((cheap, vec.copy(), n))

    # ------------------------------------------------------------------
    # 2. Deep refinement on the most promising candidates
    # allocate the remaining time fairly among the top few;
    # we keep at most 8 candidates to stay within the budget.
    deep_candidates.sort(key=lambda x: x[0])
    deep_candidates = deep_candidates[:8]

    for idx, (cheap_start, a0, n) in enumerate(deep_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.1f}] Deep refinement #{idx+1} – length {n}")

        # Allocate a sub‑budget for this candidate (≈60 s or the rest)
        now = time.time()
        remaining = deadline - now
        sub_budget = min(80.0, remaining / (len(deep_candidates) - idx))
        sub_deadline = now + sub_budget

        a = a0.copy()
        m = np.zeros_like(a)            # 1st moment for Adam
        v = np.zeros_like(a)            # 2nd moment for Adam
        beta1 = 0.9
        beta2 = 0.999

        # temperature schedule for the soft‑max surrogate
        temp = 1.0
        temp_factor = 1.01
        temp_max = 4000.0

        best_local_seq = a.copy()
        best_local_score = _cheap_score(a)
        iteration = 0

        while time.time() < sub_deadline:
            iteration += 1

            # ----- current convolution and objective (cheap) -----
            conv = _fft_self_convolve(a)
            cur_score = 2.0 * n * conv.max()

            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = a.copy()
                print(
                    f"[{time.time():.1f}] n={n} iter={iteration} "
                    f"best_local_score={best_local_score:.6f}"
                )

            # ----- gradient (soft‑max → exact as temperature grows) -----
            if temp < temp_max:
                grad = _softmax_gradient(a, conv, temp)
                temp = min(temp_max, temp * temp_factor)
            else:
                grad = _exact_gradient(a, conv)

            # ----- Adam multiplicative update -----
            lr = 0.7 / np.sqrt(iteration)   # base learning rate, decays
            a, m, v = _adam_update(a, grad, iteration, m, v, lr,
                                   beta1=beta1, beta2=beta2, eps=1e-8)

            # ----- occasional jitter to escape plateaus -----
            if iteration % 5000 == 0 and np.random.rand() < 0.2:
                a += np.random.randn(n) * 1e-6
                a = _project_onto_simplex(a, s=1.0)

            # early exit if we have already beaten a reasonable threshold
            if best_local_score <= 1.48:
                break

        # --------------------------------------------------------------
        # Evaluate the best local candidate with the true (slow) function
        try:
            true_score = evaluate_sequence(best_local_seq.tolist())
        except Exception:
            true_score = best_local_score

        if true_score < global_best_score:
            global_best_score = true_score
            global_best_seq = best_local_seq.copy()
            print(
                f"*** New global best (length={n}) → score={global_best_score:.6f}"
            )

        # Stop early if we hit the target bound
        if global_best_score <= 1.503:
            print("Target reached – terminating search.")
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a return value
    if global_best_seq is None:
        fallback_n = 1000
        global_best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(
            f"Fallback used (length={fallback_n}) → score={global_best_score:.6f}"
        )
    else:
        print(
            f"Final best score: {global_best_score:.6f} (length={len(global_best_seq)})"
        )

    return global_best_seq.tolist()
