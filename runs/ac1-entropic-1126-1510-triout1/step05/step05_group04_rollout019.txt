def build_sidon_set(n, m_target, rng):
    positions = []
    sums = set()
    max_trials_per_pos = 10000
    while len(positions) < m_target:
        for _ in range(max_trials_per_pos):
            idx = rng.integers(0, n)
            # ensure idx not already in positions
            if idx in positions:
                continue
            # candidate sums
            collides = False
            # compute sum for square
            s = idx + idx
            if s in sums:
                collides = True
            else:
                # check with each existing position
                for p in positions:
                    if idx + p in sums:
                        collides = True
                        break
            if collides:
                continue
            # succeed
            positions.append(idx)
            sums.add(s)
            for p in positions[:-1]:
                sums.add(p + idx)
            break
        else:
            # Could not find any position after many trials
            # break and reduce target
            return positions, False
    return positions, True
```

def propose_candidate(seed=12345, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5
    # Determine length
    n = 6000
    S = math.sqrt(2.0 * n)
    # Desired m
    m_target = int(math.floor(S))
    # attempt building set
    best_seq = None
    best_score = float('inf')
    while time.time() < deadline:
        positions, success = build_sidon_set(n, m_target, rng)
        if not success:
            # try smaller m
            m_target -= 1
            if m_target < 2:
                break
            continue
        # assign uniform weights
        w = S / len(positions)
        seq = np.zeros(n, dtype=float)
        seq[positions] = w
        # adjust residual to ensure sum exactly S
        residual = S - seq.sum()
        if abs(residual) > 1e-12:
            seq[positions[0]] += residual
        # evaluate
        score = evaluate_sequence(seq.tolist())
        print(f"[candidate] m={len(positions)} score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[improve] new best {best_score:.6f}")
        # Optionally try to improve by local adjustments between existing positions
        # but keep time budget.
        # break if score very low <1.01 maybe near optimum
        if best_score <= 1.02:
            break
    # Fallback: if no sequence found, use uniform sequence length maybe
    if best_seq is None:
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a low‑score step function.

The algorithm consists of three stages:

1.  Multi‑restart smooth‑max optimisation.
    We minimise a smoothed version of the hard max,
    f_μ = max(conv) + μ·log‑sum‑exp((conv‑max)/μ),
    using L‑BFGS‑B.  A geometric cooling schedule
    (μ←μ·0.7) drives μ→0, yielding an accurate
    solution of the original problem.

2.  Projected sub‑gradient descent.
    Starting from the best point of stage 1 we apply a
    rigorous sub‑gradient step for the true max‑objective,
    together with a back‑tracking line‑search and an
    aggressive iteration budget (up to several ten‑thousands
    of steps) while respecting the ℓ¹‑simplex.

3.  Large‑scale stochastic hill‑climbing.
    Random swaps, δ‑mass transfers and block‑averaging moves
    are tried for the remaining time budget.  Every move is
    accepted only if it strictly improves the exact score.
    A final interleaving‑sorting heuristic is also evaluated.

All helpers are top‑level functions; no closures or lambdas are used.
The routine respects a user‑supplied time budget (default 1000 s) and
returns the best height sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate:

        f_μ(y) = max(conv) + μ·log Σ exp((conv‑max)/μ)

    Parameters
    ----------
    y : ndarray (n,)
        Unconstrained parameters (log‑odds of the simplex).
    mu : float, >0 (μ=0 means the hard max, handled separately).
    n, S : problem size and target ℓ¹‑norm.

    Returns
    -------
    f_val : float   – smooth‑max value
    grad_y : ndarray (n,) – gradient w.r.t. y
    """
    # probabilities on the simplex, actual heights
    p = _softmax(y)                     # Σ p_i = 1, p_i ≥ 0
    a = S * p                           # Σ a_i = S

    conv = np.convolve(a, a)            # full auto‑convolution, length 2·n‑1

    # ---- smooth‑max part -------------------------------------------------
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        # guard against overflow in exp
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z                 # soft‑max weights (sum to 1)
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – uniform weight over all active indices
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # ---- gradient w.r.t. a ------------------------------------------------
    # g_a[i] = 2 * Σ_k w_k * a_{k‑i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize that returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of the true max‑objective  max(conv)  at point a.
    Returns (g, max_conv) where g is a vector of length n.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    if cnt == 0:                     # should not happen
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt            # uniform weight over active indices

    # gradient w.r.t. a (same expression as in the smooth case)
    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g, max_c

    init_ys = []

    alpha = rng.uniform(0.0005, 0.003) * (n / 2000.0)

    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

                    y,
                    method="L-BFGS-B",
                    jac=True,

                             "gtol": 1e-12,

            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    if time.time() < deadline and best_score < float("inf"):

            if sc < best_score - 1e-12:

                a = cand

    # ------------------------------------------------------------------
    # 4️⃣  Interleaving monotone‑sorting heuristic -----------------------
    if time.time() < deadline and best_score < float("inf"):

        desc = vals_desc
        _maybe_update(_project_to_simplex(desc, S))

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
