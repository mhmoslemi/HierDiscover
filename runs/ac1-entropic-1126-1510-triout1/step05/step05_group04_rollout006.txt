"""
Search for a non‑negative step‑function (list of heights) that minimizes the
provided `evaluate_sequence` score.

The algorithm consists of several phases:

1. **Parametric edge‑shape sweep** – a simple two‑exponential family
   a_i(λ,α)= (exp(-λ·i)+exp(-λ·(n‑1‑i)))^α is scanned on a grid of λ and α.
   The best shape is kept as the starting point.

2. **Smooth‑max L‑BFGS refinement** – the hard max objective
   max(conv(a,a)) is approximated by a smooth surrogate
   f_μ(a)=max+μ·log‑sum‑exp((conv‑max)/μ).  Starting from the best shape,
   μ is decreased geometrically and an L‑BFGS‑B optimisation (with analytic
   gradient) is performed at each temperature.

3. **Hard‑max sub‑gradient descent** – a true sub‑gradient of the hard
   max is computed (equal weights on all tying convolution entries) and a
   projected back‑tracking line search updates the sequence while staying on
   the ℓ¹‑simplex (Σa_i = √(2n)).

4. **Random block perturbations** – cheap random piecewise‑constant
   candidates are generated, projected onto the simplex and evaluated.

5. **Monotone sorting heuristics** – sorting the best sequence in
   descending/ascending order (followed by simplex projection) often
   improves the max‑convolution.

All steps respect the global time budget (`budget_s`).  The routine prints
progress messages and finally returns the best sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize as opt


def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _compute_smooth_obj_and_grad(
    y: np.ndarray, mu: float, n: int, S: float
) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate

        f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Parameters
    ----------
    y : ndarray, shape (n,)
        Log‑odds (unconstrained) of the simplex.
    mu : float (>0)
        Smoothing temperature.
    n : int
        Length of the step function.
    S : float
        Target ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : ndarray, shape (n,)
        Gradient w.r.t. y.
    """
    # probabilities on the simplex, then heights
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = S * p                # Σ a_i = S

    conv = np.convolve(a, a)          # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        # guard overflow / underflow
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:  # hard max – uniform weight on active indices
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. heights a: g_a[i] = 2 Σ_k w_k * a_{k-i}
    conv_wr = np.convolve(w, a[::-1])          # len 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule through the soft‑max
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable returning (value, gradient) for scipy.optimize."""
    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


def _hard_max_gradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the hard max objective max(conv(a,a)).
    Returns a vector of length n.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    w = np.zeros_like(conv, dtype=float)
    w[mask] = 1.0 / cnt
    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n - 1:n - 1 + n]
    return g


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """Search for a low‑score step function under the given time budget."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5     # safety margin

    # ----------------------------------------------------------------------
    # Problem size
    try:
        # reference sequence supplied by the environment (if any)
        ref_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n_ref = len(ref_seq)
        n = min(max(n_ref * 2, 4000), 9000)
        have_ref = True
    except Exception:
        n = 6000
        have_ref = False

    S = math.sqrt(2.0 * n)            # target ℓ¹‑norm (scale‑invariant)

    # ----------------------------------------------------------------------
    # State
    best_a = np.empty(n, dtype=float)
    best_score = float("inf")

    # ----------------------------------------------------------------------
    # Helper to evaluate and keep the best incumbent
    def _maybe_update(candidate: np.ndarray) -> None:
        nonlocal best_score, best_a
        sc = evaluate_sequence(candidate.tolist())
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 0. Uniform baseline (useful for sanity)
    uniform = np.full(n, S / n, dtype=float)
    _maybe_update(uniform)

    # ----------------------------------------------------------------------
    # 1. Parametric edge‑shape sweep
    # shape_i(λ,α) = (exp(-λ·i) + exp(-λ·(n‑1‑i)))^α
    lam_grid = np.geomspace(1e-4, 0.5, 30)          # from very flat to steep
    for lam in lam_grid:
        if time.time() > deadline:
            break
        idx = np.arange(n, dtype=float)
        base = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
        for alpha in (1.0, 1.5, 2.0):
            if time.time() > deadline:
                break
            shape = base ** alpha
            p = shape / shape.sum()
            a = p * S
            _maybe_update(a)

    # ----------------------------------------------------------------------
    # 2. Smooth‑max refinement (L‑BFGS) starting from the current best
    if best_score < float("inf"):
        # initialise log‑odds from the best vector
        y = np.log(best_a + 1e-12)

        # initial smoothing parameter (large enough to make the problem smooth)
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        mu = max(mu, 1.0)    # avoid a zero temperature

        mu_factor = 0.85
        mu_min = 1e-9
        maxiter_per_mu = 400

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    # ----------------------------------------------------------------------
    # 3. Hard‑max sub‑gradient descent with back‑tracking line search
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        it = 0
        max_local_iters = 4000
        step0 = S * 0.5 / n   # heuristic initial step size

        while time.time() < deadline and it < max_local_iters:
            grad = _hard_max_gradient(a)
            step = step0 / (1.0 + 1e-5 * it)
            improved = False

            for _ in range(12):
                a_new = a - step * grad
                a_new = _project_to_simplex(a_new, S)
                sc = evaluate_sequence(a_new.tolist())
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[subgrad] iter {it:4d} step {step:.6e} => {best_score:.6f}")
                    break
                step *= 0.5

            if not improved:
                # small random jitter to escape potential plateaus
                i, j = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a[i] * 0.05
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, S)

            it += 1

    # ----------------------------------------------------------------------
    # 4. Random block perturbations (cheap large‑scale search)
    remaining = deadline - time.time()
    if remaining > 2.0 and best_score < float("inf"):
        # a very rough estimate: ~30 evaluations per second for n~6000
        max_trials = int(min(8000, remaining * 30))
        print(f"[block] performing {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(3, 12)                     # number of blocks
            # split the index set into nb contiguous blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)                         # raw block heights
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # ----------------------------------------------------------------------
    # 5. Monotone‑sorting heuristics (often beneficial)
    if time.time() < deadline and best_score < float("inf"):
        desc = np.sort(best_a)[::-1]
        desc = _project_to_simplex(desc, S)
        _maybe_update(desc)

        asc = np.sort(best_a)
        asc = _project_to_simplex(asc, S)
        _maybe_update(asc)

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
