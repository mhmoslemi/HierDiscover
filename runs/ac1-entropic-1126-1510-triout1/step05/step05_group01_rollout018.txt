Algorithm
---------

   better asymptotic constant but costs more per iteration.

   * symmetric and asymmetric Beta(α,β) densities,
   * a few random Dirichlet draws,
   * the previous construction (if present),
   * uniform and simple exponential shapes.
   The pool size is modest (≈ 40) and each candidate is cheap‑scored
   via an FFT based self‑convolution (the “cheap score” is
   `2·n·max(conv)` which equals the true objective when the vector
   is normalised to sum = 1).
3. Keep the best few (≤ 5) candidates per length and **refine** them with a
   **projected Adam optimiser** on the simplex:
   * The optimisation variable is `z∈ℝⁿ` with `a = softmax(z)`.
   * A smooth surrogate of the `max` (soft‑max with a temperature that
     grows geometrically) supplies a gradient that is cheap to compute
     via FFTs.
   * After a certain temperature the exact sub‑gradient of the true
     maximum is used.
   * Adam’s adaptive step‑sizes together with momentum give fast
     convergence without hand‑tuned learning‑rate schedules.
4. After each inner optimisation the best local vector is evaluated
   with the real `evaluate_sequence`.  The globally best vector seen so
   far is returned when the time budget (≈ 1000 s) expires or when the
   target 1.503 is reached.

The whole search respects the time limit and prints progress messages.

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 1.0) -> np.ndarray:
    """Geometric‑like decreasing shape: a_i ∝ exp(-lam·i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    a : current probability vector (sum = 1)
    conv : self‑convolution of a
    alpha : temperature multiplier (larger → sharper)
    """
    n = a.shape[0]
    cur_max = conv.max()
    # stable soft‑max
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    # convolution of reversed a with the weights w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max, i.e. average over all active indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (remove double counting of a_i * a_i for even shifts)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _adam_step(z: np.ndarray,
               g: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.5,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on the unconstrained variable z."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


# ----------------------------------------------------------------------
def _initial_vector(kind: str, n: int) -> np.ndarray:
    """Return a feasible probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _beta_distribution(n, 0.5, 0.5)
    elif kind == "exp":
        a = _exponential_distribution(n, lam=1.0)
    elif kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _beta_distribution(n, 0.5, 0.5)
    else:
        a = _beta_distribution(n, 0.5, 0.5)

    a = _project_onto_simplex(a, s=1.0)
    return a

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 0. base length (use previous result if available)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # 1. lengths to explore (a handful around the base length)
    length_factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted(
        max(3000, int(base_n * f))
        for f in length_factors
        if int(base_n * f) >= 3000
    )
    # ensure we have at least one length
    if not lengths:
        lengths = [base_n]

    # ------------------------------------------------------------------
    # helper cheap scoring (normed to Sum=1 → score = 2·n·max(conv))
    def cheap_score(a: np.ndarray) -> float:
        return 2.0 * a.shape[0] * _fft_self_convolve(a).max()

    overall_best_seq = None               # type: np.ndarray | None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        remaining = max(0.0, deadline - time.time())
        slice_deadline = time.time() + min(remaining * 0.35, 200.0)
        print(f"[{time.time():.1f}] Exploring length n={n}")

        # --------------------------------------------------------------
        # 2. build a large pool of analytic candidates
        pool = []

        # systematic grid on (α,β) in log‑space
        alphas = np.logspace(-0.7, 0.7, num=7)   # approx 0.2 → 5
        betas = np.logspace(-0.7, 0.7, num=7)
        for a in alphas:
            for b in betas:
                pool.append(_beta_distribution(n, a, b))

        # few extra manually‑chosen shapes
        pool.append(_beta_distribution(n, 0.2, 0.8))
        pool.append(_beta_distribution(n, 0.8, 0.2))
        pool.append(_exponential_distribution(n, lam=0.5))
        pool.append(_exponential_distribution(n, lam=1.5))
        pool.append(_initial_vector("prev", n))
        pool.append(_initial_vector("uniform", n))
        pool.append(_initial_vector("random", n))
        pool.append(_initial_vector("arcsine", n))

        # cheap scores and keep the best handful
        scored = [(cheap_score(a), a) for a in pool]
        scored.sort(key=lambda x: x[0])
        starters = [a.copy() for _, a in scored[:5]]
        print(f"  top cheap scores: {[s for s, _ in scored[:5]]}")

        # --------------------------------------------------------------
        # 3. refine each starter with Adam on the simplex
        for sid, a0 in enumerate(starters):
            now = time.time()
            sub_deadline = now + min(80.0, slice_deadline - now)
            if sub_deadline - now < 5.0:
                continue

            # initialise Adam on log‑odds (softmax) variables
            a = a0.copy()
            a = np.maximum(a, 1e-15)        # avoid log(0)
            z = np.log(a)                   # unconstrained
            m = np.zeros_like(z)
            v = np.zeros_like(z)

            # optimisation hyper‑parameters
            beta1, beta2 = 0.9, 0.999
            eps = 1e-8
            lr0 = 0.6

            temperature = 1.0               # soft‑max multiplier
            temp_factor = 1.003
            temp_max = 5000.0

            best_local_a = a.copy()
            best_local_score = cheap_score(a)
            it = 0

            while time.time() < sub_deadline:
                it += 1

                # current probability vector and convolution
                a = np.exp(z)
                a /= a.sum()
                conv = _fft_self_convolve(a)

                # cheap score for monitoring
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = a.copy()
                    print(
                        f"[{time.time():.1f}] n={n} starter={sid} it={it} score={cur_score:.6f}"
                    )
                    # early stop if we have already crossed the target
                    if best_local_score <= 1.50:
                        break

                # gradient (soft‑max → exact)
                if temperature < temp_max:
                    grad_a = _softmax_gradient(a, conv, temperature)
                    temperature *= temp_factor
                else:
                    grad_a = _exact_gradient(a, conv)

                # convert gradient w.r.t a to gradient w.r.t z
                # ∂f/∂z_i = a_i (∂f/∂a_i - Σ_j a_j ∂f/∂a_j)
                dot = np.dot(a, grad_a)
                grad_z = a * (grad_a - dot)

                # Adam step (t is 1‑based)
                z, m, v = _adam_step(z, grad_z, m, v, it,
                                     lr=lr0, beta1=beta1, beta2=beta2, eps=eps)

                # occasional jitter to escape plateaus
                if it % 4000 == 0 and np.random.rand() < 0.12:
                    jitter = np.random.randn(n) * 1e-7
                    z += jitter

            # ----------------------------------------------------------
            # evaluate the best local candidate with the real black‑box
            try:
                final_score = evaluate_sequence(best_local_a.tolist())
            except Exception:
                final_score = best_local_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_a.copy()
                print(
                    f"*** Global improvement: n={n} starter={sid} → "
                    f"score={overall_best_score:.6f}"
                )
                if overall_best_score <= 1.503:
                    print("Target reached – exiting early.")
                    return overall_best_seq.tolist()

            # respect per‑length deadline
            if time.time() >= slice_deadline:
                break

        # global time check
        if time.time() >= deadline:
            break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # fallback (should never be needed)
    if overall_best_seq is None:
        n_fallback = 4000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback used.")

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
