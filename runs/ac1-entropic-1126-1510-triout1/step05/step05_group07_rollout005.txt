"""
Improved step‑function search for the `evaluate_sequence` objective.

Main ideas
----------
1. **Richer candidate families** – In addition to the uniform,
   arcsine and symmetric Beta families we explore fine‑grained mixtures
   of arcsine with a uniform background and with explicit spikes at the
   two boundaries.  Parameter grids are dense enough to catch the
   narrow optimum near the theoretical lower bound (≈1.5).

2. **Multiple Adam restarts** – A projected Adam optimiser on the
   probability simplex minimises a smooth approximation of the
   ``max`` convolution (log‑sum‑exp with an exponentially growing
   temperature).  Three short restarts from the current best shape
   allow us to escape shallow basins.

3. **Long hill‑climbing polish** – A cheap stochastic local search
   (mass swaps, jitter) works on the exact objective and usually
   extracts the last few digits of improvement.

The code respects the 1000 s budget, runs on a single CPU (the
environment may give two), makes no file or network accesses and
returns a plain list of non‑negative heights ready for the grader.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·b_k) w.r.t. a.
    For huge α we fall back to a uniform weight on all maximal entries.
    """
    n = a.shape[0]

    # soft‑max weights
    if alpha > 2_000:
        # treat every numerically maximal entry as active
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # convolution of the weight vector with reversed a
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]      # factor 2 from derivative of a*a
    return g


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) pdf evaluated on mid‑points of n bins."""
    a = max(a, 0.05)                     # avoid singularities
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    s = pdf.sum()
    if s == 0.0 or not np.isfinite(s):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= s
    return pdf

def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified list of hand‑crafted probability vectors."""
    cand = []

    # 1) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) arcsine (Beta(½,½))
    arcsine = _arcsine_distribution(n)
    cand.append(arcsine)

    # 3) symmetric Beta(a,a) for a fine grid
    for a in np.geomspace(0.05, 1.0, num=15):
        cand.append(_beta_distribution(n, a))

    # 4) mixtures arcsine + uniform (weight w on uniform)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for w in np.linspace(0.0, 0.8, 9):
        mix = (1.0 - w) * arcsine + w * uniform
        cand.append(mix)

    # 5) arcsine with spikes at both ends – scan spike fraction
    for spike_frac in np.linspace(0.0, 0.5, 21):
        mix = arcsine.copy()
        add = spike_frac / 2.0
        mix[0] += add
        mix[-1] += add
        mix = _project_onto_simplex(mix, 1.0)
        cand.append(mix)

    # 6) arcsine + uniform + spikes (three‑parameter grid, limited size)
    for w in np.linspace(0.0, 0.5, 5):
        base = (1.0 - w) * arcsine + w * uniform
        for spike_frac in (0.0, 0.15, 0.30):
            mix = base.copy()
            add = spike_frac / 2.0
            mix[0] += add
            mix[-1] += add
            mix = _project_onto_simplex(mix, 1.0)
            cand.append(mix)

    # 7) a few random Dirichlet draws (different concentration)
    rng = np.random.default_rng()
    for conc in (0.1, 0.3, 0.7, 2.0):
        for _ in range(2):
            rand = rng.dirichlet(conc * np.ones(n))
            cand.append(rand)

    # 8) the previous best solution, if it matches the length
    if prev_seq is not None and prev_seq.size == n:
        cand.append(prev_seq.copy())

    # final safety – guarantee simplex
    final = [_project_onto_simplex(c, 1.0) for c in cand]
    return final

def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the simplex using a log‑sum‑exp
    surrogate for the max‑convolution.
    Returns the best vector seen (on the true objective) and its
    internal surrogate score.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # surrogate temperature parameters
    alpha = 5.0
    alpha_mul = 1.04
    alpha_max = 1e5

    size_fft = 1 << ((3 * n - 2).bit_length())   # ≥ (2n‑1)+(n)‑1

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] it={it:,} score={best_score:.6f}")

        # gradient of smoothed max
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the simplex
        a = _project_onto_simplex(a, 1.0)

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score

def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic local search on the exact objective.
    Moves a random fraction of mass from one index to another.
    Occasionally adds tiny Gaussian jitter to escape plateaus.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue

        # move up to 40 % of the mass at i to j
        frac = rng.random() * 0.40
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter (every 4000 iterations)
        if it % 4000 == 0:
            jitter = best + rng.normal(scale=5e-7, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point required by the grader.
    Returns a list of non‑negative heights (probability vector) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # Choose a problem size.  Prefer the length of a previously known
    # solution (height_sequence_1) but stay within a safe interval.
    try:
        base_n = int(len(height_sequence_1))  # type: ignore[name-defined]
    except Exception:
        base_n = 8000
    n = max(2500, min(base_n, 15000))

    # --------------------------------------------------------------
    # Phase 0 – quick hand‑crafted candidates
    prev_seq = None
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        pass

    cand_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.20)
    candidates = _candidate_shapes(n, prev_seq)

    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        if time.time() >= cand_deadline:
            break
        scr = evaluate_sequence(cand.tolist())
        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
            print(f"[candidate] new best {best_score:.6f}")

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    # --------------------------------------------------------------
    # Phase 1 – several Adam runs (≈60 % of remaining time)
    time_left = deadline - time.time()
    adam_time = max(0.0, time_left * 0.60)
    runs = 3
    per_run = adam_time / runs if runs > 0 else 0.0

    current_best = best_seq.copy()
    current_best_score = best_score

    for r in range(runs):
        if time.time() >= deadline:
            break
        # perturb the start for runs > 0
        init = current_best.copy()
        if r > 0:
            init = init + np.random.normal(scale=0.02, size=n)
            init = _project_onto_simplex(init, 1.0)

        run_deadline = time.time() + per_run
        seq_opt, _ = _adam_optimize(init, n, run_deadline)

        # true evaluation (no surrogate)
        true_score = evaluate_sequence(seq_opt.tolist())
        if true_score < current_best_score:
            current_best_score = true_score
            current_best = seq_opt.copy()
            print(f"[Adam run {r}] improved to {current_best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – final hill‑climbing polish
    hill_deadline = deadline
    seq_final, final_score = _hill_climb(current_best, n, hill_deadline)

    # a final safety call (should never be needed)
    if final_score is None or math.isinf(final_score):
        seq_final = np.full(2000, 1.0 / 2000, dtype=np.float64)
        final_score = evaluate_sequence(seq_final.tolist())

    print(f"Final best score: {final_score:.6f} (length={len(seq_final)})")
    return seq_final.tolist()
