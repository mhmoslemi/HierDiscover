"""
Propose a low‑score step‑function.

This implementation builds on the previous Beta‑family search but
adds two major upgrades:

1.  A richer initial population (uniform, arcsine, symmetric Betas,
    random Dirichlet draws and the previous best solution interpolated
    to several lengths).

2.  A fast *exponentiated‑gradient* refinement that minimises a smooth
    approximation of the ℓ∞ self‑convolution objective.
    The smooth max is realised by a temperature‑controlled soft‑max
    over the convolution values; the temperature is increased
    gradually so that the algorithm first follows a smooth landscape
    and then concentrates on the true max.  The update is performed
    in the multiplicative (mirror‑descent) domain and projected back
    onto the probability simplex at every step.  Adaptive step‑size
    (η/√t) and occasional jitter help escape plateaus.

The whole search runs under a strict 1000 s wall‑time budget and
returns the best sequence discovered (as a plain Python list)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow/underflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()    # next power of two >= 2*n-1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum value of a * a (self‑convolution)."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Raw objective 2·n·max(a*a) for a probability vector."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)


# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over b with temperature 1/α (larger α ⇒ sharper)."""
    # stabilise by subtracting max before exponentiation
    b_max = b.max()
    exps = np.exp(alpha * (b - b_max))
    s = exps.sum()
    if s == 0.0 or np.isnan(s):
        # fallback to uniform weights
        return np.full_like(b, 1.0 / b.size)
    return exps / s

def _refine_exponential(a0: np.ndarray,
                        deadline: float,
                        alpha_start: float = 30.0,
                        alpha_max: float = 5000.0,
                        eta0: float = 0.25) -> np.ndarray:
    """
    Refine a probability vector using exponentiated‑gradient descent.
    The objective is a smooth approximation of max(a*a).  The routine
    returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    t = 0
    alpha = alpha_start

    # pre‑allocate reversed a for convolution (updated each iteration)
    while time.time() < deadline:
        t += 1

        # 1) self‑convolution
        b = _fft_self_convolve_full(a)

        # 2) smooth max weights
        w = _softmax_weights(b, alpha)

        # 3) gradient of the smoothed max
        a_rev = a[::-1]
        grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]

        # 4) multiplicative (mirror‑descent) update
        step = eta0 / math.sqrt(t)            # diminishing step size
        a = a * np.exp(-step * grad)          # element‑wise
        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
            s = 1.0
        a = a / s

        # 5) keep the best
        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] refine n={n} it={t} score={best_val:.6f}")

        # 6) schedule for the temperature (α) – increase slowly
        if t % 1500 == 0:
            alpha = min(alpha * 1.15, alpha_max)

        # 7) occasional jitter to escape plateaux
        if t % 5000 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 5e-7
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a

            a1, b1 = np.random.rand()*5.0 + 0.1, np.random.rand()*5.0 + 0.1
            a2, b2 = np.random.rand()*5.0 + 0.1, np.random.rand()*5.0 + 0.1
            lam = np.random.rand()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_score, best_seq, best_len = sc, vec.copy(), n
                print(f"   mix λ={lam:.2f} ({a1:.2f},{b1:.2f})+({a2:.2f},{b2:.2f}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If for any reason no candidate was generated, fall back to uniform.
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_len = n
        best_score = _raw_score(best_seq)
        print("Fallback uniform used.")

    print(f"Best raw score after population search: {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # 2) Local refinement on the *best* length with the exponentiated gradient.
    #    Use the remaining time budget.
    refine_deadline = deadline
    a0 = best_seq.copy()
    refined = _refine_exponential(a0, refine_deadline,
                                  alpha_start=30.0,
                                  alpha_max=5000.0,
                                  eta0=0.25)

    # ------------------------------------------------------------------
    # 3) Final evaluation with the provided black‑box function.
    final_score = evaluate_sequence(refined.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(refined)})")
    return refined.tolist()
