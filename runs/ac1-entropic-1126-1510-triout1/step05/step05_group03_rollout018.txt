def fun(z):
    a = np.exp(z - logsumexp(z))  # ensures sum=1 automatically.
    b = _fft_self_convolve(a)
    b_max = b.max()
    # compute softmax weights
    diff = b - b_max
    exp_term = np.exp(alpha * diff)
    w_sum = exp_term.sum()
    if w_sum == 0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w = exp_term / w_sum
    # compute gradient wrt a
    a_rev = a[::-1]
    grad_a = 2.0 * _fft_convolve(a_rev, w)[n-1:n-1+n]
    # gradient wrt z
    dot_ag = np.dot(a, grad_a)
    g_z = a * (grad_a - dot_ag)
    # smooth max value:
    smooth_val = (np.log(w_sum) / alpha) + b_max

#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Low‑score step‑function generator.

The idea: for a probability vector a with Σa=1 the evaluation reduces to
        2·n·max(a∗a) .
A simple family a_i ∝ (1−i/n)^p (i=0,…,n−1) has an analytically tractable
autocorrelation.  Choosing p≈log₄(n) makes the peak of the self‑convolution
scale like (p+1)²/n, so the final score ≈2·(p+1)²/n becomes very small.
The routine picks a modest size n (so that the exact np.convolve used by
evaluate_sequence stays cheap) and searches a few integer p around the
theoretically optimal value.  The best vector is returned as a plain list.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Produce a step‑function (list of non‑negative heights) with a very low
    evaluate_sequence score.  The method builds a power‑law decreasing
    distribution a_i ∝ (1−i/n)^p, tries a few p around log₄(n) and keeps the
    best one.  The final vector is returned as a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a length that keeps the exact O(n²) convolution cheap.
    # n≈5000 gives a fast exact evaluation (< 1 s) while still allowing a
    # very small score.
    n = 5000
    # Theoretical sweet‑spot: p ≈ log₄(n) = ln(n)/ln(4)
    p_opt = max(1, int(round(np.log(n) / np.log(4))))

    best_a = None
    best_fast_score = float("inf")

    print("[search] length =", n, "optimal p ≈", p_opt)

    # Search a small neighbourhood of p_opt
    for p in range(max(1, p_opt - 2), p_opt + 3):
        if time.time() >= deadline:
            break

        # power‑law decreasing shape, normalised to sum=1
        idx = np.arange(n, dtype=np.float64)
        a = (1.0 - idx / n) ** p
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum

        # fast proxy: 2·n·max(conv) (exact max can be obtained cheaply with FFT)
        max_conv = _fft_self_convolve(a).max()
        fast_score = 2.0 * n * max_conv

        print(f"   p={p:<2d} → fast_score={fast_score:.6f}")

        if fast_score < best_fast_score:
            best_fast_score = fast_score
            best_a = a.copy()

    # ------------------------------------------------------------------
    # Final exact evaluation (uses the black‑box routine).
    final_score = None
    if best_a is not None:
        try:
            final_score = evaluate_sequence(best_a.tolist())
            print("[final] evaluate_sequence score =", final_score)
        except Exception as exc:
            print("[final] evaluate_sequence raised:", exc)

    # ------------------------------------------------------------------
    # Return the best vector we have (either the exact‑evaluated one or the
    # best proxy if the black‑box failed for some reason).
    result = best_a if best_a is not None else np.full(n, 1.0 / n)
    return result.tolist()
