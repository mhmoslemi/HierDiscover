def _project_to_simplex(v, z):
    # Euclidean projection onto simplex
    v = np.maximum(v, 0)
    if z == 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # rare case; set all mass on max element
        w = np.zeros_like(v); w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    n = a.shape[0]
    # compute convolution via FFT
    size = 1
    while size < 2*n:
        size <<= 1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    conv = conv[:2*n-1]
    max_val = conv.max()
    # mask for indices near max (tolerance)
    # Use relative tolerance maybe 1e-12
    mask = conv >= max_val - 1e-12
    cnt = mask.sum()
    w = np.zeros_like(conv)
    if cnt > 0:
        w[mask] = 1.0 / cnt
    # compute gradient: g = 2 * (w * a reversed) conv
    # w (len 2n-1) conv with a_rev (len n)
    a_rev = a[::-1]

    fa_rev = np.fft.rfft(a_rev, n=size2)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size2)
    # extract central part
    start = n-1
    g = 2.0 * conv_wr[start:start+n]
    return g, max_val

"""
Propose a low–score step function.

Algorithm
---------
1.  Choose a length n (≈ 6‑9k) and the target ℓ¹‑norm S = √(2·n).
2.  Generate a diverse set of initial candidates:
    * uniform, exponential, reverse‑exponential,
    * Dirichlet samples with various concentrations,
    * the reference sequence if it exists,
    * a quick random search (hundreds of Dirichlet draws) evaluated
      with a fast FFT‑based convolution max.
    The best candidate (in terms of the cheap max‑convolution) is kept.
3.  Refine the candidate by sub‑gradient descent on the hard max
    objective.  For a current vector a we:
    * compute the auto‑convolution via FFT,
    * identify the indices attaining the current maximum,
    * build a uniform weight vector w on those indices,
    * obtain the sub‑gradient g = 2·(w ∗ a_rev) (again by FFT),
    * take a step a ← a – η·g and project back onto the simplex
      {x ≥ 0, Σx = S}.
    The step size η decays slowly.  The cheap max‑convolution is used
    to accept/reject a move; the exact score is evaluated only
    periodically (every few hundred iterations) to keep the runtime
    low.
4.  After the descent, perform a cheap large‑scale random block
    perturbation phase (splitting the index set into 3‑15 blocks,
    assigning a random value to each block, projecting onto the simplex
    and keeping improvements).
5.  Finally try monotone sorting heuristics (descending and ascending
    order) which often improve the max‑convolution.
6.  Throughout the search the best true score (as reported by
    `evaluate_sequence`) and its corresponding vector are remembered.
   The function returns that best vector as a plain Python list.

All heavy inner loops use FFT‑based convolution (O(n log n)), so a very
large number of iterations fits comfortably into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    # make a copy because we will modify it
    v = np.asarray(v, dtype=float).copy()
    # ensure non‑negativity (helps numerical stability)
    v = np.maximum(v, 0.0)

    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – put all mass on the largest entry
        w = np.zeros_like(v)
        w[np.argmax(v)] = z
        return w
    rho = np.where(cond)[0][-1] + 1      # index of last positive
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _max_conv_fft(a: np.ndarray) -> float:
    """
    Fast (FFT) computation of max_{k} (a * a)_k.
    a must be a 1‑D non‑negative array.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    conv = conv[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray):
    """
    Compute a sub‑gradient of the hard‑max objective
    f(a) = max(conv(a,a)) together with the current max value.
    Returns (g, f_val) where g has the same shape as a.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    conv = conv[: 2 * n - 1]
    f_val = float(conv.max())

    # indices where the max is achieved (tolerance for FP noise)
    eps = 1e-12
    mask = conv >= f_val - eps
    cnt = mask.sum()
    w = np.zeros_like(conv, dtype=float)
    if cnt > 0:
        w[mask] = 1.0 / cnt

    # sub‑gradient g = 2 * (w ⋆ a_rev)
    a_rev = a[::-1]
    size2 = _next_pow2(len(w) + n)
    fw = np.fft.rfft(w, n=size2)
    fa_rev = np.fft.rfft(a_rev, n=size2)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size2)
    # central part corresponds to indices n‑1 … n‑1+n‑1
    start = n - 1
    g = 2.0 * conv_wr[start : start + n]
    return g, f_val


# ----------------------------------------------------------------------
def _maybe_update_best(a: np.ndarray, best):
    """
    Evaluate the true objective with evaluate_sequence().
    If it improves, update the best container.
    `best` is a dict with keys 'a' and 'score'.
    """
    sc = evaluate_sequence(a.tolist())
    if sc < best["score"]:
        best["score"] = sc
        best["a"] = a.copy()
        print(f"[improve] new best true score = {sc:.6f}")
    return best


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a
    low `evaluate_sequence` score.  The routine respects the supplied
    time budget and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Determine problem size
    try:
        ref_len = len(height_sequence_1)  # type: ignore[name-defined]
        n = min(max(ref_len * 2, 4000), 9000)
        have_ref = True
    except Exception:
        n = 6000
        have_ref = False

    S = math.sqrt(2.0 * n)               # target ℓ¹‑norm

    # --------------------------------------------------------------
    # Container for the best true solution seen so far
    best = {"a": np.zeros(n, dtype=float), "score": float("inf")}

    # --------------------------------------------------------------
    # Helper to generate a simplex‑projected vector from a raw positive
    # array (used for deterministic initialisations).
    def _from_raw(x_raw):
        return _project_to_simplex(x_raw, S)

    # --------------------------------------------------------------
    # Deterministic diverse initial candidates
    init_candidates = []

    # Uniform (on the simplex)
    init_candidates.append(_from_raw(rng.random(n)))

    # Decreasing exponential
    alpha = rng.uniform(0.001, 0.005) * (n / 2000.0)
    init_candidates.append(_from_raw(np.exp(-alpha * np.arange(n))))

    # Increasing exponential (large values on the right)
    init_candidates.append(_from_raw(np.exp(-alpha * np.arange(n - 1, -1, -1))))

    # Random Dirichlet‑like vectors (different concentration levels)
    for conc in (0.2, 0.5, 1.0):
        gamm = rng.gamma(shape=conc, scale=1.0, size=n)
        init_candidates.append(_from_raw(gamm))

    # Reference sequence, padded or cropped to length n
    if have_ref:
        ref = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if ref.shape[0] < n:
            pad = n - ref.shape[0]
            left = pad // 2
            right = pad - left
            ref = np.concatenate([np.zeros(left), ref, np.zeros(right)])
        elif ref.shape[0] > n:
            start = (ref.shape[0] - n) // 2
            ref = ref[start : start + n]
        init_candidates.append(_project_to_simplex(ref, S))

    # --------------------------------------------------------------
    # Quick random Dirichlet search (cheap FFT evaluation only)
    cheap_best_val = float("inf")
    cheap_best_a = None
    random_search_time = min(30.0, budget_s / 4.0)  # spend at most 30 s or ¼ of budget
    rnd_start = time.time()
    while time.time() - rnd_start < random_search_time and time.time() < deadline:
        # Random concentration between 0.1 and 2.0
        conc = rng.uniform(0.1, 2.0)
        g = rng.gamma(shape=conc, scale=1.0, size=n)
        a = _project_to_simplex(g, S)
        cheap_val = _max_conv_fft(a)
        if cheap_val < cheap_best_val:
            cheap_best_val = cheap_val
            cheap_best_a = a
            # Occasionally evaluate the true objective (once every 0.5 s)
            if time.time() - rnd_start > 0.5:
                best = _maybe_update_best(a, best)
                rnd_start = time.time()  # reset so we don’t spam too often

    # Include the cheap best into the deterministic pool
    if cheap_best_a is not None:
        init_candidates.append(cheap_best_a)

    # --------------------------------------------------------------
    # Evaluate deterministic candidates with the true objective
    for idx, a in enumerate(init_candidates):
        if time.time() > deadline:
            break
        best = _maybe_update_best(a, best)

    # --------------------------------------------------------------
    # Sub‑gradient descent phase
    if best["score"] == float("inf"):
        # fallback: uniform vector
        best["a"] = _from_raw(np.ones(n))
        best["score"] = evaluate_sequence(best["a"].tolist())
        print("[fallback] used uniform start")

    a_cur = best["a"].copy()
    cheap_cur = _max_conv_fft(a_cur)

    # Step‑size schedule
    eta0 = S * 0.5 / n          # initial magnitude (≈ average mass per entry)
    eval_period = 200           # true evaluation every `eval_period` iterations
    max_iters = 200_000
    it = 0

    while time.time() < deadline and it < max_iters:
        g, max_val = _subgradient(a_cur)
        # Decaying step size (square‑root decay)
        eta = eta0 / math.sqrt(1.0 + it * 0.01)

        a_new = a_cur - eta * g
        a_new = _project_to_simplex(a_new, S)

        cheap_new = _max_conv_fft(a_new)

        if cheap_new < cheap_cur - 1e-12:
            a_cur = a_new
            cheap_cur = cheap_new
            # occasional true evaluation
            if it % eval_period == 0:
                best = _maybe_update_best(a_cur, best)
        else:
            # Small random jitter to escape plateaus
            i1, i2 = rng.choice(n, size=2, replace=False)
            delta = eta * rng.random() * a_cur[i1] * 0.1
            a_jitter = a_cur.copy()
            a_jitter[i1] = max(a_jitter[i1] - delta, 0.0)
            a_jitter[i2] += delta
            a_jitter = _project_to_simplex(a_jitter, S)
            cheap_jit = _max_conv_fft(a_jitter)
            if cheap_jit < cheap_cur:
                a_cur = a_jitter
                cheap_cur = cheap_jit

        it += 1

    # --------------------------------------------------------------
    # Large‑scale random block perturbations (cheap)
    remaining = deadline - time.time()
    if remaining > 2.0 and best["score"] < float("inf"):
        trials = int(min(6000, remaining * 130))
        print(f"[block] performing {trials} random block trials")
        for _ in range(trials):
            nb = rng.integers(3, 16)                 # number of blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)

            cheap = _max_conv_fft(cand)
            if cheap < cheap_cur - 1e-12:
                cheap_cur = cheap
                a_cur = cand
                # Evaluate true objective less frequently
                if _ := rng.integers(0, 10) == 0:
                    best = _maybe_update_best(a_cur, best)

    # --------------------------------------------------------------
    # Monotone‑sorting heuristics (often lower the max‑convolution)
    if time.time() < deadline:
        for order in ("desc", "asc"):
            if order == "desc":
                cand = np.sort(a_cur)[::-1]
            else:
                cand = np.sort(a_cur)
            cand = _project_to_simplex(cand, S)
            cheap = _max_conv_fft(cand)
            if cheap < cheap_cur - 1e-12:
                cheap_cur = cheap
                a_cur = cand
                best = _maybe_update_best(cand, best)

    # --------------------------------------------------------------
    print(f"[final] best true score = {best['score']:.6f} (length {n})")
    return best["a"].tolist()
