Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  

The algorithm works in three stages:

1. **Length selection** – a small set of lengths is explored (derived from
   a previous solution if available).

2. **Analytic candidate sweep** – for each length we generate a dense grid
   of symmetric Beta(α,α) distributions, a few skewed Beta’s, random
   Dirichlet draws and the previous construction.  All candidates are
   normalised to sum = 1 and scored cheaply via an FFT based self‑convolution
   (the denominator cancels out for normalised vectors).

3. **Logit‑Adam refinement** – the best few candidates are refined with a
   continuous optimisation on the log‑its  ``y`` (so that ``a = softmax(y)``)
   using an Adam optimiser.  The optimisation starts with a smooth soft‑max
   surrogate of the max‑convolution (temperature grows) and finishes with
   the exact sub‑gradient once the temperature is large.  The best sequence
   seen (according to the cheap score) is kept; at the end we call the
   official ``evaluate_sequence`` once to obtain the final score.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence found as a plain list of ``float``.  If no valid sequence
is produced a uniform vector is returned as a safe fallback.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth soft‑max surrogate of max(conv) w.r.t. a.
    Returns a vector of length n (same size as a).
    """
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max (uniform weight over active indices).
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms i=j (k = 2*i)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """
    Cheap surrogate of evaluate_sequence for a vector that already sums to 1:
    returns 2·n·max(conv) where conv = a*a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _logits_to_probs(y: np.ndarray) -> np.ndarray:

    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    a = exp_y / exp_y.sum()
    return a
