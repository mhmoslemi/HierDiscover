"""
Propose a low‑score step function.

The search combines three complementary phases:

1.  Smooth‑max continuation – a family of differentiable surrogates
    f_μ (max + μ·log‑sum‑exp) is minimised with L‑BFGS‑B.
    Dozens of diverse initial points (uniform, linear, exponential,
    Dirichlet‑sparse, random Gaussian and the reference solution) and
    a slow cooling schedule (μ←0.9·μ) give a thorough exploration of
    the continuous landscape.

2.  Exact hard‑max sub‑gradient polishing – starting from the best
    surrogate solution we descend on the true objective
    𝑓(a)=max (a∗a) while staying on the ℓ¹‑simplex (projection via the
    Euclidean simplex projector).  A generous iteration budget and a
    simple back‑tracking line search are used.

3.  Large‑scale random‑block perturbations – cheap block‑wise reshuffles
    of the probabilities followed by a simplex projection allow us to
    escape any remaining local minima.  The final candidate is also
    compared with its monotone‑sorted versions.

All phases respect a user‑supplied time budget (default 1000 s) and
record the best sequence discovered.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(
    y: np.ndarray, mu: float, n: int, S: float
) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate:
        f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Parameters
    ----------
    y : ndarray, shape (n,)
        Unconstrained parameters (log‑odds of the simplex).
    mu : float
        Smoothing temperature (>0).  μ = 0 corresponds to the hard max.
    n : int
        Length of the step function.
    S : float
        Target ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : ndarray, shape (n,)
        Gradient ∂f/∂y.
    """
    # probabilities on the simplex, actual heights
    p = _softmax(y)                     # Σ p_i = 1, p_i ≥ 0
    a = S * p                           # Σ a_i = S

    # full (linear) convolution a*a (length 2·n‑1)
    conv = np.convolve(a, a)

    # smooth‑max weights w_k ∝ exp((conv_k - max)/μ)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # guard overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – uniform weight on all active indices
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. heights a:
    #   g_a[i] = 2 * Σ_k w_k * a_{k-i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)

        init_ys.append(np.log(exp_inc / exp_inc.sum() + 1e-12))

    # 5 – Dirichlet‑sparse (various concentration parameters)
    for alpha in (0.2, 0.35, 0.5, 0.8):
        p_dir = rng.dirichlet([alpha] * n)
        init_ys.append(np.log(p_dir + 1e-12))

    # 6 – random Gaussian vector
    init_ys.append(rng.standard_normal(n) * 0.5)

    # 7 – reference solution (if available)
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        init_ys.append(np.log(a_ref + 1e-12))

    # -------------------------------------------------------------
    # smooth‑max continuation parameters
    mu_factor = 0.90          # slower cooling – more accurate descent
    mu_min = 1e-12
    maxiter_per_mu = 400      # a bit more L‑BFGS iterations per temperature
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}] initialising")
        y = y0.copy()

        # initialise temperature from the current max‑convolution
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        mu = max(mu, 1.0)                 # avoid μ = 0

        # continuation loop ------------------------------------------------
        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False}
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    # -------------------------------------------------------------
    # Hard‑max sub‑gradient polishing
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        it = 0
        max_local_iters = 15000
        step0 = S * 0.4 / n

        while time.time() < deadline and it < max_local_iters:
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break

            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt

            # sub‑gradient w.r.t. a
            conv_wr = np.convolve(w, a[::-1])
            g = 2.0 * conv_wr[n - 1:n - 1 + n]

            # back‑tracking line search
            step = step0 / (1.0 + 1e-5 * it)
            improved = False
            for _ in range(15):
                a_new = a - step * g
                a_new = _project_to_simplex(a_new, S)
                sc = evaluate_sequence(a_new.tolist())
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[subgrad] iter {it:5d} step {step:.6e} → {best_score:.6f}")
                    break
                step *= 0.5

            if not improved:
                # occasional random jitter to escape plateaux
                i, j = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a[i] * 0.04
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, S)

            it += 1

    # -------------------------------------------------------------
    # Random block perturbations (cheap large‑scale search)
    remaining = deadline - time.time()
    if remaining > 2.0 and best_score < float("inf"):
        max_trials = int(min(15000, remaining * 120))
        print(f"[block] {max_trials} random block trials")
        for _ in range(max_trials):
            # pick a random number of blocks between 3 and 12
            nb = rng.integers(3, 13)
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # -------------------------------------------------------------
    # Monotone‑sorting heuristics (often finish a little lower)
    if time.time() < deadline and best_score < float("inf"):
        descending = np.sort(best_a)[::-1]
        descending = _project_to_simplex(descending, S)
        _maybe_update(descending)

        ascending = np.sort(best_a)
        ascending = _project_to_simplex(ascending, S)
        _maybe_update(ascending)

    # -------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} (length {n})")
    return best_a.tolist()
