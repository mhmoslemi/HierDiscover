"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Select a few problem sizes** – based on the length of a possible
   previously found solution (`height_sequence_1`) we try a handful of
   lengths between 500 and 20 000 (larger lengths give a slightly smaller
   theoretical bound but are more expensive).

2. **Generate a rich pool of seeds** – for each length we create many
   candidates from simple parametric families:
   * symmetric power‑law   a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ   (p ∈ [0.05,1.5])
   * asymmetric power‑law a_i ∝ (i+½)⁻ˡ (n‑i‑½)⁻ʳ   (ℓ,r ∈ [0,1.2])
   * mixtures of the above with a uniform component.
   The candidates are normalised to sum = 1 and projected onto the simplex.

   The 2 n·max(conv) value is evaluated with a fast FFT; the best few
   (≈ 4) per length are kept.

3. **Adam‑optimised mirror descent** – each retained seed is refined by
   minimising the true objective 2 n·max(a∗a) on the simplex.
   * Variables are represented as logits `y = log a` to keep positivity.
   * Gradient of a smooth log‑sum‑exp surrogate for the max is used while a
     temperature `α` is moderate; after `α` grows large we switch to the exact
     sub‑gradient of the max entry.
   * Adam updates on the logits are performed; after every step the vector
     is re‑normalised onto the simplex.
   * The schedule for `α` is exponential (α←α·1.25) up to a large
     `α_max = 2000`, then the exact sub‑gradient is used.

   The optimization runs until the allocated time slice expires, keeping
   the best sequence seen.

4. **Stochastic hill climbing** – the globally best vector is finally
   polished by repeatedly moving a random fraction of mass from one index
   to another (up to 10 % of the source) whenever the true ``evaluate_sequence``
   score improves.  This cheap local search can escape tiny plateaus.

5. **Return the best sequence** – converted to a plain Python list.
   Progress and intermediate scores are printed.

The whole routine respects the 1000 s budget (default) and never
returns ``None``; in the worst case a uniform vector is used as a safety
fallback.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a∗a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max entry of a∗a (at index k)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g

# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of (1/α)·log∑exp(α·conv) w.r.t. a (soft‑max surrogate)."""
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # convolution of a_rev with w (equivalent to gradient of each conv entry)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_grad = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv_grad[start:start + a.shape[0]]
    return g

# ----------------------------------------------------------------------
def _sym_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power law: a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals

# ----------------------------------------------------------------------
def _asym_power_shape(n: int, pl: float, pr: float) -> np.ndarray:
    """Asymmetric power law: a_i ∝ (i+½)⁻ˡ·(n‑i‑½)⁻ʳ."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-pl) * (n - i - 0.5) ** (-pr)
    vals /= vals.sum()
    return vals

# ----------------------------------------------------------------------
def _mix_with_uniform(shape: np.ndarray, w: float) -> np.ndarray:
    """Linear mixture of *shape* with the uniform distribution."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = (1.0 - w) * shape + w * uniform
    return _project_onto_simplex(mixed, s=1.0)

# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Fast surrogate = 2·n·max(a∗a) (no normalisation needed)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()

# ----------------------------------------------------------------------
def _generate_candidates(n: int) -> list[np.ndarray]:
    """Create a diverse list of seed vectors for length *n*."""
    candidates = []

    # (1) Uniform and random Dirichlet seeds
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))
    rng = np.random.default_rng()
    candidates.append(rng.random(n))
    candidates[-1] /= candidates[-1].sum()

    # (2) Symmetric power‑law family with mixture
    p_grid = np.linspace(0.05, 1.5, 30)                 # finer than before
    w_grid = np.linspace(0.0, 0.6, 7)                   # up to 60 % uniform
    for p in p_grid:
        shape = _sym_power_shape(n, p)
        for w in w_grid:
            cand = _mix_with_uniform(shape, w)
            candidates.append(cand)

    # (3) Asymmetric power‑law family
    exp_grid = np.linspace(0.0, 1.2, 7)
    for pl in exp_grid:
        for pr in exp_grid:
            cand = _asym_power_shape(n, pl, pr)
            candidates.append(cand)

    # (4) Previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.size == n:
            candidates.append(_project_onto_simplex(np.maximum(prev, 0.0), s=1.0))
    except Exception:
        pass

    # keep a reasonable amount – we will later select the best few
    return candidates

# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam mirror‑descent on logits (log a) to minimise 2·n·max(a∗a).
    Returns the best vector seen (as a numpy array) and its internal score.
    """
    # initialise Adam on logits
    eps = 1e-30
    a = a0.copy()
    a = np.maximum(a, eps)
    y = np.log(a)                     # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr0 = 0.25                        # base learning‑rate

    best_seq = a.copy()
    best_score = _cheap_score(a)

    # temperature schedule for soft‑max surrogate
    alpha = 2.0
    alpha_max = 2000.0
    alpha_mul = 1.25
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- evaluate current point ----
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_seq = a.copy()

        # ---- compute gradient ----
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # gradient w.r.t. logits
        grad_y = a * grad

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        bias_correction1 = 1.0 - beta1 ** it
        bias_correction2 = 1.0 - beta2 ** it
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        lr = lr0 / math.sqrt(it)        # slowly decaying step size
        y -= lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # ---- map back to the simplex ----
        a = np.exp(y)
        a = np.maximum(a, eps)          # avoid exact zeros
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # occasional jitter to escape flat regions
        if it % 1500 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()

    return best_seq, best_score

def _local_hill_climb(a_start: np.ndarray,
                     n: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill climbing: move a random fraction of mass between two
    positions whenever the true evaluation score improves.
    Returns the best sequence found and its evaluated score.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.15    # up to 15 % of source
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur_score = evaluate_sequence(cand.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand
            # print(f"[Hill climb] it={it} score={best_score:.6f}")
    return best_seq, best_score
