"""

Algorithm
---------

   symmetric beta, triangular, exponential) on several lengths and keep
   the best (lowest exact max‑convolution) as the initial point.

   max‑convolution:
        F_β(a) = log Σ_k exp(β·(a∗a)_k) .
   The gradient is 2·conv_rev(w) where w are soft‑max weights.
   An Adam optimiser (with projection onto the probability simplex) is
   run while β is gradually increased, turning the smoothed objective
   into the true max.

3. **Exact max‑subgradient descent** – once β is large, switch to the
   exact max sub‑gradient (active‑set) with momentum and a diminishing
   step‑size.  This refines the solution further.

4. **Local random transfer hill‑climbing** – small random moves of mass
   between two positions are accepted only if they improve the exact
   objective.  Occasional jitter helps escape plateaus.

The whole search respects the supplied ``budget_s`` (≈ 1000 s) and always

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path for already feasible vectors
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) for a normalized (sum=1) sequence."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p, p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))          # shift for stability
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _generalized_arcsine_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Generalised arcsine: w_i ∝ ((i+0.5)*(n-i-0.5))^{-alpha}.
    alpha=0.5 reproduces the classical arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    vals = denom ** (-alpha)
    vals /= vals.sum()
    return vals

            print(f"[{time.time():.1f}] n={n} triangular score={sc:.6f}")

        # ---- exponential ---------------------------------------------
        a = _exponential_distribution(n, lam=0.005)
        sc = _max_convolution_score(a)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} exponential score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Include the previous best construction, if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # Fallback – should never happen.
    if overall_best_seq is None:
        n = max(500, base_n)
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _max_convolution_score(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Soft‑max Adam optimisation with gradually increasing β.
    n = overall_best_seq.shape[0]
    a = overall_best_seq.copy()
    best_seq = a.copy()
    best_score = overall_best_score

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step_size = 0.2
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for β (smoothness of max)
    beta_start = 5.0
    beta_max = 300.0
    beta_inc_per_iter = (beta_max - beta_start) / 200_000   # aim to reach β_max after ~200k iters

    iter_adam = 0

    while time.time() < deadline:
        iter_adam += 1

        # ----- convolution and soft‑max weights -----
        b = _fft_self_convolve(a)
        cur_max = b.max()
        # linear schedule for β
        beta_cur = min(beta_max, beta_start + beta_inc_per_iter * iter_adam)

        # stable soft‑max
        shifted = b - cur_max
        w = np.exp(beta_cur * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            # numerical underflow – fall back to exact max weights
            w = np.zeros_like(b)
            w[np.flatnonzero(b >= cur_max - 1e-12)] = 1.0 / (b >= cur_max - 1e-12).sum()
        else:
            w /= w_sum

        # ----- gradient of the smooth objective -----
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ----- Adam update (with projection) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_adam)
        v_hat = v / (1.0 - beta2 ** iter_adam)
        a = a - step_size * m_hat / (np.sqrt(v_hat) + eps)

        # Keep the vector on the simplex (non‑negative, sum = 1)
        a = _project_onto_simplex(a, s=1.0)

        # Periodically evaluate the *exact* objective
        if iter_adam % 500 == 0:
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam iter={iter_adam} new best {cur_score:.6f}")

        # Early stop if we have already crossed the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 5.  Exact max‑subgradient descent (makes the final push).
    a = best_seq.copy()
    best_score = best_score
    best_seq = a.copy()

    velocity = np.zeros_like(a)
    momentum = 0.9
    step0 = 0.4
    it_sub = 0

    while time.time() < deadline:
        it_sub += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_active = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_active)

        # weight vector for the active set (binary)
        w = np.zeros_like(b)
        w[active] = 1.0

        # exact sub‑gradient (same formula as in the original code)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # diagonal correction for the terms where k = 2i (see original implementation)
        even_mask = (active % 2 == 0)
        diag_idx = (active[even_mask] // 2)
        if diag_idx.size > 0:
            grad[diag_idx] -= a[diag_idx]

        # momentum + diminishing step size
        velocity = momentum * velocity + (1.0 - momentum) * grad
        eta = step0 / np.sqrt(it_sub)
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # evaluate exact objective
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Subgrad iter={it_sub} new best {cur_score:.6f}")

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 6.  Local random‑transfer hill climbing (tiny stochastic tweaks).
    a = best_seq.copy()
    cur_score = best_score
    it_hc = 0
    max_transfer = 0.04                 # max 4 % of total mass per trial

    while time.time() < deadline:
        it_hc += 1
        # gradually shrink the amount we are allowed to move
        transfer = max_transfer * (0.5 ** (it_hc / 18000.0))

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(transfer, a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-13:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Transfer iter={it_hc} improved best {best_score:.6f}")

        # occasional jitter to jump out of plateaus
        if it_hc % 6000 == 0 and rng.random() < 0.12:
            jitter = rng.normal(0.0, 1e-4, size=n)
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
