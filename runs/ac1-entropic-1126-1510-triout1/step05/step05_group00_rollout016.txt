"""
Propose a low‑score non‑negative step function.

The algorithm:
1.  Choose a few candidate lengths (up‑to ≈ 8000 points).  For each length
    generate several simple analytic families (arcsine, symmetric beta,
    triangular, exponential) plus a random Dirichlet start.
2.  For every start run a fast projected sub‑gradient descent on the simplex.
    The sub‑gradient of  max_k (a*a)_k  is obtained from the active set of
    convolution entries that attain the current maximum.  Momentum and a
    diminishing step‑size  η_t = η0/√t  are used.
3.  After the gradient phase a cheap simulated‑annealing hill‑climbing
    routine moves small amounts of mass between two random positions.
    Improvements are always kept; worse moves are occasionally accepted with
    a temperature schedule to escape plateaux.
4.  The best sequence found over all starts and lengths is returned.
   All numbers are non‑negative and the vector is projected onto the simplex
   (sum = 1) before evaluation, which matches the behaviour of
   ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, v.size + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective value  2·n·max(a*a)  under the convention sum(a)=1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the simplex.
    Returns the best sequence and its score found before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(a)

    velocity = np.zeros_like(a)
    beta = 0.9                 # momentum coefficient
    step0 = 0.8                # initial step‑size scaling
    it = 0

    # reserve roughly half of the remaining budget for the gradient phase
    subgrad_deadline = time.time() + 0.5 * (deadline - time.time())

    while time.time() < subgrad_deadline:
        it += 1

        # full self‑convolution and current maximum
        conv = _fft_self_convolve(a)
        cur_max = conv.max()

        # active set (indices attaining the max up to tiny tolerance)
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps)

        # binary weight vector for the active constraints
        w = np.zeros_like(conv)
        w[active] = 1.0

        # sub‑gradient of the max‑convolution (see header discussion)
        rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * rev[n - 1: n - 1 + n]

        # diagonal correction used in the reference implementation
        even_mask = (active % 2 == 0)
        diag_idx = (active[even_mask] // 2)
        if diag_idx.size:
            grad[diag_idx] -= a[diag_idx]

        # momentum update + diminishing step size
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a -= eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.1f}] subgrad n={n} it={it} score={cur_score:.6f}")

        if best_score <= target_score:
            break

    return best, best_score


# ----------------------------------------------------------------------
def _hillclimb_optimize(a0: np.ndarray,
                        deadline: float,
                        target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing / random‑transfer hill climbing.
    Starts from ``a0`` (already normalised) and returns the best sequence
    found before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score

    # parameters
    max_transfer = 0.05                     # at most 5 % of total mass per trial
    start_temp = 0.02
    end_temp = 1e-6
    start_time = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        # linear temperature schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = start_temp * (1.0 - elapsed / total) + end_temp * (elapsed / total)

        # gradually shrink the transferred mass
        transfer = max_transfer * (0.5 ** (it / 20000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                print(f"[{time.time():.1f}] hillclimb iter={it} best={best_score:.6f}")
        else:
            # Metropolis acceptance of uphill moves
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a_new
                cur_score = new_score

        # occasional jitter to escape flat regions
        if it % 5000 == 0 and np.random.rand() < 0.1:
            jitter = np.random.randn(n) * 1e-4
            a = best + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

        if best_score <= target_score:
            break

    return best, best_score


# ----------------------------------------------------------------------
def _optimize_one_start(a_start: np.ndarray,
                        deadline: float,
                        target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Run sub‑gradient descent followed by hill climbing for a single start.
    """
    # 1) sub‑gradient phase
    sub_deadline = deadline - 0.05 * (deadline - time.time())   # keep a tiny reserve
    best, best_score = _subgradient_optimize(a_start, sub_deadline, target_score)

    # 2) hill‑climb phase (use whatever time remains)
    if time.time() < deadline:
        best, best_score = _hillclimb_optimize(best, deadline, target_score)

    return best, best_score

    The routine respects the supplied time budget and returns the best

    """
    np.random.seed(seed)

    for _ in range(3):

            candidates.append(_beta_symmetric_distribution(n, p))

        # a random Dirichlet start – often surprisingly good
        candidates.append(np.random.dirichlet(np.ones(n)))

        # optionally try to reuse a previously stored construction (if any)
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.ndim == 1:

                if prev.size == n:

        except Exception:
            pass

        # 3) Optimise each start but stop early if we run out of time.
        for cand in candidates:

            if best_global_score <= target_score:

        if best_global_score <= target_score:
            break

    # 4) Fallback – a uniform distribution (should never be needed).
    if best_global is None:

        best_global_score = _max_convolution_score(best_global)
        print(f"[{time.time():.1f}] fallback uniform score={best_global_score:.6f}")

    print(f"Final best score: {best_global_score:.6f} (length={len(best_global)})")
    return best_global.tolist()
