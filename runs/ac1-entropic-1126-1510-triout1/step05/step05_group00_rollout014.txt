def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:2*n - 1]

def _fft_cross_convolve(p, a_rev):
    size = 1 << ((len(p) + len(a_rev) - 1) - 1).bit_length()
    P = np.fft.rfft(p, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(P * A_rev, n=size)
    return conv[:len(p) + len(a_rev) - 1]

Now the gradient: grad = 2 * conv_rev[n-1 : n-1 + n] where conv_rev = _fft_cross_convolve(p, a_rev).

def refine_sequence(a, deadline, target_score=None):

    n = len(a)
    best_seq = a.copy()
    best_score = objective(a)

    velocity = np.zeros_like(a)

    beta = 0.9
    step0 = 0.5
    α = 0.5
    α_max = 500.0

    iteration = 0

    while time.time() < deadline:
        iteration += 1

        if iteration % 1000 == 0:
            # increase α
            α = min(α_max, α * 1.1)

        # compute conv via FFT
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # compute smoothed p
        scaled = α * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = np.sum(exp_scaled)
        p = exp_scaled / sum_exp

        # compute gradient:
        a_rev = a[::-1]
        conv_rev = _fft_cross_convolve(p, a_rev)  # length 2n-1 + n -1 = 3n-2
        grad = 2.0 * conv_rev[n-1 : n-1+n]  # slice length n

        # momentum + step size
        velocity = beta * velocity + (1 - beta) * grad
        eta = step0 / np.sqrt(iteration)

        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, s=1.0)

        a = a_new

        # evaluate actual objective
        conv = _fft_self_convolve(a)  # compute again maybe okay
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] refine iter={iteration} new best {cur_score:.6f}")

        # early stop
        if target_score is not None and best_score <= target_score:
            break

    return best_seq, best_score

"""
Propose a low‑score non‑negative step function.

Algorithm
---------

1. **Exploratory sampling** – generate a variety of analytic families
   (arcsine, symmetric beta, triangular, exponential) together with
   a few random Dirichlet draws for several lengths.  Keep the best
   few candidates (by the exact objective).

2. **Annealed soft‑max gradient descent** – minimise the exact quantity
   `2*n*max(a*a)` (with `sum(a)=1`) by minimising a differentiable
   surrogate

        φ_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

   The gradient of this surrogate is cheap to obtain via FFTs.
   The smoothing parameter `α` is increased during the run, so the
   method first follows a smooth landscape and gradually approaches the
   true max‑norm.  After each update the vector is projected onto the
   probability simplex.

3. **Local random‑transfer hill climbing** – starting from the best
   sequence obtained above, repeatedly move a small amount of mass
   between two random positions and keep improvements.  Occasional
   jitter helps to escape plateaux.

The whole procedure respects the supplied time budget (≈ 1000 s) and
returns a list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families -----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p, p) on a regular grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:                     # pragma: no cover
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# FFT based convolutions -----------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_convolve(p: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """
    Convolution of p (length 2·n‑1) with a_rev (length n) via real FFT.
    Returns the full result of length ``len(p)+len(a_rev)-1``.
    """
    size = 1 << ((len(p) + len(a_rev) - 1) - 1).bit_length()
    P = np.fft.rfft(p, n=size)
    A = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(P * A, n=size)
    return conv[: len(p) + len(a_rev) - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _refine_sequence(
    a0: np.ndarray,
    deadline: float,
    target_score: float | None = None,
) -> tuple[np.ndarray, float]:
    """
    Annealed soft‑max gradient descent on the simplex.
    Returns the best sequence seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam‑style moments (optional but useful)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smoothing parameter schedule
    alpha = 0.5          # start with a smooth surrogate
    alpha_max = 400.0    # final (almost exact) max
    alpha_inc = 1.03    # multiplicative increase every 500 iters

    # step‑size schedule
    lr0 = 0.3

    best_seq = a.copy()
    best_score = _max_convolution_score(a)

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        it += 1

        # increase smoothing slowly
        if it % 500 == 0:
            alpha = min(alpha_max, alpha * alpha_inc)

        # self‑convolution
        conv = _fft_self_convolve(a)

        # --- soft‑max weights -------------------------------------------------
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)          # stabilise
        sum_exp = np.sum(exp_scaled)
        p = exp_scaled / sum_exp                           # shape (2n‑1,)

        # --- gradient of the surrogate ---------------------------------------
        a_rev = a[::-1]
        conv_rev = _fft_cross_convolve(p, a_rev)          # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]           # slice length n

        # --- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps) * m_hat

        a_new = a - step
        a_new = _project_onto_simplex(a_new, s=1.0)

        a = a_new

        # evaluate exact objective
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{now:.1f}] refine iter={it} new best {cur_score:.6f}")

        if target_score is not None and best_score <= target_score:
            break

    return best_seq, best_score

            best_seq_global = a.copy()
            print(f"[{time.time():.1f}] n={n} exponential score={sc:.6f}")

        for alpha in (0.5, 2.0):
            a = np.random.dirichlet(np.full(n, alpha))
            sc = _max_convolution_score(a)
            candidates.append((a.copy(), sc))
            if sc < best_score_global:
                best_score_global = sc
                best_seq_global = a.copy()
                print(f"[{time.time():.1f}] n={n} dirichlet(alpha={alpha}) score={sc:.6f}")

    # 3️⃣  Add the previously best known construction (if any)

            sc = _max_convolution_score(prev)
            candidates.append((prev.copy(), sc))
            if sc < best_score_global:
