def _adam_optimize(a0, n, time_deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr0=0.4
    t=0
    best_local = a.copy()
    best_score = _score(a)
    alpha = 1.0  # temperature for softmax
    alpha_factor = 1.02
    alpha_max = 5000.0

    while time.time() < time_deadline:
        t+=1
        lr = lr0 / np.sqrt(t)
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_local = a.copy()
        # compute gradient
        if alpha < alpha_max:
            grad = _softmax_gradient(a, conv, alpha)
            alpha = min(alpha_max, alpha * alpha_factor)
        else:
            grad = _exact_gradient(a, conv)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex (sum to 1)
        a = _project_onto_simplex(a, s=1.0)

    return best_local, best_score

def _inverse_sqrt_distribution(n, p):

"""
Propose a step‑function with a low value of evaluate_sequence.

Algorithm
---------
1. Choose a few problem sizes (n).  If a previous construction
   ``height_sequence_1`` exists we centre around its length, otherwise we
   start from ~8000.  Lengths are kept ≤ 20 000 so that the FFTs stay fast.

2. For each length generate a small library of analytic candidates:
   * uniform,
   * arcsine (Beta(½,½)),
   * several symmetric & skewed Beta distributions,
   * a family a_i ∝ (i·(n‑i))^{-p} (p∈{0.25,0.5,0.75,1.0}),
   * a random Dirichlet / Gamma draw,
   * the previous construction (if available).

   All vectors are projected onto the probability simplex (non‑negative,
   sum = 1) and cheaply scored with the fast FFT‑based self‑convolution.
   The best few are kept as starters.

3. Refine each starter with a projected Adam optimiser.
   * Objective: smoothed max of the convolution using the soft‑max
     surrogate (temperature α that is increased each iteration).
   * Gradient of the surrogate is computed via FFTs.
   * Adam‑updates the vector, after each step the current iterate is
     projected back onto the simplex.
   * A tiny jitter is added every few thousand iterations to escape plateaus.
   * The best iterate encountered (lowest cheap score) is kept.

4. The best refined candidate of a length is finally evaluated with the
   official ``evaluate_sequence`` function.  The globally best sequence
   across all lengths (and the whole time budget) is returned.

The whole search respects the supplied ``budget_s`` (default 1000 s); if
the target score 1.503 is reached the search stops early.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
def _inverse_sqrt_distribution(n: int, p: float) -> np.ndarray:
    """Shape a_i ∝ (i+½)^{-p}·(n‑i‑½)^{-p}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n + 0.5 - i)
    # avoid zero
    prod = np.maximum(prod, 1e-12)
    w = prod ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w


# ----------------------------------------------------------------------
def _initial_vector(init_kind: str, n: int) -> np.ndarray:
    """Produce a feasible probability vector (sum = 1)."""
    if init_kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init_kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif init_kind == "arcsine":
        a = _beta_distribution(n, 0.5, 0.5)
    elif init_kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _beta_distribution(n, 0.5, 0.5)
    else:
        a = _beta_distribution(n, 0.5, 0.5)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Cheap proxy for evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the soft‑max surrogate of max(conv)."""
    n = a.shape[0]
    cur_max = conv.max()
    # stable soft‑max
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the true max (uniform weight over active indices)."""
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (self‑product terms)
    even = (active % 2 == 0)
    diag = active[even] // 2
    if diag.size:
        grad[diag] -= a[diag]
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   target_score: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the soft‑max surrogate of max(a*a).
    Returns the best vector seen and its cheap score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.3
    iteration = 0
    best_seq = a.copy()
    best_sc = _score(a)

    # temperature for the soft‑max surrogate
    alpha = 1.0
    alpha_factor = 1.008
    alpha_max = 5000.0

    while time.time() < deadline:
        iteration += 1
        lr = lr0 / math.sqrt(iteration)

        conv = _fft_self_convolve(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = a.copy()
            # early exit if we already beat the target
            if best_sc <= target_score:
                break

        # gradient
        if alpha < alpha_max:
            grad = _softmax_gradient(a, conv, alpha)
            alpha = min(alpha_max, alpha * alpha_factor)
        else:
            grad = _exact_gradient(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter to avoid stagnation
        if iteration % 4000 == 0:
            jitter = np.random.randn(n) * 1e-7
            a += jitter
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_sc
