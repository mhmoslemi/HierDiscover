def _fw_step(a, n, size_fft):
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    mask = (conv == max_val).astype(float)
    # Compute gradient
    a_rev = a[::-1]
    # convolve mask with rev a via fft
    W = np.fft.rfft(mask, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    grad_full = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]  # shape n
    i = int(np.argmin(grad))
    return i, max_val
```

def _frank_wolfe_optimize(a0, n, size_fft, deadline):
    a = a0.copy()
    best = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    t=0
    while time.time() < deadline:
        # compute conv, mask
        conv = _fft_self_convolve(a)
        M = conv.max()
        mask = np.isclose(conv, M, atol=1e-12).astype(float)
        # gradient
        a_rev = a[::-1]
        W = np.fft.rfft(mask, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        grad_full = np.fft.irfft(W * A_rev, n=size_fft)
        grad = 2.0 * grad_full[n-1:n-1+n]
        # pick index with minimal gradient
        i = int(np.argmin(grad))
        # step size
        t += 1
        gamma = 2.0 / (t + 2.0)
        a = (1.0 - gamma) * a
        a[i] += gamma
        # maybe renormalize due to floating error
        a = a / a.sum()
        # evaluate
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
    return best, best_score

"""
Proposes a non‑negative step function (list of heights) that minimises
`evaluate_sequence`.  The search runs inside the 1000 s budget and consists
of three stages:

1. **Broad parametric warm‑start** – generate many hand‑crafted probability
   vectors (uniform, arcsine, a rich family of symmetric Beta PDFs,
   mixtures and a few random Dirichlet draws).  The best of them is kept.

2. **Multi‑restart optimisation** – repeatedly invoke a projected Adam
   optimiser (soft‑max surrogate for the max‑convolution) starting from
   random candidates.  After each Adam run a cheap Frank‑Wolfe style
   conditional‑gradient refinement is applied.  All runs are limited by
   the remaining time budget; the globally best vector is retained.

3. **Final polishing** – a stochastic hill‑climber (mass swaps, tiny
   jitter) works on the best vector found so far until the deadline.

All intermediate objective values are computed with an FFT based
self‑convolution ( O(n log n) ) which is much faster than the naïve
`np.convolve` used by `evaluate_sequence`.  The final sequence is returned
as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·b_k) with respect to a.
    For huge α this becomes a sub‑gradient of the max‑operator.
    """
    n = a.shape[0]

    # --- soft‑max weights -------------------------------------------------
    if alpha > 2_000:
        # treat every entry that is numerically maximal as active
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()                # for stability
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # --- convolution w * reversed(a) ------------------------------------
    a_rev = a[::-1]                        # reversed vector
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]        # factor 2 comes from derivative of a*a
    return g


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points (already normalised)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) PDF evaluated on mid‑points of n bins."""
    a = max(a, 0.01)                       # avoid division by zero
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0 or not np.isfinite(pdf_sum):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= pdf_sum
    return pdf


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Generate a modest list of hand‑crafted probability vectors.
    The vectors are already normalised to sum = 1.
    """
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) arcsine
    candidates.append(_arcsine_distribution(n))

    # 3) symmetric Beta(a,a) on a fine grid (including very small a)
    for a in np.linspace(0.02, 1.5, 15):
        candidates.append(_beta_distribution(n, a))

    # 4) mixtures uniform + arcsine
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    arcsine = _arcsine_distribution(n)
    for w in np.linspace(0.0, 0.4, 5):
        mix = (1.0 - w) * arcsine + w * uniform
        candidates.append(mix)

    # 5) spikes at the extremes (tiny amounts)
    for spike_frac in (0.05, 0.10, 0.15):
        mix = arcsine.copy()
        mix[0] += spike_frac / 2
        mix[-1] += spike_frac / 2
        mix = _project_onto_simplex(mix, 1.0)
        candidates.append(mix)

    # 6) previous best (if supplied)
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())

    # 7) a few random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(5):
        candidates.append(rng.dirichlet(np.ones(n)))

    # 8) very spiky two‑point distribution (mass at ends only)
    spikes = np.zeros(n, dtype=np.float64)
    spikes[0] = spikes[-1] = 0.5
    candidates.append(spikes)

    # final projection onto the simplex (guard against numerical drift)
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex.
    Returns the best vector seen and its raw (2·n·max(conv)) score.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.10          # a bit more aggressive than the original
    alpha_max = 100_000.0

    size_fft = 1 << ((3 * n - 2).bit_length())      # ≥ (2n‑1)+(n)‑1

    it = 0
    while time.time() < deadline:
        it += 1

        # current objective
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} it={it} score={best_score:.6f}")

        # gradient (soft‑max surrogate)
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base learning rate – a slightly larger constant helps escape plateaus
        step = 0.25 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection back onto the simplex
        a = _project_onto_simplex(a, 1.0)

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _frank_wolfe_optimize(init_a: np.ndarray,
                          n: int,
                          size_fft: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple conditional‑gradient (Frank‑Wolfe) optimiser.
    Uses the exact sub‑gradient of the max‑convolution.
    """
    a = init_a.copy()
    best = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()

    t = 0
    while time.time() < deadline:
        t += 1
        # compute current convolution and active set
        conv = _fft_self_convolve(a)
        M = conv.max()
        mask = np.isclose(conv, M, atol=1e-12).astype(float)   # length 2n‑1

        # sub‑gradient of max(conv) w.r.t. a:
        a_rev = a[::-1]
        W = np.fft.rfft(mask, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        grad_full = np.fft.irfft(W * A_rev, n=size_fft)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # choose vertex that minimises <grad, e_i>
        i = int(np.argmin(grad))

        # standard FW step size
        gamma = 2.0 / (t + 2.0)

        # move towards e_i
        a = (1.0 - gamma) * a
        a[i] += gamma

        # renormalise (tiny numeric drift)
        a = a / a.sum()

        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a random fraction of the mass
    from one index to another and keep the move if it improves the score.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)

        if best[i] < 1e-12:
            continue

        # move up to 30 % of the mass at i to j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter to escape shallow plateaus
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score
