    A_rev = np.fft.rfft(a_rev, n=size_fft)
    W = np.fft.rfft(w_norm, n=size_fft)
    conv_grad = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    grad_a = 2.0 * conv_grad[start:start + n]
    # chain rule
    grad_y = a * (grad_a - np.dot(grad_a, a))
    return f, grad_y

"""
propose_candidate – high‑precision search for a step‑function
that minimises 2·n·max(conv(a)) / (sum(a))**2.

The algorithm
----------------
1.  Start from the analytically good “arcsine” shape
    a_i ∝ 1/√((i+½)(n‑i‑½)).
2.  Smoothly minimise the true max‑convolution using
    L‑BFGS‑B on the surrogate

        f(a) = max(conv(a)) + (1/α)·log∑exp(α·(conv(a)–max(conv(a))))

    The gradient of the surrogate is obtained by a single
    FFT‑based convolution; the optimisation works directly in the
    log‑softmax parametrisation a_i = exp(y_i)/∑exp(y) which
    guarantees non‑negativity and the simplex constraint.
3.  A cheap stochastic hill‑climbing (tiny mass moves between two
    indices) further refines the best vector.
4.  The whole search is run for a few different lengths
    (8000, 12000, 20000) respecting the 1000 s budget and the
    best sequence found is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full (length 2·n‑1) convolution of *a* with itself via a real FFT.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _initial_arcsine(n: int) -> np.ndarray:
    """Arcsine law a_i ∝ 1/√((i+½)(n‑i‑½)), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _objective_and_gradient(y: np.ndarray,
                            n: int,
                            alpha: float,
                            size_fft: int):
    """
    Returns the smoothed max‑convolution value and its gradient w.r.t.
    the log‑variables y (where a = softmax(y)).
    """
    # a = softmax(y)
    a = np.exp(y)
    s = a.sum()
    a = a / s if s != 0.0 else np.full(n, 1.0 / n)

    # self‑convolution a*a
    A = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]

    max_conv = conv.max()
    shift = conv - max_conv                     # ≤0
    w = np.exp(alpha * shift)                  # soft‑max weights (unnormalised)
    w_sum = w.sum()
    f = max_conv + np.log(w_sum) / alpha       # smooth approximation of max

    # gradient w.r.t. a (soft‑max weighted sum of all ∂conv_i/∂a)
    w_norm = w / w_sum
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    W = np.fft.rfft(w_norm, n=size_fft)
    grad_conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    grad_a = 2.0 * grad_conv[start:start + n]   # derivative of Σ w_i·conv_i

    # chain rule: ∂f/∂y = a ∘ (grad_a – ⟨grad_a,a⟩)
    dot = np.dot(grad_a, a)
    grad_y = a * (grad_a - dot)
    return f, grad_y.astype(np.float64)


# ----------------------------------------------------------------------
def _lbfgs_func(y: np.ndarray, n: int, alpha: float, size_fft: int) -> float:
    """Wrapper for scipy.optimize – returns only the objective value."""
    f, _ = _objective_and_gradient(y, n, alpha, size_fft)
    return float(f)


def _lbfgs_grad(y: np.ndarray, n: int, alpha: float, size_fft: int) -> np.ndarray:
    """Wrapper for scipy.optimize – returns only the gradient."""
    _, g = _objective_and_gradient(y, n, alpha, size_fft)
    return g


# ----------------------------------------------------------------------
def _optimize_lbfgs(a0: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Runs L‑BFGS‑B on the smoothed surrogate, starting from a0.
    Returns the optimized vector (still normalised to sum 1) and the
    true internal score 2·n·max(conv).
    """
    # log‑softmax representation
    y0 = np.log(a0 + 1e-30)

    # smoothing parameter – large enough to approximate the true max
    alpha = max(5.0, 15.0 * n)

    # FFT size sufficient for all convolutions (a with a, and w with a_rev)
    size_fft = 1 << ((3 * n - 2).bit_length())

    # a modest iteration budget – fits comfortably in the global time budget
    max_iter = 400

    # run the optimiser
    res = minimize(fun=_lbfgs_func,
                   x0=y0,
                   args=(n, alpha, size_fft),
                   jac=_lbfgs_grad,
                   method='L-BFGS-B',
                   options={'maxiter': max_iter, 'disp': False})

    # recover the probability vector
    a_opt = np.exp(res.x)
    s = a_opt.sum()
    a_opt = a_opt / s if s != 0.0 else np.full(n, 1.0 / n)

    # true score (fast FFT)
    internal_score = 2.0 * n * np.max(_fft_self_convolve(a_opt))
    return a_opt, internal_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: move a tiny fraction of mass
    from one index to another whenever the true evaluation improves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 5 % of the source mass
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur_score = 2.0 * n * np.max(_fft_self_convolve(cand))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Entry point used by the evaluation harness.
    Searches for a non‑negative step‑function with a low
    ``evaluate_sequence`` score and returns it as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # lengths to try – all ≤ 20 000 to keep FFT sizes reasonable
    lengths = [8000, 12000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503

    for n in lengths:
        if time.time() >= deadline - 1.0:
            break

        # -----------------------------------------------------------------
        # (1) start from the analytic arcsine shape
        a0 = _initial_arcsine(n)

        # allocate a fraction of the remaining budget for the L‑BFGS phase
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(remaining * 0.4, 20.0)

        # (2) L‑BFGS optimisation on the smoothed max
        a_opt, _ = _optimize_lbfgs(a0, n, slice_deadline)

        # evaluate with the black‑box validator
        try:
            final_score = evaluate_sequence(a_opt.tolist())
        except Exception:               # should never happen
            final_score = float("inf")

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = a_opt.copy()
            print(f"*** New global best (L‑BFGS) → len={n} score={overall_best_score:.6f}")

        # -----------------------------------------------------------------
        # (3) final cheap stochastic refinement, if time permits
        if time.time() < deadline - 1.0 and overall_best_seq is not None:
            a_refined, _ = _local_refinement(overall_best_seq,
                                             len(overall_best_seq),
                                             deadline)
            try:
                refined_score = evaluate_sequence(a_refined.tolist())
            except Exception:
                refined_score = float("inf")
            if refined_score < overall_best_score:
                overall_best_score = refined_score
                overall_best_seq = a_refined.copy()
                print(f"*** After local refinement score={overall_best_score:.6f}")

        if overall_best_score <= target_score:
            break

    # ----------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a return value.
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Safety fallback used.")

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
