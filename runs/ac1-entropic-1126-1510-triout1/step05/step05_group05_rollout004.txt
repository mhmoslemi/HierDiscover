"""
Propose a step‑function (non‑negative vector) that minimizes
`evaluate_sequence`.  The algorithm consists of three stages:

1.  **Broad random search** – sample many parametric families
    (arcsine, power‑law, beta, Gaussian, mixtures) on several
    admissible lengths, project each candidate onto the probability
    simplex and keep the few best candidates.

2.  **Exponential‑gradient refinement** – for each of the top candidates
    run a mirror‑descent (exponentiated‑gradient) optimiser that
    minimises the smooth “soft‑max’’ surrogate of the convolution’s
    maximum.  The temperature starts low (smooth) and is increased
    multiplicatively, while the step size decays like 1/√t.

3.  **Systematic local hill‑climbing** – on the current best vector,
    repeatedly move a small fraction of mass from the coordinate with
    the largest gradient (most responsible for the peak) to the one
    with the smallest gradient, accepting only improving moves.

All helpers are defined at module level, there are no closures or file
IO, and the routine respects the supplied time‑budget.  The final
vector is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) distribution of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """Symmetric 1/(i·(n‑i))^p (invert=True) or (i·(n‑i))^p (invert=False)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    vals = prod ** (-p) if invert else prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) distribution on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on the index set."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """Convex combination (1‑lam)·arcsine + lam·uniform."""
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` when the vector is a
    probability vector (sum == 1): 2·n·max(a*a).
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a*a) w.r.t. a at the active index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·b_k) wrt a.
    Implements g = 2·conv_rev(w, a_rev) where w are the softmax
    weights over the convolution entries.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser that minimizes the
    smooth soft‑max surrogate of the convolution peak.  Returns the
    best vector and its exact objective value.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    # hyper‑parameters (tuned for speed / quality)
    eta0 = 1.5               # initial step‑size factor
    alpha = 5.0              # start temperature (smooth)
    alpha_mul = 1.04
    alpha_max = 5000.0
    size_fft_soft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # convolution and exact score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} exact_score={best_score:.6f}")

        # smooth gradient (soft‑max surrogate)
        grad = _gradient_softmax(a, b, alpha, size_fft_soft)

        # diminishing step‑size
        eta = eta0 / (np.sqrt(it) + 1.0)

        # exponentiated‑gradient update (with clipping to avoid overflow)
        upd = -eta * grad
        upd = np.clip(upd, -50, 50)          # safety for exp()
        a = a * np.exp(upd)
        a = np.maximum(a, 0.0)

        # renormalise onto simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # temperature schedule
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaus
        if it % 2000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing that moves mass from the index with the
    largest soft‑max gradient to the one with the smallest, accepting
    only moves that strictly improve the exact objective.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # compute full convolution and the soft‑max weights (same as in optimiser)
        b = _fft_self_convolve(a)
        # use a moderate temperature to obtain a stable direction vector
        grad = _gradient_softmax(a, b, alpha=200.0,
                                size_fft=1 << ((3 * n - 2).bit_length()))

        i = int(np.argmax(grad))   # most responsible for the peak
        j = int(np.argmin(grad))   # least responsible

        if i == j or a[i] <= 1e-15:
            continue

        # shift up to 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.1
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps

        cur_score = _objective(a_candidate)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine respects the time budget,
    runs a broad random search, refines the best candidates with an
    exponentiated‑gradient optimiser, finishes with a systematic local
    hill‑climb, and returns the best vector found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Determine admissible lengths (respecting the 20 000 upper bound)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    length_factors = (0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    })
    # ensure a few classic sizes are present
    lengths.extend([5000, 8000, 12000, 16000, 20000])
    lengths = sorted({l for l in lengths if l <= 20000})

    # --------------------------------------------------------------
    # Global bookkeeping
    best_seq = None
    best_score = float("inf")
    best_candidates = []   # (score, vector) tuples

    # -----------------------------------------------------------------
    # Optional warm‑start from a previously found solution
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if seq0.size:
            seq0 = np.maximum(seq0, 0.0)
            seq0 = _project_onto_simplex(seq0, s=1.0)
            score0 = _objective(seq0)
            print(f"Warm‑start score={score0:.6f}")
            best_seq = seq0.copy()
            best_score = score0
            best_candidates.append((score0, seq0))
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Seed the pool with a plain arcsine distribution of the base length
    arcs_seq = _arcsine_distribution(base_n)
    arcs_score = _objective(arcs_seq)
    print(f"Arcsine (n={base_n}) score={arcs_score:.6f}")
    if arcs_score < best_score:
        best_seq = arcs_seq.copy()
        best_score = arcs_score
    best_candidates.append((arcs_score, arcs_seq))

    # --------------------------------------------------------------
    # Phase 1 – random parametric search
    families = ("power_invert", "power_direct", "beta", "gauss", "mix")
    search_deadline = deadline - 250.0   # reserve ~250 s for refinements

    while time.time() < search_deadline:
        n = int(np.random.choice(lengths))
        fam = np.random.choice(families)

        if fam == "power_invert":
            p = np.random.uniform(0.2, 2.0)
            seq = _power_law_distribution(n, p, invert=True)
        elif fam == "power_direct":
            p = np.random.uniform(0.2, 2.0)
            seq = _power_law_distribution(n, p, invert=False)
        elif fam == "beta":
            a = np.random.uniform(0.1, 5.0)
            seq = _beta_distribution(n, a)
        elif fam == "gauss":
            sigma = np.random.uniform(n / 25.0, n / 2.0)
            seq = _gaussian_distribution(n, sigma)
        elif fam == "mix":
            lam = np.random.uniform(0.0, 1.0)
            seq = _mixture_arcsine_uniform(n, lam)
        else:   # fallback (should not happen)
            seq = _arcsine_distribution(n)

        seq = _project_onto_simplex(seq, s=1.0)
        score = _objective(seq)

        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            print(f"New global best (rand)  len={n} fam={fam} score={best_score:.6f}")

        # maintain a small pool of promising candidates (max 12)
        if len(best_candidates) < 12:
            best_candidates.append((score, seq))
        else:
            # replace worst if new one is better
            worst_idx = max(range(len(best_candidates)),
                            key=lambda i: best_candidates[i][0])
            if score < best_candidates[worst_idx][0]:
                best_candidates[worst_idx] = (score, seq)

    # --------------------------------------------------------------
    # Phase 2 – exponential‑gradient refinement of top candidates
    best_candidates.sort(key=lambda x: x[0])
    top_to_refine = [cand[1] for cand in best_candidates[:5]]

    for cand_seq in top_to_refine:
        if time.time() >= deadline - 30.0:
            break
        n = cand_seq.shape[0]
        remaining = deadline - time.time()
        # give each run at most 70 s (or a fraction of the remaining budget)
        slice_deadline = time.time() + min(70.0, remaining * 0.55)
        refined_seq, _ = _optimize_exponential(cand_seq, n, slice_deadline)
        refined_score = _objective(refined_seq)
        if refined_score < best_score - 1e-12:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"Refinement improved to {best_score:.6f} (n={n})")

    # --------------------------------------------------------------
    # Phase 3 – final systematic local hill climbing
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        refined_seq, _ = _local_refinement(best_seq, n, deadline)
        refined_score = _objective(refined_seq)
        if refined_score < best_score - 1e-12:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"After local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Safety net – fall back to uniform if nothing was found
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    # Final reporting
    print(f"Final best score (objective) = {best_score:.6f} (length = {best_seq.shape[0]})")
    # Return as ordinary Python list
    return best_seq.tolist()
```

This script implements a multi‑stage optimisation pipeline that respects the 1000 s budget, explores a rich family of candidate shapes, refines the most promising ones with a smooth‑max exponentiated‑gradient method, and finishes with a deterministic hill‑climbing step.  The returned list consists of non‑negative heights that, when fed to `evaluate_sequence`, yields a score that should improve upon the previous best bound (≈ 1.512).
