Algorithm (≈ 2 min per candidate):
1.  Sample a diverse set of probability vectors on a moderate length (≈1000)
    – uniform, arcsine, symmetric Beta(α,α) for many α (including very
    small α), and random mixtures of up to three Beta components.
2.  Keep the best K (K≈12) candidates w.r.t. the *raw* objective
    2 n·max(a∗a) (sum(a)=1 → identical to the black‑box score).
3.  For each kept candidate run a few iterations of the *alternating‑LP*
    scheme from the paper:
        •  given a, let M = max(a∗a)
        •  solve   max  Σb_j
          s.t.    b ≥ 0
                  a ∗ b ≤ M   (full linear convolution)
        •  scale b to sum = 1 and replace a ← (1‑t)·a + t·b  (t≈0.5)
    The LP is built with a sparse Toeplitz matrix and solved by
    `scipy.optimize.linprog` (HiGHS).  A small number of iterations
    (≈6) already reduces the ℓ∞‑convolution peak.
4.  Finish with a short projected sub‑gradient descent on the exact
    ℓ∞‑self‑convolution (the same step used in the previous baseline)
    to polish the result.
5.  Return the best sequence found (as a plain Python list).

The whole search respects the 1000 s budget, falls back to a uniform
vector if nothing is produced, and never uses negative numbers.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalized to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow/underflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum value of a*a."""
    return _fft_self_convolve_full(a).max()

# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1) × n sparse matrix that maps b ↦ f∗b (full convolution).
    Row k (0‑based) contains f_i where i+k−i = k, i.e. coeff f_i for b_{k‑i}.
    """
    n = f.shape[0]
    # i runs over 0..n‑1 (entry of f), j over 0..n‑1 (entry of b)
    i = np.arange(n)[:, None]          # shape (n,1)
    j = np.arange(n)[None, :]          # shape (1,n)
    rows = i + j                       # shape (n,n) → each row index k = i+j
    cols = np.broadcast_to(j, (n, n))
    data = np.broadcast_to(f[:, None], (n, n))
    # flatten in column‑major order so that each column (fixed j) is contiguous
    rows_flat = rows.ravel(order='F')
    cols_flat = cols.ravel(order='F')
    data_flat = data.ravel(order='F')
    mat = sparse.csr_matrix((data_flat, (rows_flat, cols_flat)),
                            shape=(2 * n - 1, n))
    return mat

# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               max_iter: int = 6,
               step: float = 0.5,
               deadline: float = 1e30) -> np.ndarray:
    """
    Alternating linear‑program refinement.
    a must be non‑negative and sum to 1.
    """
    n = a.shape[0]
    a = a.copy()
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        M = _fft_self_convolve_max(a)          # current ℓ∞‑self‑conv.
        # Build constraint matrix for a∗b ≤ M
        A = _build_conv_matrix(a)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Solve   max Σ b_j   ⇔   min -Σ b_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # abort the refinement if the LP fails
            print(f"[{time.time():.1f}] LP failed (iter {it}): {res.message}")
            break

        b = res.x
        # Normalise b back to a probability vector
        b_sum = b.sum()
        if b_sum == 0.0:
            b = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            b = b / b_sum

        # Convex combination step
        a = (1.0 - step) * a + step * b
        # Re‑project onto simplex to cure numerical drift
        a = _project_onto_simplex(a)
        # small safeguard against numerical overflow
        a = np.clip(a, 0.0, 1e6)

    # final normalisation
    a = _project_onto_simplex(a)
    return a

# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _fft_self_convolve_full(a)
    cur_max = b.max()
    # tolerance for active set
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    # convolution a_rev * w (full) gives gradient of each entry
    grad_full = np.fft.irfft(
        np.fft.rfft(a_rev) * np.fft.rfft(w, n=1 << ((a_rev.size + w.size - 1).bit_length())),
        n=1 << ((a_rev.size + w.size - 1).bit_length()))
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, cur_max

def _subgradient_refine(a0: np.ndarray,
                        max_iter: int = 2000,
                        step0: float = 0.05,
                        deadline: float = 1e30) -> np.ndarray:
    """Projected sub‑gradient descent on max(a*a)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = 2.0 * n * _fft_self_convolve_max(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        grad, cur_max = _max_subgradient_step(a)
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)
        # occasional jitter to escape plateaus
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

        cur_val = 2.0 * n * _fft_self_convolve_max(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it}  score={best_val:.6f}")

    return best_a

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence score.
    The routine builds a pool of Beta‑based candidates, refines the best ones
    with an alternating LP scheme and a finishing sub‑gradient descent.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # ------------------------------------------------------------------
    # Helper: raw objective (identical to evaluate_sequence when sum=1)
    def _raw_score(vec: np.ndarray) -> float:
        n = vec.shape[0]
        return 2.0 * n * _fft_self_convolve_max(vec)

    # ------------------------------------------------------------------
    # Length to use for the intensive refinement (balance speed / quality)
    # 1000 works well for the LP size and already gives a score close to the
    # asymptotic constant.
    n_refine = 1000

    # ------------------------------------------------------------------
    # Collect a large pool of initial candidates (raw score, vector)
    pool = []          # list of tuples (score, vector)

    def _maybe_add(vec: np.ndarray):
        vec = _project_onto_simplex(vec)
        sc = _raw_score(vec)
        pool.append((sc, vec))

    # 1) base solution if available
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base = _project_onto_simplex(base)
        if base.shape[0] != n_refine:
            # interpolate to the target length
            x_old = np.linspace(0.0, 1.0, base.shape[0])
            x_new = np.linspace(0.0, 1.0, n_refine)
            base = np.interp(x_new, x_old, base)
        _maybe_add(base)
    except Exception:
        pass

    # 2) simple baselines
    _maybe_add(np.full(n_refine, 1.0 / n_refine, dtype=np.float64))   # uniform
    _maybe_add(_beta_distribution(n_refine, 0.5, 0.5))                # arcsine

    # 3) grid of symmetric Beta(α,α)
    alphas = np.concatenate((
        np.linspace(0.02, 0.5, 12, endpoint=False),
        np.linspace(0.5, 2.0, 15),
        np.linspace(2.0, 5.0, 8)
    ))
    for a in alphas:
        if time.time() >= deadline:
            break
        vec = _beta_distribution(n_refine, a, a)
        _maybe_add(vec)

    # 4) random mixtures of up to three Betas
    mix_trials = 250
    for _ in range(mix_trials):
        if time.time() >= deadline:
            break
        k = np.random.choice([2, 3])               # number of components
        weights = np.random.rand(k)
        weights /= weights.sum()
        mix = np.zeros(n_refine, dtype=np.float64)
        for w in weights:
            a = np.random.rand() * 4.9 + 0.1
            b = np.random.rand() * 4.9 + 0.1
            mix += w * _beta_distribution(n_refine, a, b)
        _maybe_add(mix)

    # Keep the best K candidates for the expensive refinement
    K = 15
    pool.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in pool[:K]]

    print(f"[{time.time():.1f}] pool built, best raw score={pool[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Expensive refinement (LP + sub‑gradient) on the selected candidates
    best_vec = None
    best_score = float("inf")

    for idx, cand in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Refinement start for candidate {idx+1}/{len(top_candidates)}")
        # 1) alternating LP
        lp_deadline = deadline - 5.0   # keep a tiny buffer for later steps
        a_lp = _lp_refine(cand,
                          max_iter=6,
                          step=0.5,
                          deadline=lp_deadline)
        # 2) sub‑gradient polishing
        sub_deadline = deadline - 2.0
        a_ref = _subgradient_refine(a_lp,
                                    max_iter=3000,
                                    step0=0.07,
                                    deadline=sub_deadline)

        sc = _raw_score(a_ref)
        print(f"[{time.time():.1f}] candidate {idx+1} refined score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_vec = a_ref.copy()
            print(f"   ** new overall best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If nothing succeeded (unlikely), fall back to a uniform vector
    if best_vec is None:
        print("Fallback to uniform vector.")
        best_vec = np.full(n_refine, 1.0 / n_refine, dtype=np.float64)
        best_score = _raw_score(best_vec)

    # Final verification with the black‑box evaluator (clamping etc.)
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_vec)})")
    return best_vec.tolist()
