"""
Propose a low‑score non‑negative step function.

The search works in three phases:

1. **Analytic sampling** – a collection of simple families (arcsine,
   symmetric beta, triangular, exponential, Gaussian) are sampled for a
   handful of lengths (500 … ≈ 2 · base_length).  The best candidate is kept
   as a starting point.

2. **Adam‑accelerated projected gradient descent** – the exact sub‑gradient
   of the objective `max(a*a)` (the active convolution entries) is used.
   The gradient is `2 * conv_rev(active)`, i.e. the previously used
   “‑ a_i” correction is omitted (the earlier implementation under‑estimated
   the gradient).  An Adam optimiser with a diminishing learning rate is
   applied, followed by Euclidean projection onto the probability simplex.

3. **Local random‑transfer hill climbing** – small amounts of mass are moved
   between random positions; improvements are kept.  The step size of the
   transfer decays over time, and occasional jitter helps to escape plateaux.

The whole routine runs for at most ``budget_s`` seconds (≈1000 s) and returns
the best sequence found.  All numbers are non‑negative; the final sequence is
normalised to sum = 1, which does not affect the evaluation score
(`evaluate_sequence` is scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p, p) on a regular grid, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # safe evaluation of x**(p-1)*(1‑x)**(p‑1)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalised to sum = 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float = 0.2) -> np.ndarray:
    """Discrete symmetric Gaussian (truncated), normalised to sum = 1."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(a*a) assuming sum(a)=1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a list of floats (sum ≈ 1) ready for evaluation.
    """
    np.random.seed(seed)
    # give a tiny safety margin for the caller
    deadline = time.time() + budget_s - 2.0

    # ---------- 1. choose lengths ----------
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    lengths = set()
    for factor in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0):
        n = int(base_n * factor)
        if n >= 500:
            lengths.add(n)
    # extra random lengths, up to three times the base but never >20000
    for _ in range(7):
        n = np.random.randint(500, min(20000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ---------- 2. analytic families ----------
    best_seq = None               # type: np.ndarray | None
    best_score = float("inf")

    beta_ps = [0.10, 0.20, 0.30, 0.40, 0.50, 0.70, 1.00, 1.50, 2.00]
    sigma_vals = [0.12, 0.18, 0.25, 0.35]          # for Gaussian
    lam_vals = [0.003, 0.005, 0.01, 0.02]         # for exponential

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine
        a = _arcsine_distribution(n)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] n={n} arcsine   score={sc:.6f}")

        # symmetric beta
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] n={n} beta(p={p}) score={sc:.6f}")

        # triangular
        a = _triangular_distribution(n)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] n={n} triangular score={sc:.6f}")

        # exponential (several lambdas)
        for lam in lam_vals:
            if time.time() >= deadline:
                break
            a = _exponential_distribution(n, lam=lam)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] n={n} exponential(lam={lam}) score={sc:.6f}")

        # Gaussian
        for sigma in sigma_vals:
            if time.time() >= deadline:
                break
            a = _gaussian_distribution(n, sigma=sigma)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] n={n} gaussian(sigma={sigma}) score={sc:.6f}")

    # ---------- 3. incorporate previous construction ----------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # Fallback: uniform if nothing else succeeded
    if best_seq is None:
        n = max(500, base_n)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ---------- 4. projected Adam gradient descent ----------
    a = best_seq.copy()                     # keep sum = 1
    n = a.shape[0]
    m = np.zeros_like(a)                    # first moment
    v = np.zeros_like(a)                    # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.4                               # initial learning‑rate factor
    it = 0
    target = 1.503

    while time.time() < deadline:
        it += 1
        # compute convolution and active set
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)

        # construct indicator for active entries
        w = np.zeros_like(b)
        w[active] = 1.0

        # exact sub‑gradient of max(a*a)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # **no diagonal correction**

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = step0 / np.sqrt(it)                 # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # keep on the simplex
        a = _project_onto_simplex(a, s=1.0)

        # evaluate
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam iter={it} new best {cur_score:.6f}")

        if best_score <= target:
            break

    # ---------- 5. local random‑transfer hill climbing ----------
    a = best_seq.copy()
    cur_score = best_score
    it = 0
    max_transfer = 0.05                     # at most 5 % of total mass per trial

    while time.time() < deadline:
        it += 1
        transfer = max_transfer * (0.5 ** (it / 20000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] transfer iter={it} improved best {best_score:.6f}")

        # occasional jitter to escape flat regions
        if it % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 1e-4
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
