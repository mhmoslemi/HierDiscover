    # returns refined vector and its cheap score.
    while time.time() < deadline:
        # compute current conv and score
        conv = _fft_self_convolve(a)
        cur_score = 2 * a.shape[0] * conv.max()
        # if no improvement possible break
        # solve LP
        n = a.shape[0]
        M = conv.max()
        # build matrix
        A = _build_conv_matrix(a)
        # solve
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=[(0, None)] * n, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:

        g_norm = g / sum_g

2. **LP refinement** – for each kept candidate we run a short sequence
   of linear‑programming steps.  With the current vector `f` we compute
   `M = max(f*f)`.  The LP solves

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for all k
                    g_j ≥ 0

   (convolution is linear in `g`, thus the constraints are linear).
   The solution `g` is renormalised to a probability vector and mixed
   with the current `f`.  The procedure is repeated a few times; each
   iteration reduces the maximal self‑convolution and therefore the
   evaluation score.

3. **Optional gradient polishing** – if SciPy is not available we fall
   back to the exponentiated‑gradient optimiser used in the previous
   state‑of‑the‑art implementation.

The search respects the time budget (default 1000 s) and returns the
best sequence found as a plain Python list.
"""

import time
import math
import random
from typing import List, Tuple, Optional

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used for the LP refinement stage.
try:
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix, csr_matrix
    SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, v.shape[0] + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast for moderate n)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    if vals.sum() == 0.0:            # fallback to uniform
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / vals.sum()
    return vals


def _cheap_score(a: np.ndarray) -> float:
    """Score assuming Σa = 1 → 2·n·max(conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Construct the (2n‑1)×n matrix A with A[k,i] = f_i
    if 0 ≤ k‑i < n, otherwise 0.
    Then (f * g)_k = (A @ g)_k.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = lil_matrix((rows, n), dtype=np.float64)
    for i, fi in enumerate(f):
        if fi == 0.0:
            continue
        # rows i … i+n‑1 receive the value fi in column i
        A[i:i + n, i] = fi
    return A.tocsr()


def _lp_refine(
    a: np.ndarray,
    deadline: float,
    max_outer: int = 8,
    step_candidates: Tuple[float, ...] = (1.0, 0.5, 0.25, 0.1),
) -> Tuple[np.ndarray, float]:
    """
    Perform a few LP‑based refinement steps on vector a (already normalised).
    Returns the refined vector and its cheap score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _cheap_score(best_a)

    for outer in range(max_outer):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(best_a)
        M = conv.max()

        # Build the constraint matrix A for (best_a * g) ≤ M
        A = _build_conv_matrix(best_a)

        # Linear programme: maximise Σ g_j  ⇔ minimise −Σ g_j
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Solve – using HiGHS if available, otherwise the default method.
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception:            # pragma: no cover
            # If something goes wrong, abandon LP refinement.
            break

        if not res.success:
            # Infeasibility is unexpected (g = 0 is always feasible).
            break

        g = res.x
        g_sum = g.sum()
        if g_sum <= 1e-12:
            break
        g_norm = g / g_sum

        improved = False
        cur_score = best_score

        for t in step_candidates:
            candidate = (1.0 - t) * best_a + t * g_norm
            candidate = _project_onto_simplex(candidate, s=1.0)
            cand_score = _cheap_score(candidate)

            if cand_score < cur_score - 1e-12:
                best_a = candidate
                best_score = cand_score
                improved = True
                break

        if not improved:
            # No step size yielded improvement – stop LP refinement.
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Fallback optimiser (exponentiated‑gradient) – used when SciPy is absent.
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the soft‑max surrogate of max(conv)."""
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    # convolution of reversed a with w, then pick the central n entries
    size = 1 << ((a.size + w.size - 1) - 1).bit_length()
    A = np.fft.rfft(a[::-1], n=size)
    B = np.fft.rfft(w, n=size)
    conv_rev = np.fft.irfft(A * B, n=size)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


def _exponential_update(
    a: np.ndarray, grad: np.ndarray, eta: float, momentum: np.ndarray, beta: float
) -> Tuple[np.ndarray, np.ndarray]:
    """Multiplicative (exponentiated) update with simple momentum."""
    momentum = beta * momentum + (1.0 - beta) * grad
    shift = np.max(momentum)
    factor = np.exp(-eta * (momentum - shift))
    a = a * factor
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s
    return a, momentum


def _fallback_exponential_opt(
    a0: np.ndarray, deadline: float, max_iter: int = 30000
) -> Tuple[np.ndarray, float]:
    """Simple exponential‑gradient optimiser (the one used previously)."""
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    eta0 = 0.7
    temp = 1.0
    temp_factor = 1.004
    temp_max = 4000.0
    best_a = a.copy()
    best_score = _cheap_score(best_a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _fft_self_convolve(a)

        # keep best
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # gradient
        if temp < 600.0:
            grad = _softmax_gradient(a, conv, temp)
            temp = min(temp_max, temp * temp_factor)
        else:
            # fall back to exact sub‑gradient (same as in the original code)
            cur_max = conv.max()
            eps = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(conv >= cur_max - eps)
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size
            size = 1 << ((a.size + w.size - 1) - 1).bit_length()
            A = np.fft.rfft(a[::-1], n=size)
            B = np.fft.rfft(w, n=size)
            conv_rev = np.fft.irfft(A * B, n=size)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        eta = eta0 / np.sqrt(it)
        a, momentum = _exponential_update(a, grad, eta, momentum, beta_mom)

        # occasional jitter
        if it % 3000 == 0 and random.random() < 0.1:
            jitter = np.random.randn(n) * 1e-7
            a += jitter
            a = _project_onto_simplex(a, s=1.0)

        if best_score <= 1.503:          # early stop if we already hit target
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The algorithm respects the time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0            # tiny safety margin

    # ------------------------------------------------------------------
    # 0. Use a previous construction (if any) as a possible starting point.
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base_n = prev_seq.size
    except Exception:
        prev_seq = None
        base_n = 3000                                 # reasonable default length

    # ------------------------------------------------------------------
    # 1. Choose a modest set of lengths to explore.
    length_factors = (0.75, 1.0, 1.25, 1.5)
    lengths = sorted(
        max(800, int(base_n * f))
        for f in length_factors
        if int(base_n * f) >= 800
    )
    # ensure at least one length
    if not lengths:
        lengths = [base_n]

    # ------------------------------------------------------------------
    # 2. Generate analytic candidates (Beta‑laws, uniform, random, prev).
    shape_params = [
        (0.15, 0.15),
        (0.20, 0.20),
        (0.30, 0.30),
        (0.40, 0.40),
        (0.50, 0.50),          # arcsine
        (0.70, 0.70),
        (1.00, 1.00),          # uniform
        (0.15, 0.85),
        (0.85, 0.15),
    ]

    all_candidates: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        # sweep Beta shapes
        for a, b in shape_params:
            vec = _beta_distribution(n, a, b)
            score = _cheap_score(vec)
            all_candidates.append((score, vec))

        # add a uniform vector
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        all_candidates.append((_cheap_score(uniform), uniform))

        # add a random Dirichlet draw
        rnd = np.random.rand(n)
        rnd /= rnd.sum()
        all_candidates.append((_cheap_score(rnd), rnd))

        # add the previous construction (interpolated if needed)
        if prev_seq is not None:
            if prev_seq.size != n:
                x_old = np.linspace(0.0, 1.0, prev_seq.size)
                x_new = np.linspace(0.0, 1.0, n)
                vec = np.interp(x_new, x_old, prev_seq)
                vec = _project_onto_simplex(vec, s=1.0)
            else:
                vec = prev_seq.copy()
            all_candidates.append((_cheap_score(vec), vec))

    # ------------------------------------------------------------------
    # 3. Keep a small pool of the most promising starters.
    all_candidates.sort(key=lambda x: x[0])
    starters = [vec.copy() for _, vec in all_candidates[:6]]
    print(f"[{time.time():.1f}] Generated {len(all_candidates)} candidates, "
          f"keeping {len(starters)} starters.")

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 4. Refine each starter.
    for idx, start in enumerate(starters):
        n = start.shape[0]
        sub_deadline = min(deadline, time.time() + 120.0)   # ≤2 min per starter

        print(f"[{time.time():.1f}] Refining starter {idx+1}/{len(starters)} (n={n})")

        # ---- LP refinement (if SciPy is available) --------------------
        if SCIPY_AVAILABLE:
            refined, cheap = _lp_refine(start, sub_deadline)
        else:                                   # pragma: no cover
            refined, cheap = _fallback_exponential_opt(start, sub_deadline)

        # ---- Final cheap evaluation ------------------------------------
        final_cheap_score = cheap
        print(f"[{time.time():.1f}] Starter {idx+1}: cheap score = {final_cheap_score:.6f}")

        # ---- Evaluate with the official (slow) black‑box ---------------
        try:
            final_score = evaluate_sequence(refined.tolist())
        except Exception:
            final_score = final_cheap_score   # fallback to cheap value

        print(f"[{time.time():.1f}] Starter {idx+1}: official score = {final_score:.6f}")

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = refined.copy()
            print(f"*** New global best (starter {idx+1}) → {overall_best_score:.6f}")

        if overall_best_score <= 1.503:
            print("Target achieved – exiting early.")
            break

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 5. Fallback in the unlikely case that nothing succeeded.
    if overall_best_seq is None:
        n_fallback = 1000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())  # type: ignore

    print(f"Final best score: {overall_best_score:.6f} "
          f"(length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
