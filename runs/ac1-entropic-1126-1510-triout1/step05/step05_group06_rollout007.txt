assistantfinal**Solution Overview**

The goal is to minimise  

\[
E(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2},
\]

where `a` is a non‑negative vector (the step‑function heights) and `*`
denotes the discrete convolution.  
Because the denominator scales exactly like the square of the numerator,
`E(a)` is invariant under a global scaling of `a`.  Hence we can fix
\(\sum a = 1\) and simply minimise  

\[
F(a)=2\,n\,\max (a*a).
\]

The optimisation is carried out in three phases

1. **Initial shapes** – a small collection of hand‑crafted seeds
   (uniform, arcsine law, random, grid‑search over the symmetric
   power‑law family, and the previous best if available).

2. **Smooth L‑BFGS optimisation** – we replace the non‑smooth max by a
   log‑sum‑exp surrogate  

   \[
   \Phi_\alpha(a)=\frac1\alpha\log\!\sum_k e^{\alpha\, (a*a)_k},
   \]

   with a temperature `α`.  For a fixed `α` the function is smooth;
   its gradient can be written as a convolution and is obtained
   efficiently with FFTs.  The variables are re‑parameterised with a
   soft‑max (`a_i = \exp(x_i)/\sum_j\exp(x_j)`) which removes the
   positivity and simplex constraints.  We run L‑BFGS on this smooth
   problem, slowly increasing `α` (continuation method) until it is
   large enough that the surrogate is practically the true `max`.

3. **Local stochastic hill‑climbing** – after the continuous optimisation
   we perform cheap random mass transfers between two indices; each
   transfer is kept only if the true `E(a)` improves.

The whole search respects the 1000 s budget, repeatedly prints progress,
and finally returns the best sequence found.

    Gradients are obtained via FFT‑based convolutions.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) self‑convolution via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _beta_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‐law a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


def _asymmetric_power_shape(n: int, pl: float, pr: float) -> np.ndarray:
    """Asymmetric power‑law a_i ∝ (i+½)⁻ˡ·(n‑i‑½)⁻ʳ, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-pl) * (n - i - 0.5) ** (-pr)
    vals /= vals.sum()
    return vals


def _grid_search_best(n: int) -> tuple[np.ndarray, float]:
    """Very cheap scan of a few (p,w) and asymmetric families."""
    best_seq = None
    best_score = np.inf

    # symmetric power‑law + uniform mixture
    p_vals = np.linspace(0.30, 0.70, 9)
    w_vals = np.linspace(0.0, 1.0, 11)
    for p in p_vals:
        shape = _beta_power_shape(n, p)
        for w in w_vals:
            cand = (1.0 - w) * shape + w / n
            cand = _project_onto_simplex(cand)
            score = 2.0 * n * _fft_self_convolve(cand).max()

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a vector on the simplex)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a*a)_k w.r.t. a.
    Implements   g = 2·conv(a_rev , w)   (restricted to the central n entries).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + len(w) - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv_full = np.fft.irfft(A_rev * W, n=size)
    conv_full = conv_full[: n + len(w) - 1]               # exact length
    start = n - 1
    g = 2.0 * conv_full[start:start + n]
    return g


def _objective_with_alpha(x: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    """
    Smooth surrogate for the max of the self‑convolution.

    Returns (value, gradient) where the gradient is w.r.t. the
    unconstrained variable x (a = softmax(x)).
    """
    a = _softmax(x)
    conv = _fft_self_convolve(a)
    max_conv = np.max(conv)

    # soft‑max weights for the log‑sum‑exp
    shift = conv - max_conv                     # ≤ 0
    exp_shift = np.exp(alpha * shift)
    sum_exp = np.sum(exp_shift)
    w = exp_shift / sum_exp                     # length 2·n‑1, sums to 1

    # gradient wrt a
    g_a = _weighted_gradient(a, w)

    # gradient wr.t the soft‑max parameters x
    dot = np.dot(a, g_a)
    grad_x = a * (g_a - dot)

    # surrogate value: (1/α)·log∑exp(α·conv) = max + log(sum_exp)/α
    value = max_conv + np.log(sum_exp) / alpha
    return float(value), grad_x.astype(np.float64)

def _local_refinement(a_start: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: transfer a small fraction of mass
    from one index to another whenever the true evaluation score improves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.10
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur_score = 2.0 * n * np.max(_fft_self_convolve(cand))
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            a = cand
            # occasional progress output
            if it % 5000 == 0:
                print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_a, best_score

                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the best
    list of heights discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Choose a handful of lengths (large enough to be strong but cheap enough).
    base_n = 8000
    try:
        base_n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        pass
    lengths = sorted({
        max(500, int(base_n * f))
        for f in (0.75, 1.0, 1.25, 1.5, 2.0)
        if max(500, int(base_n * f)) <= 20000
    })
    if not lengths:
        lengths = [8000]

    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        remaining = deadline - time.time()
        slice_deadline = time.time() + max(10.0, remaining * 0.35)

        # --------------------------------------------------------------
        # Build a small pool of promising initial vectors
        init_candidates = []

        # a) hand‑crafted seeds
        init_candidates.append(_beta_power_shape(n, 0.5))               # arcsine
        init_candidates.append(np.full(n, 1.0 / n))                    # uniform
        random_vec = np.random.rand(n)
        random_vec /= random_vec.sum()
        init_candidates.append(random_vec)                             # random

        # b) previous best if it has the same length
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)    # type: ignore[name-defined]
            if prev.size == n:
                prev = np.maximum(prev, 0.0)
                prev = prev / prev.sum()
                init_candidates.append(prev)
        except Exception:
            pass

        # c) cheap grid‑search result
        shape_best, _ = _grid_search_best(n)
        init_candidates.append(shape_best)

        # Keep only the few most promising according to a fast cheap score
        def cheap_score(v):
            return 2.0 * n * _fft_self_convolve(v).max()
        init_candidates = sorted(init_candidates, key=cheap_score)[:4]

        # --------------------------------------------------------------
        # Optimise each candidate with the continuation L‑BFGS routine
        for a0 in init_candidates:
            if time.time() >= slice_deadline:
                break
            seq, internal_score = _optimize_shape_l_bfgs(a0, n, slice_deadline)

            # Evaluate the true black‑box score
            try:
                true_score = evaluate_sequence(seq.tolist())
            except Exception:
                true_score = internal_score

            if true_score < overall_best_score:
                overall_best_score = true_score
                overall_best_seq = seq.copy()
                print(f"*** New global best → len={n} score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break

        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Final stochastic refinement (cheap hill‑climbing)
    if overall_best_seq is not None and time.time() < deadline - 2.0:
        n = len(overall_best_seq)
        seq_refined, _ = _local_refinement(overall_best_seq, n, deadline)
        try:
            refined_score = evaluate_sequence(seq_refined.tolist())
        except Exception:
            refined_score = float("inf")
        if refined_score < overall_best_score:
            overall_best_score = refined_score
            overall_best_seq = seq_refined.copy()
            print(f"*** After local refinement score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fall‑back
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Safety fallback used.")

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
