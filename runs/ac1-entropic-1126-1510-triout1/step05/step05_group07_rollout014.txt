def _adam_optimize_subgrad(init_a, n, deadline):
    a = init_a.copy()
    best = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1,beta2 = 0.9,0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[AdamSub] it={t} best={best_score:.6f}")
        # gradient subgrad
        g,_ = _subgradient(a) # returns gradient g
        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g*g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.5 / (np.sqrt(v_hat) + eps) # base lr 0.5 maybe
        a = a - step * m_hat
        a = _project_onto_simplex(a, 1.0)
    return best, best_score

"""
Optimise the step‑function heights that minimise ``evaluate_sequence``.

Algorithm
---------
1. **Candidate generation** – a modest pool of hand‑crafted probability
   vectors (uniform, arcsine, symmetric β‑laws, mixtures,
   edge‑spikes, tapered and exponential shapes, a few Dirichlet draws,
   and the previous best if available).  All candidates are projected onto
   the simplex Σa_i = 1.

2. **Fast selection** – evaluate every candidate with the provided
   ``evaluate_sequence`` and keep the best one.

3. **Projected Adam on the exact sub‑gradient** – the objective is
   `F(a) = 2·n·max(conv(a))` where `conv(a)=a*a`.  Its sub‑gradient can be
   written in closed form:

        g_j = 2·∑_{k∈K_max} a_{k‑j}   (indices out of range are ignored),

   where `K_max` is the set of convolution positions attaining the
   maximum.  This gradient is cheap to compute (one FFT for the convolution
   plus a tiny O(|K_max|·n) loop).  Adam is applied to this sub‑gradient,
   with projections onto the simplex after each update.  The bulk of the
   time budget (≈80 %) is spent here, starting from the best candidate.

4. **Stochastic hill‑climbing polish** – a light local‑search that moves a
   random fraction of mass between two positions (and occasional tiny
   jitter) while keeping the best solution seen.

The whole procedure respects the 1000 s limit; the final list of heights
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) PDF evaluated on mid‑points of n bins."""
    a = max(a, 0.05)                          # avoid division by zero
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    s = pdf.sum()
    if s == 0 or not np.isfinite(s):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= s
    return pdf


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Hand‑crafted pool of probability vectors (all lie on the simplex)."""
    candidates = []

    # 1) Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) Arcsine (Beta(½,½))
    candidates.append(_arcsine_distribution(n))

    # 3) Symmetric Beta(a,a) for a range of a (including very small ones)
    for a in np.linspace(0.08, 0.90, 9):
        candidates.append(_beta_distribution(n, a))

    # 4) Uniform + arcsine mixtures
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    arcsine = _arcsine_distribution(n)
    for w in np.linspace(0.0, 0.5, 6):
        candidates.append((1.0 - w) * arcsine + w * uniform)

    # 5) Edge spikes (added to the arcsine shape)
    for spike_frac in (0.05, 0.10, 0.15):
        mix = arcsine.copy()
        mix[0] += spike_frac / 2
        mix[-1] += spike_frac / 2
        mix = _project_onto_simplex(mix, 1.0)
        candidates.append(mix)

    # 6) Simple tapered shape (≈ 1/i + 1/(n‑i+1))
    i = np.arange(1, n + 1, dtype=np.float64)
    taper = 1.0 / i + 1.0 / (n + 1 - i)
    taper = _project_onto_simplex(taper, 1.0)
    candidates.append(taper)

    # 7) Exponential tapers with a few decay rates
    for rate in (5e-4, 1e-3, 2e-3):
        exp_shape = np.exp(-rate * np.arange(n)) + np.exp(-rate * np.arange(n)[::-1])
        exp_shape = _project_onto_simplex(exp_shape, 1.0)
        candidates.append(exp_shape)

    # 8) A few random Dirichlet draws (various concentration parameters)
    rng = np.random.default_rng()
    for alpha in (0.2, 0.5, 1.0):
        rand = rng.dirichlet(np.full(n, alpha))
        candidates.append(rand)

    # 9) Previous best (if compatible)
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())

    # final safety: make sure everything lives on the simplex
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of `max(conv(a))` w.r.t. `a`.
    Returns (g, max_val) where
        g_j = 2·∑_{k∈K_max} a_{k‑j}   (out‑of‑range indices ignored)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2·n‑1
    max_val = conv.max()
    # use a tolerance to catch all numerically maximal entries
    mask = np.isclose(conv, max_val, atol=1e-12)
    active = np.nonzero(mask)[0]               # positions attaining the max

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)                 # inclusive
        a_slice = a[i_start:i_end + 1]        # length L
        # reversed order matches j = k - i
        rev_slice = a_slice[::-1]
        j_start = k - i_end
        j_end = k - i_start
        g[j_start:j_end + 1] += 2.0 * rev_slice
    return g, max_val


# ----------------------------------------------------------------------
def _adam_optimize_subgrad(init_a: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser using the exact sub‑gradient of the
    `max(conv)` objective.
    Returns the best vector encountered and its raw score
    `2·n·max(conv)`.
    """
    a = init_a.copy()
    best = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # base learning‑rate (tuned empirically)
    base_lr = 0.5

    it = 0
    while time.time() < deadline:
        it += 1

        # current objective
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] it={it} best={best_score:.6f}")

        # sub‑gradient of max(conv)
        g, _ = _subgradient(a)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection back onto the simplex Σa_i = 1
        a = _project_onto_simplex(a, 1.0)

    return best, best_score

def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a random fraction of the mass
    from one index to another and keep the move if it improves the score.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)

        if best[i] < 1e-14:
            continue

        # move up to 30 % of the mass at i towards j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # tiny jitter every few thousand steps to hop out of plateaus
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score
