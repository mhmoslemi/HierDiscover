```
def upsample(seq, new_n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n, endpoint=False) + 0.5/old_n
    x_new = np.linspace(0, 1, new_n, endpoint=False) + 0.5/new_n
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = seq_new / seq_new.sum()
    return seq_new
```

"""
Greedy pairwise‑exchange refinement of a symmetric probability vector.

The search starts from a few simple parametric families (Beta, power‑law,
arcsine, mixture) on a moderate size (≈2000).  The best candidate is then
improved by thousands of cheap “move ε of mass from the index that
contributes most to the current maximal self‑convolution to the index that
contributes least”.  The update keeps the vector on the simplex and
maintains (optional) symmetry.  When time permits the vector is up‑sampled
to a larger length and refined once more.  The routine returns the best
sequence found within the 1000 s budget; the objective used during the
search is exactly the one implemented by ``evaluate_sequence`` (2·n·max/
sum²).  Extensive printing reports progress and intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) distribution on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """Symmetric 1/(i·(n‑i))^p or (i·(n‑i))^p distribution."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on the index set."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """Convex combination (1‑lam)*arcsine + lam*uniform."""
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) / (sum a)^2.
    The vector is projected onto the simplex before evaluation.
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = np.max(_fft_self_convolve(a))
    return 2.0 * n * max_conv  # denominator is 1 because sum a == 1


# ----------------------------------------------------------------------
def _greedy_refine(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Pairwise‑exchange greedy improvement.
    Moves a small fraction of mass from the index that contributes most
    to the current maximal convolution to the index that contributes least.
    The vector stays on the simplex; the best ever seen is returned.
    """
    a = a0.copy()
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    it = 0
    eps_factor = 0.15          # start moving up to 15 % of source mass
    report_every = 2000

    while time.time() < deadline:
        it += 1
        # convolution and its maximal index
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))

        # deterministic segment that actually participates in the max
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)

        segment = a[j_start:j_end + 1]
        if segment.size == 0:
            # should never happen, but be safe
            continue

        j_max = int(np.argmax(segment))
        j_min = int(np.argmin(segment))
        i_src = k - j_max
        i_tgt = k - j_min

        if i_src == i_tgt:
            # pick a random different target
            i_tgt = (i_src + np.random.randint(1, n)) % n

        # amount to transfer
        eps = a[i_src] * eps_factor
        if eps <= 0.0:
            # nothing to move – pick another pair at random
            i_src = np.random.randint(0, n)
            i_tgt = (i_src + np.random.randint(1, n)) % n
            eps = a[i_src] * eps_factor
            if eps <= 0.0:
                continue

        # perform the move
        a[i_src] -= eps
        a[i_tgt] += eps

        # protect against tiny negative round‑off
        if a[i_src] < 0.0:
            a[i_tgt] += a[i_src]
            a[i_src] = 0.0

        # renormalise (the sum is preserved up to round‑off)
        a /= a.sum()

        # occasional jitter to get out of plateaus
        if it % 2500 == 0 and np.random.rand() < 0.07:
            jitter = np.random.randn(n) * 1e-4
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        # slowly shrink the step size
        eps_factor = max(0.005, eps_factor * 0.9995)

        # record best
        if it % report_every == 0:
            score = _objective(a)
            if score < best_score:
                best_score = score
                best = a.copy()
                print(f"[{time.time():.2f}] greedy it={it} score={best_score:.6f}")

    # final check
    final_score = _objective(a)
    if final_score < best_score:
        best_score = final_score
        best = a.copy()
    return best, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = new_seq / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine runs for at most
    `budget_s` seconds and returns the best sequence discovered.
    The method combines a fast parametric sweep with a greedy pairwise
    exchange refinement and a final up‑sampling stage.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin for prints

    # ------------------------------------------------------------------
    # A modest length on which we can afford many cheap iterations.
    n_small = 2000
    # initialise with arcsine – known good baseline
    a_best = _arcsine_distribution(n_small)
    best_score = _objective(a_best)
    print(f"[{time.time():.2f}] start arcsine n={n_small} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameter sweeps (very cheap – each evaluation is a single FFT)
    # ------------------------------------------------------------------
    # Beta family (a around 0.5, the arcsine case)
    for a in np.linspace(0.2, 0.8, 31):
        if time.time() >= deadline - 5.0:
            break
        seq = _beta_distribution(n_small, a)
        score = _objective(seq)
        if score < best_score:
            best_score = score
            a_best = seq.copy()
            print(f"[{time.time():.2f}] beta a={a:.3f} improved score={score:.6f}")

    # power‑law (inverse) family
    for p in np.linspace(0.2, 1.5, 27):
        if time.time() >= deadline - 5.0:
            break
        seq = _power_law_distribution(n_small, p, invert=True)
        score = _objective(seq)
        if score < best_score:
            best_score = score
            a_best = seq.copy()
            print(f"[{time.time():.2f}] power‑invert p={p:.3f} improved score={score:.6f}")

    # mixture of arcsine and uniform
    for lam in np.linspace(0.0, 1.0, 21):
        if time.time() >= deadline - 5.0:
            break
        seq = _mixture_arcsine_uniform(n_small, lam)
        score = _objective(seq)
        if score < best_score:
            best_score = score
            a_best = seq.copy()
            print(f"[{time.time():.2f}] mix lam={lam:.3f} improved score={score:.6f}")

    # ------------------------------------------------------------------
    # Greedy refinement on the best short candidate.
    # Use as much remaining time as possible, but leave a small buffer
    # for the optional up‑sampling stage.
    refine_deadline = deadline - 20.0
    if time.time() < refine_deadline:
        a_best, best_score = _greedy_refine(a_best, refine_deadline)
        print(f"[{time.time():.2f}] after greedy best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a larger length (≈8000) and a short
    # final refinement.  The up‑sampled vector inherits the shape found
    # on the small problem and typically yields a slightly lower final
    # objective because the discretisation error shrinks.
    if time.time() < deadline - 5.0:
        target_n = 8000
        a_large = _upsample(a_best, target_n)
        print(f"[{time.time():.2f}] up‑sample to n={target_n}")
        # a brief refinement (a few thousand cheap steps)
        a_large, large_score = _greedy_refine(a_large,
                                             deadline - 5.0)
        if large_score < best_score:
            a_best = a_large.copy()
            best_score = large_score
            print(f"[{time.time():.2f}] after up‑sample refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback safety net
    if a_best is None or len(a_best) == 0:
        a_best = np.full(1000, 1.0 / 1000, dtype=np.float64)
        best_score = _objective(a_best)

    print(f"[{time.time():.2f}] final best score={best_score:.6f} length={len(a_best)}")
    return a_best.tolist()

    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) / (sum a)².
    The vector is projected onto the simplex before evaluation.
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = np.max(_fft_self_convolve(a))
    return 2.0 * n * max_conv    # sum(a)=1 → denominator = 1


# ----------------------------------------------------------------------
def _greedy_refine(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Pairwise‑exchange greedy improvement.
    Moves a small fraction of mass from the index that contributes most
    to the current maximal convolution to the index that contributes least.
    The vector stays on the simplex; the best ever seen is returned.
    """
    a = a0.copy()
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    it = 0
    eps_factor = 0.15           # start with up‑to 15 % of source mass
    report_every = 2000

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))

        # indices that actually contribute to the maximal entry
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)

        segment = a[j_start:j_end + 1]
        if segment.size == 0:
            continue

        j_max = int(np.argmax(segment))
        j_min = int(np.argmin(segment))
        i_src = k - j_max
        i_tgt = k - j_min

        if i_src == i_tgt:
            i_tgt = (i_src + np.random.randint(1, n)) % n

        eps = a[i_src] * eps_factor
        if eps <= 0.0:
            # fallback random move
            i_src = np.random.randint(0, n)
            i_tgt = (i_src + np.random.randint(1, n)) % n
            eps = a[i_src] * eps_factor
            if eps <= 0.0:
                continue

        # transfer epsilon of mass
        a[i_src] -= eps
        a[i_tgt] += eps

        if a[i_src] < 0.0:
            a[i_tgt] += a[i_src]
            a[i_src] = 0.0

        a /= a.sum()

        # occasional random jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.07:
            jitter = np.random.randn(n) * 1e-4
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        eps_factor = max(0.005, eps_factor * 0.9995)   # slowly shrink step

        if it % report_every == 0:
            score = _objective(a)
            if score < best_score:
                best_score = score
                best = a.copy()
                print(f"[{time.time():.2f}] greedy it={it} score={best_score:.6f}")

    final_score = _objective(a)
    if final_score < best_score:
        best_score = final_score
        best = a.copy()
    return best, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = new_seq / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine runs for at most
    `budget_s` seconds and returns the best sequence discovered.
    The method combines a fast parametric sweep with a greedy pairwise
    exchange refinement and a final up‑sampling stage.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # keep a tiny safety margin

    # ------------------------------------------------------------------
    # Work on a modest size first – inexpensive enough for thousands of
    # cheap greedy iterations.
    n_small = 2000
    a_best = _arcsine_distribution(n_small)
    best_score = _objective(a_best)
    print(f"[{time.time():.2f}] start arcsine n={n_small} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fast parametric sweep (each trial uses a single FFT).
    # ---------------------------------------------------------------
    # Beta family (around the arcsine case a=0.5)
    for a in np.linspace(0.2, 0.8, 31):
        if time.time() >= deadline - 5.0:
            break
        seq = _beta_distribution(n_small, a)
        score = _objective(seq)
        if score < best_score:
            best_score = score
            a_best = seq.copy()
            print(f"[{time.time():.2f}] beta a={a:.3f} improved score={score:.6f}")

    # Inverted power‑law family
    for p in np.linspace(0.2, 1.5, 27):
        if time.time() >= deadline - 5.0:
            break
        seq = _power_law_distribution(n_small, p, invert=True)
        score = _objective(seq)
        if score < best_score:
            best_score = score
            a_best = seq.copy()
            print(f"[{time.time():.2f}] power‑invert p={p:.3f} improved score={score:.6f}")

    # Mixture arcsine + uniform
    for lam in np.linspace(0.0, 1.0, 21):
        if time.time() >= deadline - 5.0:
            break
        seq = _mixture_arcsine_uniform(n_small, lam)
        score = _objective(seq)
        if score < best_score:
            best_score = score
            a_best = seq.copy()
            print(f"[{time.time():.2f}] mix lam={lam:.3f} improved score={score:.6f}")

    # ------------------------------------------------------------------
    # Greedy pairwise‑exchange refinement on the best short candidate.
    # Keep a short buffer for an optional up‑sampling stage.
    refine_deadline = deadline - 20.0
    if time.time() < refine_deadline:
        a_best, best_score = _greedy_refine(a_best, refine_deadline)
        print(f"[{time.time():.2f}] after greedy best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a larger length (≈8000) and a brief final
    # refinement.  Larger n usually reduces discretisation error.
    if time.time() < deadline - 5.0:
        target_n = 8000
        a_large = _upsample(a_best, target_n)
        print(f"[{time.time():.2f}] up‑sample to n={target_n}")
        a_large, large_score = _greedy_refine(a_large, deadline - 5.0)
        if large_score < best_score:
            a_best = a_large.copy()
            best_score = large_score
            print(f"[{time.time():.2f}] after up‑sample refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – should never be needed.
    if a_best is None or a_best.size == 0:
        a_best = np.full(1000, 1.0 / 1000, dtype=np.float64)
        best_score = _objective(a_best)

    print(f"[{time.time():.2f}] final best score={best_score:.6f} length={len(a_best)}")
    return a_best.tolist()
