def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0
    best_seq = None
    best_score = float('inf')

    # try previous solution if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.maximum(prev, 0.0)
        prev = _project_onto_simplex(prev, 1.0)
        score = _objective(prev)  # using fast
        if score < best_score:
            best_score = score
            best_seq = prev.copy()
        print(f"Prev solution score {score:.6f}")
    except Exception:
        pass

    # candidate lengths
    candidate_lengths = [20000, 18000, 16000, 14000, 12000, 10000, 8000, 6000, 5000]
    # ensure within allowed by evaluate (maybe max 20000)
    candidate_lengths = [n for n in candidate_lengths if n >= 500 and n <= 20000]

    # mixture lambdas
    lambdas = np.linspace(0.0, 0.5, 51)  # 0 = pure arcsine

    for n in candidate_lengths:
        if time.time() > deadline:
            break
        arcs = _arcsine_distribution(n)
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        for lam in lambdas:
            seq = (1 - lam) * arcs + lam * uniform
            # seq already sums to 1; still ensure nonneg
            # compute objective
            score = _objective(seq)  # returns 2*n*max
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"New best: n={n} lam={lam:.3f} score={score:.6f}")

        # also test power law invert around p=0.5
        # sample p values
        for p in [0.4,0.5,0.6]:
            if time.time() > deadline:
                break
            seq = _power_law_distribution(n, p, invert=True)
            score = _objective(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"New best power p={p} n={n} score={score:.6f}")

    # Phase local refinement if time left
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        # simple random moves
        a = best_seq.copy()
        while time.time() < deadline:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps = a[i] * np.random.rand() * 0.05  # up to 5%
            if eps <= 0:
                continue
            cand = a.copy()
            cand[i] -= eps
            cand[j] += eps
            # reproject onto simplex if needed - should sum to 1 already (numerical)
            # compute objective
            score = _objective(cand)
            if score < best_score - 1e-12:
                best_score = score
                best_seq = cand.copy()
                a = cand
                print(f"Local improve score={best_score:.6f}")

    print(f"Final best score ~ {best_score:.6f} length {best_seq.shape[0]}")
    return best_seq.tolist()

"""
search_proposal.py

A time‑bounded stochastic optimiser for the “step‑function” objective.
The optimiser works in three phases:

1.  Wide random search over a handful of parametric families
    (power‑law, beta, Gaussian, arcsine‑mixture) for many different
    lengths up to the allowed maximum (20 000).  Every candidate is
    projected onto the probability simplex (∑a=1) and evaluated with a
    fast FFT based exact objective
          2·n·max(a∗a)    (the denominator (∑a)² is 1 after projection).

2.  The best few candidates are refined by an exponentiated‑gradient
    mirror‑descent on the simplex.  The surrogate for the max‑norm is a
    log‑sum‑exp with a temperature α that starts small and is increased
    until the pure sub‑gradient of the max is used.

3.  A cheap random‑move hill‑climber moves a tiny fraction of mass
    between two coordinates whenever it improves the exact objective.

All helper functions are defined at module level; no closures or file I/O
are used.  The routine respects the supplied `budget_s` (≈1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def _beta_distribution(n: int, a: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (full convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) with the vector projected onto the simplex."""
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a via FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) attained at index k (2·a_{k-i})."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """∇ of (1/α)·log‑sum‑exp(α·b) w.r.t. a (smooth max surrogate)."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    return 2.0 * conv[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimisation.
    Returns the best sequence seen and its objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)

    eta0 = 0.8               # initial step size
    alpha = 0.5              # start with a smooth surrogate
    alpha_mul = 1.04         # slow annealing
    alpha_max = 2000.0
    alpha_thr = 1000.0       # switch to true sub‑gradient near the end
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient (smooth or exact)
        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, b, alpha, size_fft)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)

        # renormalise onto simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional jitter to escape plateaus
        if it % 2000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score

def _local_refinement(a_start: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Simple random‑move hill‑climbing on the best sequence."""
    a = a_start.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.07   # up to 7 % of source mass
        if eps == 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        cur = _objective(cand)
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score
