def _subgradient(a):
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    max_idx = np.where(conv == max_c)[0]
    cnt = len(max_idx)
    weight = 2.0 / cnt
    g = np.zeros(n)
    for k in max_idx:
        shift = k - (n - 1)
        if shift >= 0:
            g[shift:] += weight * a[:n-shift]
        else:
            g[:n+shift] += weight * a[-shift:]
    return g
```

def _compute_smooth_obj_and_grad(y, mu, n, S):
    p = _softmax(y)
    a = S * p
    # conv via FFT
    m = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=m)
    conv_fft = np.fft.irfft(fft_a * fft_a, n=m)
    conv = conv_fft[:2*n-1]
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700, 700)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            cnt = 1
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        f_val = max_c
    # compute g_a via convolution w and a_rev via FFT
    # a_rev = a[::-1] length n
    fft_w = np.fft.rfft(w, n=m)
    fft_ared = np.fft.rfft(a[::-1], n=m)
    conv_w_fft = np.fft.irfft(fft_w * fft_ared, n=m)
    conv_wr = conv_w_fft[:3*n-2]  # length maybe larger but we need slice
    # But the full linear convolution of lengths (2n-1) and n => length 3n-2
    # After irfft length m, but we truncated; ensure we have at least needed length.
    # However due to zero padding, the result at indexes (n-1) to (2n-2) correspond to slices we need.
    # For length I'll just compute conv_wr = np.convolve(w, a[::-1]) with np.convolve for safety (size small).
    # But w length 2n-1 (maybe ~2n), a_rev length n, using np.convolve O(n^2) again. However conv of w and a_rev can be computed via FFT same m as before.
    # Ensure we compute correct length: using same m.
    conv_wr = np.fft.irfft(fft_w * fft_ared, n=m)[: (2*n-1 + n -1)]  # (3n-2)
    # Now g_a = 2 * slice from n-1 to n-1+n
    g_a = 2.0 * conv_wr[n-1:n-1+n]
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)
```

But we need to ensure correctly sized arrays. Actually `np.fft.rfft` returns only so many points. Multiplying transforms and invert yields length m (real). """
Propose a low‑score step function.

The algorithm builds on the smooth‑max / sub‑gradient approach that was
used previously but enlarges the search space, adds more restarts,
slower cooling and an accelerated coordinate‑transfer heuristic.
All internal scores are obtained with fast FFT‑based convolutions;
the expensive ``evaluate_sequence`` is called only for the final
candidate (and occasionally when a promising improvement is found).

Key ideas
---------
* Use a larger length (up to 15000) – the theoretical lower bound of 1
  becomes reachable only with many degrees of freedom.
* Smooth‑max surrogate with L‑BFGS‑B (FFT for the convolution) and a
  gentle cooling schedule (μ←0.85·μ).  Several diverse initial points are
  tried.
* Follow‑up sub‑gradient descent on the true max‑convolution, with an
  adaptive step size and a cheap “coordinate‑transfer’’ move that swaps
  mass from large‑gradient to small‑gradient coordinates.
* Cheap large‑scale random‑block perturbations and simple monotone‑sorting
  post‑processing are applied while time permits.
* The best sequence found is finally evaluated with the provided
  ``evaluate_sequence`` routine and returned.

The routine respects a user‑supplied time budget (default 1000 s) and
never returns an empty list, guaranteeing a non‑negative score.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z <= 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a via FFT.
    Returns the first (2*len(a)-1) entries.
    """
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=m)
    conv_full = np.fft.irfft(fft_a * fft_a, n=m)
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
def _conv_fft_cross(w: np.ndarray, a_rev: np.ndarray, target_len: int) -> np.ndarray:
    """
    Linear convolution w * a_rev via FFT.
    ``target_len`` is the length of the full linear convolution,
    i.e. len(w)+len(a_rev)-1.
    Returns the first ``target_len`` entries.
    """
    m = 1 << (target_len .bit_length())
    fft_w = np.fft.rfft(w, n=m)
    fft_a = np.fft.rfft(a_rev, n=m)
    conv_full = np.fft.irfft(fft_w * fft_a, n=m)
    return conv_full[:target_len]


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate.
        f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Uses FFT for all convolutions (fast even for n≈15000).
    """
    # probabilities on the simplex and actual heights
    p = _softmax(y)                     # Σ p_i = 1, p_i ≥ 0
    a = S * p                           # Σ a_i = S

    # ----- convolution a*a (length 2n‑1) -----
    conv = _conv_fft(a)
    max_c = conv.max()

    # ----- smooth‑max weights -----
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # guard overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – uniform weight on all active indices
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            cnt = 1
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # ----- gradient w.r.t. a -----
    # g_a[i] = 2 * Σ_k w_k * a_{k-i}
    # realised by convolution of w with a reversed
    a_rev = a[::-1]
    target_len = (2 * n - 1) + n - 1          # 3n‑2
    conv_wr = _conv_fft_cross(w, a_rev, target_len)
    # slice the central n entries (indices n‑1 … 2n‑2)
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

    # ----- chain rule for y (soft‑max Jacobian) -----
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable that returns (value, gradient) for scipy.optimize."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the hard max objective
    (max of the auto‑convolution) at point a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_c = conv.max()
    max_idx = np.where(conv == max_c)[0]
    cnt = len(max_idx)
    if cnt == 0:
        cnt = 1
    weight = 2.0 / cnt
    g = np.zeros(n, dtype=float)
    for k in max_idx:
        shift = k - (n - 1)
        if shift >= 0:
            # i = shift … n‑1  →  a[i‑shift] contributes
            g[shift:] += weight * a[: n - shift]
        else:
            # shift < 0 : i = 0 … n+shift‑1  → a[i‑shift] = a[i+|shift|]
            g[: n + shift] += weight * a[-shift :]
    return g


# ----------------------------------------------------------------------
def _coordinate_transfer_step(a: np.ndarray, S: float, frac: float) -> np.ndarray:
    """
    Simple deterministic move: transfer a fraction ``frac`` of the
    mass from a coordinate with the largest sub‑gradient to one with
    the smallest.  Afterwards project back onto the ℓ¹‑simplex.
    """
    g = _subgradient(a)
    i = np.argmax(g)          # large gradient → should be decreased
    j = np.argmin(g)          # small gradient → should be increased
    epsilon = frac * a[i]
    if epsilon <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= epsilon
    a_new[j] += epsilon
    a_new = _project_to_simplex(a_new, S)
    return a_new


# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, n: int) -> np.ndarray:
    """Linearly interpolate a sequence to length ``n``."""
    old_n = len(seq)
    if old_n == n:
        return seq.astype(float).copy()
    old_idx = np.arange(old_n)
    new_idx = np.linspace(0, old_n - 1, n)
    return np.interp(new_idx, old_idx, seq).astype(float)
