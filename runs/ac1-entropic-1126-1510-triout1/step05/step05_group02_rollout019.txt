    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()    # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum value of a*a (self‑convolution)."""
    return _fft_self_convolve_full(a).max()

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _softmax_subgradient_step(a: np.ndarray, beta: float):
    """
    Smooth (soft‑max) sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)            # length 2n‑1
    cur_max = conv.max()
    # soft‑max weights (the max shift gets weight 1, others get <1)
    delta = conv - cur_max                 # ≤ 0
    w = np.exp(beta * delta)               # may under‑flow → 0
    w_sum = w.sum()
    if w_sum == 0.0:                       # numerical safety
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

# ----------------------------------------------------------------------
def _random_sparse_candidate(n: int, k: int) -> np.ndarray:
    """k equally‑weighted spikes placed on distinct random positions."""
    pos = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[pos] = 1.0 / k
    return a

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low evaluate_sequence score.

    1. Try many random sparse vectors of different support sizes.
    2. Keep the best one and refine it with a projected soft‑max
       sub‑gradient descent (uses the exact ℓ∞ self‑convolution).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 1) initial candidates – sparse vectors
    n = 5000                         # length used in the reference run
    best_seq = None
    best_raw = float("inf")
    # we try several support sizes around √(2n) ≈ 100
    support_sizes = [90, 100, 110, 120, 130, 140]
    trials_per_size = 250

    for k in support_sizes:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Sparse search: k={k}")
        for _ in range(trials_per_size):
            a = _random_sparse_candidate(n, k)
            cur_raw = 2.0 * n * _fft_self_convolve_max(a)
            if cur_raw < best_raw:
                best_raw, best_seq = cur_raw, a.copy()
                print(f"   new best raw={best_raw:.6f} (k={k})")
            if time.time() >= deadline:
                break

    # --------------------------------------------------------------
    # 2) fallback: use previous construction if any
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if base.shape[0] > 0:
            base = _project_onto_simplex(base)
            raw_base = 2.0 * base.shape[0] * _fft_self_convolve_max(base)
            if raw_base < best_raw:
                best_raw, best_seq = raw_base, base.copy()
                print(f"   base candidate improved raw={best_raw:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3) projected soft‑max sub‑gradient descent on the best candidate
    if best_seq is None:
        # should never happen, but keep a safe guard
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    a = best_seq.copy()
    best_val = best_raw
    best_a = a.copy()
    it = 0
    beta = 0.5                # start temperature for soft‑max
    base_step = 0.6

    while time.time() < deadline:
        it += 1
        grad, cur_max = _softmax_subgradient_step(a, beta)
        step = base_step / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)                     # keep sum = 1, a_i ≥ 0

        # occasional small random jitter to escape plateaus
        if it % 2000 == 0 and np.random.rand() < 0.10:
            a += np.random.randn(n) * 3e-7
            a = _project_onto_simplex(a)

        # update temperature (make soft‑max sharper over time)
        beta = min(beta * 1.001, 200.0)

        cur_raw = 2.0 * n * cur_max
        if cur_raw + 1e-12 < best_val:
            best_val = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] iter={it:,}  raw={best_val:.6f}")

        # stop early if we are already very close to the theoretical limit
        if best_val <= 1.01:
            break

    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Final evaluated score: {final_score:.6f} (len={len(final_seq)})")
    return final_seq
