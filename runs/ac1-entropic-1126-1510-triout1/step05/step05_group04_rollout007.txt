Hybrid search for a low‑score step function.

The routine builds a diverse pool of initial candidates (uniform,
exponential, Gaussian, classic window functions, Dirichlet‑sparse,

monotone‑sorting heuristics.  All updates keep the vector non‑negative

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate:
        f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Parameters
    ----------
    y : ndarray, shape (n,)
        Unconstrained parameters (log‑odds of the simplex).
    mu : float
        Smoothing temperature (>0).  μ = 0 corresponds to the hard max.
    n : int
        Length of the step function.
    S : float
        Target ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : ndarray, shape (n,)
        Gradient ∂f/∂y.
    """
    p = _softmax(y)                 # Σ p_i = 1, p_i ≥ 0
    a = S * p                       # Σ a_i = S

    conv = np.convolve(a, a)        # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. heights a: g_a[i] = 2 * Σ_k w_k * a_{k-i}
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    inner = np.dot(g_a, p)  # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable returning (value, gradient) for L‑BFGS‑B."""
    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _max_conv_and_active_indices(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute max of a*a using FFT (O(n log n)) and return the max value
    together with all indices where it is attained.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    Fa = scipy.fft.rfft(a, n=size)
    conv = scipy.fft.irfft(Fa * Fa, n=size)[: 2 * n - 1]
    max_val = conv.max()
    idx = np.where(conv == max_val)[0]
    return float(max_val), idx


# ----------------------------------------------------------------------
def _subgradient_from_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Efficient linear‑time sub‑gradient of the hard‑max objective.
    For each active index k we add the appropriately shifted slice of a.
    The result is 2 * Σ_k a_{k‑i} over all feasible i.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        g[i] += a[k - i]
    g *= 2.0
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) with a low
    evaluate_sequence() score.  The routine respects the given time
    budget (seconds) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # --------------------- problem size -------------------------------
    try:
        ref_len = len(height_sequence_1)    # type: ignore[name-defined]
        n = min(max(ref_len * 2, 4000), 9000)
        have_ref = True
    except Exception:
        n = 6000
        have_ref = False

    S = math.sqrt(2.0 * n)                  # target ℓ¹‑norm

    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # --------------------- helper to update best ----------------------
    def _maybe_update(a_vec: np.ndarray) -> None:
        nonlocal best_score, best_a
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # --------------------- generate diverse seeds ---------------------
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0))

    # linear decreasing & increasing
    seeds.append(np.arange(n, 0, -1, dtype=float))
    seeds.append(np.arange(1, n + 1, dtype=float))

    # exponential decays (different rates)
    for rate in (0.0008, 0.0015, 0.003):
        seeds.append(np.exp(-rate * np.arange(n, dtype=float)))

    # reversed exponentials
    for rate in (0.0008, 0.0015, 0.003):
        seeds.append(np.exp(-rate * np.arange(n - 1, -1, -1, dtype=float)))

    # Gaussian bumps (different sigmas)
    for sigma_fac in (0.12, 0.2, 0.35):
        sigma = sigma_fac * n
        x = np.arange(n) - (n - 1) / 2.0
        seeds.append(np.exp(-0.5 * (x / sigma) ** 2))

    # Classic window functions
    x = np.arange(n)
    seeds.append(0.5 - 0.5 * np.cos(2 * np.pi * x / (n - 1)))               # Hann
    seeds.append(0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1)))            # Hamming
    seeds.append(
        0.42
        - 0.5 * np.cos(2 * np.pi * x / (n - 1))
        + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    )  # Blackman

    # Triangle (peak in the middle)
    half = n // 2
    tri = np.concatenate(
        [np.linspace(1, half + 1, half + 1), np.linspace(half, 1, n - half - 1)]
    )
    seeds.append(tri)

    # Sparse Dirichlet (α < 1)
    dir_alpha = rng.uniform(0.2, 0.6)
    p_dir = rng.dirichlet([dir_alpha] * n)
    seeds.append(p_dir)

    # Random positive Gaussian
    seeds.append(np.abs(rng.standard_normal(n)))

    # Reference if present
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        seeds.append(a_ref)

    # Project all seeds onto the simplex and evaluate them immediately
    init_ys = []
    for i, s in enumerate(seeds):
        a_proj = _project_to_simplex(s, S)
        _maybe_update(a_proj)
        # store log‑odds for L‑BFGS initial points
        init_ys.append(np.log(a_proj + 1e-12))

    # --------------------- smooth‑max L‑BFGS phase --------------------
    mu_factor = 0.70          # slower cooling → smoother path
    mu_min = 1e-12
    maxiter_per_mu = 200      # a moderate budget per temperature
    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{len(init_ys)}] initialising")
        y = y0.copy()
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        mu = max(mu, 1.0)

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    # --------------------- fast sub‑gradient descent -----------------
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        it = 0
        max_local_iters = 8000
        step0 = S * 0.4 / n    # heuristic initial step
        while time.time() < deadline and it < max_local_iters:
            max_c, act_idx = _max_conv_and_active_indices(a)
            # compute sub‑gradient efficiently
            g = _subgradient_from_active(a, act_idx)

            # back‑tracking line search
            step = step0 / (1.0 + 1e-4 * it)
            improved = False
            for _ in range(12):
                a_new = a - step * g
                a_new = _project_to_simplex(a_new, S)
                sc = evaluate_sequence(a_new.tolist())
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[subgrad] iter {it:5d} step {step:.3e} => {best_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # occasional jitter to escape flat regions
                i, j = rng.integers(0, n, size=2, endpoint=False)
                delta = rng.random() * a[i] * 0.04
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, S)
            it += 1

    # --------------------- random block perturbations ----------------
    remaining = deadline - time.time()
    if remaining > 2.0 and best_score < float("inf"):
        max_trials = int(min(8000, remaining * 130))
        print(f"[block] {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(3, 17)          # 3 – 16 blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # --------------------- simulated‑annealing style jitter ----------
    if time.time() < deadline and best_score < float("inf"):
        # small temperature schedule
        temperature = S * 0.02
        for _ in range(2000):
            i, j = rng.integers(0, n, size=2, endpoint=False)
            # propose swapping a fraction of mass
            delta = temperature * rng.random()
            delta = min(delta, best_a[i])
            a_trial = best_a.copy()
            a_trial[i] -= delta
            a_trial[j] += delta
            a_trial = _project_to_simplex(a_trial, S)
            sc = evaluate_sequence(a_trial.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a_trial
                temperature *= 0.9995   # slowly cool
                print(f"[anneal] improved to {best_score:.6f}")
            else:
                temperature *= 1.0005   # occasional reheating

    # --------------------- monotone sorting heuristics ---------------
    if time.time() < deadline and best_score < float("inf"):
        desc = np.sort(best_a)[::-1]
        desc = _project_to_simplex(desc, S)
        _maybe_update(desc)

        asc = np.sort(best_a)
        asc = _project_to_simplex(asc, S)
        _maybe_update(asc)

    # --------------------- final report ------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
