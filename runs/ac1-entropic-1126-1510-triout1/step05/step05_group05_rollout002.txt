"""
Propose a step‑function sequence with a very low value of
evaluate_sequence.

The algorithm is deliberately simple and deterministic:

1.  Sweep a range of lengths (up to 20 000) and generate a number of
    symmetric probability vectors:
       • pure arcsine (the known asymptotically optimal shape);
       • powers of the arcsine vector (a_i ← a_i^p);
       • convex mixtures of arcsine and the uniform distribution.
    All vectors are normalised to sum to 1 so that the objective
    coincides with 2·n·max(a∗a).

2.  Keep the best few (default 5) candidates according to this cheap
    FFT‑based objective.

3.  Refine each of those candidates with a short mirror‑descent /
    exponentiated‑gradient routine.  The update works directly on the
    simplex and uses either the exact sub‑gradient of the max‑convolution
    entry or a soft‑max surrogate once the temperature grows.

4.  Apply a lightweight random‑move hill‑climbing to the currently
    best sequence.

All heavy work is performed with NumPy/FFT, making the whole search
well within the 1000 s budget.  The final result is returned as a plain
Python list of non‑negative floats, ready for the external
`evaluate_sequence` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) / (sum a)^2.
    The vector is projected onto the simplex before evaluation.
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    # after projection sum(a) == 1, therefore denominator is 1
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the (1/α)·log‑sum‑exp(α·b_k) surrogate w.r.t. a.
    Implements the formula g = 2 * conv_rev(w, a_rev).
    """
    # soft‑max weights over the convolution entries
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    eta0 = 0.8
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 15000.0
    alpha_thr = 2000.0          # switch to sub‑gradient after this
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] it={it} improved score={best_score:.6f}")

        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, b, alpha, size_fft)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape flat plateaus
        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap random‑move hill‑climbing.
    Moves a small fraction of mass from one index to another if it improves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)
    n = a.shape[0]

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.1
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cur_score = _objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score
