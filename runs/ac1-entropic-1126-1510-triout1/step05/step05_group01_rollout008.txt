"""
Propose a low‑score step‑function.

Algorithm
---------

1. **Randomised analytic families** –  a large amount of candidate
   probability vectors are generated as mixtures of (continuous) Beta
   densities, discretised on a lattice of size `n`.  The mixture
   parameters (α,β for each component and its weight) are sampled
   logarithmically so that both very flat and very U‑shaped profiles
   are explored.  The best few candidates (according to the cheap
   score   2·n·max(a⋆a)) are kept.

2. **Gradient‑based refinement** – each retained candidate is fed to an
   exponentiated‑gradient (mirror‑descent on the simplex) optimiser.
   The surrogate loss is a soft‑max of the convolution which is cooled
   geometrically until it approximates the true maximum; after that an
   exact sub‑gradient of the max is used.  Momentum and occasional
   jitter help to escape plateaus.  The iterate is always projected on
   the probability simplex.

3. **Evaluation** – the best refined vector is finally evaluated with
   the provided ``evaluate_sequence`` black‑box; the globally best
   sequence found within the time budget is returned.

The routine respects the supplied time budget (default 1000 s) and
prints progress information.  All numbers are non‑negative and the
sequence is normalised to sum = 1, which is the scale used by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    vals_sum = vals.sum()
    if vals_sum == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / vals_sum
    return vals


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a⋆a)  (valid for sum(a)=1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    Returns a vector of length n.
    """
    n = a.shape[0]
    cur_max = conv.max()
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max, uniform on the (approximately) active set.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correct diagonal double‑counting when k = 2*i
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _exponential_update(a: np.ndarray,
                        grad: np.ndarray,
                        eta: float,
                        momentum: np.ndarray,
                        beta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Multiplicative update with simple momentum (mirror descent on the simplex).
    Returns the updated vector and momentum buffer.
    """
    momentum = beta * momentum + (1.0 - beta) * grad
    shift = np.max(momentum)
    factor = np.exp(-eta * (momentum - shift))
    a = a * factor
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    return a, momentum


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the time budget and
    returns the best sequence found (as a Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny margin for final housekeeping
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # base length – use a previously discovered construction if present
    try:
        base_n = len(height_sequence_1)     # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # Target length for the heavy part of the search.
    # We take a modestly larger lattice (up‑to 12 k) to give a finer
    # discretisation of the continuous shape.
    n_target = min(12000, max(4000, int(base_n * 1.5)))

    print(f"[{time.time():.1f}] target length n={n_target}")

    # ------------------------------------------------------------------
    # 1. Random exploration of mixtures of Beta densities.
    max_keep = 30                      # keep the best this many candidates
    best_candidates = []               # list of (score, vector) tuples

    rand_deadline = start_time + min(0.25 * budget_s, 200.0)   # at most 200 s
    rng = np.random.default_rng(seed)

    # helper to insert a candidate while keeping the list sorted
    def _maybe_add(score_val: float, vec: np.ndarray) -> None:
        if len(best_candidates) < max_keep:
            best_candidates.append((score_val, vec.copy()))
            best_candidates.sort(key=lambda x: x[0])
        elif score_val < best_candidates[-1][0]:
            best_candidates[-1] = (score_val, vec.copy())
            best_candidates.sort(key=lambda x: x[0])

    # include a few deterministic seeds
    deterministic_params = [
        (0.5, 0.5),   # arcsine
        (0.3, 0.3),
        (0.2, 0.2),
        (0.8, 0.8),
        (1.0, 1.0),   # uniform
        (0.2, 0.8),   # skewed
        (0.8, 0.2),
    ]
    for al, be in deterministic_params:
        a = _beta_distribution(n_target, al, be)
        _maybe_add(_score(a), a)

    # also try to reuse the previous best construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() == 0.0:
            prev = np.full_like(prev, 1.0 / prev.size)
        else:
            prev = prev / prev.sum()
        if prev.size != n_target:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n_target)
            prev = np.interp(x_new, x_old, prev)
            prev = _project_onto_simplex(prev, s=1.0)
        _maybe_add(_score(prev), prev)
    except Exception:
        pass

    # random mixture search
    while time.time() < rand_deadline:
        # number of mixture components (1‑3)
        k = rng.choice([1, 2, 3], p=[0.3, 0.6, 0.1])
        # log‑uniform sampling for α,β in roughly [0.01, 3]
        log_min, log_max = -2.0, 0.5
        alphas = 10 ** rng.uniform(log_min, log_max, size=k)
        betas  = 10 ** rng.uniform(log_min, log_max, size=k)
        # Dirichlet weights
        weights = rng.dirichlet(np.ones(k))
        a = np.zeros(n_target, dtype=np.float64)
        for i in range(k):
            comp = _beta_distribution(n_target, alphas[i], betas[i])
            a += weights[i] * comp
        a = _project_onto_simplex(a, s=1.0)
        _maybe_add(_score(a), a)

    print(f"[{time.time():.1f}] random phase finished – best cheap score "
          f"{best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. Gradient‑based refinement of the top few candidates.
    #    The remaining time is split roughly equally among the selected starters.
    remaining_time = max(0.0, deadline - time.time())
    if remaining_time <= 0.0:
        remaining_time = 0.0

    # select up to 5 best vectors for deeper optimisation
    starters = [vec.copy() for _, vec in best_candidates[:5]]

    overall_best_seq = None
    overall_best_score = float("inf")

    for idx, a0 in enumerate(starters):
        sub_start = time.time()
        # allocate a fraction of the remaining budget for this starter
        sub_deadline = sub_start + min(remaining_time * 0.18, 180.0)

        a = a0.copy()
        n_a = a.shape[0]
        momentum = np.zeros_like(a)
        beta_mom = 0.9
        eta0 = 0.6
        # Soft‑max temperature schedule
        temp = 5.0
        temp_factor = 1.003
        temp_max = 5000.0

        best_local_seq = a.copy()
        best_local_score = _score(a)
        iteration = 0

        while time.time() < sub_deadline:
            iteration += 1

            conv = _fft_self_convolve(a)
            cur_max = conv.max()
            cur_score = 2.0 * n_a * cur_max

            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = a.copy()
                print(f"[{time.time():.1f}] n={n_a} starter={idx} iter={iteration} "
                      f"score={cur_score:.6f}")

            # gradient (smooth → exact)
            if temp < temp_max:
                grad = _softmax_gradient(a, conv, temp)
                temp = min(temp_max, temp * temp_factor)
            else:
                grad = _exact_gradient(a, conv)

            eta = eta0 / np.sqrt(iteration)
            a, momentum = _exponential_update(a, grad, eta, momentum, beta_mom)

            # occasional jitter to shake out of plateaus
            if iteration % 2000 == 0 and rng.random() < 0.2:
                a += rng.normal(scale=1e-6, size=n_a)
                a = _project_onto_simplex(a, s=1.0)

        # final evaluation of the best local vector
        try:
            final_score = evaluate_sequence(best_local_seq.tolist())
        except Exception:          # fall back to cheap score if the black‑box misbehaves
            final_score = best_local_score

        print(f"[{time.time():.1f}] starter {idx} finished – final score={final_score:.6f}")

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = best_local_seq.copy()
            print(f"*** New global best → {overall_best_score:.6f}")

        # update remaining budget
        remaining_time = max(0.0, deadline - time.time())
        if overall_best_score <= 1.503:
            print("Target attained – exiting early.")
            return overall_best_seq.tolist()

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3. Fallback – should never be needed.
    if overall_best_seq is None:
        n_fallback = 1000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("[fallback] uniform vector used.")

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
