"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1.  Generate a small pool of analytic candidates (arcsine, symmetric
    Beta(p,p) with very small p, triangular, uniform and a few random
    Dirichlet draws) for a handful of lengths (≈ 500–4000).

2.  For each candidate run an *LP‑mixing* refinement:
    * compute M = max(a∗a) (self‑convolution via FFT);
    * build a tiny linear program that maximizes Σ g_j under the
      constraints (a∗g)_k ≤ M for the K largest convolution entries
      (K≈30).  This yields a new distribution g that respects the active
      constraints of a.
    * replace a by a convex combination (1‑t)·a + t·g (t≈0.5, reduced
      if the combination would increase M).  The maximal convolution
      value never rises, and the objective 2·n·max(a∗a) monotonically
      decreases.

3.  Keep the best distribution seen across all lengths.

4.  Perform a final lightweight hill‑climbing phase that moves a tiny
    amount of mass between two random positions while never increasing
    the objective.  This often extracts a few extra digits.

The whole search respects the 1000 s budget and returns a list that can
be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5)), normalized to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid, normalized to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # stable evaluation via logs
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    # for extremely small p the exponent may under/overflow; clip safely
    w = np.exp(logw, dtype=np.float64)
    s = w.sum()
    if s == 0 or not np.isfinite(s):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle shape, normalized to sum 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay centred in the middle, normalized."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _build_A_ub(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """Construct the constraint matrix for (a * g)_k ≤ M for chosen indices."""
    n = a.shape[0]
    rows = []
    for k in idxs:
        row = np.zeros(n, dtype=np.float64)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        if j_start <= j_end:
            j_vals = np.arange(j_start, j_end + 1, dtype=np.int64)
            row[j_vals] = a[k - j_vals]          # a_{k-j}
        rows.append(row)
    if rows:
        return np.vstack(rows)
    else:
        return np.empty((0, n), dtype=np.float64)
