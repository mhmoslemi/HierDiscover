def _adam_optimize_improved(init_a, n, deadline):
    a = init_a.copy()
    best = a.copy()
    best_score = 2 * n * np.max(_fft_self_convolve(a))
    # needed factor mus.

    # Precompute reversed indices.
    size_fft = 1 << ((3*n-2).bit_length())
    # hyperparameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    alpha = 5.0
    alpha_mul = 1.08
    alpha_max = 200000.0

    lambda_l2 = 1.0  # weight for L2 gradient
    lambda_decay = 0.9995   # per iteration

    it = 0
    while time.time() < deadline:
        it += 1

        # Fourier transforms
        A = np.fft.rfft(a, n=size_fft)
        A_rev = np.fft.rfft(a[::-1], n=size_fft)
        conv_fft = A * A
        conv = np.fft.irfft(conv_fft, n=size_fft)[: 2*n-1]

        cur_max = conv.max()
        cur_score = 2 * n * cur_max
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] iter={it} score={best_score:.6f}")

        # gradient softmax (approx max)
        if alpha > 2_000:
            max_c = cur_max
            mask = np.abs(conv - max_c) < 1e-12
            w = mask.astype(float)
            cnt = w.sum()
            if cnt > 0:
                w /= cnt
            else:
                w[np.argmax(conv)] = 1.0
        else:
            shift = conv - cur_max
            w = np.exp(alpha * shift)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # FFT of w
        W = np.fft.rfft(w, n=size_fft)

        # gradient from softmax
        grad_soft = 2.0 * np.fft.irfft(W * A_rev, n=size_fft)[n-1:n-1+n]

        # gradient L2
        grad_l2 = 2.0 * np.fft.irfft(conv_fft * A_rev, n=size_fft)[n-1:n-1+n]

        grad = grad_soft + lambda_l2 * grad_l2

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = 0.3 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto simplex
        a = _project_onto_simplex(a, 1.0)

        # schedule updates
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        lambda_l2 *= lambda_decay
        # maybe break after certain iterations? continue
    return best, best_score

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0
    # Determine n
    default_n = 15000
    try:
        base_n = int(len(height_sequence_1))

        base_n = default_n

    # Phase 0: parametric candidates

            best_score = scr
            print(f"[init] new best {best_score:.6f}")

        scr = evaluate_sequence(a.tolist())
        if scr < best_score:

    # Phase 1: Adam optimization
    adam_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.6)
    seq_adam, score_adam = _adam_optimize_improved(best_seq, n, adam_deadline)
    if score_adam < best_score:
        best_seq = seq_adam.copy()

        print(f"[adam] new best {best_score:.6f}")

    # Phase 2: hill climbing polishing
    hill_deadline = deadline
    seq_hill, score_hill = _hill_climb(best_seq, n, hill_deadline)
    if score_hill < best_score:
        best_seq = seq_hill.copy()

    # final fallback
    if best_seq is None:

"""
Proposes a non‑negative step function (list of heights) that minimises
`evaluate_sequence`.

Algorithm outline
-----------------
1. Choose a length `n` (up to 15000, preferably the length of a previous
   solution).  All candidates are probability vectors (sum = 1).

2. **Warm‑start** – generate a modest library of hand‑crafted shapes
   (uniform, arcsine, symmetric Beta, simple mixtures) and a few random
   Dirichlet draws.  Keep the best one according to the true objective.

3. **Exploratory random search** – for a short time (≈ 20 % of the budget)
   keep drawing Dirichlet vectors with a variety of concentration
   parameters.  Update the incumbent whenever a better score is found.

4. **Projected Adam optimisation** – starting from the incumbent,
   minimise a smooth surrogate of the maximum convolution:
   * a soft‑max (log‑sum‑exp) term that approximates the max,
   * an auxiliary L² term (`∑ conv²`) that drives the whole convolution
     towards a flat shape.
   The optimisation runs on the simplex using Euclidean projection.
   Learning‑rate, temperature and the weight of the L² term are
   scheduled automatically.

5. **Stochastic hill‑climbing** – cheap local moves that transfer a
   random fraction of mass from one index to another, accepting only
   improving moves (plus an occasional tiny jitter).

All work respects the 1000 s wall‑clock limit; the best sequence seen is
returned as a plain ``list[float]``.
"""

import time

import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weights that approximate the sub‑gradient of
    ``max(conv)``.  For very large ``alpha`` a hard‑max mask is used.
    """
    if alpha > 5000.0:
        max_c = conv.max()
        mask = np.abs(conv - max_c) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(conv)] = 1.0
    else:
        shift = conv - conv.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
    return w


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                      conv: np.ndarray,
                      w: np.ndarray,
                      size_fft: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate  (1/α)·log Σ exp(α·conv)
    w.r.t. the probability vector a.
    """
    # reversed a
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    grad = 2.0 * np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    return grad[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _gradient_l2(a: np.ndarray,
                 conv_fft: np.ndarray,
                 size_fft: int) -> np.ndarray:
    """
    Gradient of the L² surrogate  Σ conv²  w.r.t. a.
    """
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    grad = 2.0 * np.fft.irfft(conv_fft * A_rev, n=size_fft)
    start = a.shape[0] - 1
    return grad[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) pdf evaluated on the mid‑points of n bins."""
    a = max(a, 0.05)           # avoid singularities
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    s = pdf.sum()
    if not np.isfinite(s) or s == 0.0:
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= s
    return pdf

def _candidate_shapes(n: int, prev_seq: Optional[np.ndarray]) -> List[np.ndarray]:
    """Hand‑crafted library of probability vectors for the given length."""
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) arcsine
    candidates.append(_arcsine_distribution(n))

    # 3) symmetric Beta(a,a) on a coarse grid
    for a in np.linspace(0.25, 0.95, 13):
        candidates.append(_beta_distribution(n, a))

    # 4) mixtures of uniform and arcsine
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    arcsine = _arcsine_distribution(n)
    for w in np.linspace(0.0, 0.5, 6):
        mix = (1.0 - w) * arcsine + w * uniform
        candidates.append(mix)

    # 5) arcsine with tiny spikes at the ends
    for spike_frac in (0.10, 0.20):
        mix = arcsine.copy()
        mix[0] += spike_frac / 2
        mix[-1] += spike_frac / 2
        mix = _project_onto_simplex(mix, 1.0)
        candidates.append(mix)

    # 6) the previous best if it matches the length
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())

    # 7) a handful of random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(4):
        candidates.append(rng.dirichlet(np.ones(n)))

    # sanity – enforce the simplex
    return [_project_onto_simplex(c, 1.0) for c in candidates]

def _adam_optimize_improved(init_a: np.ndarray,
                            n: int,
                            deadline: float) -> Tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises a blend of a smooth max‑convolution
    surrogate (soft‑max) and an L² term promoting flatness.
    Returns the best vector encountered and its raw ``2·n·max(conv)`` value.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # temperature for soft‑max
    alpha = 5.0
    alpha_mul = 1.07
    alpha_max = 200_000.0

    # weight of the L² term (starts strong, decays)
    lam = 5.0
    lam_decay = 0.9995

    size_fft = 1 << ((3 * n - 2).bit_length())   # ≥ (2n‑1)+(n)‑1

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- forward transforms -----
        A = np.fft.rfft(a, n=size_fft)
        A_rev = np.fft.rfft(a[::-1], n=size_fft)
        conv_fft = A * A
        conv = np.fft.irfft(conv_fft, n=size_fft)[: 2 * n - 1]

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[adam] iter={it} score={best_score:.6f}")

        # ----- gradient computation -----
        w = _softmax_weights(conv, alpha)

        grad_soft = _gradient_softmax(a, conv, w, size_fft)
        grad_l2 = _gradient_l2(a, conv_fft, size_fft)

        grad = grad_soft + lam * grad_l2

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.3 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ----- projection back onto the simplex -----
        a = _project_onto_simplex(a, 1.0)

        # ----- schedules -----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)
        lam = max(0.0, lam * lam_decay)

    return best, best_score

def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> Tuple[np.ndarray, float]:
    """
    Tiny stochastic local search: move a random fraction of the mass from one
    index to another and keep the move only if it improves the true objective.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue

        # move up to 30 % of the mass at i to j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter to escape very shallow plateaus
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score
