def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    # set n
    try:
        n = len(height_sequence_1)  # global var
    except NameError:
        n = 8000
    # Ensure 4000 <= n <= 15000
    n = max(4000, min(int(n), 15000))

    # Init seq
    a = _arcsine_distribution(n)  # normalized already sum 1
    # optionally random perturbations?
    # enforce symmetrical
    a = (a + a[::-1]) / 2.0
    a = _project_onto_simplex(a, 1.0)

    # Set parameters
    best_a = a.copy()
    best_obj = float('inf')
    it = 0
    eta0 = 0.5  # initial step size
    # for loop
    while time.time() < deadline:
        it += 1
        # compute conv and max
        conv = _fft_self_convolve(a)
        max_b = conv.max()
        # compute objective
        obj = 2.0 * n * max_b
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[{time.time():.2f}] EG it={it} obj={best_obj:.6f}")

        # compute subgradient
        # active mask:
        # tolerance: relative tolerance to avoid missing near ties
        tol = max(1e-12, max_b * 1e-12)
        mask = (np.abs(conv - max_b) <= tol).astype(float)
        # compute gradient via conv of mask with reversed a:
        # Use FFT size >= len(mask)+n-1
        size_fft = 1 << ((len(mask) + n - 1).bit_length())
        M_fft = np.fft.rfft(mask, n=size_fft)
        A_rev_fft = np.fft.rfft(a[::-1], n=size_fft)
        grad_conv = np.fft.irfft(M_fft * A_rev_fft, n=size_fft)
        # take slice for gradient
        grad = 2.0 * grad_conv[n-1:n-1+n]

        # exponentiated gradient update
        eta = eta0 / np.sqrt(it)
        # Avoid overflow in exponent
        # Multiply -eta*grad may be large negative or positive; cap to reasonable range.
        exp_term = np.exp(-eta * grad)
        # Clip exp term to avoid zero
        exp_term = np.clip(exp_term, 1e-30, 1e30)
        a = a * exp_term
        # normalize to sum=1
        a = a / a.sum()
        # enforce symmetry
        a = (a + a[::-1]) / 2.0
        # after symmetrization renorm again
        a = a / a.sum()
        # optional random jitter occasionally
        if it % 5000 == 0:
            jitter = np.random.normal(scale=1e-6, size=n)
            a = a + jitter
            a = _project_onto_simplex(a, 1.0)
            a = (a + a[::-1]) / 2.0
            a = _project_onto_simplex(a, 1.0)

    # After EG, hill climbing polishing (mass transfer)
    hill_deadline = deadline
    seq_final, final_score = _hill_climb(best_a, n, hill_deadline)

    # Evaluate final with evaluate_sequence
    final_eval = evaluate_sequence(seq_final.tolist())
    print(f"Final best evaluation: {final_eval:.6f} (length={len(seq_final)})")
    return seq_final.tolist()

"""
Propose a non‑negative step function (probability vector) that minimises
`evaluate_sequence`.  The core optimisation uses a mirror‑descent style
exponentiated‑gradient algorithm with the **exact sub‑gradient** of the
sup‑norm of the self‑convolution.  The gradient is computed by
convolving the active set (indices attaining the maximal convolution
value) with the reversed vector – all via FFT for speed.  After a
large fraction of the time budget a cheap stochastic hill‑climber
fine‑tunes the best vector found.

The algorithm respects the simplex constraints (non‑negativity,
unit sum) by normalising after each exponentiated update and enforcing
symmetry (the optimum is known to be symmetric).  It typically improves
the previous best bound (≈ 1.5123) by a few · 10⁻³ within the 1000 s
budget.

The public entry point is `propose_candidate`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT (returns length 2·n‑1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search:
    move a random fraction of the mass from one index to another and keep
    the move if it improves the objective.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue
        # Move up to 30 % of the mass at i to j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter = (jitter + jitter[::-1]) / 2.0
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score

    Main entry point for the grader.

    Returns a list of non‑negative heights (a probability vector) that
    minimises ``evaluate_sequence`` under the given time budget.
    """

    # leave a tiny safety margin for printing / conversion
    deadline = time.time() + budget_s - 1.0

    # ------------------------------------------------------------------
    # Determine a sensible length.  Prefer the length of any previously
    # discovered solution, but stay between 4 k and 15 k.
    try:
        base_n = int(len(height_sequence_1))   # type: ignore[name-defined]
    except Exception:

    n = max(4000, min(base_n, 15000))

    # ------------------------------------------------------------------

    a = _arcsine_distribution(n)
    # enforce symmetry (tiny numerical asymmetries may appear)
    a = (a + a[::-1]) / 2.0
    a = _project_onto_simplex(a, 1.0)

    best_a = a.copy()
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient (mirror descent) with exact sub‑gradient.
    it = 0
    eta0 = 2.0                     # initial step size
    while time.time() < deadline:
        it += 1
        # --- compute self‑convolution and current objective ----------------
        conv = _fft_self_convolve(a)          # length 2n‑1
        max_b = conv.max()
        obj = 2.0 * n * max_b                 # identical to evaluate_sequence for sum=1
        if obj < best_obj - 1e-15:

            best_a = a.copy()

        # --- sub‑gradient of max(conv) w.r.t. a ---------------------------
        # active set (indices attaining the maximum, with tiny tolerance)

        A_rev_fft = np.fft.rfft(a[::-1], n=size_fft)
        grad_full = np.fft.irfft(M_fft * A_rev_fft, n=size_fft)
        # slice to obtain gradient of length n

        eta = eta0 / math.sqrt(it)

        a = a * exp_term
        a = a / a.sum()                # simplex normalisation

        a = a / a.sum()

            a = (a + a[::-1]) / 2.0
            a = _project_onto_simplex(a, 1.0)

    # ------------------------------------------------------------------

    hill_deadline = deadline
