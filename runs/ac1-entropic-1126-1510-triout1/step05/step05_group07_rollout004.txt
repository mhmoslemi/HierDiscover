"""
Proposes a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The algorithm works in several stages
while staying inside the 1000 s time budget:

1. **Warm‑start families** – uniform, arcsine, symmetric & asymmetric
   Beta distributions, mixtures with spikes and a few random Dirichlet draws.
2. **Projected Adam** – fast optimiser that uses a soft‑max surrogate for the
   max‑convolution; this gives a good coarse solution.
3. **Exact sub‑gradient descent** – the true sub‑gradient of the
   L∞‑term is obtained by convolving the active‑set mask (indices attaining the
   maximum of the self‑convolution) with the reversed vector.  The update is
   projected onto the probability simplex.
4. **Hill‑climbing polish** – cheap random mass‑swaps and occasional jitter.
5. **Optional up‑sampling** – if time remains the best solution is
   interpolated to a larger length (up to 15000) and the optimisation steps
   are repeated, giving the algorithm extra resolution.

All helper functions are defined at the top level; the only required entry
point is ``propose_candidate``.  The routine returns a plain ``list`` of
floats that satisfies the constraints of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·b_k) w.r.t. a.
    For huge α this approximates the sub‑gradient of max(b).
    """
    n = a.shape[0]

    # --- soft‑max weights -------------------------------------------------
    if alpha > 2_000:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # --- convolution w * reversed(a) ------------------------------------
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float, b: float | None = None) -> np.ndarray:
    """Beta(a,b) pdf on mid‑points of n bins, defaults to symmetric."""
    if b is None:
        b = a
    a = max(a, 0.01)
    b = max(b, 0.01)
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0 or not np.isfinite(pdf_sum):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= pdf_sum
    return pdf

def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest list of handcrafted probability vectors."""
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) arcsine (Beta(½,½))
    candidates.append(_arcsine_distribution(n))

    # 3) symmetric Beta(a,a) on a grid
    for a in np.linspace(0.22, 1.3, 15):
        candidates.append(_beta_distribution(n, a))

    # 4) asymmetric Beta(a,b) (a<b) – a few points
    for a, b in [(0.3, 0.8), (0.8, 0.3), (0.4, 1.0), (1.0, 0.4)]:
        candidates.append(_beta_distribution(n, a, b))

    # 5) mixtures uniform + arcsine
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    arcsine = _arcsine_distribution(n)
    for w in np.linspace(0.0, 0.4, 5):
        mix = (1.0 - w) * arcsine + w * uniform
        candidates.append(mix)

    # 6) arcsine + tiny spikes at the boundaries
    for spike_frac in (0.08, 0.15):
        mix = arcsine.copy()
        mix[0] += spike_frac / 2
        mix[-1] += spike_frac / 2
        mix = _project_onto_simplex(mix, 1.0)
        candidates.append(mix)

    # 7) previous best (if supplied and length matches)
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())

    # 8) a few random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(5):
        candidates.append(rng.dirichlet(np.ones(n)))

    # final projection onto simplex (just to be safe)
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final

def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """Projected Adam optimiser (soft‑max surrogate)."""
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.07
    alpha_max = 5e4

    size_fft = 1 << ((3 * n - 2).bit_length())      # ≥ (2n‑1)+(n)‑1

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] n={n} it={it} score={best_score:.6f}")

        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_onto_simplex(a, 1.0)

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score

def _subgradient_optimize(init_a: np.ndarray,
                          n: int,
                          deadline: float,
                          base_eta: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on  f(a)=2·n·max(a*a) .
    The sub‑gradient is obtained by convolving the mask of active
    indices with the reversed vector.
    """
    a = init_a.copy()
    best = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()

    size_fft = 1 << ((3 * n - 2).bit_length())
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        maxb = conv.max()
        cur_score = 2.0 * n * maxb
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Subgrad] n={n} it={it} score={best_score:.6f}")

        # mask of (almost) maximal entries
        eps = 1e-12
        mask = (conv >= maxb - eps).astype(np.float64)

        # gradient = 2 * (mask * a_rev) convolved, extracted on centre
        a_rev = a[::-1]
        W = np.fft.rfft(mask, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        conv_grad = np.fft.irfft(W * A_rev, n=size_fft)
        start = n - 1
        grad = 2.0 * conv_grad[start:start + n]

        # normalise direction
        g_norm = np.linalg.norm(grad)
        if g_norm > 0:
            direction = grad / g_norm
        else:
            direction = np.zeros_like(a)

        step_size = base_eta / np.sqrt(it)
        a = a - step_size * direction
        a = _project_onto_simplex(a, 1.0)

    return best, best_score

def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """Very cheap stochastic local search."""
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-14:
            continue

        # move a random fraction (≤30 %) of mass from i to j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score


# ----------------------------------------------------------------------
def _upscale_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    old_pos = (np.arange(old_n) + 0.5) / old_n
    new_pos = (np.arange(new_n) + 0.5) / new_n
    up = np.interp(new_pos, old_pos, a)
    up = np.maximum(up, 0.0)
    up = _project_onto_simplex(up, 1.0)
    return up

    try:
        base_n = int(len(height_sequence_1))  # type: ignore[name-defined]
    except Exception:
        base_n = 8000
    base_n = max(3000, min(base_n, 15000))

    # We'll try at most two different lengths: base_n and a larger one.
    length_plan = [base_n]
    if base_n < 13000:
        length_plan.append(min(15000, int(base_n * 1.5)))

    # Global best tracking
    best_seq_global = None
    best_score_global = float("inf")

    # --------------------------------------------------------------
    for n in length_plan:
        if time.time() >= deadline:
            break

        # ---------- Phase 0 : generate candidates and pick the best ----------
        prev_seq = None
        try:
            prev_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        except Exception:
            pass

        # if we already have a global best of a different length, upscale it
        if best_seq_global is not None and best_seq_global.shape[0] == n:
            start_candidates = [best_seq_global.copy()]
        else:
            start_candidates = _candidate_shapes(n, prev_seq)

        # quick scan of the candidates (bounded time)
        shape_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.15)
        local_best = None
        local_best_score = float("inf")
        for cand in start_candidates:
            if time.time() >= shape_deadline:
                break
            scr = evaluate_sequence(cand.tolist())
            if scr < local_best_score:
                local_best_score = scr
                local_best = cand.copy()
                print(f"[shape] n={n} new best {scr:.6f}")

        if local_best is None:
            local_best = np.full(n, 1.0 / n, dtype=np.float64)
            local_best_score = evaluate_sequence(local_best.tolist())

        # ---------- Phase 1 : Adam (soft‑max) ----------
        adam_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.45)
        seq_adam, _ = _adam_optimize(local_best, n, adam_deadline)

        # ---------- Phase 2 : Exact sub‑gradient descent ----------
        sub_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.30)
        seq_sub, sub_score = _subgradient_optimize(seq_adam, n, sub_deadline)

        # ---------- Phase 3 : Stochastic hill climbing ----------
        hill_deadline = deadline
        seq_final, final_score = _hill_climb(seq_sub, n, hill_deadline)

        # Final evaluation via the provided routine
        final_eval = evaluate_sequence(seq_final.tolist())
        print(f"[final] n={n} evaluated score {final_eval:.6f}")

        if final_eval < best_score_global:
            best_score_global = final_eval
            best_seq_global = seq_final.copy()
            print(f"[global] new best score {best_score_global:.6f} (n={n})")

        # --------------------------------------------------------------
        # If we still have time and we have not yet tried the larger length,
        # upscale the best solution we have and continue the loop.
        if len(length_plan) > 1 and n == base_n:
            # prepare for the larger length in the next iteration
            continue

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq_global is None:
        best_seq_global = np.full(4000, 1.0 / 4000, dtype=np.float64)

    print(f"Returning best sequence of length {len(best_seq_global)} with score {best_score_global:.6f}")
    return best_seq_global.tolist()
