"""
Propose a non‑negative step‑function that (hopefully) improves the current
best bound for ``evaluate_sequence``.

Algorithm
---------
1. **Random parametric search** – generate many candidates from a diverse
   collection of simple families (arcsine, power‑law, symmetric Beta,
   Gaussian, triangular, raised‑cosine and mixtures).  All vectors are
   projected onto the probability simplex (sum = 1) and evaluated with a
   fast FFT‑based exact objective
        2·n·max(a∗a)   (the denominator is 1 because ∑a=1).

2. **Exponential‑gradient refinement** – the best few candidates are
   refined with a mirror‑descent / exponentiated‑gradient routine.
   The update uses a smooth approximation of the ``max`` via a
   log‑sum‑exp with an annealed temperature α; after α becomes large,
   the true sub‑gradient of the max is used.  The step‑size η decays as
   1/√t and the iterates are constantly re‑projected onto the simplex.

3. **Simulated‑annealing hill climbing** – the current global best is
   further improved by random mass transfers between two coordinates.
   Moves that increase the objective are occasionally accepted according
   to a temperature schedule, allowing the search to escape shallow
   local minima.

The routine respects a total wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

# ----------------------------------------------------------------------
# Simplex utilities
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, sum(x)=s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution / objective helpers
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())

def _full_self_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) for a simplex‑normalised vector."""
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv          # denominator is 1 (∑a = 1)

# Gradient helpers
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k−i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g

def _gradient_softmax(a: np.ndarray, b: np.ndarray,
                     alpha: float, size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·b) w.r.t. a.
    Implements  g = 2·conv_rev(w, a_rev) where w is the softmax of α·b.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g

# Parametric families (all return a probability vector of length n)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    vals = prod ** (-p) if invert else prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals

def _beta_distribution(n: int, a: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals

def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals

def _triangular_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - mid) / mid
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

def _raised_cosine_distribution(n: int, power: float) -> np.ndarray:
    # f(x) = (0.5*(1+cos(2πx)))^power,  x∈[0,1)
    i = np.arange(n, dtype=np.float64) / n
    vals = (0.5 * (1.0 + np.cos(2.0 * np.pi * i))) ** power
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals

def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)

def _mixture_two_beta(n: int, w: float, a1: float, a2: float) -> np.ndarray:
    beta1 = _beta_distribution(n, a1)
    beta2 = _beta_distribution(n, a2)
    vals = w * beta1 + (1.0 - w) * beta2
    return _project_onto_simplex(vals, s=1.0)

def _spike_uniform_mix(n: int, spike_frac: float, spike_pos: str) -> np.ndarray:
    """Uniform plus a single spike at one endpoint (or both)"""
    base = np.full(n, (1.0 - spike_frac) / n, dtype=np.float64)
    if spike_pos == "left":
        base[0] += spike_frac
    elif spike_pos == "right":
        base[-1] += spike_frac
    elif spike_pos == "both":
        base[0] += spike_frac / 2.0
        base[-1] += spike_frac / 2.0
    return _project_onto_simplex(base, s=1.0)

        # gradient: soft‑max while alpha small, true sub‑gradient later
        if alpha < alpha_thr:
            grad = _gradient_softmax(a, b, alpha, size_fft)
        else:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)

        # step‑size schedule (1/√t)
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional occasional jitter to shake out of plateaux
        if it % 2000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _local_hill_climb(a_start: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Classic random‑move hill‑climber with a simple Simulated‑Annealing
    acceptance rule (temperature decays linearly).
    """
    a = a_start.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)

    total_budget = deadline - time.time()
    start = time.time()
    while time.time() < deadline:
        # linear temperature: start ~0.01, end ~1e‑5
        elapsed = time.time() - start
        T = max(1e-5, 0.02 * (1.0 - elapsed / total_budget))

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a random fraction of the source mass (up to 15 %)
        eps = a[i] * np.random.rand() * 0.15
        if eps <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        # safety check
        if a_candidate[i] < 0.0:
            continue

        cur_score = _objective(a_candidate)

        if cur_score < best_score:
            # strict improvement
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] hill‑climb improve score={best_score:.6f}")
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-(cur_score - best_score) / T):
                a = a_candidate
    return best_seq, best_score

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Determine a sensible base length (try to reuse a previous solution)
    try:
        base_n = len(height_sequence_1)            # type: ignore[name-defined]
        if base_n < 500:
            base_n = 8000
    except Exception:
        base_n = 8000

    # Choose a set of lengths to explore (500 .. 20000)
    length_factors = (0.5, 0.7, 0.9, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    })
    # ensure a few hand‑picked values are present
    lengths.extend([4000, 5000, 8000, 12000, 16000, 20000])
    lengths = sorted({l for l in lengths if l <= 20000})

    # ------------------------------------------------------------------
    # Global bookkeeping
    best_seq = None
    best_score = float("inf")
    best_pool: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # Seed pool with known good candidates (if any)
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if seq0.size:
            seq0 = np.clip(seq0, 0.0, None)
            seq0 = _project_onto_simplex(seq0, s=1.0)
            sc0 = _objective(seq0)
            print(f"Seed (prev) score={sc0:.6f}")
            best_seq = seq0.copy()
            best_score = sc0
            best_pool.append((sc0, seq0))
    except Exception:
        pass

    # Simple arcsine baseline
    arcs = _arcsine_distribution(base_n)
    arcs_sc = _objective(arcs)
    print(f"Baseline arcsine (n={base_n}) score={arcs_sc:.6f}")
    if arcs_sc < best_score:
        best_seq = arcs.copy()
        best_score = arcs_sc
    best_pool.append((arcs_sc, arcs))

    # ------------------------------------------------------------------
    # Phase 1 – random parametric search (≈ 65 % of total budget)
    phase1_deadline = deadline - 250.0   # reserve time for later phases
    families = (
        "arcsine", "power_inv", "power_dir", "beta",
        "gauss", "triangular", "raised_cosine",
        "mix_arcsine_unif", "mix_two_beta", "spike_mix"
    )
    while time.time() < phase1_deadline:
        n = int(np.random.choice(lengths))
        fam = np.random.choice(families)

        if fam == "arcsine":
            seq = _arcsine_distribution(n)
        elif fam == "power_inv":
            p = np.random.uniform(0.2, 1.5)
            seq = _power_law_distribution(n, p, invert=True)
        elif fam == "power_dir":
            p = np.random.uniform(0.2, 1.5)
            seq = _power_law_distribution(n, p, invert=False)
        elif fam == "beta":
            a = np.random.uniform(0.2, 6.0)
            seq = _beta_distribution(n, a)
        elif fam == "gauss":
            sigma = np.random.uniform(n / 30.0, n / 2.0)
            seq = _gaussian_distribution(n, sigma)
        elif fam == "triangular":
            seq = _triangular_distribution(n)
        elif fam == "raised_cosine":
            power = np.random.uniform(0.5, 5.0)
            seq = _raised_cosine_distribution(n, power)
        elif fam == "mix_arcsine_unif":
            lam = np.random.uniform(0.0, 1.0)
            seq = _mixture_arcsine_uniform(n, lam)
        elif fam == "mix_two_beta":
            w = np.random.uniform(0.0, 1.0)
            a1 = np.random.uniform(0.2, 5.0)
            a2 = np.random.uniform(0.2, 5.0)
            seq = _mixture_two_beta(n, w, a1, a2)
        elif fam == "spike_mix":
            spike_frac = np.random.uniform(0.01, 0.2)
            pos = np.random.choice(["left", "right", "both"])
            seq = _spike_uniform_mix(n, spike_frac, pos)
        else:
            seq = _arcsine_distribution(n)

        seq = _project_onto_simplex(seq, s=1.0)
        sc = _objective(seq)

        # keep global best
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"New global best (rand) n={n} fam={fam} score={best_score:.6f}")

        # maintain a small pool of promising candidates (size ≤ 12)
        if len(best_pool) < 12:
            best_pool.append((sc, seq))
        else:
            worst_idx = max(range(len(best_pool)),
                            key=lambda i: best_pool[i][0])
            if sc < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (sc, seq)

    # ------------------------------------------------------------------
    # Phase 2 – exponential‑gradient refinement (≈ 25 % of total budget)
    best_pool.sort(key=lambda x: x[0])
    to_refine = [cand for _, cand in best_pool[:4]]

    for idx, cand in enumerate(to_refine, start=1):
        if time.time() >= deadline - 70.0:
            break
        n = cand.shape[0]
        # allocate up to 55 s for each refinement, respecting remaining budget
        remaining = deadline - time.time()
        allotted = min(55.0, remaining * 0.5)
        slice_deadline = time.time() + allotted
        refined, _ = _optimize_exponential(cand, slice_deadline)
        refined_score = _objective(refined)
        print(f"Refinement {idx}/{len(to_refine)} n={n} → score={refined_score:.6f}")
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()
            print(f"*** Improved global best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final simulated‑annealing hill climbing (remaining time)
    if best_seq is not None and time.time() < deadline - 1.0:
        final_seq, final_score = _local_hill_climb(best_seq, deadline)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"After SA hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback safety net
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length = {best_seq.shape[0]})")
    return best_seq.tolist()
