"""
Optimisation of non‑negative step‑functions for the “max‑convolution” score.

The algorithm works in three phases:

1. **Parameter sweep** – a rich two‑parameter family
      a_i ∝ ((i+δ)*(n‑i‑δ))⁻ᵖ
   (δ>0, p>0) is mixed with a uniform part.
   A coarse grid in (p, δ, w) is evaluated with a cheap FFT‑based score
   (2·n·max conv) and the 5 best candidates are kept.

2. **Adam‑mirrored gradient descent on the simplex** – each candidate is
   refined with a projected Adam optimiser.
   Early iterations use a soft‑max surrogate of the max‑norm (temperature α)
   for a smooth gradient; later iterations switch to the true sub‑gradient
   of the max entry.  The update is projected onto the probability simplex
   after every step, guaranteeing non‑negativity and unit sum.

3. **Stochastic local hill‑climbing** – the globally best vector is finally
   polished by moving tiny amounts of mass between random indices whenever
   the true evaluation score improves.

The routine respects the 1000 s budget, prints progress, and returns the
best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho_arr.size == 0 else cssv[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) convolution of *a* with itself via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power‑of‑2
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max entry of a*a at index *k*."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of (1/α)·log ∑ exp(α·b) w.r.t. a (soft‑max surrogate)."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]                               # because convolution ≠ correlation
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _shape(p: float, delta: float, n: int) -> np.ndarray:
    """Return a symmetric power‑law shape:
          a_i ∝ ((i+δ)*(n‑i‑δ))⁻ᵖ  (δ>0)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + delta) * (n - i - delta)) ** (-p)
    vals_sum = vals.sum()
    if vals_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / vals_sum


# ----------------------------------------------------------------------
def _mix_with_uniform(shape: np.ndarray, w: float) -> np.ndarray:
    """Linear mixture of a shape with the uniform distribution."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = (1.0 - w) * shape + w * uniform
    return _project_onto_simplex(mixed, s=1.0)


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Fast internal score = 2·n·max(conv) (sum(a) is 1)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def _grid_search(n: int, deadline: float):
    """Explore a modest grid (p,δ,w) and return the top‑5 candidates."""
    candidates = []

    # fixed parameter grids
    p_vals = np.linspace(0.40, 0.60, 11)          # around the arcsine exponent 0.5
    delta_vals = np.linspace(0.2, 2.0, 10)       # soften the singularity
    w_vals = np.linspace(0.0, 0.4, 5)            # modest uniform mixing

    for p in p_vals:
        for delta in delta_vals:
            shape = _shape(p, delta, n)
            for w in w_vals:
                cand = _mix_with_uniform(shape, w)
                candidates.append(cand)
                # early abort if time is out
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # also add a few hand‑crafted seeds
    seeds = []
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    seeds.append(arcsine / arcsine.sum())
    seeds.append(np.full(n, 1.0 / n))
    seeds.append(np.random.rand(n))
    for s in seeds:
        seeds[-1] = _project_onto_simplex(s, s=1.0)

    for s in seeds:
        candidates.append(s)

    # keep the best 5 according to the cheap score
    scored = [(cand, _cheap_score(cand)) for cand in candidates]
    scored.sort(key=lambda x: x[1])
    return [c for c, _ in scored[:5]]

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex with a soft‑max surrogate.
    Returns the best vector seen and its internal score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _cheap_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # parameters for the surrogate
    alpha = 5.0                 # start temperature
    alpha_mul = 1.015
    alpha_max = 250.0

    # FFT size needed for the gradient convolution (w * a_rev)
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        # --------------------------------------------------------------
        # current objective and saving the best
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] iter={it} new best internal={best_score:.6f}")

        # --------------------------------------------------------------
        # gradient (soft‑max while alpha is moderate, then true sub‑gradient)
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # --------------------------------------------------------------
        # Adam update (additive) followed by projection onto the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # a simple decaying step‑size; 0.15 works well in practice
        eta = 0.15 / np.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter to escape plateaus
        if it % 3000 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best, best_score


# ----------------------------------------------------------------------
def _local_hillclimb(a_start: np.ndarray,
                     n: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing using the true evaluation score.
    Moves a tiny fraction of mass from i to j whenever this reduces the
    score.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = _cheap_score(a)

    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # a modest transfer – up to 5 % of the source mass
        eps = a[i] * rng.random() * 0.05
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        # project is not needed – we keep the sum constant and non‑negativity by construction
        cur_score = _cheap_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best = cand.copy()
            a = cand
    return best, best_score

          f"(length={n}, evaluate_sequence={best_overall_score:.6f})")
    return best_overall.tolist()
