def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0 # margin

    # Determine length
    target_lengths = [8000, 12000, 16000, 20000]
    best_seq = None
    best_score = float('inf')
    best_n = None

        # run optimizer

    # final evaluation
    final_score = evaluate_sequence(best_seq.tolist())

def adam_optimize(x, n, deadline):
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2
    it = 0
    best_local_seq = None
    best_local_score = float('inf')
    while time.time() < deadline:
        it += 1
        # compute a from x
        p = np.exp(x - np.max(x))  # to avoid overflow
        p = p / p.sum()
        half = 0.5 * p
        a = np.concatenate([half, half[::-1]])
        # compute convolution max
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        # compute raw score = 2 * n * cur_max
        cur_score = 2.0 * n * cur_max
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
        # compute subgradient of max wrt a
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        # compute grad_a = 2 * conv_of_rev_a_w but we ignore factor 2 maybe.
        a_rev = a[::-1]
        grad_a = 2.0 * _fft_convolve(a_rev, w)[n-1 : n-1 + n]  # shape n
        # combine for half variables
        # for each i in half indices 0..h-1
        idx = np.arange(h)
        grad_half = 0.5 * (grad_a[idx] + grad_a[n-1-idx])  # combine contributions
        # compute gradient wrt x
        sum_comb = np.dot(p, grad_half)
        grad_x = p * (grad_half - sum_comb)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** it)

def _optimize_length(n, deadline):
    half = n // 2
    best_seq = None
    best_score = float('inf')
    restart = 0
    max_restarts = 8
    while restart < max_restarts and time.time() < deadline:
        # random init
        x = np.random.randn(half) * 0.5
        # Adam optimizer
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr0 = 0.2
        it = 0
        # Determine iteration limit per restart based on remaining time
        # approximate per iteration time; we'll just loop until a limit or time.
        max_iters = 5000
        while it < max_iters and time.time() < deadline:
            it += 1
            # compute a
            # stable softmax
            max_x = np.max(x)
            exp_x = np.exp(x - max_x)
            p = exp_x / exp_x.sum()
            # half values
            half_vals = 0.5 * p
            a = np.concatenate([half_vals, half_vals[::-1]])
            # compute convolution and subgradient
            conv = _fft_self_convolve_full(a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                # optionally print
                print(f"[{time.time():.1f}] restart {restart} it {it} new best = {best_score:.6f}")
            # subgradient
            tol = max(1e-12, cur_max * 1e-12)
            active = np.where(conv >= cur_max - tol)[0]
            w = np.zeros_like(conv)
            if active.size:
                w[active] = 1.0 / active.size
            a_rev = a[::-1]
            grad_a = 2.0 * _fft_convolve(a_rev, w)[n-1 : n-1 + n]
            # combine for half
            idx = np.arange(half)
            grad_half_comb = 0.5 * (grad_a[idx] + grad_a[n-1-idx])
            sum_comb = np.dot(p, grad_half_comb)
            grad_x = p * (grad_half_comb - sum_comb)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_x
            v = beta2 * v + (1 - beta2) * (grad_x ** 2)
            it_float = float(it)  # ensure
            m_hat = m / (1 - beta1 ** it_float)
            v_hat = v / (1 - beta2 ** it_float)
            lr = lr0 * (np.sqrt(1 - beta2 ** it_float) / (1 - beta1 ** it_float))
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
            # jitter
            if it % 1000 == 0:
                x += np.random.randn(half) * 0.01
        restart += 1
    return best_seq, best_score

"""
Propose a step function by *direct* optimisation on the simplex.

The sequence a (non‑negative, Σa = 1) is parametrised as a soft‑max of
unconstrained logits x.  For a given length n we minimise

    f(a) = 2·n·max(a*a)               (self‑convolution supremum)

which is exactly the quantity returned by ``evaluate_sequence`` after the
normalisation step.  The gradient of the inner max is obtained as a
sub‑gradient by selecting the active convolution indices and convolving the
reversed a with the corresponding weight vector.  optimisation is performed
with Adam (adaptive learning‑rate, momentum) and a few random restarts.
Solutions for increasing lengths are obtained by linear interpolation of the
previous best shape – this gives a good warm‑start for larger n.
Finally a short projected sub‑gradient descent (as used in the reference
solution) refines the best candidate.

The whole routine respects the supplied time budget (default 1000 s) and
returns a plain Python list suitable for the black‑box ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max), where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    # gradient of the max (still contains a factor 2)
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    # points are taken at the mid‑points of each bin
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _adam_optimize_from_logits(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 4000,
    lr0: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logit parametrisation.
    Returns the best vector (a) found and its raw score 2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        # --- soft‑max (stable) ---
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # --- objective and raw score ---
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # a little progress report (won't flood the output)
            print(f"[{time.time():.1f}] n={n} it={it}  raw score={best_score:.6f}")

        # --- sub‑gradient of the max ---
        grad_a, _ = _max_subgradient_step(a)

        # --- chain rule through the soft‑max ---
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _optimise_length(
    n: int,
    deadline: float,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n.
    Several initialisations are tried (interpolated previous best, uniform,
    a few random Dirichlet draws).  The best of all runs is returned.
    """
    # ----- build list of start logits ---------------------------------
    init_logits = []

    # 1) warm‑start from the previous best (if lengths match or can be interpolated)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        # log‑odds; add a tiny epsilon to avoid log(0)
        eps = 1e-12
        init_logits.append(np.log(cand + eps))

    # 2) uniform start (soft‑max of zero vector)
    init_logits.append(np.zeros(n, dtype=np.float64))

    # 3) a few random Dirichlet draws – give diverse shapes
    for _ in range(3):
        alpha = np.random.rand() * 4.0 + 0.1           # α ∈ (0.1,4.1)
        dir_sample = np.random.gamma(alpha, 1.0, size=n)
        dir_sample /= dir_sample.sum()
        init_logits.append(np.log(dir_sample + 1e-12))

    # ----- run Adam from each start -----------------------------------
    best_local_seq = None
    best_local_score = float("inf")
    for ix, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, start {ix+1}/{len(init_logits)}")
        seq, score = _adam_optimize_from_logits(
            x0, n, deadline, max_iters=4000, lr0=0.12
        )
        if score < best_local_score:
            best_local_score = score
            best_local_seq = seq.copy()

    # safety fallback
    if best_local_seq is None:
        best_local_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_local_score = 2.0 * n * _fft_self_convolve_max(best_local_seq)

    return best_local_seq, best_local_score
