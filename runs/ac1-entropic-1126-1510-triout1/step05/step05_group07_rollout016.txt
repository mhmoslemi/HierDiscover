"""
Proposes a non‑negative step function (list of heights) that minimises
`evaluate_sequence`.  The algorithm consists of three stages:

1. **Dense grid search** over a flexible parametric family:
   mixtures of a symmetric Beta(α,α) density, a uniform background
   and optional point spikes at the two ends.  The grid (α ∈ [0.05,0.5],
   spike‑mass w_s ∈ [0,0.4], uniform‑mass w_u ∈ [0,0.4] with w_s+w_u≤1)
   explores far more shapes than the previous code and quickly finds a
   very good starting vector.

2. **Projected Adam optimisation** on the simplex, using the same
   soft‑max surrogate for the max‑convolution as before but with an
   aggressive temperature schedule and a slightly larger base learning
   rate.  After the first long run the procedure is restarted several
   times from fresh random Dirichlet draws; the best run is kept.

3. **Stochastic hill‑climbing** (mass swaps + occasional jitter) for a
   final polishing phase.

All work respects the 1000 s time budget.  The returned list is a plain
Python list of floats suitable for `evaluate_sequence`.
"""

import time
import math
import numpy as np
import random

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _score_seq(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (avoids clipping overhead)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    return float(2.0 * len(a) * max_conv / (s * s))


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(alpha,alpha) pdf on the mid‑points of n bins (sum=1)."""
    alpha = max(alpha, 0.05)                    # avoid division by zero
    x = (np.arange(n) + 0.5) / n                # mid‑point grid
    pdf = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0 or not np.isfinite(pdf_sum):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= pdf_sum
    return pdf


# ----------------------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
def _spike_distribution(n: int, mass: float) -> np.ndarray:
    """Put `mass/2` at each endpoint, zero elsewhere (sum=mass)."""
    a = np.zeros(n, dtype=np.float64)
    a[0] = mass / 2.0
    a[-1] = mass / 2.0
    return a


# ----------------------------------------------------------------------
def _candidate_grid(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a rich set of candidate probability vectors.
    The families are:
        - Beta(α,α) for a dense grid of α.
        - Mixtures: (1‑w_s‑w_u)·Beta + w_s·spikes + w_u·uniform
          with w_s ∈ {0,0.05,…,0.40}, w_u ∈ {0,0.05,…,0.40‑w_s}.
    """
    candidates: list[np.ndarray] = []

    # basic building blocks
    uniform = _uniform_distribution(n)

    # coarse beta grid (will be refined later)
    alphas = np.linspace(0.05, 0.5, 10)               # 0.05 … 0.5
    betas = [_beta_distribution(n, a) for a in alphas]

    # generate mixtures
    spike_weights = [0.0, 0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40]
    uniform_weights = [0.0, 0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40]

    for beta in betas:
        # pure beta
        candidates.append(beta.copy())

        for w_s in spike_weights:
            for w_u in uniform_weights:
                if w_s + w_u > 0.8:          # keep enough room for beta part
                    continue
                w_beta = 1.0 - w_s - w_u
                mix = w_beta * beta
                if w_u > 0:
                    mix += w_u * uniform
                if w_s > 0:
                    mix += _spike_distribution(n, w_s)
                # projection onto simplex (tiny numerical drift)
                mix = _project_onto_simplex(mix, 1.0)
                candidates.append(mix)

    # a few random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(12):
        rand = rng.dirichlet(np.ones(n))
        candidates.append(rand)

    # previous best (if supplied)
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(_project_onto_simplex(prev_seq, 1.0))

    # ensure all are proper probabilities
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         conv: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate (1/α)·log Σ exp(α·conv_k) w.r.t a.
    For huge α this becomes a sub‑gradient of max(conv).
    """
    n = a.shape[0]

    # soft‑max weights
    if alpha > 2_000:
        max_c = conv.max()
        mask = np.abs(conv - max_c) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(conv)] = 1.0
    else:
        shift = conv - conv.max()
        w = np.exp(alpha * shift)
        s = w.sum()
        if s == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= s

    # convolution w * reversed(a)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_w = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    grad = 2.0 * conv_w[start:start + n]   # factor 2 from derivative of a*a
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the simplex.
    Returns the best vector encountered and its corresponding 2·n·max(conv)
    score (without the denominator because sum(init_a)=1).
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    alpha = 5.0                # temperature of the soft‑max surrogate
    alpha_mul = 1.09
    alpha_max = 80_000.0

    size_fft = 1 << ((3 * n - 2).bit_length())      # ≥ (2n‑1)+(n)‑1

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] n={n} it={it} score={best_score:.6f}")

        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.3 / (np.sqrt(v_hat) + eps)   # a slightly larger base step
        a = a - step * m_hat
        a = _project_onto_simplex(a, 1.0)

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: randomly move a fraction of the mass
    from one index to another and keep the move if it improves the score.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue

        # move a random fraction (up to 25 %) from i to j
        frac = rng.random() * 0.25
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional jitter to escape shallow plateaus
        if it % 4000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score
