def fft_convolve(a, b):
    size = len(a) + len(b) - 1
    L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = A * B
    conv = np.fft.irfft(C, n=L)[:size]
    return conv

def compute_obj_and_grad(y, mu, n, S):
    p = softmax(y)
    a = S * p
    conv = fft_convolve(a, a)
    max_c = conv.max()
    # compute w:
    if mu > 0.0:
        diff = (conv - max_c) / mu
        np.clip(diff, -700, 700, out=diff)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max, uniform weighting among ties
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c
    # compute gradient w.r.t a
    a_rev = a[::-1]
    conv_wr = fft_convolve(w, a_rev)
    # conv_wr length = len(w)+len(a_rev)-1 = (2n-1)+n-1 = 3n-2
    # central part indices n-1 to 2n-2 (inclusive) => length n
    g_a = 2.0 * conv_wr[n-1:n-1+n]
    # gradient w.r.t y via softmax Jacobian
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return f_val, grad_y

"""
Propose a low‑score step function.

The algorithm consists of three phases:

1.  Smooth‑max L‑BFGS optimisation.
    We minimise a smooth approximation of max(conv(a,a)) using
    a temperature μ that is gradually cooled.  Convolutions and
    their gradients are computed with FFT (O(n log n)) which makes
    many L‑BFGS iterations cheap even for n ≈ 10 000.

2.  Hard‑max sub‑gradient descent.
    Starting from the best candidate we walk opposite to a
    sub‑gradient of the true max‑convolution objective, projecting
    onto the ℓ¹‑simplex after each step.  A simple back‑tracking line
    search and occasional random jitter are used.

3.  Large‑scale random‑block perturbations + monotone sorting.
    Cheap random block constructions are projected onto the simplex
    and accepted if they improve the objective.  Finally the best
    vector is sorted increasingly and decreasingly – these monotone
    shapes are often beneficial.

All work respects a user‑supplied time budget (default 1000 s) and
returns the best height sequence found.  The evaluation uses the
same scaling as the reference implementation (∑aᵢ = √(2 n)) so that
the objective value equals the maximal auto‑convolution M,
i.e. the score returned by ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution using FFT (real inputs)."""
    size = a.shape[0] + b.shape[0] - 1
    L = 1 << (size - 1).bit_length()               # next power of two
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = A * B
    conv = np.fft.irfft(C, n=L)[:size]
    return conv


def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective and gradient.
    The unconstrained variables y are transformed to probabilities
    p = softmax(y) and then to heights a = S·p (Σa = S).
    """
    p = _softmax(y)                     # Σp = 1, p_i ≥ 0
    a = S * p                           # Σa = S

    # convolution a*a (length 2·n‑1)
    conv = _fft_convolve(a, a)
    max_c = conv.max()

    # smooth‑max weights
    if mu > 0.0:
        diff = (conv - max_c) / mu
        np.clip(diff, -700.0, 700.0, out=diff)   # guard overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                                   # hard max (μ = 0)
        mask = conv == max_c
        cnt = int(mask.sum())
        w = np.zeros_like(conv)
        if cnt:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a:
    #   g_a[i] = 2·Σ_k w_k·a_{k‑i}
    a_rev = a[::-1]
    conv_wr = _fft_convolve(w, a_rev)                # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]             # slice length n

    # chain rule: ∂f/∂y = S·p·(g_a − ⟨p,g_a⟩)
    inner = np.dot(g_a, p)                           # scalar
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable returning (value, gradient) for scipy.optimize."""
    __slots__ = ("mu", "n", "S")
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S
    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


def _quick_score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv)/sum² (identical to evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _fft_convolve(a, a)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (sum_a * sum_a))

    Search for a non‑negative step‑function with a low evaluate_sequence
    score.  The routine respects the given time budget and returns the
    best height list it discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        # length of the reference if it exists
        ref_len = len(height_sequence_1)   # type: ignore[name-defined]
        n = min(max(ref_len * 2, 4000), 12000)
    except Exception:
        n = 8000
    S = math.sqrt(2.0 * n)                  # target ℓ¹‑norm

    best_a = np.zeros(n, dtype=np.float64)
    best_score = float("inf")

    def _maybe_update(a_vec: np.ndarray) -> None:
        """Update incumbent if the new candidate is better."""
        nonlocal best_a, best_score
        sc = _quick_score(a_vec)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # generate diverse initial points for the smooth‑max stage
    init_ys = []

    # 0 – uniform (y = 0)
    init_ys.append(np.zeros(n, dtype=np.float64))

    # 1 – Gaussian
    init_ys.append(rng.standard_normal(n) * 0.5)

    # 2 – Dirichlet (sparse)
    alpha_sparse = rng.uniform(0.05, 0.30)
    p_dir = rng.dirichlet([alpha_sparse] * n)
    init_ys.append(np.log(p_dir + 1e-12))

    # 3 – decreasing exponential
    decay = rng.uniform(0.001, 0.004) * (n / 2000.0)
    init_ys.append(-decay * np.arange(n, dtype=np.float64))

    # 4 – increasing exponential (mass on right)
    init_ys.append(-decay * np.arange(n - 1, -1, -1, dtype=np.float64))

    # 5 – reference if present (rescaled)
    if "height_sequence_1" in globals():
        ref = np.asarray(height_sequence_1, dtype=np.float64)
        if ref.shape[0] != n:
            # linear interpolation to required length
            x_old = np.linspace(0, 1, ref.shape[0])
            x_new = np.linspace(0, 1, n)
            ref = np.interp(x_new, x_old, ref)
        ref = _project_to_simplex(ref, S)
        p_ref = np.clip(ref / S, 1e-12, None)
        init_ys.append(np.log(p_ref))

    # 6 – a couple of extra random seeds
    for _ in range(2):
        init_ys.append(rng.standard_normal(n) * 0.5)

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max L‑BFGS with temperature cooling
    mu_factor = 0.85          # slower cooling than the baseline
    mu_min = 1e-9
    maxiter_per_mu = 500     # more internal L‑BFGS steps
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid+1}/{n_restart}]")
        y = y0.copy()

        # initialise temperature
        a_cur = S * _softmax(y)
        conv_cur = _fft_convolve(a_cur, a_cur)
        mu = max(conv_cur.max(), 1.0)

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False}
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max sub‑gradient descent from the best point
    if time.time() < deadline and best_score < float("inf"):
        a = _project_to_simplex(best_a, S)
        it = 0
        max_local_iters = 5000
        step0 = S * 0.4 / n

        while time.time() < deadline and it < max_local_iters:
            conv = _fft_convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = int(mask.sum())
            if cnt == 0:
                break
            w = np.zeros_like(conv)
            w[mask] = 1.0 / cnt

            # sub‑gradient
            a_rev = a[::-1]
            conv_wr = _fft_convolve(w, a_rev)
            g = 2.0 * conv_wr[n - 1:n - 1 + n]

            # back‑tracking line search
            step = step0 / (1.0 + 1e-5 * it)
            improved = False
            for _ in range(15):
                a_new = a - step * g
                a_new = _project_to_simplex(a_new, S)
                sc = _quick_score(a_new)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[subgrad] iter {it:4d} step {step:.6f} => {best_score:.6f}")
                    break
                step *= 0.5

            if not improved:
                # small random jitter to escape flat regions
                i, j = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a[i] * 0.05
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, S)
            it += 1

    # ------------------------------------------------------------------
    # Phase 3 – cheap random block perturbations
    if time.time() < deadline and best_score < float("inf"):
        remaining = deadline - time.time()
        max_trials = int(min(20000, remaining * 180))
        print(f"[blocks] {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(2, 12)                     # number of blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)                        # uniform heights
            cand = np.empty(n, dtype=np.float64)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # ------------------------------------------------------------------
    # Monotone‑sorting heuristic (often helps)
    if time.time() < deadline and best_score < float("inf"):
        sorted_desc = np.sort(best_a)[::-1]
        sorted_desc = _project_to_simplex(sorted_desc, S)
        _maybe_update(sorted_desc)

        sorted_asc = np.sort(best_a)
        sorted_asc = _project_to_simplex(sorted_asc, S)
        _maybe_update(sorted_asc)

    # ------------------------------------------------------------------
    # Final official evaluation (one call only)
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence score = {final_score:.6f}")

    return final_seq
