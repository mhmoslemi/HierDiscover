"""
Propose a step function that minimizes `evaluate_sequence`.

The algorithm works in three stages:

1. **Broad parametric sweep** – Generates candidates from several
   analytic families (arcsine, power‑law, beta, Gaussian, mixtures).  For
   each admissible length we sweep a few parameters on a grid, keep the
   best few (on the simplex) and remember the global best.

2. **Additive projected gradient descent** – Refines the top candidates.
   The update reduces the current maximal entry of the self‑convolution
   using its sub‑gradient and then projects back onto the probability
   simplex.  A simple diminishing step size (η∝1/√t) is used; the loop
   runs until the time budget for the candidate is exhausted.

3. **Targeted hill‑climbing** – Starting from the best refined sequence,
   we repeatedly move a small amount of probability mass from a coordinate
   that contributes most to the current peak towards a coordinate that
   contributes the least.  Each move is kept only if it improves the
   objective.

The final best sequence (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """Symmetric power‑law 1/(i·(n‑i))^p (invert=True) or (i·(n‑i))^p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) distribution on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on the index set."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """Convex combination (1‑λ)·arcsine + λ·uniform."""
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a using FFT."""
    return float(_fft_self_convolve(a).max())


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) / (sum a)^2.
    The vector is projected onto the simplex before evaluation,
    therefore denominator equals 1.
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _additive_refine(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected additive gradient descent that reduces the peak of the
    self‑convolution.  Uses the sub‑gradient of the current maximal
    entry.  Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution and its peak
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] additive refine it={it} score={best_score:.6f}")

        # sub‑gradient of the maximal entry
        k = int(np.argmax(b))
        grad = _subgradient_max(a, k)

        # diminishing step‑size
        eta = 0.8 / np.sqrt(it)

        # additive update, followed by simplex projection
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

    return best_seq, best_score


# ----------------------------------------------------------------------
def _targeted_refinement(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Smart hill‑climbing that always moves mass from a coordinate that
    contributes most to the current peak towards a coordinate that
    contributes the least.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        k = int(np.argmax(b))

        # gradient of the peak entry: 2·a_{k‑i}
        grad = _subgradient_max(a, k)

        src = int(np.argmax(grad))                     # biggest contributor
        # pick a destination that is not the source and has smallest contribution
        mask = np.arange(n) != src
        if not np.any(mask):
            continue
        dst = int(np.argmin(grad[mask]))
        # undo the masking offset
        dst = np.arange(n)[mask][dst]

        eps = a[src] * np.random.rand() * 0.1          # move ≤10 % of source mass
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[src] -= eps
        a_candidate[dst] += eps
        if a_candidate[src] < 0.0:
            continue

        candidate_score = _objective(a_candidate)
        if candidate_score < best_score:
            best_score = candidate_score
            best_seq = a_candidate.copy()
            a = a_candidate
            if it % 5000 == 0:
                print(f"[{time.time():.2f}] targeted improve it={it} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The search respects the given time
    budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    end_time = time.time() + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Determine a sensible set of lengths (at most 20 000)
    try:
        base_n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    length_factors = (0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    })
    # hand‑picked values
    lengths.extend([5000, 8000, 12000, 16000, 20000])
    lengths = sorted({l for l in lengths if l <= 20000})

    # ------------------------------------------------------------------
    # Global bookkeeping
    best_seq = None
    best_score = float("inf")
    best_candidates = []             # (score, sequence) tuples, max size ≈12

    # seed with a previously known solution if present
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        seq0 = np.maximum(seq0, 0.0)
        seq0 = _project_onto_simplex(seq0, s=1.0)
        sc0 = _objective(seq0)
        print(f"Previous solution score={sc0:.6f}")
        best_seq = seq0.copy()
        best_score = sc0
        best_candidates.append((sc0, seq0))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Stage 1 – coarse parametric sweep
    search_deadline = end_time - 300.0       # reserve ~300 s for refinement phases
    p_grid = np.linspace(0.35, 0.75, 31)      # power‑law exponent grid
    a_grid = np.linspace(0.2, 1.8, 25)       # beta‑parameter grid
    lam_grid = np.linspace(0.0, 0.5, 21)     # mixing weight grid
    sigma_grid_factor = np.linspace(1 / 30, 0.5, 12)  # relative sigma

    for n in lengths:
        if time.time() >= search_deadline:
            break

        # ---- arcsine (the known good shape) ----
        seq = _arcsine_distribution(n)
        seq = _project_onto_simplex(seq, s=1.0)
        sc = _objective(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"New global best (arcsine) n={n} score={best_score:.6f}")
        best_candidates.append((sc, seq))

        # ---- power‑law (inverse) ----
        for p in p_grid:
            if time.time() >= search_deadline:
                break
            seq = _power_law_distribution(n, p, invert=True)
            seq = _project_onto_simplex(seq, s=1.0)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"New global best (pow‑inv) n={n} p={p:.3f} score={best_score:.6f}")
            best_candidates.append((sc, seq))

        # ---- power‑law (direct) ----
        for p in p_grid:
            if time.time() >= search_deadline:
                break
            seq = _power_law_distribution(n, p, invert=False)
            seq = _project_onto_simplex(seq, s=1.0)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"New global best (pow‑dir) n={n} p={p:.3f} score={best_score:.6f}")
            best_candidates.append((sc, seq))

        # ---- beta family ----
        for a in a_grid:
            if time.time() >= search_deadline:
                break
            seq = _beta_distribution(n, a)
            seq = _project_onto_simplex(seq, s=1.0)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"New global best (beta) n={n} a={a:.3f} score={best_score:.6f}")
            best_candidates.append((sc, seq))

        # ---- Gaussian ----
        for rel in sigma_grid_factor:
            if time.time() >= search_deadline:
                break
            sigma = max(1.0, rel * n)
            seq = _gaussian_distribution(n, sigma)
            seq = _project_onto_simplex(seq, s=1.0)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"New global best (gauss) n={n} sigma={sigma:.1f} score={best_score:.6f}")
            best_candidates.append((sc, seq))

        # ---- mixture arcsine / uniform ----
        for lam in lam_grid:
            if time.time() >= search_deadline:
                break
            seq = _mixture_arcsine_uniform(n, lam)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"New global best (mix) n={n} lam={lam:.3f} score={best_score:.6f}")
            best_candidates.append((sc, seq))

        # keep only a small pool of the best candidates
        best_candidates.sort(key=lambda x: x[0])
        best_candidates = best_candidates[:12]

    # ------------------------------------------------------------------
    # Stage 2 – additive projected gradient refinement
    # take the top few candidates from the pool
    top_candidates = [c[1] for c in sorted(best_candidates, key=lambda x: x[0])[:4]]

    for cand in top_candidates:
        if time.time() >= end_time - 120.0:   # leave ~2 min for final hill‑climb
            break
        n = cand.shape[0]
        # allocate a slice of the remaining budget (max 60 s per run)
        remaining = end_time - time.time()
        slice_deadline = time.time() + min(60.0, remaining * 0.5)
        refined, refined_score = _additive_refine(cand, slice_deadline)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined.copy()
            print(f"Refinement improved to {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Stage 3 – targeted stochastic hill‑climbing
    if best_seq is not None and time.time() < end_time - 5.0:
        n = best_seq.shape[0]
        refined, refined_score = _targeted_refinement(best_seq, end_time - 5.0)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined.copy()
            print(f"After targeted refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in case nothing was produced
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
