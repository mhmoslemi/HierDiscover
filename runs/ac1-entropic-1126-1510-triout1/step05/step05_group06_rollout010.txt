def optimize_sequence_adam(a0, n, deadline, max_iter=100000):
    a = a0.copy()
    best_seq = a.copy()
    best_score = internal_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0
    alpha = 10.0
    alpha_mul = 1.01
    alpha_max = 200.0

    size_fft = 1 << ((3*n - 2).bit_length())
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = fft_self_convolve(a)
        cur_max = conv.max()
        cur_score_int = 2.0 * n * cur_max
        if cur_score_int < best_score - 1e-12:
            best_score = cur_score_int
            best_seq = a.copy()
        # compute gradient
        if alpha < alpha_max:
            grad = gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = subgradient_max(a, k)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.5 * m_hat / (np.sqrt(v_hat) + epsilon)  # 0.5 chosen for step size scaling
        # exponentiated
        a = a * np.exp(-step)
        # Renorm
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        # occasional jitter
        if t % 5000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()
    return best_seq, best_score

"""
Hybrid random‑shape search + Adam‑mirrored gradient + stochastic hill‑climbing.

1. Choose a moderate‑to‑large length (≤20 000) and generate many
   candidate step‑functions from flexible families:
        – Beta‑distribution discretisation (α, β > 0)
        – Asymmetric power‑law a_i ∝ (i+½)^(-p_l)·(n-i‑½)^(-p_r)
        – Uniform, arcsine and a few random vectors.
   Each candidate is normalised to sum = 1 and scored by the cheap
   surrogate 2·n·max(conv) (the true objective for sum = 1).

2. Keep the top‑k candidates (k≈10) and polish each of them with a
   projected Adam optimiser on the simplex.  Early iterations use a
   soft‑max surrogate for the ℓ∞‑norm (temperature α growing up to a
   large value) and later switch to the exact sub‑gradient of the max
   entry.  After every update the vector is projected back onto the
   probability simplex.

3. The best vector found overall is finally refined by a very cheap
   stochastic hill‑climber that moves a small amount of mass between
   two random indices whenever the true evaluation improves.

The routine respects the given time budget (default 1000 s) and prints
progress information.  The returned list is a valid step‑function
(sequence of non‑negative heights) for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) convolution of *a* with itself via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()        # next power‑of‑2
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max entry of a*a at index *k*."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the smooth surrogate (1/α)·log Σ exp(α·conv_i)
    w.r.t. a (soft‑max approximation of the ℓ∞‑norm).
    """
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_full = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv_full[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on n points, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    s = vals.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / s


# ----------------------------------------------------------------------
def _asymmetric_power_shape(n: int, pl: float, pr: float) -> np.ndarray:
    """Distribution a_i ∝ (i+½)^(-pl)·(n‑i‑½)^(-pr), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-pl) * (n - i - 0.5) ** (-pr)
    s = vals.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / s


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create a normalised (sum = 1) vector of length *n*."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "arcsine":
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a /= a.sum()
    elif kind == "beta":
        # symmetric Beta(0.5,0.5) = arcsine, kept for completeness
        a = _beta_shape(n, 0.5, 0.5)
    elif kind == "random":
        a = np.random.rand(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    else:  # fallback
        a = np.full(n, 1.0 / n, dtype=np.float64)

    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _internal_score(seq: np.ndarray) -> float:
    """Cheap surrogate: 2·n·max(conv) (exact when sum(seq)=1)."""
    n = seq.shape[0]
    conv = _fft_self_convolve(seq)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam / exponentiated‑gradient optimiser on the simplex.
    Returns the best vector seen and its internal (2·n·max) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _internal_score(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 0.1                 # base learning‑rate factor

    # Soft‑max temperature schedule
    alpha = 5.0                # start with a fairly smooth surrogate
    alpha_mul = 1.02
    alpha_max = 300.0

    size_fft = 1 << ((3 * n - 2).bit_length())
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()

        # ---------- gradient ----------
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # ---------- Adam’s moment estimates ----------
        m = beta1 * (m if t > 1 else np.zeros_like(a)) + (1.0 - beta1) * grad
        v = beta2 * (v if t > 1 else np.zeros_like(a)) + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # ---------- exponentiated update ----------
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # ---------- simplex projection ----------
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # occasional tiny random jitter to escape plateaus
        if t % 5000 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Very light stochastic hill‑climbing: move a small fraction of mass
    from one index to another whenever the *true* evaluation improves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _internal_score(a)

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer at most 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.10
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur_score = _internal_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Main entry point required by the evaluation harness.

    Searches for a step‑function with a low ``evaluate_sequence`` score.
    Returns the best list of non‑negative heights found within the allotted
    time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0       # safety margin

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction (height_sequence_1)
    # exists we start from roughly double its size, otherwise we use a
    # reasonably large default.
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000
    n = min(20000, max(5000, int(base_n * 2.0)))

    print(f"[{time.time():.2f}] Using length n = {n}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap random‑shape exploration (≈15 % of total time)
    phase_end = time.time() + max(5.0, (deadline - time.time()) * 0.15)
    top_k = 10
    candidate_tuples: list[tuple[float, np.ndarray]] = []

    def _store_candidate(seq: np.ndarray) -> None:
        score = _internal_score(seq)
        if len(candidate_tuples) < top_k:
            candidate_tuples.append((score, seq))
        else:
            # replace the worst if the new one is better
            worst_idx = max(range(len(candidate_tuples)),
                           key=lambda i: candidate_tuples[i][0])
            if score < candidate_tuples[worst_idx][0]:
                candidate_tuples[worst_idx] = (score, seq)

    # a few deterministic seeds
    for kind in ("uniform", "arcsine", "beta", "random"):
        seq = _initial_sequence(n, kind)
        _store_candidate(seq)

    # random families until time runs out
    while time.time() < phase_end:
        if np.random.rand() < 0.55:
            # Beta family
            log_alpha = np.random.uniform(np.log(0.05), np.log(5.0))
            log_beta = np.random.uniform(np.log(0.05), np.log(5.0))
            seq = _beta_shape(n, np.exp(log_alpha), np.exp(log_beta))
        else:
            # Asymmetric power‑law
            pl = np.random.uniform(0.05, 2.0)
            pr = np.random.uniform(0.05, 2.0)
            seq = _asymmetric_power_shape(n, pl, pr)
        _store_candidate(seq)

    candidate_tuples.sort(key=lambda x: x[0])
    print(f"[{time.time():.2f}] Phase 1 complete – best cheap score {candidate_tuples[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation of the best few candidates
    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503          # stop early if we reach the target

    for idx, (cheap_score, a0) in enumerate(candidate_tuples):
        # allocate a fair share of the remaining budget
        remaining = deadline - time.time()
        if remaining <= 5.0:
            break
        share = max(5.0, remaining / (len(candidate_tuples) - idx))
        cand_deadline = time.time() + share

        seq_opt, internal_score = _adam_optimize(a0, n, cand_deadline)

        # evaluate with the genuine black‑box function
        try:
            final_score = evaluate_sequence(seq_opt.tolist())
        except Exception:
            final_score = internal_score   # fallback to internal estimate

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] *** New global best → len={n} "
                  f"score={overall_best_score:.6f}")

        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic refinement on the global best
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        seq_refined, _ = _local_refinement(overall_best_seq, n, deadline)
        try:
            final_score = evaluate_sequence(seq_refined.tolist())
        except Exception:
            final_score = overall_best_score
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq_refined.copy()
            print(f"[{time.time():.2f}] *** After local refinement score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fall‑back
    if overall_best_seq is None:
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("[fallback] Using uniform vector.")

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
