def solve_lp_for_g(f, M, S):
    n = len(f)
    # Build sparse constraint matrix A shape (2n-1, n)
    rows = []
    cols = []
    data = []
    for i in range(n):
        f_i = f[i]
        if f_i == 0:
            continue
        # For k from i to i + n - 1, A[k, i] = f_i
        row_start = i
        row_end = i + n  # exclusive
        rows.extend(range(row_start, row_end))
        cols.extend([i] * n)
        data.extend([f_i] * n)
    # build csr matrix
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # objective -sum(g)
    c = -np.ones(n)
    # bounds: (0, None)
    bounds = [(0, None)] * n
    # solve
    res = opt.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    return g

def gradient_max_convolution(a, w):
    # a: length n, w: length 2n-1
    # compute convolution of w with a reversed, then slice appropriate n entries
    n = len(a)
    # compute convolution length = len(w) + n - 1 = 3n-2
    n_fft = 1 << ((len(w)+n-1).bit_length())
    W = np.fft.rfft(w, n_fft)
    A_rev = np.fft.rfft(a[::-1], n_fft)
    conv_full = np.fft.irfft(W * A_rev, n_fft)[:len(w)+n-1]
    # slice at indices (n-1) to (n-1)+(n)
    g = 2.0 * conv_full[n-1:n-1+n]
    return g

def _compute_smooth_obj_and_grad(y, mu, n, S):
    p = _softmax(y)   # shape (n,)
    a = S * p
    # compute conv = a*a via FFT
    # get length = 2n-1
    conv = _conv_fft(a)  # returns array length 2n-1
    max_c = conv.max()
    if mu > 0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700, 700)
        exps = np.exp(diff)
        Z = exps.sum()
        w = exps / Z
        fval = max_c + mu * np.log(Z)
    else:
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        fval = max_c

    # compute gradient w.r.t a
    g = _gradient_wrt_a(w, a)  # shape (n,)
    # chain rule to y
    inner = np.dot(g, p)
    grad_y = S * p * (g - inner)
    return float(fval), grad_y.astype(np.float64)

def _conv_fft(a):
    n = a.shape[0]
    n_fft = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n_fft)
    conv = np.fft.irfft(A * A, n_fft)[:2*n-1]
    return conv

def _gradient_wrt_a(w, a):
    # w shape (2n-1); a shape (n,)
    n = a.shape[0]
    len_w = w.shape[0]
    # conv of w with reversed a
    n_fft = 1 << ((len_w + n - 1).bit_length())
    W = np.fft.rfft(w, n_fft)
    A_rev = np.fft.rfft(a[::-1], n_fft)
    full = np.fft.irfft(W * A_rev, n_fft)[:len_w + n - 1]
    # slice segment [n-1 : n-1+n]
    g = 2.0 * full[n-1:n-1+n]
    return g

    deadline = time.time() + budget_s - 0.5

    try:
        ref_len = len(height_sequence_1)
        n = min(max(ref_len * 2, 4000), 9000)
        have_ref = True
    except Exception:
        n = 6000
        have_ref = False

    S = math.sqrt(2.0 * n)

    best_a = np.zeros(n)
    best_score = float('inf')

    def maybe_update(a_vec):
        nonlocal best_a, best_score
        sc = compute_score(a_vec)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] score={best_score:.6f}")

    init_ys = []

    init_ys.append(np.zeros(n))

    alpha = rng.uniform(0.001, 0.005)
    init_ys.append(-alpha * np.arange(n))

    init_ys.append(-alpha * np.arange(n-1, -1, -1))

    dir_alpha = rng.uniform(0.2, 0.6)

"""
Propose a low‑score step function.

The routine works in four stages:

1. **Fast smooth‑max optimisation** – the true objective (max of the
   auto‑convolution) is approximated by a smooth surrogate
   f_μ = max + μ·log‑sum‑exp.  A continuation on the temperature μ
   together with L‑BFGS‑B (gradient computed via FFT) drives the
   sequence towards low max‑convolution values.  Several diverse
   initial points are tried.

2. **Projected sub‑gradient descent** on the hard max objective.
   The exact sub‑gradient for max(conv) is used, with a back‑tracking
   line‑search and simplex projection.

3. **Large‑scale random block perturbations** – cheap block‑wise
   random candidates are generated, projected onto the ℓ¹‑simplex
   and kept if they improve the score.

4. **Simulated‑annealing hill climbing** – small mass‑transfer moves
   are repeatedly tried, accepting downhill moves always and uphill
   moves with a temperature schedule.  The temperature decays rapidly
   so that the search focuses on exploitation near the end.

All candidates are scored with a fast FFT‑based convolution evaluator,
which matches the behaviour of ``evaluate_sequence`` (the final output
will be re‑evaluated by the grader).  The algorithm respects the
user‑supplied time budget (default 1000 s) and prints progress
messages.  No closures or lambdas are used.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution a * a using FFT."""
    n = a.shape[0]
    n_fft = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n_fft)
    conv = np.fft.irfft(A * A, n_fft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _gradient_wrt_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of 〈w, a*a〉 w.r.t. a, i.e. g = 2·(w⋆a_rev)[n‑1:n‑1+n].

    Parameters
    ----------
    w : ndarray, shape (2·n‑1,)
        Weight vector (either uniform over max‑indices or smooth‑max weights).
    a : ndarray, shape (n,)
        Current sequence of heights.

    Returns
    -------
    g : ndarray, shape (n,)
    """
    n = a.shape[0]
    n_fft = _next_pow2(w.shape[0] + n - 1)
    W = np.fft.rfft(w, n_fft)
    A_rev = np.fft.rfft(a[::-1], n_fft)
    full = np.fft.irfft(W * A_rev, n_fft)[: w.shape[0] + n - 1]
    # slice the central n entries
    g = 2.0 * full[n - 1 : n - 1 + n]
    return g


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(
    y: np.ndarray, mu: float, n: int, S: float
) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate:
        f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Parameters
    ----------
    y : ndarray, shape (n,)
        Unconstrained parameters (log‑odds of the simplex).
    mu : float
        Smoothing temperature (>0).  μ = 0 corresponds to the hard max.
    n : int
        Length of the step function.
    S : float
        Target ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : ndarray, shape (n,)
        Gradient ∂f/∂y.
    """
    # probabilities and actual heights (Σa = S)
    p = _softmax(y)
    a = S * p

    # convolution a*a
    conv = _conv_fft(a)
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)  # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a
    g_a = _gradient_wrt_a(w, a)

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)  # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable returning (value, gradient) for scipy.optimize."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _score_via_fft(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence (ignoring the built‑in clipping).
    Returns 2·n·max(conv)/(sum(a)²).  Assumes a ≥ 0 and sum > 0.01.
    """
    a = np.asarray(a, dtype=float)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)           # safeguard, matches evaluate_sequence
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _conv_fft(a)
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (sum_a * sum_a))

    try:

        sc = _score_via_fft(a_vec)

            best_a = a_vec.copy()

    init_ys = []

    alpha = rng.uniform(0.001, 0.004) * (n / 2000.0)
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    p_uni = rng.dirichlet(np.ones(n))

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:

                    obj,
                    y,
                    method="L-BFGS-B",

            _maybe_update(a)

    if time.time() < deadline and best_score < float("inf"):

                    a = a_new
                    improved = True
                    print(f"[subgrad] iter {it:5d} step {step:.6f} → {best_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # occasional random jitter to escape flats
                i, j = rng.choice(n, size=2, replace=False)
                delta = rng.random() * a[i] * 0.05
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, S)
            it += 1

    # ------------------------------------------------------------------
    # Random block perturbations (cheap large‑scale search) ----------
    if time.time() < deadline and best_score < float("inf"):
        remaining = deadline - time.time()
        max_trials = int(min(25000, remaining * 180))
        print(f"[block] {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(3, 11)                     # 3–10 blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # ------------------------------------------------------------------
    # Simulated‑annealing hill climbing ---------------------------------
    if time.time() < deadline and best_score < float("inf"):
        a = best_a.copy()
        T = best_score * 0.12          # starting temperature
        decay = 0.9995
        iter_sa = 0
        while time.time() < deadline:
            # propose a small transfer of mass between two coordinates
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(a[i], S * 0.03)
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, S)

            sc = _score_via_fft(cand)
            dE = sc - best_score
            if dE < 0 or rng.random() < math.exp(-dE / T):
                a = cand
                if sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    print(f"[SA] iter {iter_sa:6d} T={T:.5e} → {best_score:.6f}")
            T = max(T * decay, 1e-12)
            iter_sa += 1

    # ------------------------------------------------------------------
    # Monotone‑sorting heuristics (sometimes helpful) ------------------
    if time.time() < deadline and best_score < float("inf"):
        sorted_desc = np.sort(best_a)[::-1]
        sorted_desc = _project_to_simplex(sorted_desc, S)
        _maybe_update(sorted_desc)

        sorted_asc = np.sort(best_a)
        sorted_asc = _project_to_simplex(sorted_asc, S)
        _maybe_update(sorted_asc)

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
