    # Make sure at least one constraint.
    if not active:

    # iterative addition of violated constraints
    while True:
        # Build A_ub and b_ub
        A = build_constraint_matrix(f, active)
        b = np.full(len(active), M, dtype=float)
        c = -np.ones(n)  # maximize sum(g)
        # Solve LP
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs', options={'presolve': True})

            print("[LP] sum_g zero, abort")
            break

    new_sum = new_f.sum()

        # project onto simplex to correct sums
        new_f = project_to_simplex(new_f, S)

def build_constraint_matrix(f, active):
    n = len(f)
    data = []
    rows = []
    cols = []
    for row_idx, k in enumerate(active):
        # Determine range of valid j
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # j indices: {j_start,...,j_end}
        # corresponding f indices: f_idx = k - j
        # So f values = f[k - np.arange(j_start, j_end+1)]
        js = np.arange(j_start, j_end + 1)
        coeffs = f[k - js]
        data.extend(coeffs.tolist())
        rows.extend([row_idx] * len(js))
        cols.extend(js.tolist())
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n), dtype=float)
    return A

"""
Propose a low‑score step function.

Algorithm (≈ 100 s total):
  1. Initialise several diverse ℓ¹‑simplex vectors of length *n*
     (uniform, Dirichlet‑sparse, random Gaussian, optional reference).
  2. For each start run a small “alternating LP” loop:
        * Compute M = max(conv(f,f)).
        * Build the active set of convolution indices where the
          current max is attained.
        * Solve a linear program  maximise Σ g_j  s.t.
                conv(f,g) ≤ M   (only on the active set)
                g ≥ 0.
          If the solution violates any other shift, add the violating
          shift to the active set and resolve (cut‑plane method).
        * Scale the LP solution to the fixed ℓ¹‑norm  S = √(2n)
          and update f ← (1‑t)·f + t·g_scaled  (t≈0.5, then slowly shrunk).
        * Keep the best sequence seen.
  3. Refine the best candidate with a few sub‑gradient descent steps
     (exact max‑convolution).  Afterwards perform cheap random‑block
     perturbations and a monotone‑sorting heuristic.
  4. Return the best sequence found before the time budget expires.

The core of the improvement over the previous code is the *exact* LP
step with an active‑set (cut‑plane) strategy, which reduces the
maximum auto‑convolution more aggressively than the smooth‑max /
sub‑gradient stages alone.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}, z ≥ 0.
    """
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray, active: list[int]) -> sp.csr_matrix:
    """
    Build a sparse matrix A where each row corresponds to a convolution
    inequality  (f * g)[k] ≤ M for k in *active*.
    Row k contains the coefficients f[k‑j] for the variable g_j.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row_id, k in enumerate(active):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1, dtype=int)
        coeffs = f[k - js]          # f indexed as f[k‑j]
        rows.extend([row_id] * len(js))
        cols.extend(js.tolist())
        data.extend(coeffs.tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(active), n), dtype=float)
    return A


# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, S: float, t: float, deadline: float) -> np.ndarray:
    """
    One alternating‑LP step starting from vector f (already on the simplex).
    Returns an updated vector (still on the simplex) or the original f if
    the LP fails.
    """
    n = f.shape[0]
    # current bound
    conv_self = np.convolve(f, f)
    M = conv_self.max()

    # initial active set – indices where the bound is (almost) tight
    eps_active = max(1e-12, M * 1e-10)
    active = [int(k) for k, v in enumerate(conv_self) if M - v <= eps_active]

    # cut‑plane loop
    while True:
        if time.time() > deadline:
            return f
        A = _build_constraint_matrix(f, active)
        b = np.full(len(active), M, dtype=float)

        # maximise Σ g_j  → minimise -Σ g_j
        c = -np.ones(n, dtype=float)

        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # LP failed (infeasible / unbounded)
            print("[LP] solver failure:", res.message)
            return f

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print("[LP] got zero‑sum solution – aborting step.")
            return f

        # scale to keep the ℓ¹‑norm equal to S
        scale = S / sum_g
        g_scaled = g * scale

        # check for any previously unseen violations
        conv_fg = np.convolve(f, g_scaled)
        viol = [int(k) for k, v in enumerate(conv_fg) if v > M + 1e-12]

        if not viol:
            # feasible – stop cutting
            break

        # add new violated indices to the active set
        for v in viol:
            if v not in active:
                active.append(v)

        # safeguard – we do not want an explosion of constraints
        if len(active) > 5 * n:
            print("[LP] active set grew too large – stop cutting.")
            break

    # convex combination (step size t) and re‑projection
    new_f = (1.0 - t) * f + t * g_scaled
    new_f = project_to_simplex(new_f, S)
    return new_f


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluate_sequence() score.  The routine respects the given time budget
    (seconds) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        ref_len = len(height_sequence_1)  # type: ignore[name-defined]
        have_ref = True
    except Exception:
        ref_len = 5000
        have_ref = False

    n = min(max(ref_len, 4000), 8000)  # stay comfortably below 10 k
    S = math.sqrt(2.0 * n)            # fixed ℓ¹‑norm; objective = max(conv)

    # --------------------------------------------------------------
    # generate several diverse starting points (all already on the simplex)
    starts = []

    # uniform
    starts.append(np.full(n, S / n, dtype=float))

    # sparse Dirichlet (α < 1 gives spiky vectors)
    for _ in range(3):
        a = rng.uniform(0.2, 0.6)
        probs = rng.dirichlet([a] * n)
        starts.append(S * probs)

    # absolute Gaussian – then project
    gauss = np.abs(rng.standard_normal(n))
    if gauss.sum() == 0:
        gauss[0] = 1.0
    starts.append(S * gauss / gauss.sum())

    # reference, if available
    if have_ref:
        ref = np.asarray(height_sequence_1, dtype=float)
        ref = project_to_simplex(ref, S)
        starts.append(ref)

    best_f = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # main loop over restarts
    for rid, f0 in enumerate(starts):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{len(starts)}] LP‑phase start")
        f_cur = project_to_simplex(f0, S)

        # evaluate the start
        sc0 = evaluate_sequence(f_cur.tolist())
        if sc0 < best_score:
            best_score = sc0
            best_f = f_cur.copy()
            print(f"[improve] start score = {best_score:.6f}")

        t_step = 0.5
        # a modest number of alternating‑LP iterations
        for lp_it in range(20):
            if time.time() > deadline:
                break
            # --- one LP step -------------------------------------------------
            f_next = _lp_step(f_cur, S, t_step, deadline)
            if np.allclose(f_next, f_cur):
                # no progress – shrink step size and continue
                t_step = max(t_step * 0.7, 0.05)
                continue

            # evaluate
            sc = evaluate_sequence(f_next.tolist())
            if sc < best_score:
                best_score = sc
                best_f = f_next.copy()
                print(f"[improve] after LP iter {lp_it} → {best_score:.6f}")

            # keep the better of the two as the basis for the next iteration
            conv_cur = np.convolve(f_cur, f_cur).max()
            conv_next = np.convolve(f_next, f_next).max()
            f_cur = f_next if conv_next < conv_cur else f_cur

            # slowly decay the mixing factor
            t_step = max(t_step * 0.95, 0.02)

            # early stop if we cannot improve the bound any more
            if abs(conv_next - conv_cur) < 1e-8:
                break

        # --------------------------------------------------------------
        # sub‑gradient refinement on the best vector from this restart
        if time.time() < deadline and best_f is not None:
            a = f_cur.copy()
            step0 = S * 0.4 / n
            it = 0
            while time.time() < deadline and it < 2000:
                conv = np.convolve(a, a)
                max_c = conv.max()
                mask = conv == max_c
                cnt = mask.sum()
                if cnt == 0:
                    break
                w = np.zeros_like(conv)
                w[mask] = 1.0 / cnt
                conv_wr = np.convolve(w, a[::-1])
                g_sub = 2.0 * conv_wr[n - 1:n - 1 + n]

                step = step0 / (1.0 + 1e-5 * it)
                improved = False
                for _ in range(12):
                    a_new = a - step * g_sub
                    a_new = project_to_simplex(a_new, S)
                    sc2 = evaluate_sequence(a_new.tolist())
                    if sc2 < best_score - 1e-12:
                        best_score = sc2
                        best_f = a_new.copy()
                        a = a_new
                        improved = True
                        print(f"[subgrad] restart {rid + 1} it {it:4d} step {step:.6f} → {best_score:.6f}")
                        break
                    step *= 0.5
                if not improved:
                    # small random jitter to escape flat regions
                    i, j = rng.choice(n, size=2, replace=False)
                    delta = rng.random() * a[i] * 0.03
                    a[i] -= delta
                    a[j] += delta
                    a = project_to_simplex(a, S)
                it += 1

    # --------------------------------------------------------------
    # final cheap large‑scale random block perturbations
    if best_f is not None and time.time() < deadline:
        remaining = deadline - time.time()
        max_trials = int(min(5000, remaining * 120))
        print(f"[block] {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(3, 16)
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = project_to_simplex(cand, S)
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_score = sc
                best_f = cand.copy()
                print(f"[block] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # monotone‑sorting heuristic (often helpful)
    if best_f is not None and time.time() < deadline:
        sorted_desc = np.sort(best_f)[::-1]
        sorted_desc = project_to_simplex(sorted_desc, S)
        sc = evaluate_sequence(sorted_desc.tolist())
        if sc < best_score:
            best_score = sc
            best_f = sorted_desc.copy()
            print(f"[sort‑desc] new best {best_score:.6f}")

        sorted_asc = np.sort(best_f)
        sorted_asc = project_to_simplex(sorted_asc, S)
        sc = evaluate_sequence(sorted_asc.tolist())
        if sc < best_score:
            best_score = sc
            best_f = sorted_asc.copy()
            print(f"[sort‑asc] new best {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_f.tolist() if best_f is not None else []
