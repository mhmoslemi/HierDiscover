"""
Search for a non‑negative step function that minimizes
    2·n·max(conv(a,a)) / (sum(a)^2)
(where conv is the full self‑convolution).  The algorithm works in four
phases:

1. Choose a length n (preferably the length of a previously found
   solution, bounded to a few‑ten‑thousands).
2. Generate a modest set of hand‑crafted probability vectors:
   * symmetric Beta(α,α) for a fine grid of α (including values < 0.5);
   * the discrete arcsine law (β=½);
   * simple triangular and uniform shapes;
   * mixtures of the above with small endpoint spikes;
   * a few random Dirichlet draws and the previous best if available.
   All candidates are projected onto the probability simplex and
   evaluated with a fast FFT‑based routine; the best few are retained.
3. From each of the best candidates run a short projected‑Adam optimiser
   on the simplex.  The gradient is obtained from a soft‑max approximation
   of the max‑convolution that gradually turns into the exact sub‑gradient
   (α is increased exponentially up to 1e6).  Adam’s internal state is
   projected back onto the simplex after each step.
4. A cheap stochastic hill‑climbing (mass swaps + occasional jitter)
   finishes the optimisation.

The whole search respects the 1000 s time budget and returns a plain list
of floats which evaluate to a low score with ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·b_k) w.r.t. a.
    For α large this becomes a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # ---- soft‑max weights -------------------------------------------------
    if alpha > 2_000:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # ---- convolution w * reversed(a) ------------------------------------
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]      # factor 2 from derivative of a*a
    return g


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) pdf evaluated on mid‑points of n bins."""
    a = max(a, 1e-8)          # avoid a==0 which would give infinities
    x = (np.arange(n) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    s = pdf.sum()
    if s == 0.0 or not np.isfinite(s):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= s
    return pdf


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangular distribution on n points."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - mid) / (mid + 1.0)
    vals[vals < 0] = 0.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact evaluate_sequence objective (avoids the security wrapper)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:   # mirrors the safety guard in evaluate_sequence
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest list of handcrafted probability vectors."""
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) triangular
    candidates.append(_triangular_distribution(n))

    # 3) arcsine (Beta 0.5,0.5)
    candidates.append(_arcsine_distribution(n))

    # 4) symmetric Beta for a fine grid (including a<0.5)
    alphas = np.concatenate([
        np.geomspace(0.02, 0.48, 15),   # many values below 0.5
        np.linspace(0.5, 1.0, 11)      # up to uniform
    ])
    for a in alphas:
        candidates.append(_beta_distribution(n, a))

    # 5) mixtures: arcsine + small uniform
    arcsine = _arcsine_distribution(n)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for w in np.linspace(0.0, 0.4, 5):
        mix = (1.0 - w) * arcsine + w * uniform
        candidates.append(mix)

    # 6) add tiny spikes at the boundaries to a few base shapes
    for base in (arcsine, _triangular_distribution(n)):
        for spike_frac in (0.08, 0.15):
            mix = base.copy()
            mix[0] += spike_frac / 2.0
            mix[-1] += spike_frac / 2.0
            mix = _project_onto_simplex(mix, 1.0)
            candidates.append(mix)

    # 7) previous best if supplied (and same length)
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())

    # 8) a few random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(6):
        candidates.append(rng.dirichlet(np.ones(n)))

    # ensure simplex
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex.
    Returns the best vector seen and its raw objective value.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = _objective_value(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # temperature (soft‑max) schedule
    alpha = 2.0
    alpha_mul = 1.08
    alpha_max = 1e6

    size_fft = 1 << ((3 * n - 2).bit_length())  # ≥ (2n‑1)+(n)‑1

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()          # sum(a)=1, so raw score
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[Adam] it={it} n={n} score={best_score:.6f}")

        # gradient (soft‑max surrogate of the max)
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.15 / (np.sqrt(v_hat) + eps)   # a modest base learning rate
        a = a - step * m_hat
        a = _project_onto_simplex(a, 1.0)

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic local search: repeatedly move a random fraction of the mass
    between two positions and keep the move if it improves the objective.
    Occasional tiny jitter helps escaping plateaus.
    """
    best = start_a.copy()
    best_score = _objective_value(best)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-14:
            continue
        # move up to 30 % of the mass from i to j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _objective_value(cand)

        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # jitter every few thousand iterations
        if it % 4000 == 0:
            jitter = best + rng.normal(scale=5e-7, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = _objective_value(jitter)
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point required by the grader.
    Returns a list of non‑negative heights (probability vector) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    # safety margin for final conversion / printing
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Determine a sensible length.
    try:
        base_n = int(len(height_sequence_1))   # type: ignore[name-defined]
    except Exception:
        base_n = 9000

    n = max(4000, min(base_n, 15000))

    # ------------------------------------------------------------------
    # Phase 0 – quick parametric sweep to get a few good seeds.
    prev_seq = None
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        pass

    shape_deadline = time.time() + min(35.0, (deadline - time.time()) * 0.20)
    candidates = _candidate_shapes(n, prev_seq)

    best_seq = None
    best_val = float("inf")
    for cand in candidates:
        if time.time() >= shape_deadline:
            break
        val = _objective_value(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[shape] new best {best_val:.6f} (n={n})")

    if best_seq is None:                         # fallback (should never happen)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective_value(best_seq)

    # ------------------------------------------------------------------
    # Phase 1 – short Adam runs from the top few seeds.
    # keep the three best seeds for Adam
    top_k = 3
    # recompute scores to sort (already have best_val for one)
    scored = [(cand, _objective_value(cand)) for cand in candidates]
    scored.sort(key=lambda x: x[1])
    seeds = [c for c, _ in scored[:top_k]]

    remaining = deadline - time.time()
    adam_time = remaining * 0.55
    per_run_deadline = time.time() + adam_time / max(len(seeds), 1)

    for idx, start in enumerate(seeds):
        if time.time() >= deadline:
            break
        # add a tiny random jitter before each Adam run
        rng = np.random.default_rng(seed + idx + 123)
        jitter = start + rng.normal(scale=2e-2, size=n)
        jitter = _project_onto_simplex(jitter, 1.0)

        seq, score = _adam_optimize(jitter, n, per_run_deadline)
        if score < best_val:
            best_val = score
            best_seq = seq.copy()
            print(f"[Adam run {idx}] improved to {best_val:.6f}")

        # shrink deadline for the next run
        per_run_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.5)

    # ------------------------------------------------------------------
    # Phase 2 – final hill‑climbing polishing
    hill_deadline = deadline
    seq_final, final_score = _hill_climb(best_seq, n, hill_deadline)

    # ------------------------------------------------------------------
    # Safety fallback (should never fire)
    if math.isinf(final_score) or final_score > 1e9:
        seq_final = np.full(5000, 1.0 / 5000, dtype=np.float64)
        final_score = _objective_value(seq_final)

    print(f"Final best score: {final_score:.6f} (length={len(seq_final)})")
    return seq_final.tolist()
