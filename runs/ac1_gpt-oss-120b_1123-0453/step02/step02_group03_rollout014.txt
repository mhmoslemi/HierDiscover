"""
Propose a low‑value step‑function.

The search consists of three complementary stages that are executed for a
hand‑picked list of lengths (200 … 500).  For every length we run

1️⃣ **Linear‑programming refinement** – identical to the previous solution
   but with a larger iteration budget and a slower decay of the mixing
   weight.  This monotonically reduces the peak of the auto‑convolution.

2️⃣ **Gradient‑plus‑mass‑bump phase** – we repeatedly lower the current
   maximal convolution entry by moving mass away from the indices that
   contribute most to that entry (the exact gradient of the convolution)
   and we then inject a tiny amount of mass into the smallest entries.
   The latter raises the ℓ¹‑norm while only barely affecting the peak,
   which often improves the evaluation score.

3️⃣ **Simulated‑annealing fine‑tuner** – a classic SA that perturbs a
   single coordinate at a time.  The temperature schedule cools from 1.0
   down to 1e‑4 over the remaining time budget.

The best sequence seen across all lengths (and an extra final SA on that
sequence) is returned as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

def _score(seq: np.ndarray) -> float:
    """Evaluate a candidate sequence using the supplied scorer."""
    return float(evaluate_sequence(seq.tolist()))


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix A such that (A @ g)[k] == (f * g)[k].
    Each row k contains the coefficients f[i] for those i that pair with
    g[k‑i] in the convolution.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _lp_improve(
    f: np.ndarray, target_sum: float, mix: float, rng: np.random.Generator
) -> tuple[np.ndarray | None, float | None]:
    """
    Perform one LP‑based improvement step.

    Parameters
    ----------
    f : np.ndarray
        Current sequence (non‑negative, ℓ¹‑norm = ``target_sum`` is assumed).
    target_sum : float
        Desired ℓ¹‑norm after the rescaling of the LP solution.
    mix : float
        Convex‑combination weight 0 < mix ≤ 1.
    rng : np.random.Generator
        Random generator (only needed for a warm start in CVXPY).

    Returns
    -------
    (new_f, new_score) – the updated sequence and its score,
    or ``(None, None)`` if the LP failed.
    """
    M = float(np.max(np.convolve(f, f)))
    A = _conv_matrix(f)

    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None, None

    n = f.shape[0]
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:  # ECOS is fast for medium‑size dense LPs.
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=8000,
                   abstol=1e-6, reltol=1e-6)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000)

    if g.value is None:
        return None, None

    g0 = np.maximum(g.value, 0.0)
    sum_g = float(np.sum(g0))
    if sum_g <= 1e-12:
        return None, None

    # bring g to the prescribed ℓ¹‑norm
    g_rescaled = g0 * (target_sum / sum_g)

    # convex combination with the old vector
    f_new = (1.0 - mix) * f + mix * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    new_score = _score(f_new)
    return f_new, new_score


def _simulated_annealing(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple SA optimiser that perturbs a single coordinate at a time.
    The schedule cools from ``temp_start`` to ``temp_end`` over the
    remaining time budget.
    """
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        temp = temp_start * (temp_end / temp_start) ** frac

        # propose a change to one random index
        idx = rng.integers(len(cur_seq))
        delta = rng.normal(scale=0.5) * temp * 5.0
        cand = cur_seq.copy()
        cand[idx] = float(np.clip(cand[idx] + delta, 0.0, 1000.0))

        cand_score = _score(cand)

        # Metropolis acceptance rule
        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

        # occasional tiny random re‑shuffle to avoid stagnation
        if rng.random() < 0.001:
            pert = np.maximum(cur_seq + rng.normal(scale=0.1, size=len(cur_seq)),
                              0.0)
            cur_seq = np.clip(pert, 0.0, 1000.0)
            cur_score = _score(cur_seq)

    return best_seq, best_score


def _grad_for_shift(f: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of (f*f)[k] with respect to f (ignoring the constant factor 2):
    grad_i = f[k-i] for indices where 0 ≤ k-i < n, else 0.
    """
    n = f.shape[0]
    grad = np.zeros(n, dtype=np.float64)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    for i in range(i_start, i_end + 1):
        j = k - i
        grad[i] = f[j]
    return grad


def _mass_bump(seq: np.ndarray, eps: float, frac: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add a tiny amount `eps` to the smallest `frac` fraction of entries.
    The idea is to raise the ℓ¹‑norm while scarcely affecting the peak.
    """
    n = len(seq)
    n_target = max(1, int(frac * n))
    idx_small = np.argsort(seq)[:n_target]
    new_seq = seq.copy()
    new_seq[idx_small] += eps
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    return new_seq

        best_seq = np.ones(256, dtype=np.float64)

    return [float(x) for x in best_seq]
