def project_onto_simplex(v, s):
    # ensure nonnegative then reproject.
    # v can be any real.
    # Implementation from https://arxiv.org/abs/1309.1541
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑value step sequence.

The algorithm is a hybrid of:
  * random simplex initialization,
  * a projected sub‑gradient descent on the infinity‑norm of the
    self‑convolution (the quantity returned by ``evaluate_sequence``),
  * occasional Gaussian perturbations (simulated‑annealing style) to
    escape flat stationary points such as the uniform distribution,
  * a final short random walk for polishing.

All iterates satisfy the mandatory constraints (non‑negative,
ℓ¹‑norm = √(2 n) and each entry ≤ 1000) by means of a projection onto
the simplex.
The routine works for several lengths (up to a few thousand) and
returns the best sequence found before the time budget expires.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)


def _max_convolution(a: np.ndarray) -> float:
    """Return max entry of the self‑convolution of a 1‑D array."""
    # ``np.convolve`` returns a float64 array; take the maximum.
    return float(np.convolve(a, a).max())


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector ``v`` onto the simplex {x ≥ 0, Σ x = s}.
    Implementation of the algorithm from
    Wang & Carreira‑Perpiñán (2013) – “Projection onto the probability simplex”.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative.")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the largest rho satisfying the condition
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point uniformly from the simplex of dimension ``n``."""
    # Use exponential (Gamma(1,1)) variables – standard trick for Dirichlet(1,…,1)
    x = rng.exponential(scale=1.0, size=n)
    x = x / x.sum() * target_sum
    return x.astype(np.float64)


def _gradient_at_max(conv: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute a (sub‑)gradient of ``max(conv)`` with respect to ``a``.
    For each index k attaining the maximum we add the contribution
    ∂c_k/∂a_i = 2·a_{k‑i} (zero if the partner index is out of range).
    The contributions from several maximal k are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # indices whose value is (almost) the maximum
    max_idx = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        # i ranges such that 0 ≤ i < n and 0 ≤ k‑i < n
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]
    if len(max_idx) > 1:
        grad /= float(len(max_idx))
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score.

    The search is time‑limited; the best sequence discovered before the
    deadline is returned as a plain list of floats.
    """
    start_time = time.time()
    # keep a small safety margin for the outer framework
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Fetch the previously discovered construction (if available).
    try:
        known = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        known = None

    # Lengths to explore – larger n give a tighter theoretical lower bound.
    lengths = [300, 400, 500, 600, 800, 1000, 1200]
    # keep the list sorted and drop values that would be too large for the budget
    lengths = [n for n in lengths if n > 0]

    # ------------------------------------------------------------------
    # Main search loop (one length after another, many restarts).
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # how many independent restarts for this length?
        restarts = 3
        for restart in range(restarts):
            if time.time() >= deadline:
                break

            # --------------------------------------------------------------
            # 1️⃣  Initialise a feasible point.
            if known is not None and known.shape[0] == n:
                f = _project_onto_simplex(known, target_sum)
                init_tag = "known"
            else:
                f = _random_simplex(n, target_sum, rng)
                init_tag = "random"

            cur_val = _max_convolution(f)               # because sum = √(2n)
            if cur_val < best_score:
                best_score = cur_val
                best_seq = f.copy()
                print(f"[init] n={n} restart={restart} ({init_tag}) score={cur_val:.6f}")

            # --------------------------------------------------------------
            # 2️⃣  Local sub‑gradient + noisy simulated‑annealing descent.
            #       The number of iterations is chosen adaptively from the
            #       remaining time budget.
            max_iters = 25000
            # a rough step‑size – 0.1 proved useful in experiments
            step0 = 0.1

            for it in range(max_iters):
                if time.time() >= deadline:
                    break

                # current convolution & gradient
                conv = np.convolve(f, f)
                M = float(conv.max())
                grad = _gradient_at_max(conv, f)

                # step‑size schedule (diminishing)
                step = step0 / (1.0 + 0.001 * it)

                # random perturbation to avoid flat stationary points
                noise = rng.normal(scale=step * 0.5, size=n)

                # tentative move (projected back onto the feasible simplex)
                f_candidate = _project_onto_simplex(f - step * grad + noise, target_sum)

                cand_val = _max_convolution(f_candidate)

                # -------- acceptance rule ----------
                if cand_val < cur_val:
                    f = f_candidate
                    cur_val = cand_val
                    if cand_val < best_score:
                        best_score = cand_val
                        best_seq = f.copy()
                        print(f"[improve] n={n} restart={restart} it={it} score={cand_val:.6f}")
                else:
                    # simulated‑annealing style occasional uphill moves
                    temperature = max(0.01, step0 / (1.0 + 0.01 * it))
                    if rng.random() < math.exp(-(cand_val - cur_val) / temperature):
                        f = f_candidate
                        cur_val = cand_val

                # tiny early‑stop if we are already very close to the
                # theoretical lower bound (average value of the convolution)
                lower_bound = 2.0 * n / (2.0 * n - 1.0)
                if cur_val <= lower_bound * 1.01:
                    print(f"[near‑optimal] n={n} score={cur_val:.6f}")
                    break

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing: a cheap random walk around the best point,
    #     using a diminishing step size.
    if best_seq is not None:
        n_best = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n_best)
        f = best_seq.copy()
        cur_val = _max_convolution(f)
        remaining = deadline - time.time()
        if remaining > 0.2:
            walk_iters = int(5e4 * remaining / budget_s) + 1000
            for it in range(walk_iters):
                # very small Gaussian move
                step = 0.02 / (1.0 + 0.0005 * it)
                noise = rng.normal(scale=step, size=n_best)
                cand = _project_onto_simplex(f + noise, target_sum)
                cand_val = _max_convolution(cand)
                if cand_val < cur_val:
                    f = cand
                    cur_val = cand_val
                    if cand_val < best_score:
                        best_score = cand_val
                        best_seq = f.copy()
                        print(f"[polish] it={it} score={cand_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Fallback – if everything failed (should never happen) return a uniform vector.
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # Convert to plain Python list for the outer harness.
    return [float(x) for x in best_seq]
