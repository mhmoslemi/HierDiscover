Pseudo:

def fft_convolve(a,b):
    n = len(a) + len(b) -1
    N = 1<< (n-1).bit_length()  # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)
    return conv[:n]

"""
Gradient–annealing optimizer for the step‑function objective.

The code searches for a non‑negative sequence `a` (length n) with
ℓ¹‑norm √(2 n).  For such a sequence the scorer
`evaluate_sequence` returns exactly max_k (a*a)[k] (the worst value of the
auto‑convolution).  The optimizer:

* uses a smooth approximation of the max via a soft‑max (temperature β),
* computes the gradient of the smoothed objective analytically (it is a
  convolution of the current sequence with the soft‑max weights),
* performs a projected gradient step with a simple back‑tracking line search,
* raises β gradually (annealing) so that the search first moves freely
  and then focuses on the true maximum,
* works for any length up to a few thousand elements – the cost is
  dominated by a few FFT‐based convolutions per iteration, allowing many
  thousands of iterations within the 1000 s budget,
* optionally starts from the globally supplied `height_sequence_1`,
  and finally does a short random‑perturbation polishing phase.

The returned list satisfies all scorer checks and gives a lower bound
than the previous LP‑based method (target < 1.503).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast (real) convolution via FFT.
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of real vectors a and b using FFT (real‑to‑real)."""
    n = a.shape[0] + b.shape[0] - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:n]
    return conv


# ----------------------------------------------------------------------
# Helper: enforce non‑negativity and the ℓ¹‑norm √(2 n).
# ----------------------------------------------------------------------
def _project_to_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and renormalise to have ℓ¹‑norm = target_sum."""
    x = np.clip(x, 0.0, 1000.0)
    s = np.sum(x)
    if s > 0.0:
        x = x * (target_sum / s)
    else:                                   # degenerate – fall back to uniform
        x = np.full_like(x, target_sum / x.size)
    return x


# ----------------------------------------------------------------------
# Core optimisation routine for a single length n.
# ----------------------------------------------------------------------
def _optimize_length(
    n: int,
    rng: np.random.Generator,
    deadline: float,
    init_seq: np.ndarray | None,
) -> tuple[np.ndarray, float]:
    """
    Perform the annealed gradient descent for a fixed length n.

    Returns the best sequence discovered and its max‑auto‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    # ----- initialise ----------------------------------------------------
    if init_seq is not None and init_seq.shape[0] == n:
        f = _project_to_feasible(init_seq.astype(np.float64), target_sum)
    else:
        # smooth, bell‑shaped start (Gaussian)
        x = np.linspace(-1.0, 1.0, n)
        sigma = 0.5
        f = np.exp(-0.5 * (x / sigma) ** 2)
        f = _project_to_feasible(f, target_sum)

    # keep track of the best seen so far for this n
    S = _fft_convolve(f, f)
    best_val = float(np.max(S))
    best_f = f.copy()
    it_counter = 0

    # ----- annealing schedule -------------------------------------------
    beta_values = np.linspace(0.5, 20.0, 30)   # from smooth to near‑max
    step_factor = 0.5 * target_sum            # base step size (scaled later)

    while time.monotonic() < deadline:
        for beta in beta_values:
            if time.monotonic() >= deadline:
                break

            # inner loop at the current temperature
            for _ in range(200):               # a few hundred steps per β
                if time.monotonic() >= deadline:
                    break

                # current auto‑convolution and its maximum
                S = _fft_convolve(f, f)
                M = float(np.max(S))

                # soft‑max weights (stable – subtract the max)
                exp_term = np.exp(beta * (S - M))
                w = exp_term / np.sum(exp_term)   # a probability vector, length 2n‑1

                # gradient = 2 * (f * w_rev) evaluated at positions L‑1‑i
                w_rev = w[::-1]
                grad_full = _fft_convolve(f, w_rev)          # length 3n‑2
                L = len(w)                                    # = 2n‑1
                idx = (L - 1) - np.arange(n)                 # indices of interest
                grad = 2.0 * grad_full[idx]

                # step size – modestly decreasing with β
                step = step_factor / (1.0 + beta)

                # ----- simple back‑tracking line search -------------------
                f_candidate = None
                for _ in range(5):
                    cand = f - step * grad
                    cand = _project_to_feasible(cand, target_sum)

                    S_cand = _fft_convolve(cand, cand)
                    M_cand = float(np.max(S_cand))

                    if M_cand <= M:        # accept any non‑worsening move
                        f_candidate = cand
                        M = M_cand
                        break
                    step *= 0.5               # shrink and retry

                # if all trials failed, keep the last candidate anyway
                if f_candidate is None:
                    f_candidate = cand
                f = f_candidate

                # update global best for this length
                if M < best_val:
                    best_val = M
                    best_f = f.copy()
                    print(f"[anneal] n={n:4d} β={beta:5.2f} it={it_counter:5d} M={M:.6f}")

                it_counter += 1

                # early exit if we have already crossed the target bound
                if best_val < 1.503:
                    return best_f, best_val

    return best_f, best_val


# ----------------------------------------------------------------------
# Simple random‑perturbation polish (used after the main optimisation).
# ----------------------------------------------------------------------
def _polish(seq: np.ndarray, rng: np.random.Generator,
            deadline: float, target_sum: float) -> tuple[np.ndarray, float]:
    """Gaussian perturbations while time permits."""
    best = seq.copy()
    best_val = float(np.max(_fft_convolve(best, best)))
    it = 0
    while time.monotonic() < deadline:
        it += 1
        cand = best + rng.normal(scale=0.005 * target_sum, size=best.size)
        cand = _project_to_feasible(cand, target_sum)
        M = float(np.max(_fft_convolve(cand, cand)))
        if M < best_val:
            best_val = M
            best = cand.copy()
            print(f"[polish] it={it:5d} M={M:.6f}")
            if best_val < 1.503:
                break
    return best, best_val
