def simplex_projection(v, s):
    """Project vector v onto the simplex {x>=0, sum x = s}."""
    # Implementation as described.
    # Return same shape as v.
    n = v.shape[0]
    # if sum(v) == s and all(v>=0) then return v
    # else compute
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, s):
    """Project vector v onto the simplex {x >= 0, sum x = s}."""
    # if s <= 0: return zeros
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_descent_refine(a0, target_sum, deadline, rng, max_iters=2000):
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    # compute initial max conv
    conv = np.convolve(a, a)
    best_max = conv.max()
    size_fft = 1 << ((2*n - 1).bit_length())
    size_g_fft = 1 << ((3*n - 2).bit_length())
    eta0 = 0.1 * target_sum / n   # step
    alpha0 = 1.0
    alpha_inc = 10.0  # final alpha
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        # compute conv via FFT for speed
        f_hat = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(f_hat * f_hat, size_fft)[:2*n-1]
        max_conv = conv.max()
        # maybe update best
        if max_conv < best_max:
            best_max = max_conv
            best_a = a.copy()
        # compute smooth weights
        # gradually increase alpha
        alpha = alpha0 + (alpha_inc - alpha0) * it / max_iters
        maxc = conv.max()
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient via FFT
        w_rev = w[::-1]
        a_hat = np.fft.rfft(a, size_g_fft)
        w_rev_hat = np.fft.rfft(w_rev, size_g_fft)
        grad_full = np.fft.irfft(a_hat * w_rev_hat, size_g_fft)
        # slice and reverse
        grad = 2 * grad_full[n-1:2*n-1][::-1]
        # step size schedule
        eta = eta0 / math.sqrt(it+1)
        a_new = a - eta * grad
        # project onto simplex
        a_new = simplex_projection(a_new, target_sum)
        # clip
        a_new = np.clip(a_new, 0.0, 1000.0)
        a = a_new
    # final compute conv and best_max for best_a maybe compute final
    conv_best = np.convolve(best_a, best_a)

def sa_incremental(a0, target_sum, deadline, rng, max_iters=300_000):
    n = len(a0)
    a = a0.copy()
    conv = np.convolve(a, a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    T0 = 0.2
    start = time.monotonic()
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it+=1
        elapsed = time.monotonic() - start
        T = T0 * max(0.0, 1.0 - elapsed/(deadline-start))

        # pick distinct i,j
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1

        max_transfer = min(a[j], 1000.0 - a[i])
        if max_transfer <= 0:
            continue

        # delta magnitude: random proportion of avg or max_transfer
        avg = target_sum / n
        delta = rng.random() * min(max_transfer, avg * 0.5)
        if delta == 0:
            continue

        # compute new convection
        new_conv = conv.copy()
        # linear term updates:
        # update range i:i+n
        new_conv[i:i+n] += 2.0 * delta * a
        new_conv[j:j+n] -= 2.0 * delta * a
        # quadratic term:
        idx = 2*i
        new_conv[idx] += delta*delta
        idx = 2*j
        new_conv[idx] += delta*delta
        idx = i + j
        new_conv[idx] -= 2.0 * delta*delta

        # clip to avoid slight negative due to rounding
        # new_conv = np.maximum(new_conv, 0.0)
        # compute max
        new_max = new_conv.max()
        dE = new_max - cur_max

        if dE < -1e-12 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            # accept
            a[i] += delta
            a[j] -= delta

"""
Hybrid search for a low‑value step‑function.

* The goal is to minimise 2·n·max(conv) / (sum a)², which (with a fixed
  ℓ¹‑norm) is equivalent to minimising the maximum auto‑convolution.
* Two complementary heuristics are employed:
    – A fast simulated‑annealing (SA) that moves a tiny amount of mass
      between two coordinates while updating the full convolution
      incrementally (O(n) per move).
    – A short gradient‑descent phase that smooth‑approximates the “max”
      using a log‑sum‑exp, computes the gradient by FFT‑based convolutions,
      and projects back onto the simplex {a ≥ 0, Σa = √(2n)}.
* Several lengths are tried, random restarts are performed, and the best
  sequence found before the 1000 s budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector v onto the simplex {x ≥ 0, Σx = s}.
    """
    if s <= 0:
        return np.zeros_like(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1          # zero‑based index
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Stretch an existing sequence to a longer length by nearest‑index mapping,
    then renormalise to the ℓ¹‑norm √(2·new_len).
    """
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    total = out.sum()
    if total > 0:
        target = math.sqrt(2.0 * new_len)
        out = out * (target / total)
    return out


def max_conv_full(seq: np.ndarray) -> np.ndarray:
    """Return the full convolution array (length 2·n‑1)."""
    return np.convolve(seq, seq)


# ----------------------------------------------------------------------
# Simulated‑annealing with incremental convolution updates
# ----------------------------------------------------------------------


def sa_incremental(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Perform a cheap SA on `a0`.  Mass is moved from one index to another;
    the convolution is updated in O(n) using the formula
        conv ← conv + 2·(a·d) + d·d,
    where `d` has only two non‑zero entries.
    Returns the best vector encountered and its maximal convolution value.
    """
    n = a0.size
    a = a0.copy()
    conv = max_conv_full(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    T0 = 0.2                     # initial temperature
    start = time.monotonic()
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        elapsed = time.monotonic() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # how much mass can be transferred from j to i ?
        max_transfer = min(a[j], 1000.0 - a[i])
        if max_transfer <= 1e-12:
            continue

        # typical step size: a fraction of the average entry
        avg = target_sum / n
        step = min(max_transfer, avg * 0.5)
        delta = rng.random() * step
        if delta <= 1e-12:
            continue

        # ----- incremental update of the convolution -----
        new_conv = conv.copy()

        # linear part 2·(a·d)
        new_conv[i : i + n] += 2.0 * delta * a
        new_conv[j : j + n] -= 2.0 * delta * a

        # quadratic part d·d
        new_conv[2 * i] += delta * delta
        new_conv[2 * j] += delta * delta
        new_conv[i + j] -= 2.0 * delta * delta

        # numerical safety
        new_conv = np.maximum(new_conv, 0.0)

        new_max = new_conv.max()
        dE = new_max - cur_max

        # Metropolis acceptance
        if dE < -1e-12 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            a[i] += delta
            a[j] -= delta
            conv = new_conv
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[SA] n={n} iter={it} new best max={best_max:.6f}")

        # occasional tiny jitter to escape plateaus
        if it % 5_000 == 0:
            k, l = rng.integers(0, n, size=2)
            if k != l:
                tiny = min(a[l], 1e-3)
                if tiny > 0:
                    a[k] += tiny
                    a[l] -= tiny
                    conv = max_conv_full(a)          # cheap full recompute
                    cur_max = conv.max()

    return best_a, best_max


# ----------------------------------------------------------------------
# Gradient‑descent refinement (smooth max)
# ----------------------------------------------------------------------


def gradient_refine(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 2_000,
) -> tuple[np.ndarray, float]:
    """
    A few iterations of gradient descent on a smooth approximation of the
    max‑convolution (log‑sum‑exp).  Uses FFT to obtain the gradient in
    O(n log n) time.  The vector is projected back onto the simplex after
    each step.
    """
    n = a0.size
    a = a0.copy()
    best_a = a.copy()
    best_max = max_conv_full(a).max()

    size_fft = 1 << ((2 * n - 1).bit_length())
    size_grad_fft = 1 << ((3 * n - 2).bit_length())

    eta0 = 0.1 * target_sum / n          # initial step length
    alpha_start = 1.0
    alpha_end = 15.0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # full convolution (FFT for speed)
        f_hat = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(f_hat * f_hat, size_fft)[: 2 * n - 1]

        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # smooth‑max weights (log‑sum‑exp)
        alpha = alpha_start + (alpha_end - alpha_start) * it / max_iters
        mx = conv.max()
        exp_vals = np.exp(alpha * (conv - mx))
        w = exp_vals / exp_vals.sum()                 # shape (2n‑1,)

        # gradient = 2 * (a * w_rev)[n‑1:2n‑1][::-1]
        w_rev = w[::-1]
        a_hat = np.fft.rfft(a, size_grad_fft)
        w_rev_hat = np.fft.rfft(w_rev, size_grad_fft)
        grad_full = np.fft.irfft(a_hat * w_rev_hat, size_grad_fft)
        grad = 2.0 * grad_full[n - 1 : 2 * n - 1][::-1]   # length n

        # diminishing step size
        eta = eta0 / math.sqrt(it + 1.0)
        a_new = a - eta * grad
        a_new = simplex_projection(a_new, target_sum)
        a_new = np.clip(a_new, 0.0, 1000.0)
        a = a_new

    # final evaluation
    final_max = max_conv_full(best_a).max()
    return best_a, final_max
