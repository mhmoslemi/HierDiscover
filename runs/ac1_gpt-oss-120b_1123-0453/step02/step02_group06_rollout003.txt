def simplex_projection(v, S):
    # v: array
    # ensures sum v_i = S, v_i >= 0
    # from Duchi et al.
    n = len(v)
    if v.sum() == S and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho   # Actually theta = cssv[rho-1] / rho
    # Simpler:
    theta = cssv[rho-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step sequence with a low value of
    2·n·max(a*a) / (sum(a)^2)

The evaluation function ``evaluate_sequence`` first normalises the
ℓ¹‑norm of the vector to √(2·n) and then returns max(a*a).  Hence we
only have to minimise the maximal entry of the auto‑convolution while
keeping the sum fixed.

Algorithm
---------

* **Random initialisation** – for each length a random vector is drawn
  and scaled to the required ℓ¹‑norm.  If a good shorter sequence is already
  known we also try an expanded version of it.

* **Exponentiated‑gradient (EG) descent** – a cheap first‑order method for
  optimisation on the simplex.  At each iteration the full convolution is
  computed (FFT), the index attaining the current maximum is taken, and a
  sub‑gradient of the max‑convolution is built.  The vector is updated by
  a multiplicative EG step and renormalised to the target ℓ¹‑norm.

* **Simulated‑annealing fine‑tuning** – after EG we run a short
  simulated‑annealing local search (mass transfer between two coordinates)
  which is very cheap and frequently improves the result further.

* **Optional LP refinement** – if `cvxpy` is available we perform a few
  cheap linear‑programming steps analogous to the reference code.
  The LP maximises the ℓ¹‑mass under the current maximal auto‑convolution
  bound and then mixes the solution back.

The whole procedure is repeated for several lengths (larger `n` gives a
larger search space) and for many random restarts until the 1000 s budget
is exhausted.  The best sequence found overall is returned as a plain
Python list of floats.

def max_conv_fft(seq: np.ndarray, size: int = None) -> float:
    """
    Fast (FFT) computation of max_i (seq * seq)[i].
    Returns the maximal entry of the auto‑convolution.
    """
    n = seq.size
    if size is None:
        size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, size)
    conv = np.fft.irfft(ft * ft, size)[: 2 * n - 1]
    # negative tiny values may appear because of floating‑point noise
    conv = np.maximum(conv, 0.0)
    return float(conv.max())


def full_conv_fft(seq: np.ndarray, size: int = None) -> np.ndarray:
    """
    Return the full auto‑convolution of *seq* using FFT (length 2·n‑1).
    """
    n = seq.size
    if size is None:
        size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, size)
    conv = np.fft.irfft(ft * ft, size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)
    return conv

# Simplex projection (used in the LP version, not in EG)

def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector *v* onto the simplex
        {x | x_i >= 0, sum x_i = s}.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    if n == 0:
        return v
    # make a copy to avoid mutating the input
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond][np.where(ind[cond] == rho)[0][0]] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Toeplitz‑like matrix A for (f * g) = A @ g."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        A[k, k - np.arange(i_start, i_end + 1)] = f[i_start : i_end + 1]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple:
    """
    One LP refinement step (maximise ℓ¹‑mass under the current max‑conv bound).
    Returns (new_f, new_max_conv).
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv_fft(f)

    n = f.size
    M = max_conv_fft(f)

    A = build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    if g_val.sum() < 1e-12:
        return f, M

    # renormalise to the required ℓ¹‑norm
    g_rescaled = g_val * (target_sum / g_val.sum())

    # modest mixing
    t = 0.3
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    return f_new, max_conv_fft(f_new)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (EG) descent on the simplex
# ----------------------------------------------------------------------


def eg_descent(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 20000,
) -> tuple:
    """
    Multiplicative (exponentiated‑gradient) descent for the max‑convolution
    objective on the probability simplex (scaled to *target_sum*).

    Returns (best_sequence, best_max_conv).
    """
    n = f0.size
    fft_size = 1 << ((2 * n - 1).bit_length())

    a = f0.copy()
    # ensure we start with the correct ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full_like(a, target_sum / n)
    else:
        a = a * (target_sum / a_sum)

    best_seq = a.copy()
    best_val = max_conv_fft(a, fft_size)

    # EG step size schedule η_t = η0 / sqrt(t+1)
    eta0 = 1.0

    start = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # full convolution (needed for the sub‑gradient)
        conv = full_conv_fft(a, fft_size)
        cur_max = float(conv.max())
        i_max = int(conv.argmax())

        # keep the best sequence seen so far
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            print(f"[EG] n={n} iter={it} new best max={best_val:.6f}")

        # build the (sub‑)gradient of the active max entry
        g = np.zeros_like(a)
        k_start = max(0, i_max - (n - 1))
        k_end = min(i_max, n - 1)
        ks = np.arange(k_start, k_end + 1, dtype=int)
        g[ks] = 2.0 * a[i_max - ks]

        # EG multiplicative update
        eta = eta0 / math.sqrt(it + 1)
        # to avoid overflow/underflow keep the exponent moderate
        step = -eta * g
        step = np.clip(step, -50.0, 0.0)          # g ≥ 0 → step ≤ 0
        a = a * np.exp(step)

        # renormalise to the required ℓ¹‑norm (simple scaling works because all entries are ≥0)
        s = a.sum()
        if s == 0:
            # re‑initialise if we collapsed to zero
            a = rng.random(n)
            a = a / a.sum() * target_sum
        else:
            a = a * (target_sum / s)

        # occasional tiny random jitter to escape flat plateaux
        if it % 500 == 0 and rng.random() < 0.3:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = min(a[j], target_sum * 0.02) * rng.random()
                a[i] += delta
                a[j] -= delta
                # renormalise again
                a = a * (target_sum / a.sum())

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic (simulated‑annealing) local search – same as reference
# ----------------------------------------------------------------------


def sa_search(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 50000,
) -> tuple:
    """
    Greedy‑with‑temperature local search (mass transfer between two indices).
    Returns (best_sequence, best_max_conv).
    """
    n = f0.size
    fft_size = 1 << ((2 * n - 1).bit_length())

    f = f0.copy()
    cur_max = max_conv_fft(f, fft_size)
    best_f = f.copy()
    best_max = cur_max

    T0 = 0.02
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        elapsed = time.monotonic() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_transfer = min(f[j], 1000.0 - f[i])
        if max_transfer <= 1e-12:
            continue

        # typical step magnitude ~ half of the average entry
        avg = target_sum / n
        step = min(max_transfer, avg * 0.5)
        delta = rng.random() * step
        if delta <= 1e-12:
            continue

        f_new = f.copy()
        f_new[i] += delta
        f_new[j] -= delta

        new_max = max_conv_fft(f_new, fft_size)

        dE = new_max - cur_max
        if dE < -1e-12 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            f = f_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f_new.copy()
                print(f"[SA] n={n} iter={it} new best max={best_max:.6f}")

        # occasional tiny jitter (helps escaping plateaux)
        if it % 5000 == 0:
            k, l = rng.integers(0, n, size=2)
            if k != l:
                tiny = min(f[l], 1e-2)
                if tiny > 0:
                    f[k] += tiny
                    f[l] -= tiny
                    cur_max = max_conv_fft(f, fft_size)

    return best_f, best_max


# ----------------------------------------------------------------------
# Helper to enlarge a known good sequence to a larger length
# ----------------------------------------------------------------------


def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Stretch ``seq`` to length ``new_len`` by nearest‑index mapping
    and re‑normalise to the ℓ¹‑norm √(2·new_len).
    """
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    total = out.sum()
    if total > 0:
        target = math.sqrt(2.0 * new_len)
        out = out * (target / total)
    return out


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a step sequence with a low evaluation score.
    The function respects the 1000 s time limit (≈ budget_s‑2 s for safety)
    and returns the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Use an existing construction as a seed if it exists.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim == 1 and init.size > 0:
            n0 = init.size
            target0 = math.sqrt(2.0 * n0)
            if init.sum() > 0:
                init = init * (target0 / init.sum())
                sc0 = max_conv_fft(init)
                best_seq = init.copy()
                best_score = sc0
                print(f"[init] using height_sequence_1 (n={n0}) – score={sc0:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – larger n usually gives a better bound.
    # ------------------------------------------------------------------
    lengths = [600, 800, 1000]

    # ------------------------------------------------------------------
    # Main outer loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        fft_size = 1 << ((2 * n - 1).bit_length())

        # Decide how many restarts are affordable for this length.
        # We cap at 20 restarts but stop early if we run out of time.
        restarts = 0
        while time.monotonic() < deadline and restarts < 20:
            restarts += 1

            # --------------------------------------------------------------
            # Initialise start vector
            # --------------------------------------------------------------
            if best_seq is not None and best_seq.size < n and rng.random() < 0.6:
                f0 = expand_sequence(best_seq, n)
            else:
                f0 = rng.random(n)
                f0 = f0 / f0.sum() * target_sum

            # --------------------------------------------------------------
            # Phase 1 – Exponentiated‑gradient descent
            # --------------------------------------------------------------
            f_eg, val_eg = eg_descent(f0, target_sum, deadline, rng,
                                      max_iters=15000)

            # --------------------------------------------------------------
            # Phase 2 – Light simulated‑annealing fine‑tuning
            # --------------------------------------------------------------
            if time.monotonic() < deadline:
                f_sa, val_sa = sa_search(f_eg, target_sum, deadline, rng,
                                         max_iters=8000)
            else:
                f_sa, val_sa = f_eg, val_eg

            # --------------------------------------------------------------
            # Optional LP refinement (if cvxpy is present)
            # --------------------------------------------------------------
            try:
                import cvxpy as cp  # noqa: F401
                has_cvx = True
            except Exception:
                has_cvx = False

            if has_cvx and time.monotonic() < deadline:
                f_cur, cur_val = f_sa, val_sa
                for lp_it in range(3):
                    if time.monotonic() >= deadline:
                        break
                    f_cur, cur_val = lp_refine(f_cur, target_sum, deadline)
                f_sa, val_sa = f_cur, cur_val

            # --------------------------------------------------------------
            # Record global best
            # --------------------------------------------------------------
            if val_sa < best_score - 1e-12:
                best_score = val_sa
                best_seq = f_sa.copy()
                print(
                    f"[global] n={n} restart={restarts} "
                    f"new best score={best_score:.6f}"
                )

            # early stop if we already beat the target bound
            if best_score < 1.503:
                print("[stop] target bound reached")
                break

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed, but guarantees a return.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = max_conv_fft(best_seq)
        print("[fallback] uniform sequence used, score=", best_score)

    # Convert to a plain Python list of floats.
    return [float(x) for x in best_seq]
