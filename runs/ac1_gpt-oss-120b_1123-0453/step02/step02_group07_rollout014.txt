def _refine_via_hill(seq, deadline, rng):
    a = seq.copy()
    n = a.shape[0]
    # compute step etc.
    best = a.copy()
    best_sc = evaluate_sequence(best.tolist())
    cur = a.copy()
    cur_sc = best_sc
    temperature = 0.01
    step = max(1e-8, 0.02 * np.sqrt(2.0 * n) / n)
    while time.monotonic() < deadline:
        # compute max conv and its index
        conv_full = np.convolve(cur, cur)
        max_idx = int(np.argmax(conv_full))
        # compute gradient for that index
        g = np.zeros_like(cur)
        i_start = max(0, max_idx - (n-1))
        i_end = min(max_idx, n-1)
        g[i_start:i_end+1] = cur[max_idx - np.arange(i_start, i_end+1)]
        # compute contributions
        contrib = cur * g
        # choose from indices with largest contrib and positive value
        idx_from = np.argmax(contrib)
        # choose idx_to with smallest g (prefer low contribution)
        idx_to = np.argmin(g)
        if idx_from == idx_to:
            idx_to = (idx_to + 1) % n
        delta = min(cur[idx_from] * 0.1, step)
        if delta <= 0:
            break
        cand = cur.copy()
        cand[idx_from] -= delta
        cand[idx_to] += delta
        # ensure nonneg
        if cand[idx_from] < 0:
            continue
        cand_sc = evaluate_sequence(cand.tolist())
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / temperature) if temperature > 0 else 0
            if rng.random() < prob:
                accept = True
        if accept:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                print(f"[hill refine] n={n} new best={best_sc:.6f}")
        temperature = max(1e-6, temperature * 0.999)
        step = max(1e-12, step * 0.9995)
    return best, best_sc

search for a step‑function whose self‑convolution has a small ℓ∞‑norm.
The evaluation function is homogeneous, i.e. scaling the whole vector does not
change the score.  Therefore we can freely rescale a candidate to the
standard normalisation  Σa = √(2 n) and only the shape of the vector matters.

Two simple one‑parameter families are explored:

* power law   a[i] = (i+1)^(-α)                (α∈(0,1))
* sine power  a[i] = sin(π·(i+½)/n) ** β       (β>0)

For each problem size n we sweep the parameter(s), rescale the vector to the
required ℓ¹‑norm and keep the best sequence seen.  Afterwards a light
projected simulated‑annealing (mass‑transfer) is run starting from the best
candidate to polish the answer.

The whole procedure respects the allotted wall‑clock time (default 1000 s)
and returns a plain Python list of non‑negative heights.
"""

# ----------------------------------------------------------------------
def _eval(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # evaluate_sequence expects a list of Python floats
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
def _norm_to_target(a: np.ndarray, target: float) -> np.ndarray:
    """Rescale a non‑negative vector so that its ℓ¹‑norm equals *target*."""
    s = a.sum()
    if s == 0:
        return a
    return a * (target / s)

def _power_family(n: int, alpha: float, target: float) -> np.ndarray:
    """Return a power‑law vector a[i] = (i+1)^(-α) normalised to *target*."""
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    a = idx ** (-alpha)
    return _norm_to_target(a, target)


# ----------------------------------------------------------------------
def _sine_family(n: int, beta: float, target: float) -> np.ndarray:
    """Return a sine‑power vector a[i] = sin(π·(i+½)/n)^β."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / n) ** beta
    # avoid exact zeros (they would be removed by the security checks)
    a = np.maximum(a, 1e-12)
    return _norm_to_target(a, target)


# ----------------------------------------------------------------------
def _refine(seq: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Very small projected simulated‑annealing that moves a tiny amount of
    mass from an index that contributes heavily to the current worst
    convolution entry to an index that contributes little.
    """
    n = seq.shape[0]
    cur = seq.copy()
    cur_sc = _eval(cur)
    best = cur.copy()
    best_sc = cur_sc

    temperature = 0.05
    step = max(1e-8, 0.02 * np.sqrt(2.0 * n) / n)

    while time.monotonic() < deadline:
        # full convolution and position of the current maximum
        conv = np.convolve(cur, cur)
        k = int(np.argmax(conv))

        # gradient of the maximal entry (ignoring the factor 2)
        g = np.zeros_like(cur)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        g[i_start:i_end + 1] = cur[k - np.arange(i_start, i_end + 1)]

        # contribution of each index to that entry
        contrib = cur * g
        i_from = int(np.argmax(contrib))          # heavy contributor
        i_to   = int(np.argmin(g))                # light contributor
        if i_from == i_to:
            i_to = (i_to + 1) % n

        # move a modest amount of mass
        delta = min(cur[i_from] * 0.03, step)
        if delta <= 0:
            break
        cand = cur.copy()
        cand[i_from] -= delta
        cand[i_to]   += delta
        if cand[i_from] < 0:
            continue

        cand_sc = _eval(cand)

        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                print(f"[refine] n={n} improved best={best_sc:.6f}")

        temperature = max(1e-6, temperature * 0.9995)
        step = max(1e-12, step * 0.9999)

    return best


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Generate a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine scans several problem sizes and two
    inexpensive parametric families, keeps the best construction and then
    performs a tiny hill‑climb to polish it.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Problem sizes – larger n gives a finer discretisation and usually a
    # smaller attainable maximum.
    size_candidates = (200, 300, 400, 500, 600, 800)

    for n in size_candidates:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ----- power‑law sweep -------------------------------------------------
        for alpha in np.linspace(0.45, 0.80, 36):          # step ≈ 0.01
            if time.monotonic() >= deadline:
                break
            a = _power_family(n, float(alpha), target_sum)
            sc = _eval(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[search] n={n} power α={alpha:.3f} score={sc:.6f}")
            if best_score < 1.503:
                break

        # ----- sine‑power sweep ------------------------------------------------
        for beta in np.linspace(0.5, 3.0, 26):            # step ≈ 0.1
            if time.monotonic() >= deadline:
                break
            a = _sine_family(n, float(beta), target_sum)
            sc = _eval(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[search] n={n} sine β={beta:.3f} score={sc:.6f}")
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # If a decent construction was found, perform a light refinement.
    if best_seq is not None and time.monotonic() < deadline:
        best_seq = _refine(best_seq, deadline, rng)
        best_score = _eval(best_seq)
        print(f"[final] refined best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – a uniform vector (always feasible, score = 2.0)
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _eval(best_seq)
        print("[fallback] uniform sequence used, score =", best_score)

    # Return a plain Python list (the evaluator expects a list of floats)
    return [float(x) for x in best_seq]
