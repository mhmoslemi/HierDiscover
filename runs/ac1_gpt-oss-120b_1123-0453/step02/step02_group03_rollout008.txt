"""
Propose a low‑evaluation step‑function.

The search consists of three lightweight stages that run within the 1000 s
budget:

1️⃣  **Initial seeding** – use a previously found construction (global
    variable ``height_sequence_1``) if present; otherwise start from a
    random vector whose ℓ¹‑norm equals √(2·n).

2️⃣  **Projected gradient descent** – for a handful of lengths (200…800)
    a few random restarts are performed.  At every iteration the index
    *k* that attains the current maximum of the auto‑convolution
    ``c = f∗f`` is identified.  Using the analytical gradient

        ∂c_k/∂a_i = 2·a_{k‑i}        (if 0≤k‑i<n, else 0)

    the gradient of the evaluation function
    ``E(f)=2·n·max(c)/‖f‖₁²`` is computed and a projected gradient step
    (clipping to [0,1000] and renormalisation to the fixed ℓ¹‑norm) is
    taken.  The step size is adapted on‑the‑fly; whenever the move does
    not improve the score the step is shrunk, otherwise it is slightly
    enlarged.  Whenever a better global score is encountered it is stored.

3️⃣  **Simulated‑annealing polishing** – the best sequence found by the
    gradient phase is refined with a simple SA that perturbs single
    coordinates.  The schedule cools from 1.0 down to 1e‑4 and occasional
    uphill moves are accepted according to the Metropolis rule.

The routine respects the 1000 s time limit, prints progress information
and finally returns the best sequence (plain ``list[float]``) that was
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Evaluate a candidate using the supplied ``evaluate_sequence``."""
    # ``evaluate_sequence`` is provided in the execution environment.
    return float(evaluate_sequence(seq.tolist()))


def _cost(seq: np.ndarray) -> float:
    """Fast evaluation – identical to ``evaluate_sequence`` for valid inputs."""
    n = seq.shape[0]
    if n == 0:
        return float("inf")
    S = seq.sum()
    if S < 0.01:
        return float("inf")
    M = np.max(np.convolve(seq, seq))
    return 2.0 * n * M / (S * S)


def _gradient(f: np.ndarray) -> tuple[np.ndarray, int, float]:
    """
    Compute the gradient of the evaluation function with respect to ``f``.

    Returns
    -------
    grad : np.ndarray
        Gradient vector (same shape as ``f``).
    k_max : int
        Index where the auto‑convolution attains its maximum.
    M : float
        The maximal convolution value.
    """
    n = f.shape[0]
    conv = np.convolve(f, f)
    k_max = int(np.argmax(conv))
    M = float(conv[k_max])

    S = float(f.sum())

    # dM_i = 2 * f[k_max - i]  if the partner index lies inside the array
    idx = np.arange(n)
    partner = k_max - idx
    inside = (partner >= 0) & (partner < n)
    dM = np.zeros_like(f)
    dM[inside] = 2.0 * f[partner[inside]]

    # gradient of the cost  E = 2 n M / S²
    grad = (2.0 * n / (S * S)) * (dM - 2.0 * M / S)
    return grad, k_max, M


def _gd_optimize(
    f: np.ndarray,
    target_sum: float,
    max_iters: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent for a fixed length ``len(f)``.

    Parameters
    ----------
    f : np.ndarray
        Starting sequence (non‑negative, ℓ¹‑norm = ``target_sum``).
    target_sum : float
        Desired ℓ¹‑norm after each projection.
    max_iters : int
        Upper bound on iteration count for this run.
    deadline : float
        Absolute ``time.monotonic()`` deadline – stop early if exceeded.
    rng : np.random.Generator
        Random source for occasional re‑initialisation.

    Returns
    -------
    (best_seq, best_score)
        The best sequence discovered during this optimisation run.
    """
    n = f.shape[0]
    # keep the best seen locally
    best_seq = f.copy()
    best_sc = _cost(best_seq)

    step = 0.5                # initial step size
    stagnation = 0            # number of consecutive non‑improving steps

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        grad, k_max, M = _gradient(f)

        # tentative gradient step
        f_new = f - step * grad
        f_new = np.clip(f_new, 0.0, 1000.0)

        # renormalise to the prescribed ℓ¹‑norm
        Snew = f_new.sum()
        if Snew < 1e-12:          # degenerate – re‑initialise
            f_new = rng.random(n) + 0.01
            f_new = f_new / f_new.sum() * target_sum
            Snew = target_sum
        else:
            f_new = f_new * (target_sum / Snew)

        # compute the new cost (fast version)
        sc_new = _cost(f_new)

        # decide on acceptance
        cur_sc = _cost(f)
        if sc_new <= cur_sc:
            # accept the move
            f = f_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_seq = f_new.copy()
                print(f"[GD] n={n:3d} it={it:4d} score={sc_new:.6f}")
            # gently increase step size
            step = min(step * 1.02, 2.0)
            stagnation = 0
        else:
            # move was not beneficial – shrink step
            step *= 0.7
            stagnation += 1

        # occasional random shake‑up if we are stagnating
        if stagnation >= 40:
            # add a small random perturbation and re‑normalise
            f = f + rng.normal(scale=0.02 * target_sum, size=n)
            f = np.clip(f, 0.0, 1000.0)
            f = f * (target_sum / f.sum())
            step = 0.5
            stagnation = 0

    return best_seq, best_sc


def _simulated_annealing(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple single‑coordinate SA (identical to the baseline implementation).

    The schedule cools from ``temp_start`` to ``temp_end`` over the
    remaining time budget.
    """
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        temp = temp_start * (temp_end / temp_start) ** frac

        idx = rng.integers(len(cur_seq))
        delta = rng.normal(scale=0.5) * temp * 5.0
        cand = cur_seq.copy()
        cand[idx] = float(np.clip(cand[idx] + delta, 0.0, 1000.0))

        cand_score = _score(cand)

        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

        if rng.random() < 0.001:
            pert = np.maximum(
                cur_seq + rng.normal(scale=0.1, size=len(cur_seq)), 0.0
            )
            cur_seq = np.clip(pert, 0.0, 1000.0)
            cur_score = _score(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    The function respects the 1000 s time budget, runs a multi‑restart
    projected gradient descent for several problem sizes and finishes
    with a short simulated‑annealing polishing phase.  The best
    sequence discovered is returned as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a tiny safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Initialise from a known construction if available
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            seq0 = np.clip(seq0, 0.0, 1000.0)
            sc0 = _score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using height_sequence_1 – score {sc0:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  If no start is available, fall back to a random uniform start
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 200
        best_seq = np.full(n0, math.sqrt(2.0 * n0) / n0, dtype=np.float64)
        best_score = _score(best_seq)
        print(f"[init] uniform start – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Gradient‑descent refinement for several lengths
    # ------------------------------------------------------------------
    lengths = (200, 300, 400, 500, 600, 800)
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of random restarts per length
        restarts = 5
        # allow more iterations for the medium sizes (where we have most
        # confidence that an improvement is possible)
        max_iters = 400 if n <= 400 else 200

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # Preferred initialisation: stretch the current best if lengths match,
            # otherwise use a fresh random vector.
            if best_seq is not None and best_seq.shape[0] == n:
                f0 = best_seq.copy()
                f0 += rng.normal(scale=0.01, size=n)   # break symmetry
                f0 = np.clip(f0, 0.0, 1000.0)
                f0 = f0 * (target_sum / f0.sum())
            else:
                # random start – avoid exact zeros to keep the gradient well‑defined
                f0 = rng.random(n) + 0.01
                f0 = f0 * (target_sum / f0.sum())

            # run the gradient optimiser
            f_opt, sc_opt = _gd_optimize(f0, target_sum, max_iters, deadline, rng)

            if sc_opt < best_score:
                best_seq = f_opt.copy()
                best_score = sc_opt
                print(f"[global] new best after GD (n={n}) – score {sc_opt:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing with simulated annealing
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq, best_score, deadline, rng
        )
        print(f"[final] best score after SA = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Safety fallback – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)

    return [float(x) for x in best_seq]
