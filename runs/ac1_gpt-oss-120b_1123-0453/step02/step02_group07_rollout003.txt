def reduce_max_convolution(f, target_sum, deadline, rng):
    while time < deadline:
        b = np.convolve(f,f)
        max_val = b.max()
        kmax = b.argmax()
        # compute contributions
        n = len(f)
        contrib = np.zeros(n)
        # vectorized: compute partner idx = kmax - np.arange(n)
        i = np.arange(n)
        j = kmax - i
        mask = (j >= 0) & (j < n)
        contrib[mask] = f[i[mask]] * f[j[mask]]
        # find i, i2 indices to move from.
        # if any i where i==j -> product is f_i^2, high.
        # Let's choose i_max = argmax(contrib)
        i_max = int(np.argmax(contrib))
        # choose i_min = argmin(contrib) maybe but ensure not same.
        # Or choose location with small f weight.
        # Choose candidate j index for adding mass: pick random index with low product.
        # Generate list of indices where contrib is small.
        low_indices = np.where(contrib < np.percentile(contrib, 20))[0]  # bottom 20%
        if len(low_indices)==0:
            low_indices=np.arange(n)
        i_min = int(rng.choice(low_indices))
        if i_max == i_min:
            # choose different
            i_min = int(rng.choice(np.arange(n)))
            if i_min == i_max:
                continue
        # delta is fraction of f[i_max], e.g., 5%
        delta = min(f[i_max]*0.05, target_sum*0.001)  # prevent large step
        if delta <= 1e-12: break
        # transfer delta from i_max to i_min
        f[i_max] -= delta
        f[i_min] += delta
        # ensure non-negativity
        # keep sum fixed? Already sum unchanged. Good.
        # Optional: after each move you may want to clip to max 1000.
        # Not needed.
        # Check if improvement: compute new max
        new_max = np.max(np.convolve(f,f))
        # If not improved, maybe revert with some probability
        # Actually we might want to accept only if new_max <= max_val + eps
        if new_max > max_val:
            # revert or accept with some probability (annealing)
            if rng.random() > np.exp(-(new_max-max_val)/0.01):
                # revert
                f[i_max] += delta
                f[i_min] -= delta
        # else keep
    return f

def _targeted_descent(f, target_sum, deadline, rng):
    n = len(f)
    step_factor = 0.2 # fraction of value to move each iteration
    # optionally dynamic adapt step based on max contribution
    while True:
        if time.monotonic() >= deadline:
            break
        # compute conv
        b = np.convolve(f, f)
        max_val = float(b.max())
        kmax = int(b.argmax())
        # compute contributions c_i
        i = np.arange(n)
        j = kmax - i
        mask = (j >=0) & (j < n)
        c = np.zeros(n)
        c[mask] = f[i[mask]] * f[j[mask]]
        # identify heavy index
        i_max = int(np.argmax(c))
        # find candidate light index: smallest f
        # But ensure we don't move mass from i_max to itself.
        # Choose random among low f values
        low_mask = (f < np.percentile(f, 20))
        candidates = np.where(low_mask)[0]
        if candidates.size == 0:
            candidates = np.arange(n)
        i_min = int(rng.choice(candidates))
        if i_min == i_max:
            # pick another random
            i_min = int(rng.integers(0, n))
            if i_min == i_max:
                continue
        # compute delta: fraction of f[i_max] to move, but not more than 10% maybe
        delta = f[i_max] * step_factor
        # also limit delta to target_sum*0.001 perhaps.
        delta = min(delta, target_sum * 0.001)
        if delta < 1e-12:
            break
        # apply move
        f[i_max] -= delta
        f[i_min] += delta
        # ensure not negative
        if f[i_max] < 0:
            # revert move
            f[i_max] += delta
            f[i_min] -= delta
            break
        # compute new max
        new_max = float(np.max(np.convolve(f, f)))
        # if new max not improved (>= old), revert
        if new_max >= max_val:
            # revert
            f[i_max] += delta
            f[i_min] -= delta
            # reduce step factor to try smaller moves
            step_factor *= 0.8
            if step_factor < 1e-4:
                break
        else:
            # accept improvement, maybe increase step factor modestly
            step_factor = min(step_factor * 1.05, 0.3)
    return f

"""
Propose a low‑value step sequence.

Algorithm
---------
1.  Choose several problem sizes (n ≈ 150 … 500) and, for each size,
    generate a few random positive vectors (Dirichlet‑like) that are
    normalised to ∑a = √(2 n).

2.  Apply a *targeted‐descent* routine:
    -   Compute the auto‑convolution b = a∗a and locate the worst
        lag kₘₐₓ (the index where b attains its maximum).
    -   For that lag compute the contribution of every entry
        cᵢ = aᵢ·a_{kₘₐₓ−i} (zero when the partner index is out of range).
    -   Pick a “heavy’’ index i_h from the top‑10 % of cᵢ and a
        “light’’ index i_l from the bottom‑10 % of the current masses.
    -   Transfer a small fraction of mass δ from i_h to i_l.
        The move is kept only when it strictly reduces the current
        maximum convolution value; otherwise it is rejected and the
        step size is shrunk.
    -   The loop stops after a time budget, after a fixed number of
        iterations or after a patience counter of non‑improving steps.

3.  Refine the result with a lightweight stochastic hill‑climber
    (projected simulated‑annealing) that preserves the ℓ¹‑norm.

4.  Keep the best sequence over all runs.  If a previously discovered
    construction ``height_sequence_1`` is available it is used as a warm
    start.

The whole search respects the supplied time budget (default 1000 s) and
returns a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Evaluate a candidate using the provided evaluate_sequence."""
    # evaluate_sequence is available in the execution environment
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
def _targeted_descent(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Reduce the worst entry of a∗a by moving mass from heavy contributors
    to light locations.  The ℓ¹‑norm (sum) is kept constant.
    """
    n = a.shape[0]
    # Ensure the array respects the soft caps used by evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)

    # Parameters controlling the aggressiveness of the descent
    max_iters = 5000           # safety cap
    patience = 500             # stop after this many non‑improving steps
    no_improve = 0
    step_factor = 0.5          # start with a relatively large fraction

    for _ in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # current convolution and its worst lag
        b = np.convolve(a, a)
        max_val = float(b.max())
        k_max = int(b.argmax())

        # contribution of each entry to that worst lag
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        # heavy candidates – top 10 % by contribution
        heavy_thr = np.percentile(contrib, 90)
        heavy_idx = np.where(contrib >= heavy_thr)[0]
        if heavy_idx.size == 0:
            heavy_idx = np.arange(n)
        i_heavy = int(rng.choice(heavy_idx))

        # light candidates – bottom 10 % by current mass
        light_thr = np.percentile(a, 10)
        light_idx = np.where(a <= light_thr)[0]
        if light_idx.size == 0:
            light_idx = np.arange(n)
        i_light = int(rng.choice(light_idx))

        if i_heavy == i_light:
            # pick a different light index in a cheap deterministic way
            i_light = (i_heavy + 1) % n
            if i_light == i_heavy:
                continue

        # move a fraction of the heavy mass to the light position
        delta = min(a[i_heavy] * step_factor, target_sum * 0.001)
        if delta <= 1e-12:
            break

        # apply tentative move
        a[i_heavy] -= delta
        a[i_light] += delta

        # compute new worst convolution entry
        new_max = float(np.max(np.convolve(a, a)))

        if new_max < max_val - 1e-12:               # improvement
            no_improve = 0
            # a little larger steps when we are doing well
            step_factor = min(step_factor * 1.02, 0.7)
        else:                                       # reject move
            a[i_heavy] += delta
            a[i_light] -= delta
            no_improve += 1
            step_factor = max(step_factor * 0.7, 0.02)
            if no_improve >= patience:
                break

    return a


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    A modest projected simulated‑annealing that keeps the ℓ¹‑norm fixed.
    Moves a small amount of mass from a random positive index to another.
    """
    n = a.shape[0]
    best = a.copy()
    best_sc = _score(best)

    cur = a.copy()
    cur_sc = best_sc

    temperature = 0.1
    step = max(1e-9, 0.05 * target_sum / n)

    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue

        delta = min(cur[i] * 0.05, step)
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            continue

        cand_sc = _score(cand)

        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                print(f"[hill] n={n} new best={best_sc:.6f}")

        temperature = max(1e-6, temperature * 0.9995)
        step = max(1e-12, step * 0.9999)

    return best


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) achieve a score

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if it exists.
    if "height_sequence_1" in globals():
        init = np.array(list(height_sequence_1), dtype=np.float64)
        n0 = init.size

        best_seq, best_score = init, sc0
        print("[init] height_sequence_1 used, score =", sc0)

    # ------------------------------------------------------------------
    # Problem sizes to explore – larger n gives a richer search space.
    for n in (150, 200, 250, 300, 350, 400, 450, 500):
        if time.monotonic() >= deadline:

        # A few independent restarts for each size.
        restarts = 4 if n <= 300 else 3
        for r in range(restarts):

            a = rng.random(n) + 0.001
            a = a / a.sum() * target_sum

            # Occasionally try a simple geometric shape as a different seed.

            a = _targeted_descent(a, target_sum, deadline, rng)

            # ---------- hill‑climber refinement ----------
            a = _hill_climb(a, target_sum, deadline, rng)

            if sc < best_score:

                print(f"[global] n={n} restart={r} improved best={best_score:.6f}")
