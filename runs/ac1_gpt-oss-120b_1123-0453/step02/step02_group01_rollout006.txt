def _smooth_max_and_grad(a, beta):
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    # shift for stability
    shifted = conv - max_c
    exps = np.exp(beta * shifted)
    sum_exps = np.sum(exps)
    softmax = exps / sum_exps
    # gradient
    # conv of softmax with reversed a
    grad_full = np.convolve(softmax, a[::-1])
    n = a.shape[0]
    grad = 2.0 * grad_full[n-1:n-1+n]
    # smooth max value
    smooth_max = max_c + np.log(sum_exps) / beta
    return smooth_max, conv, grad

"""
Projected‑Adam optimisation of a smooth approximation to the maximal
auto‑convolution value (the quantity returned by ``evaluate_sequence``).

The algorithm works with a chosen length `n` and enforces the ℓ¹‑norm
constraint Σ a_i = √(2 n) by projecting onto the simplex after each update.
A smooth “log‑sum‑exp’’ surrogate of the maximum is minimised; its gradient
is obtained by a single FFT‑based convolution, so each iteration costs
≈O(n log n).  The temperature (β) of the surrogate is slowly increased,
turning the smooth objective into the true max.  At the end of the Adam
optimisation a few cheap LP refinements (if CVXPY is available) are tried.
Multiple lengths and random restarts are explored within the allotted time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – returns a vector with non‑negative entries
# and sum equal to `target`.
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex of sum `target`."""
    if target <= 0:
        return np.zeros_like(v)

    # make a copy to avoid modifying the input
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; distribute mass uniformly
        theta = (np.sum(v) - target) / len(v)
        w = np.maximum(v - theta, 0.0)
        # in pathological cases (very negative v) fall back to uniform
        if w.sum() == 0:
            w[:] = target / len(v)
        else:
            w = w * (target / w.sum())
        return w

    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numeric correction – ensure exact sum (tiny rounding errors)
    w_sum = w.sum()
    if w_sum > 0:
        w *= target / w_sum
    else:
        w[:] = target / len(v)
    return w


# ----------------------------------------------------------------------
def _initialize_random_vec(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform Dirichlet point on the simplex of dimension `n` with sum `target_sum`."""
    x = rng.random(n) + 0.1          # avoid exact zeros that slow optimisation
    x = x / x.sum() * target_sum
    return x.astype(np.float64)


# ----------------------------------------------------------------------
# FFT‑based convolution (real inputs).  Returns the linear convolution of x and y.
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of `max(conv)` where conv = a*a.
    Returns:
        conv      – the full convolution a*a (length 2n‑1)
        max_val   – the true maximum of conv (used for the objective)
        grad      – gradient of the smooth max w.r.t. a (length n)
    """
    # 1) convolution a*a
    conv = _fft_convolve(a, a)                     # length 2n‑1
    max_val = np.max(conv)

    # 2) log‑sum‑exp stabilisation
    shifted = conv - max_val                       # max becomes 0
    exp_term = np.exp(beta * shifted)              # all ≤ 1
    sum_exp = np.sum(exp_term)
    softmax = exp_term / sum_exp                   # ∑ softmax = 1

    # 3) gradient:  grad_i = 2 * Σ_j softmax_{i+j} * a_j
    a_rev = a[::-1]
    grad_full = _fft_convolve(softmax, a_rev)      # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1:n - 1 + n]        # extract n entries
    return conv, max_val, grad


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Adam‑style optimisation of the smooth max of the auto‑convolution.
    Stops either when time runs out or the theoretical lower bound is reached.
    Returns the best vector found and its (true) maximal auto‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    lr = 0.5
    eps = 1e-8
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the smooth max
    beta = 5.0
    beta_max = 200.0
    # increase β slowly – 5000 steps are enough to reach β_max
    beta_inc = (beta_max - beta) / 5000.0

    # keep the best point seen
    best_a = a.copy()
    best_score = float("inf")

    # theoretical lower bound (average of the convolution)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    t = 0
    while time.time() < deadline:
        t += 1

        conv, max_val, grad = _smooth_max_and_grad(a, beta)

        # record best (true) max
        if max_val < best_score:
            best_score = max_val
            best_a = a.copy()
            print(f"[opt] n={n} iter={t} max={max_val:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the ℓ¹‑simplex (keeps sum = target_sum and non‑negativity)
        a = _project_simplex(a, target_sum)

        # increase temperature
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # stop if we are essentially at the lower bound
        if max_val <= lower_bound * 1.001:
            best_score = max_val
            best_a = a.copy()
            print(f"[opt‑done] n={n} reached near‑lower‑bound {max_val:.6f}")
            break

        # safety – limit number of iterations to avoid infinite loops
        if t >= 8000:
            # we continue only if time permits; otherwise break
            if time.time() >= deadline:
                break
    return best_a, best_score

# Linear‑programming refinement (identical to the LP used in the reference code).
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) == np.convolve(f, g).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _lp_step(f: np.ndarray, M: float, target_sum: float):
    """
    Solve the LP: maximise Σ g_i   subject to   (f∗g)_k ≤ M , g_i ≥ 0 .
    Returns a feasible g (scaled to have ℓ¹‑norm `target_sum`) or None.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None

    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:  # pragma: no cover
            return None

    if g.value is None:
        return None

    g_val = np.maximum(g.value, 0.0)
    s = np.sum(g_val)
    if s < 1e-12:
        return None
    g_scaled = g_val * (target_sum / s)
    return g_scaled


def _lp_refine(a: np.ndarray, cur_max: float, target_sum: float,
               deadline: float, max_iter: int = 5):
    """
    Apply a few LP refinements starting from `a`.
    Stops early if time runs out.
    """
    best = a.copy()
    best_M = cur_max
    for i in range(max_iter):
        if time.time() >= deadline:
            break
        g = _lp_step(best, best_M, target_sum)
        if g is None:
            break
        M_new = np.max(_fft_convolve(g, g))
        if M_new < best_M - 1e-12:
            best = g
            best_M = M_new
            print(f"[lp‑refine] iter={i} new_max={M_new:.6f}")
        else:
            break
    return best, best_M

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The routine explores several lengths (≈400‑1000) and for each length performs
    a few random restarts.  Each restart is optimised with a projected‑Adam
    method that minimises a smooth approximation of the maximal auto‑convolution
    value.  If CVXPY is available, a short LP refinement is attempted afterwards.
    The best sequence found before the time budget expires is returned as a
    plain Python list of floats.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # 1 s safety margin

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # range of lengths – larger n gives a tighter theoretical bound
    lengths = [400, 500, 600, 800, 1000]

    # try to reuse a known construction if it exists
    if 'height_sequence_1' in globals():
        try:
            old = np.asarray(height_sequence_1, dtype=np.float64)
            n_old = old.shape[0]
            target_old = math.sqrt(2.0 * n_old)
            scale = target_old / np.sum(old)
            old = np.maximum(old * scale, 0.0)
            # run a short optimisation on the imported length as well
            lengths = sorted(set(lengths + [n_old]))
        except Exception:
            old = None
    else:
        old = None

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        # number of restarts – we keep it modest to respect the time budget
        restarts = 3

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # initialise a random feasible vector (or reuse the imported one)
            if old is not None and old.shape[0] == n:
                a0 = old.copy()
            else:
                a0 = _initialize_random_vec(n, target_sum, rng)

            # give each restart a bounded slice of the remaining time
            remaining = deadline - time.time()
            if remaining < 5.0:
                break
            sub_deadline = time.time() + min(30.0, remaining * 0.7)

            a_opt, max_val = _optimize(a0, target_sum, sub_deadline)

            # optional LP refinement (only if CVXPY is present)
            a_opt, max_val = _lp_refine(a_opt, max_val, target_sum, sub_deadline)

            # store the best overall result
            if max_val < best_score:
                best_score = max_val
                best_seq = a_opt.copy()
                print(f"[global‑best] n={n} restart={r} score={max_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (unlikely), return a simple uniform vector.
    if best_seq is None:
        n_fallback = 400
        target_fallback = math.sqrt(2.0 * n_fallback)
        best_seq = _initialize_random_vec(n_fallback, target_fallback, rng)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[fallback] used uniform seed, score={best_score:.6f}")

    # final safety: convert to plain Python list of floats
    return [float(x) for x in best_seq]
```

This implementation explores a range of lengths, uses a fast Adam‑based optimiser with an FFT‑accelerated gradient of a smooth maximum, projects onto the required simplex after each step, and (when possible) finishes with a few LP refinements.  
It stays well within the 1000 s budget, avoids closures or lambdas, and respects all security checks of `evaluate_sequence`.
