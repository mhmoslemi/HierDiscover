def simplex_projection(v, s):
    # v is np.array, s > 0
    if np.sum(v) == s and np.all(v >= 0):
        return v
    # shift to zero-sum case; but use algorithm.
    u = np.sort(v)[::-1]  # descending
    css = np.cumsum(u) - s
    rho = np.where(u - css / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = css[-1] / len(v)
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
* **Convex sub‑gradient descent on the ℓ∞‑norm of the self‑convolution**
  – The score (after normalising the ℓ¹‑mass to √(2 n)) equals
    max_k (a*a)_k.  This is a convex, piecewise‑quadratic function of the
    non‑negative vector a with a fixed ℓ¹‑norm.
  – For the current a we locate the index/indices where the convolution
    attains its maximum, compute a sub‑gradient
        ∂/∂a_i max_k (a*a)_k = 2·∑_{k active} a_{k‑i},
    and take a step opposite to this sub‑gradient.
  – The updated vector is projected back onto the simplex
    {a≥0, ∑a = √(2 n)} using the standard Euclidean projection.
  – A simple back‑tracking line‑search (step size halving) ensures that
    every accepted move strictly improves the score.

* **Multi‑size search & restarts**
  – The routine is executed for several lengths (200 … 800).  For each
    length we perform a few random restarts; a good solution from a
    previous length is up‑sampled (linear interpolation) and used as a
    warm start.
  – The best sequence over all lengths is kept.

* **Polishing**
  – After the sub‑gradient phase a cheap stochastic “mass‑transfer”
    local search (similar to the previous baseline) explores tiny
    perturbations while keeping the total mass fixed.

All computations respect the 1000 s global budget.  The final result
is returned as a plain list of Python floats, ready for the external
`evaluate_sequence` checker.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global random generator – will be initialised in `propose_candidate`.
_rng: np.random.Generator | None = None


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        { x ≥ 0,  Σ x = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # Fast path for already feasible vectors
    if np.all(v >= 0) and abs(v.sum() - s) < 1e-12:
        return v.copy()

    # Sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - s
    rho_arr = u - css / (np.arange(len(v)) + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = css[-1] / len(v)
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _interpolate_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of seq to length new_n (preserves shape)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, seq)


def _subgradient_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 400,
    step0_factor: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Convex sub‑gradient descent on max_k (a*a)_k with simplex projection.
    Returns (best_sequence, best_score) found during the run.
    """
    a = a0.copy()
    n = len(a)
    # Initial score
    best_seq = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    # Initial step size (relative to the feasible ℓ¹‑norm)
    step0 = step0_factor * target_sum

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # Self‑convolution and its maximum
        conv = np.convolve(a, a)
        M = conv.max()

        # Indices where the maximum is attained (allow tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # Sub‑gradient of the max (average over all active indices)
        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = k - idx_i
            grad[idx_i] += 2.0 * a[idx_j]
        if len(active) > 1:
            grad /= float(len(active))

        # If gradient is (near) zero we cannot improve further.
        if np.linalg.norm(grad) < 1e-14:
            break

        # Diminishing step size
        step = step0 / math.sqrt(it + 1.0)

        # Back‑tracking line search (max 5 halving steps)
        improved = False
        for _ in range(5):
            a_temp = a - step * grad
            a_proj = _project_onto_simplex(a_temp, target_sum)
            sc = float(evaluate_sequence(a_proj.tolist()))
            if sc + 1e-12 < best_score:
                # Accept the move
                a = a_proj
                best_seq = a.copy()
                best_score = sc
                improved = True
                break
            else:
                step *= 0.5

        if not improved:
            # No improvement – reduce step size aggressively and continue
            step0 *= 0.7
            if step0 < 1e-6:
                break

    return best_seq, best_score


def _local_search(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic walk that conserves the ℓ¹‑mass.
    Transfers a small random amount of mass between two positions.
    """
    best = seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = len(best)

    while time.monotonic() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # Transfer a random fraction (normally distributed) of the source mass
        delta = _rng.normal(loc=0.0, scale=0.02) * max(1e-6, best[i])
        if delta == 0.0:
            continue
        cand_i = best[i] + delta
        cand_j = best[j] - delta
        if cand_i < 0.0 or cand_j < 0.0:
            continue
        cand = best.copy()
        cand[i] = cand_i
        cand[j] = cand_j
        sc = float(evaluate_sequence(cand.tolist()))
        if sc + 1e-12 < best_score:
            best_score = sc
            best = cand
            # optional log
            print(f"[local] improved score={best_score:.6f}")

    return best, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects a 1000 s overall timeout (a short safety margin
    is kept).  It combines sub‑gradient optimisation on several problem
    sizes with a final stochastic polishing step.
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # Safety margin of a couple of seconds before the external timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    overall_best_seq: np.ndarray | None = None
    overall_best_score: float = float("inf")

    # ----------------------------------------------------------------------
    # Helper: try to retrieve the historic best sequence if it exists.
    # ----------------------------------------------------------------------
    try:
        # The environment may provide a variable named `height_sequence_1`.
        historic = globals().get("height_sequence_1", None)
        if historic is not None:
            historic_arr = np.array(historic, dtype=np.float64)
            # Normalise to the required ℓ¹‑norm for its length.
            n_hist = len(historic_arr)
            target_hist = math.sqrt(2.0 * n_hist)
            historic_arr = historic_arr * (target_hist / np.sum(historic_arr))
            overall_best_seq = historic_arr.copy()
            overall_best_score = float(evaluate_sequence(overall_best_seq.tolist()))
            print(f"[init] historic sequence score={overall_best_score:.6f}")
    except Exception:
        # If anything goes wrong we simply ignore the historic data.
        overall_best_seq = None
        overall_best_score = float("inf")

    # ----------------------------------------------------------------------
    # Phase 1 – sub‑gradient optimisation on several lengths.
    # ----------------------------------------------------------------------
    # Lengths explored (increase gradually; larger n can give better bounds).
    length_candidates = (200, 250, 300, 350, 400, 500, 600, 800)

    for n in length_candidates:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)  # the normalisation used by the scorer

        # Number of random restarts per length depends on remaining time.
        # We allocate at most 3 restarts per length, but stop early if time runs out.
        max_restarts = 3
        for restart in range(max_restarts):
            if time.monotonic() >= deadline:
                break

            # Choose an initial point.
            if overall_best_seq is not None:
                # Upsample/downsample the current best to the new length.
                a0 = _interpolate_sequence(overall_best_seq, n)
                a0 = a0 * (target_sum / np.sum(a0))
            else:
                # Pure random Dirichlet‑type start.
                a0 = _rng.random(n) + 0.001  # avoid exact zeros
                a0 = a0 * (target_sum / np.sum(a0))

            # Run sub‑gradient optimisation.
            seq_opt, score_opt = _subgradient_optimize(
                a0,
                target_sum,
                deadline,
                max_iters=250,
                step0_factor=0.9,
            )

            # Keep the global best.
            if score_opt + 1e-12 < overall_best_score:
                overall_best_score = score_opt
                overall_best_seq = seq_opt.copy()
                print(f"[global] new best after n={n}, restart={restart} → {overall_best_score:.6f}")

            # Early exit if we already beat the target bound comfortably.
            if overall_best_score < 1.45:
                break

    # ----------------------------------------------------------------------
    # Phase 2 – cheap stochastic polishing (if time permits).
    # ----------------------------------------------------------------------
    if overall_best_seq is not None and time.monotonic() < deadline:
        seq_polished, score_polished = _local_search(overall_best_seq, deadline)
        if score_polished + 1e-12 < overall_best_score:
            overall_best_score = score_polished
            overall_best_seq = seq_polished.copy()
            print(f"[global] after polishing → {overall_best_score:.6f}")

    # ----------------------------------------------------------------------
    # Fallback: deterministic constant sequence (should never be used).
    # ----------------------------------------------------------------------
    if overall_best_seq is None:
        n_fallback = 256
        overall_best_seq = np.ones(n_fallback, dtype=np.float64)
        overall_best_seq = overall_best_seq * (math.sqrt(2.0 * n_fallback) / n_fallback)

    # Final safety clipping (the evaluator also clips, this is just defensive).
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    # Return a plain Python list of floats as required by the external judge.
    return [float(x) for x in overall_best_seq]
