def propose_candidate(seed=42,...):

    for n in n_list:
        if time.monotonic() > deadline: break
        target = math.sqrt(2*n)
        # Random initial f
        f = rng.random(n)
        f = f / f.sum() * target
        # parameters: mix = 0.3 maybe
        mix = 0.3
        # iteration count dynamic
        for it in range(200):
            if time.monotonic() > deadline: break
            M = np.max(np.convolve(f,f))
            A = conv_matrix(f)   # size (2n-1, n)
            # Solve LP: maximize sum(g) s.t. A@g <= M, g>=0
            g = cp.Variable(n)
            constraints = [g >= 0, A @ g <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
            # solve
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
            if g.value is None: break
            g_val = np.clip(g.value, 0, None)
            s = g_val.sum()
            if s < 1e-12:
                break
            g_rescaled = g_val * (target / s)
            # mix update with possible decreasing mix
            t = mix * (0.95 ** it) # decreasing
            f_new = (1-t)*f + t*g_rescaled
            f_new = np.clip(f_new, 0, 1000)
            score = _score(f_new.tolist())
            if score < overall_best_score:
                overall_best_score = score

                # maybe stuck, break early
                pass
            f = f_new
            # early stop if M low enough maybe

                break

            seq = overall_best_seq.copy()
        else:
            seq = f.copy()
        # local search (maybe simulated annealing)
        seq, score = simulated_annealing(seq, deadline)
        if score < overall_best_score:
            overall_best_score = score

def _lp_improve(f, target_sum, deadline, mix_start=0.3, mix_decay=0.95, max_iters=120):
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        M = np.max(np.convolve(f,f))
        A = conv_matrix(f)
        if cp is None:
            break
        g = cp.Variable(len(f))
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        if g.value is None:
            break
        g_val = np.maximum(g.value, 0)
        if g_val.sum() == 0:
            break
        g_rescaled = g_val * (target_sum / g_val.sum())
        t = mix_start * (mix_decay ** it)
        f = (1 - t) * f + t * g_rescaled
        f = np.clip(f, 0, 1000)
    return f

def _targeted_local_search(seq, deadline, rng):
    best = seq.copy()
    best_score = _score(best.tolist())
    n = len(seq)
    while time.monotonic() < deadline:
        # decide whether do random or targeted move
        if rng.random() < 0.7:
            # random move as in earlier _local_search
            i, j = rng.integers(0, n, size=2)
            if i == j: continue
            delta = rng.normal(scale=0.02) * max(1e-6, best[i])
            if delta == 0: continue
            cand_i = best[i] + delta
            cand_j = best[j] - delta
            if cand_i < 0 or cand_j < 0: continue
            cand = best.copy()
            cand[i] = cand_i
            cand[j] = cand_j
        else:
            # targeted: compute convolution, find max index
            conv = np.convolve(best, best)
            k = int(np.argmax(conv))
            # candidates for moving mass from indices that contribute strongly to this k
            # contributions: for i that produce j = k-i
            # compute contributions array:
            contrib_indices = []
            for i in range(len(best)):
                j = k - i
                if 0 <= j < n:
                    contrib = best[i] * best[j]
                    contrib_indices.append((contrib, i, j))
            # sort by contribution descending
            contrib_indices.sort(reverse=True, key=lambda x: x[0])
            # pick the pair with largest contribution where source has non-zero
            for _, i_src, _ in contrib_indices[:5]:
                if best[i_src] > 0:
                    src = i_src
                    break
            else:
                src = rng.integers(0, n)
            # choose destination far from src: farthest index or random low mass
            # pick destination with minimum value
            dest = int(np.argmin(best))
            # move delta from src to dest
            delta = rng.uniform(0, min(0.1, best[src]))
            if delta <= 0: 
                continue
            cand = best.copy()
            cand[src] -= delta
            cand[dest] += delta
        # evaluate
        sc = _score(cand.tolist())
        if sc < best_score:
            best_score = sc
            best = cand
            print(f"[local] improved to {sc:.6f}")
    return best, best_score

"""
Search for a low‑value step sequence.

The algorithm combines three ideas:

1. **Random restarts** – for several lengths `n` we start from many
   random feasible vectors (ℓ¹ norm = √(2 n)).  This mitigates getting
   stuck in a bad basin.

2. **LP refinement** – with a fixed vector `f` we solve the linear
   program  
   `max Σ g_i  s.t.  A(f)·g ≤ M,  g ≥ 0`, where `A(f)` is the convolution
   matrix and `M = max(f*f)`.  The solution `g` is rescaled to the target
   ℓ¹‑norm and mixed into `f`.  Repeating this monotone‑ly reduces the
   maximal auto‑convolution value.

3. **Targeted local search** – after the LP phase we repeatedly move a
   small amount of mass from indices that contribute heavily to the
   current worst convolution entry towards indices with very little mass.
   The move is accepted if it improves the score; uphill moves are also
   occasionally accepted using a simple simulated‑annealing schedule.

The routine respects the 1000 s overall budget (a small safety margin
is kept) and returns the best sequence found.  If CVXPY is unavailable,
the algorithm falls back to a pure random search with the same
local‑search polishing.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional LP solver
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None

# ----------------------------------------------------------------------
# Global random generator – will be initialised in `propose_candidate`.
_rng = None


def conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that (A @ g) == f * g (linear convolution).

    Parameters
    ----------
    f : np.ndarray
        One‑dimensional array of length n.

    Returns
    -------
    np.ndarray
        A matrix of shape (2*n-1, n).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill by definition of linear convolution.
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over indices of f that participate in output k
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    return A


def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq))


def _lp_refine(
    f: np.ndarray,
    target_sum: float,
    deadline: float,
    mix_start: float = 0.3,
    mix_decay: float = 0.95,
    max_iters: int = 120,
) -> np.ndarray:
    """
    Perform the LP‑based improvement for a fixed length.

    Parameters
    ----------
    f : np.ndarray
        Starting sequence (already normalised to `target_sum`).
    target_sum : float
        Desired ℓ¹ mass (normally √(2 n)).
    deadline : float
        Absolute time when the routine must stop.
    mix_start, mix_decay : float
        Parameters for the convex‑combination factor `t = mix_start *
        mix_decay**it`.
    max_iters : int
        Upper bound on the number of LP iterations.

    Returns
    -------
    np.ndarray
        The refined sequence (still non‑negative, not necessarily optimal).
    """
    if cp is None:  # pragma: no cover
        return f

    n = f.shape[0]
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # Current worst auto‑convolution entry.
        M = float(np.max(np.convolve(f, f)))

        # Linear convolution matrix (A·g = f*g).
        A = conv_matrix(f)

        # ---- Linear program -------------------------------------------------
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:  # pragma: no cover
            try:
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
            except Exception:  # pragma: no cover
                break

        if g.value is None:
            break

        g_val = np.maximum(g.value, 0.0)
        s = float(g_val.sum())
        if s < 1e-12:
            break

        # Rescale to keep the ℓ¹‑norm fixed.
        g_rescaled = g_val * (target_sum / s)

        # Convex‑combination update (step size decays geometrically).
        t = mix_start * (mix_decay ** it)
        f = (1.0 - t) * f + t * g_rescaled
        f = np.clip(f, 0.0, 1000.0)

    return f


def _targeted_local_search(
    seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic walk that tries to lower the evaluation score.

    The walk mixes two types of moves:
    * random pairwise mass transfers,
    * targeted transfers from indices that strongly contribute to the
      current maximal convolution entry towards indices with minimal mass.

    A simple simulated‑annealing schedule is used for uphill moves.
    """
    best = seq.copy()
    best_score = _score(best.tolist())
    n = best.shape[0]

    # initial temperature – a small fraction of the current score
    temperature = max(1e-3, 0.01 * best_score)

    while time.monotonic() < deadline:
        # decay temperature slowly
        temperature *= 0.9995
        move_type = rng.random()

        if move_type < 0.6:
            # ---- random transfer -----------------------------------------
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.normal(loc=0.0, scale=0.02) * max(1e-9, best[i])
            if delta == 0.0:
                continue
            cand_i = best[i] + delta
            cand_j = best[j] - delta
            if cand_i < 0.0 or cand_j < 0.0:
                continue
            cand = best.copy()
            cand[i] = cand_i
            cand[j] = cand_j
        else:
            # ---- targeted transfer ----------------------------------------
            conv = np.convolve(best, best)
            k = int(np.argmax(conv))                     # index of the peak
            # collect all (i, j) pairs that produce this peak
            contribs = []
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contribs.append((best[i] * best[j], i, j))
            if not contribs:
                continue
            # pick the source index with largest product contribution
            contribs.sort(reverse=True, key=lambda x: x[0])
            src = contribs[0][1]
            # destination: index with smallest current value
            dest = int(np.argmin(best))

            # move a modest amount of mass
            max_transfer = min(0.1 * best[src], best[src])
            if max_transfer <= 0:
                continue
            delta = rng.uniform(0.0, max_transfer)
            cand = best.copy()
            cand[src] -= delta
            cand[dest] += delta

        sc = _score(cand.tolist())
        improve = sc < best_score
        if improve or rng.random() < math.exp(-(sc - best_score) / max(temperature, 1e-12)):
            best = cand
            best_score = sc
            if improve:
                print(f"[local] improved score = {sc:.6f}")

    return best, best_score


def _random_search(
    n: int,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple Dirichlet‑style random search (fallback when CVXPY is missing).
    """
    best_seq = None
    best_score = float("inf")
    while time.monotonic() < deadline:
        vec = rng.random(n) + 0.1
        vec = vec / vec.sum() * target_sum
        sc = _score(vec.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = vec.copy()
    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Main entry point for the search.

    The function respects a 1000 s wall‑clock limit (a short safety margin
    is kept).  It tries several problem sizes, performs an LP‑based
    refinement with many random restarts and finishes with a targeted
    local search.  If linear programming is unavailable a pure random walk
    is used instead.
    """
    global _rng
    _rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Optional deterministic start from a previously saved solution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.array(height_sequence_1, dtype=np.float64)
            sc0 = _score(seq0.tolist())
            overall_best_seq = seq0.copy()
            overall_best_score = sc0
            print(f"[init] seed sequence score = {sc0:.6f}")
        except Exception:
            # ignore malformed globals
            pass

    # ------------------------------------------------------------------
    # 2) Multi‑size random‑restart + LP refinement
    # ------------------------------------------------------------------
    # Problem sizes – larger n gives more freedom but is slower.
    lengths = (200, 250, 300, 350, 400, 450, 500)
    # For each length we perform a few independent restarts.
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # Number of restarts adapts to the remaining time.
        # Roughly allocate a few seconds per restart.
        time_per_restart = 15.0
        max_restarts = max(1, int((deadline - time.monotonic()) // time_per_restart))
        max_restarts = min(max_restarts, 4)  # bound to keep total runtime reasonable

        for restart in range(max_restarts):
            if time.monotonic() >= deadline:
                break

            # ---- initialise a random feasible vector --------------------
            f = _rng.random(n) + 0.05        # avoid exact zeros
            f = f / np.sum(f) * target_sum

            # ---- LP refinement -----------------------------------------
            f_refined = _lp_refine(
                f=f,
                target_sum=target_sum,
                deadline=deadline,
                mix_start=0.3,
                mix_decay=0.95,
                max_iters=120,
            )
            sc_ref = _score(f_refined.tolist())
            if sc_ref < overall_best_score:
                overall_best_score = sc_ref
                overall_best_seq = f_refined.copy()
                print(f"[global] n={n:3d} restart={restart} LP best score={sc_ref:.6f}")

            # ---- local polishing ----------------------------------------
            if time.monotonic() < deadline:
                polished_seq, sc_polished = _targeted_local_search(
                    f_refined, deadline, _rng
                )
                if sc_polished < overall_best_score:
                    overall_best_score = sc_polished
                    overall_best_seq = polished_seq.copy()
                    print(f"[global] n={n:3d} restart={restart} polished score={sc_polished:.6f}")

            # Early exit if we have already reached a very good bound.
            if overall_best_score < 1.45:
                break
        if overall_best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # 3) Fallback: pure random search + local polishing if no LP support.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        print("[fallback] LP unavailable – using pure random search")
        for n in lengths:
            if time.monotonic() >= deadline:
                break
            target_sum = math.sqrt(2.0 * n)
            seq_rand, sc_rand = _random_search(
                n=n,
                target_sum=target_sum,
                deadline=deadline,
                rng=_rng,
            )
            if seq_rand is not None and sc_rand < overall_best_score:
                overall_best_seq = seq_rand.copy()
                overall_best_score = sc_rand
                print(f"[fallback] n={n} random best score={sc_rand:.6f}")

        # final polish of the best random candidate
        if overall_best_seq is not None and time.monotonic() < deadline:
            overall_best_seq, overall_best_score = _targeted_local_search(
                overall_best_seq, deadline, _rng
            )

    # ------------------------------------------------------------------
    # 4) Final sanity checks and return.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # deterministic safe fallback – uniform sequence of length 256
        n_def = 256
        overall_best_seq = np.full(n_def, 1.0, dtype=np.float64)

    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)
    return [float(x) for x in overall_best_seq]
