def hill_climb(a, n_iters, target_sum, rng, deadline):
    n = len(a)
    best = a.copy()
    best_score = _score(a)
    current = a.copy()
    cur_score = best_score

    for i in range(n_iters):
        if time.time() >= deadline:
            break
        # propose new sequence
        src = rng.integers(n)
        if current[src] <= 1e-12:
            continue
        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)
        # transfer amount: random fraction up to 30% of src
        frac = rng.random() * 0.3
        delta = current[src] * frac
        new_seq = current.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta
        # keep within [0,1000] already.
        # optionally add small Gaussian noise to all entries
        # compute score
        sc = _score(new_seq)
        # acceptance: if better or with probability exp(-(sc - cur_score)/temp)
        # schedule temp decreasing
        # we can use simple no-annealing: accept if better; else reject.
        if sc < cur_score:
            current = new_seq
            cur_score = sc
            if sc < best_score:
                best_score = sc
                best = new_seq.copy()
                # maybe print
                print("[hill] n={} iter={} score={:.6f}".format(n, i, sc))
                if best_score < 1.503:
                    break
        else:
            # sometimes accept worsening with small probability
            # Temperature schedule: start 0.05, decay linearly
            pass
    return best, best_score

"""
Greedy hill‑climbing search for a low‑value non‑negative step function.

The optimizer repeatedly transfers a small amount of ℓ¹‑mass from a
“heavy” index to a “light” index.  In roughly 40 % of the iterations the
move is *targeted*: we first compute the shift where the auto‑convolution
`f*f` attains its current maximum and move mass away from the index that
contributes most to that peak.  The remaining iterations use a simple
random transfer of mass between two indices.

The procedure is run for several problem sizes (200 … 1000).  For each
size we start from the current global best (or from a random vector), fix
the ℓ¹‑norm to √(2·n) – the normalisation used by `evaluate_sequence` –
and then perform a large number of hill‑climb steps while respecting the
overall time budget.  After the directed search a short burst of random
perturbations is applied to escape possible plateaus.

The best sequence found (minimum value of `evaluate_sequence`) is
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# Basic utilities
# ----------------------------------------------------------------------
def _score(seq):
    """Wrap the provided scorer – works with list or numpy array."""
    return float(evaluate_sequence(seq))


def _normalize(seq, target_sum):
    """Scale ``seq`` so that its ℓ¹‑norm equals ``target_sum``."""
    s = np.sum(seq)
    if s <= 0.0:
        # fallback to a uniform vector
        seq = np.full_like(seq, target_sum / seq.size, dtype=np.float64)
        return seq
    return seq * (target_sum / s)


# ----------------------------------------------------------------------
# Move generators
# ----------------------------------------------------------------------
def _targeted_move(seq, step_factor, rng):
    """
    Produce a candidate by moving a fraction of the mass from the index
    that contributes most to the current maximal auto‑convolution to the
    index that contributes least.
    Returns ``None`` if a sensible move cannot be constructed.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    # partner[j] is the entry that, together with seq[j], forms a term of
    # the convolution at shift ``k``.
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1)
    j_range = k - i_range

    partner = np.zeros(n, dtype=seq.dtype)
    partner[i_range] = seq[j_range]

    contrib = seq * partner
    src = int(np.argmax(contrib))
    dst = int(np.argmin(contrib))

    if src == dst or seq[src] <= 0.0:
        return None

    delta = seq[src] * (rng.random() * step_factor)
    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    return new_seq


def _random_move(seq, step_factor, rng):
    """
    Produce a candidate by moving a random fraction of the mass from a
    randomly chosen source index to a distinct destination index.
    Returns ``None`` if the source happens to be zero.

    src = rng.integers(n)
    if seq[src] <= 0.0:
        return None
    dst = rng.integers(n)
    while dst == src:
        dst = rng.integers(n)

    delta = seq[src] * (rng.random() * step_factor)
    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    return new_seq

            break

        target_sum = math.sqrt(2.0 * n)

        # ----- initialise a candidate for this size -----------------
        if best_seq is not None and best_seq.shape[0] == n:
            cand = best_seq.copy()
        elif best_seq is not None and best_seq.shape[0] < n:
            cand = np.zeros(n, dtype=np.float64)
            cand[:best_seq.shape[0]] = best_seq
        else:
            cand = rng.random(n) + 0.1

        cand = _normalize(cand, target_sum)

        # ----- hill‑climbing -------------------------------------------------
        # Allocate more iterations for smaller n (the same wall‑clock time)
        max_iters = int(20000 / (n // 100 + 1))
        cand, cand_score = _hill_climb(cand, target_sum, max_iters, rng, deadline)

        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            print("[best] new global best from n={} score={:.6f}".format(n, cand_score))
            if best_score < 1.503:
                break

        # ----- short random‑perturbation stage ------------------------------
        for _ in range(200):
            if time.time() >= deadline:
                break
            pert = cand + rng.normal(scale=target_sum * 0.02, size=n)
            pert = np.clip(pert, 0.0, 1000.0)
            pert = _normalize(pert, target_sum)
            sc = _score(pert.tolist())
            if sc < best_score:
                best_seq = pert.copy()
                best_score = sc
                cand = pert
                print("[perturb] n={} improved to {:.6f}".format(n, sc))
                if best_score < 1.503:
                    break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_score = _score(best_seq.tolist())
        print("[fallback] uniform sequence, score", best_score)

    print("Finished search – best score:", best_score, "length:", len(best_seq))
    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
