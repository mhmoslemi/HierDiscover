            j = rng.choice(low_indices)
        else:
            j = rng.integers(0, n)
        if i == j:
            j = (i+1) % n
        # maximum move size (fraction of source)
        max_move = cur[i] * step_scale
        if max_move <= 0:
            continue
        delta = rng.uniform(0, max_move)
        # propose new
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # Clip
        new = np.clip(new, 0.0, 1000.0)
        # Normalize sum to target (optional). Since scaling invariant, we can renorm to keep sum.
        # We want sum to stay same; after move sum unchanged; but due to clipping it may change small.
        # So renorm:
        s = new.sum()
        target_sum = cur.sum()
        if s > 0:
            new = new * (target_sum / s)
        # Score
        new_score = _score(new.tolist())
        # acceptance
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / max(t, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
                print(f"[opt] new best {best_score:.6f} it={iteration}")
        iteration += 1
    return best, best_score
```

"""
Propose a low‑value step sequence.

Algorithm
---------

1. **Base LP phase (n ≤ 400).**
   * For several modest sizes (200–400) we run a small linear‑program
     refinement.  With a fixed vector `f` we build the convolution matrix
     `A` (so that `A @ g == f * g`).  The LP maximises the ℓ¹‑mass of a
     new vector `g` under the constraints `g ≥ 0` and `A @ g ≤ M`,
     where `M = max(f*f)`.  After solving, `g` is rescaled to the target
     ℓ¹‑norm `√(2n)` and mixed into `f` with a decreasing step size.
   * The LP iterates (up to ~80 times) and monotonically drives down the
     worst auto‑convolution value `M`.  The best vector found is kept.

2. **Scale‑up & simulated annealing.**
   * The best vector from the LP phase is linearly interpolated to a
     larger size (up to 1000).  Interpolation preserves the overall shape
     while adjusting the ℓ¹‑norm to `√(2·N)`.
   * A custom annealing walk refines the sequence:
        – At each step we compute the convolution and locate its maximum.
        – We sample a source index weighted by its contribution to the
          maximal entry and a destination index from low‑mass locations.
        – A small amount of mass is transferred from source to destination.
        – The move is accepted if it improves the score, otherwise with
          probability `exp(-(Δscore)/T)`, where the temperature `T`
          decays exponentially.
        – After each move the vector is re‑normalised to keep the ℓ¹‑norm
          fixed (the evaluation is scale‑invariant, but keeping the norm
          constant helps the LP bound).

3. **Fallback.**
   * If `cvxpy` (or a suitable solver) is unavailable the algorithm
     skips the LP phase and starts directly from a random vector,
     proceeding to the annealing stage.

The routine respects the 1000 s budget (a small safety margin is kept)
and returns the best sequence encountered.  All values are clipped to
`[0, 1000]` as required by the evaluator.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global random generator – will be initialised in `propose_candidate`.
_rng = None  # type: np.random.Generator


# ----------------------------------------------------------------------
def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
def conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the dense convolution matrix A such that (A @ g) == f * g.
    Shape is (2*len(f)-1, len(f)).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    return A

def _lp_refine_custom(
    n: int,
    target_sum: float,
    f_start: np.ndarray,
    deadline: float,
    mix_start: float = 0.5,
    mix_end: float = 0.05,
    max_iters: int = 80,
) -> tuple[np.ndarray, float]:
    """
    LP refinement for a fixed length `n`.

    Parameters
    ----------
    n : int
        Length of the sequence.
    target_sum : float
        Desired ℓ¹‑norm (normally √(2·n)).
    f_start : np.ndarray
        Starting feasible vector (non‑negative, already normalised).
    deadline : float
        Absolute time (as returned by `time.monotonic()`) after which we must stop.
    mix_start, mix_end : float
        Step‑size schedule; the mixing coefficient decays exponentially from
        `mix_start` to `mix_end` over the iterations.
    max_iters : int
        Upper bound on LP iterations.

    Returns
    -------
    best_seq : np.ndarray
        Best sequence found (may be the start vector if LP is unavailable).
    best_score : float
        Its evaluation score.
    """
    global _rng
    f = f_start.copy()
    best_seq = f.copy()
    best_score = _score(f.tolist())

    # Try importing cvxpy once; if unavailable we skip LP entirely.
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return best_seq, best_score

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        M = float(np.max(np.convolve(f, f)))   # worst auto‑convolution entry
        A = conv_matrix(f)

        # Linear program: maximise Σ g_i  subject to g ≥ 0, A@g ≤ M
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:  # pragma: no cover
            try:
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
            except Exception:
                break

        if g.value is None:
            break

        g0 = np.maximum(g.value, 0.0)
        s_g = g0.sum()
        if s_g < 1e-12:
            break

        # Rescale to the target ℓ¹‑norm.
        g_rescaled = g0 * (target_sum / s_g)

        # Mixing coefficient decays exponentially.
        mix = mix_start * (mix_end / mix_start) ** (it / max(1, max_iters - 1))

        f_new = (1.0 - mix) * f + mix * g_rescaled
        f_new = np.clip(f_new, 0.0, 1000.0)

        sc = _score(f_new.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = f_new.copy()
            print(f"[LP] n={n:3d} it={it:2d} mix={mix:.4f} score={sc:.6f}")

        # Early exit if we already beat a very good bound.
        if best_score < 1.48:
            break

        f = f_new

    return best_seq, best_score


# ----------------------------------------------------------------------
def _resample_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linearly interpolate an existing sequence to a new length `new_n`,
    then renormalise to the target ℓ¹‑norm √(2·new_n).
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, 1000.0)

    target_sum = math.sqrt(2.0 * new_n)
    s = new_seq.sum()
    if s > 0:
        new_seq *= target_sum / s
    else:
        new_seq = np.full(new_n, target_sum / new_n)

    return new_seq


# ----------------------------------------------------------------------
def _anneal_optimize(
    init_seq: np.ndarray,
    deadline: float,
    start_temp: float = 0.2,
    min_temp: float = 1e-6,
    decay: float = 0.995,
    step_scale: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local optimisation.

    Parameters
    ----------
    init_seq : np.ndarray
        Starting non‑negative sequence (already normalised).
    deadline : float
        Absolute time limit.
    start_temp, min_temp, decay : float
        Temperature schedule (geometric decay).
    step_scale : float
        Maximum fraction of the source value that may be moved in one step.

    Returns
    -------
    best_seq, best_score
    """
    global _rng
    cur = init_seq.copy()
    n = cur.shape[0]
    target_sum = cur.sum()                # should already equal √(2n)
    cur_score = _score(cur.tolist())
    best = cur.copy()
    best_score = cur_score

    temperature = start_temp
    it = 0

    while time.monotonic() < deadline:
        # update temperature
        temperature = max(min_temp, temperature * decay)

        # Full convolution and locate the maximal entry.
        conv = np.convolve(cur, cur)
        k_max = int(np.argmax(conv))

        # Compute per‑index contributions to the maximal entry.
        contrib = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = cur[i] * cur[j]

        # Choose source index (biased towards large contribution).
        if contrib.sum() > 0:
            probs = contrib / contrib.sum()
            i = _rng.choice(n, p=probs)
        else:
            i = _rng.integers(0, n)

        # Choose destination index from the low‑mass tail.
        low_thr = np.percentile(cur, 20)
        low_idxs = np.where(cur <= low_thr)[0]
        if low_idxs.size > 0:
            j = _rng.choice(low_idxs)
        else:
            j = _rng.integers(0, n)

        if i == j:
            j = (i + 1) % n

        # Propose a transfer from i → j.
        max_transfer = cur[i] * step_scale
        if max_transfer <= 0:
            continue
        delta = _rng.uniform(0.0, max_transfer)

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # Clip and re‑normalise to keep the ℓ¹‑norm fixed.
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0:
            cand *= target_sum / s

        cand_score = _score(cand.tolist())

        # Accept if better, otherwise with Metropolis probability.
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = _rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[anneal] it={it:5d} temp={temperature:.6f} score={cand_score:.6f}")

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    The routine runs a small LP phase on modest lengths (≤400),
    then progressively up‑scales the best result and refines it
    with a simulated‑annealing walk.  It returns the best sequence
    found before the 1000‑second budget expires.
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # Reserve a small safety margin.
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – LP refinement on modest sizes.
    # ------------------------------------------------------------------
    lp_sizes = (200, 250, 300, 350, 400)
    for n in lp_sizes:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # Initialise: use the current overall best if it has the right size,
        # otherwise start from a fresh random vector.
        if overall_best_seq is not None and len(overall_best_seq) == n:
            f0 = overall_best_seq.copy()
        else:
            # Random start, avoid exact zeros.
            f0 = _rng.random(n) + 0.1
            f0 = f0 / f0.sum() * target_sum

        seq_lp, score_lp = _lp_refine_custom(
            n=n,
            target_sum=target_sum,
            f_start=f0,
            deadline=deadline,
            mix_start=0.5,
            mix_end=0.05,
            max_iters=80,
        )

        if score_lp < overall_best_score:
            overall_best_score = score_lp
            overall_best_seq = seq_lp.copy()
            print(f"[global] LP best n={n} score={score_lp:.6f}")

        # Early exit if we have already crossed the target.
        if overall_best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 2 – upscale + annealing on larger sizes.
    # ------------------------------------------------------------------
    upscale_sizes = (500, 600, 800, 1000)
    for n in upscale_sizes:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # Build an initial guess by interpolating the current best.
        if overall_best_seq is not None:
            f0 = _resample_seq(overall_best_seq, n)
        else:
            f0 = _rng.random(n) + 0.1
            f0 = f0 / f0.sum() * target_sum

        # Brief (optional) LP refinement for n ≤ 600 – it is cheap enough.
        if n <= 600:
            f0, _ = _lp_refine_custom(
                n=n,
                target_sum=target_sum,
                f_start=f0,
                deadline=deadline,
                mix_start=0.3,
                mix_end=0.02,
                max_iters=30,
            )

        # Annealing (the heavy work).
        seq_opt, score_opt = _anneal_optimize(
            init_seq=f0,
            deadline=deadline,
            start_temp=0.25,
            min_temp=1e-7,
            decay=0.997,
            step_scale=0.20,
        )

        if score_opt < overall_best_score:
            overall_best_score = score_opt
            overall_best_seq = seq_opt.copy()
            print(f"[global] anneal best n={n} score={score_opt:.6f}")

        # If we have comfortably reached the target, stop early.
        if overall_best_score < 1.50:
            break

    # ------------------------------------------------------------------
    # Fallback – in the very unlikely event nothing was produced.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 256
        overall_best_seq = np.full(n, 1.0, dtype=np.float64)
        overall_best_score = _score(overall_best_seq.tolist())
        print("[fallback] using uniform sequence")

    # Ensure final clipping and convert to plain Python list.
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)
    return [float(x) for x in overall_best_seq]
