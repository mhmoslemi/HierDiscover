def propose_candidate(seed: int = 0, budget_s: float = 1100) -> list[float]:
    import time, math, numpy as np
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Initialize sequence
    if "height_sequence_1" in globals():
        f = np.array(height_sequence_1, dtype=np.float64)
    else:
        # simple geometric initialization
        n = 300
        r = 0.99
        seq = np.ones(n, dtype=np.float64)
        f = seq
    # clip and normalize sum
    f = np.clip(f, 0.0, 1000.0)
    n = len(f)
    target_sum = math.sqrt(2.0 * n)
    if f.sum() == 0:
        f = np.full(n, target_sum / n, dtype=np.float64)
    else:
        f = f / f.sum() * target_sum

    # evaluate current best
    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"[init] n={n} init score={best_score}")

    # helper to build convolution matrix
    def build_A(f_vec):
        n_local = len(f_vec)
        rows = np.zeros((2 * n_local - 1, n_local), dtype=np.float64)
        # fill rows
        for k in range(2 * n_local - 1):
            i_start = max(0, k - (n_local - 1))
            i_end = min(k, n_local - 1)
            # indices i range
            # j = k - i
            # rows[k, j] = f[i]
            rows[k, k - np.arange(i_start, i_end + 1)] = f_vec[i_start:i_end + 1]
        return rows

    iter_count = 0
    last_improve_iter = 0
    # mixing factor
    t = 0.5

    while time.monotonic() < deadline:
        # compute current conv and max
        conv = np.convolve(f, f)
        M = conv.max()
        # build A_ub
        A = build_A(f)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        # solve LP: max sum b => min -sum b
        c = -np.ones(len(f), dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True, 'disp': False})
        if not res.success:
            print(f"[LP] iteration {iter_count} LP failed, status: {res.message}")
            break
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            print("[LP] zero sum solution, stop")
            break
        # scale to target sum
        scale = target_sum / sum_b
        g = b_opt * scale

        # convex combination
        f_new = (1 - t) * f + t * g
        # renormalize to exact sum (protect rounding)
        f_new = f_new * (target_sum / f_new.sum())

        # evaluate new candidate
        score = float(evaluate_sequence(f_new.tolist()))
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            last_improve_iter = iter_count
            print(f"[iter {iter_count}] improved score={best_score:.6f}")

        # if new max not larger than current, accept, else possible still accept? According to theory, should never increase?
        f = f_new

        iter_count += 1

        # restart if stagnated
        if iter_count - last_improve_iter > 30:
            # random perturbation to escape plateau
            # add tiny Gaussian noise to f and renormalize
            noise = rng.normal(scale=0.01, size=n)
            f = np.maximum(f + noise, 0.0)
            f = f * (target_sum / f.sum())
            print(f"[restart] iter {iter_count}")
            last_improve_iter = iter_count

    # final best sequence (clipped)
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_f.tolist()]
    final_score = float(evaluate_sequence(best_seq))
    print("[final] best evaluation =", final_score)
    return best_seq

"""
A two‑stage optimizer for the step‑function score.

Stage 1 – Linear‑programming “balancing”.
Given a non‑negative sequence f with ℓ¹‑norm √(2·n) we compute
    M = max(f∗f).
We then solve the LP

    maximise   Σ b_j
    subject to  (f∗b)_k ≤ M   for every convolution index k,
                b_j ≥ 0.

The optimum b concentrates mass where f is small while never
exceeding the current worst autocorrelation M.  After scaling b to the
required ℓ¹‑norm we move a fraction 𝑡 of the current iterate towards the
scaled vector.  The theory guarantees that the maximal autocorrelation
does not increase.  Repeating this step (typically a few dozen times)
quickly drives the score down.

Stage 2 – Fine‑grained simulated‑annealing.
When the LP stage stalls we switch to a lightweight SA that repeatedly
transfers a tiny amount of mass between two indices and keeps the move
only if the maximum autocorrelation does not grow (or improves).  The
remaining time budget is spent here, yielding a final refinement.

The whole routine works for any length n (default 300) and respects the
security checks of ``evaluate_sequence``.  It returns the best sequence
found before the time budget expires.
"""

import time
import math
import numpy as np

# SciPy is available in the execution environment.
from scipy.optimize import linprog


def _initial_sequence(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Produce a feasible start vector of length *n* with ℓ¹‑norm √(2·n).
    If a previous construction ``height_sequence_1`` exists it is reused;
    otherwise a random positive vector is drawn.
    """
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=np.float64)
        if len(seq) != n:
            # truncate or pad with zeros to the requested length
            if len(seq) > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - len(seq))])
    else:
        # random start – uniform Dirichlet (exponential) distribution
        seq = rng.exponential(scale=1.0, size=n)

    # Clip to the interval accepted by the scorer and normalise the ℓ¹‑norm.
    seq = np.clip(seq, 0.0, 1000.0)
    target_sum = math.sqrt(2.0 * n)
    if seq.sum() == 0.0:
        seq = np.full(n, target_sum / n, dtype=np.float64)
    else:
        seq = seq / seq.sum() * target_sum
    return seq


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix A with entries
        A[k, j] = f[i]   where i + j = k  (i,j in 0…n‑1)
    so that (A @ b)_k = (f∗b)_k.
    """
    n = f.shape[0]
    rows = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals               # 0 ≤ j_vals < n
        rows[k, j_vals] = f[i_vals]
    return rows


def propose_candidate(seed: int = 0, budget_s: float = 1100) -> list[float]:
    """
    Search for a low‑value step sequence within the given time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin for the final printout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # 1️⃣  Choose a length.  300 gives a good trade‑off between quality
    #    and speed for the 1000 s budget.
    n = 300

    # --------------------------------------------------------------
    # 2️⃣  Initialise the candidate.
    f = _initial_sequence(rng, n)
    target_sum = math.sqrt(2.0 * n)

    # current best (the objective for a normalised vector equals max(conv))
    conv = np.convolve(f, f)
    cur_max = conv.max()
    best_f = f.copy()
    best_max = cur_max
    print(f"[init] n={n}  score={cur_max:.6f}")

    iter_num = 0
    last_improve = 0
    t = 0.5                         # convex‑combination weight

    # --------------------------------------------------------------
    # 3️⃣  Linear‑programming stage.
    #     Run while we still have more than ~30 s left – the rest is
    #     reserved for the SA polishing phase.
    while time.monotonic() < deadline - 30.0:
        # build the LP matrix for the current f
        A = _conv_matrix(f)
        M = cur_max                       # current worst autocorrelation
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        # maximise Σb  ⇔  minimise –Σb
        c = -np.ones(n, dtype=np.float64)

        # Solve the LP (highs is the default robust solver)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=(0, None), method="highs",
                      options={"presolve": True, "disp": False})
        if not res.success:
            # Numerical trouble – give up the LP stage.
            print(f"[LP] iteration {iter_num} failed: {res.message}")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0.0:                # should never happen, but stay safe
            break

        # Scale the LP solution to the required ℓ¹‑norm.
        scale = target_sum / sum_b
        g = b_opt * scale

        # Convex combination.
        f_new = (1.0 - t) * f + t * g
        # Renormalise (tiny rounding errors may appear)
        f_new *= target_sum / f_new.sum()

        # Evaluate the new maximum autocorrelation.
        conv_new = np.convolve(f_new, f_new)
        new_max = conv_new.max()

        # Accept only non‑increasing updates (the theory guarantees this,
        # but we keep a tiny tolerance for floating‑point noise).
        if new_max <= cur_max * (1.0 + 1e-12):
            f = f_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f.copy()
                last_improve = iter_num
                print(f"[LP it={iter_num:06d}] improved score={best_max:.6f}")
        else:
            # Unexpected increase – discard the step.
            pass

        iter_num += 1
        # occasional random re‑initialisation if we have been idle too long.
        if iter_num - last_improve > 200 and time.monotonic() < deadline - 30.0:
            f = _initial_sequence(rng, n)
            cur_max = np.convolve(f, f).max()
            last_improve = iter_num
            print(f"[restart] LP it={iter_num}")

    # --------------------------------------------------------------
    # 4️⃣  Simulated‑annealing polish (uses only a few thousand cheap moves)
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        print(f"[SA] polishing for ≈{remaining:.1f}s")
        f = best_f.copy()                       # start from the LP best
        cur_max = best_max
        best_sa_f = f.copy()
        best_sa_max = cur_max

        # simple temperature schedule
        T_start = 1.0
        T_end = 1e-4
        start_sa = time.monotonic()
        it = 0
        while time.monotonic() - start_sa < remaining:
            elapsed = time.monotonic() - start_sa
            T = T_start * (T_end / T_start) ** (elapsed / remaining)

            # pick donor (must have some mass) and a random receiver
            i = rng.choice(np.where(f > 1e-12)[0]) if np.any(f > 1e-12) else rng.integers(0, n)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)

            # move a small fraction of the donor mass
            delta = f[i] * (rng.random() * 0.2 + 0.01)   # between 1 % and 21 %
            if delta <= 0.0:
                continue

            # apply move
            f[i] -= delta
            f[j] += delta

            new_max = np.convolve(f, f).max()

            # Metropolis acceptance
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = math.exp(-(new_max - cur_max) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_sa_max:
                    best_sa_max = new_max
                    best_sa_f = f.copy()
                # occasionally report progress
                if it % 10000 == 0:
                    print(f"[SA it={it:06d}] score={cur_max:.6f} T={T:.2e}")
            else:
                # roll back
                f[i] += delta
                f[j] -= delta

            it += 1

        # keep the best from SA if it beats the LP result
        if best_sa_max < best_max:
            best_max = best_sa_max
            best_f = best_sa_f.copy()
            print(f"[SA final] improved to {best_max:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Return the best sequence (clipped to the scorer’s domain).
    best_seq = [
        float(max(0.0, min(1000.0, x))) for x in best_f.tolist()
    ]
    final_score = float(evaluate_sequence(best_seq))
    print("[final] best evaluation =", final_score)
    return best_seq
