"""
Improved step‑function search.

Algorithm
---------
1.  For a set of lengths `n` (200 … 800) we keep a vector `f`
    with ℓ¹‑norm `√(2 n)`.  The current worst auto‑convolution entry is
    `M = max(f*f)`.

2.  **Linear‑programming step** – we solve the LP

        minimize   M
        subject to A_f·g ≤ M,
                   Σ g = √(2 n),
                   g ≥ 0,

    where `A_f` is the convolution matrix with fixed left operand `f`.
    The LP gives a new feasible vector `g` whose bound `M` is as low as
    possible while preserving the ℓ¹‑norm.

3.  **Optimal mixing** – the objective depends only on the convex
    combination `h(t) = (1‑t) f + t g`.  For any `t∈[0,1]`

        conv_h_h = (1‑t)²·conv_f_f + 2 t (1‑t)·conv_f_g + t²·conv_g_g .

    The maximum of this quantity is a convex function of `t`; we locate
    the optimal `t` with a tiny ternary search (≈30 evaluations).

    The new `f` becomes `h(t*)` and is re‑scaled to keep the exact target
    ℓ¹‑norm.

4.  The pair (LP → mixing) is repeated (≈40 iterations for the smaller
    sizes, fewer for the larger ones) until the time budget expires or
    the score falls below a preset threshold.

5.  After the LP phase a cheap stochastic “local search” swaps a tiny
    amount of mass between two random positions while preserving the
    total sum.  Improvements are kept.

6.  The best sequence over all lengths is returned.  If the LP solver
    cannot be imported the routine falls back to a pure random walk.

The program respects the 1000 s budget, uses only NumPy (and SciPy or
CVXPY when available) and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global RNG – will be (re)initialised in `propose_candidate`.
_rng = None


def conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense convolution matrix A (size (2n‑1) × n) such that
    (A @ g) == np.convolve(f, g) for any vector g of length n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # contributions from f[i] where i in [i_start, i_end]
        # column index j = k - i
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    return A


def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq))


def _solve_lp_min_M(A: np.ndarray, target_sum: float, deadline: float):
    """
    Solve   min M   s.t. A @ g ≤ M, Σ g = target_sum, g ≥ 0, M ≥ 0.
    Tries SciPy's HiGHS solver first; falls back to CVXPY/ECOS/​SCS.
    Returns (g, M) or (None, None) on failure.
    """
    n = A.shape[1]
    rows = A.shape[0]

    # ----------- SciPy (preferred) ------------------------------------
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0                     # minimise M

        # A_ub @ [g; M] <= 0   <=>   A @ g - M <= 0
        A_ub = np.hstack([A, -np.ones((rows, 1))])
        b_ub = np.zeros(rows)

        # equality Σ g = target_sum
        A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
        b_eq = np.array([target_sum])

        bounds = [(0.0, None)] * n + [(0.0, None)]

        # HiGHS is fast and reliable for moderate sizes.
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.monotonic())},
        )
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            M = float(res.x[-1])
            return g, M
    except Exception:  # pragma: no cover
        pass

    # ----------- CVXPY fallback ----------------------------------------
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None, None

    g_var = cp.Variable(n)
    M_var = cp.Variable()
    constraints = [
        g_var >= 0,
        A @ g_var <= M_var,
        cp.sum(g_var) == target_sum,
        M_var >= 0,
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000, verbose=False)
        except Exception:
            return None, None

    if g_var.value is None:
        return None, None

    g = np.maximum(g_var.value, 0.0)
    M = float(M_var.value)
    return g, M


def _local_search(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic walk that preserves total ℓ¹‑mass.
    Randomly transfers a tiny amount of mass between two positions.
    """
    best = seq.copy()
    best_score = _score(best.tolist())
    n = best.shape[0]

    while time.monotonic() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # Propose a small transfer proportional to the source magnitude.
        delta = _rng.normal(loc=0.0, scale=0.02) * max(1e-9, best[i])
        if delta == 0.0:
            continue
        cand_i = best[i] + delta
        cand_j = best[j] - delta
        if cand_i < 0.0 or cand_j < 0.0:
            continue
        cand = best.copy()
        cand[i] = cand_i
        cand[j] = cand_j
        sc = _score(cand.tolist())
        if sc < best_score:
            best_score = sc
            best = cand
            print(f"[local] score={sc:.6f}")
    return best, best_score

    # 1) Choose problem sizes.
    # ------------------------------------------------------------------
    n_list = [200, 300, 400, 500, 600, 800]

    # If a previous construction is available, keep it for possible reuse.
    previous_seq = globals().get("height_sequence_1")

    for n in n_list:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Initialise f.
        if previous_seq is not None and len(previous_seq) == n:
            f = np.array(previous_seq, dtype=np.float64)
            # Rescale to the exact ℓ¹‑norm required by the scorer.
            f = f / max(np.sum(f), 1e-12) * target_sum
        else:
            f = _rng.random(n) + 0.1   # avoid exact zeros
            f = f / np.sum(f) * target_sum

        # Record the starting point.
        sc_start = _score(f.tolist())
        if sc_start < overall_best_score:
            overall_best_score = sc_start
            overall_best_seq = f.copy()
            print(f"[init] n={n:3d} score={sc_start:.6f}")

        # Number of LP iterations – the larger n, the fewer iterations we afford.
        max_iters = 80 if n <= 400 else 40

        for it in range(max_iters):
            if time.monotonic() >= deadline:
                break

            # Current worst auto‑convolution entry.
            conv_f_f = np.convolve(f, f)
            M_f = float(conv_f_f.max())

            # Build convolution matrix for the current f.
            A_f = conv_matrix(f)

            # Solve the LP: minimise the bound M.
            g, M_opt = _solve_lp_min_M(A_f, target_sum, deadline)
            if g is None:
                # LP failed – abort this size.
                break

            # Compute the three convolutions that appear in the quadratic form.
            conv_f_g = np.convolve(f, g)   # = A_f @ g
            conv_g_g = np.convolve(g, g)

            # ----- line search for the optimal convex combination -----
            def max_conv(t: float) -> float:
                """Maximum entry of the convolution of (1‑t)f + t g with itself."""
                comb = ((1.0 - t) ** 2) * conv_f_f + (2.0 * t * (1.0 - t)) * conv_f_g + (t ** 2) * conv_g_g
                return float(comb.max())

            # ternary search on [0,1] (convex function)
            lo, hi = 0.0, 1.0
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if max_conv(m1) < max_conv(m2):
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0

            # Check the interval ends – sometimes the optimum is at a bound.
            for tt in (0.0, 1.0, t_opt):
                if max_conv(tt) < max_conv(t_opt):
                    t_opt = tt

            # New candidate vector (convex combination).
            f_new = (1.0 - t_opt) * f + t_opt * g
            # Re‑scale to keep the exact ℓ¹‑norm (the scorer is scale‑invariant,
            # but keeping the norm fixed makes the bound meaningful).
            sum_f_new = np.sum(f_new)
            if sum_f_new > 0:
                f_new *= (target_sum / sum_f_new)

            # Clip to the allowed interval – never needed for our magnitudes,
            # but kept for safety.
            f_new = np.clip(f_new, 0.0, 1000.0)

            # Evaluate.
            sc = _score(f_new.tolist())
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = f_new.copy()
                print(f"[LP] n={n:3d} it={it:02d} t={t_opt:.4f} score={sc:.6f}")

            # Early exit if we have already crossed a very good threshold.
            if overall_best_score < 1.45:
                break

            f = f_new

        # End of LP iterations for this size.
        if overall_best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # 2) Light‑weight local polishing (uses whatever time remains).
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_score = _local_search(overall_best_seq, deadline)
        if polished_score < overall_best_score:
            overall_best_score = polished_score
            overall_best_seq = polished_seq
            print(f"[global] after polishing: {polished_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final safety net.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # deterministic fallback – a constant sequence of length 256.
        overall_best_seq = np.ones(256, dtype=np.float64)

    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    return [float(x) for x in overall_best_seq]
