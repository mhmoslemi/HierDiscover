        s = f.sum()
        if s > 0:
            f = f / s
        else:
            f = rng.random(n)
            f = f / f.sum()
    else:
        f = rng.random(n)
        f = f / f.sum()

    f = (f + f[::-1]) / 2

"""
A fast gradient‑descent / simulated‑annealing search for a low‑value step
function.  The score to minimise is

    2·n·max(f*f) / (∑f)²

which is invariant under scaling of the vector f, so we normalise all
candidates to ℓ¹‑norm = 1.  The algorithm works as follows.

1.  Randomly choose a length n (200 … 800) and a random start vector.
    If the global variable ``height_sequence_1`` is present and has the
    same length it is used as an additional start.

2.  Repeatedly minimise the maximum entry of the auto‑convolution by a
    simple projected gradient descent.  The gradient of the quantity
    ``max(f*f)`` is easy to write down:
        ∂max / ∂f_i = 2·∑_{k∈K} f_{k−i}
    where K is the set of indices at which the convolution attains its
    maximum.  After each gradient step we (i) clip to non‑negative,
    (ii) enforce symmetry f = (f+f[::-1])/2 (the optimum can be taken
    symmetric without loss of generality) and (iii) renormalise the sum
    to 1.

3.  The step size is decreased exponentially; a tiny back‑tracking line
    search accepts the first step that actually improves the score.

4.  When the gradient makes no progress we inject a small random
    perturbation to escape plateaus.  The best vector found for the
    current length is kept.

5.  After the allotted time (≈ 1000 s) a short simulated‑annealing
    refinement is performed on the overall best candidate.

All heavy work uses only NumPy and the Fast‑Fourier‑Transform for the
convolution, so thousands of iterations per length fit easily into the
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Scoring utilities (fast, scale‑invariant)
# ----------------------------------------------------------------------


def _conv_full_fft(f: np.ndarray) -> np.ndarray:
    """Full auto‑convolution of a non‑negative vector using an FFT."""
    n = f.shape[0]
    L = 2 * n - 1
    F = np.fft.rfft(f, n=L)
    conv = np.fft.irfft(F * F, n=L)
    return conv


def _compute_score(f: np.ndarray) -> float:
    """Return 2·n·max(f*f) for a vector with ℓ¹‑norm = 1."""
    n = f.shape[0]
    conv = _conv_full_fft(f)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_max_conv(f: np.ndarray) -> np.ndarray:
    """
    Gradient of ``max(f*f)`` w.r.t. f.
    For all k where the convolution attains its maximum we add
    2·f_{k-i} to the i‑th component.
    """
    n = f.shape[0]
    L = 2 * n - 1
    # convolution (real)
    F = np.fft.rfft(f, n=L)
    conv = np.fft.irfft(F * F, n=L)
    max_val = conv.max()
    # indices where the maximum is attained (within a tiny tolerance)
    eps = 1e-12
    max_idx = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(f)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += 2.0 * f[j]
    return grad


# ----------------------------------------------------------------------
# Line‑search step (projected gradient step)
# ----------------------------------------------------------------------


def _line_search_step(f: np.ndarray, grad: np.ndarray,
                      eta: float, cur_score: float) -> tuple[np.ndarray, float]:
    """
    Try decreasing step lengths η, η/2, η/4 … until an improvement is
    observed.  Returns the best candidate (or the original f if none
    improves) together with its score.
    """
    step = eta
    for _ in range(8):
        cand = f - step * grad
        cand = np.maximum(cand, 0.0)
        # enforce symmetry – the optimum can be taken symmetric
        cand = (cand + cand[::-1]) * 0.5
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            # extremely unlikely – fall back to a fresh random vector
            cand = np.random.rand(f.shape[0])
            cand /= cand.sum()
        cand_score = _compute_score(cand)
        if cand_score < cur_score - 1e-14:
            return cand, cand_score
        step *= 0.5
    return f, cur_score


# ----------------------------------------------------------------------
# Optimisation for a single length n
# ----------------------------------------------------------------------


def _optimize_one_length(n: int, rng: np.random.Generator,
                         deadline: float,
                         init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Gradient‑descent optimisation for a fixed length n.
    Returns the best sequence encountered and its score.
    """
    max_iters = 20000                     # generous upper bound
    eta_start = 0.5
    eta_end = 0.001

    # ------------------------------------------------------------------
    # Initialise a feasible vector (ℓ¹‑norm = 1, non‑negative, symmetric)
    # ------------------------------------------------------------------
    if init_seq is not None:
        f = np.asarray(init_seq, dtype=np.float64).copy()
    else:
        f = rng.random(n) + 1e-3

    f = np.clip(f, 0.0, 1000.0)
    f = (f + f[::-1]) * 0.5                # enforce symmetry
    s = f.sum()
    if s > 0.0:
        f /= s
    else:
        f = rng.random(n)
        f /= f.sum()

    cur_score = _compute_score(f)
    best_seq = f.copy()
    best_score = cur_score

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # --------------------------------------------------------------
        # Gradient step
        # --------------------------------------------------------------
        grad = _gradient_max_conv(f)

        # schedule a geometrically decreasing step size
        frac = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** frac

        f_new, new_score = _line_search_step(f, grad, eta, cur_score)

        if new_score < cur_score - 1e-14:
            # improvement – accept the move
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = f_new.copy()
                print(f"[GD] n={n:4d} it={it:5d} score={new_score:.6f}")
        else:
            # stagnation – inject a tiny random perturbation
            if rng.random() < 0.02:
                f = f + rng.normal(scale=0.01, size=n)
                f = np.maximum(f, 0.0)
                f = (f + f[::-1]) * 0.5
                f /= f.sum()
                cur_score = _compute_score(f)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = f.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Fast simulated‑annealing polish (optional, runs at the end)
# ----------------------------------------------------------------------


def _simulated_annealing(seq: np.ndarray, cur_score: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple SA that perturbs a single coefficient at a time.
    Uses the fast internal scorer; the final sequence will still be
    evaluated with the official ``evaluate_sequence`` later.
    """
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0.0:
            break
        frac = elapsed / total
        temp = temp_start * (temp_end / temp_start) ** frac

        # propose a change of a single coordinate
        idx = rng.integers(len(cur_seq))
        delta = rng.normal(scale=0.5) * temp * 5.0
        cand = cur_seq.copy()
        cand[idx] = float(np.clip(cand[idx] + delta, 0.0, 1000.0))

        # keep symmetry & normalisation
        cand = (cand + cand[::-1]) * 0.5
        cand /= cand.sum()
        cand_score = _compute_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(
                -(cand_score - cur_score) / max(temp, 1e-12)):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

        # occasional global jitter to avoid being trapped
        if rng.random() < 0.001:
            pert = np.maximum(cur_seq + rng.normal(scale=0.1, size=len(cur_seq)), 0.0)
            pert = (pert + pert[::-1]) * 0.5
            pert /= pert.sum()
            cur_seq = pert
            cur_score = _compute_score(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The routine respects the 1000 s budget, runs several gradient‑descent
    passes for different lengths and finishes with a short annealing
    phase.  It returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Candidate lengths – larger n gives more freedom but costs a bit more
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # use the globally supplied construction if it has the right size
        init_global = None
        try:
            global height_sequence_1
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            if seq0.shape[0] == n:
                init_global = seq0.copy()
        except Exception:
            pass

        # run a few independent restarts for this length
        restarts = 3
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break
            seq, score = _optimize_one_length(
                n=n,
                rng=rng,
                deadline=deadline,
                init_seq=init_global if r == 0 else None,
            )
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[global] new best n={n} restart={r} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with simulated annealing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq, best_score, deadline, rng
        )
        print(f"[final] best score after SA = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0, dtype=np.float64)

    # Convert to the plain list required by the harness
    return [float(x) for x in best_seq]
