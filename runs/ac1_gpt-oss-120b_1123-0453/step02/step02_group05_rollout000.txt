def project_onto_simplex(v, z):

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if len(rho)==0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

For a given α the gradient can be written in closed form using
convolutions, allowing fast O(n log n) updates.  After each gradient step the
vector is projected onto the simplex {x≥0, ∑x = √(2·n)}.  α is increased
gradually (from a low value that gives a smooth landscape to a high value
that focuses on the true maximum).  An occasional Metropolis‑style
acceptance of worsening moves and random restarts help to escape local
minima.

The routine runs until the supplied time budget (≈ 1000 s) expires and
returns the best sequence seen.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, ∑x = z}
def _project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum `z`."""
    if z < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Helper: initialise a feasible sequence with ℓ¹‑norm = √(2·n)
def _initial_sequence(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Return a non‑negative vector of length `n` with sum √(2·n).
    Starts from the global `height_sequence_1` if available,
    otherwise from a random vector.
    """
    target = math.sqrt(2.0 * n)

    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=np.float64)
        if base.size < n:
            base = np.pad(base, (0, n - base.size), constant_values=0.0)
        else:
            base = base[:n]
    else:
        # small random positive numbers; the scale does not matter because we project later
        base = rng.random(n)

    # project onto the required simplex
    base = _project_onto_simplex(base, target)
    return base


# --------------------------------------------------------------
# Simple wrapper for the supplied evaluator (already imported)
def _evaluate(seq: list[float]) -> float:
    """Return the score for a candidate sequence."""
    return float(evaluate_sequence(seq))


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """
    Gradient‑descent / simulated‑annealing hybrid to find a low‑value step
    sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate wall‑clock time budget (seconds).  The function stops a
        couple of seconds before the limit to guarantee a result.

    Returns
    -------
    list[float]
        The best sequence (plain Python floats) discovered within the time
        budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------
    # Choose a problem size.
    # Larger n gives more degrees of freedom, but also higher per‑iteration cost.
    n = 400  # feel free to adjust; 400 is a good trade‑off for ~1000 s

    # ----------------------------------------------------------
    # Initialise
    f = _initial_sequence(rng, n)               # feasible vector
    target_sum = math.sqrt(2.0 * n)

    # Compute the initial objective (max of auto‑convolution)
    conv = np.convolve(f, f)
    best_val = float(conv.max())
    best_f = f.copy()

    # For acceptance of uphill moves we keep the current max
    cur_max = best_val

    # Timing / schedule
    start_time = time.monotonic()
    total_time = deadline - start_time
    it = 0
    last_improve_it = 0

    # Main optimisation loop
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time

        # ---- 1) adjust smoothing parameter α (soft‑max temperature) ----
        # start with a smooth landscape (α≈5) and end with a sharp max (α≈400)
        alpha = 5.0 + 395.0 * frac  # linear schedule 5 → 400

        # ---- 2) compute auto‑convolution and its smoothed gradient ----
        conv = np.convolve(f, f)               # length 2n‑1
        max_conv = np.max(conv)

        # stable soft‑max weights: w_k ∝ exp(α·(conv_k – max_conv))
        exp_term = np.exp(alpha * (conv - max_conv))
        p = exp_term / np.sum(exp_term)        # shape (2n‑1,)

        # gradient: g_j = 2·∑_k p_k·f_{k‑j}
        # expressed as convolution of p with reversed f
        f_rev = f[::-1]
        grad_full = np.convolve(p, f_rev)      # length (2n‑1)+n‑1 = 3n‑2
        # slice the central n entries (indices n‑1 … n‑1+n‑1)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ---- 3) step size schedule ----
        # base step size proportional to average component size
        base_lr = 0.2 * target_sum / n          # ≈ 0.01–0.02 for our n
        lr = base_lr * (1.0 - frac) + 0.001     # decay to a small constant

        # ---- 4) gradient descent step with simplex projection ----
        f_candidate = f - lr * grad
        f_candidate = _project_onto_simplex(f_candidate, target_sum)

        # ---- 5) evaluate candidate ----
        conv_cand = np.convolve(f_candidate, f_candidate)
        cand_max = float(conv_cand.max())

        # Keep the best overall solution
        if cand_max < best_val - 1e-9:
            best_val = cand_max
            best_f = f_candidate.copy()
            last_improve_it = it
            print(f"[GD] it={it:06d} best={best_val:.6f} lr={lr:.5f} α={alpha:.1f}")

        # ---- 6) Metropolis acceptance (helps escape local minima) ----
        accept = False
        if cand_max <= cur_max:
            accept = True
        else:
            # temperature schedule for uphill moves (exponential cooling)
            T_start, T_end = 1.0, 1e-3
            T = T_start * (T_end / T_start) ** frac
            prob_up = math.exp(-(cand_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob_up:
                accept = True

        if accept:
            f = f_candidate
            cur_max = cand_max
        # else keep the old f and cur_max unchanged

        # ---- 7) occasional random restart if stagnation observed ----
        if it - last_improve_it > 80_000 and frac > 0.5:
            f = rng.random(n)
            f = _project_onto_simplex(f, target_sum)
            cur_max = float(np.convolve(f, f).max())
            last_improve_it = it
            print(f"[restart] it={it:06d} new_max={cur_max:.6f}")

        it += 1

    # ----------------------------------------------------------
    # Return the best sequence found (clipped to the scorer's accepted range)
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_f.tolist()]
    final_score = _evaluate(best_seq)
    print("[final] best evaluation =", final_score)
    return best_seq
