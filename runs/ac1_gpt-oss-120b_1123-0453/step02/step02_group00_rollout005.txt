def project_onto_simplex(v, z):
    # v: 1-d numpy array, target sum = z.
    # Returns projection onto {x >=0, sum x = z}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_onto_simplex(v, s):
    # v: np.ndarray
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _lp_improve(f, target_sum, deadline, max_iters=30):
    n = f.shape[0]
    t = 0.5
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        M = np.max(np.convolve(f, f))
        A = _conv_matrix(f)  # shape (2n-1, n)
        # Solve LP maximize sum(g) subject to A@g <= M, g>=0
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-8, reltol=1e-6)
        except Exception:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000, verbose=False)
        if g.value is None:

        g0 = np.maximum(g.value, 0.0)

        g_rescaled = g0 * (target_sum / sum_g)
        # mixing

        f_new = np.clip(f_new, 0, 1000)
        # Evaluate

        # update

"""
Propose a step‑function sequence with a low evaluation score.

The search proceeds in three stages inside the allocated time budget:

1️⃣  **LP‑based improvement** (optional, requires cvxpy):
    – For a fixed length n we keep the ℓ¹‑norm = √(2·n).
    – With the current vector `f` we compute M = max(f∗f).
    – A linear program maximises Σ gᵢ under the linear constraints
      (f∗g)ₖ ≤ M, gᵢ ≥ 0.  The solution `g` is rescaled to the same ℓ¹‑norm
      and mixed with `f` (convex combination).  Repeating this step drives M
      down.

2️⃣  **Gradient‑descent refinement**:
    – Work directly on the non‑convex objective M = max(f∗f).
    – At the current worst shift k* we take the sub‑gradient
      ∂M/∂fᵢ = f_{k*−i} (if the index is valid) and perform a line‑search
      step, then project back onto the simplex {x≥0, Σx = √(2·n)}.
    – A simple back‑tracking scheme adapts the step size; random jitter
      restarts the algorithm when progress stalls.

3️⃣  **Stochastic local search** (Gaussian perturbations) as a final polish.

All stages respect the security checks of `evaluate_sequence`.  The
function returns the best list of non‑negative heights found before the
deadline (default 1000 s)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional LP backend
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Returns the (2·n‑1) × n matrix A such that (A @ g)[k] = (f ∗ g)[k].
    This matrix is used in the LP step where f is fixed and the variable
    vector is g.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _score(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # evaluate_sequence works on native Python lists.
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008) in O(n log n).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero; return a zero vector.
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# LP‑based improvement (stage 1)
# ----------------------------------------------------------------------
def _lp_improve(
    f: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 60,
) -> np.ndarray:
    """
    Perform a few LP iterations that strictly decrease max(f∗f).
    Returns the improved vector (or the original one if LP fails).
    """
    if not _HAS_CVX:
        return f

    n = f.shape[0]
    t = 0.5  # mixing weight

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        M = float(np.max(np.convolve(f, f)))
        A = _conv_matrix(f)

        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        # ECOS is fast for these dense problems; fallback to SCS.
        try:
            prob.solve(
                solver=cp.ECOS,
                warm_start=True,
                max_iters=5000,
                abstol=1e-8,
                reltol=1e-6,
            )
        except Exception:  # pragma: no cover
            prob.solve(
                solver=cp.SCS,
                warm_start=True,
                max_iters=15000,
                verbose=False,
            )

        if g.value is None:
            break

        g0 = np.maximum(g.value, 0.0)
        sum_g = np.sum(g0)
        if sum_g < 1e-12:
            break

        g_rescaled = g0 * (target_sum / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, 1000.0)
        f = f_new

        # modest adaptation of t to keep progress stable
        if it % 15 == 14:
            recent = float(np.max(np.convolve(f, f)))
            if recent > M * 1.001:
                t = max(0.1, t * 0.8)

    return f


# ----------------------------------------------------------------------
# Gradient‑descent refinement (stage 2)
# ----------------------------------------------------------------------
def _gradient_optimize(
    f: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the objective max(f∗f) while staying on the
    ℓ¹‑simplex.  Returns the best vector found and its evaluation score.
    """
    best_f = f.copy()
    best_score = _score(best_f)
    # Initial step size as a fraction of the ℓ¹‑norm.
    step_size = target_sum * 0.05

    n = len(f)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # current worst convolution entry
        conv = np.convolve(best_f, best_f)
        max_val = float(conv.max())
        if max_val < 1.503:  # early exit if already good enough
            best_score = max_val
            break

        # indices where the maximum is attained (usually one)
        max_mask = conv >= max_val - 1e-12
        max_idx = np.where(max_mask)[0]

        # compute sub‑gradient of the max‑entry w.r.t. f
        grad = np.zeros_like(best_f)
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_arr = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_arr = k - i_arr
            grad[i_arr] += best_f[j_arr]

        # --- line search -------------------------------------------------
        improved = False
        trial_step = step_size
        for _ in range(12):
            cand = best_f - trial_step * grad
            # projection onto the simplex guarantees non‑negativity and sum
            cand = _project_onto_simplex(cand, target_sum)
            cand_score = _score(cand)

            if cand_score < best_score - 1e-12:
                best_f = cand
                best_score = cand_score
                step_size = trial_step * 1.2   # slightly enlarge successful step
                improved = True
                break
            else:
                trial_step *= 0.5

        if not improved:
            # shrink step size; if it becomes too tiny inject a random jitter
            step_size *= 0.5
            if step_size < 1e-8:
                rng = np.random.default_rng()
                jitter = rng.normal(scale=target_sum * 0.02, size=n)
                cand = best_f + jitter
                cand = np.clip(cand, 0.0, 1000.0)
                cand = _project_onto_simplex(cand, target_sum)
                cand_score = _score(cand)
                if cand_score < best_score:
                    best_f = cand
                    best_score = cand_score
                step_size = target_sum * 0.05

    return best_f, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (stage 3)
# ----------------------------------------------------------------------
def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbation hill‑climbing that respects the ℓ¹‑norm.
    """
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target_sum = float(np.sum(start_seq))

    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _score(cand)

        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # keep step size when improving
        else:
            # exponentially decay step size
            step *= 0.9999
            if step < 1e-7:
                step = step_initial

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The routine explores a range of lengths, performs an optional LP‑based
    improvement (if cvxpy is available), refines the result with a
    sub‑gradient method, and finally applies a short stochastic local search.
    It respects the global time budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # Consider several lengths – larger n gives more freedom.
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    # ------------------------------------------------------------------
    # Stage 1 + 2 – LP + gradient for each length
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        restarts = 4

        for restart in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ---- initialise -------------------------------------------------
            # try to reuse a user‑provided global sequence if compatible
            init_vec = None
            if "height_sequence_1" in globals():
                hs = globals()["height_sequence_1"]
                if isinstance(hs, (list, tuple)) and len(hs) >= n:
                    init_vec = np.array(hs[:n], dtype=np.float64)

            if init_vec is None:
                init_vec = rng.random(n) + 0.1

            # scale to the required ℓ¹‑norm
            init_vec = init_vec / np.sum(init_vec) * target_sum
            init_vec = np.clip(init_vec, 0.0, 1000.0)

            # evaluate baseline
            cur_score = _score(init_vec)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = init_vec.copy()
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            # ---- optional LP phase -----------------------------------------
            if _HAS_CVX:
                vec_lp = _lp_improve(init_vec.copy(), target_sum, deadline, max_iters=80)
                lp_score = _score(vec_lp)
                if lp_score < best_score:
                    best_score = lp_score
                    best_seq = vec_lp.copy()
                    print(f"[LP]   n={n} restart={restart} score={lp_score:.6f}")

                work_vec = vec_lp
            else:
                work_vec = init_vec

            # ---- gradient refinement ----------------------------------------
            vec_opt, opt_score = _gradient_optimize(
                work_vec.copy(),
                target_sum,
                deadline,
                max_iters=300,
            )
            if opt_score < best_score:
                best_score = opt_score
                best_seq = vec_opt.copy()
                print(f"[grad] n={n} restart={restart} score={opt_score:.6f}")

            # early exit if target achieved
            if best_score < 1.503:
                print("[done] target bound reached")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback (uniform medium‑size vector)
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)

    if time.monotonic() < deadline:
        refined_seq, refined_score = _local_search(best_seq, deadline, step_initial=0.5)
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"[local] final improvement to {best_score:.6f}")

    return [float(x) for x in best_seq]
