def project_onto_simplex(v, z):
    """Project vector v onto the simplex {x >=0, sum x = z}."""
    # works for any z > 0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    # find rho = max i s.t. u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0

"""
Propose a step sequence with a very low evaluation score.

Algorithm
---------
1.  Choose a set of candidate lengths (n).  Larger n gives a
    tighter theoretical lower bound 2·n/(2·n‑1) ≈ 1.
2.  For each n generate a few initial non‑negative vectors whose
    ℓ¹‑norm equals √(2·n).  Initial vectors are obtained from
    – a uniform simplex sample,
    – an interpolation of the previously best known sequence
      (`height_sequence_1` if available),
    – a simple “spike” distribution.
3.  Improve every initial vector by a projected Adam optimiser.
    The smooth objective is a soft‑max (log‑sum‑exp) of the
    auto‑convolution values:
        F_α(f) = (1/α)·log Σ_k exp(α·(f*f)_k) .
    Its gradient can be written in closed form as a convolution,
    which is computed efficiently with `np.convolve`.  After each
    gradient step the vector is projected onto the simplex
    {x≥0, Σ x = √(2·n)}.
4.  The smoothing parameter α starts small (≈5) and is multiplied
    by 1.2 every 50 iterations, gradually turning the soft‑max
    into the true max.
5.  The best sequence seen across all lengths and random restarts
    is returned.  If the optimisation fails for any reason a very
    cheap random‑walk fallback is used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
def _evaluate(seq: np.ndarray) -> float:
    """Call the provided evaluator on a NumPy array."""
    return float(evaluate_sequence(seq.tolist()))


def _random_on_simplex(n: int, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """
    Uniformly sample a point on the simplex {x≥0, Σ x = target_sum}.
    The sampling uses exponential (Gamma(1,1)) variables.
    """
    x = rng.exponential(scale=1.0, size=n)
    x = x / x.sum() * target_sum
    return x


def _project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σ x = z},
    using the algorithm of Duchi et al. (2008).
    """
    if z < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sort descending
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – fallback
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_auto(f: np.ndarray) -> np.ndarray:
    """Auto‑convolution (f * f) using NumPy."""
    return np.convolve(f, f)


def _gradient_from_weights(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a weight vector w (length 2·n‑1) representing
    a convex combination of the auto‑convolution entries,
    return ∇ Σ_k w_k·(f*f)_k.
    The gradient equals 2·(w ⋆ f_rev) restricted to the
    original n coordinates, where ⋆ denotes convolution.
    """
    n = f.shape[0]
    # reverse f once (cost O(n))
    f_rev = f[::-1]
    # full convolution length = len(w) + n - 1 = (2n‑1)+n‑1 = 3n‑2
    full = np.convolve(w, f_rev)
    start = n - 1                     # slice start index for gradient
    grad = 2.0 * full[start:start + n]
    return grad


def _simple_fallback(budget: float, seed: int):
    """
    Very cheap random‑walk optimiser used only if the main
    optimiser cannot produce a result (e.g. missing CVX or NumPy).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget
    n = 200
    target = math.sqrt(2.0 * n)
    seq = np.full(n, target / n, dtype=np.float64)
    best_seq = seq.copy()
    best_score = _evaluate(seq)

    while time.time() < deadline:
        cand = seq + rng.normal(scale=0.05, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / cand.sum() * target
        sc = _evaluate(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            seq = cand
    return best_seq, best_score

    """
    Search for a step sequence with a low evaluation score.
    Returns a plain Python list of floats.
    """

    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    try:

        have_ref = False

    # 2) Choose a set of lengths – we increase the length gradually.
    #    Larger n gives a tighter lower bound, but also higher cost.
    lengths = [400, 600, 800, 1000, 1200]   # adapt if time runs out

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Prepare a few different starting points for this length
        starts = []

        # a) Uniform simplex sample
        starts.append(_random_on_simplex(n, rng, target_sum))

        # b) Interpolated reference sequence (if we have one)
        if have_ref:
            x_old = np.linspace(0.0, 1.0, ref_seq.size)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, ref_seq)
            interp = np.maximum(interp, 0.0)          # guard against negatives
            interp = interp / interp.sum() * target_sum
            starts.append(interp)

        # c) A “spike” – a single large entry, the rest uniform tiny values
        big = target_sum * 0.6
        rest = (target_sum - big) / (n - 1)
        spike = np.full(n, rest, dtype=np.float64)
        spike[0] = big
        starts.append(spike)

        # ------------------------------------------------------------------
        # Run the optimiser from each start
        for start_idx, f in enumerate(starts):
            if time.time() >= deadline:
                break

            # Initial evaluation
            cur_score = _evaluate(f)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[init] n={n} start={start_idx} score={cur_score:.6f}")

            # Adam optimiser hyper‑parameters
            beta1, beta2 = 0.9, 0.999
            eps = 1e-8
            lr = 0.2                           # base learning rate
            m = np.zeros_like(f)
            v = np.zeros_like(f)

            # smoothing parameter for the soft‑max
            alpha = 5.0

            # iteration budget – will stop early if we run out of time
            max_iters = 2000

            for it in range(1, max_iters + 1):
                if time.time() >= deadline:
                    break

                # ---------------------------------------------------------
                # 1. Current auto‑convolution and its maximum
                b = _conv_auto(f)
                max_b = float(b.max())

                # 2. Soft‑max weights (normalized)
                #    shift by max to avoid overflow, exp is ≤ 1
                shifted = b - max_b
                w = np.exp(alpha * shifted)
                p = w / w.sum()                # Σ p = 1

                # 3. Gradient of the smoothed objective
                grad = _gradient_from_weights(f, p)

                # ---------------------------------------------------------
                # 4. Adam update (biased moments)
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)

                # bias‑corrected estimates
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)

                step = lr * m_hat / (np.sqrt(v_hat) + eps)

                f = f - step

                # 5. Projection back onto the simplex (keeps sum = target)
                f = _project_onto_simplex(f, target_sum)

                # ---------------------------------------------------------
                # 6. New max after the update (computed once per iteration)
                new_b = _conv_auto(f)
                new_max = float(new_b.max())

                # Remember globally best sequence
                if new_max < best_score:
                    best_score = new_max
                    best_seq = f.copy()
                    print(f"[adam] n={n} start={start_idx} it={it} score={new_max:.6f}")

                # ---------------------------------------------------------
                # 7. Adaptive schedule for the smoothing parameter
                if it % 50 == 0:
                    alpha = min(alpha * 1.2, 500.0)

                # 8. Early exit if we are essentially at the theoretical bound
                lower_bound = 2.0 * n / (2.0 * n - 1.0)   # = 1 + 1/(2n‑1)
                if new_max <= lower_bound * 1.02:
                    print(f"[close] n={n} start={start_idx} it={it} score={new_max:.6f}")
                    break

    # ----------------------------------------------------------------------
    # If nothing has been found (e.g. CVXPY missing), fall back to a cheap random walk
    if best_seq is None:
        fallback_budget = max(0.0, deadline - time.time())
        best_seq, best_score = _simple_fallback(fallback_budget, seed)

    # Return a plain list of Python floats
    return [float(x) for x in best_seq]
