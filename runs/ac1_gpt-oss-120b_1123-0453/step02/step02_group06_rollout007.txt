The goal is to minimise  

\[
E(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the auto‑convolution of the non‑negative step sequence `a`.  
Because the denominator and the numerator are both quadratic in the scale of `a`,
the value of `E` is *scale‑invariant*.  
Therefore we can (and do) fix the ℓ¹‑norm to  

\[
\sum_i a_i = \sqrt{2n},
\]

so that `E(a)` reduces to the maximal entry of the convolution.  
The problem is thus: **find a non‑negative vector of length `n` with fixed ℓ¹‑norm that
makes the peak of its auto‑convolution as small as possible**.

The script below implements a **directed stochastic‑annealing** search:

* **Fast convolution** – the current maximum of the convolution is obtained with a
  real FFT (`max_conv_fft` / `full_conv_fft`).  
* **Guided move** – the offset `k*` where the current maximum occurs is found.
  For this offset we compute the contribution of each coordinate  
  `c_i = a_i·a_{k*-i}` (zero if the index is out of range).  
  High contributions are the “culprits’’ that create the peak.  
  A move therefore transfers a small amount of mass from a high‑contribution
  coordinate to a low‑contribution one.
* **Simulated‑annealing acceptance** – moves that lower the peak are always kept,
  otherwise they are accepted with probability `exp(-(Δ)/T)`.  
  The temperature `T` is linearly cooled from `T0` to `0`.
* **Multiple lengths & restarts** – the search is performed for several values of
  `n` (200 … 2000).  Each length receives a few random restarts, and a previously
  good shorter sequence can be **expanded** to a longer one (`expand_sequence`).
* **Optional LP polishing** – after the stochastic phase we try a cheap linear‑
  programming “mass‑inflation’’ step (the same idea as in the reference code)
  if `cvxpy` is available.
* **Time budget handling** – the whole routine stops a couple of seconds before
  the 1000 s limit and returns the best sequence found so far.

The algorithm is completely self‑contained, uses only NumPy (and optionally
CVXPY), and respects all constraints of the evaluation function.

```python
"""Propose a low‑value step sequence.

The algorithm searches for a non‑negative sequence a of length n with
ℓ¹‑norm √(2·n) that minimises the peak of its self‑convolution.
It combines a directed simulated‑annealing local search (mass is moved
from indices that contribute most to the current worst convolution
offset to indices that contribute least) with an optional linear‑programming
polishing step.
Multiple lengths and random restarts are tried; the best sequence found
before the time budget expires is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities --------------------------------------------
# ----------------------------------------------------------------------


def fft_size_for_len(n: int) -> int:
    """Return the smallest power of two ≥ 2·n‑1 (size needed for convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def full_conv_fft(seq: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the full convolution (seq * seq) via real FFT.
    Returns an array of length 2·len(seq)‑1.
    """
    f_hat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f_hat * f_hat, size)[: 2 * seq.size - 1]
    # tiny negative values caused by rounding are set to zero
    return np.maximum(conv, 0.0)


def max_conv_fft(seq: np.ndarray, size: int) -> float:
    """Fast maximal entry of seq*seq."""
    return float(full_conv_fft(seq, size).max())


# ----------------------------------------------------------------------
# Helpers for the directed move -----------------------------------------
# ----------------------------------------------------------------------


def contributions_at_offset(seq: np.ndarray, k: int) -> np.ndarray:
    """
    For a given offset k compute the contribution of each index i to
    the convolution value conv[k] = Σ a_i * a_{k-i}.
    The returned array `c` has length n, with c[i] = a_i·a_{k-i} if the
    second index is inside the vector, otherwise 0.
    """
    n = seq.size
    c = np.zeros(n, dtype=np.float64)

    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return c

    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    j_range = k - i_range
    c[i_range] = seq[i_range] * seq[j_range]
    return c


def directed_move(
    seq: np.ndarray,
    target_sum: float,
    fft_sz: int,
    rng: np.random.Generator,
    avg_step: float,
) -> tuple[np.ndarray, float]:
    """
    Perform one guided stochastic move:
      * identify the worst convolution offset,
      * compute per‑coordinate contributions,
      * move a small amount of mass from a high‑contribution index to a
        low‑contribution index,
      * return the new sequence and its new maximal convolution value.
    """
    n = seq.size
    # current convolution
    conv = full_conv_fft(seq, fft_sz)
    k = int(np.argmax(conv))
    cur_max = float(conv[k])

    # contributions for the worst offset
    contrib = contributions_at_offset(seq, k)

    # pick a source index with large contribution (and positive mass)
    if np.any(contrib > 0):
        i_src = int(np.argmax(contrib))
    else:
        i_src = rng.integers(0, n)

    # pick a destination index with small contribution (different from src)
    # we try a few random candidates and keep the one with the smallest
    # contribution that also respects the 0‑1000 bound.
    i_dst = None
    best_c = np.inf
    for _ in range(5):
        cand = rng.integers(0, n)
        if cand == i_src:
            continue
        if seq[cand] >= 1000.0:
            continue
        c = contrib[cand]
        if c < best_c:
            best_c = c
            i_dst = cand
    if i_dst is None:
        # fallback: random different index
        i_dst = (i_src + 1) % n
        if i_dst == i_src:
            i_dst = (i_src + 2) % n

    # amount to transfer – limited by available mass and by a safety margin
    max_transfer = min(seq[i_src], avg_step * 0.5, 1000.0 - seq[i_dst])
    if max_transfer <= 1e-12:
        return seq, cur_max

    delta = rng.random() * max_transfer
    if delta <= 1e-12:
        return seq, cur_max

    new_seq = seq.copy()
    new_seq[i_src] -= delta
    new_seq[i_dst] += delta

    # clipping just in case numerical noise pushes us out of bounds
    new_seq = np.clip(new_seq, 0.0, 1000.0)

    new_max = float(full_conv_fft(new_seq, fft_sz).max())
    return new_seq, new_max


# ----------------------------------------------------------------------
# Linear‑programming polishing (optional) -------------------------------
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build matrix A (shape (2n‑1, n)) such that (A @ g)[k] = (f * g)[k].
    This matrix is used by the LP “mass‑inflation’’ step.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        A[k, k - idx] = f[idx]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    One cheap LP step: maximise ℓ¹‑mass under the current maximal
    convolution bound and mix the result with the current vector.
    Returns the new sequence and its exact maximal convolution.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv_fft(f, fft_size_for_len(f.size))

    n = f.size
    M = max_conv_fft(f, fft_size_for_len(n))

    A = build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is fast for dense small problems
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    s = g_val.sum()
    if s < 1e-12:
        return f, M

    # rescale to the prescribed ℓ¹‑norm
    g_rescaled = g_val * (target_sum / s)

    # modest mixing
    t = 0.3
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    new_max = max_conv_fft(f_new, fft_size_for_len(n))
    return f_new, new_max


# ----------------------------------------------------------------------
# Sequence expansion ----------------------------------------------------
# ----------------------------------------------------------------------


def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Stretch a known good sequence to a longer length by nearest‑index mapping
    and renormalise to the required ℓ¹‑norm √(2·new_len).
    """
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    total = out.sum()
    if total > 0:
        out *= math.sqrt(2.0 * new_len) / total
    return out


# ----------------------------------------------------------------------
# Directed stochastic‑annealing local search ----------------------------
# ----------------------------------------------------------------------


def directed_sa(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Guided simulated‑annealing: move mass from high‑contribution indices
    to low‑contribution ones.  Returns the best sequence and its maximal
    convolution value found during the run.
    """
    n = f0.size
    sz_fft = fft_size_for_len(n)

    f = f0.copy()
    cur_max = max_conv_fft(f, sz_fft)

    best_f = f.copy()
    best_max = cur_max

    # linear cooling from T0 to 0
    T0 = 0.05
    start_time = time.monotonic()
    it = 0
    avg_step = target_sum / n

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        elapsed = time.monotonic() - start_time
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start_time))

        # perform a directed move
        f_new, new_max = directed_move(f, target_sum, sz_fft, rng, avg_step)

        dE = new_max - cur_max
        if dE < -1e-12 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            f = f_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f_new.copy()
                print(f"[SA] n={n} iter={it} new best max={best_max:.6f}")

        # occasional tiny jitter to avoid stagnation
        if it % 5000 == 0:
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n - 1)
            if i2 >= i1:
                i2 += 1
            tiny = min(f[i2], 1e-3)
            if tiny > 0:
                f[i1] += tiny
                f[i2] -= tiny
                cur_max = max_conv_fft(f, sz_fft)

    return best_f, best_max

# Main entry point ------------------------------------------------------

def propose_candidate(

    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

    except Exception:  # pragma: no cover

    # ------------------------------------------------------------------

    # Return plain Python list, as required by the evaluator.
    return [float(x) for x in best_seq]
```
