"""
subgradient_search.py

A fast heuristic for the step‑function minimisation problem.

The objective returned by ``evaluate_sequence`` (after the usual
ℓ¹‑normalisation) is exactly the maximum coefficient of the
self‑convolution of the vector.  We therefore minimise

    max_k (a * a)[k]    subject to   a_i ≥ 0 ,   Σ a_i = √(2·n)

using a projected sub‑gradient method on the convex set
(simplex).  The sub‑gradient of the max‑convolution is obtained from the
indices where the current maximum is attained.  After a number of
sub‑gradient iterations a short random‑perturbation hill‑climb is run
to escape possible plateaus.

The routine explores several lengths (200 … 1200) and keeps the best
sequence found over the whole allocated budget (≤ 1000 s).  If the
environment does not provide a good start vector it falls back to a
pure random walk.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple wrapper around the provided scorer.
# ----------------------------------------------------------------------
def _score(seq):
    """Return the evaluation of a sequence (list or ndarray)."""
    # ``evaluate_sequence`` is assumed to be globally available.
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
# ----------------------------------------------------------------------
def _project_simplex(v, s):
    """Euclidean projection of v onto the simplex of sum s."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; put the whole mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = s
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient optimisation for a fixed length n.
# ----------------------------------------------------------------------
def _subgradient_optimize(a_init, target_sum, budget, rng):
    """
    Run a projected sub‑gradient descent on ``a`` for at most ``budget`` seconds.
    Returns (best_vector, best_score) found in the time window.
    """
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = _score(a)
    start = time.monotonic()
    deadline = start + budget

    # hyper‑parameters
    step_factor = 0.05          # controls step magnitude (tuned heuristically)

    it = 0
    while time.monotonic() < deadline:
        # current convolution and its maximum
        conv = np.convolve(a, a)
        M = np.max(conv)
        # indices where the maximum is attained (usually few)
        ks = np.where(conv == M)[0]

        # compute a sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)               # inclusive
            idx = np.arange(i_start, i_end + 1)
            g[idx] += a[k - idx]                # contribution a_j where j = k-i
        g *= 2.0

        # step size that scales with the gradient norm
        norm_g = np.linalg.norm(g) + 1e-12
        eta = step_factor * target_sum / norm_g / math.sqrt(it + 1.0)

        # projected gradient step
        a_tmp = a - eta * g
        a = _project_simplex(a_tmp, target_sum)

        # evaluate
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            # optional early stop when we beat the target
            if best_sc < 1.503:
                break

        it += 1

    # ------------------------------------------------------------------
    # Short random hill‑climb to polish the best candidate.
    # ------------------------------------------------------------------
    max_hill = 800
    for _ in range(max_hill):
        if time.monotonic() >= deadline:
            break
        # pick two distinct indices and move a small amount of mass
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to transfer – a tiny fraction of the source entry
        delta = min(a[i], target_sum * 0.01) * rng.random()
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            a = cand
            if best_sc < 1.503:
                break

    return best_a, best_sc

def _fallback_random_search(budget, rng):
    """Very basic random walk – kept for robustness."""
    n = 256
    a = np.full(n, 1.0, dtype=np.float64)
    target_sum = math.sqrt(2.0 * n)

        if sc < best_sc:

            best_a = cand
