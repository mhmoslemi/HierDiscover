    # If all elements nonnegative and sum <= z, we can add slack? Actually we need equal sum z.
    # Standard algorithm:

def optimize_length(n, global_deadline, rng):
    target = math.sqrt(2.0 * n)
    # initial f: random Dirichlet (sample exponential)
    a = rng.random(n)
    f = a / a.sum() * target
    best_f = f.copy()
    conv = np.convolve(f, f)
    best_val = conv.max()
    it = 0
    max_iter = 5000 # or until time
    while time.monotonic() < global_deadline and it < max_iter:
        # compute conv and max
        conv = np.convolve(f, f)
        cur_max = conv.max()
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_f = f.copy()
            print(f"[len={n}] it={it} new best={best_val:.6f}")
        # Build A matrix for LP
        # Use f to construct Toeplitz conv matrix
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for i in range(n):
            rows = i + np.arange(n)  # shape (n,)
            A[rows, np.arange(n)] = f[i]

        # Solve LP
        c = -np.ones(n)  # maximize sum b
        ub = np.full(2*n-1, cur_max, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)]*n,
                      method='highs', options={'presolve': True})
        if not res.success:
            # abort this iteration; maybe random restart
            # random perturbation
            f = np.maximum(f + 0.01 * rng.standard_normal(n), 0)
            total = f.sum()
            if total > 0:
                f = f / total * target
            else:
                f = np.full(n, target/n)
            it += 1
            continue
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            # degenerate; random restart
            f = np.maximum(f + 0.01 * rng.standard_normal(n), 0)
            total = f.sum()
            if total > 0:
                f = f / total * target
            else:
                f = np.full(n, target/n)
            it += 1

        # scale g to meet L1 target

        t = 0.2  # constant

        # ensure nonneg, clip

        # re-normalise just in case
        s = f.sum()
        if s > 0:
            f = f / s * target
        else:
            f = np.full(n, target/n)
        it += 1
    return best_f.tolist(), best_val

def simulate_annealing(f_initial, time_limit, rng):
    f = np.array(f_initial, copy=True)
    n = len(f)
    target = math.sqrt(2.0 * n)
    # Ensure L1 sum target
    s = f.sum()
    if s > 0:
        f = f / s * target
    else:
        f = np.full(n, target / n)
    conv = np.convolve(f, f)
    cur_max = conv.max()
    best_f = f.copy()
    best_val = cur_max
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_limit:
        # Choose worst conv index
        k_max = int(np.argmax(conv))
        # Determine donor distribution using gradient
        i_start = max(0, k_max - (n-1))
        i_end = min(k_max, n-1)
        idx_range = np.arange(i_start, i_end+1)
        grad = np.zeros(n)
        grad[idx_range] = 2.0 * f[k_max - idx_range]
        prob = grad + 1e-12
        prob /= prob.sum()
        donor = rng.choice(n, p=prob)
        # Choose recipient
        recipient = rng.integers(0, n)
        while recipient == donor:
            recipient = rng.integers(0, n)
        # Amount to move
        max_delta = f[donor]
        if max_delta <= 0:
            continue
        # step fraction decreasing with time
        elapsed = time.monotonic() - start
        frac = 0.2 * (1 - elapsed / time_limit) + 0.01
        delta = rng.random() * max_delta * frac
        # Apply move with updates to conv
        conv_old = conv.copy()
        f_old = f.copy()
        # Update conv efficiently as earlier
        conv[donor:donor+n] -= 2.0 * delta * f
        conv[recipient:recipient+n] += 2.0 * delta * f
        # quadratic corrections
        conv[2*donor] += delta * delta
        conv[2*recipient] += delta * delta
        s_idx = donor + recipient
        if s_idx < len(conv):
            conv[s_idx] -= 2.0 * delta * delta
        # Update f
        f[donor] -= delta
        f[recipient] += delta
        new_max = conv.max()
        # acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # temperature schedule
            elapsed_total = time.monotonic() - start
            T = max(1e-6, 1.0 * (1 - elapsed_total / time_limit))
            prob_up = math.exp(-(new_max - cur_max) / T)
            if rng.random() < prob_up:
                accept = True
        if accept:
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_f = f.copy()
                print(f"[SA] it={it} best={best_val:.6f}")
        else:
            # rollback
            f = f_old
            conv = conv_old
        it += 1
    return best_f

def frank_wolfe_optimize(n, time_limit, rng):
    target = math.sqrt(2.0 * n)
    # initial random f: Dirichlet(1)
    a = rng.random(n)
    f = a / a.sum() * target
    best_f = f.copy()
    best_val = float('inf')
    start = time.monotonic()
    it = 0
    max_iter = 2000
    while time.monotonic() - start < time_limit and it < max_iter:
        conv = np.convolve(f, f)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_f = f.copy()
            print(f"[FW] n={n} it={it} best={best_val:.6f}")
        # Build A matrix
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for j in range(n):
            A[j:j+n, j] = f
        # Solve LP
        c = -np.ones(n)  # minimize -sum(b)
        ub = np.full(2*n-1, cur_max, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            # add small noise to f and continue
            f = f + 0.01 * rng.standard_normal(n)
            f = np.clip(f, 0.0, None)
            s = f.sum()
            if s > 0:
                f = f / s * target
            else:
                f = np.full(n, target/n)
            it += 1
            continue
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            # similar fallback
            f = f + 0.01 * rng.standard_normal(n)
            f = np.clip(f, 0.0, None)
            s = f.sum()
            if s > 0:
                f = f / s * target
            else:
                f = np.full(n, target/n)
            it += 1
            continue
        g = b * (target / sum_b)
        # step size: standard FW 2/(k+2)
        step = 2.0 / (it + 2.0)
        if step > 1.0:
            step = 1.0
        f = (1 - step) * f + step * g
        # re-normalize to keep L1 = target (avoid drift)
        s = f.sum()
        if s > 0:
            f = f / s * target
        else:
            f = np.full(n, target/n)
        it += 1
    return best_f, best_val
```

def simulate_annealing_refine(f_initial, time_limit, rng):
    f = np.array(f_initial, dtype=np.float64)
    n = len(f)
    target = math.sqrt(2.0 * n)
    # ensure sum target
    if f.sum() > 0:
        f = f / f.sum() * target
    else:
        f = np.full(n, target/n)
    conv = np.convolve(f, f)
    cur_max = conv.max()
    best_f = f.copy()
    best_val = cur_max

    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_limit:
        elapsed = time.monotonic() - start
        total = time_limit
        # temperature schedule
        T_start = 1.0
        T_end = 1e-4
        T = T_start * (T_end / T_start) ** (elapsed / total)

        # donor index selection based on gradient at worst conv index
        k_max = int(np.argmax(conv))
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * f[k_max - idx]
        prob = grad + 1e-12
        prob = prob / prob.sum()
        donor = rng.choice(n, p=prob)

        # random recipient
        recipient = rng.integers(0, n)
        while recipient == donor:
            recipient = rng.integers(0, n)

        max_delta = f[donor]
        if max_delta <= 1e-12:
            continue
        # fraction of donor mass decreasing over time
        frac = 0.2 * (1.0 - elapsed / total) + 0.01
        delta = rng.random() * max_delta * frac

            continue

        # apply move with rollback possible
        conv_old = conv.copy()
        f_old = f.copy()

        # update conv
        conv[donor:donor+n] -= 2.0 * delta * f
        conv[recipient:recipient+n] += 2.0 * delta * f

        idx2 = 2 * donor
        if idx2 < len(conv):
            conv[idx2] += delta * delta
        idx2 = 2 * recipient
        if idx2 < len(conv):
            conv[idx2] += delta * delta
        s_idx = donor + recipient
        if s_idx < len(conv):
            conv[s_idx] -= 2.0 * delta * delta

        f[donor] -= delta
        f[recipient] += delta

        new_max = conv.max()

            # rollback
            conv = conv_old
            f = f_old

        it += 1
    return best_f

"""
Two‑phase optimiser for the step‑function evaluation.

Phase 1 – Frank‑Wolfe style optimisation.
    Repeatedly solves a linear programme:
        maximise ∑b_j   subject to   f * b ≤ max(f*f)   and   b ≥ 0 .
    The optimiser mixes the current vector with the LP solution
    using the classic FW step size 2/(k+2).  This drives the
    worst convolution entry down while keeping the ℓ¹‑norm fixed
    (ℓ¹ = √(2·n)).  The LP is solved with SciPy’s HiGHS solver,
    which is efficient for the moderate sizes (n≈400) used here.

Phase 2 – Simulated‑annealing refinement.
    Starting from the best vector found in phase 1, a Metropolis
    annealing moves a small amount of mass from a “donor’’ index
    (chosen proportionally to its contribution to the worst
    convolution entry) to a random “recipient’’ index.  The move is
    accepted if it improves the maximal convolution value, otherwise
    with a temperature‑dependent probability.  The temperature cools
    exponentially over the allocated time.

Both phases respect the scorer’s safety checks (non‑negative,
clipped to [0, 1000]) and keep the ℓ¹‑norm exactly √(2·n), which makes
the evaluation equal to max(f*f).  The routine adapts its run time to
the supplied 1000 s budget and returns the best sequence found.
"""

import math
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, ∑x = target}."""
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def frank_wolfe_optimize(
    n: int,
    time_limit: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe (linear‑programming) optimisation for a fixed length n.

    Returns the best sequence encountered and its (exact) max‑convolution value.
    """
    target = math.sqrt(2.0 * n)

    # start from a random Dirichlet‑like point
    a = rng.random(n)
    f = a / a.sum() * target

    best_f = f.copy()
    best_val = float("inf")

    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_limit:
        conv = np.convolve(f, f)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_f = f.copy()
            print(f"[FW] n={n} it={it:04d} best={best_val:.6f}")

        # ---- build the convolution matrix A (size (2n‑1)×n) ----------
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j : j + n, j] = f  # column j is f shifted by j positions

        # ---- linear programme: maximise ∑b_j  -----------------------
        # (SciPy minimises, so minimise –∑b_j)
        c = -np.ones(n, dtype=np.float64)
        ub = np.full(2 * n - 1, cur_max, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # perturb and retry
            f = f + 0.01 * rng.standard_normal(n)
            f = np.clip(f, 0.0, None)
            s = f.sum()
            f = f / s * target if s > 0 else np.full(n, target / n)
            it += 1
            continue

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            f = f + 0.01 * rng.standard_normal(n)
            f = np.clip(f, 0.0, None)
            s = f.sum()
            f = f / s * target if s > 0 else np.full(n, target / n)
            it += 1
            continue

        # ---- scale to the required ℓ¹‑norm -------------------------
        g = b * (target / sum_b)

        # ---- Frank‑Wolfe step ------------------------------------
        step = 2.0 / (it + 2.0)          # classic 2/(k+2) schedule
        step = min(step, 1.0)
        f = (1.0 - step) * f + step * g

        # re‑normalise (protect against numerical drift)
        s = f.sum()
        f = f / s * target if s > 0 else np.full(n, target / n)

        it += 1

    return best_f, best_val


# ----------------------------------------------------------------------
def simulated_annealing_refine(
    f_start: list[float],
    time_limit: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Simple Metropolis annealing that moves a tiny amount of mass
    from a “donor’’ index (chosen according to the gradient of the
    worst convolution entry) to a random “recipient’’ index.
    """
    f = np.array(f_start, dtype=np.float64)
    n = len(f)
    target = math.sqrt(2.0 * n)

    # normalise to the fixed ℓ¹‑norm
    s = f.sum()
    f = f / s * target if s > 0 else np.full(n, target / n)

    conv = np.convolve(f, f)
    cur_max = conv.max()
    best_f = f.copy()
    best_val = cur_max

    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_limit:
        elapsed = time.monotonic() - start
        total = time_limit

        # temperature schedule (exponential cooling)
        T_start, T_end = 1.0, 1e-4
        T = T_start * (T_end / T_start) ** (elapsed / total)

        # ---- donor index (high contribution to worst entry) -----
        k_max = int(np.argmax(conv))
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad = np.zeros(n, dtype=np.float64)
        grad[idx] = 2.0 * f[k_max - idx]

        prob = grad + 1e-12
        prob /= prob.sum()
        donor = rng.choice(n, p=prob)

        # ---- random recipient -----------------------------------
        recipient = rng.integers(0, n)
        while recipient == donor:
            recipient = rng.integers(0, n)

        # ---- amount of mass to move ----------------------------
        max_delta = f[donor]
        if max_delta <= 1e-12:
            continue
        frac = 0.2 * (1.0 - elapsed / total) + 0.01
        delta = rng.random() * max_delta * frac
        if delta <= 0.0:
            continue

        # ---- apply the move (with rollback capability) ---------
        conv_old = conv.copy()
        f_old = f.copy()

        # linear part of the convolution update
        conv[donor : donor + n] -= 2.0 * delta * f
        conv[recipient : recipient + n] += 2.0 * delta * f

        # quadratic correction terms
        idx2 = 2 * donor
        if idx2 < len(conv):
            conv[idx2] += delta * delta
        idx2 = 2 * recipient
        if idx2 < len(conv):
            conv[idx2] += delta * delta
        s_idx = donor + recipient
        if s_idx < len(conv):
            conv[s_idx] -= 2.0 * delta * delta

        # update the sequence itself
        f[donor] -= delta
        f[recipient] += delta

        new_max = conv.max()

        # ---- Metropolis acceptance -------------------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob_up = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob_up:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_f = f.copy()
                print(f"[SA] it={it:06d} best={best_val:.6f}")
        else:
            # rollback
            conv = conv_old
            f = f_old

        it += 1
    return best_f
