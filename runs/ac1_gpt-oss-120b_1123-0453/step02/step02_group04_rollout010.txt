"""
Propose a low‑value step sequence.

Algorithm
---------
We treat the problem as a scale‑invariant minimisation of the worst
autoconvolution value max_k (f*f)[k] with the ℓ¹‑norm fixed to
√(2·n).  Starting from a random (or the supplied) sequence we perform a
simulated‑annealing style local search:

* The current worst shift k_max is identified.
* An index i that contributes heavily to that worst shift is sampled
  with probability proportional to f[i]·f[k_max−i].
* Mass ε is transferred from i to a second index j that does **not**
  influence the worst shift (i.e. k_max−j is out of range).  This
  guarantees a monotone drop of the contribution of the worst shift.
* The effect of the move on the whole convolution is updated in
  O(n) using the exact formula

        Δb = 2·ε·(shift_j – shift_i) + ε²·e_{2j}+ε²·e_{2i}−2·ε²·e_{i+j},

  where shift_i is the vector f placed at offset i.
* The move is accepted if it improves the maximum; otherwise it can be
  accepted with a Metropolis probability that slowly cools down.
* Every few hundred successful moves we recompute the full convolution
  with ``np.convolve`` to avoid drift due to floating‑point round‑off.

The search is run for several problem sizes (200‑600) within the given
time budget; the best sequence ever seen is returned.  If no progress
is made we fall back to a simple random walk.

The routine respects the 1000‑second budget, prints progress and never
uses closures or lambda‑expressions.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Full discrete convolution of a real sequence with itself."""
    return np.convolve(seq, seq)


def _max_conv(seq: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(np.max(_conv_full(seq)))


def _score(seq):
    """Score using the provided evaluation function (if it exists)."""
    # ``evaluate_sequence`` is supplied by the harness.
    return float(evaluate_sequence(seq.tolist()))


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build an initial candidate of length ``n``.
    If the global ``height_sequence_1`` exists we embed it,
    otherwise we use a random Dirichlet‑like start.
    """
    target_sum = math.sqrt(2.0 * n)

    # try to use the supplied sequence
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.ndim == 1 and init.size > 0:
            if init.size == n:
                seq = init.copy()
            elif init.size < n:
                seq = np.zeros(n, dtype=np.float64)
                seq[:init.size] = init
            else:  # longer – truncate
                seq = init[:n].copy()
            cur = np.sum(seq)
            if cur > 1e-12:
                seq *= (target_sum / cur)
            else:
                seq = rng.random(n) + 0.1
                seq /= np.sum(seq)
                seq *= target_sum
            return seq
    except Exception:
        pass

    # fresh random start
    seq = rng.random(n) + 0.1
    seq /= np.sum(seq)
    seq *= target_sum
    return seq


def _update_convolution(b: np.ndarray, f: np.ndarray, i: int, j: int,
                        eps: float) -> np.ndarray:
    """
    Return the updated convolution vector after moving a mass ``eps``
    from position ``i`` to position ``j``.
    ``b`` is the old full convolution (length 2n‑1), ``f`` the old
    sequence (length n).  The update follows the exact algebraic formula
    described in the module docstring.
    """
    n = f.shape[0]
    # start from a fresh zero vector
    delta = np.zeros_like(b)

    # 2·ε·(shift_j – shift_i)
    delta[i:i + n] -= 2.0 * eps * f
    delta[j:j + n] += 2.0 * eps * f

    # quadratic correction ε²·e_{2i}+ε²·e_{2j}−2·ε²·e_{i+j}
    ii = 2 * i
    if ii < delta.size:
        delta[ii] += eps * eps
    jj = 2 * j
    if jj < delta.size:
        delta[jj] += eps * eps
    ij = i + j
    delta[ij] -= 2.0 * eps * eps

    return b + delta


# ----------------------------------------------------------------------
# Core optimisation routine for a single length
# ----------------------------------------------------------------------
def _optimise_length(n: int, budget: float, rng: np.random.Generator):
    """
    Run the local‑search / simulated‑annealing optimizer for a fixed length
    ``n`` and return the best sequence found within ``budget`` seconds.
    """
    start = time.monotonic()
    deadline = start + budget

    # initialise
    f = _initial_sequence(n, rng)
    b = _conv_full(f)
    cur_score = float(b.max())
    best_f = f.copy()
    best_score = cur_score

    # temperature schedule – start relatively warm
    T = cur_score * 0.2 if cur_score > 0 else 0.1
    decay = 0.9999

    iteration = 0
    recompute_counter = 0

    while time.monotonic() < deadline:
        iteration += 1

        # ---- decide the move ------------------------------------------------
        # worst shift
        kmax = int(np.argmax(b))

        # indices that actually contribute to the worst shift
        i_min = max(0, kmax - (n - 1))
        i_max = min(kmax, n - 1)
        i_candidates = np.arange(i_min, i_max + 1, dtype=np.int64)
        partners = kmax - i_candidates
        contrib = f[i_candidates] * f[partners]

        if contrib.sum() > 0:
            probs = contrib / contrib.sum()
            i = int(rng.choice(i_candidates, p=probs))
        else:
            i = int(rng.choice(i_candidates))

        # choose a destination j that **does not** affect the worst shift
        # i.e. where kmax‑j is out of bounds
        left = np.arange(0, i_min, dtype=np.int64)
        right = np.arange(i_max + 1, n, dtype=np.int64)
        outside = np.concatenate((left, right), dtype=np.int64)

        if outside.size > 0:
            j = int(rng.choice(outside))
        else:
            # fallback – any other index
            j = int(rng.integers(0, n - 1))
            while j == i:
                j = int(rng.integers(0, n - 1))

        # amount to move – at most half of the source mass
        max_eps = min(f[i] * 0.5, 1e-2 * math.sqrt(2.0 * n))
        if max_eps <= 0:
            continue
        eps = float(rng.random() * max_eps)

        # ---- compute the new convolution ------------------------------------
        b_new = _update_convolution(b, f, i, j, eps)
        new_score = float(b_new.max())

        # ---- acceptance test ------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis rule – slowly cools down
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # apply the move
            f[i] -= eps
            f[j] += eps
            b = b_new
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                print(f"[improve] n={n} it={iteration} score={best_score:.6f}")

            # occasional full recomputation to curb numerical drift
            recompute_counter += 1
            if recompute_counter % 2000 == 0:
                b = _conv_full(f)

        # temperature decay
        T *= decay

        # safety break if we have already reached a respectable bound
        if best_score < 1.45:   # well below the target 1.503
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative step heights with a low evaluation score.
    The routine explores several problem sizes (200‑600) using a
    targeted simulated‑annealing search that moves mass away from the
    currently worst convolution shift.  It respects the supplied time
    budget and prints intermediate progress.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a tiny safety margin for the outer teardown
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    overall_best_seq = None
    overall_best_score = float("inf")

    # problem sizes to try – increasing size gives more freedom but costs a
    # little more per iteration
    lengths = [200, 300, 400, 500, 600]

    # Allocate time roughly equally among the lengths
    remaining_lengths = len(lengths)
    for n in lengths:
        now = time.monotonic()
        if now >= deadline:
            break
        # time slice for this length (a bit generous for larger n)
        time_slice = (deadline - now) / remaining_lengths * 0.95
        remaining_lengths -= 1

        seq, score = _optimise_length(n, time_slice, rng)

        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(f"[best] new overall best: n={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – simple random walk – in the unlikely case nothing was found
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 256
        target_sum = math.sqrt(2.0 * n)
        f = rng.random(n) + 0.1
        f /= np.sum(f)
        f *= target_sum
        best_f = f.copy()
        best_score = _max_conv(f)
        deadline_fallback = time.monotonic() + max(0.0, deadline - time.monotonic())
        while time.monotonic() < deadline_fallback:
            cand = best_f + rng.normal(scale=0.02, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand_sum = np.sum(cand)
            if cand_sum > 0:
                cand = cand * (target_sum / cand_sum)
            sc = _max_conv(cand)
            if sc < best_score:
                best_score = sc
                best_f = cand.copy()
        overall_best_seq = best_f
        overall_best_score = best_score
        print("[fallback] achieved score", best_score)

    # ------------------------------------------------------------------
    # Final normalisation (not strictly required because the score is
    # scale‑invariant, but it removes any tiny drift).
    # ------------------------------------------------------------------
    n_final = overall_best_seq.shape[0]
    target_sum = math.sqrt(2.0 * n_final)
    total = np.sum(overall_best_seq)
    if total > 0:
        overall_best_seq = overall_best_seq * (target_sum / total)

    # return a plain Python list of floats
    return [float(x) for x in overall_best_seq]
