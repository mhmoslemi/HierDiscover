def simplex_proj(v, s):
   # Duchi algorithm
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - s
   rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
   if len(rho)==0:
       theta = 0
   else:
       rho = rho[-1]
       theta = cssv[rho] / (rho+1)
   w = np.maximum(v - theta, 0)
   return w

def _proj_simplex(v, s):
    # v can be any real vector
    n = len(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all values are zero, set to uniform distribution
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑value step sequence.

The algorithm uses a smooth‑max surrogate for the maximal auto‑convolution
value and performs projected gradient descent on the simplex
{a ≥ 0, Σa = √(2 n)}.  Several random restarts and a modest schedule for the
smoothness parameter (β) are tried for a handful of problem sizes.
If a reference sequence ``height_sequence_1`` is available it is used as
a warm start for n = len(height_sequence_1).  The best sequence seen
over the whole budget is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of a vector onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_restart(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random non‑negative vector of length n with Σ = √(2 n)."""
    target = math.sqrt(2.0 * n)
    # a Dirichlet‑like draw (avoids exact zeros)
    x = rng.random(n) + 0.1
    x = x / np.sum(x) * target
    return x.astype(np.float64)


def _evaluate_score(a: np.ndarray) -> float:
    """
    Return the evaluation score for a (the same quantity that
    ``evaluate_sequence`` would return) without the safety‑clipping overhead.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = float(conv.max())
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * M / (sum_a * sum_a)


def _smooth_gradient_step(a: np.ndarray,
                          target_sum: float,
                          beta: float) -> np.ndarray:
    """
    One gradient step for the smooth‑max surrogate.
    Returns the projected candidate (still sums to ``target_sum``).
    """
    # full linear convolution a*a
    conv = np.convolve(a, a)                     # length 2n‑1
    # smooth‑max weights (soft‑max over the convolution entries)
    cmax = np.max(conv)
    # subtract the max to keep the exponentials in a safe range
    exp_term = np.exp(beta * (conv - cmax))
    ws = exp_term / np.sum(exp_term)             # length 2n‑1, sums to 1
    # gradient of the smooth‑max (derived in the analysis)
    grad = 2.0 * np.correlate(a, ws, mode='valid')
    # step size will be decided by the caller (see below)
    return grad


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step sequence with a low evaluation score.

    The routine explores a few values of n (500,…,2000) and, for each,
    performs several random restarts.  In each restart a projected
    gradient descent on a smooth approximation of the max‑convolution
    is carried out.  The best sequence found before the time budget
    expires is returned.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # 1 s safety margin
    rng = np.random.default_rng(seed)

    # Retrieve the previously discovered sequence, if present.
    known_seq = None
    if "height_sequence_1" in globals():
        try:
            known_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        except Exception:
            known_seq = None

    best_seq = None
    best_score = float("inf")

    # Problem sizes to try – larger n give more freedom to lower the max.
    lengths = [500, 800, 1000, 1200, 1600, 2000]
    restarts_per_n = 4   # can be reduced automatically if we run out of time

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)   # the ℓ¹‑norm we must enforce

        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            # ---------- initial vector ----------
            if known_seq is not None and known_seq.shape[0] == n:
                # Warm‑start from the pre‑computed best sequence (re‑scaled)
                a = known_seq.copy()
                a = a * (target_sum / np.sum(a))
            else:
                a = _random_restart(n, rng)

            cur_score = _evaluate_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            # ---------- projected gradient descent ----------
            # Learning‑rate schedule: start with a step comparable to a typical entry.
            base_lr = target_sum / n
            lr = base_lr
            min_lr = base_lr * 1e-5
            no_improve = 0
            max_iters = 2000
            for it in range(max_iters):
                if time.time() >= deadline:
                    break

                # Current maximal convolution entry (the actual objective)
                conv = np.convolve(a, a)
                M = float(conv.max())

                # Early exit if we are already very close to the theoretical lower bound.
                lower_bound = 2.0 * n / (2.0 * n - 1.0)   # = average of the convolution
                if M <= lower_bound * 1.001:
                    best_score = M
                    best_seq = a.copy()
                    print(f"[near‑lb] n={n} it={it} M={M:.6f}")
                    break

                # Smooth‑max parameter (increase gradually → tighter approximation)
                beta = 1.0 + 19.0 * (it / max_iters)   # from 1 up to 20

                grad = _smooth_gradient_step(a, target_sum, beta)
                # take a step and project back onto the simplex
                a_candidate = _proj_simplex(a - lr * grad, target_sum)

                # Evaluate the true objective after the step
                conv_new = np.convolve(a_candidate, a_candidate)
                M_new = float(conv_new.max())

                if M_new + 1e-12 < M:
                    # Accept the move
                    a = a_candidate
                    cur_score = M_new
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(f"[gd] n={n} restart={restart} it={it} "
                              f"score={cur_score:.6f} lr={lr:.3e} beta={beta:.2f}")
                    # a modest increase of the learning rate can accelerate progress
                    lr = min(base_lr * 5, lr * 1.07)
                    no_improve = 0
                else:
                    # No improvement – shrink the step and try again
                    lr *= 0.5
                    no_improve += 1
                    if lr < min_lr or no_improve >= 30:
                        # Step size too small or stuck → terminate this restart
                        break

            # End of one restart – possibly improve with a quick LP refinement
            # (this is optional and only used if cvxpy is available)
            try:
                import cvxpy as cp

                # Build the convolution matrix once (costly but done only now)
                rows = 2 * n - 1
                A = np.zeros((rows, n), dtype=np.float64)
                for k in range(rows):
                    i_start = max(0, k - (n - 1))
                    i_end = min(k, n - 1)
                    for i in range(i_start, i_end + 1):
                        col = k - i
                        A[k, col] = a[i]

                M_cur = float(np.convolve(a, a).max())
                g = cp.Variable(n)
                constraints = [g >= 0, A @ g <= M_cur]
                prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
                if g.value is not None:
                    g_val = np.maximum(g.value, 0.0)
                    scale = target_sum / np.sum(g_val)
                    g_scaled = g_val * scale
                    # line‑search for the optimal mixing factor t ∈ [0,1]
                    best_local = a
                    best_local_M = M_cur
                    for t in np.linspace(0.0, 1.0, 11):
                        mixed = (1 - t) * a + t * g_scaled
                        M_mix = float(np.convolve(mixed, mixed).max())
                        if M_mix < best_local_M:
                            best_local = mixed
                            best_local_M = M_mix
                    if best_local_M + 1e-12 < M_cur:
                        a = best_local
                        cur_score = best_local_M
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = a.copy()
                            print(f"[lp‑mix] n={n} restart={restart} score={cur_score:.6f}")
                # No need to keep the LP matrix around
            except Exception:
                pass  # cvxpy not installed or LP failed – ignore

    # ----------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found.
    if best_seq is None:
        # a trivial uniform sequence of length 200
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _evaluate_score(best_seq)

    # Return plain Python floats
    return [float(x) for x in best_seq]
