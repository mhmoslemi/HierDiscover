def proj_simplex(v, s):
    # v: array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

import time, math, numpy as np

def _project_to_simplex(v, s):
    """Project v onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError('Target sum must be non‑negative')
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _random_simplex(rng, n, s):
    """Draw a point uniformly from the simplex {x>=0, sum=x}=s."""
    # Using exponential distribution (equiv Dirichlet)
    x = rng.random(n)
    # avoid zero sum
    if x.sum() == 0:
        x[0] = 1.0
    return x / x.sum() * s

def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """
    Gradient‑descent on a smooth approximation of the max auto‑convolution.
    The loss Lα(f) = (1/α)·log∑_k exp(α·(f*f)_k) converges to the
    exact objective as α→∞.  We increase α during the run, take a gradient
    step, and project onto the ℓ¹‑simplex (fixed sum √(2·n)).  Periodic
    random restarts avoid being trapped in poor local minima.
    """
    rng = np.random.default_rng(seed)
    # guard budget
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # -------------------- choose dimension --------------------
    # Use a length that balances expressiveness and speed.
    # If a previous construction exists we keep its length,
    # otherwise we aim for n≈300.
    if "height_sequence_1" in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 200
    n = max(200, min(500, base_len * 2))   # stretch a bit but stay ≤500

    # target ℓ¹‑norm (scale‑invariant)
    target_sum = math.sqrt(2.0 * n)

    # -------------------- initialise --------------------
    # try to reuse previous best if present
    if "height_sequence_1" in globals():
        f0 = np.array(height_sequence_1, dtype=np.float64)
        if len(f0) < n:
            f0 = np.pad(f0, (0, n - len(f0)), constant_values=0.0)
        elif len(f0) > n:
            f0 = f0[:n]
        f0 = _project_to_simplex(f0, target_sum)
    else:
        f0 = _random_simplex(rng, n, target_sum)

    f = f0.copy()
    best_f = f.copy()
    # evaluate initial
    conv = np.convolve(f, f)
    best_val = conv.max()
    print(f"[init] n={n} eval={best_val:.6f}")

    # -------------------- optimisation parameters --------------------
    # smoothing schedule α : from α0 to α_max (exponential)
    alpha0 = 0.5
    alpha_max = 80.0
    # learning‑rate schedule η : from η0 down to η_min linearly
    eta0 = 0.5
    eta_min = 0.01
    # restart control
    last_improve_it = 0
    it = 0
    start_time = time.monotonic()
    total_time = deadline - start_time

    # optional momentum (Adam) buffers
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    while time.monotonic() < deadline:
        # progress fraction (0..1)
        elapsed = time.monotonic() - start_time
        prog = elapsed / total_time

        # current smoothing factor
        alpha = alpha0 * (alpha_max / alpha0) ** prog

        # full auto‑convolution
        conv = np.convolve(f, f)          # length 2n‑1
        cur_max = conv.max()
        # record best
        if cur_max < best_val:
            best_val = cur_max
            best_f = f.copy()
            last_improve_it = it

        # compute soft‑max probabilities over convolution entries
        # numerical stability: subtract the max before exponentiation
        c = conv.max()
        exp_vals = np.exp(alpha * (conv - c))
        p = exp_vals / exp_vals.sum()

        # gradient of the smooth objective (Eq. derivation in analysis)
        # grad = 2 * (p * f) convolved, using reversed f
        conv_pf = np.convolve(p, f[::-1])
        grad = 2.0 * conv_pf[n-1:n-1+n]

        # learning rate (linear interpolation)
        eta = eta0 * (1.0 - prog) + eta_min * prog

        m = beta1 * m + (1.0 - beta1) * grad

"""
Gradient‑descent optimiser for the step‑function evaluation.

The objective 2·n·max(f*f)/(sum(f)^2) is scale‑invariant, therefore we fix the
ℓ¹‑norm to √(2·n).  We minimise a smooth surrogate of the max‐convolution

        Lα(f) = (1/α)·log Σ_k exp(α·(f*f)_k) ,

which converges to the true objective as α→∞.  In each iteration we
* compute the full auto‑convolution,
* turn it into a probability vector by a soft‑max with temperature α,
* obtain the gradient ∇Lα(f) = 2·(p * f_rev) (convolution with the reversed f),
* take an Adam‑style step,
* project the result onto the simplex {x≥0, Σx = √(2·n)}.

α is increased exponentially from 0.5 to 80, the step size is decreased
linearly, and occasional random restarts prevent stagnation.  The best
feasible sequence encountered is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x>=0, sum(x)=s}
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the largest index satisfying the condition
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero after shift – fallback to uniform distribution
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: draw a point uniformly from the simplex {x>=0, sum(x)=s}
def _random_simplex(rng: np.random.Generator, n: int, s: float) -> np.ndarray:
    """Uniform Dirichlet‑type sample on the simplex with total mass s."""
    x = rng.random(n)
    if x.sum() == 0.0:
        x[0] = 1.0
    return x / x.sum() * s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """
    Optimise a non‑negative step sequence using a smooth approximation of the
    max auto‑convolution.  Returns the best sequence found before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)

    # ---- time budget ---------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- choose the length (balance expressiveness / speed) -------------
    if "height_sequence_1" in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 200
    # stretch a little but stay ≤ 500 for safety
    n = max(200, min(500, base_len * 2))

    # ---- target ℓ¹‑norm (scale‑invariant) -------------------------------
    target_sum = math.sqrt(2.0 * n)

    # ---- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        f0 = np.array(height_sequence_1, dtype=np.float64)
        if len(f0) < n:
            f0 = np.pad(f0, (0, n - len(f0)), constant_values=0.0)
        elif len(f0) > n:
            f0 = f0[:n]
        f0 = _project_to_simplex(f0, target_sum)
    else:
        f0 = _random_simplex(rng, n, target_sum)

    f = f0.copy()
    best_f = f.copy()
    conv = np.convolve(f, f)
    best_val = conv.max()
    print(f"[init] n={n} eval={best_val:.6f}")

    # ---- optimisation hyper‑parameters ---------------------------------
    alpha0 = 0.5          # start smoothing
    alpha_max = 80.0      # final smoothing (≈ max)
    eta0 = 0.5            # initial learning‑rate
    eta_min = 0.01        # final learning‑rate
    beta1 = 0.9           # Adam parameters
    beta2 = 0.999
    eps = 1e-8

    # Adam buffers
    m = np.zeros_like(f)
    v_buf = np.zeros_like(f)

    # restart control
    last_improve_it = 0
    it = 0
    start_time = time.monotonic()
    total_time = deadline - start_time

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        prog = elapsed / total_time  # 0 … 1

        # ---- smoothing factor α (exponential schedule) -----------------
        alpha = alpha0 * (alpha_max / alpha0) ** prog

        # ---- full auto‑convolution ------------------------------------
        conv = np.convolve(f, f)          # length 2·n‑1
        cur_max = conv.max()

        # ---- keep global best -----------------------------------------
        if cur_max < best_val:
            best_val = cur_max
            best_f = f.copy()
            last_improve_it = it
            print(f"[gd] it={it:07d} eval={cur_max:.6f} α={alpha:.2f}")

        # ---- soft‑max probabilities over convolution entries ----------
        c = conv.max()
        exp_vals = np.exp(alpha * (conv - c))
        p = exp_vals / exp_vals.sum()

        # ---- gradient of the smooth loss -------------------------------
        # grad = 2 * (p * f) convolved; use reversed f for correct indexing
        conv_pf = np.convolve(p, f[::-1])
        grad = 2.0 * conv_pf[n - 1 : n - 1 + n]   # slice of length n

        # ---- learning‑rate interpolation --------------------------------
        eta = eta0 * (1.0 - prog) + eta_min * prog

        # ---- Adam‑style adaptive step ----------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---- gradient step + projection onto the simplex ---------------
        f_candidate = f - step
        f_candidate = _project_to_simplex(f_candidate, target_sum)

        # accept the candidate (always feasible after projection)
        f = f_candidate

        # ---- occasional random restart if stagnating -------------------
        if it - last_improve_it > 20000 and prog > 0.3:

            m.fill(0.0)
            v_buf.fill(0.0)
            last_improve_it = it
            print(f"[restart] it={it} – new random start")

        it += 1

    # ---- return the best sequence found ---------------------------------
    best_seq = [float(x) for x in best_f.tolist()]
    final_val = evaluate_sequence(best_seq)
    print("[final] best evaluation =", final_val)
    return best_seq
