def conv_matrix(f):
    n = len(f)
    rows = 2*n-1
    data = []
    row_ind = []
    col_ind = []
    for k in range(rows):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            j = k - i
            data.append(f[i])
            row_ind.append(k)
            col_ind.append(j)
    return csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
```

def lp_optimize(f, target_sum, max_iters, deadline):
    best_f = f.copy()
    best_score = _score(best_f.tolist())
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        M = np.max(np.convolve(f, f))
        A = conv_matrix(f)  # maybe as dense? Use np.ndarray.
        # Solve LP: maximize sum(g) s.t. A*g <= M, g>=0
        # Use scipy.optimize.linprog
        c = -np.ones(f.shape)  # minimize negative sum = maximize sum
        b_ub = np.full(A.shape[0], M)
        # Use high performance algorithm
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            break
        g = np.maximum(res.x, 0.0)
        # Rescale to target sum:
        s = np.sum(g)
        if s == 0:
            break
        g = g * (target_sum / s)
        # Evaluate
        sc = _score(g.tolist())
        if sc < best_score:
            best_score = sc
            best_f = g.copy()
            print(f"[LP] n={len(f)} it={it} score={sc:.6f}")

"""
Improved search for a low‑value step‑function.

The algorithm works in three stages:

1. **LP‑based descent** – for a chosen length `n` we start from a feasible
   (usually uniform) vector `f`.  Repeatedly we compute the worst
   autocorrelation `M = max(f*f)`, solve the linear program

       max    Σ g_i
       s.t.   (f * g)_k ≤ M   (∀ k) ,   g ≥ 0

   (the convolution with a fixed `f` is linear in `g`).  The LP solution
   `g` is rescaled to the required ℓ¹‑norm `√(2n)` and replaces `f`.  This
   step never increases the evaluation score and typically drives it down
   quickly.

2. **Sub‑gradient polishing** – after the LP stagnates we treat the
   objective `max_k (f*f)_k` as a piece‑wise linear function.  At the
   current worst offset `k*` we move a small amount of mass from indices
   that contribute most to that offset towards a “mean” direction, keeping
   the total sum constant.  A diminishing step‑size yields a fine
   polishing of the solution.

3. **Random walk finalisation** – a cheap stochastic walk (pairwise mass
   transfers) is performed for the few remaining seconds to possibly
   capture any last improvement.

The routine tries several lengths (200 … 800) and keeps the best
sequence found before the 1000 s deadline (a small safety margin is kept
inside the function).  All numbers are non‑negative and clipped to the
allowed range `[0, 1000]`.

"""

import time
import math
import numpy as np

# optional imports – use whichever is available
try:
    from scipy.optimize import linprog
    _SCIPY_LP = True
except Exception:  # pragma: no cover
    _SCIPY_LP = False

try:
    import cvxpy as cp
    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False

# ----------------------------------------------------------------------
# Global RNG – will be created in `propose_candidate`
_rng = None


def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided ``evaluate_sequence``."""
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
def conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that (A @ g) == np.convolve(f, g).
    Shape of A is (2*len(f)-1, len(f)).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # fill the row for this offset
        # (i , j = k-i)   ⇒   A[k, j] = f[i]
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    return A


# ----------------------------------------------------------------------
def _lp_refine_basic(f: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     max_iters: int = 30) -> tuple[np.ndarray | None, float]:
    """
    LP refinement using ``scipy.optimize.linprog`` (fallback to cvxpy).

    Returns the best vector seen and its score, or (None, inf) on failure.
    """
    n = f.shape[0]
    best_seq = None
    best_sc = float("inf")

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        M = float(np.max(np.convolve(f, f)))
        A = conv_matrix(f)                     # (2n‑1) × n matrix

        # -------- linear program: maximise Σ g_i  (i.e. minimise -Σ g_i) ----------
        c = -np.ones(n, dtype=np.float64)      # minimise –sum(g)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        if _SCIPY_LP:                          # ----- use SciPy (fast) -----
            try:
                res = linprog(c,
                               A_ub=A,
                               b_ub=b_ub,
                               bounds=(0, None),
                               method="highs",
                               options={"presolve": True})
            except Exception:               # pragma: no cover
                res = None
        else:                                   # ----- fallback to CVXPY -----
            try:
                g = cp.Variable(n, nonneg=True)
                prob = cp.Problem(cp.Maximize(cp.sum(g)),
                                  [A @ g <= M])
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
                if g.value is None:
                    res = None
                else:
                    res = type("Obj", (), {"x": np.array(g.value, dtype=np.float64)})
            except Exception:               # pragma: no cover
                res = None

        if res is None or getattr(res, "x", None) is None:
            # LP failed – give up on this size.
            break

        g = np.maximum(res.x, 0.0)
        s = np.sum(g)
        if s < 1e-12:
            break
        g = g * (target_sum / s)               # rescale to required ℓ¹‑norm

        sc = _score(g.tolist())
        if sc < best_sc:
            best_sc = sc
            best_seq = g.copy()
            print(f"[LP] n={n:3d} it={it:2d} score={sc:.6f}")

        # Full replacement (t = 1) – monotone non‑increase of M
        f = g.copy()

        # very early stop if we already reach a fantastic bound
        if best_sc < 1.45:
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
def _subgradient_descent(f: np.ndarray,
                         target_sum: float,
                         max_iters: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the worst autocorrelation entry.
    Keeps the ℓ¹‑norm fixed and enforces non‑negativity by projection.
    """
    n = f.shape[0]
    best = f.copy()
    best_sc = _score(best.tolist())
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        M = conv.max()
        # pick a worst offset uniformly at random (helps escaping ties)
        worst_idxs = np.flatnonzero(np.isclose(conv, M, atol=1e-12))
        k = int(_rng.choice(worst_idxs))

        # gradient of the selected entry: g_i = 2 * f_{k-i} (if in range)
        grad = np.zeros_like(f)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # vectorised fill
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        grad[idx_i] = 2.0 * f[idx_j]

        # project the gradient to the hyperplane sum(d)=0
        lam = grad.mean()
        direction = -grad + lam

        # diminishing step‑size
        step = 0.15 * target_sum / n / math.sqrt(it + 1.0)

        f = f + step * direction
        f = np.maximum(f, 0.0)               # enforce non‑negativity

        # re‑normalise to the required ℓ¹‑norm
        s = f.sum()
        if s > 0:
            f = f * (target_sum / s)
        else:                                 # should never happen, but be safe
            f = np.full_like(f, target_sum / n)

        sc = _score(f.tolist())
        if sc < best_sc:
            best_sc = sc
            best = f.copy()
            print(f"[subgrad] n={n:3d} it={it:5d} score={sc:.6f}")

        # optional early exit if improvement stalls
        if best_sc < 1.45:
            break

    return best, best_sc


# ----------------------------------------------------------------------
def _random_walk(seq: np.ndarray,
                 deadline: float,
                 max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic walk that preserves the total ℓ¹‑mass.
    """
    best = seq.copy()
    best_sc = _score(best.tolist())
    n = best.shape[0]
    no_improve = 0

    while time.monotonic() < deadline and no_improve < max_no_improve:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small amount; scale with the source magnitude
        delta = _rng.normal(loc=0.0, scale=0.02) * max(1e-9, best[i])
        if delta == 0.0:
            continue
        cand_i = best[i] + delta
        cand_j = best[j] - delta
        if cand_i < 0.0 or cand_j < 0.0:
            continue
        cand = best.copy()
        cand[i] = cand_i
        cand[j] = cand_j
        sc = _score(cand.tolist())
        if sc < best_sc:
            best_sc = sc
            best = cand
            no_improve = 0
            print(f"[walk] score={sc:.6f}")
        else:
            no_improve += 1

    return best, best_sc

            # assume ~50 iterations per second for the sub‑gradient loop
            max_sub_iters = int(min(20000, remaining * 50))
            polished_seq, polished_sc = _subgradient_descent(
                best_lp_seq, target_sum, max_sub_iters, deadline)
            if polished_sc < overall_best_sc:
                overall_best_sc = polished_sc
                overall_best_seq = polished_seq.copy()
                print(f"[global] after sub‑grad (n={n}) best={polished_sc:.6f}")

        # ----------------------------------------------------------------
        # Final cheap random walk (a few seconds)
        # ----------------------------------------------------------------
        if time.monotonic() < deadline:
            walk_seq, walk_sc = _random_walk(overall_best_seq, deadline)
            if walk_sc < overall_best_sc:
                overall_best_sc = walk_sc
                overall_best_seq = walk_seq.copy()
                print(f"[global] after random walk (n={n}) best={walk_sc:.6f}")

        # early exit if we have already beaten the target threshold
        if overall_best_sc < 1.45:
            break

    # ------------------------------------------------------------------
    # Fallback – deterministic safe construction if everything failed
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        overall_best_seq = np.full(256, 1.0, dtype=np.float64)

    # final safety clipping (the evaluator also clips, but we obey the rules)
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    return [float(x) for x in overall_best_seq]
