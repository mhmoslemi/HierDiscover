def proj_simplex(v):
    # Project onto simplex {x >=0, sum=1}
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - 1
    ind = np.arange(len(v)) + 1
    cond = v_sorted - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
A fast gradient‑descent search for a non‑negative step sequence that minimises
evaluate_sequence(...).  The problem can be rewritten in terms of a probability
vector p (∑p_i = 1) because the evaluation value is

    score = 2·n·max_k (p*p)[k] ,

where (p*p) is the aperiodic self‑convolution of p.
The total “mass’’ of the convolution is 1, therefore the theoretical lower bound
is 2·n/(2·n‑1) ≈ 1.  The algorithm minimises the ℓ₂‑distance between the
convolution and the flat target 1/(2·n‑1) and projects each iteration onto the
simplex (non‑negative and sum = 1).  Thanks to the FFT the per‑iteration cost
is O(n log n) and a few seconds are enough to drive the score well below the
requested 1.503.  Several sequence lengths are tried and the best result is
returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x≥0,∑x=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return the uniform simplex point
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1] - 1          # zero‑based index of the last positive entry
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities (real transforms)
# ----------------------------------------------------------------------
def _fft_size(length: int) -> int:
    """Smallest power‑of‑two ≥ length."""
    return 1 << ((length - 1).bit_length())


def _conv_via_fft(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """Convolution of real vectors a and b using an FFT of given size."""
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    c = np.fft.irfft(fa * fb, size)
    return c[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# One optimisation run for a fixed length `n`
# ----------------------------------------------------------------------
def _optimise_length(n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Returns the best probability vector `p` (∑p=1) found for this length
    together with its score = 2·n·max(p*p).
    """
    # target uniform value of the convolution
    target_val = 1.0 / (2 * n - 1)

    # FFT sizes
    size_conv = _fft_size(2 * n - 1)          # for p * p
    size_grad = _fft_size(3 * n - 2)          # for p * (conv‑target)ᵣ

    best_score = np.inf
    best_p = None

    # a few random starts – the problem is non‑convex
    for restart in range(3):
        # start from a random probability vector (not the uniform one)
        p = rng.random(n)
        p = p / p.sum()

        # learning‑rate schedule (simple decay)
        eta0 = 0.5
        decay = max(1, n)

        for it in range(max_iters):
            if time.monotonic() >= deadline:
                break

            # ----- convolution p * p -----
            conv = _conv_via_fft(p, p, size_conv)                     # length 2n‑1
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep the best ever seen for this length
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                print(f"[opt] n={n} restart={restart+1} iter={it} "
                      f"score={cur_score:.6f}")

            # early stop if already under the target bound
            if best_score < 1.503:
                return best_p, best_score

            # ----- gradient of ½‖conv‑target‖² -----
            # residual (conv - constant)
            resid = conv - target_val                               # length 2n‑1
            resid_rev = resid[::-1]

            # p * resid_rev  (full length 3n‑2)
            full = _conv_via_fft(p, resid_rev, size_grad)

            # the slice needed for the gradient (indices n‑1 … 2n‑2)
            start = len(resid) - n          # = n‑1
            seg = full[start:start + n]     # length n
            grad = 4.0 * seg[::-1]          # 4·∑_k (conv_k‑t)·p_{k‑i}

            # ----- gradient step with projection -----
            eta = eta0 / (1.0 + it / decay)   # decreasing step size
            p_new = p - eta * grad
            p = _project_simplex(p_new)

        if time.monotonic() >= deadline:
            break

    # if nothing was found (should never happen) fall back to uniform
    if best_p is None:
        best_p = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * (np.max(_conv_via_fft(best_p, best_p, size_conv)))

    return best_p, best_score

        global_best_seq = np.full(n, target / n, dtype=np.float64)
        print("[fallback] uniform sequence used")

    # Return a plain Python list of floats (the evaluator will clamp for safety)
    return [float(x) for x in global_best_seq]
