Propose a low‑value step sequence.

Algorithm
---------
We minimise the quantity   M = max_k (a * a)_k   (auto‑convolution peak)
under the ℓ¹‑normalisation  Σ a_i = √(2 n)  and a_i ≥ 0.
For a fixed length `n` the evaluation function equals `M`.

The search proceeds as follows

1.  Choose a few lengths (300 … 1200).  
2.  For each length launch several restarts.
    Initial vectors are drawn from different families:
        * uniform random on the simplex,
        * a half‑sine shape,
        * a symmetric triangular shape,
        * the previously found best sequence (if available).
3.  For each restart run a projected sub‑gradient descent:
        * compute the convolution a*a,
        * collect the indices where the convolution is (almost) maximal,
        * the sub‑gradient of the peak is
              g_i = Σ_{k∈K} 2·a_{k‑i}   (zero if k‑i out of range),
          where `K` are the (near) maximal indices.
        * take a step opposite to the gradient and project onto the
          simplex {x ≥ 0, Σ x = √(2 n)} (Euclidean projection).
        * an adaptive line search (geometric reduction) is used:
          the step size is scaled by `step_scale / ‖g‖₂` and
          repeatedly halved until the peak improves.
        * when the peak stops improving for many iterations a tiny
          random perturbation is added to escape plateaux.
4.  Keep the best vector over all lengths and restarts.
5.  If, for any reason, no vector is produced we fall back to a very
   cheap random‑walk (as in the reference solution).

The whole routine respects the 1000 s time budget and prints a few
progress messages.  The final result is returned as a plain Python
list of floats.

# Helper utilities – all defined at top level, no closures or lambdas.
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the last index where the threshold is positive
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond][rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv(seq: np.ndarray) -> np.ndarray:
    """Full (2n‑1) auto‑convolution of a real vector."""
    return np.convolve(seq, seq)


def _random_simplex(n: int, rng: np.random.Generator, target: float) -> np.ndarray:
    """Uniform point on the simplex of dimension n with prescribed ℓ¹‑sum."""
    x = rng.random(n)
    # add a tiny constant to avoid exact zeros (not required but harmless)
    x = x + 1e-12
    x = x / np.sum(x) * target
    return x


def _sine_init(n: int, target: float) -> np.ndarray:
    """Half‑sine shape a_i = sin(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n)
    a = np.maximum(a, 0.0)
    a = a / np.sum(a) * target
    return a


def _triangular_init(n: int, target: float) -> np.ndarray:
    """Symmetric triangle (tent) shape."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / np.sum(a) * target
    return a


def _subgradient(a: np.ndarray, conv: np.ndarray, tol: float) -> np.ndarray:
    """
    Compute a sub‑gradient of the peak value of the auto‑convolution.
    `tol` is the absolute tolerance for selecting active indices.
    """
    n = a.shape[0]
    M = conv.max()
    active = np.where(conv >= M - tol)[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    return g

                        # Random jitter to escape a flat region.
                        jitter = rng.normal(scale=0.02 * target_sum, size=n)
                        a = a + jitter
                        a = np.maximum(a, 0.0)
                        a = _project_simplex(a, target_sum)
                        stagnation_counter = 0

            # end of one restart ------------------------------------------------

    # ------------------------------------------------------------------
    # Fallback: very cheap random walk if nothing has been found.
    if best_seq is None:
        n = 300
        target = math.sqrt(2.0 * n)
        a = np.full(n, target / n, dtype=np.float64)
        best_seq = a.copy()
        best_score = _evaluate(a)
        fallback_deadline = start_time + budget_s - 0.1
        while time.time() < fallback_deadline:
            cand = a + rng.normal(scale=0.05, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand / np.sum(cand) * target
            sc = _evaluate(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                a = cand
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list (the grader expects this format).
    return [float(x) for x in best_seq]
