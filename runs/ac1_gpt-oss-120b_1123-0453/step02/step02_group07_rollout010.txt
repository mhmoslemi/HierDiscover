def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    
    best_seq = None
    best_score = float('inf')
    
    # optionally use previous height_sequence_1 if present
    if 'height_sequence_1' in globals():
        init = np.array(list(height_sequence_1), dtype=np.float64)
        # scale to sum sqrt(2n)
        n0 = init.size
        target0 = math.sqrt(2.0 * n0)
        if np.sum(init) > 0:
            init = init * (target0 / np.sum(init))
        init = np.clip(init, 0.0, 1000.0)
        sc = evaluate_sequence(init.tolist())
        best_seq = init.copy()
        best_score = sc
        print("[init] using previous height_sequence_1, score =", sc)
    
    # choose a list of n to try, starting from moderate size
    ns = [200, 250, 300, 350, 400, 450, 500, 600, 800, 1000]
    for n in ns:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # try a few random restarts per n
        restarts = 3
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break
            # random initial vector: exponential distribution to get variability
            a = rng.exponential(scale=1.0, size=n)
            a = a / a.sum() * target_sum
            # optionally shuffle indices to break symmetry
            # a = rng.permutation(a)
            # main optimization loop
            eta = 0.5   # initial learning rate
            p = 6.0     # softness exponent
            # compute next power-of-two length for FFT
            m = 1 << (int(np.ceil(np.log2(2 * n - 1))))
            # iteration count could be large, break when time runs out
            it = 0
            while time.monotonic() < deadline:
                # compute convolution using FFT
                af = np.fft.rfft(a, m)
                conv = np.fft.irfft(af * af, m)[:2 * n - 1]
                max_c = conv.max()
                # evaluate using evaluate_sequence for reliable scoring every few iterations
                if (it % 20) == 0:
                    sc = evaluate_sequence(a.tolist())
                    if sc < best_score:
                        best_score = sc
                        best_seq = a.copy()
                        print(f"[n={n},restart={r}] new best score {best_score:.6f} after {it} it")
                # early exit if max_c already below target ~1.5 (score approx)
                # but we cannot guarantee score = max_c because sum fixed; we still compute full.
                # compute smoothed weights
                # add small epsilon to avoid zero
                eps = 1e-15
                w = np.power(conv + eps, p)
                Z = w.sum()
                if Z == 0:
                    break
                w_norm = w / Z
                # convolution a with w_norm
                wf = np.fft.rfft(w_norm, m)
                grad_conv = np.fft.irfft(af * wf, m)[:2 * n - 1]
                # gradient for each index (central segment)
                grad = 2.0 * p * grad_conv[n - 1 : n - 1 + n]
                # exponential update
                a = a * np.exp(-eta * grad)
                # renormalize to target sum
                s = a.sum()
                if s > 0:
                    a = a * (target_sum / s)
                else:
                    a = np.full(n, target_sum / n)
                # decay learning rate slowly
                eta *= 0.9998
                # optionally raise p as we go to sharpen approximation
                if it % 200 == 0 and p < 15:
                    p += 0.5
                it += 1
                # break condition if max_c already below some threshold
                if max_c < 1.6:
                    # maybe converge; continue for refinement
                    pass
            
            # after inner loop, final local refinement with hill climber
            # use simple hillclimber to fine tune
            a = hill_climb_refine(a, target_sum, deadline, rng)
            # final score
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[final] n={n} restart {r} improved best to {best_score:.6f}")
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break
    
    if best_seq is None:
        # fallback uniform
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] uniform", best_score)
    # ensure list of floats
    return [float(x) for x in best_seq]

"""
Propose a step‑function (non‑negative list) that minimizes the
evaluation score

    2 * n * max(conv(f,f)) / (sum(f)^2)

The method is a simple “smooth max‑convolution” optimiser:

* The objective max(conv) is approximated by the smooth functional
  J(a) = ( Σ_d conv[d]^p )^(1/p).  For p≫1 the p‑norm tightly
  approximates the ℓ∞‑norm while remaining differentiable.

* The gradient of Σ_d conv[d]^p w.r.t. a is
        ∂/∂a_i Σ_d conv[d]^p = 2·p·(a ⊛ w)[i] ,
  where w[d] = conv[d]^{p‑1} and ⊛ denotes (full) convolution.
  This can be evaluated efficiently with FFTs.

* An exponential‑mirrored (multiplicative) update keeps the vector
  non‑negative.  After each update the ℓ¹‑norm is re‑normalised to the
  target   Σ a_i = √(2 n)  (the same normalisation that makes the
  evaluation score equal to max(conv)).  Small learning‑rate decay
  and a gradual increase of p sharpen the approximation as the search
  proceeds.

* Several problem sizes are tried; for each size a few random restarts
  are performed.  The best sequence found (according to the official
  ``evaluate_sequence``) is kept and returned.

The optimiser fits the 1000 s budget: most time is spent on cheap FFT‑based
gradient steps; a lightweight stochastic hill‑climber finishes the
refinement.

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Exact score using the provided evaluate_sequence."""
    # seq is a NumPy array of floats
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
def _full_convolve(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    Returns the first (len(x)+len(y)-1) entries of the real result.
    """
    fx = np.fft.rfft(x, fft_len)
    fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(fx * fy, fft_len)
    return conv[: x.size + y.size - 1]


# ----------------------------------------------------------------------
def _gradient_step(a: np.ndarray,
                   eta: float,
                   p: float,
                   fft_len: int,
                   target_sum: float) -> np.ndarray:
    """
    Perform one mirrored‑gradient step on the smooth max‑convolution loss.
    The update is:
        a ← a * exp(-eta * g)   (g = 2·p·(a⊛w)  with w = conv^{p‑1})
    followed by ℓ¹‑normalisation to `target_sum`.
    """
    # 1. current self‑convolution
    conv = _full_convolve(a, a, fft_len)          # length 2n‑1
    # 2. weight w = conv^{p‑1}   (add a tiny epsilon for stability)
    eps = 1e-15
    w = np.power(conv + eps, p - 1.0)

    # 3. convolution a ⊛ w
    aw = _full_convolve(a, w, fft_len)           # length 3n‑2
    # central part (indices n‑1 … n‑1+n‑1) corresponds to Σ_d w[d] a_{i‑d}
    start = a.size - 1
    grad = 2.0 * p * aw[start:start + a.size]    # shape (n,)

    # 4. multiplicative mirror‑descent update, keep positivity
    a_new = a * np.exp(-eta * grad)

    # 5. re‑normalise to the required ℓ¹‑norm
    s = a_new.sum()
    if s > 0.0:
        a_new *= (target_sum / s)
    else:                                   # degenerate – fall back to uniform
        a_new = np.full_like(a, target_sum / a.size)

    return a_new


# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       rng: np.random.Generator) -> np.ndarray:
    """
    Light‑weight stochastic hill‑climber that keeps the ℓ¹‑norm fixed.
    Moves a small amount of mass from a “heavy’’ index to a “light’’ one.
    """
    n = a.size
    best = a.copy()
    best_sc = _score(best)

    cur = a.copy()
    cur_sc = best_sc

    temperature = 0.03
    step = max(1e-8, 0.01 * target_sum / n)

    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue

        # move a modest fraction of the mass at i towards j
        delta = min(cur[i] * 0.04, step)
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = _score(cand)

        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                print(f"[hill] n={n} new best {best_sc:.6f}")

        temperature = max(1e-6, temperature * 0.9997)
        step = max(1e-12, step * 0.9998)

    return best

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more freedom but costs more.
    lengths = (200, 250, 300, 350, 400, 450, 500, 600, 800, 1000)

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        fft_len = 1 << (int(np.ceil(np.log2(2 * n - 1))))   # power‑of‑two for speed

        # a few random restarts per length
        for restart in range(3):
            if time.monotonic() >= deadline:
                break

            # ------------------- Initial vector -----------------------
            # Exponential‑Gamma gives a moderate spread without huge spikes.
            a = rng.gamma(shape=0.8, scale=1.0, size=n)
            a = a / a.sum() * target_sum

            # ------------------- Gradient descent phase ----------------
            eta = 5e-5          # base step size for the mirrored update
            p = 3.0             # smooth‑max exponent, will be increased gradually
            it = 0
            while time.monotonic() < deadline:
                # full self‑convolution (needed for both score and gradient)
                conv = _full_convolve(a, a, fft_len)

                # occasional exact score check (cheap compared to an FFT)
                if it % 30 == 0:
                    sc = _score(a)
                    if sc < best_score:
                        best_score, best_seq = sc, a.copy()
                        print(f"[global] n={n} restart={restart} it={it} "
                              f"new best score {best_score:.6f}")

                    # early exit if we are already below target
                    if best_score < 1.503:
                        break

                # take a gradient step on the smooth max
                a = _gradient_step(a, eta, p, fft_len, target_sum)

                # schedule adjustments
                eta *= 0.9999                     # slowly shrink step size
                if it % 250 == 0 and p < 12.0:    # increase exponent – sharper max
                    p += 0.5
                it += 1

                # stop after a reasonable number of iterations to keep time for later phases
                if it >= 2500:
                    break

            # ------------------- Final hill‑climb refinement ----------
            a_refined = _hill_climb_refine(a, target_sum, deadline, rng)
            sc_ref = _score(a_refined)
            if sc_ref < best_score:
                best_score, best_seq = sc_ref, a_refined.copy()
                print(f"[refine] n={n} restart={restart} improved best to {best_score:.6f}")

            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – a uniform sequence is always feasible.
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] uniform sequence used, score =", best_score)

    # Return a plain Python list of floats (as required by the grader)
    return [float(v) for v in best_seq]
