def gradient_descent(a, deadline, rng, max_iters=5000):
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = _score(a.tolist())
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        idx = conv.argmax()
        # Build gradient
        grad = np.zeros_like(a)
        # compute contributions from reverse shift
        # for i where 0 <= idx-i < n
        i_start = max(0, idx - (n - 1))
        i_end = min(n-1, idx)
        # grad_i = 2 * a[idx - i]
        # vectorized: indices = np.arange(i_start, i_end+1)
        # paired = idx - indices
        # assign grad[indices] = 2 * a[paired]
        indices = np.arange(i_start, i_end+1)
        paired = idx - indices
        grad[indices] = 2.0 * a[paired]
        # step size adapt: maybe safety scaling
        step = 0.1 * S / (np.linalg.norm(grad) + 1e-12)  # normalized
        # alternative: step = 0.05 * np.mean(a) etc.
        a_new = a - step * grad
        # project onto simplex (non-negative, sum = S)
        a_new = simplex_projection(a_new, S)
        # clip
        a_new = np.clip(a_new, 0.0, 1000.0)
        new_score = _score(a_new.tolist())
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            # optionally reduce step if improvement small?
        # accept new a unconditionally (or if improved)
        a = a_new
        # early break if score < threshold e.g., 1.45
        if best_score < 1.45:
            break
    return best_a, best_score

    """Project v onto the simplex {x>=0, sum x = s}."""

def targeted_local_search(a, deadline, rng):
    n = len(a)
    S = a.sum()
    best = a.copy()
    best_score = _score(best.tolist())
    while time.monotonic() < deadline:
        conv = np.convolve(best, best)
        M = conv.max()
        idx = conv.argmax()
        # compute contributions for each i
        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)
        indices = np.arange(i_start, i_end+1)
        partners = idx - indices
        contrib = best[indices] * best[partners]
        # choose i with probability proportional to contrib
        if contrib.sum() == 0:
            break
        probs = contrib / contrib.sum()
        i = rng.choice(indices, p=probs)
        # choose t with low contribution
        # compute contributions for all t
        # For each t, product = best[t] * best[idx - t] if idx - t in range else 0
        prod = np.zeros(n)
        low_mask = np.ones(n, dtype=bool)
        # compute vectorized for valid t
        t_valid = np.arange(n)
        k_t = idx - t_valid
        valid = (k_t >= 0) & (k_t < n)
        prod[valid] = best[t_valid[valid]] * best[k_t[valid]]
        # choose t among low contributions (e.g., smallest 10%)
        sorted_t = np.argsort(prod)
        # filter out i and maybe partners
        candidates = [tt for tt in sorted_t if tt != i]
        if not candidates:

        t = candidates[0]  # smallest

        delta = min(best[i] * 0.3, 0.05 * S)  # adjust

        # project onto simplex to correct sum (if rounding diff)

        if abs(sum_cand - S) > 1e-9:
            cand = cand * (S / sum_cand)
        cand = np.clip(cand, 0.0, 1000.0)

"""
search for a low‑value step sequence.

The algorithm follows three stages:

1. **LP‑based refinement** (optional):
   For a few problem sizes we run the classic convex‑combination
   iteration that solves a linear program
        max  Σ g_i   s.t.   A_f·g ≤ M ,  g ≥ 0
   where “M = max(f*f)” and ``A_f`` is the convolution matrix of ``f``.
   The solution `g` is rescaled to keep the ℓ¹‑norm √(2 n) and mixed
   into `f`.  Repeating the step monotonically decreases the maximum
   auto‑convolution value.

2. **Targeted local polishing**:
   Starting from the best LP (or from a random Dirichlet vector
   when the LP solver is unavailable) we repeatedly:
   * locate the shift `k*` where the convolution is maximal;
   * pick an index `i` that contributes the most to that shift
     (`a_i·a_{k*‑i}`);
   * move a modest amount of mass from `i` to an index `t` that
     contributes little to the same shift;
   * re‑project onto the simplex {x ≥ 0, Σx = √(2 n)}.
   The move is kept only if it improves the score.
   This focussed walk reduces the peak much faster than a blind
   random walk.

3. **Fallback random walk** (if time remains and no LP was used).

All steps respect the 1000 s overall budget (a small safety margin
is left).  The routine returns a plain Python list of non‑negative
floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global RNG – will be (re)initialised in `propose_candidate`.
_rng = None

# ----------------------------------------------------------------------
def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
def conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the dense convolution matrix A (size (2n‑1) × n) s.t.
    (A @ g) == f * g (linear convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    return A


# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.size
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _lp_refine(
    n: int,
    target_sum: float,
    deadline: float,
    mix: float = 0.2,
    max_iters: int = 120,
) -> tuple[np.ndarray | None, float]:
    """
    Run the LP‑based improvement for a given length `n`.

    Returns `(best_sequence, best_score)` or `(None, inf)` if the LP
    solver is unavailable or the optimisation fails.
    """
    global _rng
    best_seq = None
    best_score = float("inf")

    # Random feasible start (Dirichlet‑like)
    f = _rng.exponential(scale=1.0, size=n)
    f = f / np.sum(f) * target_sum

    # Try importing a solver once
    try:
        import cvxpy as cp
    except Exception:
        # No LP capability – abort.
        return None, float("inf")

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        M = float(np.max(np.convolve(f, f)))
        A = conv_matrix(f)

        g = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g)),
                          [g >= 0, A @ g <= M])
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=5000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, warm_start=True,
                           max_iters=10000, verbose=False)
            except Exception:
                break

        if g.value is None:
            break
        g0 = np.maximum(g.value, 0.0)
        s_g = np.sum(g0)
        if s_g < 1e-12:
            break

        # Rescale to keep ℓ¹‑norm = target_sum
        g_rescaled = g0 * (target_sum / s_g)

        # Convex combination update
        f_new = (1.0 - mix) * f + mix * g_rescaled
        # Numerical safety
        f_new = np.clip(f_new, 0.0, 1000.0)

        sc = _score(f_new.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = f_new.copy()
            print(f"[LP] n={n:3d} it={it:3d} score={sc:.6f}")

        f = f_new

        # Early exit if already very good
        if best_score < 1.45:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _targeted_local_search(
    seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_no_imp: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Focused stochastic walk that reduces the peak of the auto‑convolution.
    It repeatedly transfers a small amount of mass from the index that
    contributes most to the current maximal shift to an index that
    contributes very little.
    """
    n = seq.shape[0]
    S = np.sum(seq)  # should already be √(2 n)
    best = seq.copy()
    best_score = _score(best.tolist())
    no_improve = 0

    while time.monotonic() < deadline and no_improve < max_no_imp:
        conv = np.convolve(best, best)
        M = conv.max()
        k = int(conv.argmax())

        # indices that actually appear in this shift
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1, dtype=int)
        partners = k - idxs

        # contributions a_i * a_{k-i}
        contrib = best[idxs] * best[partners]

        # pick a heavy contributor (probability ∝ contribution)
        if contrib.sum() == 0:
            break
        probs = contrib / contrib.sum()
        i = rng.choice(idxs, p=probs)
        j = k - i  # partner (may be equal to i)

        # choose a target index with tiny contribution to the same shift
        prod = np.zeros(n, dtype=float)
        t_range = np.arange(n, dtype=int)
        partner_t = k - t_range
        valid = (partner_t >= 0) & (partner_t < n)
        prod[valid] = best[t_range[valid]] * best[partner_t[valid]]
        # smallest‑contribution indices (ignore i and its partner)
        order = np.argsort(prod)
        t = None
        for cand in order:
            if cand != i and cand != j:
                t = cand
                break
        if t is None:
            break

        # amount to move – a modest fraction of the source entry
        delta = min(best[i] * 0.25, 0.03 * S)
        if delta <= 1e-12:
            no_improve += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[t] += delta
        if cand[i] < -1e-12:
            no_improve += 1
            continue

        # re‑normalise to the exact ℓ¹‑norm (projection is cheap here)
        cand = _simplex_projection(cand, S)
        cand = np.clip(cand, 0.0, 1000.0)

        sc = _score(cand.tolist())
        if sc < best_score:
            best_score = sc
            best = cand
            no_improve = 0
            print(f"[targeted] improved score={best_score:.6f}")
        else:
            no_improve += 1

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.02:
            i2, j2 = rng.integers(0, n, size=2)
            if i2 == j2:
                continue
            delta2 = rng.normal(scale=0.02) * max(1e-6, best[i2])
            if best[i2] - delta2 < 0 or best[j2] + delta2 < 0:
                continue
            cand2 = best.copy()
            cand2[i2] -= delta2
            cand2[j2] += delta2
            cand2 = _simplex_projection(cand2, S)
            cand2 = np.clip(cand2, 0.0, 1000.0)
            sc2 = _score(cand2.tolist())
            if sc2 < best_score:
                best_score = sc2
                best = cand2
                no_improve = 0
                print(f"[targeted] random jump improved score={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Main entry point called by the evaluator.
    Searches for a step‑function that minimises `evaluate_sequence`.
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # safety margin before the external timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to produce a random feasible start (Dirichlet‑like)
    # ------------------------------------------------------------------
    def random_start(n: int, target_sum: float) -> np.ndarray:
        vec = _rng.exponential(scale=1.0, size=n)
        vec = vec / np.sum(vec) * target_sum
        return vec

    # ------------------------------------------------------------------
    # Sizes to explore – larger n gives a tighter theoretical bound.
    # ------------------------------------------------------------------
    for n in (250, 300, 350, 400, 450):
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # Try to keep a known good construction if it matches the size.
        start_vec = None
        if "height_sequence_1" in globals():
            try:
                hs = np.array(height_sequence_1, dtype=float)
                if hs.size == n:
                    start_vec = hs.copy()
                    start_vec = _simplex_projection(start_vec, target_sum)
                elif hs.size > n:
                    start_vec = hs[:n].copy()
                    start_vec = _simplex_projection(start_vec, target_sum)
                else:
                    # pad with small noise
                    extra = np.zeros(n - hs.size, dtype=float)
                    start_vec = np.concatenate([hs, extra])
                    start_vec = _simplex_projection(start_vec, target_sum)
            except Exception:
                start_vec = None

        # ----------------------------------------------------------------
        # 1) LP‑based refinement (if a solver is available)
        # ----------------------------------------------------------------
        best_seq_lp = None
        best_score_lp = float("inf")
        # try a few random restarts of the LP phase
        for restart in range(3):
            if time.monotonic() >= deadline:
                break
            if start_vec is not None and restart == 0:
                f0 = start_vec.copy()
            else:
                f0 = random_start(n, target_sum)

            # run the LP refinement
            seq_lp, score_lp = _lp_refine(
                n=n,
                target_sum=target_sum,
                deadline=deadline,
                mix=0.2,
                max_iters=120,
            )
            if seq_lp is not None and score_lp < best_score_lp:
                best_score_lp = score_lp
                best_seq_lp = seq_lp.copy()
                print(f"[global] LP restart {restart} size {n} score={score_lp:.6f}")

            if best_score_lp < 1.45:
                break

        # ----------------------------------------------------------------
        # 2) If the LP stage yielded nothing, fall back to a random start.
        # ----------------------------------------------------------------
        if best_seq_lp is None:
            cur_seq = random_start(n, target_sum)
            cur_score = _score(cur_seq.tolist())
            print(f"[fallback] LP unavailable – start random size {n} score={cur_score:.6f}")
        else:
            cur_seq = best_seq_lp
            cur_score = best_score_lp

        # ----------------------------------------------------------------
        # 3) Targeted local polishing (the main improvement engine)
        # ----------------------------------------------------------------
        if time.monotonic() < deadline:
            cur_seq, cur_score = _targeted_local_search(
                cur_seq, deadline, _rng
            )
            print(f"[polish] after targeted local search size {n} score={cur_score:.6f}")

        # ----------------------------------------------------------------
        # Keep the globally best sequence seen so far.
        # ----------------------------------------------------------------
        if cur_score < overall_best_score:
            overall_best_score = cur_score
            overall_best_seq = cur_seq.copy()
            print(f"[global] new overall best size {n}: {overall_best_score:.6f}")

        # early exit when we are already very close to the target
        if overall_best_score < 1.44:
            break

    # ------------------------------------------------------------------
    # 4) Final fallback – uniform sequence (guaranteed to be feasible)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 256
        overall_best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)
        overall_best_score = _score(overall_best_seq.tolist())
        print(f"[fallback] using uniform {n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list of floats (the evaluator will clip once more).
    # ------------------------------------------------------------------
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)
    return [float(x) for x in overall_best_seq]
