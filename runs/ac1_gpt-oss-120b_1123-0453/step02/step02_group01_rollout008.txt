# Simple simplex utilities -----------------------------------------------
def _project_onto_simplex(v, target):
    """
    Project vector ``v`` onto the simplex { x ≥ 0 , sum(x) = target }.
    Implementation of the algorithm from Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become zero – fall back to uniform distribution
        return np.full(n, target / n, dtype=np.float64)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _random_restart(n, rng):
    """
    Uniform point on the simplex of dimension ``n`` with
    ℓ¹‑norm √(2·n).
    """
    target = math.sqrt(2.0 * n)
    # Dirichlet(1,…,1) → exponential variables normalised.
    x = rng.exponential(scale=1.0, size=n)
    x = x / np.sum(x) * target
    return x.astype(np.float64)

def _sinusoidal_start(n, target):
    """Half‑sine shape (non‑negative) scaled to the required ℓ¹‑norm."""
    x = np.sin(np.pi * np.arange(n) / (n - 1))
    x = np.maximum(x, 0.0)
    x = x / np.sum(x) * target
    return x.astype(np.float64)

def _quadratic_start(n, target):
    """Parabolic shape that vanishes at the ends."""
    i = np.arange(n, dtype=np.float64)
    x = i * (n - 1 - i)  # i·(n‑1‑i)
    x = np.maximum(x, 0.0)
    x = x / np.sum(x) * target
    return x.astype(np.float64)

# ----------------------------------------------------------------------
# Convolution matrix (used by the LP) ------------------------------------
def _conv_matrix(f):
    """Build a (2n‑1)×n matrix A s.t. A·g = np.convolve(f,g)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A

# ----------------------------------------------------------------------
# One LP improvement step ------------------------------------------------
def _lp_step(f, M, target_sum):
    """
    Solve the LP:
        maximise   Σ g_i
        subject to (f∗g)_k ≤ M   for all k
                   g_i ≥ 0
    The solution is returned after rescaling to have ℓ¹‑norm = target_sum.
    Returns None if cvxpy is not available or the solve fails.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:
            return None

    if g.value is None:
        return None

    g_val = np.maximum(g.value, 0.0)
    s = np.sum(g_val)
    if s < 1e-12:
        return None
    return g_val * (target_sum / s)

# ----------------------------------------------------------------------
# Very cheap random‑walk fallback (used only if everything else fails) -
def _simple_fallback(budget, seed):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget
    n = 200
    target = math.sqrt(2.0 * n)
    seq = np.full(n, target / n, dtype=np.float64)
    best_seq = seq.copy()
    best_score = _evaluate(seq.tolist())

    while time.time() < deadline:
        cand = seq + rng.normal(scale=0.05, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / np.sum(cand) * target
        sc = _evaluate(cand.tolist())
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            seq = cand
    return best_seq, best_score

# ----------------------------------------------------------------------
# Gradient descent that focuses on the current maximal convolution index.
def _grad_descent_step(a, target_sum, rng, step_factor):
    """
    One iteration of the specialised gradient descent.
    Returns the new vector and a boolean telling whether the bound improved.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = float(conv.max())
    k_max = int(np.argmax(conv))

    # gradient = 2 * a_{k_max‑i} (if the index exists)
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]

    # adaptive step size (relative to the average entry)
    avg = target_sum / n
    eta = step_factor * avg
    cand = a - eta * grad
    cand = _project_onto_simplex(cand, target_sum)

    M_new = float(np.convolve(cand, cand).max())
    improved = M_new < M - 1e-12
    return cand, improved, M_new

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low max auto‑convolution.
    Returns the best sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Try to reuse the sequence from the previous run, if it exists.
    best_seq = None
    best_score = float("inf")
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.size > 0:
            n0 = init.shape[0]
            target0 = math.sqrt(2.0 * n0)
            init = np.maximum(init, 0.0)
            init = _project_onto_simplex(init, target0)
            best_seq = init.copy()
            best_score = _evaluate(best_seq.tolist())
            print(f"[reuse] Loaded height_sequence_1 (n={n0}) score={best_score:.6f}")
    except Exception:
        pass  # ignore if the variable does not exist

    # ------------------------------------------------------------------
    # Define the set of lengths we will explore.
    # Larger n offers a better theoretical bound but each iteration costs more.
    lengths = [300, 400, 500, 600, 800]
    # Heuristic: allocate more restarts to the shorter lengths.
    restarts_per_length = {300: 6, 400: 5, 500: 4, 600: 3, 800: 2}

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # Prepare a few deterministic starts in addition to random ones.
        deterministic_starts = [
            _random_restart(n, rng),
            _sinusoidal_start(n, target_sum),
            _quadratic_start(n, target_sum),
        ]

        # If we already have a best vector of *different* length, we can
        # interpolate it to the current length – this often gives a good
        # initial point.
        if best_seq is not None and best_seq.shape[0] != n:
            # linear interpolation (simple scaling)
            x_old = np.linspace(0, 1, best_seq.shape[0])
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, best_seq)
            interp = _project_onto_simplex(interp, target_sum)
            deterministic_starts.append(interp)

        restarts = restarts_per_length.get(n, 2)
        for restart_id in range(restarts):
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------------
            # Choose a starting vector.
            if restart_id < len(deterministic_starts):
                f = deterministic_starts[restart_id].copy()
            else:
                f = _random_restart(n, rng)

            cur_M = float(np.convolve(f, f).max())
            if cur_M < best_score:
                best_score = cur_M
                best_seq = f.copy()
                print(f"[init] n={n} restart={restart_id} M={cur_M:.6f}")

            # ------------------------------------------------------------------
            # LP phase – a limited number of aggressive LP steps.
            max_lp_iters = 120
            t_mix = 0.15
            for lp_it in range(max_lp_iters):
                if time.time() >= deadline:
                    break
                conv = np.convolve(f, f)
                M = float(conv.max())
                # Solve the LP
                g_scaled = _lp_step(f, M, target_sum)
                if g_scaled is None:
                    break  # LP not available – give up this restart

                M_g = float(np.convolve(g_scaled, g_scaled).max())
                if M_g < M - 1e-12:
                    f = g_scaled
                    continue

                # otherwise try a convex combination
                cand = (1.0 - t_mix) * f + t_mix * g_scaled
                M_cand = float(np.convolve(cand, cand).max())
                if M_cand < M:
                    f = cand
                    # slowly increase the mixing factor when it helps
                    t_mix = min(0.5, t_mix * 1.05)
                    continue
                # if nothing helps, become more conservative
                t_mix = max(0.05, t_mix * 0.5)
                if t_mix <= 0.05 + 1e-12:
                    break

            # ------------------------------------------------------------------
            # Gradient‑descent refinement (the main new ingredient).
            step_factor = 0.6   # relative to the average entry
            no_improve = 0
            max_grad_iters = 2000
            for gd_it in range(max_grad_iters):
                if time.time() >= deadline:
                    break
                f, improved, M_new = _grad_descent_step(f, target_sum, rng, step_factor)
                if improved:
                    step_factor = min(1.5, step_factor * 1.07)
                    no_improve = 0
                else:
                    step_factor *= 0.70
                    no_improve += 1

                if M_new < best_score:
                    best_score = M_new
                    best_seq = f.copy()
                    print(f"[grad] n={n} restart={restart_id} it={gd_it} M={M_new:.6f}")

                # occasional stochastic perturbation to escape plateaus
                if (gd_it % 250) == 0 and gd_it > 0:
                    shake = f + rng.normal(scale=0.02 * (target_sum / n), size=n)
                    shake = np.clip(shake, 0.0, 1000.0)
                    shake = _project_onto_simplex(shake, target_sum)
                    f = shake

                # early exit if we cannot improve any more
                if no_improve >= 80:
                    break

    # ------------------------------------------------------------------
    # If nothing succeeded (e.g. CVXPY missing), fall back to a cheap random walk.
    if best_seq is None:
        fallback_budget = max(0.0, deadline - time.time())
        best_seq, best_score = _simple_fallback(fallback_budget, seed)
        print(f"[fallback] obtained score={best_score:.6f}")

    # Final sanity check – ensure we return a list of Python floats.
    return [float(v) for v in best_seq]
