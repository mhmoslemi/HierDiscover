def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.ifft(fa * fa, n=L).real[:2*n-1]
    return conv
```

```
def _max_grad(a, b, tol=1e-12):
    max_val = b.max()
    mask = (b >= max_val - tol)
    idxs = np.where(mask)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in idxs:
        # compute contributions
        i = np.arange(n)
        j = k - i
        valid = (j >= 0) & (j < n)
        grad[valid] += 2.0 * a[j[valid]]
    return grad, max_val
```

def _gd_step(a, target_sum, step):
    conv = _conv_fft(a)
    grad, max_val = _max_grad(a, conv)
    # step direction: -grad
    new = a - step * grad
    new = np.clip(new, 0.0, None)
    # Rescale to sum target
    s = new.sum()
    if s > 0:
        new = new * (target_sum / s)
    else:
        # if zero, reinitialize
        new = np.full_like(a, target_sum / len(a))
    # compute new max
    new_conv = _conv_fft(new)
    new_max = new_conv.max()
    return new, new_max
```

"""
Propose a low‑value step sequence.

The algorithm combines three complementary heuristics:

1.  **LP refinement** (when CVXPY is available) – the classic
    “maximise ℓ¹‑mass under a convolution bound” step that quickly
    reduces the worst auto‑convolution value.

2.  **Projected gradient descent** – after the LP phase we iteratively
    move mass away from the indices that contribute most to the current
    maximum of the auto‑convolution.  The gradient of the maximum is
    computed analytically (∂max/∂a_i = 2·a_{k‑i} for each active shift k).
    After each update we project onto the non‑negative orthant and
    re‑scale to keep the ℓ¹‑norm fixed to √(2·n), which makes the
    objective equal to the maximum convolution entry.

3.  **Random perturbation hill‑climb** – a short Gaussian walk that
    escapes possible plateaus.

The search is performed for several lengths (200 … 1500).  The best
sequence found over all lengths and all stages is returned.  The routine
strictly respects the supplied time budget (default 1000 s) and works
even when CVXPY is not installed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Wrapper for the provided scorer.
# ----------------------------------------------------------------------
def _score(seq):
    """Return the evaluation score for a sequence (list or ndarray)."""
    return float(evaluate_sequence(seq.tolist()))  # evaluate_sequence is globally available


# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, O(n log n)).
# ----------------------------------------------------------------------
def _conv_fft(a):
    """Return the discrete convolution a*a using a real FFT (length 2n‑1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def _conv_matrix(f):
    """Dense matrix A such that (A @ g) == f * g (full convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        cols = k - np.arange(i_start, i_end + 1)
        A[k, cols] = f[i_start : i_end + 1]
    return A


# ----------------------------------------------------------------------
# Gradient of the maximum convolution entry.
# ----------------------------------------------------------------------
def _max_grad(a, conv, tol=1e-12):
    """
    Compute a sub‑gradient of max(conv) w.r.t. a.

    Returns (grad, max_val) where
        grad_i = Σ_{k∈Active} 2·a_{k‑i}   (0 ≤ i < n)
    and Active = {k | conv[k] is within tol of the maximum}.
    """
    max_val = conv.max()
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        valid = (j >= 0) & (j < n)
        grad[valid] += 2.0 * a[j[valid]]
    return grad, max_val


# ----------------------------------------------------------------------
# LP‑driven refinement (optional, requires CVXPY).
# ----------------------------------------------------------------------
def _lp_refine(f, target_sum, deadline, max_iters=30):
    """Perform a few LP steps to reduce the worst auto‑convolution."""
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, _score(f)

    n = f.shape[0]
    t = 0.3
    decay = 0.95
    best_seq = f.copy()
    best_sc = _score(f)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        M = float(np.max(np.convolve(f, f)))
        A = _conv_matrix(f)

        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000)

        if g.value is None:
            break

        g0 = np.maximum(g.value, 0.0)
        sum_g = np.sum(g0)
        if sum_g < 1e-12:
            break

        g_rescaled = g0 * (target_sum / sum_g)
        f = np.clip((1.0 - t) * f + t * g_rescaled, 0.0, 1000.0)

        sc = _score(f)
        if sc < best_sc:
            best_sc = sc
            best_seq = f.copy()
            print(f"[LP] n={n:4d} it={it:2d} t={t:.3f} score={sc:.6f}")

        t = max(0.01, t * decay)
        if best_sc < 1.503:
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Gradient‑descent refinement (always available).
# ----------------------------------------------------------------------
def _gd_refine(a, target_sum, deadline, max_iter=2000):
    """Iteratively push mass away from the active maximal convolution entries."""
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _score(a)  # this is exactly max(conv) because of the normalisation
    step_factor = 0.2  # initial step multiplier

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        conv = _conv_fft(a)
        grad, cur_max = _max_grad(a, conv)

        # step size decays with iteration number
        step = step_factor * (target_sum / n) / (it + 1)

        # trial update
        cand = a - step * grad
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 1e-12:
            cand = cand * (target_sum / s)
        else:
            cand = np.full_like(a, target_sum / n)

        new_max = _conv_fft(cand).max()

        if new_max < cur_max:                # improvement accepted
            a = cand
            if new_max < best_val:
                best_val = new_max
                best_seq = cand.copy()
                print(f"[GD] n={n:4d} it={it:4d} max={new_max:.6f}")
            # optionally increase step size a little
            step_factor = min(1.0, step_factor * 1.05)
        else:
            # no improvement – shrink step factor
            step_factor *= 0.5
            if step_factor < 1e-6:
                break

        if best_val < 1.503:
            break

    return best_seq, best_val

            else:
                a = rng.random(n) + 0.1
                a = a / a.sum() * target_sum
        else:
            # embed a shorter best sequence if possible
            if best_seq is not None and best_seq.shape[0] < n:
                a = np.zeros(n, dtype=np.float64)
                a[: best_seq.shape[0]] = best_seq
                cur = a.sum()
                if cur > 1e-12:
                    a = a * (target_sum / cur)
                else:
                    a = rng.random(n) + 0.1
                    a = a / a.sum() * target_sum
            else:
                a = rng.random(n) + 0.1
                a = a / a.sum() * target_sum

        # ------------------------------------------------------------------
        # Stage 1 – LP refinement (if possible).
        # ------------------------------------------------------------------
        if HAS_CVX:
            a, lp_score = _lp_refine(a, target_sum, deadline, max_iters=20)
            if lp_score < best_score:
                best_score = lp_score
                best_seq = a.copy()

        # ------------------------------------------------------------------
        # Stage 2 – Gradient‑descent polishing.
        # ------------------------------------------------------------------
        a, gd_score = _gd_refine(a, target_sum, deadline, max_iter=2000)
        if gd_score < best_score:
            best_score = gd_score
            best_seq = a.copy()

        # ------------------------------------------------------------------
        # Stage 3 – Short random perturbation.
        # ------------------------------------------------------------------
        a, pert_score = _perturb_search(a, target_sum, deadline, attempts=200, sigma=0.02)
        if pert_score < best_score:
            best_score = pert_score
            best_seq = a.copy()

        print(f"[summary] n={n:4d} current best score={best_score:.6f}")

        if best_score < 1.503:
            print("Target threshold reached – stopping early.")
            break

    # ------------------------------------------------------------------
    # Fallback random walk if nothing succeeded.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_seq, fallback_score = _fallback_random_search(
            max(0.0, float(budget_s) - (time.monotonic() - start_time)),
            seed=seed,
        )
        best_seq = fallback_seq
        best_score = fallback_score

    # Return a plain Python list of floats (the evaluator expects that).
    return [float(x) for x in best_seq]
