def project_onto_simplex(v, s):
   # v: array, returns nonnegative vector with sum = s.
   # Standard algorithm: shift by lambda.
   v = np.maximum(v, 0)  # ensure nonneg
   if v.sum() == 0:
        return np.full_like(v, s / len(v))
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - s
   ind = np.arange(len(v)) + 1
   cond = u - cssv / ind > 0
   rho = ind[cond][-1]
   theta = cssv[cond][-1] / rho
   w = np.maximum(v - theta, 0)
   return w

"""
Improved search for low evaluation step sequences.

The algorithm mixes a smooth‑max gradient descent (log‑sum‑exp surrogate
for the maximal auto‑convolution entry) with occasional simplex
projection, and reuses the best sequence found so far (up‑sampling it
to larger lengths).  A small number of LP refinements (the same LP as
in the reference implementation) are applied at the end of each
gradient‑descent run.  The routine respects the 1000 s time budget and
stores the best sequence seen across several lengths and random
restarts.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = s}."""
    if s <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a sequence to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, seq)

def _evaluate(seq: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence."""
    # evaluate_sequence is assumed to be present in the execution env.
    return float(evaluate_sequence(seq.tolist()))

# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, target_sum: float):
    """Solve the LP from the reference implementation, return scaled g."""
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None

    n = f.shape[0]
    # Build matrix A such that (A @ g) == conv(f, g) (full convolution)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:  # pragma: no cover
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    s = np.sum(g_val)
    if s < 1e-12:
        return None
    return g_val * (target_sum / s)

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The search proceeds over several lengths, uses a smooth‑max gradient
    descent on the auto‑convolution peak, repeatedly projects onto the
    ℓ¹‑simplex, and finally attempts a short LP refinement.  The best
    sequence found before the time budget expires is returned as a list
    of floats.
    """
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # Lengths to explore – larger n gives more flexibility but costs more.
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    # Number of restarts per length (will be reduced automatically if we run out of time)
    restarts_per_n = 3

    # Helper to obtain an initial vector for a given length.
    def _initial_vector(n, target):
        # If we have a previously found sequence, up‑sample it.
        if best_seq is not None and len(best_seq) != n:
            vec = _upsample(best_seq, n)
        else:
            # random point on the simplex
            vec = rng.random(n) + 0.1   # avoid exact zeros
            vec = vec / np.sum(vec) * target
        # Final projection to be safe
        return _project_simplex(vec, target)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            a = _initial_vector(n, target_sum)
            cur_score = _evaluate(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            # ------- smooth‑max gradient descent ----------
            max_iters = 2500
            alpha = 8.0          # smoothing parameter for the log‑sum‑exp
            eta = 5e-4           # learning rate
            beta = 0.9           # momentum coefficient
            momentum = np.zeros_like(a)

            for it in range(max_iters):
                if time.time() >= deadline:
                    break

                conv = np.convolve(a, a)
                # soft‑max weights (log‑sum‑exp surrogate of max)
                scaled = alpha * conv
                max_scaled = np.max(scaled)
                w = np.exp(scaled - max_scaled)
                w = w / np.sum(w)

                # correlation a ⋆ w  (gives Σ_j a_j w_{i+j})
                corr_full = np.correlate(a, w, mode='full')
                start_idx = len(w) - 1
                corr = corr_full[start_idx:start_idx + n]

                grad = corr + a          # gradient of the smoothed objective
                momentum = beta * momentum + (1.0 - beta) * grad
                a_new = a - eta * momentum
                a_new = _project_simplex(a_new, target_sum)

                new_score = _evaluate(a_new)
                # keep the best ever seen
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"[gd] n={n} restart={restart} it={it} score={new_score:.6f}")

                # accept improvement, otherwise shrink step size
                if new_score < cur_score - 1e-9:
                    a = a_new
                    cur_score = new_score
                    # slowly increase smoothing to focus on the true max
                    if (it + 1) % 200 == 0 and alpha < 50.0:
                        alpha *= 1.08
                else:
                    # no improvement – contract learning rate and inject a tiny jitter
                    eta *= 0.96
                    if (it + 1) % 150 == 0:
                        jitter = rng.normal(scale=0.02 * target_sum / n, size=n)
                        a = _project_simplex(np.clip(a + jitter, 0.0, None), target_sum)

                # early exit if we are already very close to the trivial lower bound
                lower = 2.0 * n / (2.0 * n - 1.0)
                if conv.max() <= lower * 1.02:
                    break

            # ------- optional LP refinement (few steps) -------
            # after gradient descent we try a short LP based improvement.
            # Only attempt if cvxpy is available; otherwise skip.
            try:
                import cvxpy  # noqa: F401
                lp_available = True
            except Exception:
                lp_available = False

            if lp_available:
                # recompute current maximal auto‑convolution value
                conv = np.convolve(a, a)
                M = float(conv.max())
                # perform up to 30 LP steps
                for lp_it in range(30):
                    if time.time() >= deadline:
                        break
                    g_scaled = _lp_step(a, M, target_sum)
                    if g_scaled is None:
                        break
                    conv_g = np.convolve(g_scaled, g_scaled)
                    M_new = float(conv_g.max())
                    if M_new < M - 1e-12:
                        a = g_scaled
                        M = M_new
                        cur_score = float(2.0 * n * M_new / (target_sum ** 2))
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = a.copy()
                            print(f"[LP-ref] n={n} restart={restart} lp_it={lp_it} score={cur_score:.6f}")
                    else:
                        # try a convex combination as in the reference code
                        t_mix = 0.2
                        f_candidate = (1.0 - t_mix) * a + t_mix * g_scaled
                        conv_cand = np.convolve(f_candidate, f_candidate)
                        M_cand = float(conv_cand.max())
                        if M_cand < M:
                            a = f_candidate
                            M = M_cand
                            cur_score = float(2.0 * n * M_cand / (target_sum ** 2))
                            if cur_score < best_score:
                                best_score = cur_score
                                best_seq = a.copy()
                                print(f"[LP-mix] n={n} restart={restart} lp_it={lp_it} score={cur_score:.6f}")
                    # stop early if no progress
                    if abs(M - M_new) < 1e-9:
                        break

    # ------------------------------------------------------------------
    # If nothing succeeded (unlikely) fall back to a tiny random‑walk.
    if best_seq is None:
        # very cheap fallback – one short random walk of length 200
        n = 200
        target_sum = math.sqrt(2.0 * n)
        a = np.full(n, target_sum / n, dtype=float)
        best_seq = a.copy()
        best_score = _evaluate(a)
        deadline_fallback = time.time() + max(0.0, deadline - time.time())
        while time.time() < deadline_fallback:
            cand = a + rng.normal(scale=0.03, size=n)
            cand = np.clip(cand, 0.0, None)
            cand = cand / np.sum(cand) * target_sum
            sc = _evaluate(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                a = cand
        print(f"[fallback] final score {best_score:.6f}")

    # final representation as plain list
    return [float(x) for x in best_seq]
