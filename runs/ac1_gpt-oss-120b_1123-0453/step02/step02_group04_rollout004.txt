def project_simplex(v, s):
    # v: 1D np array
    # ensure non-negative after projection.
    # If sum(v) == s and all v >=0, return v.
    # else sort descending.
    # algorithm from Duchi et al. (2008).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

The goal is to minimise the value returned by `evaluate_sequence`.  
For a non‑negative sequence `a` with ℓ¹‑norm `S = Σ a_i` the scorer returns  

```
score = 2·n·max(conv(a,a)) / S²
```

If we normalise every candidate to the “standard” ℓ¹‑norm  

```
S = √(2·n)
```

the factor `2·n / S²` equals `1` and the score is exactly the
maximum entry of the auto‑convolution `a*a`.  
Hence the problem reduces to

*“Find a non‑negative vector of length `n` with fixed ℓ¹‑norm
that minimises the largest coefficient of its auto‑convolution.”*

The implementation below follows a **Frank‑Wolfe / line‑search** scheme
that is known to work well for this type of optimisation:

1. **Initialisation** – start from the best known global sequence
   (`height_sequence_1`) or from a random vector, renormalised to
   `√(2·n)`.

2. **Linear‑programming direction** – with the current vector `f`
   we solve  

```
max Σ g_i
s.t.   (f * g) ≤ M         (M = max(f*f))
       g ≥ 0
```

   This LP (implemented with CVXPY) yields a feasible direction `g`
   that can add the most ℓ¹‑mass without raising the current worst
   convolution value `M`.

3. **Optimal convex combination** – after scaling `g` to the required
   ℓ¹‑norm we compute all three convolutions  
   `c₁ = f*f , c₂ = f*g , c₃ = g*g`.  
   For a mixing coefficient `α∈[0,1]`

```
h(α) = (1−α)f + α g
conv_h = (1−α)² c₁ + 2α(1−α) c₂ + α² c₃
```

   `max(conv_h)` is a convex function of `α`.  
   We obtain the *best* `α` by a cheap two‑stage grid search
   (coarse 51 points, then a refined 21‑point interval).

   The new point `f ← h(α*)` is automatically normalised.

4. **Projection on the simplex** – floating point noise may break the
   exact ℓ¹‑norm; we re‑project onto the simplex
   `{x≥0, Σx = √(2·n)}` using the O(n log n) algorithm of
   Duchi et al.

5. **Random perturbations** – after every block of LP steps a short
   Gaussian hill‑climb (projected back onto the simplex) is performed
   to escape possible plateaus.

6. **Multiple lengths & restarts** – the procedure is repeated for a
   small set of lengths (`200 … 800`).  
   The best sequence over all runs is kept.

All work respects the 1000 s budget, prints intermediate scores and
falls back to a pure random walk if CVXPY cannot be imported.

```python
"""
A Frank‑Wolfe style optimiser for the step‑function problem.

*  Uses a linear programme to find a feasible direction that can add the
   largest ℓ¹‑mass while keeping the current worst auto‑convolution ≤ M.
*  After each LP the optimal convex mixing coefficient α is obtained by
   a cheap line‑search on the exact quadratic expression of the
   convolution.
*  The iterate is projected onto the ℓ¹‑simplex to keep the normalisation.
*  Random Gaussian perturbations are applied after each block of LP
   refinements.
*  Several problem sizes (200 … 800) are tried and the best sequence is
   returned.  If CVXPY is unavailable a simple random‑search fallback is
   used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    # shift to make entries non‑negative after projection
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: build the (2n‑1) × n convolution matrix A such that
# (A @ g)[k] = (f * g)[k] .
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        cols = k - np.arange(i_start, i_end + 1)
        A[k, cols] = f[i_start : i_end + 1]
    return A


# ----------------------------------------------------------------------
# Helper: fast convolutions via FFT (real input → real output)
# ----------------------------------------------------------------------
def _fft_conv(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Return the linear convolution a * b of length 2*len(a)-1 using FFT."""
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _auto_conv(x: np.ndarray, L: int) -> np.ndarray:
    """Return x * x (auto‑convolution) of length 2·len(x)‑1."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.shape[0] - 1]


# ----------------------------------------------------------------------
# Simple random‑search fallback (used when CVXPY cannot be imported)
# ----------------------------------------------------------------------
def _fallback_random_search(budget_s: float, seed: int = 0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    # a safe default length
    n = 300
    target_sum = math.sqrt(2.0 * n)
    seq = rng.random(n) + 0.1
    seq = seq / np.sum(seq) * target_sum
    best_seq = seq.copy()
    # compute max convolution directly (this equals the score)
    L = 1 << ((2 * n - 1).bit_length())
    best_score = np.max(_auto_conv(seq, L))

    while time.monotonic() < deadline:
        cand = best_seq + rng.normal(scale=0.05, size=n)
        cand = np.clip(cand, 0.0, None)
        cand = _project_onto_simplex(cand, target_sum)
        sc = np.max(_auto_conv(cand, L))
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print("[fallback] best score =", best_score)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging system.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.

    The optimiser works as follows:

    * try several lengths (200 … 800)
    * for each length run a Frank‑Wolfe style loop:
        – solve an LP that adds as much ℓ¹‑mass as possible without
          increasing the current worst auto‑convolution
        – compute the optimal mixing coefficient α that minimises the new
          max‑convolution (exact line‑search on a quadratic form)
        – project back onto the ℓ¹‑simplex
    * after a block of LP steps perform a short Gaussian hill‑climb
    * keep the best sequence seen over all lengths
    * if CVXPY is unavailable fall back to a pure random walk
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # small safety margin for clean termination
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try to import CVXPY – the LP part depends on it.
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp

        HAS_CVX = True
    except Exception:  # pragma: no cover
        HAS_CVX = False

    # ------------------------------------------------------------------
    # If a seed sequence is supplied by the harness, use it.
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        if init_seq.ndim == 1:
            L_init = 1 << ((2 * init_seq.shape[0] - 1).bit_length())
            # compute its score (max auto‑convolution)
            init_score = np.max(_auto_conv(init_seq, L_init))
            best_seq = init_seq.copy()
            best_score = float(init_score)
            print("[init] using provided height_sequence_1, score =", init_score)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Problem sizes to explore (increase flexibility with time).
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)  # ℓ¹‑norm used in the scorer
        L_fft = 1 << ((2 * n - 1).bit_length())  # FFT length (power of two)

        # --------------------------------------------------------------
        # Initialise f for this size.
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            f = best_seq.copy()
            f = f * (target_sum / np.sum(f))
        elif best_seq is not None and best_seq.shape[0] < n:
            f = np.zeros(n, dtype=np.float64)
            f[: best_seq.shape[0]] = best_seq
            f = f * (target_sum / np.sum(f))
        else:
            # completely fresh start
            f = rng.random(n) + 0.1
            f = f / np.sum(f) * target_sum

        # --------------------------------------------------------------
        # Main optimisation loop for this length.
        # --------------------------------------------------------------
        max_lp_iters = 60  # enough to see progress, but limited by time

        for it in range(max_lp_iters):
            if time.monotonic() >= deadline:
                break

            # current worst auto‑convolution value M
            conv_f = _auto_conv(f, L_fft)
            M = float(np.max(conv_f))

            # ----------------------------------------------------------
            # 1) Linear programme: maximise Σ g_i  subject to (f*g) ≤ M , g ≥ 0
            # ----------------------------------------------------------
            if not HAS_CVX:
                # cannot continue the LP‑based phase without CVXPY
                break

            A = _conv_matrix(f)  # (2n‑1) × n matrix

            g_var = cp.Variable(n)
            prob = cp.Problem(
                cp.Maximize(cp.sum(g_var)),
                [g_var >= 0, A @ g_var <= M],
            )
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
            except Exception:  # pragma: no cover
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000)

            if g_var.value is None:
                print(f"[LP] n={n} it={it} – solver failed")
                break

            g_raw = np.maximum(g_var.value, 0.0)
            sum_g = np.sum(g_raw)
            if sum_g < 1e-12:
                break

            # ----------------------------------------------------------
            # 2) Rescale g to the target ℓ¹‑norm
            # ----------------------------------------------------------
            g = g_raw * (target_sum / sum_g)

            # ----------------------------------------------------------
            # 3) Compute the three convolutions needed for the line search
            # ----------------------------------------------------------
            conv_g = _auto_conv(g, L_fft)
            conv_fg = _fft_conv(f, g, L_fft)

            # ----------------------------------------------------------
            # 4) Optimal mixing coefficient α (grid search + refinement)
            # ----------------------------------------------------------
            # coarse grid
            alphas = np.linspace(0.0, 1.0, 51)
            best_alpha = 0.0
            best_val = float("inf")
            for a in alphas:
                combo = ((1 - a) ** 2) * conv_f + (2 * a * (1 - a)) * conv_fg + (a ** 2) * conv_g
                val = float(np.max(combo))
                if val < best_val:
                    best_val = val
                    best_alpha = a

            # refined grid around the best coarse point
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            alphas_ref = np.linspace(lo, hi, 21)
            for a in alphas_ref:
                combo = ((1 - a) ** 2) * conv_f + (2 * a * (1 - a)) * conv_fg + (a ** 2) * conv_g
                val = float(np.max(combo))
                if val < best_val:
                    best_val = val
                    best_alpha = a

            # ----------------------------------------------------------
            # 5) Update f and re‑project onto the simplex
            # ----------------------------------------------------------
            f = (1 - best_alpha) * f + best_alpha * g
            # tiny numerical drift – re‑project to keep the exact norm
            cur_sum = np.sum(f)
            if abs(cur_sum - target_sum) > 1e-9:
                f = _project_onto_simplex(f, target_sum)

            # record progress
            if best_val < best_score:
                best_score = best_val
                best_seq = f.copy()
                print(f"[LP] n={n:3d} it={it:2d} α={best_alpha:.4f} score={best_val:.6f}")

            if best_score < 1.503:
                break

        # --------------------------------------------------------------
        # 6) Short random‑perturbation hill‑climb to polish the current best.
        # --------------------------------------------------------------
        if best_seq is None:
            continue

        perturb_iters = 250
        for _ in range(perturb_iters):
            if time.monotonic() >= deadline:
                break
            cand = best_seq + rng.normal(scale=0.02, size=best_seq.shape[0])
            cand = np.clip(cand, 0.0, None)
            cand = _project_onto_simplex(cand, target_sum)

            sc = np.max(_auto_conv(cand, L_fft))
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[perturb] n={n} score={sc:.6f}")
                if best_score < 1.503:
                    break

    # ------------------------------------------------------------------
    # If nothing succeeded (e.g. CVXPY missing) run the pure fallback.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_seq, fallback_score = _fallback_random_search(
            max(0.0, float(budget_s) - (time.monotonic() - start_time)),
            seed=seed,
        )
        best_seq = fallback_seq
        best_score = fallback_score
        print("[fallback] final score =", best_score)

    # --------------------------------------------------------------
    # Return the result as a plain Python list of floats.
    # --------------------------------------------------------------
    return [float(x) for x in best_seq]
```
