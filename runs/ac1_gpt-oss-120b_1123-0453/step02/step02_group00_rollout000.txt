"""
Gradient‑descent based refinement for the step‑function attack.

The evaluation score for a sequence `a` (with ℓ¹‑norm fixed to √(2·n))
is exactly the maximal entry of the auto‑convolution `a*a`.
We therefore minimise  M = max_k (a*a)[k]  under the constraints

    a_i ≥ 0 ,   Σ a_i = √(2·n) .

The algorithm repeatedly:
    1. Starts from a random feasible vector (several initialisations,
       also re‑using the best known construction if available).
    2. Approximates the non‑smooth objective “max” by a smooth
       soft‑max (log‑sum‑exp) with a temperature β.
    3. Computes the gradient of the soft‑max w.r.t. the coefficients.
    4. Takes a gradient step and projects back onto the simplex
       (non‑negative with the prescribed ℓ¹‑norm).
    5. Keeps the best vector seen so far.

A short stochastic hill‑climbing phase (Gaussian perturbations)
polishes the final answer.  The routine respects the global time budget
and returns the best list of floats found before the deadline.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, Σx = target}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = target}."""
    if target < 0:
        raise ValueError("target sum must be non‑negative")
    # Fast path
    if np.all(v >= 0) and abs(np.sum(v) - target) < 1e-12:
        return v

    # Duchi et al. algorithm
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be zero
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: gradient of the smooth‑max of the auto‑convolution
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return a probability vector proportional to exp(beta * conv)."""
    # Numerical stability: shift by the maximum exponent.
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    return exp_scaled / np.sum(exp_scaled)

def _gradient_autocorr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k (a*a)[k] .
    Using the identity
        grad_i = 2 * Σ_j a_j * w_{i+j}
    which can be obtained by a single convolution.
    """
    a_rev = a[::-1]
    tmp = np.convolve(a_rev, w)          # length = n + (2n‑1) – 1 = 3n‑2
    n = a.shape[0]
    # elements n‑1 … n‑1+n‑1 contain the desired sums
    grad = 2.0 * tmp[n - 1: n - 1 + n]
    return grad

# ----------------------------------------------------------------------
# Core optimisation routine (gradient descent with soft‑max)
# ----------------------------------------------------------------------
def _gd_optimize(
    n: int,
    target_sum: float,
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 500,
    beta0: float = 5.0,
    lr0: float = 0.8,
) -> Tuple[np.ndarray, float]:
    """Run the gradient‑descent loop for a fixed length `n`."""
    a = start_a.copy()
    beta = beta0
    lr = lr0
    best_a = a.copy()
    # initial score (exact max of auto‑convolution)
    best_score = float(np.max(np.convolve(a, a)))
    # iteration
    while time.monotonic() < deadline and max_iters > 0:
        # (1) auto‑convolution
        conv = np.convolve(a, a)               # length 2n‑1
        M = float(np.max(conv))

        # keep the best exact score
        if M < best_score:
            best_score = M
            best_a = a.copy()

        # (2) smooth‑max weights
        w = _smooth_weights(conv, beta)

        # (3) gradient of the smooth‑max
        grad = _gradient_autocorr(a, w)

        # (4) gradient step + projection
        a = a - lr * grad
        a = _project_to_simplex(a, target_sum)

        # (5) schedule updates
        lr *= 0.9995
        beta *= 1.001
        # keep parameters in a reasonable range
        if lr < 1e-6:
            lr = lr0
        if beta > 200.0:
            beta = 200.0

        max_iters -= 1
    return best_a, best_score

# ----------------------------------------------------------------------
# Small stochastic hill‑climbing refinement (Gaussian perturbations)
# ----------------------------------------------------------------------
def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.5,
) -> Tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing that accepts any improvement."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = float(np.max(np.convolve(best, best)))

    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_to_simplex(cand, np.sum(best))   # keep ℓ¹‑norm
        cand_score = float(np.max(np.convolve(cand, cand)))
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # exponential decay of step size
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The algorithm proceeds in three stages:
        1. Load a previously discovered construction if available.
        2. Gradient‑descent refinement on several lengths (200‑1200) and
           multiple random restarts.
        3. Final Gaussian hill‑climbing polish.

    It respects a global time budget `budget_s` (seconds) and returns
    a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # Reserve a tiny margin for the return path
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------------
    # 0️⃣  Initialise with any known good vector (if present)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        seq0 = globals()["height_sequence_1"]
        if isinstance(seq0, list) and len(seq0) > 1:
            arr0 = np.array(seq0, dtype=np.float64)
            # Clip and project to a feasible point with the proper ℓ¹‑norm
            arr0 = np.clip(arr0, 0.0, 1000.0)
            n0 = arr0.shape[0]
            target0 = math.sqrt(2.0 * n0)
            arr0 = _project_to_simplex(arr0, target0)
            score0 = float(np.max(np.convolve(arr0, arr0)))
            best_seq = arr0.copy()
            best_score = score0
            print(f"[init] Loaded external candidate n={n0} score={score0:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Gradient‑descent optimisation on a variety of lengths
    # --------------------------------------------------------------
    # Lengths are chosen to balance search power and runtime.
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200]
    # Mapping length → number of restarts (more restarts for smaller n)
    restarts_per_len = {200: 6, 300: 5, 400: 5, 500: 4, 600: 4,
                        800: 3, 1000: 2, 1200: 2}
    # Mapping length → max gradient‑descent iterations per restart
    iters_per_len = {200: 800, 300: 700, 400: 600, 500: 500,
                     600: 450, 800: 350, 1000: 300, 1200: 250}
    target_bound = 1.503

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        restarts = restarts_per_len.get(n, 3)
        max_iters = iters_per_len.get(n, 300)

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ---------- initialise ----------
            # Occasionally reuse the global best if it has the same length
            if best_seq is not None and best_seq.shape[0] == n:
                a0 = best_seq.copy()
                # add a small random jitter to avoid staying stuck
                a0 += rng.normal(scale=0.01 * target_sum, size=n)
                a0 = _project_to_simplex(a0, target_sum)
            else:

                typ = rng.integers(0, 3)
                if typ == 0:          # uniform
                    a0 = rng.random(n)
                elif typ == 1:        # exponential (many small, few large)
                    a0 = rng.exponential(scale=1.0, size=n)
                else:                 # heavy‑tailed via power transformation
                    a0 = rng.random(n) ** 2
                a0 = _project_to_simplex(a0, target_sum)

            # ---------- run GD ----------
            a_opt, score_opt = _gd_optimize(
                n=n,
                target_sum=target_sum,
                start_a=a0,
                deadline=deadline,
                max_iters=max_iters,
                beta0=5.0,
                lr0=0.8,
            )
            # Keep the best overall vector
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[GD] n={n} restart={r} best_score={best_score:.6f}")

            # Early exit if we have already crossed the target
            if best_score < target_bound:
                print("[GD] Target bound reached – exiting early.")
                return best_seq.tolist()

    # --------------------------------------------------------------

    # --------------------------------------------------------------

            best_seq = refined_seq
            best_score = refined_score
            print(f"[local] improved to {best_score:.6f}")

    # Return a plain Python list as required by the runner
    return [float(x) for x in best_seq]
