def auto_conv(f):
    n=len(f)
    N=1
    while N < 2*n-1: N <<=1
    F=np.fft.rfft(f, N)
    conv=np.fft.irfft(F*F, N)[:2*n-1]
    return conv

def simplex_projection(v, s):
    # v: numpy array, nonnegative after projection
    # Implementation from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate – gradient‑descent on a smooth approximation of the
ℓ∞‑norm of the self‑convolution.

The evaluation function is
    2·n·max_k (f*f)[k] / (sum f)^2 .
Because the objective is homogeneous we may fix the ℓ¹‑norm to 1.
For a vector a (∑a_i=1) the score reduces to 2·n·max(conv(a,a)).
The maximum of the convolution is approximated by a softmax:
    Φ_λ(a)=log∑_k exp(λ·(conv_k−max_conv)).
∇Φ_λ(a) = 2·( w * a_rev )[n−1:n−1+n]   with
    w_k = exp(λ·(conv_k−max_conv)) / ∑ exp(·).

We minimise Φ_λ by projected gradient steps (project to the simplex
∑a_i=1, a_i≥0).  λ is increased geometrically (0.5 → 12) so that the
optimiser first smooths the landscape and later focuses on the true
maximum.  After the gradient phase a brief simulated‑annealing (SA)
finetunes the best vector found.

The routine runs for the allotted time (≈1000 s) and returns the best
sequence as a plain list of floats.
"""

def _score(seq: np.ndarray) -> float:
    """Wrapper for the provided evaluator (expects a Python list)."""
    return float(evaluate_sequence(seq.tolist()))


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x)
    m = len(y)
    N = 1 << ((n + m - 1).bit_length())        # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)
    return conv[: n + m - 1]                     # truncate to linear length


def _auto_conv(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·n‑1)."""
    return _conv_fft(a, a)


def _normalize_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project onto the probability simplex (non‑negative, sum = 1).
    Uses the algorithm of Duchi et al. (O(n log n)).
    """
    # make a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    -------
    best_a : np.ndarray
        The best sequence found (ℓ¹‑norm = 1).
    best_score : float
        Corresponding evaluation value (2·n·max(conv)).
    """
    best_a = None
    best_score = float("inf")

    # schedule of smoothing parameters
    lambda_vals = [0.5, 1.0, 2.0, 4.0, 8.0, 12.0]

    # basic step size (will be scaled down later)
    base_step = 0.2

    for run in range(restarts):
        # ----- random start on the simplex -----
        a = rng.random(n)
        a = _normalize_to_simplex(a)          # sum = 1, non‑negative

        # keep the best vector seen within this restart
        run_best_a = a.copy()
        conv = _auto_conv(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        for lam in lambda_vals:
            # a modest number of internal iterations – we break early if out of time
            for it in range(5000):
                if time.monotonic() >= deadline:
                    return best_a, best_score

                # ----- convolution and soft‑max weights -----
                conv = _auto_conv(a)
                max_c = conv.max()
                # stabilize the exponentials
                exp_vals = np.exp(lam * (conv - max_c))
                w = exp_vals / exp_vals.sum()               # shape 2·n‑1, sum(w)=1

                # ----- gradient of the smooth‑max -----
                a_rev = a[::-1]
                grad_full = _conv_fft(w, a_rev)              # length 3·n‑2
                grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # slice to length n

                # ----- projected gradient step -----
                step = base_step / (math.sqrt(it + 1) * (lam + 1.0))
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)               # enforce non‑negativity
                a_new = _normalize_to_simplex(a_new)         # keep sum = 1
                a = a_new

                # ----- evaluate -----
                conv = _auto_conv(a)
                score = 2.0 * n * conv.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    # occasional progress print
                    print(f"[{n}] λ={lam:.1f} it={it:04d}  new best {best_score:.6f}")

                # keep the best of the current restart (helps with restarts)
                if score < cur_score:
                    cur_score = score
                    run_best_a = a.copy()

        # after exhausting all λ values, replace a with the best of this restart
        a = run_best_a

    return best_a, best_score


def _simulated_annealing(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    A lightweight SA that perturbs a single entry, re‑normalises to
    ℓ¹‑norm = 1 and accepts moves with the Metropolis rule.
    """
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4

    n = len(seq)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        fraction = elapsed / (deadline - start)
        temp = temp_start * (temp_end / temp_start) ** fraction

        # propose a change at a random coordinate
        idx = rng.integers(n)
        delta = rng.normal(scale=0.5) * temp * 5.0
        cand = cur_seq.copy()
        cand[idx] = max(0.0, cand[idx] + delta)

        # re‑normalise to sum = 1
        cand = _normalize_to_simplex(cand)

        cand_score = 2.0 * n * _auto_conv(cand).max()

        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

        # tiny random reshuffle occasionally to escape flat regions
        if rng.random() < 0.001:
            cand = cur_seq + rng.normal(scale=0.01, size=n)
            cand = _normalize_to_simplex(cand)
            cur_seq = cand
            cur_score = 2.0 * n * _auto_conv(cand).max()

    return best_seq, best_score

# Main entry point – the competition harness will call this

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    evaluation score under the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a one‑second safety margin
    deadline = start + max(0.0, float(budget_s) - 1.0)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Use any previously discovered construction if present
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            if seq0.ndim == 1 and seq0.size > 0:
                # normalise to ℓ¹‑norm = 1 (the score is homogeneous)
                seq0 = _normalize_to_simplex(seq0)
                sc0 = 2.0 * len(seq0) * _auto_conv(seq0).max()
                best_seq = seq0.copy()
                best_score = sc0
                print(f"[init] using height_sequence_1 – score {best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  If no prior construction, start from a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 300
        best_seq = np.full(n0, 1.0 / n0, dtype=np.float64)
        best_score = 2.0 * n0 * _auto_conv(best_seq).max()
        print(f"[init] uniform start (n={n0}) – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Core optimisation – smooth‑max gradient descent on several
    #     lengths.  Larger n gives more freedom, but each iteration is
    #     cheap (FFT based).  We stop early if the time budget is exhausted.
    # ------------------------------------------------------------------
    # a set of lengths to explore – ordered from modest to larger
    lengths = [250, 300, 350, 400, 450, 500, 600, 800]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # allocate a modest time slice per size; the inner routine respects the global deadline
        seq, sc = _smooth_max_optimize(n, deadline, rng, restarts=2)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global] improved with n={n} → score {best_score:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing – simulated annealing on the best vector found
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        cur_score = 2.0 * len(best_seq) * _auto_conv(best_seq).max()
        best_seq, cur_score = _simulated_annealing(best_seq, cur_score, deadline, rng)
        best_score = cur_score
        print(f"[final] score after SA = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Fallback – should never happen, but guarantees a return value
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # The harness expects a plain Python list of floats; the scorer
    # already clamps values to [0,1000] so we simply return the raw numbers.
    return [float(x) for x in best_seq]
