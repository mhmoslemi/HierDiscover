def project_onto_simplex(v, S):
    # v: np.ndarray
    # Return x >= 0, sum x = S, minimizing ||x - v||_2^2
    # Works for arbitrary S >0.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    # find rho = max i where u_i > cssv_i / (i+1) (i zero-index)
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        # all negative, return zeros (should not happen because S>0)
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_descent(a0, S, deadline, max_iters=10_000, step0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_M = np.max(np.convolve(a, a))
    iter = 0
    while time.monotonic() < deadline and iter < max_iters:
        g, M = gradient_of_max_conv(a)
        # step size schedule
        step = step0 / np.sqrt(iter + 1)
        v = a - step * g
        a = project_onto_simplex(v, S)
        # compute new M
        newM = np.max(np.convolve(a, a))
        if newM < best_M:
            best_M = newM
            best_a = a.copy()
        iter += 1
    return best_a, best_M

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 0.5)

    # Determine length n
    n_candidates = [200, 300, 400, 500, 600, 700]
    # We'll try each sequentially within time.

    best_seq = None
    best_score = float('inf')

    def evaluate(seq):
        # Use evaluate_sequence directly for safety
        return float(evaluate_sequence(seq.tolist()))

    def normalize(vec, target_sum):
        # ensure sum positive
        s = np.sum(vec)
        if s == 0:
            return np.full_like(vec, target_sum / len(vec))
        return vec * (target_sum / s)

    # if height_sequence_1 exists
    try:
        init_seq_global = np.array(height_sequence_1, dtype=float)
    except Exception:
        init_seq_global = None

    for n in n_candidates:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # choose initial vectors: maybe use global if same n else random
        init_vectors = []
        if init_seq_global is not None and len(init_seq_global) == n:
            init_vectors.append(init_seq_global.copy())
        # Basic uniform start

        # Random start(s)
        for _ in range(2):
            v = rng.random(n)
            init_vectors.append(normalize(v, target_sum))

        for init_idx, a0 in enumerate(init_vectors):
            if time.monotonic() >= deadline:
                break

            # ensure positivity

        seq = np.full(n, math.sqrt(2.0*n)/n, dtype=float)

def subgradient_descent_wrapper(a0, target_sum, deadline, step0=1.0, max_iters=200000):
    a = a0.copy()
    best_a = a.copy()
    conv = conv_fft(a)
    best_M = conv.max()
    n = len(a)
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # Get max conv shift(s)
        # Use idx = np.argmax(conv)
        idx = conv.argmax()
        # Compute gradient g for this idx
        g = np.zeros_like(a)
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        # Note: indices of a that contribute: i in [lo, hi]
        # g[i] = 2 * a[idx - i]
        # generate array slice
        i_range = np.arange(lo, hi + 1)
        g[lo:hi + 1] = 2.0 * a[idx - i_range]
        # step size
        step = step0 / np.sqrt(it + 1)
        v = a - step * g
        # projection onto simplex sum target_sum
        a = project_onto_simplex(v, target_sum)
        # recompute convolution
        conv = conv_fft(a)
        M = conv.max()
        if M < best_M - 1e-12:   # improvement
            best_M = M
            best_a = a.copy()
        # Optionally occasional random reset or enlarge step if stuck.
        it += 1
    return best_a, best_M

conv_fft uses FFT:

def conv_fft(a):
    n = a.shape[0]
    # compute size as power of two >= 2*n-1
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    # Keep only first 2*n-1
    return conv[:2*n - 1]

def improve_by_transfer(seq, deadline, max_iters=500000, init_step=0.5):
    # seq: numpy array
    n = len(seq)
    # compute initial conv
    conv = np.convolve(seq, seq)
    sum_a = seq.sum()
    # compute score
    best = seq.copy()
    best_conv = conv.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # step size for delta fraction may start high
    step = init_step
    it = 0
    rng = np.random.default_rng()
    while time.monotonic() < deadline and it < max_iters:
        # pick random i with a_i > 1e-12
        # we can use weighted random: choose i based on a_i magnitude (more mass).
        # For simplicity, random index, then if a[i] < 1e-12, continue.
        i = rng.integers(0, n)

        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # Determine delta fraction of a[i]
        delta = seq[i] * step * rng.random()
        # delta <= seq[i] ensured
        # Apply move
        transfer_mass(seq, conv, i, j, delta)
        # compute new max
        cur_M = conv.max()
        cur_score = float(2.0 * n * cur_M / (sum_a * sum_a))

        if cur_score < best_score:
            # keep new best

        # decay step size gradually
        step *= 0.99995

            step = init_step

    return best, best_score

"""
Search for a non‑negative step sequence with a low evaluation score.

The algorithm works in three phases:

1️⃣  Initialise several candidate vectors (including the global
    `height_sequence_1` if it is available).  The vectors are random
    positive numbers scaled to a moderate ℓ¹‑norm – the exact scale does
    not matter because the evaluation function is homogeneous.

2️⃣  A specialised hill‑climbing that moves a small amount of mass from
    one index to another (preserving the total ℓ¹‑norm).  The convolution
    is updated *incrementally* using the exact algebraic formula,
    therefore each iteration costs only O(n) instead of O(n log n)
    convolutions.  Only moves that strictly improve the score
    (2·n·max(conv)/(sum²)) are kept; the step size decays slowly and
    the procedure is restarted several times for different sequence
    lengths.

3️⃣  A final gentle Gaussian perturbation (the `_local_search` from the
    reference code) to polish the best vector found.

All intermediate vectors respect the non‑negativity and clipping rules
implemented inside `evaluate_sequence`.  The function returns the best
list of floats before the time budget expires.
"""

import time
import math

def _score(seq: np.ndarray) -> float:
    """Return the official score using the provided `evaluate_sequence`."""
    # evaluate_sequence expects a Python list.
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex
        { x >= 0 , sum(x) = target_sum }.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index `i` to index `j` (i != j, delta > 0, delta <= a[i]).
    The function updates the vector `a` **in‑place** and also updates the
    convolution `conv = np.convolve(a, a)` **in‑place** using the exact algebraic
    formula (O(n) work).
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # ----- linear part (2 * convolution(d, a)) -----
    #   conv[i : i+n]  -= 2 * delta * a
    #   conv[j : j+n]  += 2 * delta * a
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # ----- quadratic part (convolution(d, d)) -----
    # d has -delta at i and +delta at j
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta          # (-δ)²
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta    # (-δ)*(+δ) + (+δ)*(-δ)
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta          # (+δ)²

    # ----- update the vector -----
    a[i] -= delta
    a[j] += delta


def _improve_by_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that repeatedly moves a small amount of mass from one index
    to another.  The convolution is kept up‑to‑date incrementally.
    Returns the best sequence encountered and its score.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)               # exact convolution
    sum_a = a.sum()
    # score = 2·n·max(conv) / sum_a²  (scale‑invariant)
    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)
    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        # pick a source index with some positive mass
        i = rng.integers(0, n)
        if a[i] <= 1e-12:          # negligible – try again
            continue
        # pick a different target index
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # choose a fraction of the source mass to move
        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        # apply the move
        _transfer_mass(a, conv, i, j, delta)

        # evaluate the new score using the cheap formula
        cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

        if cur_score < best_score:
            # improvement – keep it
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert the move
            _transfer_mass(a, conv, j, i, delta)

        # slowly decay the step size; occasionally reset to avoid stagnation
        step *= 0.9999
        if step < 1e-5:
            step = step_init

        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that adds Gaussian noise to the whole
    vector (the technique from the reference implementation).  It is used
    only as a final polishing step.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial

    return best, best_score

def propose_candidate(seed: int = 0,

    # ------------------------------------------------------------------
    # 1️⃣  Initialise candidates (including the previously found construction)

            starts.append(init_global.copy())

                vec = rng.random(n) + 0.1
            starts.append(vec)

        # (the exact scale does not affect the score)
        target_sum = math.sqrt(2.0 * n)
        for start in starts:
            if time.monotonic() >= deadline:
                break
            s = start.sum()
            if s == 0.0:
                start[:] = target_sum / n
            else:
                start[:] = start * (target_sum / s)
            start = np.clip(start, 0.0, 1000.0)

            # ------------------------------------------------------------------

            # evaluate via the official function for safety
            official_score = _score(improved)
            if official_score < best_score:
                best_score = official_score
                best_seq = improved.copy()
                print(f"[transfer] n={n} score={best_score:.6f}")

            # early exit if we have already reached the target bound
            if best_score < 1.503:
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 3️⃣  Final Gaussian polishing (fallback if nothing was found)
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – a modest uniform vector
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    if time.monotonic() < deadline:
        refined, refined_score = _local_search(best_seq, deadline,
                                               step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] improved to {best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
