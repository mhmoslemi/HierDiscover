"""
Targeted stochastic local search for the step‑function minimisation.

The evaluation is   2·n·max(conv(a,a)) / (∑a)².
Because the search keeps the ℓ¹‑norm fixed at √(2·n), the score equals the
maximum entry of the (non‑reversed) auto‑convolution.  The algorithm
focuses on that peak:

* Each iteration finds a shift k where the convolution attains its current
  maximum.
* From the pair of indices (i,j) with i+j=k it chooses the larger coefficient
  as a source and moves a small amount of mass to a random destination.
  The convolution update is performed in O(n) using slice arithmetic, avoiding
  an FFT at every step.
* Improvements are always accepted; worse moves are accepted with a
  temperature‑dependent probability (simulated‑annealing schedule).
* Periodically the full convolution is recomputed to avoid drift.
* After the stochastic phase a short linear‑programming refinement (if
  CVXPY is available) enlarges the ℓ¹‑norm under the current worst‑convolution
  bound, often decreasing the score further.

The search runs for up to ≈1000 s, trying several lengths (200…1000) and a
few random restarts per length.  The best sequence found is returned as a
plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities -------------------------------------------------
# ----------------------------------------------------------------------


def max_conv_exact(seq: np.ndarray) -> float:
    """Exact maximum of the full convolution (O(n²) for small n)."""
    return float(np.max(np.convolve(seq, seq)))


def full_convolution(seq: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a (length 2·n‑1)."""
    return np.convolve(seq, seq)


def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Stretch ``seq`` to length ``new_len`` by nearest‑index mapping and re‑scale
    to the ℓ¹‑norm √(2·new_len).
    """
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    total = out.sum()
    if total > 0.0:
        target = math.sqrt(2.0 * new_len)
        out = out * (target / total)
    return out


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional) ------------------------------
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct matrix A (shape (2n‑1, n)) such that (A @ g)[k] = (f * g)[k].
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # indices i with j = k-i inside [0,n)
        A[k, k - np.arange(i_start, i_end + 1)] = f[i_start : i_end + 1]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple:
    """
    One LP refinement step: maximise ℓ¹‑mass under the current maximal
    (f*g) bound, then mix the result with the current vector.
    Returns (new_f, new_max_conv).
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv_exact(f)

    n = f.size
    M = max_conv_exact(f)
    A = build_conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    sum_g = g_val.sum()
    if sum_g < 1e-12:
        return f, M

    g_rescaled = g_val * (target_sum / sum_g)

    t = 0.3                      # modest mixing weight
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    return f_new, max_conv_exact(f_new)

def _targeted_move(a: np.ndarray, c: np.ndarray, rng: np.random.Generator) -> tuple:
    """
    Perform one targeted mass‑transfer move that tries to lower the current
    maximal convolution entry.  Returns (a_new, c_new, new_max, moved_flag).
    """
    n = a.size
    cur_max = c.max()
    max_idxs = np.flatnonzero(c == cur_max)
    if max_idxs.size == 0:
        return a, c, cur_max, False

    k = rng.choice(max_idxs)                     # shift where the peak occurs
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    i = rng.integers(i_min, i_max + 1)           # one index of a contributing pair
    j = k - i

    # source = larger of the two coefficients
    if a[i] >= a[j]:
        s = i
        other = j
    else:
        s = j
        other = i

    # random destination distinct from source
    d = rng.integers(0, n)
    while d == s:
        d = rng.integers(0, n)

    max_transfer = min(a[s], 1000.0 - a[d])
    if max_transfer <= 1e-12:
        return a, c, cur_max, False

    avg = a.sum() / n
    step = min(max_transfer, avg * 0.5)
    delta = rng.random() * step
    if delta <= 1e-12:
        return a, c, cur_max, False

    # ----- convolution update (O(n) slice arithmetic) -----
    a_new = a.copy()
    a_new[s] -= delta
    a_new[d] += delta

    c_new = c.copy()
    # linear part: 2·δ·(a shifted by d – a shifted by s)
    c_new[d:d + n] += 2.0 * delta * a
    c_new[s:s + n] -= 2.0 * delta * a
    # quadratic part
    c_new[2 * s] += delta * delta
    c_new[2 * d] += delta * delta
    c_new[s + d] -= 2.0 * delta * delta
    c_new = np.maximum(c_new, 0.0)

    new_max = c_new.max()
    return a_new, c_new, new_max, True

                a0 = a0 / a0.sum() * target_sum

            # ----- stochastic local search ---------------------------------------
            sub_deadline = min(deadline, time.monotonic() + 30.0)  # ≤30 s per restart
            a_loc, max_loc = targeted_sa(a0, target_sum, sub_deadline, rng)

            # ----- optional LP refinement ----------------------------------------
            try:
                import cvxpy as cp  # noqa: F401
                has_cvx = True
            except Exception:  # pragma: no cover
                has_cvx = False

            if has_cvx:
                for _ in range(4):
                    if time.monotonic() >= deadline:
                        break
                    a_loc, max_loc = lp_refine(a_loc, target_sum, deadline)
                max_loc = max_conv_exact(a_loc)  # recompute exact max

            # ----- record global best --------------------------------------------
            if max_loc < best_score - 1e-12:
                best_score = max_loc
                best_seq = a_loc.copy()
                print(
                    f"[global] n={n} restart={restarts} new best score={best_score:.6f}"
                )

            # stop early if we already beat the target
            if best_score < 1.503:
                print("[stop] target bound reached")
                break

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = max_conv_exact(best_seq)
        print("[fallback] uniform sequence used, score=", best_score)

    # Return plain Python list
    return [float(x) for x in best_seq]
