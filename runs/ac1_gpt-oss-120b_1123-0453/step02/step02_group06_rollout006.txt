def optimize_seq(seq, target_sum, deadline, rng):
    n=len(seq)
    fft_size= 1 << ((2*n-1).bit_length())
    best_seq=seq.copy()
    best_max = compute_max(seq)    
    T0 = 0.1
    start = time.monotonic()
    it=0
    while time.monotonic()<deadline and it<max_iters:
        it+=1
        conv = conv_fft(seq, fft_size)
        cur_max = float(conv.max())
        k_max = int(conv.argmax())
        if cur_max < best_max:
            best_max = cur_max
            best_seq = seq.copy()
        # compute contributions for k_max
        low = max(0, k_max - (n-1))
        high = min(k_max, n-1)
        idx_range = np.arange(low, high+1)
        contrib = seq[idx_range] * seq[k_max - idx_range]
        # choose donor index with high contribution
        # pick among top 5 perc maybe
        top_k = min(5, idx_range.size)
        if top_k > 0:
            # get sorted indices descending by contrib
            top_indices = np.argpartition(-contrib, top_k-1)[:top_k]
            donor_idx = idx_range[top_indices[rng.integers(top_k)]]
        else:
            donor_idx = rng.integers(n)
        # pick recipient distinct from donor (and maybe not in pair)
        rec_idx = rng.integers(n-1)
        if rec_idx >= donor_idx:
            rec_idx += 1
        # compute max transfer from donor
        max_transfer = min(seq[donor_idx], target_sum - seq[rec_idx])  # 1000 bound not needed
        if max_transfer < 1e-9:
            continue
        # step size: fraction of donor amount
        step = min(max_transfer, (target_sum/n) * 0.5)  # typical step magnitude
        delta = rng.random() * step
        if delta < 1e-12:
            continue
        # apply move
        seq_new = seq.copy()
        seq_new[donor_idx] -= delta
        seq_new[rec_idx] += delta
        # evaluate
        conv_new = conv_fft(seq_new, fft_size)
        new_max = float(conv_new.max())
        # acceptance
        elapsed = time.monotonic() - start
        T = T0 * max(0.0, 1.0 - elapsed/(deadline-start))
        if new_max < cur_max or rng.random() < math.exp(-(new_max - cur_max)/max(T,1e-12)):
            seq = seq_new

            tiny = min(seq[j], 1e-3)

                seq[i]+=tiny
                seq[j]-=tiny
    return best_seq, best_max

"""
Low‑value step sequence search.

The algorithm is a specialised simulated‑annealing / hill‑climbing loop
that focuses moves on the indices that actually contribute to the current
worst convolution value.  For a given length n we

1. start from a random L¹‑normalised vector (or from the previously known
   best construction if it is available);
2. repeatedly move a small amount of mass from a “high‑contributor”
   index (i.e. a position that participates heavily in the maximal
   auto‑convolution) to a low‑contributor index;
3. accept improving moves always and accept worsening moves with a
   temperature‑controlled probability (simulated‑annealing style);
4. after a block of iterations perform a cheap LP refinement (the same
   refinement used in the reference code) which often tightens the bound;
5. keep the best sequence seen over many random restarts and several
   lengths (200 … 400).  The whole search respects the 1000 s budget.

The function `propose_candidate` follows the required signature and
returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def max_conv_fft(seq: np.ndarray, size: int = None) -> float:
    """
    Return the maximal value of the linear convolution `seq * seq`
    using a real FFT.  The convolution is not circular – the length is
    ``2*len(seq)-1``.
    """
    n = seq.size
    if size is None:
        size = _next_pow2(2 * n - 1)
    f_hat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f_hat * f_hat, size)[: 2 * n - 1]
    # tiny negative numbers may appear due to rounding
    conv = np.maximum(conv, 0.0)
    return float(conv.max())


def max_conv_fft_with_idx(seq: np.ndarray, size: int = None):
    """Return (max_val, argmax, full_conv_array)."""
    n = seq.size
    if size is None:
        size = _next_pow2(2 * n - 1)
    f_hat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f_hat * f_hat, size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g)[k] = (f * g)[k].
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        A[k, k - idx] = f[idx]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float):
    """
    One LP refinement step (identical to the helper used in the previous
    submission).  It solves

        maximise   Σ g_j
        subject to g ≥ 0,
                   f * g ≤ M  (componentwise),

    where M = max(f * f).  The resulting g is rescaled to the required
    ℓ¹‑norm and mixed with the current f.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv_fft(f)

    n = f.size
    M = max_conv_fft(f)

    A = build_conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS (dense) is fast enough for the modest sizes we use
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    sum_g = g_val.sum()
    if sum_g < 1e-12:
        return f, M

    # Rescale to the target ℓ¹‑norm
    g_rescaled = g_val * (target_sum / sum_g)

    # Mix (the reference used t≈0.3; we keep the same)
    t = 0.3
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)
    return f_new, max_conv_fft(f_new)


# ----------------------------------------------------------------------
# Targeted simulated‑annealing local search
# ----------------------------------------------------------------------


def _contrib_at_offset(seq: np.ndarray, k: int):
    """
    For a given offset k, return the indices that actually contribute
    together with the contribution values a_i * a_{k-i}.
    """
    n = seq.size
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    contrib = seq[idx] * seq[k - idx]
    return idx, contrib


def _targeted_move(seq: np.ndarray, k_max: int, rng: np.random.Generator,
                   avg: float):
    """
    Produce a new sequence by moving a small amount of mass from a
    high‑contributor (to the current worst offset) to a low‑contributor.
    Returns the new sequence and the amount moved (or None if move not
    possible).
    """
    n = seq.size
    idx, contrib = _contrib_at_offset(seq, k_max)

    # donor – the index with the largest contribution to the bad offset
    donor = idx[int(np.argmax(contrib))]

    # candidate recipients – indices with (practically) zero contribution
    # at the same offset.  If none exist we fall back to a completely random
    # index that is not the donor.
    zero_mask = contrib < 1e-12
    if zero_mask.any():
        recipient = rng.choice(idx[zero_mask])
    else:
        recipient = rng.integers(0, n - 1)
        while recipient == donor:
            recipient = rng.integers(0, n - 1)

    # maximum amount that can be transferred while keeping both numbers in
    # the allowed range [0,1000].
    max_transfer = min(seq[donor], 1000.0 - seq[recipient])
    if max_transfer < 1e-12:
        return None, None

    # typical step magnitude: a fraction of the average entry
    step = min(max_transfer, avg * 0.5)
    delta = rng.random() * step
    if delta < 1e-12:
        return None, None

    new_seq = seq.copy()
    new_seq[donor] -= delta
    new_seq[recipient] += delta
    return new_seq, delta


def _local_optimize(f0: np.ndarray, target_sum: float,
                    deadline: float, rng: np.random.Generator,
                    max_iters: int = 300_000):
    """
    Core optimisation loop.  It returns the best sequence found and the
    associated maximal convolution value.
    """
    n = f0.size
    fft_sz = _next_pow2(2 * n - 1)

    cur = f0.copy()
    cur_max, cur_idx, _ = max_conv_fft_with_idx(cur, fft_sz)
    best_seq = cur.copy()
    best_max = cur_max

    # cooling schedule – start a little above zero (the objective is already
    # small) and cool linearly to near‑zero.
    T0 = 0.03
    start_time = time.monotonic()

    it = 0
    while it < max_iters and time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start_time
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start_time))

        # ---------- targeted move ----------
        new_seq, _ = _targeted_move(cur, cur_idx, rng, target_sum / n)
        if new_seq is None:
            # fall back to a completely random pair move (the old SA style)
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_tr = min(cur[j], 1000.0 - cur[i])
            if max_tr > 1e-12:
                step = min(max_tr, (target_sum / n) * 0.5)
                delta = rng.random() * step
                new_seq = cur.copy()
                new_seq[i] += delta
                new_seq[j] -= delta
            else:
                continue

        new_max = max_conv_fft(new_seq, fft_sz)

        # ---------- acceptance ----------
        dE = new_max - cur_max
        accept = False
        if dE < -1e-12:
            accept = True
        elif rng.random() < math.exp(-dE / max(T, 1e-12)):
            accept = True

        if accept:
            cur = new_seq
            cur_max = new_max
            # recompute the index of the worst offset
            cur_max, cur_idx, _ = max_conv_fft_with_idx(cur, fft_sz)

            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_seq = cur.copy()
                print(f"[SA] n={n} iter={it} new best max={best_max:.6f}")

        # occasional tiny jitter to shake out plateaus
        if it % 7_000 == 0:
            k = rng.integers(0, n)
            l = rng.integers(0, n - 1)
            if l >= k:
                l += 1
            tiny = min(cur[l], 1e-2)
            if tiny > 0:
                cur[k] += tiny
                cur[l] -= tiny
                cur_max = max_conv_fft(cur, fft_sz)

        # every few thousand steps run a cheap LP refinement (if cvxpy is present)
        if it % 12_000 == 0:
            try:
                import cvxpy  # noqa: F401
                cur, _ = lp_refine(cur, target_sum, deadline)
                cur_max = max_conv_fft(cur, fft_sz)
            except Exception:
                pass

    return best_seq, best_max


# ----------------------------------------------------------------------
# Helper to generate a reasonable random start
# ----------------------------------------------------------------------


def _random_start(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random non‑negative vector of length n whose ℓ¹‑norm equals
    `target_sum`.  The distribution is uniform on the simplex (Dirichlet(1,…,1)).
    """
    vec = rng.random(n)
    vec = vec / vec.sum() * target_sum
    return vec.astype(np.float64)

# Main entry point required by the competition
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a step sequence that minimises ``evaluate_sequence``.
    The routine respects the 1000 s limit (a safety margin of 2 s is kept)
    and returns the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction exists, seed from it.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim == 1 and init.size > 0:
            n0 = init.size
            target0 = math.sqrt(2.0 * n0)
            if init.sum() > 0:
                init = init * (target0 / init.sum())
                score0 = max_conv_fft(init)
                best_seq = init.copy()
                best_score = score0
                print(f"[init] using height_sequence_1 (n={n0}) – score={score0:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – experiments have shown that n∈[150,400] gives
    # the strongest bounds for the given budget.
    # ------------------------------------------------------------------
    lengths = [150, 180, 200, 230, 260, 300, 340, 380, 400]

    # ------------------------------------------------------------------
    # Outer loop – we keep trying new random restarts until time runs out.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length (biased to larger sizes as time permits)
        n = int(rng.choice(lengths))
        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise: either expand a previously good sequence or start fresh.
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n and rng.random() < 0.3:
            f0 = best_seq.copy()
        elif best_seq is not None and best_seq.size < n and rng.random() < 0.5:
            # stretch the current best to the new length
            old = best_seq
            out = np.zeros(n, dtype=np.float64)
            for i, val in enumerate(old):
                idx = int(i * n / old.size)
                out[idx] += val
            out = out * (target_sum / out.sum())
            f0 = out
        else:
            f0 = _random_start(n, target_sum, rng)

        # ------------------------------------------------------------------
        # Local optimisation (targeted SA + occasional LP)
        # ------------------------------------------------------------------
        f_opt, max_opt = _local_optimize(f0, target_sum, deadline, rng)

        # ------------------------------------------------------------------
        # Record the global best
        # ------------------------------------------------------------------
        if max_opt < best_score - 1e-12:
            best_score = max_opt
            best_seq = f_opt.copy()
            print(f"[global] n={n} new best score={best_score:.6f}")

        # early exit if we reach the target bound
        if best_score < 1.503:
            print("[stop] target bound reached")
            break

    # ------------------------------------------------------------------
    # Safety fallback – never triggered unless something goes terribly wrong.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = max_conv_fft(best_seq)
        print("[fallback] uniform sequence used, score=", best_score)

    # Return plain Python list
    return [float(x) for x in best_seq]
