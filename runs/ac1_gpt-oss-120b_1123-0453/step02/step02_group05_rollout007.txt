"""
Greedy‑guided simulated annealing for the max‑auto‑convolution problem.

The score is  2·n·max(f∗f)/(∑f)² .  Because the l¹‑norm is kept equal to
√(2·n), the score equals max(f∗f).  The algorithm works with a fixed length
`n` (default 400) and repeatedly moves a small amount of mass from an index
that contributes most to the current worst convolution entry to an index that
contributes the least.  The move is accepted according to a Metropolis rule
with an exponential cooling schedule; a tiny random‑move probability is kept
to escape plateaus.  Stagnation triggers a cheap restart from a fresh random
vector (the best ever found is never lost).  The best sequence seen during the
run is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a feasible non‑negative sequence of length *n* with
    ℓ¹‑norm √(2·n).  If a previous construction ``height_sequence_1`` exists,
    it is embedded (truncated or padded with zeros) before normalisation."""
    target = math.sqrt(2.0 * n)

    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seq = np.zeros(n, dtype=np.float64)
        m = min(len(old), n)
        seq[:m] = old[:m]
    else:
        seq = rng.random(n, dtype=np.float64)  # uniform random start

    s = seq.sum()
    if s <= 0.0:
        seq = rng.random(n, dtype=np.float64) + 1e-3
        s = seq.sum()
    seq = seq / s * target
    return seq


# ----------------------------------------------------------------------
def _evaluate(seq):
    """Thin wrapper around the supplied scorer."""
    return float(evaluate_sequence(seq))

        The best sequence found (plain Python floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – 400 gives a good trade‑off between flexibility and speed
    n = 400

    # initialise a feasible vector (ℓ¹‑norm = √(2·n))
    f = _initial_sequence(n, rng)          # shape (n,)
    conv = np.convolve(f, f)               # full convolution, length 2·n‑1
    cur_max = conv.max()
    best_f = f.copy()
    best_max = cur_max

    # timing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # temperature schedule
    T_start = 1.0
    T_end = 1e-4

    # stagnation handling
    it = 0
    last_improve_it = 0
    stagnation_limit = 150_000          # restart if no improvement for this many iters

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        # exponential cooling
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        # --------------------------------------------------------------
        # locate the worst convolution index
        k_max = int(np.argmax(conv))

        # contributions of each index to the worst entry
        idx = k_max - np.arange(n)                     # possible partner indices
        valid = (idx >= 0) & (idx < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[valid] = f[idx[valid]]

        # --------------------------------------------------------------
        # donor: the index that (a) has positive mass and (b) contributes most
        donor_mask = (f > 1e-12) & (contrib > 0.0)
        if donor_mask.any():
            i = int(np.argmax(contrib * donor_mask))
        else:
            # fallback – random index with positive mass
            pos = np.where(f > 1e-12)[0]
            if pos.size == 0:
                i = rng.integers(0, n)
            else:
                i = int(rng.choice(pos))

        # --------------------------------------------------------------
        # destination: index with the smallest contribution (avoid i)
        dest_mask = np.arange(n) != i
        contrib_dest = contrib[dest_mask]
        j_candidates = np.where(dest_mask)[0]
        min_val = contrib_dest.min()
        min_idxs = j_candidates[contrib_dest == min_val]
        if min_idxs.size > 1:
            j = int(rng.choice(min_idxs))
        else:
            j = int(min_idxs[0])

        # occasional random move to escape plateaus
        if rng.random() < 0.01:
            i = int(rng.choice(np.where(f > 1e-12)[0]))
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)

        # --------------------------------------------------------------
        # amount of mass to transfer
        max_delta = f[i]
        if max_delta <= 1e-12:
            # nothing to move from this donor – skip
            it += 1
            continue

        # shrinking step‑size as the deadline approaches
        frac = 0.20 * (1.0 - elapsed / total_time) + 0.01
        delta = rng.random() * max_delta * frac
        if delta <= 0.0:
            it += 1
            continue

        # --------------------------------------------------------------
        # apply the move (with the ability to roll back)
        conv_old = conv.copy()
        fi_old, fj_old = f[i], f[j]

        # linear part of the convolution update
        conv[i:i + n] -= 2.0 * delta * f
        conv[j:j + n] += 2.0 * delta * f

        # quadratic corrections (self‑terms)
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        s = i + j
        if s < conv.size:
            conv[s] -= 2.0 * delta * delta

        # update the sequence itself
        f[i] -= delta
        f[j] += delta

        new_max = conv.max()

        # --------------------------------------------------------------
        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob_up = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob_up:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f.copy()
                last_improve_it = it
                # progress report – the score equals the worst convolution entry
                print(
                    f"[SA] it={it:07d} best_max={best_max:.6f} T={T:.4e}"
                )
        else:
            # roll back
            conv = conv_old
            f[i] = fi_old
            f[j] = fj_old

        # --------------------------------------------------------------
        # cheap restart if we have been stagnant for a while
        if (it - last_improve_it > stagnation_limit) and (elapsed > 0.5 * total_time):
            # keep the best ever, start from a fresh random vector
            f = _initial_sequence(n, rng)
            conv = np.convolve(f, f)
            cur_max = conv.max()
            last_improve_it = it
            # (no change to best_f / best_max)

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (clipped to the allowed range)
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_f.tolist()]
    final_score = _evaluate(best_seq)
    print("[final] best evaluation =", final_score)
    return best_seq
