def _gradient(p, conv, max_indices):

"""
Gradient‑descent refinement on the simplex.

We minimise the evaluation score 
    S(p) = 2·n·max_k (p∗p)[k]      (p_i ≥ 0, Σ p_i = 1)

because the evaluator is scale‑invariant.  The objective is the
ℓ∞‑norm of the auto‑convolution of a probability vector.
The algorithm is a projected sub‑gradient method:

* start from a feasible probability vector (random or the previously
  discovered `height_sequence_1`);
* repeatedly
   – compute the convolution,
   – locate the index/indices where the maximum is attained,
   – form the sub‑gradient g_i = 2·Σ_{k∈K} p_{k‑i}   (only terms with
     0 ≤ k‑i < n contribute),
   – move a small step p ← p – η·g,
   – project back onto the probability simplex (clip negatives and
     renormalise);
* keep the best sequence seen during the whole run.
The step size η decreases slowly (η₀/(1+it·α)) so that the method is
aggressive at the beginning and stabilises later.  A few random
perturbations are injected every few thousand iterations to escape
possible plateaus.

The routine tries several lengths (200 … 800) while respecting the
global 1000 s budget and finally returns the best sequence found.
All numbers are non‑negative and clipped to the allowed interval
[0, 1000].
"""

import time
import math
import numpy as np

# Global random generator – will be initialised in `propose_candidate`.
_rng = None


def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq))


def _convolution(p: np.ndarray) -> np.ndarray:
    """Return the linear convolution of p with itself (real, dense)."""
    return np.convolve(p, p)


def _subgradient(p: np.ndarray, max_indices: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the objective
        max_k (p∗p)[k]
    at the current point `p`.  For each maximal index k we add the vector
    (2·p_{k‑i})_i (zero where the index is out of range) and finally
    average over all maximal indices.
    """
    n = p.shape[0]
    g = np.zeros_like(p)
    for k in max_indices:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start … i_end, j = k‑i is the partner index.
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals
        # contribution 2·p_j to each g_i
        g[i_vals] += 2.0 * p[j_vals]
    if max_indices.size > 0:
        g /= float(max_indices.size)
    return g


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Implementation follows the O(n log n) algorithm of
    Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    # make a copy to avoid side effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find the rho index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gd_optimize(p0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for a fixed length `n`.
    Returns the best probability vector encountered and its score.
    """
    global _rng
    p = p0.copy()
    best_p = p.copy()
    # Since Σ p = 1 the score simplifies to 2·n·max(conv).
    conv = _convolution(p)
    best_score = 2.0 * n * conv.max()

    # Hyper‑parameters
    eta0 = 0.3            # initial step size
    decay = 1e-4          # decay factor for step size
    max_no_improve = 3000 # stop if no improvement for a while
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and no_improve < max_no_improve:
        it += 1
        # compute current convolution and maximal indices
        conv = _convolution(p)
        max_val = conv.max()
        max_idx = np.flatnonzero(conv == max_val)

        # sub‑gradient at the current maximal indices
        g = _subgradient(p, max_idx)

        # step size schedule
        eta = eta0 / (1.0 + decay * it)

        # gradient step and projection onto simplex
        p_new = p - eta * g
        # numerical safety – clip tiny negatives before projection
        p_new = np.maximum(p_new, 0.0)
        p_new = _project_onto_simplex(p_new)

        # evaluate new score (exact formula, no call to `evaluate_sequence`)
        conv_new = _convolution(p_new)
        score_new = 2.0 * n * conv_new.max()

        # record improvement
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_p = p_new.copy()
            no_improve = 0
            print(f"[gd] n={n:4d} it={it:5d} score={best_score:.6f}")
        else:
            no_improve += 1

        p = p_new

        # occasional random jitter to escape plateaus
        if it % 2000 == 0:
            jitter = _rng.normal(scale=0.01, size=n)
            p = np.maximum(p + jitter, 0.0)
            p = _project_onto_simplex(p)

    # final sanity check – make sure we stay within the allowed range
    best_p = np.clip(best_p, 0.0, 1000.0)
    return best_p, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.
    The search consists of several independent gradient‑descent runs
    for different lengths (200 … 800) and keeps the best sequence found
    before the time budget expires.
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # safety margin – leave a couple of seconds for the caller's timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Retrieve the previously discovered construction if it exists.
    try:
        # `height_sequence_1` is expected to be a list of floats.
        prev = np.asarray(height_sequence_1, dtype=float)
        prev = np.maximum(prev, 0.0)
        if prev.size > 0:
            prev /= prev.sum()          # turn into a probability vector
    except Exception:
        prev = None

    overall_best_seq = None
    overall_best_score = float("inf")

    # Lengths to try – we start modest and increase while time permits.
    lengths = (200, 300, 400, 500, 600, 700, 800)

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # Initialise a feasible probability vector.
        if prev is not None and prev.size == n:
            p0 = prev.copy()
        else:
            # Random Dirichlet‑like start.
            p0 = _rng.random(n) + 1e-6
            p0 /= p0.sum()

        # Run the gradient descent optimiser.
        best_seq, best_score = _gd_optimize(p0, n, deadline)

        if best_score < overall_best_score:
            overall_best_score = best_score
            overall_best_seq = best_seq.copy()
            print(f"[global] new best (n={n}) → score={overall_best_score:.6f}")

        # Early stop if we have already gotten very close to the theoretical optimum.
        if overall_best_score < 1.45:
            break

    # If everything failed (unlikely), fall back to a uniform distribution.
    if overall_best_seq is None:
        n = 256
        overall_best_seq = np.ones(n, dtype=float) / n
        overall_best_score = 2.0 * n * (np.convolve(overall_best_seq, overall_best_seq).max())

    # The evaluator accepts any positive scaling; we keep the ℓ¹‑norm = 1.
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    # Convert to plain Python floats as required.
    return [float(x) for x in overall_best_seq]
