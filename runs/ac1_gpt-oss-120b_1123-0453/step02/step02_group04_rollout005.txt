                - else random.

            for iteration in range(max_lp_iters):
                M = np.max(np.convolve(f,f))
                A = _conv_matrix(f)
                # LP solve
                g = cp.Variable(n)
                constraints = [g >= 0, A @ g <= M]
                prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
                if g.value is None: break

                sum_g = g0.sum()

                # mixture

                t = max(0.01, t*decay)
                # track best
                sc = _score(f)

"""
Hybrid LP‑refinement + fast local search.

* For a set of problem sizes we keep a candidate vector `f` whose
  ℓ¹‑norm is fixed to √(2·n).  This normalisation makes the scoring
  function equal to the worst auto‑convolution value `max(f*f)`.

* With the current `f` we solve a linear program
        maximise Σ g_i
        s.t.     g_i ≥ 0,
                 (f * g)[k] ≤ M   (M = max(f*f))
  The LP adds as much ℓ¹‑mass as possible without increasing the
  worst convolution.  We then mix the solution back into `f` with a
  decreasing step size `t`.

* After a few LP steps we run a fast, FFT‑based hill‑climb that moves
  tiny amounts of mass between two coordinates while keeping the total
  sum fixed.  The move is accepted only if the maximal convolution
  (computed with an O(n log n) FFT) strictly improves.  The step size
  shrinks geometrically and the loop stops when the time budget is
  exhausted.

* Several lengths (200 … 1000) are tried; the best sequence over all
  lengths and restarts is returned.  If CVXPY is unavailable a pure
  random‑walk fallback is used.

The implementation respects the 1000‑second budget, prints intermediate
scores and never creates closures or lambda expressions.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Scoring helper – uses the supplied `evaluate_sequence` function.
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the objective value for a sequence (list‑or ndarray)."""
    # `evaluate_sequence` is provided by the judging harness.
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Build the dense convolution matrix A such that (A @ g)[k] = (f * g)[k].
# Used only inside the LP part.
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill row‑by‑row; this is O(n²) but fast enough for n ≤ 1000.
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        cols = k - np.arange(i_start, i_end + 1)
        A[k, cols] = f[i_start:i_end + 1]
    return A


# ----------------------------------------------------------------------
# Fast maximal autocorrelation using FFT (O(n log n)).
# ----------------------------------------------------------------------
def _max_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)[k] using a real‑FFT."""
    n = seq.shape[0]
    m = 2 * n - 1
    # smallest power of two ≥ m
    nfft = 1 << ((m - 1).bit_length())
    # Real FFT, point‑wise square, inverse transform
    fa = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:m]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simple FFT‑based hill climbing that preserves the ℓ¹‑norm.
# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  rng: np.random.Generator,
                  deadline: float,
                  best_obj: float) -> tuple[np.ndarray, float]:
    """
    Perform many tiny mass‑transfer moves that strictly decrease the
    maximal auto‑convolution.  The loop runs until the deadline is hit.
    Returns the possibly improved sequence and its objective value.
    """
    n = seq.shape[0]
    # Current objective (max convolution).  For a correctly normalised
    # sequence this *is* the score returned by `evaluate_sequence`.
    cur_obj = _max_convolution(seq)

    # Step size = a small fraction of the mean value.
    mean_val = float(np.mean(seq))
    step = max(0.01 * mean_val, 1e-6)

    it = 0
    while time.monotonic() < deadline:
        # pick two distinct indices
        i = rng.integers(0, n)
        if seq[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # amount to move (always ≤ seq[i])
        eps = rng.random() * step
        eps = min(eps, seq[i])

        cand = seq.copy()
        cand[i] -= eps
        cand[j] = min(cand[j] + eps, 1000.0)    # respect the upper bound

        new_obj = _max_convolution(cand)

        # Accept only strict improvements.
        if new_obj + 1e-12 < cur_obj:
            seq = cand
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                # report progress
                print(f"[local] n={n:4d} iter={it:6d} obj={cur_obj:.6f}")

        # Decay the step size periodically.
        it += 1
        if it % 5000 == 0:
            step *= 0.9
            step = max(step, 1e-9)

    return seq, best_obj


# ----------------------------------------------------------------------
# Very simple random walk fallback (used when CVXPY cannot be imported).
# ----------------------------------------------------------------------
def _fallback_random_search(budget_s: float, seed: int = 0) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    # start with a moderately long uniform vector
    n = 256
    target_sum = math.sqrt(2.0 * n)
    seq = np.full(n, target_sum / n, dtype=np.float64)

    best_seq = seq.copy()
    best_obj = _max_convolution(seq)

    while time.monotonic() < deadline:
        # propose a Gaussian perturbation
        cand = best_seq + rng.normal(scale=0.02, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        # renormalise to keep the ℓ¹‑norm fixed (helps the objective)
        s = cand.sum()
        if s > 0:
            cand *= target_sum / s

        cur = _max_convolution(cand)
        if cur < best_obj:
            best_obj = cur
            best_seq = cand.copy()
            print(f"[fallback] new obj = {best_obj:.6f}")

    return best_seq, best_obj


# ----------------------------------------------------------------------
# Main entry point required by the judging system.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.

    The method consists of:
      1. Optional warm‑start from a global `height_sequence_1`.
      2. For several lengths (200 … 1000) run a short LP‑based refinement.
      3. After the LP phase, apply a fast FFT‑based local search for the
         remaining time budget.
      4. Keep the globally best sequence and return it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the teardown
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_obj = float("inf")          # objective = max auto‑convolution

    # ------------------------------------------------------------------
    # Try to import CVXPY – the LP core depends on it.
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp
        HAS_CVX = True
    except Exception:               # pragma: no cover
        HAS_CVX = False

    # ------------------------------------------------------------------
    # If a global seed sequence is present, use it as an initial candidate.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.ndim == 1 and init.size > 0:
            init_obj = _max_convolution(init)
            best_seq = init.copy()
            best_obj = init_obj
            print(f"[init] using provided height_sequence_1, obj = {init_obj:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameter sweep over several lengths.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ---- initialise a candidate `f` for this length ----------------
        if best_seq is not None and best_seq.shape[0] == n:
            f = best_seq.copy()
            cur = f.sum()
            if cur > 1e-12:
                f *= (target_sum / cur)
            else:
                f = rng.random(n) + 0.1
                f *= target_sum / f.sum()
        elif best_seq is not None and best_seq.shape[0] < n:
            f = np.zeros(n, dtype=np.float64)
            f[:best_seq.shape[0]] = best_seq
            cur = f.sum()
            if cur > 1e-12:
                f *= (target_sum / cur)
            else:
                f = rng.random(n) + 0.1
                f *= target_sum / f.sum()
        else:
            # completely fresh start
            f = rng.random(n) + 0.1
            f *= target_sum / f.sum()

        # ----------------------------------------------------------------
        # LP‑based refinement (only if CVXPY is available)
        # ----------------------------------------------------------------
        if HAS_CVX:
            # number of LP iterations depends on n (more for smaller n)
            max_lp_iters = 80 if n <= 600 else 40
            t = 0.3                # convex‑combination weight
            decay = 0.95

            for it in range(max_lp_iters):
                if time.monotonic() >= deadline:
                    break

                # worst auto‑convolution value of the current candidate
                M = _max_convolution(f)

                # Build convolution matrix (f * g) ≤ M
                A = _conv_matrix(f)

                # Linear program: maximise Σ g_i  s.t. g ≥ 0, A g ≤ M
                g = cp.Variable(n)
                constraints = [g >= 0, A @ g <= M]
                prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

                # ECOS works well for dense problems; fall back to SCS.
                try:
                    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                               verbose=False)
                except Exception:  # pragma: no cover
                    prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000,
                               verbose=False)

                if g.value is None:
                    # Solver failed – abandon this size.
                    print(f"[LP] n={n:4d} it={it:2d} – solver failed")
                    break

                g0 = np.maximum(g.value, 0.0)
                sum_g = g0.sum()
                if sum_g < 1e-12:
                    break

                # Rescale to the target ℓ¹‑norm.
                g_rescaled = g0 * (target_sum / sum_g)

                # Convex combination with decreasing step size.
                f = np.clip((1.0 - t) * f + t * g_rescaled, 0.0, 1000.0)
                t = max(0.01, t * decay)

                # Evaluate the new candidate.
                cur_obj = _max_convolution(f)
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_seq = f.copy()
                    print(f"[LP] n={n:4d} it={it:2d} t={t:.3f} obj={cur_obj:.6f}")

                # Early stop if we have already passed the target.
                if best_obj < 1.503:
                    break

        # ----------------------------------------------------------------
        # After the LP phase, polish with a fast local search.
        # ----------------------------------------------------------------
        if best_seq is not None:
            remaining = deadline - time.monotonic()
            if remaining > 2.0:   # need a tiny safety buffer
                # Run the local search on the *current* best sequence of
                # this length (or the global best – they are the same shape).
                seq = best_seq.copy()
                seq, best_obj = _local_search(seq, rng,
                                              time.monotonic() + remaining,
                                              best_obj)
                best_seq = seq.copy()

        # optional early exit when target is reached
        if best_obj < 1.503:
            break

    # ------------------------------------------------------------------
    # If everything failed (e.g. CVXPY missing) run the simple fallback.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_seq, fallback_obj = _fallback_random_search(
            max(0.0, float(budget_s) - (time.monotonic() - start_time)),
            seed=seed
        )
        best_seq = fallback_seq
        best_obj = fallback_obj

    # ------------------------------------------------------------------
    # Final sanity: make sure the output is a plain Python list of floats.
    # ------------------------------------------------------------------
    # Clip one last time to obey the scorer's bounds.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Print final result for logging.
    print(f"[final] best objective = {best_obj:.6f}  length = {best_seq.shape[0]}")

    return [float(x) for x in best_seq]
