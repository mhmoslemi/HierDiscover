Implementation details:

function projection_simplex(v, S): returns projection onto {x >=0, sum = S}.

Standard algorithm: sort v descending, compute cumulative sums, find rho = max {j: v[j] - (cumsum[j] - S) / (j+1) > 0} etc.

Works with any S.

Pseudo:

def simplex_projection(v, S):
    n = len(v)
    u = np.sort(v)[::-1]   # descending
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v, S):
    """Project v onto the simplex {x >= 0, sum(x)=S}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if len(rho) == 0:
        # all negative, project onto zero vector with sum zero (should be zero)
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑value step sequence.

Algorithm overview
-------------------
1️⃣  **LP‑driven refinement** – identical in spirit to the reference
    method: keep the ℓ¹‑norm fixed to √(2·n) so that the score equals the
    maximal entry of the auto‑convolution.  Starting from a random vector,
    solve the linear program
        maximise   Σ gᵢ
        subject to (f * g)ₖ ≤ M   for all k,
                   gᵢ ≥ 0,
    where M = max(f * f).  Rescale g to the same ℓ¹‑norm and mix it into f.
    The LP is solved many times (up to a few hundred iterations) for several
    lengths and random restarts.

2️⃣  **Projected sub‑gradient descent** – after the LP stage the current
    sequence is further improved by a simple sub‑gradient method on the
    non‑smooth objective `M = max(conv(f,f))`.  In each iteration we
    – compute the set of shifts attaining the current maximum,
    – build the sub‑gradient analytically,
    – take a diminishing step and project back onto the simplex
      `{x≥0, Σx = √(2·n)}`.  The best vector seen during this phase is kept.

3️⃣  **Stochastic polishing** – a short hill‑climbing pass that adds
    Gaussian noise, re‑projects onto the simplex and accepts strictly
    improving candidates.  This cleans up any remaining irregularities.

All stages respect the security checks of ``evaluate_sequence`` (non‑negativity,
clipping to [0,1000]) and stop as soon as the global time budget expires.
The function finally returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# try importing cvxpy once; the LP stage works only if it is present.
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the (2·n‑1) × n matrix A such that (A @ g)[k] = (f * g)[k].
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    return A


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the scaled simplex
        {x | x_i ≥ 0, Σ x_i = s}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the largest index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be projected to zero
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(seq: np.ndarray) -> float:
    """Wrapper around the provided ``evaluate_sequence`` (expects a list)."""
    return float(evaluate_sequence(seq.tolist()))


def _subgradient_optimize(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iter: int = 2000,
    step0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for ``M = max(conv(a,a))``.
    Returns the best vector seen and its maximal convolution value.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_M = np.max(np.convolve(a, a))

    for it in range(1, max_iter + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        M = conv.max()

        if M < best_M:
            best_M = M
            best_a = a.copy()

        # indices achieving the current maximum (within a tiny tolerance)
        K = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max_k conv_k at a
        g = np.zeros_like(a)
        for k in K:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                if i == j:
                    g[i] += 2.0 * a[i]          # derivative of a_i * a_i
                else:
                    g[i] += a[j]
                    g[j] += a[i]

        # diminishing step size
        step = step0 / math.sqrt(it)

        a_new = a - step * g
        a_new = _project_simplex(a_new, target_sum)
        a = a_new

    return best_a, best_M


def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.4,
    patience: int = 1000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing.  At each iteration a Gaussian perturbation
    is added, the result is projected onto the simplex (preserving the ℓ¹‑norm)
    and accepted if it improves the maximal auto‑convolution.
    """
    rng = np.random.default_rng()
    target_sum = start_seq.sum()
    best = start_seq.copy()
    best_M = np.max(np.convolve(best, best))
    step = step_initial
    it = 0

    while time.monotonic() < deadline:
        it += 1
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = _project_simplex(cand, target_sum)
        M = np.max(np.convolve(cand, cand))
        if M < best_M:
            best = cand
            best_M = M
            # reset step size on improvement
            step = step_initial
        else:
            # decay step size
            step *= 0.9995
            if step < 1e-7:
                step = step_initial
        if it >= patience:
            # safety guard – we have already done a lot of work
            break

    return best, best_M


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The search consists of three phases:
        1. LP‑based refinement (if cvxpy is available),
        2. projected sub‑gradient descent,
        3. a short stochastic polishing pass.
    It respects the total time budget and returns a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    # Reserve a small margin for the caller to retrieve the result.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.3)

    best_seq = None
    best_M = float("inf")

    # Lengths to examine – larger n gives more freedom but costs more time.
    lengths = [400, 500, 600, 800]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        restarts = 4 if n <= 600 else 2

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ----- initialise a random vector with the required ℓ¹‑norm -----
            f = rng.random(n) + 0.01
            f = f / f.sum() * target_sum

            # Optional quick smoothing – helps the LP stage converge faster
            f = np.convolve(f, np.ones(3) / 3.0, mode="same")
            f = f / f.sum() * target_sum

            # ------------------- LP refinement -------------------
            if _HAS_CVX:
                max_lp_iters = 300
                for it in range(max_lp_iters):
                    if time.monotonic() >= deadline:
                        break

                    conv_ff = np.convolve(f, f)
                    M = conv_ff.max()

                    # keep the best vector seen so far
                    if M < best_M:
                        best_M = M
                        best_seq = f.copy()
                        print(f"[LP] n={n} rs={rs} it={it} M={M:.6f}")

                    # early exit from the LP stage if we are already very good
                    if best_M < 1.503:
                        return [float(x) for x in best_seq]

                    A = _conv_matrix(f)

                    g = cp.Variable(n, nonneg=True)
                    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])

                    try:
                        prob.solve(solver=cp.ECOS, warm_start=True,
                                   max_iters=5000, abstol=1e-8, reltol=1e-6)
                    except Exception:  # pragma: no cover
                        prob.solve(solver=cp.SCS, warm_start=True,
                                   max_iters=15000, verbose=False)

                    if g.value is None:
                        break

                    g_raw = np.maximum(g.value, 0.0)
                    sum_g = g_raw.sum()
                    if sum_g < 1e-12:
                        break

                    # rescale to keep the ℓ¹‑norm fixed
                    g_rescaled = g_raw * (target_sum / sum_g)

                    # convex combination – a fairly aggressive mixing weight
                    t = 0.7
                    f = (1.0 - t) * f + t * g_rescaled
                    f = np.clip(f, 0.0, 1000.0)
                    # numerical drift – project back onto the simplex
                    f = _project_simplex(f, target_sum)

            # ---------------- Sub‑gradient refinement ----------------
            if time.monotonic() < deadline:
                sub_max_iter = 2000
                f_opt, M_opt = _subgradient_optimize(
                    f, target_sum, deadline, max_iter=sub_max_iter, step0=1.0
                )
                if M_opt < best_M:
                    best_M = M_opt
                    best_seq = f_opt.copy()
                    print(f"[SGD] n={n} rs={rs} M={M_opt:.6f}")

    # --------------------- final polishing ---------------------
    if best_seq is None:
        # deterministic fallback – a uniform vector of moderate length
        n = 400
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    if time.monotonic() < deadline:
        refined_seq, refined_M = _local_search(best_seq, deadline, step_initial=0.3)
        if refined_M < best_M:
            best_seq = refined_seq
            best_M = refined_M
            print(f"[POLISH] improved M={best_M:.6f}")

    # Final score (for debugging / logging)
    final_score = _score(best_seq)
    print(f"Best max‑convolution ≈ {best_M:.6f}, evaluate_sequence = {final_score:.6f}")
