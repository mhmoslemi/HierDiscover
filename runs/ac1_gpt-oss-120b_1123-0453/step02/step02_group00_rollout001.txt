def proj_simplex(v, s):
    # based on algorithm from Duchi et al.
    n = len(v)
    if s < 0:
        raise ValueError
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def gradient_descent(a0, deadline, S):
    a = a0.copy()
    best_seq = a.copy()
    best_score = _score(a)
    alpha = 5.0
    step = 0.5
    while time.monotonic() < deadline and alpha <= 80:
        # number of inner iterations
        for i in range(50):
            if time.monotonic() >= deadline:
                break
            # compute conv
            conv = np.convolve(a, a)
            # smooth max weights
            maxc = np.max(conv)
            # compute softmax
            # use alpha but re-scale conv for numerical stability: subtract maxc.
            exp_vals = np.exp(alpha * (conv - maxc))
            sumexp = exp_vals.sum()
            p = exp_vals / sumexp
            # gradient
            grad = 2.0 * np.convolve(p, a[::-1], mode='full')[len(a)-1 : len(a)-1+len(a)]
            # gradient descent step with projection
            a_new = a - step * grad
            # Project onto simplex
            a_new = proj_simplex(a_new, S)
            # evaluate
            score = _score(a_new)
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                # optionally increase step if progress?
                step = min(step * 1.05, 1.0)  # accelerate
            else:
                # if no improvement, reduce step
                step *= 0.9
            a = a_new
        # after inner loop increase alpha
        alpha = min(alpha * 1.5, 80.0)
        step = max(step * 0.9, 0.001)
    return best_seq, best_score

    best_local = a.copy()
    best_score = _score(a)
    # time remaining
    step = 0.5
    alpha = 5.0
    while time.monotonic() < deadline and alpha <= 80.0:
        # for each alpha, run a fixed number of iterations, but stop early if time limited
        for _ in range(200):
            if time.monotonic() >= deadline:
                break
            # compute convolution
            conv = np.convolve(a, a)
            # compute softmax weights
            maxc = np.max(conv)
            # subtract maxc for numerical stability
            exp_vals = np.exp(alpha * (conv - maxc))
            sumexp = np.sum(exp_vals)
            if sumexp == 0:

            else:
                p = exp_vals / sumexp
            # gradient
            grad = 2.0 * np.convolve(p, a[::-1], mode='full')[len(a)-1 : len(a)-1+len(a)]

"""
Propose a low‑value step sequence.

The algorithm uses three stages:

1️⃣  **Randomised initialisations** for several lengths.
2️⃣  **Smooth‑max projected gradient descent**.
    – A smooth approximation of the max‐convolution (soft‑max) is
      minimised while keeping the ℓ¹‑norm fixed (∑a = √(2 n)) and a ≥ 0.
    – The gradient of the smooth objective is obtained analytically,
      projected onto the simplex (non‑negative + fixed sum) after each
      step.  The smoothing parameter α is increased (annealed) so that
      the optimisation approaches the true max‑convolution.
3️⃣  **Final stochastic polishing** (Gaussian hill‑climbing) to
    escape tiny local irregularities.

All heavy work stops as soon as the supplied time budget expires,
ensuring a sequence is always returned before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: project onto the simplex {x ≥ 0, sum(x)=s}
# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, Σx = s}.
    Implementation of the algorithm by Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]               # sorted in descending order
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    # Find the largest ρ such that u_ρ > (cssv_ρ)/ρ
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.max(np.where(cond)[0])
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Scoring – reproduces evaluate_sequence without the type checks
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """
    Compute the evaluation score of a sequence a (numpy array).
    The function mirrors evaluate_sequence (clipping to [0,1000] and the
    2 n·max(conv)/(sum²) formula) but runs faster because no Python‑level
    validation is performed.
    """
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    max_b = np.max(np.convolve(a, a))
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))

# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing used as a final polishing step
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbation hill‑climbing.  Keeps the best sequence seen so
    far.  The step size decays exponentially; the routine stops when the
    deadline is reached.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
        # decay / occasional reset
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_sc

# ----------------------------------------------------------------------
# Gradient descent on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max (soft‑max) of the auto‑convolution
    a * a for a given smoothing parameter α.
    """
    # auto‑convolution
    conv = np.convolve(a, a)                # length 2n‑1
    # soft‑max weights (stable via subtraction of max)
    m = np.max(conv)
    exp_vals = np.exp(alpha * (conv - m))
    sumexp = np.sum(exp_vals)
    if sumexp == 0.0:                       # ultra‑rare, fallback to uniform
        p = np.full_like(conv, 1.0 / conv.size)
    else:
        p = exp_vals / sumexp               # soft‑max probabilities, sum to 1

    # gradient = 2 * (p ⋆ a_rev) restricted to n entries
    a_rev = a[::-1]
    full = np.convolve(p, a_rev, mode='full')
    n = a.shape[0]
    grad = 2.0 * full[n - 1 : n - 1 + n]    # central slice of length n
    return grad

def _gradient_descent(a0: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max objective.
    The smoothing parameter α is annealed from low to high values.
    """
    a = a0.copy()
    best = a.copy()
    best_sc = _score(best)

    # initial step size and smoothing schedule
    step = 0.5
    alpha = 5.0
    max_alpha = 80.0

    while time.monotonic() < deadline and alpha <= max_alpha:
        # perform a batch of inner iterations for the current α
        for _ in range(200):
            if time.monotonic() >= deadline:
                break

            grad = _smooth_max_gradient(a, alpha)
            cand = a - step * grad
            cand = proj_simplex(cand, target_sum)

            sc = _score(cand)
            if sc < best_sc:
                best, best_sc = cand, sc
                # modestly increase step after a successful move
                step = min(step * 1.05, 1.0)
            else:
                # no improvement → shrink step
                step = max(step * 0.90, 1e-5)

            a = cand

        # increase the smoothness (approach true max)
        alpha = min(alpha * 1.5, max_alpha)
        # slowly decay step as we get closer to the true objective
        step = max(step * 0.9, 1e-5)

    return best, best_sc

        target_sum = math.sqrt(2.0 * n)

        # Build a small pool of seeds for this length.
        seeds: list[np.ndarray] = []

        # a) uniform vector (simple baseline)
        seeds.append(np.full(n, target_sum / n, dtype=float))

        # b) a few random vectors
        for _ in range(2):
            x = rng.random(n) + 0.01               # avoid exact zeroes
            x = x / x.sum() * target_sum
            seeds.append(x)

        # c) resized version of the globally supplied construction, if any
        if init_global is not None:
            if init_global.shape[0] == n:
                seeds.append(init_global.copy())
            else:
                if init_global.shape[0] > n:
                    tmp = init_global[:n]
                else:
                    extra = np.full(n - init_global.shape[0], 1e-6, dtype=float)
                    tmp = np.concatenate([init_global, extra])
                tmp = tmp / tmp.sum() * target_sum
                seeds.append(tmp)

        # d) resized version of the best sequence discovered so far (if any)
        if best_seq is not None:
            if best_seq.shape[0] == n:
                seeds.append(best_seq.copy())
            else:
                if best_seq.shape[0] > n:
                    tmp = best_seq[:n]
                else:
                    extra = np.full(n - best_seq.shape[0], 1e-6, dtype=float)
                    tmp = np.concatenate([best_seq, extra])
                tmp = tmp / tmp.sum() * target_sum
                seeds.append(tmp)

        # ------------------------------------------------------------------
        # 2️⃣  Optimise each seed with projected gradient descent
        # ------------------------------------------------------------------
        for start_vec in seeds:
            if time.monotonic() >= deadline:
                break
            seq, sc = _gradient_descent(start_vec, target_sum, deadline)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[GD] n={n} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  If nothing was found (unlikely) fall back to a safe uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 4️⃣  Final stochastic polishing (cheap hill‑climbing)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        refined, refined_sc = _local_search(best_seq, deadline, step_initial=0.2)
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[local] final polished score={best_score:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best_seq]
