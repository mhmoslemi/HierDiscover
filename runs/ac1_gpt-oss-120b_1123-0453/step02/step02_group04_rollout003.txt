def simplex_projection(v, target):
    # ensure v nonnegative after projection falling onto simplex.
    # For generic v (may have negative), we can follow standard algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    # find rho = max i where condition holds
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, target):
    """Project vector v onto the simplex {x >= 0, sum x = target}."""
    # Reference implementation from Duchi et al.
    # If target is zero, return zeros.
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero?
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‐function with a low evaluation value.

The algorithm consists of two stages.

1)  For each candidate length n (200 … 1000) initialise a feasible
    non‑negative sequence a with ℓ¹‑norm √(2·n).  The initial point is
    taken either from the global variable ``height_sequence_1`` (if it has
    the right length) or from a random Dirichlet draw followed by a
    projection onto the simplex.

2)  A fast first‑order method is applied to minimise
        max_k (a * a)[k]
    (which equals the objective after the normalisation).  At every
    iteration the index k where the convolution attains its maximum is
    identified; the gradient of that peak w.r.t. a is
        ∂b_k/∂a_j = 2·a_{k‑j}  (zero if the index is out of bounds).
    A step a ← a – η·grad is taken and the result is projected back onto
    the simplex (non‑negative entries with the required ℓ¹‑norm).  The
    stepsize η is adapted on‑the‑fly: it is increased when a step yields
    improvement and decreased otherwise.

    After a block of gradient steps a short stochastic hill‑climb
    (random mass transfers between two positions) is performed to escape
    possible plateaus.

The whole search respects the supplied time budget (default 1000 s) and
keeps the best sequence found over all lengths.  If the time expires
before any candidate is produced a simple random walk fallback is used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: wrapper around the provided scorer.
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the objective value for ``seq`` (list or np.ndarray)."""
    # ``evaluate_sequence`` is assumed to be globally available.
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = target}.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of given L1‑norm."""
    if target <= 0:
        return np.zeros_like(v)

    # Work on a copy to avoid side effects.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    # Find the largest rho such that u[rho] - cssv[rho]/(rho+1) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using FFT (fast for many repetitions).
# ----------------------------------------------------------------------
def _full_convolve(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a*a (full length 2·n‑1) by means of FFT.
    """
    n = a.shape[0]
    size = 2 * n                     # zero‑pad to avoid circular wrap‑around
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv.real


def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)[k]; identical to the objective after normalisation."""
    return float(np.max(_full_convolve(a)))

    # Parameters controlling the inner loops
    max_restarts = 3                     # independent random starts
    max_gd_iters = 5000                  # gradient‑descent iterations per restart
    max_perturb = 500                    # random perturbations after each restart

    best_seq = None
    best_score = float("inf")

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise a feasible sequence for this restart.
        # ------------------------------------------------------------------
        if restart == 0 and init_seq is not None:
            a = init_seq.copy()
            cur = np.sum(a)
            if cur > 1e-12:
                a = a * (target_sum / cur)
            else:
                a = rng.exponential(scale=1.0, size=n)
        else:
            a = rng.exponential(scale=1.0, size=n)

        a = _simplex_projection(a, target_sum)

        # Keep track of the current best for this length.
        cur_score = _max_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] n={n:4d} restart={restart} score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # Gradient‑descent phase (sub‑gradient on the maximal entry).
        # ------------------------------------------------------------------
        # Initial stepsize – roughly the magnitude of an entry.
        step = target_sum * 0.5 / max(1.0, n)

        for it in range(max_gd_iters):
            if time.monotonic() >= deadline - 0.05:
                break

            # Compute convolution and locate a maximal index.
            conv = _full_convolve(a)
            max_val = float(conv.max())
            # Early exit if we cannot improve further.
            if max_val <= best_score:
                # still record the best we have seen.
                if max_val < best_score:
                    best_score = max_val
                    best_seq = a.copy()
                    print(f"[best] n={n:4d} restart={restart} it={it:4d} score={max_val:.6f}")
                # we keep iterating because a different direction may reduce the peak further.

            # Choose one index where the maximum is attained.
            k = int(np.flatnonzero(conv == max_val)[0])

            # Gradient of b_k = (a*a)[k] w.r.t. a:
            #   ∂b_k/∂a_j = 2·a_{k‑j}  (if the shifted index lies inside the vector)
            grad = np.zeros_like(a)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            js = np.arange(i_start, i_end + 1, dtype=int)
            grad[js] = 2.0 * a[k - js]

            # Propose a new point and project back onto the feasible set.
            cand = _simplex_projection(a - step * grad, target_sum)
            cand_score = _max_convolution(cand)

            # Accept if we improved the maximal convolution.
            if cand_score < max_val - 1e-12:
                a = cand
                step = min(step * 1.02, target_sum)    # gently increase step
                max_val = cand_score
                if max_val < best_score:
                    best_score = max_val
                    best_seq = a.copy()
                    print(f"[gd] n={n:4d} restart={restart} it={it:4d} step={step:.5f} score={max_val:.6f}")
            else:
                step *= 0.80   # shrink step if no progress

        # ------------------------------------------------------------------
        # Short stochastic hill‑climb (mass transfer between two positions).
        # ------------------------------------------------------------------
        for _ in range(max_perturb):
            if time.monotonic() >= deadline - 0.05:
                break
            cand = best_seq.copy()
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # Transfer a small random amount from i to j (preserving the sum).
            delta = rng.random() * 0.05 * cand[i]   # up to 5 % of the mass at i
            delta = min(delta, cand[i])             # keep non‑negative
            cand[i] -= delta
            cand[j] += delta
            cand_score = _max_convolution(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[perturb] n={n:4d} restart={restart} score={cand_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Fallback random walk (used only if everything else fails).
# ----------------------------------------------------------------------
def _fallback_random_search(budget_s: float, seed: int = 0) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    n = 200
    a = np.full(n, 1.0, dtype=np.float64)
    target = math.sqrt(2.0 * n)
    a = _simplex_projection(a, target)
    best_seq = a.copy()
    best_score = _max_convolution(a)

    while time.monotonic() < deadline:
        cand = a + rng.normal(scale=0.02, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _simplex_projection(cand, target)
        sc = _max_convolution(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            a = cand
            print(f"[fallback] score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine explores several problem sizes,
    performs a simple sub‑gradient descent on the maximal autocorrelation
    peak and finishes with a short random hill‑climb.  It respects the
    supplied time budget.
    """
    start = time.monotonic()
    # keep a tiny safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Try to start from the global sequence supplied by the harness.
    # ------------------------------------------------------------------
    global_init = None
    try:
        tmp = np.array(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and np.all(np.isfinite(tmp)):
            global_init = tmp
            print(f"[global] loaded height_sequence_1, length={len(tmp)}")
    except Exception:
        global_init = None

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # A collection of candidate lengths (larger n gives more freedom).
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Produce an appropriate initial vector for this length.
        # ------------------------------------------------------------------
        if global_init is not None and global_init.shape[0] == n:
            init = global_init.copy()
        elif best_seq is not None and best_seq.shape[0] <= n:
            # embed the current best (zero‑pad) to the new length
            init = np.zeros(n, dtype=np.float64)
            init[: best_seq.shape[0]] = best_seq
        elif best_seq is not None and best_seq.shape[0] > n:
            init = best_seq[:n].copy()
        else:
            init = None

        # ------------------------------------------------------------------
        # Run the optimiser for this length.
        # ------------------------------------------------------------------
        seq, sc = _optimise_length(n, target_sum, init, deadline, rng)

        # ------------------------------------------------------------------
        # Keep the overall best.
        # ------------------------------------------------------------------
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global] new overall best length={n} score={sc:.6f}")

        # Early termination if the target has already been surpassed.
        if best_score < 1.503:
            print("[global] target reached, terminating early")
            break

    # ------------------------------------------------------------------
    # If for any reason we have not produced a candidate, use the simple fallback.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_seq, fallback_score = _fallback_random_search(
            max(0.0, float(budget_s) - (time.monotonic() - start)), seed=seed
        )
        best_seq = fallback_seq
        best_score = fallback_score
        print(f"[fallback] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list of floats as required by the harness.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
