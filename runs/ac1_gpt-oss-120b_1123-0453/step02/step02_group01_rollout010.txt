def projection_simplex(v, s):
    """Projects v onto the simplex {x >=0, sum = s}."""
    # Based on algorithm by Wang and Carreira-Perpinan
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Gradient‑descent + simplex projection optimizer for the step‑function bound.

The evaluation score (with the prescribed ℓ¹‑norm √(2·n)) is exactly the
maximum entry of the auto‑convolution.  The algorithm therefore tries to
flatten the auto‑convolution by minimizing a smooth Lᵖ‑approximation of
the max‑norm.  At each iteration we
  * compute the full convolution,
  * form weights = conv^{p‑1} (p≈4) which give a sub‑gradient of the
    Lᵖ‑norm,
  * obtain the gradient via a correlation,
  * take a step opposite to the gradient,
  * project back onto the simplex {a_i≥0, Σa_i = √(2·n)}.

Occasional random jitter prevents stagnation.  The best sequence seen
among several random restarts (and an optional warm‑start from
`height_sequence_1` if present) is returned.  The implementation respects
the 1000 s time budget and uses only NumPy (no CVXPY needed).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _evaluate(seq: np.ndarray) -> float:
    """Fast evaluation – equivalent to evaluate_sequence for a correctly
    normalised sequence (sum = √(2·n)).  The external scorer is used only
    for the final verification."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    M = float(conv.max())
    # denominator = (sum(seq))**2; we enforce the exact sum = √(2·n)
    # but we still compute the full formula for safety.
    s = float(np.sum(seq))
    return float(2.0 * n * M / (s * s))


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Implementation of the algorithm by Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, return zeros (should not happen for s>0)
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_gradient(a: np.ndarray, p: int = 4) -> np.ndarray:
    """
    Gradient of the Lᵖ‑norm of the auto‑convolution (p>1) w.r.t. a.
    Returns a vector of length len(a).  Uses correlation to avoid O(n²)
    loops.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n‑1
    # weights = conv^{p‑1}; add a tiny epsilon to avoid zero‑power issues
    eps = 1e-16
    weights = np.power(conv + eps, p - 1)
    # correlation of weights with a gives Σ_t weights[t] a[t‑i]
    corr = np.correlate(weights, a, mode='full')
    grad = 2.0 * corr[n - 1:n - 1 + n]          # slice i = 0 … n‑1
    return grad


def _optimize_one_restart(n: int,
                          init_seq: np.ndarray,
                          rng: np.random.Generator,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient‑descent optimisation for a single restart.
    Returns the best sequence found (as a NumPy array) and its evaluation
    score (the max auto‑convolution value).
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise and enforce the simplex constraint
    a = project_simplex(init_seq.astype(float), target_sum)

    best_seq = a.copy()
    best_score = _evaluate(a)

    # hyper‑parameters
    p = 4                         # Lᵖ‑norm exponent
    lr = 0.5                      # base learning‑rate
    no_improve = 0
    max_iters = 2000
    jitter_every = 120            # steps between random perturbations

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # ----- evaluate current point -----
        conv = np.convolve(a, a)
        M = float(conv.max())
        if M < best_score - 1e-10:
            best_score = M
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- compute smooth gradient -----
        grad = _smooth_gradient(a, p=p)

        # normalise step size w.r.t. gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break
        step = lr * target_sum / grad_norm

        # stochastic jitter to escape shallow plateaus
        if it % jitter_every == 0 and it > 0:
            a = a + 0.01 * target_sum * rng.standard_normal(n)

        # gradient step and projection onto the simplex
        a = project_simplex(a - step * grad, target_sum)

        # adapt learning rate if we have not improved for a while
        if no_improve > 400:
            lr *= 0.7
            no_improve = 0
        if lr < 1e-5:
            break

    return best_seq, best_score


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The routine performs several random restarts for a handful of
    lengths (up to 1000 entries).  Each restart runs a projected
    gradient descent on a smooth Lᵖ‑approximation of the max‑norm of the
    auto‑convolution.  The best sequence seen before the time budget
    expires is returned.
    """
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # helper to obtain an initial vector of a given length
    def _initial_vector(length: int) -> np.ndarray:
        # If a global warm‑start exists, use it when the length matches.
        try:
            # height_sequence_1 may be defined in the outer scope
            global height_sequence_1
            seq0 = np.asarray(height_sequence_1, dtype=float)
            if seq0.shape[0] == length:
                return seq0.copy()
        except Exception:
            pass
        # otherwise return a small random Dirichlet vector
        x = rng.random(length) + 0.01
        return x.astype(float)

    # ------------------------------------------------------------------
    # explore a palette of lengths; larger n give a tighter theoretical lower bound
    lengths = [400, 500, 600, 800, 1000]

    # Main loop – keep launching restarts while time permits
    while time.time() < deadline:
        # allocate remaining time per restart (rough estimate)
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        n = int(rng.choice(lengths))
        target = math.sqrt(2.0 * n)

        # perform a few restarts for the chosen length
        restarts = 3
        for _ in range(restarts):
            if time.time() >= deadline:
                break

            init = _initial_vector(n)
            # normalise the random start to the required ℓ¹‑norm
            init = init / np.sum(init) * target

            seq, score = _optimize_one_restart(n, init, rng, deadline)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[update] n={n} score={score:.6f}")

            # early exit if we already beat the target threshold
            if best_score < 1.503:
                break

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # fallback: if nothing was discovered (unlikely), return a uniform vector
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = _evaluate(best_seq)

    # final sanity – ensure the list format expected by the grader
    return [float(x) for x in best_seq]
