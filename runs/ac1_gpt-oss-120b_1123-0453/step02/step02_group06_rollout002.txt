def balance_convolution(f, target_sum, deadline, rng, max_iter=2000):
    n = f.size
    fft_size = 1 << ((2 * n - 1).bit_length())
    best_f = f.copy()
    best_max = max_conv_fft(f, fft_size)
    # iter
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        # compute full convolution
        conv = full_conv(f, fft_size)
        # find max
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]
        if max_val < best_max:
            best_max = max_val
            best_f = f.copy()
            # print progress
            print(f"[BAL] n={n} iter={it} max={best_max:.6f}")
        # compute partner contributions
        idx = np.arange(n)
        partner_idx = k_max - idx
        valid = (partner_idx >= 0) & (partner_idx < n)
        p = np.zeros_like(f)
        p[valid] = f[partner_idx[valid]]
        # contributions
        contrib = f * p
        # source = index with largest contribution
        i_src = int(np.argmax(contrib))
        # target: smallest p (prefer where capacity exists)
        # mask for capacity
        cap_mask = f < 1000.0
        if not np.any(cap_mask):
            break
        # Among capacity indices find minimal p
        candidate_idxs = np.where(cap_mask)[0]
        # If only one candidate same as src, pick second best or skip
        j_target = int(candidate_idxs[np.argmin(p[candidate_idxs])])
        if j_target == i_src:
            # try other candidate
            # remove src from candidates
            cand2 = candidate_idxs[candidate_idxs != i_src]
            if cand2.size == 0:
                break
            j_target = int(cand2[np.argmin(p[cand2])])
        # compute transfer amount
        max_transfer = min(f[i_src], 1000.0 - f[j_target])
        if max_transfer <= 1e-12:
            # try alternative source/target: maybe second largest contribution
            # find second source
            contrib[i_src] = -1
            i_src2 = int(np.argmax(contrib))
            max_transfer = min(f[i_src2], 1000.0 - f[j_target])
            if max_transfer <= 1e-12:
                continue
            i_src = i_src2
        # step size based on average entry
        avg = target_sum / n
        step = min(max_transfer, avg * 0.5)
        if step <= 1e-12:
            continue
        delta = rng.random() * step
        if delta <= 1e-12:

        f[j_target] += delta
        # ensure no negative due to numerical issues
        f[i_src] = max(f[i_src], 0.0)
        f[j_target] = min(f[j_target], 1000.0)

    final_max = max_conv_fft(f, fft_size)
    if final_max < best_max:
        best_f = f.copy()
        best_max = final_max
    return best_f, best_max

"""
Hybrid search for a low‑value non‑negative step sequence.

The algorithm works in three stages:

1. **Initialisation** – generate a random or structured starting vector
   (uniform, sine, triangular, exponential) and normalise it so that
   Σ a_i = √(2 n).  The normalisation makes the evaluation
   `evaluate_sequence` equal to the maximal auto‑convolution value.

2. **Targeted balancing** – repeatedly locate the index k where the
   auto‑convolution (f⋆f) is maximal and move a small amount of mass from
   the entry that contributes most to this peak to an entry that contributes
   the least.  This greedy “balance‑conv” step reduces the peak quickly.
   Each move is checked with a fast FFT‑based convolution; the loop runs
   until time runs out or a preset iteration limit is reached.

3. **Optional LP refinement** – if CVXPY is available we solve the linear
   programme 
        maximise Σ g_i   subject to   g_i ≥ 0 ,   (f⋆g)_k ≤ M  ∀k,
   where M is the current maximal convolution value.
   The solution is mixed with the current vector (t≈0.3) which often lowers
   the peak further.

The outer loop tries several lengths (200 … 1500) and many random restarts,
always keeping the best sequence found.  The whole search respects the
budget (≈ 1000 s).  The returned list of floats can be fed directly to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast)
# ----------------------------------------------------------------------


def _fft_size_for_len(n: int) -> int:
    """Next power of two ≥ 2·n‑1, used for zero‑padding in FFT."""
    return 1 << ((2 * n - 1).bit_length())


def max_conv_fft(seq: np.ndarray, size: int = None) -> float:
    """
    Return max_i (seq⋆seq)[i] computed with FFT.
    The sequence may be any length; the optional ``size`` argument can be
    pre‑computed to avoid repeated work.
    """
    n = seq.size
    if size is None:
        size = _fft_size_for_len(n)
    f_hat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f_hat * f_hat, size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)          # eliminate tiny negatives
    return float(conv.max())


def full_conv_fft(seq: np.ndarray, size: int) -> np.ndarray:
    """
    Return the full discrete convolution (seq⋆seq) as a NumPy array.
    Length is 2·n‑1.  ``size`` must be the FFT size (≥2·n‑1, power of two).
    """
    n = seq.size
    f_hat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f_hat * f_hat, size)[: 2 * n - 1]
    return np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional, requires cvxpy)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Toeplitz‑like matrix A (shape (2n‑1, n)) such that (A @ g)[k] = (f⋆g)[k].
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # indices i where 0≤i≤n‑1 and 0≤k‑i≤n‑1
        A[k, i_start : i_end + 1] = f[k - np.arange(i_start, i_end + 1)]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple:
    """
    One LP refinement step.
    Maximises ℓ¹‑mass of a new vector g under the current maximal convolution
    bound M = max(f⋆f).  The optimum g is then mixed with f (t=0.3) and the
    result is returned together with its exact maximal convolution.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv_fft(f)

    n = f.size
    M = max_conv_fft(f)

    A = build_conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    sum_g = g_val.sum()
    if sum_g < 1e-12:
        return f, M

    # Rescale to maintain the required ℓ¹‑norm
    g_rescaled = g_val * (target_sum / sum_g)

    # Mixing step – a modest convex combination improves the bound
    t = 0.3
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    return f_new, max_conv_fft(f_new)

def init_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random or structured non‑negative vector of length ``n``.
    The vector is normalised so that its ℓ¹‑norm equals √(2·n).
    """
    choice = rng.random()
    if choice < 0.25:
        # plain random
        seq = rng.random(n)
    elif choice < 0.5:
        # sine shape (smooth, zero at the ends)
        i = np.arange(n)
        seq = np.sin(np.pi * (i + 1) / (n + 1))
    elif choice < 0.75:
        # triangular – first rise then fall
        mid = n // 2
        left = np.linspace(1.0, 2.0, mid, endpoint=False)
        right = np.linspace(2.0, 1.0, n - mid)
        seq = np.concatenate([left, right])
    else:
        # exponentially decaying shape
        decay = rng.random() * 0.04 + 0.01
        seq = np.exp(-decay * np.arange(n))

    # add a tiny random perturbation to break symmetry
    seq += rng.normal(scale=0.01 * seq.mean(), size=n)
    seq = np.clip(seq, 0.0, 1000.0)

    target = math.sqrt(2.0 * n)
    tot = seq.sum()
    if tot == 0.0:
        seq[:] = target / n
    else:
        seq *= (target / tot)
    return seq


def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Stretch ``seq`` to length ``new_len`` by nearest‑index mapping and
    renormalise to the ℓ¹‑norm √(2·new_len).
    """
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    total = out.sum()
    if total > 0.0:
        target = math.sqrt(2.0 * new_len)
        out *= (target / total)
    return out


# ----------------------------------------------------------------------
# Targeted balancing of the maximal auto‑convolution
# ----------------------------------------------------------------------


def balance_convolution(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 2000,
) -> tuple:
    """
    Greedy “balance‑conv’’ local search.

    Repeatedly:
      * locate k where (f⋆f)[k] is maximal,
      * move a small amount of mass from the index contributing most to
        this peak to an index contributing least,
    until the time limit is hit or ``max_iter`` iterations are done.
    Returns the best vector seen and its maximal convolution value.
    """
    f = start.copy()
    n = f.size
    fft_size = _fft_size_for_len(n)

    # initial statistics
    best_f = f.copy()
    best_max = max_conv_fft(f, fft_size)

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        # full convolution – needed to locate the active index
        conv_full = full_conv_fft(f, fft_size)
        k_max = int(np.argmax(conv_full))
        cur_max = float(conv_full[k_max])

        # keep the best seen so far
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_f = f.copy()
            print(f"[BAL] n={n} it={it} max={best_max:.6f}")

        # partner values that determine each entry's contribution to conv[k_max]
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)
        p = np.zeros_like(f)
        p[valid] = f[partner[valid]]

        # source = entry with largest contribution f[i]*p[i]
        contrib = f * p
        i_src = int(np.argmax(contrib))

        # target = entry with smallest partner value (least influence on the peak)
        # respect the 1000‑cap for the target
        cap_mask = f < 1000.0
        if not np.any(cap_mask):
            break
        candidates = np.where(cap_mask)[0]
        j_tgt = int(candidates[np.argmin(p[candidates])])
        if j_tgt == i_src:
            # fall back to the second‑best source if necessary
            contrib[i_src] = -1.0
            i_src = int(np.argmax(contrib))
            # recompute target (ignore the source if it’s the only candidate)
            if i_src in candidates:
                candidates = candidates[candidates != i_src]
                if candidates.size == 0:
                    break
            j_tgt = int(candidates[np.argmin(p[candidates])])

        # amount we can safely transfer
        max_transfer = min(f[i_src], 1000.0 - f[j_tgt])
        if max_transfer <= 1e-12:
            continue

        # step size – a fraction of the average entry; ensures stability
        avg = target_sum / n
        step = min(max_transfer, avg * 0.5)
        delta = rng.random() * step
        if delta <= 1e-12:
            continue

        # perform the transfer
        f[i_src] -= delta
        f[j_tgt] += delta

        # occasional tiny random jitter to avoid getting stuck
        if (it + 1) % 5000 == 0:
            a, b = rng.integers(0, n, size=2)
            if a != b:
                tiny = min(f[a], 1e-2)
                if tiny > 0:
                    f[b] += tiny
                    f[a] -= tiny

    # final evaluation
    final_max = max_conv_fft(f, fft_size)
    if final_max < best_max:
        best_max = final_max
        best_f = f.copy()
    return best_f, best_max


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.

    The function respects a ~1000 s wall‑time budget, explores several
    lengths, performs many random restarts, applies the targeted balancing
    routine and (when CVXPY is present) a short LP refinement.
    The best sequence found is returned as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")
    best_n = None

    # lengths to try – larger n give a richer search space
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500]

    # optional CVXPY detection – we import once
    try:
        import cvxpy  # noqa: F401
        HAVE_CVXPY = True
    except Exception:  # pragma: no cover
        HAVE_CVXPY = False

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Determine a reasonable number of restarts for this length
        restarts = 0
        while time.monotonic() < deadline and restarts < 30:
            restarts += 1

            # ------------------------------------------------------------------
            # Choose an initial vector: either an expansion of the current best
            # or a fresh structured random start.
            # ------------------------------------------------------------------
            if best_seq is not None and best_seq.size < n and rng.random() < 0.6:
                f0 = expand_sequence(best_seq, n)
            else:
                f0 = init_sequence(n, rng)

            # ------------------------------------------------------------------
            # Main optimisation phase – balancing
            # ------------------------------------------------------------------
            f_bal, max_bal = balance_convolution(
                f0, target_sum, deadline, rng, max_iter=3000
            )

            # ------------------------------------------------------------------
            # Optional LP refinement (a few cheap steps)
            # ------------------------------------------------------------------
            if HAVE_CVXPY:
                lp_steps = 0
                while lp_steps < 3 and time.monotonic() < deadline:
                    f_bal, max_bal = lp_refine(f_bal, target_sum, deadline)
                    lp_steps += 1
                # recompute a precise max after LP
                max_bal = max_conv_fft(f_bal)

            # ------------------------------------------------------------------
            # Record the globally best sequence
            # ------------------------------------------------------------------
            if max_bal < best_score - 1e-12:
                best_score = max_bal
                best_seq = f_bal.copy()
                best_n = n
                print(
                    f"[GLOBAL] n={n} restart={restarts} new best score={best_score:.6f}"
                )
                if best_score < 1.503:
                    print("[STOP] target bound reached")
                    break

            # early termination for this length if target already met
            if best_score < 1.503:
                break

        # early stop if we already beat the target
        if best_score < 1.503:
            break

    # ----------------------------------------------------------------------
    # Fallback – should never happen, but guarantees a return value.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = max_conv_fft(best_seq)
        print("[FALLBACK] uniform sequence used, score=", best_score)

    # final sanity: convert to plain Python list of floats
    result = [float(x) for x in best_seq]
    print(f"[RESULT] best n={best_n} score={best_score:.6f} length={len(result)}")
    return result
