    n = len(seq)
    size = 1 << ((2*n-1).bit_length())
    best_seq = seq.copy()
    best_max = max_conv_fft(seq,size)
    cur_seq = seq.copy()
    cur_max = best_max
    # compute conv full for later info
    while time.time < deadline and iter < max_iters:
        # compute conv
        conv = np.fft.irfft(np.fft.rfft(cur_seq,size)*np.fft.rfft(cur_seq,size),size)[:2*n-1]
        M = conv.max()
        # compute indices close to max (tolerance)
        K = np.where(conv >= M - 1e-12)[0]
        # compute contributions
        cont = np.zeros(n)
        for k in K:

            i_end = min(k, n-1)

"""
Improved search for a low‑value step sequence.

The algorithm mixes three ideas:

* **Guided descent** – each iteration moves a small amount of mass
  from the index that contributes the most to the current worst
  convolution value to the index that contributes the least.
  The contribution of an index `i` to the maximal convolution entries
  is computed exactly (sum of `a[k‑i]` over all maximal `k`).  This gives
  a deterministic descent direction that is far more effective than a
  pure random‑move SA.

* **Occasional simulated‑annealing moves** – when the guided move cannot
  improve the bound we fall back to a temperature‑controlled random
  transfer (the classic SA step) so the search can escape plateaus.

* **LP refinement (optional)** – after the descent we run a few cheap
  convex‑programming refinements exactly as in the reference code; they
  often provide a small extra boost.

The search runs for several lengths (`n ∈ {200,…,1000}`), tries a few
different initial shapes (uniform, U‑shaped, two blocks) and keeps the
best sequence found before the time budget (≈ 1000 s) expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers ---------------------------------------------------
# ----------------------------------------------------------------------


def max_conv_fft(seq: np.ndarray, size: int = None) -> float:
    """
    Fast (FFT) computation of max_i (seq * seq)[i].
    The input is a 1‑D non‑negative array.
    """
    n = seq.size
    if size is None:
        size = 1 << ((2 * n - 1).bit_length())
    f_hat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f_hat * f_hat, size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)          # eliminate tiny negatives
    return float(conv.max())


def full_conv_fft(seq: np.ndarray, size: int = None) -> np.ndarray:
    """Full convolution (real) via FFT – used only when we need the array."""
    n = seq.size
    if size is None:
        size = 1 << ((2 * n - 1).bit_length())
    f_hat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f_hat * f_hat, size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)
    return conv


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional) -----------------------------
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Toeplitz‑like matrix A (shape (2n‑1, n)) such that (A @ g)[k] = (f*g)[k]."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        A[k, k - np.arange(i_start, i_end + 1)] = f[i_start : i_end + 1]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple:
    """One LP refinement step (maximise ℓ¹‑mass under the current max‑conv bound)."""
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv_fft(f)

    n = f.size
    M = max_conv_fft(f)

    A = build_conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    sum_g = g_val.sum()
    if sum_g < 1e-12:
        return f, M

    g_rescaled = g_val * (target_sum / sum_g)

    t = 0.3
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    return f_new, max_conv_fft(f_new)


# ----------------------------------------------------------------------
# Guided descent (deterministic + occasional SA) -----------------------
# ----------------------------------------------------------------------


def guided_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 150_000,
) -> tuple:
    """
    Deterministic descent that always moves mass from a high‑contributing
    index to a low‑contributing one.  If no improvement is possible the
    routine falls back to a tiny simulated‑annealing move.
    Returns (best_sequence, best_max_convolution).
    """
    n = a0.size
    size = 1 << ((2 * n - 1).bit_length())

    a = a0.copy()
    cur_max = max_conv_fft(a, size)
    best_a = a.copy()
    best_max = cur_max

    start = time.monotonic()
    it = 0
    T0 = 0.02                    # initial temperature for occasional SA steps

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # ----- compute full convolution and maximal entries -----
        conv = full_conv_fft(a, size)
        M = conv.max()
        # indices whose value is (almost) maximal
        mask = conv >= M - 1e-12
        max_idx = np.where(mask)[0]

        # ----- contribution of each position to the maximal entries -----
        contrib = np.zeros(n, dtype=np.float64)
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # for i in [i_start, i_end] we have a term a[i] * a[k-i]
            contrib[i_start : i_end + 1] += a[k - np.arange(i_start, i_end + 1)]

        i_src = int(contrib.argmax())
        i_dst = int(contrib.argmin())

        # if source and destination coincide we resort to a random pair
        if i_src == i_dst:
            i_src, i_dst = rng.integers(0, n, size=2)
            while i_src == i_dst:
                i_dst = rng.integers(0, n)

        max_transfer = min(a[i_src], 1000.0 - a[i_dst])
        if max_transfer > 1e-12:
            # typical step size – a fraction of the average entry
            avg = target_sum / n
            step = min(max_transfer, avg * 0.5)
            delta = rng.random() * step
            if delta > 1e-12:
                a_new = a.copy()
                a_new[i_src] -= delta
                a_new[i_dst] += delta
                new_max = max_conv_fft(a_new, size)

                if new_max < cur_max - 1e-12:               # greedy improvement
                    a = a_new
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a_new.copy()
                        print(f"[GD] n={n} iter={it} new best max={best_max:.6f}")

                    continue                     # go to next iteration

                # ----- fallback to a tiny SA move -----
                elapsed = time.monotonic() - start
                T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))
                if rng.random() < math.exp(-(new_max - cur_max) / max(T, 1e-12)):
                    a = a_new
                    cur_max = new_max
                    continue

        # ----- occasional random jitter to avoid dead‑locks -----
        if it % 5000 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                tiny = min(a[i], 1e-3)
                if tiny > 0:
                    a[i] -= tiny
                    a[j] += tiny
                    cur_max = max_conv_fft(a, size)

    return best_a, best_max


# ----------------------------------------------------------------------
# Initialisation helpers -----------------------------------------------
# ----------------------------------------------------------------------


def init_uniform(n: int, target_sum: float) -> np.ndarray:
    """Flat distribution."""
    a = np.full(n, target_sum / n, dtype=np.float64)
    return a


def init_u_shaped(n: int, target_sum: float) -> np.ndarray:
    """U‑shaped (more weight at the ends, less in the centre)."""
    idx = np.arange(n)
    dist = np.abs(idx - (n - 1) / 2.0)
    w = dist + 1.0
    a = w / w.sum() * target_sum
    return a


def init_two_blocks(n: int, target_sum: float) -> np.ndarray:
    """Two equal blocks placed at the two ends of the interval."""
    L = max(1, n // 6)                # block length – a small fraction of n
    a = np.zeros(n, dtype=np.float64)
    val = target_sum / (2 * L)
    a[:L] = val
    a[-L:] = val
    return a


def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Stretch an existing sequence to a larger length (nearest‑index mapping)."""
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    total = out.sum()
    if total > 0:
        target = math.sqrt(2.0 * new_len)
        out = out * (target / total)
    return out

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None

    # ------------------------------------------------------------------
