def compute_conv_fft(a):
    n = a.shape[0]
    # length for full convolution: 2n-1
    size = 1 << (2*n-1).bit_length()
    # zero-pad to size
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(a, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[:2*n-1]

"""
Improved search for a low‑value step function.

The algorithm works in three phases inside the total time budget
(≈1000 s):

1️⃣  **Initialisation** – we start from a random Dirichlet vector,
    normalize it so that the ℓ¹‑norm equals 1 (the score is
    scale‑invariant).  If the global variable ``height_sequence_1``
    exists we use it as a warm start.

2️⃣  **Projected gradient descent on the maximal auto‑convolution**.
    With the current sequence ``a`` we compute the full convolution
    via a fast FFT, locate the index ``k*`` where the maximum occurs
    and form the sub‑gradient of ``max(a*a)`` w.r.t. the entries.
    A simple step ``a ← a – η·∇`` is taken and the vector is projected
    back onto the probability simplex (non‑negative entries with sum 1).
    The learning rate decays as ``η₀/√t``.  This phase rapidly drives
    the maximal convolution below the uniform value (which gives a
    score of 2).

3️⃣  **Fine‑tuning with simulated annealing** – a classic SA that
    performs two kinds of moves:
        * add Gaussian noise to a single coordinate,
        * swap two coordinates.
    The move is accepted according to a Metropolis rule with a
    temperature that cools from 1.0 to 1e‑4.  The best sequence seen
    during SA (and during the gradient phase) is kept.

Both phases repeatedly use a fast FFT based convolution, so the
routine can afford thousands of iterations within the budget.  The
final sequence (a plain list of non‑negative floats) is returned.
All numbers are automatically clipped to the interval [0, 1000] by
the provided scorer.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Return the official score for a candidate sequence."""
    # ``evaluate_sequence`` is provided by the harness.
    return float(evaluate_sequence(seq.tolist()))


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution of ``a`` with itself using FFT.
    Returns the real array of length ``2·len(a)‑1``.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm from:
    Duchi, Shalev‑Shwartz, Singer, Chandra – “Efficient Projections
    onto the l1‑Ball for Learning in High Dimensions”.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1                     # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximum convolution value with respect to ``a``.
    ``k`` is the index where the maximum occurs (``np.argmax``).
    The gradient of ``max_k conv_k`` equals the gradient of that
    particular entry because only that entry is active.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1)
    j = k - i

    # Most pairs contribute twice in the convolution (i≠j)
    grad[i] = 2.0 * a[j]

    # The diagonal i==j should be counted only once
    if k % 2 == 0:
        i0 = k // 2
        if i_start <= i0 <= i_end:
            grad[i0] = a[i0]                     # overwrite the double count

    return grad


def _gradient_descent(
    a: np.ndarray,
    iters: int,
    eta0: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a limited number of projected gradient steps that try to
    minimise the maximal auto‑convolution.  The function returns the
    best sequence seen together with its official score.
    """
    n = a.shape[0]
    for it in range(1, iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = _conv_fft(a)
        M = np.max(conv)
        # because Σa = 1, the official score equals 2·n·M
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[GD] n={n:4d} it={it:5d} score={cur_score:.6f}")

        k_star = int(np.argmax(conv))
        grad = _grad_max(a, k_star)

        # step size – decays like 1/√it
        eta = eta0 / math.sqrt(it)
        a = a - eta * grad
        a = _project_simplex(a)          # keep feasibility

    return best_seq, best_score


def _simulated_annealing(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that mixes two move types:
        * Gaussian perturbation of a single entry,
        * swap of two entries.
    The temperature cools from 1.0 down to 1e‑4 linearly in log‑space.
    """
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4

    n = len(seq)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        temp = temp_start * (temp_end / temp_start) ** frac

        # 50 % chance of a perturbation, 50 % chance of a swap
        if rng.random() < 0.5:
            # perturb a single coordinate
            idx = rng.integers(n)
            delta = rng.normal(scale=0.5) * temp * 5.0
            cand = cur_seq.copy()
            cand[idx] = float(np.clip(cand[idx] + delta, 0.0, 1000.0))
        else:
            # swap two coordinates
            i, j = rng.choice(n, size=2, replace=False)
            cand = cur_seq.copy()
            cand[i], cand[j] = cand[j], cand[i]

        cand_score = _score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

        # very occasional global jitter to escape deep plateaus
        if rng.random() < 0.001:
            jitter = np.maximum(cur_seq + rng.normal(scale=0.1, size=n), 0.0)
            cur_seq = np.clip(jitter, 0.0, 1000.0)
            cur_score = _score(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The routine respects the time budget, runs a projected gradient
    descent phase followed by simulated annealing, and returns the
    best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # ------------------------------------------------------------------
    # 0️⃣  Try to use the previously discovered sequence if it exists.
    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_score = float("inf")
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            tmp = np.asarray(height_sequence_1, dtype=np.float64)
            tmp = np.clip(tmp, 0.0, 1000.0)
            sc = _score(tmp)
            best_seq = tmp.copy()
            best_score = sc
            print(f"[init] using height_sequence_1 – score {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Main optimisation loop – several independent restarts.
    # ------------------------------------------------------------------
    restart = 0
    while time.monotonic() < deadline:
        # choose problem size (larger n gives more freedom)
        n = rng.integers(350, 801)               # 350 ≤ n < 801
        # random Dirichlet start (exponential draws) and normalise to sum = 1
        a = rng.exponential(scale=1.0, size=n)
        a = _project_simplex(a)

        # If we already have a decent sequence, occasionally start from it
        if best_seq is not None and rng.random() < 0.2:
            # interpolate a little toward the current best (helps reuse good mass)
            a = 0.7 * a + 0.3 * _project_simplex(best_seq[:n] if best_seq.shape[0] >= n else np.pad(best_seq, (0, n - best_seq.shape[0])))

        # ------------------------------------------------------------------
        # 2️⃣  Projected gradient descent – a few thousand cheap steps.
        # ------------------------------------------------------------------
        gd_deadline = min(deadline, time.monotonic() + 8.0)   # at most 8 s per restart
        a, cur_score = _gradient_descent(
            a,
            iters=3000,
            eta0=0.6,
            deadline=gd_deadline,
            best_seq=best_seq if best_seq is not None else a,
            best_score=best_score,
        )
        # ------------------------------------------------------------------
        # 3️⃣  Simulated‑annealing fine‑tuner (uses remaining time of this restart)
        # ------------------------------------------------------------------
        if time.monotonic() < deadline:
            seq_sa, score_sa = _simulated_annealing(a, cur_score, deadline, rng)
            if score_sa < best_score:
                best_score = score_sa
                best_seq = np.array(seq_sa, dtype=np.float64)
                print(f"[restart {restart}] SA improved to {best_score:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(256, 1.0 / 256)

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
