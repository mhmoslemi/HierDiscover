"""
Hybrid LP + smooth gradient descent search for a low‑value step sequence.

The algorithm works in three stages:

1. **LP stage** – Repeatedly solves the linear program described in the
   statement (maximise the ℓ¹‑norm of a vector `g` under the bound
   `f∗g ≤ M`) for a few random restarts and a moderate length (n≈500).
   This quickly brings the objective down to the ~1.51 range that
   previous submissions achieved.

2. **Refinement stage** – Starting from the best vector found by the LP
   stage, we minimise the quantity `max(conv(a,a))` on the simplex
   using a smooth approximation of the max (soft‑max with an annealing
   temperature).  The gradient of the smoothed objective can be written
   compactly with a single correlation (`np.correlate`).  After each
   gradient step the vector is projected back onto the ℓ¹‑simplex with
   the required sum `√(2n)`.

3. **Escape stage** – Occasionally a small random perturbation is added
   (Gaussian noise) to avoid being trapped in a shallow local minimum.
   The best sequence seen at any time is kept.

The whole routine respects the 1000 s budget, falls back to a cheap
random‑walk if cvxpy is not available, and finally returns a plain list
of floats compatible with `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {x≥0, sum(x)=s}."""
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, return zero vector with sum s (which is impossible)
        # fallback to uniform distribution
        return np.full_like(v, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return (2n‑1)×n matrix A such that A @ g == np.convolve(f, g)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A

# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, target_sum: float):
    """Solve the LP described in the statement and return the scaled vector."""
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:
            return None

    if g.value is None:
        return None

    g_val = np.maximum(g.value, 0.0)
    s = np.sum(g_val)
    if s < 1e-12:
        return None
    # Rescale to required ℓ¹‑norm
    return g_val * (target_sum / s)

# ----------------------------------------------------------------------
def _evaluate(seq: np.ndarray) -> float:
    """Wrapper around the supplied evaluate_sequence that accepts ndarray."""
    # evaluate_sequence is assumed to be available in the global namespace
    return float(evaluate_sequence(seq.tolist()))

# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return a soft‑max weighting of the convolution entries."""
    # shift by the max to avoid overflow
    shift = conv.max()
    exp_vals = np.exp(beta * (conv - shift))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        # fallback to uniform distribution
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp

# ----------------------------------------------------------------------
def _gradient_refine(a: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Perform smooth‑max gradient descent on `a` until `deadline`.
    Returns the best vector seen during the optimisation.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _evaluate(best)

    # hyper‑parameters (they will be annealed inside the loop)
    lr = 0.2          # initial learning rate
    lr_decay = 0.9997
    beta = 0.5        # initial temperature for the soft‑max
    beta_inc = (12.0 - beta) / 5000.0   # linear annealing schedule
    noise_every = 250   # add Gaussian noise every this many iterations
    noise_scale = 0.02 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution and smooth‑max weights
        conv = np.convolve(a, a)
        weights = _softmax_weights(conv, beta)

        # gradient of the smoothed max (we move opposite to it)
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        a_new = a - lr * grad
        a = _proj_simplex(a_new, target_sum)

        # anneal parameters
        lr *= lr_decay
        beta += beta_inc

        # occasional noise injection to escape plateaus
        if it % noise_every == 0:
            a = _proj_simplex(a + rng.normal(scale=noise_scale, size=n), target_sum)

        # evaluate every few iterations (cheaper than each step)
        if it % 20 == 0:
            cur_score = _evaluate(a)
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score
                print(f"[gd] iter={it} score={cur_score:.6f}")

        # stop early if we are already very close to the theoretical lower bound
        # theoretical lower bound for any n is 2n/(2n‑1) ≈ 1 + 1/(2n‑1)
        # we stop when the score is within 0.3 % of that bound.
        lower = 2.0 * n / (2.0 * n - 1.0)
        if best_score <= lower * 1.003:
            break

    return best

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search a non‑negative step sequence that minimises ``evaluate_sequence``.

    The routine first runs a compact LP‑based search (the same scheme that
    achieved a score of ≈1.5127), then refines the best vector with a
    smooth‑max gradient descent.  If the LP solver is unavailable the
    method falls back to a pure gradient‑descent / random‑walk search.
    The final vector is returned as a plain ``list[float]``.
    """
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) LP based exploration (similar to the reference implementation)
    # ------------------------------------------------------------------
    # Try a handful of lengths, but give priority to n=500 where we have
    # a known good construction.
    lengths = [500, 400, 600, 300, 700]
    restarts_per_n = 3
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            # initialise a random feasible vector (or reuse a known construction)
            init_vec = None
            # if a global variable `height_sequence_1` exists with matching length, use it
            glob = globals().get('height_sequence_1')
            if isinstance(glob, list) and len(glob) == n:
                init_vec = np.array(glob, dtype=np.float64)
                # make sure it satisfies the ℓ¹‑norm exactly
                init_vec = _proj_simplex(init_vec, target_sum)
            else:
                # random point on the simplex (Dirichlet like)
                x = rng.random(n) + 0.1   # avoid exact zeros
                init_vec = x / x.sum() * target_sum

            f = init_vec.copy()
            cur_score = _evaluate(f)
            if cur_score < best_score:
                best_seq = f.copy()
                best_score = cur_score
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            # iterative LP improvement – limited number of steps
            max_lp_iters = 120
            t_mix = 0.2

            for it in range(max_lp_iters):
                if time.time() >= deadline:
                    break

                conv = np.convolve(f, f)
                M = float(conv.max())

                g_scaled = _lp_step(f, M, target_sum)
                if g_scaled is None:
                    # LP failed – give up this restart
                    break

                conv_new = np.convolve(g_scaled, g_scaled)
                M_new = float(conv_new.max())

                if M_new < M - 1e-12:
                    f = g_scaled
                    cur_score = 2.0 * n * M_new / (target_sum ** 2)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                        print(f"[LP] n={n} restart={restart} it={it} score={cur_score:.6f}")
                else:
                    # try a convex combination
                    f_cand = (1.0 - t_mix) * f + t_mix * g_scaled
                    conv_cand = np.convolve(f_cand, f_cand)
                    M_cand = float(conv_cand.max())
                    if M_cand < M:
                        f = f_cand
                        cur_score = 2.0 * n * M_cand / (target_sum ** 2)
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = f.copy()
                            print(f"[mix] n={n} restart={restart} it={it} score={cur_score:.6f}")
                        t_mix = min(0.5, t_mix * 1.1)
                    else:
                        t_mix = max(0.05, t_mix * 0.5)
                        if t_mix < 0.05:
                            break

                # early termination test (theoretical bound)
                lower_bound = 2.0 * n / (2.0 * n - 1.0)
                if M_new <= lower_bound * 1.01:
                    best_score = 2.0 * n * M_new / (target_sum ** 2)
                    best_seq = f.copy()

"""
Hybrid LP + smooth‑max gradient descent search for a low‑value step sequence.

The algorithm works in three stages:

1. **LP stage** – Repeatedly solves the linear program described in the
   statement (maximise the ℓ¹‑norm of a vector `g` under the bound
   `f∗g ≤ M`) for a few random restarts and a moderate length (n≈500).
   This quickly brings the objective down to the ~1.51 region.

2. **Refinement stage** – Starting from the best vector found by the LP
   stage, we minimise `max(conv(a,a))` on the simplex with a smooth
   approximation of the max (soft‑max with an annealing temperature).
   The gradient of the smoothed objective can be written compactly as
   a single correlation (`np.correlate`).  After each gradient step the
   vector is projected back onto the ℓ¹‑simplex with the required sum
   `√(2n)`.

3. **Escape stage** – Occasionally a small random perturbation (Gaussian
   noise) is added to avoid being trapped in a shallow local minimum.
   The best sequence seen at any time is retained.

If the LP solver (`cvxpy`) is unavailable the code falls back to a pure
gradient‑descent / random‑walk search.  The routine respects the 1000 s
budget, prints progress, and finally returns a plain list of floats
compatible with `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {x≥0, sum(x)=s} (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution (should never happen in practice)
        return np.full_like(v, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return a (2n‑1)×n matrix A such that A @ g == np.convolve(f, g)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, target_sum: float):
    """Solve the LP from the statement and return the rescaled feasible vector."""
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:
            return None

    if g.value is None:
        return None

    g_val = np.maximum(g.value, 0.0)
    s = np.sum(g_val)
    if s < 1e-12:
        return None
    return g_val * (target_sum / s)


# ----------------------------------------------------------------------
def _evaluate(seq: np.ndarray) -> float:
    """Convenient wrapper around the supplied `evaluate_sequence`."""
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return a soft‑max weighting of the convolution entries."""
    shift = conv.max()
    exp_vals = np.exp(beta * (conv - shift))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp


# ----------------------------------------------------------------------
def _gradient_refine(a: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Smooth‑max gradient descent on the simplex until `deadline`.
    The best vector encountered is returned.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _evaluate(best)

    # hyper‑parameters (will anneal)
    lr = 0.2
    lr_decay = 0.9997
    beta = 0.5
    beta_inc = (12.0 - beta) / 5000.0
    noise_every = 250            # add Gaussian noise every this many steps
    noise_scale = 0.02 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = np.convolve(a, a)
        weights = _softmax_weights(conv, beta)

        # grad of the smoothed max (we move opposite to it)
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        a_new = a - lr * grad
        a = _proj_simplex(a_new, target_sum)

        # annealing
        lr *= lr_decay
        beta += beta_inc

        # occasional random kick
        if it % noise_every == 0:
            a = _proj_simplex(a + rng.normal(scale=noise_scale, size=n), target_sum)

        # cheap evaluation every few iterations
        if it % 20 == 0:
            cur_score = _evaluate(a)
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score
                print(f"[gd] iter={it} score={cur_score:.6f}")

        # early stop when we are extremely close to the known lower bound
        lower = 2.0 * n / (2.0 * n - 1.0)          # theoretical lower bound
        if best_score <= lower * 1.003:
            break

    return best

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score.

    The routine:
      1. runs a compact LP‑based search (the scheme that previously gave
         ~1.5127),
      2. refines the best vector with smooth‑max gradient descent,
      3. falls back to a cheap random walk if nothing else succeeded.
    The final vector is returned as a plain Python list of floats.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # small safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – LP based exploration (similar to the reference implementation)
    # ------------------------------------------------------------------
    lengths = [500, 400, 600, 300, 700]          # try a few different sizes
    restarts_per_n = 3

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            # initialise: reuse a known construction if available
            init_vec = None
            global_ref = globals().get('height_sequence_1')
            if isinstance(global_ref, list) and len(global_ref) == n:
                init_vec = np.array(global_ref, dtype=np.float64)
                init_vec = _proj_simplex(init_vec, target_sum)
            else:
                x = rng.random(n) + 0.1                # avoid exact zeros
                init_vec = x / x.sum() * target_sum

            f = init_vec.copy()
            cur_score = _evaluate(f)
            if cur_score < best_score:
                best_seq = f.copy()
                best_score = cur_score
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            max_lp_iters = 120
            t_mix = 0.2

            for it in range(max_lp_iters):
                if time.time() >= deadline:
                    break

                conv = np.convolve(f, f)
                M = float(conv.max())

                g_scaled = _lp_step(f, M, target_sum)
                if g_scaled is None:
                    break   # LP failed – give up this restart

                conv_new = np.convolve(g_scaled, g_scaled)
                M_new = float(conv_new.max())

                if M_new < M - 1e-12:
                    f = g_scaled
                    cur_score = 2.0 * n * M_new / (target_sum ** 2)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()

                else:
                    # try a convex combination when the pure LP step does not improve
                    f_cand = (1.0 - t_mix) * f + t_mix * g_scaled
                    conv_cand = np.convolve(f_cand, f_cand)
                    M_cand = float(conv_cand.max())
                    if M_cand < M:
                        f = f_cand
                        cur_score = 2.0 * n * M_cand / (target_sum ** 2)
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = f.copy()
                            print(f"[mix] n={n} restart={restart} it={it} score={cur_score:.6f}")
                        t_mix = min(0.5, t_mix * 1.1)
                    else:
                        t_mix = max(0.05, t_mix * 0.5)
                        if t_mix < 0.05:
                            break

                # early exit if we have essentially reached the theoretical lower bound
                lower_bound = 2.0 * n / (2.0 * n - 1.0)
                if M_new <= lower_bound * 1.01:
                    best_score = 2.0 * n * M_new / (target_sum ** 2)
                    best_seq = f.copy()
                    print(f"[near_opt] n={n} score={best_score:.6f}")
                    break

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max gradient refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, dtype=np.float64)
        n = a.shape[0]
        target_sum = math.sqrt(2.0 * n)

            best_score = refined_score
            print(f"[refine] final score={refined_score:.6f}")

    # Stage 3 – fallback cheap random walk (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    return [float(x) for x in best_seq]
