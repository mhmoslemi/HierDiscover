def proj_onto_simplex(v, z=1.0):
    # v is np.ndarray
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v, sum_val):
    # shift v to nonnegative after projection.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - sum_val
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? returns zeros
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def gradient_descent(f, target_sum, deadline, max_iters=10000, eta0=0.5, decay=0.9995, min_step=1e-6):
    best_f = f.copy()
    best_M = np.max(np.convolve(f,f))
    eta = eta0

        # compute conv and max

        # compute gradient at max index

"""
Search for a low‑value step sequence.

Algorithm
---------

1️⃣ **Optional LP‑based refinement** (exactly the idea from the reference
   implementation).  For a given length `n` we keep the ℓ¹‑norm fixed to
   √(2·n) so that the evaluation reduces to the worst entry of the
   auto‑convolution.  By repeatedly solving the linear program

        maximise   Σ g_i
        subject to (f * g)_k ≤ M   for every shift k,
                   g_i ≥ 0 ,

   and mixing the solution `g` back into `f` we obtain a steadily
   decreasing bound `M = max(f*f)`.  This phase is fast for moderate
   lengths (≤ 600) and gives a good starting point.

2️⃣ **Projected gradient descent on the simplex**.  The objective
   `M(f) = max_k (f*f)_k` is nonsmooth.  We use the sub‑gradient at the
   currently worst index `k*`:
        ∂M/∂f_i = 2·f_{k*‑i}   (if the index exists, else 0).

   After a step we project back onto the ℓ¹‑simplex of radius
   `√(2·n)` (non‑negative entries, fixed sum).  An exponential step‑size
   decay and a simple back‑tracking rule keep the iteration stable.
   This phase works with thousands of variables thanks to an FFT‑based
   convolution for the max‑value.

3️⃣ **Stochastic hill‑climbing** (Gaussian perturbations) cleans up the
   best vector found in the remaining time.

All phases respect the security checks performed inside
`evaluate_sequence`.  The function returns the best list of heights
found before the deadline (default 1000 s)."""

import time
import math
import numpy as np

# Optional linear‑programming backend

def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2·n‑1) × n matrix A with (A@g)[k] = (f*g)[k]."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _score(seq: np.ndarray) -> float:
    """Wrapper around the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq.tolist()))


# ---------------------------------------------------------
# Simple ℓ¹‑simplex projection (sum = target, non‑negative)
# ---------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project `v` onto {x ≥ 0, Σ x = target}."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - target
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]

    if rho_arr.size == 0:
        # all entries will be zero after projection
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------
# FFT‑based convolution helpers (fast for n up to a few thousand)
# ---------------------------------------------------------
def _fft_conv_max(f: np.ndarray):
    """Return (max_value, argmax_index) of the full auto‑convolution f*f."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())           # next power of two
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ---------------------------------------------------------
# Projected sub‑gradient descent on the ℓ¹‑simplex
# ---------------------------------------------------------
def _gradient_descent(
    f: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2000,
    eta0: float = 0.5,
    decay: float = 0.9995,
    min_step: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for `M(f)=max_k (f*f)_k`.
    Returns the best vector seen and its associated M‑value.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_M, _ = _fft_conv_max(f)
    best_score = best_M

    eta = eta0
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        M, k = _fft_conv_max(f)

        if M < best_score:
            best_score = M
            best_f = f.copy()

        # ----- sub‑gradient at the worst index k -----
        g = np.zeros_like(f)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        if low <= high:
            i_range = np.arange(low, high + 1)
            g[i_range] = 2.0 * f[k - i_range]

        # ----- propose a new point and project onto the simplex -----
        f_candidate = _project_simplex(f - eta * g, target_sum)

        M_candidate, _ = _fft_conv_max(f_candidate)

        if M_candidate < M:                     # improvement -> accept
            f = f_candidate
            # gently increase the step size again (but stay ≤ eta0)
            eta = min(eta / decay, eta0)
        else:                                    # no improvement -> shrink step
            eta *= 0.5

        # keep the step size from vanishing
        if eta < min_step:
            eta = eta0

        # exponential decay for stability
        eta *= decay

    return best_f, best_score


# ---------------------------------------------------------
# Simple stochastic local search (Gaussian perturbations)
# ---------------------------------------------------------
def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.5,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to the required ℓ¹‑norm (the evaluation function will
        # clip anyway, but keeping the norm constant helps the optimiser)
        target = math.sqrt(2.0 * cand.shape[0])
        cand = _project_simplex(cand, target)

        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        step *= decay
        if step < 1e-7:
            step = step_initial

    return best, best_score

    Search for a non‑negative step sequence with a low evaluation score.

    The routine consists of:
    * an optional LP‑based phase (small lengths, fast convergence);
    * a projected gradient‑descent phase on larger lengths;
    * a final stochastic hill‑climbing refinement.

    Parameters

    seed : int
        Random seed for reproducibility.

    list[float]

    """
