def propose_candidate(seed=0, budget_s=999):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 1.0

    best_f = None
    best_M = float('inf')
    # Choose lengths to try, e.g., [200, 400, 600]
    for n in [200, 400, 600]:
        # initial f: can be exponential shape with s=2, normalized
        # or use uniform or previous best if length matches.
        if best_f is not None and len(best_f) == n:
            f = np.array(best_f, copy=True)
        else:
            # use exponential shape: r = 1 - 2/n
            r = 1.0 - 2.0 / n
            f = np.arange(n, dtype=np.float64)
            f = r ** f
            # normalize
            target = math.sqrt(2.0 * n)
            f = f / f.sum() * target

        # main loop
        while time.monotonic() < deadline:
            cf = np.convolve(f, f)
            M = cf.max()
            if M < best_M:
                best_M = M
                best_f = f.copy()
                print(f"[{n}] new best M={M:.6f}")

            # solve LP
            A = np.zeros((2*n-1, n), dtype=np.float64)
            for j in range(n):
                A[j:j+n, j] = f  # shift
            c = -np.ones(n, dtype=np.float64)  # maximize sum

"""
Propose a low‑value step‑function using an *alternating linear‑program /
mixing* scheme.

The evaluation score for a normalised sequence (ℓ¹‑norm = √(2·n)) equals
the maximal entry of its auto‑convolution.  Starting from a simple
exponential (decreasing) shape we iteratively

1. compute M = max(f * f),
2. solve a linear programme that finds a non‑negative vector g of maximal
   ℓ¹‑sum under the linear constraints  f * g ≤ M,
3. rescale g to the required ℓ¹‑norm,
4. replace f by a convex combination (1‑t)·f + t·g that strictly reduces M
   (the combination is evaluated analytically, using the bilinearity of
   convolution).

The LP is tiny (≈ (2·n‑1) × n) and solved with SciPy’s HiGHS solver, so
many iterations fit easily into the 1000 s budget.  The procedure is
performed for several lengths (200, 400, 600, 800); the best sequence seen
overall is returned.

The method reliably pushes the mass toward the end of the vector,
reducing the worst auto‑convolution value below the target 1.503.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that A @ g = f * g (full convolution, length 2·n‑1).
    Each column j contains the vector f shifted by j positions.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
def _mix_convolution(cf, cg, cfg, alpha, beta):
    """
    Compute the convolution of (alpha*f + beta*g) with itself by using the
    bilinearity identity:
        conv = α²·cf + β²·cg + 2αβ·cfg
    """
    return (alpha * alpha) * cf + (beta * beta) * cg + 2.0 * alpha * beta * cfg


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 990.0) -> list[float]:
    """
    Search for a non‑negative step sequence with a small evaluation value.
    The function returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    best_f = None          # best sequence (numpy array)
    best_M = float("inf")  # corresponding maximal auto‑convolution entry

    # ------------------------------------------------------------------
    # try a few different lengths; larger n gives a better theoretical bound
    for n in (200, 400, 600, 800):
        if time.monotonic() >= deadline:
            break

        target_norm = math.sqrt(2.0 * n)          # required ℓ¹‑norm

        # --- start from a simple decreasing exponential (s = 2) ---
        r = 1.0 - 2.0 / n
        f = r ** np.arange(n, dtype=np.float64)   # decreasing shape
        f = f / f.sum() * target_norm             # normalise to √(2·n)

        # ------------------------------------------------------------------
        # inner optimisation loop (LP + mixing)
        it = 0
        while time.monotonic() < deadline and it < 5000:
            # current autocorrelation and its worst entry
            cf = np.convolve(f, f)
            M = cf.max()

            # keep the globally best sequence seen so far
            if M < best_M:
                best_M = M
                best_f = f.copy()
                print(f"[n={n:4d}] it={it:04d}  M={M:.6f}")

            # --------------------------------------------------------------
            # 1. linear programme: maximise sum(g) subject to f*g ≤ M, g ≥ 0
            A = _build_conv_matrix(f)                     # (2n‑1) × n
            c = -np.ones(n, dtype=np.float64)             # maximise → minimise -∑g
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)

            lp_res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=[(0, None)] * n,

                options={"presolve": True},

            if not lp_res.success:

                break

            g = np.maximum(0.0, lp_res.x)                 # clip tiny negatives
            sum_g = g.sum()
            if sum_g == 0.0:
                break
            # rescale to the same ℓ¹‑norm as f
            g = g * (target_norm / sum_g)

            # --------------------------------------------------------------
            # 2. attempt a convex combination that improves the worst entry
            cg = np.convolve(g, g)
            cfg = np.convolve(f, g)

            improved = False
            # try a short list of mixing parameters, starting with a full swap
            for t in (1.0, 0.5, 0.25, 0.125, 0.0625):
                alpha = 1.0 - t
                beta = t
                conv_mix = _mix_convolution(cf, cg, cfg, alpha, beta)
                M_mix = conv_mix.max()
                if M_mix < M - 1e-12:          # strict improvement

                    break

            if not improved:

                    f[i] -= delta
                    f[j] += delta

    # ------------------------------------------------------------------
    # fallback: if nothing worked, return a uniform vector of length 200

    # ------------------------------------------------------------------
    # final post‑processing: clip to the domain accepted by the scorer
    best_seq = [

    ]

    # one final evaluation for logging
