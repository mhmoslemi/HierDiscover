def _project_simplex(v, s):
    # v is np.ndarray
    # return w satisfying sum w = s, w>=0, minimizing L2 distance.
    # algorithm based on sorting
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _project_onto_simplex(v, s):
    # v: np.ndarray
    # s: float sum required
    # returns w
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    # find rho where u_i > cssv_i / i

        grad[idx_range] = 2.0 * f[k_max - idx_range]
        # donor i = argmax grad
        i = int(np.argmax(grad))
        # recipient j = argmin grad where j != i. Choose among zeros
        # we can find indices where grad == grad.min()
        min_val = grad.min()
        cand = np.where(grad <= min_val + 1e-12)[0]
        # Choose random from cand not equal i
        if len(cand) > 1:
            # remove i if present
            cand = cand[cand != i]
        if len(cand) == 0:
            j = rng.integers(0, n-1)
            while j == i:
                j = rng.integers(0, n-1)
        else:
            j = int(rng.choice(cand))
        # delta
        max_delta = f[i]
        if max_delta <= 1e-12:
            continue
        # fraction maybe dynamic
        frac = max(0.01, 0.1 * (1.0 - (time.monotonic() - start_time) / (deadline - start_time)))
        delta = rng.random() * max_delta * frac
        if delta <= 0:
            continue
        # copy old conv for rollback
        conv_old = conv.copy()
        fi_old = f[i]; fj_old = f[j]
        # linear updates:
        conv[i:i+n] -= 2.0 * delta * f
        conv[j:j+n] += 2.0 * delta * f
        # quadratic refinement
        conv[2*i] += delta * delta
        conv[2*j] += delta * delta
        s = i + j
        if s < 2 * n - 1:
            conv[s] -= 2.0 * delta * delta
        # apply move
        f[i] -= delta
        f[j] += delta
        new_max = conv.max()
        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob_up = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob_up:
                accept = True
        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f.copy()
                last_improve_it = it
                eval_val = 2.0 * n * best_max / (np.sum(best_f) ** 2)
                print(f"[SA] it={it} eval={eval_val:.6f} max={best_max:.6f} T={T:.4e}")
        else:
            conv[:] = conv_old
            f[i] = fi_old
            f[j] = fj_old
        # optionally restart
        if it - last_improve_it > 50000 and (time.monotonic() - start_time) > (deadline - start_time) * 0.5:
            # reinitialize from best_f perhaps
            f = best_f.copy()
            conv = np.convolve(f, f)
            cur_max = conv.max()
            last_improve_it = it
        it += 1
    return best_f

"""
Propose a step‑function (non‑negative height sequence) with a low
evaluation value 2·n·max(f*f) / (sum(f)**2).

The algorithm has two phases:

1. **Projected sub‑gradient descent** (≈45 % of the time budget):
   – The objective is the ∞‑norm of the auto‑convolution.
   – The sub‑gradient of the active (max) convolution entries is
     computed and a step is taken opposite to it.
   – After each step the vector is projected onto the probability simplex
     of fixed ℓ¹‑norm √(2·n) (the scaling does not affect the score).

2. **Refined simulated‑annealing** (remaining time):
   – Starts from the best vector of phase‑1.
   – In each iteration the worst convolution index `k_max` is located.
   – Mass is transferred from a donor index with the largest gradient
     (2·f_{k_max‑i}) to a receiver index with the smallest gradient.
   – The move is accepted with a Metropolis rule; a standard exponential
     temperature schedule is used.
   – Occasional restarts (with added noise) avoid long stagnation.

Both phases keep the ℓ¹‑norm constant, therefore the evaluation returned
by `evaluate_sequence` equals the current maximum convolution entry.
The code works for any reasonable length (default 300) and stays well
inside the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Target simplex sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_sequence(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Return a feasible (non‑negative, ℓ¹‑norm = √(2·n)) sequence.
    If a previous construction `height_sequence_1` exists, it is used as a
    starting point (truncated or padded as required).
    Otherwise a random Dirichlet vector is drawn.
    """
    target = math.sqrt(2.0 * n)

    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        elif seq.size < n:
            pad = np.full(n - seq.size, 1e-6, dtype=np.float64)
            seq = np.concatenate([seq, pad])
    else:
        # Dirichlet with all parameters = 1 → uniform on simplex
        seq = rng.dirichlet(np.ones(n)).astype(np.float64)

    # safety clipping
    seq = np.clip(seq, 0.0, 1000.0)
    # enforce the required ℓ¹‑norm
    seq = _project_onto_simplex(seq, target)
    return seq


# ----------------------------------------------------------------------
def _subgradient_descent(seq: np.ndarray,
                         remaining: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for minimizing max(conv(seq,seq)).
    Returns the best sequence seen and its max‑convolution value.
    """
    n = seq.shape[0]
    target_sum = np.sum(seq)          # = √(2·n)
    a = seq.copy()
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_val = conv.max()

    start = time.monotonic()
    it = 0
    # linear decay of step size; never let it become zero
    while time.monotonic() - start < remaining:
        conv = np.convolve(a, a)
        cur_max = conv.max()
        # indices where the maximum is attained (active set)
        active = np.isclose(conv, cur_max, atol=1e-12)
        ks = np.where(active)[0]

        # sub‑gradient of the max‑convolution entry
        grad = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        # step size: linearly shrinking to a tiny constant
        elapsed = time.monotonic() - start
        eta = 0.4 * (1.0 - elapsed / remaining) + 1e-4

        # gradient step + projection onto simplex
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        new_val = np.max(np.convolve(a_new, a_new))
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a_new.copy()
            eval_val = 2.0 * n * best_val / (target_sum ** 2)
            print(f"[GD] it={it:06d} eval={eval_val:.6f} max={best_val:.6f}")

        a = a_new
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(seq: np.ndarray,
                         best_seq: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Classic SA which moves a small amount of mass from a high‑gradient
    index to a low‑gradient index.  The temperature decreases exponentially.
    """
    n = seq.shape[0]
    target_sum = math.sqrt(2.0 * n)                 # keeps the scale fixed
    f = seq.copy()
    conv = np.convolve(f, f)
    cur_max = conv.max()
    best_f = best_seq.copy()
    best_max = best_val

    # temperature schedule
    T_start = 1.0
    T_end = 1e-4
    start = time.monotonic()
    total_sa = deadline - start
    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        T = T_start * (T_end / T_start) ** (elapsed / total_sa)

        # worst convolution index
        k_max = int(np.argmax(conv))

        # gradient of this entry (2·f_{k_max‑i})
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad = np.zeros(n, dtype=np.float64)
        grad[idx] = 2.0 * f[k_max - idx]

        # donor = index with largest gradient, receiver = smallest gradient
        i = int(np.argmax(grad))
        # pick a receiver among the (almost) zero‑gradient positions
        min_g = grad.min()
        cand = np.where(grad <= min_g + 1e-12)[0]
        cand = cand[cand != i]                     # ensure i ≠ j
        if cand.size == 0:
            # fallback to a completely random destination
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
        else:
            j = int(rng.choice(cand))

        max_delta = f[i]
        if max_delta <= 1e-12:
            continue

        # amount to move – gets smaller as time runs out
        frac = 0.15 * (1.0 - elapsed / total_sa) + 0.01
        delta = rng.random() * max_delta * frac
        if delta <= 0.0:
            continue

        # ----- apply the move (with rollback capability) -----
        conv_old = conv.copy()
        fi_old, fj_old = f[i], f[j]

        # linear part: –2·δ·f and +2·δ·f
        conv[i:i + n] -= 2.0 * delta * f
        conv[j:j + n] += 2.0 * delta * f

        # quadratic correction
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        s = i + j
        if s < 2 * n - 1:
            conv[s] -= 2.0 * delta * delta

        # update the vector itself
        f[i] -= delta
        f[j] += delta

        new_max = conv.max()

        # ----- Metropolis acceptance -----
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob_up = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob_up:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f.copy()
                last_improve = it
                eval_val = 2.0 * n * best_max / (target_sum ** 2)
                print(f"[SA] it={it:07d} eval={eval_val:.6f} max={best_max:.6f} T={T:.2e}")
        else:
            # rollback
            conv[:] = conv_old
            f[i] = fi_old
            f[j] = fj_old

        # occasional gentle restart if we have been stagnant too long
        if it - last_improve > 200_000 and elapsed > 0.5 * total_sa:
            # perturb the best solution a little and continue
            f = best_f.copy()
            noise = rng.normal(scale=0.02 * target_sum, size=n)
            f = np.maximum(f + noise, 0.0)
            f = _project_onto_simplex(f, target_sum)
            conv = np.convolve(f, f)
            cur_max = conv.max()
            last_improve = it

        it += 1

    return best_f, best_max


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """
    Search for a step‑function with a low evaluation value.
    The routine runs for roughly ``budget_s`` seconds (leaving a couple of
    seconds margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety buffer for the caller
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Choose a problem size – a longer vector gives a lower theoretical bound.
    n = 300                     # feel free to experiment (must be ≥ 2)
    seq0 = _initial_sequence(rng, n)

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent (≈45 % of the time)
    time_left = deadline - time.monotonic()
    gd_time = time_left * 0.45
    best_seq, best_max = _subgradient_descent(seq0, gd_time, rng)

    # ------------------------------------------------------------------
    # Phase 2 – refined simulated annealing (remaining time)
    best_seq, best_max = _simulated_annealing(best_seq,
                                              best_seq,
                                              best_max,
                                              deadline,
                                              rng)

    # ------------------------------------------------------------------
    # Final clean‑up: ensure validity for the scorer
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = float(evaluate_sequence(best_seq))
    print("[final] best evaluation =", final_score)
    return best_seq
