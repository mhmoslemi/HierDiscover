def simplex_projection(v, s):
    # v shape (n,)
    # project onto simplex of sum s.
    # Implementation from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def targeted_hill_climb(a, target_sum, deadline, rng, max_iters=2000):
    n = len(a)
    best = a.copy()
    best_sc = float(evaluate_sequence(a.tolist()))
    cur = a.copy()
    cur_sc = best_sc
    temperature = 0.2
    # step size for delta
    delta_factor = 0.05
    while time.monotonic() < deadline and max_iters > 0:
        # compute M and active ks
        B = np.convolve(cur, cur)
        M = B.max()
        # choose one active k randomly
        ks = np.where(B >= M - 1e-12)[0]
        k = rng.choice(ks)
        # choose i from indices that contribute to this k (i where partner valid)
        i_choices = np.arange(max(0, k-(n-1)), min(k, n-1)+1)
        # select i where cur[i] > 0
        i_vals = i_choices[cur[i_choices] > 0]
        if i_vals.size == 0:
            break
        i = rng.choice(i_vals)
        # partner p = k - i
        p = k - i
        # We'll move delta from i or p whichever larger
        # Choose source idx = i if cur[i] >= cur[p] else p
        src = i if cur[i] >= cur[p] else p
        # choose dest index j also randomly but avoid src and ensure cur[j] less than avg
        # we could sample j uniformly then optionally adjust.
        j = rng.integers(0, n)
        while j == src:
            j = rng.integers(0, n)
        # delta = min(cur[src] * delta_factor, target_sum * delta_factor / n)
        delta = min(cur[src] * 0.1, target_sum * 0.02)  # adjust
        # apply move
        cand = cur.copy()
        cand[src] -= delta
        cand[j] += delta
        if cand[src] < 0:
            continue
        cand_sc = float(evaluate_sequence(cand.tolist()))
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            if rng.random() < math.exp(-(cand_sc - cur_sc) / temperature):
                accept = True
        if accept:
            cur, cur_sc = cand, cand_sc
            if cur_sc < best_sc:
                best, best_sc = cur.copy(), cur_sc
        # update temperature and max_iters
        temperature = max(1e-6, temperature * 0.999)
        max_iters -= 1
    return best

"""
Propose a step‐function (non‑negative vector) that minimizes the
objective used in the challenge.

Algorithm
---------

1.  For a given length `n` the sum of the coefficients is fixed to
    `S = sqrt(2 n)`.  With this normalisation the score returned by
    ``evaluate_sequence`` equals `max_k (a * a)[k]`, i.e. the ℓ∞‑norm of the
    self‑convolution.  The optimisation problem is therefore:
       minimise  max convolution  subject to  a_i ≥ 0 , Σ a_i = S.

2.  The search is performed for several problem sizes (n≈400‑800) and
    several random restarts.  Each restart consists of two phases:

    * **Sub‑gradient descent** – the ℓ∞‑objective is a maximum of quadratic
      forms, hence its sub‑gradient w.r.t. `a` can be computed analytically.
      For the current active indices `k*` (where the convolution attains its
      maximum) the sub‑gradient is
            g_i = 2 Σ_{k*} a_{k*‑i} .
      After a step `a ← a – η g` the vector is projected back onto the
      simplex `{a≥0, Σ a = S}` using the standard O(n log n) Euclidean
      projection.  A diminishing step size `η = η0 / sqrt(t+1)` is used.

    * **Targeted hill‑climbing** – once the sub‑gradient phase stalls we
      perform a focused simulated‑annealing style walk.  At each iteration
      we look at a convolution index `k` attaining the current maximum,
      compute the pairwise contributions `c_i = a_i * a_{k‑i}` (zero where
      the partner index is out of range) and move a small amount of mass
      from an index with a large contribution to one with a small
      contribution.  Moves are accepted if they improve the score or with
      a probability `exp(-(Δscore)/T)` that slowly cools down.

3.  Throughout the search the best sequence (lowest score) is kept.
    If a sequence with a score < 1.503 is found the routine stops early.

The implementation relies only on NumPy and the provided
`evaluate_sequence` function; it respects the 1000 s time budget and
outputs a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1)."""
    return np.convolve(a, a)


def _score_linf(a: np.ndarray) -> float:
    """Return the ℓ∞‑norm of the self‑convolution (the true score
    when Σa = sqrt(2·len(a)))."""
    return float(np.max(_conv_full(a)))


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgradient_descent(
    a: np.ndarray,
    target_sum: float,
    max_iters: int,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Perform sub‑gradient steps on `a` to reduce max(a*a).
    Returns the best vector encountered (not necessarily the final one).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_linf(best_a)

    # initial step size – a fraction of the ℓ₁‑norm
    eta0 = target_sum * 0.05

    for t in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # current convolution and its maximum
        conv = _conv_full(a)
        M = conv.max()

        # active indices (within numerical tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # build mask of active indices (length 2n‑1)
        mask = np.zeros_like(conv, dtype=np.float64)
        mask[active] = 1.0

        # gradient: g_i = 2 Σ_{k∈active} a_{k‑i}
        # This is a convolution of the mask with the reversed a.
        a_rev = a[::-1]
        full = np.convolve(mask, a_rev)            # length 3n‑2
        start = n - 1                               # index aligning with i=0
        g = 2.0 * full[start:start + n]

        # diminishing step size
        eta = eta0 / math.sqrt(t)

        # gradient step + projection onto the simplex
        a_new = _project_simplex(a - eta * g, target_sum)

        # track the best solution seen so far
        cur_val = np.max(np.convolve(a_new, a_new))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[SGD] n={n} iter={t} best={best_val:.6f}")

        a = a_new

    return best_a

def _targeted_hill_climb(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 5000,
) -> np.ndarray:
    """
    Focused stochastic hill‑climber.  Mass is transferred from an index
    that heavily contributes to the current worst convolution entry to an
    index that contributes little.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _score_linf(cur)
    best = cur.copy()
    best_score = cur_score

    temperature = 0.2
    # typical amount of mass moved in one step
    delta_base = max(1e-8, 0.02 * target_sum / n)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # recompute convolution only when needed
        conv = _conv_full(cur)
        M = conv.max()
        # pick a random active position (where the max is attained)
        ks = np.where(conv >= M - 1e-12)[0]
        k = rng.choice(ks)

        # partner indices for this k
        idx = np.arange(max(0, k - (n - 1)), min(k, n - 1) + 1)
        partners = k - idx
        # contribution of each idx to the current maximum
        contrib = cur[idx] * cur[partners]

        # source = index with largest contribution, dest = smallest
        src_local = np.argmax(contrib)
        dst_local = np.argmin(contrib)
        i_src = idx[src_local]
        i_dst = idx[dst_local]

        if cur[i_src] <= 0:
            continue

        # amount to move – limited to a fraction of the source value
        delta = min(cur[i_src] * 0.3, delta_base)
        if delta <= 0:
            continue

        cand = cur.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta

        # evaluate candidate (no need to recompute sum – it is conserved)
        cand_score = _score_linf(cand)

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[hill] n={n} iter={it} new best={best_score:.6f}")

        # cooling schedule
        temperature = max(1e-6, temperature * 0.9995)

    return best

def _initial_vector(
    n: int, target_sum: float, rng: np.random.Generator, kind: str
) -> np.ndarray:
    """Generate a starting vector of length n."""
    if kind == "uniform":
        a = np.full(n, target_sum / n, dtype=np.float64)
    elif kind == "random":
        a = rng.exponential(scale=1.0, size=n)
        a = a / a.sum() * target_sum
    elif kind == "edge":
        # larger values at the ends, smaller in the middle
        centre = (n - 1) / 2.0
        weights = 1.0 / (np.abs(np.arange(n) - centre) + 1.0)
        a = rng.random(n) * weights
        a = a / a.sum() * target_sum
    elif kind == "cosine":
        # half‑cosine shape, always positive
        a = np.sin(math.pi * (np.arange(n) + 0.5) / n)
        a = a / a.sum() * target_sum
    else:
        # fallback to uniform
        a = np.full(n, target_sum / n, dtype=np.float64)
    return a

            a = _subgradient_descent(a, target_sum, sg_iters, deadline, rng)

            a = _targeted_hill_climb(a, target_sum, deadline, rng, max_iters=3000)

            # final evaluation (exact, using the provided function)
            cand_score = float(evaluate_sequence(a.tolist()))
            if cand_score < best_score:
                best_score, best_seq = cand_score, a.copy()
                print(f"[global] n={n} kind={kind} improved best = {best_score:.6f}")

            if best_score < 1.503:
                # Target reached – return immediately.
                print("[early] target reached")
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # Fallback: a simple uniform sequence (always feasible)
    if best_seq is None:
        n = 400
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] uniform sequence used, score =", best_score)

    # Return as plain Python list
    return [float(x) for x in best_seq]
