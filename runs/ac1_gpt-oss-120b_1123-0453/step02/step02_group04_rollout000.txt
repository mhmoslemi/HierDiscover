    import time, math, numpy as np
    start = time.monotonic()
    deadline = start + budget_s - 0.5
    rng = np.random.default_rng(seed)
    try:
        import cvxpy as cp
        HAS_CVX=True
    except Exception:

                conv_gg = np.convolve(g_rescaled, g_rescaled)

"""
Improved step‑function search.

The algorithm builds on the LP‑based scheme that was used previously but
adds a *line‑search* step: after solving the linear program that finds a new
feasible direction `g`, we compute the best convex mixing factor α∈[0,1] that
minimises the maximal auto‑convolution of the combination
    f_new = (1‑α)·f + α·g .
Because
    (f_new * f_new) = (1‑α)²·(f*f) + 2α(1‑α)·(f*g) + α²·(g*g)
the maximum can be evaluated in O(n) for any α, so a cheap ternary search
yields a near‑optimal α.  This usually gives a larger decrease of the score
per iteration than the fixed schedule used before.

For each length `n` (200 … 600) we run many LP‑updates with the adaptive
α, keep the overall best sequence, and finish with a short random‑perturb
hill‑climb.  If CVXPY is unavailable a pure random walk fallback is used.

The returned sequence satisfies the non‑negativity and normalisation
required by `evaluate_sequence`; its objective value is the maximal
auto‑convolution, which the scorer returns directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: maximum auto‑convolution of a non‑negative vector.
# ----------------------------------------------------------------------
def _max_auto_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)[k] (discrete convolution)."""
    return float(np.max(np.convolve(seq, seq)))


# ----------------------------------------------------------------------
# Helper: build the dense convolution matrix A such that (A @ g) == f * g.
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return a (2n‑1) × n matrix representing convolution with the fixed vector f."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        cols = k - np.arange(i_start, i_end + 1)
        A[k, cols] = f[i_start:i_end + 1]
    return A


# ----------------------------------------------------------------------
# LP sub‑routine: given f and its worst auto‑convolution M, find a feasible
# direction g (sum = target_sum) that does not increase the bound.
# Returns None if the LP fails.
# ----------------------------------------------------------------------
def _lp_optimize(f: np.ndarray, M: float, target_sum: float):
    """Solve the LP  max Σ g_i  s.t. g_i≥0 and (f*g)_k ≤ M """
    try:
        import cvxpy as cp
    except Exception:          # CVXPY not available
        return None

    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000)

    if g.value is None:
        return None

    g0 = np.maximum(g.value, 0.0)
    s = np.sum(g0)
    if s < 1e-12:
        return None
    return g0 * (target_sum / s)


# ----------------------------------------------------------------------
# Compute the optimal mixing factor α (in [0,1]) that minimises the maximal
# convolution of (1‑α)·f + α·g using a ternary search.
# ----------------------------------------------------------------------
def _optimal_alpha(conv_ff: np.ndarray,
                   conv_fg: np.ndarray,
                   conv_gg: np.ndarray,
                   iters: int = 30) -> float:
    lo, hi = 0.0, 1.0
    for _ in range(iters):
        a1 = lo + (hi - lo) / 3.0
        a2 = hi - (hi - lo) / 3.0
        cand1 = (1 - a1) ** 2 * conv_ff + 2 * a1 * (1 - a1) * conv_fg + a1 ** 2 * conv_gg
        cand2 = (1 - a2) ** 2 * conv_ff + 2 * a2 * (1 - a2) * conv_fg + a2 ** 2 * conv_gg
        if cand1.max() > cand2.max():
            lo = a1
        else:
            hi = a2
    return (lo + hi) / 2.0


# ----------------------------------------------------------------------
# Very simple random‑walk fallback (used when CVXPY is missing).
# ----------------------------------------------------------------------
def _fallback_random_search(budget_s: float, seed: int = 0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    n = 256
    target = math.sqrt(2.0 * n)
    seq = np.full(n, target / n, dtype=np.float64)
    best_seq = seq.copy()
    best_score = _max_auto_convolution(seq)

    while time.monotonic() < deadline:
        cand = seq + rng.normal(scale=0.02 * target, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target / np.sum(cand))
        sc = _max_auto_convolution(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            seq = cand
    print("[fallback] best score =", best_score)
    return best_seq, best_score

# ----------------------------------------------------------------------

    # ------------------------------------------------------------------
