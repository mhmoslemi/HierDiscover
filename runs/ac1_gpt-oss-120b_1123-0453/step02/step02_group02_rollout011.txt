def project_onto_simplex(v):
    # returns w in R^n such that w>=0, sum w = 1
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho: largest i where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _gradient_step(p, beta, step_factor):
    conv = np.convolve(p,p)
    max_conv = conv.max()
    if max_conv == 0:
        return p
    ratio = conv / max_conv
    # compute weights
    w = np.exp(beta * (ratio - 1.0))
    sum_w = w.sum()
    # correlation
    corr = np.correlate(p, w, mode='full')
    start = len(conv) - 1
    grad = 2 * beta * corr[start:start + len(p)] / sum_w
    # step: p - step_factor * grad
    p_new = p - step_factor * grad
    # project onto simplex
    return _project_simplex(p_new)

    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _score_from_p(p):
    n = p.shape[0]
    conv = np.convolve(p, p)
    max_conv = conv.max()
    return 2.0 * n * max_conv

def _gd_optimize(n, deadline, restarts=4, max_iters=3000,
                 beta=20.0, step0=0.5, init_seq=None):
    global _rng
    best_seq = None
    best_score = float('inf')
    for r in range(restarts):
        if time.monotonic() >= deadline:
            break
        if init_seq is not None and r == 0:
            p = np.array(init_seq, dtype=np.float64)
            p = p / p.sum()
        else:
            p = _rng.random(n)
            p = p / p.sum()
        # Simple stochastic perturb before start
        for it in range(max_iters):
            if time.monotonic() >= deadline:
                break
            conv = np.convolve(p, p)
            max_conv = conv.max()
            score = 2.0 * n * max_conv
            if score < best_score:
                best_score = score
                best_seq = p.copy()
                print(f"[GD] n={n} restart={r} it={it} score={score:.6f}")
            # Stop if already good enough
            if best_score < 1.48:  # arbitrary early break
                break
            # gradient based on soft‑max over convolution entries
            if max_conv == 0:
                break
            ratio = conv / max_conv  # <=1
            w = np.exp(beta * (ratio - 1.0))  # max weight = 1
            sum_w = w.sum()
            if sum_w == 0:
                break
            # correlation of p with w
            corr = np.correlate(p, w, mode='full')
            start = len(conv) - 1
            grad = 2.0 * beta * corr[start:start + n] / sum_w
            # step size decreases like 1/sqrt(it+1)
            step = step0 / math.sqrt(it + 1)
            p_new = p - step * grad
            # small random jitter every 200 iterations to avoid plateaus
            if it % 200 == 0:
                jitter = _rng.normal(scale=1e-3, size=n)
                p_new = p_new + jitter
            # project back onto simplex
            p = _project_simplex(p_new)
    return best_seq, best_score

def _local_search(p, deadline, n_iter=20000):
    """Simple hill‑climbing that swaps a tiny amount of mass."""
    best = p.copy()
    best_score = _score_from_p(best)
    n = best.shape[0]
    while time.monotonic() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount moved is a fraction of source mass
        delta = _rng.uniform(-0.1, 0.1) * best[i]  # can be positive or negative
        if delta == 0:
            continue
        cand_i = best[i] - delta
        cand_j = best[j] + delta
        if cand_i < 0 or cand_j < 0:
            continue
        cand = best.copy()
        cand[i] = cand_i
        cand[j] = cand_j
        # re‑project onto simplex is unnecessary because total sum unchanged
        sc = _score_from_p(cand)
        if sc < best_score:
            best_score = sc
            best = cand
            print(f"[local] score={sc:.6f}")
    return best, best_score

    """

    """
    global _rng
    _rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    overall_best_seq = None
    overall_best_score = float('inf')

    # try to reuse a previously discovered sequence if available
    try:

        if isinstance(prev, (list, tuple, np.ndarray)):
            prev = np.asarray(prev, dtype=np.float64)
            if prev.size > 0 and np.all(prev >= 0):
                # normalise to sum 1 for the internal optimiser
                prev_norm = prev / prev.sum()
                n_prev = prev.shape[0]
                seq, score = _gd_optimize(n=n_prev, deadline=deadline,
                                          restarts=1, max_iters=4000,

                    overall_best_score = score

    except Exception:
        pass

    # a set of problem sizes to explore; larger n gives a tighter theoretical bound
    sizes = [300, 400, 500, 600, 800]

    for n in sizes:
        if time.monotonic() >= deadline:
            break
        # allocate a fraction of the remaining budget to this size
        sub_deadline = time.monotonic() + min(30.0, (deadline - time.monotonic()) * 0.2)
        seq, score = _gd_optimize(n=n, deadline=sub_deadline,

        if polished_score < overall_best_score:

            overall_best_score = polished_score

"""
Gradient–descent optimizer for the max‑auto‑convolution problem.

The evaluation score is
    2·n·max(conv(f,f)) / (sum(f))² .
Because the score is homogeneous of degree 0,
the sum of the coefficients can be fixed (we use a probability vector
∑a_i = 1).  The objective then reduces to
    2·n·max(p∗p)   with  p ≥ 0,  ∑p = 1 .
We minimise a smooth approximation of the maximum
via a soft‑max over the convolution entries, perform a projected
gradient step (projecting onto the simplex), and repeat.
Several random restarts and a cheap stochastic polishing phase are
run for a handful of lengths; the best sequence found within the
time budget is returned.

The code uses only NumPy and the supplied ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global random generator – set in ``propose_candidate``.
_rng = None


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest index where the condition holds.
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _score_from_p(p: np.ndarray) -> float:
    """Return the exact evaluation score for a normalised probability vector."""
    n = p.shape[0]
    conv = np.convolve(p, p)
    max_conv = conv.max()
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def _gd_optimize(
    n: int,
    deadline: float,
    restarts: int = 4,
    max_iters: int = 3000,
    beta: float = 20.0,
    step0: float = 0.6,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray | None, float]:
    """
    Projected gradient descent on the simplex.

    Parameters
    ----------
    n : int
        Length of the sequence.
    deadline : float
        Absolute time (monotonic clock) after which the routine must stop.
    restarts : int
        Number of independent random starts.
    max_iters : int
        Upper bound on iterations per restart.
    beta : float
        Inverse temperature for the soft‑max (larger → tighter focus on the peak).
    step0 : float
        Initial step size; the actual step decays as step0 / sqrt(t+1).
    init_seq : np.ndarray or None
        If given, the first restart starts from this vector (already normalised).

    Returns
    -------
    best_seq, best_score
        The best normalised probability vector found and its corresponding score.
    """
    global _rng
    best_seq = None
    best_score = float("inf")

    for r in range(restarts):
        if time.monotonic() >= deadline:
            break

        if init_seq is not None and r == 0:
            p = np.array(init_seq, dtype=np.float64)
            p = p / p.sum()
        else:
            p = _rng.random(n)
            p = p / p.sum()

        for it in range(max_iters):
            if time.monotonic() >= deadline:
                break

            conv = np.convolve(p, p)
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv

            if cur_score < best_score:
                best_score = cur_score
                best_seq = p.copy()
                print(f"[GD] n={n:4d} restart={r} it={it:4d} score={cur_score:.6f}")

            # early exit if we are already spectacularly good
            if best_score < 1.45:
                break

            if max_conv == 0.0:
                break

            # ----- soft‑max weights over the convolution entries -----
            ratio = conv / max_conv                     # ≤ 1, the peak has ratio 1
            w = np.exp(beta * (ratio - 1.0))            # peak weight = 1
            sum_w = w.sum()
            if sum_w == 0.0:
                break

            # ----- gradient of the soft‑max objective w.r.t. p -----
            # correlation of p with w (no reversal) gives Σ_i p_i w_{i+t}
            corr = np.correlate(p, w, mode="full")
            start = len(conv) - 1                      # index of lag 0
            grad = 2.0 * beta * corr[start:start + n] / sum_w

            # ----- step size schedule -----
            step = step0 / math.sqrt(it + 1.0)

            # occasional jitter to avoid getting stuck
            if it % 200 == 0:
                jitter = _rng.normal(scale=1e-3, size=n)
                p = p + jitter

            p_new = p - step * grad
            p = _project_simplex(p_new)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(p: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic walk that preserves the ℓ¹‑mass.
    Moves a tiny amount of mass between two random positions and keeps the
    change if it improves the score.
    """
    best = p.copy()
    best_score = _score_from_p(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random fraction of the mass at the source index
        delta = _rng.uniform(0.0, 0.1) * best[i]
        if delta == 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # the total sum stays 1, just need to stay non‑negative
        if cand[i] < 0.0:
            continue
        sc = _score_from_p(cand)
        if sc < best_score:
            best_score = sc
            best = cand
            print(f"[local] score={sc:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    The routine performs:
        1.  optional warm‑start from a previously known construction,
        2.  projected gradient descent on several lengths (300…800),
        3.  a short random walk polishing phase.

    The best sequence found before the time limit is returned.
    """

    _rng = np.random.default_rng(seed)

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 0) Warm‑start from a previously discovered sequence if it exists.
    # ------------------------------------------------------------------
    try:

        if isinstance(prev, (list, tuple, np.ndarray)):
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                # normalise to a probability vector
                prev_norm = prev_arr / prev_arr.sum()
                n_prev = prev_arr.shape[0]
                seq, scr = _gd_optimize(
                    n=n_prev,
                    deadline=deadline,
                    restarts=1,
                    max_iters=4000,
                    beta=22.0,
                    step0=0.7,
                    init_seq=prev_norm,
                )
                if seq is not None and scr < overall_best_score:
                    overall_best_seq = seq.copy()
                    overall_best_score = scr
                    print(f"[global] warm‑start improvement (n={n_prev}): {scr:.6f}")
    except Exception:
        # any problem – just skip the warm‑start
        pass

    # ------------------------------------------------------------------
    # 1) Gradient‑descent refinement on several problem sizes.
    # ------------------------------------------------------------------
    # Larger n gives a tighter theoretical lower bound (≈ 2n/(2n‑1)).
    sizes = [300, 400, 500, 600, 800]

    for n in sizes:
        if time.monotonic() >= deadline:

                                             (deadline - time.monotonic()) * 0.2)

        seq, scr = _gd_optimize(

            deadline=sub_deadline,
            restarts=3,

            beta=25.0,

        )
        if seq is not None and scr < overall_best_score:

            overall_best_score = scr
            print(f"[global] new best length {n}: {scr:.6f}")

        # stop early if we have already reached a very good bound
        if overall_best_score < 1.45:

    # ------------------------------------------------------------------
    # 2) Light‑weight polishing with a random walk.
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.monotonic() < deadline:

    # ------------------------------------------------------------------

        overall_best_score = _score_from_p(overall_best_seq)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Final safety: clip to the allowed interval and return as plain floats.
    # ------------------------------------------------------------------
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)
    return [float(x) for x in overall_best_seq.tolist()]
