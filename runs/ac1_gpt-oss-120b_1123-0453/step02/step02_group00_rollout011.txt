def proj_simplex(v, s):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)

def project_onto_simplex(v, s):
    """Project vector v onto the simplex {x >= 0, sum x = s}."""
    # v is 1-D numpy array.
    # Implementation based on sorting.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Optimization of non‑negative step sequences via up‑sampling,
projected sub‑gradient descent and simulated‑annealing refinement.
The algorithm starts from a known good solution (``height_sequence_1`` if
available), interpolates it to a larger length, keeps the ℓ¹‑norm
fixed to √(2·n) and minimises the maximal auto‑convolution entry.
Projected sub‑gradient steps drive the worst convolution down, and a
final stochastic mass‑transfer annealing cleans the result.
"""

import time
import math
import numpy as np

# ---------------------------------------------------------------------------------
# Helper utilities
# ---------------------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _max_autocorr(f: np.ndarray) -> float:
    """Return max value of the auto‑convolution f*f using FFT."""
    n = f.shape[0]
    L = _next_pow2(2 * n - 1)
    F = np.fft.rfft(f, n=L)
    conv = np.fft.irfft(F * F, n=L)[: 2 * n - 1]
    return float(conv.max())

def _score_from_max(n: int, max_b: float) -> float:
    """Convert max auto‑convolution value to the evaluation‑function score."""
    # The denominator is (sum_i a_i)². In the following code we keep the
    # ℓ¹‑norm equal to √(2·n), therefore the score equals max_b.
    # The formula is kept for clarity.
    sum_target = math.sqrt(2.0 * n)
    return 2.0 * n * max_b / (sum_target * sum_target)

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Sum 's' must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the largest rho such that u_rho > (cssv_rho / rho)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgradient_step(f: np.ndarray, sum_target: float, step: float) -> np.ndarray:
    """Perform one projected sub‑gradient step that tries to lower max(f∗f)."""
    n = f.shape[0]
    # max entry of the convolution and its index
    n_fft = _next_pow2(2 * n - 1)
    F = np.fft.rfft(f, n=n_fft)
    conv = np.fft.irfft(F * F, n=n_fft)[: 2 * n - 1]
    k = int(np.argmax(conv))

    # build the sub‑gradient for shift k
    g = np.zeros_like(f)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    idx_j = k - idx_i
    # The contribution is added twice (i → j and j → i); the diagonal case
    # receives 2·f_i automatically.
    np.add.at(g, idx_i, f[idx_j])
    np.add.at(g, idx_j, f[idx_i])

    # gradient descent step (note the minus sign)
    f_new = f - step * g
    # project back onto the simplex of the desired ℓ¹‑norm
    f_new = _project_simplex(f_new, sum_target)
    return f_new

def _run_subgradient(f0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Iteratively apply the projected sub‑gradient method until the deadline."""
    n = f0.shape[0]
    sum_target = math.sqrt(2.0 * n)
    f = f0.copy()
    best = f.copy()
    best_score = _score_from_max(n, _max_autocorr(f))
    step = 0.1          # initial step size (chosen empirically)
    decay = 0.9995      # exponential decay
    it = 0
    while time.monotonic() < deadline:
        f = _subgradient_step(f, sum_target, step)
        cur_max = _max_autocorr(f)
        cur_score = _score_from_max(n, cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best = f.copy()
            print(f"[SG] it={it} score={cur_score:.6f} max={cur_max:.6f}")
        step *= decay
        it += 1
        # occasional restart to avoid stagnation
        if it % 2000 == 0:
            # small random perturbation then re‑project
            rng = np.random.default_rng()
            pert = rng.normal(scale=step * 10.0, size=n)
            f = _project_simplex(f + pert, sum_target)
    return best, best_score

def _run_simulated_annealing(f0: np.ndarray,
                             sum_target: float,
                             best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing that preserves the ℓ¹‑norm."""
    rng = np.random.default_rng()
    n = f0.shape[0]
    cur = f0.copy()
    cur_max = _max_autocorr(cur)
    cur_score = _score_from_max(n, cur_max)

    best = cur.copy()
    best_sc = best_score

    # temperature schedule – start relatively hot and cool slowly
    temp = max(0.01, cur_score * 0.2)
    cooling = 0.9999

    while time.monotonic() < deadline:
        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose a transferable amount – at most the content of the source cell
        max_eps = cur[i]
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5  # move at most half of the source mass
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps

        # evaluate
        cand_max = _max_autocorr(cand)
        cand_score = _score_from_max(n, cand_max)

        delta = cand_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / temp):
            cur = cand
            cur_score = cand_score
            cur_max = cand_max
            if cand_score < best_sc:
                best_sc = cand_score
                best = cand.copy()
                print(f"[SA] new best {best_sc:.6f}")
        # temperature decay
        temp *= cooling
        # keep temperature from vanishing
        if temp < 1e-8:
            temp = 1e-8
    return best, best_sc

def _interpolate_to_length(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq

    except Exception:

        final_score = best_score
        print(f"[final] fallback score {final_score:.6f}")

"""
Optimization of a non‑negative step sequence.

The routine works in three stages:

1️⃣  Initialise – if the global variable ``height_sequence_1`` exists we
    reuse it, otherwise a random vector is generated.  The vector is
    interpolated to a relatively large length (default 1800) and scaled
    to the ℓ¹‑norm √(2·n); the evaluation function then equals the
    maximal entry of the auto‑convolution.

2️⃣  Projected sub‑gradient descent – at every iteration the worst shift
    ``k`` of the convolution is identified, a sub‑gradient of that
    entry is built and a step is taken opposite to it.  The resulting
    vector is projected back onto the simplex {x ≥ 0, Σx = √(2·n)}.
    This phase drives the maximal auto‑convolution value down.

3️⃣  Simulated‑annealing refinement – mass is transferred between two
    randomly chosen positions while preserving the ℓ¹‑norm.
    Moves that improve the score are always accepted; worsening moves
    are accepted with a temperature‑dependent probability.  The
    temperature cools exponentially.

All phases respect the 0 ≤ a_i ≤ 1000 clipping performed inside
``evaluate_sequence`` (the generated numbers are far below that bound).
The function returns a plain list of floats before the global deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------------- #
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------------- #

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _max_autocorr_fft(f: np.ndarray) -> float:
    """Maximum value of f⋆f using an FFT of length a power of two."""
    n = f.shape[0]
    L = _next_pow2(2 * n - 1)
    F = np.fft.rfft(f, n=L)
    conv = np.fft.irfft(F * F, n=L)[: 2 * n - 1]
    return float(conv.max())


def _score_from_max(n: int, max_b: float) -> float:
    """
    Convert the maximal auto‑convolution entry into the evaluation‑function
    score.  When Σa_i = √(2·n) the score equals ``max_b``.
    """
    sum_target = math.sqrt(2.0 * n)
    return 2.0 * n * max_b / (sum_target * sum_target)


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        { x ∈ ℝⁿ | x_i ≥ 0,  Σ x_i = s }.
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(f: np.ndarray,
                      sum_target: float,
                      step: float,
                      rng: np.random.Generator) -> np.ndarray:
    """
    One projected sub‑gradient step that tries to lower the worst
    auto‑convolution entry.
    """
    n = f.shape[0]
    L = _next_pow2(2 * n - 1)
    F = np.fft.rfft(f, n=L)
    conv = np.fft.irfft(F * F, n=L)[: 2 * n - 1]
    max_val = conv.max()
    # Pick a worst shift (break ties randomly)
    worst = np.where(conv >= max_val - 1e-12)[0]
    k = int(rng.choice(worst))

    # Sub‑gradient of entry k (vectorised with np.add.at)
    g = np.zeros_like(f)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    idx_j = k - idx_i
    # each pair (i,j) contributes to both coordinates; the diagonal
    # (i==j) receives two copies, which is the correct derivative 2·f_i.
    np.add.at(g, idx_i, f[idx_j])
    np.add.at(g, idx_j, f[idx_i])

    # Gradient descent step and projection back to the simplex
    f_new = f - step * g
    f_new = _project_simplex(f_new, sum_target)
    return f_new


def _run_subgradient(f0: np.ndarray,
                     deadline: float,

    Repeated projected sub‑gradient steps until ``deadline``.
    Returns the best vector seen and its score.

    f = f0.copy()

        cur_max = _max_autocorr_fft(f)
        cur_score = _score_from_max(n, cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best = f.copy()
            print(f"[SG] it={it:06d} score={cur_score:.6f} max={cur_max:.6f}")
        step *= decay
        it += 1
        # occasional small random shake‑up to avoid stagnation
        if it % 2000 == 0:
            pert = rng.normal(scale=step * 10.0, size=n)
            f = _project_simplex(f + pert, sum_target)
    return best, best_score


def _run_simulated_annealing(f0: np.ndarray,
                             sum_target: float,
                             best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Mass–transfer simulated annealing that keeps the ℓ¹‑norm fixed.
    Operates on the current best vector ``f0`` until ``deadline``.
    """
    rng = np.random.default_rng()
    n = f0.shape[0]
    cur = f0.copy()
    cur_max = _max_autocorr_fft(cur)
    cur_score = _score_from_max(n, cur_max)

    best = cur.copy()
    best_sc = best_score

    temp = max(0.01, cur_score * 0.2)   # start temperature
    cooling = 0.9999

    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = cur[i]
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.5       # move up to half of source mass
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps

        cand_max = _max_autocorr_fft(cand)
        cand_score = _score_from_max(n, cand_max)

        delta = cand_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / temp):
            cur = cand
            cur_score = cand_score
            cur_max = cand_max
            if cand_score < best_sc:
                best_sc = cand_score
                best = cand.copy()
                print(f"[SA] new best {best_sc:.6f}")

        temp *= cooling
        if temp < 1e-9:
            temp = 1e-9
    return best, best_sc


def _interpolate_to_length(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------------- #
# Main entry point required by the competition harness
# ----------------------------------------------------------------------------- #

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The algorithm consists of:
      • initialisation (re‑use ``height_sequence_1`` when available),
      • up‑sampling to a larger length (default 1800),
      • projected sub‑gradient descent,
      • simulated‑annealing refinement.

    The function respects the global time budget and returns a plain list
    of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # reserve a small margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------------- #
    # 1️⃣  Initialise
    # --------------------------------------------------------------------- #
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        print("[init] Using existing height_sequence_1")
    except Exception:
        # random positive start, normalised to ℓ¹‑norm √(2·len)
        base_len = 500
        base_seq = rng.random(base_len) + 0.1
        base_seq = base_seq / base_seq.sum() * math.sqrt(2.0 * base_len)
        print("[init] No external sequence – starting randomly")

    # Target length – larger n gives more freedom but also larger work.
    n_target = 1800
    if time.monotonic() + 5.0 >= deadline:   # not enough time → stay small
        n_target = base_seq.shape[0]

    # Upsample / interpolate and renormalise to the required ℓ¹‑norm.
    f = _interpolate_to_length(base_seq, n_target)
    f = np.maximum(f, 0.0)
    sum_target = math.sqrt(2.0 * n_target)
    f = f / f.sum() * sum_target

    # --------------------------------------------------------------------- #
    # 2️⃣  Projected sub‑gradient descent (≈ 60 % of remaining budget)
    # --------------------------------------------------------------------- #
    mid_deadline = start + 0.6 * (deadline - start)
    best_seq, best_score = _run_subgradient(f, mid_deadline, rng)

    # --------------------------------------------------------------------- #
    # 3️⃣  Simulated‑annealing refinement (remaining ~40 % of budget)
    # --------------------------------------------------------------------- #
    if time.monotonic() < deadline:
        best_seq, best_score = _run_simulated_annealing(
            best_seq, sum_target, best_score, deadline
        )

    # --------------------------------------------------------------------- #
    # Final sanity check with the official evaluator (tiny overhead)
    # --------------------------------------------------------------------- #
    final_seq = [float(x) for x in best_seq.tolist()]
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence => {final_val:.6f}")
    except Exception:
        # Should never happen – we fall back to the internally computed score.
        final_val = best_score
        print(f"[final] fallback score {final_val:.6f}")

    return final_seq
