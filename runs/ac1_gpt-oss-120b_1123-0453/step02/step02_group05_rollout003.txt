def _project_onto_simplex(v, s):
    if s < 0:
        raise ValueError...
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-index?
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _simplex_projection(v, s):
    # v: array, s>0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Projected gradient descent for minimizing the peak of a self‑convolution.

The evaluation function after normalisation reduces to the maximum entry
of the full auto‑convolution `conv = f * f`.  We therefore minimise
`max(conv)` under the constraints

    f_i ≥ 0 ,   Σ_i f_i = √(2·n).

The objective is approximated by a smooth “soft‑max’’:
    φ_α(f) = log Σ_k exp(α·conv_k)   (α → ∞ gives max(conv)).

∂φ_α/∂f_i = 2·Σ_k w_k·f_{k-i}
where   w_k = exp(α·conv_k)/Σ_j exp(α·conv_j)  (soft‑max weights).

A projected gradient step with a decreasing step‑size and a slowly
increasing α moves the sequence towards a flatter convolution.
Multiple random restarts (different lengths n and random initialisations)
explore the search space, and the best sequence seen overall is returned.

The code respects the 1000 s time budget, prints occasional progress
messages and finally returns a Python list of floats suitable for the
provided `evaluate_sequence` scorer.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Wang & Carreira‑Perpinán (2013).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)  # 1‑based index
    cond = u - cssv / ind > 0

    if not np.any(cond):
        # All entries will be zero; distribute mass uniformly
        return np.full_like(v, s / n)

    rho = np.where(cond)[0][-1]            # last true index
    theta = cssv[rho] / (rho + 1)          # dual variable
    w = np.maximum(v - theta, 0.0)

    # Numerical fix: ensure exact sum (tiny rounding errors)
    diff = w.sum() - s
    if abs(diff) > 1e-10:
        nz = w > 0
        if nz.any():
            w[nz] -= diff / nz.sum()
            w = np.maximum(w, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_sequence(
    n: int,
    rng: np.random.Generator,
    warm_seq: np.ndarray | None = None,
) -> np.ndarray:
    """
    Produce a feasible initial sequence of length n with ℓ¹‑norm √(2·n).

    If ``warm_seq`` is provided and has the correct length, it is
    normalised and projected onto the simplex; otherwise a random
    Dirichlet‑type vector is generated.
    """
    target = math.sqrt(2.0 * n)
    if warm_seq is not None and warm_seq.shape[0] == n:
        seq = np.clip(warm_seq.astype(np.float64), 0.0, 1000.0)
        seq = _project_onto_simplex(seq, target)
        return seq

    # Random positive vector (exponential → Dirichlet after projection)
    rand_vec = rng.exponential(scale=1.0, size=n)
    seq = _project_onto_simplex(rand_vec, target)
    return seq

def propose_candidate(seed: int = 42, budget_s: float = 1100.0) -> list[float]:
    """
    Search for a step‑function with a small peak in its auto‑convolution.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds.  The routine stops a couple
        of seconds before the limit so that the caller can still retrieve
        the result.

    Returns
    -------
    list[float]
        The best sequence discovered (as plain Python floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Global best across all restarts
    best_f = None
    best_val = np.inf    # this is max(conv) because of the normalisation
    best_n = None

    # Candidate lengths to try (feel free to adjust)
    length_options = np.array([150, 180, 200, 250, 300, 350, 400, 450, 500, 600])

    total_iters = 0
    restart_id = 0

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # Choose a length for this restart (biased towards moderate sizes)
        n = int(rng.choice(length_options))
        warm = None
        if "height_sequence_1" in globals():
            hs = np.array(height_sequence_1, dtype=np.float64)
            if hs.shape[0] == n:
                warm = hs

        f = _initial_sequence(n, rng, warm)
        target_sum = math.sqrt(2.0 * n)

        # Hyper‑parameters for this run
        lr = 0.30                     # initial step size
        alpha = 5.0                   # smoothing of max (soft‑max temperature)
        max_iter = 10_000_000         # safety cap; loop will break earlier
        no_improve = 0
        best_local_max = np.inf
        best_local_f = f.copy()
        it = 0

        # ------------------------------------------------------------------
        while time.monotonic() < deadline and it < max_iter:
            # Full auto‑convolution (full mode → length 2·n‑1)
            conv = np.convolve(f, f)
            cur_max = conv.max()

            # Record improvement (global and local)
            if cur_max < best_local_max - 1e-12:

            lr = max(lr * 0.9999, 1e-5)

        # to avoid being trapped in the same basin.
        # (No effect on the already stored global best.)
        # --------------------------------------------------------------------

    # ----------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    if best_f is None:
