def _lp_refine_advanced(n, target_sum, deadline, max_iters=60, line_grid=21):
    best_seq = None
    best_score = inf
    f = ... random positive then normalized to target_sum.

    for it in range(max_iters):
        if time.monotonic() >= deadline: break

        # Build conv matrix A
        A = conv_matrix(f)

        # Solve LP for g
        try:
            import cvxpy as cp
        except ImportError: break

        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= np.max(np.convolve(f,f))]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        if g.value is None: break

        g0 = np.maximum(g.value, 0)
        # Rescale g to target_sum
        s_g = np.sum(g0)
        if s_g == 0: break
        g_rescaled = g0 * (target_sum / s_g)

        # Compute conv arrays
        conv_f_f = np.convolve(f, f)
        conv_f_g = np.convolve(f, g_rescaled)
        conv_g_g = np.convolve(g_rescaled, g_rescaled)

        # line search
        t_vals = np.linspace(0.0, 1.0, line_grid)
        best_t = 0.0
        best_M = np.max(conv_f_f)
        for t in t_vals[1:]:
            M_t = np.max( (1-t)**2 * conv_f_f + 2*t*(1-t) * conv_f_g + t**2 * conv_g_g)
            if M_t < best_M - 1e-12:
                best_M = M_t
                best_t = t
        # update f
        f_new = (1 - best_t) * f + best_t * g_rescaled
        # Ensure non-negativity (should be)
        # Score evaluation:
        sc = _score(f_new.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = f_new.copy()
            print(f"[LPAP] n={n} it={it} t={best_t:.3f} score={sc:.6f}")
        # early exit condition
        if best_score < 1.45: break
        f = f_new

    return best_seq, best_score

def _annealing_local_search(seq, deadline):
    current = seq.copy()
    n = len(seq)
    cur_val = conv_max_fft(current)
    best = current.copy()
    best_val = cur_val
    t0 = 0.1 * cur_val  # temperature start
    t_end = 1e-6 * cur_val  # final temperature
    total_time = deadline - time.monotonic()
    start_time = time.monotonic()
    iteration = 0
    while time.monotonic() < deadline:
        # compute remaining fraction
        elapsed = time.monotonic() - start_time
        remaining_frac = max(0.0, (deadline - time.monotonic()) / total_time)
        # temperature schedule linear
        T = t0 * remaining_frac + t_end * (1 - remaining_frac)
        # decide move type
        if _rng.random() < 0.2:
            # targeted move reduce max contributions
            # compute conv array:
            conv = np.convolve(current, current)
            k_max = np.argmax(conv)
            # contributions: only i where counterpart valid
            # compute array of contributions:
            idx_i = np.arange(n)
            idx_j = k_max - idx_i
            mask = (idx_j >= 0) & (idx_j < n)
            contrib = np.zeros_like(current)
            contrib[mask] = current[mask] * current[idx_j[mask]]
            i = int(np.argmax(contrib))
            # target index where counterpart not existent:
            if k_max <= n-1:
                if k_max+1 <= n-1:
                    t = _rng.integers(k_max+1, n)
                else:
                    t = _rng.integers(0, n)  # fallback random
            else:
                low = 0
                high = k_max - (n-1)
                if high >= 0:
                    t = _rng.integers(0, high+1)
                else:
                    t = _rng.integers(0, n)
            # propose epsilon
            max_eps = min(current[i] * 0.3, 5.0)  # limited
            eps = _rng.random() * max_eps
            if eps == 0: continue
            cand = current.copy()
            cand[i] = max(0.0, cand[i] - eps)
            cand[t] = min(1000.0, cand[t] + eps)
        else:
            # simple transfer
            i = _rng.integers(0, n)
            if current[i] <= 1e-12:
                continue
            j = _rng.integers(0, n-1)
            if j >= i:
                j += 1
            # step size depends on temperature:
            max_transfer = min(current[i], 0.2 * current[i])  # up to 20%
            eps = _rng.random() * max_transfer
            cand = current.copy()
            cand[i] -= eps
            cand[j] = min(1000.0, cand[j] + eps)
        # compute new value fast via conv_max_fft
        new_val = conv_max_fft(cand)
        delta = new_val - cur_val
        if delta < -1e-12 or _rng.random() < math.exp(-delta / (T+1e-12)):
            # accept
            current = cand
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = cand.copy()
                print(f"[anneal] iter={iteration} score={ (2*len(seq))*cur_val/(np.sum(current)**2):.6f}")
        iteration += 1
    return best, best_val

""" 
Improved search for low‑value step‑function coefficients.

Algorithm
---------
1. **LP refinement with line‑search mixing**  
   For several problem sizes `n` we start from a random feasible vector
   `f` (ℓ¹‑norm = √(2 n)).  At each iteration we solve the linear program  

        max  Σ g_i   s.t.   g ≥ 0,  f * g ≤ M ,

   where `M = max(f*f)`.  The LP returns a candidate `g`.  
   Rather than using a fixed convex‑combination factor `t`, we evaluate the
   whole family `f(t) = (1‑t)f + t g` by a cheap one‑dimensional line‑search
   (quadratic interpolation of the convolution) and pick the `t` that yields
   the smallest new maximum convolution value.  This accelerates the monotone
   decrease of the objective.

2. **Simulated‑annealing local polishing**  
   Starting from the best vector obtained above we perform a stochastic walk
   that preserves the total ℓ¹‑mass.  Moves consist of transferring a small
   amount of mass between two random positions, and – with a small probability
   – a *targeted* move that explicitly reduces the contribution to the
   currently maximal autocorrelation shift.  Changes that increase the
   objective are accepted with probability `exp(‑Δ/T)`, where the temperature
   `T` is linearly cooled to zero.

Both phases respect a global 1000 s wall‑clock limit.  The routine works
without any external data; if `cvxpy` is unavailable the algorithm skips the
LP phase and falls back to the annealing stage only.

The returned sequence consists of non‑negative floats (clipped to `[0,1000]`)
and can be directly fed to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# global random generator
_rng = None


def conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the dense convolution matrix A such that (A @ g) == f * g."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    return A


def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq))


def _lp_refine_with_line_search(
    n: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 60,
    line_grid: int = 21,
) -> tuple[np.ndarray | None, float]:
    """
    Perform the LP‑based refinement for a given length `n`.

    The LP maximises the ℓ¹‑mass of a feasible `g`.  Afterwards a
    one‑dimensional line search over the mixing parameter `t` is used to
    minimise the new worst autocorrelation value.
    """
    global _rng
    try:
        import cvxpy as cp
    except Exception:
        # No LP solver – signal failure.
        return None, float("inf")

    best_seq = None
    best_score = float("inf")

    # start from a random feasible vector
    f = _rng.random(n) + 0.1
    f = f / np.sum(f) * target_sum

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # worst autocorrelation value for current f
        M = float(np.max(np.convolve(f, f)))

        # linear convolution matrix built from f
        A = conv_matrix(f)

        # LP: maximise Σ g_i  s.t. A·g ≤ M,  g ≥ 0
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
            except Exception:
                break

        if g.value is None:
            break

        g0 = np.maximum(g.value, 0.0)
        s_g = np.sum(g0)
        if s_g < 1e-12:
            break
        # rescale to keep ℓ¹‑norm equal to target_sum
        g_rescaled = g0 * (target_sum / s_g)

        # pre‑compute the three convolutions needed for the quadratic model
        conv_ff = np.convolve(f, f)
        conv_fg = np.convolve(f, g_rescaled)
        conv_gg = np.convolve(g_rescaled, g_rescaled)

        # line‑search on t ∈ [0,1]
        t_candidates = np.linspace(0.0, 1.0, line_grid)
        best_t = 0.0
        best_M = np.max(conv_ff)  # value at t = 0
        for t in t_candidates[1:]:
            # (1‑t)²·ff + 2·t·(1‑t)·fg + t²·gg
            mixed = (1 - t) ** 2 * conv_ff + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_gg
            curM = np.max(mixed)
            if curM < best_M:
                best_M = curM
                best_t = t

        # update f
        f_new = (1.0 - best_t) * f + best_t * g_rescaled
        # (the sum stays exactly target_sum)
        f_new = np.clip(f_new, 0.0, 1000.0)      # safety, should never fire

        sc = _score(f_new.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = f_new.copy()
            print(f"[LP] n={n:3d} it={it:2d} t={best_t:.3f} score={sc:.6f}")

        # early stop if we already beat a strong bound
        if best_score < 1.45:
            break

        f = f_new

    return best_seq, best_score


def _conv_max_fft(seq: np.ndarray) -> float:
    """
    Fast computation of max_{k} (seq * seq)[k] using a real FFT.
    """
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    fft_seq = np.fft.rfft(seq, n=L)
    prod = fft_seq * fft_seq
    conv = np.fft.irfft(prod, n=L)[: 2 * n - 1]
    return float(np.max(conv))

        if delta < -1e-12 or _rng.random() < math.exp(-delta / (T + 1e-12)):
            current = cand
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = cand.copy()
                print(f"[anneal] it={iteration} score={_score(best.tolist()):.6f}")

        iteration += 1

    final_score = _score(best.tolist())
    return best, final_score


def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Main entry point used by the evaluation harness.

    The routine runs a multi‑size LP refinement (with line‑search mixing) and
    finishes with a simulated‑annealing polishing stage.  The total wall‑clock
    time is kept safely below 1000 s.
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # safety margin before the external timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # (a) reuse a previously discovered construction if it exists
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore
        overall_best_seq = init.copy()
        overall_best_score = _score(init.tolist())
        print(f"[init] reused height_sequence_1, score={overall_best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # (b) LP‑based refinement on several problem sizes
    # ------------------------------------------------------------------
    size_list = (200, 300, 350, 400, 500, 600, 800)
    for n in size_list:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        for start in range(2):               # two random starts per size
            if time.monotonic() >= deadline:
                break
            seq, sc = _lp_refine_with_line_search(
                n=n,
                target_sum=target,
                deadline=deadline,
                max_iters=50,
                line_grid=21,
            )
            if seq is not None and sc < overall_best_score:
                overall_best_seq = seq.copy()
                overall_best_score = sc
                print(f"[global] new best after LP n={n} start={start}, score={sc:.6f}")

            if overall_best_score < 1.45:    # early stop
                break
        if overall_best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # (c) fallback to a simple uniform start if nothing succeeded
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 300
        uniform = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        overall_best_seq = uniform.copy()
        overall_best_score = _score(uniform.tolist())
        print(f"[fallback] using uniform start, score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # (d) Simulated‑annealing polishing (uses remaining time)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        polished, polished_score = _annealing_local_search(overall_best_seq, deadline)
        if polished_score < overall_best_score:
            overall_best_seq = polished.copy()
            overall_best_score = polished_score
            print(f"[global] after annealing best score={polished_score:.6f}")

    # final safety clipping
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    return [float(x) for x in overall_best_seq]
