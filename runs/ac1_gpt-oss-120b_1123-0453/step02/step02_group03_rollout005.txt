def _hill_climb(seq, cur_score, deadline, rng):
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = cur_score
    start = time.monotonic()
    # use schedule
    while time.monotonic() < deadline:
        # choose source index from top 20% probability, else random
        if rng.random() < 0.8:
            # choose source from indices with large values
            idxs = np.argpartition(-seq, max(5, int(0.05*n)))[:max(5, int(0.05*n))]
            i = rng.choice(idxs)
        else:
            i = rng.integers(n)
        # choose target from small values
        if rng.random() < 0.8:
            idxs_small = np.argpartition(seq, int(0.05*n))[:max(5, int(0.05*n))]
            j = rng.choice(idxs_small)
        else:
            j = rng.integers(n)
        if i == j:
            continue
        # random proportion of mass from i
        max_transfer = seq[i]
        if max_transfer <= 0:
            continue
        # choose eps fraction in (0,0.3) times max_transfer
        eps = max_transfer * rng.random() * 0.5  # up to half
        cand = seq.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_score = _score(cand)
        # acceptance: accept if better or with prob based on temperature
        # temperature schedule linear: start high (0.3) decays.
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            temp = 0.0
        else:
            temp = 0.5 * (1 - elapsed/total) + 0.001
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[HC] new best {best_score:.6f}")
    return best_seq, best_score

"""
Search for a low‑value step‑function.

The algorithm runs a short‑budget simulated‑annealing (SA) loop that
actively attacks the current worst entry of the auto‑convolution.
For each iteration it

* computes the full convolution,
* picks the pair (i,j) that contributes most to the peak,
* moves a small amount of mass from the larger of the two indices to the
  smallest entry of the whole vector (this keeps the ℓ¹‑norm constant),
* accepts the move if it improves the score or with a temperature‑dependent
  probability (standard Metropolis criterion).

The SA loop is applied to several sequence lengths (200 … 1500).  The
remaining time budget is split automatically among the lengths, and the
best sequence seen over the whole run is returned.  If a previously found
construction `height_sequence_1` is present it is used as an additional
starting point.

The implementation is completely self‑contained, uses only NumPy and the
provided `evaluate_sequence` function, and respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Evaluate a candidate sequence via the supplied scorer."""
    # `evaluate_sequence` is injected by the competition harness.
    return float(evaluate_sequence(seq.tolist()))


def _targeted_sa(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing loop that always tries to reduce the current
    maximum convolution entry.  The move preserves the ℓ¹‑norm (mass
    is transferred between two coordinates).

    Parameters
    ----------
    seq : np.ndarray
        Current non‑negative sequence.
    cur_score : float
        Its current evaluation score.
    deadline : float
        Absolute time (monotonic) after which the loop must stop.
    rng : np.random.Generator
        Random source.

    Returns
    -------
    (best_seq, best_score)
        The best sequence and its score discovered during the run.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = cur_score
    start = time.monotonic()

    # pre‑allocate a buffer for the convolution to avoid repeated allocations
    while time.monotonic() < deadline:
        # --- 1. locate the current worst convolution entry -----------------
        conv = np.convolve(seq, seq)               # length 2·n‑1
        k_max = int(np.argmax(conv))               # index of the peak
        # identify the pair (i,j) that contributes most to this peak
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)

        # initialise with the first feasible pair
        i_best = i_start
        j_best = k_max - i_best
        max_prod = seq[i_best] * seq[j_best]

        # scan the interval only once – O(n) work
        for i in range(i_start, i_end + 1):
            j = k_max - i
            prod = seq[i] * seq[j]
            if prod > max_prod:
                max_prod = prod
                i_best, j_best = i, j

        # source = the larger of the two indices, target = global minimum
        src = i_best if seq[i_best] >= seq[j_best] else j_best
        tgt = int(np.argmin(seq))

        # avoid degenerate move
        if src == tgt:
            # pick a random different target
            tgt = int(rng.integers(n))
            if tgt == src:
                continue

        # amount to transfer – up to 10 % of the source entry
        delta = seq[src] * rng.random() * 0.10
        if delta <= 1e-12:
            continue

        cand = seq.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = _score(cand)

        # --- 2. acceptance (Metropolis) ------------------------------------
        elapsed = time.monotonic() - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 1.0
        # simple linear cooling from 0.5 down to 1e‑4
        temp = 0.5 * (1.0 - frac) + 1e-4

        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best {best_score:.6f}")

        # --- 3. occasional random reshuffle to escape plateaus -------------
        if rng.random() < 0.005:
            # choose three random indices and permute their values
            idxs = rng.choice(n, size=3, replace=False)
            vals = seq[idxs].copy()
            rng.shuffle(vals)
            cand2 = seq.copy()
            cand2[idxs] = vals
            cand2_score = _score(cand2)
            if cand2_score < cur_score or rng.random() < math.exp(
                -(cand2_score - cur_score) / max(temp, 1e-12)
            ):
                seq = cand2
                cur_score = cand2_score
                if cand2_score < best_score:
                    best_score = cand2_score
                    best_seq = cand2.copy()
                    print(f"[SA] shuffle improved to {best_score:.6f}")

    return best_seq, best_score


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random start vector with ℓ¹‑norm = √(2 n)."""
    target_sum = math.sqrt(2.0 * n)
    # start from a Dirichlet‑distributed vector (positive entries)
    v = rng.random(n) + 0.01            # avoid exact zeros
    v = v / np.sum(v) * target_sum
    return v

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    The routine runs several independent SA searches on different
    sequence lengths, always respecting the global time budget.
    The best sequence discovered is returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  If a historic construction exists, evaluate it.
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            seq0 = np.clip(seq0, 0.0, 1000.0)
            sc0 = _score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] height_sequence_1 present – score {sc0:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Define a schedule of lengths – larger n gives a lower
    #      theoretical bound but costs a bit more per iteration.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500]

    # Estimate how many lengths we can still explore
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # allocate a slice of the remaining time (roughly equal per length)
        remaining = deadline - time.monotonic()
        # keep at least 5 s for the final polishing step
        time_for_this = max(5.0, remaining / (len(lengths) - lengths.index(n) + 1))

        # ------------------------------------------------------------------
        # 2️⃣  Initialise a random candidate for this length
        # ------------------------------------------------------------------
        seq = _initial_sequence(n, rng)
        cur_score = _score(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[init n={n:4d}] score {cur_score:.6f}")

        # ------------------------------------------------------------------
        # 3️⃣  Run the targeted SA optimiser for the allocated slice
        # ------------------------------------------------------------------
        seq, cur_score = _targeted_sa(seq, cur_score,
                                      time.monotonic() + time_for_this, rng)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improved n={n:4d}] new best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 4️⃣  Use any remaining time to polish the overall best sequence.
    # ----------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        best_seq, best_score = _targeted_sa(
            best_seq, best_score, deadline, rng
        )
        print(f"[final] polished best score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Safety fallback – should never be hit
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(256, 1.0, dtype=np.float64)

    # Return a plain Python list (required by the harness)
    return [float(x) for x in best_seq]
