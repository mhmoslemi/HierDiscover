def auto_conv_max(a):
    n = len(a)
    m = 2*n - 1
    size = 1 << ((m-1).bit_length())
    # Compute FFT
    fft_a = np.fft.rfft(a, n=size)
    prod = fft_a*fft_a
    conv = np.fft.irfft(prod, n=size)[:m]
    return conv, conv.max()

"""
Hybrid search for a low‑value step sequence.

The algorithm works in three phases:

  1. **Exploratory sampling** –  for several lengths (300 … 1000) we
     generate simple parameterised families (Gaussian, cosine‑window,
     triangular) and a few uniform/random vectors.  Every candidate is
     normalised to Σa = √(2 n) and its score is evaluated using a fast
     FFT‑based convolution.

  2. **Local refinement** – the currently best vector is improved by a
     greedy “mass‑transfer” step.  At each iteration we locate the
     convolution index where the maximal value is attained, compute how
     much each coefficient contributes to those active indices and move a
     small amount of mass from the highest‑contributing coefficient to
     the lowest‑contributing one.  The step size is adapted automatically.
     The refinement stops when the time budget is exhausted or no
     improvement is possible.

  3. **Fallback** – if nothing was found (unlikely) a uniform vector is
     returned.

All helper functions are defined at top level, there are no closures or
lambda expressions, and only NumPy (and the supplied evaluate_sequence)
are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers -------------------------------------------------------
def _score_via_fft(seq: np.ndarray) -> float:
    """Return 2 n·max(conv(seq,seq))/Σseq² using an FFT (fast)."""
    n = seq.shape[0]
    # length of the full convolution
    m = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << ((m - 1).bit_length())
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[:m]
    M = float(conv.max())
    return 2.0 * n * M / (float(seq.sum()) ** 2)


def _auto_conv_max_fft(seq: np.ndarray):
    """Return the full convolution and its maximal entry (FFT version)."""
    n = seq.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[:m]
    return conv, float(conv.max())


def _contributions_to_active(a: np.ndarray, conv: np.ndarray, M: float,
                             eps: float = 1e-12) -> np.ndarray:
    """
    For every coefficient a[i] compute its contribution to the
    (approximately) active convolution entries, i.e. those where
    conv[k] ≈ M.
    """
    n = a.shape[0]
    active = np.where(conv >= M - eps)[0]
    contrib = np.zeros(n, dtype=np.float64)

    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        contrib[i_range] += a[j_range]
    return contrib


def _random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a uniform point on the simplex Σx = √(2 n), x_i ≥ 0."""
    target = math.sqrt(2.0 * n)
    # Dirichlet(1,…,1) is uniform on the simplex
    x = rng.random(n)
    x = x / x.sum() * target
    return x.astype(np.float64)

            right = int(centre) + half_width + 1

            right = min(right, n)
            pos = np.arange(left, right, dtype=np.float64)

            if s == 0:

            if M_cand < M - 1e-12:
                a = a_candidate
                step_factor = min(0.9, step_factor * 1.05)   # become a bit bolder
                continue
            else:
                # reject – shrink the step and try again with a random “low” index
                step_factor = max(0.01, step_factor * 0.7)
                idx_low2 = rng.integers(n)
                if idx_low2 == idx_high:
                    continue
                a_candidate2 = a.copy()
                a_candidate2[idx_high] -= delta
                a_candidate2[idx_low2] += delta
                conv2, M2 = _auto_conv_max_fft(a_candidate2)
                if M2 < M - 1e-12:
                    a = a_candidate2
                    step_factor = min(0.9, step_factor * 1.05)

    # ------------------------------------------------------------------
    # 3.   Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 400
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Return a plain Python list (the competition harness expects this)
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
