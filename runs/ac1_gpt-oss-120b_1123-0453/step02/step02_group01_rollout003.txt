def _lp_minimize_M(f, target_sum):
    n = f.shape[0]
    A = _conv_matrix(f)
    import cvxpy as cp
    g = cp.Variable(n)
    M = cp.Variable()
    constraints = [g >= 0, A @ g <= M, cp.sum(g) == target_sum]
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None or M.value is None: return None, None
    return np.maximum(g.value, 0.0), float(M.value)

"""
Improved search for a low‑value step sequence.

The algorithm starts from several random feasible vectors (or from the
previously known best vector if available) and iteratively solves a
linear program that *directly* minimizes the maximal cross‑convolution
value M = max(f∗g) for the current vector f while keeping the ℓ¹‑norm
fixed (∑f = √(2n)).  The LP variables are the new vector g and the scalar
M; the objective is minimise M.  After solving, the new vector g is
accepted if its own auto‑convolution peak is lower than the previous
peak; otherwise a small convex combination with the old vector is tried.
The process is repeated for several lengths and random restarts,
always keeping the best sequence seen.  If CVXPY is unavailable the
routine falls back to a simple random‑walk search.

The returned sequence is a plain Python list of floats, all non‑negative,
with its score equal to the maximal auto‑convolution value (because the
ℓ¹‑norm is normalised to √(2n), the evaluation routine returns exactly
that peak).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _evaluate(seq: np.ndarray) -> float:
    """
    Wrapper around the provided evaluate_sequence function.
    Returns the score (which, after normalisation, equals the peak of the
    auto‑convolution).
    """
    # evaluate_sequence is assumed to be available in the execution environment
    return float(evaluate_sequence(seq.tolist()))


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) == np.convolve(f, g).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _lp_step_minimize_M(f: np.ndarray, target_sum: float):
    """
    Solve the linear program

        minimise   M
        subject to  A @ g <= M
                    sum(g) == target_sum
                    g >= 0

    where A is the convolution matrix of the current vector f.
    Returns a tuple (g, M) where g is the optimal feasible vector and M
    the minimal bound.  Returns (None, None) if the solver fails.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None, None

    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n)
    M = cp.Variable()
    constraints = [g >= 0, A @ g <= M, cp.sum(g) == target_sum, M >= 0]
    prob = cp.Problem(cp.Minimize(M), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:  # pragma: no cover
            return None, None

    if g.value is None or M.value is None:
        return None, None

    g_val = np.maximum(g.value, 0.0)  # guard against tiny negatives
    return g_val, float(M.value)


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative vector of length n whose entries sum to
    target_sum (uniform on the simplex, with a tiny offset to avoid exact zeros).
    """
    x = rng.random(n) + 1e-3
    return x / np.sum(x) * target_sum

                # Stop early if we are very close to the theoretical lower bound.
                lower_bound = 2.0 * n / (2.0 * n - 1.0)   # = 1 + 1/(2n‑1)
                if M <= lower_bound * 1.005:
                    print(f"[near] n={n} it={it} score={M:.6f} lower={lower_bound:.6f}")
                    break

    # ----------------------------------------------------------------------
    # Fallback: simple random walk if nothing useful was found.
    if best_seq is None:
        fallback_budget = max(0.0, deadline - time.time())
        n = 300
        target = math.sqrt(2.0 * n)
        seq = np.full(n, target / n, dtype=np.float64)
        best_seq = seq.copy()
        best_score = _evaluate(seq)
        rng = np.random.default_rng(seed + 9999)
        end_time = time.time() + fallback_budget
        while time.time() < end_time:
            cand = seq + rng.normal(scale=0.03, size=n)
            cand = np.clip(cand, 0.0, None)
            if cand.sum() == 0:
                continue
            cand = cand * (target / cand.sum())
            sc = _evaluate(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                seq = cand
        print(f"[fallback] best_score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    result = [float(x) for x in best_seq]
    print(f"Best overall score: {best_score:.6f} (n={best_n})")
    return result
