def propose_candidate(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 2)

    best_seq = None
    best_score = float('inf')

    # length list
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline: break

        target_sum = math.sqrt(2.0 * n)
        max_iters = 30000  # may adjust based on time
        # maybe compute allowed time per n: remain_time = deadline - now, allocate proportionally

        restarts = 0
        while time.monotonic() < deadline and restarts < 10:
            restarts += 1

            # initialization: try three types: random uniform; triangular; from previous best expanded
            if best_seq is not None and best_seq.size < n and rng.random() < 0.5:
                a0 = expand_sequence(best_seq, n)
            else:
                # choose random shape
                shape_type = rng.integers(0,3)
                if shape_type == 0:
                    # uniform random Dirichlet
                    a0 = rng.random(n)
                elif shape_type == 1:
                    # triangular shape
                    # construct base of length n: a_i proportional to min(i+1, n-i)
                    idx = np.arange(n)
                    tri = np.minimum(idx+1, n-idx)
                    a0 = tri.astype(float)
                else:
                    # decreasing exponential
                    a0 = np.exp(-np.linspace(0, 5, n))
                a0 = a0 / a0.sum() * target_sum

            # do gradient descent
            a = a0.copy()
            # Pre-FFT size
            fft_size = 1 << ((2 * n - 1).bit_length())
            # schedule: step size schedule diminishing
            start = time.monotonic()
            it = 0
            while time.monotonic() < deadline and it < max_iters:
                it += 1
                # compute max conv
                conv = np.fft.irfft(np.fft.rfft(a, fft_size) * np.fft.rfft(a, fft_size), fft_size)[:2*n-1]
                conv = np.maximum(conv, 0.0)
                max_val = conv.max()
                k_star = int(conv.argmax())
                # compute gradient
                g = np.zeros_like(a)
                # compute slice indices
                i_start = max(0, k_star - (n - 1))
                i_end = min(k_star, n - 1)
                # i_range = np.arange(i_start, i_end+1)
                # g[i_range] = 2 * a[k_star - i_range]
                # vectorized
                i_range = np.arange(i_start, i_end+1)
                g[i_range] = 2 * a[k_star - i_range]

                # step size: e.g., eta = 0.5 * target_sum / (it + 10) ??? We'll tune.
                # We can compute based on gradient norm
                # Use diminishing step: eta = 1.0 / (it + 1)
                eta = 0.5 / (it**0.5 + 1)  # decreasing slowly
                # a_new = a - eta * g
                a_new = a - eta * g
                # project onto simplex
                a = project_onto_simplex(a_new, target_sum)

                # occasionally compute exact conv max for checking best
                if it % 500 == 0:
                    # compute max conv for a
                    conv_full = np.fft.irfft(np.fft.rfft(a, fft_size) * np.fft.rfft(a, fft_size), fft_size)[:2*n-1]
                    conv_full = np.maximum(conv_full, 0.0)
                    cur_max = conv_full.max()
                    if cur_max < best_score:
                        best_score = cur_max
                        best_seq = a.copy()
                        print(f"[GD] n={n} restart={restarts} iter={it} new best max={best_score:.6f}")
                # maybe early break if cur_max reaches target bound
                # also break if no change? not needed.

            # after gradient descent, final update best
            conv = np.fft.irfft(np.fft.rfft(a, fft_size) * np.fft.rfft(a, fft_size), fft_size)[:2*n-1]
            cur_max = np.max(conv)
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[GD-FINAL] n={n} restart={restarts} final max={cur_max:.6f}")

            # early stop if bound below target
            if best_score < 1.503:
                print("[STOP] reached target.")
                break

        # after length loop maybe early exit also
        if best_score < 1.503:
            break

    # fallback
    if best_seq is None:
        n=200
        target=math.sqrt(2*n)
        best_seq=np.full(n, target/n)
        best_score= max_conv_exact(best_seq)

    # convert to list

def simplex_projection(v, z):
    # v: array of floats
    # z: target sum (positive)
    # returns projection onto {x >= 0, sum x = z}
    n = v.shape[0]
    if z < 0:
        raise ValueError
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    # find rho, the last index where u_i > cssv_i / (i+1)
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

propose_candidate

A fast deterministic local‑search strategy for the step‑function problem.

* Initialise many random vectors (and a few hand‑crafted shapes) with the
  required ℓ¹‑norm  √(2·n).
* Repeatedly move a small amount of mass from an index that most

  maximal convolution entry) to an index that contributes the least.
  The move is kept if it reduces the maximal convolution (or with a tiny

  non‑negativity constraints are respected by construction.

`max_b = max(seq * seq)`.  With the ℓ¹‑norm fixed to √(2·n) the scorer

# Utility: projection onto the simplex (sum = z, non‑negative)

def simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=z} (Euclidean projection)."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    # find the largest rho satisfying u[i] - cssv[i]/(i+1) > 0
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast max‑convolution using FFT (exact up to floating‑point noise)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray, fft_size: int = None) -> float:
    """Return max_k (a * a)[k] using a real FFT."""
    n = a.size
    if fft_size is None:
        fft_size = 1 << ((2 * n - 1).bit_length())
    # real FFT, multiply pointwise, inverse FFT
    fhat = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fhat * fhat, fft_size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)               # clip tiny negatives
    return float(conv.max())


# ----------------------------------------------------------------------
# Gradient of the current worst convolution entry
# ----------------------------------------------------------------------


def max_conv_and_gradient(a: np.ndarray, fft_size: int):
    """
    Compute the maximal autocorrelation value, its index and the gradient
    of that entry w.r.t. a (∂/∂a_i conv_max).
    """
    n = a.size
    # convolution via FFT
    fhat = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fhat * fhat, fft_size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)
    max_val = float(conv.max())
    k_star = int(conv.argmax())

    # gradient: g_i = 2 * a_{k* - i} if the index is valid, else 0
    g = np.zeros_like(a)
    i_start = max(0, k_star - (n - 1))
    i_end = min(k_star, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k_star - idx]

    return max_val, k_star, g, conv


# ----------------------------------------------------------------------
# Simple deterministic improvement step (mass transfer)
# ----------------------------------------------------------------------


def improve_sequence(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Starting from a0 (already normalised to target_sum), repeatedly move a
    tiny amount of mass from a “bad’’ coordinate to a “good’’ one.
    Returns the best sequence seen and its maximal convolution.
    """
    n = a0.size
    fft_sz = 1 << ((2 * n - 1).bit_length())
    a = a0.copy()
    cur_max, _, cur_grad, _ = max_conv_and_gradient(a, fft_sz)

    best_a = a.copy()
    best_val = cur_max

    # small probability to accept a worsening move (helps escape plateaux)
    accept_worsening = 0.02

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # current worst index and its gradient
        cur_max, k_star, g, _ = max_conv_and_gradient(a, fft_sz)

        # select donor (largest gradient) and receiver (smallest gradient)
        i_out = int(np.argmax(g))
        # find a donor with a positive amount of mass
        if a[i_out] <= 1e-14:
            # fallback: pick the largest‑mass element
            i_out = int(np.argmax(a))

        i_in = int(np.argmin(g))
        if i_in == i_out:
            i_in = (i_in + 1) % n

        # amount to transfer – a small fraction of the average mass
        avg = target_sum / n
        max_transfer = min(a[i_out], avg * 0.5)
        if max_transfer <= 1e-12:
            # nothing to move – inject a tiny random jitter
            p, q = rng.integers(0, n, size=2)
            if p != q and a[p] > 1e-12:
                eps = min(a[p], 1e-3)
                a[p] -= eps
                a[q] += eps
            continue

        delta = rng.random() * max_transfer
        a_candidate = a.copy()
        a_candidate[i_out] -= delta
        a_candidate[i_in] += delta

        # evaluate the candidate
        cand_max, _, _, _ = max_conv_and_gradient(a_candidate, fft_sz)

        # accept if better, otherwise sometimes accept to stay explorative
        if cand_max < cur_max - 1e-12 or rng.random() < accept_worsening:
            a = a_candidate
            cur_max = cand_max
            if cand_max < best_val - 1e-12:
                best_val = cand_max
                best_a = a.copy()
                print(f"[improve] n={n} it={it} new best max={best_val:.6f}")

        # occasional random jitter to shake the process
        if it % 500 == 0:
            p, q = rng.integers(0, n, size=2)
            if p != q and a[p] > 1e-12:
                eps = min(a[p], 1e-3)
                a[p] -= eps
                a[q] += eps

    return best_a, best_val


# ----------------------------------------------------------------------
# Optional LP refinement (same as the reference code)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Toeplitz‑like matrix A such that (A @ g)[k] = (f * g)[k]."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        A[k, k - np.arange(i_start, i_end + 1)] = f[i_start : i_end + 1]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """One LP refinement step (maximise ℓ¹‑mass under the current max‑conv bound)."""
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv_fft(f)

    n = f.size
    M = max_conv_fft(f)

    A = build_conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    s = g_val.sum()
    if s < 1e-12:
        return f, M

    # rescale to the target ℓ¹‑norm and mix
    g_rescaled = g_val * (target_sum / s)
    t = 0.3
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    return f_new, max_conv_fft(f_new)


# ----------------------------------------------------------------------
# Helper to stretch a known good sequence to a longer length
# ----------------------------------------------------------------------


def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Stretch ``seq`` to length ``new_len`` (nearest‑index mapping) and renormalise."""
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    total = out.sum()
    if total > 0:
        target = math.sqrt(2.0 * new_len)
        out = out * (target / total)
    return out


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a step sequence with a low evaluation score.
    The routine respects the 1000 s limit (≈ budget_s‑2 s for safety)
    and returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Seed with a previously found construction if it exists.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.ndim == 1 and init.size > 0:
            n0 = init.size
            target0 = math.sqrt(2.0 * n0)
            if init.sum() > 0:
                init = init * (target0 / init.sum())
                sc0 = max_conv_fft(init)
                best_seq = init.copy()
                best_score = sc0
                print(f"[init] using height_sequence_1 (n={n0}) – score={sc0:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – larger n usually gives a better bound.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        max_restarts = 15                     # keep the total time reasonable
        restarts = 0

        while time.monotonic() < deadline and restarts < max_restarts:
            restarts += 1

            # --------------------------------------------------------------
            # Initialise a fresh candidate (several possibilities).
            # --------------------------------------------------------------
            if best_seq is not None and best_seq.size < n and rng.random() < 0.5:
                a0 = expand_sequence(best_seq, n)
            else:
                choice = rng.integers(0, 4)
                if choice == 0:                     # random Dirichlet
                    a0 = rng.random(n)
                elif choice == 1:                   # symmetric triangular shape
                    idx = np.arange(n)
                    a0 = np.minimum(idx + 1, n - idx)
                elif choice == 2:                   # exponential decay
                    a0 = np.exp(-np.linspace(0, 5, n))
                else:                               # a few equally‑spaced spikes
                    m_spikes = max(2, int(0.8 * math.sqrt(2 * n)))
                    spikes = rng.choice(n, size=m_spikes, replace=False)
                    a0 = np.zeros(n, dtype=np.float64)
                    a0[spikes] = 1.0
                a0 = a0 / a0.sum() * target_sum

            # --------------------------------------------------------------
            # Deterministic improvement phase
            # --------------------------------------------------------------
            a_opt, max_val = improve_sequence(
                a0,
                target_sum,
                deadline,
                rng,
                max_iters=3000,
            )

            # --------------------------------------------------------------
            # Optional tiny LP refinement (if cvxpy present)
            # --------------------------------------------------------------
            try:
                import cvxpy  # noqa: F401
                has_cvx = True
            except Exception:  # pragma: no cover
                has_cvx = False

            if has_cvx:
                for _ in range(3):
                    if time.monotonic() >= deadline:
                        break
                    a_opt, max_val = lp_refine(a_opt, target_sum, deadline)
                max_val = max_conv_fft(a_opt)   # exact after LP

            # --------------------------------------------------------------
            # Record global best
            # --------------------------------------------------------------
            if max_val < best_score - 1e-12:
                best_score = max_val
                best_seq = a_opt.copy()
                print(
                    f"[global] n={n} restart={restarts} new best score={best_score:.6f}"
                )
            if best_score < 1.503:                 # target reached → stop early
                print("[stop] reached target bound")
                break

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: uniform sequence (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = max_conv_fft(best_seq)
        print("[fallback] uniform sequence used, score=", best_score)

    # ------------------------------------------------------------------
    # Return a plain Python list (the recreation platform will call
    # evaluate_sequence on it).
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
