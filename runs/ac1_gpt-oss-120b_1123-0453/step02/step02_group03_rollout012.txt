Gradient‑based mass rebalancing for minimizing the convolution bound.
...

import time
import math
import numpy as np

def _score(seq: np.ndarray) -> float:
    return float(evaluate_sequence(seq.tolist()))

def _conv_fft(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    # real FFT, zero‑pad to size
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(F * F, n=size)[:m]
    # Numerical noise can lead to tiny negative values – clip
    return np.maximum(conv, 0.0)

def _pairwise_move(f: np.ndarray, k_max: int, step_factor: float,
                   rng: np.random.Generator) -> np.ndarray:
    """Transfer a small amount of mass away from the pair (i,j) that
    contributes most to the current maximal convolution entry."""
    n = f.shape[0]
    i_low = max(0, k_max - (n - 1))
    i_high = min(k_max, n - 1)
    if i_low > i_high:
        return f
    idx = np.arange(i_low, i_high + 1)
    jdx = k_max - idx
    # contribution of each ordered pair (i,j) to conv[k_max] is f[i]*f[j]
    contrib = f[idx] * f[jdx]
    total = np.sum(contrib)
    if total == 0.0:
        # nothing to move – pick a random index instead
        r = rng.integers(0, n)
        delta = step_factor * f[r]
        f[r] = max(f[r] - delta, 0.0)
        # place removed mass somewhere else
        t = rng.integers(0, n)
        while t == r:
            t = rng.integers(0, n)
        f[t] += delta
        return f

    # choose an index proportionally to its contribution
    probs = contrib / total
    i = rng.choice(idx, p=probs)
    j = k_max - i
    # amount we are willing to move – a fraction of the smaller entry
    if i == j:
        max_reduce = f[i]
        delta = step_factor * max_reduce
        delta = min(delta, max_reduce)
        # pick a receiver distinct from i
        r = rng.integers(0, n)
        while r == i:
            r = rng.integers(0, n)
        f[i] -= delta
        f[r] += delta
    else:
        max_reduce = min(f[i], f[j])
        delta = step_factor * max_reduce
        delta = min(delta, max_reduce)
        # pick a receiver distinct from i and j
        r = rng.integers(0, n)
        while r == i or r == j:
            r = rng.integers(0, n)
        f[i] -= delta
        f[j] -= delta
        f[r] += 2.0 * delta
    return f

def _gradient_step(f: np.ndarray, k_max: int, step: float) -> np.ndarray:
    """Zero‑sum projected gradient step that pushes mass away from the
    indices that directly participate in the maximal convolution entry."""
    n = f.shape[0]
    grad = np.zeros_like(f)
    # range of i such that j = k_max - i lies inside the vector
    i_low = max(0, k_max - (n - 1))
    i_high = min(k_max, n - 1)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        jdx = k_max - idx
        grad[idx] = 2.0 * f[jdx]
    # projection onto the hyper‑plane sum(d)=0
    mean_grad = grad.mean()
    direction = -grad + mean_grad
    f_new = f + step * direction
    return f_new

    best construction found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    def _optimise_length(n: int, init_seq: np.ndarray | None) -> tuple[np.ndarray, float]:
        S = math.sqrt(2.0 * n)                # required ℓ¹‑norm
        if init_seq is not None and init_seq.size == n:
            f = np.clip(init_seq.astype(np.float64), 0.0, 1000.0)
            # renormalise to the correct ℓ¹‑norm

            else:

            f = rng.random(n) + 0.1
            f = f / np.sum(f) * S

        def _shake(arr):
            pert = np.maximum(arr + rng.normal(scale=0.05, size=n), 0.0)
            return np.clip(pert, 0.0, 1000.0)

        best_local_score = cur_score

Gradient‑based mass rebalancing + lightweight simulated‑annealing.

* Work on several problem sizes (250 … 800).
* Start from the previously found construction `height_sequence_1`
  when its length matches the current size, otherwise from a random
  positive vector whose ℓ¹‑norm is √(2·n).
* For each size perform many cheap iterations:
    – locate the index `k_max` where the self‑convolution is maximal;
    – either take a projected gradient step that moves mass away from
      the indices participating in that peak, or
    – execute a stochastic “pairwise” transfer that removes a small
      fraction of the product `a_i·a_{k_max‑i}` and deposits it elsewhere.
    – after every change clip to [0,1000] and renormalise the ℓ¹‑norm.
    – periodically evaluate the true score (expensive) and keep the
      best sequence seen.
* When the time budget is almost exhausted run a short simulated‑annealing
  fine‑tuner (the same routine used in the reference solution) to polish
  the best sequence found.
* Return the best sequence as a plain list of floats.

The algorithm is deliberately simple, free of external LP solvers and
fits comfortably inside the 1000 s budget while often improving the
previous best bound (≈ 1.516) toward the target 1.503.
"""

import numpy as np

# Fast linear convolution via FFT (positive‑only input, no wrap‑around)
# ----------------------------------------------------------------------
def _conv_fft(f: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution f * f using a zero‑padded FFT.
    The result length is 2·len(f)‑1.
    """
    n = f.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(F * F, n=size)[:m]
    # Numerical noise can generate tiny negatives – clip them.
    return np.maximum(conv, 0.0)

        r = rng.integers(0, n)
        delta = factor * f[r]
        f[r] = max(f[r] - delta, 0.0)
        t = rng.integers(0, n)
        while t == r:
            t = rng.integers(0, n)
        f[t] += delta
        return f

    probs = contrib / total
    i = rng.choice(idx, p=probs)
    j = k_max - i

    if i == j:
        # centre element – move a fraction of it elsewhere
        max_reduce = f[i]
        delta = factor * max_reduce
        delta = min(delta, max_reduce)
        r = rng.integers(0, n)
        while r == i:
            r = rng.integers(0, n)
        f[i] -= delta
        f[r] += delta
    else:
        max_reduce = min(f[i], f[j])
        delta = factor * max_reduce
        delta = min(delta, max_reduce)
        r = rng.integers(0, n)
        while r == i or r == j:
            r = rng.integers(0, n)
        f[i] -= delta
        f[j] -= delta
        f[r] += 2.0 * delta
    return f


def _gradient_step(f: np.ndarray, k_max: int, step: float) -> np.ndarray:
    """
    Projected gradient step that preserves the ℓ¹‑norm.
    The gradient of the maximal convolution entry w.r.t. ``f`` is
    `2 * f[k_max - i]` (zero when the partner index is out of range).
    The step direction is adjusted to have zero sum.
    """
    n = f.shape[0]
    grad = np.zeros_like(f)

    i_low = max(0, k_max - (n - 1))
    i_high = min(k_max, n - 1)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        jdx = k_max - idx
        grad[idx] = 2.0 * f[jdx]

    mean_grad = grad.mean()
    direction = -grad + mean_grad          # sum(direction) == 0
    return f + step * direction


# ----------------------------------------------------------------------
# Simple simulated‑annealing (kept from the reference solution)
# ----------------------------------------------------------------------
def _simulated_annealing(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Perturb a single coordinate at a time with an exponential cooling schedule."""
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        temp = temp_start * (temp_end / temp_start) ** frac

        idx = rng.integers(len(cur_seq))
        delta = rng.normal(scale=0.5) * temp * 5.0
        cand = cur_seq.copy()
        cand[idx] = float(np.clip(cand[idx] + delta, 0.0, 1000.0))

        cand_score = _score(cand)

        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

        # tiny random shuffle to avoid being stuck
        if rng.random() < 0.001:
            pert = np.maximum(cur_seq + rng.normal(scale=0.1, size=len(cur_seq)), 0.0)
            cur_seq = np.clip(pert, 0.0, 1000.0)
            cur_score = _score(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Core optimisation for a single length
# ----------------------------------------------------------------------
def _optimise_length(
    n: int,
    init_seq: np.ndarray | None,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Run the gradient / pairwise rebalancing loop for a fixed length ``n``.
    Returns the best sequence encountered and its score.
    """
    S = math.sqrt(2.0 * n)               # required ℓ¹‑norm

    if init_seq is not None and init_seq.size == n:
        f = np.clip(init_seq.astype(np.float64), 0.0, 1000.0)
        cur_sum = np.sum(f)
        f = f * (S / cur_sum) if cur_sum > 0 else np.full(n, S / n)
    else:
        f = rng.random(n) + 0.1          # avoid exact zeros
        f = f / np.sum(f) * S

    def _shake(arr):
        pert = np.maximum(arr + rng.normal(scale=0.05, size=n), 0.0)
        return np.clip(pert, 0.0, 1000.0)

    cur_score = _score(f)
    best_local_seq = f.copy()
    best_local_score = cur_score

    max_iters = 12000
    eval_every = 250                     # call the expensive scorer this often
    step_start = 0.5 * S / n
    step_end = 0.02 * S / n

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # locate the current worst convolution entry
        conv = _conv_fft(f)
        k_max = int(np.argmax(conv))

        # decide which elementary move to apply
        if rng.random() < 0.7:            # mostly gradient steps
            frac = it / max_iters
            step = step_start * (1 - frac) + step_end * frac
            f = _gradient_step(f, k_max, step)
        else:                             # occasional pairwise transfer
            frac = it / max_iters
            factor = 0.4 * (1 - frac) + 0.02 * frac
            f = _pairwise_move(f, k_max, factor, rng)

        # keep within the allowed box and restore the exact ℓ¹‑norm
        f = np.clip(f, 0.0, 1000.0)
        cur_sum = np.sum(f)
        if cur_sum <= 0:
            f = np.full(n, S / n)
        else:
            f = f * (S / cur_sum)

        # occasional random shake to escape flat plateaux
        if it % 1500 == 0 and rng.random() < 0.2:
            f = _shake(f)

        # periodic true evaluation
        if it % eval_every == 0:
            cur_score = _score(f)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = f.copy()
                print(f"[len={n:4d} it={it:5d}] new local best {best_local_score:.6f}")

            if best_local_score < 1.503:   # target reached – stop early
                break

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
# Main entry point demanded by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    The routine explores several problem sizes, performs fast gradient‑based
    mass rebalancing, and finishes with a short simulated‑annealing polish.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Load the global construction if present
    # ------------------------------------------------------------------
    try:
        global_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
    except Exception:
        global_seq = None

    # ------------------------------------------------------------------
    # 2️⃣  Try a handful of lengths; larger lengths give a tighter theoretical bound
    # ------------------------------------------------------------------
    trial_lengths = [250, 300, 350, 400, 500, 600, 800]

    for n in trial_lengths:
        if time.monotonic() >= deadline:
            break

        init = global_seq if (global_seq is not None and global_seq.size == n) else None
        seq_n, score_n = _optimise_length(n, init, deadline, rng)

        if score_n < best_score:
            best_score = score_n
            best_seq = seq_n.copy()
            print(f"[global] best after length {n:4d} → {best_score:.6f}")

        if best_score < 1.503:            # early exit if target already met
            break

    # ------------------------------------------------------------------
    # 3️⃣  Light simulated‑annealing refinement on the overall best
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        best_seq, best_score = _simulated_annealing(best_seq, best_score, deadline, rng)
        print(f"[final SA] best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 256
        S0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, S0 / n0, dtype=np.float64)
        best_score = _score(best_seq)

    print(f"[final] best score {best_score:.6f} length {best_seq.shape[0]}")
    return [float(x) for x in best_seq]
