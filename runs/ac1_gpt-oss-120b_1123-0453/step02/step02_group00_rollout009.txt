def project_onto_simplex(v, target):
    # v may have negative entries; we need produce x >=0 and sum=x = target
    if target < 0:
        raise ValueError
    # shift by sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-indexed
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Edge cases: if sum(v) = target and v >=0, returns same.

def subgradient_descent(f, target_sum, deadline, max_iter=10000, step0=1.0, gamma=0.0, tolerance=1e-12):
    t=0; best_f = f.copy(); best_score = _score(f)
    while time.monotonic() < deadline and t < max_iter:
        conv = np.convolve(f, f)
        M = conv.max()
        # Use argmax set
        active_mask = np.isclose(conv, M, atol=1e-12)
        # If we wish to use smoothed gradient:
        # compute softmax weights: w = np.exp(gamma*(conv - M)); w = w/w.sum()
        # else use active

        if gamma > 0:
            # smoothed gradient
            exp_vals = np.exp(gamma * (conv - M))
            w = exp_vals / exp_vals.sum()
            # compute gradient as 2 * sum_k w_k * f_{k-i}
            # this is convolution of w and f_rev, similar to earlier but weighted
            grad = 2 * np.convolve(w, f[::-1])[len(f)-1:len(f)-1+len(f)]
        else:

            u = active_mask.astype(np.float64)
            grad = 2 * np.convolve(u, f[::-1])[len(f)-1:len(f)-1+len(f)]

        # step size schedule
        step = step0 / np.sqrt(t + 1)
        # update
        f_new = f - step * grad
        # project onto simplex
        f_new = project_onto_simplex(f_new, target_sum)
        # evaluate
        score = _score(f_new)   # compute actual eval
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        # Decide acceptance: we can accept if score improves else maybe still move.
        # Typically subgradient method ensures we move regardless.
        f = f_new
        t += 1
        # occasionally increase gamma
        if t % 500 == 0 and gamma < 20:

"""
Projected sub‑gradient descent on the simplex + final stochastic polishing.

* The evaluation score (with ℓ¹‑norm fixed to √(2·n)) equals the maximum
  entry of the self‑convolution.  We therefore minimise that peak.
* For a current vector `f` we compute the set of shifts attaining the
  maximum, build the sub‑gradient of the max‑operator and take a step in
  the opposite direction.  Afterwards we project back onto the simplex
  {x ≥ 0, Σx = √(2·n)}.
* The algorithm is run for a few lengths (including the length of a
  warm‑start sequence if one is available) with several random restarts.
* When the global time budget is almost exhausted a tiny Gaussian hill‑climber
  cleans up the best candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex with a given total mass.
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of `v` onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    # Work on a copy
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:                         # all entries will be zero
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Wrapper around the provided (black‑box) evaluation function.
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Convert a NumPy array to a Python list and evaluate it."""
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Very small stochastic hill‑climber used at the very end.
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """Gaussian perturbations accepted only when the score improves."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)          # respect the guard‑rails
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Core optimisation: projected sub‑gradient descent.
# ----------------------------------------------------------------------
def _subgradient_descent(f: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000,
                         step0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Minimise max(f*f) while staying on the simplex Σf = target_sum, f ≥ 0.
    Returns the best vector seen and its score.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_sc = _score(best_f)

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)               # length 2n‑1
        M = conv.max()
        # indices where the maximum is attained (allow tiny tolerance)
        active = conv >= M - 1e-12
        cnt_active = np.count_nonzero(active)
        if cnt_active == 0:                     # should not happen
            break
        # sub‑gradient: average over the active shifts
        u = active.astype(np.float64) / cnt_active
        # (u * f_rev)[i] = Σ_k u_k * f_{k-i}
        grad = 2.0 * np.convolve(u, f[::-1])[n - 1:n - 1 + n]

        # diminishing step size
        step = step0 / math.sqrt(it + 1.0)

        # gradient step + projection back onto the simplex
        f_new = f - step * grad
        f_new = project_onto_simplex(f_new, target_sum)

        sc = _score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
            # occasional log line (won’t flood the output)
            if it % 200 == 0:
                print(f"[sgd] it={it:4d} score={sc:.6f} max={M:.6f}")

        f = f_new

        # tiny random jitter every few hundred iterations to escape plateaux
        if (it + 1) % 500 == 0:
            jitter = np.random.default_rng().normal(scale=0.005, size=n)
            f = project_onto_simplex(f + jitter, target_sum)

    return best_f, best_sc


# ----------------------------------------------------------------------
# Public entry point required by the competition harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Strategy:

      3. Finish with a short stochastic hill‑climber.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")

        else:

        sc0 = _score(seq0)
        best_seq = seq0.copy()

    else:

        sc0 = _score(seq0)
        best_seq = seq0.copy()
        best_score = sc0
        print(f"[init] uniform start (n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – we always include the warm‑start length.
    # ------------------------------------------------------------------
    lengths = sorted({200, 300, 400, 500, 600, 800, n0})
    # keep the list reasonably small to stay within the time budget
    lengths = [L for L in lengths if L <= 1000]

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent on several lengths.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        restarts = 3 if n != n0 else 2          # fewer restarts when we already have a good start

            # initialise the vector

            # run the optimiser; step0 is tuned per size
            step0 = target_sum * 0.5                # initial step magnitude

                                                 deadline,
                                                 max_iter=max_iter,
                                                 step0=step0)

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = f_opt.copy()
                print(f"[sgd] new global best – n={n} restart={rs} score={sc_opt:.6f}")

            # early exit if we have already beaten the target bound
            if best_score < 1.503:
                print("[sgd] target 1.503 reached – exiting early.")
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # Phase 2 – final stochastic polishing (very cheap, uses the remaining time)
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety net – should never happen
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    if time.monotonic() < deadline:
        refined_seq, refined_sc = _local_search(best_seq, deadline, step_initial=0.5)

            best_seq = refined_seq
            best_score = refined_sc
            print(f"[local] final improvement – score={best_score:.6f}")

    # Return plain Python list of floats
    return [float(x) for x in best_seq]
