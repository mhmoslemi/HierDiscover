Implementation of projection onto simplex: see standard algorithm:

```
def project_simplex(v):
    # Ensure non-negative and sum to 1.
    # algorithm: sort v descending, compute cumulative sums, find rho
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑value step sequence.
The algorithm runs a projected gradient descent on the probability
vector p (∑ p = 1).  The objective approximates the ℓ∞‑norm of the
self‑convolution by a soft‑max (temperature schedule).  At each step
the gradient is obtained via an FFT‑based convolution, p is updated
and projected onto the simplex.  After the descent a short stochastic
hill‑climber refines the best candidate.  Multiple problem sizes and
random restarts are tried until the time budget expires.
"""

import time
import math
import numpy as np

def _score(seq):
    """Evaluate a sequence using the provided evaluate_sequence."""
    return float(evaluate_sequence(seq))

def _next_pow2(x):
    """Small utility to get the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _convolve_fft(a, b):
    """Convolution of real vectors a and b using FFT, returns real array."""
    size = len(a) + len(b) - 1
    L = _next_pow2(size)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = A * B
    conv = np.fft.irfft(C, n=L)
    return conv[:size]

def _self_convolve(p):
    """self‑convolution p * p (real, length 2·len(p)‑1)."""
    return _convolve_fft(p, p)

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_step(p, temperature):
    """
    Compute the soft‑max weighted gradient of max_{k}(p*p)[k].
    Returns gradient (same shape as p) and the current max of the convolution.
    """
    n = p.shape[0]
    # self‑convolution
    c = _self_convolve(p)
    max_c = c.max()
    # soft‑max weights (more emphasis on the current max)
    # To avoid overflow we shift by max_c.
    exp_terms = np.exp((c - max_c) / temperature)
    w = exp_terms / exp_terms.sum()
    # gradient = 2 * (w * reverse(p)) (convolution) sliced to length n
    p_rev = p[::-1]
    grad_full = _convolve_fft(w, p_rev)  # length = (2n-1)+n-1
    grad = 2.0 * grad_full[n-1:n-1+n]
    return grad, max_c

def _optimize_distribution(p0, n, deadline, rng,
                           max_iter=5000,
                           eta_start=0.1,
                           temp_start=0.02):
    """
    Projected gradient descent (soft‑max) on the simplex.
    Returns the best probability vector found and its score (2·n·max_c).
    """
    p = p0.copy()
    best_p = p.copy()
    # compute initial score
    c = _self_convolve(p)
    best_score = 2.0 * n * c.max()
    best_it = 0

    eta = eta_start
    temperature = temp_start

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        # gradient and current max
        grad, cur_max = _gradient_step(p, temperature)

        # gradient descent update with a tiny random perturbation
        p = p - eta * grad
        # add a little jitter to escape possible symmetric traps
        jitter = rng.normal(scale=eta * 0.01, size=n)
        p = p + jitter

        # project back onto simplex (norm = 1, non‑negative)
        p = _project_simplex(p)

        # evaluate current score rapidly (no call to evaluate_sequence)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            best_it = it
            print(f"[gd] n={n} it={it} score={best_score:.6f}")

        # simple schedules
        eta *= 0.9999
        temperature = max(1e-6, temperature * 0.9995)

    # final conversion to the usual scaling (sum = sqrt(2n))
    return best_p, best_score

def _hill_climb(p, target_sum, deadline, rng,
                max_moves=20000,
                step_frac=0.02,
                cooldown=0.9995):
    """
    Very light stochastic hill‑climber that keeps the ℓ¹ norm fixed.
    Moves a small amount of mass from a “heavy” index to a “light” one.
    """
    n = p.shape[0]
    best = p.copy()
    cur = p.copy()
    # pre‑scale to the required sum
    cur = cur * (target_sum / cur.sum())
    best = cur.copy()
    best_score = _score(best.tolist())
    cur_score = best_score

    temperature = 0.05
    step = max(1e-9, step_frac * target_sum / n)

    while time.monotonic() < deadline and max_moves > 0:
        max_moves -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue
        delta = min(cur[i] * 0.03, step)
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand.tolist())
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[hc] n={n} new best={best_score:.6f}")
        temperature = max(1e-6, temperature * cooldown)
        step = max(1e-12, step * cooldown)

    return best

    """

    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    for n in (250, 300, 350, 400, 450, 500):
        if time.monotonic() >= deadline:

        target_sum = math.sqrt(2.0 * n)

        # ---------- initial vector ----------
        if warm_start is not None and warm_start.size >= n:
            p0 = warm_start[:n].copy()

    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = A * B
    conv = np.fft.irfft(C, n=L)
    return conv[:size]

def _self_convolve(p: np.ndarray) -> np.ndarray:
    """Self‑convolution p * p (length 2·len(p)‑1)."""
    return _convolve_fft(p, p)

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]                      # descending
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _gradient_step(p: np.ndarray, temperature: float):
    """
    Compute gradient of the soft‑max approximation of max_k (p*p)[k].
    Returns:
        grad – vector same shape as p,
        max_c – current maximum entry of the self‑convolution.
    """
    n = p.shape[0]
    c = _self_convolve(p)
    max_c = c.max()
    # soft‑max weights; shift by max_c to keep numbers ≤ 1
    exp_terms = np.exp((c - max_c) / temperature)
    w = exp_terms / exp_terms.sum()
    # gradient = 2 * (w * reverse(p))  (convolution)
    p_rev = p[::-1]
    grad_full = _convolve_fft(w, p_rev)          # length = (2n‑1)+n‑1
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad, max_c

# ----------------------------------------------------------------------
def _optimize_distribution(p0: np.ndarray, n: int, deadline: float,
                           rng: np.random.Generator,
                           max_iter: int = 8000,
                           eta_start: float = 0.15,
                           temp_start: float = 0.025):
    """
    Projected gradient descent on the simplex.
    Returns the best probability vector found and its score (2·n·max_c).
    """
    p = p0.copy()
    best_p = p.copy()
    best_score = 2.0 * n * _self_convolve(p).max()

    eta = eta_start
    temperature = temp_start

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        grad, _ = _gradient_step(p, temperature)

        # gradient descent with a tiny random jitter to break symmetry
        p = p - eta * grad
        jitter = rng.normal(scale=eta * 0.01, size=n)
        p = p + jitter

        # keep p on the simplex
        p = _project_simplex(p)

        cur_score = 2.0 * n * _self_convolve(p).max()
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[gd] n={n} it={it} score={best_score:.6f}")

        # schedules – very slow decay (we have a lot of time)
        eta *= 0.9999
        temperature = max(1e-6, temperature * 0.9995)

    return best_p, best_score

# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray, target_sum: float, deadline: float,
                rng: np.random.Generator,
                max_moves: int = 15000,
                step_frac: float = 0.02,
                cooldown: float = 0.9995):
    """
    Light stochastic hill‑climber that keeps the ℓ¹‑norm fixed.
    Moves a small amount of mass from a “heavy” index to a “light” one.
    """
    n = seq.shape[0]
    # ensure the required ℓ¹‑norm
    cur = seq * (target_sum / seq.sum())
    best = cur.copy()
    # score = 2·n·max_b / (sum²)  (sum = target_sum)
    best_score = 2.0 * n * _self_convolve(cur).max() / (target_sum ** 2)
    cur_score = best_score

    temperature = 0.05
    step = max(1e-9, step_frac * target_sum / n)

    while time.monotonic() < deadline and max_moves > 0:
        max_moves -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue
        delta = min(cur[i] * 0.03, step)
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = 2.0 * n * _self_convolve(cand).max() / (target_sum ** 2)

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[hc] n={n} new best={best_score:.6f}")

        temperature = max(1e-6, temperature * cooldown)
        step = max(1e-12, step * cooldown)

    return best

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score.
    Uses gradient‑descent on the simplex followed by a tiny hill‑climber.
    Several values of n are tried; the best sequence seen before the
    deadline is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # optional warm‑start from a previous construction
    warm_start = None
    if "height_sequence_1" in globals():
        warm_start = np.array(list(height_sequence_1), dtype=np.float64)
        if warm_start.sum() > 0:
            warm_start = warm_start / warm_start.sum()

    # ------------------------------------------------------------------
    # try a few problem sizes; larger n usually offers a better bound
    for n in (250, 300, 350, 400, 450, 500):
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # ----- initialise a probability vector -----
        if warm_start is not None and warm_start.size >= n:
            p0 = warm_start[:n].copy()
            p0 = p0 / p0.sum()
        else:
            p0 = rng.random(n) + 0.01               # avoid exact zeros
            p0 = p0 / p0.sum()

        # ----- gradient‑descent phase -----
        p_opt, gd_score = _optimize_distribution(p0, n, deadline, rng,
                                                max_iter=8000,
                                                eta_start=0.15,
                                                temp_start=0.025)

        # scale to the conventional ℓ¹‑norm (sqrt(2n)) before hill‑climb
        seq_scaled = p_opt * (target_sum / p_opt.sum())

        # ----- hill‑climb refinement -----
        seq_refined = _hill_climb(seq_scaled, target_sum, deadline, rng,
                                  max_moves=15000)

        final_score = _score(seq_refined.tolist())
        if final_score < best_score:
            best_score, best_seq = final_score, seq_refined.copy()
            print(f"[global] n={n} improved best score = {best_score:.6f}")

        if best_score < 1.503:         # target reached – stop early
            break

    # ------------------------------------------------------------------
    # fallback: a uniform sequence (always feasible)
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq.tolist())
        print("[fallback] uniform sequence used, score =", best_score)

    # return a plain Python list of floats
    return [float(x) for x in best_seq]
