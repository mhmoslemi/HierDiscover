def proj_simplex(v):
    # Projects v onto the probability simplex {x >= 0, sum x = 1}
    # v is a numpy array.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function sequence with a low  evaluate_sequence  score.

The algorithm works on the *probability* vector p (∑ p_i = 1) because the
objective is scale‑invariant:
    score = 2·n·max_k (p * p)[k] .
Hence we minimise the ℓ∞‑norm of the auto‑convolution of p.

Method
------
1. Choose a problem size n (default 2000, large enough to approach the
   theoretical lower bound 2n/(2n‑1) ≈ 1).
2. Initialise p either from a warm start (``height_sequence_1`` if it has
   the same length) or from a random simplex point.
3. Perform many *projected sub‑gradient* steps.
   * Compute the self‑convolution c = p * p.
   * Let M = max c and let K be the set of indices where c attains M.
   * The sub‑gradient of the objective at p is
         g_i = Σ_{k∈K} p_{k‑i}   (only when the index is valid).
   * Update   p ← p – η·g   with a diminishing step  η = 1/√t .
   * Project the result back onto the probability simplex.
   * Occasionally add a tiny random perturbation to escape flat plateaus.
4. Keep the best vector seen (lowest score) and, when time permits,
   apply a few cheap stochastic moves (swap a small mass from a high‑
   gradient index to a low‑gradient one) – essentially a lightweight
   hill‑climber.
5. Rescale the best probability vector to the required ℓ¹‑norm
   √(2 n) (the evaluate_sequence function normalises internally) and
   clip the values to the allowed range [0, 1000].

The whole search respects the external time budget (default 1000 s) and
returns a plain ``list[float]`` suitable for the evaluator.
"""

import time
import math
import numpy as np


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → uniform point
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _self_convolution(p: np.ndarray, fft_size: int) -> np.ndarray:
    """Return p * p using a real FFT (fast for n up to a few thousand)."""
    fft = np.fft.rfft(p, n=fft_size)
    conv = np.fft.irfft(fft * fft, n=fft_size)
    # keep the first 2·n‑1 entries – the true convolution length
    return conv[: 2 * p.shape[0] - 1]


def _subgradient(p: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """Compute a sub‑gradient of  max(conv)  w.r.t. the probability vector p."""
    n = p.shape[0]
    # indices where the convolution attains its maximum (within a tiny tolerance)
    tol = 1e-14
    ks = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(p)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        g[i_range] += p[j_range]
    return g


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **_kwargs):
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence(...).  The function respects the given time budget
    and returns a Python list of floats (the step heights).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 0.2)

    # ------------------------------------------------------------------
    # Choose a problem size.  Larger n gives a lower theoretical bound,
    # but we must stay fast enough.  2000 works well on the available
    # hardware within the time limit.
    n = 2000
    # ------------------------------------------------------------------
    # Initialise a probability vector p (∑p = 1).
    if "height_sequence_1" in globals():
        hs = np.array(list(height_sequence_1), dtype=np.float64)
        if hs.size == n:
            p = hs / max(hs.sum(), 1e-12)
            # add a little noise to avoid starting exactly at a stationary point
            p += rng.normal(scale=0.01, size=n)
            p = _proj_simplex(p)
        else:
            p = rng.random(n)
            p = p / p.sum()
    else:
        p = rng.random(n)
        p = p / p.sum()

    # keep the best point seen so far
    best_p = p.copy()
    # initial score (2·n·max(p*p))
    conv = np.convolve(p, p)
    best_score = 2.0 * n * conv.max()

    # pre‑compute an FFT length that is a power of two
    fft_len = 1 << ((2 * n - 1).bit_length())

    it = 0
    no_improve = 0          # counter for stagnation detection

    # ------------------------------------------------------------------
    # MAIN PROJECTED SUB‑GRADIENT LOOP
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # self‑convolution (fast)
        conv = _self_convolution(p, fft_len)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score:
            best_score = score
            best_p = p.copy()
            no_improve = 0
            print(f"[best] iter={it:5d} n={n} score={score:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of the ℓ∞‑norm of the convolution
        g = _subgradient(p, conv, M)

        # diminishing step size η_t = 1/√t  (t = iteration count)
        eta = 1.0 / math.sqrt(it)

        # gradient step followed by projection onto the simplex
        p = _proj_simplex(p - eta * g)

        # occasional random perturbation to jump off flat plateaus
        if rng.random() < 0.001:
            p = _proj_simplex(p + rng.normal(scale=0.02, size=n))

        # simple restart after a long stall (helps escape local minima)
        if no_improve > 2000 and (deadline - time.monotonic()) > 5.0:
            p = rng.random(n)
            p = p / p.sum()
            no_improve = 0

        # safety break – we never need more than a few 10⁴ iterations
        if it >= 40000:
            break

    # ------------------------------------------------------------------
    # LIGHTHEART HILL‑CLIMB REFINEMENT (optional, cheap)
    # ------------------------------------------------------------------
    temperature = 0.02
    for hc in range(3000):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(best_p, best_p)
        M = conv.max()
        cur_score = 2.0 * n * M

        # reuse the sub‑gradient as a cheap guide
        g = _subgradient(best_p, conv, M)
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if best_p[i_max] <= 0:
            continue

        delta = min(best_p[i_max] * 0.05, 0.001)  # move a tiny amount of mass
        cand = best_p.copy()
        cand[i_max] -= delta
        cand[i_min] += delta

        # evaluate candidate
        conv_c = np.convolve(cand, cand)
        cand_score = 2.0 * n * conv_c.max()

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            best_p = cand
            if cand_score < best_score:
                best_score = cand_score
                print(f"[hill] iter={hc:4d} n={n} score={cand_score:.6f}")
            temperature *= 0.9995
        else:
            temperature *= 0.9999

    # ------------------------------------------------------------------
    # FINAL RESCALING TO THE REQUIRED ℓ¹‑NORM  √(2 n)
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    seq = best_p * target_sum
    # enforce the hard caps from evaluate_sequence
    seq = np.clip(seq, 0.0, 1000.0)

    # if something went terribly wrong fall back to a uniform construction
    if np.isnan(seq).any() or np.isinf(seq).any() or seq.sum() < 0.01:
        print("[fallback] uniform construction used")
        seq = np.full(n, target_sum / n)

    return [float(x) for x in seq]
