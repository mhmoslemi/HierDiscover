The routine builds on the intuition that the score

is minimized when the self‑convolution f*f is as flat as possible.
Starting from an existing candidate (or a cheap Hann‑window) we repeatedly
move a tiny amount of ℓ¹‑mass from indices that contribute heavily to the
current maximal convolution entry to indices that contribute little.
The move strictly reduces the peak of the convolution unless a new peak
appears elsewhere – in this case a simple simulated‑annealing acceptance
rule is used.

For several problem sizes (n≈200…1200) the process is run until the time
budget (≈ 1000 s) is exhausted.  The best sequence seen over all lengths is
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: evaluate the objective locally (identical to evaluate_sequence
# without the security checks).  Using the local version avoids the
# overhead of the scorer and makes the search fast.
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    # fast convolution – numpy's implementation is C‑level and very fast for
    # the moderate sizes we use (≤ 1200).
    conv = np.convolve(seq, seq)
    max_conv = np.max(conv)
    s = np.sum(seq)
    # the scorer returns np.inf for an almost‑zero sum; we mimic that.
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Helper: generate a simple Hann window (non‑negative, zero at the ends).
# The shape is a good starting point because its autocorrelation is fairly
# flat compared with the uniform distribution.
# ----------------------------------------------------------------------
def _hann_window(n: int) -> np.ndarray:
    """Return a Hann window of length n with ℓ¹‑norm = 1."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * math.pi * i / (n - 1)))
    # normalise to unit ℓ¹‑norm – scaling does not affect the score.
    w /= np.sum(w)
    return w


# ----------------------------------------------------------------------
# Core optimisation step for a single length.
# ----------------------------------------------------------------------
def _optimise_length(init_seq: np.ndarray,
                     n: int,
                     rng: np.random.Generator,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Take ``init_seq`` (any length) and optimise a sequence of length ``n``.
    Returns the best sequence found for this length and its score.
    """
    # embed the initial sequence (centre‑align) and zero‑pad to length n
    if init_seq.size > n:
        f = init_seq[:n].copy()
    else:
        f = np.zeros(n, dtype=np.float64)
        start = (n - init_seq.size) // 2
        f[start:start + init_seq.size] = init_seq

    # a tiny random jitter avoids pathological symmetries
    f += rng.normal(scale=1e-4, size=n)
    f = np.clip(f, 0.0, 1000.0)

    # guarantee a non‑zero sum
    if np.sum(f) < 0.01:
        f += 1.0
    best_f = f.copy()
    best_score = _objective(best_f)

    # parameters for the mass‑shifting loop
    max_iters = 3000                     # generous upper bound; broken by time
    step_factor = 0.5                    # fraction of source mass moved each step
    step_decay = 0.9995                  # slowly shrink the step
    temp_initial = 1e-3                  # annealing temperature
    temp_decay = 0.9999

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # current convolution and its worst index
        conv = np.convolve(f, f)
        max_val = np.max(conv)
        k_max = int(np.argmax(conv))

        # compute the contribution weights for each index toward conv[k_max]
        # weight[i] = f[k_max - i] when the partner index exists, else 0.
        idx = np.arange(n, dtype=np.int64)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        weights = np.zeros_like(f)
        weights[mask] = f[partner[mask]]

        # source = index with largest weight, destination = smallest weight
        src = int(np.argmax(weights))
        dst = int(np.argmin(weights))

        # if the source carries virtually no mass, we are stuck – break.
        if f[src] <= 1e-12:
            break

        # amount of mass to move (never exceed the whole source entry)
        delta = step_factor * f[src]
        if delta <= 1e-12:
            break

        # perform the tentative move
        f_new = f.copy()
        f_new[src] -= delta
        f_new[dst] += delta

        # safeguard against negative entries caused by floating‑point noise
        if f_new[src] < 0.0:
            f_new[dst] += f_new[src]          # remove the small negative part
            f_new[src] = 0.0

        # evaluate
        new_score = _objective(f_new)

        # deterministic improvement or occasional uphill move (simulated annealing)
        improve = new_score < best_score
        if improve:
            f = f_new
            best_f = f_new.copy()
            best_score = new_score
            # progress output
            print(f"[len {n:4d}] it {it:4d} score {best_score:.6f}")
        else:
            # temperature decays slowly; acceptance probability = exp(-(Δ)/T)
            temperature = temp_initial * (temp_decay ** it)
            prob = math.exp(-(new_score - best_score) / (temperature + 1e-12))
            if rng.random() < prob:
                f = f_new

        # decay the step size – allows fine adjustments later
        step_factor = max(0.01, step_factor * step_decay)

        # early exit if we have already crossed the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Small random‑perturbation polishing phase (adds Gaussian noise).
    # ------------------------------------------------------------------
    for _ in range(200):
        if time.monotonic() >= deadline:
            break
        cand = best_f + rng.normal(scale=0.005, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _objective(cand)
        if sc < best_score:
            best_f = cand.copy()
            best_score = sc
            print(f"[polish len {n}] score {best_score:.6f}")
            if best_score < 1.503:
                break

    return best_f, best_score


# ----------------------------------------------------------------------
# Simple fallback random walk (used only if time remains and nothing else
# produced a result).
# ----------------------------------------------------------------------
def _fallback_random_search(budget_s: float, seed: int = 0) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    n = 300
    seq = np.full(n, 1.0, dtype=np.float64)
    best_seq = seq.copy()
    best_score = _objective(seq)

    while time.monotonic() < deadline:
        cand = seq + rng.normal(scale=0.02, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _objective(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            seq = cand
            print(f"[fallback] new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging system.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.

    The method tries several sequence lengths, starts from the globally
    supplied ``height_sequence_1`` (or a Hann window if unavailable) and
    applies a focused mass‑shifting optimisation that directly attacks the
    maximal entry of the self‑convolution.  A short random polishing phase
    follows each length.  The best sequence seen before the time budget
    expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Acquire a starting sequence from the global variable if it exists.
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        if init_seq.ndim != 1:
            raise ValueError
        # make sure the sequence is non‑negative
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        print("[init] using provided height_sequence_1")
    except Exception:
        # fall back to a Hann window of moderate size
        init_seq = _hann_window(200)
        print("[init] using generated Hann window")

    # overall best holder
    best_overall_seq = init_seq.copy()
    best_overall_score = _objective(best_overall_seq)

    # ------------------------------------------------------------------
    # Sweep over a handful of lengths; longer lengths give the algorithm
    # more freedom but also cost more time per iteration.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        seq_n, score_n = _optimise_length(init_seq, n, rng, deadline)

        if score_n < best_overall_score:
            best_overall_seq = seq_n.copy()
            best_overall_score = score_n
            print(f"[global] new best with n={n} score={best_overall_score:.6f}")

        if best_overall_score < 1.503:
            print("[goal] reached target threshold, exiting early")
            break

    # ------------------------------------------------------------------
    # If nothing decent was found (unlikely) run the generic fallback.
    # ------------------------------------------------------------------
    if best_overall_seq is None or best_overall_score == float('inf'):
        remaining = max(0.0, float(budget_s) - (time.monotonic() - start_time))
        fallback_seq, fallback_score = _fallback_random_search(remaining, seed=seed)
        best_overall_seq = fallback_seq
        best_overall_score = fallback_score

    # final safety: convert to plain Python list of floats
    return [float(x) for x in best_overall_seq]
