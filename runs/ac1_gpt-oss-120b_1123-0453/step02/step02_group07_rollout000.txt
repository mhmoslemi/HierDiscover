def pgd_optimize(a_init, target_sum, deadline):
    a = a_init.copy()
    n = len(a)
    best_score = max_conv(a)  # compute via convolution
    best_a = a.copy()
    # schedule
    alpha = 2.0
    while time.time() < deadline:
        # compute conv with FFT
        # Use current alpha
        conv = compute_conv(a)  # length 2n-1
        maxconv = np.max(conv)
        # compute smooth objective log-sum-exp
        # Use w
        shift = conv.max()
        exp_vals = np.exp(alpha * (conv - shift))
        sumexp = exp_vals.sum()
        w = exp_vals / sumexp
        # gradient
        grad = compute_gradient(w, a)
        # step size adaptation
        # we can use fixed step decreasing.
        step = 0.2 * target_sum / n  # maybe adapt
        # line search:
        for _ in range(10):
            a_new = a - step * grad
            # project onto simplex
            a_new = project_simplex(a_new, target_sum)
            # compute new max conv
            conv_new = compute_conv(a_new)
            max_new = conv_new.max()
            if max_new <= maxconv or step < 1e-12:
                a = a_new
                break
            step *= 0.5
        # Optionally increase alpha as we converge
        if maxconv - max_new < 1e-6: # no improvement
            alpha = min(alpha * 1.2, 200)
        else:
            # keep same
            pass
        # track best
        if max_new < best_score:
            best_score = max_new
            best_a = a.copy()
            print(f"[opt n={n}] new best maxconv {best_score:.6f}")
        # termination condition based on improvement count
        # Might break after many iterations w/o improvement
        if time.time() >= deadline:

Compute convolution using FFT:

def compute_conv(a):
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    # compute FFT of size L
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    return conv

def _project_simplex(v, s):
    """Euclidean projection of vector v onto simplex {x >=0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be nonnegative")
    v = np.asarray(v, dtype=np.float64)
    # shift by threshold
    # similar algorithm from Duchi
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho = np.where(u > cssv / np.arange(1, len(v)+1))[0]
    if len(rho) == 0:
        tau = 0.0
    else:
        rho = rho[-1]
        tau = cssv[rho] / (rho + 1)
    w = np.maximum(v - tau, 0.0)
    return w

"""
Propose a low‑value step sequence.

The algorithm searches for a probability vector (scaled to the canonical
ℓ¹‑norm √(2n)) that minimises the peak of its self‑convolution.
It combines three ideas:

1.  Several deterministic / random initialisations (uniform,
    U‑shaped β(½,½), triangular, Dirichlet‑like, pure random).
2.  A smooth “soft‑max” gradient descent on the objective
    f(a)= (1/α)·log∑_k exp(α·(a*a)_k) that approximates the
    max‑convolution.  The gradient is obtained by a single FFT‑based
    convolution, the iterate is projected onto the non‑negative simplex
    (fixed ℓ¹‑norm) and the smoothing parameter α is increased
    gradually (continuation method).
3.  A short stochastic hill‑climber that moves a small amount of mass
    from indices that contribute heavily to the current worst convolution
    entry to indices that contribute little, accepting occasional worsening
    moves with a slowly decreasing temperature (simulated‑annealing).

All steps respect the time budget (default 1000 s).  The best sequence
found for any length is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    v = np.asarray(v, dtype=np.float64)

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # Find the largest index rho where u_i > cssv_i / (i+1)
    rho_arr = np.where(u > cssv / np.arange(1, len(v) + 1))[0]
    if rho_arr.size == 0:
        tau = 0.0
    else:
        rho = rho_arr[-1]
        tau = cssv[rho] / (rho + 1)
    w = np.maximum(v - tau, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using an FFT (real‐valued)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _gradient_wrt_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of ϕ(a)=∑_k w_k·(a*a)_k  w.r.t. a.
    It equals 2· (w * a_rev)[i+n‑1] for i=0…n‑1.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    K = _next_pow2((2 * n - 1) + n - 1)
    fw = np.fft.rfft(w, n=K)
    fa_rev = np.fft.rfft(a_rev, n=K)
    conv = np.fft.irfft(fw * fa_rev, n=K)
    grad = 2.0 * conv[(n - 1) : (n - 1 + n)]
    return grad


# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, alpha: float) -> tuple[np.ndarray, float]:
    """
    Returns the soft‑max weights w and the corresponding smooth
    approximation of max(conv):  (1/α)·log ∑ exp(α·conv).
    """
    # shift for numerical stability
    shift = np.max(conv)
    exp_vals = np.exp(alpha * (conv - shift))
    sumexp = np.sum(exp_vals)
    w = exp_vals / sumexp
    smooth = (np.log(sumexp) + alpha * shift) / alpha
    return w, smooth


# ----------------------------------------------------------------------
def _optimize_distribution(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑descent on the smooth max‑convolution, followed by a
    stochastic hill‑climber.  Returns the best vector found and its
    (exact) max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = np.max(_conv_self_fft(best_a))

    # ----------- smooth‑gradient phase ---------------------------------
    alpha = 2.0               # start with a very smooth approximation
    step = 0.25 * target_sum / n
    max_iters = 3000

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self_fft(a)
        maxconv = np.max(conv)

        if maxconv < best_val - 1e-12:
            best_val = maxconv
            best_a = a.copy()
            print(f"[gd] n={n} it={it} maxconv={best_val:.6f}")

        # compute soft‑max weights
        w, _ = _smooth_max(conv, alpha)

        # gradient of the smooth objective
        grad = _gradient_wrt_a(w, a)

        # simple backtracking line search (non‑increase guarantee)
        step_cur = step
        for _ in range(12):
            cand = a - step_cur * grad
            cand = _project_simplex(cand, target_sum)
            cand_max = np.max(_conv_self_fft(cand))
            if cand_max <= maxconv:
                a = cand
                break
            step_cur *= 0.5
        # shrink the step size slowly
        step = max(step * 0.997, 1e-12)

        # increase smoothing parameter every few hundred iterations
        if (it + 1) % 200 == 0 and alpha < 200:
            alpha = min(alpha * 1.7, 200)

    # ----------- stochastic hill‑climber --------------------------------
    cur = best_a.copy()
    cur_val = best_val
    temperature = 0.08
    step_hc = 0.02 * target_sum / n

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0:
            continue

        delta = min(cur[i] * 0.04, step_hc)
        if delta <= 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum is already preserved, just clamp small negatives
        cand[i] = max(cand[i], 0.0)

        cand_val = np.max(_conv_self_fft(cand))

        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(cand_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_a = cand_val, cand.copy()
                print(f"[hc] n={n} new best={best_val:.6f}")

        temperature = max(1e-6, temperature * 0.9995)
        step_hc = max(1e-12, step_hc * 0.9997)

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence whose evaluation score
    (2·n·max(f*f) / (∑f)²) is as small as possible.
    The routine respects the supplied time budget and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously found global variable, if present.
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = init.size
        target0 = math.sqrt(2.0 * n0)
        init = init * (target0 / max(init.sum(), 1e-12))
        init = np.clip(init, 0.0, 1000.0)
        sc = np.max(_conv_self_fft(init))          # exact score (sum fixed)
        best_seq, best_score = init, sc
        print("[init] height_sequence_1 used, score =", best_score)

    # ------------------------------------------------------------------
    # Try a handful of problem sizes (larger n usually yields a lower bound).
    for n in (200, 250, 300, 350, 400, 450, 500):
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # A few distinct initialisers for the same n
        for init_id in range(5):
            if time.time() >= deadline:
                break

            if init_id == 0:
                a0 = np.full(n, target_sum / n)                     # uniform
            elif init_id == 1:
                # U‑shaped β(½,½) – heavy on the ends
                idx = np.arange(1, n + 1)
                a0 = 1.0 / np.sqrt(idx * (n + 1 - idx))
                a0 = a0 / a0.sum() * target_sum
            elif init_id == 2:
                # Simple triangle (peak in the middle)
                idx = np.arange(n)
                a0 = np.minimum(idx + 1, n - idx)
                a0 = a0 / a0.sum() * target_sum
            elif init_id == 3:
                # Dirichlet‑like (α = 0.3) – very spiky
                a0 = rng.gamma(0.3, 1.0, size=n)
                a0 = a0 / a0.sum() * target_sum
            else:
                # Pure random positive vector
                a0 = rng.random(n) + 0.1
                a0 = a0 / a0.sum() * target_sum

            # Run the optimiser
            a_opt, score_opt = _optimize_distribution(a0, target_sum, deadline, rng)

            if score_opt < best_score:
                best_seq, best_score = a_opt.copy(), score_opt
                print(f"[global] n={n} init={init_id} → new best maxconv={best_score:.6f}")

            # Early exit if we already beat the target bound
            if best_score <= 1.503:
                break
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – a plain uniform sequence (always feasible)
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = np.max(_conv_self_fft(best_seq))
        print("[fallback] uniform sequence used, score =", best_score)

    # Final safety – clip to allowed range and ensure positivity
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # The evaluator is scale‑invariant; we keep the canonical ℓ¹‑norm.
    return [float(v) for v in best_seq]
