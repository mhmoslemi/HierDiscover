Propose a low‑value step sequence.

Algorithm
---------
1.  **Iterated LP‑refinement**
    * For a chosen length `n` keep the ℓ¹‑norm fixed to
      `S = sqrt(2·n)`.  For a current vector `f` compute
      `M = max(f * f)`.
    * Solve the linear program
            maximise   Σ g_i
            subject to (f * g) ≤ M ,  g ≥ 0
      (the convolution matrix of `f` is built explicitly).
      The LP is solved with `scipy.optimize.linprog` (fallback to
      `cvxpy` if SciPy is unavailable).
    * Rescale `g` to the same ℓ¹‑norm `S` and mix it with `f`
            f ← (1‑t)·f + t·g        (t ≈ 0.6, slowly decayed)
      The mix is renormalised to keep the exact sum `S`.
    * Repeat until the improvement stalls or the time budget expires.
    * The procedure is performed for several lengths
      (`200, 400, 600, 800, 1000`) and a few random restarts
      (including a padded version of the best sequence seen so far).

2.  **Simulated‑annealing local refinement**
    * Starting from the best vector found in phase 1,
      repeatedly add a Gaussian perturbation, re‑normalise the sum
      to `S`, and accept the new vector if its peak autocorrelation
      (`max(f*f)`) is smaller, or with a Metropolis probability that
      slowly cools down.

      loop stops when the time budget is exhausted.

Both phases respect the security checks of `evaluate_sequence`

the best sequence discovered as a plain list of Python floats.

def _conv_max(seq: np.ndarray) -> float:
    """Return max(full convolution of `seq` with itself) using FFT."""
    n = seq.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << (size - 1).bit_length()
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[:size]
    # numerical noise can produce tiny negatives – clamp to 0
    conv = np.maximum(conv, 0.0)
    return float(conv.max())


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2·n‑1)×n matrix A with (A @ g)[k] = (f * g)[k]."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column is a shifted copy of f
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   subject to (f * g) ≤ M , g ≥ 0.
    Returns the optimal `g` (numpy array) or None on failure.
    """
    n = f.shape[0]
    A = _conv_matrix(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)           # minimise -Σg_i
    bounds = [(0.0, None) for _ in range(n)]

    if _HAS_SCI:
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
            if res.success and res.x is not None:
                return np.maximum(res.x, 0.0)
        except Exception:   # pragma: no cover
            pass

    if _HAS_CVX:
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=5000, abstol=1e-8, reltol=1e-6)
        except Exception:   # pragma: no cover
            prob.solve(solver=cp.SCS, warm_start=True,
                       max_iters=15000, verbose=False)
        if g.value is not None:
            return np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)

    return None

    """
    One LP‑based improvement step.
    Returns the new vector and the previous max‑convolution value.
    """
    M = _conv_max(f)
    g = _solve_lp_max_sum(f, M)
    if g is None:
        # LP failed – return the unchanged vector
        return f, M

    sum_g = g.sum()
    if sum_g < 1e-12:
        return f, M

    # Rescale g to the exact ℓ¹‑norm.
    g = g * (S_target / sum_g)

    # Convex combination (keeps ℓ¹‑norm exactly S_target).
    f_new = (1.0 - t) * f + t * g
    # Renormalise just in case of rounding errors.
    f_new = f_new * (S_target / f_new.sum())
    f_new = np.clip(f_new, 0.0, 1000.0)

    return f_new, M


def _annealing_refine(seq: np.ndarray,
                      S_target: float,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climber.
    Perturbs the whole vector with Gaussian noise and accepts
    improvements (or occasional uphill moves) while the temperature
    slowly cools.
    """
    best = seq.copy()
    best_M = _conv_max(best)

    # initial temperature proportional to the current peak
    temp = 0.05 * best_M
    step_std = 0.1 * S_target   # absolute scale of the Gaussian noise

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step_std, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise the ℓ¹‑norm exactly
        cand = cand * (S_target / cand.sum())
        cand_M = _conv_max(cand)

        if cand_M < best_M:
            best, best_M = cand, cand_M
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cand_M - best_M) / max(temp, 1e-12)):
                best, best_M = cand, cand_M

        # exponential cooling
        temp *= 0.9995
        step_std *= 0.9995
        # keep a minimal step size to avoid stagnation
        if step_std < 1e-8:
            step_std = 0.1 * S_target

    return best, best_M


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the global time budget and returns a plain list
    of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # tiny safety margin

    # ------------------------------------------------------------------
    # Prepare possible starting vectors
    # ------------------------------------------------------------------
    # If a previous construction is provided, keep it handy.
    prev_seq = globals().get('height_sequence_1')
    if isinstance(prev_seq, (list, tuple, np.ndarray)):
        prev_seq = np.asarray(prev_seq, dtype=np.float64)
        # strip any NaNs/negatives that could have slipped in
        prev_seq = np.maximum(prev_seq, 0.0)
    else:
        prev_seq = None

    best_seq = None
    best_score = float('inf')

    # Lengths we will explore (larger n gives more freedom)
    lengths = [200, 400, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        S_target = math.sqrt(2.0 * n)          # ℓ¹‑norm we keep fixed
        restarts = 4                           # a few different starts

        for restart in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ---- Initialise the vector f ---------------------------------
            if prev_seq is not None and prev_seq.size <= n and restart == 0:
                # pad the previous best vector centrally
                f = np.zeros(n, dtype=np.float64)
                offset = (n - prev_seq.size) // 2
                f[offset:offset + prev_seq.size] = prev_seq
                f = f * (S_target / f.sum())
                init_msg = "padded"
            else:
                # random exponential‑like start – gives many near‑zero entries
                f = rng.exponential(scale=1.0, size=n) + 0.05
                f = f * (S_target / f.sum())
                init_msg = "random"

            # ---- LP‑driven refinement ------------------------------------
            t = 0.6                         # initial mixing factor
            it = 0
            no_improve = 0
            prev_M = _conv_max(f)

            while time.monotonic() < deadline and it < 300:
                it += 1
                f, M_before = _lp_improve(f, t, S_target)
                M_after = _conv_max(f)

                # keep global best
                if M_after < best_score:
                    best_score = M_after
                    best_seq = f.copy()
                    print(f"[LP] n={n} restart={restart} it={it} "
                          f"score={M_after:.6f} M_before={M_before:.6f}")

                # early exit if we already beat the target
                if best_score < 1.503:
                    print("[LP] Target achieved – exiting early.")
                    return best_seq.tolist()

                # adapt mixing weight if progress stalls
                if M_after >= prev_M * 0.9999:
                    no_improve += 1
                    if no_improve >= 15:
                        t = max(0.1, t * 0.8)
                        no_improve = 0
                else:
                    no_improve = 0

                prev_M = M_after

    # ------------------------------------------------------------------
    # If nothing was found (unlikely), fall back to a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = best_seq * (math.sqrt(2.0 * n) / best_seq.sum())
        best_score = _conv_max(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – stochastic annealing refinement (uses remaining time)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        n = best_seq.shape[0]
        S_target = math.sqrt(2.0 * n)
        refined_seq, refined_M = _annealing_refine(best_seq, S_target,
                                                   deadline, rng)
        if refined_M < best_score:
            best_seq, best_score = refined_seq, refined_M
            print(f"[Anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result as a plain list of Python floats
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
