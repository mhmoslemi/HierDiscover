import time, math, numpy as np

def max_conv(seq):
    n = seq.size
    size = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f * f, size)[:2*n-1]
    conv = np.maximum(conv, 0.0)
    idx = int(conv.argmax())
    return float(conv[idx]), idx

def deterministic_descent(seq, target_sum, deadline, max_iters=50000):
    n = seq.size
    avg = target_sum / n
    # Ensure sum close to target (scaling invariance, but we keep sum)
    # Use projection to maintain sum:
    # After each move, adjust scaling to target sum.
    # Actually scaling invariance: we can ignore sum. But we keep sum near target for fairness.
    cur_max, cur_shift = max_conv(seq)
    best_seq = seq.copy()
    best_max = cur_max

    # precompute for performance
    while time.monotonic() < deadline and max_iters > 0:
        max_iters -= 1
        # compute contributions to current max shift
        n = seq.size
        m = cur_shift
        idx = np.arange(n)
        partner_idx = m - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        # contributions for each i (i's term in sum)
        contrib = np.zeros(n)
        contrib[mask] = seq[idx[mask]] * seq[partner_idx[mask]]
        # source: largest contribution, must have positive value
        if np.all(contrib[mask] <= 1e-12):
            break
        source = int(np.argmax(contrib))
        # target: index with smallest partner value (i.e., a_{m-i})
        partner_vals = np.zeros(n)
        partner_vals[mask] = seq[partner_idx[mask]]
        # choose among indices where a_i not too close to upper bound
        candidate_mask = (~mask) | (seq < 1000.0)  # all indices allowed if out of mask
        # choose target with minimal partner_vals
        # maybe also prefer indices where seq is small
        # combine: rank by (partner_vals, seq)
        # simple: target = argmin partner_vals
        target = int(np.argmin(partner_vals))
        if source == target:
            # try random different target
            # import random pick
            possible = np.where(np.arange(n) != source)[0]
            if possible.size == 0:
                break
            target = int(np.random.choice(possible))
        # compute max possible transfer without exceeding 1000
        max_transfer = min(seq[source], 1000.0 - seq[target])
        if max_transfer <= 1e-12:
            # try different source
            # find next best source with some mass
            sorted_src = np.argsort(-contrib)
            found = False
            for cand in sorted_src:
                if cand == source: continue
                if seq[cand] > 1e-12:
                    source = int(cand)
                    max_transfer = min(seq[source], 1000.0 - seq[target])
                    if max_transfer > 1e-12:
                        found = True
                        break
            if not found:
                break
        # choose delta: small fraction of source to keep stable
        delta = min(max_transfer, avg * 0.5)
        # try decreasing delta if fails
        improved = False
        while delta > 1e-12:
            new_seq = seq.copy()
            new_seq[source] -= delta
            new_seq[target] += delta
            new_max, new_shift = max_conv(new_seq)
            if new_max < cur_max - 1e-12:
                seq = new_seq
                cur_max = new_max
                cur_shift = new_shift
                improved = True
                if new_max < best_max:
                    best_max = new_max
                    best_seq = new_seq.copy()
                    print(f"[det] n={n} new best max={best_max:.6f}")
                break
            else:
                delta *= 0.5
        if not improved:
            # if no improvement try random jitter
            # maybe break if stuck
            # We'll continue to next iteration anyway (might adjust other source/target)
            pass
        # optional early exit if cur_max close to theoretical lower bound maybe <1.503
        if cur_max < 1.503:
            break
    return best_seq, best_max

def simplex_projection(v, target_sum):
    # v may have negative, but we want non-negative after shift.
    # Implementation based on algorithm from Duchi et al.
    # Sort descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def gradient_descent(a, target_sum, deadline, max_iters=2000, alpha_start=1.0, alpha_end=10.0):
    n = a.size
    size_corr = 1 << ((3*n - 2).bit_length())
    size_conv = 1 << ((2*n - 1).bit_length())
    # start with current best max conv
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        # compute convolution using FFT size_conv
        F = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(F * F, size_conv)[:2*n-1]
        max_c = conv.max()
        # compute weights (softmax)
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        # compute softmax numerically stable
        exp_vals = np.exp(alpha * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # gradient via correlation
        A = np.fft.rfft(a, size_corr)
        W = np.fft.rfft(w, size_corr)
        corr = np.fft.irfft(A * np.conj(W), size_corr)
        grad = 2 * corr[:n]   # shape n
        # step size schedule, e.g., eta = 0.1 * target_sum / n * (1 - it / max_iters) + epsilon
        eta = 0.1 * target_sum / n * (1.0 - it / max_iters) + 1e-5
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() == 0:
            # random reinitialize
            a_new = rng.random(n)
        # renormalize to target sum
        a_new = a_new * (target_sum / a_new.sum())
        # evaluate new max conv
        F_new = np.fft.rfft(a_new, size_conv)
        conv_new = np.fft.irfft(F_new * F_new, size_conv)[:2*n-1]
        new_max = conv_new.max()
        # accept if improvement or not?
        # Since we are using gradient descent, we may accept if new_max <= max_c + tolerance
        # else maybe revert and reduce eta
        if new_max <= max_c:
            a = a_new
            cur_max = new_max
        else:
            # maybe reduce eta? But we already have decreasing schedule.
            a = a_new  # still move?
            # Keep track if improvement
        # keep best
        if cur_max < best_max:
            best_max = cur_max; best_seq = a.copy()
        # optional early stop
        if cur_max < 1.5: break
    return best_seq, best_max

"""
Propose a low‑value step sequence.

Algorithm
---------
* Try to improve the maximal auto‑convolution directly.
* For a given length *n* initialise a random (or expanded) non‑negative
  vector whose ℓ¹‑norm is the required √(2·n).
* Run a smooth‑max gradient descent:
    - Compute the full auto‑convolution by FFT.
    - Turn the convolution into a soft‑max weight vector *w* (parameter α
      grows from 1 to 10).  The smoothed objective ≈ log‑sum‑exp of the
      convolution values.
    - The gradient of this objective is the cross‑correlation of the
      current sequence with *w* (computed efficiently by another FFT).
    - Take a short back‑tracking line search, clip negatives,
      and renormalise to keep the ℓ¹‑norm fixed (the evaluation function
      is scale‑invariant).
    - Keep the best vector seen during the run.
* After the gradient phase optionally perform a few linear‑programming
  “mass‑inflation’’ steps (the same LP used in the reference code) to
  further reduce the bound.
* Repeat the whole process for several lengths (200 … 1000) and many
  random restarts until the 1000 s budget expires.  The best found
  sequence is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities -------------------------------------------------
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def max_conv(seq: np.ndarray) -> tuple[float, int]:
    """
    Fast (FFT‑based) computation of max_i (seq * seq)[i] and its index.
    Returns (max_value, argmax_index).
    """
    n = seq.size
    size = _next_pow2(2 * n - 1)
    f_hat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f_hat * f_hat, size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)           # clip tiny negatives
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Stretch ``seq`` into ``new_len`` places by nearest‑index mapping
    and re‑normalise to the ℓ¹‑norm √(2·new_len).
    """
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    tot = out.sum()
    if tot > 0.0:
        out = out * (math.sqrt(2.0 * new_len) / tot)
    return out


# ----------------------------------------------------------------------
# LP refinement (optional – requires cvxpy) ----------------------------
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1)×n Toeplitz‑like matrix A with (A@g)[k]=(f*g)[k]."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i0 = max(0, k - (n - 1))
        i1 = min(k, n - 1)
        # columns that appear in row k:
        col = k - np.arange(i0, i1 + 1)
        A[k, col] = f[i0 : i1 + 1]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    One LP refinement step (maximise ℓ¹‑mass under the current max‑conv
    bound).  Returns updated vector and its exact max‑convolution.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv(f)[0]

    n = f.size
    M = max_conv(f)[0]

    A = build_conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    total = g_val.sum()
    if total < 1e-12:
        return f, M

    g_rescaled = g_val * (target_sum / total)
    t = 0.3
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    return f_new, max_conv(f_new)[0]


# ----------------------------------------------------------------------
# Smooth‑max gradient descent -----------------------------------------
# ----------------------------------------------------------------------


def gradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 800,
    rng: np.random.Generator | None = None,
) -> tuple[np.ndarray, float]:
    """
    Minimise the infinity‑norm of the auto‑convolution using a
    smoothed max (log‑sum‑exp) and FFT‑based gradients.
    Returns the best vector seen and its exact max‑convolution.
    """
    a = a0.copy()
    n = a.size
    size_conv = _next_pow2(2 * n - 1)
    size_corr = _next_pow2(3 * n - 2)

    cur_max, _ = max_conv(a)
    best_a = a.copy()
    best_max = cur_max

    # average entry – useful for step‑size scaling
    avg = target_sum / n
    init_eta = avg * 0.2  # modest start step

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # --- convolution and smooth‑max weights ---------------------------------
        F = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(F * F, size_conv)[: 2 * n - 1]
        max_c = conv.max()

        # increase smoothing as we progress (α from 1 to 10)
        alpha = 1.0 + 9.0 * (it / max_iters)
        exp_vals = np.exp(alpha * (conv - max_c))  # stable: max term becomes 1
        w = exp_vals / exp_vals.sum()               # probabilities over shifts

        # --- gradient (cross‑correlation a ⋆ w) ---------------------------------
        A = np.fft.rfft(a, size_corr)
        W = np.fft.rfft(w, size_corr)
        corr = np.fft.irfft(A * np.conj(W), size_corr)
        grad = 2.0 * corr[:n]   # gradient of the smoothed objective

        # --- back‑tracking line search -----------------------------------------
        eta = init_eta * (1.0 - it / max_iters) + 1e-9
        improved = False
        for _ in range(5):
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s <= 0.0:
                # fallback – re‑initialise a tiny random vector
                a_new = rng.random(n) if rng is not None else np.random.rand(n)
                a_new = a_new / a_new.sum() * target_sum
                s = target_sum
            else:
                a_new = a_new * (target_sum / s)

            new_max, _ = max_conv(a_new)
            if new_max <= cur_max:
                a = a_new
                cur_max = new_max
                improved = True
                if new_max < best_max:
                    best_max = new_max
                    best_a = a_new.copy()
                    print(f"[gd] n={n} iter={it} new best max={best_max:.6f}")
                break
            else:
                eta *= 0.5

        if not improved:
            # no progress – stop early
            break

        if cur_max < 1.503:      # early exit if we already beat the target
            break

    return best_a, best_max
