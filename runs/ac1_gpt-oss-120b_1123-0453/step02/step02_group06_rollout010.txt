def simplex_projection(v, s):
    # Projects onto simplex {x >= 0, sum x = s}
    # v: np.ndarray
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all u_i <= theta
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    beta_factor = 1.02

    tau_start = 2.0
    tau_end = 0.05

    while time.time() < deadline and iter_cnt < 30000:
        iter_cnt += 1
        # compute convolution and max
        conv_full, max_b = compute_conv_and_max(f, size)
        # Save best
        if max_b < best_max - 1e-12:
            best_max = max_b
            best_f = f.copy()

        exp_vals = np.exp(beta * (conv_full - max_b))
        w_sum = exp_vals.sum()

            weights = np.zeros_like(conv_full)
            weights[int(conv_full.argmax())] = 1.0

"""
Propose a low‑value step sequence.

The algorithm consists of two stages:

1. **Projected gradient descent with a smooth soft‑max approximation of the
   ℓ∞‑norm of the auto‑convolution.**  
   For a fixed length `n` we normalise the ℓ¹‑norm to √(2·n) so that the
   evaluation score equals the maximal convolution value.  In each iteration we
   * compute the full linear convolution `f*f` (FFT),
   * form a soft‑max weight vector on the convolution entries,
   * compute the gradient `∇ ≈ 2·(weights ⋆ f)` via an FFT‑based correlation,
   * take a small step opposite to the gradient and project onto the simplex
     `{x ≥ 0, ∑x = √(2·n)}`.

   The temperature (soft‑max sharpness) is increased slowly, while the step
   size decays linearly.  Random jitter is added every few thousand steps to
   escape plateaus.

2. **Optional LP polishing.**  
   When `cvxpy` is available we perform a few cheap linear‑programming
   refinements analogous to the reference method: maximise the ℓ¹‑mass under
   the current maximal convolution bound and mix the result with the current
   vector.

The search is performed for several lengths (`n = 200 … 1500`) with a few
restarts each.  The best sequence found before the 1000 s deadline is returned
as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (non‑negative, prescribed ℓ¹‑norm)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projects v onto the simplex {x ≥ 0, ∑x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")

    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be set to zero
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact max‑convolution (used for final scoring and LP steps)
# ----------------------------------------------------------------------
def max_conv_exact(seq: np.ndarray) -> float:
    """Return the maximum entry of the linear auto‑convolution."""
    return float(np.max(np.convolve(seq, seq)))


# ----------------------------------------------------------------------
# Helpers for the FFT based gradient descent
# ----------------------------------------------------------------------
def compute_conv_and_max(f: np.ndarray, size: int):
    """Return (full convolution f*f, max value).  Uses real FFT of given size."""
    F = np.fft.rfft(f, size)
    conv_full = np.fft.irfft(F * F, size)  # linear convolution, zero‑padded
    conv = conv_full[: 2 * f.size - 1]
    return conv, float(conv.max())


def compute_gradient(weights: np.ndarray, f: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max weighted convolution.
    g[t] = 2 * Σ_i weights[i + t] * f[i]   (0 ≤ t < n)
    implemented as a correlation via FFT.
    """
    n = f.size
    f_rev = f[::-1]
    W = np.fft.rfft(weights, size)
    F_rev = np.fft.rfft(f_rev, size)
    corr_full = np.fft.irfft(W * F_rev, size)  # linear correlation
    start = n - 1
    grad = 2.0 * corr_full[start : start + n]
    return grad


# ----------------------------------------------------------------------
# Optional LP refinement (requires cvxpy)
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Toeplitz‑like matrix A such that (A @ g)[k] = (f * g)[k]."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        A[k, k - np.arange(i_start, i_end + 1)] = f[i_start : i_end + 1]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple:
    """One LP refinement: maximise ℓ¹‑mass under the current max‑conv bound."""
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv_exact(f)

    n = f.size
    M = max_conv_exact(f)
    A = build_conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    sum_g = g_val.sum()
    if sum_g < 1e-12:
        return f, M

    g_rescaled = g_val * (target_sum / sum_g)
    t = 0.3
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)
    return f_new, max_conv_exact(f_new)


# ----------------------------------------------------------------------
# Utility to enlarge an existing sequence to a longer length
# ----------------------------------------------------------------------
def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Stretch ``seq`` to ``new_len`` by nearest‑index accumulation and renorm."""
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    total = out.sum()
    if total > 0:
        target = math.sqrt(2.0 * new_len)
        out = out * (target / total)
    return out


# ----------------------------------------------------------------------
# Core optimiser for a fixed length
# ----------------------------------------------------------------------
def optimise_f(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple:
    """Projected gradient descent with soft‑max smoothing."""
    n = f0.size
    # FFT size large enough for both conv (2n‑1) and correlation (3n‑2)
    size = 1 << ((3 * n - 2).bit_length())

    f = f0.copy()
    conv, cur_max = compute_conv_and_max(f, size)
    best_f = f.copy()
    best_max = cur_max

    start_time = time.monotonic()
    it = 0

    # schedule parameters
    beta = 1.0
    beta_max = 50.0
    beta_factor = 1.02
    tau_start = 2.0   # large initial step
    tau_end = 0.01    # final step

    while time.monotonic() < deadline and it < 20000:
        it += 1

        # recompute convolution and max
        conv, cur_max = compute_conv_and_max(f, size)

        # keep the best so far
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_f = f.copy()
            print(f"[gd] n={n} it={it} new best max={best_max:.6f}")

        # soft‑max weights on the convolution entries
        shift = conv - cur_max          # max entry becomes zero
        exp_vals = np.exp(beta * shift)  # all ≤ 1, no overflow
        w_sum = exp_vals.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback to a pure arg‑max weight
            weights = np.zeros_like(conv)
            weights[int(conv.argmax())] = 1.0
        else:
            weights = exp_vals / w_sum

        # gradient (2 * correlation of weights and f)
        grad = compute_gradient(weights, f, size)

        # linear schedule for the step‑size factor τ
        elapsed_frac = (time.monotonic() - start_time) / max(1e-12, deadline - start_time)
        tau = tau_start * (1.0 - elapsed_frac) + tau_end * elapsed_frac
        lr = tau * target_sum / n

        # gradient step with projection onto the ℓ¹‑simplex
        f = project_to_simplex(f - lr * grad, target_sum)

        # increase soft‑max sharpness
        beta = min(beta * beta_factor, beta_max)

        # occasional random jitter to avoid stagnation
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.005 * target_sum / n, size=n)
            f = project_to_simplex(f + jitter, target_sum)

    return best_f, best_max


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a step sequence with a low evaluation score.
    The function respects the ≈ 1000 s limit, performs several restarts
    over increasing lengths, combines a deterministic projected‑gradient
    descent with optional LP polishing, and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # keep a 2 s safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")   # here “score” == maximal auto‑convolution

    # ------------------------------------------------------------------
    # Optional seed from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim == 1 and init.size > 0:
            n0 = init.size
            target0 = math.sqrt(2.0 * n0)
            if init.sum() > 0:
                init = init * (target0 / init.sum())
                best_seq = init.copy()
                best_score = max_conv_exact(best_seq)
                print(f"[init] using height_sequence_1 (n={n0}) – max={best_score:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – larger n enlarges the search space but costs
    # more time per iteration.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500]

    # ------------------------------------------------------------------
    # Main outer loop over lengths and restarts
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # number of restarts per length – adapt to remaining time
        restarts = 0
        while time.monotonic() < deadline and restarts < 12:
            restarts += 1

            # --------------------------------------------------------------
            # Choose an initial point (random / exponential / symmetric / expand)
            # --------------------------------------------------------------
            init_type = rng.choice(["random", "exp", "sym", "expand"])
            if (
                init_type == "expand"
                and best_seq is not None
                and best_seq.size < n
                and rng.random() < 0.7
            ):
                f0 = expand_sequence(best_seq, n)
            elif init_type == "exp":
                alpha = rng.uniform(0.5, 3.0)
                vals = np.exp(-alpha * np.arange(n))
                f0 = vals / vals.sum() * target_sum
            elif init_type == "sym":
                half = rng.random(n // 2)
                if n % 2:
                    mid = rng.random(1)
                    f0 = np.concatenate([half, mid, half[::-1]])
                else:
                    f0 = np.concatenate([half, half[::-1]])
                f0 = f0 / f0.sum() * target_sum
            else:  # random
                f0 = rng.random(n)
                f0 = f0 / f0.sum() * target_sum

            # --------------------------------------------------------------
            # Deterministic optimisation (projected gradient + soft‑max)
            # --------------------------------------------------------------
            f_local, max_local = optimise_f(f0, target_sum, deadline, rng)

            # --------------------------------------------------------------
            # Optional linear‑programming polishing (if cvxpy is installed)
            # --------------------------------------------------------------
            try:
                import cvxpy as cp  # noqa: F401
                has_cvx = True
            except Exception:  # pragma: no cover
                has_cvx = False

            if has_cvx:
                for _ in range(3):
                    if time.monotonic() >= deadline:
                        break
                    f_local, _ = lp_refine(f_local, target_sum, deadline)
                max_local = max_conv_exact(f_local)

            # --------------------------------------------------------------
            # Record global best
            # --------------------------------------------------------------
            if max_local < best_score - 1e-12:
                best_score = max_local
                best_seq = f_local.copy()
                print(
                    f"[global] n={n} restart={restarts} "
                    f"new best max={best_score:.6f}"
                )

            # early exit if we already cross the target bound
            if best_score < 1.503:
                print("[stop] reached target bound")
                break

    # ------------------------------------------------------------------
    # Safety fallback – never used unless everything else failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = max_conv_exact(best_seq)
        print("[fallback] uniform sequence used, max=", best_score)

    # Convert to a plain list of floats for the evaluator
    return [float(x) for x in best_seq]
