       frac = elapsed / (deadline - start)
       lr = init_lr * (1 - frac) + min_lr * frac
       # decrease tau gradually
       tau = max(0.001, tau * (0.99 ** it))

"""
Gradient‑descent on a smooth approximation of the L∞‑norm
---------------------------------------------------------

We minimise the evaluation function

    E(a) = 2·n·max_k (a * a)[k] / (sum a)^2

Only the shape of the vector matters because the expression is
homogeneous of degree 0.  Hence we keep the ℓ¹‑norm fixed (to 1.0)
and minimise `max_k (a*a)[k]`.

The non‑smooth max is replaced by a soft‑max
    Lτ(a) = τ·log Σ_k exp( (a*a)[k] / τ )
with a temperature τ>0 that is gradually decreased.
For a fixed τ the gradient is

    ∂L/∂a_i = 2· Σ_k w_k·a_{k-i} ,  w_k = exp((a*a)[k]/τ)/Σ_j exp((a*a)[j]/τ)

which is a convolution of the weight vector `w` with the reversed `a`.
All convolutions are performed with the FFT (real‑valued) for speed.

The optimisation proceeds with an exponentiated‑gradient update
that automatically respects non‑negativity; after each step the
vector is projected back onto the probability simplex (ℓ¹‑norm = 1).
Multiple random restarts and occasional tiny Gaussian perturbations
help to escape local minima.  The best vector found for any length is
kept globally.  Finally the raw vector is returned – the evaluator
rescales it automatically, so the ℓ¹‑norm does not matter.

The whole search respects the 1000 s budget (a safety margin of 2 s)
and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures) ---------------------------------------
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """
    Real FFT convolution of a and b, truncated to length len(a)+len(b)-1.
    The supplied size must be a power of two ≥ len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[: len(a) + len(b) - 1]


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def _expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Simple nearest‑index stretching of ``seq`` to length ``new_len``
    followed by renormalisation to ℓ¹‑norm = 1.
    """
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    return _project_to_simplex(out)


def _softmax(x: np.ndarray, tau: float) -> np.ndarray:
    """Numerically stable soft‑max with temperature τ."""
    # x may be large negative; subtract the max first
    m = x.max()
    shifted = (x - m) / tau
    # clip to avoid overflow in np.exp
    np.clip(shifted, -700, 700, out=shifted)
    e = np.exp(shifted)
    return e / e.sum()


def _run_eg_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth max‑objective.
    Returns (best_vector, best_evaluation_score).
    """
    a = a0.copy()
    best_a = a.copy()

    # FFT size must accommodate conv(a,a) (2n‑1) and conv(w,a_rev) (3n‑2)
    size_fft = _next_pow2(3 * n - 2)

    # bookkeeping
    best_score = float("inf")
    start = time.monotonic()

    # hyper‑parameters (tuned empirically)
    lr_start = 0.5      # initial learning rate for EG update
    lr_end = 0.02       # final learning rate as time runs out
    tau_start = 0.15    # start temperature for the soft‑max
    tau_end = 0.001     # final temperature (close to true max)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # --------------------------------------------------------------
        # 1) compute auto‑convolution (probability distribution)
        # --------------------------------------------------------------
        fft_a = np.fft.rfft(a, size_fft)
        conv_full = np.fft.irfft(fft_a * fft_a, size_fft)[: 2 * n - 1]
        conv_full = np.maximum(conv_full, 0.0)          # numerical safety
        max_conv = conv_full.max()

        # current evaluation (ℓ¹‑norm = 1 ⇒ 2·n·max_conv)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[GD] n={n:4d} it={it:5d} best_score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2) schedule learning‑rate and temperature
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start
        total = deadline - start
        frac = min(1.0, elapsed / total) if total > 0 else 1.0
        lr = lr_start * (1.0 - frac) + lr_end * frac
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # --------------------------------------------------------------
        # 3) compute soft‑max weights and gradient
        # --------------------------------------------------------------
        w = _softmax(conv_full, tau)                   # shape (2n‑1,)

        # gradient = 2 * (w * a_rev) convolved, then slice the middle n entries
        a_rev = a[::-1]
        fft_w = np.fft.rfft(w, size_fft)
        fft_rev = np.fft.rfft(a_rev, size_fft)
        grad_full = np.fft.irfft(fft_w * fft_rev, size_fft)
        # length of grad_full = (2n‑1) + n‑1 = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]      # shape (n,)

        # --------------------------------------------------------------
        # 4) Exponentiated‑gradient step (keeps entries non‑negative)
        # --------------------------------------------------------------
        # a ← a * exp(‑η·grad)
        a = a * np.exp(-lr * grad)

        # Projection onto the simplex (ℓ¹‑norm = 1)
        a = _project_to_simplex(a)

        # --------------------------------------------------------------
        # 5) occasional tiny random jitter – helps to escape plateaus
        # --------------------------------------------------------------
        if (it % 1000) == 0 and rng.random() < 0.3:
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_to_simplex(a)

    return best_a, best_score

                break

            # -------------------- initialise candidate --------------------
            if best_seq is not None and best_seq.size < n and rng.random() < 0.6:
                a0 = _expand_sequence(best_seq, n)
                # the expansion already normalises to ℓ¹‑norm = 1
            else:
                a0 = rng.random(n)
                a0 = _project_to_simplex(a0)

            # -------------------- run gradient descent --------------------
            a_opt, score = _run_eg_descent(a0, n, deadline, rng)

            # -------------------- record global best --------------------
            if score < best_score - 1e-12:
                best_score = score
                best_seq = a_opt.copy()
                print(
                    f"[global] n={n:4d} restart={restart+1:2d} "
                    f"new_best_score={best_score:.6f}"
                )

            # early exit if we already beat the target bound
            if best_score < 1.503:
                print("[stop] target bound reached")
                break

        # early exit from outer loop as well
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] uniform sequence used, score =", best_score)

    # Convert to plain Python list of floats for the caller.
    return [float(x) for x in best_seq]
