def project_to_simplex(v):
    """Project onto simplex {x >=0, sum(x)=1}."""
    # Implementation using sorted v.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = len(v) - 1
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def convolve_self(a):

    conv = np.fft.irfft(fa*fa, L)[:2*n-1]

def _project_to_simplex(v):
    """Projection onto probability simplex."""
    n = v.shape[0]
    # shift v so sum is positive
    # algorithm by Wang & Carreira-Perpinan (2013).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all <=0, set to uniform minimal
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Step‑function optimizer for the “max‑convolution” objective.

The score to minimise is

    score = 2·n·max( a * a ) / (sum(a)^2)

For any non‑zero scaling of the vector the factor sum(a)^2 cancels,
so we can safely keep the ℓ¹‑norm equal to 1 and simply minimise

    2·n·max( a * a )      (a_i ≥ 0,  Σ a_i = 1)

The algorithm works as follows:

1.  **Simplex projection** – a helper that projects any real vector onto
    the probability simplex (non‑negative entries that sum to 1).

2.  **FFT based self‑convolution** – `np.fft` is used so a single
    iteration costs O(n log n) even for several thousand coefficients.

3.  **Sub‑gradient descent on the infinity norm** – at each iteration the
    index `k` where the self‑convolution attains its maximum is identified.
    The sub‑gradient of this max‑element is
        g_i = 2·a_{k‑i}   (if 0 ≤ k‑i < n, else 0).
    A step `a ← a – η·g` is taken, a small random jitter is added
    occasionally, and the updated vector is projected back onto the simplex.
    The step size η decays linearly during a run.

4.  **Multiple restarts & lengths** – for a few different lengths
    (300 … 1000) many random restarts are performed while respecting the
    global time budget. The best sequence found over all runs is returned.
    If a historic construction `height_sequence_1` exists it is used as a
    starting point.

The whole routine stays well below the 1000 s limit, uses only NumPy,
and returns a plain list of floats (all entries ≥ 0, ≤ 1000).
"""

import time
import math
import numpy as np

# Helper utilities (no closures)

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    # Find the largest rho satisfying the condition.
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive; return the uniform simplex point.
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear self‑convolution a*a using an FFT of length
    next power of two ≥ 2·len(a)‑1. Returns the real result of length
    2·n‑1.
    """
    n = a.shape[0]
    # smallest power of two ≥ 2·n‑1  →  use (2*n-2).bit_length()
    L = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def _optimize_length(
    n: int,
    rng: np.random.Generator,
    deadline: float,
    target_score: float = 1.503,
) -> tuple[np.ndarray | None, float]:
    """
    Run a series of random‑restart sub‑gradient descents for a fixed length n.
    Returns the best sequence (ℓ¹‑norm = 1) and its score.
    """
    best_seq: np.ndarray | None = None
    best_score = float("inf")
    max_iters_per_restart = 3000
    max_restarts = 20
    restarts_done = 0

    while time.monotonic() < deadline and restarts_done < max_restarts:
        restarts_done += 1

        # random start – Dirichlet‑like (exponential) distribution
        a = rng.exponential(scale=1.0, size=n)
        a = a / a.sum()                     # ℓ¹‑norm = 1

        # random initial step magnitude
        step0 = rng.uniform(0.15, 0.55)

        for it in range(max_iters_per_restart):
            if time.monotonic() >= deadline:
                break

            # 1️⃣ self‑convolution (FFT)
            conv = _convolve_self_fft(a)
            k_max = int(np.argmax(conv))
            max_val = conv[k_max]

            # 2️⃣ sub‑gradient of the max‑convolution entry
            g = np.zeros_like(a)
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] = 2.0 * a[k_max - idx]

            # 3️⃣ step size (linear decay)
            step = step0 * (1.0 - it / max_iters_per_restart)

            # 4️⃣ update + occasional jitter
            a = a - step * g
            if it % 200 == 0:   # tiny random kick every 200 steps
                a = a + rng.normal(scale=step * 0.03, size=n)

            # 5️⃣ project back onto the simplex (keeps sum = 1, non‑neg.)
            a = _project_to_simplex(a)

            # 6️⃣ current score (sum(a) == 1 ⇒ 2·n·max)
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{n}] restart {restarts_done:02d} iter {it:04d} "
                    f"score {best_score:.6f}"
                )
                # early stop if we already beat the target comfortably
                if best_score < target_score - 0.02:
                    # keep searching a bit longer – better hope!
                    pass

            # optional early exit when we are already very good
            if best_score < target_score * 0.99:
                break

    return best_seq, best_score

    # 1️⃣  Search over a few lengths.
    # ------------------------------------------------------------------
    # Lengths that give a reasonable trade‑off between flexibility and speed.
    lengths = [300, 400, 500, 600, 800, 1000]

    # Distribute the remaining time roughly equally among the lengths.
    while time.monotonic() < deadline and lengths:
        n = lengths.pop(0)
        time_left = deadline - time.monotonic()
        # give each size at most 30 s or a fraction of the remaining budget
        per_len_budget = min(30.0, time_left / (len(lengths) + 1))
        per_len_deadline = time.monotonic() + per_len_budget

        seq_n, score_n = _optimize_length(n, rng, per_len_deadline)

        if seq_n is not None and score_n < best_score:
            best_score = score_n
            best_seq = seq_n.copy()
            print(f"[global] new best – n={n} score={best_score:.6f}")

        # very early exit if we have already reached the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Safety fallback – should never trigger.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 250
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[fallback] uniform n={n} score={best_score:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best_seq]
