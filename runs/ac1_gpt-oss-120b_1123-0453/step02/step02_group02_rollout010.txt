def _lp_refine(n, target_sum, deadline, mix=0.3, max_iters=120):
    global _rng
    best_seq = None
    best_score = float('inf')
    # start from random but also include uniform distribution as candidate
    f = _rng.random(n) + 0.1
    f = f / np.sum(f) * target_sum
    # optionally also try uniform
    f_uniform = np.full(n, target_sum / n)
    f_candidates = [f, f_uniform]
    # pick best of a few random starts? Might incorporate later

    f_current = f
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        # compute M
        conv_f = np.convolve(f_current, f_current)
        M = float(conv_f.max())
        # build A
        A = conv_matrix(f_current)  # vectorized
        # solve LP
        try:
            import cvxpy as cp
        except Exception:
            break
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        if g.value is None:
            break
        g0 = np.maximum(g.value, 0)
        s_g = np.sum(g0)
        if s_g < 1e-12:
            break
        g_rescaled = g0 * (target_sum / s_g)
        # update with decreasing mix
        mix_it = max(0.05, mix * (1 - it / max_iters))
        f_new = (1 - mix_it) * f_current + mix_it * g_rescaled

        # compute score directly as max
        sc = float(np.max(np.convolve(f_new, f_new)))
        if sc < best_score:
            best_score = sc
            best_seq = f_new.copy()
            print(f"[LP] n={n} it={it} mix={mix_it:.4f} score={sc:.6f}")
        f_current = f_new
        # early exit if sc is very low
        if best_score < 1.45:
            break
    return best_seq, best_score

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm improves over the previous version by:
  * Using a vectorised construction of the convolution matrix (much faster).
  * Running many more LP refinement iterations with a decreasing mixing factor.
  * Trying several problem sizes (up to n≈800) – larger n gives more freedom.
  * Adding a *targeted* local search that explicitly reduces the worst
    convolution entry by moving mass from the strongest contributing index
    to the weakest one.
  * Falling back to a simple random walk if CVXPY is unavailable.

The whole search respects a 1000 s time budget (a small safety margin is kept
before the external timeout).  The returned sequence consists of non‑negative
floats; after the search it is clipped to the allowed interval [0, 1000].
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global RNG – will be initialised in propose_candidate.
_rng: np.random.Generator | None = None


# ----------------------------------------------------------------------
def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    # The evaluator already performs all required checks and clipping.
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
def conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the dense convolution matrix A of shape (2*n-1, n) such that
    (A @ g) == np.convolve(f, g).

    Vectorised implementation – no explicit Python loops.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # row index matrix (rows,1) and column index matrix (1,n)
    r = np.arange(rows)[:, None]          # shape (rows,1)
    c = np.arange(n)[None, :]             # shape (1,n)
    diff = r - c                          # shape (rows,n)
    mask = (diff >= 0) & (diff < n)       # valid positions
    A = np.zeros((rows, n), dtype=np.float64)
    # Populate A where mask is True: A[row, col] = f[ row - col ]
    A[mask] = f[diff[mask]]
    return A


# ----------------------------------------------------------------------
def _lp_refine(
    n: int,
    target_sum: float,
    deadline: float,
    mix: float = 0.3,
    max_iters: int = 120,
) -> tuple[np.ndarray | None, float]:
    """
    Run the linear‑programming based refinement for a given length `n`.

    Returns (best_sequence, best_score) where the score is the maximum entry
    of the auto‑convolution (equivalently the value returned by the evaluator
    when the ℓ¹‑norm equals √(2 n)).
    """
    global _rng
    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # Start from a random feasible vector (plus a uniform one for diversity).
    f = _rng.random(n) + 0.1               # avoid exact zeros
    f = f / np.sum(f) * target_sum
    f_uniform = np.full(n, target_sum / n, dtype=np.float64)

    # Use the random start; we may later switch to the uniform one if it is better.
    f_current = f

    # Attempt to import CVXPY once.
    try:
        import cvxpy as cp
    except Exception:                     # pragma: no cover
        print("[LP] cvxpy not available – skipping LP stage")
        return None, float("inf")

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # Current worst auto‑convolution entry.
        conv_f = np.convolve(f_current, f_current)
        M = float(conv_f.max())

        # Linear convolution matrix for the current f.
        A = conv_matrix(f_current)

        # ------------------------------------------------------------------
        # Solve   max Σ g_i   s.t.   A·g ≤ M ,  g ≥ 0
        # ------------------------------------------------------------------
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:  # pragma: no cover
            try:
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
            except Exception:  # pragma: no cover
                break

        if g.value is None:
            break

        g0 = np.maximum(g.value, 0.0)
        s_g = float(np.sum(g0))
        if s_g < 1e-12:
            break

        # Rescale so that the ℓ¹‑norm matches the target.
        g_rescaled = g0 * (target_sum / s_g)

        # Decreasing mixing factor – start aggressive, finish delicate.
        mix_it = max(0.02, mix * (1.0 - it / max_iters))

        # Convex‑combination update.
        f_new = (1.0 - mix_it) * f_current + mix_it * g_rescaled
        f_new = np.clip(f_new, 0.0, 1000.0)

        # The evaluation score for a correctly normalised vector is just its peak.
        sc = float(np.max(np.convolve(f_new, f_new)))

        if sc < best_score:
            best_score = sc
            best_seq = f_new.copy()
            print(f"[LP] n={n:3d} it={it:3d} mix={mix_it:.3f} score={sc:.6f}")

        # Early exit if we are already very good.
        if best_score < 1.45:
            break

        f_current = f_new

    # If the uniform start performed better, keep it.
    if best_seq is None:
        # try the uniform vector directly
        sc_u = float(np.max(np.convolve(f_uniform, f_uniform)))
        return f_uniform, sc_u
    return best_seq, best_score


# ----------------------------------------------------------------------
def _targeted_local_search(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic local walk that always tries to lower the *worst* convolution
    entry.  At each step the algorithm:
      1. finds the index k where the convolution attains its maximum,
      2. computes the contribution of every position i to that entry,
      3. moves a fraction of the mass from the biggest contributor to the
         smallest one (keeping the total ℓ¹‑mass unchanged),
      4. accepts the move only if the maximum is reduced.
    The step size shrinks over time.
    """
    global _rng
    best = seq.copy()
    best_score = float(np.max(np.convolve(best, best)))
    n = best.shape[0]
    it = 0

    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(best, best)
        max_idx = int(conv.argmax())
        # contribution of each index i to the worst entry
        contrib = np.zeros(n, dtype=np.float64)
        # vectorised version: compute partner indices
        partners = max_idx - np.arange(n)
        valid = (partners >= 0) & (partners < n)
        contrib[valid] = best[valid] * best[partners[valid]]

        src = int(np.argmax(contrib))
        dst = int(np.argmin(contrib))

        if src == dst:
            # pick a random pair as a fallback
            src, dst = _rng.choice(n, size=2, replace=False)

        max_transfer = best[src]
        if max_transfer <= 0.0:
            continue

        # step size that gradually shrinks (start ~50 % of the source weight)
        step_frac = 0.5 * (1.0 - it / 20000.0)
        step_frac = max(step_frac, 0.01)
        delta = step_frac * max_transfer

        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta

        if cand[src] < 0.0:     # should not happen, safety check
            continue

        new_score = float(np.max(np.convolve(cand, cand)))
        if new_score < best_score - 1e-12:
            best = cand
            best_score = new_score
            print(f"[targeted] iter={it} score={best_score:.6f}")

        # optional early exit if we are extremely close to the theoretical limit
        if best_score <= 1.02:
            break

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    The routine proceeds in three phases:
      1. LP‑based refinement on several problem sizes (up to n≈800).
      2. A deterministic targeted local search that focuses on the worst
         convolution entry.
      3. A final safety clipping and conversion to a plain Python list.
    The whole routine respects the 1000 s overall time limit (a small
    safety margin is kept).
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # Time limit – leave a small safety margin.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) LP‑based refinement on a range of lengths.
    # ------------------------------------------------------------------
    # The set of lengths is chosen to give us a spread of degrees of freedom.
    # Larger n allows a lower peak but makes each LP iteration slower.
    # The schedule of max_iters is tuned so that the whole LP phase finishes
    # well before the global deadline.
    candidate_ns = [200, 250, 300, 350, 400, 500, 600, 700, 800]

    for n in candidate_ns:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Heuristic: allocate more iterations for smaller n (cheaper LP solves).
        max_iters = int(250 * 200 / n)          # roughly 250 iters for n=200
        max_iters = max(40, min(max_iters, 180))  # clamp to a reasonable range

        # Mixing factor – start rather aggressive for small n.
        mix = 0.35 if n <= 400 else 0.25

        best_seq, best_score = _lp_refine(
            n=n,
            target_sum=target_sum,
            deadline=deadline,
            mix=mix,
            max_iters=max_iters,
        )

        if best_seq is not None and best_score < overall_best_score:
            overall_best_score = best_score
            overall_best_seq = best_seq.copy()
            print(f"[global] new best after size {n}: {best_score:.6f}")

        # Early stop if we already beat a very good bound.
        if overall_best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # 2) Deterministic targeted local search (if we have a candidate).
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # Fallback: uniform sequence of moderate size.
        n = 256
        overall_best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        overall_best_score = float(np.max(np.convolve(overall_best_seq, overall_best_seq)))
        print("[fallback] using uniform start sequence")

    if time.monotonic() < deadline:
        # Allocate the remaining time to the focused local optimisation.
        print("[local] starting targeted local search")
        polished_seq, polished_score = _targeted_local_search(overall_best_seq, deadline)
        if polished_score < overall_best_score:
            overall_best_score = polished_score
            overall_best_seq = polished_seq.copy()
            print(f"[global] after polishing: {polished_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final safety: clip and return a plain Python list of floats.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        overall_best_seq = np.ones(256, dtype=np.float64)

    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    # The evaluator will clip again, but we stay on the safe side.
    return [float(x) for x in overall_best_seq]
