def simplex_projection(v, s):
    # v: np.ndarray
    # returns w.
    # algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, s):
    """Project onto the simplex {x >=0, sum(x)=s}."""
    # Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all u <= 0, return zeros (should not happen)
        return np.maximum(v, 0)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Smooth‑gradient optimisation of the max‑auto‑convolution.

The objective that evaluate_sequence returns (with the usual normalisation
∑a_i = sqrt(2·n)) is simply  max_k (a * a)[k].
We therefore minimise the maximum entry of the self‑convolution.

The algorithm works as follows:

1. Choose a length n (several values are tried, up to 600).
2. Initialise a with a simple pattern (uniform, linear, or random) and
   normalise it to the required ℓ¹‑norm.
3. Repeatedly:
   * compute the full autocorrelation c = a * a
   * build a smooth approximation of the max using
         w_k = exp(β·(c_k – max(c)))   (β>0)
   * the gradient of the smooth objective is
         g = 2·conv(w , reverse(a))   (slice of length n)
   * take a step a ← a – η·g, then project back onto the simplex
     {a_i ≥ 0 , ∑a_i = target_sum}.
   * keep the best sequence seen.
   The step size η is decreased geometrically; β is kept modest
   (β=5) so the weights stay well‑behaved.
4. After the gradient phase a short random‑perturbation hill‑climb
   (Gaussian noise with decreasing variance) is applied to escape any
   remaining plateaus.
5. The best sequence over all lengths and restarts is returned.

All heavy work is done with FFT‑based convolutions (O(n log n)) and the
routine respects the 1000 s time budget.  If anything fails the code
falls back to a very cheap random walk.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, sum(x) = s}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a zero vector
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full autocorrelation using FFT (a * a)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return np.convolve(a, a) using a real FFT (fast for n≤2000)."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of two arbitrary 1‑D arrays (full mode)
# ----------------------------------------------------------------------
def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    size = x.shape[0] + y.shape[0] - 1
    m = 1 << (size.bit_length())
    fx = np.fft.rfft(x, m)
    fy = np.fft.rfft(y, m)
    conv = np.fft.irfft(fx * fy, m)[:size]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation – identical to evaluate_sequence when the arguments are
# non‑negative and ≤1000.
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_self(seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max gradient descent for a fixed length n.
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation of max(b_k).
    Returns the best sequence found and its (exact) score.
    """
    a = a0.copy()
    n = a.shape[0]
    beta = 5.0                     # smoothness parameter
    step = target_sum * 0.2        # initial step size (relative to the norm)
    decay = 0.97                   # geometric decay of step
    best_seq = a.copy()
    best_score = _fast_score(a)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # autocorrelation and smooth weights
        conv = _conv_self(a)
        max_val = conv.max()
        # w_k = exp(beta * (conv_k - max_val))  (max weight == 1)
        w = np.exp(beta * (conv - max_val))

        # gradient: g_i = 2 * sum_j w_{i+j} * a_j
        # this is a slice of the convolution w * reverse(a)
        grad_full = _conv_full(w, a[::-1])
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # gradient step + simplex projection
        a_new = a - step * grad
        a_new = _simplex_projection(a_new, target_sum)
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        new_score = _fast_score(a_new)

        # keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            # optional: print progress
            # print(f"[gd] n={n} it={it} score={new_score:.6f}")

        # accept if we have improved, otherwise shrink the step
        if new_score <= _fast_score(a):
            a = a_new
        else:
            step *= 0.5

        step = max(step * decay, target_sum * 1e-6)
        if step < target_sum * 1e-9:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Small random‑perturbation hill‑climb (used after the gradient phase)
# ----------------------------------------------------------------------
def _hill_climb(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 300,
) -> tuple[np.ndarray, float]:
    best = seq.copy()
    best_score = _fast_score(best)
    n = best.shape[0]
    sigma = target_sum * 0.02

    for _ in range(max_iter):
        if time.monotonic() >= deadline:
            break
        cand = best + rng.normal(scale=sigma, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _simplex_projection(cand, target_sum)
        sc = _fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
            sigma *= 0.9
        else:
            sigma *= 0.99
    return best, best_score
