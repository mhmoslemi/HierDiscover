"""
Propose a low‑value step sequence.

The routine improves over the previous implementation in three ways:

1. **Extended size sweep** – we also try problem sizes 800 and 1000.
2. **Line‑search mixing** – after solving the LP we test several
   mixing factors (t values) instead of a single geometric decay.
3. **Focused hill‑climb** – a cheap “mass‑transfer’’ move that steals
   probability mass from the coefficient that contributes most to the
   worst auto‑convolution entry and gives it to a random location.
   This directly attacks the peak of the convolution and often
   reduces the objective more than untargeted Gaussian noise.

All steps respect the 1000 s budget, fall back to a simple random walk
if CVXPY is unavailable and keep the best sequence found overall.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: evaluate a sequence using the provided scorer.
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Wrapper for the external evaluate_sequence."""
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Helper: build the convolution matrix A such that (A @ g)[k] = (f * g)[k].
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return a (2n‑1) × n matrix representing convolution with fixed f."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        cols = k - np.arange(i_start, i_end + 1)
        A[k, cols] = f[i_start:i_end + 1]
    return A


# ----------------------------------------------------------------------
# Helper: compute max convolution value and the index where it occurs.
# ----------------------------------------------------------------------
def _max_conv_and_index(f: np.ndarray):
    """Return (max value, index) of f*f."""
    b = np.convolve(f, f)
    idx = int(np.argmax(b))
    return float(b[idx]), idx


# ----------------------------------------------------------------------
# Helper: one focussed mass‑transfer move.
# ----------------------------------------------------------------------
def _mass_transfer_step(f: np.ndarray, worst_idx: int, rng: np.random.Generator,
                        step_frac: float) -> np.ndarray:
    """
    Transfer a small amount of mass from the coefficient that contributes most
    to the worst convolution entry to a random other position.
    The total ℓ¹‑norm stays unchanged.
    """
    n = f.shape[0]
    # contribution of each i to the worst index (ignoring the factor 2)
    i_arr = np.arange(n)
    j_arr = worst_idx - i_arr
    mask = (j_arr >= 0) & (j_arr < n)
    contrib = np.zeros_like(f)
    contrib[mask] = f[i_arr[mask]] * f[j_arr[mask]]
    donor = int(np.argmax(contrib))
    if contrib[donor] == 0.0:
        # nothing to steal – fall back to a completely random move
        donor = rng.integers(0, n)
    # choose a different recipient uniformly
    recipient = donor
    while recipient == donor:
        recipient = rng.integers(0, n)

    # amount to move – limited to a fraction of the donor value
    delta = min(f[donor] * step_frac, f[donor])
    if delta <= 0.0:
        return f.copy()

    new_f = f.copy()
    new_f[donor] -= delta
    new_f[recipient] += delta
    # enforce the hard caps required by the scorer
    np.clip(new_f, 0.0, 1000.0, out=new_f)
    return new_f


# ----------------------------------------------------------------------
# Simple random‑search fallback (used when CVXPY cannot be imported).
# ----------------------------------------------------------------------
def _fallback_random_search(budget_s: float, seed: int = 0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    n = 300
    seq = np.full(n, 1.0, dtype=np.float64)
    # scale to a reasonable ℓ¹‑norm (the exact value does not matter)
    seq = seq / np.sum(seq) * math.sqrt(2.0 * n)
    best_seq = seq.copy()
    best_score = _score(seq)

    while time.monotonic() < deadline:
        cand = seq + rng.normal(scale=0.02, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            seq = cand
    print("[fallback] best score =", best_score)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging system.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.

    The algorithm iterates over several problem sizes, performs an LP‑based
    refinement with a line‑search over the mixing factor, and then applies a
    focussed hill‑climb that moves mass away from the worst convolution entry.
    The best sequence found over all sizes and restarts is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a global seed sequence is supplied, initialise from it.
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        if init_seq.ndim != 1:
            raise ValueError
        init_score = _score(init_seq)
        best_seq = init_seq.copy()
        best_score = init_score
        print("[init] using provided height_sequence_1, score =", init_score)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Try to import CVXPY – it powers the LP‑based core.
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp
        HAS_CVX = True
    except Exception:               # pragma: no cover
        HAS_CVX = False

    # ------------------------------------------------------------------
    # Problem sizes to explore.  The larger sizes give more freedom but
    # also larger LPs; we stop early if the budget is exhausted.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)          # normalisation used by the scorer
        # initialise f
        if best_seq is not None and best_seq.shape[0] == n:
            f = best_seq.copy()
            cur = np.sum(f)
            if cur > 1e-12:
                f = f * (target_sum / cur)
        else:
            if best_seq is not None and best_seq.shape[0] < n:
                f = np.zeros(n, dtype=np.float64)
                f[:best_seq.shape[0]] = best_seq
                cur = np.sum(f)
                if cur > 1e-12:
                    f = f * (target_sum / cur)
                else:
                    f = rng.random(n) + 0.1
                    f = f / np.sum(f) * target_sum
            else:
                f = rng.random(n) + 0.1
                f = f / np.sum(f) * target_sum

        # mixing schedule – we will try several t‑values each iteration
        t_candidates = [0.3, 0.2, 0.1, 0.05]

        max_iters = 70   # LP refinements per size
        for it in range(max_iters):
            if time.monotonic() >= deadline:
                break

            # current worst auto‑convolution value and index
            M, worst_idx = _max_conv_and_index(f)

            # build the linear‑program matrix for (f * g) ≤ M
            if not HAS_CVX:
                # cannot continue without CVXPY
                break
            A = _conv_matrix(f)

            # ----- linear program -----
            g = cp.Variable(n)
            constraints = [g >= 0, A @ g <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
            except Exception:   # pragma: no cover
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000)

            if g.value is None:
                print(f"[LP] n={n} iter={it} – solver failed")
                break

            g0 = np.maximum(g.value, 0.0)
            sum_g = np.sum(g0)
            if sum_g < 1e-12:
                break

            # rescale to the target ℓ¹‑norm
            g_rescaled = g0 * (target_sum / sum_g)

            # ------------------------------------------------------------------
            # line‑search over a few mixing factors and keep the best candidate.
            # ------------------------------------------------------------------
            best_local_f = f
            best_local_score = _score(f)

            for t in t_candidates:
                cand = (1.0 - t) * f + t * g_rescaled
                cand = np.clip(cand, 0.0, 1000.0)
                sc = _score(cand)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local_f = cand

            f = best_local_f
            sc = best_local_score

            if sc < best_score:
                best_score = sc
                best_seq = f.copy()
                print(f"[LP] n={n:4d} it={it:2d} score={sc:.6f}")

            # ------------------------------------------------------------------
            # short focussed hill‑climb to polish the candidate.
            # ------------------------------------------------------------------
            hill_iters = 20
            step_frac = 0.05  # steal up to 5 % of the donor value each move
            for _ in range(hill_iters):
                if time.monotonic() >= deadline:
                    break
                cand = _mass_transfer_step(f, worst_idx, rng, step_frac)
                sc2 = _score(cand)
                if sc2 < best_score:
                    best_score = sc2
                    best_seq = cand.copy()
                    f = cand
                    print(f"[hill] n={n} it={it} new score={sc2:.6f}")
                elif sc2 < sc:
                    # accept a modest improvement even if not a global best
                    f = cand
                    sc = sc2
                # slowly shrink the transferred fraction
                step_frac = max(0.01, step_frac * 0.95)

            # early stopping if we are already clearly below the target
            if best_score < 1.45:
                break

        # ------------------------------------------------------------------
        # random Gaussian polishing after the whole LP block
        # ------------------------------------------------------------------
        if best_seq is None:
            continue
        perturb_iters = 150
        for _ in range(perturb_iters):
            if time.monotonic() >= deadline:
                break
            cand = best_seq + rng.normal(scale=0.005, size=best_seq.shape[0])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[perturb] n={len(best_seq)} score={sc:.6f}")
                if best_score < 1.45:
                    break

    # ------------------------------------------------------------------
    # If every attempt failed (e.g. CVXPY missing) fall back to random walk.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_seq, fallback_score = _fallback_random_search(
            max(0.0, float(budget_s) - (time.monotonic() - start_time)),
            seed=seed
        )
        best_seq = fallback_seq
        best_score = fallback_score

    print("[final] best score obtained =", best_score, "length =", len(best_seq))
    # Return a plain Python list – the harness expects that.
    return [float(x) for x in best_seq]
