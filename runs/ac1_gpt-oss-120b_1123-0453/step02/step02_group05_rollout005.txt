    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, budget_s-2)

    n = 200  # can be changed

        f_sum = n
    f = f / f_sum * target_sum

        M = conv_f.max()

"""
Low‑max‑convolution step‑function search.

The score after ℓ¹‑normalisation equals max(f*f).  
We minimise this quantity by a Frank‑Wolfe style iteration:

* Given the current vector `f` (non‑negative, Σf = √(2·n)), compute
  M = max(f*f).
* Solve the linear programme
        maximise   Σ g_j
        subject to g_j ≥ 0
                   (f * g)_k ≤ M   for all k
  The constraints are linear because `f` is fixed.
* Scale the LP solution to the required ℓ¹‑norm and mix it with `f`
  by an exact line search on the convex combination
        h(t) = (1‑t)·f + t·g .
  The resulting auto‑convolution is
        (h*h) = (1‑t)²·(f*f) + 2t(1‑t)·(f*g) + t²·(g*g) .
  Scanning `t∈[0,1]` yields the best reduction of the worst entry.
* Accept the improvement, keep the best ever vector and repeat until
  the time budget expires or the target value 1.503 is reached.
* If progress stalls we perform a cheap random perturbation or a full
  restart while preserving the best sequence found so far.

The routine works for any length `n`; if the global variable
`height_sequence_1` exists we start from that construction,
otherwise a random start of length 200 is used.
"""

import time
import math
import numpy as np

# SciPy is allowed and provides a fast MILP/LP solver.
try:
    from scipy.optimize import linprog
except Exception:          # pragma: no cover
    linprog = None  # fallback will use simple random moves only


def _initial_sequence(rng: np.random.Generator, n: int) -> np.ndarray:
    """Return a feasible non‑negative vector of length *n* with ℓ¹‑norm √(2·n)."""
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=np.float64)
        if len(seq) != n:

            seq = np.resize(seq, n)

    total = np.sum(seq)
    if total < 1e-12:
        seq = np.full(n, 1.0, dtype=np.float64)
        total = n
    seq = seq / total * target
    return seq


def _build_A(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2·n‑1) × n) such that (A @ g)_k = (f * g)_k
    for any non‑negative vector g.
    Column j of A is f shifted down by j rows.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_j
        subject to g ≥ 0,
                   (f * g)_k ≤ M  (for every k).
    Returns the optimal g (as a NumPy array) or None on failure.
    """
    if linprog is None:                # pragma: no cover
        return None
    A = _build_A(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)   # linprog minimises → maximise Σg
    bounds = [(0.0, None)] * f.shape[0]
    # HIGHs is the default modern solver, very fast for these modest sizes.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def _best_mix_t(conv_f: np.ndarray,
                conv_g: np.ndarray,
                conv_fg: np.ndarray) -> tuple[float, float]:
    """
    Perform a cheap line‑search on t∈[0,1] that minimises
        max( (1‑t)²·conv_f + 2t(1‑t)·conv_fg + t²·conv_g )
    Returns the best t and the corresponding maximal entry.
    """
    # a 0.02 grid (51 points) is enough for the required precision.
    ts = np.linspace(0.0, 1.0, 51)
    best_t = 0.0
    best_val = conv_f.max()
    for t in ts[1:]:
        a = (1.0 - t) ** 2
        b = 2.0 * t * (1.0 - t)
        c = t ** 2
        conv = a * conv_f + b * conv_fg + c * conv_g
        cur = conv.max()
        if cur < best_val:
            best_val = cur
            best_t = t
    return best_t, best_val

                delta = f[i] * rng.random() * 0.02
                j = rng.integers(0, n - 1)
                while j == i:
                    j = rng.integers(0, n - 1)
                f[i] -= delta
                f[j] += delta

        # ---------------------------------------------------------
        # Periodic restart if we have been idle for a while.
        if time.monotonic() - last_improve > stagnation_limit:
            f = rng.random(n)
            f = np.clip(f, 0.0, 1000.0)
            f = f / np.sum(f) * target_sum
            last_improve = time.monotonic()

        it += 1

        # Early exit if the target is already met.
        if best_M <= 1.503 + 1e-12:
            break

    # -------------------------------------------------------------
    # Return the best sequence found (plain Python floats, clipped to the
    # range accepted by the scorer).
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_f.tolist()]
    final_score = evaluate_sequence(best_seq)
    print("[final] best evaluation =", final_score)
    return best_seq
