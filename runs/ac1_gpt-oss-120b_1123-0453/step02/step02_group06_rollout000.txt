            cur_max = conv.max()
            if cur_max < best_max - 1e-12:

                best_seq = f.copy()

def gd_optimize(n, target_sum, deadline, rng, max_iters=50000):
    # init random
    f = rng.random(n)
    f = f / f.sum() * target_sum
    best_f = f.copy()
    # compute best max
    size_fft = 1 << ((2*n-1).bit_length())
    ft = np.fft.rfft(f, size_fft)
    conv = np.fft.irfft(ft * ft, size_fft)[:2*n-1]
    best_max = conv.max()
    # schedule parameters
    base_eta = target_sum * 0.1 / n # maybe
    for it in range(1, max_iters+1):
        if time.monotonic() >= deadline:
            break
        # compute conv
        ft = np.fft.rfft(f, size_fft)
        conv = np.fft.irfft(ft * ft, size_fft)[:2*n-1]
        cur_max = conv.max()
        # update best
        if cur_max < best_max:
            best_max = cur_max
            best_f = f.copy()
        # smoothing parameter alpha schedule
        # start small, gradually increase
        # e.g., alpha = 10 * (1 - np.exp(-it/5000))
        alpha = 20.0 * (1 - np.exp(-it / 2000.0))
        if alpha < 1e-8:
            alpha = 1e-8
        # compute weights w
        # shift to avoid overflow
        shift = conv.max()
        # stable computation
        exp_vals = np.exp(alpha * (conv - shift))
        w = exp_vals / exp_vals.sum()
        # correlation w and f via FFT
        N_corr = 1 << ((len(w) + n - 1).bit_length())
        W = np.fft.fft(w, N_corr)
        F = np.fft.fft(f, N_corr)

            f_new = proj_simplex(f_new + perturb, target_sum)

    return best_f, best_max

"""
A gradient‑descent + lightweight simulated‑annealing optimiser for the
self‑convolution bound.

The algorithm works on a fixed ℓ¹‑norm (√(2·n)).  For a given length *n* it
optimises the smoothed objective

    Jα(a) = log Σ_k exp(α·(a*a)[k])

which converges to the maximal autocorrelation term as α→∞.
The gradient of Jα is 2·α·corr(w,a) where w are the soft‑max weights of the
convolution and `corr` is the (linear) cross‑correlation.  Both the
convolution and the correlation are evaluated with FFTs (O(n log n)) so
thousands of iterations are cheap.

After the gradient phase a short simulated‑annealing refinement (mass
transfer moves) is applied, which often reduces the bound a little further.
Multiple lengths and random restarts are tried within the 1000 s budget,
keeping the best sequence found.

The returned list satisfies all security checks of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(seq: np.ndarray, size: int = None) -> float:
    """Maximum entry of the (linear) self‑convolution of *seq* using FFT."""
    n = seq.size
    if size is None:
        size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, size)
    conv = np.fft.irfft(ft * ft, size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)          # clip tiny negatives from round‑off
    return float(conv.max())


def max_conv_exact(seq: np.ndarray) -> float:
    """Exact (O(n²)) maximum self‑convolution – used only for final checks."""
    return float(np.max(np.convolve(seq, seq)))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = target}
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex with sum = *target*."""
    if target <= 0:
        return np.zeros_like(v)

    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero after projection
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Expand a shorter sequence to a longer one (nearest‑index mapping)
# ----------------------------------------------------------------------


def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Stretch *seq* to *new_len* by nearest‑index mapping and renormalise."""
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    total = out.sum()
    if total > 0:
        target = math.sqrt(2.0 * new_len)
        out *= target / total
    return out


# ----------------------------------------------------------------------
# Gradient‑descent optimiser (soft‑max smoothing of the max‑convolution)
# ----------------------------------------------------------------------


def gd_optimize(
    n: int,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise the maximal self‑convolution of a non‑negative vector of length *n*
    (with ℓ¹‑norm = *target_sum*) by gradient descent on a log‑sum‑exp smooth
    approximation.
    Returns the best sequence found and its maximal convolution value.
    """
    # --- initialise ----------------------------------------------------
    f = rng.random(n)
    f = f / f.sum() * target_sum

    size_fft = 1 << ((2 * n - 1).bit_length())
    best_f = f.copy()
    # initial max‑convolution (exact, cheap for the first time)
    cur_max = max_conv_fft(f, size_fft)
    best_max = cur_max

    # base step size: a fraction of the average entry
    base_eta = 0.5 * target_sum / n

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # ---- convolution and soft‑max weights -------------------------
        ft = np.fft.rfft(f, size_fft)
        conv = np.fft.irfft(ft * ft, size_fft)[: 2 * n - 1]

        cur_max = float(conv.max())
        if cur_max < best_max:
            best_max = cur_max
            best_f = f.copy()
            print(f"[GD] n={n} iter={it} new best max={best_max:.6f}")

        # smoothing parameter α – start small, increase slowly
        alpha = 20.0 * (1.0 - math.exp(-it / 2_000.0)) + 1e-8

        # soft‑max weights (numerically stable)
        shift = conv.max()
        exp_vals = np.exp(alpha * (conv - shift))
        w = exp_vals / exp_vals.sum()

        # ---- gradient = 2·α·corr(w, f) -------------------------------
        N_corr = 1 << ((len(w) + n - 1).bit_length())
        W = np.fft.fft(w, N_corr)
        F = np.fft.fft(f, N_corr)
        corr = np.fft.ifft(W * np.conj(F)).real
        # slice the central n entries: corr[n-1 : n-1+n]
        grad = 2.0 * alpha * corr[n - 1 : n - 1 + n]

        # ---- gradient step -------------------------------------------
        eta = base_eta / math.sqrt(it)                # decreasing step
        f_new = f - eta * grad

        # projection back onto the simplex (non‑negativity + fixed sum)
        f = project_simplex(f_new, target_sum)

        # occasional tiny random jitter to escape shallow plateaus
        if it % 5_000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            f = project_simplex(f + jitter, target_sum)

    return best_f, best_max


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def sa_refine(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a small amount of mass between two coordinates.
    Used after the gradient stage for a quick final polish.
    Returns the best sequence seen and its maximal convolution.
    """
    n = f0.size
    size_fft = 1 << ((2 * n - 1).bit_length())
    f = f0.copy()
    cur_max = max_conv_fft(f, size_fft)
    best_f = f.copy()
    best_max = cur_max

    T0 = 0.02
    start = time.monotonic()
    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break
        elapsed = time.monotonic() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # compute a feasible transfer amount
        max_transfer = min(f[j], 1000.0 - f[i])
        if max_transfer <= 1e-12:
            continue

        # step size roughly half the average entry
        avg = target_sum / n
        step = min(max_transfer, avg * 0.5)
        delta = rng.random() * step
        if delta <= 1e-12:
            continue

        f_new = f.copy()
        f_new[i] += delta
        f_new[j] -= delta

        new_max = max_conv_fft(f_new, size_fft)
        dE = new_max - cur_max
        if dE < -1e-12 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            f = f_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f_new.copy()
                print(f"[SA] n={n} iter={it} new best max={best_max:.6f}")

        # tiny random jitter each few thousand iterations
        if it % 5_000 == 0:
            k, l = rng.integers(0, n, size=2)
            if k != l:
                tiny = min(f[l], 1e-2)
                if tiny > 0:
                    f[k] += tiny
                    f[l] -= tiny
                    cur_max = max_conv_fft(f, size_fft)

    return best_f, best_max

                                          max_iters=15_000)

            # The evaluation metric (with our normalisation) is exactly max_opt
            score = max_opt
            if score < best_score - 1e-12:
                best_score = score
                best_seq = f_opt.copy()
                print(f"[global] n={n} restart={restarts} new best score={best_score:.6f}")

            # Early exit if we have already beaten the target bound
            if best_score < 1.503:
                print("[stop] reached target bound")
                break

    # ------------------------------------------------------------------
    # Safety fallback – guarantees a return even if everything failed.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = max_conv_exact(best_seq)
        print("[fallback] uniform sequence used, score=", best_score)

    # Convert to plain Python list of floats
    return [float(x) for x in best_seq]
