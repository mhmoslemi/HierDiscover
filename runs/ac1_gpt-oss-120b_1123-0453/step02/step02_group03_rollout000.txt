def _lp_min_max_step(f, target_sum):

    n = len(f)

    M = np.max(np.convolve(f,f))  # maybe not needed.

    A = _conv_matrix(f)   # shape (2n-1, n)

    import cvxpy as cp

    g = cp.Variable(n)

    t = cp.Variable()

    constraints = [g >= 0, cp.sum(g) == target_sum, A @ g <= t]

    prob = cp.Problem(cp.Minimize(t), constraints)

    # Options: solver=ECOS, warm_start.

    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, absvtol=1e-6, reltol=1e-6)

    if g.value is None or t.value is None: return None

    new_f = np.maximum(g.value, 0.0)  # ensure non-negative.

    # Ensure sum close to target. Might adjust:

    s = np.sum(new_f)

    if s > 0:
        new_f = new_f * (target_sum / s)

    # Clip to allowed range.

    new_f = np.clip(new_f, 0, 1000)

    return new_f

def _sa_mass_transfer(seq, cur_score, deadline, rng):

   cur_seq = seq.copy()
   best_seq = cur_seq.copy()
   best_score = cur_score

   start = time.monotonic()
   while time.monotonic() < deadline:
        # temperature schedule: linear or exponential.
        elapsed = time.monotonic() - start
        frac = elapsed / (deadline - start)
        temp = 1.0 * (0.001 ** frac)  # anneal to 0.001.

        i, j = rng.integers(len(cur_seq), size=2)
        if i == j: continue
        # max transferable value from i is cur_seq[i]
        delta_max = cur_seq[i]
        if delta_max <= 0: continue
        # propose transfer amount up to something like temp * max(1, average)
        delta = rng.uniform(0, delta_max * temp * 0.5) # scale
        # apply
        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure within bound [0,1000]
        if cand[i] < 0 or cand[j] > 1000:
            continue
        cand_score = _score(cand)
        # accept if better or metropolis
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

   return best_seq, best_score

- Initialise from global `height_sequence_1` if present, otherwise a uniform
  sequence of length 400 normalised to sum sqrt(2·n).
- Repeatedly improve the sequence with a linear programme that *minimises*
  the maximum convolution value while keeping the ℓ¹‑norm fixed.
  Each LP solution replaces the current sequence (warm‑started with the
  previous iterate), guaranteeing a non‑increase of the objective.
- After the LP phase a simulated‑annealing / mass‑transfer local search
  performs many small random redistributions of mass while preserving the
  total sum, allowing occasional uphill moves to escape local minima.
- The best sequence found before the time limit is returned.
"""

import time
import math
import numpy as np

def _score(seq: np.ndarray) -> float:
    """Return the evaluation score using the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq.tolist()))

def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1)×n matrix A such that (A @ g)[k] = (f * g)[k]."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A

def _lp_min_max_step(f: np.ndarray, target_sum: float) -> np.ndarray | None:
    """
    One LP iteration that, given the current sequence f,
    finds a new non‑negative sequence g with the same ℓ¹‑norm
    that minimises max(f * g).  The solution replaces f.
    """
    n = f.shape[0]
    A = _conv_matrix(f)

    try:
        import cvxpy as cp
    except Exception:
        return None

    g = cp.Variable(n)
    t = cp.Variable()

    # warm start with the previous iterate (helps convergence)
    g.value = f
    t.value = float(np.max(np.convolve(f, f)))

    constraints = [
        g >= 0,
        cp.sum(g) == target_sum,
        A @ g <= t,
        t >= 0,
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)

    # ECOS is fast for dense problems of this size.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, abstol=1e-6, reltol=1e-6)
    except Exception:
        # fallback to SCS if ECOS fails
        prob.solve(solver=cp.SCS, warm_start=True,
                   max_iters=15000, eps=1e-5)

    if g.value is None:
        return None

    g0 = np.maximum(g.value, 0.0)
    s = np.sum(g0)
    if s == 0:
        return None
    # Renormalise to the required ℓ¹‑norm (numerical drift)
    g0 = g0 * (target_sum / s)
    # Clip to the admissible interval required by the scorer
    g0 = np.clip(g0, 0.0, 1000.0)
    return g0

def _sa_mass_transfer(seq: np.ndarray, cur_score: float,
                     deadline: float, rng: np.random.Generator
                     ) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that preserves the total ℓ¹‑norm by
    transferring a random amount of mass between two coordinates.
    """
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4

    n = len(cur_seq)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        temp = temp_start * (temp_end / temp_start) ** frac

        # pick two distinct indices
        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        # amount we can move from i to j
        max_transfer = cur_seq[i]
        if max_transfer <= 0:
            continue

        # scale transfer size with temperature
        delta = rng.random() * max_transfer * temp * 0.5

        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta

        # enforce hard bounds
        if cand[i] < 0 or cand[j] > 1000:
            continue

        cand_score = _score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

        # occasional tiny random perturbation of all entries
        if rng.random() < 0.001:
            perturb = np.maximum(cur_seq + rng.normal(scale=0.05, size=n), 0.0)
            cur_seq = np.clip(perturb, 0.0, 1000.0)
            cur_score = _score(cur_seq)

    return best_seq, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    provided evaluation function.  The routine respects a 1000 s time
    budget, combines a convex‑programming “max‑convolution minimiser” with
    simulated‑annealing mass‑transfer, and returns the best sequence found
    as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Initialise – try to reuse the previously discovered construction
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            seq0 = np.clip(seq0, 0.0, 1000.0)
            sc0 = _score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using height_sequence_1 – score {sc0:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Fallback initialisation
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 400
        target_sum = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target_sum / n0, dtype=np.float64)
        best_score = _score(best_seq)
        print(f"[init] uniform start (n={n0}) – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Main optimisation loop (LP → SA) – may repeat with different
    #    random restarts while time permits.
    # ------------------------------------------------------------------
    n = best_seq.shape[0]
    target_sum = max(0.01, float(np.sum(best_seq)))  # keep the current ℓ¹‑norm

    # Pre‑compute how many LP iterations we can afford (rough estimate)
    # Each LP+score takes ~0.02 s on a modern CPU for n≈400 → allow ~2000.
    max_lp_iters = 250

    # Multiple random restarts – keep the best overall.
    restart = 0
    while time.monotonic() < deadline and restart < 5:
        restart += 1

        # Random initial sequence for this restart (except for the first
        # which keeps the best we already have).
        if restart == 1:
            f = best_seq.copy()
        else:
            # random positive entries, then normalise to the required sum
            f = rng.random(n) + 0.01
            f = f / np.sum(f) * target_sum

        # -----------------------------------------------------------------
        # LP phase – minimise max‑convolution while keeping ℓ¹‑norm fixed
        # -----------------------------------------------------------------
        for it in range(max_lp_iters):
            if time.monotonic() >= deadline:
                break
            f_new = _lp_min_max_step(f, target_sum)
            if f_new is None:
                break
            f = f_new
            sc = _score(f)
            if sc < best_score:
                best_score = sc
                best_seq = f.copy()
                print(f"[LP] restart={restart} it={it} score={sc:.6f}")

            # stop early if improvement stalls
            if it > 10 and abs(sc - best_score) < 1e-7:
                break

            # optional early exit if we already beat the target
            if best_score < 1.50:
                # give some slack for further polishing
                break

        # ---------------------------------------------------------------
        # SA phase – fine‑tune the sequence with mass‑transfer moves
        # ---------------------------------------------------------------
        if time.monotonic() < deadline:
            f_sa, sc_sa = _sa_mass_transfer(f, _score(f), deadline, rng)
            if sc_sa < best_score:
                best_score = sc_sa
                best_seq = f_sa.copy()
                print(f"[SA] after restart={restart} best score={best_score:.6f}")

        # If we have already obtained a score clearly below the target,
        # we can stop early.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Safety fallback – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)

    # Return plain Python list as required by the harness
    return [float(x) for x in best_seq]

   uniform sequence of length 400 whose ℓ¹‑norm equals √(2·n).
