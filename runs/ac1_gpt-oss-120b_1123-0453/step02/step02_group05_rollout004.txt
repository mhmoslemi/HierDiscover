"""
Exponential‑gradient descent for flattening the autocorrelation of a step
function.

The score  2·n·max(f*f) / (∑f)²  is homogeneous, therefore the ℓ¹‑norm of the
vector can be fixed to √(2·n) without loss of optimality.  For a fixed length
`n` the objective is the largest entry of the self‑convolution `conv = f*f`.
A sub‑gradient of `max(conv)` at the worst index `k*` is simply

    g_i = 2·f_{k*‑i}        (if 0 ≤ k*‑i < n, otherwise 0).

We perform exponentiated gradient updates

    f_i ← f_i·exp(‑η·g_i),

followed by a projection onto the simplex {f≥0, ∑f = √(2n)}.
The step size η decays like 1/√t, guaranteeing convergence for convex
problems; the objective `max(conv)` is convex (a pointwise maximum of
quadratic forms) on the non‑negative orthant, so this simple scheme is
well‑suited.

Multiple random restarts (including the previously best construction,
`height_sequence_1`, when available) are performed within the allotted
time budget.  The best sequence encountered is returned after clipping
to the range accepted by `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_l1(v: np.ndarray, target: float) -> np.ndarray:
    """Project non‑negative vector `v` onto the simplex {x≥0, ∑x=target}."""
    total = v.sum()
    if total <= 0.0:
        # fallback to a uniform distribution
        return np.full_like(v, target / v.shape[0])
    return v * (target / total)


# ----------------------------------------------------------------------
def _optimize_expgd(
    f0: np.ndarray,
    target: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient descent on a fixed length `n`.

    Returns the best vector seen and its worst convolution entry.
    """
    n = f0.shape[0]
    f = f0.copy()
    # initial evaluation
    conv = np.convolve(f, f)
    best_max = conv.max()
    best_f = f.copy()
    it = 0
    last_improve_it = 0
    eta0 = 1.0          # initial step size
    smooth = 0.01       # convex combination with the uniform vector

    uniform_val = target / n

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        k = int(np.argmax(conv))
        cur_max = conv[k]

        if cur_max < best_max - 1e-14:          # strict improvement
            best_max = cur_max
            best_f = f.copy()
            last_improve_it = it
            eval_val = 2.0 * n * cur_max / (target ** 2)
            print(
                f"[EG] n={n:03d} it={it:07d} eval={eval_val:.6f} max={cur_max:.6f}"
            )

        # --- sub‑gradient of the active max‑entry ---------------------------------
        grad = np.zeros_like(f)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] = 2.0 * f[k - idx]

        # --- step‑size schedule ----------------------------------------------------
        eta = eta0 / math.sqrt(it + 1.0)

        # --- multiplicative update -------------------------------------------------
        factor = np.exp(-eta * grad)
        f = f * factor

        # --- mild smoothing towards uniform (helps escape plateaus) ---------------
        if smooth > 0.0:
            f = (1.0 - smooth) * f + smooth * uniform_val

        # --- enforce non‑negativity -------------------------------------------------
        f = np.maximum(f, 1e-14)

        # --- project back to the ℓ¹‑simplex ----------------------------------------
        f = _project_to_l1(f, target)

        # --- occasional random jitter if we have been stagnant ----------------------
        if it - last_improve_it > 200_000:
            jitter = rng.random(n) * uniform_val * 0.01
            f = f + jitter
            f = _project_to_l1(f, target)
            last_improve_it = it

        it += 1

    return best_f, best_max


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """
    Search for a step sequence with a low evaluation score.

    The routine performs several independent runs of exponential‑gradient
    descent (starting from the best known construction when available and
    from random initialisations otherwise) within the given time budget.
    The best sequence encountered is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    # Reserve a small safety margin for the caller that prints the final score.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Choose a base length.  We keep the length of the previously discovered
    # construction (if any) to reuse the work already invested, otherwise 200.
    if "height_sequence_1" in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = base_seq.shape[0]
    else:
        n = 200
        base_seq = None

    target = math.sqrt(2.0 * n)          # ℓ¹‑norm required by the scorer

    best_f_global = None
    best_max_global = float("inf")
    run_id = 0

    # ------------------------------------------------------------------
    # Main loop: keep launching optimisation runs until time runs out.
    while time.monotonic() < deadline - 5.0:   # keep a small buffer
        remaining = deadline - time.monotonic()
        # Allocate at most 30 s for a single run (or half of the remaining time).
        run_budget = min(30.0, remaining * 0.5)
        run_deadline = time.monotonic() + run_budget

        # Initialise the starting vector.
        if run_id == 0 and base_seq is not None:
            f0 = base_seq.copy()
        else:
            # Random Dirichlet‑like start.
            rnd = rng.random(n) + 1e-6
            f0 = _project_to_l1(rnd, target)

        # Run the deterministic optimiser.
        f_opt, max_opt = _optimize_expgd(f0, target, run_deadline, rng)

        # Keep the globally best solution.
        if max_opt < best_max_global - 1e-12:
            best_max_global = max_opt
            best_f_global = f_opt.copy()
            eval_val = 2.0 * n * max_opt / (target ** 2)
            print(
                f"[GLOBAL] run={run_id} new best eval={eval_val:.6f} max={max_opt:.6f}"
            )

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback – should never happen, but guard against an empty result.
    if best_f_global is None:
        best_f_global = np.full(n, target / n)

    # Clip to the range required by `evaluate_sequence` and convert to list.
    best_seq = [
        float(max(0.0, min(1000.0, x))) for x in best_f_global.tolist()
    ]

    # Final evaluation (for logging only; the caller does not depend on it).
    final_score = evaluate_sequence(best_seq)
    print("[final] best evaluation =", final_score)

    return best_seq
