"""
Gradient‑descent flattening of the auto‑convolution.

The score to minimise is
    score = 2·n·max_k (f∗f)[k] / (∑_i f_i)^2 .
For a fixed length n we enforce the ℓ¹‑norm ∑_i f_i = √(2·n) (the
optimal normalisation used in the reference solution).  If (f∗f)
were perfectly flat its value would be 2·n/(2·n−1) and the score would
reach the theoretical lower bound 2·n/(2·n−1).

The algorithm works as follows:

* For a selection of lengths `n` (200–1500) we perform several
  independent random restarts.
* Each restart starts from a random point on the simplex of the
  required ℓ¹‑norm (or from the previously best known sequence if its
  length matches).
* We then run a simple first‑order optimisation that minimises the
  squared deviation of the convolution from the flat target value
  `target = (∑ f_i)^2 / (2·n−1)`.  The gradient of
  `E = Σ_k ( (f∗f)_k – target )²` can be written as a convolution,
  which is computed efficiently with `numpy.convolve`.
* After every gradient step we project onto the feasible set:
  - clip negative entries to zero,
  - re‑scale the vector to keep the exact ℓ¹‑norm.
* If the step does not improve the true score we try a few smaller
  step sizes; if still no improvement we inject a tiny random perturb
  to escape plateaux.
* The best sequence over all runs (according to the exact
  `evaluate_sequence` score) is returned.

The method does not rely on external solvers and explores a region
of the search space that is qualitatively different from the LP‑based
approach, giving a good chance of beating the previous best bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)

def _evaluate(seq):
    """Return the official scorer value for a list/array."""
    # `evaluate_sequence` is provided in the execution environment.
    return float(evaluate_sequence(seq.tolist()))


def _compute_score(f):
    """Compute the score 2·n·max(conv)/sum(f)^2 for a numpy vector."""
    n = f.shape[0]
    s = f.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(f, f)
    M = conv.max()
    return float(2.0 * n * M / (s * s))


def _random_restart(f_len, rng):
    """Generate a random non‑negative vector of length f_len whose sum equals √(2·f_len)."""
    target = math.sqrt(2.0 * f_len)
    # Dirichlet‑like sampling (exponential distribution gives a uniform point on the simplex)
    x = rng.random(f_len) + 1e-9
    x = x / x.sum() * target
    return x.astype(np.float64)


def _simple_fallback(budget, seed):
    """Very cheap random‑walk fallback (used only if nothing else succeeded)."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget
    n = 200
    target = math.sqrt(2.0 * n)
    seq = np.full(n, target / n, dtype=np.float64)
    best_seq = seq.copy()
    best_score = _compute_score(seq)

    while time.time() < deadline:
        cand = seq + rng.normal(scale=0.05, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 1e-12:
            cand = cand * (target / s)
        sc = _compute_score(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            seq = cand
    return best_seq, best_score
