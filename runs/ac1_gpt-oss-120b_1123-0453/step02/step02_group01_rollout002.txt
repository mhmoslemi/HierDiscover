"""
Gradient‑based search for a low‑value step sequence.

The evaluation function (after normalising the ℓ¹‑norm to √(2 n))
returns the maximum entry of the auto‑convolution,
so we minimise  max(a * a)  under:
    a_i ≥ 0,   Σ a_i = √(2 n).

The algorithm runs several random restarts for a few different
lengths.  For each restart it performs projected gradient descent on
a smooth approximation of the ℓ∞‑norm:
    f_p(a) = (1/p)·log Σ exp(p·(a*a)ₖ)      (soft‑max of the convolution).

The gradient of f_p can be written as a convolution, so each iteration
uses FFTs (O(n log n)).  After the gradient step the vector is projected
onto the simplex (non‑negative, prescribed ℓ¹‑norm).  A small symmetry
enforcement (a ← (a+a[::-1])/2) encourages a flatter autocorrelation.
If the time budget expires before the gradient loop finishes the code
falls back to a cheap random‑walk optimiser.

The routine returns the best sequence encountered as a plain list of
floats; it prints progress information and the final best score.
"""

import time
import math
import numpy as np


def _evaluate(seq: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence."""
    # evaluate_sequence is already available in the execution environment
    return float(evaluate_sequence(seq.tolist()))


def _project_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the non‑negative simplex of sum = target."""
    if target < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _simple_fallback(budget: float, seed: int):
    """
    Very basic random walk used when the main optimiser cannot be run.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget
    n = 200
    target = math.sqrt(2.0 * n)
    seq = np.full(n, target / n, dtype=np.float64)
    best_seq = seq.copy()
    best_score = _evaluate(seq)

    while time.time() < deadline:
        cand = seq + rng.normal(scale=0.05, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / np.sum(cand) * target
        sc = _evaluate(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            seq = cand
    return best_seq, best_score
