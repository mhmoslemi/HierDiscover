def conv_fft(a):
    # compute convolution a*a using FFT (real)
    L = int(2 ** np.ceil(np.log2(2 * len(a) - 1)))
    # Use np.fft.rfft for real input
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)
    return conv[:2*len(a)-1]

def _search_for_n(n, init_f, rng, deadline):
    compute target_sum = math.sqrt(2.0 * n)
    avg = target_sum / n
    if init_f is not None:
        f = np.array(init_f, dtype=np.float64)
        sum_f = f.sum()
        if sum_f > 0:
            f = f * (target_sum / sum_f)
        else:
            f = rng.random(n) + 0.1
            f = f / f.sum() * target_sum
    else:
        f = rng.random(n) + 0.1
        f = f / f.sum() * target_sum

    L = 2 * n - 1
    # initial convolution via FFT:
    conv = _conv_fft(f)
    M = conv.max()

    best_f = f.copy()
    best_M = M

    # main loop parameters
    move_scale = avg * 0.5          # initial max move size
    move_decay = 0.9999
    temperature = 0.01 * M
    temp_decay = 0.9999
    min_move_scale = avg * 0.001

    # compute max iterations based on time remaining
    # we'll loop indefinite until deadline, with break condition maybe after many declines
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # pick source index weighted by f
        # To avoid zero division if some f entries zero (unlikely), we add tiny epsilon
        prob = f / (f.sum() + 1e-12)
        i = rng.choice(n, p=prob)
        j = rng.integers(0, n)
        if i == j:
            continue

        max_d = f[i]
        # bound by move_scale
        max_move = min(max_d, move_scale)
        if max_move <= 0:
            continue
        d = rng.random() * max_move
        if d <= 0:
            continue

        # compute delta convolution
        delta = np.zeros(L, dtype=np.float64)
        # slices using view
        delta[i:i+n] -= 2.0 * d * f
        delta[j:j+n] += 2.0 * d * f
        idx2i = 2*i
        idx2j = 2*j
        idxij = i + j
        delta[idx2i] += d*d
        delta[idx2j] += d*d
        delta[idxij] -= 2.0 * d*d

        new_conv = conv + delta
        new_M = new_conv.max()

        accept = False
        if new_M < M:
            accept = True
        else:
            # Simulated annealing acceptance
            if temperature > 0:
                prob_accept = math.exp(-(new_M - M) / temperature)
                if rng.random() < prob_accept:
                    accept = True

        if accept:
            # accept move
            conv = new_conv
            f[i] -= d
            f[j] += d
            M = new_M
            if M < best_M:
                best_M = M
                best_f = f.copy()
                print(f"[search n={n}] iter={iteration} new best M={M:.6f}")
        # else reject: do nothing

        # update parameters
        move_scale = max(move_scale * move_decay, min_move_scale)
        temperature = max(temperature * temp_decay, 1e-8)

        # Early exit if no improvement for a while? Not needed.

    return best_f, best_M

"""
Propose a low‑value step sequence.

Algorithm
---------
* The evaluation of a sequence is invariant under a global scale, and after
  normalisation (ℓ¹‑norm = √(2·n)) the objective equals the maximum entry of
  the auto‑convolution f∗f.
* We treat the problem as a massive “mass‑transfer’’ optimisation:
  at each step we move a small amount of mass from an index i to an index j.
  The effect on the convolution can be computed analytically in O(n)
  (no full convolution is needed).
* Moves are accepted if they lower the current worst convolution value M,
  otherwise they are accepted with a small Boltzmann probability (simulated
  annealing) to escape plateaux.
* The search is performed for several lengths n (200 … 2000).  The best
  sequence found for a given n is used as a warm‑start for the next, larger
  n (zero‑padding then re‑normalisation).
* The whole routine respects the 1000 s budget, prints intermediate scores
  and finally returns the best list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: evaluate a candidate using the external scorer.
# ----------------------------------------------------------------------
def _score(seq):
    """Call the provided evaluate_sequence and return a float."""
    # ``evaluate_sequence`` is assumed to be present in the global namespace.
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
# Convolution via FFT (used only for the initial full convolution).
# ----------------------------------------------------------------------
def _conv_fft(f):
    """Return f∗f using a power‑of‑two FFT (real input)."""
    n = f.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    F = np.fft.rfft(f, n=L)
    conv_full = np.fft.irfft(F * F, n=L)
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
# Local stochastic search for a fixed length ``n``.
# ----------------------------------------------------------------------
def _search_n(
    n,
    init_f,
    rng,
    deadline,
    max_iters=50000,
):
    """
    Perform a mass‑transfer simulated‑annealing search for length ``n``.
    Returns the best sequence (numpy array) and its objective value (max conv).
    """
    target_sum = math.sqrt(2.0 * n)
    avg = target_sum / n

    # ---- initialise ----------------------------------------------------
    if init_f is not None:
        f = np.asarray(init_f, dtype=np.float64).copy()
        cur = f.sum()
        if cur > 0:
            f *= target_sum / cur
        else:
            f = rng.random(n) + 0.1
            f /= f.sum()
            f *= target_sum
    else:
        f = rng.random(n) + 0.1
        f /= f.sum()
        f *= target_sum

    conv = _conv_fft(f)                # length 2n‑1
    M = conv.max()

    best_f = f.copy()
    best_M = M

    # ---- annealing schedule -------------------------------------------
    move_scale = avg * 0.5              # maximal transferred mass
    move_decay = 0.9999
    min_move = avg * 0.001

    temperature = 0.01 * M
    temp_decay = 0.9999

    L = conv.shape[0]

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # ----- choose a source index weighted by its mass -------------
        prob = f / (f.sum() + 1e-12)
        i = rng.choice(n, p=prob)
        j = rng.integers(0, n)
        if i == j:
            continue

        # ----- amount to move ----------------------------------------
        max_move = min(f[i], move_scale)
        if max_move <= 0:
            continue
        d = rng.random() * max_move
        if d <= 0:
            continue

        # ----- compute convolution delta analytically -----------------
        delta = np.zeros(L, dtype=np.float64)
        # contribution of the linear part 2·d·(f shifted)
        delta[i : i + n] -= 2.0 * d * f
        delta[j : j + n] += 2.0 * d * f
        # quadratic correction
        delta[2 * i] += d * d
        delta[2 * j] += d * d
        delta[i + j] -= 2.0 * d * d

        new_conv = conv + delta
        new_M = new_conv.max()

        # ----- acceptance test ---------------------------------------
        accept = False
        if new_M < M:
            accept = True
        else:
            # Simulated annealing acceptance
            if temperature > 0:
                prob_accept = math.exp(-(new_M - M) / temperature)
                if rng.random() < prob_accept:
                    accept = True

        if accept:
            # commit move
            conv = new_conv
            f[i] -= d
            f[j] += d
            M = new_M
            if M < best_M:
                best_M = M
                best_f = f.copy()
                print(
                    f"[search] n={n:4d} it={it:6d} M={M:.6f}"
                )
        # else: reject – do nothing

        # ----- schedule updates ---------------------------------------
        move_scale = max(move_scale * move_decay, min_move)
        temperature = max(temperature * temp_decay, 1e-9)

    return best_f, best_M

# Main entry point required by the judging system.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.

    The routine runs a fast stochastic mass‑transfer optimisation for a
    collection of problem sizes, re‑using the best sequence found so far as a
    warm start for larger sizes.  It respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a small safety margin for the outer teardown
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If the harness provides a previous construction, use it as a seed.
    # ------------------------------------------------------------------
    init_global = None
    try:
        init_global = np.array(height_sequence_1, dtype=np.float64)
        if init_global.ndim != 1:
            init_global = None
        else:
            init_score = _score(init_global.tolist())
            best_seq = init_global.copy()
            best_score = init_score
            print("[init] Provided height_sequence_1, score =", init_score)
    except Exception:
        init_global = None

    # ------------------------------------------------------------------
    # Lengths to try – we start modest and gradually increase.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500, 2000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # use the best sequence found so far as a warm start (zero‑pad if needed)
        if init_global is not None:
            if init_global.shape[0] > n:
                start_f = init_global[:n].copy()
            else:
                start_f = np.zeros(n, dtype=np.float64)
                start_f[: init_global.shape[0]] = init_global
        else:
            start_f = None

        # run the stochastic search for this size
        f_best, M_best = _search_n(
            n=n,
            init_f=start_f,
            rng=rng,
            deadline=deadline,
            max_iters=50000,
        )

        # evaluate via the official scorer (should equal M_best)
        sc = _score(f_best.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = f_best.copy()
            print(f"[global] New overall best: n={n}, score={sc:.6f}")

        # store it for the next, larger length
        init_global = f_best.copy()

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: if nothing succeeded, return a simple random sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        f = rng.random(n) + 0.1
        f /= f.sum()
        f *= math.sqrt(2.0 * n)
        best_seq = f
        best_score = _score(best_seq.tolist())
        print("[fallback] score =", best_score)

    # final safety – plain Python list of floats
    return [float(x) for x in best_seq]
