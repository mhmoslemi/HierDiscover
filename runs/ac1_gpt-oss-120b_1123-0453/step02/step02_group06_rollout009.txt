def proj_simplex(v, z=target_sum):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

This projects onto simplex of sum = z. Then we ensure nonnegatives.

def gradient_descent(seq, target_sum, max_iters, deadline, rng):
    n = len(seq)
    a = seq
    step = target_sum * 0.1 / n  # small step magnitude
    best_a = a.copy()
    best_max = max_conv_fft(a)
    for it in range(max_iters):
        if time.monotonic() >= deadline: break
        conv = np.convolve(a, a)
        max_val = conv.max()
        # early exit if already lower than best global known.
        if max_val < best_max - 1e-12:
            best_max = max_val
            best_a = a.copy()
            # Could print progress.
        # Determine active set
        eps = 1e-6 * max_val  # absolute tolerance maybe.
        active = np.where(conv >= max_val - eps)[0]
        # Compute gradient
        grad = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            I = np.arange(i_start, i_end+1)
            grad[I] += a[k - I]
        # Determine step magnitude adaptively; can compute a step that reduces max
        # We'll try multiple step sizes
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            step_size = factor * step
            a_candidate = a - step_size * grad
            a_candidate = proj_simplex(a_candidate, target_sum)
            max_candidate = max_conv_fft(a_candidate)
            if max_candidate < max_val - 1e-9:
                a = a_candidate
                step = step_size  # adapt
                improved = True
                break
        if not improved:
            # reduce step
            step *= 0.5
        # occasional random restart jitter

            # small random perturb
            perturb = rng.normal(scale=target_sum * 0.001, size=n)
            a = proj_simplex(a + perturb, target_sum)
    return best_a, best_max

def gd_optimize(a0, target, deadline, rng, max_iters=5000):
    n = a0.size
    fft_size = 1 << ((2 * n - 1).bit_length())
    a = a0.copy()
    cur_max = max_conv_fft(a, fft_size)
    best_a = a.copy()
    best_max = cur_max
    # step_scale start
    step_scale = target * 0.05  # arbitrary
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        # compute convolution
        conv = np.convolve(a, a)
        max_val = conv.max()
        # active set indices
        eps = 1e-6
        active = np.where(conv >= max_val - eps)[0]
        # compute gradient
        grad = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            I = np.arange(i_start, i_end + 1)
            grad[I] += a[k - I]
        grad *= 2.0
        # compute step norm
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        # candidate step scaling factors
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step = factor * step_scale / (grad_norm + 1e-12)
            cand = a - step * grad
            cand = proj_simplex(cand, target)
            cand_max = max_conv_fft(cand, fft_size)
            if cand_max + 1e-12 < cur_max:
                a = cand
                cur_max = cand_max
                improved = True
                # maybe increase step_scale gradually
                step_scale *= 1.05
                break
        if not improved:
            # reduce step_scale to make smaller steps
            step_scale *= 0.5
        # check best
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            # optional print
            print(f"[GD] n={n} iter={it} new best max={best_max:.6f}")
        # occasional random perturb to escape local minima
        if it % 200 == 0 and rng.random() < 0.1:

            a = proj_simplex(a + perturb, target)
            cur_max = max_conv_fft(a, fft_size)
    return best_a, best_max

It searches the space of non‑negative step sequences by many random restarts, a **projected gradient descent** that directly reduces the current maximal auto‑convolution, and an optional small LP‑refinement (if `cvxpy` is available).

"""
Propose a step sequence with a low maximal auto‑convolution.

Algorithm
---------
* For a list of lengths n (200‑1000) perform many random restarts.
* Each restart:
    – initialise a non‑negative vector with ℓ¹‑norm √(2 n)
      (random, triangular or an expansion of the current global best).
    – run a projected gradient descent:
          • compute the full convolution by FFT,
          • identify the active set (indices whose value is within a tiny
            tolerance of the current maximum),
          • compute the sub‑gradient of the max‑convolution (≈ 2·Σ a_{k−i}),
          • take a step opposite to the gradient, project back onto the
            simplex (maintaining the ℓ¹‑norm),
          • adapt the step size and occasionally add a small random jitter.
    – optionally perform a few LP refinements (maximise ℓ¹‑mass under the
      current convolution bound) if cvxpy is present.
* Keep the globally best vector (lowest maximal convolution) and stop early
  if the target bound 1.503 is reached.
* Return the best sequence as a plain list of floats.
"""

def max_conv_fft(seq: np.ndarray, size: int = None) -> float:
    """
    Fast (FFT) computation of max_i (seq * seq)[i].
    """
    n = seq.size
    if size is None:
        size = 1 << ((2 * n - 1).bit_length())
    fhat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fhat * fhat, size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)          # remove tiny negative artefacts
    return float(conv.max())


def full_conv_fft(seq: np.ndarray, size: int = None) -> np.ndarray:
    """
    Return the full linear convolution (length 2*n-1) using FFT.
    """
    n = seq.size
    if size is None:
        size = 1 << ((2 * n - 1).bit_length())
    fhat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fhat * fhat, size)[: 2 * n - 1]
    return np.maximum(conv, 0.0)

# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x >= 0, sum(x) = target}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return a uniform distribution
        return np.full_like(v, target / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient‑descent optimiser -------------------------------------------
# ----------------------------------------------------------------------


def active_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the maximal auto‑convolution.
    For each active index k we add a_{k-i} to the gradient of component i.
    The true derivative is 2·a_{k-i}, so we multiply by 2 at the end.
    """
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        I = np.arange(i_start, i_end + 1)
        grad[I] += a[k - I]
    grad *= 2.0
    return grad


def gd_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 15000,
) -> tuple:
    """
    Projected gradient descent on the (smoothed) max‑convolution.
    Returns the best vector encountered and its maximal convolution.
    """
    n = a0.size
    fft_size = 1 << ((2 * n - 1).bit_length())

    a = a0.copy()
    cur_max = max_conv_fft(a, fft_size)
    best_a = a.copy()
    best_max = cur_max

    # initial step size (a reasonable fraction of the total mass)
    step_scale = target_sum * 0.05

    # tolerance for the active set (absolute)
    eps_abs = 1e-9

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = full_conv_fft(a, fft_size)
        max_val = float(conv.max())

        # keep global best for this run
        if max_val < best_max - 1e-12:
            best_max = max_val
            best_a = a.copy()
            print(f"[GD] n={n} it={it} new best max={best_max:.6f}")

        # active set (values within a tiny absolute gap of the maximum)
        active = np.where(conv >= max_val - eps_abs)[0]

        # compute sub‑gradient
        grad = active_gradient(a, active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        improved = False
        # try a handful of step‑size multipliers (geometric)
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step = factor * step_scale / (grad_norm + 1e-12)
            cand = a - step * grad
            cand = proj_simplex(cand, target_sum)
            cand_max = max_conv_fft(cand, fft_size)

            if cand_max < cur_max - 1e-12:
                a = cand
                cur_max = cand_max
                # slightly increase step scale after a successful move
                step_scale *= 1.03
                improved = True
                break

        if not improved:
            # no progress – shrink the step scale
            step_scale *= 0.5

        # occasional random jitter to escape plateaus
        if (it + 1) % 500 == 0 and rng.random() < 0.15:
            jitter = rng.normal(scale=target_sum * 0.002, size=n)
            a = proj_simplex(a + jitter, target_sum)
            cur_max = max_conv_fft(a, fft_size)

    return best_a, best_max


# ----------------------------------------------------------------------
# Optional LP refinement (requires cvxpy) -------------------------------
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct matrix A (size (2n‑1)×n) such that (A @ g)[k] = (f * g)[k].
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        A[k, k - np.arange(i_start, i_end + 1)] = f[i_start : i_end + 1]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple:
    """
    One LP refinement step: maximise ℓ¹‑mass under current max‑convolution bound.
    Returns (new_f, new_max_conv).
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv_fft(f)

    n = f.size
    M = max_conv_fft(f)

    A = build_conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    sum_g = g_val.sum()
    if sum_g < 1e-12:
        return f, M

    g_rescaled = g_val * (target_sum / sum_g)

    # modest mixing
    t = 0.3
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)
    return f_new, max_conv_fft(f_new)

def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Stretch an existing sequence to a larger length by nearest‑index mapping
    and re‑normalise to √(2·new_len).
    """
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        idx = min(idx, new_len - 1)
        out[idx] += val
    total = out.sum()
    if total > 0:
        out *= math.sqrt(2.0 * new_len) / total
    return out


def triangular_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a simple triangular (tent) shape, optionally perturbed a little.
    """
    idx = np.arange(n, dtype=np.float64)
    tri = np.minimum(idx + 1, n - idx)
    # optional random small perturbation
    tri += rng.normal(scale=0.001, size=n)
    tri = np.maximum(tri, 0.0)
    return tri

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a step sequence with a low maximal auto‑convolution.
    The function respects the 1000 s time limit (≈ budget_s‑2 s for safety)
    and returns the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Use any previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim == 1 and init.size > 0:
            n0 = init.size
            tgt0 = math.sqrt(2.0 * n0)
            if init.sum() > 0:
                init = init * (tgt0 / init.sum())
                sc0 = max_conv_fft(init)
                best_seq = init.copy()
                best_score = sc0
                print(f"[init] using height_sequence_1 (n={n0}) – score={sc0:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – larger n offers a richer search landscape.
    # ------------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000]

    # ------------------------------------------------------------------
    # Main outer loop over lengths.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Roughly equal time budget per length
        length_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) / max(1, len(lengths) - lengths.index(n)))

        restarts = 0
        while time.monotonic() < length_deadline and restarts < 30:
            restarts += 1

            # ----------------------------------------------------------------
            # Initial point for this restart.
            # ----------------------------------------------------------------
            if best_seq is not None and best_seq.size == n and rng.random() < 0.4:
                f0 = best_seq.copy()
                f0 = proj_simplex(f0 + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            elif best_seq is not None and best_seq.size < n and rng.random() < 0.5:
                f0 = expand_sequence(best_seq, n)
            else:
                if rng.random() < 0.5:
                    f0 = triangular_initial(n, rng)
                else:
                    f0 = rng.random(n)
                f0 = f0 / f0.sum() * target_sum

            # ----------------------------------------------------------------
            # Gradient‑descent optimisation.
            # ----------------------------------------------------------------
            f_opt, max_opt = gd_optimize(f0, target_sum, length_deadline, rng)

            # ----------------------------------------------------------------
            # Optional LP refinement (few cheap iterations).
            # ----------------------------------------------------------------
            try:
                import cvxpy  # noqa: F401
                has_cvx = True
            except Exception:  # pragma: no cover
                has_cvx = False

            if has_cvx:
                for _ in range(3):
                    if time.monotonic() >= length_deadline:
                        break
                    f_opt, max_opt = lp_refine(f_opt, target_sum, length_deadline)
                    max_opt = max_conv_fft(f_opt)

            # ----------------------------------------------------------------
            # Record global best.
            # ----------------------------------------------------------------
            if max_opt < best_score - 1e-12:
                best_score = max_opt
                best_seq = f_opt.copy()
                print(f"[global] n={n} restart={restarts} new best score={best_score:.6f}")

            # early exit if target already reached
            if best_score < 1.503:
                print("[stop] reached target bound")
                break

    # ------------------------------------------------------------------
    # Safety fall‑back – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = max_conv_fft(best_seq)
        print("[fallback] uniform sequence used, score=", best_score)

    # Return plain Python list.
    return [float(x) for x in best_seq]
