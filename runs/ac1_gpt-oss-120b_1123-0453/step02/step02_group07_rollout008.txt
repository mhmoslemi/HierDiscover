Implementation of projection onto simplex: The algorithm:

def project_to_simplex(v, target_sum):
    # ensure v >=0 after projection.
    # Use algorithm from Duchi et al (2008).
    # Input: v (numpy array)
    # Output: w such that w >=0, sum w = target_sum, and w is closest to v in Euclidean norm.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def improve_by_worst_hill_climb(a, target_sum, deadline, rng):
    n = len(a)
    best = a.copy()
    best_score = evaluate_sequence(...)

    cur = a.copy()
    cur_score = best_score

    # parameters
    init_step = target_sum * 0.02  # maybe bigger
    step = init_step
    temperature = 0.1  # acceptance for worse moves

    while time.monotonic() < deadline:
        # compute conv and max conv indices
        conv = np.convolve(cur, cur)
        M = np.max(conv)
        max_inds = np.flatnonzero(conv == M)  # could be multiple
        # choose one at random
        k = rng.choice(max_inds)

        # compute product contributions for each index i
        # pair index j = k - i
        contrib = np.zeros(n)
        valid = (k - np.arange(n) >=0) & (k - np.arange(n) < n)
        idx = np.arange(n)[valid]
        j = k - idx
        contrib[idx] = cur[idx] * cur[j]

        # compute heavy set: indices with large contrib (top maybe)
        # use weighted random selection
        if np.sum(contrib) == 0:
            # fallback random move
            i = rng.integers(0,n)
            j = rng.integers(0,n)
            while i == j or cur[i] <= 0:
                i = rng.integers(0,n)
                j = rng.integers(0,n)
        else:
            # choose i with probability proportional to contrib
            probs = contrib / contrib.sum()
            i = rng.choice(n, p=probs)
            # choose j with probability inversely related or random among low contributions
            # pick random index, but bias to low contrib using (max_contrib - contrib)
            low_probs = (np.max(contrib) - contrib + 1e-12)
            low_probs_sum = low_probs.sum()
            if low_probs_sum > 0:
                low_probs = low_probs / low_probs_sum
                j = rng.choice(n, p=low_probs)
            else:
                j = rng.integers(0,n)
                while j == i:
                    j = rng.integers(0,n)

        # ensure i != j, cur[i] > 0
        if i == j or cur[i] <= 0:
            continue

        # Determine delta: move proportion of cur[i] or step; e.g., delta = min(step, 0.1*cur[i])
        max_move = min(step, cur[i] * 0.5)  # don't zero out
        delta = rng.random() * max_move  # random within [0,max_move]
        if delta <= 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue

        # symmetrize optional:
        cand = (cand + cand[::-1]) / 2

        # Normalize sum to target_sum via projection onto simplex (ensures sum)
        cand = project_to_simplex(cand, target_sum)

        cand_sc = _score(cand)

        # accept scheme
        if cand_sc < cur_score or rng.random() < math.exp(-(cand_sc - cur_score)/temperature):
            cur, cur_sc = cand, cand_sc
            if cur_sc < best_score:
                best_seq, best_score = cur.copy(), cur_sc
                print(f"[improve] n={n} new best={best_score:.6f}")

        # decay parameters
        temperature = max(1e-7, temperature * 0.9995)
        step = max(1e-12, step * 0.9999)

    return best_seq

    n_candidates = [150,200,250,300,350,400,500] maybe.

    # For each n, try few restarts (maybe 3) with different init types.

    for n in n_candidates:

            break
        target_sum = math.sqrt(2.0 * n)

        # Uniform
        init_funcs.append(lambda: np.full(n, target_sum/n, dtype=np.float64))

        # For each init, produce candidate
        for init_func in init_funcs:
            if time.monotonic() >= deadline:
                break
            a = init_func().astype(np.float64)
            # ensure non negative
            a = np.maximum(a, 0.0)

            a = project_to_simplex(a, target_sum)

            a = (a + a[::-1]) / 2

            a = improve_by_worst_hill_climb(a, target_sum, deadline, rng)
            sc = _score(a)

    if best_seq is None:

        n = 200

        best_score = _score(best_seq)
        print("[fallback] uniform used, score", best_score)

    return [float(x) for x in best_seq]

import time, math, numpy as np

# define projection onto simplex
def project_simplex(v, s):
    """Project onto simplex {x >= 0, sum x = s}."""
    n = v.shape[0]
    # shift to zero
    # Use algorithm from Duchi 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? set uniform?
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def improve_seq(a, target_sum, deadline, rng):
    n = len(a)
    # parameters
    beta = 1.0
    beta_inc = (50.0 - 1.0) / 2000  # increase linearly over iterations
    # step size schedule
    step = target_sum * 0.1  # initial step
    step_factor = 0.99  # degrade each iteration

    cur_a = a.copy()
    # ensure feasible
    cur_a = project_simplex(cur_a, target_sum)
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    it = 0
    while time.monotonic() < deadline:
        # if too many iterations break
        if it > 50000:
            break
        # compute gradient via smoothed max
        smooth_val, conv_full, max_c, grad = smoothmax_grad(cur_a, beta)
        # step size adapt
        epsilon = step / (1 + it * 0.001)
        # move
        cand = cur_a - epsilon * grad
        # project onto simplex
        cand = project_simplex(cand, target_sum)
        cand_score = compute_score(cand)
        # accept if better
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

                cur_score = cand_score

        beta = min(100.0, beta + beta_inc)
        # decay step
        step *= step_factor
        it += 1

    return best_a

"""
Propose a step‑function (non‑negative list of heights) that minimizes the
provided evaluate_sequence() bound.

Algorithm
---------

1.  Choose a problem size `n` (we try several values up to about 500).
2.  Initialise a candidate vector with a few simple shapes
    (uniform, random Dirichlet, sine, cosine‑peaked, quadratic).
    The vector is projected onto the simplex
    `{x_i ≥ 0, Σ x_i = √(2 n)}` – this keeps the objective invariant to
    scaling.

3.  Perform a *projected smooth‑max* gradient descent:
    *  For a temperature‐like parameter β we form a smooth approximation of
       `max_k (a * a)[k]` using a soft‑max on the convolution values.
    *  The gradient of this smooth approximation is computed analytically
       (using a second convolution) and a step is taken in the negative
       gradient direction.
    *  After each step the vector is projected back onto the simplex.
    *  β is increased gradually (annealing) so the optimisation starts
       smooth and ends near the true max‑norm.
    *  Step size decays over time; an Adam‑style line‑search is not needed
       because the projection keeps the iterates feasible.

4.  Occasionally add a tiny random perturbation to escape shallow local
    minima.

5.  Keep the best sequence found over all initialisations and all
    problem sizes.  If a previously discovered sequence `height_sequence_1`
    exists we use it as a warm‑start.

The method is completely self‑contained, uses only NumPy (and the supplied
`evaluate_sequence`) and respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σ x = s}."""
    # Implementation of Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Compute the exact score (no clamping needed for our ranges)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_c / (sum_a ** 2))


def _smoothmax_and_grad(a: np.ndarray, beta: float):
    """
    Return a smooth approximation of max(conv) and its gradient w.r.t. a.
    The gradient is derived from the soft‑max weights.
    """
    conv = np.convolve(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # soft‑max weights (stable version)
    shifted = conv - max_c
    exp_beta = np.exp(beta * shifted)
    w = exp_beta / np.sum(exp_beta)          # length 2n‑1, sums to 1

    # gradient = 2 * Σ_k w_k * a_{k-i}
    # Compute efficiently via a second convolution.
    w_rev = w[::-1]
    conv2 = np.convolve(a, w_rev)            # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv2[2 * n - 2 - np.arange(n)]
    # smooth approximation value (not strictly needed for optimisation)
    smooth_val = max_c + np.log(np.sum(exp_beta)) / beta
    return smooth_val, max_c, grad


def _improve_sequence(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Projected smooth‑max gradient descent with simple annealing.
    Returns the best sequence found for this start.
    """
    n = a0.shape[0]
    a = _project_to_simplex(a0, target_sum)
    best = a.copy()
    best_score = _objective(best)

    cur_score = best_score

    beta = 1.0                # start with a very smooth max
    beta_inc = 0.03           # increase slowly → annealing
    step = target_sum * 0.5   # initial step magnitude
    step_decay = 0.9995

    max_iters = 30000
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # smooth max + gradient
        _, max_c, grad = _smoothmax_and_grad(a, beta)

        # normalise step size by the gradient norm
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break
        eps = step / grad_norm

        cand = a - eps * grad
        cand = _project_to_simplex(cand, target_sum)

        cand_score = _objective(cand)

        # accept only improvements (the landscape is well‑behaved after projection)
        if cand_score < cur_score:
            a, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[gd] n={n} it={it} score={best_score:.6f} beta={beta:.2f}")

        # occasionally add a tiny random shake to escape plateaus
        if (it + 1) % 1000 == 0:
            noise = rng.normal(scale=step * 0.01, size=n)
            cand2 = a + noise
            cand2 = np.maximum(cand2, 0.0)
            cand2 = _project_to_simplex(cand2, target_sum)
            cand2_score = _objective(cand2)
            if cand2_score < cur_score:
                a, cur_score = cand2, cand2_score
                if cand2_score < best_score:
                    best, best_score = cand2.copy(), cand2_score
                    print(f"[shake] n={n} it={it} score={best_score:.6f}")

        # update hyper‑parameters
        beta = min(beta + beta_inc, 80.0)
        step = max(step * step_decay, 1e-9)

    return best


# ----------------------------------------------------------------------
# Initialisation helpers (no closures, no lambdas)
def _init_uniform(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, target_sum / n, dtype=np.float64)


def _init_random(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n) + 1e-6
    return _project_to_simplex(v, target_sum)


def _init_sine(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    v = np.sin(np.pi * (i + 0.5) / n) + 1e-3
    return _project_to_simplex(v, target_sum)


def _init_cosine_peaks(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    # peaks at both ends, trough in the middle
    v = 1.0 + np.cos(np.pi * i / (n - 1)) + 1e-3
    return _project_to_simplex(v, target_sum)


def _init_quadratic(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    v = (i - (n - 1) / 2) ** 2 + 1.0
    return _project_to_simplex(v, target_sum)


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: float = 1100,
    **kwargs,
) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) attains a score
    below the target 1.503.  The routine performs several restarts with
    different problem sizes and initial shapes, then refines each start by
    projected smooth‑max gradient descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from any previously found construction.
    if "height_sequence_1" in globals():
        seq0 = np.array(list(height_sequence_1), dtype=np.float64)
        n0 = seq0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        seq0 = _project_to_simplex(seq0, target0)
        seq0 = _improve_sequence(seq0, target0, deadline, rng)
        sc0 = _objective(seq0)
        best_seq, best_score = seq0, sc0
        print("[warm] height_sequence_1 used, score =", sc0)

    # ------------------------------------------------------------------
    # Problem sizes we will explore (larger = potentially lower bound).
    size_list = (150, 200, 250, 300, 350, 400, 450, 500)

    # Initialisers to try (order matters only for speed).
    init_fns = (
        _init_uniform,
        _init_random,
        _init_sine,
        _init_cosine_peaks,
        _init_quadratic,
    )

    for n in size_list:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for init_fn in init_fns:
            if time.monotonic() >= deadline:
                break

            a0 = init_fn(n, target_sum, rng)

            # optional quick LP‑based refinement – only if cvxpy is present
            # (kept very lightweight to stay within the time budget)
            try:
                import cvxpy as cp

                M = float(np.max(np.convolve(a0, a0)))
                A = np.zeros((2 * n - 1, n))
                for k in range(2 * n - 1):
                    i_start = max(0, k - (n - 1))
                    i_end = min(k, n - 1)
                    cols = k - np.arange(i_start, i_end + 1)
                    A[k, cols] = a0[i_start : i_end + 1]
                g = cp.Variable(n)
                constr = [g >= 0, A @ g <= M]
                prob = cp.Problem(cp.Maximize(cp.sum(g)), constr)
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
                if g.value is not None:
                    g0 = np.maximum(g.value, 0.0)
                    if np.sum(g0) > 0:
                        g0 = _project_to_simplex(g0, target_sum)
                        a0 = (a0 + g0) * 0.5  # simple averaging
                        a0 = _project_to_simplex(a0, target_sum)
            except Exception:
                pass  # cvxpy not available or failed – continue without LP

            # Refine with gradient descent
            a_opt = _improve_sequence(a0, target_sum, deadline, rng)

            sc = _objective(a_opt)
            if sc < best_score:
                best_seq, best_score = a_opt, sc
                print(f"[global] n={n} new best score = {best_score:.6f}")

            if best_score < 1.503:
                # target reached – we can stop early
                break
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback to a uniform sequence if nothing succeeded.
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _objective(best_seq)
        print("[fallback] uniform sequence used, score =", best_score)

    # Return a plain Python list
    return [float(x) for x in best_seq]
