"""
Iterative linear‑programming refinement of a step function.

The objective 2·n·max(f*f)/(∑f)² is homogeneous; we therefore keep the
ℓ¹‑norm fixed to √(2·n) and minimise the worst entry of the auto‑convolution.
The algorithm follows the idea from the statement:

    • start from a feasible non‑negative vector f (ℓ¹‑norm = √(2·n));
    • compute its worst convolution value M = max(f*f);
    • build the linear convolution matrix C_f with (C_f g)_k = Σ_i f_i·g_{k‑i};
    • solve the LP    maximise Σ_j g_j
                      subject to C_f g ≤ M·1 ,   g ≥ 0.
      The solution g puts as much ℓ¹‑mass as possible without exceeding the
      current bound M.
    • rescale g to the required ℓ¹‑norm and form a convex combination
          f ← (1‑t)·f + t·g   (t shrinks with time);
      keep the new vector only if it lowers the maximum convolution.
    • repeat until the time budget expires, restarting from a fresh random
      vector after long stagnation while preserving the overall best.

The approach makes larger, systematic moves than a simple simulated‑annealing
step and converges quickly to a lower max‑convolution value.
"""

import time
import math

import numpy as np
import cvxpy as cp


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2·n‑1)×n matrix C_f with (C_f @ g)_k = Σ_i f_i·g_{k‑i}."""
    n = f.shape[0]
    rows = np.arange(2 * n - 1)[:, None]          # shape (2n‑1,1)
    cols = np.arange(n)                           # shape (1,n)
    diffs = rows - cols                            # (2n‑1, n)
    mask = (diffs >= 0) & (diffs < n)              # valid positions
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    C[mask] = f[diffs[mask]]
    return C


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a feasible starting vector of length n with ℓ¹‑norm √(2·n)."""
    # Prefer the previously discovered construction if it matches the length.

        seq = rng.random(n) + 1e-8

    target = math.sqrt(2.0 * n)

            # LP failed – fall back to a tiny random perturbation and continue
            delta = rng.random(n) * 1e-4
            f = np.clip(f + delta - delta.mean(), 0.0, 1000.0)
            f = f / np.sum(f) * target_l1
            it += 1
            continue

        g_vec = np.asarray(g.value, dtype=np.float64)

        # rescale g so that its ℓ¹‑norm equals the target (the objective
        # 2·n·max/∑² is homogeneous, this does not affect the bound)
        sum_g = g_vec.sum()
        if sum_g < 1e-12:
            # nothing useful was found – restart
            f = _initial_sequence(n, rng)
            it += 1
            continue
        g_vec = g_vec * (target_l1 / sum_g)

        # ---- convex combination ------------------------------------------------
        # temperature‑like step size that decays linearly with time
        elapsed = (deadline - (deadline - time.monotonic()))  # dummy to keep style
        # better schedule: exponential decay between T_start and T_end
        total_time = deadline - (deadline - budget_s + 2.0)
        elapsed = time.monotonic() - (deadline - total_time)
        t = T_start * (T_end / T_start) ** (elapsed / total_time)

        f_candidate = (1.0 - t) * f + t * g_vec
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)
        # re‑normalise to keep the ℓ¹‑norm exact (numerical drift)
        f_candidate = f_candidate / np.sum(f_candidate) * target_l1

        conv_candidate = np.convolve(f_candidate, f_candidate)
        M_candidate = conv_candidate.max()

        # accept only if we made progress
        if M_candidate < M:
            f = f_candidate
            if M_candidate < best_val - 1e-12:
                best_val = M_candidate
                best_f = f.copy()
                best_iter = it
                # progress report
                print(
                    f"[LP] it={it:06d} best_max={best_val:.6f} "
                    f"t={t:.4e}"
                )
                last_improve = it
        else:
            # no improvement – try a smaller step next round
            # (the schedule already shrinks, but we accelerate the decay)
            T_start = max(T_start * 0.9, T_end)

        it += 1

        # ------------------------------------------------------------------
        # occasional stochastic restart if we have been stagnant for a while
        if it - last_improve > 5000 and elapsed > total_time * 0.5:
            f = _initial_sequence(n, rng)
            last_improve = it
            # re‑initialise the temperature schedule
            T_start = 0.5

    # ----------------------------------------------------------------------
    # Return the best sequence found (plain Python list of floats)
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_f.tolist()]
    final_score = float(evaluate_sequence(best_seq))
    print("[final] best evaluation =", final_score)
    return best_seq
