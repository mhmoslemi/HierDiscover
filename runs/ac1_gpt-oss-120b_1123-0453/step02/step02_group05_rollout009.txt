def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    # choose length
    n = 500 # maybe 400?
    # initialize f
    if 'height_sequence_1' in globals():
        f = np.array(height_sequence_1, dtype=float)
        if len(f) != n:
            # we can pad or truncate
            if len(f) < n:
                # pad zeros
                f = np.concatenate([f, np.zeros(n - len(f))])
            else:
                f = f[:n]
    else:
        f = np.full(n, 1.0) # before normalization
    # normalize to sum sqrt(2n)
    target_sum = math.sqrt(2.0 * n)
    f = f / f.sum() * target_sum
    best_f = f.copy()
    best_max = np.convolve(f, f).max()
    # iteration schedule: start with t=0.1, gradually reduce
    start_time = time.monotonic()
    deadline = start_time + budget_s - 2.0
    it = 0
    while time.monotonic() < deadline:
        # compute M and conv
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # build constraint matrix A (size (2*n-1, n))
        # Use broadcast: for each i, the contribution to conv indices i:i+n adds f[i] * g.
        A = np.zeros((2*n-1, n))
        for i in range(n):
            A[i:i+n, i] = f[i]   # this adds f_i at column i for rows i..i+n-1
        # upper bound vector b = M * np.ones(2*n-1)
        b = np.full(2*n-1, M)
        # solve linear program: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n)
        # Use scipy.optimize.linprog
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            # fallback: break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # scale g to have same total mass as f
        g = g * (target_sum / sum_g)
        # step size t decreasing with time
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        t = 0.5 * (1 - elapsed/total) + 0.01   # step from 0.51 down to 0.01
        # combine
        f_new = (1-t) * f + t * g
        # evaluate new max
        M_new = np.convolve(f_new, f_new).max()
        # accept if improved; else reject small steps
        if M_new < best_max - 1e-9:
            best_max = M_new
            best_f = f_new.copy()
            # maybe reduce step quickly
            # progress log
            print(f"[Iter {it}] M={M_new:.6f}")
        # update current f: maybe always accept if M_new <= M, else revert.
        if M_new <= M:
            f = f_new
        else:
            # if not improved, maybe keep old f; maybe also try small step: keep old f
            pass

        # optionally recreate with random restart every while loops if stagnant.
        if it % 200 == 0 and best_max > 1.6: # maybe not needed

            f = best_f + rng.normal(scale=target_sum*0.01, size=n)
            f = np.clip(f, 0, None)
            f = f / f.sum() * target_sum

        it += 1
    # after loop: ensure positivity, clip, renorm
    best_f = np.clip(best_f, 0, 1000)
    best_f = best_f / best_f.sum() * target_sum
    best_seq = [float(x) for x in best_f.tolist()]
    final_score = evaluate_sequence(best_seq)
    print("[final] best evaluation =", final_score)
    return best_seq

        else:
            # Deterministic gradient descent phase
            # Identify worst convolution index
            k = int(np.argmax(conv))
            # Gradient contributions only for i in [max(0, k-(n-1)), min(k, n-1)]
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # compute gradient slice
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            grad_slice = 2.0 * f[k - idx]
            # Full gradient vector
            grad = np.zeros(n, dtype=np.float64)
            grad[idx] = grad_slice

            # donor: non‑zero f and maximal gradient
            donor_mask = (grad > 0) & (f > 1e-12)
            if not np.any(donor_mask):
                break
            i = int(np.argmax(grad * donor_mask.astype(float)))  # ensures zero for false

            recipient_mask = grad == 0
            if np.any(recipient_mask):
                j = int(rng.choice(np.where(recipient_mask)[0]))
            else:

                j = int(np.argmin(grad))

                j = rng.integers(0, n)

            max_delta = f[i]

            delta = rng.random() * max_delta * frac

            conv_old = conv.copy()
            f_i_old = f[i]
            f_j_old = f[j]

            # Update convolution efficiently (same as SA)
            conv[i:i + n] -= 2.0 * delta * f
            conv[j:j + n] += 2.0 * delta * f

            # quadratic correction
            conv[2 * i] += delta * delta
            conv[2 * j] += delta * delta
            s = i + j

    best_seq = [float(x) for x in best_f.tolist()]

"""
Hybrid LP + deterministic gradient descent optimizer for the step‑function
evaluation problem.

The evaluation score for a sequence with ℓ¹‑norm √(2·n) is exactly the
maximum entry of its auto‑convolution.  The algorithm therefore tries to
lower that worst convolution value.

Phase 1 (up to ~50 % of the time budget) solves a linear program:
    maximise  Σ g_i
    s.t.    g_i ≥ 0
            (f * g)_k ≤ M   for all k,
where M = max(f * f).  Because (f * g) is linear in g, the LP can be
solved very fast.  The optimal g concentrates the mass where f contributes
least to the active bound; mixing f←(1‑t) f + t g (with a decreasing t)
strictly reduces M in practice.

Phase 2 uses a simple deterministic “gradient‑descent” move that mimics the
sub‑gradient of the max‑convolution.  For the current worst index k we have
∂max/∂f_i = 2·f_{k‑i} (when defined).  The algorithm transfers a small
amount of mass from the index with the largest gradient (donor) to an
index with the smallest gradient (receiver).  The convolution is updated
in‑place using the same formula as the original simulated‑annealing code,
so each step is O(n).

Both phases keep the ℓ¹‑norm fixed, enforce non‑negativity and clip values
to the scorer’s admissible interval.  The best sequence encountered is
returned at the end of the allotted time.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget (seconds).  The function finishes a few
        seconds early so that the caller can still retrieve the result.

    Returns
    -------
    list[float]
        The best sequence (plain Python floats) discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1️⃣ Initialise a feasible sequence (ℓ¹‑norm = √(2·n)).
    # Use the previously found construction if available, otherwise a
    # uniform vector.
    if "height_sequence_1" in globals():
        f = np.array(height_sequence_1, dtype=np.float64)
        f = np.clip(f, 0.0, 1000.0)   # safety clipping
    else:
        # fallback – 200 entries of 1.0
        f = np.full(200, 1.0, dtype=np.float64)

    n = len(f)
    target_sum = math.sqrt(2.0 * n)
    f = f / np.sum(f) * target_sum                     # normalise ℓ¹‑norm

    # ------------------------------------------------------------------
    # 2️⃣ Initial convolution and bookkeeping.
    conv = np.convolve(f, f)                           # full convolution, length 2·n‑1
    cur_max = conv.max()
    best_f = f.copy()
    best_max = cur_max

    # ------------------------------------------------------------------
    # 3️⃣ Time handling.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    start_time = time.monotonic()
    total_time = deadline - start_time
    lp_phase_end = start_time + 0.5 * total_time      # 50 % of the budget for LP

    it = 0
    last_improve_it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop.
    while time.monotonic() < deadline:
        now = time.monotonic()

        # --------------------------------------------------------------
        # Phase 1 – Linear‑programming mixing.
        if now < lp_phase_end:
            # Build the convolution matrix A where
            #   (A @ g)_k = (f * g)_k .
            rows = 2 * n - 1
            A = np.zeros((rows, n), dtype=np.float64)
            for i in range(n):
                A[i:i + n, i] = f[i]

            M = cur_max
            b = np.full(rows, M * (1.0 + 1e-9), dtype=np.float64)   # tiny slack

            # maximise Σ g_i   → minimise -Σ g_i
            res = linprog(

                A_ub=A,

                bounds=(0, None),
                method="highs",
                options={"presolve": True, "time_limit": 1.0},
            )

            if not res.success:
                # If the LP fails (numerical issues) just skip this iteration.
                pass
            else:
                g = res.x
                sum_g = g.sum()
                if sum_g > 0.0:
                    # Scale g to the required ℓ¹‑norm.
                    g = g * (target_sum / sum_g)

                    # Mixing factor – larger at the beginning, slowly shrinks.
                    elapsed = now - start_time

                    conv_candidate = np.convolve(f_candidate, f_candidate)
                    max_candidate = conv_candidate.max()

                    if max_candidate < cur_max:

        # --------------------------------------------------------------
        # Phase 2 – Deterministic gradient‑descent moves.
        else:

            k = int(np.argmax(conv))

            # Gradient of the worst entry w.r.t. each f_i.

            # Amount of mass to move.
            max_delta = f[i]
            if max_delta <= 1e-12:
                # Nothing to move; skip.
                it += 1
                continue

            elapsed = now - start_time
            frac = max(0.01, 0.3 * (1.0 - elapsed / total_time))
            delta = rng.random() * max_delta * frac
            if delta <= 0.0:
                it += 1
                continue

            # Keep a copy for possible rollback.
            conv_old = conv.copy()
            f_i_old, f_j_old = f[i], f[j]

            # ----- update convolution (identical to the SA code) -----
            conv[i:i + n] -= 2.0 * delta * f
            conv[j:j + n] += 2.0 * delta * f

            # quadratic correction
            conv[2 * i] += delta * delta
            conv[2 * j] += delta * delta
            s = i + j
            if s < 2 * n - 1:
                conv[s] -= 2.0 * delta * delta

            # ----- update the sequence itself -----
            f[i] -= delta
            f[j] += delta

            new_max = conv.max()
            if new_max < cur_max:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_f = f.copy()
                    last_improve_it = it
                    print(f"[GD] it={it:07d} max={new_max:.6f}")
            else:
                # Undo the move – it would increase the worst entry.
                conv = conv_old
                f[i], f[j] = f_i_old, f_j_old

        # --------------------------------------------------------------
        # Occasional perturbation if we have been stagnant for a while.
        if it - last_improve_it > 200_000 and (now - start_time) > 0.5 * total_time:
            # Small random perturbation around the best sequence seen so far.
            perturb = rng.normal(scale=target_sum * 0.01, size=n)
            f = best_f + perturb
            f = np.clip(f, 0.0, None)
            f = f / f.sum() * target_sum
            conv = np.convolve(f, f)
            cur_max = conv.max()
            last_improve_it = it
            print(f"[PERTURB] it={it:07d} reset max={cur_max:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 4️⃣ Return the best sequence (clip to the scorer’s domain and re‑normalise
    #    just in case of tiny numerical drift).
    best_f = np.clip(best_f, 0.0, 1000.0)
    best_f = best_f / best_f.sum() * target_sum
    best_seq = [float(x) for x in best_f.tolist()]

    final_score = evaluate_sequence(best_seq)
    print("[final] best evaluation =", final_score)

    return best_seq
