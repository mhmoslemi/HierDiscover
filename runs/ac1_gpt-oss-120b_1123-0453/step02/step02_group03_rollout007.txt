Implementation for projection onto simplex:

def simplex_projection(v, S):

    u = np.sort(v)[::-1]

    cssv = np.cumsum(u) - S

    ind = np.arange(len(v)) + 1

    cond = u - cssv / ind > 0

    if not np.any(cond):
        # all zero
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]

    theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0)

    return w

Thus we can enforce sum S and nonnegatives.

def subgradient_optimize(f, target_sum, max_iters, deadline, rng):
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    step0 = target_sum * 0.5 # some scaling
    for it in range(1, max_iters+1):
        if time.monotonic() >= deadline:
            break
        conv = np.convolve(f, f)
        M = conv.max()
        # compute subgrad
        max_idx = np.where(conv >= M - 1e-12)[0]
        grad = np.zeros(n)
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                # j = k-i
                grad[i] += 2 * f[k - i]
        # compute step size
        eta = step0 / np.sqrt(it) / (np.linalg.norm(grad) + 1e-12) # scale to moderate
        # move opposite direction of grad
        f_new_raw = f - eta * grad
        # project onto simplex
        f_new = _project_to_simplex(f_new_raw, target_sum)
        # evaluate
        sc = _score(f_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            print(f"[SGD] n={n} it={it} score={sc:.6f}")
        # optional acceptance if not improved? maybe still update f to continue
        # Typically subgradient method updates regardless.
        f = f_new
    return best_f, best_score

"""
Optimal step‑function search (≈ 2024‑style)

The optimiser works in three phases:

1️⃣  **Initialisation** – use the global ``height_sequence_1`` if it exists,
    otherwise start from the uniform construction with ℓ¹‑norm √(2 n).

2️⃣  **LP‑driven descent** – repeatedly solve the linear programme that
    maximises the ℓ¹‑mass of a partner sequence *g* under the convolution
    bound ``(f * g) ≤ M`` where ``M = max(f*f)``.  After each LP solve we
    convex–mix the current sequence with the rescaled ``g``.  The mixing
    weight is slowly decreased; this guarantees a monotone reduction of
    the evaluation score.

3️⃣  **Projected sub‑gradient fine‑tuning** – treat the worst convolution
    entry as a (non‑convex) surrogate and move mass from indices that
    contribute heavily to it towards indices that contribute little.
    The step is a projected sub‑gradient step on the simplex
    (ℓ¹‑norm = √(2 n), non‑negative).

4️⃣  **Simulated annealing** – a final stochastic hill‑climber that
    nudges the sequence out of tiny plateaus and extracts the best
    reachable value before the time budget expires.

The whole routine respects the 1000 s budget, prints intermediate scores
and finally returns the best list of non‑negative heights found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Pass the candidate to the provided scorer."""
    # ``evaluate_sequence`` is supplied by the competition harness.
    return float(evaluate_sequence(seq.tolist()))


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix A such that (A @ g)[k] == (f * g)[k].
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _lp_improve(
    f: np.ndarray,
    target_sum: float,
    mix: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray | None, float | None]:
    """
    One LP improvement step as described in the state‑of‑the‑art paper.

    Returns the new sequence and its score, or (None, None) if the LP fails.
    """
    M = float(np.max(np.convolve(f, f)))
    A = _conv_matrix(f)

    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None, None

    n = f.shape[0]
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS works well for dense small‑to‑medium LPs.
    try:
        prob.solve(
            solver=cp.ECOS,
            warm_start=True,
            max_iters=8000,
            abstol=1e-6,
            reltol=1e-6,
        )
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000)

    if g.value is None:
        return None, None

    g0 = np.maximum(g.value, 0.0)
    sum_g = float(np.sum(g0))
    if sum_g <= 1e-12:
        return None, None

    # rescale g to keep the ℓ¹‑norm fixed
    g_rescaled = g0 * (target_sum / sum_g)

    # convex combination
    f_new = (1.0 - mix) * f + mix * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    new_score = _score(f_new)
    return f_new, new_score


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x ≥ 0, Σx = target_sum}.
    """
    # shift for numerical stability
    v_shift = v - np.max(v)  # not strictly needed but helps with large numbers
    u = np.sort(v_shift)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be zero
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v_shift - theta, 0.0)
    return w


def _subgradient_optimize(
    f: np.ndarray,
    target_sum: float,
    max_iters: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the (non‑convex) objective
    ``max_k (f*f)_k``.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_sc = _score(f)

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        M = conv.max()
        # indices where the maximum is attained (allow tiny numerical slack)
        max_idx = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of the max‑convolution entry
        grad = np.zeros(n, dtype=np.float64)
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            grad[i_start : i_end + 1] += 2.0 * f[k - idx]

        # diminishing step size (scaled by the gradient norm)
        denom = np.linalg.norm(grad) + 1e-12
        eta = (target_sum * 0.5) / math.sqrt(it) / denom

        # move opposite to the sub‑gradient and re‑project onto the simplex
        f_candidate = _project_to_simplex(f - eta * grad, target_sum)

        sc = _score(f_candidate)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f_candidate.copy()
            print(f"[SGD] it={it:4d} score={sc:.6f}")

        # always update the current iterate – this is the classic sub‑gradient rule
        f = f_candidate

        # occasional random jitter to avoid stagnation
        if it % 200 == 0:
            jitter = rng.normal(scale=0.02 * target_sum / n, size=n)
            f = _project_to_simplex(f + jitter, target_sum)

    return best_f, best_sc


def _simulated_annealing(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple temperature‑controlled hill‑climber that perturbs one coordinate
    at a time, occasionally accepting uphill moves (Metropolis rule) and
    performing tiny global shuffles.
    """
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        temp = temp_start * (temp_end / temp_start) ** frac

        # --- single‑coordinate perturbation ---------------------------------
        idx = rng.integers(len(cur_seq))
        delta = rng.normal(scale=0.5) * temp * 5.0
        cand = cur_seq.copy()
        cand[idx] = float(np.clip(cand[idx] + delta, 0.0, 1000.0))

        cand_score = _score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best {best_score:.6f}")

        # --- occasional global jitter ---------------------------------------
        if rng.random() < 0.001:
            pert = np.maximum(
                cur_seq + rng.normal(scale=0.1, size=len(cur_seq)), 0.0
            )
            cur_seq = np.clip(pert, 0.0, 1000.0)
            cur_score = _score(cur_seq)

    return best_seq, best_score

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The routine respects the time limit, combines an LP‑based phase,
    a projected sub‑gradient refinement and a final simulated‑annealing
    polishing step, and returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Initialise from a known good construction, else uniform start.
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            seq0 = np.clip(seq0, 0.0, 1000.0)
            sc0 = _score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using height_sequence_1 – score {sc0:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 400
        target_sum0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target_sum0 / n0, dtype=np.float64)
        best_score = _score(best_seq)
        print(f"[init] uniform start (n={n0}) – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  LP‑driven improvement – iterate as long as time permits.
    # ------------------------------------------------------------------
    try:
        import cvxpy  # noqa: F401 – just to check availability
        have_cvx = True
    except Exception:  # pragma: no cover
        have_cvx = False

    if have_cvx:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        f = best_seq.copy()
        mix = 0.5  # start with a relatively aggressive mixing weight
        lp_iter = 0

        while time.monotonic() < deadline - 2.0:  # keep a couple of seconds for later phases
            f_new, sc = _lp_improve(f, target_sum, mix, rng)
            if f_new is None:
                break
            f = f_new
            lp_iter += 1
            mix = max(0.02, mix * 0.994)  # very slowly decay the step size

            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = f.copy()
                print(f"[LP] it={lp_iter:4d} score={sc:.6f}")

            # sanity break if a huge number of iterations have been performed
            if lp_iter >= 1500:
                break

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient refinement (deterministic descent).
    # ------------------------------------------------------------------
    if time.monotonic() < deadline - 1.0:
        remaining = deadline - time.monotonic()
        # Roughly allocate 0.5 s per iteration → bound the number of steps
        max_iters = min(4000, int(remaining * 2))
        f_opt, sc_opt = _subgradient_optimize(
            best_seq.copy(),
            math.sqrt(2.0 * len(best_seq)),
            max_iters,
            deadline,
            rng,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = f_opt.copy()
            print(f"[Grad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Local stochastic polishing (simulated annealing).
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq, best_score, deadline, rng
        )
        print(f"[final] best score after SA = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
