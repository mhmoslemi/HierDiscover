def hill_climb_f(a, deadline, rng):
    a = a.copy()
    n = a.shape[0]
    target_sum = a.sum() # whatever
    best = a.copy()
    best_score = _score(best) # compute evaluate_sequence

    cur = a.copy()
    cur_score = best_score

    temperature = 0.1
    # initial step: maybe 0.01 * target_sum (0.01)
    step = 0.01 * target_sum

    while time.monotonic() < deadline:
        # compute conv to find max index
        conv = np.convolve(cur, cur)
        max_b = conv.max()
        # early exit if below target
        # Use max index
        k = int(conv.argmax())
        # find i that gives max product
        # i_range = max(0, k - (n-1)) .. min(k, n-1)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # compute products
        indices = np.arange(i_start, i_end + 1)
        j_vals = k - indices
        products = cur[indices] * cur[j_vals]
        # pick i that maximizes product
        idx_max = np.argmax(products)
        i = int(indices[idx_max])
        j = int(j_vals[idx_max])
        # Choose destination index to move mass to: pick random index not i or j
        dest = rng.integers(0, n)
        while dest in (i, j):
            dest = rng.integers(0, n)
        # Determine amount to move delta: small fraction of larger of a[i], a[j]
        # Use fraction maybe 0.05 or scale by step
        src = i if cur[i] > cur[j] else j
        delta = min(cur[src] * 0.1, step) # move up to 10% of src or step.
        if delta <= 0:
            continue
        # Apply move
        cand = cur.copy()
        cand[src] -= delta
        cand[dest] += delta
        # enforce nonnegative (cand already)
        # Renormalize to keep sum = target_sum
        s = cand.sum()
        if s <= 0:
            continue
        cand = cand * (target_sum / s)
        cand_score = _score(cand)
        # accept criteria
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / temperature):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print("[hill] n={} score={:.6f}".format(n, best_score))
        # temperature and step decay
        temperature = max(1e-6, temperature * 0.9995)
        step = max(1e-12, step * 0.9995)

    return best

    conv = np.convolve(a, a)
    M = conv.max()
    ks = np.flatnonzero(conv == M)
    return conv, M, ks

Then we need a function to find largest product for a given k:

def max_product_pair(a, k):
    n = len(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    # compute products vectorized
    i = np.arange(i_start, i_end + 1)
    j = k - i
    prod = a[i] * a[j]
    idx = np.argmax(prod)
    return int(i[idx]), int(j[idx]), prod[idx]

Now hill_climb procedure:

def hill_climb(a, local_deadline, rng):
    n = len(a)
    target_sum = a.sum()
    best = a.copy()
    best_sc = _score(best)
    cur = a.copy()
    cur_sc = best_sc

    temperature = 0.5
    step_frac = 0.2   # fraction of source value for move
    min_step = target_sum * 1e-6
    while time.monotonic() < local_deadline:
        conv, M, ks = _max_convolution(cur)
        # choose random k among maxima
        k = int(rng.choice(ks))
        i, j, _ = max_product_pair(cur, k)   # pair with largest product
        # choose source = index among i,j with larger a
        src = i if cur[i] >= cur[j] else j
        # destination index: choose index with smallest a (argmin) not equal src.
        dest = int(np.argmin(cur))
        if dest == src:
            # find second smallest
            dest_candidates = np.argsort(cur)
            for d in dest_candidates:
                if d != src:
                    dest = d
                    break
        # compute delta
        delta = min(cur[src] * 0.2, max(step_frac * target_sum / n, min_step))
        if delta <= 0:
            continue
        candidate = cur.copy()
        candidate[src] -= delta
        candidate[dest] += delta
        # ensure nonnegative
        if candidate[src] < 0:
            continue
        # renormalize sum to target
        s = candidate.sum()
        if s <= 0:
            continue
        candidate = candidate * (target_sum / s)
        cand_sc = _score(candidate)
        # accept if better or with prob
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            # simulate annealing
            prob = math.exp(-(cand_sc - cur_sc) / temperature) if temperature > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            cur = candidate
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best = candidate.copy()
                best_sc = cand_sc
                print("[hill] n={} best_score={:.6f}".format(n, best_sc))
        # decay temperature slowly
        temperature = max(1e-8, temperature * 0.9999)
    return best

def _simplex_projection(v):
    # Returns projection of v onto the simplex {x : x >=0, sum(x)=1}
    # using algorithm of Duchi et al.
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    """Project onto the probability simplex {x: x>=0, sum=1}."""
    # ensure v is numpy array
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max i where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
A fast stochastic sub‑gradient optimiser for the step‑function bound.

The objective is   score = 2·n·max(f*f) / (∑f)² .
Because the formula is scale‑invariant we can normalise any candidate
to an ℓ¹‑simplex (∑f = 1, fᵢ ≥ 0).  The optimisation therefore consists of
minimising  g(f)=maxₖ (f*f)[k] on the simplex.

The algorithm performs many short runs:
* start from a random Dirichlet vector (or a deterministic “U’’ shape);
* repeatedly locate the convolution index k where the current maximum is
  attained;
* compute the sub‑gradient of the corresponding quadratic form
  (gᵢ = 2·f[k‑i] for indices that contribute);
* take a diminishing step in the opposite direction and project back onto
  the simplex (the classic Euclidean projection onto the probability simplex);
* keep the best vector seen.  Occasionally a tiny random jitter is added
  to escape plateaux.

A few restarts with different dimensions (n ≈ 200 … 1200) are executed
until the global time budget (≈1000 s) is exhausted.  The best sequence
found is returned as a plain Python list; `evaluate_sequence` will give the
correct bound (the same as the internally computed score).  Simple hill‑climbing
refinement is applied at the end of every run to polish the result.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _eval_score(seq: np.ndarray) -> float:
    """Exact evaluation of the objective (no clipping needed, sum≈1)."""
    n = seq.shape[0]
    max_b = np.max(np.convolve(seq, seq))
    s = np.sum(seq)
    # protect against division by zero (should never happen)
    if s == 0.0:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D vectors")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _subgradient_opt(
    init: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the simplex.
    Returns the best vector (in terms of _eval_score) observed before the deadline.
    """
    a = _project_simplex(init.astype(np.float64, copy=True))
    n = a.shape[0]

    best_a = a.copy()
    best_sc = _eval_score(best_a)

    it = 0
    last_improve = 0
    step0 = 0.5                # initial step‑size multiplier

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # current convolution & worst index
        conv = np.convolve(a, a)
        max_b = np.max(conv)
        k = int(np.argmax(conv))

        # --------------------------------------------------------------
        # sub‑gradient of the active quadratic form
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]          # includes the diagonal case (2·a_i)

        # --------------------------------------------------------------
        # diminishing step and Euclidean projection onto the simplex
        it += 1
        eta = step0 / math.sqrt(it)
        a_new = a - eta * g
        a = _project_simplex(a_new)

        # --------------------------------------------------------------
        # evaluate
        sc = 2.0 * n * np.max(np.convolve(a, a)) / (np.sum(a) ** 2)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            last_improve = it
            print(f"[sgd] n={n} it={it} best={best_sc:.6f}")

        # --------------------------------------------------------------
        # occasional jitter to escape flat regions
        if it - last_improve > 500:
            jitter = (rng.random(n) - 0.5) * 0.001
            a = _project_simplex(a + jitter)
            last_improve = it

        # safety break in case we are extremely close to the deadline
        if time.monotonic() + 0.005 >= deadline:
            break

    return best_a


# ----------------------------------------------------------------------
def _hill_climb(
    start: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Very light simulated‑annealing hill‑climber that conserves the ℓ¹‑norm.
    Moves a small amount of mass from a “heavy” index to a “light” one.
    """
    n = start.shape[0]
    cur = start.copy()
    cur_sc = _eval_score(cur)
    best = cur.copy()
    best_sc = cur_sc

    temperature = 0.05
    step = max(1e-8, 0.02 / n)          # absolute step size (since sum≈1)

    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue

        delta = min(cur[i] * 0.1, step)          # move up to 10 % of source
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection onto simplex (keeps sum = 1)
        cand = _project_simplex(cand)

        cand_sc = _eval_score(cand)

        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                print(f"[hill] n={n} new best={best_sc:.6f}")

        temperature = max(1e-6, temperature * 0.9995)
        step = max(1e-12, step * 0.9999)

    return best


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function with a low bound.
    Uses many short sub‑gradient runs on random initialisations,
    followed by a tiny hill‑climbing refinement.
    Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    # give a small safety margin for the outer runner
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start with a previously discovered construction if available.
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        init = init / np.maximum(np.sum(init), 1e-12)          # put on simplex
        sc = _eval_score(init)
        best_seq, best_score = init, sc
        print("[init] warm start from height_sequence_1, score =", sc)

    # ------------------------------------------------------------------
    # Candidate problem sizes (larger n usually gives a better asymptotic bound)
    n_candidates = (200, 300, 400, 500, 600, 800, 1000, 1200)

    for n in n_candidates:
        if time.monotonic() >= deadline:
            break

        # allocate a slice of the remaining time to this size
        remaining = deadline - time.monotonic()
        # keep at least a few seconds for the final polishing
        slice_deadline = time.monotonic() + min(remaining * 0.25, 30.0)

        # several restarts for each n
        restarts = 4
        for rs in range(restarts):
            if time.monotonic() >= slice_deadline:
                break

            # random Dirichlet start (dense but not uniform)
            start = rng.dirichlet(np.ones(n))

            # ---- Sub‑gradient optimisation ----
            sub_opt_deadline = time.monotonic() + 8.0    # ~8 s per run
            seq = _subgradient_opt(start, sub_opt_deadline, rng)

            # ---- Light hill‑climber fine‑tuning ----
            hill_deadline = time.monotonic() + 2.0
            seq = _hill_climb(seq, hill_deadline, rng)

            sc = _eval_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[global] n={n} restart={rs} improved best = {best_score:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final sanity check – if nothing was found (unlikely) fall back to uniform.
    if best_seq is None:
        n = 200
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _eval_score(best_seq)
        print("[fallback] uniform sequence used, score =", best_score)

    # The evaluator is scale‑invariant, but returning a nicely scaled vector
    # (∑ = √(2 n)) makes the numbers a bit larger and easier to inspect.
    n_final = best_seq.shape[0]
    target_sum = math.sqrt(2.0 * n_final)
    best_seq = best_seq * (target_sum / np.sum(best_seq))
    # Clip just in case (evaluate_sequence will also clip)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print("[result] final n =", n_final, "score ≈", _eval_score(best_seq))
    return [float(x) for x in best_seq]
