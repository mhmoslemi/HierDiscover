"""
Propose a step‑function that minimizes the provided evaluation metric.
The algorithm works directly on the *probability* vector `p` (sum(p)=1).
For a fixed length `n` the evaluation simplifies to

    score = 2·n·max{ (p * p)[k] }                 (1)

because scaling of the vector does not change the ratio inside
`evaluate_sequence`.  Hence we only need to minimise the peak of the
self‑convolution of `p`.

The optimizer uses two complementary ideas:

* **Projected sub‑gradient descent** – repeatedly locate the index `k`
  where the convolution attains its maximum and move probability mass
  away from the indices that contribute most to that entry.
  The update is projected back onto the probability simplex.

* **Local stochastic refinement** – after the deterministic phase a
  lightweight hill‑climber (the same routine used in the reference
  solution) is applied to squeeze out the last improvement.

Multiple problem sizes (`n` in the range 150‑400) and several random
restarts are tried until the global time budget expires.  The best
sequence found is scaled to the required ℓ¹‑norm
`sqrt(2·n)` and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Evaluate a candidate using the given black‑box."""
    # the evaluation function expects a plain list of Python floats
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i >= 0 , Σ x_i = 1 }.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(p: np.ndarray) -> float:
    """
    Maximum entry of the full convolution of a probability vector with itself.
    """
    return np.max(np.convolve(p, p))


# ----------------------------------------------------------------------
def _gradient_step(p: np.ndarray, n: int, step: float) -> np.ndarray:
    """
    One (projected) sub‑gradient step that reduces the current maximal
    convolution entry.
    """
    conv = np.convolve(p, p)
    k_max = int(np.argmax(conv))

    # gradient of the active convolution entry:
    #  g_i = 2 * p_{k_max-i}  (if the index exists)
    g = np.zeros_like(p)
    low = max(0, k_max - (n - 1))
    high = min(k_max, n - 1)
    idx = np.arange(low, high + 1)
    g[idx] = 2.0 * p[k_max - idx]

    # take a step opposite to the gradient and project onto the simplex
    p_new = p - step * g
    return _simplex_projection(p_new)


# ----------------------------------------------------------------------
def _optimize_distribution(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_restarts: int = 4,
    max_iter: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Return the probability vector (sum = 1) that gives the lowest
    `2·n·max(p*p)` seen before the deadline.
    """
    best_p = None
    best_obj = float("inf")

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # ----- initialise -------------------------------------------------
        if restart == 0:
            p = np.full(n, 1.0 / n)               # uniform start
        elif restart == 1:
            # geometric start – r tuned to n
            r = math.exp(-1.0 / (0.4 * n))          # roughly 0.99 for n≈200
            a = np.power(r, np.arange(n, dtype=np.float64))
            p = a / a.sum()
        else:
            # random Dirichlet (alpha=0.5 -> sparse)
            a = rng.gamma(0.5, 1.0, size=n)
            p = a / a.sum()

        # ----- sub‑gradient descent ---------------------------------------
        step0 = 0.04                                 # initial step size
        for it in range(max_iter):
            if time.monotonic() >= deadline:
                break

            obj = 2.0 * n * _conv_max(p)              # current objective
            if obj < best_obj:
                best_obj = obj
                best_p = p.copy()
                print(f"[grad] n={n} restart={restart} it={it} obj={obj:.6f}")

            # schedule: linear decay, never let step become zero
            step = max(1e-6, step0 * (1.0 - it / max_iter))
            p = _gradient_step(p, n, step)

            # occasional stochastic jitter to escape flat regions
            if it % 250 == 0 and it > 0:
                noise = rng.normal(scale=0.01, size=n)
                p = _simplex_projection(p + noise)

    return best_p, best_obj


# ----------------------------------------------------------------------
def _hill_climb(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Lightweight projected simulated‑annealing hill‑climber that keeps
    the ℓ¹‑norm fixed.  This is the same routine used in the reference
    solution and works well as a final polishing step.
    """
    n = seq.shape[0]
    best_f = seq.copy()
    best_sc = _score(best_f)
    cur_f = seq.copy()
    cur_sc = best_sc

    temperature = 0.05
    step = max(1e-8, 0.02 * target_sum / n)   # initial movement size

    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_f[i] <= 0:
            continue

        delta = min(cur_f[i] * 0.03, step)
        if delta <= 0:
            continue

        cand = cur_f.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue

        cand_sc = _score(cand)

        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_f, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_f, best_sc = cand.copy(), cand_sc
                print(f"[hill] n={n} new best={best_sc:.6f}")

        temperature = max(1e-6, temperature * 0.9995)
        step = max(1e-12, step * 0.9999)

    return best_f


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a step‑function (list of non‑negative heights) whose score
    is as small as possible within the allotted time.  The routine tries
    several problem sizes, runs a deterministic gradient optimiser
    followed by a short hill‑climber, and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # give a small safety margin for the outer harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try a handful of lengths; larger n gives a lower theoretical limit
    # but also makes each iteration slower.  The range 150‑400 works well.
    for n in (150, 200, 250, 300, 350, 400):
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ----- deterministic optimisation --------------------------------
        p_opt, _ = _optimize_distribution(
            n=n,
            deadline=deadline,
            rng=rng,
            max_restarts=5,
            max_iter=3000,
        )
        if p_opt is None:
            continue

        # scale to the required ℓ¹‑norm
        f = p_opt * target_sum
        f = np.clip(f, 0.0, 1000.0)

        # ----- final polishing with hill‑climber -------------------------
        f = _hill_climb(f, target_sum, deadline, rng)

        sc = _score(f)
        if sc < best_score:
            best_score, best_seq = sc, f.copy()
            print(f"[global] n={n} improved best score = {best_score:.6f}")

        # early exit if we already beat the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: uniform distribution (always feasible)
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] uniform sequence used, score =", best_score)

    # Return a plain Python list of floats for the external harness
    return [float(x) for x in best_seq]
