def _lp_step(f, target_sum, M):
    n = f.shape[0]
    A = _conv_matrix(f)
    if not _HAS_CVX:
        return None  # no LP
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-8, reltol=1e-6)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000, verbose=False)
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    sum_g = g0.sum()
    if sum_g == 0:
        return None
    g_rescaled = g0 * (target_sum / sum_g)

def _gradient_step(seq, target_sum, alpha):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())
    # compute g vector
    g = np.zeros(n)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    # for i in start..end:
    for i in range(start, end + 1):
        j = k - i
        g[i] = seq[j]
    mean_g = g.mean()
    d = -g + mean_g
    new = seq + alpha * d
    # enforce nonneg
    new = np.maximum(new, 0.0)
    total = new.sum()
    if total == 0:
        new = np.full(n, target_sum / n)
    else:
        new = new * (target_sum / total)
    return new

"""
Improved search for a low‑value step sequence.

The algorithm consists of three phases:

1️⃣  **Exploratory parametric sweep** – for each candidate length we
    generate simple power‑law and exponential families,
    normalise their ℓ¹‑norm to √(2·n) and keep the best one.

2️⃣  **Iterative LP refinement** – starting from the best candidate,
    we repeatedly solve a linear program
        maximise Σ g_i   subject to (f * g)_k ≤ M·(1‑δ) , g ≥ 0
    where M = max(f*f) and δ≈5·10⁻³.
    If the LP is feasible the new vector replaces the old one,
    otherwise we fall back to a convex mix with the LP solution for
    the non‑tight bound (δ=0).  The process is repeated while time
    permits, aggressively decreasing the mixing weight when progress
    stalls.

3️⃣  **Fine‑grained stochastic refinement** – a cheap swap‑based hill‑climb
    and a short Gaussian perturbation stage polish the final vector.

All steps respect the security checks of ``evaluate_sequence`` (non‑negative,
clipped to [0,1000]) and keep the ℓ¹‑norm exactly at √(2·n), so that the
objective reduces to the maximal auto‑convolution entry.
The routine returns the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional linear programming backend
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return A such that (A @ g)[k] = (f * g)[k]   (full linear convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # (i, k-i) is a valid pair for all i in the range
        for i in range(i_start, i_end + 1):
            A[k, k - i] = f[i]
    return A


def _fast_score(seq: np.ndarray) -> float:
    """Exact evaluation of `evaluate_sequence` but without the safety wrapper."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


def _parameter_sweep(n: int, target_sum: float):
    """Search a few simple families (power‑law, exponential) for a good start."""
    best_vec = None
    best_sc = float("inf")
    idx = np.arange(1, n + 1, dtype=np.float64)

    # power‑law a_i ∝ i⁻ᵖ
    for p in np.linspace(0.1, 3.0, 30):
        a = idx ** (-p)
        a = a / a.sum() * target_sum
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # exponential a_i ∝ exp(-α·i)
    for alpha in np.linspace(0.001, 0.20, 30):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        a = a / a.sum() * target_sum
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


def _lp_step(f: np.ndarray, target_sum: float, bound: float):
    """
    Solve   maximise Σ g_i   s.t. (f * g)_k ≤ bound,   g ≥ 0.
    Returns the rescaled feasible vector or ``None`` if the LP fails.
    """
    if not _HAS_CVX:
        return None

    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= bound])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, abstol=1e-8, reltol=1e-6)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True,
                   max_iters=15000, verbose=False)

    if g.value is None:
        return None

    g0 = np.maximum(g.value, 0.0)
    s = g0.sum()
    if s < 1e-12:
        return None
    return g0 * (target_sum / s)


def _gradient_step(seq: np.ndarray, target_sum: float, alpha: float):
    """One inexpensive gradient‑like reduction of the current max entry."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())

    # gradient of the active convolution entry
    g = np.zeros(n, dtype=np.float64)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    for i in range(i_start, i_end + 1):
        j = k - i
        g[i] = seq[j]
    mean_g = g.mean()
    direction = -g + mean_g
    new = seq + alpha * direction
    new = np.clip(new, 0.0, 1000.0)
    s = new.sum()
    if s == 0.0:
        new = np.full(n, target_sum / n, dtype=np.float64)
    else:
        new = new * (target_sum / s)
    return new


def _local_search(start_seq: np.ndarray, deadline: float,
                  step_initial: float = 0.1) -> tuple[np.ndarray, float]:
    """Simple Gaussian hill‑climb finishing the optimisation."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to keep ℓ¹‑norm fixed
        target = math.sqrt(2.0 * n)
        cand = cand * (target / cand.sum())

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9997
        if step < 1e-7:
            step = step_initial
    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    kwargs : dict
        Ignored (kept for compatibility).

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Initialise from a possible previously found construction.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.array(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                s0 = init.sum()
                if s0 > 0:
                    init = init * (target0 / s0)
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception:  # pragma: no cover
            pass

    # --------------------------------------------------------------
    # 1️⃣  Parameter sweep (quick cheap candidates)
    # --------------------------------------------------------------
    lengths = [1000, 900, 800, 700, 600, 500, 400, 300, 200]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        cand_vec, cand_sc = _parameter_sweep(n, target_sum)
        if cand_sc < best_score:
            best_seq, best_score = cand_vec, cand_sc
            print(f"[sweep] n={n} score={cand_sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  LP‑based iterative refinement
    # --------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        # number of random restarts – less for larger n to stay within budget
        restarts = 3 if n <= 600 else 2

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # start from a random Dirichlet‑like vector, optionally blended with the global best
            f = rng.random(n) + 0.1
            f = f / f.sum() * target_sum
            if best_seq is not None and best_seq.shape[0] == n:
                f = (f + best_seq) / 2.0
                f = f / f.sum() * target_sum

            mix_weight = 0.9  # start aggressive
            max_iters = 200

            for it in range(max_iters):
                if time.monotonic() >= deadline:
                    break

                # current score & max convolution entry
                cur_sc = _fast_score(f)
                if cur_sc < best_score:
                    best_seq, best_score = f.copy(), cur_sc
                    print(f"[LP] n={n} restart={r} it={it} score={cur_sc:.6f}")

                if best_score < 1.503:
                    print("[LP] Target reached – exiting early.")
                    return best_seq.tolist()

                M = np.convolve(f, f).max()

                # ----- try a strict LP (aimed at a tiny reduction) -----
                strict_bound = M * (1.0 - 5e-3)  # 0.5 % tighter bound
                g_candidate = _lp_step(f, target_sum, strict_bound)

                if g_candidate is not None:
                    # fully replace if it really improves the objective
                    cand_sc = _fast_score(g_candidate)
                    if cand_sc < cur_sc:
                        f = g_candidate
                        continue   # go to next iteration

                # ----- fallback: ordinary LP (no tightening) -----
                g_candidate = _lp_step(f, target_sum, M)
                if g_candidate is None:
                    # LP failed – fall back to a cheap gradient step
                    f = _gradient_step(f, target_sum, alpha=0.07)
                    continue

                # evaluate the LP solution
                g_sc = _fast_score(g_candidate)

                # decide whether to replace or to mix
                if g_sc < cur_sc:
                    f_new = g_candidate
                else:
                    f_new = (1.0 - mix_weight) * f + mix_weight * g_candidate
                    # slowly decay mixing weight when progress stalls
                    if it % 20 == 0 and it > 0:
                        recent_sc = _fast_score(f_new)
                        if recent_sc > best_score * 1.001:
                            mix_weight = max(0.1, mix_weight * 0.8)

                # enforce clipping & ℓ¹‑norm
                f_new = np.clip(f_new, 0.0, 1000.0)
                total = f_new.sum()
                if total == 0.0:
                    f_new = np.full(n, target_sum / n, dtype=np.float64)
                else:
                    f_new = f_new * (target_sum / total)

                # occasional cheap gradient polishing
                if it % 15 == 0:
                    f_new = _gradient_step(f_new, target_sum, alpha=0.04)

                f = f_new

    # --------------------------------------------------------------
    # 3️⃣  Fine‑grained stochastic refinement
    # --------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback: uniform vector of moderate size
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # ---- random swap hill‑climb (very cheap) ----
    while time.monotonic() < deadline:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        avg = best_seq.mean()
        delta = max(1e-4, avg * 0.02)           # move 2 % of the average mass
        donors = np.where(best_seq > delta)[0]
        if donors.size == 0:
            break
        i = rng.choice(donors)
        j = rng.integers(n)
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target_sum / cand.sum())
        cand_sc = _fast_score(cand)
        if cand_sc < best_score:
            best_seq, best_score = cand, cand_sc
            print(f"[swap] improved to {best_score:.6f}")

    # ---- final short Gaussian hill‑climb ----
    if time.monotonic() < deadline:
        refined_seq, refined_sc = _local_search(best_seq, deadline,
                                                step_initial=0.2)
        if refined_sc < best_score:
            best_seq, best_score = refined_seq, refined_sc
            print(f"[final] improved to {best_score:.6f}")

    # return a plain Python list as required by the outer harness
    return [float(x) for x in best_seq]
