"""
Gradient‑descent + local polishing step‑function optimizer.

The score returned by `evaluate_sequence` is, after normalising the L1‑norm
to √(2 n), exactly the maximal entry of the auto‑convolution `a*a`.
Therefore we try to minimise  max(a*a)  under the constraints
a_i ≥ 0 and  Σ a_i = √(2 n).

* **Smooth‑max gradient descent** – we replace the hard max by a
  soft‑max (log‑sum‑exp) with a temperature α.  The gradient of the
  smooth‑max is a simple correlation, which can be computed with
  `np.correlate`.  A momentum term and an exponential schedule for α
  (slowly sharpening the approximation) guide the search.

* **Random restarts** – for each problem size several independent
  starts are performed.  If the global variable `height_sequence_1`
  (a previously found construction) is available it is used as a
  warm‑start for the matching length.

* **Final stochastic walk** – after the gradient phase a cheap
  “transfer‑mass” local search (the same idea as the previous
  baseline) tries to polish the best sequence.

The routine respects the 1000 s budget (a small safety margin is kept)
and returns the best list of non‑negative floats found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global random generator – will be initialised in `propose_candidate`.
_rng = None


def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq))


def _smooth_weights(conv: np.ndarray, alpha: float) -> tuple[np.ndarray, float]:
    """
    Compute soft‑max weights for the convolution vector `conv`.

    Returns (weights, max_value) where `weights` sum to 1 and concentrate
    on the entries close to the (hard) maximum when `alpha` is large.
    """
    max_val = float(conv.max())
    # Numerical stability: subtract the maximum before exponentiating.
    shifted = conv - max_val
    # alpha may be large; exponentiate safely.
    exps = np.exp(alpha * shifted)
    weight_sum = exps.sum()
    # The sum is always >= 1 because the term at the maximum is 1.
    weights = exps / weight_sum
    return weights, max_val


def _gradient_descent_optimize(
    n: int,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 12000,
    init_f: np.ndarray | None = None,
) -> tuple[np.ndarray | None, float]:
    """
    Gradient‑descent minimisation of the smooth‑max of the auto‑convolution.

    Returns (best_sequence, best_score) found during the run.
    """
    # initial feasible vector
    if init_f is not None:
        f = np.asarray(init_f, dtype=np.float64).copy()
        # protect against pathological inputs
        if f.shape != (n,):
            f = rng.random(n)
    else:
        f = rng.random(n) + 1e-12  # avoid exact zeroes

    f = np.maximum(f, 0.0)
    f_sum = f.sum()
    if f_sum <= 0.0:
        f = rng.random(n) + 1e-12
        f_sum = f.sum()
    f = f * (target_sum / f_sum)

    best_seq = None
    best_score = float("inf")

    # hyper‑parameters
    lr = 1.0                     # initial learning rate
    alpha = 5.0                  # start with a smooth approximation
    beta = 0.9                   # momentum coefficient
    momentum = np.zeros_like(f)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        weights, max_val = _smooth_weights(conv, alpha)

        # keep the best seen so far (hard max is the true score)
        if max_val < best_score:
            best_score = max_val
            best_seq = f.copy()
            print(f"[GD] n={n:3d} it={it:5d} score={best_score:.6f}")

        # gradient of the smooth max w.r.t. f:
        #   ∇_f  (softmax) = 2 * (weights ⋆ f)   (correlation)
        grad = 2.0 * np.correlate(weights, f, mode='valid')
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # gradient step
        f = f - lr * momentum
        # enforce non‑negativity
        f = np.maximum(f, 0.0)

        # renormalise to the required ℓ¹‑norm
        s = f.sum()
        if s <= 0.0:
            # re‑initialise if we accidentally collapsed to zero
            f = rng.random(n) + 1e-12
            s = f.sum()
        f = f * (target_sum / s)

        # simple schedules
        lr *= 0.9999               # slowly decay the step size
        alpha = min(alpha * 1.001, 50.0)   # sharpen the soft‑max gradually

    return best_seq, best_score


def _local_search(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic transfer‑mass walk that preserves the total ℓ¹‑mass.
    Only non‑worsening moves are kept (pure hill‑climbing).
    """
    best = seq.copy()
    best_score = _score(best.tolist())
    n = best.shape[0]

    # a modest temperature that slowly cools
    temp = 0.05
    while time.monotonic() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose a relative transfer based on the source magnitude
        delta = _rng.normal(loc=0.0, scale=0.02) * max(1e-12, best[i])
        if delta == 0.0:
            continue
        cand_i = best[i] + delta
        cand_j = best[j] - delta
        if cand_i < 0.0 or cand_j < 0.0:
            continue
        cand = best.copy()
        cand[i] = cand_i
        cand[j] = cand_j
        sc = _score(cand.tolist())
        # accept if improvement or with a tiny probability when
        # the temperature is still non‑zero (simulated annealing flavour)
        if sc < best_score or _rng.random() < math.exp(-(sc - best_score) / max(temp, 1e-12)):
            best_score = sc
            best = cand
            print(f"[local] score={best_score:.6f}")
            # cool down a bit after a successful move
            temp *= 0.9995
        else:
            # otherwise slowly cool
            temp *= 0.9999

    return best, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Main entry point called by the evaluation harness.

    The algorithm performs:
      1. a few gradient‑descent runs on several problem sizes,
      2. optional warm‑starts from a previously discovered construction,
      3. a final stochastic polishing walk.
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # Respect the external time limit with a safety margin.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 0) Use the previously found construction if it exists.
    # ------------------------------------------------------------------
    overall_best_seq = None
    overall_best_score = float("inf")

    prev_seq_obj = globals().get("height_sequence_1")
    if isinstance(prev_seq_obj, (list, tuple, np.ndarray)):
        try:
            prev_seq = np.asarray(prev_seq_obj, dtype=np.float64)
            if prev_seq.ndim == 1 and prev_seq.size > 0:
                sc = _score(prev_seq.tolist())
                overall_best_seq = prev_seq.copy()
                overall_best_score = sc
                print(f"[init] previous best score={sc:.6f}")
        except Exception:  # pragma: no cover
            overall_best_seq = None

    # ------------------------------------------------------------------
    # 1) Gradient‑descent phase on a range of lengths.
    # ------------------------------------------------------------------
    # The longer the sequence the larger the search space, but also the
    # lower the achievable peak seems to be.  We try a modest ladder.
    lengths = (200, 300, 400, 500, 600, 700, 800)

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # determine a sensible number of restarts for the current size
        restarts = 3 if n <= 500 else 2

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # warm‑start from the global previous best if the dimensions match
            init_vec = None
            if overall_best_seq is not None and overall_best_seq.shape[0] == n:
                init_vec = overall_best_seq.copy()

            best_seq, best_score = _gradient_descent_optimize(
                n=n,
                target_sum=target_sum,
                deadline=deadline,
                rng=_rng,
                max_iters=12000,
                init_f=init_vec,
            )

            if best_seq is not None and best_score < overall_best_score:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(f"[global] new best after n={n} restart={r}: {overall_best_score:.6f}")

            # early exit if we have already broken the target bound
            if overall_best_score < 1.45:
                break

        if overall_best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # 2) Light‑weight stochastic polishing (if time permits).
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_score = _local_search(overall_best_seq, deadline)
        if polished_score < overall_best_score:
            overall_best_score = polished_score
            overall_best_seq = polished_seq.copy()
            print(f"[global] after polishing: {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback – deterministic constant sequence.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 256
        overall_best_seq = np.full(n, 1.0, dtype=np.float64)
        overall_best_score = _score(overall_best_seq.tolist())
        print("[fallback] using deterministic constant sequence")

    # Final safety clipping (the evaluator also does this).
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    # Return a plain Python list of floats as required.
    return [float(x) for x in overall_best_seq]
