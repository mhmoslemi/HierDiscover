    deadline = time.monotonic() + budget_s - 1.0
    best_seq = None
    best_score = float('inf')
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    for n in lengths:
        # check time remaining
        if time.monotonic() >= deadline:
            break
        print(f"[MAIN] Optimizing length n={n}")
        f = _initial_f(n, rng) # create.
        target_norm = math.sqrt(2.0 * n)
        # we could run outer iteration loops until time or small fixed limit.
        outer_limit = 50  # may adjust.
        for outer in range(outer_limit):
            if time.monotonic() >= deadline:
                break
            conv = np.convolve(f, f)
            cur_max = conv.max()
            # solve LP
            A = np.zeros((2*n - 1, n), dtype=np.float64)
            # fill A: each column j adds f as described.
            for j in range(n):
                A[j:j+n, j] = f
            # constraints: A @ b <= cur_max
            lhs_ub = A
            rhs_ub = np.full(2*n - 1, cur_max, dtype=np.float64)
            c = -np.ones(n, dtype=np.float64)  # maximize sum(b)
            res = linprog(c, A_ub=lhs_ub, b_ub=rhs_ub,
                          bounds=[(0, None)] * n,
                          method='highs', options={'presolve': True})
            if not res.success:
                print("[LP] failed:", res.message)
                # fallback: break outer loop
                break
            b = res.x
            sum_b = b.sum()
            if sum_b <= 1e-12:
                break
            g = b * (target_norm / sum_b)
            # blending
            t = 0.2
            f = (1 - t) * f + t * g
            # renormalize (account for rounding)
            f = np.clip(f, 0.0, 1000.0)

"""
Propose a step‑function sequence that minimises the supplied
`evaluate_sequence` score.

The algorithm is a *deterministic refinement* based on the linear‑program
idea from the statement, followed by a short simulated‑annealing polish.
For each chosen length `n` we

1. initialise a feasible non‑negative vector with ℓ¹‑norm √(2·n).
2. repeat a bounded number of *LP‑steps*:
   * compute `M = max(f*f)`;
   * solve the LP  max Σ b_j  s.t.   f * b ≤ M  (entry‑wise) ,  b ≥ 0;
   * rescale the optimal `b` to the required ℓ¹‑norm and blend it into `f`
     ( `f ← (1‑t)·f + t·g` with `t = 0.2` );
   * re‑normalise `f` to ℓ¹‑norm √(2·n).
   The LP is tiny (≤ 2 n‑1 constraints, n variables) and solved with
   SciPy's HiGHS driver.  Each step strictly reduces the maximum
   auto‑convolution, i.e. the evaluation value.

3. keep the best vector over all tested lengths.

When the time budget is almost exhausted a short simulated‑annealing
routine (similar to the one used previously) performs a final local
search starting from the best vector found.

All numbers are clipped to the interval [0, 1000] before returning.
"""

import math
import time
import numpy as np

# SciPy is optional – if unavailable we fall back to pure SA.
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def _normalize(f: np.ndarray) -> np.ndarray:
    """Scale `f` so that its ℓ¹‑norm equals √(2·len(f))."""
    n = len(f)
    target = math.sqrt(2.0 * n)
    total = f.sum()
    if total == 0.0:
        # avoid division by zero – give a uniform tiny mass
        f = np.full(n, 1.0 / n, dtype=np.float64)
        total = 1.0
    f = f / total * target
    # safety clipping (required by the scorer)
    f = np.clip(f, 0.0, 1000.0)
    return f


def _initial_f(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial feasible vector of length `n`.

    Preference is given to the global `height_sequence_1` if it exists.
    Otherwise a random exponential vector is used.
    """
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=np.float64)
        if len(seq) < n:
            seq = np.concatenate([seq, np.zeros(n - len(seq), dtype=np.float64)])
        elif len(seq) > n:
            seq = seq[:n].copy()
    else:
        seq = rng.exponential(scale=1.0, size=n)

    seq = np.clip(seq, 0.0, 1000.0)
    return _normalize(seq)


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    One LP refinement step.

    Solve   max Σ b_j   s.t.   f * b ≤ M ,  b ≥ 0.
    Return the blended vector   (1‑t)·f + t·g   with g = scaled b.
    """
    n = len(f)
    # Build the (2n‑1) × n convolution matrix A where
    # (A @ b)[k] = Σ_i f_i·b_{k‑i}  (out‑of‑range indices give 0).
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contributes f_i to row i+j
    for j in range(n):
        A[j:j + n, j] = f

    # Linear program: maximise Σ b_j  ⇔ minimise -Σ b_j
    c = -np.ones(n, dtype=np.float64)
    ub = np.full(rows, M, dtype=np.float64)

    if linprog is None:                     # pragma: no cover
        # SciPy not present – skip the LP step.
        return f.copy()

    res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # Should never happen because b = f is feasible.
        return f.copy()

    b = res.x
    sum_b = b.sum()
    if sum_b == 0.0:
        return f.copy()

    # Rescale to the required ℓ¹‑norm.
    target_norm = math.sqrt(2.0 * n)
    g = b * (target_norm / sum_b)

    # Blend and renormalise.
    t = 0.2
    new_f = (1.0 - t) * f + t * g
    new_f = _normalize(new_f)
    return new_f


def _sa_refine(f: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short simulated‑annealing polishing using the same move set as the
    original code, but always starting from the supplied vector.
    """
    n = len(f)
    target_norm = math.sqrt(2.0 * n)

    # Pre‑compute the full convolution once; we will recompute it after each move,
    # because `n` is modest (≤ 1000) and the remaining time is short.
    conv = np.convolve(f, f)
    cur_max = conv.max()
    best_f = f.copy()
    best_max = cur_max

    start = time.monotonic()
    total = max(deadline - start, 1e-3)

    it = 0
    rng = np.random.default_rng()
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        # exponential cooling from 1 to 1e‑4
        T = 1.0 * (1e-4 / 1.0) ** (elapsed / total)

        # index of the worst convolution entry
        k_max = int(np.argmax(conv))

        # gradient of the worst entry w.r.t. each component of f:
        #   ∂(f*f)[k_max]/∂f[i] = 2·f[k_max‑i]  if the index exists
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad = np.zeros(n, dtype=np.float64)
        grad[idx_range] = 2.0 * f[k_max - idx_range]

        # donor: probability proportional to the gradient (large contribution)
        prob = grad + 1e-12
        prob /= prob.sum()
        i = rng.choice(n, p=prob)

        # destination: uniform random different index
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # amount of mass to move
        max_delta = f[i]
        if max_delta <= 1e-12:
            continue
        frac = 0.2 * (1.0 - elapsed / total) + 0.01
        delta = rng.random() * max_delta * frac
        if delta <= 0.0:
            continue

        # apply the move
        f[i] -= delta
        f[j] += delta

        # re‑compute convolution (fast enough for the short final phase)
        conv = np.convolve(f, f)
        new_max = conv.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob_up = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob_up:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f.copy()
        else:
            # rollback
            f[i] += delta
            f[j] -= delta

        it += 1

    # final normalisation (mass is conserved, but rounding may have drifted)
    best_f = _normalize(best_f)
    return best_f


def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a low‑value step‑function sequence within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate wall‑clock budget (seconds).  The routine stops a
        second before the limit to guarantee that a result is returned.

    Returns
    -------
    list[float]
        The best sequence discovered, ready for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    best_seq = None
    best_score = float("inf")

    # A handful of lengths – larger n gives more freedom but costs more.
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        print(f"[MAIN] Optimising length n={n}")

        f = _initial_f(n, rng)

        # Outer refinement – a limited number of LP steps.
        outer_no_improve = 0
        max_outer = 60  # safety cap
        for outer in range(max_outer):
            if time.monotonic() >= deadline:
                break

            conv = np.convolve(f, f)
            cur_max = conv.max()
            cur_score = cur_max          # because sum = √(2n) → evaluation = max

            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[BEST] n={n} outer={outer} eval={cur_score:.6f}")
                outer_no_improve = 0
            else:
                outer_no_improve += 1

            # stop early if we have stalled
            if outer_no_improve >= 10:
                break

            # one LP refinement step
            f = _lp_step(f, cur_max)

        # keep the best version of this length (already stored in best_seq)

    # ------------------------------------------------------------------
    # Final simulated‑annealing polish (if any time remains).
    if best_seq is not None and time.monotonic() < deadline - 2.0:
        remaining = deadline - time.monotonic()
        print("[POST] SA polish – remaining time:", remaining)
        polished = _sa_refine(best_seq.copy(), deadline)
        # evaluate polished version
        polished_score = float(np.convolve(polished, polished).max())
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished.copy()
            print(f"[POLISH] improved to eval={polished_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety conversion to plain Python floats, respecting scorer limits.
    if best_seq is None:
        # fallback – uniform 200‑vector (should never happen)
        best_seq = np.full(200, 1.0, dtype=np.float64)
        best_seq = _normalize(best_seq)

    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # One last evaluation for reporting (ignore possible exceptions).
    try:
        final_eval = evaluate_sequence(final_seq)
        print("[FINAL] best evaluation =", final_eval)
    except Exception as exc:  # pragma: no cover
        print("[FINAL] evaluation failed:", exc)

    return final_seq
