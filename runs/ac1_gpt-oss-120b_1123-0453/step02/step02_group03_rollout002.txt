def improve_by_gradient(seq, target_sum, steps, rng):
    for _ in range(steps):
        conv = np.convolve(seq, seq)
        max_val = np.max(conv)
        kmax = np.argmax(conv)
        # compute contributions
        contrib = np.zeros_like(seq)
        # For each i, if j = kmax - i in range, add seq[i] * seq[j]
        for i in range(len(seq)):
            j = kmax - i
            if 0 <= j < len(seq):
                contrib[i] += seq[i] * seq[j]

        # Determine indices with highest contributions

        src = rng.choice(idxs)
        # choose random target among other indices
        possible_targets = np.arange(len(seq))
        possible_targets = possible_targets[possible_targets != src]
        tgt = rng.choice(possible_targets)

    max_val = conv.max()

    contrib = np.zeros(n)
    # For each i, j = k - i.
    # Only if 0 <= j < n.

    idx_j = k - idx_i
    valid = (idx_j >=0) & (idx_j < n)

assistantfinal**Solution Overview**

The goal is to minimise the score returned by `evaluate_sequence`.  
For a sequence `a` with ℓ¹‑norm `‖a‖₁ = √(2·n)` the score simplifies to  

```
score = max ( a * a )                     # a * a = discrete autocorrelation
```

Hence we have to **reduce the maximal autocorrelation value** while keeping the sum
fixed.  
The programme below uses a **simulated‑annealing optimiser** that works directly on
the autocorrelation:

* the autocorrelation is obtained by an FFT‑based convolution (`O(n log n)`),  
* at each step the lag `k` with the current maximal value is identified,  
* the contribution of every index to this lag is `a[i]·a[k‑i]`.  
  A source index with a large contribution donates a small amount of its mass to a
  target index with a small contribution – this directly attacks the peak,
* the move is accepted according to a temperature schedule (Metropolis rule).  

The optimiser is run several times on different lengths (`n∈{200,300,400,600,800,1000}`) and
with different random initialisations (uniform, exponential, exponential‑draw).
The best sequence found within the 1000 s budget is returned.

The code contains only top‑level functions, respects the required
`propose_candidate` signature and prints progress information.

```python
"""
Simulated‑annealing optimiser for the step‑function problem.

The score (for a normalised sequence with ℓ¹‑norm = √(2·n))
equals the maximal autocorrelation value

    max_k ( (a * a)[k] ) .

We therefore minimise this maximum while keeping the sum fixed.
The optimiser repeatedly:

1. evaluates the autocorrelation via a fast FFT,
2. selects the lag k that attains the current maximum,
3. computes for every index i the contribution c_i = a[i]·a[k‑i],
4. moves a small amount of mass from an index with large c_i to an index
   with small c_i,
5. accepts the move with a Metropolis rule (temperature decays with time).

Multiple restarts with different lengths (200 … 1000) and different
initial shapes are performed until the overall time budget (≈1000 s) is
exhausted.  The best sequence found is returned as a plain list of
floats (all entries ≥ 0, ≤ 1000).
"""

def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Autocorrelation a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = seq.shape[0]
    # Next power of two ≥ 2·n‑1 for efficient FFT.
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)
    return conv[: 2 * n - 1]


def _max_autocorr(seq: np.ndarray) -> float:
    """Maximum entry of the autocorrelation."""
    conv = _conv_fft(seq)
    return float(conv.max())


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative start vector of length n.
    Three families are mixed: uniform, deterministic exponential,
    and exponential‑draw (Dirichlet‑like).  The vector is normalised
    to ℓ¹‑norm = sqrt(2·n).
    """
    choice = rng.integers(3)
    if choice == 0:
        seq = np.full(n, 1.0, dtype=np.float64)
    elif choice == 1:
        # deterministic exponential decay with a random rate
        rate = rng.uniform(0.4, 2.5)
        seq = np.exp(-rate * np.arange(n, dtype=np.float64))
    else:
        # exponential draw – yields a Dirichlet‑like random vector
        seq = rng.exponential(scale=1.0, size=n) + 1e-12

    # keep everything non‑negative and normalise the ℓ¹‑norm
    target_sum = math.sqrt(2.0 * n)
    seq = seq * (target_sum / seq.sum())
    return seq


def _propose_move(seq: np.ndarray,
                  target_sum: float,
                  rng: np.random.Generator,
                  temperature: float) -> tuple[np.ndarray, float, bool]:
    """
    One neighbourhood move for SA.

    * Find the lag k where the autocorrelation is maximal.
    * Compute per‑index contributions c_i = a[i]·a[k‑i].
    * Choose a source index among the largest contributors,
      a target among the smallest, and shift a random amount of mass.
    * Return the (possibly) updated sequence, its new max autocorrelation
      and a flag telling whether the move was accepted.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    max_val = float(conv.max())
    # indices where the peak occurs (allow a tiny tolerance)
    ks = np.where(conv >= max_val * 0.999999)[0]
    k = rng.choice(ks)

    # contributions for this lag
    contrib = np.zeros(n, dtype=np.float64)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    j_idx = k - i_idx
    contrib[i_idx] = seq[i_idx] * seq[j_idx]

    # pick source (large contribution) and target (small contribution)
    # at least one element must be selectable
    top_frac = 0.15
    low_frac = 0.15
    n_top = max(1, int(top_frac * n))
    n_low = max(1, int(low_frac * n))

    src_cand = np.argpartition(contrib, -n_top)[-n_top:]
    src = int(rng.choice(src_cand))

    tgt_cand = np.argpartition(contrib, n_low)[:n_low]
    # ensure source ≠ target
    if tgt_cand.size > 1:
        tgt_cand = tgt_cand[tgt_cand != src]
    tgt = int(rng.choice(tgt_cand))

    # amount to transfer – at most half of the source value
    max_transfer = seq[src] * 0.5
    if max_transfer <= 0.0:
        # nothing to move – treat as a rejected move
        return seq, max_val, False
    delta = rng.random() * max_transfer

    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[tgt] += delta
    # safety clipping before renormalisation
    new_seq = np.clip(new_seq, 0.0, 1000.0)

    # renormalise the ℓ¹‑norm (the score is homogeneous, any scaling works)
    cur_sum = new_seq.sum()
    if cur_sum > 0.0:
        new_seq = new_seq * (target_sum / cur_sum)

    new_max = _max_autocorr(new_seq)

    # Metropolis acceptance
    if new_max < max_val:
        accept = True
    else:
        prob = math.exp(-(new_max - max_val) / max(temperature, 1e-12))
        accept = rng.random() < prob

    if accept:
        return new_seq, new_max, True
    else:
        return seq, max_val, False


def _sa_optimize(start_seq: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing optimiser for a single problem size.

    Parameters
    ----------
    start_seq : np.ndarray
        Initial vector (already normalised to the required ℓ¹‑norm).
    target_sum : float
        Desired ℓ¹‑norm (√(2·n)); the optimiser keeps this value.
    deadline : float
        Absolute time after which the optimiser must stop.
    rng : np.random.Generator
        Random number generator.

    Returns
    -------
    best_seq, best_score
        The best sequence discovered and its maximal autocorrelation.
    """
    cur_seq = start_seq.copy()
    cur_score = _max_autocorr(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score

    # temperature schedule (geometric decay)
    temp_start = 0.3
    temp_end = 5e-7
    start_time = time.monotonic()

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        if total <= 0.0:
            break
        frac = elapsed / total
        temperature = temp_start * (temp_end / temp_start) ** frac

        cur_seq, cur_score, _ = _propose_move(cur_seq, target_sum, rng, temperature)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[SA] new best max {best_score:.6f} (n={len(start_seq)})")

    return best_seq, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low evaluation
    score.  The routine respects the time budget, tries several lengths,
    performs simulated‑annealing on each, and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # list of problem sizes to explore (larger n offers more freedom)
    sizes = [200, 300, 400, 600, 800, 1000]

    # --------------------------------------------------------------
    # 1️⃣  Try to improve a previously supplied construction, if any.
    # --------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        try:
            seq0 = np.asarray(global_seq, dtype=np.float64)
            n0 = len(seq0)
            target0 = math.sqrt(2.0 * n0)
            seq0 = np.clip(seq0, 0.0, 1000.0)
            seq0 = seq0 * (target0 / max(seq0.sum(), 1e-12))
            sub_dead = min(deadline, time.monotonic() + 30.0)
            seq_opt, sc_opt = _sa_optimize(seq0, target0, sub_dead, rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = seq_opt.copy()
                print(f"[init] improved existing height_sequence_1 → {best_score:.6f}")
        except Exception:
            pass  # ignore malformed global variable

    # --------------------------------------------------------------
    # 2️⃣  Main multi‑size SA loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a size (cycling through the list gives a balanced exploration)
        n = rng.choice(sizes)

        target_sum = math.sqrt(2.0 * n)
        init_seq = _initial_sequence(n, rng)

        # give each SA run a time slice of ~20 s (adjusted if we are close to the end)
        remaining = deadline - time.monotonic()
        run_time = min(20.0, remaining * 0.6)  # leave some slack for later runs
        if run_time < 0.5:
            break
        run_deadline = time.monotonic() + run_time

        seq_opt, sc_opt = _sa_optimize(init_seq, target_sum, run_deadline, rng)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = seq_opt.copy()
            print(f"[global] new best score {best_score:.6f} (n={n})")

    # --------------------------------------------------------------
    # Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best_score = _max_autocorr(best_seq)
        print("[fallback] uniform start used")

    # Final sanity check using the supplied scorer (optional, cheap)
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best score reported by evaluator = {final_score:.6f}, "
          f"length = {len(best_seq)}")

    # Return as plain Python list of floats
    return [float(x) for x in best_seq]
