def auto_convolve(a):
    n = a.shape[0]
    n_fft = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, n_fft)
    conv = np.fft.irfft(fft_a * fft_a, n_fft)[:2*n-1]
    return conv

def convolve_full(x, y):
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n).bit_length())
    fft_x = np.fft.rfft(x, n_fft)
    fft_y = np.fft.rfft(y, n_fft)
    conv = np.fft.irfft(fft_x * fft_y, n_fft)[:n]
    return conv

import math, time, numpy as np

def _full_convolution(x, y):
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n_fft)
    fy = np.fft.rfft(y, n_fft)
    conv = np.fft.irfft(fx * fy, n_fft)
    return conv[:n]

def _auto_convolve(a):
    return _full_convolution(a, a)

def _project_simplex(v, target_sum):
    # Duchi et al. algorithm
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    # shift by max to improve numerical stability?
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _gradient_descent_optimize(n, rng, deadline, init_seq=None):
    target = math.sqrt(2.0 * n)
    # initialise
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        a = a * (target / np.sum(a))
    else:
        # random point on simplex
        x = rng.random(n) + 0.1
        a = x / np.sum(x) * target

    best_a = a.copy()
    best_M = np.max(_auto_convolve(a))

    alpha = 0.5
    alpha_max = 5.0
    step = target * 0.01  # initial additive step size
    step_min = target * 1e-6
    it = 0
    max_iters = 10000

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _auto_convolve(a)
        M = conv.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
            print(f"[gd] n={n} it={it} M={M:.6f}")

        # ----- gradient of smooth max -----
        # smooth max via softmax with temperature alpha
        shift = conv - np.max(conv)  # stability
        w = np.exp(alpha * shift)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # compute gradient: 2 * correlation(w, a)
        rev_a = a[::-1]
        corr = _full_convolution(w, rev_a)  # length 3n-2
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        improved = False

        for _ in range(8):

            cand = _project_simplex(cand, target)

"""
Gradient‑descent search for a low‑value step sequence.

The score returned by ``evaluate_sequence`` reduces (when the ℓ¹‑norm is
fixed to √(2·n)) to the maximal entry of the auto‑convolution.
The algorithm works directly on that quantity:

* fix a length n and the target ℓ¹‑norm = √(2·n)
* start from a random point on the simplex (or from a known good
  construction)
* repeatedly minimise a *smooth* approximation of the maximum
  (soft‑max with a temperature α) by an additive gradient step.
  The gradient of the smooth max is a correlation that can be computed
  by FFTs in O(n log n).
* after each step the vector is projected back onto the non‑negative
  simplex (projection onto the ℓ¹‑simplex).
* a simple back‑tracking line‑search adapts the step size; occasional
  random perturbations help escaping plateaus.
* the procedure is run for several lengths and random restarts within
  the allotted time.  The best sequence found is returned.

No external files, network access or closures are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (full linear convolution)
def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x∗y (length = len(x)+len(y)-1)."""
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n_fft)
    fy = np.fft.rfft(y, n_fft)
    conv = np.fft.irfft(fx * fy, n_fft)
    return conv[:n]

def _auto_convolve(a: np.ndarray) -> np.ndarray:
    """Auto‑convolution of a with itself (full size 2·len(a)-1)."""
    return _full_convolution(a, a)


# ----------------------------------------------------------------------
# Projection onto the non‑negative simplex {x≥0, Σx = target}
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = target}."""
    if target < 0:
        raise ValueError("target must be non‑negative")

    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)

    # Find ρ = max{i | u_i > cssv_i / i}
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]                 # last true index
        theta = cssv[rho] / (rho + 1)
    else:
        # Fallback (numerical safety): distribute uniformly
        theta = (np.sum(v) - target) / len(v)

    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation for a single length n
def _gradient_descent_optimize(
    n: int,
    rng: np.random.Generator,
    deadline: float,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max gradient descent on the max auto‑convolution.
    Returns the best sequence found and its max‑convolution value.
    """
    target = math.sqrt(2.0 * n)

    # ---- initialise a feasible point on the simplex ----
    if init_seq is not None:
        a = np.asarray(init_seq, dtype=np.float64)
        a = a * (target / np.sum(a))
    else:
        # random point on the simplex (slightly perturbed to avoid exact zeros)
        x = rng.random(n) + 0.1
        a = x / np.sum(x) * target

    best_a = a.copy()
    best_M = np.max(_auto_convolve(a))

    # hyper‑parameters
    alpha = 0.5            # temperature for the smooth max
    alpha_max = 5.0
    step = target * 0.01   # initial additive step size
    step_min = target * 1e-6
    max_iters = 10000
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _auto_convolve(a)
        M = conv.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
            print(f"[gd] n={n} it={it} M={M:.6f}")

        # ----- smooth‑max weights (soft‑max) -----
        shift = conv - np.max(conv)                 # for numerical stability
        w = np.exp(alpha * shift)
        w_sum = np.sum(w)
        if w_sum == 0:                              # extremely unlikely
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # ----- gradient of the smooth max -----
        # grad = 2 * correlation(w, a)
        rev_a = a[::-1]
        corr_full = _full_convolution(w, rev_a)    # length 3·n‑2
        start = n - 1
        grad = 2.0 * corr_full[start:start + n]

        # ----- back‑tracking line search (additive) -----
        improved = False
        step_try = step
        for _ in range(8):                          # at most 8 attempts
            cand = a - step_try * grad
            cand = _project_simplex(cand, target)
            cand_M = np.max(_auto_convolve(cand))
            if cand_M < M:
                a = cand
                step = step_try * 1.1               # modestly increase step
                improved = True
                break
            step_try *= 0.5
        if not improved:
            # small stochastic perturbation to escape shallow minima
            noise = rng.normal(scale=0.005 * target / n, size=n)
            a = a + noise
            a = _project_simplex(a, target)
            step = max(step * 0.9, step_min)

        # ----- update temperature -----
        alpha = min(alpha_max, alpha * 1.001)

        # occasional random‑restart of a tiny fraction (helps for very long runs)
        if it % 500 == 0 and rng.random() < 0.15:
            a = _project_simplex(rng.random(n), target)

    return best_a, best_M


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score.

    The method explores several lengths (larger n give a tighter theoretical
    bound) and, for each length, runs a smooth‑max gradient descent with
    projection onto the ℓ¹‑simplex.  The best sequence observed before the
    time budget expires is returned as a plain list of floats.
    """
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.1)   # keep a tiny safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_M = float("inf")

    # lengths to try – more points give a lower theoretical limit
    lengths = [800, 600, 500, 400, 300, 200]
    runs_per_len = 3

    for n in lengths:
        if time.time() >= deadline:
            break
        for run in range(runs_per_len):
            if time.time() >= deadline:
                break

            # try to reuse a previously discovered construction if it matches n
            init_seq = None
            if "height_sequence_1" in globals():
                hs = globals()["height_sequence_1"]
                if isinstance(hs, list) and len(hs) == n:
                    init_seq = np.asarray(hs, dtype=np.float64)

            seq, M = _gradient_descent_optimize(n, rng, deadline, init_seq=init_seq)

            if M < best_M:
                best_M = M
                best_seq = seq.copy()
                print(f"[best] n={n} run={run} M={M:.6f}")

            # early termination if we have already reached the target bound
            if best_M <= 1.503:
                print("[target reached]")
                deadline = time.time()                # stop any further work
                break

    # Fallback – uniform distribution (should never happen, but keeps the contract)
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    # Convert to plain Python list (evaluate_sequence expects a list)
    return [float(v) for v in best_seq]
