    # assume s > 0
    if s < 0:
        raise ValueError
    # shape v 1d
    n = v.shape[0]

    u = np.sort(v)[::-1]

"""
Subgradient-based flattening of the autocorrelation.

This method repeatedly moves mass away from the index that
currently attains the maximal convolution value.  The direction of
the sub‑gradient of the max‑convolution is computed analytically,
the step is taken and the vector is projected back onto the
simplex (non‑negative entries with the prescribed ℓ¹‑norm).  Several
random restarts and a short simulated‑annealing polishing phase
are executed within the time budget.  Deterministic shapes (sine,
quadratic, …) are also tried as cheap warm‑starts.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = s}."""
    if s < 0:
        raise ValueError("radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray):
    """Return max value of a*a and the index where it is attained."""
    # use FFT for speed; length is 2*len(a)-1
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())

    conv = np.fft.irfft(af * af)[: 2 * n - 1]

    return float(conv[idx]), idx


def _subgrad_optimize(a0: np.ndarray, target_sum: float, deadline: float,
                      rng: np.random.Generator):
    """Run a simple sub‑gradient descent on max(a*a)."""
    n = a0.shape[0]
    a = _project_simplex(a0, target_sum)

    best_a = a.copy()
    best_val, _ = _max_convolution(a)

    it = 0
    last_improve = 0
    # step size base
    base_step = target_sum / n
    while time.monotonic() < deadline and it < 20000:
        # compute max convolution and index
        max_val, k = _max_convolution(a)

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
        idx_j = k - idx_i
        g[idx_i] = 2.0 * a[idx_j]

        # diminishing step size
        eta = base_step * 0.5 / math.sqrt(it + 1.0)

        a_new = a - eta * g
        a = _project_simplex(a_new, target_sum)

        # occasional tiny random jitter to escape flat plateaus
        if it % 500 == 0 and it > 0:
            jitter = (rng.random(n) - 0.5) * (target_sum * 0.001 / n)
            a = _project_simplex(a + jitter, target_sum)

        # keep the best seen so far
        if max_val < best_val - 1e-12:
            best_val = max_val
            best_a = a.copy()
            last_improve = it

        # stop if no improvement for a while
        if it - last_improve > 4000:
            break

        it += 1

    return best_a, best_val


def _hill_climb(a0: np.ndarray, target_sum: float, deadline: float,
                rng: np.random.Generator):
    """Lightweight simulated annealing that keeps the ℓ¹‑norm fixed."""
    n = a0.shape[0]

    cur_sc = float(evaluate_sequence(cur.tolist()))
    best = cur.copy()
    best_sc = cur_sc

    temperature = 0.05
    step = max(1e-8, 0.02 * target_sum / n)

            if cand_sc < best_sc:

                print(f"[polish] new best={best_sc:.6f}")

Subgradient flattening of the autocorrelation.

The score 2·n·max(a*a)/(∑a)² is invariant under scaling, therefore we
fix ∑a = √(2·n).  With this normalisation the objective reduces to
minimise the maximum entry of the convolution a*a.  The algorithm
repeatedly:

1. computes the index k where the current convolution attains its
   maximum,
2. evaluates the (sub‑)gradient of the active entry
   (g_i = 2·a_{k-i} for the contributing i),
3. takes a step a ← a – η·g and projects back onto the non‑negative
   simplex {x≥0, ∑x = √(2n)}.

Several random restarts are performed for a range of problem sizes.
Deterministic “shape” candidates (sine, cosine, quadratic, linear)
are tried first as cheap warm‑starts.  After the sub‑gradient phase a
short simulated‑annealing hill‑climber refines the best vector.  The
whole search respects the supplied time budget (≈1000 s) and returns
the best sequence found.

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
