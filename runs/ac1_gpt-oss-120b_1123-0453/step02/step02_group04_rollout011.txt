def project_onto_simplex(v, s):
    # based on algorithm from https://arxiv.org/pdf/1101.6081.pdf
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _hill_climb(f, target_sum, deadline, rng):
    n = f.shape[0]
    # precompute n_fft for FFT
    L = 2 * n - 1
    n_fft = 1 << ((L-1).bit_length())
    # compute initial convolution and M
    F = np.fft.rfft(f, n=n_fft)
    conv = np.fft.irfft(F*F, n=n_fft)[:L]
    M = conv.max()
    best_f = f.copy()
    best_M = M

    # parameters
    patience = 0
    while time.monotonic() < deadline:
        # gradient step
        k_max = int(np.argmax(conv))
        # compute gradient inside support
        grad = np.zeros(n, dtype=np.float64)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        slice_len = i_end - i_start + 1
        if slice_len > 0:
            grad[i_start:i_end+1] = 2.0 * f[i_start:i_end+1][::-1]
        # Determine bound for step
        pos = grad > 1e-12
        if np.any(pos):
            # maximal step to keep positivity
            max_alpha = np.min(f[pos] / grad[pos])
            # try a few step fractions
            for factor in (0.5, 0.2, 0.1, 0.05):
                alpha = factor * max_alpha
                if alpha <= 0:
                    continue
                y = f - alpha * grad
                # project onto simplex
                y = _project_simplex(y, target_sum)
                # compute new conv max quickly using FFT
                Y = np.fft.rfft(y, n=n_fft)
                conv_y = np.fft.irfft(Y*Y, n=n_fft)[:L]
                M_y = conv_y.max()
                if M_y < best_M - 1e-8:   # improvement
                    best_M = M_y
                    best_f = y.copy()
                    f = y
                    conv = conv_y
                    M = M_y
                    # early exit for this iteration
                    break
        # random transfer step
        # choose i weighted by f^2 maybe, to target large contributions to M
        # We use importance sampling: probabilities prop to f
        if np.sum(f) > 0:
            i = rng.choice(n, p=f/np.sum(f))
        else:
            i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # delta proportion of f[i], limited
        delta = min(f[i] * 0.1, target_sum * 0.01)
        y = f.copy()
        y[i] -= delta
        y[j] += delta
        # enforce nonneg via projection (though direct ensures nonneg)
        # no need to project onto simplex because sum unchanged
        # compute conv of y
        Y = np.fft.rfft(y, n=n_fft)
        conv_y = np.fft.irfft(Y*Y, n=n_fft)[:L]
        M_y = conv_y.max()
        if M_y < best_M - 1e-8:
            best_M = M_y
            best_f = y.copy()
            f = y
            conv = conv_y
            M = M_y
        # occasional random perturb to escape stagnation
        if rng.random() < 0.001:
            # random Gaussian noise small
            y = f + rng.normal(scale=0.01, size=n)
            y = np.clip(y, 0.0, None)
            y = _project_simplex(y, target_sum)
            Y = np.fft.rfft(y, n=n_fft)
            conv_y = np.fft.irfft(Y*Y, n=n_fft)[:L]
            M_y = conv_y.max()
            if M_y < best_M:
                best_M = M_y
                best_f = y.copy()
                f = y
                conv = conv_y
        # optional early stop if best_M close to lower bound 1+epsilon
        if best_M < 1.003:  # near theoretical lower bound
            break
    return best_f, best_M

def _project_simplex(v, s):
    # following algorithm: https://github.com/ams3/algorithms/blob/master/ProjectionSimplex.py
    n = len(v)
    if np.sum(v) == s and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero? return zeros?
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‐value step sequence.

The algorithm is a pure gradient / simulated‑annealing search that tries to
flatten the auto‑convolution of a non‑negative sequence while keeping the
ℓ¹‑norm fixed (∑aᵢ = √(2·n)).  For a fixed norm the evaluation function reduces
to the maximum entry of the auto‑convolution, so the optimisation problem is
equivalent to minimising that peak value.

* The current best sequence (if any) is taken from the global
  ``height_sequence_1`` variable.
* It is up‑sampled to larger lengths (200 → 400 → 800 …) by linear
  interpolation; after each up‑sampling the ℓ¹‑norm is re‑normalised.
* For each length we run a hill‑climbing loop until the time budget
  expires:
    - Compute the full auto‑convolution via FFT (O(n log n)).
    - Identify the index k where the convolution is maximal.
    - Build the (sub‑)gradient of that maximal entry w.r.t. the vector a.
    - Take a projected‑gradient step (projection onto the simplex
      {x ≥ 0, ∑x = √(2·n)}) and accept the step if the peak value improves.
    - Occasionally perform random “mass‑transfer’’ moves and tiny Gaussian
      perturbations to escape plateaus.
* The best sequence found over all lengths is returned as a plain list of
  floats.  All numbers are non‑negative and far below the 1000‑limit, so the
  scorer’s clipping never interferes.
"""

import math
import numpy as np

# Helper: evaluate a sequence using the supplied scorer.
# ----------------------------------------------------------------------
def _score(seq):
    """Return the objective value for ``seq`` (list or np.ndarray)."""
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = s}
# ----------------------------------------------------------------------
def _project_simplex(v, s):
    """
    Euclidean projection of vector ``v`` onto the simplex
    {x : x_i >= 0, sum_i x_i = s}.
    """
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    n = v.shape[0]
    # Fast path: already feasible
    if np.all(v >= 0) and abs(v.sum() - s) < 1e-12:
        return v.copy()
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find rho = max { j | u_j - cssv_j / j > 0 }
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All components will be zero
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Auto‑convolution via FFT
# ----------------------------------------------------------------------
def _auto_convolution(f, n_fft):
    """Return the full linear convolution f * f (length 2·len(f)‑1)."""
    F = np.fft.rfft(f, n=n_fft)
    conv = np.fft.irfft(F * F, n=n_fft)[: 2 * f.shape[0] - 1]
    return conv


def _conv_max_and_idx(f, n_fft):
    """Maximum entry of f*f and its index."""
    conv = _auto_convolution(f, n_fft)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv


# ----------------------------------------------------------------------
# Gradient / random‑move hill climb for a fixed length
# ----------------------------------------------------------------------
def _hill_climb(f, target_sum, deadline, rng):
    """
    Perform a bounded‑time hill‑climbing minimisation of
    max(f*f) while keeping f ≥ 0 and sum(f) = target_sum.
    Returns the best vector found and its max‑convolution value.
    """
    n = f.shape[0]
    L = 2 * n - 1
    n_fft = 1 << ((L - 1).bit_length())        # next power of two

    # initial state
    conv = _auto_convolution(f, n_fft)
    best_conv = conv
    best_M = float(conv.max())
    best_f = f.copy()
    best_score = _score(best_f.tolist())

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1. Gradient step on the currently worst convolution entry
        # --------------------------------------------------------------
        k_max = int(np.argmax(conv))
        # gradient of (f*f)[k_max] w.r.t. f (sub‑gradient)
        grad = np.zeros_like(f)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        if i_start <= i_end:
            # grad[i] = 2 * f[k_max - i] on the support interval
            rev_slice = f[i_start:i_end + 1][::-1]          # f[k_max-i]
            grad[i_start:i_end + 1] = 2.0 * rev_slice

        # maximum feasible step without creating negatives
        pos = grad > 1e-12
        if np.any(pos):
            max_alpha = np.min(f[pos] / grad[pos])
            # try a few decreasing step fractions
            for frac in (0.5, 0.2, 0.1, 0.05, 0.02):
                alpha = frac * max_alpha
                if alpha <= 0:
                    continue
                y = f - alpha * grad
                y = _project_simplex(y, target_sum)
                conv_y = _auto_convolution(y, n_fft)
                M_y = float(conv_y.max())
                if M_y + 1e-9 < best_M:          # strict improvement
                    best_M = M_y
                    best_f = y.copy()
                    best_conv = conv_y
                    f = y
                    conv = conv_y
                    best_score = _score(best_f.tolist())
                    print(f"[grad] n={n:4d} it={it:<5d} M={best_M:.6f} score={best_score:.6f}")
                    break   # accept and continue with fresh gradient

        # --------------------------------------------------------------
        # 2. Random mass‑transfer move (small “Robin‑hood’’ step)
        # --------------------------------------------------------------
        if f.sum() > 0:
            i = rng.choice(n, p=f / f.sum())
        else:
            i = rng.integers(n)
        j = rng.integers(n)
        if i != j:
            delta = min(f[i] * 0.08, target_sum * 0.01)  # modest amount
            y = f.copy()
            y[i] -= delta
            y[j] += delta
            # Projection not needed – sum is unchanged and values stay ≥0
            conv_y = _auto_convolution(y, n_fft)
            M_y = float(conv_y.max())
            if M_y + 1e-9 < best_M:
                best_M = M_y
                best_f = y.copy()
                best_conv = conv_y
                f = y
                conv = conv_y
                best_score = _score(best_f.tolist())
                print(f"[transfer] n={n:4d} it={it:<5d} M={best_M:.6f} score={best_score:.6f}")

        # --------------------------------------------------------------
        # 3. Tiny Gaussian perturbation (occasionally, to escape plateaus)
        # --------------------------------------------------------------
        if rng.random() < 0.001:
            y = f + rng.normal(scale=0.005 * target_sum, size=n)
            y = np.clip(y, 0.0, None)
            y = _project_simplex(y, target_sum)
            conv_y = _auto_convolution(y, n_fft)
            M_y = float(conv_y.max())
            if M_y + 1e-9 < best_M:
                best_M = M_y
                best_f = y.copy()
                best_conv = conv_y
                f = y
                conv = conv_y
                best_score = _score(best_f.tolist())
                print(f"[perturb] n={n:4d} it={it:<5d} M={best_M:.6f} score={best_score:.6f}")

        # --------------------------------------------------------------
        # Early exit if we already hit the theoretical lower bound
        # --------------------------------------------------------------
        lower_bound = (2.0 * n) / (2 * n - 1)   # = A² / (2n‑1) with A = √(2n)
        if best_M <= lower_bound * 1.001:      # a tiny tolerance
            print(f"[stop] reached near‑theoretical bound at n={n}")
            break

    return best_f, best_M


# ----------------------------------------------------------------------
# Main entry point required by the judging system.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.
    The routine uses a gradient‑based hill‑climber that works on several
    problem sizes, progressively increasing the length of the step function.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # small safety margin for the outer teardown
    deadline = start_time + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # Load a supplied initial construction if available.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.ndim == 1 and init.size > 0:
            n0 = init.shape[0]
            target0 = math.sqrt(2.0 * n0)
            init = init * (target0 / init.sum())
            best_seq = init.copy()
            best_score = _score(best_seq.tolist())
            print("[init] loaded height_sequence_1, n =", n0,
                  "score =", best_score)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # If no initial data, start from a small random vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 200
        target0 = math.sqrt(2.0 * n0)
        best_seq = rng.random(n0) + 0.1
        best_seq = best_seq / best_seq.sum() * target0
        best_score = _score(best_seq.tolist())
        print("[init] random start, n =", n0, "score =", best_score)

    # ------------------------------------------------------------------
    # Choose a schedule of increasing lengths.
    # Larger n gives a better theoretical bound but each iteration is slower.
    # ------------------------------------------------------------------
    lengths = [200, 400, 800, 1200, 1600]   # stop early if time runs out
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # allocate a time slice for this length (roughly equal shares)
        remaining = deadline - time.monotonic()
        time_for_this = remaining * 0.9 / (len(lengths) - lengths.index(n))
        slice_deadline = time.monotonic() + time_for_this

        # ----- initialise vector of length n -----
        if best_seq.shape[0] == n:
            f = best_seq.copy()
        else:
            # linear interpolation / zero‑padding to grow the vector
            x_old = np.linspace(0.0, 1.0, best_seq.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            f = np.interp(x_new, x_old, best_seq)
        target_sum = math.sqrt(2.0 * n)
        f = f * (target_sum / f.sum())   # re‑normalise ℓ¹‑norm

        # ----- hill‑climbing phase -----
        f_opt, M_opt = _hill_climb(f, target_sum, slice_deadline, rng)

        # ----- evaluate with the official scorer -----
        sc = _score(f_opt.tolist())
        print(f"[final] n={n:4d}  M={M_opt:.6f}  score={sc:.6f}")

        if sc < best_score:
            best_score = sc
            best_seq = f_opt.copy()
            print("[update] new global best, score =", best_score)

    # ------------------------------------------------------------------
    # If, for any reason, we have not produced a candidate, fall back
    # to a very simple random walk (never happens in practice).
    # ------------------------------------------------------------------
    if best_seq is None:
        # simple fallback – the old random‑walk routine
        rng = np.random.default_rng(seed)
        n = 256
        seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = seq.copy()
        best_score = _score(best_seq.tolist())
        print("[fallback] using trivial uniform start, score =", best_score)

    # ------------------------------------------------------------------
    # Return a plain Python list of floats as required by the harness.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
