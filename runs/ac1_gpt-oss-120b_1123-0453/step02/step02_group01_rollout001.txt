import time, math, numpy as np

def _evaluate(seq): return float(evaluate_sequence(seq.tolist()))

def _conv_matrix(f):
    n = len(f)
    rows = 2*n-1
    A = np.zeros((rows,n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            col = k - i
            A[k,col] = f[i]
    return A

def _lp_minimize_M(f, target_sum):
    import cvxpy as cp
    n = f.shape[0]
    A = _conv_matrix(f)
    g = cp.Variable(n)
    Mvar = cp.Variable()
    constraints = [g >= 0,
                   A @ g <= Mvar,
                   cp.sum(g) == target_sum]
    prob = cp.Problem(cp.Minimize(Mvar), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:
            return None, None
    if g.value is None or Mvar.value is None:
        return None, None
    g_val = np.maximum(g.value, 0.0)
    return g_val, float(Mvar.value)

def _optimal_mix(f, g, Cff, Cgg, Cfg, current_M):
    # ternary search on [0,1]
    lo = 0.0
    hi = 1.0
    best_M = current_M
    best_t = 0.0
    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        comb1 = (1-t1)**2 * Cff + 2*t1*(1-t1) * Cfg + t1**2 * Cgg
        M1 = np.max(comb1)
        comb2 = (1-t2)**2 * Cff + 2*t2*(1-t2) * Cfg + t2**2 * Cgg
        M2 = np.max(comb2)
        if M1 < M2:
            hi = t2
            if M1 < best_M:
                best_M = M1
                best_t = t1
        else:
            lo = t1
            if M2 < best_M:
                best_M = M2
                best_t = t2
    # evaluate at final point
    t_opt = (lo+hi)/2
    comb_opt = (1-t_opt)**2 * Cff + 2*t_opt*(1-t_opt) * Cfg + t_opt**2 * Cgg
    M_opt = np.max(comb_opt)
    if M_opt < best_M:
        best_M = M_opt
        best_t = t_opt
    # if best_t still 0 (no improvement), we can keep as None
    if best_M >= current_M - 1e-12:
        return None, None
    f_mix = (1 - best_t) * f + best_t * g
    return f_mix, best_M

"""
Propose a low‑value step sequence.

The algorithm improves on the previous LP‑based scheme by
* solving a *min‑M* linear program (instead of max‑sum),
* performing an *exact* line‑search for the optimal convex
  combination of the current vector and the LP solution,
* using Dirichlet random restarts and a small perturbation
  mechanism to escape plateaus,
* trying several lengths (up to 1000) with a time‑budget guard.

The routine returns the best sequence found before the budget
expires.  All helper functions are defined at top level so that
they have no closures, as required by the competition rules.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _evaluate(seq: np.ndarray) -> float:
    """Wrapper for the provided ``evaluate_sequence`` function."""
    # The environment already supplies evaluate_sequence.
    return float(evaluate_sequence(seq.tolist()))


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the Toeplitz matrix A (size (2n‑1)×n) such that
    A @ g == np.convolve(f, g) for any g of the same length.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _lp_minimize_M(f: np.ndarray, target_sum: float):
    """
    Solve the linear program

        minimise   M
        subject to A @ g <= M,
                   sum(g) == target_sum,
                   g >= 0,

    where A is the convolution matrix of ``f``.
    Returns a feasible ``g`` and the optimal value ``M``.
    If cvxpy is not available or the solver fails, (None, None) is returned.
    """
    try:
        import cvxpy as cp
    except Exception:                     # pragma: no cover
        return None, None

    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n)
    Mvar = cp.Variable()
    constraints = [
        g >= 0,
        A @ g <= Mvar,
        cp.sum(g) == target_sum,
    ]
    prob = cp.Problem(cp.Minimize(Mvar), constraints)

    # Try a fast interior‑point solver first; fall back to SCS if needed.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:                     # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:                 # pragma: no cover
            return None, None

    if g.value is None or Mvar.value is None:
        return None, None

    g_val = np.maximum(g.value, 0.0)       # guard against tiny negatives
    return g_val, float(Mvar.value)


def _optimal_mix(f: np.ndarray, g: np.ndarray,
                 Cff: np.ndarray, Cgg: np.ndarray, Cfg: np.ndarray,
                 current_M: float):
    """
    Find the mixing factor t ∈ (0,1) that minimises the infinity norm of
    the convolution of (1‑t)·f + t·g with itself.
    Returns a pair (new_vector, new_M) or (None, None) if no improvement.
    """
    lo, hi = 0.0, 1.0
    best_M = current_M
    best_t = None

    # Ternary search – the objective is convex on [0,1].
    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0

        comb1 = (1 - t1) ** 2 * Cff + 2 * t1 * (1 - t1) * Cfg + t1 ** 2 * Cgg
        M1 = np.max(comb1)

        comb2 = (1 - t2) ** 2 * Cff + 2 * t2 * (1 - t2) * Cfg + t2 ** 2 * Cgg
        M2 = np.max(comb2)

        if M1 < M2:
            hi = t2
            if M1 < best_M:
                best_M, best_t = M1, t1
        else:
            lo = t1
            if M2 < best_M:
                best_M, best_t = M2, t2

    # Final refinement at the centre of the last interval.
    t_opt = (lo + hi) / 2.0
    comb_opt = (1 - t_opt) ** 2 * Cff + 2 * t_opt * (1 - t_opt) * Cfg + t_opt ** 2 * Cgg
    M_opt = np.max(comb_opt)
    if M_opt < best_M:
        best_M, best_t = M_opt, t_opt

    if best_t is None or best_M >= current_M - 1e-12:
        return None, None

    new_vec = (1.0 - best_t) * f + best_t * g
    return new_vec, best_M


def _dirichlet_start(n: int, target_sum: float, rng: np.random.Generator):
    """Sample a random point on the simplex using a Dirichlet distribution."""
    x = rng.dirichlet(np.ones(n))
    return x.astype(np.float64) * target_sum

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score.

    The method explores several lengths (up to 1000), performs a few
    random restarts per length, and iteratively solves a min‑M LP
    together with an optimal convex‑mixing line‑search.  When progress
    stalls a small random perturbation is applied.  The best sequence
    found before the time budget expires is returned as a plain list.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # keep a tiny margin

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try a handful of lengths; larger lengths give a tighter theoretical bound.
    lengths = [300, 400, 500, 600, 800, 1000]
    restarts_per_n = 3          # a few independent starts per length
    max_lp_iters = 120          # LP steps per restart (adjusted by time)

    # If the platform already supplied a good construction, reuse it.
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        global_seq = np.asarray(global_seq, dtype=np.float64)
    else:
        global_seq = None

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            # --------------------------------------------------------------
            # Initialise the current vector.
            if restart == 0 and global_seq is not None and len(global_seq) == n:
                f = global_seq.copy()
                f = np.maximum(f, 0.0)
                f = f / np.sum(f) * target_sum
                print(f"[init] using global seed n={n}")
            else:
                f = _dirichlet_start(n, target_sum, rng)

            conv_f = np.convolve(f, f)
            M = float(conv_f.max())
            cur_score = 2.0 * n * M / (target_sum ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            # --------------------------------------------------------------
            # Main improvement loop.
            no_progress = 0

                M_g = float(conv_g.max())

                if M_g < M - 1e-12:
                    # Direct improvement.
                    f = g
                    M = M_g
                    conv_f = conv_g
                    cur_score = 2.0 * n * M / (target_sum ** 2)
                    improved = True
                    no_progress = 0
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                        print(f"[LP] n={n} restart={restart} it={it} score={cur_score:.6f}")
                else:

                    Cff = conv_f
                    Cgg = conv_g
                    Cfg = np.convolve(f, g)
                    f_mix, M_mix = _optimal_mix(f, g, Cff, Cgg, Cfg, M)
                    if f_mix is not None:
                        f = f_mix
                        M = M_mix
                        conv_f = np.convolve(f, f)
                        cur_score = 2.0 * n * M / (target_sum ** 2)
                        improved = True
                        no_progress = 0
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = f.copy()
                            print(f"[mix] n={n} restart={restart} it={it} score={cur_score:.6f}")

                if not improved:
                    # 4) No improvement – apply a tiny random perturbation.
                    no_progress += 1
                    if no_progress >= 5:
                        perturb = rng.normal(scale=0.02, size=n)
                        f = np.clip(f + perturb, 0.0, None)
                        f = f / np.sum(f) * target_sum
                        conv_f = np.convolve(f, f)
                        M = float(conv_f.max())
                        cur_score = 2.0 * n * M / (target_sum ** 2)
                        no_progress = 0
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = f.copy()
                            print(f"[perturb] n={n} restart={restart} it={it} score={cur_score:.6f}")

                # Early exit when we are essentially at the theoretical lower bound.
                lower_bound = 2.0 * n / (2.0 * n - 1.0)      # average of the convolution

                        print(f"[near‑lb] n={n} score={cur_score:.6f}")

    if best_seq is None:

        best_score = _evaluate(best_seq)

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    return [float(v) for v in best_seq]
