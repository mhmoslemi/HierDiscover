def project_onto_simplex(v, target):
    # Duchi et al algorithm
    if v.sum() == target and np.all(v>=0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def compute_convolution_fft(f):
    n = f.shape[0]
    # next power of two length for FFT
    size = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(F * F, n=size)
    # keep first 2n-1 elements
    return conv[:2 * n - 1]

def project_onto_simplex(v, target):
    # Duchi et al. Simplex projection for nonnegative vectors
    if target <= 0:
        return np.zeros_like(v)
    # ensure v is float
    v = np.asarray(v, dtype=np.float64)
    # shift by max? Actually algorithm sorts descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    # find rho
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # due to rounding may not sum exactly target, adjust
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        # distribute diff proportionally among positive entries
        pos = w > 0
        if pos.any():
            w[pos] += diff / np.count_nonzero(pos)
        else:
            # all zero case
            w = np.full_like(v, target / len(v))
    return w

"""
Simple sub‑gradient descent for the autocorrelation minimisation problem.

The evaluation score for a non‑negative vector `a` of length `n` with
ℓ¹‑norm `S = √(2n)` is exactly the maximal entry of its discrete
autocorrelation `a*a`.  The algorithm therefore tries to minimise that
maximum.

Approach
--------
* Initialise a few candidate vectors (Dirichlet draw, exponential,
  power‑law, or a padded version of the previously discovered best
  sequence `height_sequence_1`).
* For each candidate run a projected sub‑gradient descent:
    - compute the full convolution by FFT,
    - find the shift `k*` with the current maximal value,
    - the sub‑gradient at this shift is `2 * shift_reverse(a, k*)`,
    - take a step `a ← a – η·grad`, clip negatives, and re‑scale to the
      required ℓ¹‑norm (`S`).  This keeps the feasibility of the
      candidate.
    - adapt the learning rate: increase after a successful step,
      decrease after a failure.
    - occasional random jitter helps escape plateaus.
* Keep the best vector seen over all runs.
* A short stochastic “local search” (tiny mass transfers) is performed
  at the end for polishing.

The routine respects the global time limit (`budget_s`) and returns a
plain Python list of floats.  All numbers are non‑negative and well
below the validator’s 1000 limit.
"""

import time
import math
import numpy as np

_rng = None  # will be initialised in `propose_candidate`


def _conv_max_fft(vec: np.ndarray) -> float:
    """Return max entry of the discrete autocorrelation of `vec` using FFT."""
    n = vec.shape[0]
    # length for a power‑of‑two FFT that covers the whole convolution
    size = 1 << ((2 * n - 1).bit_length())
    # real FFT is a bit faster
    f_hat = np.fft.rfft(vec, n=size)
    conv = np.fft.irfft(f_hat * f_hat, n=size)[: 2 * n - 1]
    return float(conv.max())


def _shift_reverse(vec: np.ndarray, shift: int) -> np.ndarray:
    """Return the vector `v` with v[i] = vec[shift-i] if index in range, else 0."""
    n = vec.shape[0]
    offset = shift - np.arange(n)
    mask = (offset >= 0) & (offset < n)
    out = np.zeros_like(vec)
    out[mask] = vec[offset[mask]]
    return out


def _project_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = target}.
    This is the O(n log n) algorithm of Duchi et al.
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]           # descending order
    cssv = np.cumsum(u) - target
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical guard – ensure the sum is exactly `target`
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        pos = w > 0
        if pos.any():
            w[pos] += diff / np.count_nonzero(pos)
        else:
            w = np.full_like(v, target / len(v))
    return w


def _reorder_zigzag(arr: np.ndarray) -> np.ndarray:
    """Place the largest entries alternately at the leftmost and rightmost positions."""
    sorted_vals = np.sort(arr)[::-1]  # descending
    n = len(arr)
    out = np.empty(n, dtype=arr.dtype)
    l, r = 0, n - 1
    toggle = True
    for v in sorted_vals:
        if toggle:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        toggle = not toggle
    return out


def _subgradient_descent(
    init: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that directly minimises
    `max(conv(a))`.  Returns the best vector and its score.
    """
    n = init.shape[0]
    cur = init.copy()
    cur_score = _conv_max_fft(cur)
    best_vec = cur.copy()
    best_score = cur_score

    # initial learning rate – about one‐tenth of the average entry
    lr = max(target_sum / (10.0 * n), 1e-8)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = _conv_max_fft(cur)  # cheap – we already have the max
        # we need the full convolution only to locate the maximal shift
        # recompute it once per iteration (still cheap)
        n_full = 1 << ((2 * n - 1).bit_length())
        f_hat = np.fft.rfft(cur, n=n_full)
        full_conv = np.fft.irfft(f_hat * f_hat, n=n_full)[: 2 * n - 1]
        k_star = int(np.argmax(full_conv))

        # sub‑gradient of the active quadratic term
        grad = 2.0 * _shift_reverse(cur, k_star)

        # step + projection onto the ℓ¹‑simplex
        cand = cur - lr * grad
        cand = np.maximum(cand, 0.0)
        total = cand.sum()
        if total > 0:
            cand = cand * (target_sum / total)
        else:
            cand = np.full_like(cur, target_sum / n)

        cand_score = _conv_max_fft(cand)

        if cand_score < cur_score - 1e-12:
            # successful move
            cur, cur_score = cand, cand_score
            if cand_score < best_score - 1e-12:
                best_vec, best_score = cand.copy(), cand_score
                no_improve = 0
                lr *= 1.02                 # slightly larger step after improvement
            else:
                no_improve += 1
                lr *= 1.01
        else:
            # reject – shrink step size
            lr *= 0.5
            no_improve += 1

        it += 1

        # occasional random jitter to escape flat plateaus
        if no_improve >= 200:
            jitter = _rng.normal(scale=target_sum * 0.001, size=n)
            cur = cur + jitter
            cur = np.maximum(cur, 0.0)
            cur = cur * (target_sum / cur.sum())
            cur_score = _conv_max_fft(cur)
            no_improve = 0

        # stop early if we are already very good
        if best_score < 1.45:
            break
        if lr < 1e-10:
            break

    return best_vec, best_score


def _local_polish(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Very cheap random mass‑transfer walk (keeps ℓ¹‑norm)."""
    best = seq.copy()
    best_score = _conv_max_fft(best)
    n = len(best)

    while time.monotonic() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose a tiny transfer proportional to the source magnitude
        delta = _rng.normal(scale=0.02) * max(1e-9, best[i])
        if delta == 0.0:
            continue
        cand_i = best[i] + delta
        cand_j = best[j] - delta
        if cand_i < 0.0 or cand_j < 0.0:
            continue
        cand = best.copy()
        cand[i] = cand_i
        cand[j] = cand_j
        sc = _conv_max_fft(cand)
        if sc < best_score:
            best, best_score = cand, sc
            # optional progress output
            print(f"[polish] improved score={best_score:.6f}")

    return best, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low autocorrelation peak.
    The routine uses several random initialisations and a projected
    sub‑gradient descent, followed by a tiny stochastic polish.
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # safety margin before the external timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Choose a set of problem sizes.
    # ------------------------------------------------------------------
    # Larger n gives more freedom, but we keep the total work bounded.
    n_candidates = [300, 400, 500, 600, 800, 1000]

    for n in n_candidates:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # several restarts for each size
        restarts = 6
        for _ in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ------------------------------------------------------------
            # a) initialise a candidate vector – several heuristics
            # ------------------------------------------------------------
            strategy = _rng.choice(["dirichlet", "exp", "power", "prev"])
            if strategy == "dirichlet":
                init = _rng.exponential(scale=1.0, size=n)
            elif strategy == "exp":
                lam = _rng.uniform(0.01, 0.15)
                init = np.exp(-lam * np.arange(n))
            elif strategy == "power":
                p = _rng.uniform(0.5, 2.0)
                init = (np.arange(1, n + 1) ** (-p))
            else:  # "prev"
                # Use the global `height_sequence_1` if it exists
                try:
                    prev = np.asarray(height_sequence_1, dtype=float)
                except Exception:
                    prev = None
                if prev is not None and prev.size > 0:
                    if prev.size <= n:
                        # embed at a random offset
                        init = np.zeros(n, dtype=float)
                        start = _rng.integers(0, n - prev.size + 1)
                        init[start:start + prev.size] = prev
                    else:
                        init = prev[:n].copy()
                else:
                    init = _rng.exponential(scale=1.0, size=n)

            # normalise to the required ℓ¹‑norm
            init = np.maximum(init, 0.0)
            if init.sum() == 0.0:
                init = np.full(n, 1.0)
            init = init / init.sum() * target_sum

            # random optional zig‑zag reordering (helps to break symmetry)
            if _rng.random() < 0.5:
                init = _reorder_zigzag(init)

            # ------------------------------------------------------------
            # b) run the sub‑gradient descent
            # ------------------------------------------------------------
            best_vec, best_sc = _subgradient_descent(init, target_sum, deadline)

            if best_sc < overall_best_score:
                overall_best_score = best_sc
                overall_best_seq = best_vec.copy()
                print(f"[global] new overall best n={n} score={best_sc:.6f}")

            # very early exit if we already beat a strong target
            if overall_best_score < 1.45:
                break

    # ------------------------------------------------------------------
    # 2) Light‑weight polishing if time permits.
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_score = _local_polish(overall_best_seq, deadline)
        if polished_score < overall_best_score:
            overall_best_score = polished_score
            overall_best_seq = polished_seq.copy()
            print(f"[global] after polishing score={polished_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback to a deterministic safe construction.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        overall_best_seq = np.full(256, 1.0, dtype=float)
        overall_best_score = _conv_max_fft(overall_best_seq)
        print("[fallback] using uniform safe sequence")

    # final clipping – values are far from 1000 but we stay on the safe side
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    return [float(x) for x in overall_best_seq]
