def compute_score(a):

    n = len(a)

"""
Simulated annealing based search for low‑value step sequence.
...
"""
import time, math, numpy as np

_rng = None

def compute_score(a: np.ndarray) -> float:
    """Return the evaluation score for a sequence (same as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))

def _anneal_sequence(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing optimisation for a fixed length."""
    global _rng
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sum(a)  # already normalized
    best = a.copy()
    best_score = compute_score(a)
    avg = target_sum / n
    sigma = max(avg * 0.5, 1e-6)          # spread of proposed transfer
    # temperature schedule – start relatively high, decay exponentially
    T0 = best_score * 0.05 if best_score > 0 else 1.0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        T = T0 * (0.9995 ** it)

        # 20 % of the time we try a “targeted” move that attacks the current maximal convolution entry.
        if _rng.random() < 0.2:
            conv_full = np.convolve(a, a)
            k_max = int(np.argmax(conv_full))
            idx = np.arange(n)
            partner = k_max - idx
            valid = (partner >= 0) & (partner < n)
            contrib = np.zeros(n)
            contrib[valid] = a[idx[valid]] * a[partner[valid]]
            total_contrib = contrib.sum()
            if total_contrib > 0:
                probs = contrib / total_contrib
                i = _rng.choice(n, p=probs)
            else:
                i = _rng.integers(0, n)
            # pick a destination that is *not* the same index
            j = _rng.integers(0, n - 1)
            if j >= i:
                j += 1
        else:
            # generic move – pick source proportional to its current mass
            probs = a / target_sum
            i = _rng.choice(n, p=probs)
            j = _rng.integers(0, n - 1)
            if j >= i:
                j += 1

        # propose a transfer delta from i to j
        delta = _rng.normal(0.0, sigma)
        # enforce non‑negativity of both entries
        if delta > a[i]:
            delta = a[i]
        if -delta > a[j]:
            delta = -a[j]
        if abs(delta) < 1e-14:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # keep everything inside the allowed box
        if a_new[i] < 0.0 or a_new[j] < 0.0 or a_new[i] > 1000.0 or a_new[j] > 1000.0:
            continue

        sc = compute_score(a_new)

        # Metropolis acceptance
        if sc < best_score or (T > 0.0 and _rng.random() < math.exp(-(sc - best_score) / T)):
            a = a_new
            if sc < best_score:
                best_score = sc
                best = a_new.copy()
                # slowly shrink the proposal width when we improve
                sigma = max(sigma * 0.99, avg * 0.01)
                print(f"[anneal] n={n:3d} it={it:6d} score={best_score:.6f}")

        # occasional cooling of sigma even without improvement
        if it % 5000 == 0:
            sigma = max(sigma * 0.9, avg * 0.005)

        # early stop if we already hit a very good bound
        if best_score < 1.45:
            break

    return best, best_score

def _local_search(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Pure greedy hill‑climb that only accepts improving swaps."""
    global _rng
    best = a.copy()
    best_score = compute_score(best)
    n = a.shape[0]
    while time.monotonic() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a tiny amount from the larger entry towards the smaller one
        if best[i] > best[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        delta = _rng.normal(0.0, 0.02) * max(1e-6, best[src])
        if delta == 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        if cand[src] < 0.0 or cand[dst] < 0.0:
            continue
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand
            print(f"[local] score={best_score:.6f}")
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Search for a low‑score step function.
    The routine runs a simulated‑annealing optimisation for several
    problem sizes, keeps the best sequence found and finishes with a
    short greedy polishing phase.
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # give a tiny safety margin before the external timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    overall_best_seq = None
    overall_best_score = float('inf')

    # problem sizes – we start modest and grow while time permits
    sizes = [250, 300, 350, 400, 450, 500, 550, 600]

    for n in sizes:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # a handful of diverse initialisations
        inits = []

        # uniform
        inits.append(np.full(n, target_sum / n))

        # random positive vector
        rnd = _rng.random(n) + 0.001
        rnd = rnd / rnd.sum() * target_sum
        inits.append(rnd)

        # linearly increasing (triangular) shape
        tri = np.arange(1, n + 1, dtype=np.float64)
        tri = tri / tri.sum() * target_sum
        inits.append(tri)

        # linearly decreasing – mirror of the previous one (mass near the ends)
        rev_tri = tri[::-1]
        inits.append(rev_tri)

        # block: half of the coordinates carry the whole mass
        half = max(1, n // 2)
        block = np.concatenate([np.full(half, target_sum / half), np.zeros(n - half)])
        block = block / block.sum() * target_sum
        inits.append(block)

        # If a historic construction exists, try to reuse it (rescaled)
        try:
            if 'height_sequence_1' in globals():
                hist = np.array(height_sequence_1, dtype=float)
                if hist.size != n:
                    # truncate or pad with zeros
                    if hist.size > n:
                        hist = hist[:n]
                    else:
                        hist = np.concatenate([hist, np.zeros(n - hist.size)])
                hist = np.clip(hist, 0.0, 1000.0)
                if hist.sum() > 0:
                    hist = hist / hist.sum() * target_sum
                    inits.append(hist)
        except Exception:
            pass

        for start in inits:
            if time.monotonic() >= deadline:
                break
            seq, sc = _anneal_sequence(start, deadline)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = seq.copy()
                print(f"[global] n={n:3d} new best score={sc:.6f}")

            # early exit if we have already crossed the target comfortably
            if overall_best_score < 1.45:
                break

    # final polish – a fast greedy walk that only accepts improvements
    if overall_best_seq is not None and time.monotonic() < deadline:
        polished, polished_score = _local_search(overall_best_seq, deadline)
        if polished_score < overall_best_score:
            overall_best_score = polished_score
            overall_best_seq = polished
            print(f"[global] after polishing: {polished_score:.6f}")

    # fallback – should never happen, but guard against a completely empty result
    if overall_best_seq is None:
        n = 256
        overall_best_seq = np.full(n, 1.0, dtype=np.float64)

    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    # Return a plain Python list of floats as required by the framework
    return [float(x) for x in overall_best_seq.tolist()]

"""
Simulated‑annealing search for a low‑value step sequence.

The algorithm explores several problem sizes, starts each run from a
different simple seed (uniform, random, triangular, block, historic
construction), and runs a temperature‑controlled Monte‑Carlo walk that
preserves the ℓ¹‑norm (the normalisation √(2 n) used by the scorer).  A
small “targeted” move tries to reduce the current maximal auto‑convolution
entry, while the rest of the moves are generic mass transfers.  The walk
accepts uphill steps with a Metropolis probability, allowing it to escape
local minima.  When time is low a final greedy polishing walk keeps only
improving swaps.  All numbers are kept in the allowed interval
[0, 1000] and the routine respects the 1000 s budget.
"""

import time
import math
import numpy as np

# Global RNG – initialised in `propose_candidate`.
_rng = None


def _score(seq: np.ndarray) -> float:
    """
    Compute the same score as `evaluate_sequence` for a NumPy vector.
    The sequence is clipped to the validator’s limits before the
    convolution is performed.
    """
    a = np.clip(seq, 0.0, 1000.0)
    n = a.shape[0]
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    conv = np.convolve(a, a)
    max_conv = float(np.max(conv))
    return float(2.0 * n * max_conv / (s * s))


def _anneal_sequence(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing optimisation for a fixed length.

    * The ℓ¹‑norm of `a0` is assumed to be the target (√(2 n)).
    * At each iteration a small amount of mass is moved from one index
      to another while the total sum stays unchanged.
    * 20 % of the proposals are *targeted*: they pick a source index that
      contributes the most to the current maximal convolution entry.
    * The Metropolis rule accepts uphill moves with probability
      exp(−Δ/temperature).  Temperature decays exponentially.
    * The function stops when the global deadline is reached or when a
      very good bound (< 1.45) has been found.
    """
    global _rng
    a = a0.copy()
    n = a.shape[0]
    target_sum = float(np.sum(a))               # already normalised
    best = a.copy()
    best_score = _score(a)

    avg = target_sum / n
    sigma = max(avg * 0.5, 1e-6)                 # scale of the proposed transfer
    T0 = best_score * 0.05 if best_score > 0 else 1.0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        T = T0 * (0.9995 ** it)                  # exponential cooling

        # ----- choose source (i) and destination (j) -----------------
        if _rng.random() < 0.2:                  # targeted move
            conv_full = np.convolve(a, a)
            k_max = int(np.argmax(conv_full))

            idx = np.arange(n)
            partner = k_max - idx
            valid = (partner >= 0) & (partner < n)

            contrib = np.zeros(n)
            contrib[valid] = a[idx[valid]] * a[partner[valid]]
            total = float(np.sum(contrib))

            if total > 0.0:
                probs = contrib / total
                i = _rng.choice(n, p=probs)
            else:
                i = _rng.integers(0, n)

            # pick a different destination at random
            j = _rng.integers(0, n - 1)
            if j >= i:
                j += 1
        else:                                    # generic move
            probs = a / target_sum
            i = _rng.choice(n, p=probs)
            j = _rng.integers(0, n - 1)
            if j >= i:
                j += 1

        # ----- propose a transfer delta from i to j -------------------
        delta = _rng.normal(0.0, sigma)
        # keep both entries non‑negative
        if delta > a[i]:
            delta = a[i]
        if -delta > a[j]:
            delta = -a[j]
        if abs(delta) < 1e-14:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # keep everything inside the allowed box
        if a_new[i] < 0.0 or a_new[j] < 0.0 or a_new[i] > 1000.0 or a_new[j] > 1000.0:
            continue

        sc = _score(a_new)

        # ----- Metropolis acceptance ---------------------------------
        if sc < best_score or (T > 0.0 and _rng.random() < math.exp(-(sc - best_score) / T)):
            a = a_new
            if sc < best_score:
                best_score = sc
                best = a_new.copy()
                sigma = max(sigma * 0.99, avg * 0.01)   # slightly shrink step size on improvement
                print(f"[anneal] n={n:3d} it={it:6d} score={best_score:.6f}")

        # occasional cooling of sigma (even without improvement)
        if it % 5000 == 0:
            sigma = max(sigma * 0.9, avg * 0.005)

        # early exit if already very good
        if best_score < 1.45:
            break

    return best, best_score


def _local_search(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that only keeps strictly improving swaps.
    This is used as a cheap polishing phase after the annealing.
    """
    global _rng
    best = a.copy()
    best_score = _score(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue

        # transfer a tiny amount from the larger entry towards the smaller one
        if best[i] > best[j]:
            src, dst = i, j
        else:
            src, dst = j, i

        delta = _rng.normal(0.0, 0.02) * max(1e-6, best[src])
        if delta == 0.0:
            continue

        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        if cand[src] < 0.0 or cand[dst] < 0.0:
            continue

        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best = cand
            print(f"[local] score={best_score:.6f}")

    return best, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.

    The routine runs simulated‑annealing on several lengths,
    keeps the best sequence found and finishes with a short greedy polishing.
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # safety margin before the external timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    overall_best_seq = None
    overall_best_score = float("inf")

    # problem sizes – start modest, grow while time permits
    sizes = [250, 300, 350, 400, 450, 500, 550, 600]

    for n in sizes:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ---- diverse initialisations ---------------------------------
        inits = []

        # uniform
        inits.append(np.full(n, target_sum / n))

        # random positive vector
        rnd = _rng.random(n) + 0.001
        rnd = rnd / rnd.sum() * target_sum
        inits.append(rnd)

        # linearly increasing (triangular)
        tri = np.arange(1, n + 1, dtype=np.float64)
        tri = tri / tri.sum() * target_sum
        inits.append(tri)

        # linearly decreasing – mass near the ends
        rev_tri = tri[::-1]
        inits.append(rev_tri)

        # block – half the positions carry the whole mass
        half = max(1, n // 2)
        block = np.concatenate([np.full(half, target_sum / half), np.zeros(n - half)])
        block = block / block.sum() * target_sum
        inits.append(block)

        # historic construction, if available
        try:
            if "height_sequence_1" in globals():
                hist = np.array(height_sequence_1, dtype=float)
                if hist.size != n:
                    if hist.size > n:
                        hist = hist[:n]
                    else:
                        hist = np.concatenate([hist, np.zeros(n - hist.size)])
                hist = np.clip(hist, 0.0, 1000.0)
                if hist.sum() > 0.0:
                    hist = hist / hist.sum() * target_sum
                    inits.append(hist)
        except Exception:
            pass

        # ---- run annealing from each start -----------------------------
        for start in inits:
            if time.monotonic() >= deadline:
                break
            seq, sc = _anneal_sequence(start, deadline)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = seq.copy()
                print(f"[global] n={n:3d} new best score={sc:.6f}")

            if overall_best_score < 1.45:
                break

    # ---- final greedy polishing ---------------------------------------
    if overall_best_seq is not None and time.monotonic() < deadline:
        polished, polished_score = _local_search(overall_best_seq, deadline)
        if polished_score < overall_best_score:
            overall_best_score = polished_score
            overall_best_seq = polished
            print(f"[global] after polishing: {polished_score:.6f}")

    # ---- fallback (should never be needed) -----------------------------
    if overall_best_seq is None:
        n = 256
        overall_best_seq = np.full(n, 1.0, dtype=np.float64)

    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    # Return a plain list of Python floats as required by the platform.
    return [float(x) for x in overall_best_seq.tolist()]
