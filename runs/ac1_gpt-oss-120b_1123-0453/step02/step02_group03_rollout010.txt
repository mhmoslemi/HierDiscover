"""
Improved step‑function search.

The algorithm proceeds in three stages for each candidate length *n*:

1️⃣ **Population‑based exploration (simple GA)** – a small population of
   non‑negative sequences (all normalised to ‖f‖₁ = √(2 n)) is evolved.
   Crossover, Gaussian mutation and a selection of the best individuals
   drive the population toward lower evaluation scores.

2️⃣ **Linear‑programming refinement** – starting from the best GA individual,
   the classic “LP‑enlarge‑mass” step is applied many times:
   for the current f we solve
       maximise   Σ g_i    s.t.   f * g ≤ M   , g ≥ 0,
   where M = max(f * f).  The optimiser g is rescaled to the same ℓ¹‑norm
   as f and mixed with f (convex combination).  Repeating this step
   monotonically reduces the maximal self‑convolution entry.

3️⃣ **Fine‑tuned simulated annealing** – a final SA hill‑climber makes
   small local perturbations and occasional uphill moves to escape plateaux.

The routine tries several lengths (200 … 400).  All work respects the global
time budget (default 1000 s) and returns the best sequence found as a plain
list of floats (each entry ≥ 0 and ≤ 1000).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Wrapper around the provided scorer."""
    return float(evaluate_sequence(seq.tolist()))


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix A such that (A @ g)[k] == (f * g)[k].
    Each row k contains the coefficients f[i] for those i that pair with
    g[k‑i] in the convolution.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _lp_improve(
    f: np.ndarray,
    target_sum: float,
    mix: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray | None, float | None]:
    """
    One LP‑based improvement step.

    Parameters
    ----------
    f : np.ndarray
        Current sequence (non‑negative, ℓ¹‑norm = ``target_sum``).
    target_sum : float
        Desired ℓ¹‑norm after the rescaling step.
    mix : float
        Convex‑combination weight; 0 < mix ≤ 1.
    rng : np.random.Generator
        Random generator (used only for a warm start inside CVXPY).

    Returns
    -------
    (new_f, new_score) or (None, None) if the LP fails.
    """
    M = float(np.max(np.convolve(f, f)))
    A = _conv_matrix(f)

    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None, None

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is fast for dense problems; fall back to SCS on failure.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=8000, abstol=1e-6, reltol=1e-6)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000)

    if g.value is None:
        return None, None

    g0 = np.maximum(g.value, 0.0)
    sum_g = float(np.sum(g0))
    if sum_g <= 1e-12:
        return None, None

    # Rescale to keep the ℓ¹‑norm fixed.
    g_rescaled = g0 * (target_sum / sum_g)

    # Convex combination with the current f.
    f_new = (1.0 - mix) * f + mix * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    new_score = _score(f_new)
    return f_new, new_score


def _simulated_annealing(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple SA optimiser that perturbs a single coordinate at a time.
    The temperature cools from 1.0 down to 1e‑4 over the remaining time.
    """
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        temp = temp_start * (temp_end / temp_start) ** frac

        idx = rng.integers(len(cur_seq))
        delta = rng.normal(scale=0.5) * temp * 5.0
        cand = cur_seq.copy()
        cand[idx] = float(np.clip(cand[idx] + delta, 0.0, 1000.0))

        cand_score = _score(cand)

        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

        # tiny random reshuffle to avoid stagnation
        if rng.random() < 0.001:
            pert = np.maximum(
                cur_seq + rng.normal(scale=0.1, size=len(cur_seq)), 0.0
            )
            cur_seq = np.clip(pert, 0.0, 1000.0)
            cur_score = _score(cur_seq)

    return best_seq, best_score


def _ga_optimize(
    n: int,
    popsize: int,
    generations: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Very light‑weight genetic algorithm for a fixed length ``n``.
    The population is kept normalised to √(2 n); each fitness evaluation
    uses the supplied ``evaluate_sequence`` function.
    """
    target_sum = math.sqrt(2.0 * n)

    # ---- initial population ------------------------------------------------
    pop = []
    for _ in range(popsize):
        seq = rng.random(n)
        seq = seq / seq.sum() * target_sum
        pop.append(seq)

    # add a deterministic alternating pattern (could be a good seed)
    alt = np.zeros(n)
    alt[::2] = 1.0
    if alt.sum() > 0:
        alt = alt / alt.sum() * target_sum
        pop.append(alt)

    # evaluate
    scores = [_score(s) for s in pop]
    best_idx = int(np.argmin(scores))
    best_seq = pop[best_idx].copy()
    best_score = scores[best_idx]
    print(f"[GA init] n={n} best init score {best_score:.6f}")

    # ---- evolution ---------------------------------------------------------
    for gen in range(generations):
        if time.monotonic() >= deadline:
            break

        # keep the top ~20 % as elites
        elite_cnt = max(2, popsize // 5)
        elite_idx = np.argsort(scores)[:elite_cnt]
        elite = [pop[i] for i in elite_idx]

        # new generation
        new_pop = elite.copy()           # preserve elites
        while len(new_pop) < popsize:
            p1, p2 = rng.choice(elite, size=2, replace=True)
            alpha = rng.random()
            child = alpha * p1 + (1.0 - alpha) * p2
            # mutation
            child += rng.normal(scale=0.05, size=n)
            child = np.clip(child, 0.0, 1000.0)

            # renorm to the target ℓ¹‑norm
            s = child.sum()
            if s < 1e-12:
                child = rng.random(n) + 1e-3
                s = child.sum()
            child *= (target_sum / s)

            new_pop.append(child)

        pop = new_pop
        scores = [_score(s) for s in pop]

        cur_best_idx = int(np.argmin(scores))
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score - 1e-9:
            best_score = cur_best_score
            best_seq = pop[cur_best_idx].copy()
            print(f"[GA] n={n} gen={gen} new best {best_score:.6f}")

        # early exit if we are already exceptionally good
        if best_score < 1.4:
            break

    return best_seq, best_score


def _lp_refine(
    f: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Perform many LP improvement steps on ``f`` until time runs out or
    ``max_iters`` is reached.  The mixing factor slowly decays.
    """
    n = f.shape[0]
    target_sum = f.sum()
    mix = 0.30
    best_seq = f.copy()
    best_score = _score(f)

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        f_new, sc = _lp_improve(best_seq, target_sum, mix, rng)
        if f_new is None:
            break
        best_seq = f_new
        if sc < best_score - 1e-9:
            best_score = sc
            print(f"[LP refine] n={n} it={it:4d} score={best_score:.6f}")
        # slowly shrink the mixing weight – this behaves like a diminishing
        # step size in a gradient descent.
        mix = max(0.01, mix * 0.997)
        it += 1
        # if no improvement for a while we can stop early
        if it % 200 == 0 and best_score - sc < 1e-8:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a low‑value step‑function sequence within the given time budget.

    The algorithm combines a very simple genetic algorithm, a long LP‑refinement
    phase and a final simulated‑annealing fine‑tuner.  The best sequence found
    (as a plain list) is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Use a globally supplied construction if it exists.
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            seq0 = np.clip(seq0, 0.0, 1000.0)
            sc0 = _score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using height_sequence_1 – score {sc0:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Main search over several lengths.
    # ------------------------------------------------------------------
    lengths = [200, 250, 300, 350, 400]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.monotonic()
        # give each length at most 0.45 of the remaining budget
        sub_deadline = time.monotonic() + min(remaining * 0.45, 120.0)

        # ---- GA stage -------------------------------------------------
        popsize = 20
        generations = 150
        seq, sc = _ga_optimize(n, popsize, generations, rng, sub_deadline)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[overall] GA yielded new best score {best_score:.6f}")

        # ---- LP refinement --------------------------------------------
        # Give a generous amount of time for LP; reuse the global deadline.
        seq_lp, sc_lp = _lp_refine(seq, deadline, rng, max_iters=1500)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = seq_lp.copy()
            print(f"[overall] LP refinement new best {best_score:.6f}")

        # ---- local SA -------------------------------------------------
        if best_seq is not None and time.monotonic() < deadline:
            best_seq, best_score = _simulated_annealing(
                best_seq, best_score, deadline, rng
            )
            print(f"[overall] after SA best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Safety fallback – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 200
        best_seq = np.full(n0, math.sqrt(2.0 * n0) / n0, dtype=np.float64)
        best_score = _score(best_seq)

    print(f"[final] best score reported {best_score:.6f} (length {len(best_seq)})")
    return [float(x) for x in best_seq]
