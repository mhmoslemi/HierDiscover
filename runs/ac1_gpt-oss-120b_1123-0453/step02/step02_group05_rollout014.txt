def improve_sequence(f, n_iter=10):
    for iter in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        # Build matrix A where A[k,j] = f[k-j] if 0 <= k-j < n else 0
        n = len(f)
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n-1)
            # for each j (the variable index) we have coefficient f[i] where i = k - j
            # Equivalent: for j in [i_start,i_end], coefficient = f[k - j]
            for j in range(i_start, i_end+1):
                A[k, j] = f[k - j]
        # Linear program: maximize sum(g) subject to A @ g <= M * np.ones(2*n-1), g >=0
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2*n-1)]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True)
        if g.value is None:
            break
        sum_f = np.sum(f)
        sum_g = np.sum(g.value)
        if sum_g <= sum_f + 1e-8:
            # cannot improve sum
            break
        # set f to g (or combine)
        f = g.value  # maybe after rebalancing?
    return f

def _run_sa(f, deadline):
    n = len(f)
    conv = np.convolve(f, f)
    cur_max = conv.max()
    best_f = f.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    # schedule parameters
    total_time = deadline - time.monotonic()
    start = time.monotonic()
    it = 0
    last_improve_it = 0
    # adjust parameters
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        T = max(1e-6, (1.0 - elapsed/total_time))
        # pick worst index
        k_max = int(np.argmax(conv))
        # gradient vector
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx_range = np.arange(i_start, i_end + 1)
        grad = np.zeros(n)
        grad[idx_range] = 2.0 * f[k_max - idx_range]
        # probability distribution proportional to grad
        prob = grad + 1e-12
        prob /= prob.sum()
        i = rng.choice(n, p=prob)
        # pick destination index random
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        # delta amount
        max_delta = f[i]
        if max_delta <= 0:
            continue
        # step size shrink near end
        frac = 0.2 * (1.0 - elapsed / total_time) + 0.01
        delta = rng.random() * max_delta * frac
        # apply update
        # store old conv for rollback
        conv_old = conv.copy()
        fi_old = f[i]
        fj_old = f[j]

        # subtract contributions from i
        conv[i:i+n] -= 2.0 * delta * f
        # add contributions from j (using original f)
        conv[j:j+n] += 2.0 * delta * f

        # self-term corrections: conv[2*i] added delta^2 ; conv[2*j] added delta^2; conv[i+j] subtract 2 delta^2 (if within length)
        conv[2*i] += delta * delta
        conv[2*j] += delta * delta
        s = i + j
        if s < len(conv):
            conv[s] -= 2.0 * delta * delta

        # update f
        f[i] -= delta
        f[j] += delta

        new_max = conv.max()

        # acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob_up = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob_up:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f.copy()
                last_improve_it = it
                # compute objective
                obj = 2.0 * n * best_max / (np.sum(best_f) ** 2)
                print(f"[SA] it={it} obj={obj:.6f} max={best_max:.6f}")
        else:
            # rollback
            conv[:] = conv_old
            f[i] = fi_old
            f[j] = fj_old

        it += 1
        # optional restart if stagnates
        if it - last_improve_it > 200000 and elapsed > 0.5 * total_time:
            # restart
            f = np.random.random(n)
            f = f / f.sum()
            conv = np.convolve(f, f)
            cur_max = conv.max()
            last_improve_it = it
    # compute objective
    best_obj = 2.0 * n * best_max / (np.sum(best_f) ** 2)
    return best_f, best_obj

"""
Propose a step‑function (non‑negative sequence) with a low value of

        2·n·max(f*f) / (sum(f)**2)

The routine works in two stages.

1. **Fast simulated‑annealing (SA)** – starting from a feasible
   sequence (the historic best `height_sequence_1` if available,
   otherwise a uniform vector) the algorithm repeatedly moves a
   small amount of mass from an index that contributes most to the
   worst convolution entry to a random destination.  The move is
   accepted with a Metropolis rule.  The temperature is decreased
   linearly in time.  The best vector seen during the SA run is kept.

2. **Linear‑programming improvement** – with the SA result fixed, we
   solve the LP

        maximise   Σ g_i
        subject to g_i ≥ 0
                   (f * g)_k ≤ M   for all k

   where `M` is the current worst convolution value.
   The solution `g` has a larger ℓ¹‑norm while not increasing the
   worst convolution.  Replacing `f` by `g` therefore reduces the
   evaluation score.  The LP is solved with CVXPY/ECOS; a few
   iterations are performed until the time budget is exhausted.

Both phases respect the scorer’s limits (non‑negative, ≤1000).  The
function returns the best sequence discovered before the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _evaluate(seq: list[float]) -> float:
    """Wrap the provided scorer."""
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
def _initial_sequence() -> np.ndarray:
    """Create a feasible starting sequence.
    Uses the historic best if it exists, otherwise a uniform vector.
    The vector is scaled to ℓ¹‑norm = 1 (the absolute scale does not
    affect the score because the evaluator is homogeneous).
    """
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=np.float64)
    else:
        seq = np.full(200, 1.0, dtype=np.float64)

    # safety – clip to the interval accepted by the scorer
    seq = np.where(np.isnan(seq) | np.isinf(seq), 0.0, seq)
    seq = np.clip(seq, 0.0, 1000.0)

    # normalise to ℓ¹‑norm = 1  (any positive constant works)
    if seq.sum() == 0.0:
        seq[:] = 1.0
    seq = seq / np.sum(seq)
    return seq


# ----------------------------------------------------------------------
def _run_sa(f0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing optimiser for a fixed length vector.

    Parameters
    ----------
    f0 : np.ndarray
        Initial non‑negative vector with sum(f0) = 1.
    deadline : float
        Absolute time (``time.monotonic()``) at which the run must stop.

    Returns
    -------
    best_f : np.ndarray
        Best vector found.
    best_obj : float
        Corresponding evaluation score (identical to ``2·n·max/(sum**2)``).
    """
    rng = np.random.default_rng()
    f = f0.copy()
    n = len(f)
    conv = np.convolve(f, f)                    # full convolution, length 2·n‑1
    cur_max = conv.max()
    best_f = f.copy()
    best_max = cur_max
    best_obj = 2.0 * n * best_max / (np.sum(best_f) ** 2)

    start = time.monotonic()
    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        # linear cooling – temperature goes from 1.0 to 1e‑4
        T = 1.0 * (1e-4 / 1.0) ** (elapsed / (deadline - start + 1e-12))

        # --------------------------------------------------------------
        # choose the index that currently limits the maximum convolution
        k_max = int(np.argmax(conv))

        # gradient of the (unique) worst entry: g_i = 2·f_{k_max‑i}
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad = np.zeros(n, dtype=np.float64)
        grad[idx] = 2.0 * f[k_max - idx]

        # donor index – probability proportional to the gradient magnitude
        prob = grad + 1e-12
        prob /= prob.sum()
        i = rng.choice(n, p=prob)

        # random destination, different from i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # amount of mass to move (fraction of donor, shrinks near the end)
        max_delta = f[i]
        if max_delta <= 1e-14:
            continue
        frac = 0.2 * (1.0 - elapsed / (deadline - start + 1e-12)) + 0.01
        delta = rng.random() * max_delta * frac
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # apply the move (in‑place, keep a copy for possible rollback)
        conv_old = conv.copy()
        fi_old, fj_old = f[i], f[j]

        # linear part of the convolution update
        conv[i:i + n] -= 2.0 * delta * f
        conv[j:j + n] += 2.0 * delta * f

        # quadratic corrections (self‑terms)
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        s = i + j
        if s < conv.size:
            conv[s] -= 2.0 * delta * delta

        # update the sequence itself
        f[i] -= delta
        f[j] += delta

        new_max = conv.max()

        # --------------------------------------------------------------
        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob_up = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob_up:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f.copy()
                last_improve = it
                best_obj = 2.0 * n * best_max / (np.sum(best_f) ** 2)
                print(f"[SA] it={it:07d} obj={best_obj:.6f} max={best_max:.6f} T={T:.2e}")
        else:
            # rollback
            conv[:] = conv_old
            f[i] = fi_old
            f[j] = fj_old

        # occasional full recomputation (prevents drift from rounding)
        if it % 5000 == 0:
            conv = np.convolve(f, f)
            cur_max = conv.max()

        # cheap restart if no improvement for a long while
        if (it - last_improve > 200_000) and (elapsed > 0.5 * (deadline - start)):
            f = _initial_sequence()
            conv = np.convolve(f, f)
            cur_max = conv.max()
            last_improve = it

        it += 1

    best_obj = 2.0 * n * best_max / (np.sum(best_f) ** 2)
    return best_f, best_obj


# ----------------------------------------------------------------------
def _run_lp_improve(f0: np.ndarray, start_deadline: float,
                    final_deadline: float) -> tuple[np.ndarray, float]:
    """Linear‑programming improvement of a vector.

    The LP maximises the ℓ¹‑norm while keeping the worst
    convolution (f * g) below the current bound `M`.  If the new
    vector has a larger sum, the evaluation score decreases.

    Parameters
    ----------
    f0 : np.ndarray
        Starting vector (sum may be any positive number).
    start_deadline : float
        Current time (used only for diagnostics).
    final_deadline : float
        Absolute time at which the whole routine must stop.

    Returns
    -------
    best_f : np.ndarray
        Vector with the lowest score found by the LP stage.
    best_obj : float
        Its evaluation score.
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print("[LP] cvxpy not available – skipping LP stage.")
        return f0, 2.0 * len(f0) * np.max(np.convolve(f0, f0)) / (np.sum(f0) ** 2)

    n = len(f0)
    f = f0.copy()
    best_f = f.copy()
    best_obj = 2.0 * n * np.max(np.convolve(f, f)) / (np.sum(f) ** 2)

    # we will run at most a handful of LP iterations (each cheap)
    while time.monotonic() < final_deadline:
        conv = np.convolve(f, f)
        M = conv.max()

        # build matrix A where (A @ g)[k] = Σ_i f_i * g_{k-i}
        row_idx = np.arange(2 * n - 1)[:, None]          # (2n‑1,1)
        col_idx = np.arange(n)[None, :]                  # (1,n)
        i_idx = row_idx - col_idx                         # shape (2n‑1,n)
        valid = (i_idx >= 0) & (i_idx < n)
        A = np.where(valid, f[i_idx], 0.0)                # (2n‑1, n)

        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2 * n - 1)]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        except Exception as exc:
            print("[LP] solver error:", exc)
            break

        if g.value is None:
            print("[LP] LP infeasible – aborting LP stage.")
            break

        g_val = np.array(g.value, dtype=np.float64)
        sum_f = np.sum(f)
        sum_g = np.sum(g_val)

        # if the LP cannot increase the ℓ¹‑norm we are done
        if sum_g <= sum_f * (1.0 + 1e-7):
            break

        # adopt the new vector (it keeps the worst convolution ≤ M)
        f = g_val
        cur_obj = 2.0 * n * np.max(np.convolve(f, f)) / (np.sum(f) ** 2)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_f = f.copy()
            print(f"[LP] improved obj={best_obj:.6f}  sum={np.sum(f):.4f}")

        # a very short pause – the loop will stop automatically when
        # the overall time budget expires.
    return best_f, best_obj


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """
    Search for a step‑function that minimises the supplied evaluation
    function.  The routine runs a fast simulated‑annealing phase and,
    if CVXPY is available, a subsequent linear‑programming refinement.
    It respects the 1000 s time budget (the caller reserves a small
    margin).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate total time budget (seconds).  The function stops a
        couple of seconds before the limit so that the caller can still
        retrieve the result.

    Returns
    -------
    list[float]
        The best sequence found (plain Python floats, clipped to the
        scorer’s interval [0, 1000]).
    """
    rng = np.random.default_rng(seed)
    # leave a tiny safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1. initialise a feasible vector (ℓ¹‑norm = 1)
    f0 = _initial_sequence()

    # ------------------------------------------------------------------
    # 2. simulated‑annealing (use ~70 % of the allotted time)
    sa_deadline = min(deadline, time.monotonic() + 0.7 * (deadline - time.monotonic()))
    best_f, best_obj = _run_sa(f0, sa_deadline)

    # ------------------------------------------------------------------
    # 3. linear‑programming refinement (if time permits)
    if time.monotonic() < deadline:
        best_f, best_obj = _run_lp_improve(best_f,
                                           start_deadline=time.monotonic(),
                                           final_deadline=deadline)

    # ------------------------------------------------------------------
    # 4. final post‑processing
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_f.tolist()]
    final_score = _evaluate(best_seq)
    print("[final] best evaluation =", final_score)
    return best_seq
