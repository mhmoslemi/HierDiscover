def project_onto_simplex(v, s):
    # v: np.ndarray
    # ensures non-negative, sum = s
    # algorithm from Duchi
    # If sum(v) == s and all v >=0, return v

"""
Gradient–Projection Search for Low‑Maximum Autocorrelation

The evaluator returns the maximal entry of the autocorrelation
(f * f) when the ℓ¹‑norm of *f* is normalised to √(2 n).  Hence the
objective is simply to minimise this maximum.

The algorithm works independently of external LP solvers:

1. **Projected (sub)‑gradient descent**
   * For a given length *n* we keep the ℓ¹‑norm fixed to √(2 n).
   * In each iteration we compute the convolution, locate the index
     where the maximum is attained, and form the sub‑gradient of that
     entry (∇ (f*f)[k] = 2 · f reversed and shifted).
   * A back‑tracking line search proposes a step; the resulting vector
     is projected onto the simplex { x ≥ 0, ∑x = √(2 n) } (Euclidean
     projection).  The step size is reduced until the maximum autocorrelation
     strictly decreases.
   * The best vector seen during the run is retained.

2. **Multiple random restarts**
   * The descent is executed a few times from different random starts
     (exponential distribution) and from the best known global
     construction (``height_sequence_1`` if available).
   * The best result over all restarts and all tested lengths is kept.

3. **Light‑weight polishing**
   * Finally a cheap stochastic pair‑wise mass‑transfer walk (the
     previous local search) is applied for a few seconds to squeeze
     any remaining improvement.

The whole routine respects the 1000 s budget (a small safety margin
is left).  All numbers are non‑negative and never exceed the allowed
upper bound of 1000.  The returned value is a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global RNG – will be initialised in `propose_candidate`.
_rng: np.random.Generator | None = None


def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    # The evaluator already applies all required checks and clipping.
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex { x ≥ 0, Σx = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and abs(v.sum() - s) <= 1e-12:
        return np.maximum(v, 0.0)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would become zero; return zero vector.
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(f: np.ndarray) -> tuple[float, np.ndarray]:
    """Return the maximal autocorrelation value and the full convolution."""
    conv = np.convolve(f, f)
    return float(np.max(conv)), conv


# ----------------------------------------------------------------------
def _gradient_of_max_entries(f: np.ndarray, conv: np.ndarray,
                            eps: float = 1e-12) -> np.ndarray:
    """
    Compute the sub‑gradient of the (possibly non‑unique) maximal
    entries of `conv = f*f`.  If several indices attain the maximum,
    their gradients are summed.
    """
    n = f.shape[0]
    M = np.max(conv)
    # Choose all indices whose value is within `eps` of the maximum.
    idxs = np.where(conv >= M - eps)[0]
    grad = np.zeros_like(f)

    for idx in idxs:
        start = max(0, idx - (n - 1))
        end = min(idx, n - 1)
        js = np.arange(start, end + 1, dtype=int)
        # gradient contribution: 2 * f[idx - j] for each j in the range
        grad[js] += 2.0 * f[idx - js]

    return grad


# ----------------------------------------------------------------------
def _optimize_length(
    n: int,
    deadline: float,
    max_restarts: int = 4,
    max_iters: int = 5000,
) -> tuple[np.ndarray | None, float]:
    """
    Run projected (sub)‑gradient descent for a fixed length `n`.
    Returns the best sequence found and its score (max autocorrelation).
    """
    target_sum = math.sqrt(2.0 * n)
    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # Use an existing good construction if available.
    global height_sequence_1
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.size >= n:
                seq0 = cand[:n].copy()
                seq0 = np.maximum(seq0, 0.0)
                if seq0.sum() > 0:
                    seq0 = seq0 / seq0.sum() * target_sum
                    sc0 = _score(seq0.tolist())
                    if sc0 < best_score:
                        best_score = sc0
                        best_seq = seq0.copy()
                        print(f"[init] length={n} using height_sequence_1 score={sc0:.6f}")
        except Exception:
            pass

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # Random start – exponential distribution gives a few larger entries.
        f = np.maximum(_rng.exponential(scale=1.0, size=n), 0.0)
        f = f / f.sum() * target_sum

        # If we already have a decent start, reuse it on the first restart.
        if restart == 0 and best_seq is not None:
            f = best_seq.copy()

        cur_score = _score(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[restart] n={n} restart={restart} init_score={cur_score:.6f}")

        # --------------------- gradient descent loop ---------------------
        base_step = 1.0
        it = 0
        while time.monotonic() < deadline and it < max_iters:
            it += 1
            M, conv = _max_convolution(f)

            # Compute sub‑gradient of the (possibly multiple) maximal entries.
            grad = _gradient_of_max_entries(f, conv)
            grad_norm_sq = float(np.dot(grad, grad))
            if grad_norm_sq == 0.0:
                break

            # Initial step size (back‑tracking will shrink it if needed).
            step = base_step * M / grad_norm_sq
            improved = False

            # Back‑tracking line search (max 20 reductions).
            for _ in range(20):
                f_candidate = f - step * grad
                f_candidate = _project_onto_simplex(f_candidate, target_sum)
                cand_score = _score(f_candidate.tolist())
                if cand_score < cur_score - 1e-12:
                    improved = True
                    break
                step *= 0.5

            if not improved:
                # Could not improve – shrink the global step factor.
                base_step *= 0.5
                if base_step < 1e-6:
                    break
                continue

            # Accept the improvement.
            f = f_candidate
            cur_score = cand_score

            # Record the best overall for this length.
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[grad] n={n} restart={restart} iter={it} score={cur_score:.6f}")

            # If we are making good progress, gently increase step size.
            base_step = min(base_step * 1.1, 5.0)

            # Early exit if we have already reached a very low bound.
            if best_score < 1.45:
                break

        # End of one restart.
        if best_score < 1.45:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic pair‑wise mass transfer that preserves the ℓ¹‑norm.
    Keeps improvements; runs until the deadline is reached.
    """
    best = seq.copy()
    best_score = _score(best.tolist())
    n = best.shape[0]

    while time.monotonic() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # Transfer a small random amount from i to j.
        delta = _rng.normal(loc=0.0, scale=0.02) * max(1e-9, best[i])
        if delta == 0.0:
            continue
        cand_i = best[i] + delta
        cand_j = best[j] - delta
        if cand_i < 0.0 or cand_j < 0.0:
            continue
        cand = best.copy()
        cand[i] = cand_i
        cand[j] = cand_j
        sc = _score(cand.tolist())
        if sc < best_score:
            best_score = sc
            best = cand
            print(f"[local] improved score={sc:.6f}")

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """

    """

    _rng = np.random.default_rng(seed)

    # Subtract a tiny safety margin before the external timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # --------------------------------------------------------------
    # 1) Gradient‑descent phase on several problem sizes.
    # --------------------------------------------------------------
    # Sizes that have historically given good results.
    for n in (200, 250, 300, 350, 400, 450, 500):
        if time.monotonic() >= deadline:
            break
        seq, sc = _optimize_length(n, deadline, max_restarts=4, max_iters=8000)
        if seq is not None and sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"[global] new best from n={n}: score={sc:.6f}")

        # Early stop if we have already crossed a very good threshold.
        if overall_best_score < 1.45:
            break

    # --------------------------------------------------------------
    # 2) Polishing with a cheap local walk (if time permits).
    # --------------------------------------------------------------
    if overall_best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_score = _local_search(overall_best_seq, deadline)
        if polished_score < overall_best_score:
            overall_best_score = polished_score

    # 3) Fallback – deterministic safe sequence.
    # --------------------------------------------------------------

    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)
