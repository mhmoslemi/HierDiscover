def simplex_projection(v, s):
    # v is nonnegative? we need nonnegative constraint after projection.
    # Implementation from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - s) / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Improved search for a low‑value step sequence.

The algorithm builds on the LP‑based scheme that produced the
previous best score (≈1.5127) but adds two key upgrades:

1. **Optimal convex combination** – after solving the LP for a
   partner `g` we compute the mixing factor `t∈[0,1]` that
   minimises  max ( (1‑t)f + t g ) ∗ ( (1‑t)f + t g )
   by a 1‑D convex ternary search.  This replaces the fixed
   mixing factor `t=0.2` used before and yields much larger
   improvements per iteration.

2. **Multiple‑restart, multi‑length strategy** – we run the
   LP+optimal‑mixing loop for several values of `n` (400–800)
   and a few random restarts per length.  The best sequence
   encountered over the whole budget is returned.

If CVXPY is unavailable the routine falls back to a simple
random‑walk optimiser.  All helper functions are defined at the
top level, there are no closures or lambdas, and the total run
time is limited to the supplied budget (default 1000 s)."""

# Helper utilities
def _evaluate(seq: np.ndarray) -> float:
    """Fast wrapper for the supplied evaluate_sequence."""
    # evaluate_sequence is assumed to be already imported in the exec env
    return float(evaluate_sequence(seq.tolist()))


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a (possibly negative) vector onto the
    simplex {x ≥ 0, sum(x) = s}.  Implements the algorithm of Duchi et al.
    """
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    # shift to non‑negative by clipping then find appropriate offset
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - s) / (np.arange(len(v)) + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_restart(f_len: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random non‑negative vector of length `f_len` whose sum equals √(2·f_len)."""
    target = math.sqrt(2.0 * f_len)
    # Dirichlet(1) – uniform on the simplex
    x = rng.random(f_len) + 0.1          # avoid exact zeros for numeric stability
    x = x / x.sum() * target
    return x.astype(np.float64)


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) == np.convolve(f, g).
    Used inside the LP – the matrix depends linearly on ``f``.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i ranges over indices of f that appear in output entry k
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _lp_step(f: np.ndarray, M: float, target_sum: float):
    """
    Solve the LP:
        maximise   Σ g_i
        subject to (f∗g)_k ≤ M  (for all k)
                  g_i ≥ 0
    The solution is then scaled to the required ℓ¹‑norm ``target_sum``.
    Returns ``None`` if the LP cannot be solved.
    """
    try:
        import cvxpy as cp
    except Exception:               # CVXPY missing – caller must handle fallback
        return None

    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:               # fall back to a slower solver if needed
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:
            return None

    if g.value is None:
        return None

    g_val = np.maximum(g.value, 0.0)          # eliminate tiny negatives
    s = g_val.sum()
    if s < 1e-12:
        return None
    # Scale to the target ℓ¹‑norm
    g_scaled = g_val * (target_sum / s)
    return g_scaled.astype(np.float64)


def _optimal_mix(A: np.ndarray, B: np.ndarray, C: np.ndarray):
    """
    Find the mixing coefficient t∈[0,1] that minimises
        max_k ((1‑t)²·A_k + 2·t·(1‑t)·B_k + t²·C_k)
    using ternary search (the objective is convex in t).
    Returns (t_opt, M_opt) where M_opt is the achieved maximum.
    """
    lo, hi = 0.0, 1.0
    # 35 iterations are enough for double precision
    for _ in range(35):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        conv1 = (1 - m1) ** 2 * A + 2 * m1 * (1 - m1) * B + (m1 ** 2) * C
        conv2 = (1 - m2) ** 2 * A + 2 * m2 * (1 - m2) * B + (m2 ** 2) * C
        if conv1.max() < conv2.max():
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    M_opt = ((1 - t_opt) ** 2 * A + 2 * t_opt * (1 - t_opt) * B + (t_opt ** 2) * C).max()
    return t_opt, M_opt


def _simple_fallback(budget: float, rng: np.random.Generator):
    """
    Very cheap random walk used when the LP machinery is unavailable.
    Works on a fixed length (n=300) and respects all security checks.
    """
    deadline = time.time() + budget
    n = 300
    target = math.sqrt(2.0 * n)
    seq = np.full(n, target / n, dtype=np.float64)
    best_seq = seq.copy()
    best_score = _evaluate(seq)

    while time.time() < deadline:
        cand = seq + rng.normal(scale=0.04, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / cand.sum() * target
        sc = _evaluate(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            seq = cand
    return best_seq, best_score

                if M_cand < M - 1e-12:
                    f = f_cand
                    M = M_cand
                    cur_score = float(M)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                        print(f"[mix] n={n} restart={restart} it={it} score={cur_score:.6f}")
                    continue

                # As a last resort perform a tiny random perturbation (simulated‑annealing flavour).
                f_rand = f + rng.normal(scale=0.02, size=n)
                f_rand = np.clip(f_rand, 0.0, 1000.0)
                f_rand = f_rand / f_rand.sum() * target_sum
                M_rand = np.convolve(f_rand, f_rand).max()
                if M_rand < M - 1e-12:
                    f = f_rand
                    M = M_rand
                    cur_score = float(M)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                        print(f"[rand] n={n} restart={restart} it={it} score={cur_score:.6f}")
                    continue

                # No progress – give up on this restart after a few stagnant steps.
                # The loop will exit naturally after max_lp_iters.
                # (We deliberately do *not* break early to keep the time budget reasonable.)

    # ------------------------------------------------------------------
    # If nothing was found (e.g. CVXPY unavailable), fall back to a cheap random walk.
    if best_seq is None:
        fallback_budget = max(0.0, deadline - time.time())
        best_seq, _ = _simple_fallback(fallback_budget, rng)

    # Convert the result to a plain Python list of floats (required by the scorer).
    return [float(x) for x in best_seq]
