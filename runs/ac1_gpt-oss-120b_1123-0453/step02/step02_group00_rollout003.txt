def conv_max(f):
    n = len(f)
    size = 1 << (2*n-1).bit_length()  # next pow2 for efficient FFT
    # Actually compute using np.fft.rfft to length size.
    # pad f to size
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(F * F, n=size)  # full linear convolution size
    # conv length size. The linear convolution correct segment is first 2*n-1 entries.
    conv = conv[:2*n-1]
    max_b = np.max(conv)
    return max_b

def simulated_annealing(seq, target_sum, deadline):

    n = len(seq)
    best_seq = seq.copy()
    best_score = _fast_score(seq)  # compute max conv

    current_seq = seq.copy()
    current_score = best_score

    # Determine total time for this SA block
    total_time = deadline - time.monotonic()
    start = time.monotonic()
    # Set T0 as (current_score - ideal?) maybe 0.1 * current_score.
    T0 = max(0.001, current_score * 0.1)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = elapsed / total_time
        # Temperature linear schedule decreasing to 0
        T = T0 * (1 - frac)

        # Step fraction proportion to temperature
        step_frac = 0.2 * (1 - frac) + 0.001  # minimal step

        # Choose move type
        if np.random.rand() < 0.5:
            # Random pair transfer
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            # Determine max amount we can move
            max_transfer = min(current_seq[i], step_frac * target_sum)
            max_transfer = min(max_transfer, 1000.0 - current_seq[j])
            if max_transfer <= 0:
                continue
            delta = np.random.rand() * max_transfer
            cand_seq = current_seq.copy()
            cand_seq[i] -= delta
            cand_seq[j] += delta
        else:
            # Directed move using max convolution index
            conv = conv_fft(current_seq)
            km = np.argmax(conv)
            # contributions
            # compute vector h: for i = 0..n-1 where j=km-i in range
            # Use reversed array for j indexing.
            # For speed we can compute vector of valid j.
            # Let's vectorize:
            i_vals = np.arange(n)
            j_vals = km - i_vals
            mask = (j_vals >= 0) & (j_vals < n)
            h = np.zeros(n)
            h[mask] = current_seq[i_vals[mask]] * current_seq[j_vals[mask]]
            # find i_max, i_min
            # choose i_max = argmax(h), i_min = argmin(h) among those with mask true.
            if not np.any(mask):
                continue
            valid_indices = i_vals[mask]
            h_valid = h[mask]
            i_max = valid_indices[np.argmax(h_valid)]
            i_min = valid_indices[np.argmin(h_valid)]
            if i_max == i_min:
                continue
            max_transfer = min(current_seq[i_max], step_frac * target_sum)
            max_transfer = min(max_transfer, 1000.0 - current_seq[i_min])
            if max_transfer <= 0:
                continue
            delta = np.random.rand() * max_transfer
            cand_seq = current_seq.copy()
            cand_seq[i_max] -= delta
            cand_seq[i_min] += delta

        max_b = conv_fft(cand_seq)

        # acceptance
        if cand_score < current_score:
            accept = True
        else:
            # compute acceptance probability
            if T > 0:
                prob = math.exp(-(cand_score - current_score) / T)
                accept = np.random.rand() < prob
            else:
                accept = False

        if accept:

            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()
                # Log improvement
                print(f"[SA] n={n} iter={int(elapsed*1000)} score={cand_score:.6f} T={T:.6f}")

        if np.random.rand() < 0.05:
            # gradient descent step (optional)
            # compute km again
            km = np.argmax(conv_fft(current_seq))
            rev = np.zeros_like(current_seq)
            # g_i = 2 * seq[km - i] for valid i
            for i in range(n):

                if 0 <= j < n:
                    rev[i] = current_seq[j] * 2.0
            # step size based on temperature
            alpha = step_frac * 0.5
            grad_seq = current_seq - alpha * rev
            # clip non negative
            grad_seq = np.clip(grad_seq, 0.0, 1000.0)
            # maintain sum
            sum_grad = np.sum(grad_seq)

Search for a low‑value non‑negative step sequence.

The algorithm consists of three phases:

1️⃣  Warm‑start – if a global variable ``height_sequence_1`` is present,
    it is resized (by linear interpolation) to the current length and
    rescaled so that the ℓ¹‑norm equals √(2·n).  Otherwise a random vector
    with the same ℓ¹‑norm is created.

2️⃣  Simulated‑annealing (SA) – repeated pairwise mass transfers and
    “directed’’ moves that shift mass away from the indices that contribute
    most to the current maximal auto‑convolution entry.  The SA schedule
    (linear cooling, step size proportional to temperature) drives the
    evaluation ``2·n·max(f*f)/‖f‖₁²`` down while keeping the sum of the
    coefficients fixed.  Fast evaluation is performed with an FFT‑based
    convolution, because the original ``evaluate_sequence`` is much slower.

3️⃣  Final stochastic polishing – a simple Gaussian perturbation hill‑climb
    (similar to the reference code) cleans up the best candidate found so
    far.

All helper functions are top‑level and the only public entry point is
``propose_candidate`` which returns a plain Python list of floats.
"""

import numpy as np

# ----------------------------------------------------------------------
# Helper functions – fast convolution and scoring
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``seq`` with itself using a real FFT.
    Returns an array of length ``2·n‑1``.
    """
    n = seq.shape[0]
    # next power of two for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` but with an FFT‑based convolution.
    The sequence is assumed already to lie in the interval [0, 1000].
    """
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Local refinement (Gaussian perturbation)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that adds zero‑mean Gaussian noise.
    The sum of the coefficients is *not* forced to stay constant – the
    true ``evaluate_sequence`` will re‑normalise the vector automatically.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing on a fixed length
# ----------------------------------------------------------------------


def _simulated_annealing(seq: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    SA that preserves the ℓ¹‑norm (by moving mass between two entries)
    and constantly tries to lower the evaluation score.
    """
    n = seq.shape[0]
    # Make sure the starting vector respects the constraints
    seq = np.clip(seq, 0.0, 1000.0)
    cur_sum = seq.sum()
    if cur_sum > 0:
        seq = seq * (target_sum / cur_sum)
    cur_score = _fast_score(seq)

    best_seq = seq.copy()
    best_score = cur_score

    start_time = time.monotonic()
    total_time = deadline - start_time
    # linear cooling from T0 to 0
    T0 = max(1e-4, cur_score * 0.08)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time
        T = T0 * (1.0 - frac)          # temperature
        step_frac = 0.25 * (1.0 - frac) + 1e-5   # decreasing transfer size

        # ------- choose a move ----------------------------------------
        if rng.random() < 0.55:
            # random pairwise transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            # maximum transferable mass without violating bounds
            max_transfer = min(seq[i], step_frac * target_sum)
            max_transfer = min(max_transfer, 1000.0 - seq[j])
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            cand = seq.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # directed move – shift mass away from the indices that
            # dominate the current worst convolution entry
            conv = _conv_full_fft(seq)
            km = int(conv.argmax())
            # products contributing to the current worst entry
            idx = np.arange(n)
            jdx = km - idx
            mask = (jdx >= 0) & (jdx < n)
            if not np.any(mask):
                continue
            prod = np.zeros(n)
            prod[mask] = seq[idx[mask]] * seq[jdx[mask]]
            i_max = int(idx[mask][np.argmax(prod[mask])])
            i_min = int(idx[mask][np.argmin(prod[mask])])
            if i_max == i_min:
                continue
            max_transfer = min(seq[i_max], step_frac * target_sum)
            max_transfer = min(max_transfer, 1000.0 - seq[i_min])
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            cand = seq.copy()
            cand[i_max] -= delta
            cand[i_min] += delta

        # ------- evaluate candidate ------------------------------------
        cand_score = _fast_score(cand)

        # ------- acceptance -------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(cand_score - cur_score) / T)
                accept = rng.random() < prob

        if accept:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[SA] n={n:4d} iter={int(elapsed*1000):7d} "
                      f"score={cand_score:.6f} T={T:.5f}")

        # occasional cheap gradient‑like step (no extra conv needed)
        if rng.random() < 0.04:
            conv = _conv_full_fft(seq)
            km = int(conv.argmax())
            rev = np.zeros_like(seq)
            # gradient of the worst entry: 2·f[km‑i] (if in range)
            idx = np.arange(n)
            jdx = km - idx
            in_range = (jdx >= 0) & (jdx < n)
            rev[in_range] = 2.0 * seq[jdx[in_range]]
            alpha = step_frac * 0.4
            grad_seq = seq - alpha * rev
            grad_seq = np.clip(grad_seq, 0.0, 1000.0)
            # re‑normalise the ℓ¹‑norm
            s = grad_seq.sum()
            if s > 0:
                grad_seq = grad_seq * (target_sum / s)
            grad_score = _fast_score(grad_seq)
            if grad_score < cur_score:
                seq = grad_seq
                cur_score = grad_score
                if grad_score < best_score:
                    best_seq = grad_seq.copy()
                    best_score = grad_score
                    print(f"[SA‑grad] n={n:4d} iter={int(elapsed*1000):7d} "
                          f"score={grad_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Interpolation utility – enlarge a known shape to a longer length
# ----------------------------------------------------------------------


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of ``seq`` to length ``new_n``.
    The output is clipped to [0,1000] but otherwise untouched.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return np.clip(new_seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the supplied time budget and returns a plain
    Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # a tiny safety margin so that the caller can still read the result
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.3)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # lengths to explore – a progression from modest to larger sizes
    lengths = [500, 600, 800, 1000]

    # retrieve the global reference sequence if it exists
    known_seq = globals().get("height_sequence_1", None)
    if known_seq is not None:
        known_seq = np.asarray(known_seq, dtype=np.float64)

    # map to keep the best result for each dimension (useful for interpolation)
    best_by_len: dict[int, np.ndarray] = {}

    # allocate roughly 70 % of the time budget to SA, the rest to a final polish
    total_sa_time = (deadline - time.monotonic()) * 0.70
    time_per_len = total_sa_time / max(1, len(lengths))

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------- initialise -------------------------------------------------
        if known_seq is not None and known_seq.shape[0] == n:
            init_seq = known_seq.copy()
        elif best_by_len:
            # take the best sequence we already have (the one with closest size)
            # and interpolate it to the new length
            # pick the longest already‑computed sequence <= n
            src_n = max(k for k in best_by_len.keys() if k <= n)
            init_seq = _interpolate(best_by_len[src_n], n)
        else:
            # purely random start
            init_seq = rng.random(n) + 0.05

        # normalise ℓ¹‑norm to √(2·n)
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        cur_sum = init_seq.sum()
        if cur_sum > 0:
            init_seq = init_seq * (target_sum / cur_sum)

        # ------- simulated annealing ----------------------------------------
        sa_deadline = time.monotonic() + time_per_len
        cand_seq, cand_score = _simulated_annealing(init_seq,
                                                    target_sum,
                                                    sa_deadline,
                                                    rng)
        best_by_len[n] = cand_seq.copy()

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[global] new overall best: score={best_score:.6f} (n={n})")

        # early exit if we already beat the target
        if best_score < 1.503:
            print("[global] target reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Final polishing with a simple Gaussian hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined_seq, refined_score = _local_search(best_seq, deadline,
                                                   step_initial=0.5)
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"[final] improved to {best_score:.6f}")

    # safety fallback – should never trigger
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = best_seq * (math.sqrt(2.0 * n) / best_seq.sum())

    # Return as a plain Python list
    return [float(v) for v in best_seq]
