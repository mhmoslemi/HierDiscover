"""
Propose a low‑value step sequence.

The algorithm performs many short, independent stochastic searches.
Each search starts from a random (exponential) vector whose ℓ¹‑norm is
fixed to the theoretical optimum √(2 n).  It then applies a *targeted
hill‑climber* that repeatedly:

1. evaluates the self‑convolution,
2. finds the index k where the convolution attains its maximum,
3. measures for every position i how much it contributes to that
   worst entry (product a_i·a_{k‑i}),
4. moves a small amount of mass from the biggest contributor to a
   location that contributes little (or is empty).

The move is accepted if it improves the score, otherwise it may be
accepted with a tiny Metropolis probability (simulated‑annealing) to
escape local minima.  The step size slowly shrinks and the temperature
is annealed.  Several different sequence lengths (n≈150‑800) and a few
random restarts are tried within the allotted time; the best sequence
found is returned.

The implementation evaluates the objective itself (max (conv)) which
is much faster than calling the provided ``evaluate_sequence`` during
the search, but the final list is returned in the required format.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence (identical for valid inputs)."""
    n = len(seq)
    if n == 0:
        return math.inf
    b = np.convolve(seq, seq)
    m = np.max(b)
    s = seq.sum()
    if s < 0.01:
        return math.inf
    return float(2.0 * n * m / (s * s))


# ----------------------------------------------------------------------
def _targeted_hill_climb(a: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         rng: np.random.Generator) -> np.ndarray:
    """
    Hill‑climber that moves mass away from indices that heavily influence
    the current worst convolution entry.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_sc = _fast_score(best)

    temperature = 1.0            # annealing temperature
    step_factor = 0.12           # fraction of the source mass moved each step
    it = 0

    # pre‑compute the “average” mass per entry – useful for picking a light
    # destination.
    avg_mass = target_sum / n

    while True:
        if time.monotonic() >= deadline:
            break
        if it >= 20000:                     # safety cap
            break
        it += 1

        # --------------------------------------------------------------
        # 1) current convolution and worst index
        conv = np.convolve(cur, cur)
        k_max = int(conv.argmax())          # index where max occurs

        # --------------------------------------------------------------
        # 2) contribution of each position to the worst entry
        # contribution_i = cur[i] * cur[k_max - i]   (0 if partner out of range)
        idx = k_max - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        partner = np.zeros_like(cur)
        partner[valid] = cur[idx[valid]]
        contrib = cur * partner               # ordered contribution (i,j) and (j,i)

        # pick a donor – the index with the largest contribution that still
        # holds positive mass
        if np.any(contrib > 0):
            donor = int(np.argmax(contrib))
        else:                                 # fall‑back: random positive index
            donor_candidates = np.where(cur > 0)[0]
            if donor_candidates.size == 0:
                break
            donor = int(rng.choice(donor_candidates))

        # --------------------------------------------------------------
        # 3) pick a receiver – an index that contributes little to the worst
        #    entry (or is simply light overall)
        light = np.where(cur <= avg_mass * 0.5)[0]
        if light.size > 0:
            receiver = int(rng.choice(light))
        else:
            # if all entries are “heavy”, pick a random different index
            receiver = int(rng.integers(0, n))
            while receiver == donor:
                receiver = int(rng.integers(0, n))

        # --------------------------------------------------------------
        # 4) compute transfer amount
        delta = cur[donor] * step_factor       # a fraction of the donor's mass
        if delta <= 0.0:
            continue
        # keep the move small enough not to make the donor negative
        delta = min(delta, cur[donor] * 0.9)

        cand = cur.copy()
        cand[donor] -= delta
        cand[receiver] += delta

        # small numerical cleanup – keep non‑negativity
        if cand[donor] < 0.0:
            continue

        cand_sc = _fast_score(cand)

        # --------------------------------------------------------------
        # 5) acceptance rule (Metropolis)
        accept = False
        if cand_sc < best_sc:
            accept = True
        else:
            # exponential acceptance probability, temperature slowly decays
            prob = math.exp(-(cand_sc - best_sc) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
                print(f"[hill‑target] n={n} it={it} score={best_sc:.6f}")

        # --------------------------------------------------------------
        # 6) schedule
        temperature = max(1e-6, temperature * 0.9995)
        step_factor = max(0.01, step_factor * 0.9997)

        # occasional renormalisation to fix tiny drift
        if it % 200 == 0:
            cur_sum = cur.sum()
            if cur_sum > 0:
                cur = cur * (target_sum / cur_sum)

    return best


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: float = 1100,
                      **kwargs):
    """
    Return a list of non‑negative heights whose evaluation score is as low
    as possible (target < 1.503).  The routine performs many short,
    independent searches with the targeted hill‑climber described above.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the outer harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction is available, use it as a warm start.
    if "height_sequence_1" in globals():
        init = np.array(list(height_sequence_1), dtype=np.float64)
        n0 = init.size
        target0 = math.sqrt(2.0 * n0)
        init = init * (target0 / max(init.sum(), 1e-12))
        init = np.clip(init, 0.0, 1000.0)
        sc0 = _fast_score(init)
        best_seq, best_score = init, sc0
        print("[init] warm start from height_sequence_1, score =", sc0)

    # ------------------------------------------------------------------
    # Search over several problem sizes.  Larger n give a richer search
    # space but also cost more per iteration – we cap the size to keep
    # the total running time well below the budget.
    candidate_sizes = (160, 200, 250, 300, 350, 400, 450, 500, 600)

    for n in candidate_sizes:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of independent restarts for this length – we keep it low
        # when time is running out.
        max_trials = 4
        for trial in range(max_trials):
            if time.monotonic() >= deadline:
                break

            # ---- random initial vector (exponential to avoid many exact zeros)
            a0 = rng.exponential(scale=1.0, size=n) + 1e-6
            a0 = a0 / a0.sum() * target_sum

            # ---- improve it with the targeted hill‑climber
            improved = _targeted_hill_climb(a0, target_sum, deadline, rng)

            sc = _fast_score(improved)
            if sc < best_score:
                best_score, best_seq = sc, improved.copy()
                print(f"[global] n={n} trial={trial} new best={best_score:.6f}")

            if best_score < 1.503:         # target met – stop early
                break
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: a simple uniform sequence – always feasible.
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _fast_score(best_seq)
        print("[fallback] using uniform sequence, score =", best_score)

    # Return a plain Python list of floats (the external harness will call
    # evaluate_sequence on it).
    return [float(x) for x in best_seq]
