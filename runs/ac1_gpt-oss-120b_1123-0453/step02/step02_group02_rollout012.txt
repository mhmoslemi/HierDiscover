def _targeted_local_search(seq, deadline):
    a = np.array(seq, dtype=np.float64)
    n = len(a)
    target_sum = a.sum()   # Already normalized.
    best_score = _score(a.tolist())
    best_seq = a.copy()
    # parameters:
    # step fraction scale initial e.g., 0.2; gradually reduce.
    init_frac = 0.2
    # we shall run while time left and iteration < max_iters (maybe 200000)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute conv and find argmax
        conv = np.convolve(a, a)
        M = conv.max()
        kmax = conv.argmax()
        # Build pair indices contributing.
        # i ranges from max(0, kmax - (n-1)) to min(kmax, n-1)
        i_start = max(0, kmax - (n - 1))
        i_end = min(kmax, n - 1)
        # Choose random i in this range
        i = _rng.integers(i_start, i_end + 1)
        j = _rng.integers(0, n)
        if i == j:
            continue
        # Determine allowable delta: at most a[i] (can't go negative)
        max_delta = a[i]
        # Choose delta = fraction * max_delta
        # step size factor based on iteration: decaying
        frac = init_frac / (1 + it * 0.0001)   # decaying a bit
        delta = frac * max_delta * (_rng.random() - 0.5)   # allow +-?
        # Actually we want move mass from i to j, not from j to i; delta positive reduces i and adds to j.
        # So delta = _rng.random() * max_delta?
        delta = _rng.random() * max_delta * 0.1  # small.
        # new values
        new_i = a[i] - delta
        new_j = a[j] + delta
        if new_i < 0: continue
        cand = a.copy()
        cand[i] = new_i
        cand[j] = new_j
        # No need to renormalize sum because we transferred mass.
        # Evaluate score
        sc = _score(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            a = cand
            # optionally print
            print(f"[targeted] n={n} score={sc:.6f} it={it}")
        it += 1
```

def _smooth_gradient_descent(a, target_sum, deadline, max_iters=5000):
    n = len(a)
    best = a.copy()
    best_score = _score(best.tolist())
    step0 = target_sum * 0.05   # initial step size
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        k = conv.argmax()
        # compute gradient of max conv
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = a.copy()
        grad[mask] += a[idx[mask]]
        # step size
        step = step0 / (1 + it * 0.01)
        a_new = a - step * grad
        # project onto simplex
        a_new = _project_simplex(a_new, target_sum)

            best = a_new.copy()
            a = a_new
            print(f"[grad] n={n} it={it} score={sc:.6f}")
        else:
            # maybe accept if not degrade large? Not.
            a = a_new
    return best, best_score

def _project_simplex(v, s):
    # v is numpy array, s >0
    # algorithm: sort descending
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Annealed gradient descent over the simplex.

The evaluator is scale‑invariant: for a fixed length `n` the score
    2·n·max(conv(a,a)) / (sum(a)²)
is minimised by reducing the largest entry of the auto‑convolution.
We therefore fix the ℓ¹‑norm to `sqrt(2·n)` (the normalisation used by the
scorer) and minimise the maximal convolution entry.

The algorithm works as follows for a chosen length `n`.

1. Initialise a non‑negative vector `a` with the required ℓ¹‑norm.
   Several random restarts are performed; if the global variable
   `height_sequence_1` is present it is used as a privileged start.

2. At each iteration we compute the full convolution `c = a*a`.
   A temperature `T` controls a soft‑max weighting
        w_k ∝ exp((c_k – max(c)) / T)
   which smoothly approximates the indicator of the maximal index.
   The gradient of the smoothed objective is
        ∇ = a + (w ∗ a_rev)[n‑1 : n‑1+n]
   where `a_rev` is `a` reversed and `∗` denotes ordinary convolution.
   A diminishing step `η` moves `a` opposite to the gradient, after which
   `a` is projected onto the simplex {x ≥ 0, ∑x = sqrt(2·n)}.
   The temperature is multiplied by a factor < 1 each iteration, so the
   optimisation gradually focuses on the true maximum.

3. The best vector seen during the run is kept.  After the annealed phase
   a cheap stochastic local‑search (mass swaps) is applied for a few
   seconds to polish the result.

The whole routine respects the 1000 s budget, tries several lengths
(`200 … 500`), and finally returns the best sequence found.
All numbers are non‑negative and clipped to the allowed interval [0, 1000].
"""

import time
import math
import numpy as np

# Global RNG – initialised in `propose_candidate`.
_rng = None

# ----------------------------------------------------------------------
def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex
        { x | x_i ≥ 0,  Σ x_i = s }.
    Implementation follows the classic O(n log n) algorithm.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Shift by the mean to improve numerical stability.
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero; fallback
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _annealed_gradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Annealed (soft‑max) gradient descent on the simplex.
    Returns the best sequence seen and its score.
    """
    n = a0.shape[0]
    a = a0.copy()
    best = a.copy()
    best_score = _score(best.tolist())

    # Initial temperature – a fraction of the current max convolution.
    conv = np.convolve(a, a)
    T = max(conv.max() * 0.5, 1e-6)

    # Base step size – proportional to the ℓ¹‑norm.
    step0 = target_sum * 0.5

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # Full convolution and its maximum.
        conv = np.convolve(a, a)
        maxc = conv.max()

        # Annealing schedule.
        T = max(1e-8, T * 0.985)          # slowly cool down

        # Soft‑max weights over convolution entries.
        # Shift by maxc to avoid overflow.
        exp_shift = (conv - maxc) / T
        # Clamp extreme values for safety.
        exp_shift = np.clip(exp_shift, -80.0, 0.0)
        w = np.exp(exp_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # Gradient of the smoothed max: grad = a + (w * a_rev) slice.
        a_rev = a[::-1]
        # Convolution length = (2n‑1) + n‑1 = 3n‑2.
        tmp = np.convolve(w, a_rev)
        grad_shift = tmp[n - 1 : n - 1 + n]       # exactly n entries
        grad = a + grad_shift

        # Diminishing step size.
        step = step0 / (1.0 + it * 0.01)

        # Gradient step and projection onto the simplex.
        a_new = a - step * grad
        a_new = _project_simplex(a_new, target_sum)

        # Evaluate.
        sc = _score(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best = a_new.copy()
            print(f"[anneal] n={n:4d} it={it:4d} score={sc:.6f}")

        a = a_new

    return best, best_score


# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic walk that preserves the ℓ¹‑mass.
    Randomly transfers a tiny amount of mass between two indices.
    """
    best = seq.copy()
    best_score = _score(best.tolist())
    n = best.shape[0]

    while time.time() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # Propose a transfer proportionally to the source magnitude.
        delta = _rng.normal(loc=0.0, scale=0.02) * max(1e-12, best[i])
        if delta == 0.0:
            continue
        cand_i = best[i] + delta
        cand_j = best[j] - delta
        if cand_i < 0.0 or cand_j < 0.0:
            continue
        cand = best.copy()
        cand[i] = cand_i
        cand[j] = cand_j
        sc = _score(cand.tolist())
        if sc < best_score:
            best_score = sc
            best = cand
            print(f"[local] score={sc:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
def _search_length(
    n: int,
    deadline: float,
    restarts: int = 4,
) -> tuple[np.ndarray | None, float]:
    """
    Run several random restarts of the annealed gradient descent for a
    given length `n`.  Returns the best sequence and its score, or (None,
    inf) if the time budget is exhausted.
    """
    target_sum = math.sqrt(2.0 * n)
    best_seq = None
    best_score = float("inf")

    for r in range(restarts):
        if time.time() >= deadline:
            break

        # Initialise: uniform + Gaussian noise, then project onto simplex.
        base = np.full(n, target_sum / n, dtype=np.float64)
        noise = _rng.normal(loc=0.0, scale=0.2, size=n)
        a0 = base * (1.0 + noise)
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0, target_sum)

        seq, sc = _annealed_gradient_descent(a0, target_sum, deadline)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[search] n={n:4d} restart={r} new best={sc:.6f}")

    return best_seq, best_score

            print(f"[init] using height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception:   # pragma: no cover
            overall_best_seq = None
            overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Search over a list of lengths.
    # ------------------------------------------------------------------
    for n in (200, 250, 300, 350, 400, 450, 500):
        if time.time() >= deadline:
            break
        seq_n, score_n = _search_length(n, deadline, restarts=4)
        if seq_n is not None and score_n < overall_best_score:
            overall_best_score = score_n
            overall_best_seq = seq_n.copy()
            print(f"[global] new best after size {n}: {score_n:.6f}")

        # Early stop if we already beat a strong bound.
        if overall_best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # 2) Final polishing with a lightweight random walk.
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.time() < deadline:
        polished, polished_score = _local_search(overall_best_seq, deadline)
        if polished_score < overall_best_score:
            overall_best_seq = polished
            overall_best_score = polished_score
            print(f"[global] after polishing: {polished_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback – a uniform vector if everything else failed.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 256
        target_sum = math.sqrt(2.0 * n)
        overall_best_seq = np.full(n, target_sum / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq.tolist())
        print("[fallback] returning uniform construction")

    # Final safety clipping.
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    return [float(x) for x in overall_best_seq.tolist()]
