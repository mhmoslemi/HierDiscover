def subgradient_descent(f, target_sum, max_iter, deadline):
    n = len(f)
    step = 0.01 * target_sum  # initial step
    best_f = f.copy()
    best_score = _score(f)
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        conv = np.convolve(f, f)
        M = conv.max()
        # early termination if M close to best known target; but we want to improve.
        # find active shifts
        active = np.where(np.abs(conv - M) < 1e-12)[0]  # tolerance.

        # decay step

Implementation details: Use the algorithm for projection onto a simplex of sum S > 0.

Algorithm for projection onto simplex:

Given vector v (any real). Want find x = argmin ||x - v||_2 such that x_i >= 0 and sum x_i = S.

Solution: sort v descending, find rho = max {j | v_j - (1/(j+1)) (sum_{i=1}^{j+1} v_i - S) > 0} ; compute theta = (sum_{i=1}^{rho+1} v_i - S) / (rho+1); set x_i = max(v_i - theta, 0). That's O(n log n). For n up to few thousand, fine.

Implementation:

```
def project_simplex(v, s):
    n = v.shape[0]
    if s < 0:
        raise ValueError
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def conv_fft(f):
    n = f.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(F*F, n=size)[:2*n-1]

The algorithm uses a projected sub‑gradient descent on the max‑auto‑convolution
objective, followed by a short stochastic hill‑climbing refinement.

import time
import math
import numpy as np

def _auto_convolve_fft(f):
    """Return the auto‑convolution f∗f using a power‑of‑two FFT."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(F * F, n=size)[: 2 * n - 1]
    return conv

def _project_onto_simplex(v, s):
    """Project vector v onto the simplex {x≥0 , sum x = s}."""
    # Implementation of the algorithm from
    # "Efficient Projections onto the l1‑Ball for Learning in High Dimensions"
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgradient_descent(f0, target_sum, deadline, max_iter=2000):
    """Projected sub‑gradient descent on max(f∗f)."""
    n = f0.shape[0]
    f = f0.copy()
    best_f = f.copy()

    conv = _auto_convolve_fft(f)

"""
Projection‑based sub‑gradient search for low‑value step sequences.

The algorithm fixes the ℓ¹‑norm of the vector to √(2·n), because then
evaluate_sequence(f) = max(f∗f).  For a given length n the program
optimises this maximum by a projected sub‑gradient descent, starting
from several random (and a deterministic) initial vectors.  The best
vector discovered over a range of lengths (up to a few thousand) is
then polished with a simple stochastic hill‑climber.  All work respects
the global time budget (≈1000 s) and the security checks performed by
``evaluate_sequence``.
"""

import time
import math
import numpy as np

def _auto_convolve_fft(f: np.ndarray) -> np.ndarray:
    """Return the full auto‑convolution f∗f using a power‑of‑two FFT."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    # rfft automatically pads with zeros when n < size
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(F * F, n=size)[: 2 * n - 1]
    return conv

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x≥0, sum x = s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / len(v)            # all entries become zero
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgradient_descent(f0: np.ndarray, target_sum: float,
                         deadline: float, max_iter: int = 2000):
    """
    Projected sub‑gradient descent that minimises max(f∗f).

    The objective equals the maximal entry of the auto‑convolution because
    the ℓ¹‑norm of the vector is kept equal to `target_sum = √(2·n)`.
    """
    n = f0.shape[0]
    f = f0.copy()
    best_f = f.copy()
    conv = _auto_convolve_fft(f)
    best_score = float(conv.max())

    # Step‑size scaling – 5 × average entry, then 1/√t decay.
    step0 = 5.0 * target_sum / n

    it = 0
    while it < max_iter and time.monotonic() < deadline:
        it += 1
        conv = _auto_convolve_fft(f)
        M = conv.max()
        # Shifts attaining the current maximum
        active = np.where(np.abs(conv - M) <= 1e-12)[0]

        # Sub‑gradient of the max‑shift(s)
        grad = np.zeros_like(f)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                grad[i] += f[j]
                grad[j] += f[i]

        # Gradient step (move opposite to increase of M)
        step = step0 / math.sqrt(it)
        f = f - step * grad

        # Project back onto the feasible simplex
        f = _project_onto_simplex(f, target_sum)

        cur_score = float(M)           # because target_sum² = 2 n
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[SGD] iter={it:4d} n={n:4d} score={cur_score:.6f} M={M:.6f}")

        if best_score <= 1.502:        # early exit if target already met
            break

    return best_f, best_score

                  deadline: float,

    return best, best_score
