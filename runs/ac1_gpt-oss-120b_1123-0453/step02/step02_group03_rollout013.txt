def hill_climb(seq, deadline, rng):
    cur_seq = seq.copy()
    cur_score = _score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    start = time.monotonic()
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total_allowed = deadline - start
        frac = elapsed / total_allowed
        T = temp_start * (temp_end / temp_start) ** frac
        scale = max(1e-6, T * init_scale) # scaling amplitude
        # pick two indices
        i, j = rng.integers(0, len(seq), size=2)
        if i == j: continue
        delta = scale * rng.normal()
        # limit delta relative to a_i and a_j
        if delta > 0:
            delta = min(delta, cur_seq[i]) # i loses
        else:
            delta = max(delta, -cur_seq[j]) # i gains (since delta negative, moving from j to i)
        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonneg and clip
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[HC] new best {best_score:.6f} at iter {iteration}")
        # occasionally mutate entire vector partially
        if rng.random() < 0.001:
            # random restart:
            pert = np.maximum(cur_seq + rng.normal(scale=0.1 * np.max(cur_seq), size=len(seq)), 0.0)
            cur_seq = np.clip(pert,0,1000)
            cur_score = _score(cur_seq)
        iteration += 1
    return best_seq, best_score

def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.5  # safety margin
    best_seq = None
    best_score = float('inf')
    lengths = [200, 300, 400, 500, 600, 800, 1000]
    # try each length for some allocated time proportionally
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # allocate time: e.g., remaining_time/(remaining_lengths)
        remaining_lengths = len(lengths) - lengths.index(n)
        rem_time = deadline - time.monotonic()
        allocate = rem_time / remaining_lengths
        # generate initial seq: random Dirichlet distribution scaled
        init = rng.random(n) + 1e-3
        init = init / np.sum(init) * math.sqrt(2.0 * n)
        # optionally shape to edge heavy:
        # we can also try multiple start patterns: uniform, edge, random.
        # We'll run hillclimb multiple times with different inits.
        # For each start pattern:
        for pattern in range(3):
            # produce different initial pattern:
            if pattern == 0:
                seq0 = init.copy()
            elif pattern == 1:
                # edge heavy: two spikes at ends
                seq0 = np.full(n, 1e-6)
                half = math.sqrt(2.0 * n) / 2
                seq0[0] = half
                seq0[-1] = half
                seq0 = np.clip(seq0, 0.0, 1000.0)
            else:
                # Gaussian shape
                x = np.arange(n)
                center = (n-1)/2
                sigma = n/8

                gauss = gauss + 1e-6
                seq0 = gauss

                best_seq = seq.copy()

    # fallback if none found
    if best_seq is None:

def targeted_hill_climb(seq, deadline, rng):
    cur_seq = seq.copy()
    cur_score = _score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    n = len(seq)
    start = time.monotonic()
    iter_count = 0

    # Define temperature schedule (maybe just a cooling factor)
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        # compute conv once every few iterations? maybe each iteration
        conv = np.convolve(cur_seq, cur_seq)
        M = float(conv.max())
        idx_max = int(conv.argmax())

        # Determine candidate indices that contribute to idx_max:
        # i in [max(0, idx_max-(n-1)), min(n-1, idx_max)]
        i_start = max(0, idx_max-(n-1))
        i_end = min(n-1, idx_max)
        # choose i randomly from that range weighted by a_i (higher more likely)
        # compute weights = cur_seq[i_start:i_end+1]
        weights = cur_seq[i_start:i_end+1]
        if np.sum(weights)>0:
            probs = weights / np.sum(weights)
            i = rng.choice(np.arange(i_start, i_end+1), p=probs)
        else:
            i = rng.integers(i_start, i_end+1)
        j = idx_max - i
        # choose move type
        move_type = rng.integers(2)
        # step scale based on temperature
        temp = 1.0 - (elapsed/total)  # decreasing linearly from 1 to 0
        step_scale = temp * 0.5 * np.mean(cur_seq)  # maybe fraction of avg

        if move_type == 0:
            # move epsilon from i to random other index
            epsilon = min(step_scale * rng.random(), cur_seq[i])
            if epsilon <= 0:
                continue
            k = rng.integers(0, n)
            # ensure k != i
            while k == i:
                k = rng.integers(0, n)
            cand = cur_seq.copy()
            cand[i] -= epsilon
            cand[k] += epsilon
        else:
            # move epsilon from both i and j (if i!=j) to two random others
            epsilon = min(step_scale * rng.random(), cur_seq[i])
            # For j
            epsilon_j = min(step_scale * rng.random(), cur_seq[j])
            # target indices
            k1 = rng.integers(0, n)
            while k1 == i:
                k1 = rng.integers(0, n)
            k2 = rng.integers(0, n)
            while k2 == j or (k2==k1):
                k2 = rng.integers(0, n)
            cand = cur_seq.copy()
            cand[i] -= epsilon
            cand[k1] += epsilon
            cand[j] -= epsilon_j
            cand[k2] += epsilon_j

        # clip
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        iter_count += 1

        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[T-HC] iter {iter_count} new best {best_score:.6f}")

        # occasional global perturbation
        if rng.random() < 0.001:
            # random small perturbation to all entries
            perturb = rng.normal(scale=0.01*np.max(cur_seq), size=n)
            cand = cur_seq + perturb
            cand = np.maximum(cand, 0.0)
            cand = np.clip(cand, 0.0, 1000.0)
            cur_seq = cand
            cur_score = _score(cur_seq)

    return best_seq, best_score

def gradient_descent_step(seq, eta):
    conv = np.convolve(seq, seq)
    M = conv.max()
    active_ks = np.flatnonzero(conv >= M - 1e-12)  # indices where conv == M
    # compute gradient
    grad = np.zeros_like(seq)
    n = len(seq)
    for k in active_ks:
        # for i from max(0, k-(n-1)) to min(k, n-1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            # derivative w.r.t a_i: a_j + (a_i if i==j else a_i)
            # Actually we add a_j + a_i
            grad[i] += seq[j] + seq[i]
    # Now we have gradient to increase M. To decrease M, move opposite direction.
    # We'll adjust seq by -eta * grad
    new_seq = seq - eta * grad
    # enforce nonnegativity
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    # rescale to keep sum same (or at least non zero)
    total = new_seq.sum()
    if total < 1e-6:
        return seq
    # optional: keep sum same as original (some constant)
    new_seq = new_seq * (seq.sum() / total)
    return new_seq

    cur_seq = seq.copy()
    cur_score = _score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    n = len(seq)

    start = time.monotonic()
    iteration = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        # cooling schedule
        frac = elapsed / total
        # base step size
        base = np.mean(cur_seq) * 0.1
        eta = base * (1 - frac)  # linearly decreasing
        # compute conv and gradient
        conv = np.convolve(cur_seq, cur_seq)
        M = float(conv.max())
        active_ks = np.flatnonzero(conv >= M - 1e-12)
        # compute gradient
        grad = np.zeros_like(cur_seq)
        for k in active_ks:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                grad[i] += cur_seq[j] + cur_seq[i]
        # update
        cand = cur_seq - eta * grad
        cand = np.clip(cand, 0.0, 1000.0)
        # Rescale to preserve sum (optional)
        s = np.sum(cand)
        if s > 0:
            cand *= (np.sum(cur_seq) / s)
        cand_score = _score(cand)

        # acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(1e-6, (1-frac))):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[GD] n={n} it={iteration} new best {best_score:.6f}")

        # occasional random jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=0.01*np.mean(cur_seq), size=n)
            cand2 = cur_seq + jitter
            cand2 = np.clip(cand2, 0.0, 1000.0)

                    best_seq = cand2.copy()
                    print(f"[GD] jitter new best {best_score:.6f}")

def _targeted_hill_climb(seq, deadline, rng):
    cur = seq.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score
    n = len(seq)
    start = time.monotonic()
    it = 0
    while time.monotonic() < deadline:
        # compute conv and max
        conv = np.convolve(cur, cur)
        M = float(conv.max())
        idx_max = int(conv.argmax())
        # decide move type
        if rng.random() < 0.7:  # targeted
            # construct product list
            i_start = max(0, idx_max - (n - 1))
            i_end = min(n - 1, idx_max)
            # compute products
            i_vals = np.arange(i_start, i_end+1)
            j_vals = idx_max - i_vals
            products = cur[i_vals] * cur[j_vals]
            # find index with max product
            max_idx = int(np.argmax(products))
            i = i_vals[max_idx]
            j = j_vals[max_idx]
            # epsilon selection
            max_removal = min(cur[i], cur[j]) if i != j else cur[i]
            if max_removal <= 0:
                continue
            epsilon = max_removal * rng.random() * 0.5  # up to half
            # choose target indices far away
            # generate random indices not equal to i/j
            t1 = rng.integers(0, n)
            while t1 == i:
                t1 = rng.integers(0, n)
            if i != j:
                t2 = rng.integers(0, n)
                while t2 == j or t2 == t1:
                    t2 = rng.integers(0, n)
                cand = cur.copy()
                cand[i] -= epsilon
                cand[t1] += epsilon
                cand[j] -= epsilon
                cand[t2] += epsilon
            else:
                t2 = rng.integers(0, n)
                while t2 == t1:
                    t2 = rng.integers(0, n)
                cand = cur.copy()
                cand[i] -= epsilon
                cand[t1] += epsilon/2
                cand[t2] += epsilon/2
        else:
            # random transfer move similar to earlier algorithm
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
            delta = cur[i] * 0.1 * rng.random()
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        # clip and ensure non-negative
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        # acceptance using simulated annealing schedule
        elapsed = time.monotonic() - start
        temp = 1.0 - elapsed / (deadline - start)
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temp, 1e-10)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[THC] n={n} it={it} best {best_score:.6f}")
        it += 1
    return best, best_score

"""
Targeted hill‑climbing for the step‑function attack.

The algorithm repeatedly picks the index *k* where the auto‑convolution
`f*f` attains its maximum.  It then reduces the dominant product
`f[i]·f[k‑i]` by moving a small amount of mass from the two coordinates
`i` and `k‑i` to two random locations (keeping the ℓ¹‑norm constant).
A simulated‑annealing acceptance rule allows occasional uphill moves,
and a tiny random jitter prevents stagnation.

The search is performed for several lengths `n` (200 … 1000).  Each
length receives a time slice; the best sequence found overall is
returned.  All values are forced into the interval `[0,1000]` as required
by the evaluator.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Wrapper around the provided ``evaluate_sequence``."""
    # ``evaluate_sequence`` is available in the execution environment.
    return float(evaluate_sequence(seq.tolist()))


def _targeted_hill_climb(
    seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that focuses on the peak of the auto‑convolution.

    Parameters
    ----------
    seq : np.ndarray
        Starting sequence (non‑negative, length ≥ 2).
    deadline : float
        Absolute time (time.monotonic) after which the routine stops.
    rng : np.random.Generator
        Random number generator.

    Returns
    -------
    best_seq, best_score
        The best sequence discovered and its score.
    """
    cur = seq.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score
    n = len(seq)

    start = time.monotonic()
    it = 0
    # Counters for stagnation detection
    since_improve = 0

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1) analyse the current peak of the convolution
        # --------------------------------------------------------------
        conv = np.convolve(cur, cur)               # length 2·n‑1
        M = float(conv.max())
        k_max = int(conv.argmax())

        # --------------------------------------------------------------
        # 2) propose a move that tries to lower the dominant products
        # --------------------------------------------------------------
        if rng.random() < 0.7:  # 70 % of the time do a targeted move
            # indices that participate in the maximal lag
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_vals = np.arange(i_start, i_end + 1, dtype=int)
            j_vals = k_max - i_vals

            # product a_i * a_j for each pair (i,j) that contributes to k_max
            products = cur[i_vals] * cur[j_vals]

            # choose the pair with the largest product
            max_pair = int(np.argmax(products))
            i = int(i_vals[max_pair])
            j = int(j_vals[max_pair])

            # amount we can safely move (a fraction of the smaller entry)
            max_removal = cur[i] if i == j else min(cur[i], cur[j])
            if max_removal <= 0.0:
                # fallback to a cheap random transfer
                i, j = rng.integers(0, n, size=2)
                while j == i:
                    j = rng.integers(0, n)
                max_removal = cur[i]

            epsilon = max_removal * rng.random() * 0.5  # ≤ 50 % of the source
            # pick two distinct target positions far from the sources
            t1 = rng.integers(0, n)
            while t1 == i:
                t1 = rng.integers(0, n)
            if i != j:
                t2 = rng.integers(0, n)
                while t2 == j or t2 == t1:
                    t2 = rng.integers(0, n)
                cand = cur.copy()
                cand[i] -= epsilon
                cand[t1] += epsilon
                cand[j] -= epsilon
                cand[t2] += epsilon
            else:
                # i == j, we split the transferred mass among two targets
                t2 = rng.integers(0, n)
                while t2 == t1:
                    t2 = rng.integers(0, n)
                cand = cur.copy()
                cand[i] -= epsilon
                cand[t1] += epsilon * 0.5
                cand[t2] += epsilon * 0.5
        else:
            # 30 % : a simple random mass transfer (standard SA move)
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)
            delta = cur[src] * rng.random() * 0.1  # up to 10 % of source
            cand = cur.copy()
            cand[src] -= delta
            cand[dst] += delta

        # --------------------------------------------------------------
        # 3) keep the sequence inside the allowed box and preserve ℓ¹‑norm
        # --------------------------------------------------------------
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise the ℓ¹‑norm to the original one (scaling does not
        # change the score, but it guards against drifting to zero)
        tot = float(cand.sum())
        if tot > 0.0:
            cand *= (float(cur.sum()) / tot)

        cand_score = _score(cand)

        # --------------------------------------------------------------
        # 4) simulated‑annealing acceptance
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start
        total = deadline - start
        temp = max(1e-12, 1.0 - elapsed / total)          # linear cooling
        accept = (cand_score < cur_score) or (
            rng.random()
            < math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
        )
        if accept:
            cur = cand
            cur_score = cand_score
            since_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[THC] n={n:4d} it={it:5d} new best {best_score:.6f}")
        else:
            since_improve += 1

        # --------------------------------------------------------------
        # 5) occasional jitter to escape deep plateaus
        # --------------------------------------------------------------
        if since_improve > 5000 or rng.random() < 0.001:
            jitter = rng.normal(scale=0.01 * np.mean(cur) + 1e-6, size=n)
            cur = np.clip(cur + jitter, 0.0, 1000.0)
            cur_score = _score(cur)
            since_improve = 0
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
                print(f"[THC] jittered new best {best_score:.6f}")

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    The routine explores several problem sizes, applying a targeted
    hill‑climber that concentrates on the peak of the auto‑convolution.
    It respects the 1000 s time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)   # small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # (optional) initialise from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            seq0 = np.clip(seq0, 0.0, 1000.0)
            sc0 = _score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using global height_sequence_1 – score {sc0:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # search over a handful of lengths
    # ------------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # allocate a fair share of the remaining budget to this length
        rem_lengths = len(lengths) - lengths.index(n)
        time_for_n = (deadline - time.monotonic()) / rem_lengths

        # ----- a uniform start (sum = sqrt(2·n)) -----------------------
        uniform = np.full(
            n, math.sqrt(2.0 * n) / n, dtype=np.float64
        )  # ℓ¹‑norm = √(2·n)
        sub_deadline = time.monotonic() + time_for_n * 0.6
        seq, sc = _targeted_hill_climb(uniform, sub_deadline, rng)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[global] new best {best_score:.6f} (uniform) n={n}")

        # ----- a random Dirichlet‑like start -------------------------
        if time.monotonic() < deadline:
            rand = rng.random(n) + 1e-3
            rand = rand / np.sum(rand) * math.sqrt(2.0 * n)
            sub_deadline = time.monotonic() + time_for_n * 0.4
            seq, sc = _targeted_hill_climb(rand, sub_deadline, rng)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[global] new best {best_score:.6f} (random) n={n}")

    # ------------------------------------------------------------------
    # fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        best_seq = np.full(
            n, math.sqrt(2.0 * n) / n, dtype=np.float64
        )
        best_score = _score(best_seq)

    print(f"[final] best score {best_score:.6f} – length {len(best_seq)}")
    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
