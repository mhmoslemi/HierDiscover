"""
Search for a low‑evaluation step function.

The algorithm works in three stages:

1️⃣  **Initialisation** – take the globally supplied `height_sequence_1`
    (if it exists) or start from a uniform vector whose ℓ¹‑norm equals
    √(2·n).  The vector is stored in ``best_seq`` together with its
    current best score.

2️⃣  **Smooth‑max gradient descent** – minimise a differentiable
    approximation of the peak of the auto‑convolution.
    We use a soft‑max (controlled by a temperature `α`) of the
    convolution values and update the sequence by projected gradient
    descent (non‑negative, fixed ℓ¹‑norm).  The FFT is used for fast
    convolutions and for the gradient (the formula
    `∂J/∂a_i = 2·∑_k w_k·a_{k‑i}`).

3️⃣  **Fine‑tuned simulated annealing** – after the gradient phase we run
    a standard SA that perturbs a single coordinate or swaps mass
    between two coordinates while preserving the total sum.  The SA
    accepts uphill moves with a temperature schedule.

Both optimisation phases respect the global time budget (default
1000 s).  The best sequence found is returned as a plain list of
floats (all entries are clipped to the interval [0, 1000]).

The implementation avoids heavy external solvers and uses only NumPy
and the provided ``evaluate_sequence`` (for the final verification).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level – no closures, no lambdas)
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    # size for FFT – next power‑of‑two for speed
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    # numerical noise may produce tiny negatives → clip
    return np.clip(conv, 0.0, None)


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute soft‑max weights w_k = exp(alpha·conv_k) / Σexp(...).
    The subtraction of the maximum avoids overflow.
    """
    # avoid overflow/underflow
    scaled = alpha * conv
    max_val = np.max(scaled)
    exp_term = np.exp(scaled - max_val)
    w = exp_term / np.sum(exp_term)
    return w


def _gradient_via_fft(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2·∑_j a_j·w_{i+j}  (i = 0…n‑1) using convolution.
    The result is the slice conv[n‑1 : n‑1+n] of the convolution
    w * a_rev.
    """
    n = a_rev.shape[0]
    m = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, m)
    fa = np.fft.rfft(a_rev, m)
    conv = np.fft.irfft(fw * fa, m)[: len(w) + n - 1]
    start = n - 1
    return conv[start : start + n]


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of a candidate (same formula as ``evaluate_sequence``
    but without the safety wrapper).  The sequence must already satisfy
    the non‑negativity and upper‑bound constraints.
    """
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_full_fft(seq)
    max_b = conv.max()
    n = seq.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a non‑negative vector onto the simplex {x ≥ 0, Σx = target_sum}.
    Simple scaling works because we never create negative entries after
    clipping; we just renormalise the total mass.
    """
    s = v.sum()
    if s == 0.0:
        # distribute uniformly to avoid division by zero
        return np.full_like(v, target_sum / v.shape[0])
    return v * (target_sum / s)


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------


def _smooth_max_descent(
    start_seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate of the
    auto‑convolution peak.

    Parameters
    ----------
    start_seq : np.ndarray
        Initial non‑negative sequence (will be copied).
    target_sum : float
        Desired ℓ¹‑norm (kept fixed by projection).
    deadline : float
        Absolute time (monotonic) after which the routine must stop.
    rng : np.random.Generator
        Random source (used only for occasional jitter to avoid stagnation).

    Returns
    -------
    (best_seq, best_score)
        The best sequence discovered and its fast‑computed score.
    """
    seq = start_seq.copy()
    n = seq.shape[0]
    # hyper‑parameters (tuned empirically)
    alpha = 0.8          # initial soft‑max temperature
    lr = 0.5 * target_sum / n  # initial learning rate
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    start = time.monotonic()
    while time.monotonic() < deadline:
        # --- compute convolution and soft‑max weights -----------------
        conv = _conv_full_fft(seq)
        w = _softmax_weights(conv, alpha)

        # --- gradient (2 * convolution of w with reversed a) -------
        grad = 2.0 * _gradient_via_fft(w, seq[::-1])

        # --- gradient step (projected) -----------------------------
        seq = seq - lr * grad
        seq = np.clip(seq, 0.0, 1000.0)
        seq = _project_to_simplex(seq, target_sum)

        # occasional jitter to escape shallow plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=0.01 * target_sum, size=n)
            seq = np.clip(seq + jitter, 0.0, 1000.0)
            seq = _project_to_simplex(seq, target_sum)

        # --- evaluate -----------------------------------------------
        sc = _fast_score(seq)
        if sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print(f"[GD] iter≈{int(time.monotonic()-start)} score={sc:.6f}")

        # --- schedule ------------------------------------------------
        # slowly increase the focus on the true max (larger α)
        alpha = min(alpha * 1.008, 12.0)
        # slowly decay the step size
        lr *= 0.9995

        # safety stop if we are not improving for a while
        if time.monotonic() - start > (deadline - start) * 0.9:
            # final few seconds we just break early
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Simulated annealing (fine‑tuner)
# ----------------------------------------------------------------------


def _simulated_annealing(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Classical SA that perturbs either a single entry or moves a
    random amount of mass between two entries while keeping the total
    sum fixed.  The temperature cools from 1.0 to 1e‑4 over the
    remaining time budget.
    """
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4
    n = cur_seq.shape[0]
    target_sum = cur_seq.sum()

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        # exponential cooling
        temp = temp_start * (temp_end / temp_start) ** frac

        # decide move type
        if rng.random() < 0.15:          # mass‑exchange move
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.normal(scale=0.2 * target_sum) * temp
            cand = cur_seq.copy()
            cand[i] = max(cand[i] + delta, 0.0)
            cand[j] = max(cand[j] - delta, 0.0)
        else:                            # single‑coordinate perturbation
            idx = rng.integers(n)
            delta = rng.normal(scale=0.5 * target_sum) * temp
            cand = cur_seq.copy()
            cand[idx] = max(cand[idx] + delta, 0.0)

        # re‑normalise to keep the ℓ¹‑norm exact
        cand = _project_to_simplex(cand, target_sum)

        cand_score = _fast_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

        # tiny random reshuffle occasionally
        if rng.random() < 0.001:
            pert = np.maximum(cur_seq + rng.normal(scale=0.05 * target_sum, size=n), 0.0)
            cur_seq = _project_to_simplex(pert, target_sum)
            cur_score = _fast_score(cur_seq)

    return best_seq, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Find a step‑function sequence with a low evaluation score.
    Uses a combination of gradient descent on a smooth surrogate
    of the max‑convolution and a final simulated‑annealing fine‑tuner.
    The routine respects the overall time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # 1️⃣  Initialise from a known good construction if available
    # --------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_score = float("inf")
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            seq0 = np.clip(seq0, 0.0, 1000.0)
            # enforce a valid ℓ¹‑norm (the evaluator will do it anyway)
            sc0 = _fast_score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using height_sequence_1 – score {sc0:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # fallback: uniform start (length 300)
    # --------------------------------------------------------------
    if best_seq is None:
        n0 = 300
        target_sum = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target_sum / n0, dtype=np.float64)
        best_score = _fast_score(best_seq)
        print(f"[init] uniform start – score {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smooth‑max gradient descent (primary optimiser)
    # --------------------------------------------------------------
    # allocate roughly 60 % of the time to the descent phase
    gd_deadline = start + 0.60 * (deadline - start)

    # we will try a few different lengths (including the current one)
    lengths_to_try = [best_seq.shape[0], 350, 400, 450, 500]
    for n in lengths_to_try:
        if time.monotonic() >= gd_deadline:
            break
        # re‑scale current best to the new length if needed
        if n != best_seq.shape[0]:
            # embed by simple linear interpolation onto a finer grid
            x_old = np.linspace(0, 1, best_seq.shape[0])
            x_new = np.linspace(0, 1, n)
            new_seq = np.interp(x_new, x_old, best_seq)
            new_seq = np.clip(new_seq, 0.0, 1000.0)
            new_seq = _project_to_simplex(new_seq, best_seq.sum())
        else:
            new_seq = best_seq.copy()

        # run gradient descent on this length
        seq_gd, score_gd = _smooth_max_descent(
            new_seq,
            target_sum=new_seq.sum(),
            deadline=gd_deadline,
            rng=rng,
        )
        if score_gd < best_score:
            best_score = score_gd
            best_seq = seq_gd.copy()
            print(f"[gd] length={n} improved score={score_gd:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing refinement (remaining time)
    # --------------------------------------------------------------
    if time.monotonic() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq, best_score, deadline, rng
        )
        print(f"[final] best score after SA = {best_score:.6f}")

    # --------------------------------------------------------------
    # Safety fallback – should never happen
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)

    # final verification with the official scorer (ensures compliance)
    final_sc = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] official evaluation = {final_sc:.6f}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
