def solve_lp_for_g(f, M):
    n = len(f)
    # Build A_ub matrix
    rows = []
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    # Build A in dense format
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k-(n-1))
        j_end = min(k, n-1)
        # for j in range(j_start, j_end+1):
        #   A[k,j] = f[k-j]
        # Use vectorized:
        idx = np.arange(j_start, j_end+1)
        A[k, idx] = f[k - idx]
    # Solve LP: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None) for _ in range(n)]
    # Using scipy.optimize.linprog:
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None
    target = math.sqrt(2.0 * n)
    g = b * (target / sum_b)
    return g

"""
Iterative linear‑programming / gradient descent search.

The objective 2·n·max(f*f)/(sum(f)**2) is homogeneous, so we keep the
ℓ¹‑norm fixed to √(2·n).  For a fixed vector f we compute M = max(f*f) and
solve the linear program

    maximise   Σ b_j
    subject to   (f * b)_k ≤ M   for all k = 0,…,2n‑2
                 b_j ≥ 0

The solution b tries to spread the mass as evenly as possible while
respecting the current worst convolution value M.  After rescaling b to
the same ℓ¹‑norm we convex‑combine it with the current f (mixing factor t
that shrinks from ≈0.5 to ≈0.01).  A cheap deterministic “gradient”
step then moves a small amount of mass from the coefficient that
contributes most to the current worst convolution entry to the smallest
coefficient.  The process repeats until the time budget expires.
Both the LP step and the gradient step are very cheap for n≈200‑400,
so many iterations fit into the 1000 s limit.

The best sequence seen during the run (as a plain Python list) is
returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _initial_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).

    If a global `height_sequence_1` of the same length exists it is used
    (after clipping).  Otherwise a Dirichlet random vector is generated.
    """
    if "height_sequence_1" in globals() and len(height_sequence_1) == n:
        vec = np.array(height_sequence_1, dtype=np.float64)
    else:
        # Dirichlet (uniform) random vector, sum = 1
        vec = rng.dirichlet(np.ones(n, dtype=np.float64))
    # Clip to the domain accepted by the scorer
    vec = np.clip(vec, 0.0, 1000.0)
    # Normalise to ℓ¹‑norm = √(2·n)
    target = math.sqrt(2.0 * n)
    vec = vec / np.sum(vec) * target
    return vec


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that
        (A @ b)[k] = (f * b)[k]   for k = 0,…,2n‑2 .
    Row k contains the slice of f that aligns with index k of the
    convolution.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        idx = np.arange(j_start, j_end + 1, dtype=np.int64)
        # f[k‑j] is defined because 0 ≤ k‑j ≤ n‑1
        A[k, idx] = f[k - idx]
    return A


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP that maximises Σ b_j subject to (f * b) ≤ M and b ≥ 0.
    Returns the vector b rescaled to ℓ¹‑norm √(2·n), or None on failure.
    """
    n = f.shape[0]
    A = _build_convolution_matrix(f)
    rows = 2 * n - 1
    # Linear programme: maximise Σ b_j  → minimise -Σ b_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]
    # Upper bounds for each convolution entry
    b_ub = np.full(rows, M, dtype=np.float64)

    # Use the HiGHS solver (fast and robust)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return None
    target = math.sqrt(2.0 * n)
    g = b * (target / sum_b)
    return g


def _gradient_step(f: np.ndarray, cur_max: float, step_frac: float) -> np.ndarray:
    """
    Deterministic improvement: move a small amount of mass from the
    coefficient that most contributes to the current worst convolution
    entry to the smallest coefficient.

    step_frac – fraction of the donor mass to move (≈0.01–0.05).
    """
    n = f.shape[0]
    conv = np.convolve(f, f)
    k_max = int(np.argmax(conv))

    # indices that appear in (f*f)[k_max]
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)

    # contribution of f[i] to the peak is f[k_max‑i]; the donor is the i
    # with maximal contribution.
    contributions = f[k_max - idx]
    donor = idx[np.argmax(contributions)]

    # pick a recipient with smallest current value (different from donor)
    candidates = np.where(np.arange(n) != donor)[0]
    if candidates.size == 0:
        return f
    recipient = candidates[np.argmin(f[candidates])]

    max_delta = f[donor]
    delta = max_delta * step_frac
    if delta <= 0.0:
        return f

    # Apply the move
    f_new = f.copy()
    f_new[donor] -= delta
    f_new[recipient] += delta

    # Accept only if the peak does not increase
    new_max = np.convolve(f_new, f_new).max()
    if new_max <= cur_max:
        return f_new
    return f

    """

            cur_val = cur_max

                best_f = f.copy()
                last_improve_it = it

                    f"[improve] n={n} it={it:06d} val={cur_val:.6f} "

                )

            if it - last_improve_it > 5000 and elapsed > 0.5 * total_time:
