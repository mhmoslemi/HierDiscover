def project_onto_simplex(v, S):
    # v: array; returns w: nonneg vector sum S that is close to v in Euclidean sense.
    if np.sum(v) == S and np.all(v>=0): return v
    # Ensure non-negative: w = v - tau, where tau chosen s.t. sum(max(v - tau, 0)) = S.
    # algorithm: sort v descending as u.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v, s):
    # standard algorithm.
    # ensure v a numpy array.
    # If sum of v equals s and all non-negative, return v.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: all zero? set equal distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_descent(f, target_sum, deadline, max_iters=5000):
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_f = f.copy()
    cur_score = best_score
    step = target_sum * 0.1
    patience = 0
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        # compute conv and max
        conv = np.convolve(cur_f, cur_f)
        max_val = conv.max()
        cur_score = max_val  # since sum fixed
        # preserve best
        if cur_score < best_score:
            best_score = cur_score
            best_f = cur_f.copy()
            patience = 0
            print(f"[GD] n={n} it={it} score={cur_score:.6f}")
        else:
            patience += 1
        # if patience > 200 and step small break
        if patience > 200 and step < 1e-8:
            break
        # determine active offsets near max
        eps = max_val * 1e-9 + 1e-12
        active = np.where(conv >= max_val - eps)[0]
        # compute gradient
        grad = compute_gradient(cur_f, active)
        # try step sizes
        improved = False
        for _ in range(10):
            f_candidate = cur_f - step * grad
            # project onto simplex
            f_candidate = project_onto_simplex(f_candidate, target_sum)
            # ensure nonneg and maybe clip
            # compute score
            new_conv = np.convolve(f_candidate, f_candidate)
            new_score = new_conv.max()
            if new_score < cur_score:
                cur_f = f_candidate
                cur_score = new_score
                improved = True
                # increase step a little
                step *= 1.02
                break
            else:
                step *= 0.5
        if not improved:
            # if cannot improve, maybe random perturb
            # add small noise

""" 
Search for a non‑negative step sequence minimising the custom evaluation
function.  The algorithm consists of:

* Optional warm‑start from the global `height_sequence_1`.
* A projected gradient descent on the ℓ¹‑norm simplex that iteratively
  reduces the maximal entry of the auto‑convolution.  At each step the
  active (maximising) offsets are identified, the sub‑gradient of the
  worst convolution entry is computed, a line‑search step is taken and
  the new vector is projected back onto the simplex.
* Several random restarts and several lengths (200 … 1200) are performed
  while respecting the global time budget.
* The best candidate is finally polished by a short stochastic
  hill‑climbing (Gaussian perturbations).

All numbers are kept non‑negative and below the clipping bound
(1000) required by `evaluate_sequence`.  The function returns the best
list of floats found before the deadline.
"""

import time
import math
import numpy as np

# Helper: projection onto the simplex {x >= 0, sum x = s}
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # shift by decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case – return uniform distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _compute_score_from_vector(f: np.ndarray) -> float:
    """Return the evaluation score for a vector f (no clipping)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float('inf')
    return float(2.0 * n * max_val / (sum_val * sum_val))

def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Return indices where conv is (almost) equal to its maximum."""
    # relative tolerance – very tight to keep the active set small
    eps = max_val * 1e-9 + 1e-12
    return np.where(conv >= max_val - eps)[0]

def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (f*f)[k] w.r.t. f for the active offsets."""
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = f[j]
        # default contribution 2*val
        grad[i] += 2.0 * vals
        # subtract the over‑counted self contribution (i == j)
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= vals[mask]   # resulting in a single f_i term
    return grad

def _line_search_step(f: np.ndarray, grad: np.ndarray,
                      step: float, target_sum: float,
                      cur_score: float) -> tuple[np.ndarray, float, float]:
    """
    Try to move along -grad with the supplied step.
    Returns (new_f, new_score, new_step). The step is shrunk if the move does not improve.
    """
    # Attempt a few reductions of the step size.
    for _ in range(10):
        f_candidate = f - step * grad
        f_candidate = _project_onto_simplex(f_candidate, target_sum)
        # Clip to the limits enforced by evaluate_sequence (optional)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)
        score = _compute_score_from_vector(f_candidate)
        if score < cur_score:
            # Successful move – slightly increase the step for next round.
            return f_candidate, score, step * 1.02
        # otherwise reduce step and try again
        step *= 0.5
    # No improvement – keep original vector and a smaller step.
    return f, cur_score, step * 0.5

def _gradient_descent(f0: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """Projected gradient descent that reduces the worst auto‑convolution entry."""
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)
    step = target_sum * 0.1   # initial step size
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # current convolution and score
        conv = np.convolve(f, f)
        max_val = np.max(conv)
        cur_score = max_val   # because the ℓ¹‑norm is fixed to target_sum

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
            print(f"[GD] it={it} n={len(f)} score={cur_score:.6f}")
        else:
            no_improve += 1

        # stop if we have stalled for a while
        if no_improve > 300:
            break

        # active offsets (usually only the centre)
        active = _active_offsets(conv, max_val)

        # sub‑gradient of the active max entry
        grad = _gradient_submax(f, active)

        # line‑search step
        f, cur_score, step = _line_search_step(f, grad, step, target_sum, cur_score)

    return best_f, best_score

def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """Simple Gaussian hill‑climbing used for the final polishing stage."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑project onto the ℓ¹‑simplex that matches the original sum
        target_sum = np.sum(start)  # keep the same ℓ¹‑norm
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """Search for a low‑value step sequence within the allotted time."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.3)   # small safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start from the globally supplied construction if available.
    # ------------------------------------------------------------------
    h1 = globals().get('height_sequence_1')

        seq = np.array(h1, dtype=np.float64)

        # ensure a feasible ℓ¹‑norm
        n = len(seq)

        if np.sum(seq) == 0:
            seq = np.full(n, target_sum / n, dtype=np.float64)
        else:

        score = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[init] using height_sequence_1, n={n}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Main optimisation – projected gradient descent with several sizes.
    # ------------------------------------------------------------------

            else:

            # early stop if we have already crossed the required bound
            if best_score < 1.503:
                print("[early] target achieved")
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # 3️⃣  Final stochastic polishing (if time permits).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback uniform construction of moderate size
        n = 400
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

        refined, refined_score = _stochastic_refinement(best_seq, deadline, step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # Convert to plain Python list, clip once more for safety.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

"""
Search for a low‑value non‑negative step sequence.

The algorithm consists of three stages:

1️⃣ Warm‑start from the global `height_sequence_1` if it exists.
2️⃣ Projected gradient descent on the ℓ¹‑simplex that directly reduces the
   maximal entry of the auto‑convolution.  In each iteration the active
   (maximising) offsets are identified, the sub‑gradient of the worst
   convolution entry is computed, a line‑search step is taken and the new
   vector is projected back onto the simplex (∑ aᵢ = √(2 n), aᵢ ≥ 0).
   Several lengths (200 … 1200) and random restarts are processed while
   respecting the global time budget.
3️⃣ A short stochastic hill‑climbing (Gaussian perturbations) polish the
   best candidate before returning it.

All numbers are kept non‑negative and below the clipping bound required
by `evaluate_sequence`.  The function returns the best list of floats
found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0,  Σ x_i = s}.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Edge case – return a uniform distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """
    Exact evaluation of the provided `evaluate_sequence` formula
    (without the internal clipping, which is unnecessary for our values).
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where the convolution is (almost) maximal."""
    eps = max_val * 1e-9 + 1e-12
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the worst auto‑convolution entry with respect to f,
    summed over all active offsets.
    """
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = f[j]
        # default contribution 2 * f_{k-i}
        grad[i] += 2.0 * vals
        # self‑pair (i == j) should be counted only once
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= vals[mask]   # 2*val – val = val
    return grad


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _gradient_descent(f0: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent that minimises the maximum auto‑convolution
    entry while keeping f on the ℓ¹‑simplex (∑f = target_sum, f ≥ 0).
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)
    step = target_sum * 0.1               # initial step size
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        cur_score = _compute_score_from_vector(f)

        # record the best vector seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
            print(f"[GD] it={it} n={len(f)} score={cur_score:.6f}")
        else:
            no_improve += 1

        if no_improve > 300:                # stagnation → stop early
            break

        # identify the active (maximising) offsets and compute a sub‑gradient
        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)
        grad = _gradient_submax(f, active)

        # propose a move
        f_candidate = _project_onto_simplex(f - step * grad, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)   # safety clip
        cand_score = _compute_score_from_vector(f_candidate)

        if cand_score < cur_score:
            # successful step – slightly increase the step size
            f = f_candidate
            step *= 1.02
        else:
            # no improvement – shrink the step size
            step *= 0.5
            # keep the old f

    return best_f, best_score


def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing used for the final polishing stage.
    The ℓ¹‑norm is preserved by a projection onto the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value step sequence within the allotted time.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.3)   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start from the previously found construction, if any.
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.array(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        n = len(seq)
        target_sum = math.sqrt(2.0 * n)
        if np.sum(seq) == 0:
            seq = np.full(n, target_sum / n, dtype=np.float64)
        else:
            seq = seq * (target_sum / np.sum(seq))
        score = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[init] using height_sequence_1, n={n}, score={score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Main optimisation – projected gradient descent with restarts.
    # --------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000, 1200]   # diverse sizes
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for restart in range(5):
            if time.monotonic() >= deadline:
                break

            # initialise a candidate vector
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
            else:
                f0 = rng.random(n) + 0.01                # avoid exact zeros
                f0 = f0 / np.sum(f0) * target_sum

            # allow at most ~8 s for each descent run
            sub_deadline = min(deadline, time.monotonic() + 8.0)
            f_opt, score_opt = _gradient_descent(f0,
                                                 target_sum,
                                                 sub_deadline,
                                                 max_iters=5000)

            if score_opt < best_score:
                best_score = score_opt
                best_seq = f_opt.copy()
                print(f"[update] n={n} restart={restart} score={score_opt:.6f}")

            if best_score < 1.503:          # target reached → early exit
                print("[early] target achieved")
                return [float(x) for x in best_seq]

    # --------------------------------------------------------------
    # 3️⃣ Final stochastic polishing (if time permits).
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback to a modest uniform construction
        n = 400
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline,
                                                         step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # Safety clip before returning (evaluate_sequence also does this)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
