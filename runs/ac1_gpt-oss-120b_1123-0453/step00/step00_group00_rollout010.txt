        # convert to python list for the evaluator

    lo, hi = 0.5, 2.0

    val2, seq2 = score_for_p(p2)

    # keep track of best
    if val1 < best_score:

    if val2 < best_score:
        best_score, best_seq = val2, seq2

    # log

import numpy as np
import time, math

def build_power_sequence(p, n):
    """Return a monotone sequence a_i = (i+1)^p scaled to max 1000."""
    idx = np.arange(1, n+1, dtype=np.float64)
    seq = idx ** p
    # Avoid division by zero if p negative (not here). Scale to max 1000.
    scale = 1000.0 / seq[-1]
    seq = seq * scale
    return seq

def evaluate_exponent(p, n):
    """Return (score, seq) for exponent p."""
    seq = build_power_sequence(p, n)
    # evaluate_sequence expects a python list; convert.
    score = float(evaluate_sequence(seq.tolist()))
    return score, seq

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a step‑function using a power‑law shape a_i∝(i+1)^p.
    The exponent p is optimised with a golden‑section search (initial
    interval [0.5, 2.0]) to minimise the provided `evaluate_sequence`.
    The routine respects the time budget and returns the best sequence found.
    """
    # fast RNG for reproducibility – not used directly but kept for compatibility
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Length of the sequence – use the known construction if available.
    try:
        n = int(len(height_sequence_1))
        if n == 0:
            raise Exception
    except Exception:
        n = 256

    # Golden‑section search parameters
    lo, hi = 0.5, 2.0
    phi = (math.sqrt(5) - 1) / 2.0  # ≈0.618
    # initial interior points
    p1 = lo + (1 - phi) * (hi - lo)
    p2 = lo + phi * (hi - lo)

    val1, seq1 = evaluate_exponent(p1, n)
    val2, seq2 = evaluate_exponent(p2, n)

    best_score = float('inf')
    best_seq = None

    if val1 < best_score:
        best_score, best_seq = val1, seq1
    if val2 < best_score:
        best_score, best_seq = val2, seq2

    print(f"Initial: p1={p1:.4f} score={val1:.6f}, p2={p2:.4f} score={val2:.6f}")

    # Iterative refinement
    while time.monotonic() < deadline:
        # stop if interval is tiny
        if hi - lo < 1e-5:
            break

        if val1 > val2:
            # the minimum lies to the right of p1
            lo = p1
            p1, val1, seq1 = p2, val2, seq2
            p2 = lo + phi * (hi - lo)
            val2, seq2 = evaluate_exponent(p2, n)
        else:
            # the minimum lies to the left of p2
            hi = p2
            p2, val2, seq2 = p1, val1, seq1
            p1 = lo + (1 - phi) * (hi - lo)
            val1, seq1 = evaluate_exponent(p1, n)

        # keep the best seen so far
        if val1 < best_score:
            best_score, best_seq = val1, seq1
        if val2 < best_score:
            best_score, best_seq = val2, seq2

        print(f"Interval [{lo:.6f}, {hi:.6f}] best={best_score:.6f}")

"""
Search for a low‑value step function.

The idea is to use a simple analytic family
    a_i = (i+1)^p   (i = 0 … n‑1)
scaled so that the largest entry equals 1000 – this respects the
hard limits imposed by ``evaluate_sequence``.  For p=0 the sequence is
uniform and the score is exactly 2.  As p increases the large values are
concentrated towards the end of the vector, which reduces the biggest
diagonal of the self‑convolution and therefore lowers the objective.
The optimal p lies between 1 and 1.5; to find it we run a cheap
golden‑section search on the interval [0.5, 2.0] while respecting the
overall time budget.  After the 1‑D optimisation a few cheap random
block perturbations are tried as a final local search.

The function `propose_candidate` is the entry point used by the
evaluation harness.
"""

import math
import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------

def build_power_sequence(p: float, n: int) -> np.ndarray:
    """
    Return a monotone sequence a_i = (i+1)^p, scaled to have max value 1000.
    All entries are non‑negative and fit inside the evaluator’s limits.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    raw = idx ** p                                      # power law
    scale = 1000.0 / raw[-1]                            # make the last entry exactly 1000
    return raw * scale


def evaluate_exponent(p: float, n: int):
    """
    Compute the evaluation score for a given exponent p and length n.
    Returns a tuple (score, sequence_ndarray).
    """
    seq = build_power_sequence(p, n)
    # evaluate_sequence expects a plain Python list of floats
    score = float(evaluate_sequence(seq.tolist()))
    return score, seq


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a step function by optimising the power‑law exponent.
    A golden‑section search over p ∈ [0.5, 2.0] is performed, followed
    by a brief random block‑perturbation local search, all bounded by
    the supplied time budget.  The best sequence found is returned as
    a list of floats.
    """
    # rng is kept only for the final local search (kept for reproducibility)
    rng = np.random.default_rng(seed)

    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------
    # Determine the length of the sequence we should produce
    # --------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        if n == 0:
            raise Exception
    except Exception:
        # fallback – a modest size that always works
        n = 256

    # --------------------------------------------------------
    # Golden‑section search on the exponent p
    # --------------------------------------------------------
    lo, hi = 0.5, 2.0
    phi = (math.sqrt(5) - 1) / 2.0               # ≈0.61803398875

    # internal points
    p1 = lo + (1 - phi) * (hi - lo)              # ≈ lo + 0.381966…
    p2 = lo + phi * (hi - lo)                    # ≈ lo + 0.618034…

    val1, seq1 = evaluate_exponent(p1, n)
    val2, seq2 = evaluate_exponent(p2, n)

    best_score = float('inf')
    best_seq = None

    if val1 < best_score:
        best_score, best_seq = val1, seq1
    if val2 < best_score:
        best_score, best_seq = val2, seq2

    print(f"[init] p1={p1:.5f} score={val1:.6f} | p2={p2:.5f} score={val2:.6f}")

    while time.monotonic() < deadline:
        # stop early if the interval has become tiny
        if hi - lo < 1e-5:
            break

        # the side with the larger value can be discarded
        if val1 > val2:
            lo = p1
            p1, val1, seq1 = p2, val2, seq2
            p2 = lo + phi * (hi - lo)
            val2, seq2 = evaluate_exponent(p2, n)
        else:
            hi = p2
            p2, val2, seq2 = p1, val1, seq1
            p1 = lo + (1 - phi) * (hi - lo)
            val1, seq1 = evaluate_exponent(p1, n)

        # update the global best
        if val1 < best_score:
            best_score, best_seq = val1, seq1
        if val2 < best_score:
            best_score, best_seq = val2, seq2

        print(f"[search] interval=[{lo:.6f},{hi:.6f}] best={best_score:.6f}")

    # --------------------------------------------------------
    # Final short local refinement (optional, cheap)
    # --------------------------------------------------------
    parent = best_seq.copy() if best_seq is not None else np.ones(n, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(parent.tolist()))

    # simple block perturbation – similar to the baseline but very short
    while time.monotonic() < deadline:

            best_score = score

    print(f"[final] best score achieved: {best_score:.6f}")

    # Return a plain Python list of floats as required by the harness
    return [float(x) for x in parent]
