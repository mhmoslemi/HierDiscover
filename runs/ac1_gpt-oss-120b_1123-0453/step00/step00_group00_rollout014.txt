"""
Hybrid search for a low‑score step‑function.

The evaluator is invariant under a global scaling of the sequence, but it is
sensitive to the *shape* of the coefficients because the score is
   2·n·max_k (a*a)[k] / (∑a_i)^2 .
The algorithm mixes three ideas:

1. **Random Dirichlet samples** – a fast way to generate many different
   non‑negative vectors that sum to one.
2. **Extreme‑pairing heuristic** – after drawing a random simplex point,
   the values are sorted and placed on alternating positions
   (large on even indices, small on odd indices).  This anti‑aligns large
   entries with their reversed counterpart and tends to lower the peak of
   the convolution.
3. **Targeted mass‑transfer moves** – the current worst convolution index
   `k_max` is identified; the pair (i, j = k_max‑i) that contributes most
   to that peak is located and a small amount of mass is moved from both
   entries to two random locations.  This directly flattens the
   convolution.

All candidates are clipped to the [0, 1000] range used by the official
`evaluate_sequence` function.  The best sequence found before the time
budget (≈ 1000 s) expires is returned.

The routine respects the required signature `propose_candidate(seed,
budget_s, **kwargs)`.
"""

import numpy as np
import time

def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation of the block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional averaging with the reversed vector
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    # ----------------------------------------------------------------------
