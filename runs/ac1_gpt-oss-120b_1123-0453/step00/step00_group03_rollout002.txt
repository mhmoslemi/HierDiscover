def simplex_projection(v, S):
    # v: np.ndarray
    # Return projected onto {w >=0, sum w = S}
    # If S == 0: return zeros.
    # Use sort descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        theta = cssv[-1] / len(v)
    else:
        rho = ind[cond][-1]
        theta = cssv[rho-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑Descent based local search with simplex projection.

The algorithm keeps the ℓ1‑norm of the sequence constant (∑a_i = 1) so that
the objective 
    2·n·max(conv(a,a)) / (∑a_i)²
is scale‑invariant.  Each iteration:

1. Compute the full auto‑convolution a*a and locate the index k where it attains
   its maximum M.
2. The sub‑gradient of the objective w.r.t. a at this index is proportional to
   (S·a_{k‑i} ‑ M), where S = ∑a_i.  The descent direction is therefore
   d_i = M ‑ S·a_{k‑i}.
3. Take a small step a ← a + α·d and project back onto the simplex
   {x ≥ 0, ∑x = S}.  Projection is performed with the O(n log n) algorithm
   of Duchi et al.
4. Occasionally (≈30 %) a random block perturbation (as in the reference
   implementation) is applied for diversification; the perturbed vector is
   re‑scaled to keep the ℓ1‑norm unchanged.
5. The best sequence seen so far (according to the official evaluate_sequence)
   is stored and returned after the time budget expires.

The method uses only NumPy and the provided `evaluate_sequence` function,
runs entirely in‑memory and respects the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project the vector v onto the simplex {x ≥ 0, ∑x = s}.
    Implements the algorithm from:
    Duchi, Shalev‑Shwartz, Singer & Chandra, ICML 2008.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # If the sum is already s and all entries are non‑negative we are done.
    if np.abs(v.sum() - s) < 1e-12 and np.all(v >= 0):
        return v.copy()
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    # Find the rho index: largest i such that u_i > (cssv_i) / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1  # zero‑based
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical cleaning: ensure sum equals s (within tolerance)
    diff = w.sum() - s
    if abs(diff) > 1e-10:
        # Distribute the tiny residual uniformly over positive components
        pos = w > 0
        if np.any(pos):
            w[pos] -= diff / pos.sum()
            w = np.maximum(w, 0.0)
    return w


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = base.astype(np.float64).copy()
    n = candidate.size
    # block width between n/200 and n/12 (inclusive)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation on the selected block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetrisation
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def local_score(seq: np.ndarray) -> float:
    """
    Fast local computation of the evaluation function.
    Mirrors evaluate_sequence but avoids the defensive checks.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    n = seq.size
    return 2.0 * n * max_b / (total * total)

    ):

    S_target = 1.0
    a = base / base.sum() * S_target               # current iterate
    best = a.copy()

        else:

            cur_score = local_score(a)

    # ------------------------------------------------------------------
