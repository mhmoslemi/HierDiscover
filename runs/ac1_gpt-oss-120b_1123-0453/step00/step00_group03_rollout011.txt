"""
Propose a step‑function sequence that minimizes the evaluation metric.

Algorithm
---------
1.  Use a theoretically motivated starting shape:
    a_i = C_i = binom(2i,i) / 4**i  (i = 0 .. n-1).
    This sequence yields a nearly constant self‑convolution and a
    score ≈ π/2 ≈ 1.57, already better than the uniform 2.0 bound.

2.  Iteratively improve the shape while keeping the total sum fixed.
    For the current vector f:
        •  Let M = max(f * f).
        •  Solve the linear program
                maximise  Σ g_i
                subject  A_f @ g ≤ M   (A_f encodes the convolution f * g)
                         g_i ≥ 0
           where A_f is the (2n−1)×n Toeplitz matrix of f.
        •  Scale g so that Σ g = Σ f (preserves the normalisation used by the
           evaluation function).
        •  Mix f and g with several step sizes t∈{0.1,0.3,0.5,0.7,0.9}:
                candidate = (1−t)·f + t·g
           Keep the best candidate (lowest evaluation score).

    This is a Frank–Wolfe‐type update: we move towards a direction that
    increases the total mass in parts of the sequence that contribute little
    to the current maximal convolution, thereby reducing the maximum.

3.  When the time budget is almost exhausted, fall back to a cheap
    random block‑perturbation local search (the method used in the
    reference implementation) to squeeze out any remaining improvement.

The whole routine runs under the provided time budget (≈1000 s) and
returns the best sequence found.  All numbers are non‑negative and
clipped to the range required by ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper: block perturbation (fallback/local‑search)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width between n//200 and n//12 (inclusive)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # optional symmetrisation
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Build the (2n‑1)×n convolution matrix A_f  (row k contains f_i where i+j=k)
# ----------------------------------------------------------------------
def build_conv_matrix(f):
    """Return a sparse matrix A such that A @ g = f * g (full convolution)."""
    n = f.size
    data = []
    offsets = []
    # each diagonal offset = -d (sub‑diagonal) contains the constant f[d]
    for d in range(n):
        data.append(np.full(n, f[d], dtype=np.float64))
        offsets.append(-d)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a low‑score step‑function sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget (seconds).  The function stops a few
        seconds before this limit to return the best sequence found.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise ``f`` – the starting sequence.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is supplied by the runner; use its length.
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)

    n = int(base.size)                     # keep the original length
    # The analytically motivated coefficients: C_i = binom(2i,i)/4**i
    f = np.empty(n, dtype=np.float64)
    f[0] = 1.0
    for i in range(1, n):
        # recurrence  C_{i} = C_{i-1} * (2*i-1)/(2*i)
        f[i] = f[i - 1] * (2 * i - 1) / (2 * i)
    # Ensure adherance to the evaluator's clipping rules
    f = np.clip(f, 0.0, 1000.0)

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main deterministic improvement loop (Frank‑Wolfe style)
    # ------------------------------------------------------------------
    iter_cnt = 0
    while time.monotonic() < deadline:
        # --- 1) current max convolution M ---------------------------------
        conv_f = np.convolve(best, best)
        M = float(conv_f.max())

        # --- 2) build linear program ---------------------------------------
        A = build_conv_matrix(best)                       # sparse (2n-1)×n
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)                 # maximise sum(g)

        # Solve the LP; use HiGHS (fast, supports sparse A)
        try:

                c, A_ub=A, b_ub=b,
                bounds=[(0, None)] * n,

        except Exception as exc:

            break

        if not res.success:

            print(f"[LP] failed (msg={res.message}) after {iter_cnt} iterations")
            break

        g = res.x
        sum_best = float(best.sum())
        sum_g = float(g.sum())
        if sum_g == 0.0:

            break

        # --- 3) scale g to keep the total sum unchanged --------------------
        scale = sum_best / sum_g
        g = g * scale
        g = np.clip(g, 0.0, 1000.0)

        improved = False
        # --- 4) try a few mixing coefficients --------------------------------
        for t in (0.1, 0.3, 0.5, 0.7, 0.9):
            cand = (1.0 - t) * best + t * g
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                improved = True
                print(f"[iter {iter_cnt}] t={t:.2f} -> new best {best_score:.6f}")
        iter_cnt += 1

        # If none of the mixes improved the score, break early to save time.
        if not improved:
            print(f"[iter {iter_cnt}] no improvement, terminating deterministic phase")
            break

        # Safety net: avoid excessive iterations if time is short.
        if iter_cnt >= 30:   # more than enough for convergence
            break

    # ------------------------------------------------------------------
    # Final random‑local‑search polishing (uses the original block
    # perturbation scheme).  The time left is typically a few seconds.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[random] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return [float(x) for x in best]
