def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    n = len(height_sequence_1)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(n, dtype=np.float64)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial score: {best_score:.6f}")

    # Phase 1: param sweep over exponent p and scaling
    # Determine number of samples based on remaining time.
    # e.g., 200 samples or until time < deadline - ...
    sample_count = 0
    while time.monotonic() < deadline - 2.0 and sample_count < 500:
        # random exponent
        p = rng.uniform(0.0, 8.0)

        scale = rng.uniform(0.5, 1.0) * 1000.0  # ensures edges near clipping

        score = float(evaluate_sequence(cand.tolist()))
        sample_count += 1
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"New best via exponent {p:.3f}, score {best_score:.6f}")

    # Phase 2: local refinement using gradient descent
    # We'll run while time left
    while time.monotonic() < deadline:
        # compute convolution
        b = np.convolve(best, best)
        max_idx = int(np.argmax(b))
        max_b_val = b[max_idx]
        # gradient for max anti-diagonal
        grad = np.zeros_like(best)
        # for each i where k-i in range
        start_i = max(0, max_idx - (n-1))
        end_i = min(n-1, max_idx)
        # i ranges
        # For i in [start_i, end_i], j = max_idx - i
        # grad[i] = 2 * best[j]

        if np.isfinite(score) and score < best_score:

"""
A hybrid random‑parameter + gradient descent search for a low‑autocorrelation
non‑negative step function.  The algorithm:
 1. Start from the provided sequence (or uniform if invalid).
 2. Sample many “U‑shaped” families a_i = (i/(n‑1))^p + ((n‑1-i)/(n‑1))^p,
    possibly mixed with a uniform component and scaled so that the edges hit
    the clipping bound 1000.  Keep the best.
 3. Refine the best candidate by a simple projected gradient step that
    reduces the currently maximal convolution entry; occasional random block
    perturbations are added to escape plateaus.
 4. The loop runs until the allocated time budget expires and returns the
    best sequence found.
"""

import numpy as np
import time

def generate_u_shape(n: int, p: float) -> np.ndarray:
    """Return a symmetric “U‑shape’’ whose edges are 1 and centre ≈2·(0.5)^p."""
    i = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    norm = (i / (n - 1)) ** p
    rev = ((n - 1 - i) / (n - 1)) ** p
    return norm + rev

def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Create an O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # Choose a width that is a sensible fraction of n.
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # Random multiplicative factor drawn from log‑normal distribution.
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # Occasionally enforce symmetry (helps the convolution).
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # Respect the hard limits imposed by evaluate_sequence().
    return np.clip(candidate, 0.0, 1000.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.
    The search consists of a fast random‑parameter sweep followed by a
    lightweight projected‑gradient refinement.  The function returns a plain
    Python list of floats (the best sequence found before the time budget
    expires).
    """
    rng = np.random.default_rng(seed)
    # Leave a small safety margin for the outer harness.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise the current best sequence.
    # ------------------------------------------------------------------
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_arr = None

    if base_arr is None or base_arr.ndim != 1 or base_arr.size == 0 \
            or not np.isfinite(base_arr).all():
        # Fallback: a constant sequence of length 256 (any length works).
        n = 256
        best = np.ones(n, dtype=np.float64)
    else:
        n = base_arr.size
        best = np.clip(base_arr, 0.0, 1000.0)

    # Ensure a modest total sum (evaluate_sequence rejects very tiny sums).
    current_sum = float(best.sum())
    if current_sum < 0.1:
        best *= (0.1 / max(current_sum, 1e-12))

    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Start: n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random parameter sweep over a simple 1‑parameter family.
    # ------------------------------------------------------------------
    # We try a few hundred candidates very quickly.  The amount of work
    # depends on the remaining time.
    sample_limit = 400
    samples_done = 0
    while time.monotonic() < deadline - 2.0 and samples_done < sample_limit:
        # Exponent p controls how strongly the sequence favours the edges.
        p = rng.uniform(0.0, 8.0)          # p=0 → uniform, larger → more “U‑shaped’’
        shape = generate_u_shape(n, p)

        # Mix the shape with a uniform component – this gives a richer
        # family without extra cost.
        mix = rng.random()
        candidate = mix * shape + (1.0 - mix) * np.ones(n, dtype=np.float64)

        # Random scaling; we deliberately push the edge values up to the
        # clipping bound 1000 to increase contrast.
        scale = rng.uniform(0.5, 1.0) * 1000.0
        candidate = np.clip(candidate * scale, 0.0, 1000.0)

        # Guard against the “sum too small’’ rejection.
        if candidate.sum() < 0.1:
            candidate *= (0.1 / max(candidate.sum(), 1e-12))

        score = float(evaluate_sequence(candidate.tolist()))
        samples_done += 1
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[Sweep] new best: score={best_score:.6f} (p={p:.3f}, mix={mix:.2f})")

    # ------------------------------------------------------------------
    # Phase 2 – very cheap projected gradient descent + occasional block
    #          perturbations.  It runs until the deadline.
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # Compute the full convolution once.
        conv = np.convolve(best, best)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # Gradient of the active anti‑diagonal (2*a_{k-i} on the admissible range).
        grad = np.zeros_like(best)
        i_start = max(0, max_idx - (n - 1))
        i_end   = min(max_idx, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = max_idx - i_vals
        grad[i_vals] = 2.0 * best[j_vals]

        # A simple step size proportional to the typical magnitude.
        step = 0.02 * (best.mean() if best.mean() > 0 else 1.0)
        candidate = best - step * grad

        # Small random jitter to keep the search stochastic.
        jitter = rng.normal(0.0, best.mean() * 0.005, size=n)
        candidate = candidate + jitter

        # Projection onto the feasible box.
        candidate = np.clip(candidate, 0.0, 1000.0)

        # Prevent the “sum too close to zero’’ rejection.
        if candidate.sum() < 0.1:
            candidate *= (0.1 / max(candidate.sum(), 1e-12))

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[Grad {iteration}] improved to {best_score:.6f} (max_idx={max_idx})")

        # Occasionally try a larger random block move – this often jumps out of
        # tiny plateaus that pure gradient steps cannot escape.
        if rng.random() < 0.12:
            cand2 = make_block_perturbation(best, rng)
            if cand2.sum() < 0.1:          # safety guard
                cand2 *= (0.1 / max(cand2.sum(), 1e-12))
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = cand2
                best_score = score2
                print(f"[Block {iteration}] improved to {best_score:.6f}")

    print(f"Finished. Best score={best_score:.6f}")
    # Return a plain Python list as required by the harness.
    return [float(x) for x in best.tolist()]

"""
A two‑phase optimizer for the “step‑function” problem.

Phase 1 – Random‑parameter sweep.
    We generate many candidates from a simple one‑parameter family
    a_i = (i/(n‑1))**p + ((n‑1‑i)/(n‑1))**p (U‑shaped, edges = 1,
    centre ≈ 2·0.5**p) and mix it with a uniform component.
    Each candidate is scaled so that the edge values hit the clipping
    bound 1000, then evaluated.  The best candidate is kept.

Phase 2 – Fast projected‑gradient refinement.
    Starting from the best candidate we repeatedly
        * compute the full convolution,
        * locate the anti‑diagonal with the maximal value,
        * take the gradient of that entry (2·a_{k‑i} on the admissible
          indices) and take a small step opposite to the gradient,
        * add a tiny random jitter,
        * project back onto the feasible box [0,1000] and keep the move
          if it improves the score.
    Occasionally a larger random block perturbation (the same routine
    used in the reference code) is tried to escape plateaus.

The whole loop runs until the supplied time budget is exhausted and
returns the best sequence found as a plain Python list of floats.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper: a simple “U‑shaped” family of non‑negative sequences.
# ----------------------------------------------------------------------
def generate_u_shape(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shape: edges are 1, centre ≈2·(0.5)**p."""
    i = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    left = (i / (n - 1)) ** p
    right = ((n - 1 - i) / (n - 1)) ** p
    return left + right


# ----------------------------------------------------------------------
# Helper: block‑wise multiplicative perturbation (identical to the reference).
# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative O(n) perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    if rng.random() < 0.35:               # occasional symmetrisation
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

# Main entry point required by the framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) discovered before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise the current best sequence.
    # ------------------------------------------------------------------
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_arr = None

    if (
        base_arr is None

        n = 256
        best = np.ones(n, dtype=np.float64)
    else:
        n = int(base_arr.size)
        best = np.clip(base_arr, 0.0, 1000.0)

    # Ensure a non‑trivial total mass (evaluate_sequence aborts for sum < 0.01).
    if best.sum() < 0.1:
        best *= (0.1 / max(best.sum(), 1e-12))

    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[Init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – fast random‑parameter sweep.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    sweep_limit = 400          # hard cap on number of random candidates
    sweeps_done = 0

    while time.monotonic() < deadline - 2.0 and sweeps_done < sweep_limit:
        p = rng.uniform(0.0, 8.0)                     # shape exponent
        shape = generate_u_shape(n, p)

        # Mix the U‑shape with a flat component for a richer family.
        mix = rng.random()
        cand = mix * shape + (1.0 - mix) * np.ones(n, dtype=np.float64)

        # Random scaling – pushes edge values up to the clipping bound.
        scale = rng.uniform(0.5, 1.0) * 1000.0
        cand = np.clip(cand * scale, 0.0, 1000.0)

        if cand.sum() < 0.1:                         # avoid the “sum too small’’ guard
            cand *= (0.1 / max(cand.sum(), 1e-12))

        score = float(evaluate_sequence(cand.tolist()))
        sweeps_done += 1
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[Sweep] p={p:.3f}, mix={mix:.2f}, score={best_score:.6f}")

    iteration = 0
    while time.monotonic() < deadline:

        grad = np.zeros_like(best)

        cand = np.clip(cand, 0.0, 1000.0)

        # Guard against the “sum too close to zero’’ rejection.
        if cand.sum() < 0.1:

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

            print(f"[Grad {iteration}] k={k_max}, score={best_score:.6f}")

            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = cand2
                best_score = score2
                print(f"[Block {iteration}] score={best_score:.6f}")

    print(f"[Done] best score={best_score:.6f}")
    return [float(x) for x in best.tolist()]
