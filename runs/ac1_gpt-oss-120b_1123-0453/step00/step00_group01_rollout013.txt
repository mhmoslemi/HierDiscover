def propose_candidate(seed=42, budget_s=1100):
   rng = np.random.default_rng(seed)
   deadline = time.monotonic() + max(0.0, budget_s - 3)
   best_seq = None
   best_score = float('inf')
   
   # initial candidate: uniform length 1000
   n0 = 1024
   candidate = np.full(n0, 1.0 / n0)
   best_seq = candidate
   best_score = evaluate_sequence(candidate.tolist())
   
   # maybe evaluate height_sequence_1 baseline if exists.
   try:
       base = np.asarray(height_sequence_1, dtype=np.float64)
       base = np.clip(base,0,1000)
       if base.ndim==1 and base.size>0 and np.isfinite(base).all():
           base_score = evaluate_sequence(base.tolist())
           if base_score < best_score:
               best_seq = base.copy()
               best_score = base_score
   except NameError:
       pass

   # main loop
   while time.monotonic() < deadline:
       # random length
       n = rng.integers(50, 5000)
       # random distribution type
       typ = rng.choice(['exp','power','dirichlet','uniform'])
       if typ == 'exp':
           r = rng.uniform(0.5,0.999)
           a = r**np.arange(n)
       elif typ == 'power':
           p = rng.uniform(0.1,2.0)
           a = 1.0 / (np.arange(1,n+1)**p)
       elif typ == 'dirichlet':
           alpha = rng.uniform(0.1,2.0)
           a = rng.gamma(alpha, 1.0, size=n)
       else: # uniform
           a = np.ones(n)
       # optionally random shuffle
       if rng.random() < 0.7:
           rng.shuffle(a)

       # normalize sum
       total = a.sum()
       if total < 1e-12:
           continue
       a = a / total

       # optionally apply smoothing
       if rng.random() < 0.3:
           a = (a + np.convolve(a, np.ones(3)/3, mode='same')) / 2

       # evaluate
       score = evaluate_sequence(a.tolist())
       if np.isfinite(score) and score < best_score:
           best_seq = a.copy()
           best_score = score
           print(f"New best score {best_score:.6f} with length {n}")

       # local hill climbing if we have best
       if best_seq is not None and rng.random() < 0.2:
           # copy best
           cand = best_seq.copy()
           # attempt small perturbations
           for _ in range(5):
               # choose two indices
               i, j = rng.choice(cand.size, size=2, replace=False)
               # perturb amount proportionally
               delta = rng.normal(0.0, 0.01) * cand[i]  # relative change
               # adjust ensuring keep nonnegative and sum unchanged
               new_i = cand[i] + delta
               new_j = cand[j] - delta
               if new_i < 0 or new_j < 0:
                   continue
               cand[i] = new_i
               cand[j] = new_j
               # evaluate
               score2 = evaluate_sequence(cand.tolist())
               if np.isfinite(score2) and score2 < best_score:
                   best_seq = cand.copy()
                   best_score = score2
                   print(f"  Local improvement to {best_score:.6f}")
   # final output: ensure list of floats
   if best_seq is None:
       # fallback uniform some length
       best_seq = np.full(256, 1.0/256)
   return [float(x) for x in best_seq]

"""
Propose a step function that minimizes

    2 * n * max(conv(f, f)) / (sum(f)**2)

using a small LP‑based refinement (the “f → g” update described
in the statement) together with random restarts.  The algorithm
runs until the time budget expires and returns the best sequence
found.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _score(seq: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for already‑validated data.
    Returns the objective value; ``np.inf`` for illegal inputs.
    """
    if seq.ndim != 1 or seq.size == 0:
        return np.inf
    if not np.isfinite(seq).all():
        return np.inf
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    n = seq.size
    max_b = np.max(np.convolve(seq, seq))
    return float(2.0 * n * max_b / (s * s))


def _build_A(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix that maps a vector g to the convolution
    f * g (linear in g).  Entry A[k, i] = f[i] if 0 ≤ k‑i < n, else 0.
    """
    n = f.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]
    return A


def _solve_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Maximise the sum of a non‑negative vector g under the linear constraints
    (f * g)[k] ≤ M for every k.  Returns the optimal g or ``None`` on failure.
    """
    n = f.size
    A = _build_A(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    # maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    # HiGHS is fast for medium‑size LPs
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        return None
    return res.x


def _refine(f: np.ndarray, t: float = 0.5, steps: int = 5) -> np.ndarray:
    """
    Perform a few (f → g) refinement steps.  After solving the LP for g,
    we blend the current f with the normalised g using mixing factor t.
    """
    f = np.asarray(f, dtype=np.float64)
    for _ in range(steps):
        M = np.max(np.convolve(f, f))
        g = _solve_g(f, M)
        if g is None:
            break
        s = np.sum(g)
        if s <= 0.0:
            break
        g_norm = g / s
        f = (1.0 - t) * f + t * g_norm
    return f


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a sequence with a low evaluation score.
    The routine uses random Dirichlet starts, the LP refinement described
    above and occasional local perturbations.  It stops when the time
    budget runs out and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    def _update(candidate: np.ndarray) -> None:
        nonlocal best_seq, best_score
        sc = _score(candidate)
        if np.isfinite(sc) and sc < best_score:
            best_seq = candidate.copy()
            best_score = sc
            print(f"New best score {best_score:.6f} with length {len(candidate)}")

    # start from a few simple baselines
    for n0 in (64, 128, 256):
        _update(np.full(n0, 1.0 / n0))

    # include the previously discovered construction if it exists
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            _update(base)
    except NameError:
        pass

    # main random‑restart loop
    while time.monotonic() < deadline:
        # pick a length (small enough for fast LP solves)
        n = int(rng.integers(30, 200))
        # draw a random Dirichlet vector (alpha controls concentration)
        alpha = float(rng.uniform(0.2, 5.0))
        f = rng.gamma(alpha, 1.0, size=n)
        f /= f.sum()                     # normalise to sum = 1

        # LP refinement
        t_mix = float(rng.uniform(0.3, 0.7))
        n_iter = int(rng.integers(3, 7))
        f_ref = _refine(f, t=t_mix, steps=n_iter)
        _update(f_ref)

        # occasional cheap hill‑climbing from the current best
        if best_seq is not None and rng.random() < 0.2:
            cand = best_seq.copy()
            # try a few small mass transfers
            for _ in range(3):
                i, j = rng.choice(cand.size, size=2, replace=False)
                delta = rng.normal(0.0, 0.02) * cand[i]
                if delta == 0.0:
                    continue
                new_i = cand[i] + delta
                new_j = cand[j] - delta
                if new_i < 0.0 or new_j < 0.0:
                    continue
                cand[i] = new_i
                cand[j] = new_j
            if cand.sum() > 0.0:
                cand = cand / cand.sum()
                _update(cand)

    # fallback – should never happen, but keep a safe default
    if best_seq is None:
        best_seq = np.full(256, 1.0 / 256)

    # convert to plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
