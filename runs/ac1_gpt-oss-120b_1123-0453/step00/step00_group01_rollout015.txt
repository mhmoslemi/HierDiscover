def project_simplex(v):
    # ensure v sums to 1 and each component >=0
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

Propose a low–score step sequence.

The algorithm starts from the analytically good “arcsine” shape
a_i ∝ 1/√(i+1) (the Beta(0.5, 1) distribution), which already yields a
score ≈ π/2 ≈ 1.57 < 2.  A simple projected sub‑gradient descent then
reduces the maximal convolution entry further.  At each iteration we
(1) compute the linear convolution of the current sequence with itself
    using an FFT (O(n log n)), (2) locate the index k where the convolution
    attains its maximum, (3) take the sub‑gradient of that maximal entry
    (g_i = 2·a_{k‑i} where defined), (4) step in the negative gradient
    direction with a diminishing step‑size, (5) project back onto the
    probability simplex (non‑negative entries summing to 1).  The best
    sequence seen during the allotted time is returned.

All numbers are non‑negative; scaling does not affect the evaluation

def _conv_fft(a):
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=m)
    B = A * A
    conv = np.fft.irfft(B, n=m)[: 2 * n - 1]
    return conv

    """Sub‑gradient of (a*a)[k] w.r.t. a (only the active k)."""
    n = a.size

    if i0 <= i1:
        i = np.arange(i0, i1 + 1, dtype=np.int64)
        j = k - i
        g[i] = 2.0 * a[j]
    return g

    a /= a.sum()                     # force unit sum → sum_a = 1

    best_a = a.copy()
    conv = _conv_fft(a)
    best_max = conv.max()
    print(f"[init] n={n}, initial max conv = {best_max:.6e}")

    # --- sub‑gradient descent -------------------------------------------------
    it = 0
    step0 = 0.5                     # initial step‑size scale
    while time.monotonic() < deadline:
        # current convolution and its maximum
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # compute sub‑gradient of the active maximal entry
        g = _gradient_for_max(a, k_max)

        # diminishing step size (1/√iteration)
        eta = step0 / np.sqrt(it + 1.0)

        # gradient step & projection onto the simplex (non‑negative, sum=1)
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            # degenerate case → re‑initialise uniformly
            a_new = np.full(n, 1.0 / n)

        # evaluate the new point (only the max entry is needed)
        conv_new = _conv_fft(a_new)
        max_new = conv_new.max()

        # accept if we made progress; otherwise occasionally perturb
        if max_new < max_val:
            a = a_new
            if max_new < best_max:
                best_max = max_new
                best_a = a_new.copy()
                print(f"[iter {it:5d}] improved max conv = {best_max:.6e}")
        else:
            # occasional random block perturbation to escape flat regions
            if rng.random() < 0.01:
                width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.12))
                a_pert = a.copy()
                a_pert[start:start + width] *= factor
                a_pert = np.clip(a_pert, 0.0, None)
                a_pert /= a_pert.sum()
                a = a_pert

        it += 1

    result = [float(x) for x in best_a]

    # final reporting (uses the provided evaluate_sequence if available)
    try:
        final_score = evaluate_sequence(result)
        print(f"[final] evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:
        print(f"[final] could not compute official score: {exc}")

    return result
