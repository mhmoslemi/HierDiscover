def propose_candidate(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    # get initial sequence
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = None
    if base is None or base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        n = 256
        base = np.ones(n, dtype=np.float64)
    parent = np.clip(base, 0.0, 1000.0)
    best = parent.copy()
    best_score = evaluate_sequence(best.tolist())
    current = best.copy()
    curr_score = best_score
    n = len(current)
    it = 0
    no_improve_iters = 0
    while time.monotonic() < deadline:
        it += 1
        # temperature
        t_remain = deadline - time.monotonic()
        # proportion left
        prop = t_remain / (budget_s)  # maybe approximate.
        T = max(0.0, prop) * 0.1  # start with 0.1 decreasing to 0
        # choose move type
        if rng.random() < 0.5:
            candidate = make_block_perturbation(current, rng)
        else:
            candidate = gradient_move(current, rng)
        # evaluate
        score = evaluate_sequence(candidate.tolist())
        if not np.isfinite(score):
            continue
        # update best
        if score < best_score:
            best = candidate.copy()
            best_score = score
            no_improve_iters = 0
            print(f"Iter {it}: new best {best_score:.6f}")
        else:
            no_improve_iters += 1
        # accept move for current
        if score < curr_score:
            current = candidate
            curr_score = score
        else:
            # maybe accept worse with probability
            if T > 0 and rng.random() < np.exp((curr_score - score) / T):
                current = candidate
                curr_score = score
        # occasional restart if stuck
        if no_improve_iters > 5000:
            # restart from best or random
            if rng.random() < 0.5:
                current = best.copy()
            else:
                # random perturb of best
                current = make_block_perturbation(best, rng)

            no_improve_iters = 0

def gradient_move(seq, rng):
    n = len(seq)
    a = np.asarray(seq, dtype=np.float64)
    # compute convolution to find max shift
    b = np.convolve(a, a)
    k_max = int(b.argmax())
    M = float(b[k_max])
    S = float(a.sum())
    if S == 0:
        # fallback to block perturbation
        return make_block_perturbation(seq, rng)
    target = M / S
    # partner values
    # compute indices of partner: j = k_max - i
    # We'll create array partner = np.where(0 <= j < n, a[j], 0)
    indices = np.arange(n)
    j = k_max - indices
    partner = np.where((j >= 0) & (j < n), a[j], 0.0)
    # high = indices where partner > target + 1e-12 and a>0
    high_mask = (partner > target) & (a > 1e-12)
    low_mask = (partner < target)
    high_indices = indices[high_mask]
    low_indices = indices[low_mask]
    if high_indices.size == 0 or low_indices.size == 0:
        # fallback to block perturbation
        return make_block_perturbation(seq, rng)
    i = int(rng.choice(high_indices))
    j_idx = int(rng.choice(low_indices))
    # step size: epsilon fraction of a[i] (or a[j] maybe)
    eps = 0.1  # maybe adapt based on iteration or magnitude
    delta = eps * a[i]
    # ensure delta not larger than a[i]
    delta = min(delta, a[i])
    cand = a.copy()
    cand[i] -= delta
    cand[j_idx] += delta
    # optionally ensure non-negative and clip
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

def gradient_move(seq, rng, epsilon):
    n = len(seq)
    a = np.asarray(seq, dtype=np.float64)
    b = np.convolve(a, a)
    # find max shift
    k_max = int(np.argmax(b))
    M = float(b[k_max])
    S = float(a.sum())
    if S == 0:
        return make_block_perturbation(seq, rng)
    target = M / S
    indices = np.arange(n)
    j = k_max - indices
    partner = np.where((j >= 0) & (j < n), a[j], 0.0)
    high_mask = (partner > target) & (a > 1e-12)
    low_mask = (partner < target)
    high_idxs = indices[high_mask]
    low_idxs = indices[low_mask]
    if high_idxs.size == 0 or low_idxs.size == 0:
        # fall back to block perturbation
        return make_block_perturbation(seq, rng)
    i = int(rng.choice(high_idxs))
    j_idx = int(rng.choice(low_idxs))
    # compute delta
    delta = epsilon * a[i]
    # ensure not exceed a[i]
    delta = min(delta, a[i])
    cand = a.copy()
    cand[i] -= delta
    cand[j_idx] += delta
    # clip
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

    We start from the given global `height_sequence_1` (or a uniform fallback),

    The loop runs until the time budget (≈1000 s) expires and the best sequence

            cur_score = score

            cur_score = float(evaluate_sequence(current.tolist()))
            no_improve = 0

    return [float(x) for x in best]

"""
Hybrid stochastic optimizer for the self‑convolution bound.

The algorithm starts from the global `height_sequence_1` (or a uniform fallback)
and repeatedly applies two types of moves:

1. **Block perturbation** – multiplicative scaling of a random contiguous block
   (optionally mirrored).  After the perturbation the whole vector is rescaled
   to keep the total sum unchanged, which stabilises the evaluation function.

2. **Gradient‑guided mass transfer** – the current maximal convolution entry
   `k*` is identified.  Indices whose partner `a[k*‑i]` is larger than the
   target value `M / sum(a)` are considered “high”, those with a smaller partner
   are “low”.  A small amount of mass is moved from a high index to a low index,
   which tends to lower the maximal convolution value.

An annealing schedule allows occasional uphill moves, and a simple restart
policy prevents stagnation.  The search runs until the time budget (≈1000 s)
expires and returns the best sequence found.
"""

import numpy as np
import time

def make_block_perturbation(base, rng):
    """Return a non‑negative O(n) block perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # choose a block width; ensure at least 1 element
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))          # [min_w, max_w]
    start = int(rng.integers(0, max(1, n - width + 1)))  # start position
    stop = min(n, start + width)

    # multiplicative perturbation of the block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor

    # optional symmetric averaging (helps to keep the shape smooth)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])

    # keep values inside the allowed range
    cand = np.clip(cand, 0.0, 1000.0)

    # renormalise to the original L1 norm (preserves the scale of the score)
    orig_sum = np.sum(base)
    cur_sum = np.sum(cand)
    if cur_sum > 0.0:
        cand *= (orig_sum / cur_sum)
    cand = np.clip(cand, 0.0, 1000.0)
    return cand


def gradient_move(seq, rng, epsilon):
    """
    Guided move that shifts a fraction epsilon of mass from an index that
    contributes a lot to the current maximal convolution entry to an index
    that contributes little.
    """
    a = np.asarray(seq, dtype=np.float64)
    n = a.size

    # convolution and the position of its maximum
    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))
    M = float(conv[k_max])
    S = float(a.sum())
    if S < 1e-12:
        # degenerate – fall back to a random block perturbation
        return make_block_perturbation(seq, rng)

    target = M / S                     # desired partner value
    idx = np.arange(n)
    partner_idx = k_max - idx
    partner = np.where((partner_idx >= 0) & (partner_idx < n), a[partner_idx], 0.0)

    high_mask = (partner > target) & (a > 1e-12)
    low_mask = partner < target

    high = idx[high_mask]
    low = idx[low_mask]

    if high.size == 0 or low.size == 0:
        # no clear direction – use a block perturbation instead
        return make_block_perturbation(seq, rng)

    i = int(rng.choice(high))          # source (high contribution)
    j = int(rng.choice(low))           # destination (low contribution)

    # move a small fraction of the source mass to the destination
    delta = epsilon * a[i]
    delta = min(delta, a[i])           # stay non‑negative

    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta

    cand = np.clip(cand, 0.0, 1000.0)
    return cand


def propose_candidate(seed=42, budget_s=1100):
    """Search for a step‑function sequence with a low evaluation score.

    The routine runs for roughly ``budget_s`` seconds (≈1000 s) and returns
    the best sequence discovered.  It mixes random block perturbations with
    the gradient‑guided mass‑transfer moves described above.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin so the caller does not cut us off
    deadline = start_time + max(0.0, float(budget_s) - 0.5)
    total_time = deadline - start_time

    # ----- initialise -------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base = None
    if base is None or base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        # simple uniform fallback (length 512 keeps the search cheap enough)
        n = 512
        base = np.ones(n, dtype=np.float64)
    parent = np.clip(base, 0.0, 1000.0)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    current = best.copy()
    cur_score = best_score

    n = len(current)
    it = 0
    no_improve = 0

    # annealing parameters (tuned empirically)
    init_temp = 0.05
    final_temp = 0.0005

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        remaining = max(0.0, deadline - now)
        prop = remaining / total_time

        # temperature drops linearly
        temperature = final_temp + (init_temp - final_temp) * prop
        # step size for the gradient move (larger early on)
        eps = 0.01 + 0.2 * prop   # in [0.01, 0.21]

        # ---- choose a move ------------------------------------------------
        if rng.random() < 0.5:
            cand = make_block_perturbation(current, rng)
        else:
            cand = gradient_move(current, rng, eps)

        # ---- evaluate ------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            continue

        # ---- record best ---------------------------------------------------
        if score < best_score:
            best = cand.copy()
            best_score = score
            no_improve = 0
            print(f"Iter {it:8d}: new best = {best_score:.6f}")

        else:
            no_improve += 1

        # ---- accept/reject for the current state -------------------------
        if score < cur_score:
            current = cand
            cur_score = score
        else:
            # occasional uphill moves (simulated annealing)
            if temperature > 0.0 and rng.random() < np.exp((cur_score - score) / temperature):
                current = cand
                cur_score = score

        # ---- restart if we are stuck for a long time ----------------------
        if no_improve > 5000:
            # bounce back to the best known point with a fresh perturbation
            current = make_block_perturbation(best, rng)
            cur_score = float(evaluate_sequence(current.tolist()))
            no_improve = 0
            print(f"Iter {it:8d}: restart – best still {best_score:.6f}")

    # final result
    return [float(x) for x in best]
