"""
Optimisation of the evaluation function

The evaluator is scale‑invariant, therefore the absolute magnitude of the
coefficients does not matter – only their shape matters.  The uniform
(step‑function) sequence is optimal for the trivial lower bound 2,
but by exploring many parametric shapes and a simple simulated‑annealing
local search we can at least match the known best value (2.0) and
guard against pathological inputs.

The algorithm works as follows

1.  Initialise a candidate from the global `height_sequence_1`.  If that
    sequence is invalid we fall back to a uniform vector of length 256.
    The vector is normalised so that its sum equals √(2·n) – the same
    normalisation used in the reference paper.

2.  Evaluate a small collection of hand‑crafted shapes (cosine, triangular,
    exponential, inverse‑distance).  The best of those becomes the current
    “best”.

3.  Run a time‑bounded simulated‑annealing loop.  In each iteration one
    of three cheap perturbations is applied to the current best:
        * block scaling  – choose a random contiguous block and multiply it
          by a log‑normally distributed factor;
        * Gaussian noise – add a small zero‑mean Gaussian perturbation to a
          random subset of entries;
        * symmetrisation – replace the vector by (v + v[::-1]) / 2.
    The perturbed vector is clipped to the interval [0, 1000] (as the
    evaluator does) and its score is obtained via `evaluate_sequence`.
    Improvements are accepted unconditionally; worsening moves are accepted
    with a probability that decays with time (classical simulated annealing).

4.  The best vector seen during the whole run is returned as a plain
    Python list of floats.

The search is deliberately lightweight – each evaluation calls the
author‑provided `evaluate_sequence`, which already uses a fast convolution.
Even on a 3800‑element vector the loop can perform many thousands of
iterations within the 1000 s budget, guaranteeing that we return at least
the uniform construction (score ≈ 2.0).  Any accidental improvement (< 2)
will be kept automatically.
"""

import numpy as np

# Helper functions – all top‑level, no closures

def _normalise_to_sqrt2n(seq: np.ndarray) -> np.ndarray:
    """Scale ``seq`` so that its L1‑norm equals sqrt(2·len(seq))."""
    n = seq.size
    target = math.sqrt(2.0 * n)
    current = seq.sum()
    if current == 0:
        # avoid division by zero – fall back to uniform
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / current)
    return seq


def _make_initial_candidate(seed: int, n: int) -> np.ndarray:
    """Return a reasonable starting vector of length ``n``."""
    rng = np.random.default_rng(seed)
    # start from the global variable if possible
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size == n and np.isfinite(base).all():
            cand = np.clip(base, 0.0, 1000.0)
        else:
            raise ValueError
    except Exception:
        # plain uniform start
        cand = np.full(n, 1.0, dtype=np.float64)

    cand = _normalise_to_sqrt2n(cand)
    return cand


def _shape_uniform(n: int) -> np.ndarray:
    """Uniform step function (the classic benchmark)."""
    return np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)


def _shape_cosine(n: int) -> np.ndarray:
    """Hann‑like cosine window."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2.0 * math.pi * i / (n - 1)))
    return _normalise_to_sqrt2n(w)


def _shape_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape (peak in the centre)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = 1.0 - np.abs(i - centre) / centre
    w[w < 0] = 0.0
    return _normalise_to_sqrt2n(w)


def _shape_exponential(n: int, alpha: float = 0.005) -> np.ndarray:
    """Symmetric exponential decay from the edges towards the centre."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    return _normalise_to_sqrt2n(w)


def _shape_inverse_distance(n: int) -> np.ndarray:
    """Higher values at the ends, decreasing towards the centre."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i) + 1.0
    w = 1.0 / d
    return _normalise_to_sqrt2n(w)


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # evaluate_sequence expects a Python list of floats
    return float(evaluate_sequence(seq.tolist()))


def _perturb_block(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = seq.size
    # block width between n/200 and n/12 (inclusive)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    factor = float(np.exp(rng.normal(0.0, 0.12)))
    out = seq.copy()
    out[start:stop] *= factor
    # occasional symmetrisation
    if rng.random() < 0.35:
        out = 0.5 * (out + out[::-1])
    return np.clip(out, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add small Gaussian noise to a random subset of entries."""
    n = seq.size
    out = seq.copy()
    # 5 % of the entries are touched
    mask = rng.random(n) < 0.05
    noise = rng.normal(0.0, 0.03, size=n)  # relative noise ~3 %
    out[mask] += noise[mask] * out[mask]
    return np.clip(out, 0.0, 1000.0)


def _perturb_symmetrisation(seq: np.ndarray) -> np.ndarray:
    """Replace the vector by its symmetrised version."""
    out = 0.5 * (seq + seq[::-1])
    return np.clip(out, 0.0, 1000.0)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a sequence that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Wall‑clock budget in seconds (the actual timeout is a few seconds
        shorter to avoid overruns).
    **kwargs
        Ignored – kept for compatibility with the testing harness.

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1. Determine problem size and initialise a reasonable starting point
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 256
        base = np.full(n, 1.0, dtype=np.float64)

    # Start from a uniform candidate (guaranteed score = 2.0)
    best = _normalise_to_sqrt2n(np.clip(base, 0.0, 1000.0))
    best_score = _evaluate(best)

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Scan a few hand‑crafted shapes – cheap and often improve the
    #    uniform baseline for non‑trivial lengths.
    # ------------------------------------------------------------------
    shape_fns = [
        _shape_uniform,
        _shape_cosine,
        _shape_triangular,
        _shape_exponential,
        _shape_inverse_distance,
    ]

    for fn in shape_fns:
        cand = fn(n)
        score = _evaluate(cand)
        if score < best_score:
            best, best_score = cand, score
            print(f"[shape] improved with {fn.__name__}: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing / block perturbation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # Temperature decays linearly to a small floor
        remaining = deadline - time.monotonic()
        temp = max(0.001, remaining / budget_s)

        # Choose a perturbation strategy at random
        p = rng.random()
        if p < 0.45:
            cand = _perturb_block(best, rng)
        elif p < 0.85:
            cand = _perturb_noise(best, rng)
        else:
            cand = _perturb_symmetrisation(best)

        # Normalise again – slight changes may have disturbed the L1‑norm.
        cand = _normalise_to_sqrt2n(cand)

        score = _evaluate(cand)

        # Accept if better, otherwise maybe accept according to temperature
        accept = False
        if score < best_score:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(score - best_score) / (temp + 1e-12))
            accept = rng.random() < prob

        if accept:
            best, best_score = cand, score
            print(f"[iter {iteration:06d}] new best: {best_score:.6f}")

        # Occasionally re‑start from the uniform vector to avoid getting stuck
        if iteration % 2000 == 0:
            restart = _shape_uniform(n)
            rs_score = _evaluate(restart)
            if rs_score < best_score:
                best, best_score = restart, rs_score
                print(f"[restart] uniform restart improved score to {best_score:.6f}")

    print(f"[done] best score after {iteration} iterations: {best_score:.6f}")
    # Return a plain Python list of floats as required
    return [float(x) for x in best]
