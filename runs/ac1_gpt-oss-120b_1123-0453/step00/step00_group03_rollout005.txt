def propose_candidate(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # get base from global variable 'height_sequence_1', fallback to uniform.
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        n = 256
        parent = np.ones(n, dtype=np.float64)
    n = parent.size

    # ensure values in bounds.
    parent = np.clip(parent, 0.0, 1000.0)

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # optional: evaluate some random restarts in parallel? We can just loop.

    # define parameters
    # SA parameters
    initial_temp = 0.1  # scale of acceptance for increases relative to score difference
    final_temp = 1e-4
    # We'll compute iteration number guess = (deadline - time.monotonic()) / (approx time per iteration). Approx 0.0001 sec? Not known.
    # We'll just compute temperature based on elapsed fraction.

    iter = 0
    while time.monotonic() < deadline:
        # compute convolution and gradient at max offset
        conv = np.convolve(best_seq, best_seq)
        max_val = conv.max()
        max_idx = conv.argmax()
        offset = max_idx - (n - 1)  # offset can be negative
        # compute gradient
        # Use vectorized compute: g = np.zeros(n)
        g = np.zeros(n, dtype=np.float64)
        if offset >= 0:
            # i+offset valid for i in [0, n-offset-1]; i-offset valid for i in [offset, n-1]
            # contributions:
            g[:n - offset] += best_seq[offset:]  # a_{i+offset}
            g[offset:] += best_seq[:n - offset]  # a_{i-offset}
        else:
            off = -offset
            # i+offset valid for i in [off, n-1]; i-offset valid for i in [0, n-off-1]
            g[off:] += best_seq[:n - off]  # i+offset = i - off => a_{i-off}
            g[:n - off] += best_seq[off:]  # i-offset = i + off => a_{i+off}
        # Now select i high and j low
        # Determine probabilities
        # High: weighted by g (plus epsilon)
        g_sum = g.sum()
        if g_sum == 0:
            # all zero? then skip gradient, just random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        else:
            # sample i weighted by g
            prob_i = g / g_sum
            i = rng.choice(n, p=prob_i)
            # sample j weighted by 1/(g+eps)
            eps = 1e-12
            inv = 1.0 / (g + eps)
            inv_sum = inv.sum()
            prob_j = inv / inv_sum
            j = rng.choice(n, p=prob_j)
            # ensure i!=j
            if i == j:
                j = (j + 1) % n
        # compute transfer amount
        # Max transferable from i is best_seq[i] (can't go negative)
        max_transfer = best_seq[i]
        if max_transfer <= 0:
            # pick another i maybe random
            continue
        # Choose fraction
        frac = rng.uniform(0.01, 0.3)  # up to 30% of mass
        delta = max_transfer * frac
        # ensure j capacity
        capacity = 1000.0 - best_seq[j]
        if delta > capacity:
            delta = capacity
        if delta <= 0:
            continue
        # apply move
        candidate = best_seq.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # optional: ensure non-negative (clamp)
        candidate = np.clip(candidate, 0.0, 1000.0)

        # evaluate
        cand_score = float(evaluate_sequence(candidate.tolist()))
        # compute temperature based on elapsed fraction
        elapsed = time.monotonic() - (deadline - float(budget_s) + 3.0)  # Not accurate.
        # Actually compute fraction remaining:
        fraction = (deadline - time.monotonic()) / (float(budget_s) - 3.0)  # Not robust
        # For temperature schedule: linearly decreasing from init to final.
        T = final_temp + (initial_temp - final_temp) * fraction
        # decide acceptance
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            # acceptance prob
            if T > 0:
                prob = np.exp(-(cand_score - best_score) / (T * best_score + 1e-12))
                if rng.random() < prob:
                    accept = True
        if accept:
            best_seq = candidate
            best_score = cand_score
            # optionally log improvement
            print(f"Iter {iter}: Improved score to {best_score:.6f}")
        iter += 1

    # final best sequence
    return [float(x) for x in best_seq]

```python
import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Simulated‑annealing / gradient‑guided local search for the step‑function heights.

    The algorithm starts from the supplied global ``height_sequence_1`` (or from a
    random sequence if it is unavailable) and repeatedly moves a small amount of
    mass from a “high‑correlation’’ index to a “low‑correlation’’ index.
    The move choice is based on the gradient of the current maximal
    autocorrelation term (the quantity that dominates the evaluation function).
    Occasionally a larger random block perturbation (the one used in the
    reference code) is applied to escape local minima.  Accepted moves are
    governed by a simple simulated‑annealing schedule that gradually reduces
    the temperature, so the search quickly focuses on improvements while still
    allowing occasional uphill steps.

    The function runs until ``budget_s`` seconds have elapsed (minus a small
    safety margin) and returns the best sequence found as a plain Python list
    of floats.
    """
    #--- Setup ----------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)

    # Load initial sequence (global variable provided by the harness)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base = None

    if base is None or base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        # Fallback: uniform length 256
        n = 256
        base = np.ones(n, dtype=np.float64)
    else:
        n = base.size

    # Clamp to the limits enforced by ``evaluate_sequence``
    base = np.clip(base, 0.0, 1000.0)

    # Create a random alternative start (same length) to diversify the search
    random_start = rng.random(n) * 100.0  # scale not important (clamp later)
    random_start = np.clip(random_start, 0.0, 1000.0)

    # Choose the better of the two as the initial “best’’ candidate
    candidates = [base, random_start]
    best_seq = None
    best_score = np.inf
    for cand in candidates:
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # Safety fallback – in the unlikely case both candidates are invalid
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Parameters for the simulated‑annealing schedule
    T_start = 0.12          # initial temperature (relative to objective)
    T_end   = 5e-4          # final temperature
    total_budget = deadline - start_time

    # Helper: block perturbation similar to the reference implementation
    def block_perturb(seq, rng_local):
        """Return a new sequence obtained by scaling a random block."""
        cand = seq.copy()
        width = int(rng_local.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng_local.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = float(np.exp(rng_local.normal(0.0, 0.12)))
        cand[start:stop] *= factor
        # Occasionally symmetrize to inject extra structure
        if rng_local.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        return np.clip(cand, 0.0, 1000.0)

    # Helper: compute the gradient of the maximal autocorrelation term
    def max_corr_gradient(seq):
        """Return (offset, gradient array) for the current maximal convolution."""
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        offset = max_idx - (n - 1)   # signed shift that attains max
        g = np.zeros_like(seq)
        if offset >= 0:
            # i+offset term
            g[:n - offset] += seq[offset:]
            # i-offset term
            g[offset:] += seq[:n - offset]
        else:
            off = -offset
            g[off:] += seq[:n - off]
            g[:n - off] += seq[off:]
        return offset, g

    # Main optimisation loop
    it = 0
    last_improve_it = 0
    while time.monotonic() < deadline:
        # Occasionally perform a big block move (probability decays slowly)
        if rng.random() < 0.03:
            cand = block_perturb(best_seq, rng)
        else:
            # Gradient‑guided small transfer move
            offset, grad = max_corr_gradient(best_seq)

            # Sampling probabilities: high gradient → source, low gradient → sink
            grad_sum = grad.sum()
            if grad_sum == 0:
                # pure random move if gradient vanished (should be rare)
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
            else:
                prob_src = grad / grad_sum
                i = rng.choice(n, p=prob_src)
                # Inverse‑gradient weight for the destination
                inv = 1.0 / (grad + 1e-12)
                prob_dst = inv / inv.sum()
                j = rng.choice(n, p=prob_dst)
                if i == j:
                    j = (j + 1) % n

            # Transfer a fraction of the mass at i to j
            src_val = best_seq[i]
            if src_val <= 0.0:
                # No mass to move – skip this iteration
                it += 1
                continue
            # Random fraction between 1 % and 30 % of the source value
            frac = rng.uniform(0.01, 0.30)
            delta = src_val * frac
            # Respect the upper bound of 1000 at the destination
            capacity = 1000.0 - best_seq[j]
            if delta > capacity:
                delta = capacity
            if delta <= 0.0:
                it += 1
                continue

            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            # Numerical safety – enforce the allowed range
            cand = np.clip(cand, 0.0, 1000.0)

        # Evaluate the candidate
        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            # Invalid candidate – ignore
            it += 1
            continue

        # Temperature based on remaining time
        elapsed = time.monotonic() - start_time
        fraction = max(0.0, (total_budget - elapsed) / total_budget)
        T = T_end + (T_start - T_end) * fraction

        # Acceptance rule (simulated annealing)
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            # Probabilistic uphill move
            if T > 0.0:
                prob = np.exp(-(cand_score - best_score) / (T + 1e-12))
                if rng.random() < prob:
                    accept = True

        if accept:
            best_seq = cand
            best_score = cand_score
            last_improve_it = it
            # Log progress occasionally (every 1000 accepted moves)
            if it % 1000 == 0:
                print(f"[{it}] New best score: {best_score:.6f}")

        # Simple restart heuristic: if we have gone many iterations
        # without improvement, re‑seed from a random perturbation of the current best.
        if it - last_improve_it > 20000:
            # create a fresh candidate by a moderate block perturbation
            best_seq = block_perturb(best_seq, rng)
            best_score = float(evaluate_sequence(best_seq.tolist()))
            last_improve_it = it
            print(f"[{it}] Restart – score reset to {best_score:.6f}")

        it += 1

    # Return the best sequence found (as a plain Python list)
    return [float(x) for x in best_seq]
```
