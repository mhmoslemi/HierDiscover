def partial_score(seq):
   seq = np.clip(seq,0,1000) # ensure non-negative and limit.
   sum_a = np.sum(seq)
   if sum_a < 0.01:
       return np.inf
   # Using FFT compute convolution:
   n = seq.shape[0]
   size = 2*n - 1
   # Find next power of two for speed:
   # Use np.fft.rfft for real signals.
   m = 1 << (size-1).bit_length()
   # zero pad both seq to length m.
   a_fft = np.fft.rfft(seq, n=m)
   b_fft = a_fft * a_fft
   conv = np.fft.irfft(b_fft)[:size]
   max_b = np.max(conv)
   return 2 * n * max_b / (sum_a**2)

"""
Propose a sequence that minimizes the provided evaluation function.

The strategy is based on the known (continuous) arcsine distribution:
    a_i ≈ 1 / sqrt(i+1)

For this shape the self‑convolution is almost constant, giving an
asymptotic score close to π/2 ≈ 1.57.  Starting from this analytical
construction we perform a fast gradient‑like local search that repeatedly
reduces the coefficient responsible for the current maximal convolution
value while keeping the total sum of the sequence unchanged.

Key ideas
---------
* Initialise a non‑negative vector `a` with the arcsine profile.
* Compute the convolution by FFT (O(n log n)) – far faster than the
  `np.convolve` used in the reference evaluator.
* Find the index `k_max` where the convolution attains its maximum.
  The derivative of this maximum w.r.t. `a_i` is `a_{k_max-i}`.
  Large products `a_i * a_{k_max-i}` are the main contributors to the
  current worst term.
* Transfer a small fraction of mass from indices with the largest
  contributions to indices with the smallest contributions.  This keeps
  the overall sum constant but lowers the peak of the convolution.
* Occasionally apply a random block perturbation (the technique used in
  the reference solution) to escape shallow local minima.
* Keep the best sequence seen so far and return it when the time budget
  expires.

The algorithm uses only NumPy and the standard library, runs within the
provided time budget, and respects all constraints of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_convolution(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full linear convolution `seq * seq` using FFT."""
    # Real FFT of the zero‑padded sequence
    fft_seq = np.fft.rfft(seq, n=fft_len)
    # Autocorrelation in the frequency domain (element‑wise square)
    prod = fft_seq * fft_seq
    # Inverse transform – keep only the first 2·n‑1 entries
    conv_full = np.fft.irfft(prod)[: 2 * seq.size - 1]
    return conv_full


def _score_from_fft(seq: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate for `evaluate_sequence`.  It reproduces the exact
    arithmetic (up to the tiny numerical errors of FFT) used by the
    evaluator, but avoids the O(n²) direct convolution.
    """
    n = seq.size
    conv = _fft_convolution(seq, fft_len)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(seq))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _initial_arcsine(n: int) -> np.ndarray:
    """Return the arcsine profile a_i = 1 / sqrt(i+1) for i = 0 … n‑1."""
    return 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 1.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds.  The function stops a few seconds before the
        limit to avoid a hard termination.
    **kwargs
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size -------------------------------------------------
    # ------------------------------------------------------------------
    # Use the length of the previously known construction if it exists.
    # Fallback to a reasonable default (≈5 000) if the global is missing.
    global height_sequence_1
    try:
        n = int(len(height_sequence_1))
        if n == 0:
            raise ValueError
    except Exception:
        n = 4806  # a size already used by the reference solution

    # ------------------------------------------------------------------
    # Pre‑compute the FFT length (next power of two) – reused each iteration.
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())  # smallest power of two ≥ conv_len

    # ------------------------------------------------------------------
    # Initialise with the analytical arcsine distribution.
    # ------------------------------------------------------------------
    a = _initial_arcsine(n)
    a = np.clip(a, 0.0, 1000.0)            # respect the 0‑1000 bounds
    best = a.copy()
    best_score = _score_from_fft(best, fft_len)
    print(f"[init] n={n}, initial score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – stops a few seconds before the deadline.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    iteration = 0
    # hyper‑parameters (tuned empirically)
    donor_fraction = 0.005          # proportion of indices considered donors/receivers
    move_fraction = 0.06            # how much of a donor's mass is moved per step
    random_block_prob = 0.015       # probability of a random block perturbation

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) Compute the current convolution and locate its maximum.
        # --------------------------------------------------------------
        conv = _fft_convolution(a, fft_len)
        k_max = int(np.argmax(conv))

        # --------------------------------------------------------------
        # 2) Gradient‑like information: contribution of each a_i to the
        #    peak at `k_max` is a_i * a_{k_max-i}.
        # --------------------------------------------------------------
        idx = k_max - np.arange(n)                 # partner indices for each i
        mask = (idx >= 0) & (idx < n)               # valid partners only
        partner = np.empty_like(a)
        partner.fill(0.0)
        partner[mask] = a[idx[mask]]
        contributions = a * partner                  # product = a_i * a_{k_max-i}

        # --------------------------------------------------------------
        # 3) Choose a donor (large contribution) and a receiver
        #    (small contribution) and transfer a small amount of mass.
        # --------------------------------------------------------------
        top_k = max(1, int(n * donor_fraction))    # number of extreme indices
        # donors: indices with the largest contributions
        donor_idx = np.argpartition(contributions, -top_k)[-top_k:]
        # receivers: indices with the smallest contributions
        recv_idx = np.argpartition(contributions, top_k)[:top_k]

        # Pick a random donor and receiver from the respective sets
        donor = int(rng.choice(donor_idx))
        receiver = int(rng.choice(recv_idx))

        # Amount to move – a modest fraction of the donor's current value
        delta = a[donor] * move_fraction
        if delta > 0:
            a[donor] -= delta
            a[receiver] += delta

        # Ensure numerical bounds after the transfer
        np.clip(a, 0.0, 1000.0, out=a)

        # --------------------------------------------------------------
        # 4) Evaluate the new candidate (fast surrogate) and keep it
        #    if it improves the best score seen so far.
        # --------------------------------------------------------------
        cur_score = _score_from_fft(a, fft_len)
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            print(f"[iter {iteration:5d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5) Occasionally perform a random block perturbation to
        #    diversify the search (similar to the reference code).
        # --------------------------------------------------------------
        if rng.random() < random_block_prob:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            a[start:stop] *= factor
            # optional symmetric smoothing – keeps the search space rich
            if rng.random() < 0.35:
                a[:] = 0.5 * (a + a[::-1])
            np.clip(a, 0.0, 1000.0, out=a)

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python floats).
    # ------------------------------------------------------------------
    print(f"[final] best score≈{best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]
