    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)
    best_seq = None
    best_score = np.inf

    # Define list of lengths to try: from small to large (maybe [128,256,512,1024,2048])
    lengths = [128,256,512,1024,2048]
    # Optionally also use height_sequence_1 length (global variable if available)
    if 'height_sequence_1' in globals():
        lengths.append(len(height_sequence_1))

    for n in lengths:
        # generate a few seed sequences:
        # uniform
        seq = np.ones(n) / n
        # linear decreasing
        seq_lin = np.arange(n, 0, -1, dtype=np.float64)
        seq_lin = seq_lin / seq_lin.sum()
        # exponential decay
        lam = 0.01 * n
        seq_exp = np.exp(-np.arange(n) / lam)
        seq_exp = seq_exp / seq_exp.sum()
        # Triangular shape (mirror)
        seq_tri = np.concatenate([np.arange(1, n//2+1), np.arange(n//2,0,-1)])[:n]
        seq_tri = seq_tri / seq_tri.sum()
        # Initialize list of seeds
        seed_seqs = [seq, seq_lin, seq_exp, seq_tri]
        # Might also add random uniform series
        for _ in range(2):
            seq_rand = rng.random(n)
            seq_rand /= seq_rand.sum()
            seed_seqs.append(seq_rand)
        # If height_sequence_1 is same length, use its normalized version
        if 'height_sequence_1' in globals() and len(height_sequence_1) == n:
            seq_h = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            seq_h = seq_h / seq_h.sum() if seq_h.sum() > 0 else np.ones(n)/n
            seed_seqs.append(seq_h)

        for base in seed_seqs:
            # start local optimization
            cur = np.copy(base)
            cur_score = float(evaluate_sequence(cur.tolist()))
            # track local best
            local_best = cur.copy()
            local_best_score = cur_score
            # Simulated annealing schedule
            iter_count = 0
            while time.monotonic() < deadline:
                # temperature decreasing with iterations
                T = max(0.001, 1.0 - (time.monotonic() - deadline + budget_s) / budget_s)
                # propose
                new_seq = propose_move(cur, rng)
                new_score = float(evaluate_sequence(new_seq.tolist()))
                # acceptance
                if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(T, 1e-12)):
                    cur = new_seq
                    cur_score = new_score
                # update local best
                if cur_score < local_best_score:
                    local_best = cur.copy()
                    local_best_score = cur_score
                # also update global best
                if local_best_score < best_score:
                    best_seq = local_best.copy()
                    best_score = local_best_score
                    print(f'New global best: {best_score:.6f} at length {n}')
                # iteration cutoff maybe
                iter_count += 1
                if iter_count % 1000 == 0 and time.monotonic() >= deadline:
                    break

    if best_seq is None:

        best_seq = np.ones(256) / 256
    # Ensure nonnegative, clip to range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

def propose_move(current, rng):

    typ = rng.integers(0,3)

def propose_move(seq, rng):
    n = seq.size
    new = seq.copy()
    # randomly decide move type
    move_type = rng.integers(0, 4)
    if move_type == 0:
        # transfer small amount from random i to random j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        delta = new[i] * rng.uniform(0.0, 0.1)  # up to 10% of source
        new[i] -= delta
        new[j] += delta
    elif move_type == 1:
        # multiply a random block by factor
        width = max(1, n // rng.integers(10, 50))  # random width
        start = rng.integers(0, max(1, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.05))
        new[start:start+width] *= factor
    elif move_type == 2:
        # perturb all entries with small Gaussian noise
        noise = rng.normal(0.0, 0.01, size=n)
        new = np.maximum(new + noise, 0.0)
    else:
        # reflection: make sequence more symmetric
        if rng.random() < 0.5:
            new = 0.5 * (new + new[::-1])
    # Re-normalize sum to 1
    total = new.sum()
    if total > 0:
        new = new / total
    else:
        new = np.ones(n) / n
    return new

def perturb_sequence(seq, rng):
    n = seq.size
    candidate = seq.copy()
    # random block scaling or transfer
    move_type = rng.integers(0, 3)
    if move_type == 0:
        # small transfer between two random indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        delta = candidate[i] * rng.uniform(0.0, 0.1)
        candidate[i] -= delta
        candidate[j] += delta
    elif move_type == 1:
        # block scaling
        width = max(1, n // rng.integers(10, 50))
        start = rng.integers(0, max(1, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.05))
        candidate[start:start+width] *= factor
    else:
        # add small Gaussian noise
        noise = rng.normal(0.0, 0.01, size=n)
        candidate = np.maximum(candidate + noise, 0.0)
    # renormalize to keep sum ~1
    total = candidate.sum()
    if total > 0:
        candidate = candidate / total
    return candidate

"""
Linear decreasing step function with optional hill‑climbing refinement.

The base construction uses a simple arithmetic progression
a_i ∝ (n - i) (i = 0 … n‑1) which is normalized to sum = 1.
For large n the evaluation value tends to 4/3 ≈ 1.333…, well

If time permits (budget_s seconds) a cheap local search is run:
random pairwise transfers or block scalings are proposed and
kept only when they improve the objective
`evaluate_sequence`.  The search respects the non‑negative
and sum‑greater‑than‑0.01 constraints required by the evaluator.
The best sequence found before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def linear_decreasing(n: int) -> np.ndarray:
    """
    Return a normalized linearly decreasing sequence of length n:
        a_i = (n - i) / sum_{j=1}^{n} j
    The sum of the returned array is exactly 1.0.
    """
    # descending integers  n, n-1, ..., 1
    arr = np.arange(n, 0, -1, dtype=np.float64)
    arr /= arr.sum()                     # normalization to unit sum
    return arr


def perturb_sequence(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a modest random perturbation of ``seq`` while keeping the sum = 1.
    Three simple move types are used:
      * transfer a fraction of mass from one index to another,
      * multiply a random contiguous block by a small factor,
      * add tiny Gaussian noise to all entries.
    The result is clipped to stay non‑negative and re‑normalized.
    """
    n = seq.size
    cand = seq.copy()

    move_type = rng.integers(0, 3)

    if move_type == 0:
        # transfer a random fraction from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n - 1)

        # take at most 10 % of the source entry
        delta = cand[i] * rng.uniform(0.0, 0.1)
        cand[i] -= delta
        cand[j] += delta

    elif move_type == 1:
        # scale a random block by a log‑normal factor
        # block width is between n/200 and n/12 (as in the reference code)
        min_w = max(1, n // 200)
        max_w = max(2, n // 12)
        width = rng.integers(min_w, max_w + 1)
        start = rng.integers(0, max(1, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.08))
        cand[start:start + width] *= factor

    else:
        # add very small Gaussian noise, then keep non‑negative
        noise = rng.normal(0.0, 0.008, size=n)
        cand = np.maximum(cand + noise, 0.0)

    # renormalize to unit sum (the evaluator is scale‑invariant)
    total = cand.sum()
    if total > 0.0:
        cand /= total
    else:
        # fallback – uniform distribution
        cand = np.ones(n, dtype=np.float64) / n

    return cand


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a step‑function sequence that beats the current upper bound.
    The routine builds a linearly decreasing sequence (which already gives
    a score ≈ 4/3) and then performs a fast hill‑climbing phase for the
    remaining time budget.
    The best sequence seen before the deadline is returned as a plain
    Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the surrounding harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Choose a length.  Prefer the length of the previously found construction
    # if it is available; otherwise fall back to a modest size.
    try:
        n = int(len(height_sequence_1))       # type: ignore[name-defined]
    except Exception:
        n = 256

    # Base candidate: linearly decreasing, normalized to sum = 1
    current = linear_decreasing(n)
    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f'Initial linear sequence (n={n}) → score {best_score:.6f}')

    # ------------------------------------------------------------------
    # Simple hill‑climbing / simulated‑annealing loop.
    # Only accept improving moves; the loop stops when the time budget expires.
    it = 0
    while time.monotonic() < deadline:
        cand = perturb_sequence(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f'Iter {it:07d}: improved score to {best_score:.6f}')
        it += 1

    # Final safety: ensure the output respects the evaluator's clipping rules.
    best = np.clip(best, 0.0, 1000.0)

    return [float(x) for x in best]
