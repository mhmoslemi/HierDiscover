"""
Step‑function optimizer.

The algorithm is a lightweight projected gradient descent on the
∞‑norm of the self‑convolution, combined with occasional random block
perturbations (the “make_block_perturbation” routine from the starter
code) to escape shallow local minima.

Key ideas
---------
* The evaluation 2·n·max(f*f)/(∑f)² is invariant under uniform scaling,
  therefore we keep the vector normalised to sum = 1 (except for the final
  clipping performed by ``evaluate_sequence``).
* For a given index *k* where the convolution attains its current maximum,
  the sub‑gradient of ``max(f*f)`` w.r.t. a_i is Σ_{j|i+j=k} a_j.
  This can be computed in O(n) by a simple slice operation.
* A step of the form  f ← f – η·g  (η = learning‑rate) is projected back
  onto the simplex (clip to non‑negative, renormalise).  The step reduces
  the maximal convolution entry while keeping the vector feasible.
* Every few iterations we apply a random block perturbation (multiplying a
  contiguous block by a log‑normal factor) and optionally enforce symmetry
  (f ← (f+f[::-1])/2).  This provides exploration and often improves the
  objective.
* The best sequence seen so far is returned after the allotted time.

The method runs only with NumPy and the already‑imported ``evaluate_sequence``.
"""

import time
import numpy as np


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project ``v`` onto the probability simplex (non‑negative, sum = 1)."""
    s = v.sum()
    if s <= 0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return np.clip(v, 0.0, None) / s


def make_block_perturbation(base, rng):
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # choose a block width between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal factor
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

    symmetrize_prob = 0.25  # probability to enforce symmetry after an update

    iteration = 0
    log_interval = 200

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) Gradient step aimed at the current maximal convolution index.
        # --------------------------------------------------------------
        conv = np.convolve(cur_seq, cur_seq)           # full convolution
        max_val = conv.max()
        # all indices achieving the maximum (sub‑gradient is sum over them)
        peak_idxs = np.where(conv >= max_val - 1e-12)[0]

        # Compute sub‑gradient g_i = Σ_{k in peaks} a_{k-i} (if in range)
        grad = np.zeros_like(cur_seq)
        n = cur_seq.size
        for k in peak_idxs:
            # i ranges where j = k - i is a valid index
            i_start = max(0, k - (n - 1))

            cur_score = cand_score
            no_improve_counter = 0

            )

        # 3) Learning‑rate schedule and possible restart.
        # --------------------------------------------------------------
        lr = max(lr_min, lr * lr_decay)

        if no_improve_counter >= restart_threshold:
            # Random restart: take the current best, perturb it a bit and resume.
            cur_seq = best_seq.copy()
            cur_seq = make_block_perturbation(cur_seq, rng)
            cur_seq = _normalize(cur_seq)
            cur_score = float(evaluate_sequence(cur_seq.tolist()))
            no_improve_counter = 0
            # Re‑inflate learning rate a little after a restart.
            lr = max(lr_min, lr * 2.0)

        # Periodic progress logging (does not affect performance much).
        if iteration % log_interval == 0:

    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations. Best score = {best_score:.9f}")
    return [float(x) for x in best_seq]
