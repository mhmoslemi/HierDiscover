def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # Initialize best with uniform:
    best = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    if best.ndim != 1 or best.size == 0 or not np.isfinite(best).all():
        best = np.ones(n, dtype=np.float64)
    best_score = evaluate_sequence(best.tolist())
    if not np.isfinite(best_score):
        best_score = float('inf')
    # Also optionally generate initial random candidate:
    # generate a random binary with p=0.5
    candidate = rng.integers(0, 2, size=n).astype(np.float64)
    if candidate.sum() < 0.01:
        candidate[0] = 1.0
    score = evaluate_sequence(candidate.tolist())
    if np.isfinite(score) and score < best_score:
        best = candidate
        best_score = score
    # Main loop
    while time.monotonic() < deadline:
        # Randomly decide to generate new random candidate or perturb best
        if rng.random() < 0.2:
            # fresh random
            candidate = rng.random(n)  # uniform [0,1)
            # optionally sparsify: set a proportion to 0
            mask = rng.random(n) < 0.5
            candidate = candidate * mask
        else:
            # perturb best using block scaling near peak.
            candidate = make_block_perturbation(best, rng)
        # Clip and ensure sum > 0.01
        candidate = np.clip(candidate, 0.0, 1000.0)
        # ensure sum not too small
        if candidate.sum() < 0.01:
            candidate[0] = 0.01 - candidate.sum()
        score = evaluate_sequence(candidate.tolist())
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"New best {best_score:.6f} at {time.monotonic():.2f}")
    return best.tolist()

"""
Hybrid local‑search + active‑set linear programming.

The evaluation 2·n·max(b)/(sum(a)^2) is invariant to scaling,
so we only need a shape of the step heights that flattens the
autoconvolution a*a.  A classical construction a_i ∝ 1/√(i+1) makes
the interior of the convolution almost constant, giving a score of
π/2 ≈ 1.571.  We start from this “inverse‑sqrt’’ shape and repeatedly
solve a tiny linear program:

    maximise   Σ g_j
    subject to (f * g)_k ≤ M   for the K largest convolution entries k
               g_j ≥ 0

where f is the current sequence, M = max(f*f) and K≈10.
The LP finds a new non‑negative sequence g that can carry more total
mass without increasing the current worst convolution entry.
After rescaling g to the same ℓ₁‑norm as f we blend
    f ← (1‑t)·f + t·g   (t≈0.2)
which strictly reduces the objective or leaves it unchanged.
If the LP fails or makes no progress we fall back to the original
block‑perturbation routine.  The loop runs until the time budget expires
and the best sequence found is returned.

The algorithm exploits the fact that only a handful of constraints are
active (those attaining the current maximum), so each LP is tiny
(≈10 constraints × n variables) and solves quickly even for n≈5000.
"""

import time
import numpy as np

# optional imports – the code works without SciPy (using only random perturbations)
try:
    import scipy.optimize
    import scipy.sparse
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False


def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                    # make it roughly symmetric sometimes
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def _build_sparse_A(f, active_idx):
    """Construct the sparse matrix A for (f * g)_k ≤ M,
    only for the rows indexed by ``active_idx`` (sorted)."""
    n = f.size
    m = len(active_idx)
    rows = []
    cols = []
    data = []

    # map each active k to its row number (0…m‑1)
    k_to_row = {k: i for i, k in enumerate(active_idx)}

    # for each i we place f[i] on the diagonal offset i within the selected rows
    for i, fi in enumerate(f):
        lo = i                     # smallest k that uses column j=0
        hi = i + n - 1             # largest k that uses column j=n‑1
        # find slice of active_idx that lies in [lo, hi]
        start = np.searchsorted(active_idx, lo, side='left')
        end = np.searchsorted(active_idx, hi, side='right')
        for pos in range(start, end):
            k = active_idx[pos]
            j = k - i               # column index of g
            rows.append(pos)        # row = position in active_idx
            cols.append(j)
            data.append(fi)

    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n), dtype=np.float64)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a sequence of non‑negative heights that (hopefully) beats the target.
    The routine mixes a deterministic “inverse‑sqrt’’ start, a tiny LP step
    that raises the ℓ₁‑norm under a fixed max‑convolution bound, and random
    block perturbations when the LP is not applicable.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --- initialise ---------------------------------------------------------
    # Length of the problem (use the provided reference sequence if present)
    n = int(len(height_sequence_1)) if 'height_sequence_1' in globals() else 256

    # Inverse‑sqrt shape (one‑sided) – this already gives ≈π/2 ≈ 1.57 score
    inv_sqrt = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    f = inv_sqrt / inv_sqrt.sum()          # normalised ℓ₁‑norm = 1

    # If the reference sequence exists and looks better, start from it
    if 'height_sequence_1' in globals():
        ref = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if ref.ndim == 1 and ref.size == n and np.isfinite(ref).all():
            ref_score = float(evaluate_sequence(ref.tolist()))
            cur_score = float(evaluate_sequence(f.tolist()))
            if ref_score < cur_score and np.isfinite(ref_score):
                f = ref.copy()
                cur_score = ref_score
        else:
            ref = None
    else:
        ref = None
        cur_score = float(evaluate_sequence(f.tolist()))

    best = f.copy()
    best_score = cur_score
    print(f"Initial score = {best_score:.6f}")

    # algorithm parameters
    mix_rate = 0.2          # t in the description
    K_active = 12           # number of constraints fed to the LP
    lp_tol = 1e-9           # tiny tolerance for linprog (default)

    while time.monotonic() < deadline:
        # ------------------------------------------------- LP improvement step
        if _HAS_SCIPY:
            # convolution with itself
            conv = np.convolve(f, f)
            M = conv.max()

            # choose the K largest indices (active set)
            if K_active < conv.size:
                top_idx = np.argpartition(-conv, K_active)[:K_active]
            else:
                top_idx = np.arange(conv.size)
            active_idx = np.sort(top_idx)  # required to be sorted for the builder

            # Build sparse A, RHS = M
            A = _build_sparse_A(f, active_idx)
            b_ub = np.full(A.shape[0], M, dtype=np.float64)

            # LP: maximise sum(g)  <=> minimise -sum(g)
            c = -np.ones(n, dtype=np.float64)

            bounds = [(0.0, None)] * n

            try:
                res = scipy.optimize.linprog(
                    c, A_ub=A, b_ub=b_ub, bounds=bounds,
                    method='highs',
                    options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())}
                )
            except Exception:   # pragma: no cover
                res = None

            if res is not None and res.success:
                g = res.x
                sum_f = f.sum()
                sum_g = g.sum()
                if sum_g > 0:
                    # rescale g to keep the ℓ₁‑norm unchanged
                    g = g * (sum_f / sum_g)

                    # convex combination
                    new_f = (1.0 - mix_rate) * f + mix_rate * g
                    new_f = np.clip(new_f, 0.0, 1000.0)

                    new_score = float(evaluate_sequence(new_f.tolist()))
                    if np.isfinite(new_score) and new_score < best_score:
                        best = new_f.copy()
                        best_score = new_score
                        print(f"LP step improved score to {best_score:.6f}  (active={len(active_idx)})")
                    # Accept the step as the new current point – it never hurts the bound
                    f = new_f
                    continue          # skip the random perturbation this iteration

        # ------------------------------------------------- fallback random perturbation
        cand = make_block_perturbation(f, rng)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            print(f"Random perturb improved score to {best_score:.6f}")

        # move to the new candidate regardless (helps exploration)
        f = cand

    # -----------------------------------------------------------------------
    print(f"Finished. Best score = {best_score:.6f}")
    return [float(x) for x in best]
