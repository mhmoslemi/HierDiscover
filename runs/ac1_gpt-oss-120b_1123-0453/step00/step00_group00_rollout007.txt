"""
Propose a candidate step-function using an LP‑based refinement.

The evaluation score (with a sum fixed to √(2n)) equals the maximal value of
the self‑convolution `a*a`.  Starting from a uniform distribution we repeatedly
solve a linear program

    maximise   sum(g)
    subject to   a * g  ≤  M    (component‑wise)
                g ≥ 0

where `M = max(a*a)`.  The LP yields a vector `g` that puts as much mass as
possible into positions where `a` is small while respecting the current
convolution bound.  By mixing the old sequence `a` with the new one

    a ← (1‑t)·a + t·g   (t is a small step size)

the maximal convolution value never increases – a classic “exchange”
iteration for the Kober–Schoenberg inequality.  After each mix we renormalise
the sum to √(2n) (the factor used in the original optimisation) and keep the
best sequence seen.

If, for any reason, the LP becomes numerically unstable we fall back to the
simple block‑perturbation local search from the starter code.

The routine runs until the allotted time budget is exhausted (≈1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Helper: a cheap local‑search perturbation (fallback when the LP fails)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a list of non‑negative heights (a step‑function) that minimises
    `evaluate_sequence`.  The method is an LP‑based iterative refinement
    described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for post‑processing / output
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Choose a working length.
    # If the global `height_sequence_1` exists we keep its length but never
    # exceed 5000 (to keep the LP tractable).  Otherwise fall back to 2000.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        if n > 5000:
            n = 5000
    except Exception:
        n = 2000

    # Normalised sum S = sqrt(2n).  With this normalisation the evaluation
    # score equals max(convolution) and scaling of the whole vector does not
    # change the score.
    S = math.sqrt(2.0 * n)

    # Start from a uniform distribution – the obvious feasible point.
    a = np.full(n, S / n, dtype=np.float64)

    # Keep the best sequence seen so far.
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    iteration = 0
    # Candidate step sizes – smaller values give finer adjustments.
    step_candidates = (0.2, 0.1, 0.05, 0.02, 0.01)

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1) compute current convolution bound M
        # --------------------------------------------------------------
        conv = np.convolve(a, a)
        M = conv.max()

        # --------------------------------------------------------------
        # 2) Build the convolution matrix C (size (2n‑1) × n) where
        #    C[k, j] = a[k‑j]  if 0 ≤ k‑j < n, else 0.
        #    This matrix expresses (a * g)[k] = C[k, :]·g.
        # --------------------------------------------------------------
        # Each diagonal offset –d (d = 0 … n‑1) is a constant vector
        # filled with a[d].  Using `diags` creates a CSR matrix with
        # exactly n·n non‑zeros (≈23 M for n=4800) – still acceptable.
        data = [np.full(n, val, dtype=np.float64) for val in a]
        offsets = [-d for d in range(n)]
        C = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
        del data  # free temporary list early

        # --------------------------------------------------------------
        # 3) Linear programme: maximise sum(g)  ⇔ minimise –sum(g)
        # --------------------------------------------------------------
        c = np.full(n, -1.0, dtype=np.float64)          # objective: -∑g
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # upper bound M for each row
        bounds = [(0.0, None)] * n                       # g_j ≥ 0

        # use HiGHS (default) – fast and reliable for medium‑sized LPs
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'disp': False})

        # free the convolution matrix – it is the biggest temporary object
        del C

        if not res.success:
            # LP failed (rare).  Do a cheap random move and continue.
            cand = make_block_perturbation(a, rng)
            scr = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(scr) and scr < best_score:
                best_seq = cand.copy()
                best_score = scr
            a = cand
            continue

        g = res.x
        # --------------------------------------------------------------
        # 4) Rescale g to have the required sum √(2n) (the algorithm in the
        #    reference normalises every new vector that way).
        # --------------------------------------------------------------
        g_sum = g.sum()
        if g_sum > 0:
            g *= (S / g_sum)

        # --------------------------------------------------------------
        # 5) Try a few mixing factors and keep the best improvement.
        # --------------------------------------------------------------
        improved = False
        for t in step_candidates:
            a_new = (1.0 - t) * a + t * g
            # renormalise to avoid tiny drift caused by floating point errors
            a_new *= (S / a_new.sum())
            scr = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(scr) and scr < best_score:
                best_seq = a_new.copy()
                best_score = scr
                a = a_new
                improved = True
                break

        if not improved:
            # If no mixing improved the score, try a small random perturbation.
            cand = make_block_perturbation(a, rng)
            scr = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(scr) and scr < best_score:
                best_seq = cand.copy()
                best_score = scr
            a = cand

        iteration += 1
        if iteration % 5 == 0:
            print(f"Iter {iteration:4d}: best_score={best_score:.6f}, n={n}")

    # final sanity: make sure we return a plain Python list of floats
    return [float(x) for x in best_seq]
