"""...

import numpy as np, time, math

def conv_self(a):
    n = a.shape[0]
    size = 1 << int(np.ceil(np.log2(2*n - 1)))
    # compute FFT for real input
    fa = np.fft.rfft(a, size)
    # elementwise product
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_score(a, n=None):
    if n is None:
        n = a.shape[0]
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = conv_self(a)
    max_b = b.max()
    return float(2.0 * n * max_b / (S * S))

def compute_gradient(k_max, a):
    n = a.shape[0]
    idx = k_max - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    grad = np.zeros_like(a)
    grad[valid] = 2.0 * a[idx[valid]]
    diag = valid & (idx == np.arange(n))
    grad[diag] = a[diag]  # correct diag case
    return grad

def block_perturb(a, rng):
    n = a.shape[0]
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    b = a.copy()
    b[start:stop] *= factor
    if rng.random() < 0.35:
        b = 0.5 * (b + b[::-1])
    np.clip(b, 0.0, 1000.0, out=b)
    return b

def transfer_candidate(a, grad, rng):
    n = a.shape[0]
    # Number of extreme indices to consider (1-2% of n)
    top_k = max(1, int(0.02 * n))
    # Order indices by gradient
    order = np.argsort(grad)
    low_idx = order[:top_k]
    high_idx = order[-top_k:]
    hi = int(rng.choice(high_idx))
    lo = int(rng.choice(low_idx))
    # ensure distinct
    if hi == lo:
        # pick another low index
        lo_candidates = low_idx[low_idx != hi]
        if lo_candidates.size == 0:
            return a.copy()
        lo = int(rng.choice(lo_candidates))
    # Determine amount to move; use a fraction of the source value
    amount = float(rng.uniform(0.03, 0.12) * a[hi])
    # protect from reducing below zero
    amount = min(amount, a[hi])
    b = a.copy()
    b[hi] -= amount
    b[lo] += amount
    # occasional symmetrization
    if rng.random() < 0.03:
        b = 0.5 * (b + b[::-1])
    np.clip(b, 0.0, 1000.0, out=b)
    return b

    """Simulated‑annealing search that mixes random block perturbations with
    a gradient‑guided transfer of mass between the most and least damaging
    positions (those that contribute most to the current peak of the
    self‑convolution).  The algorithm keeps the best sequence seen within
    the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # initialise from the provided global construction if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    # enforce the bounds required by evaluate_sequence
    np.clip(parent, 0.0, 1000.0, out=parent)

    n = parent.size
    # current solution and best known solution
    current = parent.copy()
    current_score = compute_score(current, n)
    best = current.copy()
    best_score = current_score

    # main loop
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # decide move type
        if rng.random() < 0.75:
            # gradient‑guided mass transfer
            b = conv_self(current)
            k_max = int(np.argmax(b))
            grad = compute_gradient(k_max, current)
            candidate = transfer_candidate(current, grad, rng)
        else:
            # broader random perturbation
            candidate = block_perturb(current, rng)

        # quick evaluation (no call to evaluate_sequence)
        cand_score = compute_score(candidate, n)

        # simple hill‑climbing acceptance with a tiny simulated‑annealing component
        if cand_score < current_score:
            current = candidate
            current_score = cand_score
            if cand_score < best_score:
                best = candidate.copy()
                best_score = cand_score
        else:
            # temperature decays linearly to near zero
            remaining = deadline - time.monotonic()
            total = deadline - start_time
            T = max(1e-4, remaining / total) * 0.05   # a small scale
            if rng.random() < math.exp(-(cand_score - current_score) / T):
                current = candidate
                current_score = cand_score

        # occasional progress report (once per second)
        if iteration % 200 == 0:
            elapsed = time.monotonic() - start_time
            print(f"Iter {iteration:6d} | time {elapsed:5.1f}s | best {best_score:.6f}")

    # final report
    print(f"Finished after {time.monotonic() - start_time:.1f}s, best score {best_score:.6f}")
    # convert to plain Python list of floats as required
    return [float(x) for x in best]

"""
Algorithm:   *Initial guess* – a power‑law decreasing sequence a_i = (i+1)^(-0.5)
               (this shape gives a near‑constant self‑convolution and a
                score ≈ π/2 ≈ 1.57, already far below the naïve uniform bound 2).

            *Local optimisation* – a simulated‑annealing / hill‑climbing loop
               that repeatedly
                – evaluates the full self‑convolution,
                – finds the offset k where the convolution is maximal,
                – computes the sub‑gradient of that peak (how each coefficient
                  contributes to the peak),
                – moves a small amount of mass from the “high‑gradient”
                  coefficients to the “low‑gradient” ones (keeping the total
                  sum constant), and optionally applies a random block
                  perturbation.
               The best sequence seen within the time budget is returned.

            The score is scale‑invariant, therefore we never need to rescale
            the sequence – we only keep the values within the allowed interval
            [0, 1000].

            The implementation uses FFT‑based convolution (O(n log n)) which
            is fast enough to evaluate thousands of candidates within the
            typical 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << int(np.ceil(np.log2(2 * n - 1)))   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray, n: int) -> float:
    """Score = 2*n*max(conv)/sum(a)^2 (exactly the evaluate_sequence formula)."""
    s = a.sum()
    if s < 0.01:                     # protect against division by zero
        return float("inf")
    b = conv_self(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_gradient(k_max: int, a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the peak b[k_max] = Σ a_i a_{k_max-i}
    w.r.t. a.  For each i:
        grad[i] = 2*a[k_max-i]  (if the paired index exists)
        but when i == k_max-i the coefficient appears only once,
        so grad[i] = a[i] in that case.
    """
    n = a.shape[0]
    idx = k_max - np.arange(n)                # partner index for each i
    valid = (idx >= 0) & (idx < n)

    grad = np.zeros_like(a)
    grad[valid] = 2.0 * a[idx[valid]]
    diag = valid & (idx == np.arange(n))
    grad[diag] = a[diag]                       # correct the double‑count on diagonal
    return grad


def block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """A simple O(n) random block‑scale perturbation (similar to the reference)."""
    n = a.shape[0]
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    b = a.copy()
    b[start : start + width] *= factor
    if rng.random() < 0.35:                 # occasional symmetrisation
        b = 0.5 * (b + b[::-1])
    np.clip(b, 0.0, 1000.0, out=b)
    return b


def transfer_candidate(a: np.ndarray, grad: np.ndarray,
                       rng: np.random.Generator) -> np.ndarray:
    """
    Move a small amount of mass from a high‑gradient index to a low‑gradient
    index (keeping the total sum unchanged).  The amount transferred is a
    random fraction of the source value.
    """
    n = a.shape[0]
    top_k = max(1, int(0.015 * n))           # 1–2 % of the indices
    order = np.argsort(grad)

    low_idx = order[:top_k]
    high_idx = order[-top_k:]

    hi = int(rng.choice(high_idx))
    lo = int(rng.choice(low_idx))

    if hi == lo:                             # extremely unlikely, but safe
        return a.copy()

    # transfer a random portion (3–12 %) of the source mass
    portion = float(rng.uniform(0.03, 0.12))
    amount = portion * a[hi]
    amount = min(amount, a[hi])

    b = a.copy()
    b[hi] -= amount
    b[lo] += amount

    # occasional tiny symmetrisation to keep the shape smooth
    if rng.random() < 0.03:
        b = 0.5 * (b + b[::-1])

    np.clip(b, 0.0, 1000.0, out=b)
    return b


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Build a candidate step‑function (list of non‑negative heights) that
    minimises the provided evaluation function.  The method uses a
    power‑law initialization followed by a fast simulated‑annealing /
    hill‑climbing optimisation.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # a safety margin

    # ------------------------------------------------------------------
    # Determine the length n.  Prefer the length of the provided construction
    # (height_sequence_1) but fall back to a modest default.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)

    n = base.size
    np.clip(base, 0.0, 1000.0, out=base)

    # ------------------------------------------------------------------
    # 1)  Choose an initial shape from the family a_i = (i+1)^(-p)
    #     Scan a few exponents around p=0.5 and keep the best start.
    best_start = None
    best_score = float("inf")
    for p in np.linspace(0.40, 0.60, 21):
        a = (np.arange(n, dtype=np.float64) + 1.0) ** (-p)
        np.clip(a, 0.0, 1000.0, out=a)
        sc = compute_score(a, n)
        if sc < best_score:
            best_score = sc
            best_start = a.copy()
    # keep the best of the small grid as the current solution
    current = best_start
    current_score = best_score

    # also keep a global best (may improve later)
    best = current.copy()
    best_score = current_score

    # ------------------------------------------------------------------
    # 2)  Main optimisation loop
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # a) evaluate the current convolution and locate the peak
        b = conv_self(current)
        max_b = b.max()
        k_max = int(b.argmax())

        # b) compute the sub‑gradient of that peak
        grad = compute_gradient(k_max, current)

        # c) generate a neighbour (mostly a directed transfer, sometimes a block)
        if rng.random() < 0.75:
            cand = transfer_candidate(current, grad, rng)
        else:
            cand = block_perturb(current, rng)

        # d) evaluate the neighbour
        cand_score = compute_score(cand, n)

        # e) acceptance rule (hill‑climbing with a tiny temperature schedule)
        improved = cand_score < current_score
        if not improved:
            # linear temperature that vanishes at the end of the budget
            remaining = deadline - time.monotonic()
            total = deadline - start
            temperature = max(1e-6, remaining / total) * 0.07
            prob = math.exp(-(cand_score - current_score) / temperature)
            improved = rng.random() < prob

        if improved:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # occasional progress report (once per ~200 iterations)
        if iteration % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"Iter {iteration:6d} | time {elapsed:5.1f}s | "
                  f"best {best_score:.6f} | cur {current_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Search finished after {time.monotonic() - start:.1f}s "
          f"with best score {best_score:.6f}")
    return [float(x) for x in best]
