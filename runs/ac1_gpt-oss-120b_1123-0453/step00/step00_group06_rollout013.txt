import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """...
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Initialize sequence length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = base.size
    except Exception:
        n = 2000

    # Initial random positive sequence
    seq = rng.random(n)  # uniform [0,1)
    # Scale so max close to 500 (to give headroom for clipping)
    seq = seq * 500.0
    seq = np.clip(seq, 0.0, 1000.0)

    # ensure sum >0.01
    if seq.sum() < 0.01:
        seq += 0.01

    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    current_seq = seq.copy()
    current_score = best_score

    # Parameters
    lr0 = 0.001   # learning rate
    decay = 0.9999
    lr = lr0
    iter_count = 0
    last_report = time.monotonic()
    report_interval = 10.0

    while time.monotonic() < deadline:
        # compute self convolution
        conv = np.convolve(current_seq, current_seq)
        # target constant (average of convolution)
        total = current_seq.sum()
        target = total * total / (2 * n - 1)

        # gradient of squared deviation loss (approx)
        diff = conv - target
        # gradient (4 * conv(diff, seq[::-1]) slice)
        grad_full = np.convolve(diff, current_seq[::-1])
        grad = 4.0 * grad_full[n-1:n-1+n]

        # gradient step
        candidate = current_seq - lr * grad
        # project onto feasible domain
        candidate = np.clip(candidate, 0.0, 1000.0)

        # evaluate
        cand_score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(cand_score) and cand_score < current_score:
            current_seq = candidate
            current_score = cand_score
            if cand_score < best_score:
                best_seq = candidate
                best_score = cand_score
        else:
            # occasionally accept a worse move to inject diversity
            if rng.random() < 0.001:
                current_seq = candidate
                current_score = cand_score

        # occasionally apply a random block perturbation (like baseline)
        if iter_count % 200 == 0:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            block = current_seq[start:stop] * np.exp(rng.normal(0.0, 0.12))
            current_seq[start:stop] = np.clip(block, 0.0, 1000.0)

        iter_count += 1
        lr *= decay  # slowly reduce learning rate

        # periodic reporting
        if time.monotonic() - last_report > report_interval:
            print(f'Iter {iter_count}, best {best_score:.6f}, current {current_score:.6f}, lr {lr:.6e}')
            last_report = time.monotonic()

    print('Finished search, best score:', best_score)
    return [float(x) for x in best_seq]

"""
Gradient‑descent based search for a step‑function that minimises
evaluate_sequence().  The algorithm
    * starts from a random non‑negative vector (or the provided
      `height_sequence_1`);
    * repeatedly evaluates its self‑convolution;
    * computes the deviation of the convolution from the flat target
      (the average value of the convolution);
    * takes a small projected gradient step that reduces this deviation;
    * occasionally injects a random block‑perturbation (the baseline
      strategy) to escape plateaus;
    * keeps the best sequence seen so far.

The objective 2·n·max(b)/(sum(a)²) is scale‑invariant, therefore the search
focuses only on the *shape* of the vector, not on its absolute magnitude.
The code respects all limits (non‑negative, ≤1000) and returns the best
sequence before the time budget expires.
"""

import time
import numpy as np


def _initial_sequence(rng: np.random.Generator, n: int) -> np.ndarray:
    """Return a feasible starting sequence of length n."""
    # try to reuse the provided sequence, otherwise random initialise
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size == n and np.isfinite(base).all():
            seq = base.copy()
        else:
            raise ValueError
    except Exception:
        # random positive values, scaled to leave headroom for the optimiser
        seq = rng.random(n) * 500.0
    seq = np.clip(seq, 0.0, 1000.0)
    # make sure we have a non‑trivial sum
    if seq.sum() < 0.01:
        seq += 0.01
    return seq


def _flat_target(seq: np.ndarray) -> float:
    """Desired (flat) value of the convolution: the average of all entries."""
    total = seq.sum()
    return (total * total) / (2 * seq.size - 1)


def _gradient(seq: np.ndarray, target: float) -> np.ndarray:
    """
    Gradient of the squared‑error loss
        L = Σ_k ( (a*a)[k] – target )²
    with respect to the vector a.
    The analytic form is
        ∂L/∂a_i = 4 * Σ_j ( (a*a)[i+j] – target ) * a_j .
    This can be computed with a single convolution.
    """
    conv = np.convolve(seq, seq)          # (a*a)  length 2n‑1
    diff = conv - target                  # deviation from the flat target
    # convolution of diff with the reversed a gives the summed term
    grad_full = np.convolve(diff, seq[::-1])
    n = seq.size
    # slice the part that corresponds to ∂L/∂a_i  (see analysis notes)
    grad = 4.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence().  The function runs for roughly `budget_s` seconds
    (leaving a tiny safety margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------
    # initialise
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size if base.ndim == 1 and base.size > 0 else 2000
    except Exception:
        n = 2000

    seq = _initial_sequence(rng, n)

    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    current_seq = seq.copy()
    current_score = best_score

    # learning‑rate schedule
    lr = 1e-3
    decay = 0.9999

    iteration = 0
    last_report = time.monotonic()
    REPORT_EVERY = 5.0  # seconds

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- compute convolution and the flat target -----
        target = _flat_target(current_seq)

        # ----- gradient step -----
        grad = _gradient(current_seq, target)

        # simple projected gradient update
        candidate = current_seq - lr * grad
        candidate = np.clip(candidate, 0.0, 1000.0)

        # ----- occasionally perform a random block perturbation -----
        if iteration % 250 == 0:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            block = candidate[start:stop] * np.exp(rng.normal(0.0, 0.12))
            candidate[start:stop] = np.clip(block, 0.0, 1000.0)

        # ----- evaluate -----
        cand_score = float(evaluate_sequence(candidate.tolist()))

        # accept if strictly better; otherwise keep the old vector
        if np.isfinite(cand_score) and cand_score < current_score:
            current_seq = candidate
            current_score = cand_score
            if cand_score < best_score:
                best_seq = candidate
                best_score = cand_score
        else:
            # small probability to accept a worse move (helps escape local minima)
            if rng.random() < 0.001:
                current_seq = candidate
                current_score = cand_score

        # ----- learning‑rate decay and bookkeeping -----
        lr *= decay
        iteration += 1

        # ----- periodic progress report -----
        now = time.monotonic()
        if now - last_report > REPORT_EVERY:
            print(
                f"Iter {iteration:6d} | best {best_score:.6f} | "
                f"current {current_score:.6f} | lr {lr:.2e}"
            )
            last_report = now

    # final report
    print(f"Search finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(x) for x in best_seq]
