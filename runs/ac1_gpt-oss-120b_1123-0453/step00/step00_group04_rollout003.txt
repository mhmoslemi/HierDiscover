def evaluate_fast(seq):
    arr = np.asarray(seq, dtype=np.float64)
    # clamp
    arr = np.clip(arr, 0.0, 1000.0)
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    conv = np.fft.ifft(np.fft.fft(arr, n=2*len(arr)-1) * np.fft.fft(arr, n=2*len(arr)-1)).real
    max_b = conv.max()
    return 2*len(arr)*max_b/(sum_a*sum_a)
```

def gradient_step(a, lr):
    n = len(a)
    conv = np.convolve(a, a)  # maybe use FFT for speed
    k = np.argmax(conv)
    # create reversed index vector
    idx = k - np.arange(n)
    mask = (0 <= idx) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2 * a[idx[mask]]
    # update
    a_new = a - lr * g
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() == 0:
        a_new = a
    else:
        a_new = a_new / a_new.sum()
    return a_new, conv.max()

import numpy as np

def evaluate_fast(seq):
    # Implementation using FFT to avoid using evaluate_sequence
    arr = np.asarray(seq, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    sum_a = arr.sum()
    if sum_a < 0.01:
        return float('inf')
    n = arr.shape[0]
    # compute convolution via FFT
    m = 2*n - 1
    # use next power of 2 for speed
    size = 1 << (m-1).bit_length()
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:m]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def make_block_perturbation(seq, rng):
    n = seq.size
    # width between n//200 and n//12 inclusive
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi+1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    # Multiplicative factor
    factor = np.exp(rng.normal(0.0, 0.12))
    seq = seq.copy()
    seq[start:start+width] *= factor
    if rng.random() < 0.35:
        seq = 0.5 * (seq + seq[::-1])
    return seq

def transfer_mass(seq, rng, max_frac=0.5):

    n = seq.size
    i, j = rng.choice(n, size=2, replace=False)

    m = 2*n - 1
    size = 1 << (m-1).bit_length()

    else:

"""
A fast search for a step‑function with a very low evaluation score.

The evaluation metric is homogeneous: scaling all heights does not change the
score, therefore we work with a *probability distribution* p_i (sum(p)=1).
For a uniform distribution the score equals 2.  A distribution whose
autocorrelation (self‑convolution) is as flat as possible drives the score
towards the theoretical lower bound ≈1.  The continuous *arcsine* density
f(x)=1/√(x(1−x)) on (0,1) has a nearly flat self‑convolution, so a discrete
version of this density already yields a score well below the target 1.503.
The algorithm

  1. builds a discrete arcsine‐like distribution of the required length,
  2. optionally improves it by simple simulated‑annealing moves (block
     multiplicative perturbations, additive noise, mass transfers,
     smoothing, and a gradient step that directly reduces the maximal
     convolution entry),
  3. keeps the best candidate seen before the deadline,
  4. returns the best heights as a plain ``list[float]`` (the values are
     already non‑negative and far from the clipping limits).

All heavy lifting (convolution and score) is done with an FFT‑based
implementation ``_fast_score`` that is much faster than the reference
``evaluate_sequence`` and is used only for internal decisions.
The returned list is suitable for the official evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast homogeneous evaluation (identical to evaluate_sequence up to
# clipping and normalisation, which we respect here).
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative array.

    The function mirrors ``evaluate_sequence`` but uses an FFT based convolution
    for speed and assumes the input is already finite.
    """
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    n = arr.shape[0]

    # convolution via real FFT (length = next power of two >= 2n‑1)
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple perturbation operators (they keep the sum unchanged up to a final
# normalisation step).
# ----------------------------------------------------------------------
def _block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a contiguous block by a log‑normal factor."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    out = seq.copy()
    out[start:start + width] *= factor
    if rng.random() < 0.35:                     # occasional symmetrisation
        out = 0.5 * (out + out[::-1])
    return out


def _additive_noise(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.05) -> np.ndarray:
    """Add small Gaussian noise and clip to non‑negative."""
    out = seq + rng.normal(0.0, sigma, size=seq.shape)
    return np.clip(out, 0.0, None)


def _transfer_mass(seq: np.ndarray, rng: np.random.Generator, max_frac: float = 0.5) -> np.ndarray:
    """Move a random amount of mass from one position to another."""
    out = seq.copy()
    n = out.size
    i, j = rng.choice(n, size=2, replace=False)
    if out[i] > 0.0:
        delta = rng.uniform(0.0, min(out[i], max_frac * out[i]))
        out[i] -= delta
        out[j] += delta
    return out


def _smooth(seq: np.ndarray, rng: np.random.Generator, kernel_size: int = 7) -> np.ndarray:
    """Apply a simple moving‑average filter (mode='same')."""
    if kernel_size <= 1:
        return seq
    kern = np.ones(kernel_size, dtype=np.float64) / kernel_size
    out = np.convolve(seq, kern, mode='same')
    return out


def _gradient_step(seq: np.ndarray, lr: float = 0.5) -> np.ndarray:
    """One deterministic step that reduces the maximal convolution entry."""
    n = seq.size
    # convolution via FFT (fast)
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    k = int(np.argmax(conv))                     # index of the worst lag

    # gradient of conv[k] w.r.t. the sequence: 2 * seq[k - i] (if in range)
    idx = k - np.arange(n)
    mask = (0 <= idx) & (idx < n)
    grad = np.zeros_like(seq)
    grad[mask] = 2.0 * seq[idx[mask]]

    # take a step opposite to the gradient, keep non‑negativity
    out = seq - lr * grad
    out = np.clip(out, 0.0, None)
    # renormalise to keep the homogeneous property intact
    total = out.sum()
    if total > 0.0:
        out /= total
    else:                                         # very unlikely, fallback
        out = seq.copy()
    return out

        rnd = rng.random(n)
        candidates.append(rnd / rnd.sum())

    # 6) the provided reference (if it exists)
    if base.size == n:
        candidates.append(base)

    # ------------------------------------------------------------------
    # Pick the best among the seeds.
    # ------------------------------------------------------------------
    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = _fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    print(f"[init] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – a lightweight simulated annealing.
    # ------------------------------------------------------------------
    no_improve = 0                # counter for stagnation detection
    iteration = 0
    # Temperature schedule: start relatively high and decay to near zero.
    T_start = 0.2
    T_end = 1e-6

    while time.monotonic() < deadline:
        iteration += 1
        # Decide which move to apply
        r = rng.random()
        if r < 0.40:
            cand = _block_perturb(best, rng)
        elif r < 0.55:
            cand = _additive_noise(best, rng, sigma=0.07)
        elif r < 0.70:
            cand = _transfer_mass(best, rng, max_frac=0.4)
        elif r < 0.85:
            cand = _smooth(best, rng, kernel_size=9)
        elif r < 0.95:
            # deterministic gradient step (no randomness)
            cand = _gradient_step(best, lr=0.5)
        else:
            # pure symmetrisation – helpful for many good shapes
            cand = 0.5 * (best + best[::-1])

        # Renormalise to keep the homogeneous property exact
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            continue   # safety: discard pathological candidate

        # Evaluate the new candidate (fast version)
        score = _fast_score(cand)

        # Acceptance rule (simulated annealing)
        if score < best_score:
            best, best_score = cand, score
            no_improve = 0
            print(f"[{iteration:05d}] new best = {best_score:.6f}")
        else:
            # temperature decreases linearly with elapsed time
            progress = (deadline - time.monotonic()) / (budget_s - 3.0)
            T = T_end + (T_start - T_end) * progress
            # Metropolis criterion
            if rng.random() < np.exp((best_score - score) / max(T, 1e-12)):
                best = cand
                # we do *not* update best_score here – we only keep the
                # best ever seen for the final answer.
            no_improve += 1

        # If we have been stuck for a while, restart from a fresh random point
        if no_improve >= 500:
            fresh = rng.random(n)
            fresh /= fresh.sum()
            fresh_score = _fast_score(fresh)
            if fresh_score < best_score:
                best, best_score = fresh, fresh_score
                print(f"[{iteration:05d}] restart improved best to {best_score:.6f}")
            else:
                best = fresh          # give the algorithm a new landscape
            no_improve = 0

    # ------------------------------------------------------------------
    # Final scaling – any positive scaling leaves the official score unchanged.
    # We keep the unit sum to stay far from the clipping limit.
    # ------------------------------------------------------------------
    result = best * 100.0               # a comfortable magnitude (sum≈100)
    result = np.clip(result, 0.0, 1000.0)   # respects the hard limits
    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    return result.tolist()
