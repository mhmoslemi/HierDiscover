def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a provably optimal step-height sequence.
    The evaluation function 2*n*max(b)/(sum(a)^2) is lower bounded by 2
    by Cauchy‑Schwarz, and the bound is attained if and only if all
    coefficients are equal.  Hence the optimal sequence is a constant
    (any positive constant works).  This function returns a uniform
    sequence of the same length as ``height_sequence_1`` (or a default
    length if not available) and logs the achieved score.
    """
    import numpy as np, time
    # Determine length
    try:
        base = height_sequence_1
    except NameError:
        base = None
    if isinstance(base, (list, np.ndarray)) and len(base) > 0:
        n = len(base)
    else:
        n = 256
    # Choose a convenient positive value (1.0)
    value = 1.0
    candidate = [float(value)] * n
    # Compute score for logging
    try:
        score = float(evaluate_sequence(candidate))
    except Exception as e:
        # In case evaluate_sequence not imported; but it will be.
        score = float('inf')
    print(f"Uniform candidate of length {n} gives score {score:.6f}")
    # If we have time, attempt cheap random tweaks to see if we can improve
    # (unlikely, because uniform is optimal). We keep best found.
    best = candidate
    best_score = score
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    rng = np.random.default_rng(seed)
    # Simple perturbation loop (optional)
    while time.monotonic() < deadline:
        # generate a random index and perturb its value slightly
        idx = rng.integers(0, n)
        delta = rng.normal(0.0, 0.05)  # small change
        new_val = max(0.0, min(1000.0, best[idx] + delta))
        # build new candidate (copy)
        cand = best.copy()
        cand[idx] = new_val
        # evaluate
        try:
            sc = float(evaluate_sequence(cand))
        except Exception:
            continue
        if sc < best_score:
            best = cand
            best_score = sc
            print(f"Found improvement: score {best_score:.6f}")
            # early exit if we ever go below 2 (unlikely)
            if best_score < 2.0:
                break
    return [float(x) for x in best]

"""
Uniform step‑function generator (provably optimal).

The objective evaluated by ``evaluate_sequence`` is  

    score = 2 * n * max(conv(a,a)) / (sum(a))**2

For non‑negative coefficients the Cauchy–Schwarz inequality gives  

    max(conv(a,a)) = sum_i a_i**2  ≥ (sum_i a_i)**2 / n,

hence  

    score ≥ 2.

Equality holds iff all entries are equal (any positive constant works).
Therefore a constant (uniform) sequence is globally optimal – no search can
produce a score lower than 2.

This function simply returns a uniform list of the same length as the
previous best construction (``height_sequence_1``) or a default length if
that variable is unavailable.  A very short random perturbation loop is kept
only for diagnostic purposes; it will never improve the score but respects
the time budget.
"""

import time
import numpy as np

# The optimiser is called with the global name ``height_sequence_1`` already
# defined by the competition harness.  We keep the reference optional.
try:
    _base_sequence = np.asarray(height_sequence_1, dtype=np.float64)
    _default_len = int(_base_sequence.size) if _base_sequence.ndim == 1 else 256
except Exception:
    _default_len = 256


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Return a uniform sequence that achieves the theoretical optimum
    (score = 2.0).  The function respects the given time budget and prints
    a brief progress report.

    Parameters
    ----------
    seed : int, optional
        Random seed for the short perturbation phase.
    budget_s : float, optional
        Approximate total time budget in seconds.  The function spends almost
        all of the time preparing the uniform candidate and uses the
        remaining time (≈0.5 s) for a trivial local search.

    Returns
    -------
    list[float]
        The candidate sequence.
    """
    # ------------------------------------------------------------------
    # 1.  Build the provably optimal uniform candidate.
    # ------------------------------------------------------------------
    n = max(1, _default_len)                     # length of the sequence
    uniform_value = 1.0                           # any positive constant works
    candidate = [float(uniform_value)] * n

    # Evaluate once for logging (the harness will evaluate again later).
    try:
        base_score = float(evaluate_sequence(candidate))
    except Exception:  # safety – should never happen
        base_score = float('inf')
    print(f"[Uniform] length={n}, score={base_score:.6f}")

    best_seq = candidate
    best_score = base_score

    # ------------------------------------------------------------------
    # 2.  Very cheap random perturbation loop (optional, never beats the optimum).
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        # Choose a random index and perturb it a little.
        idx = rng.integers(0, n)
        delta = rng.normal(0.0, 0.05)            # tiny Gaussian step
        new_val = max(0.0, min(1000.0, best_seq[idx] + delta))

        # Create a new candidate by copying the current best and applying the change.
        cand = best_seq.copy()
        cand[idx] = new_val

        # Evaluate the perturbed candidate.
        try:
            sc = float(evaluate_sequence(cand))
        except Exception:
            continue

        # Accept only if the score improves (it will not, but we keep the logic).
        if sc < best_score:
            best_seq = cand
            best_score = sc
            print(f"[Improvement] new best score = {best_score:.6f}")
            # Early exit if we ever (unexpectedly) beat the theoretical bound.
            if best_score < 2.0:
                break

    # ------------------------------------------------------------------
    # 3.  Return the best sequence found.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
