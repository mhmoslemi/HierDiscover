def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Generate a nonnegative step-function sequence aiming to minimize max convolution.
    Uses an exponential decay profile derived from analytic approximation of the optimal shape.
    """
    import numpy as np, time, math

    # Determine sequence length
    try:
        n = len(height_sequence_1)
        base_len = n
    except Exception:
        base_len = 2000
        n = base_len

    # solve for alpha where e^alpha - 1 = 2 alpha
    # Use Newton method with initial guess 1.3
    alpha = 1.3
    for _ in range(20):
        f = math.exp(alpha) - 1 - 2 * alpha
        df = math.exp(alpha) - 2
        alpha = alpha - f / df
    # compute r = exp(-alpha/n)
    r = math.exp(-alpha / n)
    diff = 1 - r
    # generate sequence
    idx = np.arange(n, dtype=np.float64)
    seq = diff * np.power(r, idx)
    # clip to ensure nonnegativity and bound
    seq = np.clip(seq, 0.0, 1000.0)
    # Evaluate occasionally for debugging
    try:
        score = evaluate_sequence(seq.tolist())
        print(f"Generated exponential sequence length {n}, alpha={alpha:.6f}, score={score:.6f}")
    except Exception as e:
        print("Evaluation error:", e)

    # optional local refinement: short random hill climbing within remaining time
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    rng = np.random.default_rng(seed + 1)
    while time.monotonic() < deadline:
        # propose by scaling a random contiguous block
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 100))
        stop = min(n, start + width)
        factor = math.exp(rng.normal(0.0, 0.08))
        cand = best_seq.copy()
        cand[start:stop] *= factor
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_seq = cand
            best_score = score
            print(f"Improved score to {best_score:.6f} at time {time.monotonic():.2f}")
    return [float(x) for x in best_seq]

def propose_candidate(...):
    n = len(height_sequence_1) or default.
    best_seq = None, best_score = inf.

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)

    while time.monotonic() < deadline:
        p = rng.uniform(0.3, 0.7) 
        delta = rng.uniform(0.0, 2.0) # shift
        idx = np.arange(n, dtype=np.float64)
        seq = (idx + delta) ** (-p)
        # ensure not infinite at idx+delta=0; delta>0.
        # set values for all terms when idx+delta >0.
        # Normalize:
        total = seq.sum()
        if total == 0:
            continue
        seq = seq / total
        # Evaluate via evaluate_sequence (or quick compute).
        try:
            score = float(evaluate_sequence(seq.tolist()))
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best score {best_score:.6f} with p={p:.4f}, delta={delta:.4f}")
    if best_seq is None:
        # fallback: uniform
        best_seq = np.ones(n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("Fallback uniform", best_score)

    return [float(x) for x in best_seq]

"""
Propose a candidate step‑function sequence.

The core idea is to start from the *arcsine* shape
    a_i ∝ 1/√(i+1)                     (i = 0,…,n‑1)
which is the discrete analogue of the continuous function
    f(t)=1/√t   on [0,1].
For this shape the self‑convolution is (asymptotically) constant,
so the ratio 2·n·max(b)/(∑a)² approaches π/2 ≈ 1.57 – well
below the naïve uniform bound (2.0).  From this analytically motivated
starting point we run a cheap stochastic local‑search (block
perturbations) to push the score a little further while staying
within the 1000 s time budget.

The function respects all contest rules:
* no closures, no lambdas,
* only top‑level helpers,
* uses only numpy and the already‑imported evaluate_sequence,
* prints progress for debugging.
"""
import math
import time
import numpy as np

def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # block width is between n/200 and n/12 (as in the reference code)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal jitter
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring – it does not hurt the arcsine shape
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce the global bounds required by evaluate_sequence
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a sequence of non‑negative heights that (hopefully) improves
    on the current bound.  The algorithm:

    1.  Choose a length `n`.  If the global ``height_sequence_1`` exists
        we reuse its length, otherwise a modest default (2000) is used.
    2.  Initialise with the arcsine shape  a_i ∝ 1/√(i+1)  and normalise
        to L1‑norm 1 (the absolute scale does not affect the score).
    3.  Repeatedly apply cheap random block perturbations (the same
        primitive used by the reference code) while keeping the best
        candidate seen so far.
    4.  Return the best sequence as a plain Python list of floats.
    """
    # ------------------------------------------------------------------
    # 1) Determine the problem size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # pylint: disable=undefined-variable
        if n == 0:
            raise Exception
    except Exception:
        # fallback length – small enough for fast experimentation
        n = 2000

    # ------------------------------------------------------------------
    # 2) Build the arcsine (1/√i) starting point.
    # ------------------------------------------------------------------
    # indices start at 1 to avoid division by zero
    idx = np.arange(1, n + 1, dtype=np.float64)
    base = 1.0 / np.sqrt(idx)
    base = base / np.sum(base)                     # L1‑normalise (scale is irrelevant)

    # ------------------------------------------------------------------
    # 3) Local stochastic search (time‑bounded).
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Keep the best solution seen so far.
    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # Simple block‑perturbation hill‑climbing loop.
    while time.monotonic() < deadline:
        cand = _make_block_perturbation(best_seq, rng)

        # fast evaluation – the official evaluator is authoritative,
        # but the formula is cheap and identical for non‑clipped values.
        # We still call the official function once the candidate wins.
        # This saves many expensive FFT‑based convolutions.
        cand_sum = np.sum(cand)
        if cand_sum < 1e-12:               # degenerate – reject quickly
            continue
        # compute convolution via FFT (O(n log n))
        size = int(2 ** np.ceil(np.log2(2 * n - 1)))
        fa = np.fft.rfft(cand, size)
        conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
        max_b = np.max(conv)
        cand_score = 2.0 * n * max_b / (cand_sum ** 2)

        if cand_score < best_score:
            # verify with the official checker (handles clipping, NaNs,…)
            official_score = float(evaluate_sequence(cand.tolist()))
            if not np.isfinite(official_score):
                continue
            if official_score < best_score:
                best_seq = cand
                best_score = official_score
                print(f"[improve] new best score {best_score:.6f} "
                      f"at t={time.monotonic():.2f}")

    # ------------------------------------------------------------------
    # 4) Return the (hopefully) improved sequence.
    # ------------------------------------------------------------------
    print(f"[final] best score {best_score:.6f} after "
          f"{time.monotonic():.2f} seconds")
    return [float(x) for x in best_seq]
