def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5
    n = len(height_sequence_1) if defined else 256
    # initialize sequence
    parent = np.full(n, 1.0 / n, dtype=np.float64)
    if 'height_sequence_1' in globals(): ...
    # optionally take best from existing if better
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # parameters
    learning_rate = 0.2
    decay = 0.9995
    iter_count = 0
    while time.monotonic() < deadline:
        # compute convolution of best
        conv_best = np.convolve(best, best)
        idx_max = np.argmax(conv_best)
        # generate candidate via targeted move
        # find i with max product
        i_start = max(0, idx_max - (n-1))
        i_end = min(idx_max, n-1)
        i_vals = np.arange(i_start, i_end+1)
        j_vals = idx_max - i_vals
        products = best[i_vals] * best[j_vals]
        max_pair_idx = np.argmax(products)
        i = i_vals[max_pair_idx]
        j = j_vals[max_pair_idx]
        # delta
        delta = min(best[i], best[j]) * learning_rate
        if delta > 0:
            cand = best.copy()
            cand[i] -= delta
            cand[j] -= delta
            # pick two other indices to receive delta
            p = rng.integers(0, n)
            q = rng.integers(0, n)
            cand[p] += delta
            cand[q] += delta
            # clip
            np.clip(cand, 0, 1000, out=cand)
            # compute score
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
        # occasionally random perturb
        if rng.random() < 0.05:
            block = make_block_perturbation(best, rng)
            score = float(evaluate_sequence(block.tolist()))
            if np.isfinite(score) and score < best_score:
                best = block
                best_score = score
        # update learning_rate
        learning_rate *= decay
        iter_count += 1
        if iter_count % 1000 == 0:
            print(f"Iter {iter_count}, best_score={best_score:.6f}, lr={learning_rate:.6f}")
    print(f"Finished after {iter_count} iterations, best_score={best_score}")
    return best.tolist()

    n = len(seq)
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)

"""Propose a step‐function that lowers the evaluation score.

The algorithm combines three ideas:
1.  A fast “gradient” move that tries to reduce the current maximal
    convolution coefficient.  For the index `k_max` where the
    convolution `c = f*f` is maximal, the sub‑gradient of `max(c)`
    w.r.t. `f` is `g_i = 2*f_{k_max-i}` (zero when the index is out of
    bounds).  Moving a little in the opposite direction reduces the
    peak.  After each move we clip to the allowed range and renormalise
    the vector (the score is scale‑invariant).

2.  Random block perturbations (the same routine used in the reference
    solution) to inject diversity and escape plateaus.

3.  A simple simulated‑annealing acceptance rule so occasional
    non‑improving moves are accepted early on, helping the search to
    explore a larger region of the space.

The search runs until the allocated time budget is exhausted, keeping
the best sequence seen.  All heavy work (convolution, score) is done
with NumPy which is fast enough for vectors of a few thousand entries.
"""
import numpy as np
import time
from math import exp

# ----------------------------------------------------------------------
# Helper: fast scoring (identical to evaluate_sequence but without the
# security checks).  All values are already clipped to [0,1000] and
# non‑negative, so the function is safe and much faster.
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative clipped sequence."""
    s = seq.sum()
    if s < 1e-12:                     # protect against division by zero
        return np.inf
    conv = np.convolve(seq, seq)
    return float(2.0 * len(seq) * conv.max() / (s * s))

# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Create one O(n) non‑negative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # width of the block – between n/200 and n/12, at least 1 element
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative Gaussian noise on the selected block
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    # with probability 0.35 make the sequence (approximately) symmetric
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# ----------------------------------------------------------------------
def _gradient_move(seq: np.ndarray, lr: float, rng: np.random.Generator) -> np.ndarray:
    """One deterministic move that reduces the current maximal convolution."""
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())               # index of the current peak
    # build the sub‑gradient: g_i = 2 * seq_{k-i}  (if the partner index exists)
    grad = np.zeros_like(seq)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_vals = np.arange(i_start, i_end + 1, dtype=int)
    j_vals = k - i_vals
    grad[i_vals] = 2.0 * seq[j_vals]

    # step size (learning rate) scaled by the current magnitude of the gradient
    step = lr * np.mean(grad) if grad.any() else 0.0
    if step == 0.0:
        return seq

    new_seq = seq - step * grad
    # keep non‑negative numbers
    np.clip(new_seq, 0.0, None, out=new_seq)

    # renormalise to keep the total sum constant (scale does not matter for the score)
    total = new_seq.sum()
    if total > 0.0:
        new_seq *= seq.sum() / total
    else:
        # fall back to a tiny random perturbation if we lost all mass
        new_seq = np.abs(rng.normal(0.0, 1e-6, size=n))
        new_seq *= seq.sum() / new_seq.sum()

    # final clipping to obey the evaluator’s hard limits
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score step function within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock time (seconds) available for the search.  The function
        returns the best sequence found before the deadline.
    kwargs
        Ignored – present only to match the expected signature.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise the search vector -------------------------------------------------
    # Prefer the previously discovered construction if it looks sane,
    # otherwise fall back to a uniform distribution.
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            a0 = np.clip(base, 0.0, 1000.0)
        else:
            a0 = None
    else:
        a0 = None

    if a0 is None or a0.size == 0:
        n = 256
        a0 = np.ones(n, dtype=np.float64)
    else:
        n = a0.size

    # normalise to unit sum – the objective is scale invariant
    a0 = a0.astype(np.float64)
    a0_sum = a0.sum()
    if a0_sum == 0.0:
        a0 = np.full_like(a0, 1.0 / n)
    else:
        a0 = a0 / a0_sum

    best = a0.copy()
    best_score = _fast_score(best)

    # ------------------------------------------------------------------
    # parameters for the stochastic search
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    it = 0
    # learning‑rate schedule for the gradient moves
    lr = 0.5
    lr_decay = 0.9995

    # Simulated‑annealing temperature (controls acceptance of worse moves)
    temperature = 0.1
    temp_decay = 0.9998

    # Main optimisation loop ----------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # Choose a move type
        if rng.random() < 0.6:
            # gradient‑based move targeting the current peak
            candidate = _gradient_move(best, lr, rng)
        else:
            # random block perturbation (large‑scale exploration)
            candidate = make_block_perturbation(best, rng)

        # Occasionally inject a completely fresh random start (helps escape deep minima)
        if rng.random() < 0.02:
            candidate = rng.dirichlet(np.ones(n))
            # keep the same overall scale as the current best
            candidate *= best.sum()
            np.clip(candidate, 0.0, 1000.0, out=candidate)

        # Compute the (fast) score
        cand_score = _fast_score(candidate)

        # Simulated‑annealing acceptance rule
        accept = False
        if np.isfinite(cand_score):
            if cand_score < best_score:
                accept = True
            else:
                # probability of upward move
                prob = exp(-(cand_score - best_score) / max(temperature, 1e-12))
                if rng.random() < prob:
                    accept = True

        if accept:
            best = candidate
            best_score = cand_score

        # periodic reporting
        if it % 5000 == 0:
            elapsed = time.monotonic() - (deadline - float(budget_s) + 5.0)
            print(f"[{elapsed:6.2f}s] iter {it:,d} – best score {best_score:.6f}")

        # decay schedule
        lr *= lr_decay
        temperature *= temp_decay

    # Final sanity check – make sure the vector respects the limits
    np.clip(best, 0.0, 1000.0, out=best)

    # Convert to plain Python list as required by the external harness
    return [float(x) for x in best]
