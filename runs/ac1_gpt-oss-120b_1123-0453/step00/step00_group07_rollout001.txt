"""
Propose a step‑function that is provably optimal for ``evaluate_sequence``.

The evaluation function is
    E(a) = 2·n·max(conv(a,a)) / (sum(a))² .
For any non‑negative a, the maximal convolution entry is at least
    Σ a_i²  (the centre of the convolution).
By Cauchy–Schwarz,
    Σ a_i² ≥ (Σ a_i)² / n,
so
    E(a) ≥ 2·n·(Σ a_i)²/(n·(Σ a_i)²) = 2.

Equality holds exactly for a constant sequence (all entries equal and
non‑zero).  Consequently the best achievable score is 2.0, and any
constant vector attains this bound.

The function below therefore returns a uniform vector (value 1.0) whose
length matches the previously discovered ``height_sequence_1`` when that
is available; otherwise it falls back to length 256.
The code respects the supplied time budget, prints the obtained score,
and returns the sequence as a plain ``list[float]``.
"""

import time
import numpy as np

def _uniform_candidate(length: int, value: float = 1.0) -> np.ndarray:
    """Create a uniform array of given length, clipped to the allowed range."""
    arr = np.full(length, value, dtype=np.float64)
    # The evaluator clamps values to [0, 1000]; this makes the intention explicit.
    return np.clip(arr, 0.0, 1000.0)

    """
    Return a sequence that minimises ``evaluate_sequence``.

    The routine follows a deterministic strategy:
    * Determine a suitable length (use the existing global ``height_sequence_1`

    * Log the resulting score for reference.
    * Return the candidate as a list of python floats.

    Parameters
    ----------
    seed : int, optional
        Random seed (kept only to match the required signature).
    budget_s : float, optional
        Time budget in seconds; not needed for the optimal solution but

    **kwargs :
        Additional unused arguments.

    Returns
    -------
    list[float]
        The uniform step function.
    """
    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is expected to be a global variable containing
        # a previously discovered sequence.  We use its length if it looks
        # reasonable; otherwise we fall back to a modest default.
        base = np.asarray(height_sequence_1, dtype=np.float64)
