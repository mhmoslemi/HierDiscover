"""
propose_candidate
-----------------
A fast sub‑gradient descent optimizer for the max‑convolution objective.

The evaluation function to be minimized is

    2 * n * max(a * a) / (sum(a) ** 2)

where `max(a * a)` is the maximum entry of the full discrete convolution
of the (non‑negative) sequence with itself.  Because the objective
is scale‑invariant we keep the ℓ₁‑norm fixed to √(2 n) – after this
normalisation the objective equals the maximal convolution entry.
The optimizer repeatedly:
    * evaluates the full convolution via an FFT (O(n log n)),
    * locates the index where it attains its maximum,
    * builds the sub‑gradient (which for a given index `s` is simply
      `grad[i] = 2 * a[s-i]` whenever `0 ≤ s-i < n`),
    * takes a back‑tracked step opposite to the gradient, projects onto
      the non‑negative orthant, and re‑normalises the sum.
Occasional random block perturbations are injected to escape shallow
local minima.  The best sequence seen during the allotted time is
returned.

The routine works for any length (the current benchmark uses the
global `height_sequence_1` of length 4806) and respects the
evaluate_sequence() security checks.
"""

import numpy as np
import time
import math

# --------------------------------------------------------------
# Helper utilities (no closures, no lambdas as required)
# --------------------------------------------------------------

def _target_sum(num_items: int) -> float:
    """Desired ℓ₁‑norm (the optimizer keeps the sum fixed to √(2 n))."""
    return math.sqrt(2.0 * num_items)


def _fft_len(num_items: int) -> int:
    """Length of the zero‑padded FFT (next power of two ≥ 2·n‑1)."""
    needed = 2 * num_items - 1
    return 1 << (needed - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return full linear convolution a*a using a real FFT."""
    # rfft computes the DFT of the real input up to the Nyquist freq.
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # The true linear convolution has length 2·n‑1.
    return conv[: 2 * a.size - 1]


def _max_conv_and_index(a: np.ndarray, fft_len: int):
    """Maximum convolution value and its index for a given sequence."""
    conv = _full_convolution_fft(a, fft_len)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv  # also return the full array if needed


def _score_from_seq(a: np.ndarray) -> float:
    """Exact objective (identical to evaluate_sequence) using our own fast code."""
    n = a.size
    if n == 0:
        return float("inf")
    # enforce the same clipping as evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    max_b, _, _ = _max_conv_and_index(a, _fft_len(n))
    return 2.0 * n * max_b / (s * s)


def _subgradient_for_max(a: np.ndarray, s_max: int) -> np.ndarray:
    """Sub‑gradient of max‑convolution w.r.t. a (for the bin `s_max`)."""
    n = a.size
    grad = np.zeros_like(a)
    # i contributes via a[i] * a[s_max-i] if the partner index lies in range.
    i = np.arange(n)
    j = s_max - i
    mask = (j >= 0) & (j < n)
    grad[i[mask]] = 2.0 * a[j[mask]]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Like the baseline code: multiply a random block by a log‑normal factor."""
    n = a.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    out = a.copy()
    out[start:stop] *= scale
    if rng.random() < 0.35:
        out = 0.5 * (out + out[::-1])
    out = np.clip(out, 0.0, 1000.0)
    return out

# Main entry point required by the competition framework
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises
    evaluate_sequence.  The routine runs for roughly ``budget_s`` seconds
    (a small safety margin is kept) and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate wall‑clock time budget in seconds.
    **kwargs
        Ignored – part of the required signature.

    Returns
    -------
    list[float]
        The best sequence discovered (as plain Python floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (use the known global if available)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if base.size == 0:
            raise ValueError
    except Exception:
        # fallback – a modest length that still gives a non‑trivial search
        base = np.ones(256, dtype=np.float64)

    n = int(base.size)
    target = _target_sum(n)
    fft_len = _fft_len(n)

    # ------------------------------------------------------------------
    # Initialise a pool of starting points (random, uniform, previous best)
    # ------------------------------------------------------------------
    candidates = []

    # 1) Start from the previously discovered sequence (clipped & normalised)
    cand = np.clip(base, 0.0, 1000.0)
    if cand.sum() > 0:
        cand = cand / cand.sum() * target
    else:
        cand = np.full_like(cand, target / n)
    candidates.append(cand)

    # 2) Full uniform vector (the trivial constant solution)
    uniform = np.full(n, target / n, dtype=np.float64)
    candidates.append(uniform)

    # 3) A purely random start
    rnd = rng.random(n)
    rnd = rnd / rnd.sum() * target
    candidates.append(rnd)

    # Keep the best of the initial pool
    best_seq = None
    best_score = float("inf")
    for idx, seq in enumerate(candidates):
        sc = _score_from_seq(seq)
        if np.isfinite(sc) and sc < best_score:
            best_seq = seq.copy()
            best_score = sc
            print(f"[init] candidate {idx} -> score {sc:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop (sub‑gradient descent + occasional random kicks)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)  # safety margin
    step = max(0.5 * target / n, 1e-4)   # initial step size (roughly 0.5·mean)

    # Statistics for diagnostics
    iter_no = 0
    accept_counter = 0
    while time.monotonic() < deadline:
        iter_no += 1

        # ------------------------------------------------------------------
        # 1) Evaluate current max‑convolution and its location
        # ------------------------------------------------------------------
        cur_max, s_max, _ = _max_conv_and_index(best_seq, fft_len)

        # ------------------------------------------------------------------
        # 2) Compute sub‑gradient for the offending bin
        # ------------------------------------------------------------------
        grad = _subgradient_for_max(best_seq, s_max)

        # ------------------------------------------------------------------
        # 3) Back‑track line search on the quadratic sub‑gradient step
        # ------------------------------------------------------------------
        improved = False
        step_try = step
        for _ in range(12):  # at most 12 back‑tracking attempts
            # tentative update
            cand = best_seq - step_try * grad
            cand = np.clip(cand, 0.0, 1000.0)
            s = cand.sum()
            if s <= 0.0:          # pathological – re‑initialise randomly
                cand = rng.random(n)
                cand = cand / cand.sum() * target
                s = target
            else:
                cand = cand * (target / s)   # renormalise ℓ₁‑norm

            # cheap evaluation (max‑convolution only)
            new_max, _, _ = _max_conv_and_index(cand, fft_len)

            if new_max < cur_max - 1e-12:   # a strict improvement
                best_seq = cand
                cur_max = new_max
                improved = True
                break
            else:
                step_try *= 0.5   # shrink and retry

        # adapt step size for next iteration
        if improved:
            accept_counter += 1
            step = min(step * 1.2, target)  # allow a modest increase
        else:
            step = max(step * 0.7, 1e-6)    # become more cautious

        # ------------------------------------------------------------------
        # 4) Occasionally inject a random block perturbation (exploration)
        # ------------------------------------------------------------------
        if (iter_no % 250) == 0 or not improved:
            pert = _random_block_perturb(best_seq, rng)
            pert = pert / pert.sum() * target
            pert_max, _, _ = _max_conv_and_index(pert, fft_len)
            if pert_max < cur_max:
                best_seq = pert
                cur_max = pert_max
                # keep the current step size unchanged (trust the perturb)

        # ------------------------------------------------------------------
        # 5) Record best score seen so far (full objective)
        # ------------------------------------------------------------------
        score = 2.0 * n * cur_max / (target * target)   # equal to cur_max after normalisation
        if score < best_score:
            best_score = score
            # keep a pristine copy (already clipped & normalised)
            best_seq = np.clip(best_seq, 0.0, 1000.0)

        # occasional progress report
        if (iter_no % 1000) == 0:
            print(f"[iter {iter_no:5d}] best_score={best_score:.6f} "
                  f"cur_max={cur_max:.6f} step={step:.6f} accepts={accept_counter}")

    # ------------------------------------------------------------------
    # Final validation via the official evaluate_sequence (just for safety)
    # ------------------------------------------------------------------
    final_list = [float(x) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_list)
    print(f"[final] best_score (internal) = {best_score:.6f}, "
          f"official score = {final_score:.6f}")

    return final_list
