            cand = best.copy()
            b = np.convolve(cand, cand)
            max_k = int(np.argmax(b))

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f'New best: {best_score:.6f}')
            # optionally reduce block width for refined search
    return best.tolist()

def random_candidate(n, rng, S, max_val, alpha):
    raw = rng.gamma(alpha, 1.0, size=n)
    arr = raw / raw.sum() * S
    arr = np.clip(arr, 0, max_val)
    # re-normalize after clipping
    sum_arr = arr.sum()
    if sum_arr == 0:
        # fallback to uniform
        arr = np.full(n, S / n)
    else:
        arr = arr / sum_arr * S
    return arr

def reduce_max(seq, target_M, rng):
    for _ in range(10):
        b = np.convolve(seq, seq)
        max_b = b.max()
        if max_b <= target_M:
            break
        max_k = int(np.argmax(b))
        # compute contributions: for i in range(max(0, max_k-(n-1)), min(n-1, max_k)+1)
        i_start = max(0, max_k - (len(seq) - 1))
        i_end = min(len(seq) - 1, max_k)
        # compute product contributions for each i
        contributions = [(i, seq[i]*seq[max_k - i]) for i in range(i_start, i_end+1)]
        # sort by contribution descending
        contributions.sort(key=lambda x: x[1], reverse=True)
        # pick top few indices (maybe top 2)
        top = contributions[:2]

                f = 1/f
            seq[i] *= f
        # ensure nonnegatives

    return seq

"""
Search for a step‑function (sequence of non‑negative heights) that minimises
the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a hybrid of three cheap heuristics:

1. **Random Dirichlet sampling** – draws a very sparse vector (Gamma
   distribution with a small shape parameter), caps every entry at
   sqrt(target) and renormalises it to the required ℓ₁ norm
   (√(2 n)).  Sparse vectors tend to have few pairwise products,
   consequently a smaller convolution peak.

2. **Block perturbation** – perturbs the current best vector by scaling a
   random contiguous block (the routine used in the reference code).
   This explores the neighbourhood of the best candidate.

3. **Targeted reduction** – repeatedly looks at the index where the
   convolution attains its maximum, scales down the few entries that
   contribute most to that peak and re‑normalises the vector.  This
   directly attacks the worst‑case term.

All three moves are tried repeatedly until the time budget expires.
Whenever a better sequence is found its score is printed.
The function returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: block perturbation (taken from the reference implementation)
def make_block_perturbation(base, rng):
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # width of the block – a fraction of the total length
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # with some probability, enforce mirror symmetry (helps sometimes)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Helper: generate a random sparse candidate via a Dirichlet‑like draw
def random_dirichlet_candidate(n, rng, sum_target, max_val, alpha):
    """
    Produce a random non‑negative vector of length ``n`` whose ℓ₁ norm is
    ``sum_target``.  The entries are drawn from a Gamma distribution
    (shape=alpha) to obtain a sparse distribution, then clipped to
    ``max_val`` and finally re‑normalised.
    """
    # Gamma draws (Gamma(alpha, 1) == Dirichlet when normalised)
    raw = rng.gamma(alpha, 1.0, size=n)
    if raw.sum() == 0.0:
        raw = np.ones(n)
    vec = raw / raw.sum() * sum_target
    # enforce an absolute upper bound – this keeps single squares ≤ target
    vec = np.clip(vec, 0.0, max_val)
    # renormalise again (clipping changes the total mass)
    s = vec.sum()
    if s == 0.0:
        vec = np.full(n, sum_target / n)
    else:
        vec = vec / s * sum_target
    return vec


# ----------------------------------------------------------------------
# Helper: reduce the current maximum of the self‑convolution
def reduce_max(seq, target_M, rng, steps=5):
    """
    Simple deterministic “gradient‑free” step: look at the index of the
    largest convolution entry, scale down the two coordinates that
    contribute most to it, and renormalise to the original ℓ₁ norm.
    Repeated a few times (``steps``) or until the peak is ≤ target.
    """
    n = len(seq)
    sum_target = seq.sum()
    for _ in range(steps):
        b = np.convolve(seq, seq)
        max_b = b.max()
        if max_b <= target_M:
            break
        k = int(np.argmax(b))
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # contributions a[i] * a[k-i]; keep the two largest ones
        contributions = [(i, seq[i] * seq[k - i]) for i in range(i_min, i_max + 1)]
        contributions.sort(key=lambda x: x[1], reverse=True)
        # Scale the two biggest contributors down (geometric move)
        for i, _ in contributions[:2]:
            # a modest shrink factor, centred at 1.0
            factor = np.exp(rng.normal(0.0, 0.06))
            if factor > 1.0:
                factor = 1.0 / factor
            seq[i] *= factor
        # keep non‑negative and respect the global 0‑1000 clamp
        seq = np.clip(seq, 0.0, 1000.0)
        # renormalise to keep the ℓ₁ norm (scaling does not affect the score)
        s = seq.sum()
        if s == 0.0:
            seq = np.full(n, sum_target / n)
        else:
            seq = seq / s * sum_target
    return seq
