function refine(seq, steps=200, eta_start=0.5):
    a = seq.copy()
    for s in range(steps):
        conv = np.convolve(a, a)
        kmax = np.argmax(conv)
        # compute gradient
        grad = np.zeros_like(a)
        low = max(0, kmax - (a.size - 1))
        high = min(a.size-1, kmax)
        idx = np.arange(low, high+1)
        grad[idx] = 2 * a[kmax - idx]
        # optional add small epsilon to avoid zero
        grad += 1e-12
        # step size schedule
        eta = eta_start * (0.9 ** s)
        a = a * np.exp(-eta * grad)
        # renormalize: sum to 1
        total = a.sum()
        if total > 0:
            a = a / total
        # optionally clip to small lower bound
        a = np.clip(a, 0.0, None)
    return a

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a nonnegative sequence that minimizes evaluate_sequence.
    Uses a small linear programming refinement (alternating convex) on
    a moderate-size problem (n=256) combined with random restarts and
    simple gradient descent.
    """
    import time, numpy as np
    from scipy.optimize import linprog
    from scipy.sparse import diags

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Choose problem size
    n = min(256, len(globals().get('height_sequence_1', [])) or 256)
    # If height_sequence_1 is too short, default 256.

    best = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial uniform score: {best_score:.6f}")

    # Helper: evaluate and optionally update best
    def try_update(candidate):
        nonlocal best, best_score
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate.copy()
            best_score = score
            print(f"New best score: {best_score:.6f}")
        return

    t = 0.3  # mixing step size
    max_restarts = 1000  # will not reach due to time limit
    iteration = 0

    while time.monotonic() < deadline:
        # random restart with Dirichlet
        alpha = np.full(n, 0.5)  # concentration <1 for variability
        f = rng.dirichlet(alpha)  # sum=1
        # optional small smoothing: clip
        f = np.clip(f, 0.0, None)

        # ensure not trivial uniform
        # Perform a few LP refinement steps
        prev_M = None
        for step in range(5):
            # compute convolution max
            conv = np.convolve(f, f)
            M = conv.max()
            # Stop if M not improving significantly
            if prev_M is not None and M >= prev_M * 0.9999:
                break
            prev_M = M

            # Build LP matrix A
            # A_ub rows = 2n-1, cols = n, each row k has f_i at column j where i + j = k
            # Use diags with offsets 0..n-1
            offsets = np.arange(n, dtype=int)
            # Each diagonal data is f (size n)
            data = np.tile(f, (n, 1))
            A = diags(data, offsets, shape=(2*n-1, n), format='csr')

            # RHS vector of M
            b_ub = np.full(2*n-1, M, dtype=np.float64)

            # Objective: maximize sum(g) -> minimize -sum(g)
            c = -np.ones(n, dtype=np.float64)

            # Solve LP
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True})
            if not res.success:
                # fallback: break out of inner loop
                break
            g = res.x
            sum_g = g.sum()
            if sum_g <= 0:
                break
            # scale g to same total mass as f (here both are 1)
            # but we keep sum=1
            # mix
            f = (1 - t) * f + t * g
            # optionally renormalize to sum=1 (due to rounding)
            f_sum = f.sum()
            if f_sum != 0:
                f = f / f_sum

        # after refinement, evaluate candidate
        try_update(f)

        # occasional gradient descent refinement on best
        if iteration % 20 == 0:
            # apply gradient descent to current best
            candidate = best.copy()
            # simple gradient descent with adaptive step
            for gd_step in range(30):
                conv = np.convolve(candidate, candidate)
                kmax = int(conv.argmax())
                grad = np.zeros_like(candidate)
                i_start = max(0, kmax - (n-1))
                i_end = min(n-1, kmax)
                idx = np.arange(i_start, i_end+1)
                grad[idx] = 2.0 * candidate[kmax - idx]
                # step size adaptation
                eta = 0.5 / (gd_step + 1)
                candidate = candidate * np.exp(-eta * grad)
                # renormalize

"""
Propose a step‑function sequence with a lower evaluation score.

The core idea is a small convex‐programming refinement (the “alternating LP”
described in the statement) applied to a moderate problem size (n≈256).  Starting
from many random Dirichlet draws we repeatedly:

1.  Solve   max Σ g_i   subject to   f≥0 , Σ f_i = 1 and
    (f * g)[k] ≤ M = max(f*f) for every convolution index k.
    This is a linear program; the constraints are encoded via a sparse Toeplitz
    matrix built with ``scipy.sparse.diags``.

2.  Mix the obtained g back into f with a small step size t (here t=0.3) and
    renormalise to keep Σ f_i = 1.  The theory guarantees that the maximal
    convolution value never increases, so the objective J = 2·n·M steadily
    improves.

3.  After a few LP steps we evaluate the candidate with the provided
    ``evaluate_sequence`` and keep the best one seen so far.  Occasionally we
    polish the best candidate by a simple sub‑gradient descent on the active
    convolution index.

The whole search is bounded by the supplied time budget (default 1100 s);
the algorithm aborts early if the deadline is reached.  The returned list
contains non‑negative numbers and is guaranteed to be a valid input for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP described in the problem statement for a given f and bound M.
    Returns the optimal g (may be the same length as f).  If the LP fails, the
    original f is returned unchanged.
    """
    n = f.size
    # Build the Toeplitz‑like constraint matrix:
    #   (f * g)[k] = sum_i f_i * g_{k-i} ≤ M   for k = 0 .. 2n‑2
    offsets = np.arange(n, dtype=int)                # 0 … n‑1
    data = np.tile(f, (n, 1))                        # each diagonal = f
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)                     # maximise Σ g_i  → minimise -Σ g_i
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method="highs", options={"presolve": True})
    if not res.success:
        return f.copy()  # fall back to the current vector
    return res.x


def _gradient_descent(a: np.ndarray, steps: int = 30, eta0: float = 0.5) -> np.ndarray:
    """
    Simple sub‑gradient descent on the active convolution index.
    The step size decays as eta = eta0/(step+1).
    """
    n = a.size
    for s in range(steps):
        conv = np.convolve(a, a)
        k = int(conv.argmax())
        # gradient of the active quadratic term: 2 * a_{k-i}
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]
        eta = eta0 / (s + 1)
        a = a * np.exp(-eta * grad)          # multiplicative update keeps positivity
        total = a.sum()
        if total > 0:
            a = a / total                     # re‑normalise to Σ a_i = 1
    return a
