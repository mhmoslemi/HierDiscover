def proj_simplex(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    # algorithm from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max i s.t. u_i - (cssv_i / i) > 0
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    Tries to reduce the max self-convolution value under a fixed ℓ1 norm.
    """
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Determine sequence length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = base.size
    except Exception:
        n = 6000  # fallback length
        base = None

    # Initialize candidate: arcsine solution on length n
    a = arcsine_sequence(n)  # already sum to 1

    # optionally mix with provided base
    if base is not None:
        # evaluate both and keep better
        cand1 = base.copy()
        cand2 = a.copy()
        score1 = float(evaluate_sequence(cand1.tolist()))
        score2 = float(evaluate_sequence(cand2.tolist()))
        if score1 < score2:
            a = cand1 / np.sum(cand1)  # normalize to sum=1 for consistency
            best = cand1.copy()
            best_score = score1
        else:
            best = cand2.copy()
            best_score = score2
    else:
        best = a.copy()
        best_score = float(evaluate_sequence(a.tolist()))

    # Ensure best sums to 1 (to keep scale invariance)
    best_sum = np.sum(best)
    if best_sum <= 0:
        best = np.ones(n) / n
        best_sum = 1.0
    best /= best_sum

    a = best.copy()
    step = 0.1
    iteration = 0

    while time.monotonic() < deadline:
        # Compute convolution and max index
        conv = np.convolve(a, a)
        max_val = conv.max()
        s_max = int(conv.argmax())
        # Gradient of the max entry
        idx = np.arange(n)
        match = s_max - idx
        mask = (match >= 0) & (match < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[match[mask]]
        # Take a gradient step with projection onto the simplex
        cand = a - step * g
        # Add a little isotropic noise every few iterations
        if iteration % 50 == 0:
            cand += rng.normal(scale=0.001, size=n)
        # Simplex projection (l1 = 1, non‑negative)
        cand = proj_simplex(cand)
        # Evaluate candidate (scale invariance -> we keep sum = 1)
        cand_score = 2.0 * n * np.max(np.convolve(cand, cand))  # sum(cand)=1
        # Keep the best
        if cand_score < best_score:
            best = cand.copy()
            best_score = float(cand_score)
            a = cand.copy()
        else:
            # occasionally accept a move to avoid stagnation
            if rng.random() < 0.001:
                a = cand.copy()
        # Decay step size slowly
        step *= 0.9999
        iteration += 1
        # occasional random block perturbation to escape flat regions
        if iteration % 500 == 0:
            pert = make_block_perturbation(best, rng)
            pert = np.asarray(pert, dtype=np.float64)
            # renormalize to sum=1
            if np.sum(pert) > 0:
                pert = pert / np.sum(pert)
                pert_score = 2.0 * n * np.max(np.convolve(pert, pert))
                if pert_score < best_score:
                    best = pert.copy()
                    best_score = float(pert_score)
                    a = pert.copy()
        # print progress occasionally
        if iteration % 2000 == 0:
            print(f"Iter {iteration}, best_score={best_score:.6f}, step={step:.6f}")

    # Return the best sequence found as a plain Python list of floats
    return [float(x) for x in best]

def proj_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    # shift to ensure sum = 1
    # algorithm from https://stanford.edu/~jduchi/projects/DuchiShallow.pdf
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent optimizer for the step‑function “max‑convolution” score.

The score to be minimised is
    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm starts from the analytically known “arcsine” sequence
a_i ∝ C(2i,i)/4^i (central binomial coefficients) which already beats the
uniform sequence (score ≈ π/2).  From this point we repeatedly

1. compute the convolution,
2. locate the index s where the convolution attains its maximum,
3. take a sub‑gradient of this maximal entry (∂/∂a_i  = 2·a_{s‑i}),
4. move a little opposite to the gradient and project back onto the
   probability simplex (non‑negative, sum = 1).

Occasional tiny random noises and block‑perturbations are injected to
escape shallow local minima.  The loop stops when the allotted time is
exhausted (≈ 1 000 s) and the best sequence found is returned.
"""

import time
import numpy as np


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (sum = 1, v ≥ 0)."""
    # Duchi et al. “Efficient Projections onto the L1‑Ball for Learning in
    # High Dimensions”, ICML 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i > (cssv_i / i)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_sequence(n: int) -> np.ndarray:
    """
    Return the “arcsine’’ sequence of length n:
        a_i = C(2i,i) / 4^i , then normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # a_i = a_{i‑1} * (2i‑1) / (2i)
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()          # put the vector onto the simplex
    return a


def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (via FFT)."""
    n = x.shape[0]
    size = 2 * n - 1
    # real FFT is a bit faster for real‑valued data
    f = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv


def make_block_perturbation(base, rng):
    """A cheap O(n) non‑negative perturbation used as a diversification step."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) improves on the
    current best score.  The method starts from the arcsine distribution,
    performs projected gradient descent on the maximal convolution entry,
    and occasionally injects random block perturbations for exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------------
    # Determine the problem size (length of the step function)
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        # fallback length – reasonably large but not too expensive
        n = 6000
        base = None

    # ----------------------------------------------------------------------
    # Initial candidates: arcsine and (if available) the provided one
    # ----------------------------------------------------------------------
    a_arc = arcsine_sequence(n)                       # already on simplex
    best = a_arc.copy()
    best_score = float(2.0 * n * np.max(self_convolution(best)))

    if base is not None:
        # normalise the given sequence to sum = 1  (scale does not change score)
        tmp = base.copy()
        if tmp.sum() == 0:
            tmp = np.ones_like(tmp)
        tmp = np.maximum(tmp, 0.0)
        tmp /= tmp.sum()
        score_tmp = float(2.0 * n * np.max(self_convolution(tmp)))
        if score_tmp < best_score:
            best = tmp.copy()
            best_score = score_tmp

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    a = best.copy()
    step = 0.15               # initial step size (along the negative gradient)
    decay = 0.9999            # gradual step‑size decay
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # 1) current convolution and the index of its maximum
        conv = self_convolution(a)
        max_val = conv.max()
        s_max = int(conv.argmax())

        # 2) sub‑gradient of the maximal entry (∂/∂a_i = 2·a_{s_max‑i})
        idx = np.arange(n)
        mate = s_max - idx
        mask = (mate >= 0) & (mate < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[mate[mask]]

        # 3) gradient step + tiny isotropic noise every few iterations
        cand = a - step * g
        if it % 30 == 0:
            cand += rng.normal(scale=0.001, size=n)

        # 4) project back onto the simplex (keeps sum = 1 and non‑negativity)
        cand = proj_simplex(cand)

        # 5) evaluate the candidate (the sum is 1 → score = 2·n·max(conv))
        cand_score = float(2.0 * n * np.max(self_convolution(cand)))

        # 6) keep the best solution found so far
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            a = cand.copy()
            no_improve = 0
        else:
            no_improve += 1
            # occasional random restart if we stagnate for a while
            if no_improve > 2000:
                a = arcsine_sequence(n)
                no_improve = 0

        # 7) every few hundred iterations try a block perturbation (exploration)
        if it % 500 == 0 and it > 0:
            pert = make_block_perturbation(best, rng)
            pert = pert / np.sum(pert)               # re‑normalise to the simplex
            pert_score = float(2.0 * n * np.max(self_convolution(pert)))
            if pert_score < best_score:
                best = pert.copy()
                best_score = pert_score
                a = pert.copy()
                no_improve = 0

        # 8) decay step size and count iteration
        step *= decay
        it += 1

        # periodic progress report
        if it % 2000 == 0:
            print(f"Iter {it:6d} | best_score = {best_score:.6f} | step = {step:.6f}")

    # ----------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ----------------------------------------------------------------------
    print(f"Finished after {it} iterations, best score = {best_score:.6f}")
    return [float(x) for x in best]
