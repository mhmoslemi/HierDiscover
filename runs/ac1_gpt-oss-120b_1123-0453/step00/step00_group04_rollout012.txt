def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 3.0

    n = len(height_sequence_1) # assume provided; fallback to default length 2000
    # start with sqrt shape
    base = np.array([1.0/np.sqrt(i+1) for i in range(n)], dtype=np.float64)
    # optionally normalize to sum=1 for readability
    base = base/np.sum(base)   # scaling irrelevant; keep for stability.
    base = np.clip(base, 0.0, 1000.0)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f'Initial score: {best_score:.6f}')
    
    # Use L = 3 phases with decreasing temperature
    temperature = 1.0
    while time.monotonic() < deadline:
        # decide move type
        move_type = rng.choice(['block_mul', 'transfer'])
        if move_type == 'block_mul':
            # similar to previous but using smaller width maybe.
            candidate = best.copy()
            n = candidate.size
            width = int(rng.integers(max(1, n // 300), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = np.exp(rng.normal(0.0, 0.07))   # smaller perturbation
            candidate[start:stop] *= factor
        else:  # transfer
            i, j = rng.choice(len(best), size=2, replace=False)
            # amount to transfer
            amount = best[i] * rng.random() * 0.2  # move up to 20% of source
            if amount > best[i]:
                amount = best[i]
            candidate = best.copy()
            candidate[i] -= amount
            candidate[j] += amount
        
        # optionally symmetrize with 0.3 prob
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        # simulated annealing acceptance
        if np.isfinite(score) and (score < best_score or rng.random() < np.exp((best_score - score) / temperature)):
            best = candidate
            if score < best_score:
                best_score = score
                print(f'New best: {best_score:.6f} at {time.monotonic():.1f}s')
        # gradually cool
        temperature *= 0.999995  # decay per iteration

    print(f'Final best: {best_score:.6f}')
    return [float(x) for x in best]

"""
Propose a step‑function (non‑negative sequence) that tries to minimise
evaluate_sequence.  The construction is based on the continuous function
a(x) ~ 1/sqrt(x) (the “arcsine” shape) which makes the self‑convolution
almost flat, giving a theoretical score ≈ π/2 ≈ 1.57.  Starting from this
shape we run a fast simulated‑annealing / greedy search that repeatedly
moves a small amount of mass from a highly‑contributing index (with respect
to the current maximal convolution entry) to a weakly‑contributing index.
Occasional random block scalings and symmetrisation steps diversify the
search.  The best sequence seen before the time budget expires is returned.
"""

import numpy as np
import time

def _score(arr: np.ndarray) -> float:
    """Fast replica of evaluate_sequence (without the safety checks)."""
    n = arr.size
    if n == 0:
        return np.inf
    conv = np.convolve(arr, arr)
    max_b = float(conv.max())
    sum_a = float(arr.sum())
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) lower the
    evaluation score.  The routine respects the given time budget (in
    seconds) and prints occasional progress messages.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 1. Determine the length of the sequence we shall optimise.
    # -----------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))          # provided by the host
        if n == 0:
            raise Exception
    except Exception:
        n = 2000                                 # fallback length

    init = 1.0 / np.sqrt(idx + 1.0)               # monotone decreasing
    init = 0.5 * (init + init[::-1])              # make it symmetric
    init = np.clip(init, 0.0, 1000.0)

    best = init.copy()
    best_score = _score(best)
    print(f'initial score = {best_score:.6f} (n = {n})')

    # 3. Simulated‑annealing / greedy search.
    #    At every iteration we either
    #      * move a small amount of mass from a “high‑contrib” index to
    #        a “low‑contrib” index (targeted move), or
    #      * apply a random block multiplication (exploratory move).
    #    Occasionally we also symmetrise the whole vector.

        scr = _score(candidate)

        # ---- Acceptance ----
        improve = np.isfinite(scr) and scr < best_score
        if improve or (np.isfinite(scr) and rng.random() < np.exp((best_score - scr) / temperature)):
            best = candidate
            if improve:
                best_score = scr
                print(f'it {it:06d}: new best = {best_score:.6f}')
        # geometric cooling (very gentle)
        temperature *= 0.999997

    print(f'finished after {it} iterations, best score = {best_score:.6f}')
    # convert to plain python floats (as required by the runner)
    return [float(x) for x in best]
