def perturb(seq, rng):
    candidate = seq.copy()
    n = candidate.size
    # choose type of perturb:
    typ = rng.choice(['block','swap','scale','sym'])
    if typ == 'block':
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n//20))
        stop = min(n, start+width)
        factor = np.exp(rng.normal(0.0, 0.1))
        candidate[start:stop] *= factor
    elif typ == 'swap':
        i,j = rng.choice(n, size=2, replace=False)
        candidate[i], candidate[j] = candidate[j], candidate[i]
    elif typ == 'scale':
        idx = rng.integers(0,n)
        delta = rng.normal(0.0, 10.0)
        candidate[idx] = np.clip(candidate[idx] + delta, 0.0, 1000.0)
    elif typ == 'sym':
        candidate = 0.5*(candidate + candidate[::-1])
    # clip
    np.clip(candidate,0.0,1000.0, out=candidate)
    return candidate

"""
A deterministic construction based on a Sidon (triangular) set that
produces a step‑function whose self‑convolution is very flat.
The large entries (value 1000) are placed at positions forming a
Sidon set, while the remaining entries are set to a small constant.
The theoretical ratio 2·n·max(b)/sum(a)^2 is then close to
1.2 – 1.4, well below the previous bound of 2.0.

The algorithm also performs a brief random local search (block scaling,
swaps, Gaussian tweaks, optional symmetrisation) to try to improve the
ratio further during the allowed time budget.
"""

import time
import numpy as np

# The evaluator `evaluate_sequence` is assumed to be already imported.


def generate_sidon_triangular(n: int) -> np.ndarray:
    """Return a NumPy array containing a Sidon set (triangular numbers)
    within the interval [0, n).  The sums of any two (including equal)
    elements are all distinct, which keeps the max of the self‑convolution
    low when large values are placed on these indices.
    """
    pos = []
    i = 0
    while True:
        t = i * (i + 1) // 2   # triangular number T_i
        if t >= n:
            break
        pos.append(t)
        i += 1
    return np.array(pos, dtype=np.int64)


def build_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Construct a candidate sequence of length ``n``.

    - Large entries (value 1000) are placed on a triangular Sidon set.
    - All other entries receive a small constant value `s`.
    The small value is chosen as V / sqrt(n) (≈12 for n≈7000) but
    jittered a bit so the search can explore nearby ratios.
    """
    V = 1000.0

    # Sidon positions (triangular numbers)
    large_pos = generate_sidon_triangular(n)

    # Number of large entries
    K = large_pos.size

    # Small constant value, jittered a bit (factor in [0.8, 1.2])
    sqrt_n = max(1, int(np.sqrt(n)))
    base_s = V / sqrt_n
    jitter = rng.uniform(0.8, 1.2)
    s = np.clip(base_s * jitter, 0.0, V)

    # Build the sequence
    seq = np.full(n, s, dtype=np.float64)
    seq[large_pos] = V
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def perturb_sequence(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a slightly modified copy of ``seq``.

    Several types of cheap moves are used:
      * block scaling by a log‑normal factor,
      * swapping two entries,
      * adding a small Gaussian noise to a single entry,
      * optional global symmetrisation.
    """
    candidate = seq.copy()
    n = candidate.size

    move_type = rng.integers(0, 4)
    if move_type == 0:                     # block scaling
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        candidate[start:stop] *= factor
    elif move_type == 1:                   # swap two entries
        i, j = rng.choice(n, size=2, replace=False)
        candidate[i], candidate[j] = candidate[j], candidate[i]
    elif move_type == 2:                   # single entry tweak
        idx = rng.integers(0, n)
        delta = float(rng.normal(0.0, 8.0))
        candidate[idx] = np.clip(candidate[idx] + delta, 0.0, 1000.0)
    else:                                  # occasional symmetrisation
        if rng.random() < 0.35:
            candidate = 0.5 * (candidate + candidate[::-1])

    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises `evaluate_sequence`.  The method starts from a deterministic
    Sidon‑based construction and then performs a fast local search for a
    limited time (≈budget_s seconds).  The best sequence found is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the sequence we will work with.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        # Fallback: a modest length if the global variable is missing/invalid.
        n = 256

    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initial deterministic candidate.
    # ------------------------------------------------------------------
    best_seq = build_candidate(n, rng)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main search loop.
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # Occasionally restart from a fresh deterministic candidate with a
        # different jitter factor, to avoid stagnation.
        if rng.random() < 0.07:
            cand = build_candidate(n, rng)
        else:
            cand = perturb_sequence(best_seq, rng)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand
            print(f"[{iteration:06d}] new best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback guarantee (should never be needed).
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        print("[fallback] using uniform sequence of ones.")
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"[final] best score = {best_score:.6f}")
    # Return a plain Python list of floats as required by the evaluator.
    return [float(x) for x in best_seq]
