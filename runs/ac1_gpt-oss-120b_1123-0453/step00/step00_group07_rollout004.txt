```
import cvxpy as cp

def solve_lp(f, M, active_ks, upper=1000.0):
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0, g <= upper]
    for k in active_ks:
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # f slice
        f_slice = f[i_start:i_end+1][::-1]  # reversed
        g_slice = g[i_start:i_end+1]
        constraints.append(cp.sum(cp.multiply(f_slice, g_slice)) <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-6, reltol=1e-6)
    if g.value is None:
        return None
    return np.array(g.value, dtype=np.float64)
```

"""...
Algorithm:
- start from an existing sequence (height_sequence_1) if present, else uniform.
- Iteratively improve using a linear‑programming “boosting’’ step:
    • Compute the current autocorrelation peak M and the set of indices where it is attained.
    • Solve the LP max Σ g_i subject to g_i≥0, g_i≤1000 and (f*g)[k]≤M for the active k.
      The LP finds a new non‑negative sequence g that can be added without violating the current peak.
    • Mix the old sequence f and the LP solution g (average or smaller mixing weight) and keep the
      mixture if it improves the evaluation score.
- If the LP step does not give improvement, fall back to a cheap block perturbation (the
  original method) to keep escaping local minima.
- The loop runs until the allotted time expires (≈budget_s‑3 s). The best sequence found is
  returned as a Python list of floats.
"""

import numpy as np
import time

try:
    import cvxpy as cp
    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def make_block_perturbation(base, rng):
    """Return one O(n)-memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def _solve_lp_for_g(f, M, active_ks, upper=1000.0, max_iters=2000):
    """
    Solve  max Σ g_i  subject to
          0 ≤ g_i ≤ upper,
          (f * g)[k] ≤ M  for all k in ``active_ks``.
    Returns a NumPy array with the solution or ``None`` on failure.
    """
    n = f.shape[0]
    g = cp.Variable(n)

    constraints = [g >= 0, g <= upper]

    # Pre‑compute the constant slices for each active shift.
    for k in active_ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # Slice length
        length = i_end - i_start + 1
        if length <= 0:
            continue
        # f slice reversed because convolution uses f[i] * g[k-i]
        f_slice = f[i_start:i_end + 1][::-1]
        g_slice = g[i_start:i_end + 1]
        # Linear constraint Σ f_i * g_{k-i} ≤ M
        constraints.append(cp.sum(cp.multiply(cp.Constant(f_slice), g_slice)) <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=max_iters, verbose=False)
    except Exception:  # pragma: no cover
        # ECOS may raise on bad data; try a different solver if available.
        try:
            prob.solve(solver=cp.SCS, max_iters=max_iters, verbose=False)
        except Exception:
            return None

    if g.value is None:
        return None
    # Clip to the allowed range just in case of small numerical violations.
    return np.clip(np.array(g.value, dtype=np.float64), 0.0, upper)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine runs for roughly ``budget_s`` seconds
    (minus a small safety margin) and returns the best sequence found as a list
    of Python ``float`` objects.
    """
    rng = np.random.default_rng(seed)
    # leave a safety margin for the runner
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initial state – use the globally supplied construction if it looks sane.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    best_n = best.shape[0]
    print(f"Initial length {best_n}, score {best_score:.6f}")

    iteration = 0
    # main optimisation loop
    while time.monotonic() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        # 1) Compute current autocorrelation peak M and a modest active set.
        # ------------------------------------------------------------------
        conv_f = np.convolve(best, best)
        M = conv_f.max()
        # Choose shifts whose value is ≥ 0.995 · M (tunable).
        # Keep at most 150 shifts to keep the LP small.
        active = np.where(conv_f >= M * 0.995)[0]
        if active.size > 150:
            # keep the top values instead of by index order
            top_idx = np.argsort(-conv_f)[:150]
            active = np.sort(top_idx)

        # ------------------------------------------------------------------
        # 2) Try the LP “boost’’ step if a LP solver is available.
        # ------------------------------------------------------------------
        new_candidate = None
        if _CVX_AVAILABLE:
            g = _solve_lp_for_g(best, M, active, upper=1000.0, max_iters=1000)
            if g is not None:
                # Mix the old and the LP solution.
                # Several mixing coefficients are tried; the first that yields
                # an improvement is kept.
                for t in (0.5, 0.3, 0.15, 0.07, 0.03):
                    mixed = (1.0 - t) * best + t * g
                    mixed = np.clip(mixed, 0.0, 1000.0)
                    score = float(evaluate_sequence(mixed.tolist()))
                    if score < best_score:
                        new_candidate = mixed
                        best_score = score
                        best = mixed
                        print(f"Iter {iteration:4d} – LP mix t={t:.2f} improved score to {best_score:.6f}")
                        break
        # ------------------------------------------------------------------
        # 3) If the LP path did not improve, fall back to cheap block
        #    perturbations (the original method).  Accept only improvements.
        # ------------------------------------------------------------------
        if new_candidate is None:
            cand = make_block_perturbation(best, rng)
            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"Iter {iteration:4d} – block perturb improved score to {best_score:.6f}")

        # ------------------------------------------------------------------
        # 4) Occasionally inject a random “restart’’ to avoid stagnation.
        # ------------------------------------------------------------------
        if iteration % 30 == 0 and rng.random() < 0.25:
            # randomize a fraction of the entries multiplicatively
            noise = np.exp(rng.normal(0.0, 0.25, size=best.shape))
            restart = np.clip(best * noise, 0.0, 1000.0)
            restart_score = float(evaluate_sequence(restart.tolist()))
            if restart_score < best_score:
                best = restart
                best_score = restart_score
                print(f"Iter {iteration:4d} – restart perturb improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score {best_score:.6f}")
    return [float(x) for x in best]

"""
Search for a non‑negative step‑function (height sequence) that minimises
``evaluate_sequence``.

Approach
--------
1. Start from the global construction ``height_sequence_1`` (or a uniform fallback).
2. Repeatedly:
   * Compute the current autocorrelation peak `M` and keep a modest
     “active’’ set of shifts where the convolution is close to `M`.
   * Solve the linear programme
        maximise   Σ g_i
        subject to 0 ≤ g_i ≤ 1000
                   (f * g)[k] ≤ M   for every active shift k
     where `f` is the current sequence.
     The LP adds as much total mass as possible without increasing the
     already‑maximal convolution value.
   * Scale the LP solution down (if necessary) so that its total sum does not
     exceed the current sum of `f`.  This guarantees the constraint stays valid.
   * Mix the old sequence and the LP solution (averaging or a small convex
     combination).  If the mixture yields a lower evaluation score it is kept.
   * If the LP step did not improve the score, fall back to a cheap
     block‑perturbation (the method used in the reference implementation).
   * Occasionally inject a random multiplicative perturbation to escape
     possible plateaus.
3. Stop after the allotted time (≈ budget_s seconds) and return the best
   sequence found as a list of Python floats.

The algorithm is a practical realisation of the “boosting’’ step described
in the reference text: the LP increases the sum while keeping the maximal
convolution bounded, and a convex combination reduces that bound.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Optional cvxpy import – the LP step is the core of the improvement.
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _CVX_AVAILABLE = True
except Exception:                     # pragma: no cover
    _CVX_AVAILABLE = False


def make_block_perturbation(base, rng):
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _solve_lp_for_g(f, M, active_ks, upper=1000.0, max_iters=1500):
    """
    Solve the linear programme

        maximise   Σ g_i
        subject to g_i ∈ [0, upper]
                   (f * g)[k] ≤ M   for each k ∈ active_ks

    and return the optimal vector `g` (as a NumPy array) or ``None`` on
    failure.
    """
    n = f.shape[0]
    g = cp.Variable(n)

    constraints = [g >= 0, g <= upper]

    # One linear constraint per active shift.
    for k in active_ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        length = i_end - i_start + 1
        if length <= 0:
            continue
        # f slice is reversed because (f * g)[k] = Σ_i f_i * g_{k-i}.
        f_slice = f[i_start:i_end + 1][::-1]
        g_slice = g[i_start:i_end + 1]
        constraints.append(
            cp.sum(cp.multiply(cp.Constant(f_slice), g_slice)) <= M
        )

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=max_iters, verbose=False)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, max_iters=max_iters, verbose=False)
        except Exception:
            return None

    if g.value is None:
        return None
    return np.clip(np.array(g.value, dtype=np.float64), 0.0, upper)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a list of floats representing the best height sequence found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    # Keep a few seconds margin for the runner.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------
    #  Initialise from the global construction if it looks sane.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():

        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:

    parent = np.clip(parent, 0.0, 1000.0)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Start: length={best.size}, score={best_score:.6f}")

    iteration = 0

    # --------------------------------------------------------------
    #  Main optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ----- 1) current autocorrelation and active set -----------------
        conv_f = np.convolve(best, best)
        M = conv_f.max()
        # Shifts whose value is at least 99.5 % of the peak.
        active = np.where(conv_f >= M * 0.995)[0]
        if active.size > 150:                     # keep the LP small
            active = np.argsort(-conv_f)[:150]
            active = np.sort(active)

        improved = False

        # ----- 2) LP “boost” step (if cvxpy is available) ---------------
        if _CVX_AVAILABLE:
            g_raw = _solve_lp_for_g(best, M, active, upper=1000.0)
            if g_raw is not None:
                sum_f = best.sum()
                sum_g = g_raw.sum()
                # Scale down if needed so that the constraint (f·g) ≤ M stays valid.
                if sum_g > 0:
                    scale = 1.0
                    if sum_g > sum_f:
                        scale = sum_f / sum_g
                    g = g_raw * scale
                else:
                    g = g_raw

                # Try several mixing coefficients.
                for t in (0.5, 0.3, 0.15, 0.07, 0.03):
                    mixed = (1.0 - t) * best + t * g
                    mixed = np.clip(mixed, 0.0, 1000.0)

                    score = float(evaluate_sequence(mixed.tolist()))
                    if score < best_score:
                        best = mixed
                        best_score = score
                        improved = True
                        print(
                            f"Iter {iteration:4d} – LP mix t={t:.2f} "
                            f"score={best_score:.6f}"
                        )
                        break

        # ----- 3) fallback: cheap block perturbation --------------------
        if not improved:
            cand = make_block_perturbation(best, rng)
            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                improved = True
                print(
                    f"Iter {iteration:4d} – block perturb improved "
                    f"score={best_score:.6f}"
                )

        # ----- 4) occasional random “restart’’ to escape plateaus ------
        if iteration % 30 == 0 and rng.random() < 0.25:
            noise = np.exp(rng.normal(0.0, 0.25, size=best.shape))
            restart = np.clip(best * noise, 0.0, 1000.0)
            restart_score = float(evaluate_sequence(restart.tolist()))
            if restart_score < best_score:
                best = restart
                best_score = restart_score
                print(
                    f"Iter {iteration:4d} – restart perturb improved "
                    f"score={best_score:.6f}"
                )

        # Early exit if we have already beaten the target (optional).
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration}.")
            break

    print(f"Finished after {iteration} iterations, best score {best_score:.6f}")
    return [float(x) for x in best]
