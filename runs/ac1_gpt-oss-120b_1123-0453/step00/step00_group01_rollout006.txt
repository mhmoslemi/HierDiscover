"""
Propose a candidate step‑function sequence that minimizes
evaluate_sequence(seq).

The algorithm combines three ideas:

1. **Random sparse constructions** –  Choose a random subset of indices
   (size ≈ n**α with 0.5 ≤ α ≤ 0.9) and give each selected position a
   constant height so that the total ℓ₁‑norm equals √(2 n).  For
   α ≈ 0.75 the number of chosen positions is ≈ n⁰·⁷⁵, which yields a
   convolution maximum close to the average value and drives the score
   towards the target ≈ 1.5.

2. **Exponential (dense) constructions** –  Sample i.i.d. exponential
   random variables, normalise them to the same ℓ₁‑norm and evaluate.
   Occasionally this yields a better (more “smooth”) distribution.

3. **Local refinement** –  Starting from the best sequence found so far,
   perform cheap block‑wise perturbations (`make_block_perturbation`) and
   accept improvements.  This mimics a simple hill‑climbing step.

The loop runs until the allotted budget (default ≈ 1100 s) is exhausted.
All intermediate scores are printed to aid debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: cheap block perturbation (the same idea as the reference code)
def make_block_perturbation(base, rng):
    """Return a single O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # choose a block width between n/200 and n/12 (inclusive)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal noise on the whole block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional cheap symmetrisation (helps sometimes)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

    Parameters

    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds; the function stops a few
        seconds before this limit.
    **kwargs : dict
        Ignored – kept for compatibility with the testing harness.

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """

    # but fixing it makes the random constructions easier).
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Best solution seen so far (start from the uniform baseline)
    best = np.full(n, target_sum / n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial (uniform) score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to build a binary‑mask candidate.
    def binary_candidate(alpha):
        """Random binary mask with ~n**alpha ones (all ones have equal height)."""
        m = max(1, int(round(n ** alpha)))
        # draw `m` distinct indices without replacement
        idx = rng.choice(n, size=m, replace=False)
        cand = np.zeros(n, dtype=np.float64)
        height = target_sum / m                     # makes ℓ₁‑norm = target_sum
        cand[idx] = height
        return cand

    # Helper for an exponential (dense) candidate.
    def exponential_candidate():
        """Random positive vector (Exponential), normalised to target_sum."""
        vals = rng.exponential(scale=1.0, size=n)
        # avoid the pathological case of all zeros
        if vals.sum() == 0.0:
            vals[0] = 1.0
        cand = vals / vals.sum() * target_sum
        return cand

    # ------------------------------------------------------------------
    # Main search loop
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        if iteration % 50 == 0 and iteration > 0:

            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:

            # binary mask – pick exponent α uniformly in a useful range
            alpha = rng.uniform(0.55, 0.90)
            cand = binary_candidate(alpha)
        else:

            cand = exponential_candidate()

            cand *= jitter

        cand = np.clip(cand, 0.0, 1000.0)

        # Evaluate

            # but we guard against pathological numerical issues.
            continue

        # Keep the best.

        if best_score <= 1.503:

            break

    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    # Return as a plain Python list (the evaluator expects that).
    return [float(x) for x in best]
