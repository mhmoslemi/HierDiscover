def project_onto_simplex(v):
    """Project vector v onto the probability simplex (non-negative, sum=1)."""
    # v may contain negative values.
    # Implementation based on algorithm from reference (Duchi et al.)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):

    u = np.sort(v)[::-1]

        return np.maximum(v, 0)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‐flattening search for a step–function that minimizes
evaluate_sequence().

The objective 2·n·max(conv)/sum² is invariant to scaling, therefore we
optimise a probability vector w (∑w=1, w≥0).  The convolution w*w is the
distribution of the sum of two independent draws from w.  We minimise the
peak of this distribution by penalising any entry of the convolution that
exceeds the global average 1/(2n‑1).  The penalty is quadratic, giving a
smooth gradient that can be computed by a single extra convolution.
After each gradient step we project back onto the probability simplex.
Occasionally we inject a random “block perturbation” to escape possible
plateaus.  The best vector found during the allotted time is returned as a
list of non‑negative heights (the scale does not matter – we keep the sum
≈1 so the evaluator works without clipping).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0,  Σ_i x_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    u = np.sort(v)[::-1]                 # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1          # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform simplex point
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_block_perturbation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    A cheap O(n) perturbation used for occasional exploration:
    select a random contiguous block and scale it by a log‑normal factor.
    With probability 0.35 we also symmetrise the sequence.
    """
    n = seq.size
    # width is between n/200 and n/12 (as in the reference code)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    out = seq.copy()
    out[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        out = 0.5 * (out + out[::-1])
    out = np.clip(out, 0.0, 1000.0)
    # re‑normalise to keep the sum ≈ 1 (scaling does not affect the score)
    total = out.sum()
    if total == 0.0:
        out = np.full_like(out, 1.0 / n)
    else:
        out = out / total
    return out

    target_avg = 1.0 / (2 * n - 1)

        best_w = best_w / best_w.sum()

    print(f"[final] iterations={iteration}, best score={best_score:.6f}")
    return best_w.tolist()
