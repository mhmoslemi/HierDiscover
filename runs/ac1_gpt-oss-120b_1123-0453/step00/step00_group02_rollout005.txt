def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    # Start from known sequence if appropriate
    parent = np.asarray(globals().get('height_sequence_1', None), dtype=np.float64)
    # Validate parent
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        n = 8192
        a = np.full(n, 1.0/n, dtype=np.float64)
    else:
        # Use its length if not too small else pad.
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size < 4096:
            n = 4096
        else:
            n = max(parent.size, 4096)
        # pad or truncate to n
        if parent.size < n:
            a = np.zeros(n, dtype=np.float64)
            a[:parent.size] = parent
        else:
            a = parent[:n].copy()
        # Normalize to sum = 1
        s = a.sum()
        if s <= 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s
    # ensure sum 1
    a = a / a.sum()
    best = a.copy()
    # Compute initial score using own evaluation
    size = 1 << ((2*n-1).bit_length())
    def compute_score(vec):
        s = vec.sum()
        if s < 1e-12:
            return np.inf
        # compute convolution using FFT for speed
        f_fft = np.fft.rfft(vec, size)
        conv = np.fft.irfft(f_fft * f_fft, n=size)[:2*n-1]
        maxb = conv.max()
        return float(2.0 * n * maxb / (s * s))
    best_score = compute_score(best)
    # Precompute uniform vector for smoothing
    uniform = np.full(n, 1.0/n, dtype=np.float64)

    # Helper: random block perturbation similar to given, for occasional jumps
    def random_block(vec):
        # choose block width between n//200 and n//12
        lo = max(1, n // 200)
        hi = max(2, n // 12)
        width = int(rng.integers(lo, hi+1))
        start = int(rng.integers(0, n - width + 1))
        stop = start + width
        new = vec.copy()
        factor = np.exp(rng.normal(0.0, 0.12))
        new[start:stop] *= factor
        if rng.random() < 0.35:
            new = 0.5 * (new + new[::-1])
        new = np.clip(new, 0.0, 1000.0)
        # renormalize to sum 1
        if new.sum() == 0:
            new = uniform.copy()
        else:
            new /= new.sum()
        return new

    # Main loop
    iter_cnt = 0
    last_improve_iter = 0
    while time.monotonic() < deadline:
        # compute convolution and gradient
        a_fft = np.fft.rfft(a, size)
        conv_full = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        maxb = conv_full.max()
        # current score
        score = float(2.0 * n * maxb / (a.sum() * a.sum()))
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"Iter {iter_cnt}: new best score {best_score:.6f}")
            last_improve_iter = iter_cnt

        # Compute average
        avg = (a.sum() ** 2) / (2 * n - 1)  # equals 1/(2n-1) if sum=1
        r = conv_full - avg
        # gradient of variance of convolution
        r_fft = np.fft.rfft(r, size)
        grad_full = np.fft.irfft(r_fft * a_fft, n=size)[:n]  # conv(r, a) length n
        g = 4.0 * grad_full

        # adaptive step size
        # scale step so max |alpha * g_i| ~ 0.1 maybe
        max_abs_g = np.max(np.abs(g))
        if max_abs_g == 0:
            step = 0.0
        else:
            step = 0.1 / max_abs_g

        a = a - step * g
        # keep nonnegative
        a = np.clip(a, 0.0, None)
        # small mixing with uniform to avoid collapse
        eps = 0.005
        a = (1 - eps) * a + eps * uniform
        # renormalize
        s = a.sum()
        if s == 0:
            a = uniform.copy()
        else:
            a = a / s

        # occasional random jump to escape stagnation
        if iter_cnt % 200 == 0 and iter_cnt > 0:
            # if no improvement in last 500 iter, try random perturb
            if iter_cnt - last_improve_iter > 500:
                a = random_block(a)

        iter_cnt += 1

    # Return best as list of floats (scale back to original magnitude? keep as is)
    result = [float(x) for x in best]
    print(f"Finished after {iter_cnt} iterations, best score {best_score:.6f}")
    return result

"""
Propose a step‑function (non‑negative sequence) that minimizes the
evaluate_sequence metric.

The algorithm works on a normalized probability vector `a` (sum = 1) of
length `n`.  It repeatedly

1. computes the auto‑convolution `b = a*a` by FFT,
2. measures the deviation `r = b - avg` where `avg = sum(a)^2/(2n‑1)`,
3. forms the gradient of the variance of `b` (which drives the
   convolution towards a flat shape) as `g = 4 * (r * a)`,
4. takes a small projected gradient step,
5. enforces non‑negativity, mixes a tiny amount of the uniform vector
   (to keep the iterate away from the boundary), and re‑normalises.

Occasionally a random block perturbation (the same kind used in the
reference code) is applied to escape stagnation.  The best sequence
encountered (according to the exact metric) is remembered and returned.

The procedure respects the 1000‑second budget, works with any starting
sequence supplied in the global `height_sequence_1`, and never uses
closures or lambdas.
"""

import numpy as np
import time

def _evaluate_metric(vec: np.ndarray) -> float:
    """
    Compute the metric used by evaluate_sequence for a given vector.
    The vector is assumed to be a list/array of non‑negative numbers.
    """
    n = vec.size
    s = vec.sum()
    if s < 0.01:
        return float("inf")
    # size for FFT: next power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    f_fft = np.fft.rfft(vec, size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def random_block_perturbation(base: np.ndarray,
                              rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative perturbation of ``base`` by scaling a random
    contiguous block and optionally symmetrising it.
    The result is re‑normalised to have sum = 1.
    """
    n = base.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    cand = base.copy()
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)

    total = cand.sum()
    if total == 0.0:
        cand = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cand = cand / total
    return cand


def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs) -> list[float]:
    """
    Search for a sequence with a low evaluate_sequence score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate wall‑clock budget (seconds).  The function stops a few
        seconds before the limit to avoid termination.

    Returns
    -------
    list[float]
        The best coefficient sequence found (non‑negative, sum > 0.01).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise the search vector `a` (normalise to sum = 1)
    # ------------------------------------------------------------------
    global height_sequence_1
    parent = np.asarray(
        globals().get("height_sequence_1", []),
        dtype=np.float64,
    )
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        # no sensible starting point – fall back to uniform
        n = 8192
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # keep the length of the supplied construction (or pad to a useful size)
        n = max(parent.size, 4096)
        if parent.size < n:
            a = np.zeros(n, dtype=np.float64)
            a[: parent.size] = parent
        else:
            a = parent[:n].copy()
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

    # uniform vector for occasional smoothing
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # pre‑compute the FFT size (next power of two >= 2*n‑1)
    fft_size = 1 << ((2 * n - 1).bit_length())

    # keep the best sequence seen so far
    best = a.copy()
    best_score = _evaluate_metric(best)
    print(f"Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    iter_cnt = 0
    last_improve = 0
    eps_mix = 0.005         # strength of uniform mixing
    while time.monotonic() < deadline:
        # ----- convolution and current metric ---------------------------------
        a_fft = np.fft.rfft(a, fft_size)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_size)[: 2 * n - 1]
        max_b = conv_full.max()
        cur_score = float(2.0 * n * max_b / (a.sum() * a.sum()))
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"Iter {iter_cnt:6d} – new best: {best_score:.6f}")
            last_improve = iter_cnt

        # ----- gradient of the variance of the convolution --------------------
        # average value of the convolution (exact for sum(a)=1)
        avg = (a.sum() ** 2) / (2 * n - 1)          # = 1/(2n‑1) when sum=1
        r = conv_full - avg                         # deviation from the average

        r_fft = np.fft.rfft(r, fft_size)
        # convolution of r with a  (length n)
        grad_full = np.fft.irfft(r_fft * a_fft, n=fft_size)[:n]
        grad = 4.0 * grad_full                      # gradient of variance

        # ----- adaptive projected gradient step -------------------------------
        max_abs = np.max(np.abs(grad))
        if max_abs > 0.0:
            step = 0.1 / max_abs                    # keep the update modest
        else:
            step = 0.0
        a = a - step * grad
        a = np.clip(a, 0.0, None)                   # enforce non‑negativity

        # ----- small mixing with the uniform distribution (helps stability) -
        a = (1.0 - eps_mix) * a + eps_mix * uniform

        # ----- re‑normalise to stay on the simplex ---------------------------
        total = a.sum()
        if total == 0.0:
            a = uniform.copy()
        else:
            a = a / total

        # ----- occasional random jump if we have stagnated -------------------
        if iter_cnt % 200 == 0 and iter_cnt > 0:
            if iter_cnt - last_improve > 500:
                a = random_block_perturbation(a, rng)

        iter_cnt += 1

    print(f"Finished after {iter_cnt} iterations – best score {best_score:.6f}")
    # convert to plain Python floats for the evaluator
    return [float(x) for x in best]
