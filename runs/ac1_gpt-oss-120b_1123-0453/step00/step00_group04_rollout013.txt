import numpy as np
import time

def make_block_perturbation(base, rng):
    """
    Return a cheap O(n) perturbation of ``base`` by scaling a random block
    and optionally symmetrising.  Used for local refinement.
    """
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce hard limits required by the evaluator
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step‑function (a list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The core idea is to explore power‑law shaped
    sequences a_i = (i+offset)^{-alpha} – these give a near‑flat self‑convolution
    and can reach values well below the current bound of 2.0.  Random parameters
    (length, exponent, offset) are sampled until the time budget expires; the
    best sequence is kept.  Occasionally a cheap block‑perturbation is applied
    to the current best to squeeze out further improvements.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the caller
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # start from the provided construction if it looks sane
    try:
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"initial score {best_score:.6f} (length {best.size})")

    # --------------------------------------------------------------
    # Parameter‑space search: power‑law sequences
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # length – a few thousand gives a good trade‑off between quality
        # and the O(n²) convolution cost.
        n = int(rng.integers(2000, 8001))               # 2 000 ≤ n ≤ 8 000
        # exponent > 0.5 yields a decreasing shape with a flat convolution.
        alpha = rng.uniform(0.55, 0.85)                  # 0.55 ≤ α ≤ 0.85
        # offset prevents the first element from dominating the max‑convolution.
        offset = int(rng.integers(1, 31))               # 1 ≤ offset ≤ 30

        # a_i = (i + offset)^{-alpha}
        idx = np.arange(n, dtype=np.float64) + offset
        candidate = idx ** (-alpha)

        # tiny random multiplicative noise – helps escape plateaus
        if rng.random() < 0.25:
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            candidate *= noise

        # enforce the hard limits of the evaluator
        candidate = np.clip(candidate, 0.0, 1000.0)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"new best {best_score:.6f}  n={n}  alpha={alpha:.4f}  off={offset}")

        # occasionally run a few cheap local moves on the current best
        if best_score < 1.6 and rng.random() < 0.07:

            if np.isfinite(local_score) and local_score < best_score:
                best = local
                best_score = local_score
                print(f"local improvement {best_score:.6f}")

    # final sanity clip (the evaluator repeats the clipping anyway)
    best = np.clip(best, 0.0, 1000.0)
    print(f"final best score {best_score:.6f}  length {best.size}")
    return [float(x) for x in best]
