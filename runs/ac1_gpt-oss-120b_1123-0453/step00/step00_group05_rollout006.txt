""" 
search for a step–function (non‑negative sequence) that minimizes
evaluate_sequence().  

The algorithm consists of three stages  

1. **Random seeding** – generate many independent random vectors, optionally
   smooth them and optionally enforce symmetry.  Every candidate is evaluated
   with evaluate_sequence() and the best one is kept.

2. **Local, gradient‑guided refinement** – for the current best sequence the
   convolution max is located.  A weight for each position is computed as the
   total contribution of that entry to every maximal convolution term.  Mass is
   moved from a high‑weight (bad) index to a low‑weight (good) index; the move
   size is a random fraction of the source entry.  The move is accepted if it
   improves the score, otherwise it may be accepted with a small probability
   (simulated annealing).

3. **Exploratory block perturbations** – occasionally a “block perturbation”
   (the routine used in the reference implementation) is applied to escape
   local minima.

The search runs until the allocated time budget (``budget_s``) is exhausted
(leaving a few seconds margin for the final return).  All numbers are kept in
the interval [0, 1000] and the returned sequence is a plain list of floats. 
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``.

    A random contiguous block is multiplied by exp(N(0,0.12)).
    With probability 0.35 the result is symmetrised.
    The output is clipped to the interval [0,1000].
    """
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # width of the block is chosen between n/200 and n/12  (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def compute_weight(seq, conv, max_val):
    """Compute a simple sub‑gradient of the max‑convolution.

    For each index i, weight[i] = sum_{k : conv[k]==max_val} seq[k-i]
    (terms outside the array are ignored).  This equals the amount by which
    decreasing seq[i] would reduce the maximal convolution value.
    """
    n = seq.size
    mask = conv == max_val
    if not np.any(mask):
        return np.zeros_like(seq)
    max_pos = np.where(mask)[0]
    weight = np.zeros(n, dtype=np.float64)

    for k in max_pos:
        # indices i that participate in convolution term k
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i = np.arange(start, end + 1, dtype=np.int64)
        j = k - i
        weight[i] += seq[j]
    return weight


def random_candidate(n, rng):
    """Produce a fresh random sequence of length ``n``."""
    # start from a plain uniform draw
    cand = rng.random(n)

    # optionally smooth with a short moving average
    if rng.random() < 0.5:
        width = int(rng.integers(1, 20))
        kernel = np.ones(width, dtype=np.float64) / float(width)
        cand = np.convolve(cand, kernel, mode='same')

    # optional symmetrisation (helps to keep the convolution shape symmetric)
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])

    return np.clip(cand, 0.0, 1000.0)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a sequence of non‑negative heights with low evaluate_sequence().

    The routine respects the time budget (``budget_s`` seconds).  The returned
    object is a plain list of floats suitable for evaluate_sequence().
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the final return
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine problem size – prefer the historic sequence length if
    # it exists, otherwise fall back to a modest default.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base_seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        n = base_seq.size
        if n == 0:
            n = 256
            base_seq = np.ones(n, dtype=np.float64)
    else:
        n = 256
        base_seq = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Phase 1 – massive random sampling
    # ------------------------------------------------------------------
    best = random_candidate(n, rng)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] random start score = {best_score:.6f}")

    # Use most of the allocated time for this phase, keep a little for
    # later refinement.
    phase1_deadline = deadline - 30.0  # reserve ~30 s for phase 2
    while time.monotonic() < phase1_deadline:
        cand = random_candidate(n, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[rand] new best = {best_score:.6f}  (t={time.monotonic():.1f}s)")

    # ------------------------------------------------------------------
    # Phase 2 – local, gradient‑guided search with occasional block moves
    # ------------------------------------------------------------------
    # Simple simulated‑annealing temperature schedule
    temperature = max(best_score * 0.02, 1e-6)

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 2a) gradient‑guided transfer of mass
        # --------------------------------------------------------------
        conv = np.convolve(best, best)
        max_val = conv.max()
        weight = compute_weight(best, conv, max_val)

        # indices sorted by weight (high → low)
        hi_order = np.argsort(-weight)
        lo_order = np.argsort(weight)

        # find a source with a non‑zero amount
        src = None
        for i in hi_order:
            if best[i] > 1e-9:
                src = i
                break
        if src is None:
            break  # nothing to move

        # find a destination different from source
        dst = lo_order[0] if lo_order[0] != src else lo_order[1]

        # move a random fraction of the source amount
        frac = rng.uniform(0.05, 0.20)          # 5 % – 20 %
        delta = best[src] * frac
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = float(evaluate_sequence(cand.tolist()))
        accept = False
        if np.isfinite(cand_score):
            if cand_score < best_score:
                accept = True
            else:
                # Metropolis acceptance
                prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
                if rng.random() < prob:
                    accept = True

        if accept:
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[move] improved to {best_score:.6f} (t={time.monotonic():.1f}s)")
            else:
                # even if worse we keep the candidate for the SA step
                best = cand

        # --------------------------------------------------------------
        # 2b) occasional block perturbations (helps escape traps)
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            cand_blk = make_block_perturbation(best, rng)
            blk_score = float(evaluate_sequence(cand_blk.tolist()))
            if np.isfinite(blk_score) and blk_score < best_score:
                best = cand_blk
                best_score = blk_score
                print(f"[blk] new best {best_score:.6f} (t={time.monotonic():.1f}s)")

        # cool the temperature slowly
        temperature *= 0.9995
        # avoid temperature becoming exactly zero
        temperature = max(temperature, 1e-9)

    print(f"[final] best score = {best_score:.6f}")
    return [float(x) for x in best]
