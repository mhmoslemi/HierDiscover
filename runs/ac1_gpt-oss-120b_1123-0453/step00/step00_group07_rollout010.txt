"""
Propose a candidate sequence that improves the current bound (2.0).

Algorithm
---------
1. Choose a moderate length (≤ 512) so that a dense linear program can be
   solved quickly.  If the global variable ``height_sequence_1`` is available,
   the first 512 entries are used as a starting point; otherwise a uniform
   vector is taken.

2. Normalise the vector to L1‑norm 1 (the evaluation is scale‑invariant).

3. Repeatedly perform a *linear‑programming improvement step* :

   * Let ``M = max(conv(f, f))``.
   * Construct the matrix ``A`` where ``(A @ g)[k] = (f * g)[k]``.
     ``A`` has shape ``(2n-1, n)`` and can be filled by
     ``A[j:j+n, j] = f`` for every column ``j``.
   * Solve the LP

        maximise   Σ g_i
        subject to A @ g ≤ M  (element‑wise)
                   g_i ≥ 0

     (implemented with ``scipy.optimize.linprog``).  The solution ``g``
     has the largest possible L1‑norm while keeping the convolution with
     the current ``f`` below the present maximal value ``M``.

   * Rescale ``g`` to unit L1‑norm and replace ``f`` by a convex combination
     ``(1‑t)·f + t·g`` (``t`` = 0.2).  If ``g`` already yields a smaller
     ``max(conv(g,g))`` we take ``f = g`` directly – this accelerates the
     flattening of the convolution.

4. Keep the best vector seen during the search, stop when the time budget
   expires or when the evaluation drops below the target (≈1.503).

5. Finally scale the best vector to a moderate amplitude (≈500) and clip
   it to the allowed range ``[0, 1000]`` before returning.  Scaling does
   not affect the score because the evaluation function is homogeneous.

The whole routine runs entirely in memory, uses only NumPy and SciPy, and
fits comfortably into the 1000 s time limit.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    _has_linprog = True
except Exception:
    _has_linprog = False  # fallback to simple random search if needed


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights (floats) that aim to minimise
    ``evaluate_sequence``.  The function respects the time budget, prints
    progress information and uses a small linear‑programming based local
    optimisation loop.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Choose a working length (small enough for a dense LP)
    # ------------------------------------------------------------------
    global height_sequence_1
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)

    # We cap the length at 512 to keep the LP cheap.
    max_len = 512
    if base.size >= max_len:
        f = base[:max_len].copy()
    else:
        # Pad with a small constant to reach max_len
        pad = np.full(max_len - base.size, 1e-3, dtype=np.float64)
        f = np.concatenate([base, pad])
    n = f.size

    # Clip to the limits enforced by ``evaluate_sequence`` and normalise.
    f = np.clip(f, 0.0, 1000.0)
    if f.sum() == 0.0:
        f = np.ones_like(f)
    f = f / f.sum()                     # L1‑norm = 1  (scale‑invariant)

    # ------------------------------------------------------------------
    # Helper to evaluate the current vector.
    # ------------------------------------------------------------------
    def score_vec(v):
        return float(evaluate_sequence(v.tolist()))

    best_f = f.copy()
    best_score = score_vec(best_f)
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Main optimisation loop (LP‑based)
    # ------------------------------------------------------------------
    mix_t = 0.2          # convex‑mixing weight
    it = 0
    while time.monotonic() < deadline and it < 200:
        it += 1

        # current maximal convolution coefficient
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # Build the convolution matrix A such that (A @ g) = f * g
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f

        b_ub = np.full(rows, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  (linprog minimises)

        if not _has_linprog:
            # Simple random fallback – perturb a random block.
            cand = f.copy()
            width = max(1, n // 20)
            start = rng.integers(0, n - width + 1)
            cand[start:start + width] *= np.exp(rng.normal(0.0, 0.15))
            cand = np.clip(cand, 0.0, 1000.0)
            cand /= cand.sum()
            new_f = (1 - mix_t) * f + mix_t * cand

            res = linprog(c, A_ub=A, b_ub=b_ub,

            if not res.success:

            if g.sum() == 0.0:

            conv_gg = np.convolve(g, g)
            if conv_gg.max() < M:

            else:

        new_f = np.clip(new_f, 0.0, 1000.0)

        if best_score < 1.503:

            break

        f = new_f

    # ------------------------------------------------------------------
    # 3. Post‑process: bring values into a comfortable numeric range.
    #    Scaling does not change the score because ``evaluate_sequence`` is
    #    homogeneous (it only depends on ratios).
    # ------------------------------------------------------------------
    best_f = np.clip(best_f * 500.0, 0.0, 1000.0)

    print(f"[final] best score = {best_score:.6f}, length = {best_f.size}")
    return best_f.tolist()
