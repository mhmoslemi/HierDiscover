"""
Propose a sequence of non‑negative heights that minimizes the given
`evaluate_sequence` function.

The algorithm works in two stages:

1. **Linear‑programming blending**
   - Start from a random simplex vector `f` of length `n` (the sum is
     normalised to √(2 n) because scaling does not change the objective).
   - Compute `M = max(f * f)`, i.e. the largest coefficient of the convolution
     of `f` with itself.
   - Build the convolution matrix `A` such that `A @ g = f * g` (full
     convolution, linear in `g`).
   - Solve the linear programme  
        maximise   Σ g_i               (increase the total mass)  
        subject to A @ g ≤ M   and   g ≥ 0
     (`scipy.optimize.linprog` with the HiGHS solver).  
     The LP returns an extreme point `g` that respects the current bound `M`
     but is typically very sparse and non‑uniform.
   - Blend the current vector with the LP solution:
        f ← (1‑t)·f + t·g ,   with a step size `t` that slowly decreases.
   - Renormalise `f` to keep Σ f_i = √(2 n) (this does not affect the
     objective).  Repeat the LP‑blend loop as long as time permits.
   - The maximal convolution value `M` tends to shrink and the evaluation
     score improves.

2. **Fallback random perturbations**
   - If SciPy is unavailable or a linear programme fails, fall back to a
     simple block‑perturbation local search (the method used in the reference
     code).

The procedure is run for a few different lengths (`n = 120, 200, 400`);
the best sequence found over all runs is returned.  All operations stay
within the allowed memory limits and the function respects the total time
budget supplied via `budget_s`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – if unavailable we simply fall back to a pure‑Numpy
# local search.
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:                                   # pragma: no cover
    SCIPY_AVAILABLE = False

# Helper used for the pure‑Numpy fallback (identical to the reference).
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------

#   (A @ g)[k] = (f * g)[k]  for all k = 0 … 2n‑2.
# ----------------------------------------------------------------------
def build_conv_matrix(f):
    """Return dense matrix A (shape (2n‑1, n)) such that A @ g = conv(f,g)."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j holds a copy of f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine uses an LP‑based blending strategy
    (when SciPy is available) together with a simple random‑perturbation

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Helper to evaluate a numpy vector and update the global best.
    # ------------------------------------------------------------------
    def try_update(candidate):
        nonlocal best_seq, best_score
        try:

        except Exception:                     # pragma: no cover
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate.copy()

    # First try the LP‑based method for a few modest lengths.

    lengths = [120, 200, 400]   # small enough for fast LP solves

        it = 0

        # improvement to avoid wasting time.
        no_improve = 0

            conv_ff = np.convolve(f, f)
            M = conv_ff.max()

            # Build the linear constraints A @ g ≤ M
            if SCIPY_AVAILABLE:

                c = -np.ones(n, dtype=np.float64)               # minimise -Σg

                res = linprog(c, A_ub=A, b_ub=b,

            s = cand.sum()
            if s > 0:

            else:

                cand = rng.exponential(scale=1.0, size=n)
                cand = cand / cand.sum() * target_sum

            cand = np.clip(cand, 0.0, 1000.0)

            # Evaluate and possibly record a new best.
            score = try_update(cand)

            # Stagnation detection.
            if score is not None and score < best_score - 1e-8:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 300:
                    break

            f = cand
            it += 1

    # ------------------------------------------------------------------
    # If the LP route never produced a candidate (e.g. SciPy missing),
    # fall back to the original random‑local‑search approach.
    # ------------------------------------------------------------------
    if best_seq is None:

        parent = np.ones(256, dtype=np.float64)
        best_seq = parent.copy()
        best_score = float('inf')

            cand = make_block_perturbation(best_seq, rng)
            try_update(cand)

    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(256, dtype=np.float64)

    # Return a plain Python list of floats as required.
    return [float(x) for x in best_seq]
