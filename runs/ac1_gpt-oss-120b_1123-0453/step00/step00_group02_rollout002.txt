import numpy as np, time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Simple optimizer that exploits the proven lower bound of 2 for the evaluation
    function.  The optimum is attained by any constant (uniform) non‑negative
    sequence, because for such a sequence max(conv)=sum_i a_i^2 and the
    Cauchy–Schwarz inequality makes the ratio 2.  The function therefore
    returns a uniform sequence of appropriate length (taken from the global
    ``height_sequence_1`` if present) and performs a brief stochastic search
    around it to verify that no improvement is found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
        if n == 0:
            n = 256
    except Exception:
        n = 256

    # Uniform initial candidate
    const = 1.0  # any positive constant
    best = np.full(n, const, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial uniform candidate length {n}, score {best_score}")

    # Simple stochastic perturbation search (optional)
    def perturb(seq):
        # choose a random block and multiply by exp(normal)
        candidate = seq.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        candidate[start:stop] *= factor
        # optionally symmetrize
        if rng.random() < 0.35:
            candidate = 0.5 * (candidate + candidate[::-1])
        return np.clip(candidate, 0.0, 1000.0)

    # Run search
    while time.monotonic() < deadline:
        cand = perturb(best)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score - 1e-12:

    """
    Produce a block‑wise multiplicative perturbation of ``seq``.
    This is the same idea as the reference ``make_block_perturbation``.
    """
    n = seq.size
    # Choose a block width between n/200 and n/12 (inclusive)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    # Choose a start index for the block
    start = int(rng.integers(0, max(1, n - width + 1)))

    factor = float(np.exp(rng.normal(0.0, 0.12)))

    if rng.random() < 0.35:

    return np.clip(candidate, 0.0, 1000.0)
