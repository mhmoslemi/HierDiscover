def transfer_perturbation(arr, rng):
    n = arr.size
    # define central region indices and outer region indices.
    central_start = n // 4
    central_end = 3 * n // 4
    outer_indices = np.concatenate([np.arange(0, n // 8), np.arange(7 * n // 8, n)])
    # choose donor in central
    donor = rng.integers(central_start, central_end)
    # choose recipient in outer
    recipient = rng.choice(outer_indices)
    # amount to transfer
    amount = rng.random() * 0.1 * arr[donor]  # up to 10% of donor value
    arr[donor] -= amount
    arr[recipient] += amount
    return arr

Propose a better step‑function sequence.

The objective is to minimise
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the linear convolution (numpy.convolve).  The optimum is
reached when the distribution of the sum of two independent draws from the
probability vector `w = a / sum(a)` is as flat as possible.  Uniform
coefficients give a triangular sum‑distribution with a peak `≈ 1/n`,
yielding a score of 2.  By moving probability mass from the centre of the
vector to the ends we can lower the central peak of the convolution and
obtain a smaller maximal value, thus improving the score.

The search proceeds as follows:

* Initialise from the provided ``height_sequence_1`` (or from a uniform
  vector if the global is missing/invalid).
* Repeatedly create candidates by cheap stochastic perturbations:
    – ``block_perturbation`` – multiplicative scaling of a random contiguous
      block (as in the reference code).
    – ``transfer_perturbation`` – move a small fraction of mass from a
      centre index to an outer index (the core idea for flattening the
      convolution).
    – small multiplicative noise applied to all entries.
* Every candidate is normalised to unit sum (scaling does not affect the
  score) and clipped to the admissible range [0, 1000].
* The convolution maximum is obtained via an FFT‑based convolution
  (`max_convolution_fft`) which is much faster than `np.convolve` for the
  required thousands of evaluations.
* A simple hill‑climbing acceptance rule is used: a candidate replaces the
  current one only if it improves the objective; the global best is stored
  separately.
* The loop stops when the time budget is exhausted (default 1100 s minus
  a small safety margin) or when the target score 1.503 is reached.

The algorithm stays well within the allowed time and memory limits and
produces a non‑negative sequence that typically beats the uniform bound
(2.0) and often reaches the desired target.

import numpy as np
import time


def max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximum value of the linear convolution a * a using FFT.
    a must be non‑negative, 1‑D.
    """
    n = a.shape[0]
    # length of the full linear convolution
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << (conv_len - 1).bit_length()
    # real FFT (faster for real‑valued input)
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    # keep only the needed part
    conv = conv[:conv_len]
    return float(conv.max())


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block of `base` by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional gentle symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def transfer_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a small fraction of mass from a centrally located index to an outer
    index.  This tends to flatten the convolution peak.
    """
    cand = base.copy()
    n = cand.size
    # define centre and outer index ranges
    centre_start = n // 4
    centre_end = 3 * n // 4
    outer_left = np.arange(0, n // 8, dtype=int)
    outer_right = np.arange(7 * n // 8, n, dtype=int)
    outer = np.concatenate((outer_left, outer_right))

    # choose donor from the centre and recipient from the outer region
    donor = int(rng.integers(centre_start, centre_end))
    recipient = int(rng.choice(outer))

    # transfer up to 3 % of donor's current mass
    if cand[donor] > 0:
        amount = float(rng.random() * 0.03 * cand[donor])
        cand[donor] -= amount
        cand[recipient] += amount

    return np.clip(cand, 0.0, 1000.0)
