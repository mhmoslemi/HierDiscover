"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv(a, a) = np.convolve(a, a).

The algorithm combines three ideas:

1. **Scale invariance** – the objective does not depend on the overall
   magnitude of the vector, so we keep the sum fixed (≈1) during the search.
2. **Sub‑gradient descent on the worst convolution bin** – at each
   iteration we locate the index `k` where the convolution is maximal.
   The gradient of that bin w.r.t. the coefficients is
   `∂/∂a_i conv_k = 2 * a_{k-i}` (zero when the partner index is out of range).
   We move the coefficients opposite to this gradient, i.e. we reduce the
   values that contribute most to the current peak.
3. **Exploratory random block perturbations** – occasionally a small random
   multiplicative block is applied (the same idea used in the reference code)
   to escape shallow local minima.

The loop runs until the supplied time budget (default 1100 s) is exhausted,
always keeping the best sequence seen so far.  All operations are O(n) per
iteration (numpy convolution is cheap for n≈5 000), so many thousands of
updates can be performed in the given time budget.

The returned sequence respects the security checks of `evaluate_sequence`
(only non‑negative finite numbers, clipped to the allowed range).
"""

import time
import numpy as np

def _initial_vector(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Produce a starting vector of length n.
    If the global `height_sequence_1` exists and looks valid it is used,
    otherwise a random Dirichlet draw is taken.
    The vector is normalised to have unit sum (scale invariance).
    """
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size == n and np.isfinite(base).all():
            # keep the overall shape but normalise the mass
            a = np.clip(base, 0.0, 1000.0)
            if a.sum() < 0.01:
                a = np.ones_like(a)
            a = a / a.sum()
            return a
    except NameError:
        pass

    # fallback: random positive vector, normalised to sum = 1
    a = rng.random(n) + 0.01          # avoid exact zeros at start
    a = a / a.sum()
    return a


def _conv_max(a: np.ndarray) -> float:
    """Return the maximum entry of a*a (full convolution)."""
    return np.max(np.convolve(a, a))


def _subgradient_step(a: np.ndarray, max_idx: int, lr: float) -> np.ndarray:
    """
    Perform one sub‑gradient descent step that reduces the contribution of
    the convolution entry at `max_idx`.
    The derivative of conv[max_idx] w.r.t. a[i] is 2 * a[max_idx - i] (if in range).
    """
    n = a.shape[0]
    idx = np.arange(n)
    partner = max_idx - idx
    mask = (partner >= 0) & (partner < n)

    # gradient = 2 * a[partner] for valid positions, 0 otherwise
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[partner[mask]]

    # a - lr * grad, keep non‑negative
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0.0, None)

    # renormalise to keep the sum constant (scale invariance)
    s = a_new.sum()
    if s < 1e-12:                    # degenerated – restart randomly
        a_new = np.random.random(n)
        a_new = a_new / a_new.sum()
    else:
        a_new = a_new / s
    return a_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply the same block‑perturbation used in the reference solution.
    This is a cheap way to inject stochasticity.
    """
    n = a.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    candidate = a.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor

    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])

    # keep within the range enforced by evaluate_sequence
    candidate = np.clip(candidate, 0.0, 1000.0)

    # renormalise – the absolute scale does not matter for the objective
    candidate = candidate / candidate.sum()
    return candidate


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a sequence with a low evaluation score.
    The function respects the allowed time budget and returns the best
    sequence found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # length of the sequence – use the length of the provided construction
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000                     # fallback length if the global is missing

    # ----- initial population -------------------------------------------------
    a = _initial_vector(rng, n)
    cur_score = float(evaluate_sequence(a.tolist()))
    best = a.copy()
    best_score = cur_score
    print(f"[init] length={n}, score={best_score:.6f}")

    it = 0
    # -------------------------------------------------------------------------
    while time.monotonic() < deadline:
        # 1. locate the current worst convolution bin
        conv = np.convolve(a, a)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # 2. sub‑gradient step aimed at lowering that peak
        # learning rate decays slowly with the iteration count
        lr = 0.12 / (1.0 + it * 0.0008)
        a_step = _subgradient_step(a, max_idx, lr)

        # 3. occasional random block perturbation to diversify the search
        if rng.random() < 0.025:
            a_step = _random_block_perturb(a_step, rng)

        # 4. evaluate the new candidate (exact evaluation required for scoring)
        score = float(evaluate_sequence(a_step.tolist()))
        if not np.isfinite(score):
            # discard pathological candidates
            a_step = _initial_vector(rng, n)
            score = float(evaluate_sequence(a_step.tolist()))

        # 5. acceptance rule – keep the new point if it improves, otherwise
        #    accept with a small probability to allow uphill moves.
        if score < cur_score or rng.random() < 0.03:
            a = a_step
            cur_score = score

        # 6. update the overall best solution
        if score < best_score:
            best = a_step.copy()
            best_score = score
            print(f"[iter {it:06d}] new best score = {best_score:.6f}")

        # 7. restart if we have been stuck for a long time (optional)
        if it % 15000 == 0 and it > 0 and best_score > 1.9:

            a = _initial_vector(rng, n)
            cur_score = float(evaluate_sequence(a.tolist()))
            print(f"[restart] iter {it}, score reset to {cur_score:.6f}")

        it += 1

    print(f"[finished] best score obtained = {best_score:.6f}")
    # Convert to plain list of Python floats as required by the harness
    return [float(x) for x in best]
