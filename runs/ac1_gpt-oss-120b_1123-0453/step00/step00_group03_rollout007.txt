import numpy as np
import time

def compute_score(seq: np.ndarray) -> float:
    """
    Compute the evaluation function used in the challenge.
    Equivalent to evaluate_sequence for a pre‑clipped, finite, non‑negative array.
    Returns float('inf') for degenerate cases.
    """
    n = seq.shape[0]
    if n == 0:
        return float('inf')
    # pre-condition: seq already clipped to [0,1000] and finite
    b = np.convolve(seq, seq)
    max_b = b.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic search for a step‑function height sequence that minimizes
    2·n·max(convolution)/sum^2.
    The algorithm mixes three simple moves:
    * pairwise mass transfer,
    * block multiplicative perturbation (the original baseline),
    * a gradient–descent step that explicitly reduces the current maximal
      convolution coefficient.
    An annealing schedule is used so the search can escape local minima.
    The best sequence seen before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)

    # initialise the sequence -------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    # a non‑zero sum is required
    if base.sum() < 0.01:
        base[:] = 1.0

    n = base.shape[0]

    # current solution, its score and the best ever seen
    curr = base.copy()
    curr_score = compute_score(curr)
    best = curr.copy()
    best_score = curr_score

    # annealing parameters
    T_start = 0.05
    T_end = 1e-4

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # choose a move type ---------------------------------------------------
        r = rng.random()
        if r < 0.35:
            # ---- pairwise transfer -------------------------------------------------
            i, j = rng.choice(n, size=2, replace=False)
            # transfer at most 30 % of the donor value
            max_transfer = curr[i] * rng.random()
            delta = max_transfer * rng.random()
            cand = curr.copy()
            cand[i] = max(cand[i] - delta, 0.0)
            cand[j] = min(cand[j] + delta, 1000.0)

        elif r < 0.55:
            # ---- block multiplicative perturbation -------------------------------
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            cand = curr.copy()
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:start + width] *= factor
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])

        elif r < 0.85:
            # ---- gradient descent step --------------------------------------------
            # compute the convolution of the current vector once
            conv = np.convolve(curr, curr)
            max_b = conv.max()
            idxs = np.where(conv == max_b)[0]

            # build a sub‑gradient of the maximal entry (sum of contributions)
            grad = np.zeros_like(curr)
            for idx in idxs:
                lo = max(0, idx - (n - 1))
                hi = min(n - 1, idx)
                i_range = np.arange(lo, hi + 1)
                j_range = idx - i_range
                grad[i_range] += 2.0 * curr[j_range]

            # step size proportional to the typical magnitude of the vector
            # (the factor 0.02 works reasonably well in practice)
            step = 0.02 * (curr.mean() + 1e-12)
            cand = curr - step * grad
            cand = np.clip(cand, 0.0, 1000.0)

        else:
            # ---- symmetrisation / smoothing ---------------------------------------
            cand = 0.5 * (curr + curr[::-1])

        # occasional random scaling – helps to escape deep plateaux
        if rng.random() < 0.01:
            scale = 0.5 + rng.random()  # between 0.5 and 1.5
            cand = np.clip(cand * scale, 0.0, 1000.0)

        # ensure we stay away from the degenerate zero‑sum case
        if cand.sum() < 0.01:
            # add a tiny constant everywhere
            cand += 1e-3

        # evaluate ---------------------------------------------------------------
        cand_score = compute_score(cand)

        # ignore non‑finite scores
        if not np.isfinite(cand_score):
            continue

        # always keep the best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # simulated annealing acceptance for the current walk
        progress = (time.monotonic() - start_time) / (deadline - start_time)
        T = T_start * (1.0 - progress) + T_end * progress

        if cand_score < curr_score:
            curr = cand
            curr_score = cand_score
        else:
            # accept worse solutions with probability exp(-(Δ)/T)
            delta = cand_score - curr_score
            if rng.random() < np.exp(-delta / (T + 1e-12)):
                curr = cand
                curr_score = cand_score

        # occasional diagnostics – printed roughly every 30 seconds
        if iteration % 1000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.1f}s] best={best_score:.6f}  curr={curr_score:.6f}")

    # final conversion – the competition harness expects a plain Python list
    return [float(x) for x in best]

"""
search algorithm for the step‑function optimisation challenge.

The goal is to minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

over non‑negative sequences a (length n).

The implementation mixes three cheap local moves:

* **pairwise mass transfer** – move a random amount from one index to another;
* **block multiplicative perturbation** – the baseline “multiply a random block”
  operation used in the reference solution;
* **gradient‑descent step** – explicitly reduces the currently maximal
  convolution coefficient by moving against its sub‑gradient.

The walk is driven by a simple simulated‑annealing schedule so that occasional
worse moves are accepted, helping the search escape local minima.  The best
sequence observed before the deadline is returned.

All helper functions are defined at module level; no closures or lambdas are
used.  The code respects the 1000‑second time budget by checking the deadline
each iteration.
"""

import time
import numpy as np


def _score(seq: np.ndarray) -> float:
    """
    Compute the exact evaluation used by the challenge.
    The input is assumed to be a 1‑D finite, non‑negative numpy array that is
    already clipped to the [0, 1000] range.
    """
    n = seq.size
    if n == 0:
        return float("inf")
    # full convolution
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Stochastic multi‑move optimiser.

    Parameters
    ----------
    seed : int
        Seed for the random number generator.
    budget_s : float
        Approximate time budget in seconds; the function stops a few seconds
        before this limit to avoid a hard termination.
    kwargs : dict
        Ignored – present only for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best height sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------ #
    # initialise the sequence                                            #
    # ------------------------------------------------------------------ #
    if "height_sequence_1" in globals():

    else:
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    if base.sum() < 0.01:
        base[:] = 1.0

    n = base.shape[0]

    # current solution, its score and the best ever seen
    curr = base.copy()
    curr_score = _score(curr)
    best = curr.copy()
    best_score = curr_score

    # annealing schedule (linear from T_start to T_end)
    T_start = 0.05
    T_end = 1e-4

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # ---------------------------------------------------------------- #
        # pick a move type                                                #
        # ---------------------------------------------------------------- #
        move_type = rng.random()
        if move_type < 0.35:

            # pairwise mass transfer                                      #
            # ----------------------------------------------------------- #

            donor = curr[i]
            delta = donor * rng.random() * rng.random() * 0.3
            cand = curr.copy()
            cand[i] = max(cand[i] - delta, 0.0)
            cand[j] = min(cand[j] + delta, 1000.0)

            # ----------------------------------------------------------- #
            # block multiplicative perturbation (baseline)                #
            # ----------------------------------------------------------- #
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start_idx = int(rng.integers(0, max(1, n - width + 1)))

    # -------------------------------------------------------------------- #
    # return the best sequence found                                        #
    # -------------------------------------------------------------------- #
    return [float(v) for v in best]
