"""
Propose a step‑height sequence that minimizes the evaluation function.

The algorithm is a hybrid of random block perturbations, a simple
gradient‑like step that reduces the current maximum of the self‑convolution,
and occasional mass‑swaps.  All moves keep the vector non‑negative and are
renormalised to a fixed L1‑norm (the objective is scale invariant, so the
choice of norm does not matter).  The best sequence seen so far is returned.

The code runs for ``budget_s`` seconds (default 1100 s) and prints progress
whenever a new best score is found.
"""

import time
import numpy as np

def _objective(a: np.ndarray) -> float:
    """
    Compute 2*n*max(convolution)/sum(a)^2.
    This matches ``evaluate_sequence`` after the same clipping and
    non‑negativity guarantees.
    """
    n = a.shape[0]
    b = np.convolve(a, a)          # linear convolution, length 2n‑1
    max_b = np.max(b)
    s = np.sum(a)
    if s < 1e-12:                  # protective – shouldn't happen
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Scale a random contiguous block by a log‑normal factor.
    Occasionally symmetrises the whole vector.
    """
    n = base.size
    cand = base.copy()
    # block width roughly between n/300 and n/12 (adjustable)
    width = int(rng.integers(max(1, n // 300), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.09)))   # smaller sigma than original
    cand[start:stop] *= scale
    if rng.random() < 0.33:                      # gentle symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _gradient_step(a: np.ndarray, rng: np.random.Generator,
                   progress: float) -> np.ndarray:
    """
    Reduce the current maximum convolution entry.

    The index k where the convolution is maximal is found, then all
    coefficients that contribute to that entry are multiplied by
    (1 - beta), where beta decays with the remaining time.
    The vector is finally renormalised to unit L1 norm.
    """
    n = a.size
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))

    # indices i for which a[i] participates in (a*a)[k]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)

    # beta shrinks linearly as time runs out (progress∈[0,1])
    beta = 0.02 * (1.0 - progress) + 1e-5

    a_new = a.copy()
    a_new[mask] *= (1.0 - beta)

    # renormalise to keep the scale fixed (objective is homogeneous)
    s = np.sum(a_new)
    if s == 0.0:
        a_new[:] = 1.0 / n
    else:
        a_new /= s
    return a_new


def _swap_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Transfer a random fraction of mass from one position to another.
    """
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    donor = a[i]
    # move at most half of the donor mass
    eps = rng.random() * donor * 0.5
    a_new = a.copy()
    a_new[i] -= eps
    a_new[j] += eps
    # renormalise (still unit L1 because we moved mass)
    return a_new


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided ``evaluate_sequence`` function.

    The search combines random block perturbations, a simple gradient step
    that pushes down the current maximal convolution value and occasional
    mass‑swap moves.  The best sequence seen before the time budget expires
    is returned as a Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise a feasible start vector (use the global variable if it exists)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:                # fallback if the global is missing / bad
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    n = base.size

    # work with a unit‑L1 normalisation (scale does not affect the score)
    total = np.sum(base)
    if total == 0.0:
        a = np.full_like(base, 1.0 / n)
    else:
        a = base / total

    best = a.copy()
    best_score = _objective(best)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)
    it = 0

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time)   # 0 → 1

        rnd = rng.random()
        if rnd < 0.55:                     # block‑scale move (most frequent)
            cand = _block_perturbation(best, rng)
        elif rnd < 0.85:                   # gradient‑like move
            cand = _gradient_step(best, rng, progress)
        else:                              # simple swap move
            cand = _swap_move(best, rng)

        # keep unit L1 norm (objective is homogeneous)
        s = np.sum(cand)
        if s == 0.0:
            cand[:] = 1.0 / n
        else:
            cand /= s

        score = _objective(cand)

        if score < best_score:
            best = cand
            best_score = score
            print(f"[{it:07d}] new best score = {best_score:.6f}")

        # every few thousand iterations, try a fresh random direction to

        if it % 3000 == 0:
            rand = rng.random(n)
            rand /= rand.sum()
            mixed = 0.6 * best + 0.4 * rand
            mixed /= mixed.sum()
            mixed_score = _objective(mixed)
            if mixed_score < best_score:
                best = mixed
                best_score = mixed_score
                print(f"[{it:07d}] mixed improvement = {best_score:.6f}")

    # Return the best sequence found (as plain Python list of floats)
    return [float(x) for x in best]
