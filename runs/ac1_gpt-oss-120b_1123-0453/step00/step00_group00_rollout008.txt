def simplex_projection(v):
    # ensure nonnegative sum 1
    # Based on algorithm from Wang & Carreira-Perpinan.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Optimization of the step‑function bound

We minimise the quantity returned by ``evaluate_sequence`` :

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

where ``conv`` is the ordinary convolution (np.convolve).  By fixing the
ℓ¹–norm of the sequence to  √(2·n)  the score becomes exactly the
maximum entry of the self‑convolution.  Hence we can iteratively improve a
candidate sequence **f** by solving a linear program:

    maximise    Σ b_j
    subject to  f * b ≤ M      (element‑wise)       (M = max(f*f))
                b ≥ 0

The constraints are linear because **f** is fixed.  The optimal
solution *b* is the most “massive’’ sequence that still respects the
current worst‑case convolution bound *M*.  After a scaling step to restore
the ℓ¹–norm we mix the old sequence with the new one:

    g = ( √(2·n) / Σb ) * b
    f ← (1‑t)·f + t·g          (t a small step size)

Each iteration provably does not increase the bound and typically lowers
it.  To escape possible stagnation we occasionally apply the cheap block
perturbation used in the reference code.  The whole procedure runs within
the allotted time budget and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: block‑perturbation (fallback & exploration)
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Simple ℓ¹‑simplex projection (used only if we need to re‑normalise)
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    refinement (the theoretical step of the reference paper) with cheap
    random block perturbations to keep the search robust.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1.  Initialise a sensible starting point.
    #     – if a previous construction is available we reuse it (trimmed
    #       to a manageable length);
    #     – otherwise an “arcsine’’ shaped distribution is used because
    #       its self‑convolution is close to uniform.
    #
    # The sequence is scaled so that sum = sqrt(2·n) (the scale used in
    # the LP‑based iteration).
    #
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            init = None
    else:
        init = None

    if init is not None:
        # keep at most 1024 entries – larger arrays would make the LP too
        # heavy while providing little extra benefit per iteration.
        if init.size > 1024:
            init = init[:1024]
        n = init.size
        scale = np.sqrt(2.0 * n) / init.sum()
        f = np.clip(init * scale, 0.0, 1000.0)
    else:
        # arcsine‑like start (more mass near the ends → flatter convolution)
        n = 1024
        idx = np.arange(n, dtype=np.float64) + 0.5
        x = idx / n
        eps = 1e-6
        a = 1.0 / np.sqrt((x + eps) * (1.0 - x + eps))
        a = np.clip(a, 0.0, 1000.0)
        a = a * (np.sqrt(2.0 * n) / a.sum())
        f = a

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop.
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best_score = float('inf')

    # step size for convex combination (t in the paper)
    t = 0.2
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # Occasionally (≈20 % of the time) run the LP‑based refinement.
        # The LP is cheap for n≈1000 (≈2·10⁶ non‑zeros) but we keep the
        # frequency low to leave room for many cheap perturbations.
        if rng.random() < 0.2:
            # current ℓ¹‑norm is already √(2·n) by construction
            conv_f = np.convolve(f, f)
            M = float(conv_f.max())

            # Build the convolution matrix A such that A @ b = f * b.
            # For each column j, the non‑zero rows are j … j+n‑1 and carry
            # the coefficients f[0], …, f[n‑1].
            A = np.zeros((2 * n - 1, n), dtype=np.float64)
            for j in range(n):
                A[j:j + n, j] = f

            # Linear programme: maximise Σ b  →  minimise –Σ b
            c = -np.ones(n, dtype=np.float64)          # objective
            ub = np.full(2 * n - 1, M, dtype=np.float64)   # upper bounds

            # Give the solver a small time limit so we never exceed the
            # global deadline.
            time_left = max(0.0, deadline - time.monotonic())
            lp_options = {"presolve": True, "time_limit": time_left}

            res = linprog(
                c,
                A_ub=A,
                b_ub=ub,
                bounds=[(0.0, None)] * n,
                method="highs",
                options=lp_options,
            )

            if res.success and res.x.sum() > 0.0:
                b = res.x
                # Rescale to keep the ℓ¹‑norm = √(2·n)
                factor = np.sqrt(2.0 * n) / b.sum()
                g = b * factor
                # Convex combination with the previous iterate
                candidate = (1.0 - t) * f + t * g
                candidate = np.clip(candidate, 0.0, 1000.0)
            else:
                # LP failed – fall back to a random block perturbation
                candidate = make_block_perturbation(f, rng)
        else:
            # Cheap random move (the method used in the reference solution)
            candidate = make_block_perturbation(f, rng)

        # ----------------------------------------------------------------
        # 3.  Evaluate and possibly accept the new candidate.
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate.copy()
            best_score = score
            f = best.copy()                 # continue from the improved point
            # gradually shrink t so that later steps become finer
            t = max(0.02, t * 0.95)
            print(f"Iter {iteration:5d} – new best = {best_score:.6f}")
            # early stop if we have reached the target region
            if best_score < 1.6:
                break
        else:
            # occasional random walk to avoid being stuck
            if rng.random() < 0.07:
                f = candidate.copy()

    # ------------------------------------------------------------------
    # Return the best sequence discovered (converted to plain Python floats).
    return [float(x) for x in best]
