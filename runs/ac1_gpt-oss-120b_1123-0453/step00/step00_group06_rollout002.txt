"""
step‑function optimizer

This implementation improves upon the reference block‑perturbation search.
It starts from the known construction (``height_sequence_1`` if present) and
from a few randomly generated seeds.  The core idea is a greedy “mass‑shift”
step: locate the convolution index that currently attains the maximum,
compute each element’s contribution to that index, and move a small amount
of mass from a high‑contributing position to a low‑contributing one.
The move is always accepted when it improves the score, otherwise it is
accepted with a tiny Metropolis probability (simulated‑annealing style) to
escape plateaus.  Every few hundred iterations a random block perturbation
(a la the reference code) is applied to diversify the search.  The best
sequence seen before the time budget expires is trimmed of leading/trailing
zeros (they only increase the ``n`` factor) and returned.

The algorithm runs entirely in NumPy, uses only the supplied
``evaluate_sequence`` for the exact objective, and respects the 1000‑second
budget via a monotonic clock.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # pick a block width between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    if rng.random() < 0.35:                     # optional mirroring
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


def contributions_to_shift(seq: np.ndarray, shift: int) -> np.ndarray:
    """
    For a given shift index ``shift`` compute the vector
    ``c_i = a_i * a_{shift-i}`` (zero where the index is out of bounds).
    This is the derivative of the convolution value at ``shift`` w.r.t. each
    entry of ``seq``.
    """
    n = seq.size
    i_start = max(0, shift - (n - 1))
    i_end = min(n - 1, shift)
    # slice of valid indices
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = shift - i
    contrib = seq[i] * seq[j]
    out = np.zeros_like(seq)
    out[i] = contrib
    return out


def trim_zeros(seq: np.ndarray) -> np.ndarray:
    """Remove leading and trailing zeros (or near‑zeros) – they only hurt the score."""
    nz = np.where(seq > 1e-12)[0]
    if nz.size == 0:
        return np.array([0.0], dtype=np.float64)
    return seq[nz[0] : nz[-1] + 1]


def random_initial_sequences(rng: np.random.Generator):
    """Yield a few diverse starting points."""
    # 1. the global best if it exists
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            yield np.clip(base, 0.0, 1000.0)
    except Exception:
        pass

    # 2. a uniform vector (size 256)
    yield np.full(256, 1.0, dtype=np.float64)

    # 3. exponential decay – various lengths
    for n in (512, 1024, 2048):
        seq = np.exp(-np.arange(n) / (n / 5.0))
        seq *= 5.0                           # scale to a reasonable magnitude
        yield np.clip(seq, 0.0, 1000.0)

    # 4. random positive vectors
    for n in (512, 1024, 2048):
        seq = rng.random(n) * rng.uniform(0.5, 10.0)
        yield np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point used by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Hill‑climbing / gradient‑driven optimizer for the step‑function score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock budget (seconds). The function returns the best sequence
        found before the budget expires.

    Returns
    -------
    list[float]
        The sequence of non‑negative heights achieving the smallest score.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a tiny margin for finalisation
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise: evaluate a handful of seeds and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    for seq in random_initial_sequences(rng):
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = seq
            best_score = score
            print(f"[init]   score={best_score:.6f}, len={best_seq.size}")

    if best_seq is None:
        # fallback – a tiny constant vector
        best_seq = np.ones(64, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # work on a copy; keep the globally best for restarts
    current = best_seq.copy()
    cur_score = best_score

    # parameters controlling the search dynamics
    step_fraction = 0.12      # what fraction of the source element we move
    temperature_start = 0.02  # relative scale for Metropolis acceptance
    iteration = 0
    last_report = start_time

    while time.monotonic() < deadline:
        n = current.size
        # --------------------------------------------------------------
        # 1) compute convolution and locate the maximal entry
        # --------------------------------------------------------------
        conv = np.convolve(current, current)
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])
        sum_a = float(current.sum())
        # safeguard (should never happen)
        if sum_a < 1e-12:
            break

        # direct evaluation matches the exact one (no NaNs, no clip needed)
        cur_score = 2.0 * n * max_val / (sum_a * sum_a)

        # --------------------------------------------------------------
        # 2) contributions of each coordinate to the current maximum
        # --------------------------------------------------------------
        contrib = contributions_to_shift(current, max_idx)

        # pick a source index weighted by its contribution (high chance to be reduced)
        if contrib.sum() > 0:
            probs_out = contrib / contrib.sum()
            i_out = int(rng.choice(n, p=probs_out))
        else:
            i_out = int(rng.integers(0, n))

        # pick a destination index where the contribution is small
        # weight = (max_contrib - contrib)  -> prefers low contributors
        max_contrib = float(contrib.max())
        probs_in = (max_contrib - contrib) + 1e-12
        probs_in /= probs_in.sum()
        i_in = int(rng.choice(n, p=probs_in))

        if i_out != i_in:
            # amount we move – bounded by source availability and destination capacity
            avail = float(current[i_out])
            cap = 1000.0 - float(current[i_in])
            if avail > 0 and cap > 0:
                delta = min(avail * step_fraction, cap)
                # add a bit of stochasticity
                delta *= float(rng.random())
                if delta > 0:
                    candidate = current.copy()
                    candidate[i_out] -= delta
                    candidate[i_in] += delta
                    # evaluate exact objective
                    cand_score = float(evaluate_sequence(candidate.tolist()))
                    # acceptance rule
                    improve = cand_score < cur_score
                    # temperature cools linearly with time
                    elapsed = (time.monotonic() - start_time) / (deadline - start_time)
                    temperature = temperature_start * max(0.0, 1.0 - elapsed)
                    accept = improve or (temperature > 0 and
                                         rng.random() < math.exp(-(cand_score - cur_score) / max(1e-12, temperature)))
                    if accept:
                        current = candidate
                        cur_score = cand_score
                        # record global best
                        if cand_score < best_score:
                            best_seq = candidate.copy()
                            best_score = cand_score
                            print(f"[{time.monotonic()-start_time:5.1f}s] new best {best_score:.6f} (len={best_seq.size})")

        # --------------------------------------------------------------
        # 3) occasional global perturbation for diversification
        # --------------------------------------------------------------
        iteration += 1
        if iteration % 500 == 0:
            if rng.random() < 0.35:
                pert = make_block_perturbation(current, rng)
                pert_score = float(evaluate_sequence(pert.tolist()))
                if pert_score < cur_score or rng.random() < 0.05:
                    current = pert
                    cur_score = pert_score
                    if pert_score < best_score:
                        best_seq = pert.copy()
                        best_score = pert_score
                        print(f"[{time.monotonic()-start_time:5.1f}s] perturbed best {best_score:.6f}")

        # --------------------------------------------------------------
        # 4) periodic progress report (lightweight)
        # --------------------------------------------------------------
        if time.monotonic() - last_report > 30.0:
            last_report = time.monotonic()
            print(f"[{last_report - start_time:5.1f}s] current best {best_score:.6f}")

    # ------------------------------------------------------------------
    # final tidy‑up: cut leading / trailing zeros (they only increase ``n``)
    # ------------------------------------------------------------------
    if best_seq is not None:
        best_seq = trim_zeros(best_seq)
    else:
        best_seq = np.zeros(1, dtype=np.float64)

    # final safety clip (required by evaluate_sequence)
    np.clip(best_seq, 0.0, 1000.0, out=best_seq)

    print(f"Finished – best score {best_score:.6f}, length {best_seq.size}")
    return [float(x) for x in best_seq]
