"""
Hybrid shape‑search + simulated‑annealing optimizer for the step‑function
evaluation problem.

* Phase 1 – systematic sweep of simple parametrised families (exponential,
  Gaussian, triangular, random).  The best shape found becomes the seed for
  the second phase.
* Phase 2 – local refinement using two stochastic moves:
        – block perturbation (multiplicative noise on a contiguous segment)
        – elementwise log‑normal multiplicative noise.
  An annealing schedule accepts occasional uphill moves to escape local minima.
* The algorithm respects all constraints (non‑negative, ≤1000) and works
  within the time budget supplied via ``budget_s`` (≈ 1000 s).
"""

import math
import time
from typing import List

import numpy as np


def generate_exponential(n: int, q: float) -> np.ndarray:
    """Symmetric exponential shape: a[i] = q**|i‑center| (q∈(0,1])."""
    i = np.arange(n)
    center = (n - 1) / 2.0
    return np.exp(np.log(q) * np.abs(i - center))


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle of the array."""
    i = np.arange(n)
    center = (n - 1) / 2.0
    return np.exp(-((i - center) ** 2) / (2.0 * sigma ** 2))


def generate_triangular(n: int, width: float) -> np.ndarray:
    """
    Symmetric triangular hat.
    ``width`` is the full base width (0 < width ≤ n).  Height is 1 at the centre,
    linearly decreasing to 0 at the edges of the base.
    """
    i = np.arange(n)
    center = (n - 1) / 2.0
    half = width / 2.0
    return np.clip(1.0 - np.abs(i - center) / half, 0.0, None)

    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    if rng.random() < 0.35:               # optional symmetrisation
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


def multiplicative_noise(seq: np.ndarray, rng: np.random.Generator,
                         sigma: float = 0.05) -> np.ndarray:
    """
    Log‑normal elementwise perturbation and L1‑renormalisation.
    ``sigma`` controls the strength of the noise.
    """
    factor = np.exp(rng.normal(0.0, sigma, size=seq.shape))
    candidate = seq * factor
    # Renormalise to keep the sum (and thus scaling invariance) stable.
    sum_orig = np.sum(seq)
    if sum_orig > 0:
        candidate *= (sum_orig / np.sum(candidate))
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> List[float]:
    """
    Main entry point for the optimiser.
    Returns a list of non‑negative heights that (hopefully) improves the
    current best bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initialise with the provided global sequence or a simple fallback.
    # ------------------------------------------------------------------
    fallback = np.ones(256, dtype=np.float64)
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
                fallback = np.clip(tmp, 0.0, 1000.0)
        except Exception:
            pass

    best = fallback.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Phase 1 – systematic exploration of a few parametric families.
    # ------------------------------------------------------------------
    # Allocate roughly 20 % of the total budget to this deterministic sweep.
    phase1_end = time.monotonic() + 0.2 * (deadline - time.monotonic())
    lengths = [512, 1024, 2048, 4096]          # modest sizes for cheap evaluation

    for n in lengths:
        i = np.arange(n)
        center = (n - 1) / 2.0

        # -- Exponential family --
        for q in [0.90, 0.93, 0.95, 0.97, 0.99, 0.995]:
            if time.monotonic() >= phase1_end:
                break
            seq = np.exp(np.log(q) * np.abs(i - center))
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score = seq, score

        # -- Gaussian family --
        for sigma_factor in [0.08, 0.12, 0.18, 0.25, 0.33, 0.45]:
            if time.monotonic() >= phase1_end:
                break
            sigma = sigma_factor * n
            seq = np.exp(-((i - center) ** 2) / (2.0 * sigma ** 2))
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score = seq, score

        # -- Triangular family --
        for width_factor in [0.4, 0.6, 0.8, 1.0]:
            if time.monotonic() >= phase1_end:
                break
            width = width_factor * n
            half = width / 2.0
            seq = np.clip(1.0 - np.abs(i - center) / half, 0.0, None)
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score = seq, score

        # -- Random uniform seeds (few samples) --
        for _ in range(5):
            if time.monotonic() >= phase1_end:
                break
            seq = rng.random(n)
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score = seq, score

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing style local refinement.
    # ------------------------------------------------------------------
    # Temperature schedule – starts warm, cools to near zero.
    T_start, T_end = 0.05, 1e-5
    local_start = time.monotonic()
    total_local = max(0.0, deadline - local_start)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - local_start
        frac = elapsed / total_local if total_local > 0 else 1.0
        temp = T_start * (1.0 - frac) + T_end * frac

        # Choose a perturbation method.
        if rng.random() < 0.6:
            cand = block_perturb(best, rng)
        else:
            # sigma for log‑normal noise adapts with temperature.
            sigma_noise = max(0.01, temp * 5.0)
            cand = multiplicative_noise(best, rng, sigma=sigma_noise)

        # Occasionally enforce symmetry – empirically useful for this objective.
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])

        # Evaluate.
        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            continue

        # Acceptance rule.
        if score < best_score:
            best, best_score = cand, score
            # Light progress reporting (throttled).
            if int(best_score * 1000) % 5 == 0:
                print(f"[{time.monotonic():.1f}s] New best: {best_score:.6f} (len={len(best)})")
        else:
            # Simulated‑annealing uphill acceptance.
            if rng.random() < math.exp(-(score - best_score) / (temp + 1e-12)):
                best, best_score = cand, score

    # ------------------------------------------------------------------
    # Return the best sequence discovered as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
