    contributions = a[idxs] * a[k_max - idxs]
    # find i with largest contribution

def block_perturbation(a, rng):

    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

def gradient_mass_transfer(a, rng, epsilon_factor=0.1, min_eps=1e-5):
    n = a.shape[0]
    # Compute convolution to find k_max
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))
    # Determine i range
    i_start = max(0, k_max - (n-1))
    i_end = min(n-1, k_max)
    idxs = np.arange(i_start, i_end+1)
    # contributions
    contrib = a[idxs] * a[k_max - idxs]
    # find index with largest contribution
    max_pos = idxs[np.argmax(contrib)]
    # find index with smallest value among all (or minimal contribution)
    # we avoid moving to the same index.
    # Choose candidate j with probability: either random low a or random anywhere.
    # Pick j randomly from full range excluding max_pos
    j = int(rng.integers(0, n-1))
    while j == max_pos:
        j = int(rng.integers(0, n-1))
    # epsilon as fraction of a[max_pos]
    eps = min(epsilon_factor * a[max_pos], a[max_pos] * 0.5)  # don't zero it entirely
    if eps <= 0:
        return a  # no move
    a_new = a.copy()
    a_new[max_pos] -= eps
    a_new[j] += eps
    # Clip
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new
```

"""
Heuristic construction of a step sequence with a very low evaluation score.

The idea is to pair each entry a[i] with a[n‑1‑i] so that their
product is constant after normalisation.  Write a[i] = e^{t_i},
a[n‑1‑i] = e^{-t_i}.  After scaling the whole vector to sum 1 the
self‑convolution at the centre (k = n‑1) becomes

    (a*a)[n‑1] = n * s² ,  with  s = 1 / Σ_i (e^{t_i}+e^{-t_i})

where Σ_i runs over the ⌊n/2⌋ pairs.  The evaluation score turns out to be

    score = 2 / (average_{i} cosh(t_i))²

so increasing the spread of the t_i’s (larger |t_i|) reduces the score.
Choosing t_i uniformly in [-L, L] gives average cosh = sinh(L)/L and
score = 2·L² / sinh(L)².  For L≈2 the score is ≈0.6, far below the
target 1.503.

The function below builds several candidates (deterministic linear t_i,
randomised draws, a few L values) within the given time budget and
returns the best one.
"""

import time
import numpy as np

def compute_score(seq: np.ndarray, n: int) -> float:
    """Fast replica of evaluate_sequence (no type checks)."""
    a = np.clip(seq, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def anti_diag_exp_sequence(n: int, L: float, rng: np.random.Generator,
                           use_random_t: bool = False) -> np.ndarray:
    """
    Build the anti‑diagonal exponential construction.

    Parameters
    ----------
    n : int
        Length of the sequence.
    L : float
        Half‑range of the logarithmic parameters t_i ∈ [-L, L].
    rng : np.random.Generator
        Random number generator.
    use_random_t : bool
        If True, draw t_i uniformly at random; otherwise use a
        deterministic linear spacing.

    Returns
    -------
    seq : np.ndarray
        Normalised non‑negative sequence (sum equals 1).
    """
    m = n // 2                     # number of independent pairs
    if use_random_t:
        t = rng.uniform(-L, L, size=m)
    else:
        # deterministic equally‑spaced values covering [-L, L]
        if m > 1:
            t = np.linspace(-L, L, m, dtype=np.float64)
        else:
            t = np.array([0.0], dtype=np.float64)

    seq = np.empty(n, dtype=np.float64)

    # fill the paired entries
    for i in range(m):
        val = np.exp(t[i])
        seq[i] = val
        seq[n - 1 - i] = 1.0 / val

    # middle element for odd length
    if n % 2 == 1:
        seq[m] = 1.0

    # normalise to sum 1 (the evaluation function is scale‑invariant)
    total = seq.sum()
    if total > 0.0:
        seq /= total
    else:
        seq = np.ones(n, dtype=np.float64) / n
    return seq


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a sequence of non‑negative heights that achieves a low
    evaluation score.  The method tries several values of the spread
    parameter L (both deterministic and randomised) and keeps the best
    result seen before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Length of the problem – fall back to 256 if the global is missing
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0:
            n = base.size
        else:
            n = 256
    else:
        n = 256

    # start with the trivial constant sequence (score = 2)
    best_seq = np.ones(n, dtype=np.float64) / n
    best_score = compute_score(best_seq, n)

    # deterministic L values (linear spacing of t_i)
    L_vals = [0.6, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]
    for L in L_vals:
        if time.monotonic() >= deadline:
            break
        seq = anti_diag_exp_sequence(n, L, rng, use_random_t=False)
        sc = compute_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq
            print(f"[deterministic] L={L:.2f}  score={sc:.6f}")

    # random trials – a few thousand at most
    trials = 0
    while time.monotonic() < deadline and trials < 2000:
        L = rng.uniform(0.6, 3.0)          # spread chosen on the fly
        seq = anti_diag_exp_sequence(n, L, rng, use_random_t=True)
        sc = compute_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq
            print(f"[random] L={L:.2f}  new best score={sc:.6f}")
        trials += 1

    # final clipping to satisfy the official evaluator
    final = np.clip(best_seq, 0.0, 1000.0).tolist()
    print(f"Final best score: {best_score:.6f}")
    return [float(x) for x in final]
