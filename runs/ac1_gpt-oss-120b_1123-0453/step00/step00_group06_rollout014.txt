import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Generate decreasing polynomial sequences and pick the best.
    The sequence a_i = ((n - i) / n) ** p yields a low max convolution.
    We try several integer powers p >=1 and return the one with minimal
    evaluate_sequence score.
    """
    # Ensure reproducibility
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine length to use: try to keep the existing length if possible
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
        if n == 0:
            raise ValueError
    except Exception:
        n = 256
    # Powers to explore (including linear, quadratic, cubic, etc.)
    powers = [1, 2, 3, 4, 5, 6]

    best_seq = None
    best_score = np.inf
    for p in powers:
        # build decreasing polynomial sequence
        idx = np.arange(n, dtype=np.float64)
        seq = ((n - idx) / n) ** p
        # Scale to keep max <= 1000 (not required after clipping but safe)
        max_val = seq.max()
        if max_val > 0:
            seq = seq * (1000.0 / max_val)
        # evaluate
        sc = float(evaluate_sequence(seq.tolist()))
        print(f"p={p}, score={sc:.6f}")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # If time allows, perform small random perturbations around the best solution
    while time.monotonic() < deadline:
        # jitter a few random positions
        candidate = best_seq.copy()
        # pick a random block width
        width = rng.integers(1, max(2, n // 50))
        start = rng.integers(0, n - width + 1)
        # apply log-normal perturbation
        factor = np.exp(rng.normal(0.0, 0.05, size=width))
        candidate[start:start+width] *= factor
        # keep values within bounds
        candidate = np.clip(candidate, 0.0, 1000.0)
        sc = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate
            print(f"Improved via perturbation: score={sc:.6f}")
    # Return result as plain list of floats
    return [float(x) for x in best_seq]

    a_i = ((n - i) / n) ** p   (i = 0 … n‑1)

    for p in powers:
        # decreasing polynomial: a_i = ((n-i)/n)**p   (max = 1)
        idx = np.arange(n, dtype=np.float64)
        seq = ((n - idx) / n) ** p          # values in (0,1]

        # scale to keep the biggest entry ≤1000 (clipping in the evaluator
        # would also achieve this, but keeping the scale explicit avoids
        # unnecessary loss of precision).
        seq *= 1000.0 / seq.max()

        score = float(evaluate_sequence(seq.tolist()))
        print(f"[grid] p={p:<2d} -> score={score:.6f}")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # If the grid search failed (should not happen), fall back to a
    # simple uniform sequence – this guarantees a finite return.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] uniform sequence, score =", best_score)

    # ------------------------------------------------------------------
    # Light‑weight local search: repeatedly rescale a random block.
    # This is very cheap (O(n) per iteration) and may discover tiny
    # improvements without risking a huge time consumption.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        candidate = best_seq.copy()
        # Choose a random block width (1 % – 2 % of n)
        width = rng.integers(1, max(2, n // 50))
        start = rng.integers(0, n - width + 1)

        # Log‑normal multiplicative perturbation (small sigma → fine tuning)
        factor = np.exp(rng.normal(0.0, 0.05, size=width))
        candidate[start:start + width] *= factor

        # Enforce the evaluator's bounds explicitly
        candidate = np.clip(candidate, 0.0, 1000.0)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate
            print(f"[local] improved score = {score:.6f}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
```
