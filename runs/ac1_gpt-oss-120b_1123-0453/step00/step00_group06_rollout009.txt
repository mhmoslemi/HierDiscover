"""
Propose a step‐function (non‑negative sequence) that minimises
        2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm is a simple projected sub‑gradient / mass‑relocation search:

* Start from the provided `height_sequence_1` (or a random Dirichlet if it
  is invalid) and normalise it to a probability vector (sum = 1).
* Repeatedly:
    – Compute the linear convolution a*a and locate the index `k_max`
      where the maximum occurs.
    – The contribution of entry i to that maximum is a[i] * a[k_max‑i]
      (zero when the partner index is out of bounds).
    – Choose the pair (i, j=k_max‑i) with the largest contribution and
      move a small fraction of its mass to two random positions.
      The move is projected back onto the probability simplex
      (non‑negative, sum = 1) which keeps the objective scale‑invariant.
    – If the modification improves the true `evaluate_sequence` score,
      keep it as the new current best.
    – Every few hundred iterations a cheap “block perturbation” (the
      original perturbation operator) is tried to escape possible local
      minima.
* The loop runs until the allocated time budget (≈1000 s) expires.
* The best sequence found is returned as a plain Python list of floats.

All operations are O(n log n) (FFT based convolution) and use only
standard NumPy; no external optimisation packages are required.
"""

import math
import time
import numpy as np

# -----------------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)

def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_max_and_full(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return (max(conv(a,a)), full convolution) using FFT.
    The convolution is linear (length 2*len(a)-1).
    """
    n = a.shape[0]
    n_fft = next_pow_two(2 * n - 1)
    # real FFT is slightly faster
    fa = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(fa * fa, n=n_fft)[: 2 * n - 1]
    return float(conv.max()), conv


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
        { w >= 0 , sum w = 1 }.
    Standard O(n) algorithm (see Duchi et al., 2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical guard – renormalise
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


def random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a random probability vector (Dirichlet(1,…,1))."""
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Mimic the block perturbation used in the reference code."""
    candidate = base.copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    candidate = np.clip(candidate, 0.0, 1000.0)
    # re‑normalise to keep the scale invariant
    if candidate.sum() == 0:
        candidate = random_dirichlet(rng, n)
    else:
        candidate = candidate / candidate.sum()
    return candidate
