def _is_valid_start(seq):
    """Check that the provided starting sequence is a finite 1‑D array."""
    try:
        arr = np.asarray(seq, dtype=np.float64)
    except Exception:  # pragma: no cover
        return False
    if arr.ndim != 1 or arr.size == 0:
        return False
    if not np.isfinite(arr).all():
        return False
    return True


def block_perturbation(base, rng):
    """
    Scale a random contiguous block of ``base`` by a log‑normal factor.
    The block width is chosen between n//200 and n//12 (inclusive) where
    n = len(base).  The result is clipped to the allowed interval [0,1000].
    """
    candidate = np.copy(base)
    n = candidate.size
    # ensure block size is at least 1
    min_width = max(1, n // 200)
    max_width = max(2, n // 12)
    width = int(rng.integers(min_width, max_width + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # log‑normal scaling (median 1, moderate variance)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # optional symmetrisation of this candidate
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce the security clipping
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


def noise_perturbation(base, rng):
    """
    Add small Gaussian noise to a random subset of positions.
    The number of perturbed indices is a small fraction of the length.
    """
    candidate = np.copy(base)
    n = candidate.size
    # roughly 0.5 % of entries are touched (but at least 1)
    k = max(1, int(rng.uniform(0.0, 0.005) * n))
    idx = rng.choice(n, size=k, replace=False)
    noise = rng.normal(0.0, 0.05, size=k)  # tiny noise
    candidate[idx] += noise
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


def symmetrise(candidate):
    """Replace the sequence by the average of itself and its reversal."""
    cand = 0.5 * (candidate + candidate[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

        parent = np.full(uniform_len, 1.0, dtype=np.float64)

    best = np.copy(parent)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[Init] Length={best.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing loop.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    it = 0
    # Temperature schedule: start relatively high, decay to near‑zero.
    max_temp = 0.5

    while time.monotonic() < deadline:
        it += 1
        # Choose a perturbation operator
        r = rng.random()
        if r < 0.5:
            cand = block_perturbation(best, rng)
        elif r < 0.8:
            cand = noise_perturbation(best, rng)
        else:
            cand = symmetrise(best)

        # Occasionally rotate the whole vector (zero‑cost but changes the
        # convolution pattern slightly).
        if rng.random() < 0.05:
            shift = int(rng.integers(0, cand.size))
            cand = np.roll(cand, shift)

        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            continue

        # Acceptance rule (simulated annealing)
        improve = score < best_score
        if improve:
            best = cand
            best_score = score
            print(f"[Iter {it}] New best score: {best_score:.6f}")
        else:
            # Linear temperature decay
            elapsed = time.monotonic() + (budget_s - deadline) - (deadline - budget_s + 3.0)
            fraction = (time.monotonic() - (deadline - budget_s + 3.0)) / (budget_s - 3.0)
            temp = max_temp * (1.0 - fraction)
            if temp > 1e-9:
                prob = np.exp(-(score - best_score) / temp)
                if rng.random() < prob:
                    best = cand
                    best_score = score
                    print(f"[Iter {it}] Accepted worse (prob={prob:.3e}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Fallback safety: if no valid candidate was discovered, return a
    #    uniform sequence (theoretical optimum).
    # ------------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        fallback_len = 512
        best = np.full(fallback_len, 1.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))
        print("[Fallback] Using uniform sequence.")

    print(f"[Done] Final best score: {best_score:.6f}, length={best.size}")
    # Convert to plain Python floats for the caller.
    return [float(x) for x in best]
