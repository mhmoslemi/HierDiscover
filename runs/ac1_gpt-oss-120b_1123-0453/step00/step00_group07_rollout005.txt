def compute_gradient(a, w):
    n = len(a)
    L = len(w)  # should be 2n-1
    # Reverse w
    w_rev = w[::-1]
    # convolution via FFT
    size = 1 << ((L + n - 1).bit_length())
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(A * W, size)[:L + n - 1]  # length 3n-2
    offset = L - 1
    g = 2.0 * conv[offset: offset + n]
    return g

"""
Simple gradient‑based optimizer for the step‑function evaluation problem.

The optimizer works on a *fixed* length sequence (the length of the
provided `height_sequence_1`).  The objective after normalising the sum to
1 is exactly

    score = 2 * n * max(conv(a, a))

so minimising the maximum of the auto‑convolution is sufficient.
A smooth approximation of the max (log‑sum‑exp) is used together with an
Adam update.  After each update the vector is projected onto the simplex
(sum = 1, a_i ≥ 0).  The algorithm runs as long as the time budget
allows and keeps the best sequence according to the exact
`evaluate_sequence` evaluator.  When the optimizer stalls a cheap
block‑perturbation heuristic (the one from the reference solution) is
applied to escape local minima.

The whole routine finishes within the allocated 1000 s and returns a
list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: block perturbation (the reference “make_block_perturbation”)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetry enforcement
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Small utility: next power‑of‑two (for FFT sizes)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – returns the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

# Main entry point required by the test harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The routine uses a smooth‑max gradient descent
    (Adam) followed by occasional block perturbations.  It respects the
    time budget and always returns a valid (finite) sequence.
    """
    rng = np.random.default_rng(seed)
    # give a small safety margin so we do not overrun the time limit
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = base_seq.size
    else:
        n = 6733          # default taken from the statement
        base_seq = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Normalisation: we keep the L1‑norm fixed to 1.0 (scale‑invariant).
    # ------------------------------------------------------------------
    target_sum = 1.0
    # start from the provided construction, but normalise it.
    base_seq = np.clip(base_seq, 0.0, 1000.0)
    if base_seq.ndim != 1 or base_seq.size != n or not np.isfinite(base_seq).all():
        base_seq = np.ones(n, dtype=np.float64) / n
    else:

        if s < 1e-12:

    eps_smooth_start = 0.4

        if run_id == 0:

        else:

            a = np.full(n, target_sum / n)

            # linearly anneal the temperature

            grad = 2.0 * corr_full[:n][::-1]    # shape (n,)

            t_adam += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----- 5) Projection: non‑negative and sum = 1 ------------
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, target_sum / n)
            else:
                a = a * (target_sum / s)

            # optional learning‑rate decay
            lr *= lr_decay

            # ----- 6) Periodic exact evaluation -----------------------
            if t_adam % 30 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[run {run_id:02d}] step {t_adam:05d} "
                        f"new best = {best_score:.6f}"
                    )

            # early stop if gradient becomes tiny (stalled)
            if np.linalg.norm(grad) < 1e-12:
                break

        # ----- End of one Adam run ------------------------------------
        run_id += 1

        # After each run we try a cheap block perturbation around the best
        # found so far – this mirrors the reference heuristic and often
        # helps to escape local minima.
        if time.monotonic() < deadline:
            cand = make_block_perturbation(best_seq, rng)
            cand_score = _exact_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = np.asarray(cand, dtype=np.float64)
                print(f"[perturb] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the allowed range).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
