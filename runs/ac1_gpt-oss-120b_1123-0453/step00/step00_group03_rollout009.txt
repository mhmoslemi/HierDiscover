Propose a step‑function (non‑negative sequence) that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The routine starts from a geometric (exponential) profile, which is already
better than the uniform “flat” sequence (score ≈ 1.8 < 2).  It then runs a
small number of linear‑programming refinements based on the classic
Frank–Wolfe style iteration described in the problem statement:

    * Let f be the current sequence and M = ||f * f||_∞.
    * Solve the LP          maximise   Σ g_i
                              subject to g_i ≥ 0
                                         (f * g)_k ≤ M  for all k
      (the convolution f*g is linear in g, therefore the constraints are
       linear).
    * Rescale g so that Σ g_i = √(2 n) (the same L1‑norm that the
      evaluation uses for a “normalized” vector).  Mix f and g:
          f ← (1‑t)·f + t·g   with a small t (here 0.1).

The LP guarantees that the new M cannot increase, and the mixing drives
the sequence towards a shape with a smaller convolution peak.  The loop
runs until the time budget expires, keeping the best‑scoring sequence
encountered.  If cvxpy or a suitable solver is unavailable the function
falls back to the simple block‑perturbation hill‑climber used in the
starter code.

The implementation respects the 1000 s budget, uses only O(n²) auxiliary
memory (a dense convolution matrix of size (2n‑1)×n) and works for any
length n (defaults to the length of the provided global

# Helper: a cheap block perturbation (fallback / occasional random tweak)

def _make_block_perturbation(base, rng):
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    s = int(rng.integers(0, max(1, n - w + 1)))
    e = min(n, s + w)
    cand[s:e] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    # Give a small safety margin for the bookkeeping/printing overhead.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Determine the sequence length (use the global reference if present)
    # --------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        if n == 0:
            raise Exception
    except Exception:
        n = 256          # a safe default when the global is missing

    target_sum = np.sqrt(2.0 * n)            # the “normalisation” used in the paper

    # --------------------------------------------------------------
    # Build an exponential start: a_i ∝ r^i   (r = exp(-c / n))
    # --------------------------------------------------------------
    c_start = 1.2                           # empirically good for n≈2500
    r = np.exp(-c_start / n)
    f = np.power(r, np.arange(n), dtype=np.float64)
    f = f / np.sum(f) * target_sum          # enforce Σf = √(2n)

    best_seq = f.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length={n}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try to import cvxpy; if that fails we will use a pure stochastic
    # fallback (the block‑perturbation hill‑climber).
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp
        have_cvx = True
    except Exception:                      # pragma: no cover
        have_cvx = False
        print("[info] cvxpy not available – using pure stochastic search.")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1. Compute the current convolution bound M = max(f * f)
        # --------------------------------------------------------------
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # --------------------------------------------------------------
        # 2. If cvxpy is usable, perform one LP refinement step
        # --------------------------------------------------------------
        if have_cvx:
            # Build the (2n‑1)×n matrix T that satisfies   T @ g = f * g
            rows = conv_ff.size               # = 2*n - 1
            T = np.zeros((rows, n), dtype=np.float64)
            for i in range(n):
                T[i:i + n, i] = f[i]          # each column is a shifted copy of f[i]

            # Linear programme: maximise Σ g_i  subject to  T @ g ≤ M , g ≥ 0
            g = cp.Variable(n, nonneg=True)
            constraints = [T @ g <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True,
                           max_iters=5000, abstol=1e-9, reltol=1e-9, feastol=1e-9)
            except Exception as exc:         # pragma: no cover
                print("[warn] LP solver failed:", exc)
                have_cvx = False
                continue

            if g.value is None:              # pragma: no cover
                print("[warn] LP returned no solution – terminating LP phase.")
                have_cvx = False
                continue

            g_val = np.asarray(g.value, dtype=np.float64).ravel()
            sum_g = g_val.sum()
            if sum_g == 0.0:                  # pragma: no cover
                # Degenerate solution – fallback to random tweak
                raise RuntimeError

            # Rescale to keep the L1‑norm fixed (Σ = √(2n))
            g_norm = g_val * (target_sum / sum_g)

            # ----------------------------------------------------------
            # 3. Mix the new candidate with the current sequence.
            #    A small mixing factor t guarantees non‑increase of M.
            # ----------------------------------------------------------
            t_mix = 0.10                       # conservative step size
            f_new = (1.0 - t_mix) * f + t_mix * g_norm
        else:
            # ----------------------------------------------------------
            # No LP – perform a cheap random block perturbation.
            # ----------------------------------------------------------
            f_new = _make_block_perturbation(f, rng)

        # --------------------------------------------------------------
        # 4. Evaluate the candidate; keep it if it improves the score.
        # --------------------------------------------------------------
        score = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = f_new.copy()
            best_score = score
            print(f"[iter {iteration:04d}] improved score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5. Update the working sequence (even if not improving) to let
        #    the LP keep moving forward.
        # --------------------------------------------------------------
        f = f_new

        # occasional random shake‑up to avoid stagnation (once every ~50 iters)
        if iteration % 47 == 0 and not have_cvx:
            f = _make_block_perturbation(f, rng)

    print(f"[done] best score found = {best_score:.6f}")
    # Ensure the output is a plain Python list of floats as required.
    return [float(x) for x in best_seq]
