    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # leave margin

    candidate = best_seq.copy()

    T0 = 1.0

        op = rng.integers(0, 4)
        if op == 0:
            new = block_perturb(candidate, rng)
        elif op == 1:
            new = single_scale(candidate, rng)
        elif op == 2:

                best_seq = new
                best_score = new_score

def gradient_step(seq, rng):
    n = len(seq)
    # compute convolution to locate max
    b = np.convolve(seq, seq)
    k = int(np.argmax(b))
    # find indices participating
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    # contributions for each i
    contrib = np.zeros(n, dtype=np.float64)
    contrib[mask] = seq[mask] * seq[j[mask]]
    # find top contributor
    i_max = int(np.argmax(contrib))
    j_max = int(k - i_max)
    # decide reduction factor
    factor = np.exp(rng.normal(-0.2, 0.05))  # less than 1
    candidate = seq.copy()
    candidate[i_max] = max(0.0, candidate[i_max] * factor)
    # also reduce j_max with some probability
    if rng.random() < 0.5:
        candidate[j_max] = max(0.0, candidate[j_max] * factor)
    # increase some random index not equal to i_max/j_max
    idx_inc = int(rng.integers(0, n))
    inc = rng.normal(0.0, 5.0)  # small positive
    if inc > 0:
        candidate[idx_inc] = min(1000.0, candidate[idx_inc] + inc)
    # clip
    candidate = np.clip(candidate, 0.0, 1000.0)
    return candidate
```

def generate_sidon_positions(n, m, rng):
    positions = []
    sumset = set()
    attempts = 0
    while len(positions) < m and attempts < 10000:  # limit loops
        pos = rng.integers(0, n)
        # compute sums with existing positions (including pos+pos)
        collisions = False
        for p in positions:
            if (pos + p) in sumset:
                collisions = True
                break
        if not collisions and (pos + pos) not in sumset:
            positions.append(int(pos))
            for p in positions:
                sumset.add(pos + p)
        attempts += 1
    return positions
```

"""
Propose a step‑function sequence that beats the 2.0 upper bound.

The idea is to flatten the autocorrelation (the convolution a*a) by
* putting a few large values (1000) at positions forming a Sidon set,
  i.e. all pairwise sums are distinct.  This guarantees that any shift
  of the convolution receives at most one ordered pair of large
  entries, so the peak contributed by the large entries is at most
  2·1000² = 2 000 000.
* filling the remaining positions with a modest constant value (≈20).
  The contribution of the many small entries to any shift is then
  well below the peak coming from the large values.

With roughly √n large entries (≈60 for n≈3800) the sum of the
coefficients is large while the maximal convolution value stays
near 2 000 000, giving an evaluation ≈0.8 << 1.5.

The function builds such a sequence, checks its score and falls back
to a simple uniform sequence if something goes wrong.
"""
import time
import numpy as np

def _generate_sidon_positions(n: int, m: int, rng: np.random.Generator) -> list[int]:
    """
    Greedy construction of a Sidon set (all i+j are distinct) of size m
    inside {0,…,n‑1}.  Returns the list of positions, possibly shorter
    than m if the construction fails after many attempts.
    """
    positions: list[int] = []
    sums: set[int] = set()
    attempts = 0
    max_attempts = 50000
    while len(positions) < m and attempts < max_attempts:
        pos = int(rng.integers(0, n))
        # check for collisions with existing sums (including self‑sum)
        if (pos + pos) in sums:
            attempts += 1
            continue
        collision = False
        for p in positions:
            if (pos + p) in sums:
                collision = True
                break
        if collision:
            attempts += 1
            continue
        # no collision -> accept the position
        for p in positions:
            sums.add(pos + p)
        sums.add(pos + pos)
        positions.append(pos)
        attempts = 0  # reset after a successful addition
    return positions
