iteration (the “LP‑perturbation” method described in the statement).

The algorithm:
  * Start from a slightly perturbed uniform vector (length ≤256 for speed).
  * Compute M = max(conv(f,f)).

  * Repeat while time permits, keeping the best vector seen.

The LP is tiny (≤256 variables, ≤511 constraints), so many iterations

def _make_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build the (2n‑1) × n matrix A such that A @ g = conv(f,g)."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):

        g_start = k - i_end               # first column of g that participates

    return A

            cand *= target_sum / np.sum(cand)
            score = float(evaluate_sequence(cand.tolist()))
            if score < best_score:
                best_seq, best_score = cand, score
                print(f"[{iteration}] Random fallback improved score to {best_score:.6f}")
            continue

        A = _make_conv_matrix(f)

        # ----- solve the LP: maximise sum(g) subject to conv(f,g) ≤ M -----
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2 * n - 1), g >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        # Warm‑start is cheap for this tiny problem; ECOS is fast for LPs.
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:          # fall back to a more robust (but slower) solver
            prob.solve(solver=cp.SCS, verbose=False)

        if g.value is None:        # solver failed – exit the LP loop
            break

        sum_g = float(np.sum(g.value))
        if sum_g <= 0.0:
            break

        # ----- rescale g to keep the total mass unchanged -----
        sum_f = float(np.sum(f))
        scale = sum_f / sum_g
        g_scaled = np.array(g.value, dtype=np.float64) * scale

        # ----- convex update -----
        f_new = (1.0 - t) * f + t * g_scaled
        f_new = np.clip(f_new, 0.0, 1000.0)

        # ----- evaluate the candidate -----
        new_score = float(evaluate_sequence(f_new.tolist()))
        if math.isfinite(new_score) and new_score < best_score:
            best_seq, best_score = f_new.copy(), new_score
            print(f"[{iteration}] Improved score to {best_score:.6f}")

        # ----- decide whether to accept the update -----
        # Theory guarantees non‑increase of the max‑convolution for any
        # 0 < t ≤ 1, but we keep a small safety margin.
        if new_score <= best_score * 1.001:
            f = f_new
        else:
            # No progress – shrink the step size a little.
            t = max(0.05, t * 0.7)

        # Early exit if we have already achieved the target region.
        if best_score < 1.503:
            break

    # Return the best sequence found (as plain Python floats).
    return [float(v) for v in best_seq]
