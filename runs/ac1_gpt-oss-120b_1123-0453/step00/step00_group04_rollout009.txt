def propose_targeted_move(a, rng, step_factor):
    n = a.size
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))
    # Compute product contributions for the max shift
    # create array of length n: product[i] = a[i] * a[k_max - i] if in range else 0
    # using mask
    i = np.arange(n)
    j = k_max - i
    mask = (j >= 0) & (j < n)
    prod = np.zeros(n)
    prod[mask] = a[i[mask]] * a[j[mask]]
    i_max = int(np.argmax(prod))
    partner = k_max - i_max
    # find index with smallest a (lowest value)
    j_min = int(np.argmin(a))
    # ensure we are not moving from same index to itself
    if j_min == i_max:
        # pick second smallest
        sorted_idx = np.argsort(a)
        for idx in sorted_idx:
            if idx != i_max:
                j_min = idx
                break
    delta = step_factor * a[i_max]
    if delta <= 0:
        return None
    cand = a.copy()
    cand[i_max] -= delta
    cand[j_min] += delta
    # ensure non-negatives
    # (cand[i_max] might become negative due to floating errors; clip)
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

"""
A simple heuristic search for a step‑function that minimizes the
evaluation metric:

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works with a fixed length (the length of the provided
`height_sequence_1`).  It repeatedly mutates the current best sequence
using two complementary moves:

* **Block perturbation** – a multiplicative change of a random contiguous
  block (the same idea as the reference implementation).
* **Exchange move** – a small amount of mass is transferred from a pair
  of positions whose product contributes heavily to the current maximum
  convolution value to a pair whose product is small.  This directly
  reduces the peak while keeping the total sum unchanged.

Both mutation types are cheap (O(n) memory, O(n log n) time for the
convolution) and are evaluated with the authoritative
`evaluate_sequence` function.  The step size of the exchange move is
gradually decreased (simulated‑annealing style) and a few random
re‑starts are performed if the search gets stuck.

The implementation is fully self‑contained, uses only NumPy and the
standard library, and respects the required 1000 s time budget.
"""

import numpy as np
import time
from typing import List

# ----------------------------------------------------------------------
# Helper functions (must be top level – no closures)
# ----------------------------------------------------------------------


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative O(n)‑memory perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # block width between n/200 and n/12 (inclusive)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal noise on the block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # occasional symmetrisation (helps to balance the reverse pair products)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def exchange_move(a: np.ndarray, rng: np.random.Generator, step_factor: float) -> np.ndarray:
    """
    Transfer a small amount of mass from a high‑product pair (i, n‑1‑i)
    to a low‑product pair (j, n‑1‑j).  The total sum stays unchanged,
    and the move explicitly tries to reduce the current peak of the
    convolution.
    """
    n = a.size
    # product of each element with its mirror (these are the contributions
    # to the centre of the convolution)
    pair_prod = a * a[::-1]

    # pick the highest and the lowest products
    i_high = int(np.argmax(pair_prod))
    j_low = int(np.argmin(pair_prod))

    i_partner = n - 1 - i_high
    j_partner = n - 1 - j_low

    # ensure we are not moving zero mass
    delta = step_factor * min(a[i_high], a[i_partner] if i_partner != i_high else a[i_high])
    if delta <= 0.0:
        return None

    cand = a.copy()
    # subtract from the high pair
    cand[i_high] -= delta
    if i_partner != i_high:
        cand[i_partner] -= delta
    else:
        # the middle element (odd length) appears only once in the centre
        pass

    # add to the low pair
    cand[j_low] += delta
    if j_partner != j_low:
        cand[j_partner] += delta

    # clip to keep everything in the allowed range
    cand = np.clip(cand, 0.0, 1000.0)
    return cand


def random_dirichlet_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point on the simplex (sum = 1)."""
    seq = rng.dirichlet(np.ones(n))
    return np.clip(seq, 0.0, 1000.0)

    Search for a sequence of non‑negative heights that minimizes the provided
    ``evaluate_sequence`` function.  The function returns the best sequence it

    Parameters

    seed : int, optional
        Random seed for reproducibility.

    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

        parent = np.ones(256, dtype=np.float64)

    total = float(parent.sum())

        total = float(parent.sum())
    a = parent / total  # now sum(a) == 1.0

    best = a.copy()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    iteration = 0
    no_improve = 0
    step_factor = 0.12          # initial magnitude for the exchange move
    decay = 0.9994              # slow decay of step_factor each iteration

    n = a.size

    while time.monotonic() < deadline:
        iteration += 1

            cand = exchange_move(best, rng, step_factor)
            if cand is None:
                # fallback – simple block perturbation
                cand = make_block_perturbation(best, rng)
        else:
            # pure block perturbation (exploration)
            cand = make_block_perturbation(best, rng)

        # evaluate the candidate
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            # should never happen because evaluate_sequence already guards,
            # but keep the loop robust
            continue

        # accept only improvements (strictly smaller score)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            no_improve = 0
            print(f"[{iteration:06d}] new best: {best_score:.6f}")
        else:
            no_improve += 1

        # occasional random restart when we have been idle for a while
        if no_improve > 8000:
            # re‑initialise with a fresh Dirichlet vector of the same length
            cand = random_dirichlet_sequence(n, rng)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                print(f"[{iteration:06d}] restart improved to {best_score:.6f}")
                no_improve = 0
            else:
                # accept the restart anyway to escape a flat region
                best = cand
                no_improve = 0

        # slowly shrink the step size – analogous to a cooling schedule
        step_factor = max(0.001, step_factor * decay)

    print(f"[done] iterations={iteration}, best_score={best_score:.6f}")
    # Return the best sequence found (as plain Python floats)
    return [float(x) for x in best]
