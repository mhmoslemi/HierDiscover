"""
Propose a candidate step‑function sequence.

The evaluation metric is  

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where *conv* denotes the linear (full) convolution.  
For any non‑negative sequence the maximum of the convolution is attained at
lag 0 and equals sum_i a_i².  By the Cauchy–Schwarz inequality

    sum_i a_i² >= (sum_i a_i)² / n

with equality iff all entries are equal.  Consequently

    score >= 2

for every admissible sequence, and the bound 2 is attained by any constant
sequence.  Hence a uniform (constant) sequence is provably optimal.

The function below returns such a constant sequence.  Its length is taken
from the supplied global `height_sequence_1` (fallback: 256).  The constant
value is chosen as `sqrt(2·n) / n` so that the sum of the sequence equals
`sqrt(2·n)`, the normalisation used in the reference algorithm – any other
positive constant would give the same score.

The implementation is deliberately simple: no search, no loops, and it
fits comfortably inside the allowed time budget.
"""

import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Return a uniform sequence that achieves the theoretical optimum
    (score = 2).  The length is derived from the global `height_sequence_1`
    if present; otherwise a default length of 256 is used.

    Parameters
    ----------
    seed : int (unused)
        Kept for compatibility with the evaluation harness.
    budget_s : float (unused)
        Kept for compatibility; the function does not need an iterative
        budget because the solution is known analytically.
    **kwargs :
        Ignored.

    Returns
    -------
    list[float]
        The candidate sequence.
    """
    # Determine the target length.
    try:
        # height_sequence_1 is provided by the surrounding environment.
        base = height_sequence_1               # type: ignore[name-defined]
        n = int(np.asarray(base).size)
        if n <= 0:
            n = 256
    except Exception:
        # Fallback if the global is missing or malformed.
        n = 256

    # Constant value – any positive constant works; we choose the normalised
    # value sqrt(2 n) / n to keep the sum at sqrt(2 n).
    const_val = math.sqrt(2.0 * n) / n
    # Clamp to the range enforced by the evaluator.
    const_val = max(0.0, min(1000.0, const_val))

    # Build the sequence.
    sequence = [float(const_val) for _ in range(n)]

    # Optional diagnostic print – harmless for the scoring system.
    try:
        # evaluate_sequence is assumed to be available in the global scope.
        score = evaluate_sequence(sequence)   # type: ignore[name-defined]
        print(f"[propose_candidate] length={n}, const={const_val:.6f}, score={score:.12f}")
    except Exception as exc:
        # If evaluation is unavailable (e.g., during dry‑run) we simply skip.
        print(f"[propose_candidate] diagnostic evaluation failed: {exc}")

    return sequence
