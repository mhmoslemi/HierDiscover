def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    try:
        import cvxpy as cp
        HAS_CVX = True
    except Exception:
        HAS_CVX = False
    
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Small function to evaluate
    def evaluate(seq):
        return float(evaluate_sequence(seq.tolist()))
    
    # Fallback: use simple block perturbation if cvxpy unavailable
    def simple_search():
        nonlocal best_seq, best_score
        # start from constant
        n = 200
        seq = np.full(n, 1.0, dtype=np.float64)
        seq = seq / np.sum(seq) * math.sqrt(2*n)  # normalize to target sum
        best_seq = seq.copy()
        best_score = evaluate(seq)
        while time.monotonic() < deadline:
            candidate = seq + rng.normal(scale=0.1, size=n)
            candidate = np.clip(candidate, 0.0, 1000.0)
            score = evaluate(candidate)
            if score < best_score:
                best_seq = candidate.copy()
                best_score = score
                seq = candidate
        return
    
    # Primary optimization using LP
    if HAS_CVX:

        lengths = [50, 100, 200]  # moderate sizes
        for n in lengths:
            # initialize random f
            f = rng.random(n)
            # ensure non-zero sum
            if np.sum(f) < 1e-8:
                f = np.ones(n)
            target_sum = math.sqrt(2.0 * n)
            f = f / np.sum(f) * target_sum
            # iterative improvement
            t = 0.3
            for iteration in range(20):

                    break

                M = float(np.max(conv))

                A = np.zeros((2*n-1, n), dtype=np.float64)
                for k in range(2*n-1):
                    i_start = max(0, k-(n-1))
                    i_end = min(k, n-1)

                    #     col = k - i
                    #     A[k, col] = f[i]

"""
Propose a low‑value step sequence by iteratively enlarging its ℓ¹‑mass
while keeping the worst auto‑convolution entry fixed.
The core of the method is a small linear program:

    max   Σ g_i
    s.t.  (f * g)_k ≤ M   for all k
          g_i ≥ 0

where * is the linear (full) convolution, f is the current sequence,
M = max(f*f) is its worst convolution value, and g is the new candidate.
Because the constraints are linear in g, the program is a standard LP
(handled by CVXPY/ECOS).  After solving we rescale g so that the total
sum equals the target √(2·n), and replace f by a convex combination
(1−t) f + t g.  Repeating this step drives the ratio
2·n·max(f*f)/sum(f)² downwards – the quantity the scorer returns.
If CVXPY is unavailable or the LP fails to converge, a simple
random‑perturbation fallback is used.

The routine respects the security checks of `evaluate_sequence`,
clips values to the allowed interval [0, 1000] and returns the best
sequence found before the time budget expires.
"""
import time
import math
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a sequence of non‑negative heights with a low evaluation score.

    The implementation tries a few LP‑based refinements of a random
    start vector; if that is not possible it falls back to random
    local search.  The function respects the time budget.
    """
    # ------------------------------------------------------------------
    # Helper: evaluate a candidate (global evaluate_sequence is provided)
    def _score(seq):
        return float(evaluate_sequence(seq.tolist()))

    # ------------------------------------------------------------------
    # Helper: build convolution matrix A where (A @ g)[k] = (f * g)[k]
    def _conv_matrix(f):
        n = len(f)
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                col = k - i                 # index of g that pairs with f[i]
                A[k, col] = f[i]
        return A

    # ------------------------------------------------------------------
    # Simple random‑perturbation fallback (similar to the provided baseline)
    def _simple_search():
        nonlocal best_seq, best_score
        print("[fallback] simple random search")
        n = 200
        # start with a uniform sequence that already yields score == 2.0
        seq = np.full(n, 1.0, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        best_seq = seq.copy()
        best_score = _score(seq)
        while time.monotonic() < deadline:
            cand = seq + rng.normal(scale=0.05, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best_seq = cand.copy()
                best_score = sc
                seq = cand
        print("[fallback] best score =", best_score)

    # ------------------------------------------------------------------
    # Main routine
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")

    # Try to use CVXPY for the LP‑based improvement
    try:
        import cvxpy as cp
        HAS_CVX = True
    except Exception:                     # pragma: no cover
        HAS_CVX = False

    if HAS_CVX:
        # try a few problem sizes (small enough for a fast LP)
        for n in (50, 100, 200):
            if time.monotonic() >= deadline:
                break

            # target ℓ¹‑norm (the original code normalises to √(2n))
            target_sum = math.sqrt(2.0 * n)

            # start from a random non‑negative vector with the correct sum
            f = rng.random(n) + 0.1               # avoid exact zeros
            f = f / np.sum(f) * target_sum

            # mixing parameter – small enough to keep monotone improvement
            t = 0.3

            for it in range(30):
                if time.monotonic() >= deadline:
                    break

                # current worst auto‑convolution value
                M = float(np.max(np.convolve(f, f)))

                # build the (2n‑1)×n convolution matrix for constraints
                A = _conv_matrix(f)

                # LP: maximize Σ g_i  subject to A·g ≤ M, g ≥ 0
                g = cp.Variable(n)
                constraints = [g >= 0, A @ g <= M]
                prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

                # ECOS is fast for dense LPs; fall back to SCS if needed
                try:
                    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
                except Exception:                     # pragma: no cover
                    prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000)

                if g.value is None:
                    break   # LP failed – abort this size

                g0 = np.maximum(g.value, 0.0)          # numerical safety
                sum_g = np.sum(g0)
                if sum_g <= 1e-12:
                    break

                # rescale g so that it has the same ℓ¹‑norm as f
                g_rescaled = g0 * (target_sum / sum_g)

                # convex combination step
                f_new = (1.0 - t) * f + t * g_rescaled
                f_new = np.clip(f_new, 0.0, 1000.0)

                # evaluate the new candidate
                sc = _score(f_new)
                if sc < best_score:
                    best_score = sc
                    best_seq = f_new.copy()
                    print(f"[LP] n={n} it={it} score={sc:.6f}")

                # stop early if we already beat the target
                if best_score < 1.51:
                    break

                f = f_new

    # If no LP run succeeded (or CVXPY unavailable) use the fallback
    if best_seq is None:
        _simple_search()

    # Final safety: ensure we return a plain Python list of floats
    if best_seq is None:
        # a deterministic safe fallback
        best_seq = np.ones(256, dtype=np.float64)

    return [float(x) for x in best_seq]
