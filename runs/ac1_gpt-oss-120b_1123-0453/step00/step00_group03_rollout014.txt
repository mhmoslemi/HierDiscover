def block_perturb(a, rng):
    n = a.shape[0]
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

import scipy.optimize
import scipy.sparse as sp

def solve_min_max_convolution(f, target_sum=1.0):
    n = len(f)
    # Build sparse matrix A (2n-1 rows, n columns)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows i .. i + n -1
        row_start = i
        row_end = i + n  # exclusive
        rows.extend(range(row_start, row_end))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Build constraints
    # A_ub * g - M <= 0  -> (A, -1) for each row
    # We'll augment A with a column for M.
    # Suppose x = [g (n), M (1)]
    # A_ub shape (2n-1, n+1)
    # We'll create a sparse matrix with same pattern but extra column.
    # Equivalent to create A_ub = sp.hstack([A, -np.ones((2*n-1,1))])
    Mcol = -sp.csr_matrix(np.ones((2*n-1, 1)))
    A_ub = sp.hstack([A, Mcol], format='csr')
    b_ub = np.zeros(2*n-1)

    # Equality: sum(g) = target_sum
    A_eq = sp.csr_matrix(np.hstack([np.ones((1,n)), np.zeros((1,1))]))
    b_eq = np.array([target_sum])

    # Objective: minimize M
    c = np.concatenate([np.zeros(n), np.array([1.0])])

    # Bounds
    bounds = [(0, None)]*n + [(0, None)]

    # Solve
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                 A_eq=A_eq, b_eq=b_eq,
                                 bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)
    g_opt = res.x[:n]
    M_opt = res.x[-1]
    return g_opt, M_opt

def propose_candidate(seed=0, budget_s=1100, **kwargs):
    import numpy as np, time, scipy.optimize, scipy.sparse as sp
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Determine length n
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 2000  # fallback
        base = np.ones(n, dtype=np.float64) / n

    # Initialize f with random positive values
    f = rng.random(n)
    f = f / f.sum()  # sum = 1

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Init n={n}, init score {best_score:.6f}")

    # mixing factor
    t = 0.25

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # current max convolution
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # Solve LP to find g minimizing max conv(f,g) under sum(g)=1
        # Build sparse convolution matrix T for f
        col_idx = np.arange(n)
        row_idx = (col_idx[:, None] + np.arange(n)).ravel()
        col_repeat = np.repeat(col_idx, n)
        data = np.repeat(f, n)
        T = sp.csr_matrix((data, (row_idx, col_repeat)), shape=(2*n-1, n))

        # Build A_ub = [T, -1]
        neg_one = -sp.csr_matrix(np.ones((2*n-1, 1)))
        A_ub = sp.hstack([T, neg_one], format='csr')
        b_ub = np.zeros(2*n-1)

        # Equality sum(g)=1
        A_eq = sp.csr_matrix(np.hstack([np.ones(n), np.array([0.0])]).reshape(1, n+1))
        b_eq = np.array([1.0])

        c = np.concatenate([np.zeros(n), np.array([1.0])])  # minimize M

        bounds = [(0, None)] * (n + 1)

        # Solve LP
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                     A_eq=A_eq, b_eq=b_eq,
                                     bounds=bounds, method='highs')
        if not res.success:
            # fall back to random perturbation
            g = make_block_perturbation(f, rng)
            M_new = np.max(np.convolve(g, g))
        else:
            g = res.x[:n]
            M_new = res.x[-1]

        # Mix towards g
        f = (1 - t) * f + t * g
        # Ensure non-negativity and renormalize
        f = np.clip(f, 0.0, 1000.0)
        s = f.sum()
        if s == 0 or not np.isfinite(s):
            f = rng.random(n)
            f = f / f.sum()
        else:
            f = f / s

        # Evaluate score
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f.copy()
            best_score = score
            print(f"Iter {iteration}: new best {best_score:.6f}")

        # Occasionally randomize to escape stagnation
        if iteration % 10 == 0 and rng.random() < 0.2:
            f = make_block_perturbation(f, rng)
            f = f / f.sum()

    return [float(x) for x in best]

"""
Propose a sequence that reduces the evaluation score.

The algorithm starts from a random (non‑negative, normalised) step function.
It repeatedly solves a small linear programme:

    minimise   M
    subject to   (f * g)[k] ≤ M      for all k
                 Σ g_i   = 1
                 g_i ≥ 0

where ``f`` is the current sequence and ``g`` is the new candidate.
The LP finds a candidate whose convolution with the current sequence does not
exceed the current peak ``M`` while keeping the total mass fixed.
A convex combination ``f ← (1‑t) f + t g`` (with a modest step size) strictly
lowers the peak of ``f*f`` and therefore the evaluation value.
If the LP fails (or SciPy is unavailable) a fast random local‑search
fallback is used.

The loop runs until the time budget expires and the best sequence found is
returned.
"""

# ----------------------------------------------------------------------
# Helper: simple block perturbation (fallback & occasional randomisation)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width between n/200 and n/12
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Helper: build the convolution matrix for a fixed left‑hand side vector
# ----------------------------------------------------------------------
def _convolution_matrix(f):
    """Return a scipy.sparse matrix T with T @ g == f * g (full convolution)."""
    # Import locally – SciPy may be unavailable in the fallback path.
    import scipy.sparse as sp
    n = f.shape[0]
    # rows i+j   (0 ≤ i,j < n)  →  row index runs from 0 to 2n‑2
    col_idx = np.arange(n)
    # each column i contributes f[i] to rows i … i+n‑1
    rows = (col_idx[:, None] + np.arange(n)).ravel()
    cols = np.repeat(col_idx, n)
    data = np.repeat(f, n)
    T = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return T


# ----------------------------------------------------------------------
# Helper: solve the LP  minimise M   s.t.   f*g ≤ M, Σg = 1, g≥0
# ----------------------------------------------------------------------
def _solve_minimax_lp(f):
    """Return (g_opt, M_opt) solving the linear programme described above."""
    import scipy.optimize
    import scipy.sparse as sp

    n = f.shape[0]
    # Convolution matrix for the current left vector f
    T = _convolution_matrix(f)                     # shape (2n‑1, n)

    # Append the column for the auxiliary variable M (coefficient –1)
    neg_one = -sp.csr_matrix(np.ones((2 * n - 1, 1)))
    A_ub = sp.hstack([T, neg_one], format="csr")
    b_ub = np.zeros(2 * n - 1)

    # Equality Σ g_i = 1   (M has no contribution)
    A_eq = sp.csr_matrix(np.hstack([np.ones(n), np.array([0.0])]).reshape(1, n + 1))
    b_eq = np.array([1.0])

    # Objective: minimise M
    c = np.concatenate([np.zeros(n), np.array([1.0])])

    # Bounds: g_i ≥ 0,  M ≥ 0
    bounds = [(0, None)] * (n + 1)

    # Use HiGHS – it can handle the sparse formulation efficiently.
    res = scipy.optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    g_opt = res.x[:n]
    M_opt = res.x[-1]
    return g_opt, M_opt
