"""
Greedy binary construction of a step function with low convolution peak.

The algorithm builds a 0/1 sequence (height values) by adding positions one by
one.  For each candidate position the resulting maximum value of the full
self‑convolution (the quantity that the evaluation function depends on) is
computed.  The candidate that yields the smallest possible new maximum is
chosen, thus keeping the autocorrelation as flat as possible while the number
of non‑zero entries grows.  Since the objective
    2·n·max(b) / (sum(a))²
depends only on the shape of the sequence, a binary vector with many
‘1’s and a low convolution peak gives a score well below the uniform baseline
(2.0).  The search runs until the time budget expires and returns the best
sequence seen.
"""
import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Time budget: leave a small margin for cleanup
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----------------------------------------------------------------------
    # Determine the length n from the provided global variable, or fall back.
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = base.size
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Start from the given height sequence as a fallback; its score is the
    # initial best score.
    # ----------------------------------------------------------------------
    parent = np.clip(base, 0.0, 1000.0)
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial score (parent) = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Greedy binary construction.
    # ----------------------------------------------------------------------
    # a binary vector a, stored as a boolean mask
    selected_mask = np.zeros(n, dtype=bool)
    selected = []                       # list of indices with value 1
    b_counts = np.zeros(2 * n - 1, dtype=np.int32)   # convolution counts
    current_max = 0                     # max(b_counts)

    # Parameters for the greedy loop
    SAMPLE_SIZE = 200   # how many random candidates to evaluate per iteration
    LOG_EVERY = 5       # evaluate and log every LOG_EVERY additions

    iteration = 0
    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # Choose a set of candidate positions that are still unselected.
        # ------------------------------------------------------------------
        remaining = np.where(~selected_mask)[0]
        if remaining.size == 0:
            break
        # Randomly sample a subset (or all if very few left)
        cand_size = min(SAMPLE_SIZE, remaining.size)
        cand_indices = rng.choice(remaining, size=cand_size, replace=False)

        # ------------------------------------------------------------------
        # For each candidate compute the resulting new maximum of the
        # convolution (without actually applying the update).
        # ------------------------------------------------------------------
        best_i = None
        best_new_max = None

        # Pre‑convert selected list to a NumPy array for fast arithmetic
        sel_arr = np.array(selected, dtype=np.int64) if selected else None

        for i in cand_indices:
            # Contribution of the new self‑pair (i,i)
            new_max_i = b_counts[2 * i] + 1

            if sel_arr is not None and sel_arr.size > 0:
                # All ordered pairs (i, j) with j already selected
                sums = i + sel_arr                     # integer array
                # These sums get +2 each
                vals = b_counts[sums] + 2
                # The worst of these may dominate the new max
                mx = vals.max()
                if mx > new_max_i:
                    new_max_i = mx

            # The overall new maximum is the larger of the current maximum
            # and the new contributions for this candidate.
            new_max = new_max_i if new_max_i > current_max else current_max

            # Keep the candidate that yields the smallest new maximum.
            if best_new_max is None or new_max < best_new_max:
                best_new_max = new_max
                best_i = i
                # early exit if we have found a candidate that does not raise the max
                if new_max == current_max:
                    break

        # ------------------------------------------------------------------
        # Apply the best candidate.
        # ------------------------------------------------------------------
        i = int(best_i)
        # update convolution counts
        b_counts[2 * i] += 1
        if sel_arr is not None and sel_arr.size > 0:
            sums = i + sel_arr
            b_counts[sums] += 2
        selected_mask[i] = True
        selected.append(i)
        iteration += 1

        # Keep current maximum up‑to‑date
        if best_new_max > current_max:
            current_max = best_new_max

        # --------------------------------------------------------------
        # Compute the objective value from the current state.
        # --------------------------------------------------------------
        K = len(selected)
        # The evaluation function depends only on n, max(b) and sum(a)=K.
        score = 2.0 * n * float(current_max) / (K * K)

        # Occasionally evaluate the exact function (mostly for debugging)
        if iteration % LOG_EVERY == 0:
            # Build the explicit sequence (float 0.0 / 1.0)
            seq = np.zeros(n, dtype=np.float64)
            seq[selected] = 1.0
            exact = float(evaluate_sequence(seq.tolist()))
            print(
                f"Iter {iteration:4d}: K={K}, max_b={current_max}, "
                f"approx_score={score:.6f}, exact_score={exact:.6f}"
            )
        # Update best solution if we improved
        if score < best_score:

"""
Greedy binary construction of a step function with low convolution peak.

The algorithm builds a 0/1 sequence (height values) by adding positions one by
one.  For each candidate position the resulting maximum value of the full
self‑convolution (the quantity that the evaluation function depends on) is
computed.  The candidate that yields the smallest possible new maximum is
chosen, thus keeping the autocorrelation as flat as possible while the number
of non‑zero entries grows.  Since the objective

    2·n·max(b) / (sum(a))²

depends only on the shape of the sequence, a binary vector with many ‘1’s and a
low convolution peak gives a score well below the uniform baseline (2.0).
The search runs until the time budget expires and returns the best sequence
seen.
"""

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a sequence of non‑negative heights that minimizes evaluate_sequence.

    The function performs a greedy construction of a binary (0/1) sequence.
    It repeatedly adds a new index that raises the maximum of the self‑convolution
    as little as possible, thereby flattening the convolution and reducing the
    evaluation score.  The best sequence encountered during the run is returned.
    """
    rng = np.random.default_rng(seed)
    # leave a small margin for cleanup before the external timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----------------------------------------------------------------------
    # Determine the length of the target sequence.
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = base.size
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Use the provided sequence as a fallback initial best solution.
    # ----------------------------------------------------------------------
    parent = np.clip(base, 0.0, 1000.0)
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial score (parent) = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Greedy binary construction.
    # ----------------------------------------------------------------------
    selected_mask = np.zeros(n, dtype=bool)   # True where a_i = 1
    selected = []                              # list of indices with value 1
    b_counts = np.zeros(2 * n - 1, dtype=np.int32)  # convolution (ordered) counts
    current_max = 0                             # max(b_counts)

    SAMPLE_SIZE = 200   # number of random candidates examined each iteration
    LOG_EVERY = 5       # evaluate and log progress every LOG_EVERY additions

    iteration = 0
    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # Choose a random subset of still‑unused positions to evaluate.
        # ------------------------------------------------------------------
        remaining = np.where(~selected_mask)[0]
        if remaining.size == 0:
            break
        cand_sz = min(SAMPLE_SIZE, remaining.size)
        cand_indices = rng.choice(remaining, size=cand_sz, replace=False)

        # ------------------------------------------------------------------
        # Find the candidate that yields the smallest possible new maximum.
        # ------------------------------------------------------------------
        best_i = None
        best_new_max = None
        sel_arr = np.array(selected, dtype=np.int64) if selected else None

        for i in cand_indices:
            # self‑pair (i,i) contributes +1 to b[2*i]
            new_max_i = b_counts[2 * i] + 1

            if sel_arr is not None and sel_arr.size > 0:
                # ordered pairs (i, j) with j already selected add +2 to b[i+j]
                sums = i + sel_arr                # vector of sums
                vals = b_counts[sums] + 2
                mx = vals.max()
                if mx > new_max_i:
                    new_max_i = mx

            # overall new maximum after inserting i
            new_max = new_max_i if new_max_i > current_max else current_max

            if best_new_max is None or new_max < best_new_max:
                best_new_max = new_max
                best_i = i
                # perfect candidate – does not increase the current maximum
                if new_max == current_max:
                    break

        # ------------------------------------------------------------------
        # Apply the best candidate.
        # ------------------------------------------------------------------
        i = int(best_i)
        b_counts[2 * i] += 1
        if sel_arr is not None and sel_arr.size > 0:
            sums = i + sel_arr
            b_counts[sums] += 2
        selected_mask[i] = True
        selected.append(i)
        iteration += 1

        if best_new_max > current_max:
            current_max = best_new_max

        # --------------------------------------------------------------
        # Compute the current objective (approximation, exact evaluation is cheap).
        # --------------------------------------------------------------
        K = len(selected)                       # sum of a_i (binary → K)
        approx_score = 2.0 * n * float(current_max) / (K * K)

        if iteration % LOG_EVERY == 0:
            # build the explicit sequence for an exact check
            seq = np.zeros(n, dtype=np.float64)
            seq[selected] = 1.0
            exact = float(evaluate_sequence(seq.tolist()))
            print(
                f"Iter {iteration:4d}: K={K}, max_b={current_max}, "
                f"approx_score={approx_score:.6f}, exact_score={exact:.6f}"
            )

        # Keep the best sequence seen so far.
        if approx_score < best_score:
            best_score = approx_score
            best_seq = np.zeros(n, dtype=np.float64)
            best_seq[selected] = 1.0
            print(f"*** New best approx score {best_score:.6f} at iter {iteration}")

        # Stop early if we already beat the target bound.
        if best_score < 1.503:
            print("Target bound reached – exiting early.")
            break

    # ----------------------------------------------------------------------
    # Return the best sequence as a plain Python list of floats.
    # ----------------------------------------------------------------------
    return [float(x) for x in best_seq]
