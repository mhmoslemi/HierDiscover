"""
Step‑function optimisation for the `evaluate_sequence` metric.

The algorithm works in two phases:

1. **Parametric sweep** – generate candidate sequences from a few simple
   families (uniform, linear, quadratic, power‑law and exponential decay) for
   several lengths.  The exponential family is parametrised by the product
   `c = ε·n` (ε = 1 – r where r is the decay factor).  For each candidate the
   evaluation function is called and the best sequence so far is remembered.

2. **Local refinement** – starting from the best candidate a small projected
   gradient descent is performed.  The gradient of the maximal convolution
   entry `b_max` is simply `g_i = 2·a_{k_max‑i}` (zero when the index is out of
   range).  A step in the negative gradient direction is taken, the result is
   projected back onto the probability simplex (non‑negative, unit sum) and
   kept if it improves the score.  Occasionally a tiny random jitter is added
   to escape plateaus.

The method respects all security checks of `evaluate_sequence`, runs within
the supplied time budget and returns a list of floats suitable for the final
evaluation.
"""

import numpy as np
import time


def generate_exponential(n, c):
    """
    Length‑n exponentially decaying sequence a_i = r**i,
    where r = exp(-c / n).  The result is normalised to sum = 1.
    """
    r = np.exp(-c / n)
    # Avoid under‑flow for very large n
    powers = r ** np.arange(n, dtype=np.float64)
    s = powers.sum()
    if s == 0.0:
        return np.ones(n, dtype=np.float64) / n
    return powers / s


def generate_linear(n):
    """Monotonically decreasing linear sequence (n … 1) normalised to sum = 1."""
    seq = np.arange(n, 0, -1, dtype=np.float64)
    return seq / seq.sum()


def generate_quadratic(n):
    """Quadratically decreasing sequence ((n – i)²) normalised to sum = 1."""
    seq = np.arange(n, 0, -1, dtype=np.float64) ** 2
    return seq / seq.sum()


def generate_powerlaw(n, p):
    """
    Power‑law decreasing sequence (i+1)^{-p} (reversed so that the first
    element is the largest) normalised to sum = 1.
    """
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))[::-1]
    return seq / seq.sum()

            print(f"Quadratic n={n} →  {score:.6f}")

        # power‑law families
        for p in (0.5, 1.0, 1.5, 2.0):
            pw = generate_powerlaw(n, p)
            score = float(evaluate_sequence(pw.tolist()))
            if score < best_score:
                best, best_score = pw.copy(), score
                print(f"Power‑law p={p} n={n}  →  {score:.6f}")

        # exponential families
        for c in exp_c_vals:
            exp_seq = generate_exponential(n, c)
            score = float(evaluate_sequence(exp_seq.tolist()))
            if score < best_score:
                best, best_score = exp_seq.copy(), score
                print(f"Exponential c={c:.2f} n={n}  →  {score:.6f}")

    # ----------------------------------------------------------------------
    # 2️⃣  Local gradient‑descent refinement
    # ----------------------------------------------------------------------
    step = 0.02                     # learning rate
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # full convolution and the index of the maximal entry
        conv = np.convolve(best, best)
        max_idx = int(np.argmax(conv))

        # gradient of the maximal convolution entry w.r.t. the sequence
        grad = np.zeros_like(best)
        lo = max(0, max_idx - (best.size - 1))
        hi = min(best.size - 1, max_idx)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        j = max_idx - i
        grad[i] = 2.0 * best[j]

        # projected gradient step (simple simplex projection)
        new_seq = best - step * grad
        new_seq = np.clip(new_seq, 0.0, None)
        tot = new_seq.sum()
        if tot < 1e-12:                               # should never happen, but guard
            new_seq = np.ones_like(best) / best.size
            tot = 1.0
        new_seq /= tot

        # evaluate
        score = float(evaluate_sequence(new_seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = new_seq.copy(), score
            print(f"Iter {iteration:05d} improved → {best_score:.6f}")
            step *= 0.995                # slowly decay step size for stability
            continue

        # occasional random jitter to leave flat regions
        if rng.random() < 0.005:
            jitter = np.exp(rng.normal(0.0, 0.08, size=best.size))
            new_seq = best * jitter
            new_seq = np.clip(new_seq, 0.0, None)
            new_seq /= new_seq.sum()
            s2 = float(evaluate_sequence(new_seq.tolist()))
            if s2 < best_score:
                best, best_score = new_seq.copy(), s2
                print(f"Iter {iteration:05d} jitter → {best_score:.6f}")

        # keep the previous best if the current move was not improving
        # (the variable ``best`` is unchanged)

    print(f"Finished with best score {best_score:.6f}")
    # Return a plain Python list – the evaluator will clip again if needed.
    return [float(x) for x in best]
