def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # init a
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size==0 or not np.isfinite(base).all():
            n = 1024
            a = np.ones(n) / n
        else:
            a = np.clip(base, 0.0, 1000.0)
            # normalize to sum 1 to keep scaling invariant
            total = a.sum()
            if total <= 0:
                a = np.ones_like(a) / a.size
            else:
                a = a / total
    else:
        n = 1024
        a = np.ones(n) / n
    n = a.shape[0]
    best_a = a.copy()
    best_score = float('inf')
    lr = 0.02
    iter_idx = 0
    # Precompute size for FFT of a*a: conv length = 2n-1, pad size
    conv_len = 2*n - 1
    size_conv = 1 << (conv_len - 1).bit_length()
    # For gradient conv M * a_rev length conv_len + n -1 = 3n-2
    grav_len = conv_len + n - 1
    size_grad = 1 << (grav_len - 1).bit_length()
    # main loop
    while time.monotonic() < deadline:
        # compute convolution a*a
        fft_a = np.fft.fft(a, n=size_conv)
        b_full = np.fft.ifft(fft_a * fft_a).real[:conv_len]
        max_b = b_full.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Iter {iter_idx}, new best score {best_score:.6f}")
        # active mask
        # tolerance: relative small fraction of max, e.g., 1e-8 * max or absolute eps
        eps = max(1e-12, max_b * 1e-8)
        active = (b_full >= max_b - eps).astype(float)
        # if active all zeros? keep uniform?
        if active.sum() == 0:
            active[ np.argmax(b_full) ] = 1.0
        # compute gradient via convolution of active mask with reversed a
        a_rev = a[::-1]
        fft_active = np.fft.fft(active, n=size_grad)
        fft_rev = np.fft.fft(a_rev, n=size_grad)
        conv_active_rev = np.fft.ifft(fft_active * fft_rev).real
        # extract gradient slice
        # conv result shape size_grad; valid indices start at n-1, length n
        start = n - 1
        grad = 2.0 * conv_active_rev[start:start+n]
        # multiplicative update
        a = a * np.exp(-lr * grad)
        # clip tiny values to avoid underflow
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total <= 0:
            a = np.ones(n) / n
        else:
            a = a / total
        # random jitter occasionally to escape
        if iter_idx % 50 == 0 and rng.random() < 0.2:
            jitter = rng.normal(0.0, 0.02, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
        # gradually decay learning rate
        if iter_idx % 100 == 0 and lr > 0.001:
            lr *= 0.98
        iter_idx += 1
    # return best candidate as list
    # We'll scale up maybe to typical magnitude (like sum= sqrt(2n)?) but not needed
    result = best_a * 1.0  # keep as is
    return [float(x) for x in result]

"""
Propose a step‐function sequence that minimizes
      F(a) = 2·n·max(a*a) / (sum(a))²
using an iterative “flatten the convolution” scheme.

The algorithm:
* Start from the given `height_sequence_1` (or a uniform distribution).
* Normalise the sequence to have unit sum (scaling does not affect the score).
* Repeatedly:
    – Compute the full self‑convolution b = a * a (FFT).
    – Identify the indices where b attains (or is very close to) its maximum.
    – Build a binary mask of those active positions.
    – Compute the gradient of the maximal convolution w.r.t. a:
          g_j = 2 * Σ_{k active} a_{k‑j} .
      This is a short convolution of the mask with the reversed a,
      performed efficiently with FFT.
    – Update a multiplicatively: a ← a·exp(‑lr·g) (keeps positivity).
    – Renormalise a to sum 1.
    – Occasionally inject a small random jitter to escape plateaus.
* Keep the best sequence (lowest objective) seen so far.
* Stop before the allotted time runs out and return the best sequence.

The method drives the convolution towards a flat shape,
thus reducing max(b) and the evaluation score below the uniform value 2.0.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative heights (a step function) with low score.

    The function runs a time‑limited optimisation that flattens the
    self‑convolution of the sequence.  The returned list will be evaluated
    by the provided ``evaluate_sequence`` function.

    rng = np.random.default_rng(seed)
    # Give ourselves a little safety margin before the external timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Initialise the sequence -------------------------------------------------

        or not np.isfinite(base).all()
    ):
        # fallback: a modest size uniform vector
        n = 1024
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = np.clip(base, 0.0, 1000.0)
        total = a.sum()
        if total <= 0.0:
            a = np.ones_like(a, dtype=np.float64) / a.size
        else:
            a = a / total
        n = a.shape[0]

    # Pre‑compute FFT sizes (next power of two) for the two convolutions we need.
    conv_len = 2 * n - 1                     # length of a*a
    size_conv = 1 << (conv_len - 1).bit_length()   # power‑of‑2 >= conv_len

    # For gradient: mask (len conv_len) * a_rev (len n) -> length conv_len+n-1
    grad_len = conv_len + n - 1
    size_grad = 1 << (grad_len - 1).bit_length()

    # ------------------------------------------------------------------
    # Optimisation loop ----------------------------------------------------
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = float('inf')
    lr = 0.02                # learning‑rate for the multiplicative update
    iter_cnt = 0

    while time.monotonic() < deadline:
        # ----- convolution a * a -------------------------------------------------
        fft_a = np.fft.fft(a, n=size_conv)
        b_full = np.fft.ifft(fft_a * fft_a).real[:conv_len]

        max_b = b_full.max()
        # Since sum(a) == 1, the objective is simply 2·n·max_b
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[iter {iter_cnt}] new best score = {best_score:.6f}")

        # ----- active set (indices where convolution is maximal) -----------------
        # a tiny tolerance avoids missing the max due to floating error
        eps = max(1e-12, max_b * 1e-8)
        active_mask = (b_full >= max_b - eps).astype(np.float64)

        # If for some numerical reason no entry is active, force the max index.
        if active_mask.sum() == 0.0:
            active_mask[np.argmax(b_full)] = 1.0

        a_rev = a[::-1]

        fft_rev = np.fft.fft(a_rev, n=size_grad)

        start = n - 1

        a = a * np.exp(-lr * grad)

        a = np.maximum(a, 0.0)
