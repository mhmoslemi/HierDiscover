def simplex_projection(v):
    """Project vector v onto the probability simplex (nonnegative entries sum to 1)."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

This is O(n log n).

def linear_convolve(a, b):
    size = len(a) + len(b) - 1
    n_fft = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    c = np.fft.irfft(A * B, n=n_fft)[:size]
    return c

```python
"""
Propose candidate step function heights using projected gradient descent
to minimize the max self‑convolution (the evaluation function). The
algorithm works in the space of probability vectors (L1‑norm = 1)
because the objective is scale‑invariant. Starting from a random
Dirichlet vector (or the provided `height_sequence_1`) we repeatedly
compute the convolution, form a soft‑max weighting of the convolution
values and take a gradient step that reduces the weighted sum.  After
each step the vector is projected back onto the simplex (non‑negative,
unit sum).  Several restarts are performed while the time budget lasts
and the best sequence found is returned.
"""

import numpy as np
import time

def _linear_convolve(x, y):
    """Linear (non‑circular) convolution using real FFT."""
    size = len(x) + len(y) - 1
    n_fft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:size]
    # tiny negative numbers caused by rounding are removed
    return np.clip(conv, 0.0, None)

def _score_from_vector(v):
    """Fast version of evaluate_sequence (the input must be non‑negative)."""
    s = np.sum(v)
    if s < 1e-12:
        return np.inf
    b = _linear_convolve(v, v)
    max_b = np.max(b)
    return 2.0 * len(v) * max_b / (s * s)

def _project_simplex(v):
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    # Close‑form algorithm – O(n log n) due to sorting.
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero, fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a coefficient sequence that minimises the provided
    ``evaluate_sequence`` function.  The method uses projected gradient
    descent on the self‑convolution L∞‑norm (approximated by a
    soft‑max) and performs cheap restarts while the time budget lasts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------------
    # Determine the length of the sequence we will work with.
    # ``height_sequence_1`` may exist – if it does, keep its size,
    # otherwise fall back to a moderately large default.
    # --------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = base.size if base.ndim == 1 and base.size > 0 else 0
    except Exception:
        n = 0

    if n == 0:
        n = 2048                      # a power‑of‑two size makes FFT cheap

    # Keep the best sequence seen across all restarts.
    best_vec = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
    best_score = _score_from_vector(best_vec)

    # --------------------------------------------------------------------
    # Main optimisation loop – many short restarts.
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- initialise a fresh random point on the simplex ----------------
        a = rng.dirichlet(np.ones(n))
        # small amount of the previously best vector is mixed in – helps
        # to keep good patterns discovered early.
        if rng.random() < 0.3:
            a = 0.7 * a + 0.3 * best_vec
            a = _project_simplex(a)

        # hyper‑parameters for this restart
        lr = 0.4                     # learning rate
        beta = 5.0                   # soft‑max sharpness
        beta_max = 300.0
        iter_limit = 2000
        it = 0

        while it < iter_limit and time.monotonic() < deadline:
            # ----- convolution, peak & current score -----------------------
            b = _linear_convolve(a, a)
            max_b = np.max(b)
            score = 2.0 * n * max_b              # sum(a) == 1 by construction
            if score < best_score:
                best_score = score
                best_vec = a.copy()
                # occasional progress report
                print(f"Iter {it:6d}  new best score {best_score:.6f}")

            # ----- gradient of the soft‑max approximation ------------------
            # g is a probability distribution that concentrates on the
            # currently largest convolution entries.
            w = np.exp(beta * (b - max_b))        # max(b) = max_b
            g = w / w.sum()

            # grad = 2 * (g * a_rev)  (linear convolution)
            grad_full = _linear_convolve(g, a[::-1])
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # ----- simple back‑tracking line search -------------------------
            a_new = a - lr * grad
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() == 0.0:
                # restart from uniform if we hit the origin
                a_new = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a_new = a_new / a_new.sum()

            new_score = _score_from_vector(a_new)

            if new_score < score:
                # accept and optionally increase step size slightly
                a = a_new
                lr = min(lr * 1.05, 2.0)
            else:
                # reject – shrink the step and try again
                lr = lr * 0.5

            # slowly sharpen the soft‑max to focus on the true peak
            beta = min(beta_max, beta * 1.001)

            it += 1

    # --------------------------------------------------------------------
    # Return the best sequence we have discovered.
    # The evaluation function works with any positive scaling, therefore we
    # keep the unit‑sum vector unchanged.
    # --------------------------------------------------------------------
    result = [float(x) for x in best_vec]
    # sanity check – guard against a pathological zero vector
    try:
        if not np.isfinite(evaluate_sequence(result)):
            # fall back to a uniformly distributed vector if something went
            # wrong (the evaluator would otherwise return ``inf``).
            result = [float(1.0 / n) for _ in range(n)]
    except Exception:
        result = [float(1.0 / n) for _ in range(n)]
    return result

"""
Propose a step–function that improves on the trivial uniform construction.

The algorithm follows the iterative linear‑programming scheme described in the
statement.  Starting from a normalized vector `f` (∑f = 1), we compute the
maximum value 

    M = max_k (f * f)[k]  (linear convolution)

and then solve the linear program

    maximise   Σ g_i
    subject to  g_i ≥ 0
                (f * g)[k] ≤ M   for all k = 0 … 2n‑2

The solution `g₀` is an extreme point that pushes mass as far as possible
without exceeding the current bound `M`.  After a simple rescaling (here the
sum is already 1) we move a small step toward `g₀` :

    f ← (1‑t)·f + t·g₀ ,   0 < t ≤ 1

and repeat.  The max convolution usually drops after several iterations,
producing a sequence whose evaluation score

    score = 2·n·max(f*f)/(∑f)²

becomes smaller than the uniform value 2.  The loop is limited by the
available time budget; the best sequence observed is returned.

The implementation uses only NumPy, SciPy (for the Toeplitz matrix and the
LP solver) and plain Python, respects all contest rules and prints occasional
progress messages.
"""

import time
import numpy as np
import scipy.linalg as la
import scipy.optimize as opt


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast and reasonably accurate)."""
    size = len(x) + len(y) - 1
    n_fft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:size]
    # round‑off errors may produce tiny negatives – clip them
    return np.clip(conv, 0.0, None)


def _score_from_vector(v: np.ndarray) -> float:
    """Fast analogue of evaluate_sequence for a non‑negative vector."""
    s = v.sum()
    if s < 1e-12:
        return np.inf
    b = _linear_convolve(v, v)
    return 2.0 * len(v) * b.max() / (s * s)


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A @ g = f * g  (linear
    convolution).  Entry A[k, j] = f[k‑j] if 0 ≤ k‑j < n, otherwise 0.
    """
    n = f.size
    # first column: f followed by zeros
    col = np.concatenate([f, np.zeros(n - 1)])
    # first row: only the first element of f is non‑zero
    row = np.concatenate([f[:1], np.zeros(n - 1)])
    return la.toeplitz(col, row)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a coefficient sequence with a low evaluation score.
    The function is called by the grader; it must return a list of
    non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------------
    # Choose a working length.  If a previous construction is available,
    # keep its length; otherwise fall back to a modest size that keeps the LP
    # cheap (≈ 500 variables).
    # --------------------------------------------------------------------
    try:
        # `height_sequence_1` is supplied by the environment.
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = base.size
        f = np.clip(base, 0.0, None)
        if f.sum() == 0.0:
            f = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            f = f / f.sum()                     # normalise to L1 = 1
    except Exception:
        # no usable global sequence – start from a uniform vector
        n = 512                                # a comfortable size for the LP
        f = np.full(n, 1.0 / n, dtype=np.float64)

    best_f = f.copy()
    best_score = _score_from_vector(best_f)

    # algorithmic parameters
    t = 0.3               # initial mixing step size
    t_min = 0.02
    decay = 0.90          # geometric decay of `t`
    max_outer_iters = 200

    outer = 0
    while time.monotonic() < deadline and outer < max_outer_iters:
        # ----------------------------------------------------------------
        # current bound M = max_k (f * f)[k]
        # ----------------------------------------------------------------
        conv_ff = _linear_convolve(f, f)
        M = conv_ff.max()
        cur_score = 2.0 * n * M                 # ∑f = 1 => denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{outer:3d}] new best score {best_score:.6f}")

        # ----------------------------------------------------------------
        # Build the linear‑programming matrix A and solve:
        #
        #   maximise  1ᵀ g
        #   s.t.      A @ g ≤ M  ,  g ≥ 0
        #
        # (A is the convolution matrix for the current `f`.)
        # ----------------------------------------------------------------
        A = _build_conv_matrix(f)                     # shape (2n‑1, n)
        c = -np.ones(n)                               # linprog minimises cᵀx
        ub = np.full(2 * n - 1, M)

        # bounds (g_i ≥ 0)
        bounds = [(0, None) for _ in range(n)]

        # solve the LP – the default HiGHS solver is fast for these sizes
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.monotonic())},
        )

        if not res.success:
            # if the LP fails (should be rare) break out of the loop
            print(f"[{outer:3d}] LP failed (status: {res.message}), terminating.")
            break

        g = res.x
        s_g = g.sum()
        if s_g <= 0.0:
            # degenerate – fall back to uniform (should never happen)
            g = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            g = g / s_g                               # keep the L1‑norm = 1

        # ---------------------------------------------------------------
        # Take a convex step toward the LP solution and re‑normalise.
        # ---------------------------------------------------------------
        f = (1.0 - t) * f + t * g
        f = np.clip(f, 0.0, None)
        f /= f.sum()                                 # maintain L1 = 1

        # slowly reduce the step size – this stabilises the iteration
        t = max(t_min, t * decay)

        outer += 1

    # --------------------------------------------------------------------
    # Return the best vector found.  The evaluator accepts any non‑negative
    # numbers; we keep the L1‑norm = 1 (well above the 0.01 safety limit).
    # --------------------------------------------------------------------
    result = [float(x) for x in best_f]
    # final sanity check – if for some reason the sequence is rejected,
    # fall back to a simple uniform construction.
    try:
        if not np.isfinite(evaluate_sequence(result)):
            result = [float(1.0 / n) for _ in range(n)]
    except Exception:
        result = [float(1.0 / n) for _ in range(n)]

    return result
