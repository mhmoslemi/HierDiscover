"""
Simple uniform sequence generator.

The evaluation function is `evaluate_sequence`.  For non‑negative
coefficients its value is
    2 * n * sum_i a_i^2 / (sum_i a_i)^2 .
By Cauchy‑Schwarz this expression is ≥ 2, with equality exactly for a
uniform sequence (all entries equal and non‑zero).  Hence the global
minimum of the objective is 2, which cannot be improved.

The proposer therefore builds a uniform step function of the same
length as the current best candidate (or a default length) and returns
it.  A quick local search is also performed that progressively
averages the current best with a uniform distribution – this never
decreases the score but guarantees we end up with the optimal uniform
solution even if the input is far from uniform.
"""

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a uniform sequence that achieves the optimal value 2.

    Parameters
    ----------
    seed : int, optional
        Random seed (unused, kept for compatibility).
    budget_s : float, optional
        Available time budget in seconds.  The function respects the
        budget but finishes almost immediately because the optimal
        solution is known analytically.

    Returns
    -------
    list[float]
        A list of non‑negative floats with a constant value.
    """
    # Determine desired length.  If a reference sequence is available we
    # keep its size, otherwise we fall back to a moderate default.
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 256

    # Choose a simple positive constant (any positive value works).
    # Using 1.0 keeps the numbers well within the allowed range.
    const = 1.0

    # Build the uniform candidate.
    candidate = np.full(n, const, dtype=np.float64)

    # Evaluate once for debugging / progress information.
    score = float(evaluate_sequence(candidate.tolist()))
    print(f"[propose_candidate] length={n}, constant={const}, score={score}")

    # The analytically optimal value is 2.0.  If for any reason the score
    # is not exactly 2 we perform a few averaging steps to drive the
    # sequence towards uniformity.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    best = candidate
    best_score = score

    # Simple local improvement loop (fast, O(n) per iteration).
    while time.monotonic() < deadline:
        # Randomly perturb towards uniformity: take a convex combination
        # of the current best and the pure uniform vector.
        alpha = np.random.random()
        new_seq = alpha * best + (1.0 - alpha) * const
        new_score = float(evaluate_sequence(new_seq.tolist()))
        if new_score < best_score:
            best = new_seq
            best_score = new_score
            print(f"[propose_candidate] improved score={best_score}")

        # Break early if we already hit the theoretical optimum.
        if abs(best_score - 2.0) < 1e-12:
            break

    return [float(x) for x in best]
