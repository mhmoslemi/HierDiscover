"""Propose a low‑score step function.

The algorithm builds a sparse “core’’ of large values that form a Sidon
(set of integers whose pairwise sums are all distinct).  The remaining
indices receive a small uniform value w = v / m where v is the core
height and m is the core size.  For a Sidon set the largest
convolution entry is the core–core term (≈2·v²); the cross terms and the
filler‑filler term are kept ≤2·v² by the choice w=v/m, so the evaluation
≈4·n/(m+(n‑m)/m)².  With n≈2500 this is minimal near m≈√n≈50,
yielding a score ≈1.0 – 1.5, well below the target 1.503.

The search repeatedly constructs random Sidon sets (by greedy insertion in a
random order) and keeps the best sequence found within the time budget.
A few simple post‑scalings are also tried.  The function respects all
security checks in `evaluate_sequence` and uses only allowed libraries.
"""

import numpy as np
import time

# ------------------------------------------------------------
# helper: greedy construction of a Sidon set (B2‑set)
# ------------------------------------------------------------
def random_sidon_set(length: int, rng: np.random.Generator) -> list[int]:
    """
    Return a list of indices forming a Sidon set (all sums i+j are distinct).
    The search order is a random permutation of 0..length‑1.
    """
    order = rng.permutation(length)
    positions = []                     # chosen indices
    used_sums = set()                  # all i+j that have appeared (i<=j)
    for p in order:
        # test if adding p creates a duplicated sum
        conflict = False
        # check sum with each already chosen element
        for q in positions:
            if (p + q) in used_sums:
                conflict = True
                break
        # also the sum with itself must be new
        if not conflict and (2 * p) in used_sums:
            conflict = True
        if conflict:
            continue
        # accept p: update the set of sums
        for q in positions:
            used_sums.add(p + q)
        used_sums.add(2 * p)
        positions.append(p)
    return positions

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a sequence of non‑negative heights that aims to minimise
    `evaluate_sequence`.  The method searches for a large Sidon set
    (the “core’’) and fills the rest with a tiny uniform value.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size (length of the sequence)
    # --------------------------------------------------------
    try:
        n = len(height_sequence_1)                      # provided global

        n = 256
        base_len = n

    # Initialise best solution

    best_seq = np.ones(n, dtype=np.float64)              # safe fallback
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial fallback score: {best_score:.6f}")

    # --------------------------------------------------------
    # Search loop (time‑bounded)
    # --------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # --- construct a random Sidon set (the core) -----------------
        core = random_sidon_set(n, rng)
        m = len(core)
        if m < 2:                     # need at least two large entries

        # --- core height v = 1, filler height w = v / m ----------------
        v = 1.0
        w = v / m

        seq[core] = v

        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"Iter {iteration:5d}: new best {best_score:.6f}  (core size {m})")

            if best_score <= 1.503:

        # (scale v while keeping w = v/m)
        for scale in (0.5, 0.75, 1.25, 1.5, 2.0):
            v2 = v * scale
            w2 = v2 / m
            cand = np.full(n, w2, dtype=np.float64)
            cand[core] = v2
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:

                best_seq = cand.copy()
                print(f"Iter {iteration:5d}: scaled best {best_score:.6f}")

        # --------------------------------------------------------
        # occasional random tweak of a non‑core entry (small local search)

        if iteration % 50 == 0:

    # --------------------------------------------------------

    print(f"Search finished after {iteration} iterations. Best score: {best_score:.6f}")
