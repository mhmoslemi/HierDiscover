            raise Exception

    # Build linear decreasing (or increasing) integer sequence 1..n

    # Normalize so that sum = sqrt(2 * n) as in the standard normalization
    total = float(np.sum(base))
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / total
    seq = base * scale
    # Apply clipping to respect evaluator's bounds
    seq = np.clip(seq, 0.0, 1000.0)
    # Calculate score for logging
    try:
        score = evaluate_sequence(seq.tolist())
    except Exception as e:
        score = float('inf')
    print(f"Generated linear sequence length={n}, scale={scale:.6e}, score={score:.6f}")
    # Ensure we respect the deadline
    deadline = start + max(0.0, float(budget_s) - 3.0)
    while time.monotonic() < deadline:
        # No time to improve further; just return the constructed sequence
        break
    return [float(x) for x in seq]
