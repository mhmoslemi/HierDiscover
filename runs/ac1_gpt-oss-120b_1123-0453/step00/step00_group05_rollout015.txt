def backtrack(start, selected, sums):
    if len(selected) == m:
        return selected
    # remaining needed = m - len(selected)
    # max possible positions = max_n - start
    # if max_n - start < remaining_needed: return None (prune)
    for x in range(start, max_n):
        # compute new sums set
        ok = True
        new_sums = set()
        for y in selected:
            s = x + y
            # for unordered pairs, we need s added only once, but since we include both x+y and y+x produce same s; we can just compute s.
            if s in sums:
                ok = False
                break
            new_sums.add(s)
        # also need x + x
        if x*2 in sums:
            ok = False
        else:
            new_sums.add(x*2)
        if not ok: continue
        # add
        selected.append(x)
        sums.update(new_sums)
        res = backtrack(x+1, selected, sums)
        if res:
            return res
        # undo
        selected.pop()
        for s in new_sums:
            sums.remove(s)
    return None

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Choose length n
    n = 200  # or dynamic based on time
    # Initialize sequence to uniform distribution (each 1/n)
    seq = np.full(n, 1.0/n, dtype=np.float64)

    # Track best
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    # Simulated annealing schedule
    temperature = 1.0  # initial temperature
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # temperature schedule: cool down gradually
        temperature = max(0.001, temperature * 0.9999)
        # propose modification
        # choose type: either transfer or block perturbation
        if rng.random() < 0.7:
            # transfer mass between two random indices
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # amount to transfer
            delta = (rng.random() - 0.5) * 0.2  # relative change up to +-0.1 of average?
            # ensure delta does not make i negative
            # actual value to move = delta * seq[i] maybe? Let's pick absolute
            move = delta * seq[i]  # could be positive or negative
            if move == 0:
                continue
            new_i = seq[i] - move
            new_j = seq[j] + move
            if new_i < 0:
                new_i = 0
                move = seq[i]  # move all
                new_j = seq[j] + move
            # create candidate copy
            candidate = seq.copy()
            candidate[i] = new_i
            candidate[j] = new_j
        else:
            # block perturbation like original
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            candidate = seq.copy()
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            candidate[start:stop] *= factor
        # clip values to [0,1000] as evaluate seq will do but we keep similar
        candidate = np.clip(candidate, 0.0, 1000.0)

        # optionally normalize sum to 1 (scaling unchanged but maintain sum not too small)
        # we can keep it as is; scaling invariance ensures eval unchanged

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate.copy()
            seq = candidate  # move to new state
            # print progress
            print(f"Iter {iteration}: new best {best_score:.6f}")
            # optionally reset temperature
            temperature = 1.0
        else:
            # maybe accept with probability
            if temperature > 0 and np.isfinite(score):
                accept_prob = np.exp(-(score - best_score) / temperature)
                if rng.random() < accept_prob:
                    seq = candidate
            # else keep old seq

    # After loop, finalize best sequence
    # Ensure it's a list of floats
    return [float(x) for x in best_seq]

"""
Propose a low‑value step‑function sequence.

The evaluation function is  2·n·max(conv(a,a)) / (sum(a))² .
For a fixed L¹‑norm the value depends only on the shape of the
vector.  Uniform vectors give a value of exactly 2, but by reshaping
the mass we can lower the maximal convolution term and thus obtain
values below 2 (the theoretical lower bound is 1).

The algorithm works in two stages:

1. **Exploration** – a quick grid over several simple parametric families
   (uniform, linear ramps, exponentials, Gaussians) for different lengths.
   The best candidate from this stage is kept as a start point.

2. **Iterated local search** – starting from the best candidate we
   repeatedly modify the vector while keeping the total sum constant.
   The modification focuses on the index where the convolution attains its
   current maximum: a small amount of mass is moved away from the
   participating positions and given to a random other position.
   Random block‑wise scaling and occasional restarts are also performed.
   The search stops when the allocated time budget expires.

All operations respect the security checks inside ``evaluate_sequence``:
values are kept non‑negative, clipped to [0,1000] and the length never
exceeds a few hundred, so each evaluation is cheap.

The function returns the best sequence found within the time limit.
"""

import time
import numpy as np

def _eval(seq):
    """Helper that calls the external evaluate_sequence safely."""
    return float(evaluate_sequence(seq.tolist()))


def _grid_search(best_seq, best_score, deadline):
    """Try a few parametric families with various lengths."""
    for n in (30, 40, 50, 60, 80, 100):
        # uniform
        cand = np.ones(n, dtype=np.float64)
        sc = _eval(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"grid uniform n={n} → {sc:.6f}")

        # linear ramp up
        cand = np.arange(1, n + 1, dtype=np.float64)
        sc = _eval(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"grid linear up n={n} → {sc:.6f}")

        # linear ramp down
        cand = np.arange(n, 0, -1, dtype=np.float64)
        sc = _eval(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"grid linear down n={n} → {sc:.6f}")

        # exponential families
        for r in (0.6, 0.7, 0.8, 0.9, 0.95):
            cand = np.power(r, np.arange(n, dtype=np.float64))
            sc = _eval(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"grid exp r={r:.2f} n={n} → {sc:.6f}")

            cand = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
            sc = _eval(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"grid rev‑exp r={r:.2f} n={n} → {sc:.6f}")

        # Gaussian‑like bumps
        x = np.arange(n, dtype=np.float64)
        for sigma in (n / 5.0, n / 4.0, n / 3.0):
            mu = (n - 1) / 2.0
            cand = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            sc = _eval(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"grid gauss sigma={sigma:.1f} n={n} → {sc:.6f}")

        if time.monotonic() >= deadline:
            break
    return best_seq, best_score


def _active_indices(seq, k_max):
    """Indices i for which j = k_max‑i is a valid partner (both non‑zero)."""
    n = len(seq)
    act = []
    for i in range(n):
        j = k_max - i
        if 0 <= j < n and seq[i] > 0 and seq[j] > 0:
            act.append(i)
    return act


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step‑function giving a low value of the evaluation function.

    The routine runs for roughly ``budget_s`` seconds (a few seconds are kept
    as a safety margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----- start from a simple uniform vector (fallback) -----
    init_seq = np.ones(60, dtype=np.float64)          # length 60 is a good trade‑off
    best_seq = init_seq.copy()
    best_score = _eval(best_seq)
    print(f"initial uniform (n={len(best_seq)}) → {best_score:.6f}")

    # ----- quick grid search over simple families -----
    best_seq, best_score = _grid_search(best_seq, best_score, deadline)

    # ----- local improvement loop -----
    cur_seq = best_seq.copy()
    cur_score = best_score

    iteration = 0
    temperature = 1.0          # simulated‑annealing style temperature
    while time.monotonic() < deadline:
        iteration += 1
        temperature = max(0.001, temperature * 0.9995)

        # compute convolution and locate the current worst (largest) index
        conv = np.convolve(cur_seq, cur_seq)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # Decide which type of move to apply
        if rng.random() < 0.6:
            # ------ mass‑transfer move focused on the offending sum ------
            active = _active_indices(cur_seq, k_max)
            if len(active) >= 2:
                i = rng.choice(active)
                # pick a random destination that is *not* the partner of i for k_max
                # (to avoid trivially undoing the move)
                forbidden = {k_max - i}
                candidates = [idx for idx in range(len(cur_seq))
                              if idx not in forbidden]
                j = rng.choice(candidates)
                # amount to move – a modest fraction of the source mass
                delta = rng.random() * 0.4 * cur_seq[i]
                if delta > cur_seq[i]:
                    delta = cur_seq[i]
                cand = cur_seq.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                # fall back to a small random scaling move
                width = max(1, len(cur_seq) // rng.integers(30, 80))
                start = rng.integers(0, max(1, len(cur_seq) - width + 1))
                factor = float(np.exp(rng.normal(0.0, 0.08)))
                cand = cur_seq.copy()
                cand[start:start + width] *= factor
        else:
            # ------ block‑wise random scaling (as in the reference code) ------
            width = max(1, len(cur_seq) // rng.integers(30, 80))
            start = rng.integers(0, max(1, len(cur_seq) - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.08)))
            cand = cur_seq.copy()
            cand[start:start + width] *= factor

        # enforce the hard limits of evaluate_sequence
        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate the candidate
        cand_score = _eval(cand)

        # accept if better, otherwise maybe accept with a small probability
        improved = np.isfinite(cand_score) and cand_score < cur_score
        if improved or (np.isfinite(cand_score) and
                        rng.random() < np.exp(-(cand_score - cur_score) / temperature)):
            cur_seq, cur_score = cand, cand_score
            if cand_score < best_score:
                best_seq, best_score = cand.copy(), cand_score
                print(f"it {iteration:6d}  new best {best_score:.6f}")

        # occasional random restart to escape deep local minima
        if iteration % 1500 == 0:
            n_new = rng.integers(30, 120)
            cur_seq = np.abs(rng.normal(0.0, 1.0, size=n_new))
            cur_seq = np.clip(cur_seq, 0.0, 1000.0)
            cur_score = _eval(cur_seq)

    print(f"search finished after {iteration} iterations – best score {best_score:.6f}")
    # Convert to plain Python list of floats as required
    return [float(v) for v in best_seq]
