def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

        current_sum = n
    f *= (target_sum / current_sum)
    # Keep best
    best = f.copy()
    best_score = evaluate_sequence(best.tolist())

    # Precompute indices for constraint matrix building (depends on f values per iteration)
    # We'll rebuild A each iteration as f changes.
    # Additional small t.
    t = 0.2
    iteration = 0

    while time.monotonic() < deadline:
        # Compute convolution and M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Build constraint matrix A (sparse)
        rows = []
        cols = []
        data = []
        # For each k but we only need rows where M > 0 maybe all.
        # Use loops
        for k in range(2 * n - 1):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # j = k - i.
            for i in range(i_start, i_end + 1):
                j = k - i
                rows.append(k)
                cols.append(j)
                data.append(f[i])
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        # Solve LP
        c = -np.ones(n)  # maximize sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        try:

        except Exception as e:
            print(f'LP failed: {e}')
            break
        if res.status != 0:
            # No feasible improvement
            print(f'LP did not converge: status {res.status}, message: {res.message}')
            break

"""
Propose a candidate step‑function that improves on the baseline (value 2.0).

The algorithm implements the iterative scheme described in the statement:

1. Start from an existing (non‑uniform) sequence `f` and normalise it so that
   sum(f) = sqrt(2·n).  With this normalisation the evaluation function equals
   the maximal convolution value M = max(f * f).

2. Solve the linear programme
       maximise   Σ g_i
       subject to  g_i ≥ 0
                   (f * g)_k ≤ M   for every convolution index k .
   The constraints are linear in `g` because `f` is fixed.
   This finds a vector `g` that can increase the total mass while keeping the
   same upper bound `M`.

3. If the LP yields a larger mass (`Σ g > Σ f`) we scale `g` back to the
   prescribed total mass (sqrt(2·n)) and move a fraction `t` towards it:
          f ← (1‑t)·f + t·g .
   The move never increases the maximal convolution, and after a few steps the
   bound `M` typically drops.

4. Repeat steps 1‑3 until the time budget expires or no further improvement
   is possible, keeping the best sequence seen.

The LP is solved with SciPy's high‑performance “highs” solver using a sparse
constraint matrix.  The whole routine stays well within the 1000 s budget for
moderate sequence lengths (≤ 1500).  The returned list of floats can be fed
directly to `evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# The global `height_sequence_1` may or may not be present.
# If it exists we use a slice of it as the initial non‑uniform guess.
# Otherwise we fall back to a simple random start.

def _initial_sequence(rng, max_len=1500):
    """Return a non‑negative starting sequence of length ≤ max_len."""
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size >= 2:
            # pick a random contiguous block (to keep locality)
            length = min(base.size, max_len)
            start = rng.integers(0, max(1, base.size - length + 1))
            seq = base[start:start + length].copy()
            return np.clip(seq, 0.0, 1000.0)
    # fallback: random Dirichlet vector (always non‑negative)
    length = max_len
    seq = rng.dirichlet(np.ones(length))
    return seq

def _build_constraint_matrix(f):
    """
    Build the sparse matrix A (shape (2·n‑1, n)) such that
        (A @ g)[k] = (f * g)[k]   for all k.
    Each row k contains the coefficients f[i] for those i where
    0 ≤ k‑i < n (i.e. where g_{k‑i} participates).
    """
    n = f.size
    rows = []
    cols = []
    data = []
    # Loop over output index k of the convolution
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over the first factor, the second index is j = k‑i
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The algorithm follows the LP‑based iterative scheme
    described above and returns the best sequence found before the time budget
    runs out.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----------------------------------------------------------------------
    # 1. initialise a non‑uniform sequence and normalise its sum
    # ----------------------------------------------------------------------
    f = _initial_sequence(rng, max_len=1500)
    n = f.size
    # normalise to sqrt(2·n) – this makes the evaluation equal to max convolution
    target_sum = np.sqrt(2.0 * n)
    f_sum = f.sum()
    if f_sum <= 0.0:
        f = np.ones(n, dtype=np.float64)
        f_sum = n
    f = f * (target_sum / f_sum)

    # keep track of the best solution seen
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"initial length={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2. iteration loop (LP + convex combination)
    # ----------------------------------------------------------------------
    t = 0.2                     # step size for convex blending
    iteration = 0
    while time.monotonic() < deadline:
        # current maximal convolution value M
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # build constraint matrix A that encodes (f * g)_k ≤ M
        A = _build_constraint_matrix(f)

        # linear programme:
        #   maximise   sum(g)   <=> minimise  -sum(g)
        #   subject to  A @ g ≤ M,   g ≥ 0
        c = -np.ones(n, dtype=np.float64)          # objective
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        # solve – we give the solver a modest time limit so that we never
        # block past the overall deadline.
        time_left = deadline - time.monotonic()
        opts = {"presolve": True, "mem_guess": n * n}
        if time_left > 0:
            # the 'highs' method honour the 'time_limit' keyword in recent SciPy
            opts["time_limit"] = time_left

        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options=opts,
        )

        if res.status != 0:
            # infeasible or solver failure – stop the loop
            print(f"LP stopped (iter {iteration}) – status {res.status}: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()

        # If the LP could not increase total mass we are at a (local) optimum.
        if sum_g0 <= target_sum * (1.0 + 1e-12):
            print(f"no sum increase after iteration {iteration} (sum_g={sum_g0:.6f})")
            break

        # Scale g so that it again has the prescribed total mass.
        scale = target_sum / sum_g0
        g = g0 * scale

        # Convex combination – this never raises the maximal convolution.
        f = (1.0 - t) * f + t * g

        # Numerical sanitising (clip and re‑normalise to avoid drift)
        f = np.clip(f, 0.0, 1000.0)
        f_sum = f.sum()
        if f_sum > 0.0:
            f = f * (target_sum / f_sum)

        # ------------------------------------------------------------------
        # 3. evaluate current candidate
        # ------------------------------------------------------------------
        cur_score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score:
            best = f.copy()
            best_score = cur_score
            print(f"iter {iteration:3d}: improved score = {best_score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            print(f"target reached after {iteration} iterations (score={best_score:.6f})")
            break

        iteration += 1

    # final sanity – ensure positivity and the required float format
    result = [float(x) for x in best]
    print(f"returning sequence of length {len(result)} with final score {best_score:.6f}")
    return result
