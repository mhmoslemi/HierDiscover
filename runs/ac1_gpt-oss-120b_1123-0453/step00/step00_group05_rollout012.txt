    * single‑index scaling;
    * redistribution of mass between two indices;
    * occasional regeneration of a fresh sparse pattern.
    After each mutation the score is evaluated with a fast NumPy
    implementation (identical to the official evaluator).  Improvements
    are kept.

4.  When the time budget expires the best vector is returned as a plain
    Python list (the outer framework will run the official `evaluate_sequence`
    on it).  Progress prints give the current best score and elapsed time.

The combination of a well‑chosen geometric start and aggressive exploration
of sparse random patterns readily beats the uniform baseline (score ≈ 2)
and often reaches the target region around 1.5.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast equivalent of `evaluate_sequence` (no security checks).
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """
    Return the same value that `evaluate_sequence` would compute,
    but without the overhead of type checks.  The input `seq`
    is assumed to be a 1‑D NumPy array of floats.
    """
    # Apply the same clipping that the official evaluator does.
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """Geometric progression a_i = r**i, scaled to have L1 = sum_target."""
    # Solve e^t = 1 + 2 t  (t ≈ 1.256) and set δ = t / n, r = 1-δ.
    t = 1.0
    for _ in range(20):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)          # safety for very small n
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:                 # fall back to uniform if something goes wrong
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    cand = (sum_target / raw_sum) * raw
    return cand


def _sparse_random_candidate(n: int, sum_target: float, rng: np.random.Generator) -> np.ndarray:
    """Create a random sparse vector (a few non‑zero entries) and scale it."""
    # Choose a sparsity level roughly between 5 % and 25 % of the positions.
    k = rng.integers(int(n * 0.05), int(n * 0.25) + 1)
    if k == 0:                     # very small n – fall back to uniform
        return np.full(n, sum_target / n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    # Positive values drawn from a Γ( shape=1, scale=1 ) distribution
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= (sum_target / total)
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    `evaluate_sequence` score.  The routine runs until the supplied time
    budget expires and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)   # safe fallback
    n = int(base.size)
    sum_target = np.sqrt(2.0 * n)               # normalization used by the paper

    # ------------------------------------------------------------------
    # Initialise best solution with a few deterministic candidates
    # ------------------------------------------------------------------
    best = _geometric_candidate(n, sum_target)
    best_score = _fast_score(best)
    # reversed version sometimes does better
    rev = best[::-1].copy()
    rev_score = _fast_score(rev)
    if rev_score < best_score:
        best, best_score = rev, rev_score

    # a quick sparse try (helps avoid a local optimum)
    sparse = _sparse_random_candidate(n, sum_target, rng)
    sp_score = _fast_score(sparse)
    if sp_score < best_score:
        best, best_score = sparse, sp_score

    print(f"[init] n={n}, target‑sum={sum_target:.3f}, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # Randomly pick a mutation strategy
        mode = rng.integers(0, 5)

        if mode == 0:
            # Block multiplicative perturbation around the current best
            cand = best.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.12))
            cand[start:start + width] *= factor
            if rng.random() < 0.2:                # occasional symmetrisation
                cand = 0.5 * (cand + cand[::-1])

        elif mode == 1:
            # Scale a single coordinate
            cand = best.copy()
            i = rng.integers(n)
            factor = np.exp(rng.normal(0.0, 0.09))
            cand[i] *= factor

        elif mode == 2:
            # Transfer a fraction of mass from one index to another
            cand = best.copy()
            i, j = rng.choice(n, size=2, replace=False)
            # move at most 40 % of the smaller entry from i to j
            eps = rng.random() * 0.4
            delta = eps * min(cand[i], cand[j])
            if cand[i] > cand[j]:
                cand[i] -= delta
                cand[j] += delta
            else:
                cand[j] -= delta
                cand[i] += delta

        elif mode == 3:
            # Replace the whole vector by a fresh sparse random candidate
            cand = _sparse_random_candidate(n, sum_target, rng)

        else:
            # Power‑law style candidate (a_i ∝ (i+1)^{-p})
            cand = np.arange(1, n + 1, dtype=np.float64)
            p = rng.uniform(0.2, 2.0)
            cand = np.power(cand, -p)
            cand_sum = cand.sum()
            if cand_sum > 0.0:
                cand *= (sum_target / cand_sum)

        # Clip to the legal interval (the fast scorer does the same)
        cand = np.clip(cand, 0.0, 1000.0)

        # Fast evaluation (identical to the official one for valid inputs)
        cur_score = _fast_score(cand)

        # Keep improvement
        if cur_score < best_score:
            best, best_score = cand, cur_score
            elapsed = time.monotonic() - (deadline - budget_s + 3.0)
            print(f"[{elapsed:5.1f}s] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    # A final call to the official evaluator guarantees that the
    # security wrapper has been applied (clipping, NaN/inf checks, etc.).
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f}")
    return final
