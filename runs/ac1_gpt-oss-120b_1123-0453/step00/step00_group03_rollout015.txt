    while time.monotonic() < deadline:
        # propose candidate by swapping pieces, small random scaling or random block scaling like before
        candidate = best.copy()
        # randomly choose few indices and adjust
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        candidate[start:stop] *= factor

"""
Propose a candidate step‑function sequence that improves over the trivial
uniform construction.  The search proceeds in three phases:

1.  Initialise a few simple analytic families (uniform, linear, exponential
    and mixtures of uniform+exponential) and keep the best one.
2.  Refine the current best by random block perturbations (the same idea as
    the reference implementation) for the remaining time.
3.  Return the best sequence found.

All candidates are clipped to the allowed range [0, 1000] before evaluation.
The function works with any global `height_sequence_1` (its length is used as
the target length); if it is missing a fallback length of 256 is used.
"""
import time
import math
import numpy as np

def make_block_perturbation(base, rng):
    """Create a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width is a reasonable fraction of the sequence length
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal noise on a single block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring to keep the shape balanced
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a low‑score coefficient sequence.

    The routine first evaluates a handful of analytically generated families
    (uniform, linear, exponential and mixtures).  The best result is then
    locally improved with random block perturbations until the time budget
    expires.  The returned list is guaranteed to be non‑negative and bounded
    by 1000, satisfying the checks performed by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1. Determine the problem size (use the global height_sequence_1 if it exists)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # pylint: disable=undefined-variable
        if n == 0:
            raise ValueError
    except Exception:
        n = 256                     # fallback length

    # ------------------------------------------------------------------
    # 2. Helper to evaluate and possibly update the current best solution
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    def try_candidate(seq, description):
        nonlocal best_seq, best_score
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = seq.copy()
            best_score = score
            print(f"New best ({description}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Analytic families
    # ------------------------------------------------------------------
    # Uniform (all ones, later scaled)
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform = np.clip(uniform, 0.0, 1000.0)
    try_candidate(uniform, "uniform")

    # Linear increase 1,2,3,...
    linear = np.arange(1, n + 1, dtype=np.float64)
    linear = linear * (1000.0 / linear.max())
    try_candidate(linear, "linear")

    # Exponential families a_i = r**i  (scaled to max 1000)
    # We search over a range of decay parameters α = -ln r * n.
    alphas = np.linspace(0.2, 5.0, 25)          # coarse sweep of decay strength
    for alpha in alphas:
        r = math.exp(-alpha / n)                # r = exp(-α/n)  ->  r∈(0,1)
        exp_seq = np.exp(np.log(r) * np.arange(n, dtype=np.float64))
        exp_seq = exp_seq * (1000.0 / exp_seq.max())
        try_candidate(exp_seq, f"exp α={alpha:.2f}")

        # Mixtures of uniform and exponential (convex combination)
        for w in np.linspace(0.0, 1.0, 5)[1:-1]:   # avoid the trivial endpoints
            mix = (1.0 - w) * np.ones(n) + w * exp_seq / 1000.0
            mix = mix * (1000.0 / mix.max())
            try_candidate(mix, f"mix α={alpha:.2f} w={w:.2f}")

    # ------------------------------------------------------------------
    # 4. Local refinement using block perturbations
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen, but protect against it
        best_seq = np.ones(n, dtype=np.float64) * 1.0

    while time.monotonic() < deadline:
        cand = make_block_perturbation(best_seq, rng)
        try_candidate(cand, "perturb")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found
    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length {n})")
    return [float(x) for x in best_seq]
