        # iterate
        while time.monotonic() < deadline:
            # propose modification:
            # choose source index i proportional to p (more weight lines)
            source = rng.choice(n, p=p) # weighted by p
            # choose destination j uniformly maybe from region far from source
            # define far: absolute distance > n//4

            # ensure dest != source

            p_new[source] -= delta
            p_new[dest] += delta

                continue
            # compute conv

            # acceptance
            if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / temperature):
                p = p_new
                conv = conv_new
                cur_score = new_score
                # update best
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = p.copy()
                    print(f"New best score {best_score:.6f} at n={n}")
            # cooling schedule
            temperature *= 0.9999

"""
Propose a step‑function (list of non‑negative heights) that minimizes
evaluate_sequence.  The algorithm works on the “probability” viewpoint:
if a = (a_i) and S = sum(a_i) then evaluate_sequence(a) = 2·n·max(p∗p)
where p_i = a_i / S and n = len(a).  The scale of a does not matter, only
the shape of p.  We therefore search over probability vectors p
(sum = 1, p_i ≥ 0) and try to reduce the peak of the self‑convolution p∗p.

The search consists of many short independent runs.
For each run a random length n (≈128–1024) and a random initial shape
(e.g. uniform, decreasing power‑law, Dirichlet‑sampled) are generated.
A cheap hill‑climber is then applied:

* compute the convolution c = p∗p,
* locate the index k where c is maximal,
* the contribution of entry i to that peak is p_i·p_{k−i};
  the entry with the largest contribution is taken as a *source*,
* a *target* index with a small probability mass is chosen,
* a small amount of mass Δ is moved from source → target, preserving the sum,
* the new peak is evaluated; the move is kept only if it improves the
  objective (otherwise it is reverted).

The move size Δ is a random fraction (5‑20 %) of the source weight,
yielding a smooth annealing‑like behaviour.  Each run performs a few
thousand iterations (or stops early if no improvement is seen) and the
best sequence over all runs is retained.  Finally the best probability
vector is returned as the list of heights (the values already satisfy
the bounds required by evaluate_sequence).

The method is simple, does not rely on external solvers, and works well
within the 1000 s time budget.
"""

import numpy as np
import time
from typing import List

# ----------------------------------------------------------------------
# Helper evaluation (exactly the same as evaluate_sequence when sum(p)=1)
# ----------------------------------------------------------------------
def _local_score(p: np.ndarray) -> float:
    """Return 2·n·max(p∗p) for a probability vector p (sum(p)=1)."""
    n = p.size
    conv = np.convolve(p, p)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple hill‑climber that tries to lower the peak of the convolution
# ----------------------------------------------------------------------
def _hill_climb(p: np.ndarray, rng: np.random.Generator,
                max_iters: int = 2000) -> (np.ndarray, float):
    """Return a locally improved copy of p and its local score."""
    n = p.size
    cur = p.copy()
    cur_score = _local_score(cur)
    best = cur.copy()
    best_score = cur_score
    stale = 0  # number of consecutive non‑improving steps

    for _ in range(max_iters):
        conv = np.convolve(cur, cur)
        k = int(conv.argmax())                # index of the current peak

        # all i that actually contribute to this peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        partner = k - idx
        contrib = cur[idx] * cur[partner]      # contribution of each i

        # source = index with the largest contribution to the peak
        src = int(idx[contrib.argmax()])

        # target = an index with relatively little mass (to keep the simplex)
        low_mask = cur < cur.mean()
        if low_mask.any():
            cand = np.where(low_mask)[0]
        else:
            cand = np.arange(n)
        tgt = int(rng.choice(cand))
        if tgt == src:                         # avoid a no‑op move
            continue

        # move a random fraction of the source mass to the target
        delta = cur[src] * rng.uniform(0.05, 0.20)
        delta = min(delta, cur[src])           # stay non‑negative

        cur[src] -= delta
        cur[tgt] += delta

        new_score = _local_score(cur)

        if new_score < cur_score:              # keep improving moves only
            cur_score = new_score
            stale = 0
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert the move
            cur[src] += delta
            cur[tgt] -= delta
            stale += 1
            if stale > 500:                    # stop if stuck
                break

    return best, best_score

# Main entry point required by the platform

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> List[float]:
    """
    Search for a height sequence with a low evaluate_sequence value.
    Uses many short random runs + a cheap hill‑climber on the
    probability‑vector representation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Repeatedly start a new random run until time runs out
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # random length (keep it modest for speed)
        n = int(rng.integers(128, 1025))

        # ---- pick a random initial shape --------------------------------
        shape_type = rng.choice(['uniform', 'decreasing', 'increasing',
                                 'power', 'dirichlet'])
        if shape_type == 'uniform':
            p0 = np.full(n, 1.0 / n)

        elif shape_type == 'decreasing':
            expo = rng.uniform(0.3, 2.0)
            w = (np.arange(1, n + 1, dtype=float) ** -expo)
            p0 = w / w.sum()

        elif shape_type == 'increasing':
            expo = rng.uniform(0.3, 2.0)
            w = (np.arange(1, n + 1, dtype=float) ** -expo)[::-1]
            p0 = w / w.sum()

        elif shape_type == 'power':
            alpha = rng.uniform(0.0, 2.0)
            w = (np.arange(1, n + 1, dtype=float) ** -alpha)
            p0 = w / w.sum()

        else:  # dirichlet – highly non‑uniform start
            conc = rng.uniform(0.1, 1.5)        # concentration parameter
            a = rng.gamma(conc, 1.0, size=n)
            p0 = a / a.sum()

        # ---- local optimisation -----------------------------------------
        # number of hill‑climbing steps adapts to remaining time
        remaining = deadline - time.monotonic()
        max_iters = int(min(3000, max(200, 2000 * remaining / budget_s)))
        p_opt, local_score = _hill_climb(p0, rng, max_iters)

        # ---- exact evaluation (the platform’s scorer) -------------------
        # Scaling does not matter; we keep sum = 1 so the official scorer
        # yields the same value as _local_score.
        try:
            exact = float(evaluate_sequence(p_opt.tolist()))
        except Exception:                      # defensive – should never happen
            exact = np.inf

        if np.isfinite(exact) and exact < best_score:
            best_score = exact
            best_seq = p_opt.copy()
            print(f"[{time.monotonic():.2f}] New best = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (guaranteed finite score = 2.0)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = np.full(n, 1.0 / n)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback used, score = {best_score:.6f}")

    # Clip to the limits enforced by evaluate_sequence (0 ≤ h ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return a plain Python list of floats
    return [float(x) for x in best_seq]
