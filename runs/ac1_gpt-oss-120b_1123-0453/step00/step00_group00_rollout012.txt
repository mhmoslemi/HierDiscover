def propose_candidate(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-3.0)

    # load global height_sequence_1 and ensure numeric
    base = np.asarray(height_sequence_1, dtype=np.float64)
    if base.ndim !=1 or base.size==0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)
    # Clip to allowed range
    base = np.clip(base, 0.0, 1000.0)
    # Ensure sum not too small: if sum<0.01, set uniform
    if base.sum() < 0.01:
        base[:] = 1.0

    # Normalization: keep sum of base constant; compute target_sum
    target_sum = base.sum()
    n = base.size

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Precompute average bound for early pruning
    def score_est(seq):
        # compute max conv via FFT
        n = len(seq)
        # compute size for FFT
        size = 1 << ((2*n-1).bit_length())
        # compute convolution
        fft_a = np.fft.rfft(seq, n=size)
        conv = np.fft.irfft(fft_a*fft_a, n=size)
        max_b = conv[:2*n-1].max()
        sum_a = seq.sum()
        if sum_a < 0.01:
            return float('inf')
        return 2*n*max_b/(sum_a**2)

    # keep best early bound
    # Use iteration loop
    iteration = 0
    while time.monotonic() < deadline:
        # choose move
        move_type = rng.integers(0, 3)
        candidate = best.copy()
        if move_type == 0:
            # block multiplicative perturbation similar to earlier algo
            candidate = make_block_perturbation(candidate, rng) # need to define
        elif move_type == 1:
            # pairwise swap
            candidate = perturb_swap(candidate, rng)
        else:
            # random small additive noise
            candidate = perturb_noise(candidate, rng)

        # Ensure bounds
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # Renormalize to maintain sum (optional)
        sum_cand = candidate.sum()
        if sum_cand < 0.01:
            continue
        # Optionally rescale to target_sum
        # But scaling cancels; but keep to avoid huge variation
        candidate *= (target_sum / sum_cand)

        # quick estimated score
        est_score = score_est(candidate)
        if est_score >= best_score:
            continue

        # evaluate exact
        exact_score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(exact_score) and exact_score < best_score:
            best = candidate
            best_score = exact_score
            print(f"Iter {iteration}: new best score {best_score:.6f}")
        iteration += 1

    print(f"Finished with best score {best_score:.6f}")
    return [float(x) for x in best]

"""
Hybrid stochastic optimizer for the step‑function evaluation problem.

The algorithm proceeds in three stages:

1. **Parametric warm‑start** – we try a family of analytic shapes
   a_i = (i+1)^{-p} (including the classic p=0.5 = 1/√i) and keep the
   best one (fast FFT based scoring).

2. **Local search** – a mixture of cheap perturbations:
   * block‑wise multiplicative scaling,
   * pairwise mass exchange,
   * small Gaussian noise,
   * a specialised “flatten” move that reduces the contribution of the
     currently most‑violated convolution index.

   After each move we compute a cheap upper bound using FFT; only when
   the bound beats the current best do we call the authoritative
   ``evaluate_sequence`` function.

3. **Time‑budget handling** – the loop runs until the allotted time is
   exhausted (the caller supplies ``budget_s``).  Progress is printed
   periodically.

All numbers are clipped to the interval [0,1000] and the sum is kept
away from zero (scaling does not affect the objective, but we avoid the
``sum_a < 0.01`` guard).

The function ``propose_candidate`` returns the best sequence found.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------

def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Fast (FFT) computation of max(conv(seq, seq)).  Returns a float."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    # real‑to‑real FFT (rfft) is faster for real data
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # keep only the valid linear‑convolution part
    return float(conv[: 2 * n - 1].max())


def _score_estimate(seq: np.ndarray) -> float:
    """Return the evaluation function value (using the fast convolution)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    # block width roughly between n/200 and n/12 (as in the reference)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring (helps symmetry)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from i -> j."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1  # ensure i != j
    if cand[i] <= 0.0:
        return cand
    # move at most 20 % of the source mass
    delta = cand[i] * float(rng.random() * 0.2)
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add small zero‑mean Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    # choose a modest block (≈5 % of the array)
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    # 5 % of the typical magnitude as std‑dev
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)


def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Reduce the contribution of the currently worst convolution index."""
    cand = seq.copy()
    n = cand.size
    # compute full convolution (fast) and locate the maximal index
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())

    # indices that actually participate in this sum
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    # contribution of each position to the offending sum
    contrib = cand[idx] * cand[partner]

    # pick a source index with large contribution
    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand

    # pick a destination index far from the offending pair, preferably with
    # a small contribution (or simply random)
    # avoid the symmetric partner of src when possible
    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))

    # move a modest fraction of the source mass to the destination
    delta = cand[src] * float(0.05 + 0.15 * rng.random())  # 5‑%‑20 %
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Stochastic optimiser for the step‑function evaluation problem.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the caller subtracts a small safety margin).

    Returns
    -------
    list[float]
        The best sequence found before the time limit.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start: try a few parametric shapes (including the classic 1/sqrt)
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_seq = np.ones(256, dtype=np.float64)

    if base_seq.ndim != 1 or base_seq.size == 0 or not np.isfinite(base_seq).all():
        base_seq = np.ones(256, dtype=np.float64)

    base_seq = np.clip(base_seq, 0.0, 1000.0)

    n = base_seq.size
    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # try a few analytic families
    # a_i = (i+1)^(-p)  with p in [0.3, 0.6]
    for p in np.linspace(0.30, 0.60, 7):
        idx = np.arange(n, dtype=np.float64) + 1.0
        cand = np.power(idx, -p)
        cand = np.clip(cand, 0.0, 1000.0)
        est = _score_estimate(cand)
        if est < best_score:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = cand
                best_score = exact
                print(f"[warm‑start] p={p:.3f} → new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Main stochastic loop
    # ------------------------------------------------------------------
    it = 0
    report_interval = 1000
    move_functions = [_make_block_perturbation,
                      _perturb_swap,
                      _perturb_noise,
                      _flatten_move]

    while time.monotonic() < deadline:
        # pick a move at random
        move = rng.choice(move_functions)
        candidate = move(best, rng)

        # keep the sum roughly comparable (scaling does not change the score)
        s = float(candidate.sum())
        if s < 0.01:
            continue
        # normalise to the original sum (optional – keeps numbers well‑scaled)
        candidate *= (best.sum() / s)

        # quick upper‑bound using the fast FFT based estimator
        est_score = _score_estimate(candidate)

        # accept only if we have a genuine chance to improve
        if est_score < best_score - 1e-12:
            exact_score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(exact_score) and exact_score < best_score:
                best = candidate
                best_score = exact_score
                print(f"[{it:07d}] improvement: {best_score:.6f}")

        it += 1
        if it % report_interval == 0:
            print(f"[{it:07d}] current best = {best_score:.6f}")

    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    return [float(x) for x in best]
