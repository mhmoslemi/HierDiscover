def fast_score(a):
    # Clip to [0,1000]
    a = np.clip(a, 0.0, 1000.0)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = np.convolve(a, a)
    max_b = np.max(c)
    n = len(a)
    return 2.0 * n * max_b / (sum_a * sum_a)

def move_block_scale(a, rng):
    n = a.shape[0]
    width = int(rng.integers(max(1, n//200), max(2, n//12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    b = a.copy()
    b[start:start+width] *= factor
    return b

def move_pair_swap(a, rng):
    n = a.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a.copy()
    total = a[i] + a[j]
    new_i = total * rng.random()
    b = a.copy()
    b[i] = new_i
    b[j] = total - new_i
    return b

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non-negative sequence that minimizes the evaluation function.

    The algorithm starts from the provided ``height_sequence_1`` (or a random
    uniform start) and performs a simulated‑annealing / gradient‑descent style
    local search.  Moves consist of block scaling, pairwise mass swaps,
    small Gaussian perturbations, a symmetrisation step and a dedicated
    ``gradient move`` that explicitly reduces the contribution to the
    currently maximal convolution entry.  The temperature slowly cools to
    focus the search on improvements.  The best sequence found before the
    time budget expires is returned as a plain ``list[float]``.
    """
    import numpy as np
    import time

    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 2.0)   # leave a couple seconds margin
    deadline = start_time + total_budget

    # Initialise the sequence -------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        base = np.clip(base, 0.0, 1000.0)
    except Exception:
        # fallback: uniform vector of length 256
        base = np.full(256, 1.0, dtype=np.float64)

    n = base.shape[0]

    # initialise current and best solution
    curr = base.copy()
    curr_score = fast_score(curr)
    best = curr.copy()
    best_score = curr_score

    # pre‑compute FFT length for this n
    _ = fast_score(np.zeros_like(curr))  # just to fill the cache

    # temperature schedule
    T_start = best_score * 0.2 + 1e-3   # a small positive start temperature
    # counters
    it = 0
    no_improve = 0
    # ---------------------------------------------------------------

"""
A lightweight simulated‑annealing / local‑search optimizer for the
non‑negative step‑function problem.

The algorithm starts from the provided `height_sequence_1` (or a uniform
fallback) and repeatedly proposes modifications of five types:

    0 – Scale a random contiguous block (multiplicative perturbation).
    1 – Swap mass between two random positions while keeping the total
        sum unchanged.
    2 – Add small Gaussian noise to a random small subset of entries.
    3 – **Gradient move**: identify the index `k` where the current
        convolution is maximal, compute the contribution of each entry
        to that maximum and move a fraction of the most harmful mass to
        random other positions.
    4 – Symmetrise the current sequence (average with its reverse).

A simple temperature schedule (linear cooling) is used; worse moves are
accepted with a Metropolis probability early on, allowing the search to
escape local minima.  Whenever a new best value (according to the fast
evaluation routine) is found it is printed.  If the search stalls for a
while a fresh random start is generated to diversify the search.

The evaluation routine `fast_score` reproduces the behaviour of
`evaluate_sequence` but uses an FFT‑based convolution for speed and avoids
the expensive Python checks in the reference implementation.

The function returns a plain ``list[float]`` that can be fed directly to
`evaluate_sequence` by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper cache for the FFT length (shared by all calls)
_fft_len_cache: dict[int, int] = {}


def fast_score(arr: np.ndarray) -> float:
    """
    Compute the same value as ``evaluate_sequence`` but fast.
    The array is clipped to the allowed interval [0, 1000] and the sum
    is checked for the ``<0.01`` guard.
    """
    a = np.clip(arr, 0.0, 1000.0)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float("inf")

    n = a.shape[0]
    # length of the zero‑padded FFT (next power of two ≥ 2*n‑1)
    m = _fft_len_cache.get(n)
    if m is None:
        m = 1 << ((2 * n - 1).bit_length())
        _fft_len_cache[n] = m

    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_b = float(np.max(conv))

    return 2.0 * n * max_b / (sum_a * sum_a)

# Move primitives -------------------------------------------------------
def move_block_scale(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = a.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    b = a.copy()
    b[start : start + width] *= factor
    return b


def move_pair_swap(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of the mass between two positions."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a.copy()
    total = a[i] + a[j]
    new_i = total * rng.random()
    b = a.copy()
    b[i] = new_i
    b[j] = total - new_i
    return b


def move_noise(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add tiny Gaussian noise to a few random entries."""
    n = a.size
    k = int(rng.integers(1, max(2, n // 50)))
    idx = rng.choice(n, size=k, replace=False)
    b = a.copy()
    b[idx] = np.clip(b[idx] + rng.normal(0.0, 0.1, size=k), 0.0, 1000.0)
    return b


def move_symmetrize(a: np.ndarray) -> np.ndarray:
    """Replace the sequence by the average of itself and its reversal."""
    return 0.5 * (a + a[::-1])


def move_gradient(a: np.ndarray, k_max: int, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the contribution to the currently maximal convolution entry.
    The gradient of ``c_k = sum_i a_i a_{k-i}`` w.r.t. ``a_i`` is
    2 * a_{k-i}` (or ``a_i`` when ``i = k-i``).  We take the indices with
    the largest absolute gradient, shrink them by a random fraction and
    redistribute the removed mass uniformly to other random positions.
    """
    n = a.size
    i_arr = np.arange(n)
    j_arr = k_max - i_arr
    valid = (j_arr >= 0) & (j_arr < n)

    # gradient contribution (sub‑gradient of the max‑convolution)
    g = np.zeros(n, dtype=np.float64)
    g[valid] = 2.0 * a[j_arr[valid]]
    self_mask = valid & (i_arr == j_arr)
    g[self_mask] = a[i_arr[self_mask]]

    # number of coordinates to touch (at most 1 % of the vector)
    k = int(rng.integers(1, max(2, n // 100)))
    if k >= n:
        idxs = np.arange(n)
    else:
        idxs = np.argpartition(g, -k)[-k:]

    total_mass = float(a[idxs].sum())
    if total_mass <= 0.0:
        return a.copy()

    # fraction of the selected mass to move (up to 50 %)
    frac = rng.random() * 0.5
    moved = total_mass * frac

    b = a.copy()
    # shrink the chosen positions proportionally
    b[idxs] = b[idxs] * (1.0 - frac)

    # add the removed mass to random other locations
    other = rng.choice(n, size=k, replace=False)
    add_per = moved / k
    b[other] += add_per
    return b


def random_init(rng: np.random.Generator, n: int) -> np.ndarray:
    """Create a fresh random non‑negative start (exponential distribution)."""
    a = rng.exponential(scale=1.0, size=n)
    # make sure we are not all zeros
    if a.sum() < 1e-8:
        a[0] = 1.0
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise the step‑function coefficients.

    The routine runs for roughly ``budget_s`` seconds (a small safety
    margin is subtracted) and returns the best sequence found.
    """

    start_time = time.monotonic()

    time_budget = max(0.0, float(budget_s) - 2.0)
    deadline = start_time + time_budget

    # initialise a starting vector
    try:

        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        base = np.clip(base, 0.0, 1000.0)
    except Exception:
        # fallback – a modest uniform vector
        base = np.full(256, 1.0, dtype=np.float64)

    n = base.size
    # current solution and the overall best
    curr = base.copy()
    curr_score = fast_score(curr)

    best = curr.copy()
    best_score = curr_score

    # pre‑compute the FFT length for this dimension
    _ = fast_score(np.zeros_like(curr))   # fills the cache

    # temperature schedule (linear cooling)
    T_start = max(1e-3, best_score * 0.2)

    it = 0
    no_improve = 0

    print(f"[init] n={n}, initial score={best_score:.6f}")

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # recompute the index of the maximal convolution entry
        fa = np.fft.rfft(np.clip(curr, 0.0, 1000.0), n=_fft_len_cache[n])
        conv_full = np.fft.irfft(fa * fa, n=_fft_len_cache[n])[: 2 * n - 1]
        k_max = int(np.argmax(conv_full))

        # --------------------------------------------------------------
        # propose a candidate by picking one of the move types
        move_type = rng.integers(0, 5)   # 0..4
        if move_type == 0:
            cand = move_block_scale(curr, rng)
        elif move_type == 1:
            cand = move_pair_swap(curr, rng)
        elif move_type == 2:
            cand = move_noise(curr, rng)
        elif move_type == 3:
            cand = move_gradient(curr, k_max, rng)
        else:
            cand = move_symmetrize(curr)

        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = fast_score(cand)

        # --------------------------------------------------------------
        # temperature (linear decay)
        elapsed = time.monotonic() - start_time
        T = T_start * (1.0 - elapsed / time_budget)
        if T < 1e-12:
            T = 0.0

        # acceptance rule (Metropolis)
        if cand_score < curr_score:
            accept = True
        else:
            accept = (T > 0.0) and (rng.random() < np.exp(-(cand_score - curr_score) / T))

        if accept:
            curr = cand
            curr_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{it:6d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # occasional random restart to avoid deep stagnation
        if no_improve >= 5000 and (deadline - time.monotonic()) > 5.0:
            curr = random_init(rng, n)
            curr_score = fast_score(curr)
            no_improve = 0
            # (the best found so far is kept)

        it += 1

    # Final verification with the reference evaluator (for logging only)
    try:
        official = evaluate_sequence(best.tolist())
        print(f"[final] best score reported by evaluate_sequence = {official:.6f}")
    except Exception:
        # In the isolated grading environment the function may not be
        # importable; ignore any errors.
        pass

    # Return a plain Python list as required by the harness
    return best.tolist()
