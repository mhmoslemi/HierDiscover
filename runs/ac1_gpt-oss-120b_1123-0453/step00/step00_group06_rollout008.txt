        idx = np.arange(n, dtype=np.float64)
        a = np.power(idx + 1.0, -alpha)
        # Add optional random perturbation
        a += 0.0  # remain
        return a

    best = None

    for _ in range(5):  # few random initial alphas

    step_frac = 0.2  # fraction of donor amount per move
    while time.monotonic() < deadline:

        max_idx = int(conv.argmax())

        contributions = np.zeros_like(best)

        receiver = int(np.argmin(contributions + 1e-12))  # ensure finite

"""
Propose a low‑value step‑function for the given evaluation routine.

The algorithm works as follows:

1. Start from a power‑law “arcsine‑like” sequence a_i = (i+1)^‑α with
   α≈0.5 (the analytically known construction gives an objective≈π/2≈1.57).
2. Repeatedly compute the full convolution c = a*a and locate the index k
   where c is maximal.
3. For that k compute the contribution of each position i to c[k]:
       contrib[i] = a[i] * a[k‑i]   (zero when the partner index is out of range).
   High contribution entries increase the maximal value, low contribution
   entries are good places to receive mass.
4. Move a small fraction of the mass from the donor (largest contribution)
   to the receiver (smallest contribution) while preserving the total sum.
   The move is accepted only if the objective improves.
5. Occasionally (≈2 % of iterations) apply a random block perturbation
   (the same primitive used by the reference implementation) to escape
   local minima.
6. If no improvement is seen for a while the algorithm re‑initialises from
   a fresh power‑law with a slightly perturbed exponent α∈[0.45,0.55].
7. The whole search stops when the time budget (default 1000 s) is exhausted,
   returning the best sequence found.

All numbers are kept non‑negative and clipped to the range [0,1000] which
matches the security checks performed inside ``evaluate_sequence``.
"""

import time
import numpy as np

def make_power_law(n: int, alpha: float) -> np.ndarray:
    """Return a symmetric power‑law sequence a_i = (i+1)^‑α (i=0..n‑1)."""
    idx = np.arange(n, dtype=np.float64)
    a = np.power(idx + 1.0, -alpha)
    # keep values small enough for the safety clip; scaling is irrelevant for the objective
    return np.clip(a, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce one O(n)‑memory non‑negative perturbation of ``base``.
    It multiplies a randomly chosen contiguous block by exp(N(0,σ²))
    and with 35 % probability mirrors the whole vector.
    """
    candidate = base.copy()
    n = candidate.size
    # block width roughly between n/200 and n/12 (as in the reference code)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional reflection to keep the sequence roughly symmetric
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def contributions_to_max(seq: np.ndarray, k: int) -> np.ndarray:
    """
    For a given sequence ``seq`` and a convolution index ``k``,
    return the array of contributions of each element to the value
    (seq*seq)[k] (i.e. a_i * a_{k‑i} when the partner index exists,
    otherwise 0).
    """
    n = seq.size
    i = np.arange(n, dtype=np.int64)
    j = k - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i[mask]] * seq[j[mask]]
    return contrib
