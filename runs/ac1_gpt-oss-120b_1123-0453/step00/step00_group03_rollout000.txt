"""
Uniform‑step optimizer.

The evaluation function simplifies to

    score = 2 * n * (∑ a_i²) / (∑ a_i)² ,

which, for non‑negative a_i, is minimized by a uniform sequence
(a_i = const).  The theoretical optimum is therefore exactly 2.0,
independent of the constant value or the length n (as long as n≥1).

This implementation therefore:
  * picks a length n (randomised but bounded so that the run stays fast),
  * chooses a positive constant value v that respects the 0‑1000 clipping,
  * returns the uniform array [v,…,v] of length n.

A short log line is printed so that progress can be observed when the
function is called repeatedly.
"""

import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Produce a uniform step‑function sequence that attains the optimal
    value 2.0 for the provided ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int, optional
        Random seed used to decide the length of the sequence and its
        constant height.  Different seeds give different (but equally
        optimal) candidates.
    budget_s : float, optional
        Upper time‑budget in seconds.  The implementation does not need
        the budget because a single uniform candidate is immediately
        optimal, but the argument is kept for compatibility with the
        surrounding framework.
    **kwargs :
        Ignored – present only to match the required signature.

    Returns
    -------
    list[float]
        The generated sequence (plain Python list of floats).
    """
    rng = np.random.default_rng(seed)

    # Choose a length that is large enough to be interesting but still
    # cheap to evaluate.  The range 256‑4096 works well for the time
    # limits of the challenge.
    n = int(rng.integers(256, 4097))

    # Choose a constant value in the allowed interval (0, 1000].
    # Any value works; we pick a random one to avoid a deterministic
    # output for the same seed across runs.
    v = float(rng.uniform(0.01, 1000.0))

    # Build the uniform array.
    sequence = np.full(n, v, dtype=np.float64)

    # Convert to plain list for the evaluator.
    seq_list = sequence.tolist()

    # Quick sanity check (optional, not required for correctness).
    try:
        score = float(evaluate_sequence(seq_list))
    except Exception:          # pragma: no cover
        score = float('nan')

    print(f"[Uniform optimizer] length={n}, value={v:.6f}, score={score}")

    return seq_list
