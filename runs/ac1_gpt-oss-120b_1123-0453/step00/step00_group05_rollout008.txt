"""
LP‑based iterative flattening of the auto‑convolution.
Starts from a random (or provided) non‑negative vector, normalises to sum 1,
and repeatedly solves a linear program that finds, under the current
convolution bound M = max (f*f), a vector g with maximal L1‑mass.
After normalising g we replace f by a convex combination of f and g.
The procedure reduces the largest convolution coefficient and drives the
objective 2·n·max(f*f) towards its theoretical lower bound ≈1.
The method runs under the given time budget and returns the best
sequence found.
"""
import numpy as np, time, scipy.linalg, scipy.optimize

def make_block_perturbation(base, rng):
    """Return a simple O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # perturb a random contiguous block
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetrisation
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

def _conv_matrix(f):
    """Build the (2n‑1) × n matrix A with A g = conv(f,g)."""
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # each column i is the vector f shifted down by i rows
    for i in range(n):
        A[i:i + n, i] = f
    return A

def _metric(a):
    """Return the objective 2·n·max(conv(a,a)) (assuming sum(a)=1)."""
    n = a.shape[0]
    max_b = np.max(np.convolve(a, a))
    return 2.0 * n * max_b

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    N = 256

            f = start[:N].copy()
        else:

            f = np.zeros(N, dtype=np.float64)
            f[: start.size] = start
    else:
        f = rng.random(N, dtype=np.float64)

    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 1e-12:
        f[:] = 1.0 / N
    else:
        f /= s

    best_f = f.copy()
    best_val = _metric(best_f)

    iteration = 0
    # main optimisation loop ------------------------------------------------
    while time.monotonic() < deadline:
        n = f.shape[0]
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=[(0, None)] * n,
                                    method='highs', options={'presolve': True})
        if res.success:
            g = res.x
            sum_g = g.sum()
            if sum_g > 0:

                g_norm = g / sum_g
                # form a convex combination – the step size decays slowly
                t = 0.5 * (0.99 ** iteration)
                cand = (1.0 - t) * f + t * g_norm
                cand /= cand.sum()                 # re‑normalise (numerical safety)

                val = _metric(cand)
                if val < best_val:
                    best_val = val
                    best_f = cand.copy()
                    f = cand.copy()
                    iteration = 0   # reset counter after a successful move
                    continue
        # -----------------------------------------------------------------
        # If the LP did not give an improvement, fall back to a cheap
        # random block perturbation (the strategy used in the reference code).
        # -----------------------------------------------------------------
        cand = make_block_perturbation(f, rng)
        cand_sum = cand.sum()
        if cand_sum > 0:

                best_f = cand.copy()

    # Return the best vector that was discovered.
    # --------------------------------------------------------------------
    # Convert to plain Python floats – the caller expects a list.

"""
LP‑based iterative flattening of the auto‑convolution.

The algorithm starts from a non‑negative vector (taken from the supplied
`height_sequence_1` when possible, otherwise a random one) and normalises
it to ℓ₁‑norm 1.  For a current sequence `f` we compute the bound

    M = max(conv(f, f))

and solve the linear programme

    maximise   1ᵀ g
    subject to  conv(f, g) ≤ M ,   g ≥ 0 .

Because `conv(f, g)` is linear in `g` this is a genuine LP.  Its optimum
adds as much ℓ₁‑mass as possible in the “holes’’ left by the present
convolution bound.  After normalising `g` we replace `f` by a convex
combination ` (1‑t)·f + t·g ` (the step size `t` decays slowly).  The
maximum convolution coefficient of `f` never grows and usually shrinks,
driving the objective

    2·n·max(conv(f, f))

towards the theoretical lower bound ≈ 1.  The loop runs until the time
budget expires and the best sequence found is returned.
"""

import numpy as np
import time
import scipy.optimize


def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # random contiguous block
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _conv_matrix(f):
    """
    Build the (2n‑1) × n matrix A such that A @ g == np.convolve(f, g).
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _metric(a):
    """Objective 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.shape[0]
    return 2.0 * n * np.max(np.convolve(a, a))
