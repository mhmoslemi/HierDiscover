def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine initial length n
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)
    n = base.size

    # Clip to [0,1000]
    parent = np.clip(base, 0.0, 1000.0)
    # Normalize sum to sqrt(2n) (optional) to keep numbers moderate
    sum_parent = np.sum(parent)
    target_sum = np.sqrt(2.0 * n)
    if sum_parent > 0:
        parent = parent * (target_sum / sum_parent)
    else:
        parent = np.ones(n, dtype=np.float64) * (target_sum / n)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial score: {best_score:.6f}")

    # also keep current sequence for exploring, start from best
    current = best.copy()
    cur_score = best_score

    # counters for stagnation
    no_improve_iters = 0

    # Main loop
    while time.monotonic() < deadline:
        # Randomly decide move type
        if rng.random() < 0.7:
            # mass transfer guided by current maximal convolution peak
            cand = current.copy()
            # compute convolution
            b = np.convolve(cand, cand)

            i_high = int(np.argmax(contrib))
            i_low = int(np.argmin(contrib))

                cand = make_block_perturbation(cand, rng)
            else:

                delta = float(rng.uniform(0.0, min(0.5, 0.1) * max_delta))

                cand[i_low] += delta

                cand = np.clip(cand, 0.0, 1000.0)

            cand = make_block_perturbation(current, rng)

            continue

        if score < best_score:
            best = cand.copy()
            best_score = score
            current = cand.copy()
            cur_score = score
            no_improve_iters = 0

    print(f"Finished. Best score {best_score:.6f}")

"""
Propose a near‑optimal step‑function for the evaluation metric.

The objective is to minimise
    score = 2·n·max(conv(a,a)) / (sum(a)**2)

For a non‑negative vector a the convolution at index k is
    (a*a)_k = Σ_i a[i]·a[k‑i] .
If we choose a_i ≈ 1/√(i+α) (α>0) then the continuous analogue of the
convolution is

    ∫₀ᵏ dx / √(x+α)·√(k‑x+α) = π                      (independent of k)

so the discrete auto‑convolution is almost flat.  Consequently the
maximal value of the convolution is close to the average and the
score approaches π/2 ≈ 1.57, well below the naïve uniform bound 2.0.

The algorithm:

1.  Determine the length `n` from the provided `height_sequence_1`
    (fallback to 256).
2.  Search a small 2‑dimensional grid for the parameters
    `p` (the power, centred on 0.5) and `α` (offset) that give the
    smallest score for the family a_i = (i+α)^{-p}.
3.  Using the best candidate as a starting point, run a cheap
    hill‑climbing phase: repeatedly move a small amount of mass from a
    position that contributes most to the current maximal convolution
    entry to a position that contributes least.  Keep any improvement.
4.  Return the best sequence found.

All steps respect the evaluation function’s constraints (non‑negative,
clipping to [0,1000]) and run within the allotted time budget.
"""

import numpy as np
import time


def make_block_perturbation(base, rng):
    """
    Return a cheap O(n) non‑negative perturbation of ``base``.
    The perturbation multiplies a random contiguous block by a log‑normal
    factor and possibly mirrors the whole vector.  Values are clipped to
    the interval [0,1000].
    """
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # choose a block width in a reasonably large range
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetry (helps escape plateaus)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

                score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    print(f"grid p={p:.3f} α={alpha:.3f} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) optional hill‑climbing from the best analytic vector.
    #    Move a small amount of mass from a "high‑contributor"
    #    to a "low‑contributor" for the current maximal convolution index.
    # ------------------------------------------------------------------
    if best_seq is None:               # should never happen, but be safe
        best_seq = np.ones(n, dtype=np.float64)

    current = best_seq.copy()
    cur_score = best_score
    max_iters = 10_000                 # safety cap; real stop is time‑based

    while time.monotonic() < deadline and max_iters > 0:
        # compute full convolution and locate the peak
        conv = np.convolve(current, current)
        k_max = int(np.argmax(conv))

        # contributions of each element to that peak
        i = np.arange(n, dtype=np.int64)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(current)
        contrib[mask] = current[i[mask]] * current[j[mask]]

        # donor – index with the largest contribution
        donor = int(np.argmax(contrib))
        # receiver – index with the smallest contribution (different from donor)
        receiver = int(np.argmin(contrib))
        if donor == receiver:
            # fallback to a random block move if the contributions are flat
            candidate = make_block_perturbation(current, rng)
        else:
            # transfer a small fraction of the donor’s mass
            available = current[donor]
            if available <= 0.0:
                candidate = make_block_perturbation(current, rng)
            else:
                delta = float(rng.uniform(0.0, min(0.5, 0.10) * available))
                if delta < 1e-12:
                    delta = available * 0.01
                candidate = current.copy()
                candidate[donor] -= delta
                candidate[receiver] += delta
                candidate = np.clip(candidate, 0.0, 1000.0)

        # evaluate the proposal
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < cur_score:
            # keep improvement
            current = candidate
            cur_score = score
            if score < best_score:
                best_score = score
                best_seq = candidate.copy()
                print(f"hill‑climb improved score to {best_score:.6f}")
        else:
            # occasional random jump to avoid dead‑ends
            if rng.random() < 0.001:
                current = candidate
                cur_score = score
        max_iters -= 1

    print(f"Finished search – best score achieved: {best_score:.6f}")
    # Return a plain Python list of floats as required by the grader.
    return [float(x) for x in best_seq]
