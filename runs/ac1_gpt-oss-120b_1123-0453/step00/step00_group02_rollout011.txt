"""
Propose a candidate step‑function heights sequence.

The algorithm treats the heights as a probability distribution (they are
scaled to sum to 1, which does not affect the evaluation because the
objective is homogeneous).  Starting from several random initialisations
it performs a simple projected sub‑gradient descent that reduces the peak
of the auto‑convolution (the quantity `max(a*a)`).  After each gradient
step the vector is projected onto the probability simplex (non‑negative,
sum = 1).  Whenever a new best score is found the global best is updated.
A few occasional block‑perturbations (the baseline heuristic) are applied
as a fallback to escape shallow minima.

The routine respects the allotted time budget, stops before the deadline,
and returns the best sequence discovered.
"""

import numpy as np
import time
from typing import List

# Helper: projection onto the probability simplex {x >= 0, sum(x) = 1}
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the largest rho satisfying the KKT condition
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: single gradient‑descent step on the max‑convolution objective
def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Perform one sub‑gradient step that tries to lower max(convolve(a, a)).
    Returns the projected new vector.
    """
    n = a.size
    # full convolution (real numbers)
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))                 # index of the current worst peak
    # gradient of max‐convolution w.r.t. a_i = 2 * a_{k-i} (if index in range)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    # take a step in the negative gradient direction
    a_new = a - step * g
    # project back onto the simplex (keeps sum == 1 and non‑negative)
    return _proj_simplex(a_new)


# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """One cheap O(n) perturbation of ``base`` (fallback heuristic)."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> List[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The method combines projected sub‑gradient
    descent on the max‑convolution with occasional block perturbations.
    It returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine problem size -------------------------------------------------
    # Use at most 2000 coefficients (fast enough for many evaluations) but
    # keep the user‑provided length if it is already small.
    max_n = 2000
    try:
        # height_sequence_1 is supplied by the platform; fall back if missing.
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_seq = np.ones(256, dtype=np.float64)

    if base_seq.ndim != 1 or base_seq.size == 0:
        base_seq = np.ones(256, dtype=np.float64)

    n = int(min(max_n, base_seq.size))
    # Take a contiguous slice of the provided sequence (if longer) and clip.
    start = 0
    parent = np.clip(base_seq[start:start + n], 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Initial candidates (probability vectors, sum == 1)
    candidates = []

    # (1) Uniform distribution – serves as a simple reference.
    candidates.append(np.full(n, 1.0 / n))

    # (2) Truncated user sequence, renormalised to a probability vector.
    if parent.size == n:
        prob = parent.copy()
        prob_sum = prob.sum()
        if prob_sum > 1e-12:
            prob = prob / prob_sum
        else:
            prob = np.full(n, 1.0 / n)
        candidates.append(prob)

    # (3) Several random Dirichlet draws (sparse‑ish and dense‑ish)
    for _ in range(3):
        # α < 1 favours sparsity, α > 1 favours uniformity.
        alpha = rng.uniform(0.3, 2.0)
        vec = rng.dirichlet(np.full(n, alpha))
        candidates.append(vec)

    # Keep the best one seen so far.
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop ------------------------------------------------
    # We iterate over the pool of initial vectors, perform a limited number
    # of sub‑gradient steps for each, and keep the best overall result.
    for init_idx, a in enumerate(candidates):
        if time.monotonic() > deadline - 0.5:
            break

        # Ensure we start on the simplex.
        a = _proj_simplex(a)

        cur_score = float(evaluate_sequence(a.tolist()))
        if cur_score < best_score:
            best_seq = a.copy()
            best_score = cur_score
            print(f"[Init {init_idx}] initial score {cur_score:.6f}")

        # Gradient‑descent parameters.
        step = 0.2               # initial step size
        decay = 0.995            # geometric decay per iteration
        max_iters = 2000

        for it in range(max_iters):
            if time.monotonic() > deadline - 0.5:
                break

            # One sub‑gradient step.
            a_next = _subgradient_step(a, step)

            # Evaluate.  Using evaluate_sequence (the authoritative evaluator)
            # keeps the comparison fair.
            score_next = float(evaluate_sequence(a_next.tolist()))

            # Keep the global best if we improve.
            if score_next < best_score:
                best_seq = a_next.copy()
                best_score = score_next
                print(f"[Init {init_idx}, iter {it}] new best {best_score:.6f}")

            # Accept the move if it improves the local objective;
            # otherwise keep the old vector (simple greedy descent).
            if score_next < cur_score:
                a = a_next
                cur_score = score_next
            else:
                # Occasionally try a tiny random jitter to escape flat spots.
                if rng.random() < 0.01:
                    jitter = rng.normal(0.0, 0.001, size=n)
                    a = _proj_simplex(a + jitter)

            # Decrease the step size.
            step *= decay

        # ------------------------------------------------------------------
        # After finishing the descent for this seed, perform a few cheap
        # block perturbations (the baseline approach) starting from the
        # current best for additional exploration.
        for _ in range(30):
            if time.monotonic() > deadline - 0.5:
                break
            cand = make_block_perturbation(best_seq, rng)
            # Re‑normalise to the same scale (sum == 1) before evaluation.
            cand = np.clip(cand, 0.0, 1000.0)
            s = cand.sum()
            if s > 0:
                cand = cand / s
            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[Perturb] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback: if nothing was ever accepted, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Scale back to a “reasonable” magnitude (the evaluator is scale‑invariant,
    # but keeping numbers away from extreme 0 helps numeric stability).
    # We map the simplex vector to the range [0, 1] (already the case) and
    # optionally multiply by a constant – here we keep it unchanged.
    result = [float(x) for x in best_seq.tolist()]
    print(f"Final best score: {best_score:.6f} (n={n})")
    return result
