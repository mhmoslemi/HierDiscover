"""
Propose a low‐score step‑function using a Sidon set construction.

The evaluation function is
    score = 2 * n * max(conv(seq, seq)) / (sum(seq)^2)

For a non‑negative sequence the convolution maximum is at least the
largest self‑product a_i*a_i (i.e. a_i^2).  By placing a set of large
entries (value 1000) on a Sidon set – a set whose pairwise sums are all
different – we guarantee that *no* convolution index contains more than
one product of two large entries.  All remaining positions are filled
with a small constant ε.  Choosing

    ε = 1000 / sqrt(N)

makes the three contributions to the convolution maximum

* large–large      : 1000²
* ε–ε              : ε²·N = 1000²
* large–ε          : #large·1000·ε = 1000²

so the maximal convolution value is ≈ 3·10⁶.  The total sum is

    Σ a_i = (#large)·1000 + ε·N  ≈ 2·1000·sqrt(N),

giving an evaluation score ≈ 1.5, well below the previous bound 2.0.
The function searches a few values of the Sidon set size (parameter `p`)
within the time budget and returns the best sequence found.
"""

import time
import math
import numpy as np

def sidon_positions(p: int) -> np.ndarray:
    """Return a Sidon set (B₂ sequence) of size `p` inside [0, p*(p+1))."""
    i = np.arange(p, dtype=np.int64)
    # classic construction a_i = i·p + (i² mod p)
    return i * p + (i * i) % p


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a step‑function sequence that minimise `evaluate_sequence`.

    The algorithm:
      * Randomly pick a Sidon set size `p` (200 ≤ p ≤ 800).
      * Build a sequence of length N = p·(p+1).
      * Fill every entry with ε = 1000 / sqrt(N).
      * Replace the Sidon positions by the maximal allowed height 1000.
      * Evaluate the sequence; keep the best one seen before the time budget runs out.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    M = 1000.0                # maximal allowed height
    best_seq = None
    best_score = float('inf')

    # try a handful of different Sidon sizes while we have time
    while time.monotonic() < deadline:
        # choose a size p – we keep it moderate so that convolution is fast
        p = int(rng.integers(200, 801))          # 200 ≤ p ≤ 800
        N = p * (p + 1)                          # length of the sequence

        # ε chosen exactly as 1000 / sqrt(N)
        eps = M / math.sqrt(N)

        # initialise the whole sequence with ε
        seq = np.full(N, eps, dtype=np.float64)

        # set the Sidon positions to the maximal height
        pos = sidon_positions(p)
        # pos is guaranteed to be < N
        seq[pos] = M

        # evaluate the candidate
        try:
            score = float(evaluate_sequence(seq.tolist()))
        except Exception:                        # safety – should never happen
            continue

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best: score={best_score:.6f}, p={p}, N={N}, eps={eps:.6f}")

        # stop early if we already beat the target
        if best_score < 1.503:
            break

    # Fallback in the very unlikely event that nothing was found
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("Fallback to a uniform sequence (score={:.6f})".format(best_score))

    print(f"Final best score: {best_score:.6f}")
    return [float(x) for x in best_seq]
