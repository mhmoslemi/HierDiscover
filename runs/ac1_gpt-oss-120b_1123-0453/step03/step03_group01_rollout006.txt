def improve_via_subgradient(a, deadline):
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(a, target_sum)   # ensure sum
    # Precompute conv
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_score = np.max(conv)   # this is score.
    it = 0
    eta0 = 1.0  # initial step size; will adapt.
    while time.monotonic() < deadline and it < 1_000_000:
        conv = np.convolve(a, a)
        k_max = np.argmax(conv)
        # compute gradient
        grad = np.zeros_like(a)
        # idx range
        start_i = max(0, k_max - (n - 1))
        end_i = min(n - 1, k_max)
        # for i in [start_i, end_i], j = k_max - i
        grad[start_i:end_i+1] = 2.0 * a[k_max - np.arange(start_i, end_i+1)]
        # step size schedule:
        # Use decreasing step: eta = eta0 / sqrt(it+1)
        eta = eta0 / np.sqrt(it + 1)
        a_new = a - eta * grad
        # ensure nonnegative via projection onto simplex
        a_new = _project_onto_simplex(a_new, target_sum)
        # evaluate conv after projection
        conv_new = np.convolve(a_new, a_new)
        cur_score = np.max(conv_new)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_new.copy()
            # optionally reduce eta0 to accelerate? Not needed.
        a = a_new
        it += 1
        if it % 1000 == 0:
            # maybe log progress
            print(f"[sgd] n={n} it={it} score={best_score:.6f}")
            if best_score < 1.503:
                break
    return best_a, best_score

def _targeted_hill_climb(a, conv, deadline,
                         max_iters=200_000,
                         fraction_range=(0.2, 0.8)):
    n = a.shape[0]
    best_a = a.copy()
    best_score = np.max(conv)
    rng = np.random.default_rng()
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        kmax = int(np.argmax(conv))
        # Determine overlapping indices
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        if lo > hi:
            # shouldn't happen
            break
        idx = np.arange(lo, hi+1)
        # compute contributions a[i] * a[kmax - i]
        prod = a[idx] * a[kmax - idx]
        # choose source with largest contribution
        src_idx = int(idx[int(np.argmax(prod))])
        # source mass may be small; if zero, break
        if a[src_idx] < 1e-12:
            # find next maybe random
            # break
            break
        # target region
        if kmax < n-1:
            target_region = np.arange(kmax+1, n)
        else:
            target_region = np.arange(0, kmax - (n-1))
        if target_region.size == 0:
            break
        # pick target index with minimal a
        tgt_idx = int(target_region[int(np.argmin(a[target_region]))])
        # choose fraction
        frac = rng.random() * (fraction_range[1] - fraction_range[0]) + fraction_range[0]
        delta = a[src_idx] * frac
        if delta <= 1e-12:
            break
        # apply transfer
        _transfer_mass(a, conv, src_idx, tgt_idx, delta)
        # compute new max
        cur_score = np.max(conv)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert
            _transfer_mass(a, conv, tgt_idx, src_idx, delta)
        it += 1
        # early exit if best_score already good
        if best_score < 1.503:
            break
    return best_a, best_score

"""
Propose a non‑negative step sequence with a low evaluation score.

Algorithm outline
-----------------
1.  Choose several candidate lengths (300 … 1200).  For each length create a
    handful of initial vectors:
        * uniform vector,
        * random exponential vectors,
        * a smooth sinusoidal “bell” shape,
        * the previously discovered global variable `height_sequence_1`
          if its length matches.
    Each start is projected onto the simplex
        {x ≥ 0 , Σ x = √(2 n)} – the evaluation is scale invariant.

2.  **Targeted hill‑climbing** – the core optimiser.
    Repeatedly move a fraction of mass from the index that contributes the
    most to the current maximal convolution entry to an index that does not
    affect that entry (i.e. lies outside the overlap region).
    The convolution is kept *incrementally* up‑to‑date with O(n) work per
    move using the exact algebraic update, so millions of iterations are
    cheap.  The step fraction is randomised and slowly decays; moves that
    do not improve the score are reverted.

3.  After the targeted phase a short Gaussian perturbation
    (`_local_search`) is applied as a polish step.

4.  The best sequence found (across all lengths and starts) is returned.
    If the score falls below the target 1.503 the search stops early.

All helper functions are top‑level, use only NumPy and the provided
`evaluate_sequence`, and respect the non‑negativity / clipping rules
implemented there.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official score – wrapper around the supplied `evaluate_sequence`."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        { x ≥ 0 , Σ x = target_sum }.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer ``delta`` from index ``i`` to ``j`` (i ≠ j, delta > 0,
    delta ≤ a[i]).  Both ``a`` and its convolution ``conv = np.convolve(a,a)``
    are updated **in‑place** using the exact algebraic formula (O(n) work).
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part: 2·a*δ
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part: δ·δ on the three self‑indices
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta          # (-δ)²
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta    # (-δ)*(+δ) + (+δ)*(-δ)
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta          # (+δ)²

    # finally move the mass in the vector
    a[i] -= delta
    a[j] += delta


def _targeted_hill_climb(a: np.ndarray,
                         conv: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         frac_low: float = 0.2,
                         frac_high: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing that always attacks the current maximal
    convolution entry.  Mass is moved from the index that contributes most
    to that entry to an index that does **not** affect it (outside the
    overlap region).  The move is kept only if the maximal value decreases.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = np.max(conv)               # with sum fixed, score == max(conv)

    rng = np.random.default_rng()
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        kmax = int(np.argmax(conv))

        # indices that overlap with offset kmax
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        idx = np.arange(lo, hi + 1)
        # contribution of each overlapping index to conv[kmax]
        prod = a[idx] * a[kmax - idx]
        src = int(idx[int(np.argmax(prod))])

        # choose a target that does *not* overlap with kmax
        if kmax < n - 1:
            # any index > kmax lies outside the overlap
            target_region = np.arange(kmax + 1, n)
        else:
            # indices < kmax-(n-1) are outside
            target_region = np.arange(0, kmax - (n - 1))

        if target_region.size == 0:
            break

        # pick the sparsest location in that region
        tgt = int(target_region[int(np.argmin(a[target_region]))])

        # random fraction of the source mass
        frac = rng.random() * (frac_high - frac_low) + frac_low
        delta = a[src] * frac
        if delta <= 1e-12:
            break

        # try the move
        _transfer_mass(a, conv, src, tgt, delta)
        cur_score = np.max(conv)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert – we keep a monotone decreasing sequence of scores
            _transfer_mass(a, conv, tgt, src, delta)

        # occasional early exit
        if best_score < 1.503:
            break

        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Gentle stochastic polishing – add Gaussian noise to the whole vector,
    accept only improvements.  Uses the official ``evaluate_sequence`` for
    the exact score.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------

    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    lengths = [300, 400, 500, 600, 800, 1000, 1200]

        target_sum = math.sqrt(2.0 * n)

        # a small collection of diverse starting vectors
        starts: list[np.ndarray] = []

        # 1) uniform
        starts.append(np.full(n, target_sum / n, dtype=float))

        x = np.arange(n, dtype=float)

        starts.append(bell)

        # 3) a few exponential random vectors

            if time.monotonic() < deadline:

            # evaluate with the official function (covers clipping etc.)
            official_score = _score(improved)

            if official_score < best_score:
                best_score = official_score
                best_seq = improved.copy()

    if best_seq is None:

        best_seq = np.full(n, target_sum / n, dtype=float)

        conv = np.convolve(best_seq, best_seq)
        best_seq, _ = _targeted_hill_climb(best_seq.copy(),
                                           conv.copy(),
                                           deadline,
                                           max_iters=300_000)

    if time.monotonic() < deadline:
        best_seq, final_score = _local_search(best_seq, deadline, step_initial=0.2)
        if final_score < best_score:
            best_score = final_score
            print(f"[final] improved to {best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
