The algorithm consists of two stages:

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact score used by `evaluate_sequence` (no internal clipping)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    # avoid overflow / underflow
    max_c = np.max(conv)
    shifted = (conv - max_c) / beta
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    return w


def _weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k * conv_k(a,a) w.r.t. `a`.
    `weights` must have length 2·len(a)‑1 and sum to 1.
    """
    n = len(a)
    grad = np.zeros_like(a)
    # loop over the convolution offsets – cheap for n up to a few thousand
    for k, wk in enumerate(weights):
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = a[j]
        grad[i] += wk * 2.0 * vals
        # diagonal entries (i == j) were added twice, we need only one copy
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= wk * vals[mask]
    return grad


def _adam_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 20000,
    patience: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation of the smoothed max‑convolution.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = len(a)
    # Adam hyper‑parameters
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # ----- compute smoothed gradient -----
        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        # smoothing parameter – a few percent of the current maximal value
        beta = max_conv * 0.07 + 1e-12
        w = _softmax_weights(conv, beta)
        grad = _weighted_gradient(a, w)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cur_score = _compute_score_from_vector(a)
        cand_score = _compute_score_from_vector(a_candidate)

        # accept if there is an improvement
        if cand_score < cur_score:
            a = a_candidate
        # record the globally best vector
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            print(f"[Adam] n={n} it={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score


def _stochastic_refinement(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.3,
    swap_prob: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Gaussian hill‑climbing + occasional random pairwise swaps.
    Keeps the ℓ¹‑norm fixed via simplex projection.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            # random swap of two coordinates
            i, j = rng.integers(0, len(best), size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # Gaussian perturbation
            cand = best + rng.normal(scale=step, size=best.shape)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # enlarge step a little when we improve
            step = min(step_initial, step * 1.02)
        else:
            # otherwise slowly decay the step size
            step *= 0.995

        # keep step within reasonable bounds
        if step < 1e-6:
            step = step_initial

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the time budget.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣ Warm‑start from any previously found construction.
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        if np.sum(seq) == 0:
            seq = np.full_like(seq, 1.0 / len(seq))
        else:
            seq = seq / np.sum(seq)          # normalise to sum = 1
        score = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[warm] loaded height_sequence_1, n={len(seq)} score={score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Main optimisation – Adam descent with several lengths/restarts.
    # --------------------------------------------------------------
    # A range of lengths – larger n gives a lower theoretical bound.
    lengths = [600, 800, 1000, 1200, 1500, 1800, 2000, 2500]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = 1.0                      # we keep the ℓ¹‑norm = 1

        # at most three restarts per length (more for small n)
        restarts = 5 if n <= 1200 else 3
        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a candidate
            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
            else:
                a0 = rng.random(n) + 1e-6          # avoid exact zeros
                a0 = a0 / np.sum(a0) * target_sum

            # allocate a slice of the remaining budget (≈ 10 s per run)
            sub_deadline = min(deadline, time.monotonic() + 10.0)
            a_opt, sc_opt = _adam_descent(a0, target_sum, sub_deadline,
                                          max_iters=20000, patience=500)

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[update] n={n} restart={rs} score={sc_opt:.6f}")

            # early exit if we beat a comfortable margin
            if best_score < 1.45:
                print("[early] satisfactory score reached")
                break

        if best_score < 1.45:
            break

    # --------------------------------------------------------------
    # 2️⃣ Final stochastic polishing (if time permits).
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: modest uniform sequence
        n = 800
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline,
                                                         step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # final safety clipping (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # --------------------------------------------------------------
    # Return a plain Python list – the evaluation script will call
    # `evaluate_sequence` on it afterwards.
    # --------------------------------------------------------------
    return [float(x) for x in best_seq]
