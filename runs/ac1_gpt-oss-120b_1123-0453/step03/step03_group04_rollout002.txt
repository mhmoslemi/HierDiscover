import time, math, numpy as np

def _linear_convolve(x, y):
    n = len(x) + len(y) - 1
    size = np.fft.next_fast_len(n)
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[:n]
    return conv

def _smooth_weights(conv, beta):
    # conv: np.ndarray length 2n-1
    max_c = conv.max()
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / exp_term.sum()
    return w

def _gradient_smooth(a, w):
    # a: (n,), w: (2n-1,)
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)  # length 3n-2
    n = len(a)
    segment = corr[n-1:2*n-1]   # n elements
    grad = 2.0 * segment[::-1]
    return grad

def _fast_score(seq):
    conv = np.convolve(seq, seq)  # fine for small n
    max_b = conv.max()
    return max_b

"""
Search for a low‑value step sequence.

The routine works in three stages:

0️⃣  Use an existing construction (if present) as a starter.
1️⃣  Generate cheap parametric candidates (power‑law, exponential) for
    a range of lengths and keep the best.
2️⃣  For each length run several random restarts of a *smooth‑max*
    gradient descent.  The objective  max(a∗a) is approximated by a
    differentiable soft‑max (log‑sum‑exp).  Gradients are obtained by
    FFT‑accelerated convolutions, the iterate is projected onto the
    feasible set (non‑negative, ℓ¹‑norm = √(2 n)).  The schedule for the
    smoothness parameter β is increased gradually, and a modest amount
    of random jitter is injected to avoid stagnation.

Finally a very cheap random‑mass‑swap hill‑climb is applied.
The best sequence found before the time budget expires is returned as a
plain Python list, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return a good FFT length (power‑of‑2 fallback if needed)."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (speed‑optimal for many calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fast_score(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` after normalisation (max‑auto‑conv)."""
    conv = np.convolve(seq, seq)          # cheap for the moderate sizes we use
    return float(conv.max())


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and enforce ℓ¹‑norm = √(2·n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / total)
    return seq


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries (stable version)."""
    cmax = conv.max()
    # subtract the max to avoid overflow
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    For a_i the derivative is 2·∑_j w_{i+j}·a_j.
    This is a cross‑correlation that can be obtained by one FFT convolution.
    """
    w_rev = w[::-1]                                   # reverse the weight vector
    corr = _linear_convolve(a, w_rev)                 # length = 3·n‑2
    n = a.shape[0]
    # extract the middle n entries (indices n‑1 … 2·n‑2) and reverse
    segment = corr[n - 1 : 2 * n - 1]                  # shape (n,)
    grad = 2.0 * segment[::-1]
    return grad


def _optimize_sequence(init_seq: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Smooth‑max gradient descent with an annealing schedule for β.
    Returns the best vector encountered and its true max‑auto‑convolution.
    """
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    best_local = a.copy()
    best_score = _fast_score(a)

    it = 0
    beta = 1.0                       # start with a very smooth approximation
    base_step = (target / n) * 5.0   # step roughly five times the average entry

    while time.monotonic() < deadline and it < 1500:
        # ----- evaluate current point -----
        conv = _linear_convolve(a, a)               # length 2·n‑1
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # ----- smooth‑max and gradient -----
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # ----- line‑search‑like step (simple decay) -----
        step = base_step / (1.0 + it * 0.02)
        a = a - step * grad

        # ----- projection onto the feasible set -----
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # ----- occasional stochastic jitter to avoid plateaus -----
        if it % 200 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.03
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # ----- anneal β (make the surrogate sharper) -----
        if it % 100 == 0 and beta < 250.0:
            beta *= 1.45

        it += 1

    return best_local, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.

    The method combines a quick parametric sweep, several random restarts,
    and a smooth‑max gradient descent accelerated with FFTs.  All vectors
    are kept on the ℓ¹‑sphere required by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the global best
    # ------------------------------------------------------------------
    def _update_global(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = _fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global best] n={len(seq)} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Warm‑start from an existing construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            cand = _normalize(cand)
            _update_global(cand)
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (power‑law & exponential families)
    # ------------------------------------------------------------------
    lengths = [1500, 1200, 1000, 800, 600, 500, 400, 300, 200, 150, 100]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(1, n + 1, dtype=np.float64)

        # power‑law a_i ∝ i^{‑p}
        for p in np.linspace(0.2, 3.0, 20):
            a = idx ** (-p)
            a = a / a.sum() * target
            a = _normalize(a)
            _update_global(a)

        # exponential a_i ∝ exp(-α·i)
        for alpha in np.linspace(0.001, 0.15, 20):
            a = np.exp(-alpha * np.arange(n, dtype=np.float64))
            a = a / a.sum() * target
            a = _normalize(a)
            _update_global(a)

    # ------------------------------------------------------------------
    # 2️⃣  Random restarts + smooth‑max gradient descent
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # fewer restarts when n is large (each run is more expensive)
        restarts = 6 if n <= 600 else 3

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # Dirichlet‑like random start
            start = rng.gamma(1.0, size=n)
            start = start / start.sum() * target

            # blend with current global best when dimensions match
            if best_seq is not None and best_seq.shape[0] == n:
                start = (start + best_seq) / 2.0
                start = _normalize(start)

            seq_opt, sc_opt = _optimize_sequence(start, deadline, rng)
            _update_global(seq_opt)

            if best_score < 1.503:   # target already achieved
                print("[early exit] target reached")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 3️⃣  Final cheap refinement – random mass swaps
    # ------------------------------------------------------------------
    if best_seq is not None:
        while time.monotonic() < deadline:
            n = best_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            delta = best_seq[i] * 0.03
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _normalize(cand)
            _update_global(cand)
    else:
        # fallback – a simple uniform vector
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # Return a plain Python list as required by the outer harness
    return [float(x) for x in best_seq]
