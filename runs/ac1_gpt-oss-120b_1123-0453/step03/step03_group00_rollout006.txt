"""
Step‑function minimiser.

The search uses a *conditional‑gradient* (Frank‑Wolfe) scheme that
alternates between:
    • Computing M = max_k (f*f)_k .
    • Solving a linear programme    maximise Σ g_j   s.t.   f*g ≤ M , g ≥ 0 .
The LP finds a vector g with a larger ℓ¹‑norm than f while keeping the
mixed convolution bounded by the current worst value M.
After scaling g back to the same ℓ¹‑norm as f we take a small convex
step f ← (1‑t)f + t g.  Repeating this reduces the maximal
auto‑convolution entry and therefore the evaluation score.

The outer loop tries several lengths and random restarts, keeps the best
candidate and finishes with a tiny stochastic hill‑climbing polish.
All vectors are kept non‑negative, summed to √(2 n) (any scaling works
because the objective is homogeneous) and clipped to 0…1000 to satisfy
the security checks of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# optional – SciPy is available on the platform used for the competition
try:
    from scipy.optimize import linprog          # linear programming
    from scipy.linalg import toeplitz           # building the convolution matrix
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s}.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # sort descendingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # edge case – return a uniform distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """
    Exact evaluation of ``evaluate_sequence`` (the clipping inside the
    original function is unnecessary for the values we generate).
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Tiny Gaussian hill‑climbing used for the final polishing stage.
    The ℓ¹‑norm is preserved by a projection onto the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Conditional‑gradient optimisation (Frank‑Wolfe for the min‑max problem)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Full convolution matrix C such that C @ g == np.convolve(f, g)
    (shape: (2·n‑1, n)).
    """
    n = len(f)
    col = np.r_[f, np.zeros(n - 1)]
    row = np.r_[f[0], np.zeros(n - 1)]
    return toeplitz(col, row)


def _solve_lp_for_g(f: np.ndarray, M: float) -> tuple[np.ndarray, bool]:
    """
    Solve   maximise Σ g_j   subject to   f * g ≤ M   (component‑wise) and g ≥ 0.
    Returns the optimal g and a success flag.
    """
    n = len(f)
    A = _build_conv_matrix(f)                      # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)    # right‑hand side
    c = -np.ones(n, dtype=np.float64)              # maximise Σ g  → minimise –Σ g
    bounds = [(0.0, None)] * n

    # ``highs`` is the default modern solver in SciPy ≥1.6
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return np.zeros_like(f), False
    return res.x, True


def _conditional_gradient(f0: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          max_iters: int = 100) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style optimisation that reduces the maximal auto‑convolution
    entry while keeping the vector on the ℓ¹‑simplex ( Σ a_i = target_sum ).
    """
    if not _SCIPY_AVAILABLE:          # safety net – fall back to the
        raise RuntimeError("SciPy not available, cannot run LP‑based method")

    f = f0.copy()
    # normalise to the requested ℓ¹‑norm
    f = f * (target_sum / np.sum(f))

    best = f.copy()
    best_score = _compute_score_from_vector(best)

    t = 0.2                # initial convex‑combination weight

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # current worst auto‑convolution value
        conv_ff = np.convolve(f, f)
        M = np.max(conv_ff)

        # LP: find a vector with a larger ℓ¹‑norm that respects the same bound M
        g0, ok = _solve_lp_for_g(f, M)
        if not ok:
            break

        sum_g0 = np.sum(g0)

        # If the LP cannot increase the ℓ¹‑norm we have reached a stationary point
        if sum_g0 <= target_sum * (1.0 + 1e-12):
            # nothing better can be achieved under the present bound
            break

        # Scale g back to the simplex (the scaling factor is ≤ 1)
        scale = target_sum / sum_g0
        g = g0 * scale

        # Convex update
        f = (1.0 - t) * f + t * g

        # Re‑project onto the simplex (numerical drift)
        f = _project_onto_simplex(f, target_sum)

        # Clamp to the range accepted by ``evaluate_sequence`` (just a safeguard)
        f = np.clip(f, 0.0, 1000.0)

        score = _compute_score_from_vector(f)
        if score < best_score - 1e-12:
            best_score = score
            best = f.copy()
            # slowly reduce the step size to refine the neighbourhood
            t = max(0.01, t * 0.99)
        else:
            # no improvement – be more conservative
            t = max(0.01, t * 0.9)

    return best, best_score

# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the allotted time.
    Returns a plain Python list of floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.3)   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction, if any.
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        n = len(seq)
        target_sum = math.sqrt(2.0 * n)
        seq = seq * (target_sum / np.sum(seq))
        score = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[init] used height_sequence_1, n={n}, score={score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Main search – conditional gradient with several lengths.
    # --------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000, 1200, 1500]   # a reasonable spread
    max_restarts = 4                                         # per length

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for restart in range(max_restarts):
            if time.monotonic() >= deadline:
                break

            # ----- initialise a candidate vector --------------------------------
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
            else:
                # random power‑law shape + tiny noise
                exponent = rng.uniform(-2.0, 2.0)
                idx = np.arange(n, dtype=np.float64) + 1.0
                f0 = idx ** exponent
                f0 += rng.normal(scale=0.01, size=n)      # small perturbation
                f0 = np.clip(f0, 0.0, None)
                f0 = f0 / np.sum(f0) * target_sum

            # ----- run the conditional‑gradient optimiser ----------------------
            sub_deadline = min(deadline, time.monotonic() + 12.0)   # per run budget
            try:
                f_opt, score_opt = _conditional_gradient(f0,
                                                         target_sum,
                                                         sub_deadline,
                                                         max_iters=120)
            except Exception:   # fallback – if something goes wrong use the old method
                # graceful degradation to the gradient‑descent code from the
                # previous submission (omitted here for brevity)
                continue

            if score_opt < best_score:
                best_score = score_opt
                best_seq = f_opt.copy()
                print(f"[update] n={n} restart={restart} score={score_opt:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                print("[early] target achieved")
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # 2️⃣ Final stochastic polishing (if time permits).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: a modest uniform construction
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline,
                                                         step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[refine] polished score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best candidate (clipped – evaluate_sequence also does this)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[final] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
