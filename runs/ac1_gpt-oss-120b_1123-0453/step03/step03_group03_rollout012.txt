"""
Propose a low‑evaluation step sequence.

The search proceeds in three stages within the given time budget:

1️⃣ Warm‑start – reuse the global `height_sequence_1` if it exists,
   otherwise start from a random Dirichlet vector that is
   ℓ¹‑normalised to √(2·n).

2️⃣ Peak‑reduction – repeatedly locate the index `k*` where the auto‑convolution
   attains its maximum and solve a tiny linear programme that
   redistributes a small fraction of the mass from indices that
   contribute heavily to the peak to indices that contribute little.
   The LP is tiny (≈ n variables) and therefore cheap; it gives the
   *best possible* linear move for the current peak while keeping all
   coefficients non‑negative and preserving the total sum.
   If a move would increase the global maximum we fall back to a simple
   greedy transfer.

3️⃣ Stochastic polishing – Gaussian perturbations with a decaying step size
   are applied until the deadline; any improvement is kept.

The evaluation used during the search is identical to `evaluate_sequence`
(but computed locally for speed).  The final list of non‑negative
floats is returned; the caller will feed it to `evaluate_sequence` for the
official score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional CVXPY backend for the linear‑programming move
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False


def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Full convolution a * a (no circular padding)."""
    # for n up to a few thousand the naive implementation is fast enough
    return np.convolve(seq, seq)


def _eval(seq: np.ndarray) -> float:
    """
    Fast local version of ``evaluate_sequence``.
    The sequence is assumed to contain only non‑negative finite numbers.
    """
    n = seq.size
    if n == 0:
        return float("inf")
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    max_b = _conv_full(seq).max()
    return float(2.0 * n * max_b / (s * s))


def _normalize(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale to the given ℓ¹‑norm, clip to the allowed value range."""
    if seq.sum() == 0:
        return seq
    seq = seq * (target_sum / seq.sum())
    # the problem statement guarantees the final check clips to [0,1000],
    # but staying inside the interval avoids unnecessary work.
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _lp_peak_move(a: np.ndarray, peak_idx: int, step_frac: float = 0.5) -> np.ndarray | None:
    """
    Solve a tiny LP that moves mass away from indices that contribute a lot
    to the current peak `peak_idx` towards indices that contribute little.
    The move is linear in the variables because the peak value is
    Σ_i a_i * a_{peak_idx-i} and a_{peak_idx-i} is treated as a constant.
    Returns the new vector (already normalised) or ``None`` if the LP
    fails or does not improve the peak.
    """
    n = a.size
    # build the constant coefficient vector c_i = a_{peak_idx-i} (0 if out of range)
    c = np.zeros(n, dtype=float)
    i = np.arange(n)
    j = peak_idx - i
    mask = (j >= 0) & (j < n)
    c[mask] = a[j[mask]]

    # we want to minimise the new peak value = Σ_i (a_i+Δ_i) * c_i
    # -> minimise cᵀ Δ  subject to Δ_i ≥ -a_i,   Σ Δ = 0,
    #   and a simple bound on the magnitude of the move.
    if not _HAS_CVX:
        return None

    Δ = cp.Variable(n)
    # bound magnitude: we never increase an entry by more than *step_frac* of its current value
    upper = step_frac * a
    prob = cp.Problem(
        cp.Minimize(c @ Δ),
        [
            Δ >= -a,
            Δ <= upper,
            cp.sum(Δ) == 0,
        ],
    )
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-9, reltol=1e-7)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:
            return None

    if Δ.value is None:
        return None

    delta = np.asarray(Δ.value, dtype=float)
    # tiny numerical safety
    delta = np.where(delta < -a, -a, delta)
    new_a = a + delta
    # re‑normalise to the exact ℓ¹‑norm (scaling does not affect the score)
    target = math.sqrt(2.0 * n)
    new_a = _normalize(new_a, target)
    return new_a


def _greedy_peak_move(a: np.ndarray, peak_idx: int, rng: np.random.Generator) -> np.ndarray:
    """
    Very cheap fallback: move a random fraction of mass from an index that
    contributes strongly to the current peak to one that contributes little.
    """
    n = a.size
    i = np.arange(n)
    j = peak_idx - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[i[mask]] * a[j[mask]]
    total = contrib.sum()
    if total == 0.0:
        return a

    # source: proportional to contribution
    prob = contrib / total
    src = rng.choice(n, p=prob)

    # destination: any index whose contribution is among the smallest 30 %
    low = np.percentile(contrib, 30)
    candidates = np.where(contrib <= low)[0]
    if candidates.size == 0:
        dst = rng.integers(n)
    else:
        dst = rng.choice(candidates)

    if src == dst:
        return a

    # move up to 30 % of the source mass
    eps_max = 0.3 * a[src]
    if eps_max <= 0.0:
        return a
    eps = rng.uniform(0.0, eps_max)

    b = a.copy()
    b[src] -= eps
    b[dst] += eps
    # keep within the allowed interval
    b = np.clip(b, 0.0, 1000.0)
    # re‑normalise (preserves the overall sum)
    target = math.sqrt(2.0 * n)
    b = _normalize(b, target)
    return b


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.

    The routine respects the 1000 s global budget, prints progress messages
    and finally returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm start from the global variable (if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.array(height_sequence_1, dtype=float)
            n0 = tmp.size
            target0 = math.sqrt(2.0 * n0)
            tmp = _normalize(tmp, target0)
            sc0 = _eval(tmp)
            best_seq = tmp.copy()
            best_score = sc0
            print(f"[init] global height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception as e:  # pragma: no cover
            print("[init] could not reuse height_sequence_1:", e)

    # ------------------------------------------------------------------
    # 1️⃣ Main optimisation loop – try several lengths and restarts
    # ------------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000]  # more lengths give more flexibility
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)       # we keep the ℓ¹‑norm fixed
        restarts = 5 if n <= 600 else 3

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise – uniform Dirichlet (slightly perturbed uniform)
            a = rng.random(n) + 0.1
            a = _normalize(a, target_sum)

            # if we already possess a good vector of the same length, start from it
            if best_seq is not None and best_seq.size == n:
                a = best_seq.copy()

            # --------------------------------------------------------------
            # local search on this starting point
            # --------------------------------------------------------------
            max_iters = 500 if n <= 600 else 300
            for it in range(max_iters):
                if time.monotonic() >= deadline:
                    break

                conv = _conv_full(a)
                cur_max = float(conv.max())
                peak_idx = int(np.argmax(conv))

                # keep the globally best sequence seen so far
                cur_score = _eval(a)
                if cur_score < best_score - 1e-9:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[best] n={n} restart={rs} it={it} score={cur_score:.6f}")

                # stop early if we are already below the target bound
                if best_score < 1.503:
                    print("[early] target reached – exiting.")
                    return best_seq.tolist()

                # ----------------------------------------------------------
                # try a linear‑programming mass redistribution
                # ----------------------------------------------------------
                new_a = None
                if _HAS_CVX:
                    new_a = _lp_peak_move(a, peak_idx, step_frac=0.4)

                # if LP failed or did not improve the global maximum, fall back
                if new_a is None:
                    new_a = _greedy_peak_move(a, peak_idx, rng)

                # evaluate the new candidate only if its peak actually decreased
                new_conv = _conv_full(new_a)
                new_max = float(new_conv.max())
                if new_max < cur_max - 1e-12:
                    a = new_a
                else:
                    # occasional random jitter to escape shallow plateaus
                    if rng.random() < 0.02:
                        jitter = rng.normal(scale=0.1, size=n)
                        a_jit = a + jitter
                        a_jit = _normalize(np.clip(a_jit, 0.0, 1000.0), target_sum)
                        a = a_jit

            # end of one restart
    # ------------------------------------------------------------------
    # 2️⃣ Stochastic polishing (Gaussian perturbations)
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – uniform vector of length 400
        n = 400
        best_seq = np.full(n, 1.0, dtype=float)
        best_seq = _normalize(best_seq, math.sqrt(2.0 * n))
        best_score = _eval(best_seq)

    cur = best_seq.copy()
    n = cur.size
    step = 0.5
    while time.monotonic() < deadline:
        cand = cur + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, math.sqrt(2.0 * n))
        cand_score = _eval(cand)
        if cand_score < best_score:
            cur = cand
            best_seq = cand.copy()
            best_score = cand_score
            print(f"[polish] improved to {best_score:.6f}")
            if best_score < 1.503:
                print("[polish] target reached – exiting.")
                break
        step *= 0.9995
        if step < 1e-5:
            step = 0.5

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} (length {best_seq.size})")
    return [float(x) for x in best_seq]
