def project_to_simplex(v, target_sum):
    # ensures nonnegative and sum = target_sum
    n = len(v)
    # shift such that sum = target_sum after thresholding
    # algorithm typical: sort descending, compute cumulative sum.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v, target_sum):
    """Project v onto the simplex {x >= 0, sum x = target_sum}."""
    # Based on algorithm by Wang & Carreira-Perpinan (2013)
    # Shift so that sum(v) = target_sum after thresholding.
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]

        # all entries zero?
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

"""
Propose a low‑value step sequence.

The algorithm is deliberately different from the previous LP‑based approach.
It consists of three stages:

1️⃣  **Broad parametric sweep** – simple analytic families (U‑shaped
    exponentials, power‑laws, centre‑heavy shapes and random Dirichlet draws)
    are generated for several lengths.  Each candidate is normalised to
    ℓ¹‑norm = √(2·n) so that the evaluation score coincides with the
    maximal auto‑convolution entry.  The best candidate is retained.

2️⃣  **Projected sub‑gradient descent** – starting from the best candidates
    (and a few random perturbations) we minimise the maximal convolution
    value directly.  At each iteration we compute the full convolution with
    an FFT, identify the active index/indices attaining the current maximum,
    form the sub‑gradient (2·a_{k‑i}) and take a small step opposite to it.
    The iterate is projected back onto the simplex
    {x ≥ 0, Σx = √(2·n)}.  The step size decays slowly and occasional
    Gaussian jitter helps escape plateaus.

3️⃣  **Polishing hill‑climb** – a short Gaussian perturbation phase
    (implemented in ``_local_search``) refines the best vector found
    by the descent.

All stages respect the safety checks of ``evaluate_sequence`` (non‑negative,
clipped to [0,1000]) and keep the ℓ¹‑norm exactly at the prescribed value.
The routine stops before the time budget expires and returns the best
sequence discovered.
"""

import time

def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real inputs)."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Exact objective of evaluate_sequence for a normalised vector."""
    n = seq.shape[0]
    conv = _conv_fft(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:          # mirrors the safety check in evaluate_sequence
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


def project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    # Find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero after projection
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_shape(n: int, shape: str, param: float) -> np.ndarray:
    """Return a non‑negative vector of length n for the given analytic shape."""
    idx = np.arange(n, dtype=np.float64)
    d = np.minimum(idx, n - 1 - idx)          # distance to the nearest edge

    if shape == "exp_u":                      # high at edges, low in centre
        a = np.exp(-param * d)
    elif shape == "exp_center":               # high in centre, low at edges
        dc = np.abs(idx - (n - 1) / 2.0)
        a = np.exp(-param * dc)
    elif shape == "power_u":                  # (d+1)^-p, edges heavy
        a = (d + 1) ** (-param)
    elif shape == "power_center":              # centre‑heavy power law
        dc = np.abs(idx - (n - 1) / 2.0)
        a = (dc + 1) ** (-param)
    elif shape == "triangular":               # classic triangle (peak centre)
        a = np.minimum(idx + 1, n - idx)
    else:                                      # fallback uniform
        a = np.ones_like(idx)

    return a


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """Simple Gaussian hill‑climb (no external imports)."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)
    target = math.sqrt(2.0 * n)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target / cand.sum())        # re‑normalise to √(2n)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        # slowly shrink the perturbation size
        step *= 0.9995
        if step < 1e-7:
            step = step_initial
    return best, best_sc

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step sequence with a low evaluation score."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a possibly previously found construction.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.ndim == 1 and init.size > 0:
                n0 = init.size
                target0 = math.sqrt(2.0 * n0)
                init = init * (target0 / init.sum())
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception as e:               # pragma: no cover
            print("[init] failed:", e)

    # ------------------------------------------------------------------
    # 1️⃣  Broad parametric sweep on several lengths.
    # ------------------------------------------------------------------
    lengths = [2000, 1600, 1500, 1200, 1000, 800, 600, 500, 400, 300]
    shape_families = [
        ("exp_u", np.linspace(0.001, 0.6, 25)),
        ("exp_center", np.linspace(0.001, 0.6, 25)),
        ("power_u", np.linspace(0.2, 5.0, 25)),
        ("power_center", np.linspace(0.2, 5.0, 25)),
        ("triangular", [0.0]),                # dummy param, ignored
    ]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # deterministic families
        for shape, param_grid in shape_families:
            for p in param_grid:
                if time.monotonic() >= deadline:
                    break
                a = generate_shape(n, shape, float(p))
                a = a / a.sum() * target
                sc = _fast_score(a)
                if sc < best_score:
                    best_seq, best_score = a.copy(), sc
                    print(f"[sweep] shape={shape:12s} p={p:6.3f} n={n:4d} score={sc:.6f}")

        # a few random Dirichlet draws
        for _ in range(5):
            if time.monotonic() >= deadline:
                break
            a = rng.gamma(shape=1.0, scale=1.0, size=n) + 1e-8
            a = a / a.sum() * target
            sc = _fast_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[rand]  n={n:4d} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent from a handful of starts.
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety fallback – uniform of moderate size
        n_fallback = 300
        target = math.sqrt(2.0 * n_fallback)
        best_seq = np.full(n_fallback, target / n_fallback, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # build a small pool of starting points
    start_pool = [best_seq.copy()]
    # perturbations of the current best
    for _ in range(3):
        p = best_seq.copy()
        p += rng.normal(scale=0.02 * np.mean(best_seq), size=p.shape)
        p = np.clip(p, 0.0, 1000.0)
        p = p * (math.sqrt(2.0 * p.shape[0]) / p.sum())
        start_pool.append(p)

    # add a few fresh random vectors (different lengths)
    for n in lengths[:3]:                     # only the three largest lengths
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        for _ in range(2):
            a = rng.gamma(shape=1.0, scale=1.0, size=n) + 1e-8
            a = a / a.sum() * target
            start_pool.append(a)

    # descent parameters
    max_iters_per_run = 5000
    for idx, seq in enumerate(start_pool):
        if time.monotonic() >= deadline:
            break
        n = seq.shape[0]
        target = math.sqrt(2.0 * n)

        # step size roughly 5‑10 % of a typical entry
        step = max(1e-4, 0.08 * target / n)

        cur_best = seq.copy()
        cur_best_sc = _fast_score(cur_best)

        it = 0
        while time.monotonic() < deadline and it < max_iters_per_run:
            it += 1

            conv = _conv_fft(cur_best)
            max_val = conv.max()
            # active set (indices attaining the maximal value, with a tiny tolerance)
            tol = 1e-12
            active = np.where(conv >= max_val - tol)[0]

            # sub‑gradient of the active max‑convolution entry
            grad = np.zeros_like(cur_best)
            for k in active:
                i0 = max(0, k - (n - 1))
                i1 = min(k, n - 1)
                idx_range = np.arange(i0, i1 + 1)
                grad[i0:i1 + 1] += cur_best[k - idx_range]
            grad *= 2.0

            # gradient‑step (trying to *decrease* the maximal entry)
            new = cur_best - step * grad

            # occasional jitter to escape flat regions
            if rng.random() < 0.04:
                new += rng.normal(scale=step * 0.5, size=n)

            # project back onto the feasible simplex
            new = project_to_simplex(new, target)

            new_sc = _fast_score(new)

            # keep the best local vector
            if new_sc < cur_best_sc:
                cur_best, cur_best_sc = new, new_sc
                if new_sc < best_score:
                    best_seq, best_score = new.copy(), new_sc
                    print(f"[desc] start{idx} it={it:4d} n={n:4d} score={new_sc:.6f}")

            # simple acceptance rule (accept improving steps, otherwise keep old)
            if new_sc <= cur_best_sc:
                cur_best = new
                cur_best_sc = new_sc

            # slowly decay the learning rate
            step *= 0.9997
            if step < 1e-6:
                step = 1e-6

    # ------------------------------------------------------------------
    # 3️⃣  Final short Gaussian hill‑climb for polishing.
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined_seq, refined_sc = _local_search(best_seq.copy(), deadline,
                                                step_initial=0.2)
        if refined_sc < best_score:
            best_seq, best_score = refined_seq, refined_sc
            print(f"[final] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence as plain Python list.
    # ------------------------------------------------------------------
    if best_seq is None:
        # (Should never happen, but keep a deterministic fallback.)
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    return [float(x) for x in best_seq]
