"""
Optimisation of a non‑negative step sequence for the target function
`evaluate_sequence`.  The score equals the peak value of the linear
auto‑convolution (the L¹‑sum is irrelevant because the objective is
scale‑invariant).  The algorithm consists of three stages:

1️⃣ Warm‑start from a previously discovered construction (`height_sequence_1`).

2️⃣ Fast projected Adam optimisation on the ℓ¹‑simplex.  The loss is the
   current maximum auto‑convolution entry; the sub‑gradient is obtained
   by summing the contributions of all active (maximising) shifts.
   Adam’s adaptive learning rates and a projection onto the simplex keep
   the walk stable and accelerate convergence.

3️⃣ A temperature‑controlled simulated‑annealing hill‑climber that
   explores the neighbourhood of the best candidate while preserving the
   ℓ¹‑norm exactly (mass is moved from one index to another).  Up‑hill
   moves are occasionally accepted, allowing the search to escape shallow
   local minima.

The routine respects the global time budget (default 1000 s) and returns
the best list of floats discovered before the deadline.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0,  Σ x_i = s}.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Edge case – return a uniform distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(f: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` (the clipping performed inside that
    function is irrelevant for the values we generate).
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where the convolution attains (almost) the maximal value."""
    eps = max_val * 1e-12 + 1e-14
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the worst auto‑convolution entry w.r.t. `f`.
    The gradient is summed over all active offsets.
    """
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        vals = f[j]
        # each term contributes f[i] * f[j]; derivative w.r.t. f[i] is f[j]
        grad[i] += vals
        # and symmetrically w.r.t. f[j]
        grad[j] += f[i]
        # The diagonal i == j has been added twice; correct it:
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= f[i[mask]]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _adam_descent(f0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation that minimises the peak auto‑convolution.
    The iterate is always projected onto the simplex Σ a_i = target_sum,
    a_i ≥ 0.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)

    m = np.zeros_like(f)          # first moment
    v = np.zeros_like(f)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                      # base learning rate
    it = 0
    no_improve = 0

    while it < max_iters and time.monotonic() < deadline:
        it += 1

        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)

        # sub‑gradient of the *maximum* convolution entry
        grad_max = _gradient_submax(f, active)

        # full gradient of the objective 2·n·max/Σ²
        sum_f = np.sum(f)
        grad_obj = (2.0 * len(f) / (sum_f * sum_f)) * grad_max

        # Adam update (gradient descent direction)
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = _project_onto_simplex(f_candidate, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cur_score = _score(f)
        cand_score = _score(f_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_f = f_candidate.copy()
            no_improve = 0
            print(f"[Adam] it={it:4d} n={len(f)} score={cand_score:.6f}")
        else:
            no_improve += 1

        # accept the candidate if it improves the raw objective,
        # otherwise keep the old point (Adam already adapts the step)
        if cand_score < cur_score:
            f = f_candidate
        else:
            # optional tiny back‑off to avoid stagnation
            if no_improve > 200:
                lr *= 0.7
                no_improve = 0

        if no_improve > 800:               # early stopping
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Simulated annealing polishing
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Simple temperature‑controlled hill‑climber.
    One move consists of transferring a small amount of mass from one
    coordinate to another (exact L¹‑norm is preserved).
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.05 * target_sum / n          # typical move size

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # amount to move – drawn from an exponential distribution for
        # a broad range of scales, but capped so we never make a coordinate
        # negative.
        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # maintain the hard [0,1000] bounds (projection is cheap)
        cand = np.clip(cand, 0.0, 1000.0)

        # Because we performed an exact mass transfer the L¹‑norm is
        # already preserved; nevertheless a small numerical drift may
        # appear – re‑project onto the simplex once in a while.
        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)

        # accept if better, otherwise maybe accept uphill
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            # metropolis criterion
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # temperature schedule
        T *= decay
        if T < 1e-8:
            T = init_temp

        # gradually shrink the typical step size
        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.
    The routine respects the given time budget and returns a plain
    Python list suitable for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # a small safety margin – we want to finish *before* the budget
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start from a previously found construction, if any.
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64).copy()
        seq = np.clip(seq, 0.0, 1000.0)
        n = len(seq)
        target_sum = math.sqrt(2.0 * n)
        if np.sum(seq) == 0.0:
            seq = np.full(n, target_sum / n)
        else:
            seq = seq * (target_sum / np.sum(seq))
        s = _score(seq)
        best_seq = seq.copy()
        best_score = s
        print(f"[init] warm‑start from height_sequence_1, n={n}, score={s:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Main optimisation – Adam on several lengths and restarts.
    # --------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000, 1200, 1500, 1800]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of random restarts for the current length
        restarts = 12 if n <= 1000 else 8
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a starting point
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
                # add a tiny jitter to escape possible plateaus
                f0 += rng.normal(scale=0.001 * target_sum, size=n)
                f0 = np.clip(f0, 0.0, None)
                f0 = _project_onto_simplex(f0, target_sum)
            else:
                f0 = rng.random(n)
                f0 = _project_onto_simplex(f0, target_sum)

            # allocate a few seconds for each descent run
            sub_deadline = min(deadline, time.monotonic() + 7.0)
            f_opt, score_opt = _adam_descent(f0,
                                             target_sum,
                                             sub_deadline,
                                             max_iters=4000)

            if score_opt < best_score:
                best_score = score_opt
                best_seq = f_opt.copy()
                print(f"[update] n={n} restart={r} score={score_opt:.6f}")

            # early exit if we have already beaten the target
            if best_score < 1.503:
                print("[early] target achieved during Adam phase")
                return [float(x) for x in best_seq]

    # --------------------------------------------------------------
    # 3️⃣ Final polishing with simulated annealing (if time permits).
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _simulated_annealing(best_seq, deadline,
                                                     init_temp=0.2,
                                                     decay=0.9997)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final] annealed score={refined_score:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never be needed) – a simple uniform construction.
    # --------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] using uniform construction")

    # Clip to the range required by `evaluate_sequence` (extra safety)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
