"""
Propose a step sequence with a low evaluation score.

Algorithm
---------
1. **Warm‑start by a cheap parametric sweep** – try power‑law shapes
   a_i ∝ (i+1)^p (and its reverse) for a single length (default 1000).
   The best of these cheap candidates is taken as the initial vector.

2. **Simulated annealing on the exact objective** – repeatedly move a
   small amount of mass from one index to another (keeping the ℓ¹‑norm
   constant).  The new vector is accepted if it improves the score or
   with a temperature‑dependent probability.  The temperature is
   exponentially cooled; after a large number of idle iterations the
   current state is re‑initialised around the best solution seen so far.

3. **Return the best sequence** found before the time budget expires.
   The score is evaluated with the same formula as ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper functions (fast, no security checks)
# --------------------------------------------------------------

def _max_conv(seq: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution ``seq * seq``
    computed with an FFT (O(n log n)).  The sequence is assumed to be
    already clipped to the allowed range [0, 1000] and to have a
    positive ℓ¹‑norm.
    """
    n = seq.shape[0]
    # length needed for linear convolution (next power of two for speed)
    N = 1 << ((2 * n - 1).bit_length())
    # real‑FFT is enough because the input is real
    ft = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(ft * ft, n=N)[: 2 * n - 1]
    return float(conv.max())


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` (no type checks, no prints).
    Returns ``np.inf`` when the ℓ¹‑norm is smaller than 0.01.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    total = float(seq.sum())
    if total < 0.01:
        return float("inf")
    n = seq.shape[0]
    max_c = _max_conv(seq)
    return float(2.0 * n * max_c / (total * total))

# Main search routine
# --------------------------------------------------------------

                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the global time budget and returns a plain list
    of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # safety margin

    # 0. Choose a working length (larger n gives a lower theoretical bound)
    # --------------------------------------------------------------
    # If a previous best exists, reuse its length, otherwise try 1000.
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
        if n < 100:
            n = 100
    else:
        n = 1000

    # --------------------------------------------------------------
    # 1. Simple parametric sweep to obtain a decent starting vector
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # try a few power‑law exponents (both forward and reversed)
    exponents = np.linspace(-3.0, 0.0, 31)  # from steeply decreasing to flat
    for p in exponents:
        # raw shape (i+1)^p, i=0..n-1
        base = (np.arange(1, n + 1, dtype=np.float64) ** p)
        for rev in (False, True):
            seq = base[::-1] if rev else base
            # normalise only to make the scale well behaved (score is invariant)
            seq = seq / seq.sum()
            # ensure we stay in the allowed range
            seq = np.clip(seq, 0.0, 1000.0)
            sc = _fast_score(seq)

            best_seq = seq0.copy()
            print(f"[global] reused old sequence score={sc0:.6f}")

    # --------------------------------------------------------------
    # 3. If still nothing (unlikely), fall back to a random vector
    # --------------------------------------------------------------
    if best_seq is None:
        seq = rng.random(n) + 0.1
        seq = seq / seq.sum()
        best_seq = seq.copy()
        best_score = _fast_score(seq)
        print(f"[fallback] random start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing refinement on the true objective
    # --------------------------------------------------------------
    seq = best_seq.copy()
    total_sum = seq.sum()                     # constant during the SA moves
    n = seq.shape[0]

    cur_max = _max_conv(seq)
    cur_score = 2.0 * n * cur_max / (total_sum * total_sum)

    # temperature schedule
    T0 = max(cur_max * 0.5, 1e-6)              # initial temperature
    T = T0
    decay = 0.99994
    it = 0
    last_improve = 0
    max_idle = 250_000                         # restart after this many idle steps

    while time.monotonic() < deadline:
        # ------- propose a mass transfer between two positions -------
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # maximal transferable amount without violating [0,1000]
        max_delta = min(seq[i], 1000.0 - seq[j])
        if max_delta <= 0.0:
            continue

        # scale the proposal size with the current temperature
        frac = min(0.5, T / T0 + 1e-12)
        delta = rng.random() * max_delta * frac

        # apply the move
        seq[i] -= delta
        seq[j] += delta

        # evaluate new state
        new_max = _max_conv(seq)
        new_score = 2.0 * n * new_max / (total_sum * total_sum)

        # acceptance rule (Metropolis)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur_score = new_score
            cur_max = new_max
            if new_score < best_score:
                best_score = new_score
                best_seq = seq.copy()
                last_improve = it
                print(f"[SA] iter={it} best_score={best_score:.6f}")
                if best_score < 1.503:                     # target reached
                    return best_seq.tolist()
        else:
            # revert move
            seq[i] += delta
            seq[j] -= delta

        # temperature annealing
        T *= decay
        it += 1

        # occasional restart if we have been idle for too long
        if it - last_improve > max_idle:
            seq = best_seq.copy()
            # tiny Gaussian jitter to escape flat regions
            noise = rng.normal(scale=0.01, size=n)
            seq = np.clip(seq + noise, 0.0, 1000.0)
            # re‑normalise the ℓ¹‑norm (invariance, but keep numeric stability)
            seq = seq * (total_sum / seq.sum())
            cur_score = _fast_score(seq)
            cur_max = _max_conv(seq)
            T = T0
            last_improve = it

    # --------------------------------------------------------------
    # 5. Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f} length={len(best_seq)}")
    return [float(x) for x in best_seq]
