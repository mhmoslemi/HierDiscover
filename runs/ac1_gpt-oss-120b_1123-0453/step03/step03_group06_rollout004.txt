import time, math, numpy as np

def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[:2*n-1]

def _fast_score(seq):
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a*sum_a))

def _proj_simplex(v, s):
    # Projection onto simplex {x >=0, sum x = s}
    # algorithm from Duchi et al. (2008)
    n = v.shape[0]
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    # Find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _gradient_of_max(seq):
    n = seq.shape[0]
    conv = _conv_full_fft(seq)
    max_val = conv.max()
    # indices that are close to max
    eps = 1e-12 * max_val if max_val > 0 else 1e-12
    idxs = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(seq)
    # for each k in idxs accumulate gradient
    for k in idxs:
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        rev = np.zeros_like(seq)
        rev[mask] = seq[j[mask]]
        g = 2.0 * rev
        if k % 2 == 0:
            ic = k // 2
            if 0 <= ic < n:
                g[ic] = rev[ic]   # half weight
        grad += g
    grad /= len(idxs)
    return max_val, grad, conv

def _grad_descent(seq, target_sum, deadline, rng):
    n = seq.shape[0]
    cur = seq.copy()
    # ensure sum and non-neg
    cur = np.maximum(cur, 0.0)
    cur = _proj_simplex(cur, target_sum)
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score
    step = target_sum * 0.2   # initial step about 20% of average magnitude
    no_improve = 0
    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1
        max_val, grad, conv = _gradient_of_max(cur)
        # direction: move opposite to gradient
        cand = cur - step * grad
        cand = np.clip(cand, 0.0, None)
        cand = _proj_simplex(cand, target_sum)
        cand_score = _fast_score(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            step = step * 1.05  # a little larger step after success
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[GD] n={n:4d} iter={iter_count:6d} score={cand_score:.6f}")
        else:

            no_improve += 1
        # occasional random restart if stuck

Gradient‑descent optimizer for the max‑self‑convolution objective.

The score returned by ``evaluate_sequence`` (with the ℓ¹‑norm fixed to √(2·n))
equals the maximal entry of the full linear convolution f*f.
Hence we minimise  max_k (f*f)[k]  under the constraints

    f_i ≥ 0   and   Σ_i f_i = √(2·n).

The algorithm proceeds as follows:

1️⃣  **Initialisation** – for each considered length n we try a few

    gradient is taken.  After the step we project back onto the simplex
    (non‑negative entries, fixed ℓ¹‑norm).  A simple back‑tracking line
    search shrinks the step when the move does not improve the score.

3️⃣  **Random restarts** – if the descent stalls for many iterations we
    jitter the current best solution and continue.

4️⃣  **Final polishing** – a short Gaussian hill‑climb (similar to the
    reference code) cleans up the best candidate.

The search explores several lengths (up to 2000) – larger sequences
provide more flexibility and often yield a lower maximal convolution.
All helper functions are top‑level; the only public entry point is
``propose_candidate`` which returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution using a real FFT
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of ``seq`` with itself (full length)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Scoring – exact analogue of ``evaluate_sequence`` (faster)
# ----------------------------------------------------------------------


def _fast_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv)/‖seq‖₁²  (identical to evaluate_sequence)."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector ``v`` onto the probability simplex
    scaled to sum ``s`` (see Duchi et al., 2008).
    """
    if s < 0:
        raise ValueError("Target simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the inequality
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the worst convolution entry
# ----------------------------------------------------------------------


def _gradient_of_worst(seq: np.ndarray):
    """
    Returns (worst_value, gradient, full_convolution).
    The gradient is the sub‑gradient of the maximal convolution entry.
    """
    n = seq.shape[0]
    conv = _conv_full_fft(seq)
    max_val = conv.max()
    # indices that achieve (or are numerically very close to) the maximum
    eps = 1e-12 * max_val if max_val > 0 else 1e-12
    idxs = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(seq)
    i = np.arange(n)

    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        # rev[i] = seq[j] where defined, otherwise 0
        rev = np.zeros_like(seq)
        rev[mask] = seq[j[mask]]
        grad += 2.0 * rev                 # ∂(f*f)[k]/∂f_i = 2·f_{k‑i}

    grad /= len(idxs)                     # average if several ties
    return max_val, grad, conv


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (core optimiser)
# ----------------------------------------------------------------------


def _grad_descent(
    init_seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.
    Returns the best sequence seen and its score.
    """
    n = init_seq.shape[0]
    cur = _proj_simplex(np.maximum(init_seq, 0.0), target_sum)
    cur_score = _fast_score(cur)

    best = cur.copy()
    best_score = cur_score

    # initial step size – a few multiples of the average entry
    avg = target_sum / n
    step = avg * 5.0
    no_improve = 0
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        max_val, grad, _ = _gradient_of_worst(cur)

        # candidate move opposite to the gradient
        cand = cur - step * grad
        cand = np.clip(cand, 0.0, None)
        cand = _proj_simplex(cand, target_sum)

        cand_score = _fast_score(cand)

        if cand_score < cur_score:                     # successful move
            cur = cand
            cur_score = cand_score
            step = step * 1.05                         # gently increase step
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[GD] n={n:4d} it={iteration:6d} score={cand_score:.6f}"
                )
        else:                                          # reject – shrink step
            step *= 0.5
            no_improve += 1

        # random jitter when stuck for a while
        if no_improve >= 200:
            jitter = rng.normal(scale=step, size=n)
            cand = best + jitter
            cand = np.clip(cand, 0.0, None)
            cand = _proj_simplex(cand, target_sum)
            cand_score = _fast_score(cand)
            cur = cand
            cur_score = cand_score
            no_improve = 0

    return best, best_score


# ----------------------------------------------------------------------
# Simple Gaussian hill‑climb used for final polishing
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray, deadline: float, step_initial: float = 0.5):
    """Gaussian perturbation hill‑climb (keeps ℓ¹‑norm fixed)."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial
    target = best.sum()

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _proj_simplex(cand, target)       # preserve the sum
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The algorithm explores several lengths, runs a projected sub‑gradient
    descent for each, and finishes with a short stochastic polishing.
    """
    rng = np.random.default_rng(seed)

    # safety margin so that the caller can still read the result
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.3)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # Lengths to try – larger n gives more flexibility.
    lengths = [800, 1200, 1600, 2000]

    # If a global sequence exists, keep it for possible interpolation.
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, np.ndarray)):
        global_seq = np.asarray(global_seq, dtype=np.float64)
    else:
        global_seq = None

    # Roughly 70 % of the time budget for the core optimiser.
    remaining = deadline - time.monotonic()
    core_budget = remaining * 0.70
    per_len = core_budget / max(1, len(lengths))

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise a starting shape (exponential family, possibly interpolated)
        # ------------------------------------------------------------------
        if global_seq is not None and global_seq.shape[0] == n:
            init_seq = global_seq.copy()
        else:
            # try a few decay rates and keep the most promising
            best_init = None
            best_init_score = float("inf")
            idx = np.arange(n, dtype=np.float64)
            for lam in np.linspace(0.90, 0.999, 8):
                cand = (lam ** idx)
                cand = _proj_simplex(cand, target_sum)
                sc = _fast_score(cand)
                if sc < best_init_score:
                    best_init_score = sc
                    best_init = cand
            # add a tiny random perturbation to break symmetry
            init_seq = best_init + 0.005 * rng.random(n)
            init_seq = _proj_simplex(init_seq, target_sum)

        # ------------------------------------------------------------------
        # Run the projected sub‑gradient descent for this length
        # ------------------------------------------------------------------
        sub_deadline = time.monotonic() + per_len
        cand_seq, cand_score = _grad_descent(init_seq, target_sum, sub_deadline, rng)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[global] New overall best: score={best_score:.6f} (n={n})")

        # early exit if we have already passed the target
        if best_score < 1.503:
            print("[global] Target reached – exiting early.")
            break

    # ----------------------------------------------------------------------
    # Final stochastic polishing (uses the exact fast_score)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _local_search(best_seq, deadline, step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] Improved to {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = _proj_simplex(best_seq, math.sqrt(2.0 * n))

    # Return as a plain Python list of floats
    return [float(v) for v in best_seq]
