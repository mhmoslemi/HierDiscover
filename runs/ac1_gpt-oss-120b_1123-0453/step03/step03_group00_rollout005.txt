def _lp_improve(f, target_sum, deadline, max_outer_iters=15):
    n = len(f)
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)
    iter_count = 0
    while time.monotonic() < deadline and iter_count < max_outer_iters:
        M = np.max(np.convolve(best_f, best_f))
        # Solve LP
        g = _solve_lp_for_g(best_f, M)
        if g is None:
            break
        sum_g = np.sum(g)
        if sum_g <= target_sum * 1.001:
            # no significant increase; break
            break
        # Scale to target sum
        g = g * (target_sum / sum_g)
        # Mix
        t = 0.2
        improved = False
        while t > 1e-4:
            new_f = (1 - t) * best_f + t * g
            M_new = np.max(np.convolve(new_f, new_f))

                best_f = new_f
                improved = True
                break

        # keep best score
        cur_score = _compute_score_from_vector(best_f)
        if cur_score < best_score:
            best_score = cur_score
        else:
            # Not improving score; maybe continue iteration still?
            pass
        iter_count += 1
    return best_f, best_score

"""
LP‑guided refinement for the step‑function attack.

The algorithm works on a fixed ℓ¹‑norm (∑aᵢ = √(2 n)).
For a given sequence f we compute M = max(f*f).  Keeping M fixed,
the linear program

    maximise   Σ gᵢ
    subject to (f * g)ₖ ≤ M   for all k = 0 … 2 n‑2
               gᵢ ≥ 0

finds the “most massive” vector that still respects the same
convolution bound.  After normalising g to the desired ℓ¹‑norm we
mix it with the current f (small step size) – this never increases M.
Repeating the LP‑mix step monotonically reduces the maximal
auto‑convolution entry, i.e. the evaluation score.
A few stochastic hill‑climbs polish the final vector.

The whole search explores several lengths (300 … 1200) and a handful
of random restarts, always respecting the 1000 s time budget.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities (same as the previous baseline – kept for polishing)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x | x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact formula used by ``evaluate_sequence`` (no internal clipping)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    s = np.sum(f)
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (s * s))


def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """Gaussian hill‑climbing that keeps the ℓ¹‑norm fixed."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# LP‑based improvement
# ----------------------------------------------------------------------


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   subject to   (f * g)_k ≤ M  (for all k)
    and g_i ≥ 0.  Returns the optimal g or None if the LP failed.
    """
    n = len(f)
    # Build the (2n‑1) × n constraint matrix.
    # Column i is f placed starting at row i (zero‑padded on both sides).
    cols = []
    for i in range(n):
        col = np.concatenate([np.zeros(i), f, np.zeros(n - 1 - i)])
        cols.append(col)
    A = np.column_stack(cols)               # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises; we minimise -Σg_i to maximise Σg_i.
    c = -np.ones(n, dtype=np.float64)

    # Highs is fast and reliable for moderate‑size LPs.
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method="highs",
                  options={"presolve": True, "time_limit": 30})
    if not res.success:
        return None
    return res.x


def _lp_improve(f0: np.ndarray,
                target_sum: float,
                deadline: float,
                max_outer_iters: int = 12) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑mix steps.
    Each outer iteration:
        * compute M = max(f*f)
        * solve the LP for a more massive g under the same bound M
        * normalise g to the target ℓ¹‑norm
        * mix g into f with a step size that never increases M
    Returns the best vector seen and its score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)
    outer = 0

    while time.monotonic() < deadline and outer < max_outer_iters:
        outer += 1
        M = np.max(np.convolve(f, f))

        g = _solve_lp_for_g(f, M)
        if g is None:
            break

        sum_g = np.sum(g)
        # If the LP cannot increase the ℓ¹‑norm, we are at a fixed point.
        if sum_g <= target_sum * (1.0 + 1e-6):
            break

        # Scale g to the required ℓ¹‑norm.
        g = g * (target_sum / sum_g)

        # --- mixing ----------------------------------------------------
        t = 0.2                     # initial step size
        mixed = False
        while t > 1e-5:
            f_candidate = (1.0 - t) * f + t * g
            M_new = np.max(np.convolve(f_candidate, f_candidate))
            # Accept only if the max‑convolution does not increase.
            if M_new <= M * (1.0 + 1e-12):
                f = f_candidate
                mixed = True
                break
            t *= 0.5
        if not mixed:
            # No admissible step size → stop.
            break

        cur_score = _compute_score_from_vector(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            # early exit if we already beat the target
            if best_score < 1.503:
                return best_f, best_score

    return best_f, best_score

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.
    Uses an LP‑guided iterative refinement and a final stochastic polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)  # leave a tiny margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Optional warm‑start from the previously discovered construction.
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        f = np.asarray(h1, dtype=np.float64)
        n0 = len(f)
        target0 = math.sqrt(2.0 * n0)
        if np.sum(f) == 0:
            f = np.full(n0, target0 / n0)
        else:
            f = f * (target0 / np.sum(f))
        score0 = _compute_score_from_vector(f)
        best_seq = f.copy()
        best_score = score0
        print(f"[init] height_sequence_1 used, n={n0}, score={score0:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000, 1200]
    for n in lengths:

        target_sum = math.sqrt(2.0 * n)

            if time.monotonic() >= deadline:

            # ----------------------------------------------------------------

            # ----------------------------------------------------------------
