def _project_simplex(v, S):
    # ensure sum of nonnegative entries equals S.
    # Standard algorithm: sort descending.
    # Code:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v, s):
    # returns projection onto {x >= 0, sum x = s}
    # based on algorithm by Wang & Carreira-Perpinan
    n = v.shape[0]
    u = np.sort(v)[::-1]

        return np.maximum(v, 0)

    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Projected‑gradient descent on a smooth approximation of the
max‑auto‑convolution objective.

The evaluation function (after normalising the ℓ¹‑norm to √(2·n))
is exactly the maximal entry of the auto‑convolution.
We minimise this quantity by:

1.  Starting from a smooth “semicircle” shape (or from the
    previously discovered `height_sequence_1`) and scaling it to the
    required ℓ¹‑norm.
2.  Repeatedly smoothing the max with a log‑sum‑exp (temperature T)
    and taking a gradient step of the smoothed objective.
    The gradient of the smoothed max is a weighted correlation of the
    current sequence with the weights `w = exp((conv‑max)/T)`.
3.  After each step we project the vector onto the simplex
    `{x≥0, Σx = √(2·n)}` – this keeps the sum fixed and respects the
    non‑negativity constraint.
4.  The temperature and step‑size are annealed linearly; the whole
    routine is run for several sequence lengths (500 → 2500) while
    respecting the global time budget.
5.  A final short Gaussian‑perturbation hill‑climb refines the best
    candidate.

The algorithm is completely deterministic apart from the supplied
seed and fits comfortably into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT)
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``seq`` with itself (real FFT)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Exact analogue of ``evaluate_sequence`` but using the fast FFT."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simplex projection (onto x≥0, Σx = S)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Target simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) of the auto‑convolution
# ----------------------------------------------------------------------


def _grad_corr(seq: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(seq*seq)_k w.r.t. seq.
    It equals 2·correlation(seq, w) and is computed via FFT.
    """
    n = seq.shape[0]
    rev = seq[::-1]                     # reversed sequence
    size = 1 << ((rev.shape[0] + w.shape[0] - 1).bit_length())
    rev_fft = np.fft.rfft(rev, n=size)
    w_fft = np.fft.rfft(w, n=size)
    conv_rev = np.fft.irfft(rev_fft * w_fft, n=size)
    # correlation at lag j is conv_rev[n‑1+j]; we need j=0…n‑1
    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max gradient descent with simplex projection
# ----------------------------------------------------------------------


def _gradient_descent(start_seq: np.ndarray,
                     target_sum: float,
                     time_budget: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed max‑convolution objective.
    Returns the best sequence found within the allotted time.
    """
    seq = start_seq.copy()
    n = seq.shape[0]

    cur_score = _fast_score(seq)
    best_seq = seq.copy()
    best_score = cur_score

    start = time.monotonic()
    deadline = start + time_budget

    # initial temperature based on the spread of the convolution
    conv = _conv_full_fft(seq)
    T0 = max(1e-8, 0.5 * (conv.max() - conv.min()))
    base_step_factor = 0.6  # relative to the target ℓ¹‑norm

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = elapsed / time_budget

        # ---- temperature schedule (linear cooling) ----
        T = T0 * (1.0 - frac) + 1e-8

        # ---- smooth‑max weights ----
        conv = _conv_full_fft(seq)
        mx = conv.max()
        w = np.exp((conv - mx) / T)            # smooth max
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum                         # normalise for stability

        # ---- gradient ----
        grad = _grad_corr(seq, w)

        # ---- step size (scaled to keep updates comparable) ----
        gnorm = np.linalg.norm(grad)
        if gnorm == 0:
            break
        step_factor = base_step_factor * (1.0 - frac) + 1e-3
        step = step_factor * target_sum / gnorm

        cand = seq - step * grad
        cand = _project_simplex(cand, target_sum)

        cand_score = _fast_score(cand)

        # store best ever
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[GD] n={n:4d} it={int(elapsed*1000):7d} "
                  f"score={cand_score:.6f} T={T:.5e}")

        # accept if better, otherwise Metropolis acceptance
        if cand_score < cur_score:
            seq, cur_score = cand, cand_score
        else:
            # occasional uphill moves while temperature is still high
            if rng.random() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
                seq, cur_score = cand, cand_score

        # tiny random jitter to avoid stagnation
        if rng.random() < 0.01:
            jitter = rng.normal(scale=0.005 * target_sum, size=n)
            seq = np.clip(seq + jitter, 0.0, 1000.0)
            seq = _project_simplex(seq, target_sum)
            cur_score = _fast_score(seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final stochastic polishing (Gaussian hill‑climb)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing with Gaussian perturbations, always re‑projected
    onto the simplex (sum is the original ℓ¹‑norm).
    """
    rng = np.random.default_rng()
    target_sum = float(start_seq.sum())
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        step *= 0.9996
        if step < 1e-8:
            step = step_initial
    return best, best_score

# Interpolation helper – enlarge a known shape to a longer length

def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return np.clip(new_seq, 0.0, 1000.0)

        else:
            # smooth “semicircle’’ shape as a generic good starting point
            x = np.linspace(-1.0, 1.0, n)
            init_seq = np.sqrt(np.clip(1.0 - x ** 2, 0.0, None))
            # add a tiny random jitter to break symmetry
            init_seq += rng.random(n) * 1e-3

        # enforce the ℓ¹‑norm (simplex) constraint
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        cur = init_seq.sum()
        if cur > 0:
            init_seq = init_seq * (target_sum / cur)

        # ------------------- gradient descent -------------------
        gd_deadline = time.monotonic() + time_per_len
        cand_seq, cand_score = _gradient_descent(init_seq,
                                                 target_sum,
                                                 gd_deadline - time.monotonic(),
                                                 rng)
        best_by_len[n] = cand_seq.copy()

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[global] new overall best: score={best_score:.6f} (n={n})")

        # early exit if the target is already beaten
        if best_score < 1.503:
            print("[global] target reached – exiting early.")
            break

    # ------------------- final polishing -------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined_seq, refined_score = _local_search(best_seq, deadline,
                                                   step_initial=0.5)
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"[final] improved to {best_score:.6f}")

    # safety fallback (should never happen)
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = best_seq * (math.sqrt(2.0 * n) / best_seq.sum())

    # return a plain Python list
    return [float(v) for v in best_seq]
