"""
Hill‑climbing / simulated‑annealing search for a step sequence
that minimises   2·n·max(conv(a,a)) / (sum(a))².

The algorithm works on a single length (default 500, but larger
lengths are also tried).  It repeatedly moves a small amount of
mass from an index that contributes heavily to the current maximal
convolution entry towards a random other index.  The move is kept
if it reduces the objective; otherwise it may be accepted with a
temperature‑dependent probability (annealing).  When progress stalls
a tiny Gaussian perturbation is applied to escape plateaus.  The best
sequence seen throughout the whole budget is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Compute the evaluation function for a given sequence.
    This is exactly the same as ``evaluate_sequence`` but without the
    safety checks (the sequence is already non‑negative and well‑scaled).
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    M = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * M / (s * s))


def _dirichlet_start(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex with sum = target_sum."""
    return rng.dirichlet(np.ones(n)) * target_sum


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑value step sequence.

    The routine performs a long hill‑climbing/annealing run on length 500
    (the length that gave the best known score) and afterwards tries a few
    larger lengths with a short budget.  The best sequence found before the
    time budget expires is returned as a plain Python list.
    """
    start_time = time.time()
    total_budget = max(0.0, float(budget_s))
    # keep a tiny safety margin
    deadline = start_time + total_budget - 0.05

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate and possibly update the global best.
    def _maybe_update(seq: np.ndarray, score: float, tag: str = ""):
        nonlocal best_seq, best_score
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            print(f"[global {tag}] new best score = {best_score:.6f} (len={len(seq)})")

    # ------------------------------------------------------------------
    # Primary search: length 500 (most time is spent here).
    primary_n = 500
    target = math.sqrt(2.0 * primary_n)

    # Try to start from a previously discovered construction, if present.
    seed_seq = globals().get("height_sequence_1")
    if isinstance(seed_seq, (list, tuple, np.ndarray)):
        seed_arr = np.asarray(seed_seq, dtype=np.float64)
        if seed_arr.shape[0] == primary_n:
            a = seed_arr.copy()
            a = np.clip(a, 0.0, None)
            a = a * (target / a.sum())
            print("[init] using supplied global sequence (n=500)")
        else:
            a = _dirichlet_start(primary_n, target, rng)
    else:
        a = _dirichlet_start(primary_n, target, rng)

    curr_score = _objective(a)
    _maybe_update(a, curr_score, "init")

    # main hill‑climbing loop
    no_progress = 0
    iter_cnt = 0
    while time.time() < deadline:
        # ------------------- 1. compute current convolution and its peak ------------
        conv = np.convolve(a, a)
        M = conv.max()
        k_max = int(conv.argmax())               # index where the peak occurs

        # ------------------- 2. contributions to the peak ----------------------
        # contribution of a_i to conv[k_max] is a_i * a_{k_max-i} (if the index exists)
        idx = k_max - np.arange(primary_n)
        mask = (idx >= 0) & (idx < primary_n)
        contrib = np.zeros(primary_n, dtype=np.float64)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick a source index i proportionally to its contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(primary_n, p=probs)
        else:
            i = rng.integers(primary_n)

        # pick a destination index j (different from i)
        j = rng.integers(primary_n - 1)
        if j >= i:
            j += 1

        # ------------------- 3. propose a small mass transfer --------------------
        max_eps = a[i]                     # cannot move more than we have at i
        if max_eps <= 0:
            # nothing to move – try another iteration
            continue

        elapsed = time.time() - start_time
        frac = elapsed / total_budget
        # step factor slowly decays from ~0.20 to ~0.005
        step_factor = 0.20 * (1.0 - frac) + 0.005
        eps = max_eps * rng.random() * step_factor

        b = a.copy()
        b[i] -= eps
        b[j] += eps
        # re‑normalise to keep the prescribed sum (scaling does not affect the objective)
        b = b * (target / b.sum())

        # ------------------- 4. evaluate the proposal ---------------------------
        new_score = _objective(b)

        # ------------------- 5. accept / reject ---------------------------------
        improved = False
        if new_score < curr_score - 1e-12:
            improved = True
        else:
            # simulated‑annealing acceptance
            # temperature also decays linearly to a tiny value
            temp = 0.001 * (1.0 - frac) + 1e-8
            if rng.random() < math.exp(-(new_score - curr_score) / temp):
                improved = True

        if improved:
            a = b
            curr_score = new_score
            no_progress = 0
            if new_score < best_score - 1e-12:
                _maybe_update(a, new_score, f"it{iter_cnt}")
        else:
            no_progress += 1

        # ------------------- 6. occasional perturbation -------------------------
        if no_progress >= 400:
            # small Gaussian shake to escape a plateau
            sigma = 0.01 * target / primary_n
            a = a + rng.normal(scale=sigma, size=primary_n)
            a = np.clip(a, 0.0, None)
            a = a * (target / a.sum())
            curr_score = _objective(a)
            no_progress = 0
            if curr_score < best_score - 1e-12:
                _maybe_update(a, curr_score, f"pert{iter_cnt}")

        iter_cnt += 1

    print(f"[summary] primary length {primary_n} completed – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Secondary search on a few larger lengths (short runs, just in case)
    extra_lengths = [600, 800, 1000, 1200, 1500, 2000]
    for n in extra_lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        a = _dirichlet_start(n, target, rng)

        curr_score = _objective(a)
        _maybe_update(a, curr_score, f"extra_init_n{n}")

        # very short inner loop (a few thousand iterations)
        inner_iters = 3000
        no_progress = 0
        for it in range(inner_iters):
            if time.time() >= deadline:
                break

            conv = np.convolve(a, a)
            k_max = int(conv.argmax())

            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib = np.zeros(n, dtype=np.float64)
            contrib[mask] = a[mask] * a[idx[mask]]
            total_contrib = contrib.sum()

            if total_contrib > 0:
                probs = contrib / total_contrib
                i = rng.choice(n, p=probs)
            else:
                i = rng.integers(n)

            j = rng.integers(n - 1)
            if j >= i:
                j += 1

            max_eps = a[i]
            if max_eps <= 0:
                continue

            # a faster decaying step for the short runs
            step_factor = 0.15 * (1 - it / inner_iters) + 0.002
            eps = max_eps * rng.random() * step_factor

            b = a.copy()
            b[i] -= eps
            b[j] += eps
            b = b * (target / b.sum())

            new_score = _objective(b)
            if new_score < curr_score - 1e-12:
                a = b
                curr_score = new_score
                no_progress = 0
                if new_score < best_score - 1e-12:
                    _maybe_update(a, new_score, f"extra_n{n}_it{it}")
            else:
                no_progress += 1
                if no_progress >= 300:
                    sigma = 0.005 * target / n
                    a = a + rng.normal(scale=sigma, size=n)
                    a = np.clip(a, 0.0, None)
                    a = a * (target / a.sum())
                    curr_score = _objective(a)
                    no_progress = 0
                    if curr_score < best_score - 1e-12:
                        _maybe_update(a, curr_score, f"extra_n{n}_pert{it}")

        print(f"[summary] extra length {n} best local score = {curr_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe construction.
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _objective(best_seq)
        print("[fallback] using uniform sequence")

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    # return a plain Python list as required by the harness
    return [float(v) for v in best_seq]
