"""
Gradient‑descent + simplex projection + final hill‑climb search.

* Scale‑invariant objective → enforce Σ a = √(2 n) so that the official
  score is simply max(a * a).

* The nonsmooth “max” is replaced by a smooth soft‑max
  μ·log∑exp(conv/μ).  Its gradient can be written in closed form using a
  convolution, which we evaluate with `np.convolve`.  We run a projected
  Adam optimiser on the simplex.

* After a short gradient phase we polish the best vector with the
  transfer‑mass hill‑climber from the reference code and a tiny Gaussian
  local‑search.  The routine respects the 1000 s budget and stops early
  once a score < 1.503 is found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official score (the evaluator works with a plain Python list)."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        { x >= 0 , sum(x) = target_sum }.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """In‑place move of `delta` from index i to j, updating the full convolution."""
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    delta = min(delta, a[i])

    # linear part  – 2·δ·a shifted
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part (δ at i, –δ at j)
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta

    a[i] -= delta
    a[j] += delta


def _improve_by_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that repeatedly moves a tiny amount of mass from one index
    to another while keeping the ℓ¹‑norm fixed.
    Returns the best sequence encountered and its (official) score.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)
    sum_a = a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

    rng = np.random.default_rng()
    step = step_init
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        _transfer_mass(a, conv, i, j, delta)
        cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert
            _transfer_mass(a, conv, j, i, delta)

        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Gentle Gaussian perturbation (exact same idea as the reference code).
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score


def _gradient_descent(start_seq: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex with a smooth approximation
    of the max‑convolution.  The function returns the best vector seen
    (according to the official score) and its score.
    """
    a = _project_onto_simplex(start_seq, target_sum)
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.3

    # smoothing parameter μ – start relatively large and shrink
    conv = np.convolve(a, a)
    mu = max(conv.max() - conv.min(), 0.5)

    best_a = a.copy()
    best_score = np.max(conv)          # because sum = √(2n)

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        cur_max = np.max(conv)

        # keep the best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights
        shift = (conv - cur_max) / mu          # ≤ 0
        # avoid under‑/overflow
        exp_vals = np.exp(np.clip(shift, -700, 0))
        w = exp_vals / exp_vals.sum()          # length = 2n‑1

        # gradient of the smooth‑max
        w_rev = w[::-1]
        rev = np.convolve(a, w_rev)            # length = 3n‑2
        idx = (len(w) - 1) - np.arange(n)      # grab the needed slice
        grad = 2.0 * rev[idx]                  # ∂/∂a of μ·log∑exp

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a = _project_onto_simplex(a_new, target_sum)

        # anneal μ and the learning‑rate very gently
        mu = max(mu * 0.9995, 0.02)
        lr = max(lr * 0.9999, 0.02)

        it += 1

    # return the best one we saw (official score = max(conv) because of scaling)
    return best_a, best_score

    """

    best_score = float("inf")
