"""
Propose a low‑score step‑function sequence.

The algorithm consists of three stages:

1️⃣  **Diversified restarts** – for several lengths (200 … 1000) we
    initialise a random vector on the simplex (or the provided
    `height_sequence_1`).  The ℓ¹‑norm is fixed to √(2 n) because the
    evaluation score is scale‑invariant.

2️⃣  **Focused stochastic hill‑climbing** – we keep the exact
    convolution `conv = a*a` and repeatedly move a small amount of mass
    from a “source’’ index to a “destination’’ index.

    * With probability 0.8 we pick the current peak offset `k_max`
      (`conv[k_max] = max(conv)`) and move mass from an index whose
      partner `a[k_max‑i]` is large (large contribution to the peak) to
      an index whose partner is tiny.  This directly attacks the active
      constraint that determines the score.

    * With probability 0.2 we perform a completely random transfer.
      Both moves are accepted if they lower the peak value; otherwise
      they may be accepted with a small probability (simulated annealing)
      to escape local minima.

    The convolution is updated *incrementally* in O(n) using the
    `_transfer_mass` routine, so millions of moves are affordable.

3️⃣  **Gaussian polishing** – a short stochastic local search that adds
    Gaussian noise to the best vector found and retains any improvement
    (the official `evaluate_sequence` is called only here).

The whole search stops after the allocated budget (default 1000 s) or
as soon as a score below the target 1.503 is discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index `i` to index `j` (i != j, delta > 0,
    delta <= a[i]).  Updates `a` and the full self‑convolution `conv`
    in‑place (O(n) work).

    The convolution `conv` has length `2*n-1` where `n = len(a)`.
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part: 2 * delta * a (added to shifted copies)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part: convolution of the delta‑vector with itself
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta          # (-δ)^2
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta    # (-δ)*(+δ) + (+δ)*(-δ)
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta          # (+δ)^2

    # update the underlying vector
    a[i] -= delta
    a[j] += delta


def _hill_climb(a: np.ndarray,
                target_sum: float,
                deadline: float,
                max_iters: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Focused stochastic hill‑climbing that tries to minimise the peak
    of the self‑convolution while preserving the ℓ¹‑norm.

    Returns the best vector encountered and its peak convolution value.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)               # exact convolution, length 2*n-1
    best_a = a.copy()
    best_peak = conv.max()
    rng = np.random.default_rng()
    eps = 1e-12

    # annealing schedule
    temperature = 0.1
    min_temp = 1e-6
    cooling = 0.9999

    # typical transfer fraction (adjusts the size of the move)
    step_factor = 0.5

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        old_peak = conv.max()
        k_max = int(conv.argmax())

        # ----- decide move type -----
        if rng.random() < 0.8:
            # -------- targeted move ----------
            idx = np.arange(n)
            partner = k_max - idx
            mask = (partner >= 0) & (partner < n)
            # g[i] = a[partner[i]]  (contribution of index i to the peak)
            g = np.zeros(n)
            g[mask] = a[partner[mask]]

            # viable sources: positive mass and non‑zero contribution
            src_mask = (a > eps) & (g > eps)
            if np.any(src_mask):
                src_candidates = np.where(src_mask)[0]
                scores = g[src_candidates] * a[src_candidates]
                src = int(src_candidates[int(np.argmax(scores))])
            else:
                # fall back to any positive entry
                src_candidates = np.where(a > eps)[0]
                if src_candidates.size == 0:
                    break
                src = int(rng.choice(src_candidates))

            # destination – index with the smallest contribution to the peak
            dst = int(np.argmin(g))
            if dst == src:
                # pick any other index
                others = np.where(np.arange(n) != src)[0]
                dst = int(rng.choice(others))
        else:
            # -------- random move ----------
            src_candidates = np.where(a > eps)[0]
            if src_candidates.size == 0:
                break
            src = int(rng.choice(src_candidates))
            dst = int(rng.integers(0, n - 1))
            if dst >= src:
                dst += 1

        # amount of mass to shift
        delta = a[src] * step_factor * rng.random()
        if delta <= 0.0:
            it += 1
            continue

        # apply the transfer
        _transfer_mass(a, conv, src, dst, delta)
        new_peak = conv.max()
        diff = new_peak - old_peak

        # acceptance test (deterministic improvement / simulated annealing)
        accept = False
        if diff <= 0.0:
            accept = True
        else:
            # probability to accept a worsening move
            prob = math.exp(-diff / max(temperature, min_temp))
            if rng.random() < prob:
                accept = True

        if accept:
            # keep the new state; update the best record if we improved globally
            if new_peak < best_peak - 1e-12:
                best_peak = new_peak
                best_a = a.copy()
        else:
            # revert the move
            _transfer_mass(a, conv, dst, src, delta)

        # anneal temperature slowly
        temperature = max(min_temp, temperature * cooling)
        it += 1

    return best_a, best_peak


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Gentle Gaussian perturbation based polishing.
    Uses the official `evaluate_sequence` to verify improvements.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score

            best_score = refined_score
            print(f"[final polishing] improved to {best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
