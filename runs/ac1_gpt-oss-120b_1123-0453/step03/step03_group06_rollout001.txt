def lp_improve(f, target_sum, t, deadline):
    # returns new f after blending.
    n = len(f)
    conv_f_f = _conv_full_fft(f)
    M = conv_f_f.max()
    # Build constraint matrix
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    # Solve LP
    c = -np.ones(n)  # minimize -sum(g)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 1.0})
    if not res.success:
        # fallback: return f unchanged
        return f, M
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return f, M
    # Scale g0 to target sum
    scale = target_sum / sum_g0
    g = g0 * scale
    # blend
    new_f = (1 - t) * f + t * g
    # Ensure positivity and clip
    new_f = np.clip(new_f, 0.0, 1000.0)
    # Re-normalize to target sum (optional, maybe enforce)
    s = new_f.sum()
    if s > 0:
        new_f = new_f * (target_sum / s)
    # Evaluate new M
    new_M = _conv_full_fft(new_f).max()
    return new_f, new_M

"""
Linear‑programming iterative refinement for low‑autocorrelation step sequences.

The algorithm follows the scheme described in the problem statement:
1. Start from a feasible non‑negative sequence (the provided
   `height_sequence_1` if available, otherwise a random one) and
   normalise its ℓ¹‑norm to √(2·n).
2. Repeatedly:
   * compute M = max(f * f) (full linear convolution);
   * solve a linear programme that maximises Σ g_j  subject to
        g_j ≥ 0   and   (f * g)(k) ≤ M  for all k.
     The LP variables are the new heights g.
   * rescale g to the required ℓ¹‑norm, blend it with the current
     sequence (convex combination with a gradually decreasing weight t)
     and renormalise.
   * keep the best sequence seen so far.
3. After the time for LP steps is exhausted, perform a short
   Gaussian‑perturbation hill‑climb to polish the result.

The LP step guarantees that the maximal autocorrelation never increases,
so the score monotonically decreases.  The final stochastic polishing
helps to escape the piecewise‑linear LP frontier.

The whole search respects the 1000 s budget; the returned list is ready
to be handed to `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution and scoring (exact analogue of evaluate_sequence)
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``seq`` with itself using a real FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Compute 2·n·max(conv)/‖seq‖₁² – identical to ``evaluate_sequence``."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Build the linear‑programming constraint matrix for a fixed f
# ----------------------------------------------------------------------


def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return a (2·n‑1) × n matrix A such that (A @ g)[k] = (f * g)[k].
    Small eps is added to avoid entirely zero columns.
    """
    n = f.shape[0]
    # Protect against exact zeros – they would make the LP unbounded.
    f_eps = np.maximum(f, 1e-12)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f_eps
    return A


# ----------------------------------------------------------------------
# One LP‑based improvement step
# ----------------------------------------------------------------------


def _lp_update(f: np.ndarray, target_sum: float, t: float) -> tuple[np.ndarray, float]:
    """
    Perform a single linear‑programming update:
      – maximise Σ g_j   subject to   g ≥ 0  and  (f * g) ≤ M,
    then blend the solution with the current f using weight t.
    Returns the new sequence and its current score.
    """
    n = f.shape[0]

    # current worst autocorrelation
    M = _conv_full_fft(f).max()

    # build A such that (A @ g) == f * g
    A = _build_constraint_matrix(f)

    # linear programme: maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Solve with HiGHS (fast and reliable). No time limit is set – the
    # problem size (≤ 1000 variables) is tiny.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # If the LP fails for any reason we simply keep the old sequence.
        return f, _fast_score(f)

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0.0:
        return f, _fast_score(f)

    # Scale to the required ℓ¹‑norm (the objective is scale‑invariant).
    g = g0 * (target_sum / sum_g0)

    # Convex blend:  f_new = (1‑t)·f + t·g
    f_new = (1.0 - t) * f + t * g
    f_new = np.clip(f_new, 0.0, 1000.0)

    # Renormalise to keep the sum exactly target_sum (helps numerical stability)
    s = f_new.sum()
    if s > 0.0:
        f_new = f_new * (target_sum / s)

    return f_new, _fast_score(f_new)


# ----------------------------------------------------------------------
# Simple Gaussian hill‑climbing polish
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing that adds zero‑mean Gaussian noise.
    The sum is *not* forced to stay constant – ``evaluate_sequence``
    will renormalise automatically, which is fine because the score is
    scaling‑invariant.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # exponential cooling of the step size
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The function runs an LP‑based iterative refinement for a fixed length
    (taken from ``height_sequence_1`` when available, otherwise 500) and
    finishes with a short stochastic polishing phase.
    """
    rng = np.random.default_rng(seed)
    # leave a tiny safety margin for the caller to read the result
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------------
    # Initialise – use the global reference if it exists
    # --------------------------------------------------------------
    known_seq = globals().get("height_sequence_1", None)
    if known_seq is not None:
        f = np.asarray(known_seq, dtype=np.float64).copy()
    else:
        n = 500
        f = rng.random(n) + 0.05          # tiny offset avoids exact zeros
    n = f.shape[0]

    target_sum = math.sqrt(2.0 * n)

    # Clip and normalise to the required ℓ¹‑norm
    f = np.clip(f, 0.0, 1000.0)
    cur = f.sum()
    if cur > 0.0:
        f = f * (target_sum / cur)

    best_seq = f.copy()
    best_score = _fast_score(best_seq)

    # --------------------------------------------------------------
    # LP‑driven refinement
    # --------------------------------------------------------------
    # Allocate ~80 % of the remaining budget to the LP loop, the rest
    # will be used for the final polish.
    polish_time = 5.0                     # seconds reserved for polishing
    lp_deadline = deadline - polish_time
    start_time = time.monotonic()
    total_lp_time = max(0.0, lp_deadline - start_time)

    # linear schedule for the blending weight t: from 0.5 down to 0.02
    def _t_for_fraction(frac: float) -> float:
        # frac ∈ [0,1] – fraction of total LP time elapsed
        return max(0.02, 0.5 * (1.0 - frac))

    iteration = 0
    while time.monotonic() < lp_deadline:
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_lp_time if total_lp_time > 0 else 1.0
        t = _t_for_fraction(frac)

        f, cur_score = _lp_update(f, target_sum, t)

        if cur_score < best_score:
            best_seq = f.copy()
            best_score = cur_score
            print(f"[LP] iter={iteration:4d} t={t:.4f} score={best_score:.6f}")

        # early exit if we have already passed the target
        if best_score < 1.503:
            print("[LP] target achieved – stopping early.")
            break

        iteration += 1

    # --------------------------------------------------------------
    # Final stochastic polishing
    # --------------------------------------------------------------
    if time.monotonic() < deadline:
        refine_deadline = time.monotonic() + min(polish_time, deadline - time.monotonic())
        best_seq, best_score = _local_search(best_seq, refine_deadline, step_initial=0.3)
        print(f"[final] polished score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        n = 500
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = best_seq * (math.sqrt(2.0 * n) / best_seq.sum())

    # Return a plain Python list
    return [float(v) for v in best_seq]
