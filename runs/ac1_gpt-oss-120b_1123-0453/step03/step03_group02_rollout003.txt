            sub_deadline = time.monotonic() + min(30.0, (deadline - time.monotonic()) * 0.5)
            f_opt, M_opt = _gradient_descent_momentum(f_start, target_sum, sub_deadline, max_iters=8000)

"""
Propose a low‑value step sequence.

The algorithm works in three phases:

1️⃣  **Initialization** – load the previously discovered sequence (if any) and
    pad / re‑project it to several target lengths.

2️⃣  **Projected momentum descent** – for each length run a sub‑gradient descent
    with a heavy‑ball momentum term on the ℓ¹‑simplex (fixed ℓ¹‑norm
    √(2·n)).  The sub‑gradient is taken at the worst convolution index.
    The step size is adapted on‑the‑fly and the iterate is always projected
    back onto the simplex.

3️⃣  **Simulated‑annealing polishing** – starting from the best vector found
    we perform an exponential‑cooling annealing with Gaussian perturbations.
    Candidate moves are projected onto the simplex and accepted either if they
    improve the score or with a temperature‑dependent probability.

All phases use only NumPy and standard library primitives, run under the
global time budget and return the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – evaluate a sequence (already provided in the environment)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Wrap the external evaluate_sequence function."""
    # evaluate_sequence expects a list of Python floats
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Helper 2 – projection onto the non‑negative simplex Σx = target
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = target}."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - target
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 3 – fast full convolution and its maximum (FFT based)
# ----------------------------------------------------------------------
def _fft_conv_max(f: np.ndarray):
    """
    Return (max_value, argmax_index) of the full auto‑convolution f*f.
    Uses an FFT of size power‑of‑two for speed.
    """
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Helper 4 – projected gradient descent with momentum
# ----------------------------------------------------------------------
def _gradient_descent_momentum(
    f_start: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 8000,
    eta0: float = 1.0,
    beta: float = 0.85,
    min_eta: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the ℓ¹‑simplex with heavy‑ball momentum.

    Returns the best vector seen (as a NumPy array) and its worst
    convolution value (i.e. max_k (f*f)_k) observed during the run.
    """
    n = f_start.shape[0]
    f = f_start.copy()
    best_f = f.copy()
    best_M, _ = _fft_conv_max(f)

    eta = eta0
    v = np.zeros_like(f)

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        M, k = _fft_conv_max(f)

        # keep the best ever seen
        if M < best_M:
            best_M = M
            best_f = f.copy()

        # sub‑gradient at the worst index k
        g = np.zeros_like(f)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        if low <= high:
            idxs = np.arange(low, high + 1)
            g[idxs] = 2.0 * f[k - idxs]

        # heavy‑ball update
        v = beta * v - eta * g
        f_candidate = _project_simplex(f + v, target_sum)

        M_cand, _ = _fft_conv_max(f_candidate)

        if M_cand < M:
            # improvement – keep the momentum term
            f = f_candidate
            # slightly reward larger steps (but keep them bounded)
            eta = min(eta0, eta / 0.9)
        else:
            # failure – discard momentum and shrink step size
            v = np.zeros_like(v)
            eta *= 0.7
            if eta < min_eta:
                eta = eta0

    return best_f, best_M


# ----------------------------------------------------------------------
# Helper 5 – simulated annealing polishing
# ----------------------------------------------------------------------
def _annealing_search(
    start_seq: np.ndarray,
    deadline: float,
    init_temp: float = 0.8,
    decay: float = 0.996,
    step_scale: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Simple exponential‑cooling annealing that perturbs the current best
    vector with Gaussian noise, re‑projects onto the simplex and accepts
    moves probabilistically.
    """
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target = math.sqrt(2.0 * n)

    cur = start_seq.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    temp = init_temp
    cur_step = step_scale

    while time.monotonic() < deadline:
        cand = cur + rng.normal(scale=cur_step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target)

        cand_score = _score(cand)
        delta = cand_score - cur_score

        if delta < 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # exponential cooling
        temp *= decay
        cur_step *= decay
        if cur_step < 1e-7:
            cur_step = step_scale

    return best, best_score

    """

    Returns the best sequence (as a Python list) found before the deadline.
    """

    start_time = time.monotonic()
    # keep a small safety margin so the harness never sees a timeout
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Load a previously found construction if it exists

    if "height_sequence_1" in globals():

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    for n in lengths:
        if time.monotonic() >= deadline:

        # --------------------------------------------------------------
        starts = []

    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
