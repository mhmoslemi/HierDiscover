def project_simplex(v, target_sum):
    # v: np.ndarray
    # ensure nonneg via algorithm.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    # find rho
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        # all negative? then set uniform
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def smooth_max_and_grad(a, mu):
    conv = np.convolve(a, a)
    max_val = conv.max()
    # compute softmax weights
    # shift for numerical stability
    shifted = conv - max_val
    exp_shift = np.exp(shifted / mu)
    w = exp_shift / exp_shift.sum()
    # compute gradient = 2 * cross_correlation(w, a)
    # cross-corr g_i = sum_j w_{i+j} a_j
    # Use FFT:
    N = int(2 ** np.ceil(np.log2(len(w) + len(a) - 1)))
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a, n=N)
    # cross-correlation uses W * conj(A) then ifft.
    corr = np.fft.irfft(W * np.conj(A), n=N)[:len(a)]
    grad = 2 * corr
    return max_val, grad

"""
Propose a low‑value step sequence.

The algorithm consists of three stages:

1️⃣  **Parameter sweep** – for each target length a short sweep over
    simple families (power‑law, exponential) supplies a quick
    candidate that is already normalised to the required ℓ¹‑norm
    (ℓ¹‑norm = √(2·n) makes the evaluation equal to
    max (a * a)).

2️⃣  **Smoothed projected gradient descent** – we minimise a smooth
    approximation of the max‑convolution using the log‑sum‑exp
    (temperature μ).  The gradient of the smoothed max is a
    cross‑correlation that can be evaluated by an FFT in
    O(N log N).  After each gradient step we project back onto the
    simplex (non‑negative entries with the prescribed sum).  The
    temperature μ is decreased geometrically, turning the smooth
    objective into the true max‑objective.

3️⃣  **Fine‑grained stochastic polishing** – a cheap random‑swap
    hill‑climb followed by a short Gaussian perturbation stage
    improves the final vector.

All steps respect the security checks of ``evaluate_sequence``:
entries stay non‑negative, are far below 1000, and the ℓ¹‑norm is
exactly √(2·n).  The routine keeps the best vector seen so far and
returns it before the 1000 s deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _fast_score(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a normalised sequence."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _parameter_sweep(n: int, target_sum: float):
    """Search simple families (power‑law, exponential) for a good start."""
    best_vec = None
    best_sc = float("inf")
    idx = np.arange(1, n + 1, dtype=np.float64)

    # a_i ∝ i⁻ᵖ
    for p in np.linspace(0.1, 3.0, 30):
        a = idx ** (-p)
        a = a / a.sum() * target_sum
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # a_i ∝ exp(‑α·i)
    for alpha in np.linspace(0.001, 0.20, 30):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        a = a / a.sum() * target_sum
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    { w ≥ 0, Σ w = target_sum }.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        # all entries would become zero – fall back to uniform
        return np.full_like(v, target_sum / v.size)

    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_gradient_step(a: np.ndarray,
                          mu: float,
                          lr: float,
                          target_sum: float):
    """
    One step of projected gradient descent on the smooth max:
    maximise  φ_μ(a) = μ·log Σ_k exp( (a*a)_k / μ )
    (equivalently minimise it).

    Returns the new vector, the current max‑convolution entry and the
    gradient (for logging only).
    """
    # full auto‑convolution
    conv = np.convolve(a, a)
    max_c = conv.max()

    # soft‑max weights, numerically stable
    shifted = conv - max_c          # max becomes 0
    exp_w = np.exp(shifted / mu)
    w = exp_w / exp_w.sum()        # Σ w_k = 1

    # gradient = 2 * cross‑correlation(w, a)
    rev_a = a[::-1]                # reversed a
    N = int(2 ** np.ceil(np.log2(len(w) + len(a) - 1)))
    # FFT‑based convolution (real → real)
    W = np.fft.rfft(w, n=N)
    A_rev = np.fft.rfft(rev_a, n=N)
    full = np.fft.irfft(W * A_rev, n=N)   # length N, real
    start = a.size - 1
    grad = 2.0 * full[start:start + a.size]

    # gradient descent step (we minimise the smooth max)
    a_new = a - lr * grad

    # project back onto the simplex with the required ℓ¹‑norm
    a_new = _project_onto_simplex(a_new, target_sum)

    return a_new, max_c, grad


def _swap_hill_climb(seq: np.ndarray,
                     target_sum: float,
                     rng: np.random.Generator,
                     time_deadline: float,
                     best_score: float):
    """Very cheap random‑swap hill‑climb."""
    n = seq.shape[0]
    current = seq.copy()
    cur_sc = _fast_score(current)

    while time.monotonic() < time_deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # move a small amount of mass from i to j
        delta = max(1e-4, 0.02 * current.mean())
        if current[i] < delta:
            continue
        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < cur_sc:
            current, cur_sc = cand, cand_sc
            if cand_sc < best_score:
                best_score = cand_sc
                # expose the improvement via a print (optional)
                print(f"[swap] improved to {cand_sc:.6f}")
    return current, cur_sc, best_score


def _gaussian_local_search(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.2):
    """
    Short Gaussian hill‑climb (similar to the one in the reference code).
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = best.size
    target = math.sqrt(2.0 * n)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target / cand.sum())
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9997
        if step < 1e-7:
            step = step_initial
    return best, best_sc

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  initialise from a previously discovered construction, if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.size
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = _project_onto_simplex(init, target0)
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Parameter sweep over a handful of lengths
    # ------------------------------------------------------------------
    lengths = [500, 600, 700, 800, 900, 1000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        vec, sc = _parameter_sweep(n, target)
        if sc < best_score:
            best_seq, best_score = vec, sc
            print(f"[sweep] n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed projected gradient descent (multiple restarts)
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # number of random restarts – a little less for the largest n
        restarts = 3 if n <= 800 else 2

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # pick an initial point – blend a random Dirichlet with the global best if sizes match
            a = rng.random(n) + 0.01
            a = a / a.sum() * target
            if best_seq is not None and best_seq.shape[0] == n:
                a = (a + best_seq) / 2.0
                a = a / a.sum() * target

            # gradient‑descent parameters
            mu = 0.2                 # temperature for the log‑sum‑exp smoothing
            lr = target * 0.03       # learning rate (relative to the ℓ¹‑norm)
            max_iters = 2000

            for it in range(max_iters):
                if time.monotonic() >= deadline:
                    break

                a, cur_max, _ = _smooth_gradient_step(a, mu, lr, target)

                # after the projection the ℓ¹‑norm is exactly ``target``,
                # therefore ``cur_max`` is the true evaluation score
                if cur_max < best_score:
                    best_seq, best_score = a.copy(), cur_max
                    print(f"[gd] n={n} r={r} it={it} score={cur_max:.6f}")

                # cooling schedule (geometric decay)
                mu *= 0.985
                # occasional small increase of the learning rate when mu becomes tiny
                if mu < 1e-4:
                    mu = 1e-4
                # early exit if we are already below the target
                if best_score < 1.503:
                    print("[gd] target reached – exiting early.")
                    return best_seq.tolist()

    # ------------------------------------------------------------------
    # 3️⃣  Fine‑grained stochastic polishing
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – a modest‑size uniform vector
        n = 400
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # ----- cheap random‑swap hill‑climb -----
    best_seq, _tmp_sc, best_score = _swap_hill_climb(
        best_seq, math.sqrt(2.0 * best_seq.size), rng, deadline, best_score
    )

    # ----- short Gaussian hill‑climb -----
    if time.monotonic() < deadline:
        refined, refined_sc = _gaussian_local_search(best_seq, deadline, step_initial=0.2)
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[final] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
