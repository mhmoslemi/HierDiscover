def simulated_annealing(f0, deadline):
    f = f0.copy()
    n = len(f)
    target_sum = np.sum(f)
    step = target_sum * 0.05   # initial
    T = step   # approximate scale of objective improvement
    best = f.copy()
    best_score = compute_score(f)
    while time.monotonic() < deadline:
        # propose
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction: move from j->i
        max_transfer = min(step, f[j])
        delta = rng.random() * max_transfer
        # maybe also allow reverse with 0.5 chance
        if rng.random() < 0.5:
            # move opposite
            delta = -delta
        # apply
        f_candidate = f.copy()
        if delta >= 0:
            f_candidate[i] += delta
            f_candidate[j] -= delta
        else:
            delta = -delta
            # require delta <= f[i]; ensure
            if delta > f[i]:
                delta = f[i]
            f_candidate[i] -= delta
            f_candidate[j] += delta
        # compute score
        score = compute_score(f_candidate)
        # acceptance
        if score < best_score:
            best_score = score
            best = f_candidate.copy()
        if score < compute_score(f):
            accept = True
        else:
            # accept with prob exp(-(score-cur)/T)
            accept = rng.random() < math.exp(-(score - compute_score(f))/max(T, 1e-12))
        if accept:
            f = f_candidate
        # decay step and temperature
        step *= 0.9999
        T *= 0.9995
        # optionally occasionally reset step
        if step < target_sum*1e-6:
            step = target_sum*0.05
            T = step
    return best, best_score

            score = compute_score(vec)
            population.append([vec, score])

                child = parent.copy()

                child[j] -= delta

            # compute score

"""
Multi‑length stochastic optimisation for the step‑function bound.

The search explores several sequence lengths (400 … 1200).  For each length a
small population of candidates is kept on the ℓ¹‑simplex
(∑aᵢ = √(2 n), aᵢ ≥ 0).  Candidates are mutated by Gaussian perturbations,
projected back onto the simplex and evaluated with the exact
`evaluate_sequence` objective (implemented locally for speed).  Better
mutants replace their parents; occasional “crossover’’ (averaging two
candidates) adds diversity.  The best vector found for each length is then
polished with a few steps of projected gradient descent and a short
Gaussian hill‑climb.  The overall best sequence (lowest bound) is returned.

All helper functions are top‑level, there are no closures or lambda‑expressions.
The routine respects the overall time budget (default 1000 s) and never
produces negative or >1000 entries, so `evaluate_sequence` will never reject
the result.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact objective used by `evaluate_sequence` (no internal clipping)."""
    n = len(f)
    conv = np.convolve(f, f)               # length 2n‑1
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where the auto‑convolution is (almost) maximal."""
    eps = max_val * 1e-9 + 1e-12
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the worst auto‑convolution entry w.r.t. f,
    summed over all active offsets (the same routine as in the previous
    submission).
    """
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = f[j]
        # each pair contributes 2·f_j, but (i==j) counted once
        grad[i] += 2.0 * vals
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= vals[mask]   # 2·val – val = val
    return grad


def _gradient_descent(f0: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the max‑convolution objective."""
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)
    step = target_sum * 0.1
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        cur_score = _compute_score_from_vector(f)

        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:          # early stop on stagnation
            break

        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)
        grad = _gradient_submax(f, active)

        f_candidate = _project_onto_simplex(f - step * grad, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)
        cand_score = _compute_score_from_vector(f_candidate)

        if cand_score < cur_score:
            f = f_candidate
            step *= 1.02
        else:
            step *= 0.5
    return best_f, best_score


def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """Simple Gaussian hill‑climbing while staying on the ℓ¹‑simplex."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Population based stochastic search
# ----------------------------------------------------------------------


def _initial_vector(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Return a random feasible vector; a few deterministic patterns are mixed in."""
    mode = rng.integers(0, 4)
    if mode == 0:       # uniform‑like
        v = rng.random(n) + 0.01
    elif mode == 1:     # exponential decay
        lam = rng.uniform(0.0, 5.0 / n)
        v = np.exp(-lam * np.arange(n))
    elif mode == 2:     # linear increase
        v = np.arange(1, n + 1, dtype=np.float64)
    else:               # random Dirichlet‑style
        v = rng.random(n) + 0.01
    v = _project_onto_simplex(v, target_sum)
    return v


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the allotted time.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.3)   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣ Try a few different lengths; allocate time proportionally.
    # ------------------------------------------------------------------
    lengths = [400, 600, 800, 1000, 1200]
    n_lengths = len(lengths)

    for idx_len, n in enumerate(lengths):
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        pop_size = 6
        population = []

        # Warm‑start from the global construction if it matches the length.
        h1 = globals().get("height_sequence_1")
        if isinstance(h1, (list, tuple, np.ndarray)) and len(h1) == n:
            vec = np.array(h1, dtype=np.float64)
            vec = np.clip(vec, 0.0, 1000.0)
            if np.sum(vec) > 0:
                vec = _project_onto_simplex(vec, target_sum)
            else:
                vec = np.full(n, target_sum / n)
            population.append([vec, _compute_score_from_vector(vec)])

        # Fill the rest of the population with random feasible vectors.
        while len(population) < pop_size:
            vec = _initial_vector(n, target_sum, rng)
            population.append([vec, _compute_score_from_vector(vec)])

        # Sort by score (ascending).
        population.sort(key=lambda x: x[1])

        # Assign a slice of the overall remaining budget to this length.
        remaining = deadline - time.monotonic()
        # give roughly an equal share, but leave a bit for polishing later.
        sub_deadline = time.monotonic() + min(remaining * 0.7,
                                              max(4.0, remaining / (n_lengths - idx_len)))

        step = target_sum * 0.2          # initial mutation magnitude
        temperature = step * 10.0

        # ------------------------------------------------------------------
        # 2️⃣ Evolutionary loop (mutations + occasional crossover).
        # ------------------------------------------------------------------
        while time.monotonic() < sub_deadline:
            # pick a parent uniformly
            parent_idx = rng.integers(pop_size)
            parent, parent_score = population[parent_idx]

            # --- mutation -------------------------------------------------
            child = parent + rng.normal(scale=step, size=n)
            child = _project_onto_simplex(child, target_sum)
            child = np.clip(child, 0.0, 1000.0)
            child_score = _compute_score_from_vector(child)

            # accept if improvement or with Metropolis probability
            if child_score < parent_score or \
               rng.random() < math.exp(-(child_score - parent_score) / max(temperature, 1e-12)):
                population[parent_idx][0] = child
                population[parent_idx][1] = child_score
                # maintain sorted order cheaply
                population.sort(key=lambda x: x[1])

                if child_score < best_score:
                    best_score = child_score
                    best_seq = child.copy()
                    print(f"[new global] n={n} score={best_score:.6f}")

            # --- occasional crossover ------------------------------------
            if rng.random() < 0.07:
                a_idx, b_idx = rng.choice(pop_size, size=2, replace=False)
                a = population[a_idx][0]
                b = population[b_idx][0]
                child = (a + b) / 2.0
                child = _project_onto_simplex(child, target_sum)
                child_score = _compute_score_from_vector(child)
                if child_score < population[-1][1]:
                    population[-1][0] = child
                    population[-1][1] = child_score

            # decay schedule
            step *= 0.9997
            temperature *= 0.9995
            if step < target_sum * 1e-5:
                step = target_sum * 0.2
                temperature = step * 10.0

        # ------------------------------------------------------------------
        # 3️⃣ Polish the best vector found for this length.
        # ------------------------------------------------------------------
        if best_seq is not None:
            # a) short projected gradient descent
            gd_deadline = min(deadline, time.monotonic() + 5.0)
            f_opt, score_opt = _gradient_descent(best_seq, target_sum,
                                                 gd_deadline,
                                                 max_iters=3000)
            if score_opt < best_score:
                best_seq = f_opt
                best_score = score_opt
                print(f"[gd polish] n={n} score={best_score:.6f}")

            # b) final stochastic refinement (if time permits)
            if time.monotonic() < deadline:
                refined, refined_score = _stochastic_refinement(best_seq,
                                                                deadline,
                                                                step_initial=0.3)
                if refined_score < best_score:
                    best_seq = refined
                    best_score = refined_score
                    print(f"[final refine] n={n} score={best_score:.6f}")

        if best_score < 1.503:
            print("[target achieved] early exit")
            break

    # ------------------------------------------------------------------
    # 4️⃣ Fallback in the unlikely event no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 400
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the range required by evaluate_sequence).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
