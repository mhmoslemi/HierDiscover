def _simulated_annealing(a_start, deadline):
    a = a_start.copy()
    cur_score = _compute_score_from_vector(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = len(a)
    # Determine initial step (scale) = some fraction of average value
    avg = np.mean(a)
    step = avg * 0.2
    t0 = 0.1 * cur_score
    decay = 0.9995
    while time.monotonic() < deadline:
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        # delta limited by a[i]
        max_delta = min(step, a[i])
        if max_delta <= 0:
            continue
        delta = rng.uniform(-max_delta, max_delta) # signed; if negative, moving from j to i
        # We'll adjust both ways: if delta>0, move from i to j; else from j to i
        if delta >= 0:
            new_i = a[i] - delta
            new_j = a[j] + delta
        else:
            d = -delta
            new_i = a[i] + d
            new_j = a[j] - d
        # Ensure nonnegative
        if new_i < 0 or new_j < 0:
            continue
        new_a = a.copy()
        new_a[i] = new_i
        new_a[j] = new_j
        # clip to bounds
        new_a = np.clip(new_a, 0.0, 1000.0)
        # due to rounding might break sum; project back onto simplex to be safe
        new_a = _project_onto_simplex(new_a, np.sum(a))  # sum unchanged
        new_score = _compute_score_from_vector(new_a)
        if new_score < cur_score:
            a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
        else:
            temp = t0 * (decay ** (time.monotonic() - start_time))
            if np.exp(-(new_score - cur_score) / (temp + 1e-12)) > rng.random():
                a = new_a
                cur_score = new_score
        step *= 0.9999
        if step < 1e-8:
            step = avg * 0.2
    return best_a, best_score

"""
Propose a low‑value non‑negative step sequence.

Algorithm Overview
------------------
1️⃣ Warm‑start: If a previously found sequence ``height_sequence_1`` exists,
   it is normalised to the ℓ¹‑simplex (∑ aᵢ = √(2 n)) and used as the initial
   incumbent.

2️⃣ Randomised projected‑gradient descent:
   * A variety of lengths (400 … 3000) and several random initialisations
     (uniform, exponential, sinusoidal, √‑shaped and sparse “spike” patterns)
     are explored.
   * In each iteration the auto‑convolution is computed (FFT for n>1024),
     the active offsets where the maximum is attained are identified,
     and the exact sub‑gradient of the worst convolution entry is built.
   * A step is taken along the (negative) gradient, the iterate is projected
     back onto the ℓ¹‑simplex and the step size is adapted (increase on
     improvement, halve on failure).  Stagnation triggers early termination.
   * The best vector over all restarts and lengths is retained.

3️⃣ Simulated‑annealing polishing:
   * Starting from the best candidate, a simple pairwise mass‑swap
     neighbourhood is explored while preserving the ℓ¹‑norm.
   * Moves are accepted if they improve the objective or, with a
     temperature‑controlled probability, if they worsen it.
   * Temperature decays exponentially; the step‑size of the swaps also
     shrinks.

The code respects the 1000 s time budget, checks the deadline frequently,
and always returns a list of non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x | x_i≥0, Σx_i = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Edge case – return a uniform distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full convolution (FFT for larger n)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full discrete auto‑convolution a*a."""
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a)
    # FFT path – next power‑of‑two size for speed
    m = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (max entry of auto‑convolution)
# ----------------------------------------------------------------------


def _max_conv_score(a: np.ndarray) -> float:
    """Score = max entry of a*a (equal to evaluate_sequence after normalisation)."""
    conv = _full_convolution(a)
    return float(np.max(conv))


# ----------------------------------------------------------------------
# Active offsets (indices attaining (almost) the maximum)
# ----------------------------------------------------------------------


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    eps = max_val * 1e-12 + 1e-15
    return np.where(conv >= max_val - eps)[0]


# ----------------------------------------------------------------------
# Sub‑gradient of the worst auto‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_max_conv(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max_{k} (a*a)[k] (sum over all active k)."""
    n = len(a)
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        # each pair (i,j) contributes 2*a_j to ∂/∂a_i
        grad[i] += 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent with adaptive step size
# ----------------------------------------------------------------------


def _projected_gradient_descent(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Minimise max auto‑convolution entry while staying on the ℓ¹‑simplex.
    Returns the best vector seen and its score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _max_conv_score(f)

    # initial step size – a small fraction of the typical coordinate size
    step = target_sum * 0.05
    stagnation_counter = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # compute current max convolution and active offsets
        conv = _full_convolution(f)
        cur_max = float(np.max(conv))
        active = _active_offsets(conv, cur_max)

        # sub‑gradient of the worst entry
        grad = _subgradient_max_conv(f, active)

        # propose a move
        f_cand = _project_onto_simplex(f - step * grad, target_sum)
        f_cand = np.clip(f_cand, 0.0, 1000.0)  # safety clip
        cand_score = _max_conv_score(f_cand)

        if cand_score < best_score - 1e-12:
            # successful improvement
            best_f = f_cand.copy()
            best_score = cand_score
            f = f_cand
            step *= 1.07  # modest step increase
            stagnation_counter = 0
            # occasional debug output
            if it % 200 == 0:
                print(f"[GD] it={it} n={len(f)} score={cand_score:.6f}")
        else:
            # no improvement – shrink step
            step *= 0.5
            stagnation_counter += 1
            if stagnation_counter >= 400:
                break  # give up on this run

    return best_f, best_score


# ----------------------------------------------------------------------
# Simulated annealing / hill‑climbing polish
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple pairwise mass‑swap annealing that respects the ℓ¹‑norm.
    Returns the best vector found and its score.
    """
    rng = np.random.default_rng()
    n = len(start)
    cur = start.copy()
    cur_score = _max_conv_score(cur)
    best = cur.copy()
    best_score = cur_score

    # temperature schedule – start relatively high and decay
    T0 = cur_score * 0.12
    decay = 0.9995
    step_scale = np.mean(cur) * 0.20  # initial magnitude of swaps

    start_time = time.monotonic()
    while time.monotonic() < deadline:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(step_scale, cur[i])  # cannot move more than what i holds
        if max_delta <= 0:
            continue
        delta = rng.uniform(-max_delta, max_delta)

        # apply signed transfer
        if delta >= 0:
            new_i = cur[i] - delta
            new_j = cur[j] + delta
        else:
            d = -delta
            new_i = cur[i] + d
            new_j = cur[j] - d

        if new_i < 0 or new_j < 0:
            continue

        cand = cur.copy()
        cand[i] = new_i
        cand[j] = new_j
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑project to the exact simplex (tiny numerical drift)
        cand = _project_onto_simplex(cand, np.sum(start))

        cand_score = _max_conv_score(cand)

        # acceptance test
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # Metropolis rule
            elapsed = time.monotonic() - start_time
            T = T0 * (decay ** elapsed / 0.1)  # a slowly decaying temperature
            prob = math.exp(-(cand_score - cur_score) / (T + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # slowly shrink the perturbation magnitude
        step_scale *= 0.9997
        if step_scale < 1e-8:
            step_scale = np.mean(start) * 0.20

    return best, best_score


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------


def _random_initial_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random non‑negative vector of length n (not normalised)."""
    mode = rng.integers(5)
    if mode == 0:  # uniform Dirichlet‑like
        vec = rng.random(n) + 0.001
    elif mode == 1:  # exponential distribution – sparsity
        vec = rng.exponential(scale=1.0, size=n) + 0.001
    elif mode == 2:  # sinusoidal bump
        x = np.arange(n)
        vec = np.sin(np.pi * (x + 0.5) / n) + 0.001
    elif mode == 3:  # sqrt‐shaped (large in centre, small at edges)
        x = np.arange(n)
        vec = np.sqrt((x + 1) * (n - x)) + 0.001
    else:  # few spikes with a tiny baseline
        vec = np.full(n, 0.001)
        k = rng.integers(1, max(2, n // 6))
        idx = rng.choice(n, size=k, replace=False)
        vec[idx] = rng.random(k) + 0.2
    return vec


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step sequence within the allotted time.
    Returns the best sequence as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a tiny safety margin for the final polish
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start from any previously found construction.
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        n0 = len(seq)
        target0 = math.sqrt(2.0 * n0)
        if np.sum(seq) > 0:
            seq = seq * (target0 / np.sum(seq))
        else:
            seq = np.full(n0, target0 / n0, dtype=np.float64)
        score0 = _max_conv_score(seq)
        best_seq = seq.copy()
        best_score = score0
        print(f"[init] used height_sequence_1 (n={n0}) → score={score0:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Main optimisation – randomised projected gradient descent.
    # --------------------------------------------------------------
    # a list of lengths to explore (order shuffled each outer loop)
    length_pool = [400, 600, 800, 1000, 1500, 2000, 2500, 3000]

    while time.monotonic() < deadline - 5.0:   # reserve a few seconds for polishing
        rng.shuffle(length_pool)
        for n in length_pool:
            if time.monotonic() >= deadline - 5.0:
                break
            target_sum = math.sqrt(2.0 * n)

            # adapt number of restarts to size (larger n → fewer restarts)
            if n <= 1000:
                max_restarts = 12
                max_iters = 4000
            elif n <= 2000:
                max_restarts = 7
                max_iters = 3000
            else:
                max_restarts = 4
                max_iters = 2000

            for restart in range(max_restarts):
                if time.monotonic() >= deadline - 5.0:
                    break

                # initialise a candidate vector
                if best_seq is not None and len(best_seq) == n:
                    f0 = best_seq.copy()
                else:
                    f0 = _random_initial_vector(n, rng)
                    f0 = f0 / np.sum(f0) * target_sum

                sub_deadline = min(deadline, time.monotonic() + 8.0)  # ~8 s per descent
                f_opt, score_opt = _projected_gradient_descent(
                    f0, target_sum, sub_deadline, max_iters=max_iters
                )

                if score_opt < best_score - 1e-12:
                    best_score = score_opt
                    best_seq = f_opt.copy()
                    print(
                        f"[update] n={n} restart={restart} score={score_opt:.6f}"
                    )

                # early exit if we already beat the target threshold
                if best_score < 1.503:
                    print("[early] target threshold reached")
                    break
            if best_score < 1.503:
                break

    # --------------------------------------------------------------
    # 3️⃣ Final stochastic polishing (simulated annealing).
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined_seq, refined_score = _simulated_annealing(best_seq, deadline)
        if refined_score < best_score - 1e-12:
            best_seq, best_score = refined_seq, refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # safety clipping (evaluate_sequence also clips)
    if best_seq is None:
        # fallback: a modest uniform construction
        n_fallback = 500
        target_sum = math.sqrt(2.0 * n_fallback)
        best_seq = np.full(n_fallback, target_sum / n_fallback, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
