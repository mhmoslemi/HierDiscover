Propose a candidate sequence for minimizing the evaluation function.

The algorithm consists of three stages:

1️⃣ **Projected sub‑gradient descent** – we minimise the convex objective
   `max(conv)` where `conv = a * a` is the discrete convolution of the
   (non‑negative) step heights `a`.  The sum of the heights is fixed to
   `√(2 n)` (the normalisation used by the reference solution) and the
   vector is projected onto the ℓ¹‑simplex after each sub‑gradient step.
   The sub‑gradient of the active convolution entry `k` is known analytically:
       g_i = a_i + a_{k‑i}   (with a_{j}=0 for out‑of‑range indices).
   Fast convolution via FFT (O(n log n)) is used at every iteration.

2️⃣ **Mass‑transfer hill climbing** – after the sub‑gradient phase we
   apply the incremental transfer algorithm from the reference code.
   It moves a small amount of mass between two coordinates while keeping
   the ℓ¹‑norm unchanged, updating the full convolution incrementally.
   This often yields a small further reduction of the maximum entry.

3️⃣ **Gentle Gaussian polishing** – a final stochastic local‑search
   (Gaussian perturbations) removes leftover irregularities.

All intermediate vectors respect the non‑negativity and clipping rules
implemented inside `evaluate_sequence`.  The routine explores several
sequence lengths (up to 1000) and many random restarts within the allotted
time budget (default 1000 s).  The best sequence found is returned as a
plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Exact official score (calls the provided evaluate_sequence)."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex
        { x >= 0 , sum(x) = target_sum }.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Convolution a*a using a real FFT (fast for many iterations)."""
    n = a.shape[0]
    # length needed for full convolution
    needed = 2 * n - 1
    # next power of two for efficient FFT
    L = 1 << (needed - 1).bit_length()
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)
    return conv[:needed]


# ----------------------------------------------------------------------
# Incremental mass‑transfer hill‑climber (from the reference code)
# ----------------------------------------------------------------------


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index `i` to index `j` (i != j, delta > 0).
    Updates `a` and the full convolution `conv` in‑place using the exact
    algebraic formula (O(n) work).
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part 2 * conv(d, a)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part conv(d, d) where d = -δe_i + δe_j
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta

    # update vector
    a[i] -= delta
    a[j] += delta


def _improve_by_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that repeatedly moves a small amount of mass while keeping the ℓ¹‑norm.
    Returns the best sequence encountered and its official score.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)
    sum_a = a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)
    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        _transfer_mass(a, conv, i, j, delta)
        cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            _transfer_mass(a, conv, j, i, delta)   # revert

        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that adds Gaussian noise to the whole
    vector (as used in the reference solution).  Used as a final polishing step.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------


def _subgradient_descent_one(start_vec: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Run a projected sub‑gradient descent starting from `start_vec`.
    The simplex target sum is √(2 n).  Returns the best vector seen and the
    associated score (computed analytically, not via evaluate_sequence).
    """
    S = math.sqrt(2.0 * n)
    a = _project_onto_simplex(start_vec, S)   # ensure exact sum
    best_vec = a.copy()
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max / (S * S)

    rng = np.random.default_rng()
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = _conv_fft(a)
        k = int(conv.argmax())
        cur_max = conv[k]
        cur_score = 2.0 * n * cur_max / (S * S)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            # occasional progress report
            print(f"[sgd] n={n} it={it} score={best_score:.6f}")

        # diminishing step size
        step = 0.5 / math.sqrt(it + 1)

        # sub‑gradient of the active convolution entry
        g = a.copy()
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1)
        g[idx] += a[k - idx]

        # gradient step and projection
        a = a - step * g
        a = _project_onto_simplex(a, S)

        # early exit if we already beat the target bound
        if best_score < 1.503:
            break

    return best_vec, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the global time budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    # reserve a small margin for post‑processing / printing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Gather candidate initialisations (including any global hint)
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        init_global = np.array(height_sequence_1, dtype=float)
    except Exception:
        init_global = None

    lengths = [200, 300, 400, 500, 600, 800, 1000]   # larger n gives a lower theoretical bound

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        S = math.sqrt(2.0 * n)

        # generate a handful of starts for this length
        starts: list[np.ndarray] = []

        if init_global is not None and init_global.shape[0] == n:
            starts.append(init_global.copy())

        # uniform start (all entries equal, already feasible)
        starts.append(np.full(n, S / n, dtype=float))

        # a few random starts (exponential gives sparse vectors, uniform dense)
        for _ in range(3):
            if rng.random() < 0.5:
                vec = rng.exponential(scale=1.0, size=n)
            else:
                vec = rng.random(n) + 0.1
            starts.append(vec)

        # ------------------------------------------------------------------
        # 2️⃣  Run the sub‑gradient optimiser for each start
        # ------------------------------------------------------------------
        for start in starts:
            if time.monotonic() >= deadline:
                break

            # Normalise to the simplex with the required ℓ¹‑norm
            start = _project_onto_simplex(start, S)

            # sub‑gradient descent
            sgd_vec, sgd_score = _subgradient_descent_one(start, n, deadline,
                                                          max_iters=80_000)
            # official evaluation (safety check)
            official_score = _score(sgd_vec)
            if official_score < best_score:
                best_score = official_score
                best_seq = sgd_vec.copy()
                print(f"[sgd‑best] n={n} score={best_score:.6f}")

            # Early exit if we have already beaten the target
            if best_score < 1.503:
                return best_seq.tolist()

            # ------------------------------------------------------------------
            # 3️⃣  Refine with the incremental mass‑transfer hill‑climber
            # ------------------------------------------------------------------
            improved_vec, _ = _improve_by_transfer(sgd_vec, deadline,
                                                   max_iters=150_000,
                                                   step_init=0.5)
            refined_score = _score(improved_vec)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = improved_vec.copy()
                print(f"[transfer‑best] n={n} score={best_score:.6f}")

            if best_score < 1.503:
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 4️⃣  Final Gaussian polishing (in case nothing better was found)
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – modest uniform vector of length 200
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    if time.monotonic() < deadline:
        refined, refined_score = _local_search(best_seq, deadline,
                                               step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final‑polish] improved to {best_score:.6f}")

    # Return a plain list as required by the evaluator
    return [float(x) for x in best_seq]
