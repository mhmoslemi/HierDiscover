def _project_simplex(v, z=1):
    # v is 1D array
    # Projects onto {x: x>=0, sum x = z}
    # from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]

    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑score step sequence.

The algorithm follows several complementary stages:

1️⃣  **Initial candidates** – a fast parametric sweep (power‑law,
    exponential and simple logarithmic families) for several lengths.

2️⃣  **Intensive descent** – for each promising candidate we run a long
    projected sub‑gradient descent that directly reduces the maximal
    auto‑convolution entry.  The descent uses the exact sub‑gradient of
    the active convolution entry and projects onto the non‑negative
    simplex of the required ℓ¹‑norm.

3️⃣  **Simulated‑annealing polishing** – a cheap mass‑transfer move
    (donor → receiver) is performed many times; worse moves are accepted
    with a temperature that decays linearly to the deadline.

4️⃣  **Inflation trick** – the best vector is “inflated’’ (each entry is
    split into *k* copies and the whole vector is rescaled to the larger
    ℓ¹‑norm).  This often loosens the convolution peak.  Inflated vectors
    are again refined by stages 2 & 3.

The whole search respects the security checks of ``evaluate_sequence``:
all entries are non‑negative, clipped to [0, 1000] and the ℓ¹‑norm is
exactly √(2 n).  The routine stops before the time budget expires and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` (without the safety wrapper)."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    # Duchi et al. (2008) – O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / (np.arange(1, len(u) + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _parameter_sweep(n: int, target_sum: float):
    """Search a handful of simple families and return the best vector."""
    best_vec = None
    best_sc = float("inf")
    idx = np.arange(1, n + 1, dtype=np.float64)

    # power‑law a_i ∝ i⁻ᵖ
    for p in np.linspace(0.1, 3.0, 30):
        a = idx ** (-p)
        a = a / a.sum() * target_sum
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # exponential a_i ∝ exp(-α·i)
    for alpha in np.linspace(0.001, 0.20, 30):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        a = a / a.sum() * target_sum
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # simple logarithmic decay a_i ∝ 1 / (i + c)
    for c in np.linspace(0.0, 5.0, 20):
        a = 1.0 / (idx + c)
        a = a / a.sum() * target_sum
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


def _subgradient_step(seq: np.ndarray, target_sum: float, lr: float) -> np.ndarray:
    """
    One projected sub‑gradient step aimed at decreasing the maximal
    convolution entry.  The direction is ``-g + mean(g)`` where
    ``g_i = seq_{k_max - i}`` for the active offset ``k_max``.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())

    # contribution vector g for the active offset
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    j = k - i_range
    g = np.zeros_like(seq)
    g[i_range] = seq[j]

    # zero‑sum direction (preserves total mass before projection)
    direction = -g + g.mean()

    new_seq = seq + lr * direction
    # enforce the hard bounds before projection
    new_seq = np.clip(new_seq, 0.0, 1000.0)

    # project back onto the ℓ¹‑simplex of radius target_sum
    total = new_seq.sum()
    if total == 0.0:
        new_seq = np.full(n, target_sum / n, dtype=np.float64)
    else:
        new_seq = _project_to_simplex(new_seq, target_sum)

    return new_seq


def _intensive_descent(start_seq: np.ndarray,
                       deadline: float,
                       max_iters: int = 5000,
                       start_lr: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Long projected sub‑gradient descent with a geometrically decaying
    learning rate.  Returns the best vector encountered.
    """
    n = start_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    lr = start_lr
    lr_decay = 0.9997

    while time.monotonic() < deadline and max_iters > 0:
        seq = _subgradient_step(seq, target_sum, lr)
        sc = _fast_score(seq)
        if sc < best_sc:
            best_sc, best_seq = sc, seq.copy()
        lr *= lr_decay
        max_iters -= 1

    return best_seq, best_sc


def _simulated_annealing(start_seq: np.ndarray,
                         start_score: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Cheap SA that moves a random amount of mass from a donor index to a
    receiver index.  Worse moves are accepted with probability
    ``exp(-Δ/T)`` where temperature T decays linearly to zero at the
    deadline.
    """
    n = start_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    cur_seq = start_seq.copy()
    cur_sc = start_score
    best_seq = cur_seq.copy()
    best_sc = cur_sc

    # a modest initial temperature; the scale of the objective is ≈1.
    T0 = 0.005

    while time.monotonic() < deadline:
        # pick donor (must have some mass) and receiver
        donor = rng.integers(n)
        if cur_seq[donor] < 1e-9:
            continue
        receiver = rng.integers(n)
        if donor == receiver:
            continue

        # amount to move – up to 30 % of the donor’s mass
        delta = cur_seq[donor] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        cand = cur_seq.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        # clipping and re‑projection (the sum is unchanged unless clipping)
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total != target_sum:
            cand = _project_to_simplex(cand, target_sum)

        cand_sc = _fast_score(cand)
        Δ = cand_sc - cur_sc
        elapsed = time.monotonic() - start_time
        remaining = max(0.0, deadline - time.monotonic())
        # linear temperature schedule
        T = T0 * remaining / (deadline - start_time)

        if Δ < 0 or rng.random() < math.exp(-Δ / max(T, 1e-12)):
            cur_seq, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_seq = cand_sc, cand.copy()

    return best_seq, best_sc


def _inflate_sequence(seq: np.ndarray, factor: int) -> np.ndarray:
    """
    Replace each entry by ``factor`` copies, then rescale the whole vector
    so that the ℓ¹‑norm becomes √(2·(factor·len(seq))).
    """
    if factor <= 1:
        return seq.copy()
    # repeat each entry ``factor`` times
    repeated = np.repeat(seq / factor, factor)
    # required ℓ¹‑norm after inflation
    n_new = len(seq) * factor
    target = math.sqrt(2.0 * n_new)
    # the repeat sum is still ``seq.sum()``; scale to the new target
    scale = target / seq.sum()
    inflated = repeated * scale
    return inflated

    if "height_sequence_1" in globals():

            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = _project_to_simplex(init, target0)
                sc0 = _fast_score(init)
                best_seq, best_score = init.copy(), sc0
                print(f"[init] length={n0} score={sc0:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep for a handful of lengths.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        vec, sc = _parameter_sweep(n, target)
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[sweep] n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Intensive descent on the current best (and a few random restarts).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a uniform vector of moderate size
        n = 400
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # keep a small pool of promising vectors for diversification
    pool = [(best_seq.copy(), best_score)]

    # a few random Dirichlet restarts at the same lengths
    for n in [500, 600, 800]:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        for _ in range(2):
            rnd = rng.random(n) + 0.1
            rnd = _project_to_simplex(rnd, target)
            sc = _fast_score(rnd)
            pool.append((rnd, sc))

    # intensive descent on each pool element
    for idx, (vec, sc0) in enumerate(pool):
        if time.monotonic() >= deadline:
            break
        vec_refined, sc_refined = _intensive_descent(vec, deadline,
                                                     max_iters=8000,
                                                     start_lr=0.5)
        if sc_refined < best_score:
            best_seq, best_score = vec_refined.copy(), sc_refined
            print(f"[descent] pool#{idx} improved to {sc_refined:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polishing on the current best.
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        best_seq, best_score = _simulated_annealing(best_seq,
                                                    best_score,
                                                    deadline,
                                                    rng)
        print(f"[SA] after polishing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Inflation trick: try longer sequences built from the current best.
    # ------------------------------------------------------------------
    for factor in [2, 3]:
        if time.monotonic() >= deadline:
            break
        inflated = _inflate_sequence(best_seq, factor)
        sc_infl = _fast_score(inflated)
        if sc_infl < best_score:
            best_seq, best_score = inflated, sc_infl
            print(f"[inflate] factor={factor} score={sc_infl:.6f}")

        # run a quick descent on the inflated candidate
        if time.monotonic() < deadline:
            infl_ref, infl_sc = _intensive_descent(best_seq, deadline,
                                                   max_iters=4000,
                                                   start_lr=0.4)
            if infl_sc < best_score:
                best_seq, best_score = infl_ref, infl_sc
                print(f"[inflate‑descent] factor={factor} improved to {infl_sc:.6f}")

        # final SA on the inflated version
        if time.monotonic() < deadline:
            best_seq, best_score = _simulated_annealing(best_seq,
                                                        best_score,
                                                        deadline,
                                                        rng)
            print(f"[inflate‑SA] factor={factor} final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final tiny Gaussian hill‑climb (very cheap, just in case).
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        step = 0.2
        while time.monotonic() < deadline:
            cand = best_seq + rng.normal(scale=step, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _project_to_simplex(cand, target)
            cand_sc = _fast_score(cand)
            if cand_sc < best_score:
                best_seq, best_score = cand, cand_sc
                print(f"[final‑gauss] improved to {cand_sc:.6f}")
            step *= 0.9995
            if step < 1e-7:
                step = 0.2

    print(f"[total] best score achieved = {best_score:.6f}")
    return [float(x) for x in best_seq]
