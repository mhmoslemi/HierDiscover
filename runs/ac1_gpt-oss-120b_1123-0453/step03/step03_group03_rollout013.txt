def project_to_simplex(v, S):
    # Clip to non-negative?
    v = np.maximum(v, 0)  # Ensure nonnegative? But we need to handle unconstrained.
    # Actually projection onto simplex with nonneg constraints can be done by sorting.
    # Use standard algorithm:
    # Shift v to satisfy sum = S.
    # Input: vector v (may have negative). Return w = argmin ||w - v||_2^2 s.t. w >= 0, sum w = S.
    # Implementation: sort v descending, find rho etc.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, S):
    # ensure sum(v) = S and v >= 0
    # Standard algorithm: sort v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        # All entries zero? fallback: set all to zero except one value = S
        w = np.zeros_like(v)
        w[0] = S
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Using this projection will also correct any overshoot.

A fast hill‑climbing / annealing optimiser for the auto‑convolution bound.

The optimiser works inside the given time budget (default 1000 s) and
does not rely on external LP solvers.  It proceeds as follows:

1. Initialise the current vector from the global `height_sequence_1`

   vector is normalised so that its ℓ¹‑norm equals √(2·n).

2. Repeatedly locate the current maximal convolution entry.  The
   contribution of each coefficient to that peak is
   `c_i = a_i * a_{k‑i}` (where `k` is the index of the peak).
   The algorithm tries to move a fraction of the mass from a
   high‑contribution position to a low‑contribution position.
   Several transfer fractions are evaluated and the best move is
   kept.

3. If a move does not improve the peak, a small Gaussian perturbation
   is performed and the resulting vector is projected back onto the
   simplex (non‑negative, fixed ℓ¹‑norm).  This provides a simple
   simulated‑annealing effect and helps escape shallow local minima.

4. The best vector ever seen is tracked; if its evaluation drops below
   the target bound (1.503) the search stops early.

All vectors respect the security checks performed by
`evaluate_sequence` (non‑negative, values ≤ 1000).  The function returns
the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex with prescribed sum S.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, sum(x)=S}."""
    # Standard O(n log n) algorithm.
    if S <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - S
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / len(v)
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(seq: np.ndarray) -> float:
    """Return max(f*f) for a non‑negative sequence."""
    # For a correctly normalised vector the evaluation equals this value.
    conv = np.convolve(seq, seq)
    return float(conv.max())
