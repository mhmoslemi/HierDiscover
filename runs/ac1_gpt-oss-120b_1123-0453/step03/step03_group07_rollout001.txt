def subgradient(f, target_sum):
    conv = conv_full(f)
    M = conv.max()
    idxs = np.where(conv >= M - 1e-12)[0]
    grad = np.zeros_like(f)
    for i in idxs:
        # compute contributions
        start = max(0, i - (len(f)-1))
        end = min(i, len(f)-1)
        # For k in start..end: index within f
        # Contribute 2*f[i - k] to grad[k]
        # vectorized:
        ks = np.arange(start, end+1)
        grad[ks] += 2 * f[i - ks]
    grad = grad / len(idxs)  # average
    return conv, M, grad

def project_simplex(v, S):
    # From Duchi et al. 2008.
    n = v.shape[0]
    if S < 0:
        raise ValueError
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1   # zero-indexed
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def subgradient_optimize(n, init_f, S, max_iters, step0, deadline):
    f = init_f.copy()
    conv = conv_full(f)
    best_M = conv.max()
    best_f = f.copy()
    # maintain previous M to monitor stagnation
    no_improve = 0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv, M, grad = subgradient(f)
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # step size schedule
        alpha = step0 / np.sqrt(it)

        # gradient descent step
        f = f - alpha * grad

        # projection onto simplex
        f = project_simplex(f, S)

        # occasional small random perturbation if stuck
        if no_improve >= 200:
            # Add a small random noise
            perturb = np.random.default_rng().normal(scale=0.01 * S / n, size=n)
            f = np.clip(f + perturb, 0, None)
            f = project_simplex(f, S)
            no_improve = 0

"""
Propose a low‑value step sequence.

This implementation replaces the previous LP‑based heuristic with a
Projected Sub‑gradient method on the convex objective  

    M(f) = max_i (f * f)[i]                (convolution with itself)

subject to the simplex constraints  

    f_i ≥ 0 ,  Σ f_i = √(2 n) .

The algorithm:
* tries several lengths (up to 2000) and a few random Dirichlet restarts,
* for each restart runs a projected sub‑gradient descent with a
  diminishing step‑size and occasional tiny random perturbations,
* keeps the best vector found (its score is exactly the maximal
  convolution value, which equals ``evaluate_sequence``’s return).

All helper functions are defined at top level so that there are no
closures, and the total running time is limited by the supplied budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full(f: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of f with itself using FFT.
    Returns an array of length 2·len(f)‑1.
    """
    n = f.shape[0]
    # next power of two for efficient FFT
    L = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(f, n=L)
    conv = np.fft.irfft(F * F, n=L)
    return conv[: 2 * n - 1]


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x | x ≥ 0 , Σ x = s}.
    Implementation of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.maximum(v, 0.0)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(f: np.ndarray):
    """
    Sub‑gradient of M(f)=max_i (f*f)[i] at the current vector f.
    Returns (conv, M, grad) where
        conv = full convolution f*f,
        M    = max(conv),
        grad = averaged gradient of the active (maximal) entries.
    """
    n = f.shape[0]
    conv = _conv_full(f)
    M = conv.max()
    # indices where the max is attained (within numerical tolerance)
    idxs = np.where(conv >= M - 1e-12)[0]
    grad = np.zeros_like(f)
    for i in idxs:
        start = max(0, i - (n - 1))
        end = min(i, n - 1)
        ks = np.arange(start, end + 1)
        # derivative of (f*f)[i] w.r.t. f[ks] is 2 * f[i‑ks]
        grad[ks] += 2.0 * f[i - ks]
    grad = grad / len(idxs)  # any convex combination is a sub‑gradient
    return conv, M, grad


def _subgradient_optimize(n: int,
                          init_f: np.ndarray,
                          target_sum: float,
                          max_iters: int,
                          step0: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for a fixed length n.
    Returns the best vector seen and its maximal convolution value.
    """
    f = init_f.copy()
    conv = _conv_full(f)
    best_M = conv.max()
    best_f = f.copy()

    no_improve = 0
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv, M, grad = _subgradient(f)

        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # diminishing step size
        alpha = step0 / math.sqrt(it)

        # gradient step and projection onto the simplex {x ≥ 0, Σx = target_sum}
        f = f - alpha * grad
        f = _project_simplex(f, target_sum)

        # occasional tiny random jitter if we are stuck
        if no_improve >= 200:
            perturb = rng.normal(scale=0.01 * target_sum / n, size=n)
            f = np.clip(f + perturb, 0.0, None)
            f = _project_simplex(f, target_sum)
            no_improve = 0

    return best_f, best_M


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score.

    The routine explores a range of lengths (up to 2000) and several
    random Dirichlet restarts per length.  For each start it runs a
    projected sub‑gradient descent on the maximal autocorrelation,
    keeping the best sequence found before the time budget expires.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # tiny safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Lengths to explore – larger n give a lower theoretical lower bound.
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500, 1800, 2000]

    for n in lengths:
        if time.time() >= deadline:

        target_sum = math.sqrt(2.0 * n)

        # a few independent restarts for each length
        restarts = 3 if n <= 1000 else 2

        for r in range(restarts):
            if time.time() >= deadline:

            # ----------------------------------------------------------
            # Initialise the current vector.
            global_seq = globals().get("height_sequence_1")
            if (r == 0 and isinstance(global_seq, (list, tuple, np.ndarray))

                f0 = np.maximum(f0, 0.0)

            else:

            # Rough estimate: 0.0004 s per iteration for n≈500, scaling with n·log n.

            )

            # ----------------------------------------------------------
            # Evaluate (the evaluation function is exactly M for the

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found.
    if best_seq is None:
        n = 200

        best_score = float(evaluate_sequence(best_seq.tolist()))
        best_n = n
        print("[fallback] uniform sequence used")

    print(f"Best overall score: {best_score:.6f} (length {best_n})")
    # Return a plain Python list of floats as required.
    return [float(v) for v in best_seq]
