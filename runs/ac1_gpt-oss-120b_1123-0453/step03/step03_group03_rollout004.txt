def simplex_projection(v, s):
    """Projects onto the probability simplex of sum s and non-negative entries."""
    if s < 0:
        raise ...
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v, s):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def grad_max_convolution(f, alpha):
    conv = np.convolve(f, f)
    # smoothed weights
    # compute stable exp
    z = alpha * conv
    max_z = np.max(z)
    w = np.exp(z - max_z)
    w /= w.sum()
    # compute convolution of w with reversed f
    rev = f[::-1]
    conv_w_rev = np.convolve(w, rev)
    grad = 2 * conv_w_rev[len(f)-1 : len(f)-1 + len(f)]
    return grad

"""
Gradient‑based flattening of the auto‑convolution.

The evaluation of a step sequence is simply the maximum entry of the
auto‑convolution when the ℓ¹‑norm is fixed to √(2·n).  The algorithm
starts from the best known construction (if available) and then
repeatedly reduces the peak by a projected (sub‑)gradient descent on a
smooth approximation of the max‑operator.  A simple line‑search
selects a step size, the iterate is projected onto the simplex
{a_i≥0, Σa_i=√(2·n)} and the smoothing parameter is slowly increased so
that the optimisation first moves mass globally and later focuses on the
currently highest peak.  A short stochastic polishing stage is performed
until the time budget expires.

The method is completely self‑contained, uses only numpy and the
provided ``evaluate_sequence`` for the final scoring, and respects all
security checks (non‑negative, ≤1000).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smoothed_gradient(f: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        φ_α(f) = (1/α)·log Σ_k exp(α·(f*f)[k])
    w.r.t. ``f``.
    The formula reduces to a convolution of the weight vector with the
    reversed ``f``.
    """
    # auto‑convolution
    conv = np.convolve(f, f)                     # length 2·n‑1
    # stable soft‑max weights
    z = alpha * conv
    z_max = np.max(z)
    w = np.exp(z - z_max)
    w /= w.sum()

    # gradient = 2·(w * f_rev) evaluated at the original indices
    rev = f[::-1]
    tmp = np.convolve(w, rev)                    # length 3·n‑2
    n = f.shape[0]
    grad = 2.0 * tmp[n - 1 : n - 1 + n]
    return grad


def _max_convolution(f: np.ndarray) -> float:
    """Return max_k (f*f)[k].  For a correctly normalised vector this is the score."""
    return float(np.max(np.convolve(f, f)))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine follows a three‑stage approach:

    1.  Initialise from the global ``height_sequence_1`` (if present) or a
        random simplex point.
    2.  Projected (sub‑)gradient descent on a smoothed max‑convolution,
        with an adaptive step‑size and a slowly increasing smoothing
        parameter.
    3.  Short stochastic polishing with Gaussian noise.

    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Initialise – try a few different lengths
    # ------------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000]   # longer vectors often give more flexibility
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)    # ℓ¹‑norm that makes the score equal to max‑conv

        # ----- initial vector -------------------------------------------------
        if "height_sequence_1" in globals():
            try:
                seq0 = np.array(height_sequence_1, dtype=float)
                if seq0.size != n:
                    # truncate or pad with zeros to match the chosen length
                    if seq0.size > n:
                        seq0 = seq0[:n]
                    else:
                        seq0 = np.concatenate([seq0, np.zeros(n - seq0.size)])
                f = _project_to_simplex(seq0, target_sum)
            except Exception as e:                     # pragma: no cover
                print("[init] fallback – cannot reuse global sequence:", e)
                f = rng.random(n)
                f = _project_to_simplex(f, target_sum)
        else:
            f = rng.random(n)
            f = _project_to_simplex(f, target_sum)

        cur_max = _max_convolution(f)
        if cur_max < best_score:
            best_score = cur_max
            best_seq = f.copy()
            print(f"[init] n={n} initial score={cur_max:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣  Projected smoothed‑max gradient descent
        # ------------------------------------------------------------------
        # hyper‑parameters (chosen to be robust across lengths)
        alpha = 1.0                     # start with a very smooth approximation
        eta0 = target_sum * 0.15        # initial step size (relative to the sum)
        eta = eta0
        max_iter = 3000                # generous upper bound; will stop early by time

        for it in range(max_iter):
            if time.monotonic() >= deadline:
                break

            # compute the (sub)gradient
            grad = _smoothed_gradient(f, alpha)

            # simple back‑tracking line search – at most 5 reductions of η
            improved = False
            for _ in range(5):
                cand = _project_to_simplex(f - eta * grad, target_sum)
                cand_max = _max_convolution(cand)
                if cand_max < cur_max - 1e-9:          # require a tiny improvement
                    f = cand
                    cur_max = cand_max
                    improved = True
                    # keep the overall best
                    if cur_max < best_score:
                        best_score = cur_max
                        best_seq = f.copy()
                        print(f"[gd] n={n} it={it} score={cur_max:.6f}")
                    break
                else:
                    eta *= 0.5                         # shrink step and retry
            if not improved:
                # no progress – recover η and inject a small random perturbation
                eta = eta0
                if it % 120 == 0:                     # every few hundred iterations
                    noise = rng.normal(scale=0.01 * target_sum, size=n)
                    f = _project_to_simplex(f + noise, target_sum)
                    cur_max = _max_convolution(f)

            # slowly increase the smoothing – the objective becomes sharper
            alpha = min(alpha * 1.015, 40.0)

            # occasional gentle increase of the step size (helps escape plateaus)
            if it % 80 == 0:
                eta = min(eta0, eta * 1.08)

            # early exit if we already beat the target bound
            if best_score < 1.503:
                print("[gd] target bound reached – exiting early.")
                return best_seq.tolist()

        # after a full run for this length keep the result as a warm‑start for
        # the next (larger) length
        best_seq = best_seq  # already holds the best found

    # ------------------------------------------------------------------
    # 3️⃣  Final stochastic polishing (Gaussian jitter)
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – uniform vector of length 500
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _max_convolution(best_seq)

    cur = best_seq.copy()
    n = cur.shape[0]
    target_sum = math.sqrt(2.0 * n)
    step = 0.35                                            # initial perturbation scale

    while time.monotonic() < deadline:
        cand = cur + rng.normal(scale=step, size=n)
        cand = _project_to_simplex(cand, target_sum)
        cand_max = _max_convolution(cand)
        if cand_max < best_score:
            cur = cand
            best_seq = cand.copy()
            best_score = cand_max
            print(f"[polish] improved to {best_score:.6f}")
        # exponential decay of the perturbation amplitude
        step *= 0.9996
        if step < 1e-5:
            step = 0.35

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} length={best_seq.size}")
    return [float(x) for x in best_seq]
