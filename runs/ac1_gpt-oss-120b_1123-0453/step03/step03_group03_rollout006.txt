"""
Propose a candidate step‑function sequence with a low auto‑convolution bound.

Algorithm
---------
1. **Initialisation** – start from the globally stored `height_sequence_1`
   (if available) or from a random non‑negative vector of a given length.
   The ℓ¹‑norm is always normalised to √(2·n) so that the evaluation score
   equals the maximal auto‑convolution entry `M`.

2. **Linearised LP move (core driver)** – at each iteration we solve a small
   linear programme that searches for a direction `d` (with Σdᵢ = 0,
   aᵢ+dᵢ ≥ 0) that minimises a linearised upper bound on the new peak

        M'  =  maxₖ ( (a*a)ₖ + 2·Σᵢ aₖ₋ᵢ·dᵢ ).

   The LP variables are the `n` adjustments `dᵢ` and the scalar `M'`.
   The problem is convex, bounded (M' ≥ 0) and cheap to solve with
   `scipy.optimize.linprog`.  The obtained direction is applied with a
   back‑tracking line search (λ∈{1,0.8,0.5,…}) until the *actual* max
   convolution decreases.  This step aggressively flattens the dominant
   peaks while keeping the ℓ¹‑norm fixed.

3. **Fallback greedy peak reduction** – if the LP does not yield an
   improvement we perform a cheap greedy move: move a fraction of mass from
   an index that contributes most to the current peak to an index that
   contributes little.  The move is kept only when it lowers the score.

4. **Stochastic polishing** – when the time budget is almost exhausted we
   apply decreasing‑amplitude Gaussian perturbations and keep any improvement.

The whole search runs under the provided time budget (default 1000 s) and
returns the best sequence found (list of `float`).  All numbers are clipped
to the interval [0, 1000] and the ℓ¹‑norm is kept at √(2·n) throughout,
satisfying the security checks of `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional imports – the code works without them
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    _HAS_SCIPY = True
except Exception:               # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
def _conv(seq: np.ndarray) -> np.ndarray:
    """Full convolution a*a (real, non‑negative)."""
    return np.convolve(seq, seq)


def _score(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # evaluate_sequence expects a Python list
    return float(evaluate_sequence(seq.tolist()))


def _norm_to_target(seq: np.ndarray) -> np.ndarray:
    """Scale the vector so that Σaᵢ = √(2·n).  Also clip to the allowed range."""
    n = seq.size
    target = math.sqrt(2.0 * n)
    # protect against zero‑sum vectors
    s = np.sum(seq)
    if s == 0.0:
        seq = np.full_like(seq, target / n)
    else:
        seq = seq * (target / s)
    # final clipping (the security checks also do it, but we keep the vector
    # inside the admissible box to avoid surprises)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _linear_move(seq: np.ndarray, epsilon: float = 0.5) -> np.ndarray | None:
    """
    Solve a linear programme that finds a direction d (Σd=0, d_i ≥ -a_i) which
    minimises the linearised peak bound M' = max_k (c_k + 2·Σ_i a_{k‑i} d_i).

    Parameters
    ----------
    seq : np.ndarray
        Current non‑negative sequence (already normalised to target ℓ¹‑norm).
    epsilon : float
        Trust‑region factor – the step size λ will be limited to [0,1] in a
        back‑tracking line search, but the LP itself is unbounded; we keep
        this argument only for possible future bounds.

    Returns
    -------
    np.ndarray | None
        A new sequence `seq + λ·d` that improves the actual score, or `None`
        if the LP fails or no improvement is found.
    """
    n = seq.size
    # compute the current convolution and the linear coefficients
    conv = _conv(seq)
    # build matrix A with shape (2n‑1, n) where A[k,i] = 2·a_{k‑i} if valid
    k_grid = np.arange(2 * n - 1)[:, None]          # (2n‑1,1)
    i_grid = np.arange(n)[None, :]                 # (1,n)
    idx = k_grid - i_grid                          # (2n‑1,n)
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # use advanced indexing for speed
    A[mask] = 2.0 * seq[idx[mask]]

    # ----- linear programme (variables: d_0 … d_{n‑1}, M) -----
    # objective: minimise M  (coefficients 0 … 0, 1)
    c_obj = np.concatenate([np.zeros(n), [1.0]])

    # inequality:  A·d - M ≤ -conv
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1), dtype=np.float64)])
    b_ub = -conv

    # equality: Σ d_i = 0
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([0.0])

    # bounds: d_i ≥ -a_i (to keep a_i + d_i ≥ 0), M ≥ 0
    bounds = [(-seq[i], None) for i in range(n)] + [(0.0, None)]

    if not _HAS_SCIPY:
        return None

    # solve with HiGHS (fast interior‑point/simplex)
    try:
        res = _opt.linprog(
            c=c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception:               # pragma: no cover
        return None

    if not res.success:
        return None

    d = res.x[:n]                    # direction (may be large)
    # back‑tracking line search – keep Σa unchanged, clip afterwards,
    # and finally re‑normalise to the exact ℓ¹‑target.
    cur_score = _score(seq)
    for lam in (1.0, 0.8, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01):
        cand = seq + lam * d
        # clipping to the allowed box
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑normalise – this does not change the sum because Σd = 0,
        # but clipping may have altered it slightly.
        cand = _norm_to_target(cand)
        cand_score = _score(cand)
        if cand_score < cur_score - 1e-9:
            return cand
    # no improvement found
    return None


def _greedy_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """
    One-step greedy mass transfer that tries to reduce the current peak.
    Returns a better sequence or None if no improvement.
    """
    n = seq.size
    conv = _conv(seq)
    k_max = int(np.argmax(conv))
    # contributions of each index to the peak at k_max
    i = np.arange(n)
    j = k_max - i
    valid = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[valid] = seq[i[valid]] * seq[j[valid]]
    total = contrib.sum()
    if total == 0.0:
        return None

    # choose a source index with probability proportional to its contribution
    prob_src = contrib / total
    src = rng.choice(n, p=prob_src)

    # choose a destination where contribution is small
    low_thr = np.percentile(contrib, 30)
    cands = np.where(contrib <= low_thr)[0]
    dst = rng.choice(cands) if cands.size > 0 else rng.integers(n)

    if src == dst:
        return None

    # move up to 30 % of the source mass
    max_move = 0.3 * seq[src]
    if max_move <= 0.0:
        return None
    move = rng.uniform(0.0, max_move)

    cand = seq.copy()
    cand[src] -= move
    cand[dst] += move
    cand = np.clip(cand, 0.0, 1000.0)
    cand = _norm_to_target(cand)
    if _score(cand) < _score(seq) - 1e-9:
        return cand
    return None
