def proj_simplex(v, S):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v, S):
    # ensure v is a float array
    if S < 0:
        raise ValueError
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries zero?
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Edge Cases: If all v are zero, cond will be false; then theta=0 and w=0.

def greedy_peak_reduction(seq, deadline, rng):
    best_seq = seq.copy()
    best_score = compute_score(seq)
    while time.monotonic() < deadline:
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])
        # compute contributions as outer product slice
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(seq)
        contrib[mask] = seq[i[mask]] * seq[j[mask]]
        total = contrib.sum()
        if total == 0: break
        # select i_sel weighted by contrib
        prob = contrib / total
        i_sel = rng.choice(n, p=prob)
        # choose destination low contribution
        low_thresh = np.percentile(contrib, 30)
        low_candidates = np.where(contrib <= low_thresh)[0]
        if low_candidates.size == 0:
            # pick random different index
            j_sel = rng.integers(n)
        else:
            j_sel = rng.choice(low_candidates)
        if i_sel == j_sel:
            continue
        eps_max = seq[i_sel] * 0.3
        if eps_max <= 0:
            continue
        eps = rng.uniform(0, eps_max)
        cand = seq.copy()
        cand[i_sel] -= eps
        cand[j_sel] += eps
        # project onto simplex? Actually mass preserved; after moving, sum unchanged. Non-negative because eps <= seq[i_sel].
        cand = np.clip(cand, 0, 1000)
        cand_score = compute_score(cand)
        if cand_score < best_score - 1e-9:

    return best_seq, best_score

Candidate generator for the auto‑convolution bound.

Algorithm
---------
1.  Initialise with the best known global sequence (if any) or with a
    random Dirichlet vector, normalised to the fixed ℓ¹‑norm
    √(2·n) (the sum is kept constant, therefore the evaluation reduces to the
    maximal value of the auto‑convolution).

2.  **Projected gradient descent on a smooth approximation of the max‑norm.**
    For a current vector `f` we compute its auto‑convolution `c = f*f`.
    The soft‑max weights
        wₖ = exp(β·(cₖ – max(c))) / Σ exp(β·(c – max(c)))
    give a smooth surrogate of the max‑operator.
    The gradient of the surrogate Lβ is
        ∇Lβ = 2·(w * rev(f))[n‑1 : n‑1+n]               (· is convolution)

    simplex {x≥0, Σx = √(2·n)}.  β is increased slowly (to sharpen the
    approximation) while η is adapted based on whether the step improved the
    objective.  Multiple restarts and several vector lengths are tried while
    respecting the global time budget.

3.  **Greedy peak reduction** – after the gradient phase we repeatedly move a
    small fraction of mass from an index that contributes a lot to the current
    maximal convolution entry to an index with little contribution, accepting
    only improving moves.

4.  **Stochastic polishing** – final Gaussian perturbations with an
    exponentially decaying step size, each perturbation being projected back onto
    the feasible simplex.

The routine logs progress, respects the 1000 s budget, and returns the best
sequence found.

    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Work on a copy to avoid side effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_descent(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of the max
    auto‑convolution value.  Returns the best vector encountered and its
    score.
    """
    n = f0.size
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)

    # Hyper‑parameters (chosen experimentally)
    step = 0.5          # initial step size
    beta = 5.0          # initial soft‑max temperature
    max_beta = 80.0
    max_iter = 5000

    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        # ----- compute soft‑max weights and gradient -----
        conv = np.convolve(f, f)
        max_val = conv.max()
        # stable soft‑max (max_val is subtracted)
        exp_vals = np.exp(beta * (conv - max_val))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0:
            # Numerical under‑flow – fall back to uniform weights
            w = np.full_like(exp_vals, 1.0 / exp_vals.size)
        else:
            w = exp_vals / sum_exp

        # Gradient = 2 * (w * rev(f))[n-1 : n-1+n]
        rev_f = f[::-1]
        grad_full = np.convolve(w, rev_f)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- gradient step with projection -----
        cand = f - step * grad
        cand = project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)

        # ----- accept if improvement -----
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_f = cand.copy()
            f = cand
            step *= 1.02         # slightly accelerate
            no_improve = 0
        else:
            # reject, shrink step size
            step *= 0.90
            no_improve += 1

        # ----- schedule -----
        beta = min(max_beta, beta * 1.01)

        # occasional logging
        if it % 200 == 0:
            print(
                f"[gd] n={n} it={it} best={best_score:.6f} step={step:.4f} beta={beta:.2f}"
            )

        # early exit if target already reached
        if best_score < 1.503:
            print(f"[gd] target reached (score={best_score:.6f})")
            break

        # occasional random shake‑up to escape flat regions
        if no_improve >= 30:
            shake = rng.normal(scale=0.02, size=n)
            f = project_onto_simplex(f + shake, target_sum)
            no_improve = 0

    return best_f, best_score


def greedy_peak_reduction(
    seq: np.ndarray, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a modest amount of mass from indices that heavily
    contribute to the current maximal convolution entry to low‑contributing
    indices.  Accept only improving moves.
    """
    n = seq.size
    best_seq = seq.copy()
    best_score = _score(best_seq)

    while time.monotonic() < deadline:
        conv = np.convolve(best_seq, best_seq)
        max_idx = int(np.argmax(conv))

        # contribution of each index to the peak at max_idx
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(best_seq)
        contrib[mask] = best_seq[i[mask]] * best_seq[j[mask]]

        total = contrib.sum()
        if total == 0.0:
            break

        # pick source index weighted by contribution
        prob_src = contrib / total
        src = rng.choice(n, p=prob_src)

        # pick a destination with low contribution (30‑th percentile)
        low_thr = np.percentile(contrib, 30)
        low_cands = np.where(contrib <= low_thr)[0]
        if low_cands.size == 0:
            dst = rng.integers(n)
        else:
            dst = rng.choice(low_cands)

        if src == dst:
            continue

        # move up to 30 % of the source mass
        eps_max = best_seq[src] * 0.30
        if eps_max <= 0.0:
            continue
        eps = rng.uniform(0.0, eps_max)

        cand = best_seq.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _score(cand)
        if cand_score < best_score - 1e-9:
            best_seq = cand
            best_score = cand_score
            print(f"[greedy] improved to {best_score:.6f}")

        if best_score < 1.503:
            print("[greedy] target reached")
            break

    return best_seq, best_score


def stochastic_polishing(
    seq: np.ndarray, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Final local search: Gaussian perturbations with exponentially
    decreasing step size, each perturbation projected back onto the feasible
    simplex.
    """
    n = seq.size
    target_sum = math.sqrt(2.0 * n)
    cur = seq.copy()
    best_score = _score(cur)
    step = 0.4

    while time.monotonic() < deadline:
        cand = cur + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)
        if cand_score < best_score:
            cur = cand
            best_score = cand_score
            print(f"[polish] improved to {best_score:.6f}")

        step *= 0.9997
        if step < 1e-6:
            step = 0.4

        if best_score < 1.503:
            print("[polish] target reached")
            break

    return cur, best_score

def propose_candidate(
    seed: int = 0,
    budget_s: float = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step sequence with a low auto‑convolution
    bound.  The routine respects the global time budget, uses several
    heuristic stages (gradient descent, greedy peak reduction, stochastic
    polishing) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the caller
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from the globally available sequence (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.array(height_sequence_1, dtype=float)
            n0 = seq0.size
            target0 = math.sqrt(2.0 * n0)
            if seq0.sum() > 0:
                seq0 = np.clip(seq0 * (target0 / seq0.sum()), 0.0, 1000.0)
            sc0 = _score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] global height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception as e:  # pragma: no cover
            print("[init] could not use global height_sequence_1:", e)

    # ------------------------------------------------------------------
    # Define lengths to explore (starting with the current best length)
    # ------------------------------------------------------------------
    lengths = []
    if best_seq is not None:
        lengths.append(best_seq.size)
    # additional lengths – increasing flexibility
    lengths.extend([400, 500, 600, 800, 1000, 1500, 2000])
    # ensure uniqueness while preserving order
    seen = set()
    lengths = [x for x in lengths if not (x in seen or seen.add(x))]

    # ------------------------------------------------------------------
    # Phase 1 – projected gradient descent on several lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        restarts = 4 if n <= 800 else 2

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise the vector
            if best_seq is not None and best_seq.size == n:
                f0 = best_seq.copy()
            else:
                # Dirichlet‑like random start (adds a slight bias toward non‑zero entries)
                f0 = rng.random(n) + 0.1
                f0 = f0 / f0.sum() * target_sum

            f_opt, sc_opt = gradient_descent(f0, target_sum, deadline, rng)

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = f_opt.copy()
                print(
                    f"[phase1] n={n} restart={rs} improved score={best_score:.6f}"
                )

            # early exit if we already beat the target
            if best_score < 1.503:
                print("[phase1] target reached – exiting early")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – greedy peak reduction (fine‑tuning)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector of length 400
        n = 400
        best_seq = np.full(n, 1.0, dtype=float)
        best_seq = best_seq / best_seq.sum() * math.sqrt(2.0 * n)

    best_seq, best_score = greedy_peak_reduction(best_seq, deadline, rng)

    # ------------------------------------------------------------------
    # Phase 3 – stochastic polishing (local search)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        best_seq, best_score = stochastic_polishing(best_seq, deadline, rng)

    print(
        f"[final] best length={best_seq.size} score={best_score:.6f}"
    )
    # Return a plain Python list of floats
    return [float(x) for x in best_seq]
