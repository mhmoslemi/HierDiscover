def _normalize(seq, target):
    total = seq.sum()
    if total > 0:

def _fast_score(seq):
    conv = np.convolve(seq, seq)

def generate_candidates(n, target, rng):
    candidates = []
    # uniform
    a = np.full(n, target/n)
    candidates.append(a)
    # triangular
    x = np.arange(n, dtype=float)
    t = 1 - np.abs(x - (n-1)/2) / ((n-1)/2)
    a = t
    a[a<0] = 0
    a = a / a.sum() * target
    candidates.append(a)
    # power law
    idx = np.arange(1, n+1, dtype=float)
    for p in np.linspace(0.1, 3.0, 20):
        a = idx ** (-p)
        a = a / a.sum() * target
        candidates.append(a)
    # exponential
    for alpha in np.linspace(0.001, 0.2, 15):
        a = np.exp(-alpha * np.arange(n, dtype=float))
        a = a / a.sum() * target
        candidates.append(a)
    # beta shapes: generate a,b random
    for _ in range(10):
        a_param = rng.uniform(0.2,5.0)
        b_param = rng.uniform(0.2,5.0)
        x = np.linspace(0,1,n)
        a_vec = (x**(a_param-1))*((1-x)**(b_param-1))

        a = a / a.sum() * target
        candidates.append(a)
    # random Dirichlet
    for _ in range(5):
        a = rng.dirichlet(np.ones(n)) * target
        candidates.append(a)

def refine(seq, target, deadline, rng):
    best = seq.copy()
    best_score = _fast_score(best)
    n = len(seq)
    # Inlined use function to compute objective = max_b
    while time.monotonic() < deadline:
        # gradient step approach
        conv = np.convolve(best, best)
        max_val = conv.max()
        k = int(conv.argmax())
        # range of i
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)

        # contributions for each i in range
        idxs = np.arange(i_start, i_end+1)
        partners = best[k - idxs]
        contributions = best[idxs] * partners

        # donor index with largest contribution
        donor = idxs[np.argmax(contributions)]
        # choose receiver index that minimizes partner value
        # we can consider globally minimal partner
        # compute partner for donor and for all positions; but use global min a_i
        # Actually we need a partner for candidate receiver to compute effect on max, should pick index where partner is minimal for this k
        # compute partner values for all positions 0..n-1: partner_all = ... wait partner for position j in k offset is best[k - j] if valid else 0
        # For indices outside range i_start..i_end, they are not part of this convolution term; moving mass there does not affect current max.
        # So we can pick receiver outside the range, making no increase in current max. That's good.
        # Choose any index j not in idxs, maybe where best[j] is minimal.
        outside = np.setdiff1d(np.arange(n), idxs)
        if outside.size > 0:
            receiver = int(outside[np.argmin(best[outside])])
        else:
            # fallback: choose index with min partner within range
            receiver = int(idxs[np.argmin(partners)])
        # ensure donor != receiver
        if donor == receiver:
            # choose another random index
            receiver = int(rng.integers(n))
            if receiver == donor:
                continue

        # epsilon selection
        # compute potential reduction factor
        partner_donor = best[k - donor]
        if 0 <= k - receiver < n:
            partner_receiver = best[k - receiver]
        else:
            partner_receiver = 0.0
        if partner_donor <= partner_receiver:
            # cannot reduce max via this move; skip to random small move
            # random small perturb
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            eps = min(best[i] * 0.3, target * 0.001)
            if eps <= 0:
                continue
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
        else:
            # choose epsilon based on donor amount and partner difference
            # typical step: reduce donor by fraction 0.2 of its mass or 0.3, whichever smaller
            max_eps = best[donor] * 0.2
            # also limited by partner diff sensitivity: we can compute required eps to reduce M by fraction.
            eps = min(max_eps, target * 0.001)  # ensure small
            cand = best.copy()
            cand[donor] -= eps
            cand[receiver] += eps

        # enforce non-negative and clipping
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target)

        cand_score = _fast_score(cand)
        if cand_score < best_score - 1e-9:
            best, best_score = cand, cand_score
            continue

        # gradient descent fallback (direction from subgradient)
        # compute subgradient g at active k (derivative of b_k)
        g = np.zeros(n)
        for i in range(i_start, i_end+1):
            j = k - i
            g[i] = best[j]
        # direction -g + mean(g)
        mean_g = g.mean()
        direction = -g + mean_g
        # step size bound to keep non-negative
        neg = direction < 0
        if np.any(neg):
            max_alpha = np.min(best[neg] / -direction[neg])
        else:
            max_alpha = 1.0
        # pick small alpha
        alpha = min(0.1 * max_alpha, 0.05 * target / n)
        cand = best + alpha * direction
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target)
        cand_score = _fast_score(cand)
        if cand_score < best_score - 1e-9:
            best, best_score = cand, cand_score
            continue

        # occasional smoothing
        if rng.random() < 0.05:
            kernel = np.array([0.25,0.5,0.25])
            smooth = np.convolve(best, kernel, mode='same')
            # enforce non-negative, clip
            smooth = np.clip(smooth, 0.0, 1000.0)
            smooth = _normalize(smooth, target)
            smooth_score = _fast_score(smooth)
            if smooth_score < best_score:
                best, best_score = smooth, smooth_score
                continue

        # if none improved, maybe random restart
        # occasional random descent
        if rng.random() < 0.01:
            # random dirichlet
            cand = rng.dirichlet(np.ones(n)) * target
            cand_score = _fast_score(cand)
            if cand_score < best_score:
                best, best_score = cand, cand_score

def swap_hill_climb(seq, target, deadline, rng):
    best = seq.copy()
    best_score = _fast_score(best)
    n = len(seq)
    while time.monotonic() < deadline:
        # choose donor index with >0 mass and receiver index
        donors = np.where(best > 0)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _normalize(cand, target)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
    return best

Improved contender for the step‑function minimisation challenge.

Algorithm outline
-----------------
1️⃣ **Rich parametric sweep** –  for each candidate length we generate many
    analytic families (uniform, triangular, power‑law, exponential,
    Beta‑distribution, arcsine (Beta(½,½)) and a few random Dirichlet draws).
    Every candidate is L¹‑normalised to √(2·n); the best‑scoring one is kept.

2️⃣ **Targeted refinement** –  starting from the best vector we repeatedly
    *identify the convolution index k where the auto‑convolution attains its
    maximum* and perform a *mass‑transfer* step:
        - move a small amount of weight from the index that contributes most
          to the active entry to an index that contributes least (or lies
          outside the active support, thus not increasing that entry);
        - if this does not decrease the objective we fall back to a cheap
          projected gradient step on the sub‑gradient of the active entry;
        - occasional smoothing (0.25/0.5/0.25 kernel) and random restarts keep
          the search from stagnating.

    The loop stops when the time budget expires.

3️⃣ **Fine‑grained stochastic polishing** – a cheap swap‑based hill‑climb
    followed by a short Gaussian perturbation stage polish the final vector
    to squeeze out the last bit of improvement.

def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``vec`` to have ℓ¹‑norm exactly ``target_sum`` (preserving non‑negativity)."""
    s = vec.sum()
    if s <= 0.0:
        # fall back to uniform distribution
        n = vec.shape[0]
        return np.full(n, target_sum / n, dtype=np.float64)
    return vec * (target_sum / s)


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` but without the safety wrapper.
    Returns the value 2·n·max(conv)/(sum(seq)²).  For sequences that are
    ℓ¹‑normalised to √(2·n) this equals the maximal auto‑convolution entry.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


def _beta_family(n: int, a: float, b: float, target_sum: float) -> np.ndarray:
    """Beta‑distribution shape a_i ∝ x^{a‑1} (1‑x)^{b‑1},  x∈(0,1)."""
    # shift a tiny bit away from the boundaries to avoid infinities
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    w = np.maximum(w, 1e-12)      # protect against exact zeros
    w = w / w.sum() * target_sum
    return w


def _generate_candidates(n: int, target_sum: float, rng: np.random.Generator):
    """Return a list of plausible start vectors for length ``n``."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, target_sum / n, dtype=np.float64))

    # 2. simple triangular (peak in the centre)
    x = np.arange(n, dtype=np.float64)
    tri = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
    tri[tri < 0] = 0.0
    cand.append(_normalize(tri, target_sum))

    # 3. power‑law a_i ∝ i⁻ᵖ   (i starts at 1)
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.1, 3.0, 20):
        a = idx ** (-p)
        cand.append(_normalize(a, target_sum))

    # 4. exponential a_i ∝ e^{-α·i}
    for alpha in np.linspace(0.001, 0.20, 15):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand.append(_normalize(a, target_sum))

    # 5. beta family, broad range of parameters (including <1)
    for _ in range(12):
        a_param = rng.uniform(0.1, 5.0)
        b_param = rng.uniform(0.1, 5.0)
        cand.append(_beta_family(n, a_param, b_param, target_sum))

    # 6. explicit arcsine shape (Beta(½,½))
    cand.append(_beta_family(n, 0.5, 0.5, target_sum))

    # 7. sine‑powered shapes a_i ∝ sin^p(π·i/(n+1))
    for p in np.linspace(0.5, 3.0, 10):
        a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1))) ** p
        cand.append(_normalize(a, target_sum))

    # 8. a few random Dirichlet draws (always respect the ℓ¹‑norm)
    for _ in range(5):
        a = rng.dirichlet(np.ones(n)) * target_sum
        cand.append(a)

    return cand

def _refine_sequence(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 5000,
) -> np.ndarray:
    """
    Heuristic that repeatedly reduces the maximum entry of the auto‑convolution.
    The method alternates between a focussed *mass‑transfer* move (donor↔receiver)
    and a cheap projected gradient step on the active convolution entry.
    """
    n = seq.shape[0]
    best = _normalize(seq, target_sum)
    best_sc = _fast_score(best)

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        conv = np.convolve(best, best)
        max_val = conv.max()
        k = int(conv.argmax())

        # --- indices that participate in the active convolution entry ---
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1, dtype=np.int64)

        # contribution of each index to the active entry
        partners = best[k - idxs]                 # a_{k-i}
        contributions = best[idxs] * partners      # a_i * a_{k-i}

        donor = idxs[np.argmax(contributions)]

        # try to pick a receiver that *does not* increase the active entry
        # (i.e. an index outside the active support, preferably with tiny mass)
        outside = np.setdiff1d(np.arange(n, dtype=np.int64), idxs, assume_unique=True)
        if outside.size > 0:
            receiver = int(outside[np.argmin(best[outside])])
        else:
            # fall back to the index with smallest partner value within the support
            receiver = int(idxs[np.argmin(partners)])

        if donor == receiver:
            # degenerate – pick a random different index
            receiver = int(rng.integers(n))
            if receiver == donor:
                continue

        # how much mass can safely be moved?
        max_move = min(best[donor] * 0.4, target_sum * 0.002)
        if max_move <= 0.0:
            continue

        # check if the move is *promising* for the active entry
        partner_donor = best[k - donor]
        if 0 <= k - receiver < n:
            partner_receiver = best[k - receiver]
        else:
            partner_receiver = 0.0

        if partner_donor > partner_receiver:
            eps = max_move
            cand = best.copy()
            cand[donor] -= eps
            cand[receiver] += eps
        else:
            # no clear gain – try a tiny random redistribution instead
            i = int(rng.choice(np.where(best > 0)[0]))
            j = int(rng.integers(n))
            eps = min(best[i] * 0.2, target_sum * 0.001)
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps

        # re‑clip and re‑normalise
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc - 1e-12:
            best, best_sc = cand, cand_sc
            continue

        # --- fallback: one projected gradient step on the active entry ---

        # gradient of b_k w.r.t. a (for the active k)
        grad = np.zeros(n, dtype=np.float64)
        for i in range(i_start, i_end + 1):
            j = k - i
            grad[i] = best[j]

        # direction that keeps the ℓ¹‑norm fixed (subtract mean)
        direction = -grad + grad.mean()

        # feasible step size (stay non‑negative)
        neg = direction < 0
        if np.any(neg):
            max_alpha = np.min(best[neg] / -direction[neg])
        else:
            max_alpha = 1.0

        # a modest step (the exact size is tuned empirically)
        alpha = min(0.08 * max_alpha, 0.04 * target_sum / n)

        cand = best + alpha * direction
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc - 1e-12:
            best, best_sc = cand, cand_sc
            continue

        # occasional smoothing to erase tiny spikes
        if rng.random() < 0.04:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(best, kernel, mode="same")
            smooth = np.clip(smooth, 0.0, 1000.0)
            smooth = _normalize(smooth, target_sum)
            smooth_sc = _fast_score(smooth)
            if smooth_sc < best_sc:
                best, best_sc = smooth, smooth_sc
                continue

        # if nothing worked, inject a fresh random Dirichlet candidate (rarely)
        if rng.random() < 0.01:
            cand = rng.dirichlet(np.ones(n)) * target_sum
            cand_sc = _fast_score(cand)
            if cand_sc < best_sc:
                best, best_sc = cand, cand_sc
                continue

    return best


# ----------------------------------------------------------------------
# Final stochastic polishing
# ----------------------------------------------------------------------


def _swap_hill_climb(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """Very cheap hill‑climb that swaps a small delta between two indices."""
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target_sum * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
    return best


def _gaussian_local_search(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """Short Gaussian perturbation loop (annealing‑style)."""
    best = seq.copy()
    best_sc = _fast_score(best)
    step = target_sum * 0.02

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape[0])
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
            step *= 0.9
        else:
            step *= 0.99
            if step < 1e-6:
                step = target_sum * 0.02
    return best


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  The routine stops a little before the limit
        to avoid a hard timeout.
    kwargs : dict
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.3)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Initialise from a previously discovered construction, if any.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = _normalize(init, target0)
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception:  # pragma: no cover
            pass

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep over several lengths.
    # --------------------------------------------------------------
    # The lengths are ordered from large to small – larger n gives more flexibility.
    lengths = [1000, 800, 600, 500, 400, 300, 200]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # generate a modest fleet of candidates
        candidates = _generate_candidates(n, target, rng)

        # quick evaluation of the fleet
        for cand in candidates:
            if time.monotonic() >= deadline:
                break
            sc = _fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"[sweep] n={n} new best {sc:.6f}")

        # --------------------------------------------------------------
        # 2️⃣  Focused refinement for the current length.
        # --------------------------------------------------------------
        # allocate a slice of the remaining budget to this length
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break
        # at most 30 s per length, or the remaining time if smaller
        refine_deadline = time.monotonic() + min(30.0, remaining * 0.5)

        # start point: either the global best of this length or a fresh Dirichlet draw
        if best_seq is not None and best_seq.shape[0] == n:
            start = best_seq.copy()
        else:
            start = rng.dirichlet(np.ones(n)) * target

        refined = _refine_sequence(start, target, refine_deadline, rng)
        refined_sc = _fast_score(refined)
        if refined_sc < best_score:
            best_seq, best_score = refined.copy(), refined_sc
            print(f"[refine] n={n} improved to {refined_sc:.6f}")

        # a couple of random restarts (very cheap) – can discover a completely
        # different basin of attraction
        for restart in range(2):
            if time.monotonic() >= deadline:
                break
            start = rng.dirichlet(np.ones(n)) * target
            sub_deadline = time.monotonic() + min(12.0, deadline - time.monotonic())
            refined = _refine_sequence(start, target, sub_deadline, rng)
            refined_sc = _fast_score(refined)
            if refined_sc < best_score:
                best_seq, best_score = refined.copy(), refined_sc
                print(f"[restart] n={n} run {restart} new best {refined_sc:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final stochastic polishing on the globally best vector.
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: a modest uniform vector
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # ensure the stored best respects the precise ℓ¹‑norm (tiny drift may have occurred)
    n = best_seq.shape[0]
    target = math.sqrt(2.0 * n)
    best_seq = _normalize(best_seq, target)

    # cheap swap‑based hill‑climb
    best_seq = _swap_hill_climb(best_seq, target, deadline, rng)

    # short Gaussian hill‑climb (if time permits)
    if time.monotonic() < deadline:
        best_seq = _gaussian_local_search(best_seq, target, deadline, rng)

    final_score = _fast_score(best_seq)
    print(f"[final] best score {final_score:.6f}   length {best_seq.shape[0]}")

    # return a plain Python list
    return [float(x) for x in best_seq]
