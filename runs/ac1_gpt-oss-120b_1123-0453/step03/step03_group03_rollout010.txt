def _gradient_step(seq, target_sum, lr):
    n = seq.size
    conv = np.convolve(seq, seq)
    M = conv.max()
    # tolerance
    tol = max(1e-12, 1e-12 * M)
    d = (conv >= M - tol).astype(float)
    grad = 2.0 * np.convolve(d, seq)[:n]
    # compute update magnitude
    step = lr * target_sum / n  # step per unit of gradient
    new_seq = seq - step * grad
    # enforce constraints
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    # renormalise
    total = new_seq.sum()
    if total < 0.01:
        # fallback uniform
        new_seq = np.full_like(seq, target_sum / n)
    else:
        new_seq = new_seq * (target_sum / total)
    return new_seq, M

"""
Gradient‑descent optimiser for the auto‑convolution bound.

The search proceeds in three phases inside the allotted time:

1️⃣  Initialise a non‑negative sequence of a chosen length (default 2000).
    If a previous good sequence exists (global `height_sequence_1`) it is
    interpolated to the target length, otherwise a random start is used.
    The vector is normalised so that its ℓ¹‑norm equals √(2·n); this makes the
    evaluation equal to the maximal entry of the auto‑convolution.

2️⃣  Projected gradient descent on the objective  maxₖ (a∗a)ₖ.
    The gradient of the max‑entry is computed analytically:
        ∂/∂aⱼ maxₖ (a∗a)ₖ = 2·∑_{k∈K} a_{k‑j} ,
    where K is the (small) set of indices attaining the current maximum.
    After a step we clip to [0,1000] and re‑scale to keep the ℓ¹‑norm fixed.
    A simple back‑tracking on the learning rate and occasional random restarts
    avoid stagnation.

3️⃣  Stochastic polishing: Gaussian perturbations with an exponentially
    decaying standard deviation are applied; any improvement is kept.
    This phase finishes the search with a quick local refinement.

The routine prints progress information and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score_seq(seq: np.ndarray) -> float:
    """Exact replica of ``evaluate_sequence`` (without the safety clipping)."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def _clip_and_rescale(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to the allowed interval and enforce the required ℓ¹‑norm."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:                      # fall back to a uniform vector
        seq = np.full_like(seq, target_sum / seq.size)
        total = seq.sum()
    seq = seq * (target_sum / total)
    return seq


def _initial_seq(
    n: int,
    rng: np.random.Generator,
    seed_seq: np.ndarray | None = None,
) -> np.ndarray:
    """Create a starting vector of length ``n``.

    If ``seed_seq`` is provided it is linearly interpolated to the new length,
    otherwise a random positive vector is generated.
    """
    if seed_seq is not None:
        old_n = seed_seq.size
        if old_n == n:
            seq = seed_seq.copy()
        else:
            # linear interpolation to the new grid
            x_old = np.arange(old_n, dtype=float)
            x_new = np.linspace(0.0, old_n - 1.0, n)
            seq = np.interp(x_new, x_old, seed_seq)
        # small random jitter to prevent being stuck on a saddle point
        jitter = rng.normal(scale=0.01 * np.mean(seq), size=n)
        seq = seq + jitter
    else:
        seq = rng.random(n) + 0.1          # avoid exact zeros
    return seq


def _gradient_step(
    seq: np.ndarray,
    target_sum: float,
    lr: float,
) -> tuple[np.ndarray, float]:
    """One projected gradient step aimed at reducing max(conv(seq,seq)).

    Returns the updated sequence and the current maximal convolution entry.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    M = conv.max()

    # indices where the maximum is attained (a tiny tolerance avoids
    # numerical artefacts)
    tol = 1e-12
    max_idxs = np.where(conv >= M - tol)[0]

    # analytic gradient:  ∂M/∂a_j = 2·∑_{k∈K} a_{k‑j}
    grad = np.zeros_like(seq)
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] += 2.0 * seq[k - i_range]

    # a simple step size: lr·(target_sum / n) gives a comparable update
    step = lr * target_sum / n
    new_seq = seq - step * grad
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    new_seq = _clip_and_rescale(new_seq, target_sum)
    return new_seq, M


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Overall time budget in seconds (the function will stop a little early).
    kwargs : dict
        Ignored – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # reserve a small safety margin for the caller
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Choose a working length.
    # Larger lengths give more freedom to flatten the convolution.
    # ------------------------------------------------------------------
    n = 2000                     # default length; can be tuned if time is tight
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Prepare a seed sequence (global variable if present)
    # ------------------------------------------------------------------
    seed_seq = None
    if "height_sequence_1" in globals():
        try:
            seed_seq = np.array(height_sequence_1, dtype=float)
        except Exception:
            seed_seq = None

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop – several random restarts with gradient descent
    # ------------------------------------------------------------------
    restarts = 5
    for restart in range(restarts):
        if time.monotonic() >= deadline:
            break

        # initialise (interpolated) sequence for this restart
        seq = _initial_seq(n, rng, seed_seq)
        seq = _clip_and_rescale(seq, target_sum)

        cur_score = _score_seq(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[init] restart={restart} score={cur_score:.6f}")

        # gradient descent parameters
        lr = 5.0                     # initial learning rate
        decay = 0.995                # multiplicative decay when no improvement
        min_lr = 0.01
        no_improve = 0
        max_no_improve = 200        # after this many idle steps we restart

        it = 0
        while time.monotonic() < deadline:
            it += 1
            seq, M = _gradient_step(seq, target_sum, lr)
            cur_score = _score_seq(seq)

            if cur_score < best_score - 1e-7:
                best_score = cur_score
                best_seq = seq.copy()
                print(
                    f"[grad] restart={restart} it={it} "
                    f"score={cur_score:.6f} M={M:.6f} lr={lr:.4f}"
                )
                # reset stagnation counter and give the learning rate a tiny boost
                no_improve = 0
                lr = max(lr, 1.0)
            else:
                no_improve += 1
                lr *= decay
                if lr < min_lr or no_improve >= max_no_improve:
                    # restart from a fresh random point to escape a plateau
                    seq = _initial_seq(n, rng, seed_seq)
                    seq = _clip_and_rescale(seq, target_sum)
                    lr = 5.0
                    no_improve = 0
                    print(f"[restart] restart={restart} it={it}")

            # early exit if we have already reached a very good bound
            if best_score < 1.2:
                print("[grad] early exit – bound sufficiently low")
                break

    # ------------------------------------------------------------------
    # Phase 3 – stochastic polishing (Gaussian perturbations)
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback (uniform vector)
        best_seq = np.full(n, target_sum / n, dtype=float)

    cur = best_seq.copy()
    sigma = max(0.1 * cur.mean(), 1e-3)    # initial perturbation scale
    while time.monotonic() < deadline:
        cand = cur + rng.normal(scale=sigma, size=n)
        cand = _clip_and_rescale(cand, target_sum)
        cand_score = _score_seq(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            cur = cand
            print(f"[polish] improved to {cand_score:.6f} sigma={sigma:.6f}")
        sigma *= 0.9995                     # exponential decay
        if sigma < 1e-6:
            sigma = max(0.1 * cur.mean(), 1e-3)

    print(f"[final] best_score={best_score:.6f} length={n}")
    return [float(x) for x in best_seq]
