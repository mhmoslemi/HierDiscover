def project_onto_simplex(v, z):
    # v is vector might have negative values; try to find w = argmin ||w - v||^2 s.t. w>=0, sum w = z.
    # algorithm (Duchi et al.):

        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - z
        rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)

- return w.

    size = 1 << ((len(x) + len(y) - 1).bit_length()) # next power of 2

    fy = np.fft.rfft(y, n=size)
    return np.fft.irfft(fx*fy, n=size)[:len(x)+len(y)-1]

Gradient‑descent based flattening of the auto‑convolution.

We treat the normalized vector a (ℓ1‑norm = √(2·n)) as a probability
distribution (after division by the sum).  The evaluation ratio reduces
to max_k (a*a)[k]; the goal is to make the convolution as flat as
possible.  The code runs a smoothed‑max optimisation: for a fixed
temperature β we minimise ϕβ(a)= (1/β)·log∑ₖ exp(β·(a∗a)[k]).

simplex (non‑negative entries with prescribed ℓ1‑norm).  The temperature

import time, math

def _project_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum x = z}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    # find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution(a: np.ndarray) -> float:
    """Exact max‑entry of the auto‑convolution of a."""
    return np.convolve(a, a).max()

def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return soft‑max weights wₖ ∝ exp(β·convₖ)."""
    # lock the scale for numerical stability
    shift = conv.max()
    ex = np.exp(beta * (conv - shift))
    sum_ex = ex.sum()
    if sum_ex == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return ex / sum_ex

def _gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """Gradient of the smoothed max w.r.t. a (∇ϕβ)."""
    # grad = 2 * (weights ⋆ a); we obtain it via conv with reversed a
    cross = np.convolve(weights, a[::-1], mode='full')

    return grad

    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray) -> float:
    """Exact maximal entry of the auto‑convolution of a."""
    return np.convolve(a, a).max()


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights wₖ ∝ exp(β·convₖ) (numerically stable)."""
    shift = conv.max()
    ex = np.exp(beta * (conv - shift))
    s = ex.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return ex / s


def _gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """∇Φβ(a) = 2·(w ⋆ a) computed via convolution with a reversed."""
    cross = np.convolve(w, a[::-1], mode='full')
    n = a.shape[0]
    return 2.0 * cross[n - 1 : n - 1 + n]


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence`` score.

    The routine combines a smoothed‑max gradient descent (with annealing),
    a few deterministic and random starts, and a final cheap stochastic
    polishing phase.  It respects the 1000 s time limit and returns the
    best vector found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Lengths to try – larger n gives a lower theoretical lower bound.
    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 1600, 2000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1️⃣  Build a modest pool of initial vectors.
        # --------------------------------------------------------------
        init_vectors = []

        idx = np.arange(1, n + 1, dtype=np.float64)

        # power‑law family  aᵢ ∝ i^{‑p}
        for p in np.linspace(0.2, 4.0, 25):
            a = idx ** (-p)
            a = a / a.sum() * target_sum
            init_vectors.append(a)

        # exponential family  aᵢ ∝ exp(‑α·i)
        for alpha in np.linspace(0.001, 0.25, 25):
            a = np.exp(-alpha * np.arange(n, dtype=np.float64))
            a = a / a.sum() * target_sum
            init_vectors.append(a)

        # a few purely random Dirichlet draws
        for _ in range(5):
            a = rng.random(n) + 0.01
            a = a / a.sum() * target_sum
            init_vectors.append(a)

        # --------------------------------------------------------------
        # 2️⃣  Smoothed‑max gradient descent from each start.
        # --------------------------------------------------------------
        for start_id, start_vec in enumerate(init_vectors):
            if time.monotonic() >= deadline:
                break

            a = start_vec.copy()
            # enforce (temporary) symmetry – helps flatten the convolution
            a = (a + a[::-1]) * 0.5

            beta = 0.5          # initial temperature
            beta_max = 30.0
            beta_mul = 1.30     # multiplicative increase every few iterations
            lr_factor = 0.5     # roughly 0.5 / ||∇|| step length
            max_iters = 2500

            for it in range(max_iters):
                if time.monotonic() >= deadline:
                    break

                cur_score = _max_convolution(a)
                if cur_score < best_score:
                    best_seq, best_score = a.copy(), cur_score
                    print(f"[{n}] start={start_id} it={it} score={cur_score:.6f}")

                # anneal temperature gradually
                if it % 30 == 0 and beta < beta_max:
                    beta = min(beta_max, beta * beta_mul)

                conv = np.convolve(a, a)
                w = _smooth_weights(conv, beta)
                g = _gradient(a, w)

                norm = np.linalg.norm(g)
                if norm == 0.0:
                    break
                step = lr_factor / (norm + 1e-12)

                a_new = a - step * g
                # restore symmetry (keeps the problem balanced)
                a_new = (a_new + a_new[::-1]) * 0.5
                a = _project_simplex(a_new, target_sum)

                # occasional cheap projection to keep numerical errors tiny
                if it % 50 == 0:
                    a = _project_simplex(a, target_sum)

        # --------------------------------------------------------------
        # 3️⃣  Local stochastic polishing for the best vector of this length.
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            a = best_seq.copy()
            # ----- swap‑based hill climbing (tiny mass moves) -----
            swaps_without_improve = 0
            while time.monotonic() < deadline and swaps_without_improve < 1500:
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                delta = a[i] * 0.03
                if delta <= 0.0:
                    continue
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, target_sum)
                cand_score = _max_convolution(cand)
                if cand_score < best_score:
                    a = cand
                    best_seq, best_score = cand.copy(), cand_score
                    swaps_without_improve = 0
                else:
                    swaps_without_improve += 1

            # ----- short Gaussian perturbation phase -----
            step = 0.2 * target_sum / n
            while time.monotonic() < deadline:
                cand = a + rng.normal(scale=step, size=n)
                cand = np.clip(cand, 0.0, 1000.0)
                cand = _project_simplex(cand, target_sum)
                cand_score = _max_convolution(cand)
                if cand_score < best_score:
                    a = cand
                    best_seq, best_score = cand.copy(), cand_score
                    step *= 0.95
                else:
                    step *= 0.992
                if step < 1e-7:
                    break

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 400
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _max_convolution(best_seq)

    # Clip to the range required by ``evaluate_sequence`` (won’t change the score)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return as a plain Python list.
    return [float(x) for x in best_seq]
