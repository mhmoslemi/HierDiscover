def convolve(a):
    n = a.shape[0]
    size = 1
    while size < 2*n-1:
        size <<=1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)
    conv = conv[:2*n-1]  # real part
    return conv

Projection onto simplex:

def proj_simplex(v, s=1.0):
    """Project onto the simplex {x>=0, sum x = s}."""
    # algorithm from https://doi.org/10.1.1.470.7127
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    """Projection of v onto the probability simplex scaled by s."""
    # Ensure v is 1‑D np.ndarray of floats

Sub‑gradient optimiser for the auto‑convolution bound.

We work with the ℓ¹‑normalisation used by ``evaluate_sequence``:
   Σ a_i = √(2·n)
so the evaluation equals the maximal entry of a*a.
The algorithm starts from the global variable ``height_sequence_1`` (if it
exists) or from a uniform vector, then repeatedly:
  • computes the index k* where the convolution attains its maximum,
  • builds a sub‑gradient of the max‑functional at k*,
  • takes a few trial steps in the opposite direction, projecting each
    trial back onto the simplex (non‑negative, prescribed ℓ¹‑norm),
  • keeps the best trial if it improves the peak.
If no improvement is found a small Gaussian jitter is applied to escape
local minima.  A final polishing phase with vanishing Gaussian steps
finishes the search before the time budget expires.
All numbers stay non‑negative and well below the clipping bound 1000.
"""

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x ∈ ℝⁿ | x_i ≥ 0 , Σ x_i = s}.
    Implementation follows the O(n) algorithm from
    Duchi et al., ICML 2008.
    """
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the equal‑mass vector
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full auto‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.

        size <<= 1
