def _project_onto_simplex(v, z=target_sum):
    """Project v onto simplex {x >= 0, sum x = z}."""
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    # find rho where condition holds
    rho = ind[cond][-1]  # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
A fast projected sub‑gradient optimizer for the “max self‑convolution” problem.

The objective (after normalising the sum of the coefficients to √(2 n))
is simply   M = max_k (f∗f)[k] .   This is a convex, Lipschitz function.
We iteratively

1. locate the index k where the convolution attains its maximum,
2. compute a sub‑gradient of M at the current f,
3. take a step opposite to the sub‑gradient,
4. project the result back onto the simplex
   { x ≥ 0 , Σ x = √(2 n) }.

The projection is performed with the O(n log n) algorithm of
Duchi‑Shalev‑Shwartz‑Singer‑Chandra (2008).  An exponential‑time‑budget
guard makes sure we stop before the supplied 1000 s run out.
Random Dirichlet restarts and occasional tiny Gaussian perturbations
help escaping plateaus.  The best sequence seen over all lengths and
restarts is returned.

The code is pure NumPy, has no closures, and respects the competition
rules.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _max_conv_and_index(f: np.ndarray):
    """
    Return (M, idx) where M = max_k (f∗f)[k] and idx is a maximising index.
    The convolution is computed by FFT (O(n log n)).
    """
    n = f.shape[0]
    # next power of two ≥ 2n‑1 for efficient FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    # real FFT – faster than complex for real data
    fft_f = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(fft_f * fft_f, n=size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    M = float(conv[idx])
    return M, idx


def _gradient_at_max(f: np.ndarray, k: int):
    """
    Sub‑gradient of max_k (f∗f)[k] at the active index k.
    For the auto‑convolution the gradient is
        g_i = 2 f_{k‑i}   (if 0 ≤ k‑i < n)  else 0.
    """
    n = f.shape[0]
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=np.intp)
    grad[idx] = 2.0 * f[k - idx]
    return grad


def _project_onto_simplex(v: np.ndarray, z: float):
    """
    Euclidean projection of v onto the simplex
        S = { x | x_i ≥ 0 , Σ x_i = z }.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if z < 0:
        raise ValueError("target sum must be non‑negative")
    # shift by the mean to avoid negative target handling later
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection – give uniform distribution
        theta = (np.sum(v) - z) / len(v)
        w = np.maximum(v - theta, 0.0)
        return w
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _dirichlet_start(n: int, target_sum: float, rng: np.random.Generator):
    """
    Sample a random point on the simplex using a symmetric Dirichlet(1,…,1).
    The result automatically satisfies Σ = target_sum and x_i ≥ 0.
    """
    x = rng.dirichlet(np.ones(n))
    return x.astype(np.float64) * target_sum


def _subgradient_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 5000,
    rng: np.random.Generator = None,
):
    """
    Projected sub‑gradient descent for the max‑convolution objective.
    Returns the best vector seen and its objective value (M).
    """
    n = f0.shape[0]
    f = f0.copy()
    best_f = f.copy()
    best_M, _ = _max_conv_and_index(f)
    no_improve = 0
    # initial step‑size (tuned heuristically)
    step0 = max(0.5, target_sum / 5.0)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        M, k = _max_conv_and_index(f)

        # record improvement
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            no_improve = 0
            print(f"[subgrad] n={n} it={it} M={M:.6f}")
        else:
            no_improve += 1

        # stop if we are essentially at the known lower bound
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        if M <= lower_bound * 1.01:
            break

        # compute sub‑gradient at the active index
        grad = _gradient_at_max(f, k)

        # diminishing step size
        eta = step0 / math.sqrt(it + 1)

        # gradient step and projection onto the simplex
        f_tmp = f - eta * grad
        f = _project_onto_simplex(f_tmp, target_sum)

        # occasional random jitter to escape plateaus
        if no_improve >= 30:
            jitter = rng.normal(scale=0.02 * target_sum / n, size=n)
            f = _project_onto_simplex(np.clip(f + jitter, 0.0, None), target_sum)
            no_improve = 0

    return best_f, best_M

    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # tiny safety margin
    rng = np.random.default_rng(seed)

    best_score = float("inf")

    # Try to reuse a previously discovered construction if available.
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        global_seq = np.asarray(global_seq, dtype=np.float64)
    else:
        global_seq = None

    # Lengths to explore – larger n usually give a tighter bound.
    lengths = [200, 300, 400, 500, 600, 800, 1000]

            break

                f_start = global_seq.copy()
                f_start = np.maximum(f_start, 0.0)
                f_start = f_start / np.sum(f_start) * target_sum
                print(f"[init] using provided global sequence (n={n})")
            else:
                f_start = _dirichlet_start(n, target_sum, rng)

            # ------------------------------------------------------------------
            # run the sub‑gradient optimiser
            # give each restart a generous chunk of the remaining budget
            remaining = deadline - time.time()
            # we allow up to 5000 iterations, but stop early if time is tight
            max_iters = 5000
            f_opt, score = _subgradient_optimize(
                f_start, target_sum, deadline, max_iters=max_iters, rng=rng
            )

            # ------------------------------------------------------------------
            # keep the overall best
            if score < best_score:
                best_score = score
                best_seq = f_opt.copy()
                print(f"[best] n={n} restart={restart} score={score:.6f}")

    # ----------------------------------------------------------------------
    # safety fallback – produce a uniform sequence if nothing was found
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = float(np.max(np.convolve(best_seq, best_seq)))
        print("[fallback] using uniform construction")

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    # convert to plain list of native Python floats as required by the judge
    return [float(v) for v in best_seq]
