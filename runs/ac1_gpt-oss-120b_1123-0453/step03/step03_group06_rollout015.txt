def _project_simplex(v, z):
    # v: numpy array
    # returns projection onto {x >= 0, sum x = z}
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if len(rho)==0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def _coefficients_flat_conv(n):
    """Return coefficients a_0..a_{n-1} such that (a*a)_k = 1 for k=0..2n-2."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for m in range(1, n):
        # sum_{i=1}^{m-1} a_i * a_{m-i}
        if m == 1:

    return a

    ∀k : Σ_{i+j=k} a_i a_j = 1   (0 ≤ k ≤ 2n‑2)

    The construction yields the optimal lower bound 2·n/(2·n‑1) for the
    evaluation function.  If a previous sequence is stored in the global
    variable ``height_sequence_1`` its length is reused; otherwise a
    length of 500 is used.
    """
    # The seed is irrelevant – the algorithm is deterministic.
    _ = seed

    # ------------------------------------------------------------------
    # Choose the length.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            n = len(prev)                     # reuse the earlier size
        except Exception:
            n = 500
    else:
        n = 500                               # default size used by the reference code

    # ------------------------------------------------------------------
    # Compute the exact coefficients that give a constant convolution.
    # ------------------------------------------------------------------
    coeffs = _flat_convolution_coeffs(n)

    # ------------------------------------------------------------------
    # Scale to satisfy the ℓ¹‑norm required by the score:
    #     Σ a_i = √(2·n)
    # (the evaluate_sequence routine clamps values to [0,1000] after this,
    #  but our numbers are far below that bound.)
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    current_sum = float(coeffs.sum())
    if current_sum == 0.0:                     # fallback – should never happen
        seq = np.full(n, target_sum / n, dtype=np.float64)
    else:
        seq = coeffs * (target_sum / current_sum)

    # ------------------------------------------------------------------
    # Enforce the hard limits used by evaluate_sequence.
    # ------------------------------------------------------------------
    seq = np.clip(seq, 0.0, 1000.0)

    # Return a plain Python list as required.
    return [float(v) for v in seq]
