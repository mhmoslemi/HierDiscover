import time, math, numpy as np

def _initialize_sequence(n, seed_seq=None, rng=None):
    target_sum = math.sqrt(2.0 * n)
    if seed_seq is not None:
        seq = np.array(seed_seq, dtype=np.float64)
        if len(seq) != n:
            # either truncate or pad
            if len(seq) > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.full(n - len(seq), 0.0)])
        seq = np.clip(seq, 0.0, 1000.0)
        total = np.sum(seq)
        if total == 0:
            seq = np.full(n, target_sum / n)
        else:
            seq = seq * (target_sum / total)
    else:
        # random positive values
        if rng is None:
            rng = np.random.default_rng()
        seq = rng.random(n) + 0.01  # avoid zeros
        seq = seq / np.sum(seq) * target_sum
    return seq

def _run_sa(seq, deadline):
    # seq as np.ndarray, sum already normalized
    n = len(seq)
    target_sum = np.sum(seq)
    # compute convolution
    conv = np.convolve(seq, seq)
    cur_max = float(conv.max())
    # tracking best
    best_seq = seq.copy()
    best_max = cur_max
    # start temperature based on cur_max
    start_T = cur_max * 0.5 + 1e-8   # avoid zero
    avg = target_sum / n
    rng = np.random.default_rng()
    start_time = time.monotonic()
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        if total <= 0: break
        frac = elapsed / total
        T = start_T * (0.001 ** frac)  # exponential cool
        sigma = avg * (T / start_T) * 2.0  # scale, 2x typical
        # Choose two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # propose delta
        raw = rng.normal(scale=sigma)
        # clip delta to keep non-negativity
        delta = raw
        # ensure bounds
        if delta > seq[i]:
            delta = seq[i]
        if -delta > seq[j]:
            delta = -seq[j]
        if abs(delta) < 1e-12:
            continue
        # indices in conv
        idx_i2 = 2 * i
        idx_j2 = 2 * j
        idx_ij = i + j
        # pre-save old conv entries
        old_i2 = conv[idx_i2]
        old_j2 = conv[idx_j2]
        old_ij = conv[idx_ij]
        # compute changes
        d = delta
        # using old a_i, a_j
        a_i = seq[i]
        a_j = seq[j]
        change_i2 = -2.0 * d * a_i + d * d
        change_j2 = 2.0 * d * a_j + d * d
        change_ij = 2.0 * d * (a_i - a_j) - 2.0 * d * d
        # apply temporary changes
        conv[idx_i2] = old_i2 + change_i2
        conv[idx_j2] = old_j2 + change_j2
        conv[idx_ij] = old_ij + change_ij
        new_max = float(conv.max())
        delta_E = new_max - cur_max
        accept = False
        if delta_E < -1e-12:
            accept = True
        else:
            # Probabilistic acceptance
            if T > 1e-12:
                prob = math.exp(-delta_E / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            # commit changes to seq
            seq[i] = a_i - d
            seq[j] = a_j + d
            cur_max = new_max
            # record best
            if cur_max < best_max:
                best_max = cur_max
                best_seq = seq.copy()
        else:
            # revert conv
            conv[idx_i2] = old_i2
            conv[idx_j2] = old_j2
            conv[idx_ij] = old_ij
    return best_seq, best_max

                    cur_max = cand_max

                        best_m = cand_max
                        best_a = cand.copy()
                # shrink step
                step *= 0.999
            return best_a, best_m

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return [float(x) for x in best_seq]

"""
Simulated‑annealing optimizer for the step‑function bound.

The algorithm keeps the ℓ¹‑norm fixed (‖a‖₁ = √(2 n)) and searches
for a non‑negative vector `a` that minimizes the maximal entry of the
auto‑convolution `a*a`.  At each iteration a small amount of mass is
transferred between two random coordinates; the new maximal convolution
value is evaluated fast with an FFT‑based convolution (O(n log n)).  An
exponential cooling schedule implements simulated annealing, allowing
up‑hill moves early on and converging to a greedy descent later.
The best vector found (over several sequence lengths and random
restarts) is returned.
"""

import time
import math
import numpy as np

def _fft_max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max entry of a*a using an FFT of length ``fft_len``."""
    # Real FFT for speed
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # Only the first 2*len(a)-1 entries are relevant
    needed = 2 * len(a) - 1
    return float(np.max(conv[:needed]))


def _initialize_sequence(n: int, seed_seq=None, rng=None) -> np.ndarray:
    """Create a starting non‑negative sequence of length n with ℓ¹‑norm √(2 n)."""
    target = math.sqrt(2.0 * n)
    if seed_seq is not None:
        seq = np.array(seed_seq, dtype=np.float64)
        if len(seq) != n:
            if len(seq) > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - len(seq))])
        seq = np.clip(seq, 0.0, 1000.0)
        s = np.sum(seq)
        if s == 0.0:
            seq = np.full(n, target / n)
        else:
            seq = seq * (target / s)
        return seq
    # random start
    if rng is None:
        rng = np.random.default_rng()
    seq = rng.random(n) + 1e-3
    seq = seq / np.sum(seq) * target
    return seq


def _simulated_annealing(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run a temperature‑controlled random walk that transfers mass between two
    coordinates while keeping the ℓ¹‑norm fixed.
    Returns the best vector seen and its maximal auto‑convolution entry.
    """
    n = len(a)
    target_sum = np.sum(a)               # already √(2 n)
    avg = target_sum / n
    # FFT length: next power of two ≥ 2n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())
    cur_max = _fft_max_convolution(a, fft_len)

    best_seq = a.copy()
    best_val = cur_max

    rng = np.random.default_rng()
    start_time = time.monotonic()
    total_time = max(1e-12, deadline - start_time)

    # initial temperature proportional to current max
    init_temp = max(cur_max * 0.5, 1e-4)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time
        # exponential cooling, final temp ~1e‑4 · init_temp
        T = init_temp * (0.0001 ** frac)

        # step size proportional to the average height and the current temperature
        step = avg * (T / init_temp) * 3.0   # 3 · avg at the start, shrinks to ≈0

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose a transfer amount
        delta = rng.normal(scale=step)
        # clip to keep entries non‑negative
        if delta > a[i]:
            delta = a[i]
        if -delta > a[j]:
            delta = -a[j]
        if abs(delta) < 1e-12:
            continue

        # apply the move temporarily
        a[i] -= delta
        a[j] += delta

        new_max = _fft_max_convolution(a, fft_len)

        # Metropolis acceptance
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 1e-12:
                prob = math.exp(-(new_max - cur_max) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta

    return best_seq, best_val


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the allotted time.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.05)   # small safety margin

    # try a few different lengths – the longer the vector the more degrees of freedom
    length_candidates = [300, 500, 800, 1000, 1200, 1500]
    # warm‑start with a previously found construction if available
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        n0 = len(h1)
        if n0 not in length_candidates:
            length_candidates.append(n0)

    length_candidates = sorted(set(length_candidates))

    best_seq = None
    best_score = float("inf")

    # Allocate roughly equal time to each length
    remaining = deadline - time.monotonic()
    per_len = max(0.5, remaining / max(1, len(length_candidates)))

    for n in length_candidates:
        if time.monotonic() >= deadline:
            break
        sub_deadline = min(deadline, time.monotonic() + per_len)

        # initialise – use the historic sequence when the size matches
        seed_seq = h1 if isinstance(h1, (list, tuple, np.ndarray)) and len(h1) == n else None
        a0 = _initialize_sequence(n, seed_seq=seed_seq, rng=rng)

        # run the annealer
        best_local, val_local = _simulated_annealing(a0, sub_deadline)

        if val_local < best_score:
            best_score = val_local
            best_seq = best_local.copy()
            print(f"[update] n={n} score={val_local:.6f}")

        # keep a little spare time for a final cheap hill‑climb
        remaining = deadline - time.monotonic()
        per_len = max(0.5, remaining / max(1, len(length_candidates) - (length_candidates.index(n)+1)))

    # ------------------------------------------------------------------
    # Optional short deterministic polishing (projected gradient descent)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        # a very small number of gradient steps to clean up any remaining wiggle
        import math as _m
        def _project_simplex(v, s):
            u = np.sort(v)[::-1]
            cssv = np.cumsum(u) - s
            ind = np.arange(len(v)) + 1
            cond = u - cssv / ind > 0
            if not np.any(cond):
                return np.full_like(v, s / len(v))
            rho = np.where(cond)[0][-1]
            theta = cssv[rho] / (rho + 1)
            return np.maximum(v - theta, 0.0)

        def _gradient_step(a):
            n = len(a)
            conv = np.convolve(a, a)
            max_val = np.max(conv)
            # active offsets
            eps = max_val * 1e-9 + 1e-12
            active = np.where(conv >= max_val - eps)[0]
            grad = np.zeros_like(a)
            for k in active:
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                i = np.arange(i_start, i_end + 1)
                j = k - i
                vals = a[j]
                grad[i] += 2.0 * vals
                mask = (i == j)
                if np.any(mask):
                    grad[i[mask]] -= vals[mask]
            return grad

        a = best_seq.copy()
        target_sum = np.sum(a)
        for _ in range(30):
            if time.monotonic() >= deadline:
                break
            g = _gradient_step(a)
            step_sz = target_sum * 0.05
            a_new = _project_simplex(a - step_sz * g, target_sum)
            a_new = np.clip(a_new, 0.0, 1000.0)
            # evaluate
            cur = _fft_max_convolution(a, 1 << ((2*len(a)-1).bit_length()))
            nxt = _fft_max_convolution(a_new, 1 << ((2*len(a_new)-1).bit_length()))
            if nxt < cur:
                a = a_new
                if nxt < best_score:
                    best_score = nxt
                    best_seq = a.copy()

    if best_seq is None:
        # fallback uniform construction
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)

    # final safety clip (evaluate_sequence also does this)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return [float(x) for x in best_seq]

            u = np.sort(v)[::-1]
            cssv = np.cumsum(u) - s
            ind = np.arange(len(v)) + 1
            cond = u - cssv / ind > 0
            if not np.any(cond):
                return np.full_like(v, s / len(v))
            rho = np.where(cond)[0][-1]
            theta = cssv[rho] / (rho + 1)
            return np.maximum(v - theta, 0.0)

        # sub‑gradient of the worst auto‑convolution entry
        def _subgrad(f: np.ndarray) -> np.ndarray:
            n = len(f)
            conv = np.convolve(f, f)
            max_val = np.max(conv)
            eps = max_val * 1e-9 + 1e-12
            active = np.where(conv >= max_val - eps)[0]
            grad = np.zeros_like(f)
            for k in active:
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                i = np.arange(i_start, i_end + 1)
                j = k - i
                vals = f[j]
                grad[i] += 2.0 * vals
                mask = (i == j)
                if np.any(mask):
                    grad[i[mask]] -= vals[mask]
            return grad

        a = best_seq.copy()
        target_sum = np.sum(a)
        fft_len = 1 << ((2 * len(a) - 1).bit_length())
        cur = _fft_max_convolution(a, fft_len)

        for _ in range(30):
            if time.monotonic() >= deadline:
                break
            g = _subgrad(a)
            step_sz = target_sum * 0.05
            a_candidate = _project_onto_simplex(a - step_sz * g, target_sum)
            a_candidate = np.clip(a_candidate, 0.0, 1000.0)
            nxt = _fft_max_convolution(a_candidate, fft_len)
            if nxt < cur - 1e-12:
                a = a_candidate
                cur = nxt
                if nxt < best_score:
                    best_score = nxt
                    best_seq = a.copy()

    # fallback in the unlikely case nothing was found
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
