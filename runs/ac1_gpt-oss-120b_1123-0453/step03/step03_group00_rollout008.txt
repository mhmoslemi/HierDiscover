import scipy.optimize as opt.

Define function _linear_oracle(f, M):

   n = len(f)

   rows = 2*n-1

   A = np.zeros((rows, n))

   for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        A[k, j_range] = f[i_range]

   b = np.full(rows, M)

   c = -np.ones(n)   # maximize sum(g)
   bounds = [(0, None)] * n

   res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
   if res.success:
       g = res.x
   else:
       # fallback: return f (no change)
       g = f.copy()
   return g

def _lp_descent(f0, deadline):
    f = f0.copy()
    n = len(f)
    S = np.sum(f)
    conv_ff = np.convolve(f, f)
    M = np.max(conv_ff)
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        # build A, solve LP
        g = _linear_oracle(f, M)  # returns g (non-negative)
        sum_g = np.sum(g)
        if sum_g < 1e-12:
            break
        g = g * (S / sum_g)   # scale to sum S

        # compute convolutions
        conv_fg = np.convolve(f, g)
        conv_gg = np.convolve(g, g)

        # line-search t
        best_t = 0.0
        best_Mt = M
        # candidate steps; exclude t=0 (no change)
        for t in [0.1,0.25,0.5,0.75,0.9]:
            # compute combined conv
            # Using formula: (1-t)^2 * conv_ff + 2*t*(1-t)*conv_fg + t**2 * conv_gg
            combined = (1-t)**2 * conv_ff + 2*t*(1-t)*conv_fg + t**2 * conv_gg
            Mt = np.max(combined)
            if Mt < best_Mt - 1e-12:
                best_Mt = Mt
                best_t = t

        if best_t == 0.0:
            # cannot improve, break
            break

        # update
        f = (1 - best_t) * f + best_t * g
        conv_ff = (1 - best_t)**2 * conv_ff + 2*best_t*(1 - best_t) * conv_fg + best_t**2 * conv_gg
        M = best_Mt

        cur_score = _compute_score_from_vector(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[LP] n={n} it={it} t={best_t:.3f} score={cur_score:.6f}")
        # else no improvement? we still improved M but maybe not enough. Continue
        it += 1

    return best_f, best_score

    Apply conditional gradient (LP oracle) refinement to reduce max convolution.
    Returns best sequence and its score seen during refinement.
    """
    f = f_initial.copy()

    S = target_sum
    conv_ff = np.convolve(f, f)
    M = np.max(conv_ff)
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)
    it = 0
    while time.monotonic() < deadline and it < max_iter:
        # Build A matrix for constraints

        for k in range(2 * n - 1):

            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1)

        c = -np.ones(n, dtype=np.float64)

            break

        sum_g = np.sum(g)
        if sum_g <= 0:
            break

        g = g * (S / sum_g)

        conv_gg = np.convolve(g, g)

"""
Optimisation pipeline for the step‑function attack.

The programme consists of three stages:

1️⃣  **Warm‑start & projected gradient descent** – a fast Euclidean
    projection onto the ℓ¹‑simplex is used together with a sub‑gradient
    of the worst auto‑convolution entry.  Several lengths and random
    restarts are processed while respecting the global budget.

2️⃣  **Conditional‑gradient (Frank–Wolfe) refinement** – for the best
    candidate a linear‑programming oracle
    ``max Σ gᵢ  s.t.  f * g ≤ M`` (where ``*`` denotes the discrete
    convolution) is solved with SciPy’s high‑performance “highs”
    solver.  The returned direction is combined with the current
    vector by an exact line‑search that minimises the new maximal
    convolution.  This step is very effective at shaving the final
    bound.

3️⃣  **Stochastic polishing** – a short Gaussian hill‑climbing phase
    (projection back onto the simplex) removes any remaining tiny
    imperfections.

All numbers are kept non‑negative and below the clipping bound used by
``evaluate_sequence``.  The function ``propose_candidate`` returns the
best sequence discovered before the deadline.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x | x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Edge case – return a uniform distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact evaluation of the objective (without the internal clipping)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:   # same protection as in evaluate_sequence
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where the convolution is (almost) maximal."""
    eps = max_val * 1e-9 + 1e-12
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the worst auto‑convolution entry with respect to ``f``.
    The contributions of all active offsets are summed.
    """
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = f[j]
        # each pair (i,j) appears twice except the diagonal i=j
        grad[i] += 2.0 * vals
        mask = (i == j)
        if np.any(mask):
            # diagonal should be counted only once
            grad[i[mask]] -= vals[mask]
    return grad


# ----------------------------------------------------------------------
# Stage 1 – projected gradient descent (same idea as the reference code)
# ----------------------------------------------------------------------


def _gradient_descent(f0: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the ℓ¹‑simplex that reduces the
    maximal entry of the auto‑convolution.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)
    step = target_sum * 0.1           # initial step size (absolute)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        cur_score = _compute_score_from_vector(f)

        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
            print(f"[GD] it={it:04d} n={len(f)} score={cur_score:.6f}")
        else:
            no_improve += 1

        if no_improve > 500:          # stagnation → stop early
            break

        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)
        grad = _gradient_submax(f, active)

        f_candidate = _project_onto_simplex(f - step * grad, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)
        cand_score = _compute_score_from_vector(f_candidate)

        if cand_score < cur_score:
            f = f_candidate
            step *= 1.02     # a little optimism
        else:
            step *= 0.5      # back‑off
    return best_f, best_score


# ----------------------------------------------------------------------
# Stage 2 – LP‑based conditional gradient refinement
# ----------------------------------------------------------------------


def _linear_oracle(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Linear‑programming oracle:
        maximise   Σ g_i
        s.t.       f * g ≤ M   (component‑wise)
                   g_i ≥ 0
    Returns the optimal ``g`` (may be dense).  If the LP fails,
    ``None`` is returned.
    """
    n = len(f)
    rows = 2 * n - 1

    # Build the (rows × n) matrix A such that A @ g == f * g
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        A[k, j_range] = f[i_range]

    # Linear program data
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  ⇔ minimise –Σ g_i
    b = np.full(rows, M, dtype=np.float64)    # upper bound M for every convolution entry
    bounds = [(0.0, None)] * n                 # g_i ≥ 0

    # Solve with HiGHS (fast interior‑point / simplex hybrid)
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True})

    if res.success:
        return res.x
    else:
        print(f"[LP] failed – {res.message}")
        return None


def _lp_refine(f_initial: np.ndarray,
               target_sum: float,
               deadline: float,
               max_iter: int = 15) -> tuple[np.ndarray, float]:
    """
    Conditional‑gradient (Frank–Wolfe) refinement that repeatedly calls the
    linear‑programming oracle, rescales the direction to the required ℓ¹‑norm
    and performs an exact line‑search on the maximal auto‑convolution.
    """
    f = f_initial.copy()
    n = len(f)
    S = target_sum
    conv_ff = np.convolve(f, f)
    M = np.max(conv_ff)

    best_f = f.copy()
    best_score = _compute_score_from_vector(f)

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        # --- LP oracle ------------------------------------------------
        g = _linear_oracle(f, M)
        if g is None:
            break

        sum_g = np.sum(g)
        if sum_g <= 1e-12:
            break
        # Rescale to respect the ℓ¹‑norm constraint (the LP does not enforce it)
        g = g * (S / sum_g)

        # --- exact line‑search on the convex combination -------------
        conv_fg = np.convolve(f, g)
        conv_gg = np.convolve(g, g)

        best_t = 0.0
        best_Mt = M
        # a tiny grid of mixing parameters – enough for a quick improvement
        for t in (0.1, 0.25, 0.5, 0.75, 0.9):
            mixed = ((1 - t) ** 2) * conv_ff + (2 * t * (1 - t)) * conv_fg + (t ** 2) * conv_gg
            Mt = np.max(mixed)
            if Mt < best_Mt - 1e-12:
                best_Mt = Mt
                best_t = t

        if best_t == 0.0:       # no improving step found
            break

        # update the iterate
        f = (1 - best_t) * f + best_t * g
        conv_ff = ((1 - best_t) ** 2) * conv_ff \
                  + (2 * best_t * (1 - best_t)) * conv_fg \
                  + (best_t ** 2) * conv_gg
        M = best_Mt

        cur_score = _compute_score_from_vector(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[LP] it={it} n={n} t={best_t:.2f} score={cur_score:.6f}")

    return best_f, best_score

# ----------------------------------------------------------------------


def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that preserves the ℓ¹‑norm via a simplex
    projection.  The step size slowly decays.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score_from_vector(cand)

        if cand_score < best_score:
            best = cand
            best_score = cand_score

        step *= 0.9995
        if step < 1e-8:
            step = step_initial

    return best, best_score
