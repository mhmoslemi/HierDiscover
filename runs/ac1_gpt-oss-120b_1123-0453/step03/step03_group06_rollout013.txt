        • a gradient‑like step that follows the (sub)gradient of the
          worst entry.
    The optimiser follows a simple simulated‑annealing schedule
    (linear cooling, decreasing step size) and always keeps the best
    sequence seen.

3️⃣  **Final polishing** – a short Gaussian hill‑climb (zero‑mean
    perturbations) is performed on the overall best sequence.

The whole search respects the supplied time budget (default 1000 s)
and returns a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution / scoring (exact analogue of evaluate_sequence)
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of ``seq`` with itself using a real FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(f*f)/‖f‖₁² (identical to ``evaluate_sequence``)."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.clip(np.interp(new_x, old_x, seq), 0.0, 1000.0)


def _normalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and scale so that ℓ¹‑norm equals ``target_sum``."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s > 0:
        seq = seq * (target_sum / s)
    return seq


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def _uniform_seq(n: int, target_sum: float) -> np.ndarray:
    return np.full(n, target_sum / n, dtype=np.float64)


def _dirichlet_seq(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    seq = rng.dirichlet(np.ones(n))
    return _normalise(seq, target_sum)


def _exp_seq(n: int, target_sum: float, r: float) -> np.ndarray:
    """Monotone exponential a_i = r**i."""
    raw = r ** np.arange(n, dtype=np.float64)
    return _normalise(raw, target_sum)


def _rev_exp_seq(n: int, target_sum: float, r: float) -> np.ndarray:
    """Exponential decreasing from the right side."""
    raw = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return _normalise(raw, target_sum)


def _pow_seq(n: int, target_sum: float, p: float) -> np.ndarray:
    """Power law a_i = (i+1)**(-p)."""
    raw = (np.arange(1, n + 1, dtype=np.float64)) ** (-p)
    return _normalise(raw, target_sum)


def _edge_seq(n: int, target_sum: float, p: float) -> np.ndarray:
    """Edge‑heavy: weight ~ 1/((i+1)*(n-i))**p."""
    i = np.arange(n, dtype=np.float64) + 1.0
    raw = 1.0 / ((i * (n - i + 1)) ** p)
    return _normalise(raw, target_sum)


def _gaussian_seq(n: int, target_sum: float, sigma_factor: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    raw = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    return _normalise(raw, target_sum)


# ----------------------------------------------------------------------
# Local optimiser (SA + gradient‑like moves)
# ----------------------------------------------------------------------


def _optimise(seq: np.ndarray,
              target_sum: float,
              deadline: float,
              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing / gradient hybrid.
    Maintains the ℓ¹‑norm (by normalisation after each move) and
    constantly tries to lower the evaluation score.
    """
    n = seq.shape[0]
    cur = np.clip(seq, 0.0, 1000.0)
    cur_sum = cur.sum()
    if cur_sum > 0:
        cur = cur * (target_sum / cur_sum)

    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.monotonic()
    total = deadline - start
    # relatively generous initial temperature
    T0 = max(1e-6, cur_score * 0.5)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = elapsed / total
        T = T0 * (1.0 - frac)                     # linear cooling
        step_frac = 0.30 * (1.0 - frac) + 1e-5    # decreasing mass‑transfer size

        # ------- pick a move -------------------------------------------------
        if rng.random() < 0.55:
            # random pairwise transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_transfer = min(cur[i], step_frac * target_sum)
            max_transfer = min(max_transfer, 1000.0 - cur[j])
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # directed move based on the worst convolution entry
            conv = _conv_full_fft(cur)
            km = int(conv.argmax())
            # compute the contribution of each i to that entry
            idx = np.arange(n)
            jdx = km - idx
            mask = (jdx >= 0) & (jdx < n)
            if not np.any(mask):
                continue
            prod = np.zeros(n, dtype=np.float64)
            prod[mask] = cur[idx[mask]] * cur[jdx[mask]]
            i_max = int(idx[mask][np.argmax(prod[mask])])
            i_min = int(idx[mask][np.argmin(prod[mask])])
            if i_max == i_min:
                continue
            max_transfer = min(cur[i_max], step_frac * target_sum)
            max_transfer = min(max_transfer, 1000.0 - cur[i_min])
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            cand = cur.copy()
            cand[i_max] -= delta
            cand[i_min] += delta

        # ------- evaluate candidate -----------------------------------------
        cand_score = _fast_score(cand)

        # ------- acceptance -------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(cand_score - cur_score) / T)
                accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[optim] n={n:4d} time={elapsed:6.2f}s score={cand_score:.6f}")

        # occasional cheap gradient‑like step (no extra conv)
        if rng.random() < 0.04:
            conv = _conv_full_fft(cur)
            km = int(conv.argmax())
            rev = np.zeros_like(cur)
            idx = np.arange(n)
            jdx = km - idx
            in_range = (jdx >= 0) & (jdx < n)
            rev[in_range] = 2.0 * cur[jdx[in_range]]
            alpha = step_frac * 0.4
            cand = cur - alpha * rev
            cand = np.clip(cand, 0.0, 1000.0)
            # re‑normalise ℓ¹‑norm
            s = cand.sum()
            if s > 0:
                cand = cand * (target_sum / s)
            cand_score = _fast_score(cand)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                    print(f"[grad] n={n:4d} time={elapsed:6.2f}s score={cand_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Final Gaussian polishing (simple hill‑climb)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """Gaussian perturbation hill‑climb (no ℓ¹‑norm fixing – final eval will clip)."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` score.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.4)  # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # lengths to explore – we give preference to larger vectors
    lengths = [500, 800, 1200, 1600, 2000]

    # keep the best sequence for each length (useful for interpolation)
    best_by_len: dict[int, np.ndarray] = {}

    # known global reference, if any
    known_seq = globals().get("height_sequence_1", None)
    if known_seq is not None:
        known_seq = np.asarray(known_seq, dtype=np.float64)

    # allocate ~80 % of the time to the optimisation loops
    total_opt_time = (deadline - time.monotonic()) * 0.80
    time_per_len = total_opt_time / max(1, len(lengths))

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # 1️⃣  build a pool of starting candidates
        # ------------------------------------------------------------------
        candidates: list[np.ndarray] = []

        # a) use known sequence (interpolated) if size matches
        if known_seq is not None and known_seq.shape[0] == n:
            candidates.append(_normalise(known_seq.copy(), target_sum))

        # b) interpolate from the best already‑computed length (if any)
        if best_by_len:
            src_n = max(k for k in best_by_len.keys() if k <= n)
            candidates.append(_normalise(_interpolate(best_by_len[src_n], n), target_sum))

        # c) a few hand‑crafted families
        #   – uniform (with tiny jitter)
        uni = _uniform_seq(n, target_sum) + rng.normal(scale=0.001, size=n)
        candidates.append(_normalise(uni, target_sum))

        #   – Dirichlet random (several draws)
        for _ in range(3):
            candidates.append(_dirichlet_seq(n, target_sum, rng))

        #   – exponential shapes
        for r in (0.96, 0.97, 0.98, 0.99, 0.995):
            candidates.append(_exp_seq(n, target_sum, r))
            candidates.append(_rev_exp_seq(n, target_sum, r))

        #   – power‑law decreasing
        for p in (0.5, 0.8, 1.0, 1.3, 1.6, 2.0):
            candidates.append(_pow_seq(n, target_sum, p))

        #   – edge‑heavy (inverse of product of distances from both ends)
        for p in (0.5, 1.0, 1.5, 2.0):
            candidates.append(_edge_seq(n, target_sum, p))

        #   – Gaussian bumps
        for sigma in (0.15, 0.25, 0.35):
            candidates.append(_gaussian_seq(n, target_sum, sigma))

        # evaluate all init candidates and keep the best few
        scored = [(cand, _fast_score(cand)) for cand in candidates]
        scored.sort(key=lambda x: x[1])
        top_initial = [seq for seq, sc in scored[:3]]   # keep up to three seeds

        # ------------------------------------------------------------------
        # 2️⃣  optimisation on the selected seeds
        # ------------------------------------------------------------------
        len_deadline = time.monotonic() + time_per_len
        for seed_seq in top_initial:
            if time.monotonic() >= len_deadline:
                break
            cand_seq, cand_score = _optimise(seed_seq, target_sum, len_deadline, rng)
            best_by_len[n] = cand_seq.copy()
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()
                print(f"[global] new best ⇒ n={n} score={best_score:.6f}")

            # early stop if we have already crossed the target
            if best_score < 1.503:
                print("[global] target achieved – terminating early.")
                break
        if best_score < 1.503:
            break

    # ----------------------------------------------------------------------
    # 3️⃣  final Gaussian polishing on the overall best candidate
    # ----------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polished, polished_score = _local_search(best_seq, deadline, step_initial=0.5)
        if polished_score < best_score:
            best_seq = polished
            best_score = polished_score
            print(f"[final] improved to {best_score:.6f}")

    # safety net – should never be needed
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = _normalise(best_seq, math.sqrt(2.0 * n))

    # return as plain Python list
    return [float(v) for v in best_seq]
