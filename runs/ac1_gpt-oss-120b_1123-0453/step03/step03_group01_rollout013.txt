"""
Propose a step sequence with a low evaluation score.

The search uses:
1. Several random or heuristic initializations (uniform, exponential,
   previously found best).
2. A fast O(n) incremental hill‑climber that repeatedly moves a tiny
   amount of mass between indices.  Most moves are random, but a small
   fraction are *targeted*: they look at the convolution index where the
   current maximum occurs, pick the pair of entries that contributes the
   largest product to that index and shift a bit of mass away from the
   larger entry towards a low‑mass index.  This concentrates the search
   on the true bottleneck and tends to lower the maximum convolution
   faster than blind random moves.
3. A brief Gaussian perturbation phase to polish the best vector found.

The whole routine respects the 1000 s time budget and returns the best
sequence (a plain Python list of non‑negative floats) discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official score using the provided evaluate_sequence."""
    return float(evaluate_sequence(seq.tolist()))


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index `i` to index `j` (i != j, delta > 0, delta <= a[i]).
    Updates the vector `a` and the convolution `conv = np.convolve(a, a)` *in‑place*
    using the exact algebraic formula (O(n) work).
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part : 2 * conv(delta, a)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part : conv(delta, delta)
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta          # (-δ)²
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta    # (-δ)*(+δ) + (+δ)*(-δ)
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta          # (+δ)²

    # update the vector itself
    a[i] -= delta
    a[j] += delta


def _targeted_move(a: np.ndarray,
                   conv: np.ndarray,
                   n: int,
                   rng: np.random.Generator,
                   step: float) -> tuple[int, int, float]:
    """
    Produce a move that targets the current maximum of the convolution.
    Returns a tuple (src, dst, delta).  The caller must apply the move
    using `_transfer_mass` and revert it if it is not improving.
    """
    # index of current maximum
    k = int(np.argmax(conv))

    # find the pair (i, j) with i + j = k that yields the largest product
    i_start = max(0, k - n + 1)
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    partner = k - idx
    products = a[idx] * a[partner]
    max_pos = int(np.argmax(products))
    i = int(idx[max_pos])
    j = int(partner[max_pos])

    # choose the larger of the two as the source
    src = i if a[i] >= a[j] else j

    # pick a destination with relatively low mass
    # simple strategy: sample a few candidates and keep the smallest
    cand = rng.integers(0, n, size=8)
    # ensure we do not pick the source itself
    cand = cand[cand != src]
    if cand.size == 0:
        dst = (src + 1) % n
    else:
        # choose the index with the smallest current value
        dst = int(cand[np.argmin(a[cand])])

    delta = a[src] * step * rng.random()
    return src, dst, delta


def _improve_by_targeted(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         step_init: float = 0.5,
                         target_prob: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that moves a little amount of mass from one index to another.
    A fraction of the moves (``target_prob``) are *targeted* at the current
    convolution peak (see `_targeted_move`).  The convolution is kept
    up‑to‑date incrementally, so each iteration costs O(n).

    Returns the best sequence encountered and its (scale‑invariant) score.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)
    sum_a = a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        # decide whether to perform a targeted move
        if rng.random() < target_prob:
            src, dst, delta = _targeted_move(a, conv, n, rng, step)
        else:
            # plain random transfer
            src = rng.integers(0, n)
            if a[src] <= 1e-12:
                it += 1
                continue
            dst = rng.integers(0, n - 1)
            if dst >= src:
                dst += 1
            delta = a[src] * step * rng.random()
            if delta <= 0.0:
                it += 1
                continue

        # apply the move
        _transfer_mass(a, conv, src, dst, delta)

        cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert
            _transfer_mass(a, conv, dst, src, delta)

        # decay step slowly
        step *= 0.9997
        if step < 1e-6:
            step = step_init

        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Gentle stochastic polishing: add Gaussian noise to the whole vector
    and accept the move if it strictly improves the official score.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-8:
            step = step_initial

    return best, best_score

    best_seq: np.ndarray | None = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

    lengths = [300, 400, 500, 600, 800, 1000]

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------

        seeds = []

        for start in seeds:

        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Gentle stochastic polishing: add Gaussian noise to the whole vector
    and keep the change if it strictly improves the official score.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-8:
            step = step_initial

    return best, best_score

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the global time budget (`budget_s` seconds) and
    returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)

    # leave a short safety margin for the caller
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Possibly reuse a known good construction
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        init_global = np.array(height_sequence_1, dtype=float)
    except Exception:
        init_global = None

    # ------------------------------------------------------------------
    # 2️⃣  Explore several lengths (larger n gives more freedom)
    # ------------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # 2.1  Build a handful of diverse initial vectors
        # ------------------------------------------------------------------
        starts: list[np.ndarray] = []

        # uniform baseline
        starts.append(np.full(n, target_sum / n, dtype=float))

        # exponential decay – more mass at the beginning
        starts.append(rng.exponential(scale=1.0, size=n))

        # random uniform with a small offset to avoid exact zeros
        starts.append(rng.random(n) + 0.1)

        # reversed exponential (mass at the end)
        starts.append(rng.exponential(scale=1.0, size=n)[::-1])

        # include the global construction if it matches the length
        if init_global is not None and init_global.shape[0] == n:
            starts.append(init_global.copy())

        for start in starts:
            if time.monotonic() >= deadline:
                break

            # Normalise to the target ℓ¹‑norm (the evaluation is scale‑invariant)
            s = start.sum()
            if s == 0.0:
                start[:] = target_sum / n
            else:
                start = start * (target_sum / s)

            start = np.clip(start, 0.0, 1000.0)

            # ------------------------------------------------------------------
            # 2.2  Focused hill‑climbing with targeted moves
            # ------------------------------------------------------------------
            improved, _ = _improve_by_targeted(start, deadline,
                                                max_iters=300_000,
                                                step_init=0.5,
                                                target_prob=0.15)

            official_score = _score(improved)
            if official_score < best_score:
                best_score = official_score
                best_seq = improved.copy()
                print(f"[improve] n={n} score={best_score:.6f}")

            # stop early once a comfortable margin is reached
            if best_score < 1.47:
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – modest uniform vector
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    if time.monotonic() < deadline:
        refined, refined_score = _local_search(best_seq, deadline,
                                               step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] improved to {best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
