"""
Optimisation of a non‑negative step sequence for the ``evaluate_sequence`` score.

The score for a normalised sequence (∑a = √(2 n)) equals the
maximum entry of its full self‑convolution.  We therefore minimise
the peak of the convolution directly.

Algorithm
---------
1.  Choose several lengths (400 … 1000).  For each length generate a few
    diverse initial vectors (uniform, half‑sine, exponential, and the
    previously discovered global ``height_sequence_1`` if it matches the
    length).  All vectors are projected onto the simplex
    {a ≥ 0, ∑a = √(2 n)}.

2.  Gradient‑descent on a smooth approximation of the max‑function.
    For the current vector a we compute b = a * a and a soft‑max
    weight w_k ∝ exp(α·b_k).  The gradient of the smooth objective is

        ∇ = 2·corr(a, w)   (cross‑correlation of a with w)

    which we obtain cheaply by a single convolution of the reversed a
    with w.  After a step a ← proj_simplex(a – η·∇) we evaluate the true
    score max(b).  The step size η is adapted (increase on success,
    decrease on failure) and the smoothing parameter α is slowly raised
    (α ← min(α·1.001, α_max)), so the optimisation first moves
    smoothly and then focuses on the true peak.

3.  After the gradient phase a few iterations of the specialised
    mass‑transfer hill‑climber (the original “transfer” routine) are run
    to fine‑tune the solution, followed by a gentle Gaussian perturbation
    (the original “local search”).

The routine respects the 1000 s time budget, prints progress, and returns
the best list of floats found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official score via the provided evaluator (list → float)."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho_pos = np.where(rho_arr > 0)[0]
    if rho_pos.size == 0:
        theta = 0.0
    else:
        rho = rho_pos[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(a: np.ndarray) -> float:
    """Fast internal score = max of full self‑convolution (∑a is fixed)."""
    conv = np.convolve(a, a)
    return float(conv.max())


def _gradient_step(a: np.ndarray,
                   alpha: float,
                   target_sum: float,
                   step: float) -> tuple[np.ndarray, float, bool]:
    """
    Perform one projected gradient step.
    Returns (new_a, new_score, improved).
    """
    n = a.shape[0]

    # full self‑convolution
    b = np.convolve(a, a)

    # soft‑max weights (smooth approximation of max)
    xb = alpha * b
    max_xb = xb.max()
    exp_xb = np.exp(xb - max_xb)               # stabilize
    w = exp_xb / exp_xb.sum()

    # gradient = 2 * cross‑correlation(a, w)
    ar = a[::-1]                               # reversed a
    conv_rev = np.convolve(ar, w)              # length = 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]    # slice length n

    # tentative update and projection onto simplex
    cand = _project_onto_simplex(a - step * grad, target_sum)

    # evaluate true score (max of convolution)
    cand_score = _compute_score(cand)

    cur_score = b.max()
    improved = cand_score < cur_score
    return cand, cand_score, improved


def _improve_by_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Very fast hill‑climbing that moves a tiny amount of mass from one index
    to another while keeping the ℓ¹‑norm fixed.  Used as a polishing step.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)
    sum_a = a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        # transfer mass i → j (in‑place, updates conv)
        # linear part
        conv[i:i + n] -= 2.0 * delta * a
        conv[j:j + n] += 2.0 * delta * a
        # quadratic part
        length = 2 * n - 1
        ii = 2 * i
        if ii < length:
            conv[ii] += delta * delta
        ij = i + j
        if ij < length:
            conv[ij] -= 2.0 * delta * delta
        jj = 2 * j
        if jj < length:
            conv[jj] += delta * delta
        # update vector
        a[i] -= delta
        a[j] += delta

        cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert
            conv[i:i + n] += 2.0 * delta * a
            conv[j:j + n] -= 2.0 * delta * a
            # quadratic revert
            if ii < length:
                conv[ii] -= delta * delta
            if ij < length:
                conv[ij] += 2.0 * delta * delta
            if jj < length:
                conv[jj] -= delta * delta
            a[i] += delta
            a[j] -= delta

        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Stochastic Gaussian perturbation followed by projection.
    Acts as a final polishing step.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑normalise to the required ℓ¹‑norm
        target_sum = math.sqrt(2.0 * len(cand))
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand * (target_sum / cand_sum)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` score.
    Uses a smooth‑max gradient descent, followed by the classic
    mass‑transfer and Gaussian polishing stages.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the caller
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Choose lengths and generate starting points
    # ------------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000]   # a range giving decent flexibility
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # collect diverse initial vectors
        starts: list[np.ndarray] = []

        # uniform
        starts.append(np.full(n, target_sum / n, dtype=float))

        # half‑sine (always positive)
        x = np.arange(n, dtype=float)
        sine = np.sin(np.pi * (x + 0.5) / n) + 0.001
        starts.append(sine * (target_sum / sine.sum()))

        # exponential random
        for _ in range(2):
            vec = rng.exponential(scale=1.0, size=n)
            starts.append(vec * (target_sum / vec.sum()))

        # optionally include the previously discovered global sequence
        try:
            global height_sequence_1
            gv = np.array(height_sequence_1, dtype=float)
            if gv.shape[0] == n:
                starts.append(gv * (target_sum / gv.sum()))
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 2️⃣  Gradient‑descent on smooth max
        # ------------------------------------------------------------------
        for start in starts:
            if time.monotonic() >= deadline:
                break

            a = start.copy()
            cur_score = _compute_score(a)

            # parameters
            alpha = 0.5                 # start with a very soft approximation
            alpha_max = 30.0
            step = 0.2 * math.sqrt(2.0 / n)   # reasonable initial magnitude
            step_min = 1e-7
            step_max = 0.5

            # simple adaptive loop
            while time.monotonic() < deadline:
                # one gradient step
                cand, cand_score, improved = _gradient_step(a, alpha, target_sum, step)

                if improved:
                    a = cand
                    cur_score = cand_score
                    # encourage larger steps after a success
                    step = min(step_max, step * 1.02)
                else:
                    # back‑off
                    step = max(step_min, step * 0.5)

                # keep the best seen globally
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[n={n}] new best score {best_score:.6f}")

                # increase smoothing slowly
                alpha = min(alpha_max, alpha * 1.001)

                # occasional early exit if we already beat the target bound
                if best_score < 1.503:
                    return best_seq.tolist()

                # stop inner loop if step becomes extremely tiny (converged)
                if step < 1e-8:
                    break

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing with the original hill‑climber & Gaussian search
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – modest uniform vector
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    # run a few rounds of the specialized mass‑transfer hill‑climber
    if time.monotonic() < deadline:
        improved, imp_score = _improve_by_transfer(best_seq, deadline,
                                                   max_iters=150_000,
                                                   step_init=0.5)
        if imp_score < best_score:
            best_seq = improved
            best_score = imp_score
            print(f"[transfer polish] improved to {best_score:.6f}")

    # final Gaussian perturbation
    if time.monotonic() < deadline:
        refined, refined_score = _local_search(best_seq, deadline,
                                               step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[gaussian polish] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the answer
    # ------------------------------------------------------------------
    print(f"Finished. Best score {best_score:.6f} (length {len(best_seq)})")
    return [float(x) for x in best_seq]
