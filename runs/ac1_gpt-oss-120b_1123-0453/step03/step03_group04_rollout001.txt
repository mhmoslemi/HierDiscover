"""
Sidon‑set based construction for a low‑value step sequence.

The evaluation metric (with the ℓ¹‑norm fixed to √(2·n)) equals the
maximum entry of the auto‑convolution of the sequence.  By choosing a
set of positions whose unordered pairwise sums are all distinct (a
Sidon set) we guarantee that each convolution entry receives at most
one ordered pair contribution, i.e. a weight of at most 2.  If the set
contains *m* equally weighted points, the maximal convolution value is

    max b ≤ 2·(S/m)²   with   S = √(2·n)

so the score is bounded by 4 n / m².  For a target score < 1.503 we need
m ≳ 1.65·√n, which is feasible because a Sidon set can contain up to
≈2·√n elements (the classic combinatorial bound).

The algorithm:
1. Try a few lengths *n* (500‑1000).  For each *n* compute the
   theoretical maximal Sidon size `m_max`.
2. Starting from the largest possible *m* (but not larger than the
   bound needed for the target score) repeatedly attempt to build a
   Sidon set by a random greedy procedure.
3. When a set of size *m* is found, build a vector that puts the
   value `v = √(2·n)/m` on the chosen positions and 0 elsewhere.
   Its ℓ¹‑norm is exactly √(2·n), therefore the evaluation equals the
   maximal auto‑convolution entry.
4. Keep the best sequence found; stop early if a score < 1.503 is
   obtained.

If no Sidon set can be constructed within the time budget the routine
falls back to a simple power‑law sweep (the method used previously).

The whole search respects the security checks of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_sidon_size(n: int) -> int:
    """
    Upper bound on the size of a Sidon set in {0,…,n‑1}:
    m(m+1)/2 ≤ 2n‑1  ⇒  m ≤ floor((-1 + sqrt(16n-7))/2)
    """
    return int(math.floor((-1.0 + math.sqrt(16.0 * n - 7.0)) / 2.0))


def _required_m_for_target(n: int, target_score: float = 1.503) -> int:
    """
    Minimal m that guarantees 4 n/m² ≤ target_score
    (since the Sidon construction yields max b ≤ 4 n/m²).
    """
    m = int(math.ceil(math.sqrt(4.0 * n / target_score)))
    # a small safety margin
    return max(m, 1)


def _attempt_sidonic_set(n: int, m: int, rng: np.random.Generator,
                        max_trials: int = 200) -> list[int] | None:
    """
    Random greedy construction of a Sidon set of size *m* in {0,…,n‑1}.
    Returns the list of indices or ``None`` if all trials fail.
    """
    candidates = np.arange(n, dtype=np.int32)

    for _ in range(max_trials):
        rng.shuffle(candidates)
        chosen = []
        sums = set()          # stores unordered sums i+j (i≤j)

        for x in candidates:
            # quick rejection using already stored sums
            conflict = False
            for y in chosen:
                if (x + y) in sums:
                    conflict = True
                    break
            if conflict:
                continue
            if (2 * x) in sums:     # self‑sum would clash
                continue

            # accept x, update sets
            for y in chosen:
                sums.add(x + y)      # y ≤ x is not required; unordered is enough
            sums.add(2 * x)
            chosen.append(x)

            if len(chosen) >= m:
                return chosen
        # this trial did not reach the desired size – try another random order
    return None


def _build_sequence(n: int, indices: list[int], target_sum: float) -> np.ndarray:
    """
    Build the step sequence: equal weight on the selected indices,
    zero elsewhere, with ℓ¹‑norm exactly ``target_sum``.
    """
    seq = np.zeros(n, dtype=np.float64)
    if not indices:
        return seq
    v = target_sum / len(indices)
    seq[indices] = v
    return seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence whose evaluation score is
    as low as possible.  The method builds near‑Sidon sets to obtain a
    rigorous bound on the maximal auto‑convolution entry.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – try Sidon constructions for several lengths.
    # ------------------------------------------------------------------
    lengths = [1000, 900, 800, 700, 600, 500, 400, 300, 200]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # theoretical maximal Sidon size for this n
        m_max = _max_sidon_size(n)

        # minimal size needed to beat the target score (rounded up)
        m_min_needed = _required_m_for_target(n, target_score=1.503)

        # we never need to search below the smaller of (m_max, m_min_needed)
        m_target = min(m_max, max(m_min_needed, int(1.8 * math.sqrt(n))))

        # gradually reduce the target size until we succeed or give up
        for m in range(m_target, m_min_needed - 1, -1):
            if time.monotonic() >= deadline:
                break

            # a few random attempts to find a Sidon set of the desired size
            indices = _attempt_sidonic_set(n, m, rng, max_trials=300)

            if indices is None:
                # failed for this m – try a smaller set
                continue

            seq = _build_sequence(n, indices, target_sum)

            # sanity: enforce the checks performed by evaluate_sequence
            # (clipping, non‑negativity, sum > 0.01)
            if seq.sum() < 0.01:
                continue

            # actual score (uses the provided security wrapper)
            sc = evaluate_sequence(seq.tolist())

            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[sidon] n={n} m={m} score={sc:.6f}")

            # early exit if we already beat the prescribed bound
            if best_score < 1.503:
                print("[sidon] Target reached – returning early.")
                return best_seq.tolist()

        # end‑for m
    # end‑for n

    # ------------------------------------------------------------------
    # Phase 2 – fallback: simple parametric sweep (similar to the
    # previous state‑of‑the‑art) in case the Sidon approach failed.
    # ------------------------------------------------------------------
    if best_seq is None:
        # start from a modest power‑law candidate
        n = 500
        target_sum = math.sqrt(2.0 * n)
        idx = np.arange(1, n + 1, dtype=np.float64)

        # try a few exponents; keep the best one
        for p in np.linspace(0.1, 3.0, 40):
            a = idx ** (-p)
            a = a / a.sum() * target_sum
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_seq, best_score = a.copy(), sc

        print(f"[fallback] best after sweep score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    if best_seq is None:
        # absolute fallback – uniform vector of length 200
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    return [float(x) for x in best_seq]
