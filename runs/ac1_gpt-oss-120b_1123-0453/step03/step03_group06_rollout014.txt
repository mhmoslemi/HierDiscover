def _lp_step(seq, target_sum, t, deadline):
    # seq is current f; M = max convolution
    n = len(seq)
    conv = _conv_full_fft(seq)
    M = conv.max()
    # Build matrix A_ub for constraints (seq * g) <= M
    # We need for each k (0..2n-2) compute sum_i seq[i] * g[k-i] <= M
    # That is convolution of seq (fixed) with g (variables).
    # We can construct Toeplitz matrix: for each k, row has seq[i] where i within range.
    # Use convolution property: g vector length n.
    # We'll construct matrix rows as length n arrays.
    A = np.zeros((2*n-1, n), dtype=float)
    for k in range(2*n-1):
        # i iterates 0..n-1, require 0<=k-i<n -> i in [max(0,k-(n-1)), min(k,n-1)]
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # For these i, coefficient = seq[i]
        A[k, i_start:i_end+1] = seq[i_start:i_end+1]
    # inequality: A @ g <= M
    bounds = [(0.0, None) for _ in range(n)]
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return seq  # fallback
    g0 = res.x
    sum_g = g0.sum()
    if sum_g > 0:
        g = g0 * (target_sum / sum_g)
    else:
        g = g0
    # convex combination
    new_seq = (1 - t) * seq + t * g
    # ensure bounds 0..1000 and renorm to target sum
    new_seq = np.clip(new_seq, 0, 1000)
    new_seq = new_seq * (target_sum / new_seq.sum())
    return new_seq
```

"""
Lightweight hybrid optimiser for the “max‑auto‑convolution” problem.

The score of a non‑negative sequence a (length n) after the built‑in
`evaluate_sequence` is  

    score = 2·n·max_k (a * a)[k] / (sum_i a_i)^2 .

If we keep the ℓ¹‑norm fixed to √(2 n) the factor 2·n/(sum²) is 1,
so minimising the score is exactly the same as minimising the largest
entry of the (full) auto‑convolution.  The optimiser works in three
stages:

1️⃣  **Initialisation** – reuse a previously found sequence (global
    ``height_sequence_1``), otherwise interpolate a shorter best
    solution or start from a random Dirichlet‑style vector.

2️⃣  **Hybrid descent** – within the allotted time for a given length
    we repeatedly perform one of the following cheap moves:
    * a *gradient step* that reduces the currently worst convolution
      entry (sub‑gradient of the max‑convolution);
    * a *random pairwise mass transfer* that respects the ℓ¹‑norm;
    * an occasional *linear‑programming* improvement (exact LP:
      maximise the ℓ¹‑norm of a vector g under the constraints
      (f * g) ≤ max(f * f), g ≥ 0) and then mix the result with the
      current sequence.

   The step size decays with time, and every few iterations a tiny
   directed move (transfer from the index with largest gradient weight
   to the one with smallest) is performed.  The best vector seen so
   far is kept.

3️⃣  **Final polishing** – a simple Gaussian hill‑climb (zero‑mean
   perturbations) refines the best candidate until the overall time
   budget expires.

All heavy work uses FFT‑based convolution (`np.fft.rfft`) and the
implementation is fully vectorised, runs on a single thread and stays
well within the 1000 s limit.  The returned sequence consists of
non‑negative floats and respects the 0 ≤ value ≤ 1000 safety checks of
`evaluate_sequence`.
"""

import math
import time
import numpy as np

# optional linear‑programming support (SciPy is allowed)
try:
    from scipy.optimize import linprog
    _SCIPY_LP = True
except Exception:      # pragma: no cover
    _SCIPY_LP = False


# ----------------------------------------------------------------------
# Fast convolution and scoring (exact analogue of evaluate_sequence)
# ----------------------------------------------------------------------
def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``seq`` with itself (real FFT)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` – assumes seq∈[0,1000]."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Interpolation utility – enlarge a known shape to a longer length
# ----------------------------------------------------------------------
def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return np.clip(new_seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Gradient of the worst convolution entry (sub‑gradient of max)
# ----------------------------------------------------------------------
def _max_gradient(seq: np.ndarray):
    """
    Returns (grad, k_max, max_val) where ``grad`` is the sub‑gradient of
    the maximal entry of the full auto‑convolution of ``seq``.
    """
    n = seq.shape[0]
    conv = _conv_full_fft(seq)
    k = int(conv.argmax())
    grad = np.zeros_like(seq)
    # contribution of a_i to conv[k] comes from a_{k-i} (if inside) and
    # a_{i-k} (if inside).  The loop is O(n) and cheap for n≤2000.
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] += seq[j]
        j2 = i - k
        if 0 <= j2 < n:
            grad[i] += seq[j2]
    return grad, k, float(conv.max())


# ----------------------------------------------------------------------
# One gradient‑descent step that keeps the ℓ¹‑norm fixed (up to re‑norm)
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, target_sum: float, step: float) -> np.ndarray:
    """
    Takes a single gradient‑descent move aimed at reducing the current
    worst convolution entry.  ``step`` controls the maximum absolute
    change of a single coordinate.
    """
    grad, _, _ = _max_gradient(seq)
    gmax = np.max(np.abs(grad))
    if gmax == 0.0:
        return seq.copy()
    delta = -step * grad / gmax                     # scaled so max|Δ|=step
    cand = seq + delta
    cand = np.clip(cand, 0.0, 1000.0)
    s = cand.sum()
    if s > 0.0:
        cand = cand * (target_sum / s)             # re‑normalise ℓ¹‑norm
    return cand


# ----------------------------------------------------------------------
# Random pairwise mass transfer (keeps the ℓ¹‑norm)
# ----------------------------------------------------------------------
def _random_transfer(seq: np.ndarray,
                     target_sum: float,
                     frac: float,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Transfers a random amount of mass from one random index to another.
    ``frac`` ≈ maximal fraction of the total mass moved in one step.
    """
    n = seq.shape[0]
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return seq
    max_transfer = min(seq[i], frac * target_sum, 1000.0 - seq[j])
    if max_transfer <= 0.0:
        return seq
    delta = rng.random() * max_transfer
    cand = seq.copy()
    cand[i] -= delta
    cand[j] += delta
    s = cand.sum()
    if s > 0.0:
        cand = cand * (target_sum / s)
    return cand


# ----------------------------------------------------------------------
# Linear‑programming improvement (exact LP as described in the reference)
# ----------------------------------------------------------------------
def _lp_improve(seq: np.ndarray,
                target_sum: float,
                mix: float = 0.2) -> np.ndarray:
    """
    Solves the LP:
        maximise   ∑ g_j
        subject to (seq * g)[k] ≤ M   ∀k,
                   g_j ≥ 0,
    where M = max(seq * seq).  The optimal g is mixed with ``seq``
    using convex coefficient ``mix`` (≈0.2) and re‑normalised.
    If SciPy is unavailable the function returns the input unchanged.
    """
    if not _SCIPY_LP:                     # pragma: no cover
        return seq

    n = seq.shape[0]
    conv = _conv_full_fft(seq)
    M = float(conv.max())

    # Build the Toeplitz matrix A where A[k,i] = seq[i] if 0 ≤ k-i < n else 0.
    # First column = seq padded with zeros, first row = seq[0] followed by zeros.
    col = np.concatenate([seq, np.zeros(n - 1)])
    row = np.concatenate([seq[:1], np.zeros(n - 1)])
    A = np.lib.stride_tricks.as_strided(col,
                                        shape=(2 * n - 1, n),
                                        strides=(col.strides[0], col.strides[0]))
    # Slightly faster than scipy.linalg.toeplitz for our fixed size.
    # The above creates the required Toeplitz matrix implicitly.

    # LP: minimise -∑g  (i.e. maximise ∑g)
    c = -np.ones(n, dtype=np.float64)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'tol': 1e-9})
    if res.status != 0 or not res.success:      # pragma: no cover
        return seq

    g0 = np.maximum(res.x, 0.0)               # numerical safety
    sumg = g0.sum()
    if sumg > 0.0:
        g = g0 * (target_sum / sumg)
    else:
        g = g0

    new_seq = (1.0 - mix) * seq + mix * g
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq = new_seq * (target_sum / s)
    return new_seq


# ----------------------------------------------------------------------
# Local stochastic polishing (Gaussian hill‑climb)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climb that repeatedly adds zero‑mean Gaussian noise.
    The ℓ¹‑norm is *not* forced to stay constant – the final
    ``evaluate_sequence`` will automatically clamp and renormalise.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # slowly anneal the perturbation size
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best, best_score

    if best_seq is None:
        # fallback – uniform vector of length 500
        n = 500
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = best_seq * (math.sqrt(2.0 * n) / n)

    if time.monotonic() < deadline:
        refined_seq, refined_score = _local_search(best_seq, deadline, step_initial=0.5)
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"[final] polished best score = {best_score:.6f}")

    # return a plain Python list as required by the harness
    return [float(v) for v in best_seq]
