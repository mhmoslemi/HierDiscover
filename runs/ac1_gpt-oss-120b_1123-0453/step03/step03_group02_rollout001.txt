def _softmax_weights(conv, beta):
    # conv: numpy array of shape (2n-1)
    # returns same shape probabilistic weights
    if beta <= 0:
        # uniform
        w = np.full_like(conv, 1.0/conv.size)
        return w
    scaled = beta*conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    w = exp_scaled / sum_exp
    return w
```

def _exp_gradient_descent(f0, target_sum, deadline, max_iters=5000, eta0=0.5, beta0=0.1, beta_inc=1.001, decay=0.9995):
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)
    eta = eta0
    beta = beta0
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # compute convolution
        conv = _fft_full_conv(f)   # returns np array
        M = np.max(conv)
        # evaluate maybe store best
        if M < best_score:
            best_score = M
            best_f = f.copy()
        # compute softmax weights
        w = _softmax_weights(conv, beta)
        # compute gradient via convolution
        grad = 2.0 * _conv_linear(w, f)[len(f)-1:len(f)-1+len(f)]
        # update via exponentiated gradient
        f = f * np.exp(-eta * grad)
        # renorm
        f = _normalize_to_sum(f, target_sum)
        # occasionally adjust eta
        eta *= decay
        if eta < 1e-6:
            eta = eta0
        # increase beta gradually to focus on max
        beta *= beta_inc
        # early break if best_score already good.
        if best_score < 1.503:
            break
    return best_f, best_score

Algorithm
---------
1. For a given length `n` we fix the ℓ¹‑norm to
   `S = sqrt(2·n)`.  With this scaling the evaluation

   (local search) polishes the best vector found.

All helpers are top‑level, no closures are used, and the routine
stops a few hundred milliseconds before the global deadline to
guarantee a return.
"""

import time
import math
import numpy as np
from typing import List

# ---------------------------------------------------------
# Basic helpers (FFT based)
# ---------------------------------------------------------
def _fft_full_conv(f: np.ndarray) -> np.ndarray:
    """Full auto‑convolution of a real vector via FFT."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    return conv


def _conv_linear(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (both real) via FFT."""
    size = 1 << ((a.shape[0] + b.shape[0] - 1).bit_length())
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    conv = np.fft.irfft(A * B, size)[: a.shape[0] + b.shape[0] - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(β·conv_i) / Σ exp(β·conv_j)."""
    if beta <= 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    scaled = beta * conv
    max_scaled = np.max(scaled)               # for numerical stability
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_scaled / sum_exp


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project non‑negative vector onto {x ≥ 0, Σx = target}."""
    v = np.maximum(v, 0.0)
    total = np.sum(v)
    if total <= 0.0:
        v = np.full_like(v, target / v.shape[0])
        return v
    v = v * (target / total)
    v = np.clip(v, 0.0, 1000.0)          # respect the evaluate_sequence caps
    total = np.sum(v)
    if total <= 0.0:
        v = np.full_like(v, target / v.shape[0])
    else:
        v = v * (target / total)
    return v


def _score(seq: np.ndarray) -> float:
    """Wrapper around the provided `evaluate_sequence`."""
    # `evaluate_sequence` expects a Python list.
    return float(evaluate_sequence(seq.tolist()))


# ---------------------------------------------------------
# Smoothed exponential‑gradient descent
# ---------------------------------------------------------
def _exp_gradient_descent(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 5000,
    eta0: float = 0.5,
    eta_decay: float = 0.99,
    beta0: float = 0.2,
    beta_factor: float = 1.015,
) -> tuple[np.ndarray, float]:
    """
    Minimise max(f*f) using a soft‑max surrogate and
    exponentiated‑gradient updates, always staying on the ℓ¹‑simplex.
    Returns the best vector seen and its (true) max‑convolution value.
    """
    f = f0.copy()
    best_f = f.copy()
    best_val = _score(f)        # true max, not the surrogate

    eta = eta0
    beta = beta0

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # full auto‑convolution and true max
        conv = _fft_full_conv(f)
        cur_max = float(np.max(conv))

        if cur_max < best_val:
            best_val = cur_max
            best_f = f.copy()

        # soft‑max weights for the current temperature
        w = _softmax_weights(conv, beta)

        # gradient = 2 * (w * f) evaluated at positions i+n‑1
        grad_full = _conv_linear(w, f)                     # length = 3n‑2
        grad = 2.0 * grad_full[f.shape[0] - 1 : f.shape[0] - 1 + f.shape[0]]

        # exponentiated‑gradient step (all entries stay positive)
        f = f * np.exp(-eta * grad)

        # project back to the simplex of fixed ℓ¹‑norm
        f = _normalize_to_sum(f, target_sum)

        # schedule updates
        eta *= eta_decay
        if eta < 1e-6:
            eta = eta0
        beta *= beta_factor

        # early exit if we have already crossed the target
        if best_val < 1.503:
            break

    return best_f, best_val


# ---------------------------------------------------------
# Simple stochastic local search (Gaussian perturbations)
# ---------------------------------------------------------
def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step0: float = 0.5,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """Gaussian hill‑climbing around `start_seq`."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step0
    n = best.shape[0]
    target = math.sqrt(2.0 * n)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize_to_sum(cand, target)

        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        step *= decay
        if step < 1e-7:
            step = step0

    return best, best_score


# ---------------------------------------------------------
# Main entry point
# ---------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The search consists of:
      * several random / padded starts,
      * smoothed exponential‑gradient descent on a few lengths,
      * a final Gaussian local search for polishing.

    The function returns the best vector found before the allotted time.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a safety margin so we always have time to return a result
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # Helper to possibly use the previously discovered construction
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = seq.shape[0]
        target0 = math.sqrt(2.0 * n0)
        seq = _normalize_to_sum(seq, target0)
        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
        print(f"[init] Loaded previous sequence (n={n0}) → score={sc:.6f}")

    # -----------------------------------------------------------------
    # 0️⃣  Define candidate lengths and per‑length budget
    # -----------------------------------------------------------------
    candidate_lengths = [800, 1000, 1200, 1500, 1800, 2000]
    # Give each length roughly the same share of remaining time
    remaining = deadline - time.monotonic()
    per_len_budget = remaining / max(1, len(candidate_lengths))

    for n in candidate_lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        # -----------------------------------------------------------------
        # Generate a list of starting vectors for this length
        # -----------------------------------------------------------------
        starts = []

        # a) Pad the current global best if it exists
        if best_seq is not None:
            if best_seq.shape[0] == n:
                starts.append(best_seq.copy())
            elif best_seq.shape[0] < n:
                padded = np.zeros(n, dtype=np.float64)
                padded[: best_seq.shape[0]] = best_seq
                padded = _normalize_to_sum(padded, target_sum)
                starts.append(padded)

        # b) Uniform vector (often a good baseline)
        uniform = np.full(n, target_sum / n, dtype=np.float64)
        starts.append(uniform)

        # c) A few random Dirichlet‑style starts
        for _ in range(3):
            rand = rng.random(n) + 0.01
            rand = _normalize_to_sum(rand, target_sum)
            starts.append(rand)

        # -----------------------------------------------------------------
        # Run the exponential‑gradient search for each start
        # -----------------------------------------------------------------
        for start_id, f_start in enumerate(starts):
            if time.monotonic() >= deadline:
                break

            # allocate a slice of the remaining budget for this run
            now = time.monotonic()
            remaining_time = deadline - now
            # give this run at most half of the remaining time (leaves room for later runs)
            run_deadline = now + min(remaining_time * 0.5, per_len_budget)

            f_opt, max_val = _exp_gradient_descent(
                f0=f_start,
                target_sum=target_sum,
                deadline=run_deadline,
                max_iters=8000,
                eta0=0.6,
                eta_decay=0.985,
                beta0=0.3,
                beta_factor=1.02,
            )
            sc = _score(f_opt)  # true evaluation (max‑conv)
            if sc < best_score:
                best_score = sc
                best_seq = f_opt.copy()
                print(f"[expGD] n={n} start={start_id} → score={sc:.6f} max_conv={max_val:.6f}")

            if best_score < 1.503:
                print("[expGD] Desired target reached – exiting early.")
                return best_seq.tolist()

    # -----------------------------------------------------------------
    # 1️⃣  Final polishing via stochastic hill‑climbing
    # -----------------------------------------------------------------
    if best_seq is None:
        # fallback: a modest uniform vector
        n = 600
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = _normalize_to_sum(best_seq, math.sqrt(2.0 * n))
        best_score = _score(best_seq)

    if time.monotonic() < deadline:
        refined_seq, refined_score = _local_search(best_seq, deadline, step0=0.6)
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"[local] polished → score={best_score:.6f}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
