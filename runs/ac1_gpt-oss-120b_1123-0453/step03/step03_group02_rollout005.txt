def _full_convolution(f):
    n = f.shape[0]; size = 1 << ((2*n - 1).bit_length())
    F = np.fft.rfft(f, size); conv = np.fft.irfft(F*F, size)[:2*n-1]
    return conv

def _maxconv(f):
    conv = _full_convolution(f)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx

Gradient for worst shift subgradient:

def _subgrad_worst(f, k):
    n = len(f)
    g = np.zeros_like(f)
    i_start = max(0, k-(n-1))
    i_end = min(k, n-1)
    if i_start <= i_end:
        # For i in i_start..i_end, contribution 2*f[k-i]
        idxs = np.arange(i_start, i_end+1)
        g[idxs] = 2.0 * f[k - idxs]
    return g

"""
A fast multi‑phase optimizer for the autocorrelation‑maximisation problem.

Phases
------
1. Initialise from the previously discovered sequence (if any) and evaluate it.
2. For several problem sizes (n = 500 … 2000) run a projected Adam optimiser:
   * The objective is M(f)=max_k (f*f)_k (the maximum entry of the full
     self‑convolution).  For a fixed ℓ¹‑norm Σ f_i = √(2n) the evaluation
     function reduces exactly to M(f).
   * At each iteration we locate the worst shift k* and use the sub‑gradient
     g_i = 2·f_{k*–i} (when the index exists, else 0).  Adam’s first‑ and
     second‑moment accumulators give a smoother update than plain sub‑gradient.
   * After the step we project back onto the non‑negative simplex
     {x ≥ 0, Σ x = √(2n)}.
3. A final stochastic Gaussian hill‑climbing polish (projected local search).

The code respects the security checks inside ``evaluate_sequence`` and stays
well within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Wrapper around the provided `evaluate_sequence`."""
    # ``evaluate_sequence`` expects a plain Python list.
    return float(evaluate_sequence(seq.tolist()))


def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector `v` onto the simplex {x ≥ 0, Σ x = target}."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]

    if rho_arr.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv(f: np.ndarray):
    """
    Return (max_value, argmax_index) of the full auto‑convolution f*f.
    Uses an FFT of length power‑of‑two for speed.
    """
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _subgradient_worst(f: np.ndarray, worst_k: int) -> np.ndarray:
    """
    Sub‑gradient of M(f)=max_k (f*f)_k at the worst shift `worst_k`.
    The gradient is 2·f_{k–i} on the indices that contribute to the worst
    convolution entry, zero elsewhere.
    """
    n = f.shape[0]
    grad = np.zeros_like(f)
    low = max(0, worst_k - (n - 1))
    high = min(worst_k, n - 1)
    if low <= high:
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] = 2.0 * f[worst_k - idx]
    return grad


def _adam_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 8000,
    alpha: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for ``M(f)=max_k (f*f)_k``.
    Returns the best vector seen and its associated max‑convolution value.
    """
    n = f0.shape[0]
    f = f0.copy()
    m = np.zeros_like(f)
    v = np.zeros_like(f)

    best_f = f.copy()
    best_M, _ = _max_conv(f)
    best_score = best_M

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        M, k = _max_conv(f)

        # sub‑gradient at the worst shift
        grad = _subgradient_worst(f, k)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = _project_simplex(f_candidate, target_sum)

        M_cand, _ = _max_conv(f_candidate)

        # keep global best
        if M_cand < best_score:
            best_score = M_cand
            best_f = f_candidate.copy()

        # accept the candidate if it does not increase the max‑convolution
        if M_cand <= M:
            f = f_candidate

    return best_f, best_score


def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.3,
    decay: float = 0.9995,
    min_step: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that respects the ℓ¹‑simplex.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial
    target_sum = float(np.sum(best))

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)

        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        step *= decay
        if step < min_step:
            step = step_initial

    return best, best_score

            best_seq = polished_seq
            best_score = polished_score
            print(f"[polish] improved → score={best_score:.6f}")

    # Return a plain Python list as required by the evaluation harness
    return [float(x) for x in best_seq]
