def _gradient_descent(f, target, n, rng, deadline):
    # f is initial vector
    f_cur = f.copy()
    lam = 0.1
    # steps size schedule
    max_steps = 500  # but break earlier if time

    for step in range(max_steps):
        if time.time() >= deadline:

        w = exp_vals / np.sum(exp_vals)

"""
Projected gradient descent on a smoothed ℓ∞‑objective.

The algorithm works on a fixed length *n* and a target ℓ₁‑sum
    S = √(2 n).
All iterates live on the simplex {x≥0, Σx=S}.  A
soft‑max approximation of the convolution maximum provides a smooth
gradient; a simple convex step with projection is performed.  The
temperature (soft‑max factor) is slowly increased, driving the
approximation from the average convolution towards the true maximum.
After the “annealing” phase a short sub‑gradient refinement (true
ℓ∞‑subgradient) is applied.  Several random restarts and several
sequence lengths (up to 1000) are explored within the given time
budget.  The best sequence found is returned as a plain list of
floats.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, Σx=target}."""
    if target <= 0:
        return np.zeros_like(v)

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    # Find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero after projection
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)

    # Numerical fix: enforce exact sum = target
    sum_w = w.sum()
    if sum_w > 0:
        w *= target / sum_w
    else:
        w.fill(target / len(v))

    return w


def _dirichlet_start(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a point uniformly from the simplex of dimension *n*."""
    x = rng.dirichlet(np.ones(n))
    return x.astype(np.float64) * target


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Return a soft‑max weight vector that approximates the ℓ∞‑indicator."""
    # Numerical stability: subtract the maximum before exponentiating.
    maxc = float(np.max(conv))
    exps = np.exp(lam * (conv - maxc))
    sum_exp = float(np.sum(exps))
    if sum_exp == 0.0:
        # Under‑flow – fall back to a pure arg‑max weight.
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
        return w
    return exps / sum_exp


def _gradient_descent(
    f: np.ndarray,
    target: float,
    n: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """
    Smoothed ℓ∞‑descent.

    The temperature `lam` starts low (smooth) and is increased
    slowly.  After each gradient step the vector is projected back
    onto the simplex.
    """
    f_cur = f.copy()
    lam = 0.1             # initial temperature
    # Number of full iterations – we will break early if time runs out.
    max_iters = 3000

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = np.convolve(f_cur, f_cur)               # length 2n‑1
        w = _softmax_weights(conv, lam)                # soft‑max on convolution

        # Gradient of the smoothed max w.r.t. f:
        #   ∂/∂f_k  Σ_i w_i (f*f)_i = 2· Σ_i w_i f_{i‑k}
        grad = 2.0 * np.correlate(w, f_cur, mode="valid")

        # Step size – shrink as temperature grows.
        alpha = (target / n) * 0.35 / (lam + 1.0)
        f_next = f_cur - alpha * grad
        f_cur = _project_simplex(f_next, target)

        # Anneal temperature (very gentle increase).
        lam = min(lam * 1.025 + 0.01, 250.0)

        # Occasionally report progress (every 200 iterations).
        if (it + 1) % 200 == 0:
            M = float(np.max(np.convolve(f_cur, f_cur)))
            score = 2.0 * n * M / (target ** 2)   # equals M when sum==target
            print(f"[grad] n={n} iter={it+1:4d} λ={lam:.2f} score={score:.6f}")

    return f_cur


def _subgradient_refinement(
    f: np.ndarray,
    target: float,
    n: int,
    deadline: float,
) -> np.ndarray:
    """
    Sub‑gradient descent on the true ℓ∞‑objective.
    Uses the active set of indices attaining the maximum.
    """
    f_cur = f.copy()
    max_iters = 800

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = np.convolve(f_cur, f_cur)
        M = float(np.max(conv))

        # Active set – indices whose value is within a tiny tolerance of the max.
        eps = max(M * 1e-9, 1e-12)
        active = np.where(conv >= M - eps)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[int(np.argmax(conv))] = 1.0
        else:
            w[active] = 1.0 / active.size

        grad = 2.0 * np.correlate(w, f_cur, mode="valid")
        # Diminishing step size.
        alpha = (target / n) * 0.25 / math.sqrt(it + 1.0)
        f_next = f_cur - alpha * grad
        f_cur = _project_simplex(f_next, target)

        if (it + 1) % 200 == 0:
            M = float(np.max(np.convolve(f_cur, f_cur)))
            score = 2.0 * n * M / (target ** 2)
            print(f"[subg] n={n} iter={it+1:4d} score={score:.6f}")

    return f_cur


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a small evaluation score.

    The routine explores several lengths (up to 1000) and a handful of
    random restarts per length.  Each restart performs:
        1. Smoothed ℓ∞‑gradient descent with a temperature schedule.
        2. A short true ℓ∞‑sub‑gradient refinement.
    The best sequence seen before the time budget expires is returned.
    """
    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # Lengths to try – larger lengths give a lower theoretical bound.
    lengths = [300, 400, 500, 600, 800, 1000]
    restarts_per_len = 3

    # Attempt to reuse a globally supplied construction if it matches the length.
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        global_seq = np.asarray(global_seq, dtype=np.float64)
    else:
        global_seq = None

    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        for restart in range(restarts_per_len):
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------------
            # Initialise the vector.
            if restart == 0 and global_seq is not None and len(global_seq) == n:
                f = global_seq.copy()
                f = np.maximum(f, 0.0)
                f = f / np.sum(f) * target
                print(f"[init] Re‑using global seed n={n}")
            else:
                f = _dirichlet_start(n, target, rng)

            # Record the initial score.
            M0 = float(np.max(np.convolve(f, f)))
            score0 = 2.0 * n * M0 / (target ** 2)   # equals M0 when sum==target
            if score0 < best_score:
                best_score = score0
                best_seq = f.copy()
                print(f"[init] n={n} restart={restart} score={score0:.6f}")

            # ------------------------------------------------------------------
            # 1) Smoothed gradient descent (annealing).
            f = _gradient_descent(f, target, n, rng, deadline)

            # 2) Sub‑gradient polishing.
            f = _subgradient_refinement(f, target, n, deadline)

            # ------------------------------------------------------------------
            # Final evaluation of this candidate.
            M_final = float(np.max(np.convolve(f, f)))
            final_score = 2.0 * n * M_final / (target ** 2)

            if final_score < best_score:
                best_score = final_score
                best_seq = f.copy()
                print(f"[best] n={n} restart={restart} final_score={final_score:.6f}")

    # ----------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was produced.
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = 2.0 * n * np.max(np.convolve(best_seq, best_seq)) / (target ** 2)

    # Verify the score with the provided evaluation function (extra safety).
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        pass

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    return [float(v) for v in best_seq]
