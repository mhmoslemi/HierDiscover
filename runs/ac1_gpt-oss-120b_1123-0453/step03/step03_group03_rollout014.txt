```
def project_simplex(v, s):
    # Ensure v is 1D np.ndarray
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find rho: largest i where u_i > cssv_i / i
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

import time, math, numpy as np

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto the simplex {x >=0, sum x = s}."""
    # Implementation of the algorithm from
    # https://doi.org/10.1137/080738970
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_from_f(f: np.ndarray) -> float:
    """Return the evaluation score for a fixed‑norm vector f."""
    n = f.shape[0]
    sum_a = f.sum()
    if sum_a < 0.01:
        return float("inf")
    max_b = np.max(np.convolve(f, f))
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _softmax_gradient(f: np.ndarray, lam: float):
    """
    Compute the gradient of the smoothed max (log‑sum‑exp) of the auto‑convolution.
    Returns (grad, current_max).
    """
    conv = np.convolve(f, f)
    cur_max = conv.max()
    # soft‑max weights (numerically stable)
    shift = conv - cur_max          # max term becomes 0
    w = np.exp(lam * shift)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # gradient = 2 * (w * f) cross‑correlation.
    rev_f = f[::-1]
    grad_full = np.convolve(w, rev_f)                     # length 3n‑2
    n = f.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]            # slice of length n
    return grad, cur_max

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project *v* onto the simplex {x ≥ 0, Σ x = s}."""
    # algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.size + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_f(f: np.ndarray) -> float:
    """Score = 2·n·max(f*f) / (Σ f)²  (the same as ``evaluate_sequence``)."""
    n = f.shape[0]
    s = f.sum()
    if s < 0.01:
        return float("inf")
    m = np.max(np.convolve(f, f))
    return float(2.0 * n * m / (s * s))


def _softmax_gradient(f: np.ndarray, lam: float):
    """
    Gradient of the smoothed maximum
        L(f) = (1/lam)·log Σ_i exp(lam·(f*f)_i)
    Returns (grad, current_max).
    """
    conv = np.convolve(f, f)               # length 2·n‑1
    cur_max = conv.max()

    # soft‑max weights – numerically stable because max term → 0
    shift = conv - cur_max
    w = np.exp(lam * shift)
    ws = w.sum()
    if ws == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / ws

    # grad = 2·(W·f) where W_{jk}=w_{j+k}
    rev_f = f[::-1]
    grad_full = np.convolve(w, rev_f)        # length 3·n‑2
    n = f.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice of length n
    return grad, cur_max


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Gradient‑descent optimiser for the auto‑convolution bound.

    The algorithm works on a fixed ℓ¹‑norm (Σ a_i = √(2·n), which makes the
    evaluation score equal to the maximum entry of a*a).  Starting from the
    global ``height_sequence_1`` when it exists (otherwise a random vector) it
    repeatedly:
        * evaluates a smoothed‑max gradient,
        * takes a step opposite to the gradient,
        * projects back onto the simplex,
        * adapts the step size and the smoothing parameter,
    keeping the best vector seen.  After the main loop a tiny stochastic
    polishing is performed until the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # a small safety margin – the caller reads the result after the function returns
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # The previously discovered construction uses n = 500 and already achieves a
    # score ≈ 1.511.  We concentrate on this length, because the score is scale‑free.
    lengths = [500]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # a few independent restarts – each gets many inner iterations
        for restart in range(4):
            if time.monotonic() >= deadline:
                break

            # --------------------------------------------------------------
            # Initialise the vector
            # --------------------------------------------------------------
            if restart == 0 and "height_sequence_1" in globals():
                try:
                    seq0 = np.array(height_sequence_1, dtype=float)
                    if seq0.size != n:
                        # truncate or pad to the required length
                        if seq0.size > n:
                            seq0 = seq0[:n]
                        else:
                            seq0 = np.concatenate([seq0,
                                                   np.zeros(n - seq0.size, dtype=float)])
                    f = seq0.copy()
                except Exception:
                    f = rng.random(n)
            else:
                f = rng.random(n)

            f = np.clip(f, 0.0, 1000.0)
            f = f * (target_sum / f.sum())

            cur_score = _score_from_f(f)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            # --------------------------------------------------------------
            # Main gradient‑descent loop
            # --------------------------------------------------------------
            step = 0.5          # initial step magnitude
            lam = 1.0           # smoothing of the max
            it = 0
            no_improve = 0
            max_no_improve = 300

            while time.monotonic() < deadline and it < 5000:
                it += 1

                # gradient of the smoothed max
                grad, cur_max = _softmax_gradient(f, lam)

                # take a step and project onto the simplex
                cand = f - step * grad
                cand = _project_simplex(cand, target_sum)

                cand_max = np.max(np.convolve(cand, cand))

                # accept if we really improved the true max
                if cand_max < cur_max - 1e-9:
                    f = cand
                    cur_max = cand_max
                    no_improve = 0
                    step = min(step * 1.05, 5.0)          # gently enlarge step on success
                else:
                    no_improve += 1
                    step *= 0.7                            # shrink step on failure
                    if step < 1e-7:
                        step = 0.5
                    if no_improve >= max_no_improve:
                        # stagnation – abort this restart
                        break

                # very slow increase of the smoothing parameter (approaches true max)
                lam = min(lam * 1.001, 30.0)

                cur_score = _score_from_f(f)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = f.copy()
                    print(f"[gd] n={n} restart={restart} it={it} score={best_score:.6f} M={cur_max:.6f}")

                if best_score < 1.503:
                    print("[gd] target reached – exiting early.")
                    return best_seq.tolist()

    # ------------------------------------------------------------------
    # Fallback (should never happen, but keeps the function safe)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0, dtype=float)
        best_seq = best_seq / best_seq.sum() * math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Final tiny stochastic polishing
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    step = 0.2
    while time.monotonic() < deadline:
        cand = cur + rng.normal(scale=step, size=cur.shape)
        cand = _project_simplex(cand, math.sqrt(2.0 * cur.shape[0]))
        cand_score = _score_from_f(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            cur = cand
            print(f"[polish] improved to {best_score:.6f}")
        step *= 0.9995
        if step < 1e-6:
            step = 0.2

    return [float(x) for x in best_seq]
