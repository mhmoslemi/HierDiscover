    # But scaling doesn't affect score. However we keep sum moderate to avoid clipping.
    # ensure sum > 0.01.
    
    best_seq = seq.copy()
    best_score = _score(best_seq)
    cur_seq = seq.copy()
    cur_score = best_score
    
    # Precompute n
    n = len(seq)
    
    # iterations
    iter_count = 0
    # SA schedule parameters
    T0 = 0.0015
    T_end = 1e-6
    # compute total allowed iterations approx:
    # We can adjust temperature linearly in log scale based on iter fraction.
    while time.monotonic() < deadline:
        # compute conv and max
        conv = np.convolve(cur_seq, cur_seq)
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])
        # compute contributions as defined
        i_vals = np.arange(n)
        j_vals = max_idx - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = cur_seq[i_vals[mask]] * cur_seq[j_vals[mask]]
        total_contrib = contrib.sum()
        # choose source
        if total_contrib > 0 and rng.random() < 0.9:
            prob = contrib / total_contrib
            i_sel = rng.choice(n, p=prob)
        else:
            i_sel = rng.integers(n)
        # choose dest low contribution
        low_thresh = np.percentile(contrib, 30)
        low_candidates = np.where(contrib <= low_thresh)[0]
        if low_candidates.size > 0:
            j_sel = rng.choice(low_candidates)
        else:
            j_sel = rng.integers(n)
        if i_sel == j_sel:
            continue
        # epsilon fraction up to 30% of source value
        source_val = cur_seq[i_sel]
        if source_val <= 0:
            continue
        eps = rng.uniform(0, 0.3) * source_val
        if eps <= 0:
            continue
        cand = cur_seq.copy()
        cand[i_sel] -= eps
        cand[j_sel] += eps
        # clip ensures non negative
        cand = np.clip(cand, 0.0, 1000.0)
        # evaluate
        cand_score = _score(cand)
        # Simulated annealing acceptance based on cur_score or best?
        # We want to accept if candidate improves current seq or by temperature.
        accept = False
        if cand_score < cur_score:
            accept = True
        else:

            elapsed = time.monotonic() - start
            frac = elapsed / (deadline - start)
            T = T0 * (1 - frac) + T_end * frac
            # acceptance probability
            prob_accept = math.exp(-(cand_score - cur_score) / T) if T > 0 else 0.0
            accept = rng.random() < prob_accept

starts from the best known construction (``height_sequence_1`` if
available).  At each iteration we

* compute the auto‑convolution and locate the shift with the maximal
  value,
* estimate each coefficient’s contribution to that peak,
* move a modest amount of mass from a high‑contribution index to a
  low‑contribution one,
* accept the move if it improves the score or with a temperature‑based
  probability (to escape local minima).

When the temperature has cooled we finish with a short Gaussian
polishing phase.  The routine runs until the supplied time budget (≈
1000 s) is exhausted and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – thin wrapper around the provided evaluation function
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """
    Evaluate a candidate sequence using the given ``evaluate_sequence``.
    The function expects a plain Python list, hence the conversion.
    """
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The search uses simulated‑annealing‑style mass transfers together
    with a final Gaussian polishing stage.  It respects the supplied
    time budget (safety margin 0.5 s) and returns the best sequence
    discovered as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin for the caller
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise a starting sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.array(height_sequence_1, dtype=float)
            # Ensure the sequence obeys the clipping rules used by the
            # evaluator – values >1000 are capped and negatives are raised to 0.
            seq0 = np.clip(seq0, 0.0, 1000.0)
            # Avoid the pathological case of an almost‑zero sum
            if seq0.sum() < 0.01:
                seq0 = np.full_like(seq0, 1.0)
        except Exception:  # pragma: no cover
            seq0 = None
    else:
        seq0 = None

    if seq0 is None:
        # fallback: a slightly perturbed uniform vector of length 500
        n = 500
        seq0 = rng.random(n) + 0.1
        seq0 = np.clip(seq0, 0.0, 1000.0)

    n = seq0.size
    cur_seq = seq0.copy()
    cur_score = _score(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score

    print(f"[init] length={n} start_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    T_initial = 0.012          # initial temperature (approx. typical Δscore)
    T_final = 1e-7             # final temperature
    max_no_improve = 3000      # after this many iterations without a new best, we perturb

    it = 0
    no_improve_counter = 0

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # 1.  Compute auto‑convolution and locate the dominant peak
        # ------------------------------------------------------------------
        conv = np.convolve(cur_seq, cur_seq)
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])

        # ------------------------------------------------------------------
        # 2.  Compute each coefficient's contribution to that peak
        # ------------------------------------------------------------------
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(cur_seq)
        contrib[mask] = cur_seq[i[mask]] * cur_seq[j[mask]]
        total_contrib = contrib.sum()

        # ------------------------------------------------------------------
        # 3.  Choose a source index (high contribution) and a destination
        # ------------------------------------------------------------------
        if total_contrib > 0 and rng.random() < 0.9:
            # weighted choice favouring large contributors
            prob = contrib / total_contrib
            src = rng.choice(n, p=prob)
        else:
            src = rng.integers(n)

        # destination: pick from the lower 30 % of contributions
        low_thr = np.percentile(contrib, 30)
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size > 0:
            dst = rng.choice(low_candidates)
        else:
            dst = rng.integers(n)

        if src == dst:
            continue

        src_val = cur_seq[src]
        if src_val <= 0.0:
            continue

        # ------------------------------------------------------------------
        # 4.  Propose moving a fraction of the source mass to the destination
        # ------------------------------------------------------------------
        eps = rng.uniform(0.0, 0.3) * src_val   # up to 30 % of the source value
        if eps <= 0.0:
            continue

        cand = cur_seq.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _score(cand)

        # ------------------------------------------------------------------
        # 5.  Acceptance rule (annealed)
        # ------------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # linear temperature schedule based on elapsed fraction
            elapsed = time.monotonic() - start_time
            frac = elapsed / (deadline - start_time)
            T = T_initial * (1.0 - frac) + T_final * frac
            if T > 0.0:
                prob_accept = math.exp(-(cand_score - cur_score) / T)
                accept = rng.random() < prob_accept

        if accept:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-9:
                best_seq = cand.copy()
                best_score = cand_score
                no_improve_counter = 0
                print(f"[SA] iter={it} new best {best_score:.6f}")
                # early exit if we already beat the target bound
                if best_score < 1.503:
                    print("[SA] target achieved – exiting early.")
                    return best_seq.tolist()
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1

        # ------------------------------------------------------------------
        # 6.  Random restart / perturbation when stuck
        # ------------------------------------------------------------------
        if no_improve_counter >= max_no_improve:
            # perturb the current best a little and continue
            perturb = rng.normal(scale=0.05 * best_seq.mean(), size=n)
            cur_seq = best_seq + perturb
            cur_seq = np.clip(cur_seq, 0.0, 1000.0)
            cur_score = _score(cur_seq)
            no_improve_counter = 0
            print(f"[restart] iter={it} re‑seeded from best {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Final Gaussian polishing (local search) – runs if time permits
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        cur = best_seq.copy()
        step = max(0.1, cur.mean() * 0.1)
        while time.monotonic() < deadline:
            cand = cur + rng.normal(scale=step, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = _score(cand)
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                cur = cand.copy()
                print(f"[polish] improved to {best_score:.6f}")
                if best_score < 1.503:
                    print("[polish] target reached – exiting.")
                    break
            # gradual cooling of the perturbation size
            step *= 0.9995
            if step < 1e-6:
                step = max(0.1, cur.mean() * 0.1)

    print(f"[final] best score = {best_score:.6f} (length={n})")
    return [float(x) for x in best_seq]
