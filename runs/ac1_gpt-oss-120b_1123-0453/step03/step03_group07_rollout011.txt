"""
Simple‐gradient “peak‑shaving’’ search.

* Normalises every candidate to the standard sum √(2 n) so that the
  evaluation equals the peak value of the aperiodic auto‑convolution.
* Starts from a diverse pool of seeds (global best, Dirichlet
  draws, exponential, block‑uniform shapes) for several lengths.
* Repeatedly picks the current maximal convolution entry, follows the
  (sub‑)gradient of that entry (∂c_k/∂a_i = 2·a_{k‑i}) and projects back
  onto the non‑negative simplex.
* A tiny back‑tracking line‑search finds a step that really lowers the
  peak; if progress stalls the step size is reduced and, after many
  attempts, a small random perturbation is applied to escape plateaus.
* The whole search stops when the 1000 s budget is exhausted; the best
  sequence ever seen is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _normalize(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to non‑negative, then scale so that the sum is exactly *target_sum*."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        # fallback – uniform distribution
        v = np.full_like(v, target_sum / v.size, dtype=np.float64)
    else:
        v = v * (target_sum / s)
    return v


def _conv_max(v: np.ndarray) -> float:
    """Return max_{k} (v * v)[k] (full aperiodic convolution)."""
    c = np.convolve(v, v)
    return float(c.max())


def _gradient_of_peak(v: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry with index *peak_idx*.
    For aperiodic convolution c_k = Σ_i v_i·v_{k‑i},
    ∂c_k/∂v_i = 2·v_{k‑i}  (0 if the index is out of range).
    """
    n = v.shape[0]
    g = np.zeros_like(v)

    i_min = max(0, peak_idx - (n - 1))
    i_max = min(peak_idx, n - 1)
    for i in range(i_min, i_max + 1):
        j = peak_idx - i
        if 0 <= j < n:
            g[i] += 2.0 * v[j]
    return g


def _dirichlet_start(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex via a Dirichlet(1,…,1) distribution."""
    x = rng.dirichlet(np.ones(n))
    return _normalize(x.astype(np.float64), target_sum)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The method explores several lengths, starts from many diverse seeds,
    and repeatedly performs a gradient‑step that reduces the current
    maximal convolution entry.  Random perturbations keep the search
    from getting stuck.  The best sequence found before the time budget
    expires is returned.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # tiny safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Lengths to try – larger n usually allow a smaller peak.
    lengths = [300, 400, 500, 600, 800, 1000]
    # Number of independent starts per length.
    starts_per_n = 6
    # Gradient‑descent parameters.
    max_outer_iters = 2000          # safety cap per start
    patience_before_perturb = 30    # steps without improvement before a random kick

    # make the global seed (if any) available
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        global_seq = np.asarray(global_seq, dtype=np.float64)
    else:
        global_seq = None

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)      # the normalisation used in the competition

        # ----------------------- build a pool of initial vectors -------------
        pool = []

        # 1) reuse the provided construction when the length matches
        if global_seq is not None and len(global_seq) == n:
            f = _normalize(global_seq.copy(), target_sum)
            pool.append(f)

        # 2) several random Dirichlet draws
        for _ in range(2):
            pool.append(_dirichlet_start(n, target_sum, rng))

        # 3) exponential shapes – a useful asymmetric family
        for lam in [0.001, 0.005, 0.02]:
            x = np.exp(-lam * np.arange(n))
            pool.append(_normalize(x, target_sum))

        # 4) block‑uniform shapes (mass concentrated at one end)
        for frac in [0.2, 0.5, 0.8]:
            k = max(1, int(frac * n))
            x = np.zeros(n, dtype=np.float64)
            x[:k] = 1.0
            pool.append(_normalize(x, target_sum))

        # --------------------------------------------------------------------
        for start_id, f in enumerate(pool):
            if time.time() >= deadline:
                break

            # initial evaluation (the evaluation equals the peak when normalised)
            cur_peak = _conv_max(f)
            cur_score = cur_peak
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[init] n={n} start={start_id} score={cur_score:.6f}")

            # ----------- main gradient‑descent / perturbation loop -------------
            step_factor = 0.5                # controls the initial step size
            no_improve = 0
            it = 0

            while it < max_outer_iters and time.time() < deadline:
                it += 1

                # locate the current worst (maximal) convolution entry
                conv_f = np.convolve(f, f)
                peak_idx = int(np.argmax(conv_f))
                peak_val = float(conv_f.max())

                # compute sub‑gradient of that entry
                grad = _gradient_of_peak(f, peak_idx)

                # try a few back‑tracking step sizes (geometric series)
                improved = False
                base_step = step_factor * target_sum / n
                for mul in (1.0, 0.5, 0.25, 0.125):
                    step = base_step * mul
                    cand = f - step * grad
                    cand = _normalize(cand, target_sum)

                    cand_peak = _conv_max(cand)
                    if cand_peak + 1e-12 < cur_score:          # improvement
                        f = cand
                        cur_score = cand_peak
                        cur_peak = cand_peak
                        step_factor = min(step_factor * 1.05, 3.0)   # allow larger steps later
                        improved = True
                        no_improve = 0
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = f.copy()
                            print(f"[gd] n={n} start={start_id} it={it} score={cur_score:.6f}")
                        break

                if not improved:
                    # no gain – shrink the step and count stagnation
                    step_factor *= 0.7
                    no_improve += 1

                    if no_improve >= patience_before_perturb:
                        # a random kick to hop out of a flat region
                        perturb = rng.normal(scale=0.02 * target_sum, size=n)
                        f = _normalize(f + perturb, target_sum)
                        cur_score = _conv_max(f)
                        no_improve = 0
                        print(f"[perturb] n={n} start={start_id} it={it} score={cur_score:.6f}")

                # early stop if we are already extremely close to the theoretical
                # lower bound 2 n/(2 n‑1) ≈ 1 for large n
                lower_bound = 2.0 * n / (2.0 * n - 1.0)
                if cur_score <= lower_bound * 1.02:
                    # good enough – keep the best we have for this length
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                    break

    # ----------------------------------------------------------------------
    # Safety fallback – should never be needed, but keeps the function total.
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _conv_max(best_seq)

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    # Return a plain Python list as required by the competition harness.
    return [float(v) for v in best_seq]
