"""
Propose a step‑function that minimizes the auto‑convolution bound.

Main ideas
----------
1. **LP refinement (if cvxpy is available)** –  
   For a current vector `f` we solve the linear programme  

        minimise   M
        s.t.       (f * g)[k] ≤ M   for all k           (linear in g)
                    g ≥ 0
                    Σ g = √(2·n)                    (fixed ℓ¹‑norm)

   The optimum `g` is a direction that can lower the worst
   auto‑convolution peak.  Rather than mixing with the small
   discrete set of weights used previously, we perform a fine
   grid search on the mixing coefficient `t∈[0,1]` and keep the
   best resulting sequence.

2. **Smart greedy peak reduction** –  
   For the current maximum index `k*` we compute the contribution
   `c_i = a_i·a_{k*‑i}` of each coefficient.  A source index is
   sampled proportionally to `c_i` and a destination is taken among
   the low‑contribution indices.  The optimal amount of mass to
   move is found by a tiny ternary search on the (convex) function  

        M(t) = max_k (conv_f[k] + t·s_k) ,

   where the slope `s_k = a_{k‑dst} – a_{k‑src}`.  The move is
   accepted only if the true evaluation score improves.

3. **Final stochastic polishing** –  
   Small Gaussian perturbations with an exponentially decaying
   step size are applied until the global deadline is reached.

All steps respect the security checks of ``evaluate_sequence``:
non‑negative entries, clipping to ``[0,1000]`` and a positive ℓ¹‑norm.
The routine works under the 1000 s budget and returns the best
sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional LP backend
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2·n‑1)×n Toeplitz matrix A with (A@g)[k] = (f*g)[k]."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # column index = k - i
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _manual_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv) / (sum(seq)²)  (identical to evaluate_sequence)."""
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    max_b = float(conv.max())
    n = seq.size
    return 2.0 * n * max_b / (sum_a * sum_a)


def _solve_min_m_lp(f: np.ndarray, target_sum: float):
    """
    Solve  minimise M
           s.t.   (f * g) ≤ M,
                  g ≥ 0,
                  Σ g = target_sum.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable(nonneg=True)

    prob = cp.Problem(cp.Minimize(M), [A @ g <= M, cp.sum(g) == target_sum])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, abstol=1e-8, reltol=1e-6)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True,
                   max_iters=20000, verbose=False)

    if g.value is None or M.value is None:
        return None, None
    g_opt = np.maximum(g.value, 0.0)
    return g_opt, float(M.value)


def _best_mix(f: np.ndarray, g: np.ndarray, steps: int = 21):
    """
    Search the mixing coefficient t∈[0,1] on a uniform grid (steps points)
    and return the sequence with the smallest max‑convolution value.
    """
    best_seq = f
    best_M = np.max(np.convolve(f, f))

    ts = np.linspace(0.0, 1.0, steps)
    for t in ts[1:]:          # skip t=0 (keeps f)
        cand = (1.0 - t) * f + t * g
        cand = np.clip(cand, 0.0, 1000.0)
        curM = np.max(np.convolve(cand, cand))
        if curM < best_M:
            best_M = curM
            best_seq = cand
    return best_seq, best_M


def _optimal_eps(seq: np.ndarray, conv: np.ndarray, src: int, dst: int):
    """
    For a move “src → dst” compute (approximately) the ε∈[0,seq[src]]
    that minimises the new max‑convolution value.
    Returns (ε_opt, M_opt) where M_opt = max(conv + ε·s).
    """
    n = seq.shape[0]
    L = 2 * n - 1
    k = np.arange(L)

    # a_{k‑src} and a_{k‑dst} (zero outside the original range)
    a_src = np.where((k - src >= 0) & (k - src < n), seq[k - src], 0.0)
    a_dst = np.where((k - dst >= 0) & (k - dst < n), seq[k - dst], 0.0)

    slope = a_dst - a_src
    max_eps = seq[src]
    if max_eps <= 0.0:
        return 0.0, conv.max()

    lo, hi = 0.0, max_eps
    # ternary search on a convex piecewise‑linear function
    for _ in range(25):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        m1 = np.max(conv + t1 * slope)
        m2 = np.max(conv + t2 * slope)
        if m1 < m2:
            hi = t2
        else:
            lo = t1
    eps_opt = (lo + hi) / 2.0
    M_opt = np.max(conv + eps_opt * slope)
    return eps_opt, M_opt

        j_arr = max_idx - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros_like(seq)
        contrib[mask] = seq[i_arr[mask]] * seq[j_arr[mask]]
        total = contrib.sum()
        if total == 0.0:
            break

        # source index – probability proportional to its contribution
        prob_src = contrib / total
        src = rng.choice(n, p=prob_src)

        # destination index – prefer low contribution to the peak
        low_thr = np.percentile(contrib, 30)
        low_cands = np.where(contrib <= low_thr)[0]
        if low_cands.size == 0:
            dst = rng.integers(n)
        else:
            dst = rng.choice(low_cands)

        if src == dst:
            continue

        # optimal amount of mass to move (ternary search on M)
        eps_opt, _ = _optimal_eps(seq, conv, src, dst)
        if eps_opt <= 0.0:
            continue

        cand = seq.copy()
        cand[src] -= eps_opt
        cand[dst] += eps_opt
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _manual_score(cand)
        if cand_score < best_score - 1e-9:
            seq = cand
            best_seq = cand.copy()
            best_score = cand_score
            print(f"[greedy] improved to {best_score:.6f}")

        if best_score < 1.503:
            print("[greedy] target bound reached.")
            break

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic polishing
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        cur = best_seq.copy()
        step = 0.4
        while time.monotonic() < deadline:
            cand = cur + rng.normal(scale=step, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = _manual_score(cand)
            if cand_score < best_score:
                cur = cand
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[local] improved to {best_score:.6f}")
            step *= 0.9997
            if step < 1e-6:
                step = 0.4   # restart step size if it becomes too small

    return [float(x) for x in best_seq.tolist()]
