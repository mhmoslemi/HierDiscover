def gradient_descent(f, target_sum, deadline, max_iters=2000, eta0=1.0):
    n = len(f)
    v = np.zeros_like(f)
    beta = 0.9
    best_f = f.copy()
    best_M, _ = _fft_conv_max(f)
    best_score = best_M
    eta = eta0
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        M, k = _fft_conv_max(f)
        if M < best_score:
            best_score = M
            best_f = f.copy()
        # compute subgradient at k
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        g = np.zeros_like(f)
        if low <= high:
            idx = np.arange(low, high+1)
            g[idx] = 2.0 * f[k - idx]
        # momentum update
        v = beta * v + (1 - beta) * g
        # candidate
        f_candidate = _project_simplex(f - eta * v, target_sum)
        M_candidate, _ = _fft_conv_max(f_candidate)
        if M_candidate < M:
            f = f_candidate
            # maybe increase step a bit
            eta = min(eta * 1.05, eta0)
        else:
            eta *= 0.5
        # ensure eta within bounds
        if eta < 1e-6:
            eta = eta0
        # optional decay?
        eta *= 0.9995
    return best_f, best_score

def simulated_annealing(start_seq, deadline, init_temp=0.5, decay=0.9995):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_M, _ = _fft_conv_max(cur)
    best = cur.copy()
    best_M = cur_M
    temp = init_temp
    while time.monotonic() < deadline:
        # propose move: pick two indices i,j, transfer delta
        i = rng.integers(len(cur))
        j = rng.integers(len(cur))
        if i == j: continue
        # delta magnitude random up to some fraction of cur[i]
        max_delta = cur[i] * 0.5
        if max_delta <= 0:
            continue
        delta = rng.uniform(0, max_delta)
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure non-negative:
        # Already we reduced i, increased j, so others unchanged, keep non-negative.
        # Project? Not needed because sum unchanged and nonneg.
        # compute M
        M_cand, _ = _fft_conv_max(cand)
        if M_cand < cur_M or rng.random() < math.exp(-(M_cand - cur_M) / max(temp, 1e-9)):
            cur = cand
            cur_M = M_cand
            if cur_M < best_M:
                best_M = cur_M
                best = cur.copy()
        temp *= decay
        if temp < 1e-6:
            temp = init_temp
    return best, best_M

def propose_candidate(seed=0, budget_s=1000):

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 0.5)
    best_seq = None
    best_score = float('inf')
    # load previous if exists
    if 'height_sequence_1' in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        # project onto simplex with its length
        n0 = seq.shape[0]
        target0 = math.sqrt(2.0 * n0)
        seq = _project_simplex(seq, target0)
        score0 = _score(seq)
        best_seq = seq
        best_score = score0
        print(f"[init] loaded height_sequence_1 n={n0} score={score0:.6f}")
    # define length candidates
    lengths = [800, 1200, 1800, 2500, 3500, 5000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # collect start vectors
        starts = []
        # embed previous best if size differs
        if best_seq is not None:
            if best_seq.shape[0] == n:
                starts.append(best_seq.copy())
            else:

                x_old = np.linspace(0, 1, best_seq.shape[0])

                interp = np.interp(x_new, x_old, best_seq)

                starts.append(interp)

            f0 = rng.random(n) + 0.1
            f0 = f0 / np.sum(f0) * target_sum
            starts.append(f0)
        # decreasing linear start
        lin = np.linspace(1.0, 0.0, n)
        lin = _project_simplex(lin, target_sum)
        starts.append(lin)
        # iterate over starts
        for idx, f_start in enumerate(starts):
            if time.monotonic() >= deadline:
                break
            # allocate sub-deadline portion: a fraction of remaining time
            remaining = deadline - time.monotonic()

"""
Propose a low‑evaluation step sequence.

The algorithm works in three phases:

1️⃣ **Initialisation** – start from a previously discovered sequence (if any)
   and create several diverse restarts (random, linear, interpolated longer
   versions).

2️⃣ **Projected (momentum) sub‑gradient descent** – minimise the worst entry
   of the auto‑convolution (which, after fixing the ℓ¹‑norm to √(2·n),
   equals the evaluation score).  After each sub‑gradient step the vector is
   projected onto the simplex {x ≥ 0, Σx = √(2·n)}.  A simple momentum term,
   adaptive step size and early stopping are used.

3️⃣ **Simulated‑annealing polish** – a cheap stochastic hill‑climber that
   repeatedly moves a random amount of mass from one coordinate to another.
   Moves that improve the maximal convolution are always accepted; worse
   moves are accepted with a temperature‑dependent probability.

All phases respect the security checks inside `evaluate_sequence`.  The
routine stops a few hundred milliseconds before the global time budget
to guarantee a return.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Convenient wrapper for the provided `evaluate_sequence`."""
    # `evaluate_sequence` expects a Python list.
    from __main__ import evaluate_sequence  # injected by the harness

    return float(evaluate_sequence(seq.tolist()))


def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project the vector `v` onto the simplex {x ≥ 0, Σx = target}."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending order
    cssv = np.cumsum(u) - target
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = -1
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


def _fft_conv_max(f: np.ndarray):
    """Return (max_entry, argmax_index) of the full auto‑convolution f∗f."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def _gradient_descent_momentum(
    f_start: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 5000,
    eta0: float = 1.0,
    beta: float = 0.9,
    min_eta: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Minimise `M(f) = max_k (f*f)_k` using projected sub‑gradient descent
    with a simple momentum term.
    Returns the best vector seen and its associated M‑value.
    """
    n = f_start.shape[0]
    f = f_start.copy()
    v = np.zeros_like(f)                # momentum accumulator
    eta = eta0

    best_f = f.copy()
    best_M, _ = _fft_conv_max(f)
    best_val = best_M

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        M, k = _fft_conv_max(f)

        if M < best_val:
            best_val = M
            best_f = f.copy()

        # ------ sub‑gradient at the worst index k ------
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        g = np.zeros_like(f)
        if low <= high:
            idx = np.arange(low, high + 1)
            g[idx] = 2.0 * f[k - idx]

        # ------ momentum update ------
        v = beta * v + (1.0 - beta) * g

        # ------ tentative step and projection ------
        f_cand = _project_simplex(f - eta * v, target_sum)

        M_cand, _ = _fft_conv_max(f_cand)

        # ------ simple acceptance rule + adaptive step size ------
        if M_cand < M:
            f = f_cand
            eta = min(eta * 1.05, eta0)          # gently increase if improving
        else:
            eta *= 0.5

        if eta < min_eta:
            eta = eta0

        # slight exponential decay for stability
        eta *= 0.9995

    return best_f, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (stochastic hill‑climbing)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start: np.ndarray,
    deadline: float,
    init_temp: float = 0.8,
    decay: float = 0.9997,
    min_temp: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Randomly move mass between two coordinates while the total ℓ¹‑norm is
    kept fixed.  Moves are accepted using a temperature schedule.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_M, _ = _fft_conv_max(cur)
    best = cur.copy()
    best_M = cur_M
    temp = init_temp

    n = cur.shape[0]

    while time.monotonic() < deadline:
        # choose two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a random fraction of the content of i towards j
        delta = rng.random() * cur[i] * 0.5
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # all entries stay non‑negative, sum unchanged
        M_cand, _ = _fft_conv_max(cand)

        # Metropolis acceptance
        if M_cand < cur_M or rng.random() < math.exp(-(M_cand - cur_M) / max(temp, 1e-12)):
            cur = cand
            cur_M = M_cand
            if cur_M < best_M:
                best_M = cur_M
                best = cur.copy()

        temp = max(temp * decay, min_temp)

    return best, best_M

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:

    Search for a non‑negative step sequence with a low evaluation score.

    The routine consists of:

      • several runs of projected momentum sub‑gradient descent on lengths

      • a final simulated‑annealing polishing stage.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

        target_sum = math.sqrt(2.0 * n)

        # (b) decreasing linear profile (often a good seed)
        lin = np.linspace(1.0, 0.0, n)

            best_score = refined_score
            print(f"[SA] polished → score={best_score:.6f}")

    # Return a plain Python list (required by the harness)
    return [float(x) for x in best_seq]
