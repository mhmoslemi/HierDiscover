def _adam_optimize(seq0, target_sum, deadline, beta_schedule, iters_per_beta):
    n = len(seq0)
    f = seq0.copy()
    best = f.copy()
    best_score = _fast_score(f)
    # Adam parameters
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            if time.monotonic() >= deadline:
                return best, best_score
            # compute gradient
            conv = np.convolve(f, f)
            max_c = conv.max()
            shifted = conv - max_c
            exp_terms = np.exp(beta * shifted)
            sum_exp = exp_terms.sum()
            if sum_exp == 0:
                continue
            p = exp_terms / sum_exp
            grad = 2.0 * np.convolve(p, f)[:n]
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            # Determine learning rate
            # Choose base lr = 0.05 * target_sum (or compute via scaling)
            lr = 0.05 * target_sum / (np.linalg.norm(grad) + 1e-12)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            f_new = f - step
            # Clip and project onto simplex
            f_new = np.clip(f_new, 0.0, 1000.0)
            f_new = _project_simplex(f_new, target_sum)
            # Evaluate
            score = _fast_score(f_new)
            if score < best_score:
                best = f_new.copy()
                best_score = score
            # Accept if improvement or maybe criterion for monotonic improvement
            if score <= _fast_score(f):
                f = f_new
            else:
                # Optionally allow acceptance with probability? Use simulated annealing
                # Let's ignore for simplicity.
                pass
            if best_score < 1.503:
                return best, best_score
    return best, best_score

def _smoothmax_grad(seq: np.ndarray, beta: float):
    conv = np.convolve(seq, seq)
    max_c = conv.max()
    exp_terms = np.exp(beta * (conv - max_c))
    sum_exp = exp_terms.sum()
    if sum_exp == 0:
        p = np.zeros_like(conv)
    else:
        p = exp_terms / sum_exp
    grad = 2.0 * np.convolve(p, seq)[:seq.shape[0]]
    return grad

"""
Propose a low‑value step sequence.

Algorithm
---------
1. **Initialisation** – If a previous best sequence (``height_sequence_1``) exists,
   normalise it to the ℓ¹‑norm √(2·n) and use it as a seed.
   Otherwise start from a random non‑negative vector on the simplex.

2. **Heavy‑weight smooth‑max optimisation** – For each chosen length ``n`` a
   projected Adam‑style optimiser minimises a smooth approximation of the
   ℓ∞‑norm of the auto‑convolution:
       φβ(f) = (1/β)·log Σk exp(β·(f∗f)k)
   The schedule β = [2,4,8,16,32,64,128] is used, with many inner iterations
   (≈2000) so that the optimiser converges to a high‑quality local minimum.
   After each step the iterate is projected onto the simplex {x≥0, Σx = √(2·n)}.

3. **Random restarts & multiple lengths** – The optimiser is run on several
   lengths (200 … 3000) and several random restarts per length.  The best
   sequence over the whole search is kept.

4. **Final local polishing** – A simple stochastic hill‑climbing phase with
   Gaussian perturbations cleans up any remaining slack.

The evaluation function is scale‑invariant, therefore the objective that is
optimised is simply the maximum entry of the auto‑convolution after normalising
the ℓ¹‑norm.  The procedure respects all security checks performed by
``evaluate_sequence`` (clipping to [0,1000] and enforcing non‑negativity).

The routine stops when the time budget expires (default 1000 s) and returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` (no type checks).
    Returns np.inf for a sum smaller than 0.01.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    max_b = float(conv.max())
    n = seq.shape[0]
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _random_simplex_vector(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a random non‑negative vector of length ``n`` whose ℓ¹‑norm equals ``target_sum``.
    """
    v = rng.random(n) + 1e-3      # avoid exact zeros
    v = v / v.sum() * target_sum
    return v


def _smoothmax_grad(seq: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max φβ(f) = (1/β)·log Σ exp(β·(f∗f)_k) .
    Returns a vector of length len(seq) (first n entries of 2·conv(p,seq)).
    """
    conv = np.convolve(seq, seq)               # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                     # for numerical stability
    exp_terms = np.exp(beta * shifted)
    sum_exp = exp_terms.sum()
    if sum_exp == 0.0:
        p = np.zeros_like(conv)
    else:
        p = exp_terms / sum_exp
    # gradient = 2·conv(p, seq) – we need only the first n entries
    grad_full = np.convolve(p, seq)            # length 3n‑2
    grad = 2.0 * grad_full[:seq.shape[0]]
    return grad


def _adam_optimize(init_seq: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   rng: np.random.Generator,
                   beta_schedule=None,
                   inner_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimiser for the smooth‑max objective.
    Returns the best sequence encountered and its score.
    """
    if beta_schedule is None:
        beta_schedule = [2, 4, 8, 16, 32, 64, 128]

    n = init_seq.shape[0]
    f = init_seq.copy()
    best = f.copy()
    best_score = _fast_score(best)
    cur_score = best_score

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    t = 0

    for beta in beta_schedule:
        for _ in range(inner_iters):
            if time.monotonic() >= deadline:
                return best, best_score

            grad = _smoothmax_grad(f, beta)
            grad_norm = np.linalg.norm(grad) + 1e-12

            # Adam moment updates
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # Step size – a small base multiplied by the adaptive factor
            base_lr = 0.05 * target_sum
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = f - step
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _project_simplex(cand, target_sum)

            cand_score = _fast_score(cand)

            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

            # Accept only if we improve the current iterate (monotone descent)
            if cand_score <= cur_score:
                f = cand
                cur_score = cand_score
            else:
                # occasional random shake‑up to escape flat regions
                if rng.uniform() < 0.001:
                    noise = rng.normal(scale=0.02 * target_sum, size=n)
                    f = np.clip(f + noise, 0.0, 1000.0)
                    f = _project_simplex(f, target_sum)

            # early exit if we have already passed the target
            if best_score < 1.503:
                return best, best_score

    return best, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing respecting the ℓ¹‑norm.
    """
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best = start_seq.copy()
    best_score = _fast_score(best)

    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995               # slow exponential decay
        if step < 1e-7:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. Seed from a previously discovered construction, if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        if seq0.sum() > 0:
            seq0 = seq0 * (target0 / seq0.sum())
            sc0 = _fast_score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using global height_sequence_1 (n={n0}) score={sc0:.6f}")

    # --------------------------------------------------------------
    # 1. Explore a range of lengths with many random restarts
    # --------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1500, 2000, 2500, 3000]
    # shrink the list if we are already close to the deadline
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of restarts per length – more for smaller n to keep total time reasonable
        restarts = 3 if n <= 1000 else 2

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a random point on the simplex
            f0 = _random_simplex_vector(n, target_sum, rng)

            # run the Adam optimiser
            seq_opt, sc_opt = _adam_optimize(f0, target_sum, deadline, rng,
                                             beta_schedule=[2, 4, 8, 16, 32, 64, 128],
                                             inner_iters=2000)

            if sc_opt < best_score:
                best_seq = seq_opt.copy()
                best_score = sc_opt
                print(f"[opt] n={n} restart={r} score={sc_opt:.6f}")

            # early termination if we have reached the target bound
            if best_score < 1.503:
                print("[opt] target achieved – exiting early.")
                print(f"[final] best score={best_score:.6f} length={len(best_seq)}")
                return [float(x) for x in best_seq]

    # --------------------------------------------------------------
    # 2. Final stochastic refinement (Gaussian perturbations)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined_seq, refined_sc = _local_search(best_seq, deadline, step_initial=0.4)
        if refined_sc < best_score:
            best_seq = refined_seq
            best_score = refined_sc
            print(f"[local] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Fallback in the unlikely event that nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[final] best score={best_score:.6f} length={len(best_seq)}")
    return [float(x) for x in best_seq]
