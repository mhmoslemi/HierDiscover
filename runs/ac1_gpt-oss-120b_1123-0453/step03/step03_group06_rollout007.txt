"""
propose_candidate
~~~~~~~~~~~~~~~~~

A two‑phase optimiser for the “step‑function’’ problem.

Phase 1 – deterministic refinement.
    Starting from a (possibly random) non‑negative vector we repeatedly solve
    a *linear program*:

        maximise    sum(g)
        subject to  g ≥ 0
                    f * g ≤ M   (component‑wise)

    where ``f`` is the current sequence, ``M = max(f*f)`` and ``*`` denotes a
    linear (full) convolution.  The LP pushes mass into indices that are
    *inactive* for the current worst auto‑convolution entry while respecting
    the same maximal value ``M``.  After solving we scale ``g`` to the target
    ℓ¹‑norm (√(2·n)) and mix it with the current sequence using a small step
    size.  A line‑search over decreasing mixing factors guarantees that the
    new maximal auto‑convolution never exceeds the old one, so the evaluation
    value (which equals ``max(f*f)`` after normalisation) monotonically
    decreases.

Phase 2 – stochastic polishing.
    A fast hill‑climber adds zero‑mean Gaussian noise to the best candidate,
    always re‑normalising the ℓ¹‑norm.  The hill‑climb stops when the time
    budget expires.

The code explores several lengths (500 … 1000), interpolates the best
known shape when moving to a larger length, and keeps the overall best
sequence.  All heavy work (convolution, LP) is vectorised and limited to
the remaining time budget, so the function always returns a valid list
before the 1000 s deadline.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution (full linear convolution using a real FFT)
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``seq`` with itself (real FFT)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Exact analogue of ``evaluate_sequence`` using the FFT helper."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Linear‑programming helpers
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix ``T`` such that ``T @ g`` equals the full
    convolution ``f * g``.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)
    idx = np.arange(n)
    for i in range(n):
        T[i + idx, idx] = f[i]       # fills the i‑th diagonal band
    return T


def _lp_step(f: np.ndarray, M: float, target_sum: float,
             rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve   max sum(g)   s.t.   g≥0  and  f * g ≤ M .
    The result is returned *scaled* to have ℓ¹‑norm = ``target_sum``.
    If the LP fails or yields zero mass, ``None`` is returned.
    """
    n = f.shape[0]
    T = _build_conv_matrix(f)                     # shape (2n‑1, n)
    c = -np.ones(n)                               # maximise sum(g) ⇒ minimise –sum(g)
    bounds = [(0.0, None)] * n
    b_ub = np.full(T.shape[0], M)

    # “highs” is fast and robust for modestly sized LPs.
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                  method="highs", options={"presolve": True})
    if not res.success:
        return None

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1e-12:
        return None

    # Scale to the prescribed ℓ¹‑norm.
    scale = target_sum / sum_g
    g_scaled = g * scale
    return g_scaled


# ----------------------------------------------------------------------
# Deterministic LP‑based refinement (Phase 1)
# ----------------------------------------------------------------------


def _lp_refine(seq: np.ndarray, target_sum: float,
               deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve ``seq`` by solving the LP described above.
    The routine stops when the time budget expires, when no improvement
    can be found, or after a safety cap on iterations.
    Returns the best sequence encountered and its maximal auto‑convolution value.
    """
    cur = seq.copy()
    cur = np.clip(cur, 0.0, 1000.0)
    cur = cur * (target_sum / cur.sum())
    cur_max = _conv_full_fft(cur).max()

    best_seq = cur.copy()
    best_max = cur_max

    it = 0
    max_iters = 150                     # safety cap; usually far fewer are needed.

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # Solve the LP for the current shape.
        g_scaled = _lp_step(cur, cur_max, target_sum, rng)
        if g_scaled is None:
            break

        improved = False
        # Try a short line‑search over mixing factors.
        for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02):
            cand = (1.0 - t) * cur + t * g_scaled
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand * (target_sum / cand.sum())
            cand_max = _conv_full_fft(cand).max()
            if cand_max < cur_max - 1e-12:          # strict improvement
                cur = cand
                cur_max = cand_max
                improved = True
                if cur_max < best_max:
                    best_seq = cur.copy()
                    best_max = cur_max
                break

        if not improved:
            # Stuck – inject a tiny random perturbation to escape possible
            # flat regions (the “shake’’ step).  The perturbation is tiny so as
            # not to destroy the current quality.
            eps = 1e-3 * target_sum
            cand = cur + rng.normal(scale=eps, size=cur.shape)
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand * (target_sum / cand.sum())
            cur = cand
            cur_max = _conv_full_fft(cur).max()
            # continue the loop – the shake may enable a subsequent LP step
    return best_seq, best_max


# ----------------------------------------------------------------------
# Stochastic polishing (Phase 2)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that does not enforce a constant ℓ¹‑norm.
    The score is recomputed with ``_fast_score`` after each perturbation.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # exponential cooling of the step size
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Interpolation helper – enlarge a known shape
# ----------------------------------------------------------------------


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a 1‑D array to a new length.
    The result is clipped to the admissible range [0, 1000].
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return np.clip(new_seq, 0.0, 1000.0)

# ----------------------------------------------------------------------

    """
