def _adam_optimize(a0, target_sum, deadline, max_iters=2000, lr=0.1, eps_factor=0.5, eps_decay=0.995, min_eps=1e-4):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    best_a = a.copy()
    best_score = float('inf')
    while time.monotonic() < deadline and t < max_iters:
        t+=1
        # compute conv via FFT
        conv = _conv_full(a)  # returns size 2n-1
        M = conv.max()
        score = M  # after normalization sum = target
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # compute weights
        eps = max(min_eps, M * eps_factor)
        # stability shift
        maxc = conv.max()
        w = np.exp((conv - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient
        grad = _gradient_weights(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # update
        a = a - step
        # project onto simplex
        a = _project_simplex(a, target_sum)
        # decay epsilon factor
        eps_factor *= eps_decay
    return best_a, best_score
```

"""
Propose a low‑score step‑function sequence.

Algorithm
---------

1. **Warm start** – if a previous construction ``height_sequence_1`` is
   present it is projected onto the ℓ¹‑simplex of the required norm
   (∑a = √(2·n)) and used as the first candidate.

2. **Smoothed‑max Adam optimisation**  
   For a chosen length `n` we minimise the maximum entry of the
   auto‑convolution (`max (a*a)`) under the ℓ¹‑norm constraint.
   The non‑smooth max is approximated by a soft‑max with a temperature
   `ε`.  The gradient of the smoothed objective is the correlation of the
   current weight vector with the coefficients, which is computed efficiently
   by FFTs.  Adam (with bias‑correction) together with a projection onto the
   simplex yields fast descent.

3. **Multiple lengths & restarts** – the optimiser is run on several
   lengths (600 … 2000) and with a few random restarts (including the
   up‑sampled best vector from the previous length).  The best vector ever
   seen (according to ``evaluate_sequence``) is retained.

4. **Final stochastic polishing** – a cheap Gaussian hill‑climbing
   finishes the run.

All steps respect the security checks inside ``evaluate_sequence``.
The function stops a few hundred milliseconds before the global budget
to guarantee a return.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: project onto the ℓ¹‑simplex {x ≥ 0, Σx = target}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σ x = target}."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - target
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]

    if rho_arr.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution) as a 1‑D ndarray."""
    n = a.shape[0]
    # FFT size large enough for both convolution (2n‑1) and later correlation
    size = 1 << ((3 * n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max:  ∇ = 2 · corr(a, w)
# ----------------------------------------------------------------------
def _gradient_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇M_ε(a) = 2·corr(a, w) where
        corr_i = Σ_j a_j·w_{j+i}   (zero outside the range)
    Both arguments are real, `w` has length 2·n‑1.
    """
    n = a.shape[0]
    L = w.shape[0]
    size = 1 << ((n + L - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    Fw = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(Fa * np.conj(Fw), n=size)
    return 2.0 * corr[:n]


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2000,
    lr: float = 0.2,
    eps_factor: float = 0.3,
    eps_decay: float = 0.998,
    min_eps: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed maximum of the auto‑convolution.
    Returns the best vector observed and its (unsmoothed) max‑convolution value.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    best_a = a.copy()
    best_M = float("inf")
    t = 0

    while time.monotonic() < deadline and t < max_iters:
        t += 1
        # full convolution (linear) and its maximum
        conv = _full_convolution(a)
        M = float(conv.max())

        if M < best_M:
            best_M = M
            best_a = a.copy()

        # soft‑max weights over the convolution entries
        eps = max(min_eps, M * eps_factor)
        # subtract max for numerical stability
        shifted = conv - conv.max()
        w = np.exp(shifted / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient (correlation of a with the weight vector)
        g = _gradient_weights(a, w)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step

        # keep feasibility: non‑negative and correct ℓ¹‑norm
        a = _project_simplex(a, target_sum)

        # gradually tighten the temperature
        eps_factor *= eps_decay

    return best_a, best_M


# ----------------------------------------------------------------------
# Very cheap local stochastic refinement (Gaussian perturbations)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray, deadline: float, step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to the ℓ¹‑norm required by the evaluation function
        target = math.sqrt(2.0 * cand.shape[0])
        cand = _project_simplex(cand, target)

        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best, best_score = cand, sc
        step *= 0.9999
        if step < 1e-7:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Wrapper for the given evaluation routine
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the value of ``evaluate_sequence`` for a NumPy array."""
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The routine performs a few restarts of a smoothed‑max Adam optimiser
    on several lengths (up to a few thousand entries) and finishes with
    a short stochastic polishing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Load previous construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = seq.shape[0]
        target0 = math.sqrt(2.0 * n0)
        seq = _project_simplex(seq, target0)
        best_seq = seq.copy()
        best_score = _score(seq)
        print(f"[init] loaded height_sequence_1 (n={n0}) → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Optimisation on several lengths
    # ------------------------------------------------------------------
    # lengths are chosen to give a good trade‑off between flexibility
    # and computational cost.
    lengths = [600, 800, 1000, 1200, 1500, 1800, 2000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Build a list of starting points for this length
        # ------------------------------------------------------------------
        starts = []

        # keep the best vector seen so far (possibly up‑sample it)
        if best_seq is not None:
            if best_seq.shape[0] == n:
                starts.append(best_seq.copy())
            else:
                # linear interpolation to the new size
                x_old = np.linspace(0.0, 1.0, best_seq.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                up = np.interp(x_new, x_old, best_seq)
                up = _project_simplex(up, target_sum)
                starts.append(up)

        # a few random restarts
        for _ in range(3):
            rnd = rng.random(n) + 0.1
            rnd = _project_simplex(rnd, target_sum)
            starts.append(rnd)

        # ------------------------------------------------------------------
        # Run the Adam optimiser from each start
        # ------------------------------------------------------------------
        for start_id, a0 in enumerate(starts):
            if time.monotonic() >= deadline:
                break

            # allocate a reasonable slice of the remaining time to this run
            remaining = deadline - time.monotonic()
            sub_deadline = time.monotonic() + min(remaining * 0.5, 8.0)  # at most ~8 s per start

            a_opt, M_opt = _adam_optimize(
                a0,
                target_sum=target_sum,
                deadline=sub_deadline,
                max_iters=2500,
                lr=0.2,
                eps_factor=0.3,
                eps_decay=0.998,
                min_eps=1e-5,
            )
            sc = _score(a_opt)
            if sc < best_score:
                best_score, best_seq = sc, a_opt.copy()
                print(f"[adam] n={n} start={start_id} → score={sc:.6f}  maxconv={M_opt:.6f}")

            if best_score < 1.503:
                print("[early] target reached – exiting")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 2️⃣  Final stochastic polishing
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – simple uniform vector of moderate length
        n = 800
        best_seq = _project_simplex(rng.random(n) + 0.1, math.sqrt(2.0 * n))
        best_score = _score(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _local_search(best_seq, deadline, step_initial=0.3)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[local] polished → score={best_score:.6f}")

    # Return a plain Python list as required by the evaluation harness
    return [float(x) for x in best_seq]
