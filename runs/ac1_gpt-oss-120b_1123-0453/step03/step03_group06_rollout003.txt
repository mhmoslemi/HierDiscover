"""
Step‑function optimizer – LP‑based descent.

The search works on a fixed length `n`.  The evaluation of a sequence
`a` with ℓ¹‑norm fixed to √(2·n) reduces to minimizing the
maximum entry of the auto‑convolution `a*a`.  Starting from a random
(or warm‑start) vector we repeat:

1. Compute the current worst autocorrelation value `M`.
2. Solve a linear program

       maximize   Σ b_j
       subject to  (a * b)_k ≤ M   for all k
                   b_j ≥ 0

   The matrix of the constraints is the Toeplitz convolution matrix of
   `a`.  The LP returns a vector `b` that can carry *much* more mass
   than `a` while respecting the same upper bound `M`.

3. Rescale `b` to have the required ℓ¹‑norm (√(2·n)) and replace `a`
   with the new vector if the bound improves; otherwise blend the two
   vectors.

The loop is stopped by the time budget or as soon as the score drops
below the target (≈1.503).  A short Gaussian hill‑climb is performed
at the end to polish the result.

All heavy work (FFT convolution, LP building) is vectorised and uses
SciPy/NumPy only; the whole routine comfortably fits into the 1000 s
budget.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Fast convolution (full linear convolution) – used throughout
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``seq`` with itself using a real FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _score_from_maxconv(max_conv: float, n: int) -> float:
    """
    The evaluation function of the competition after normalising the
    ℓ¹‑norm to √(2·n) simplifies to ``max_conv`` itself.
    """
    return max_conv

# ----------------------------------------------------------------------

    best = seq.copy()
    # note: the true ``evaluate_sequence`` normalises the sum automatically,
    # but we keep the sum fixed for speed.
    best_score = _conv_full_fft(best).max()

    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑normalise to the same ℓ¹‑norm (required for a fair comparison)
        s = cand.sum()
        if s > 0:
            cand = cand * (best.sum() / s)
        cand_score = _conv_full_fft(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best


# ----------------------------------------------------------------------
# Main optimiser – LP descent on a single length
# ----------------------------------------------------------------------


def _optimize_length(n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     warm_start: np.ndarray | None = None,
                     target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Returns the best sequence found for a fixed length ``n`` together
    with its evaluation score.  The function respects the given deadline.
    """
    target_sum = math.sqrt(2.0 * n)

    # ----- initial vector -------------------------------------------------
    if warm_start is not None and warm_start.shape[0] == n:
        a = warm_start.astype(np.float64).copy()
    else:
        a = rng.random(n) + 0.05          # small random positive values
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0:
        a = a * (target_sum / s)

    best_a = a.copy()
    best_score = float("inf")
    iteration = 0

    # ----- pre‑computed index arrays for the Toeplitz matrix -------------
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + j[None, :]).ravel()          # shape n*n
    cols = np.tile(j, n)                               # shape n*n
    shape = (2 * n - 1, n)

    # ----- main LP‑based loop --------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # current worst autocorrelation
        conv_a = _conv_full_fft(a)
        M = conv_a.max()
        cur_score = _score_from_maxconv(M, n)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[LP] n={n:4d} iter={iteration:5d} M={M:.6f} score={cur_score:.6f}")

        # early exit if target reached
        if cur_score < target_score:
            print("[LP] target achieved – terminating early.")
            break

        # ----- build the constraint matrix A_ub (depends on current a) -----
        data = np.repeat(a, n)            # each a[i] repeats n times
        A_ub = csr_matrix((data, (rows, cols)), shape=shape)

        # ----- linear program – maximise Σ b_j under (a*b)_k ≤ M ---------
        c = -np.ones(n, dtype=np.float64)     # minimise -Σ b_j  → maximise Σ b_j
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=(0, None),
                      method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if not res.success:
            # LP failed – fall back to a tiny random perturbation and continue
            print("[LP] linprog failed, applying random jitter.")
            jitter = rng.normal(scale=0.01, size=n)
            a = np.clip(a + jitter, 0.0, 1000.0)
            a = a * (target_sum / a.sum())
            continue

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            # nothing useful – break out
            break

        # ----- scale b to the required ℓ¹‑norm ---------------------------
        b = b_opt * (target_sum / sum_b)

        # evaluate the candidate
        M_new = _conv_full_fft(b).max()

        # accept if it strictly improves the bound, otherwise blend
        if M_new < M:
            a = b
        else:
            # linear blend (t can be tuned); we keep it modest to stay stable
            t = 0.5
            a = (1.0 - t) * a + t * b
            a = np.clip(a, 0.0, 1000.0)
            a = a * (target_sum / a.sum())

    return best_a, best_score


# ----------------------------------------------------------------------
# Public entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The algorithm runs a short LP‑based descent on several candidate
    lengths and finishes with a tiny Gaussian polishing step.
    """
    rng = np.random.default_rng(seed)
    # leave a tiny safety margin for the caller
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.3)

    # --- keep the best overall candidate ---------------------------------
    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # lengths to explore – larger lengths give a larger search space
    lengths = [400, 500, 600, 800, 1000]

    # optional warm‑start from a previously discovered construction
    warm = globals().get("height_sequence_1", None)
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)

    # allocate most of the budget for the LP phase; the rest for polishing
    total_time = deadline - time.monotonic()
    lp_fraction = 0.85
    polish_fraction = 0.15
    lp_deadline = time.monotonic() + total_time * lp_fraction

    # ---------------------------------------------------------
    # Loop over the chosen lengths, respecting the global deadline
    # ---------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # give each length a fair share of the remaining LP budget
        remaining = lp_deadline - time.monotonic()
        if remaining <= 0:
            break
        per_len_deadline = time.monotonic() + remaining / (len(lengths) - lengths.index(n))

        # initialise warm start for this length
        if warm is not None and warm.shape[0] == n:
            warm_start = warm.copy()
        elif best_seq is not None:
            # interpolate the current best to the new length
            old_n = best_seq.shape[0]
            old_x = np.linspace(0.0, 1.0, old_n)
            new_x = np.linspace(0.0, 1.0, n)
            warm_start = np.interp(new_x, old_x, best_seq)
        else:
            warm_start = None

        cand_seq, cand_score = _optimize_length(n,
                                                per_len_deadline,
                                                rng,
                                                warm_start=warm_start)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[global] new overall best: score={best_score:.6f} (n={n})")
            if best_score < 1.503:
                # we have reached the target – stop all work
                deadline = time.monotonic() + 0.2   # tiny remainder for polishing
                break

    # ---------------------------------------------------------
    # Final polishing – short Gaussian hill‑climb
    # ---------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polish_deadline = deadline
        best_seq = _gaussian_polish(best_seq, polish_deadline, step_initial=0.3)
        # recompute the score after polishing (mainly for logging)
        best_score = _conv_full_fft(best_seq).max()
        print(f"[final] polished score={best_score:.6f}")

    # safety fallback – should never happen
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = best_seq * (math.sqrt(2.0 * n) / best_seq.sum())

    # return a plain Python list as required by the harness
    return [float(v) for v in best_seq]
