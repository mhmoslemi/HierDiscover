"""
Gradient‑flattening search for a low‑value step sequence.

The algorithm works in three stages:

1️⃣  Warm‑start – if a previous construction ``height_sequence_1`` exists,
    it is interpolated to the current length and scaled to the required
    ℓ¹‑norm (√(2·n)).  Otherwise a random non‑negative vector is used.

2️⃣  Repeated gradient‑descent passes – the loss
    ``L = Σ (f*f – μ)²`` (μ is the average of the full auto‑convolution)
    is minimised by a simple projected gradient step.  The gradient is
    obtained analytically by a single FFT‑based correlation, thus each
    iteration costs O(n log n).  After each step the vector is clipped to
    [0,1000] and re‑normalised to keep the ℓ¹‑norm fixed.

    Many short passes are performed with fresh random restarts; the best
    candidate found for the current length is kept.

3️⃣  Final stochastic polishing – a short Gaussian hill‑climb (the same
    routine used in the reference solution) cleans up the best overall
    sequence.

The whole routine respects the supplied time budget (default 1000 s) and
returns a plain Python list of floats that can be handed to the provided
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, full linear convolution)
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``seq`` with itself using a real FFT.
    Returns an array of length ``2·n‑1``.
    """
    n = seq.shape[0]
    # size large enough for linear convolution (next power of two)
    size = 1 << ((2 * n - 1).bit_length())
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast analogue of ``evaluate_sequence`` – the same mathematical
    expression, but using the FFT‑based convolution.
    The caller must guarantee that ``seq`` already lies in [0,1000].
    """
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Interpolation – enlarge a known shape to a longer length
# ----------------------------------------------------------------------


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of ``seq`` to length ``new_n``.
    Output is clipped to [0,1000].
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return np.clip(new_seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Simple Gaussian hill‑climb (final polishing)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing that adds zero‑mean Gaussian noise.
    The ℓ¹‑norm is *not* forced to stay constant – the true
    ``evaluate_sequence`` will re‑normalise automatically.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Gradient descent on the L2‑error of the auto‑convolution
# ----------------------------------------------------------------------


def _gradient_descent(init_seq: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise  Σ (f*f – μ)²  where  μ = S²/(2n‑1)  by a projected gradient step.
    The method uses a single FFT‑based correlation to obtain the exact
    gradient of the loss.  After each step the vector is clipped to
    [0,1000] and re‑scaled to keep the ℓ¹‑norm equal to ``target_sum``.
    """
    n = init_seq.shape[0]
    size = 1 << ((3 * n - 2).bit_length())          # for linear correlation
    mu = target_sum * target_sum / (2 * n - 1)       # desired average of conv

    # step schedule – start relatively aggressive, then decay with sqrt(iter)
    step0 = target_sum * 0.03 / n                     # heuristic initial step

    cur = init_seq.copy()
    cur_score = _fast_score(cur)
    best_seq = cur.copy()
    best_score = cur_score

    it = 0
    last_improve = 0
    while time.monotonic() < deadline:
        # FFT of current vector
        f_fft = np.fft.rfft(cur, n=size)

        # full auto‑convolution (linear)
        conv = np.fft.irfft(f_fft * f_fft, n=size)[: 2 * n - 1]

        # error w.r.t. uniform target
        err = conv - mu

        # correlation = f ⋆ err  (gradient of the L2 loss is 4·corr)
        err_fft = np.fft.rfft(err, n=size)
        corr = np.fft.irfft(f_fft * np.conj(err_fft), n=size)[:n]

        grad = 4.0 * corr                     # ∇L

        # diminishing step size
        step = step0 / math.sqrt(it + 1.0)

        # candidate update
        cand = cur - step * grad
        cand = np.clip(cand, 0.0, 1000.0)

        s = cand.sum()
        if s > 0:
            cand = cand * (target_sum / s)

        cand_score = _fast_score(cand)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                last_improve = it
        else:
            # occasional random jitter to escape flat plateaus
            if rng.random() < 0.02:
                jitter = rng.normal(scale=step * 0.5, size=n)
                cand = cur + jitter
                cand = np.clip(cand, 0.0, 1000.0)
                s = cand.sum()
                if s > 0:
                    cand = cand * (target_sum / s)
                cand_score = _fast_score(cand)
                if cand_score < cur_score:
                    cur = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_seq = cand.copy()
                        best_score = cand_score
                        last_improve = it

        # simple progress output (optional, can be silenced)
        if it % 500 == 0:
            print(f"[GD] n={n:4d} iter={it:6d} score={cur_score:.6f}")

        it += 1

        # safety: if we have seen no improvement for a long while, inject noise
        if it - last_improve > 2000:
            cur = cur + rng.normal(scale=step * 0.1, size=n)
            cur = np.clip(cur, 0.0, 1000.0)
            cur = cur * (target_sum / cur.sum())
            cur_score = _fast_score(cur)
            last_improve = it

    return best_seq, best_score

        target_sum = math.sqrt(2.0 * n)

        # --------- initialise candidate ----------
        if known_seq is not None and known_seq.shape[0] == n:
            init_seq = known_seq.copy()
        elif best_by_len:
            # take the longest already‑computed sequence ≤ n and interpolate
            src_n = max(k for k in best_by_len.keys() if k <= n)
            init_seq = _interpolate(best_by_len[src_n], n)
        else:
            # purely random start – add a tiny offset to avoid exact zeros
            init_seq = rng.random(n) + 1e-3

        # enforce constraints and target ℓ¹‑norm
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        s = init_seq.sum()
        if s > 0:
            init_seq = init_seq * (target_sum / s)

        # --------- perform a number of short gradient runs ----------
        # each run gets a slice of the per‑length budget
        len_deadline = time.monotonic() + time_per_len
        # we try a few random restarts – the exact number adapts to the time left
        while time.monotonic() < len_deadline:
            # occasional jitter around current best for the same length
            if best_by_len.get(n) is not None and rng.random() < 0.5:
                start = best_by_len[n] + rng.normal(scale=0.02 * target_sum, size=n)
                start = np.clip(start, 0.0, 1000.0)
                start = start * (target_sum / start.sum())
            else:
                start = init_seq.copy()

            run_deadline = time.monotonic() + min(0.6, len_deadline - time.monotonic())
            cand_seq, cand_score = _gradient_descent(start,
                                                     target_sum,
                                                     run_deadline,
                                                     rng)

            best_by_len[n] = cand_seq.copy()

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()
                print(f"[global] new overall best: score={best_score:.6f} (n={n})")

            # early stop if we have already beaten the target
            if best_score < 1.503:
                print("[global] target reached – exiting early.")
                break

        # if the global target was reached, stop exploring further lengths
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final polishing with a simple Gaussian hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_score = _local_search(best_seq,
                                                    deadline,
                                                    step_initial=0.5)
        if polished_score < best_score:
            best_seq = polished_seq
            best_score = polished_score
            print(f"[final] improved to {best_score:.6f}")

    # safety fallback – should never happen
    if best_seq is None:
        n = 800
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = best_seq * (math.sqrt(2.0 * n) / best_seq.sum())

    # Return as a plain Python list (float conversion for safety)
    return [float(v) for v in best_seq]
