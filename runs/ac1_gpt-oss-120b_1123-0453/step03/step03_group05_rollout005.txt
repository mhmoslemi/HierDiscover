    # for each beta

        # number of inner iterations limited by remaining time
        inner_max = 300
        for i in range(inner_max):
            if time.monotonic() >= deadline:
                break
            # compute conv and gradient
            conv = np.convolve(a, a)
            max_c = conv.max()
            shift = conv - max_c
            exp_term = np.exp(beta * shift)
            sum_exp = exp_term.sum()
            if sum_exp <= 0:
                p = np.full_like(conv, 1/len(conv))
            else:

"""
Propose a low‑value step sequence.

Algorithm
---------
1. Initialise from the already‑found global sequence `height_sequence_1`
   (if present) – it already has a good score.

2. Run a projected Adam optimiser on the smooth‑max approximation of the
   worst convolution entry:
        φβ(a) = (1/β)·log Σk exp(β·(a∗a)k) .
   The gradient of φβ is 2·conv(p, a) where p_i ∝ exp(β·(a∗a)_i).
   β is increased geometrically (1 → 128) so the optimisation focuses on
   the true max as it proceeds.  After every Adam step the vector is
   projected onto the probability simplex {x≥0, Σx = √(2·n)}.

3. A fast stochastic hill‑climbing (Gaussian perturbations) refines the
   best vector found by the optimiser.

4. The whole procedure is time‑boxed (default 1000 s).  The best sequence
   discovered before the deadline is returned as a plain `list[float]`.

All operations respect the security checks performed by
`evaluate_sequence` (non‑negative, clipped to [0,1000], ℓ¹‑norm > 0).
"""

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``x`` and ``y`` using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    conv = np.fft.irfft(Z, N)[:n]
    return conv.astype(np.float64)


def _fast_score_from_max(n: int, total: float, max_conv: float) -> float:
    """
    Exact score used by ``evaluate_sequence`` when the ℓ¹‑norm is ``total``
    and the maximal auto‑convolution entry is ``max_conv``.
    """
    return float(2.0 * n * max_conv / (total * total))


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max objective
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   beta_schedule=None,
                   max_inner=200) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser.
    Returns the best vector seen and its exact score.
    """
    if beta_schedule is None:
        beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128]

    n = start_vec.shape[0]
    a = start_vec.copy()

    # keep the best seen
    conv = _conv_full(a, a)
    best_max = conv.max()
    best_score = _fast_score_from_max(n, target_sum, best_max)
    best_vec = a.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    # learning rate (scaled with the target ℓ¹‑norm)
    lr = 0.05 * target_sum

    for beta in beta_schedule:
        for _ in range(max_inner):
            if time.monotonic() >= deadline:
                return best_vec, best_score

            # ----- gradient of the smooth‑max ----------
            conv = _conv_full(a, a)                    # length 2n‑1
            max_c = conv.max()
            shifted = conv - max_c                     # for numerical stability
            exp_shift = np.exp(beta * shifted)
            sum_exp = exp_shift.sum()
            if sum_exp == 0.0:
                p = np.full_like(conv, 1.0 / conv.size)
            else:
                p = exp_shift / sum_exp                # probability vector, Σp = 1

            # grad = 2·conv(p, a) (first n components)
            grad_full = _conv_full(p, a)               # length 3n‑2
            grad = 2.0 * grad_full[:n]

            # ----- Adam step ---------------------------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            # projection onto the simplex (non‑negative, Σ = target_sum)
            a = np.clip(a, 0.0, 1000.0)
            a = _project_simplex(a, target_sum)

            # ----- evaluate exact score -----------------
            conv = _conv_full(a, a)
            cur_max = conv.max()
            cur_score = _fast_score_from_max(n, target_sum, cur_max)

            if cur_score < best_score:
                best_score = cur_score
                best_vec = a.copy()
                print(f"[Adam] n={n} beta={beta} score={cur_score:.6f}")

            if best_score < 1.503:          # early exit if target already met
                return best_vec, best_score

    return best_vec, best_score


# ----------------------------------------------------------------------
# Simple stochastic local refinement
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbation hill‑climbing, keeping the ℓ¹‑norm fixed.
    """
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best = start_seq.copy()
    # exact score with the known sum
    cur_max = np.convolve(best, best).max()
    best_score = _fast_score_from_max(n, target_sum, cur_max)

    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)
        cand_max = np.convolve(cand, cand).max()
        cand_score = _fast_score_from_max(n, target_sum, cand_max)

        if cand_score < best_score:
            best = cand
            best_score = cand_score

        step *= 0.9999
        if step < 1e-7:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the global time budget and returns a plain list of
    floats (the heights of the step function).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. Seed from the previous best if it exists
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seq0 = np.clip(seq0, 0.0, 1000.0)
        n0 = seq0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        # project onto the correct simplex (preserves ℓ¹‑norm)
        seq0 = _project_simplex(seq0, target0)
        cur_max = np.convolve(seq0, seq0).max()
        sc0 = _fast_score_from_max(n0, target0, cur_max)
        best_seq = seq0.copy()
        best_score = sc0
        print(f"[init] using global height_sequence_1 (n={n0}) score={sc0:.6f}")

    # --------------------------------------------------------------
    # 1. Main optimisation loop (Adam + local refinement)
    # --------------------------------------------------------------
    # a set of lengths to explore – larger sequences tend to allow a lower bound
    lengths = [500, 600, 800, 1000, 1200]

    while time.monotonic() < deadline:
        # pick a length (randomised to avoid bias)
        n = int(rng.choice(lengths))
        target_sum = math.sqrt(2.0 * n)

        # initialise a starting vector
        if best_seq is not None and best_seq.shape[0] == n:
            start_vec = best_seq.copy()
        else:
            # Dirichlet random vector with the prescribed ℓ¹‑norm
            start_vec = rng.dirichlet(np.ones(n)) * target_sum

        # ----- Adam optimisation on the smooth‑max objective -----
        opt_vec, opt_score = _adam_optimize(start_vec, target_sum, deadline)

        # ----- final stochastic refinement -----
        if time.monotonic() < deadline:
            opt_vec, opt_score = _local_search(opt_vec, deadline, step_initial=0.4)

        if opt_score < best_score:
            best_score = opt_score
            best_seq = opt_vec.copy()
            print(f"[global] new best score={best_score:.6f} length={n}")

        # stop early if we have already breached the target
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # 2. Fallback in the extremely unlikely case that nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score_from_max(n, target,
                                          np.convolve(best_seq, best_seq).max())
        print("[fallback] using uniform vector")

    print(f"[final] best score={best_score:.6f} length={len(best_seq)}")
    # convert to plain Python floats
    return [float(x) for x in best_seq]
