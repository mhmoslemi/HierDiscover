def _targeted_hill_climb(a, deadline, max_iters=200000, step_init=0.5):
    n = a.shape[0]
    conv = np.convolve(a, a)
    sum_a = a.sum()
    # best so far
    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv) / (sum_a**2)

    step = step_init
    rng = np.random.default_rng()
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # find index of max conv
        k_max = int(np.argmax(conv))
        # compute gradient contributions
        # vector of gradient values g_i = a[k_max - i] if valid else 0
        # plus double for diagonal.
        g = np.zeros_like(a)
        # compute overlapping indices:
        start_i = max(0, k_max - (n - 1))
        end_i = min(n - 1, k_max)
        # for i in [start_i, end_i], counterpart j = k_max - i
        # set g[i] = a[j] (where j = k_max - i)
        # if i == j, set g[i] = 2 * a[i] (diagonal)
        idx_i = np.arange(start_i, end_i+1)
        idx_j = k_max - idx_i
        # mask for diagonal
        diag_mask = (idx_i == idx_j)
        g[idx_i[~diag_mask]] = a[idx_j[~diag_mask]]
        g[idx_i[diag_mask]] = 2 * a[idx_i[diag_mask]]

        # source: index with large a_i * g_i
        # consider only where a_i > 0
        weighted = a * g
        # to avoid self loops or zero weights, we find source among indices with weight > 0
        if np.all(weighted <= 0):
            # fallback random move
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
        else:
            i = int(np.argmax(weighted))
            # dest: index with minimal g, pref non-negative a? we can pick a random i where g is minimal
            # Choose among indices where g is minimal; if multiple, random selection.
            min_g = np.min(g)
            candidates = np.where(g == min_g)[0]
            # ensure dest != i
            if candidates.size == 1 and candidates[0] == i:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
            else:
                # choose random dest not equal to i, prefer one not involved in k_max
                dest_candidates = candidates[candidates != i]
                if dest_candidates.size == 0:
                    # fallback random dest
                    j = rng.integers(0, n-1)
                    if j >= i:
                        j += 1
                else:
                    j = int(rng.choice(dest_candidates))

        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue
        # transfer
        _transfer_mass(a, conv, i, j, delta)
        cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # maybe reset step size or continue
            # optionally enlarge step a bit
            # step = max(step, step_init)
        else:
            # revert
            _transfer_mass(a, conv, j, i, delta)

        # Adjust step
        step *= 0.9995
        if step < 1e-5:
            step = step_init
        it += 1
    return best_a, best_score

def solve_lp_for_f(f, M):
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    # Build A as list of constraints
    constraints = []
    for k in range(2*n-1):
        coeff = np.zeros(n)
        # for j in range(n):
        #   i = k - j
        #   if 0 <= i < n:
        #       coeff[j] = f[i]

        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        # i = k - j

        coeff[j_start:j_end+1] = f[k - np.arange(j_start, j_end+1)]
        constraints.append(coeff @ g <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

from scipy.optimize import linprog

def solve_lp_max_sum(f, M):
    n = len(f)
    # Build A_ub matrix
    rows = []
    for k in range(2*n-1):
        coeff = np.zeros(n)
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        # For these j, i = k - j
        idx = np.arange(j_start, j_end+1)
        coeff[j_start:j_end+1] = f[k - idx]
        rows.append(coeff)
    A_ub = np.vstack(rows)
    b_ub = np.full(2*n-1, M)
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        return None
```

"""
Propose a low‑score step sequence.

Strategy
--------
1. **Initial candidates** – for each chosen length we build several
   diverse starts (uniform, exponential, sparse spikes, sinusoidal,
   and the historic `height_sequence_1` if it exists).  All candidates are
   scaled to the canonical ℓ¹‑norm   Σa = √(2·n); the objective is
   invariant under scaling, so this normalisation lets us treat the
   score simply as `max(convolution)`.

2. **LP‑based alternating refinement** – given a vector `f`,
   we solve a linear programme

        maximise    Σ g_j
        subject to  (f * g)[k] ≤ M   for every lag k
                    g_j ≥ 0 ,

   where `M = max(f*f)`.  The programme pushes mass into indices where
   `f` is small, thereby flattening the autocorrelation.  The solution
   `g₀` is rescaled to the canonical sum and mixed with `f`:
   `f ← (1‑t)·f + t·g`.  A few such iterations (t≈0.3) already give a
   strong reduction of the peak.

3. **Targeted hill‑climbing** – with the current vector we repeatedly
   move a small amount of mass from an index that contributes most to
   the *current* maximal convolution entry toward an index that contributes
   least.  The convolution is updated incrementally (O(n) per move) using
   an exact algebraic formula, so millions of moves are cheap.

4. **Final Gaussian polishing** – a light stochastic local‑search
   (Gaussian perturbations) is applied to the best vector found.

All steps respect the non‑negativity and clipping rules of
`evaluate_sequence`.  The routine stops as soon as the target bound
`1.503` is reached or when the time budget expires.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official score – wrappers the provided `evaluate_sequence`."""
    # `evaluate_sequence` expects a plain Python list
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index i to j (i != j, delta>0) and update the
    self‑convolution `conv = a * a` in‑place.
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part: 2·delta·a
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part: the convolution of the delta‑vector with itself
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta

    # update the underlying vector
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
# Linear‑programming refinement (alternating minimisation)
# ----------------------------------------------------------------------


def _solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   maximise Σ g_j   subject to   (f * g)[k] ≤ M   (∀k) ,   g ≥ 0.
    Returns the optimal g (as a NumPy array) or None on failure.
    """
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        coeff = np.zeros(n)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        coeff[j_start:j_end + 1] = f[k - idx]    # f_i where i = k - j
        rows.append(coeff)

    A_ub = np.vstack(rows)                       # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n)                              # maximise → minimise -Σg
    bounds = [(0, None)] * n

    # a short time limit keeps the total budget under control
    res = linprog(c,
                  A_ub=A_ub, b_ub=b_ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None


def _lp_refine(start: np.ndarray,
               deadline: float,
               max_iters: int = 8,
               t: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Alternating LP refinement.
    Returns the best vector encountered and its (cheap) score.
    """
    n = start.shape[0]
    target_sum = math.sqrt(2.0 * n)
    a = start.copy()
    # Ensure the initial sum is exactly the target (tiny floating drift)
    a = _project_onto_simplex(a, target_sum)

    best_a = a.copy()
    cur_conv = np.convolve(a, a)
    best_score = 2.0 * n * np.max(cur_conv) / (target_sum ** 2)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # current peak
        M = np.max(cur_conv)

        # solve LP
        g0 = _solve_lp_max_sum(a, M)
        if g0 is None:
            break
        sum_g0 = np.sum(g0)
        if sum_g0 <= 0.0:
            break

        # rescale to the canonical ℓ¹‑norm and mix
        scaling = target_sum / sum_g0
        g = g0 * scaling
        a = (1.0 - t) * a + t * g
        a = _project_onto_simplex(a, target_sum)

        # new convolution & score
        cur_conv = np.convolve(a, a)
        cur_score = 2.0 * n * np.max(cur_conv) / (target_sum ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # If we did not improve, shrink the mixing factor a bit
            t *= 0.9
            if t < 0.05:
                break   # stop early – further mixing unlikely to help

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill climbing (mass transfer)
# ----------------------------------------------------------------------


def _targeted_hill_climb(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that always moves mass away from indices that contribute
    most to the *current* maximal convolution entry.
    Returns the best vector encountered and its cheap score.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)               # exact convolution
    target_sum = a.sum()                   # remains constant during transfers

    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv) / (target_sum ** 2)

    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        # ----- locate the current peak and its gradient -----
        k_max = int(np.argmax(conv))

        # gradient g_i = ∂max(conv)/∂a_i = a_{k_max‑i} (or 2·a_i on the diagonal)
        g = np.zeros_like(a)
        start_i = max(0, k_max - (n - 1))
        end_i = min(n - 1, k_max)
        idx_i = np.arange(start_i, end_i + 1)
        idx_j = k_max - idx_i
        diag = idx_i == idx_j
        # off‑diagonal contributions
        g[idx_i[~diag]] = a[idx_j[~diag]]
        # diagonal contributions (i == j)
        g[idx_i[diag]] = 2.0 * a[idx_i[diag]]

        # ----- choose source and destination indices -----
        weighted = a * g
        if np.all(weighted <= 0):
            # fallback to a random move
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
        else:
            i = int(np.argmax(weighted))
            # destination: index with the smallest gradient (prefer zero)
            min_g = np.min(g)
            candidates = np.where(g == min_g)[0]
            # avoid picking the same index
            cand = candidates[candidates != i]
            if cand.size == 0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
            else:
                j = int(rng.choice(cand))

        # ----- transfer a small fraction -----
        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue
        _transfer_mass(a, conv, i, j, delta)

        cur_score = 2.0 * n * np.max(conv) / (target_sum ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert the move
            _transfer_mass(a, conv, j, i, delta)

        # schedule step
        step *= 0.9995
        if step < 1e-5:
            step = step_init
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Final gentle Gaussian polishing
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Stochastic Gaussian perturbation (identical to the reference implementation).
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats; the caller will feed it to
    `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    # reserve a small margin for the caller
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Choose a set of lengths (larger n gives more freedom)
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # 2️⃣  Assemble a handful of diverse initial vectors
        # ------------------------------------------------------------------
        inits: list[np.ndarray] = []

        # historic construction, if available
        if "height_sequence_1" in globals():
            try:
                hs = np.array(height_sequence_1, dtype=float)
                if hs.shape[0] == n:
                    inits.append(hs.copy())
            except Exception:
                pass

        # uniform
        inits.append(np.full(n, target_sum / n, dtype=float))

        # exponential random
        inits.append(rng.exponential(scale=1.0, size=n))

        # sparse spikes
        sparse = np.zeros(n, dtype=float)
        m = max(1, n // 20)
        idx = rng.choice(n, size=m, replace=False)
        sparse[idx] = rng.exponential(scale=5.0, size=m)
        inits.append(sparse)

        # sinusoidal envelope
        x = np.linspace(0, np.pi, n)
        sinusoid = (np.sin(x) ** 2) * rng.uniform(0.5, 1.5) + 0.01
        inits.append(sinusoid)

        # a few extra random draws
        for _ in range(2):
            vec = rng.exponential(scale=1.0, size=n) if rng.random() < 0.5 else rng.random(n) + 0.1
            inits.append(vec)

        for start in inits:
            if time.monotonic() >= deadline:
                break

            # normalise to the canonical ℓ¹‑norm (does not affect the ratio)
            s = start.sum()
            if s <= 0.0:
                start[:] = target_sum / n
            else:
                start[:] = start * (target_sum / s)
            start = np.clip(start, 0.0, 1000.0)

            # --------------------------------------------------------------
            # 3️⃣  LP‑based alternating refinement
            # --------------------------------------------------------------
            refined_lp, _ = _lp_refine(start, deadline,
                                       max_iters=8, t=0.3)
            lp_score = _score(refined_lp)
            if lp_score < best_score:
                best_score = lp_score
                best_seq = refined_lp.copy()
                print(f"[lp]    n={n}  score={best_score:.6f}")

            # --------------------------------------------------------------
            # 4️⃣  Targeted mass‑transfer hill climbing
            # --------------------------------------------------------------
            improved, _ = _targeted_hill_climb(refined_lp, deadline,
                                               max_iters=150_000,
                                               step_init=0.5)
            hc_score = _score(improved)
            if hc_score < best_score:
                best_score = hc_score
                best_seq = improved.copy()
                print(f"[hc]    n={n}  score={best_score:.6f}")

            # early exit if we have reached the target bound
            if best_score < 1.503:
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 5️⃣  Final Gaussian polishing (fallback if nothing better found)
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – modest uniform vector
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    if time.monotonic() < deadline:
        refined, refined_score = _local_search(best_seq, deadline,
                                               step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] improved to {best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
