def guided_hill_climb(a, deadline, temp_init=0.1, temp_decay=0.9995, step_init=0.5, max_iters=1e7):
    # a is numpy array with sum S = sqrt(2*n)
    n = a.shape[0]
    sum_a = a.sum()
    conv = np.convolve(a, a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max() / (sum_a * sum_a)
    cur_a = a.copy()
    cur_conv = conv
    cur_score = best_score
    rng = np.random.default_rng()
    temp = temp_init
    step = step_init
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # compute max and ks
        max_val = cur_conv.max()
        ks = np.flatnonzero(cur_conv == max_val)
        # compute contribution per index
        contr = np.zeros(n)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # i range
            i_range = np.arange(lo, hi+1)
            j_range = k - i_range
            contr[i_range] += cur_a[j_range]
        # source index: highest contr * cur_a maybe weighted, but we want high contribution.
        # choose source idx with large contr and non-zero a.
        # compute score_i = contr * cur_a (since derivative is 2*contr and contribution proportionate to a_i).
        score_i = contr * cur_a
        # mask zeros
        if np.all(score_i == 0):
            # if all zero (rare), fallback to random i
            src = rng.integers(0, n)
        else:
            src = int(np.argmax(score_i))
        # target idx: low contr (could be zero)
        dst = int(np.argmin(contr))
        if src == dst:
            # try random pair
            src = rng.integers(0, n)
            dst = rng.integers(0, n-1)
            if dst >= src:
                dst += 1
        # move delta: choose fraction of src mass
        delta = cur_a[src] * step * rng.random()
        if delta <= 0:
            continue
        # perform move
        _transfer_mass(cur_a, cur_conv, src, dst, delta)
        new_max = cur_conv.max()
        new_score = 2.0*n*new_max/(sum_a*sum_a)
        # acceptance
        delta_score = new_score - cur_score # cur_score is previous step's max? Actually we compare to cur_score, not best_score.
        # but we want to adapt. Use Metropolis with temperature.
        accepted = False
        if new_score < cur_score:
            accepted = True
        else:
            # if uphill, accept with probability
            if rng.random() < math.exp(-(new_score - cur_score)/temp):
                accepted = True
        if accepted:
            cur_score = new_score
            # Update best
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            # revert move
            _transfer_mass(cur_a, cur_conv, dst, src, delta)
        # decay temperature and step
        temp *= temp_decay
        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1
    return best_a, best_score

"""
Guided Simulated‑Annealing hill‑climber for the convolution‐max objective.

The routine works in three phases:

1️⃣  For several lengths (up to a few thousand) a random start is drawn and
    projected onto the ℓ¹‑simplex Σa = √(2 n).  The start may also be the
    previously obtained construction (global variable ``height_sequence_1``).

2️⃣  A *guided* mass‑transfer SA loop is run.  At each iteration the
    convolution is kept up‑to‑date incrementally (O(n) per move).  The
    indices that realise the current maximum of ``a*a`` are identified;
    a contribution vector tells how much each coefficient participates in
    those tight constraints.  Mass is moved from a high‑contributing index
    to a low‑contributing one.  The move is accepted if it improves the
    cheap score or with a Metropolis probability that slowly cools.

    The cheap score is
        2·n·max(conv) / (sum(a)²)      (scale‑invariant, equals the official
    evaluation when Σa = √(2 n)).
    Whenever a new best cheap score is found the official ``evaluate_sequence``
    is called to obtain the true score.

3️⃣  After the time budget a final gentle Gaussian polishing is performed
    (only accepting improvements).

The whole search respects the non‑negativity and clipping rules of
``evaluate_sequence`` and returns the best list found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official score via the provided ``evaluate_sequence``."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer ``delta`` from index i to j (i != j, delta > 0) and update
    the convolution ``conv = np.convolve(a, a)`` in‑place.
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part: 2 * conv(delta, a)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part: conv(delta, delta)
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta          # (-δ)²
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta    # cross terms
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta          # (+δ)²

    # finally move the mass in the vector
    a[i] -= delta
    a[j] += delta


def _contribution_vector(a: np.ndarray, k_idxs: np.ndarray) -> np.ndarray:
    """
    For a set of convolution indices ``k_idxs`` (where the current maximum
    is attained) compute, for every position i, the sum of a[j] with
    j = k – i (when the index is inside the array).  This is exactly the
    gradient of the max‑convolution w.r.t. a[i] (up to the constant 2).
    """
    n = a.shape[0]
    contrib = np.zeros(n, dtype=float)
    for k in k_idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1, dtype=int)
        j_range = k - i_range
        contrib[i_range] += a[j_range]
    return contrib


def _guided_hill_climb(start: np.ndarray,
                       deadline: float,
                       temp_init: float = 0.1,
                       temp_decay: float = 0.9995,
                       step_init: float = 0.5,
                       max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing that moves a small amount of mass
    from a high‑contributing index to a low‑contributing one.
    Returns the best sequence seen and its *official* score.
    """
    a = start.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)            # exact convolution (length 2n‑1)
    sum_a = a.sum()                     # constant throughout the run
    # cheap score (scale‑invariant)
    cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

    best_a = a.copy()
    best_score = _score(best_a)         # official score

    rng = np.random.default_rng()
    temp = temp_init
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        # 1. locate tight constraints (indices where max is attained)
        max_val = conv.max()
        ks = np.flatnonzero(conv == max_val)

        # 2. compute contribution of each coefficient to those constraints
        contrib = _contribution_vector(a, ks)

        # 3. pick source (large contribution & non‑zero mass) and target
        #    (small contribution).  Tie‑break with randomness to avoid cycles.
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src_scores = contrib[src_candidates] * a[src_candidates]
        src = int(src_candidates[np.argmax(src_scores)])

        dst = int(np.argmin(contrib))
        if src == dst:
            dst = rng.integers(0, n - 1)
            if dst >= src:
                dst += 1

        # 4. choose an amount of mass to transfer
        delta = a[src] * step * rng.random()
        if delta <= 0.0:
            continue

        # 5. apply the move (incremental conv update)
        _transfer_mass(a, conv, src, dst, delta)

        new_score = 2.0 * n * conv.max() / (sum_a * sum_a)   # cheap

        # 6. Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill, accept with probability exp(-(Δ)/temp)
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            # keep track of the best *official* score
            if new_score < best_score:
                # verify with the official evaluator (tiny overhead, only on improvements)
                official = _score(a)
                if official < best_score:
                    best_score = official
                    best_a = a.copy()
                    print(f"[improve] n={n} score={best_score:.6f}  it={it}")
        else:
            # revert the move
            _transfer_mass(a, conv, dst, src, delta)

        # 7. annealing & step‑size schedule
        temp *= temp_decay
        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1

        # early exit if we have already crossed the target bound
        if best_score < 1.503:
            break

    return best_a, best_score


def _local_search(start: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Gentle Gaussian perturbation (accept only improvements).
    Used as a final polishing step.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑normalise to the same ℓ¹‑norm (scale invariance makes it optional,
        # but keeping the sum constant avoids numerical drift)
        # we simply project onto the simplex with the original target sum
        target = best.sum()
        cand = _project_onto_simplex(cand, target)

        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the supplied time budget and returns a plain
    Python list of floats (the best vector discovered before timeout).
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin for the caller
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Gather possible initial vectors (including the global one)
    # ------------------------------------------------------------------
    try:
        global_init = np.array(height_sequence_1, dtype=float)
    except Exception:
        global_init = None

    # try several lengths – larger n gives more flexibility
    lengths = [500, 800, 1200, 1600, 2000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # a few diversified starts for the current length
        start_vectors = []

        if global_init is not None and global_init.shape[0] == n:
            start_vectors.append(global_init.copy())

        # uniform start (all entries equal)
        start_vectors.append(np.full(n, target_sum / n, dtype=float))

        # a few random starts (exponential – tends to be sparse)
        for _ in range(4):
            if rng.random() < 0.6:
                vec = rng.exponential(scale=1.0, size=n)
            else:
                vec = rng.random(n) + 0.1
            vec = _project_onto_simplex(vec, target_sum)
            start_vectors.append(vec)

        for start in start_vectors:
            if time.monotonic() >= deadline:
                break
            # ------------------------------------------------------------------
            # 2️⃣  Guided simulated‑annealing hill climbing
            # ------------------------------------------------------------------
            candidate, cand_score = _guided_hill_climb(start, deadline,
                                                       temp_init=0.1,
                                                       temp_decay=0.9995,
                                                       step_init=0.5,
                                                       max_iters=2_000_000)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = candidate.copy()
                print(f"[new‑best] n={n} score={best_score:.6f}")

            if best_score < 1.503:                     # target reached
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 3️⃣  Final Gaussian polishing (if we still have time)
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – moderate uniform vector of length 500
        n = 500
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    if time.monotonic() < deadline:
        refined, refined_score = _local_search(best_seq, deadline, step_initial=0.4)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final‑polish] improved to {best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
