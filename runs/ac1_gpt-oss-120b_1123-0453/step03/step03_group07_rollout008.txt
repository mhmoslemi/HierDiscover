def simplex_projection(v, s):

    n = len(v)
    if s < 0:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero?
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Propose a low‑value step sequence.

The algorithm is a fast gradient‑descent / simplex‑projection search.
For a fixed length n we keep the ℓ₁‑norm (the sum of the heights) equal to
√(2n); with this normalisation the evaluation score returned by

* start from several Dirichlet samples (and possibly from the previously

* at each iteration compute the convolution (via FFT) and locate the index
  (or indices) where the maximum is attained;
* the gradient of the max‑convolution with respect to the vector is a

The search is cheap (FFT‑based convolution, O(n log n) per iteration) and

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    # Shift by the maximum to avoid numerical issues
    v_shift = v - np.max(v)
    u = np.sort(v_shift)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = -1
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1) if rho >= 0 else 0.0
    w = np.maximum(v_shift - theta, 0.0)
    return w

def _conv_fft_full(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a∗a using FFT (real output)."""
    n = a.shape[0]
    # Next power of two ≥ 2n‑1 for speed & exact integer convolution
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    conv = _conv_fft_full(a)
    return float(conv.max())

def _gradient_of_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the max‑convolution (with respect to a).
    For each index k where conv[k] attains the global maximum we add
    contributions a_j to the gradient of a_i whenever i+j=k.
    Diagonal terms (i=j) contribute 2·a_i.
    """
    n = a.shape[0]
    max_val = float(conv.max())
    # tolerance to catch multiple active anti‑diagonals
    tol = 1e-12
    active = np.where(np.abs(conv - max_val) <= tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # iterate over i in the active anti‑diagonal
        for i in range(i_start, i_end + 1):
            j = k - i
            if i == j:
                g[i] += 2.0 * a[i]
            else:
                g[i] += a[j]
    return g

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the time budget (in seconds) and returns a list
    of floats representing the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # small safety margin

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – a mix of moderate and large sizes.
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    # Try to reuse a previously discovered sequence if it matches the length.
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        global_seq = np.asarray(global_seq, dtype=np.float64)
    else:
        global_seq = None

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)          # normalisation used by the evaluator

        # Number of independent random restarts for this length.
        restarts = 3

        for restart in range(restarts):
            if time.time() >= deadline:
                break

            # ---- Initialise -------------------------------------------------
            if restart == 0 and global_seq is not None and len(global_seq) == n:
                a = global_seq.copy()
                a = np.maximum(a, 0.0)
                a = a / a.sum() * target_sum
                print(f"[init] using global seed n={n}")
            else:
                a = rng.dirichlet(np.ones(n)) * target_sum

            # Enforce symmetry from the start (helps the later optimisation).
            a = (a + a[::-1]) / 2.0

            conv_full = _conv_fft_full(a)
            cur_M = float(conv_full.max())
            cur_score = cur_M                      # because sum = sqrt(2n)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            # ---- Gradient descent / local search ---------------------------
            # Limit the number of inner iterations by the remaining time.
            it = 0
            max_iters = 5000                       # generous upper bound
            while it < max_iters and time.time() < deadline:
                it += 1

                # Compute gradient of the current maximum.
                g = _gradient_of_max(a, conv_full)
                norm_g = np.linalg.norm(g)
                if norm_g == 0.0:
                    break

                # Step size schedule – decays like 1/√(it).
                step_scale = 0.12 * target_sum   # empirical factor
                alpha = step_scale / math.sqrt(it) / norm_g

                a_candidate = a - alpha * g
                a_candidate = np.maximum(a_candidate, 0.0)

                # Project back onto the simplex (preserve the ℓ₁‑norm).
                a_candidate = _simplex_projection(a_candidate, target_sum)

                # Enforce symmetry again (averaging with its reverse).
                a_candidate = (a_candidate + a_candidate[::-1]) / 2.0

                conv_new = _conv_fft_full(a_candidate)
                M_new = float(conv_new.max())

                # Accept only strict improvements.
                if M_new < cur_M - 1e-12:
                    a = a_candidate
                    conv_full = conv_new
                    cur_M = M_new
                    cur_score = M_new
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(f"[grad] n={n} restart={restart} it={it} score={cur_score:.6f}")
                    continue

                # No improvement – occasionally apply a tiny random shake.
                if it % 40 == 0:
                    perturb = rng.normal(scale=0.02 * target_sum, size=n)
                    a = np.clip(a + perturb, 0.0, None)
                    a = _simplex_projection(a, target_sum)
                    a = (a + a[::-1]) / 2.0
                    conv_full = _conv_fft_full(a)
                    cur_M = float(conv_full.max())
                    cur_score = cur_M
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(f"[perturb] n={n} restart={restart} it={it} score={cur_score:.6f}")

                # Early exit if we are already very close to the trivial lower bound.
                # (average value of the convolution)
                avg_bound = 2.0 * n / (2.0 * n - 1.0)
                if cur_M <= avg_bound * 1.01:
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(f"[near‑lb] n={n} it={it} score={cur_score:.6f}")
                    break

            # end of inner loop for a given restart
        # end of restarts loop for length n
    # ------------------------------------------------------------------

    # Fallback – uniform sequence if nothing was discovered (should never happen).
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _max_convolution(best_seq)

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    # Convert to plain Python list of floats as required.
    return [float(v) for v in best_seq]
