"""
Improved search for a low‑value step sequence.

The algorithm is based on the linear‑programming “mix‑with‑a‑better‑partner”
strategy described in the problem statement:

1.  Start from a random non‑negative vector on the ℓ¹‑simplex
    (∑ a_i = √(2 n)).
2.  Let M = max_k (a∗a)_k .  Solve the LP

        maximise   Σ g_i
        subject to  (a ∗ g)_k ≤ M   for all k = 0 … 2n‑2
                    0 ≤ g_i ≤ 1000

    (a ∗ g) is linear in g, therefore the constraints are linear.
    The LP finds a “partner” g that can carry as much mass as possible
    without raising the worst auto‑convolution entry beyond the current
    bound M.

3.  Rescale g to the required ℓ¹‑norm (√(2 n)), mix it with a
    (convex combination with mixing factor t) and recompute the new M.
    Repeat the LP‑mix step a few times – each iteration can only lower M.

4.  Perform several random restarts for a few different lengths
    (200 … 1200) while respecting the global time budget.
    The best sequence found is returned.  A short projected‑gradient
    polishing stage (identical to the previous submission) is added at
    the end for a final tweak.

All numbers are kept non‑negative and ≤ 1000, so the call to
`evaluate_sequence` will never trigger the internal clipping.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (simple, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _max_convolution(a: np.ndarray) -> float:
    """Return max entry of the (full) auto‑convolution of a."""
    return float(np.max(np.convolve(a, a)))


def _score_from_sequence(a: np.ndarray) -> float:
    """Exact value of the evaluation function (scale‑invariant)."""
    n = len(a)
    max_b = _max_convolution(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# LP‑based descent (the core of the new method)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (f ∗ g)_k = (A @ g)_k.
    Column i of A contains f_i on rows i … i+n‑1.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    One LP step: maximise Σ g_i subject to (f ∗ g) ≤ M and box bounds.
    Returns the rescaled partner vector g (∑ g_i = target_sum) or None
    if the LP fails.
    """
    n = len(f)
    A = _build_conv_matrix(f)
    # upper bound for each g_i (required to keep the LP bounded)
    ub = np.full(n, 1000.0)
    bounds = [(0.0, ub_i) for ub_i in ub]

    # maximise Σ g_i  →  minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c,
                  A_ub=A,
                  b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})

    if not res.success:
        return None

    g0 = res.x
    s0 = np.sum(g0)
    if s0 <= 0:
        return None

    # rescale to the required ℓ¹‑norm
    g = g0 * (target_sum / s0)
    # clip to the evaluator's hard bound and re‑project onto the simplex
    g = np.clip(g, 0.0, 1000.0)
    g = _project_onto_simplex(g, target_sum)
    return g


def _lp_descent(initial: np.ndarray,
                target_sum: float,
                max_outer_iters: int = 30,
                t_initial: float = 0.5,
                deadline: float = 1e9) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP‑mix step.  Returns the best vector seen and
    its corresponding score (max‑convolution based).
    """
    f = initial.copy()
    best_f = f.copy()
    best_score = _score_from_sequence(f)
    t = t_initial

    for outer in range(max_outer_iters):
        if time.monotonic() >= deadline:
            break

        cur_M = _max_convolution(f)

        # solve the LP and obtain a partner vector
        g = _lp_step(f, cur_M, target_sum)
        if g is None:
            break

        # convex combination
        f_new = (1.0 - t) * f + t * g

        # ensure feasibility (tiny numeric drift)
        f_new = np.clip(f_new, 0.0, 1000.0)
        f_new = _project_onto_simplex(f_new, target_sum)

        new_score = _score_from_sequence(f_new)

        # keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            # be a bit bolder with the mixing factor when we make progress
            t = min(0.9, t * 1.1)
        else:
            # no improvement → shrink the step
            t = max(0.05, t * 0.7)

        f = f_new

    return best_f, best_score

def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    eps = max_val * 1e-9 + 1e-12
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = f[j]
        grad[i] += 2.0 * vals
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= vals[mask]
    return grad


def _projected_gradient_descent(f0: np.ndarray,
                                target_sum: float,
                                deadline: float,
                                max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Simple projected sub‑gradient descent on the max‑convolution."""
    f = f0.copy()
    best_f = f.copy()
    best_score = _score_from_sequence(f)
    step = target_sum * 0.1
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        cur_score = _score_from_sequence(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)
        grad = _gradient_submax(f, active)

        f_candidate = _project_onto_simplex(f - step * grad, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)
        cand_score = _score_from_sequence(f_candidate)

        if cand_score < cur_score:
            f = f_candidate
            step *= 1.02
        else:
            step *= 0.5

    return best_f, best_score


def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """Very light Gaussian hill‑climbing polish."""
    rng = np.random.default_rng()
    target_sum = np.sum(start)
    best = start.copy()
    best_score = _score_from_sequence(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _score_from_sequence(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial

    return best, best_score

        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score_from_sequence(best_seq)

    # ------------------------------------------------------------------
    # 4️⃣  Short gradient‑based polishing (if time permits).
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        gd_deadline = deadline - 0.2
        target_sum = np.sum(best_seq)            # should already equal √(2n)
        f_refined, score_refined = _projected_gradient_descent(best_seq,
                                                               target_sum,
                                                               gd_deadline,
                                                               max_iters=4000)
        if score_refined < best_score:
            best_seq = f_refined
            best_score = score_refined
            print(f"[gd‑refine] score={score_refined:.6f}")

    # final stochastic polish (tiny budget, optional)
    if time.monotonic() < deadline:
        final_seq, final_score = _stochastic_refinement(best_seq,
                                                        deadline,
                                                        step_initial=0.3)
        if final_score < best_score:
            best_seq = final_seq
            best_score = final_score
            print(f"[final‑refine] score={final_score:.6f}")

    # ensure compliance with the evaluator's clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] final score ≈ {best_score:.6f}, length = {len(best_seq)}")
    return [float(x) for x in best_seq]
