    a = rng.random(n) + 0.01 # avoid zero.

    a = a / a.sum() * target_sum

    best_a = a.copy()

    best_sc = _fast_score(a)

    # schedule parameters
    t_start = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # compute conv
        conv = np.convolve(a, a)
        maxc = conv.max()
        # compute beta schedule
        # linear schedule: betas from 1 to 150
        beta = 1.0 + 149.0 * (it / max_iters)  # up to 150
        # compute softmax weights
        exponents = beta * (conv - maxc)  # subtract maxc to avoid overflow
        # use np.exp safely
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        grad = 2.0 * np.correlate(w, a, mode='valid')  # length n

        # step size schedule: start high then decay
        lr = 0.5 / (1.0 + it * 0.001)  # adjust

        a_new = a - lr * grad

        a_new = np.clip(a_new, 0.0, 1000.0)

        total = a_new.sum()
        if total > 0.0:
            a_new = a_new * (target_sum / total)
        else:
            a_new = np.full(n, target_sum / n, dtype=np.float64)

        # accept if improves or else maybe keep better
        sc = _fast_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # optional early exit if best_sc already below target e.g., 1.4.

    return best_a, best_sc

"""
Proposes a low‑value step sequence for the ``evaluate_sequence`` score.

The algorithm works in three stages:

1️⃣  **Broad parametric sweep** – for a range of lengths we generate
    simple families (power‑law, exponential, mixed) with a random offset.
    The vectors are normalised to ‖a‖₁ = √(2·n) (the scale used by the
    evaluation function).  The best candidate of the sweep is kept.

2️⃣  **Smooth‑max gradient descent** – starting from the best candidate
    we minimise a smooth approximation of the ∞‑norm
        φ_β(a) = log Σ_k exp(β·(a*a)_k) .
    Its gradient can be written `∇φ = 2·corr(w,a)`, where `w` is the
    soft‑max over the auto‑convolution entries.  A projected gradient step
    (non‑negative, fixed ℓ¹‑norm) is performed with a decreasing step‑size
    and a β that grows from 1 to ≈200.  The best point found during the
    descent is retained.

3️⃣  **Fine‑grained stochastic polishing** – a cheap swap hill‑climb and
    a short Gaussian hill‑climb further polish the final vector.

All steps respect the security checks of ``evaluate_sequence``:
non‑negative entries, clipping to [0,1000] and exact ℓ¹‑norm
normalisation, so the final objective equals the maximal auto‑convolution
entry.  The routine stops as soon as the 1000‑second budget expires and
returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact evaluation of ``evaluate_sequence`` without wrapper."""
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:               # should never happen (we normalise)
        return float("inf")
    return 2.0 * len(seq) * max_b / (sum_a ** 2)


def _smooth_optimize(start_seq: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of the
    infinity norm of the auto‑convolution.
    """
    a = start_seq.copy()
    best = a.copy()
    best_sc = _fast_score(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # auto‑convolution and smooth‑max weights
        conv = np.convolve(a, a)
        maxc = conv.max()
        beta = 1.0 + 199.0 * (it / max_iters)          # β grows from 1 to 200
        w = np.exp(beta * (conv - maxc))               # shift to avoid overflow
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (2·correlation)
        grad = 2.0 * np.correlate(w, a, mode="valid")  # length = n

        # projected gradient step (simple decreasing step‑size)
        lr = 0.5 / (1.0 + it * 0.01)
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, 1000.0)

        # re‑normalise ℓ¹‑norm exactly to target_sum
        s = a_new.sum()
        if s > 0.0:
            a_new = a_new * (target_sum / s)
        else:
            a_new = np.full(n, target_sum / n, dtype=np.float64)

        sc = _fast_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best = a_new.copy()

        a = a_new

    return best, best_sc


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """Very cheap Gaussian hill‑climb used as a final polish."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)

        # keep ℓ¹‑norm exact
        target = math.sqrt(2.0 * n)
        cand = cand * (target / cand.sum())

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9997
        if step < 1e-7:
            step = step_initial
    return best, best_sc


def _random_family(n: int,
                   target_sum: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    Generate a random vector from one of three simple families:
        - Power‑law   a_i ∝ (i+off)^‑p
        - Exponential a_i ∝ exp(‑α·i)  (optional offset inside the exponent)
        - Mixed       a_i ∝ (i+off)^‑p · exp(‑α·i)

    The vector is clipped to [0,1000] and normalised to the required ℓ¹‑norm.
    """
    idx = np.arange(n, dtype=np.float64)

    family = rng.choice([0, 1, 2])        # 0=power, 1=exp, 2=mixed
    if family == 0:                      # pure power‑law
        p = rng.uniform(0.2, 2.0)
        off = rng.uniform(0.1, 10.0)
        a = (idx + off) ** (-p)
    elif family == 1:                    # pure exponential
        alpha = rng.uniform(0.001, 0.05)
        a = np.exp(-alpha * idx)
    else:                                 # mixed family
        p = rng.uniform(0.2, 2.0)
        alpha = rng.uniform(0.001, 0.05)
        off = rng.uniform(0.0, 5.0)
        a = (idx + off) ** (-p) * np.exp(-alpha * idx)

    # normalise ℓ¹‑norm exactly to target_sum
    s = a.sum()
    if s == 0.0:
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a = a * (target_sum / s)

    a = np.clip(a, 0.0, 1000.0)
    return a
