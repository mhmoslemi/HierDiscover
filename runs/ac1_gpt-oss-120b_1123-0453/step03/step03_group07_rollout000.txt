def _project_onto_simplex(v, s):
    """Project v onto the simplex {x >= 0, sum x = s}."""
    if s < 0:
        raise ValueError("s must be non-negative")
    # shift v by sort
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All zero?
        rho = 0
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _lp_minimize_M(f, target_sum):
    try:
        from scipy.optimize import linprog
    except Exception:
        return None, None
    n = len(f)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # f[i] contributes to column (k - i)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    # Build linprog matrices
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    # Inequality: A @ g - M <= 0
    A_ub = np.hstack([A, -np.ones((rows, 1), dtype=np.float64)])
    b_ub = np.zeros(rows, dtype=np.float64)
    # Equality: sum(g) = target_sum
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1,1), dtype=np.float64)])
    b_eq = np.array([target_sum], dtype=np.float64)
    # bounds: g_i >=0, M >=0
    bounds = [(0, None)] * n + [(0, None)]
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None, None
    x = res.x
    g = np.maximum(x[:n], 0.0)
    M_val = x[-1]
    return g, float(M_val)

"""
Propose a low‑value step sequence.

This implementation builds on the previous LP‑based scheme but adds
* a fast linear‑programming step via SciPy’s high‑performance “highs”
  solver (fallback to cvxpy);
* a projected sub‑gradient descent that directly reduces the maximum
  of the self‑convolution;
* an exact convex‑mix line‑search between the current vector and the LP
  solution;
* adaptive restarts, larger candidate lengths (up to 1500) and a
  simple annealing‑like perturbation when progress stalls.

All helpers are defined at top level, use no closures and only standard
libraries together with numpy, scipy and cvxpy.  The search respects a
global time budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided ``evaluate_sequence``."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x >= 0, sum x = s}.
    Reference: Duchi et al., “Efficient Projections onto the l1‑Ball”.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find rho = max {j | u_j - cssv_j / j > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max_conv(f: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of max_i (f*f)[i] with respect to f.
    The sub‑gradient is the sum of gradients of all active maximal entries.
    """
    n = f.shape[0]
    conv = np.convolve(f, f)
    M = conv.max()
    active = np.where(conv >= M - eps)[0]  # indices where the max is attained
    grad = np.zeros_like(f)
    for i in active:
        lo = max(0, i - (n - 1))
        hi = min(n - 1, i)
        # For every j in [lo, hi] we add 2*f[i-j] to grad[j]
        # because d/d f_j (f*f)[i] = f[i-j] + f[i-j] = 2*f[i-j]
        # (the term with j = i-j contributes 2*f_j as well)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * f[i - j]
    return grad


def _lp_minimize_M(f: np.ndarray, target_sum: float):
    """
    Solve the linear program
        minimize   M
        subject to A @ g <= M,
                   sum(g) == target_sum,
                   g >= 0,
    where A is the convolution matrix of ``f``.
    Returns a feasible g and the optimal M, or (None, None) on failure.
    """
    # Try SciPy's high‑performance linear program solver first
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix A (rows x n)
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]

    # Linear program data
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0                     # objective: minimise M
    A_ub = np.hstack([A, -np.ones((rows, 1), dtype=np.float64)])  # A g - M <= 0
    b_ub = np.zeros(rows, dtype=np.float64)

    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([target_sum], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(0.0, None)]  # g_i >= 0, M >= 0

    if linprog is not None:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True})
            if res.success and res.x is not None:
                x = res.x
                g = np.maximum(x[:n], 0.0)
                M_val = float(x[-1])
                return g, M_val
        except Exception:   # pragma: no cover
            pass

    # Fallback to cvxpy if SciPy failed or is unavailable
    try:
        import cvxpy as cp
    except Exception:       # pragma: no cover
        return None, None

    g = cp.Variable(n)
    Mvar = cp.Variable()
    constraints = [
        g >= 0,
        A @ g <= Mvar,
        cp.sum(g) == target_sum,
        Mvar >= 0,
    ]
    prob = cp.Problem(cp.Minimize(Mvar), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:  # pragma: no cover
            return None, None

    if g.value is None or Mvar.value is None:
        return None, None

    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    M_val = float(Mvar.value)
    return g_val, M_val


def _optimal_mix(f: np.ndarray, g: np.ndarray,
                 Cff: np.ndarray, Cgg: np.ndarray, Cfg: np.ndarray,
                 current_M: float):
    """
    Exact line‑search for the optimal convex combination (1‑t)·f + t·g.
    Returns (new_vector, new_M) if a strictly better mixture exists,
    otherwise (None, None).
    """
    lo, hi = 0.0, 1.0
    best_M = current_M
    best_t = None

    # Ternary search – the max‑convolution is a convex function of t.
    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0

        comb1 = (1 - t1) ** 2 * Cff + 2 * t1 * (1 - t1) * Cfg + t1 ** 2 * Cgg
        M1 = np.max(comb1)

        comb2 = (1 - t2) ** 2 * Cff + 2 * t2 * (1 - t2) * Cfg + t2 ** 2 * Cgg
        M2 = np.max(comb2)

        if M1 < M2:
            hi = t2
            if M1 < best_M:
                best_M, best_t = M1, t1
        else:
            lo = t1
            if M2 < best_M:
                best_M, best_t = M2, t2

    # Final refinement
    t_opt = (lo + hi) / 2.0
    comb_opt = (1 - t_opt) ** 2 * Cff + 2 * t_opt * (1 - t_opt) * Cfg + t_opt ** 2 * Cgg
    M_opt = np.max(comb_opt)
    if M_opt < best_M:
        best_M, best_t = M_opt, t_opt

    if best_t is None or best_M >= current_M - 1e-12:
        return None, None

    new_vec = (1.0 - best_t) * f + best_t * g
    return new_vec, best_M


def _dirichlet_start(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex using a Dirichlet distribution."""
    x = rng.dirichlet(np.ones(n))
    return x.astype(np.float64) * target_sum

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The search proceeds over several candidate lengths (up to 1500),
    performs a few random restarts per length, and iteratively applies:
      * an LP that directly minimises the convolution bound,
      * a projected sub‑gradient descent on the max‑convolution,
      * an exact convex‑mix line‑search,
      * occasional Gaussian perturbations to escape stagnation.

    The best sequence discovered before the time budget expires is returned.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Candidate lengths – larger n generally permit lower scores but cost more.
    lengths = [300, 400, 500, 600, 800, 1000, 1200, 1500]

    # Number of independent restarts per length (adjusted later if time is low)
    base_restarts = 4
    max_lp_iters = 250          # maximum LP‑mix iterations per restart

    # Try to reuse a previously discovered sequence if available.
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        global_seq = np.asarray(global_seq, dtype=np.float64)
    else:
        global_seq = None

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # Adjust restarts based on remaining time and problem size.
        remaining = deadline - time.time()
        restarts = max(1, int(base_restarts * (remaining / budget_s)))
        restarts = min(restarts, 8)   # don’t explode the number of restarts

        for restart in range(restarts):
            if time.time() >= deadline:
                break

            # --------------------------------------------------------------
            # Initialise the current vector.
            if restart == 0 and global_seq is not None and len(global_seq) == n:
                f = global_seq.copy()
                f = np.maximum(f, 0.0)
                f = f / np.sum(f) * target_sum
                print(f"[init] using global seed n={n}")
            else:
                f = _dirichlet_start(n, target_sum, rng)

            conv_f = np.convolve(f, f)
            M = float(conv_f.max())
            cur_score = 2.0 * n * M / (target_sum ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            # --------------------------------------------------------------
            # Main optimisation loop.
            no_progress = 0
            for it in range(max_lp_iters):
                if time.time() >= deadline:
                    break

                # 1) Periodically call the LP‑based improvement (every 7th iteration)
                if it % 7 == 0:
                    g, M_lp = _lp_minimize_M(f, target_sum)
                    if g is not None:
                        # 2) Attempt the optimal convex mix between f and the LP solution.
                        Cff = conv_f
                        Cgg = np.convolve(g, g)
                        Cfg = np.convolve(f, g)
                        f_mix, M_mix = _optimal_mix(f, g, Cff, Cgg, Cfg, M)
                        if f_mix is not None:
                            f = f_mix
                            M = M_mix
                            conv_f = np.convolve(f, f)
                            cur_score = 2.0 * n * M / (target_sum ** 2)
                            no_progress = 0
                            if cur_score < best_score:
                                best_score = cur_score
                                best_seq = f.copy()
                                print(f"[mix‑LP] n={n} restart={restart} it={it} score={cur_score:.6f}")
                            continue  # skip gradient step this iteration

                # 3) Sub‑gradient descent on the active max‑convolution entries.
                grad = _subgradient_max_conv(f)
                if np.linalg.norm(grad) > 0:
                    # Adaptive step size – larger for higher dimensions.
                    eta = 0.4 * target_sum / (np.linalg.norm(grad) + 1e-12)
                    f_candidate = f - eta * grad
                    f_candidate = _project_onto_simplex(f_candidate, target_sum)
                else:
                    f_candidate = f.copy()

                conv_candidate = np.convolve(f_candidate, f_candidate)
                M_candidate = float(conv_candidate.max())
                if M_candidate < M - 1e-12:
                    # Accept the gradient step.
                    f = f_candidate
                    M = M_candidate
                    conv_f = conv_candidate
                    cur_score = 2.0 * n * M / (target_sum ** 2)
                    no_progress = 0
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                        print(f"[grad] n={n} restart={restart} it={it} score={cur_score:.6f}")
                    continue

                # 4) No improvement – stagnation handling.
                no_progress += 1
                if no_progress >= 5:
                    # Apply a modest random perturbation and re‑project.
                    perturb = rng.normal(scale=0.03 * target_sum / n, size=n)
                    f = np.clip(f + perturb, 0.0, None)
                    f = _project_onto_simplex(f, target_sum)
                    conv_f = np.convolve(f, f)
                    M = float(conv_f.max())
                    cur_score = 2.0 * n * M / (target_sum ** 2)
                    no_progress = 0
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                        print(f"[perturb] n={n} restart={restart} it={it} score={cur_score:.6f}")

                # 5) Early exit if we are already very close to the theoretical lower bound.
                lower_bound = 2.0 * n / (2.0 * n - 1.0)   # average of the convolution
                if M <= lower_bound * 1.01:
                    cur_score = 2.0 * n * M / (target_sum ** 2)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                        print(f"[near‑lb] n={n} score={cur_score:.6f}")
                    break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event nothing was discovered.
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _evaluate(best_seq)

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    # Return a plain Python list of floats as required.
    return [float(v) for v in best_seq]
