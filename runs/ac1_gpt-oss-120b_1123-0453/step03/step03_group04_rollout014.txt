Propose a low‑value step sequence for `evaluate_sequence`.

The search proceeds in four phases:

1️⃣  **Initialisation** – optionally reuse the previous best sequence
    (`height_sequence_1`).  Normalise it to the ℓ¹‑norm `sqrt(2·n)`.

2️⃣  **Broad parametric sweep** – for several lengths (up to 2000) generate
    simple families (exponential, power‑law, Hann/Hamming/Blackman windows,
    Kaiser when SciPy is available).  After normalisation each candidate is
    scored by an FFT‑based exact computation of the maximal auto‑convolution.
    The best candidate becomes the global incumbent.

3️⃣  **Multi‑scale refinement** – repeatedly up‑sample the incumbent (≈1.5×)
    using linear interpolation, renormalise and improve it with a fast
    stochastic local optimiser:
        * a cheap random “move mass between two positions” step,
        * a gradient step that reduces the current maximal convolution entry,
        * simulated‑annealing style uphill acceptance to escape shallow minima.
    The optimiser runs until the time budget expires or a length limit
    (5000) is reached.  Each successful improvement updates the global best.

4️⃣  **Final polishing** – a short Gaussian hill‑climb (similar to the
    original implementation) gives a last chance for tiny gains.

All vectors are kept non‑negative, clipped to the interval [0,1000] and
renormalised to `sum = sqrt(2·n)`, so the evaluation score equals the
maximal convolution entry.  The algorithm stays well within the 1000 s
budget and aims to beat the target value 1.503.

def _normalize(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and rescale to have exactly `target_sum`."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total == 0.0:
        seq = np.full_like(seq, target_sum / seq.shape[0])
    else:
        seq = seq * (target_sum / total)
    return seq


def _max_convolution_fft(seq: np.ndarray) -> float:
    """Return max entry of the linear auto‑convolution (exact via FFT)."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power‑of‑2 ≥ 2n‑1
    ft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(ft * ft, n=m)            # linear convolution, length m
    return float(conv[: 2 * n - 1].max())


def _full_convolution_fft(seq: np.ndarray) -> np.ndarray:
    """Return full linear auto‑convolution (length 2n‑1)."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(ft * ft, n=m)
    return conv[: 2 * n - 1]


def _gradient_step(seq: np.ndarray, target_sum: float, alpha: float) -> np.ndarray:
    """
    One cheap gradient‑like step that reduces the currently maximal convolution entry.
    The step size `alpha` is an absolute amount (not a relative factor).
    """
    n = seq.shape[0]
    conv = _full_convolution_fft(seq)
    k = int(conv.argmax())

    # gradient of conv[k] w.r.t. seq_i is seq_{k-i}
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    j_range = k - i_range
    g = np.zeros_like(seq)
    g[i_range] = seq[j_range]

    # direction that preserves the total sum (mean(g) subtraction)
    direction = -g + g.mean()
    new_seq = seq + alpha * direction
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    new_seq = _normalize(new_seq, target_sum)
    return new_seq


def _scale_up(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to a longer length `new_n`."""
    old_n = len(seq)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, seq)


def _local_optimize(
    start_seq: np.ndarray,
    target_sum: float,

    rng: np.random.Generator,

    """

        – 85 % random pair moves,
        – 15 % gradient steps,
        – simulated‑annealing uphill acceptance,
        – occasional reduction of step size.
    Returns the best sequence seen before `deadline` and its score.
    """
    n = start_seq.shape[0]
    best = start_seq.copy()
    best_sc = _max_convolution_fft(best)

    cur = best.copy()
    cur_sc = best_sc

    # initial parameters

    T = max(1e-4, best_sc * 0.05)              # temperature for uphill moves
    T_min = 1e-6
    cooling = 0.9995

    # --------------------------------------------------------------
    sweep_lengths = [500, 800, 1000, 1500, 2000]
    for n in sweep_lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        candidates = []

        # uniform
        candidates.append(np.full(n, target / n, dtype=np.float64))

        # exponential family
        for alpha in np.linspace(0.0005, 0.03, 30):
            a = np.exp(-alpha * np.arange(n, dtype=np.float64))
            a = a / a.sum() * target
            candidates.append(a)

        # power‑law family
        idx = np.arange(1, n + 1, dtype=np.float64)
        for p in np.linspace(0.1, 4.0, 30):
            a = idx ** (-p)
            a = a / a.sum() * target
            candidates.append(a)

        # Hann window
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
        a = a / a.sum() * target
        candidates.append(a)

        # Hamming window
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
        a = a / a.sum() * target
        candidates.append(a)

        # Blackman window
        a = (0.42 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
             + 0.08 * np.cos(4.0 * np.pi * np.arange(n) / (n - 1)))
        a = a / a.sum() * target
        candidates.append(a)

        # Kaiser windows (if SciPy is available)
        if _HAS_KAISER:
            i = np.arange(n, dtype=np.float64)
            arg = (2.0 * i) / (n - 1) - 1.0
            for beta in [2, 4, 6, 8, 10, 12]:
                a = _sp.i0(np.pi * beta * np.sqrt(1.0 - arg ** 2)) / _sp.i0(np.pi * beta)
                a = a / a.sum() * target
                candidates.append(a)

        # evaluate all candidates
        for cand in candidates:
            sc = _max_convolution_fft(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"[sweep] n={n} improved to {best_score:.6f}")

    # fallback if nothing was generated
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # 2️⃣  Multi‑scale refinement (increase length gradually).
    # --------------------------------------------------------------
    max_n = 5000
    current_seq = best_seq
    current_n = len(current_seq)

    while time.monotonic() < deadline and current_n < max_n:
        # increase length by roughly 1.5× (but not beyond max_n)
        next_n = min(max_n, int(current_n * 1.5))
        if next_n == current_n:
            break

        # up‑sample and re‑normalise
        up_seq = _scale_up(current_seq, next_n)
        target = math.sqrt(2.0 * next_n)
        up_seq = _normalize(up_seq, target)

        # allocate a time slice for this size (≤30 s or 30 % of remaining)
        remain = deadline - time.monotonic()
        slice_deadline = time.monotonic() + min(30.0, remain * 0.3)

        # improve the up‑sampled sequence
        up_seq_opt, up_sc = _local_optimize(up_seq, target, slice_deadline, rng)

        # record improvement
        if up_sc < best_score:
            best_seq, best_score = up_seq_opt.copy(), up_sc
            print(f"[scale] n={next_n} improved to {best_score:.6f}")

        current_seq, current_n = up_seq_opt, next_n

        # early exit if we are already very good
        if best_score < 1.45:
            break

    # --------------------------------------------------------------
    # 3️⃣  Final Gaussian hill‑climb polishing.
    # --------------------------------------------------------------
    if time.monotonic() < deadline:
        target = math.sqrt(2.0 * len(best_seq))
        refined_seq, refined_sc = _local_optimize(best_seq, target, deadline, rng)
        if refined_sc < best_score:
            best_seq, best_score = refined_seq, refined_sc
            print(f"[final] final best {best_score:.6f}")

    # Return plain Python list as required by the external harness
    return [float(v) for v in best_seq]
