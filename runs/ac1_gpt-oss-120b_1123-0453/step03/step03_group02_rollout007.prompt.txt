def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2·n‑1) × n matrix A with (A@g)[k] = (f*g)[k]."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A

def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project `v` onto {x ≥ 0, Σ x = target}."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - target
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]

    if rho_arr.size == 0:
        # all entries will be zero after projection
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w
