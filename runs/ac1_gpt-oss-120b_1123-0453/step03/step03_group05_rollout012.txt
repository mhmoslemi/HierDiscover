        target_sum_one = 1.0   # simplex
        for restart in range(restarts_per_length):
            if time.monotonic() >= deadline:
                break
            # initialize w
            w = rng.random(n)

            # optional: start from previous best if similar length?

            # For now simple.

            # subgradient descent
            # step schedule:
            eta0 = 0.8
            iter = 0
            max_iters = 200000   # large, but we will break by time.
            while time.monotonic() < deadline and iter < max_iters:
                # compute convolution via FFT
                size = 1 << ((2*n - 1).bit_length())
                fw = np.fft.rfft(w, n=size)
                conv = np.fft.irfft(fw * fw, n=size)[:2*n-1]
                max_val = conv.max()

                score = 2.0 * n * max_val
                if score < best_score:
                    best_score = score
                    best_seq = w.copy()

                if best_score < target:
                    return best_seq.tolist()

                # gradient for max entry
                idx = kmax - np.arange(n)
                mask = (idx >= 0) & (idx < n)
                grad = np.zeros_like(w)
                grad[mask] = 2.0 * w[idx[mask]]
                # learning rate
                eta = eta0 / np.sqrt(iter + 1)
                # exponentiated update

"""
Sub‑gradient / exponentiated‑gradient optimiser for the
autocorrelation‑maximisation problem.

The evaluation metric is scale–invariant; we therefore work with a
probability vector w (∑ w_i = 1).  For a fixed length n the score is

    score(w) = 2·n·max_k (w * w)_k .

The algorithm repeatedly

* evaluates the full linear convolution via a fast FFT,
* finds the shift k* where the convolution is maximal,
* computes a sub‑gradient of the max‑entry (which is simply
  2·w reflected around k*),
* performs an exponentiated‑gradient step,
* re‑normalises to the simplex.

Multiple random restarts, a few different lengths and a final
Gaussian hill‑climbing phase are used.  The routine stops as soon as
the target score 1.503 is reached or the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (fast replica of the public evaluate_sequence)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq))  (seq is assumed non‑negative,
    sum(seq) > 0.01).  No clipping or type checks – exactly the same
    formula used by evaluate_sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max() / (s * s))


def _fft_convolution(v: np.ndarray, size: int) -> np.ndarray:
    """Return the linear convolution v*v using an FFT of length *size*.
    *size* is a power of two ≥ 2·len(v)-1."""
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fv * fv, n=size)
    return conv[: 2 * v.shape[0] - 1]


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _local_search(start: np.ndarray, deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """Very short Gaussian hill‑climbing (keeps ∑=1 via simplex projection)."""
    rng = np.random.default_rng()
    n = start.shape[0]
    best = start.copy()
    best_sc = _fast_score(best)
    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, 1.0)
        sc = _fast_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best, best_sc


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step sequence with a low evaluation score.

    The algorithm runs a fast sub‑gradient optimiser (exponentiated‑gradient
    on the simplex) for several lengths and random restarts, followed by a
    very brief stochastic refinement.  As soon as the target score (1.503)
    is reached the current best vector is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    TARGET = 1.503

    # --------------------------------------------------------------
    # 0.  Warm‑start from the global best (if it exists)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.size > 0:
            w0 = seq0.clip(0.0, 1000.0)
            if w0.sum() > 0:
                w0 = w0 / w0.sum()
                sc0 = _fast_score(w0)
                best_seq = w0.copy()
                best_score = sc0
                print(f"[init] using global height_sequence_1 (n={len(w0)}) "
                      f"score={sc0:.6f}")

    # --------------------------------------------------------------
    # 1.  Main optimisation – several lengths, several restarts
    # --------------------------------------------------------------
    lengths = [500, 800, 1200, 1600, 2000]       # varied problem sizes
    max_restarts = 5
    eta0 = 0.8                                   # initial exponentiated‑gradient step
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        size_fft = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
        target_sum = 1.0                               # we work with probabilities

        # a small extra restart using the previous best if length matches
        extra_restart = 0
        if best_seq is not None and best_seq.shape[0] == n:
            extra_restart = 1

        for restart in range(max_restarts + extra_restart):
            if time.monotonic() >= deadline:
                break

            # ---- initialise -------------------------------------------------
            if extra_restart and restart == 0 and best_seq is not None:
                w = best_seq.copy()
            else:
                w = rng.random(n) + 0.1          # Dirichlet‑like start
                w = w / w.sum()

            it = 0
            max_iters = 200_000                  # hard cap per restart
            while time.monotonic() < deadline and it < max_iters:
                # ---- convolution (FFT) ------------------------------------
                conv = _fft_convolution(w, size_fft)
                max_val = conv.max()
                kmax = int(conv.argmax())
                score = 2.0 * n * max_val          # exact score for sum=1

                # keep the globally best vector
                if score < best_score:
                    best_score = score
                    best_seq = w.copy()
                    print(f"[sgd] n={n:4d} restart={restart} it={it:6d} "
                          f"score={score:.6f}")

                if best_score < TARGET:
                    return best_seq.tolist()

                # ---- sub‑gradient of the max‑entry -------------------------
                idx = kmax - np.arange(n)          # mirrored indices
                mask = (idx >= 0) & (idx < n)
                grad = np.zeros_like(w)
                grad[mask] = 2.0 * w[idx[mask]]    # ∇ max = 2·w reflected

                # ---- exponentiated‑gradient step ---------------------------
                eta = eta0 / math.sqrt(it + 1)     # slowly diminishing step
                w = w * np.exp(-eta * grad)        # multiplicative update
                # re‑normalise to the simplex (exact projection)
                s = w.sum()
                if s == 0.0:
                    w = np.full(n, 1.0 / n, dtype=np.float64)
                else:
                    w = w / s

                # occasional random jitter to avoid cycling
                if it % 5000 == 0 and it > 0:
                    jitter = rng.normal(scale=0.001, size=n)
                    w = np.clip(w + jitter, 0.0, 1.0)
                    w = _project_simplex(w, 1.0)

                it += 1

    # --------------------------------------------------------------
    # 2.  Final short stochastic refinement
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_sc = _local_search(best_seq, deadline, step_initial=0.4)
        if refined_sc < best_score:
            best_seq = refined
            best_score = refined_sc
            print(f"[local] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 200
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[final] best score={best_score:.6f} length={len(best_seq)}")
    # return a plain list of floats – scaling is irrelevant (score is scale‑free)
    return [float(x) for x in best_seq]
