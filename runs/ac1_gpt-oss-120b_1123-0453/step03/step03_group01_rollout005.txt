def improve_f(f, target_sum, deadline):
    best_f = f
    conv_f = np.convolve(f,f)
    M = np.max(conv_f)
    iteration = 0
    while time < deadline:
        C = build_conv_matrix(best_f)
        # solve LP
        p = linprog(c = -np.ones(n), A_ub=C, b_ub=np.full(rows, M), bounds=[(0, None)]*n, method='highs')
        if not p.success: break
        g = p.x
        sum_g = g.sum()
        if sum_g == 0: break
        g *= target_sum / sum_g

        improved = False

            M_new = np.max(conv_new)

                best_f = new_f
                M = M_new
                improved = True
                break
        if not improved:
            break
        # maybe also early stopping after some iterations or no improvement.
        iteration += 1
    return best_f, M

def _lp_iterate(f: np.ndarray, target_sum: float, deadline: float,
                max_iters: int = 20) -> tuple[np.ndarray, float]:
    n = len(f)
    rows = 2 * n - 1
    best_f = f
    # compute initial M
    conv = np.convolve(best_f, best_f)
    M = np.max(conv)
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        # build matrix C
        C = np.zeros((rows, n), dtype=np.float64)
        for j in range(n):
            C[j:j+n, j] = best_f
        # Solve LP (max sum g)
        c = -np.ones(n)
        b_ub = np.full(rows, M)
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        g *= target_sum / sum_g
        # try mixing
        improved = False
        # order of step sizes large to small
        for t in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            new_f = (1 - t) * best_f + t * g
            conv_new = np.convolve(new_f, new_f)
            M_new = np.max(conv_new)
            if M_new < M - 1e-12:
                best_f = new_f
                M = M_new
                improved = True
                break
        if not improved:
            # cannot improve further; break
            break
    return best_f, M

def _gradient_descent(f, target_sum, deadline, max_iters=2000, eta0=0.5):
    n = len(f)
    f_current = f.copy()
    conv = np.convolve(f_current, f_current); M = conv.max()
    step = eta0
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        idx = np.argmax(conv)
        # compute gradient: shift vector
        grad = np.zeros(n)
        # compute a_{idx - j} for j
        # For j in 0..n-1:
        low = max(0, idx - (n-1))
        high = min(idx, n-1)
        # For j such that idx - j in [0, n-1], assign grad[j] = 2 * f_current[idx - j]
        # Let's compute vector by using slice.
        j_start = low
        j_end = high
        # Actually j runs from max(0, idx-(n-1)) to min(idx, n-1)

        # grad[j] = 2 * f_current[k]
        # Equivalent to grad[low:high+1] = 2 * f_current[idx - np.arange(low, high+1)]
        js = np.arange(low, high+1)
        ks = idx - js

            step *= 0.7

"""
Hybrid optimizer for the non‑negative step‑function problem.

The routine works in three stages:

1️⃣  Initialise several random candidates on the simplex
    (∑a_i = √(2 n)).  One of the copies may be the previously
    discovered global `height_sequence_1`.

2️⃣  For each candidate run an *LP‑based* improvement:
    * fix the current vector f,
    * compute M = max(f * f),
    * solve   max ∑g_j   subject to   f * g ≤ M  and g ≥ 0,
    * rescale g to the same ℓ¹ norm as f,
    * replace f ← (1‑t) f + t g with a moderate step size t.
    This is the algorithm described in the statement and reduces the
    maximal convolution value monotonically.

3️⃣  Polish the best‑found vector with cheap hill‑climbing
    (mass‑transfer moves that keep the ℓ¹ norm) and a final Gaussian
    perturbation.  All operations respect the non‑negativity and clipping
    rules of `evaluate_sequence`.  The whole search stops before the
    supplied time budget (default 1000 s).

The evaluation function is homogeneous; after the ℓ¹‑normalisation
`∑a_i = √(2 n)` the score equals `max( a * a )`.  The optimizer therefore
works with the cheaper “max‑convolution” quantity and only calls the
official `evaluate_sequence` at the very end.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official (slow) score – used only for the final verification."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        { x ≥ 0 , Σx = target_sum } .
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the (2 n‑1) × n matrix C such that C @ g = f * g
    (convolution, zero‑padded outside the interval).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    # column j contributes a shifted copy of f starting at position j
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_improve(f: np.ndarray,
                target_sum: float,
                deadline: float,
                max_iters: int = 25,
                step: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based improvement described in the problem statement.
    Returns the improved vector and its maximal convolution value.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    best_f = f.copy()
    conv = np.convolve(best_f, best_f)
    M = conv.max()                     # current maximal convolution

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # build the convolution matrix for the current `best_f`
        C = _build_conv_matrix(best_f)

        # maximise 1ᵀ g  (i.e. minimise –1ᵀ g)   subject to   C g ≤ M,  g ≥ 0
        res = linprog(
            c=-np.ones(n),
            A_ub=C,
            b_ub=np.full(rows, M),
            bounds=[(0, None)] * n,
            method="highs",
        )
        if not res.success:
            # LP failed – abort the LP phase
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # rescale g to the same ℓ¹‑norm as f (i.e. `target_sum`)
        g = g * (target_sum / sum_g)

        # convex combination f ← (1‑t) f + t g
        new_f = (1.0 - step) * best_f + step * g

        # recompute objective
        new_conv = np.convolve(new_f, new_f)
        new_M = new_conv.max()

        if new_M < M - 1e-12:          # strict improvement
            best_f = new_f
            M = new_M
            # gradually reduce the mixing factor – we want finer steps later
            step = max(step * 0.9, 0.02)
        else:
            # no progress – try a smaller step once and give up if it fails
            step *= 0.5
            if step < 1e-3:
                break

    return best_f, M


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index `i` to index `j` (i ≠ j, δ > 0,
    δ ≤ a[i]).  The vectors ``a`` and its self‑convolution ``conv`` are
    updated in‑place in O(n) time.
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part: 2·δ·a contributes to the convolution
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part: convolution of the transfer vector with itself
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta          # (-δ)²
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta    # (-δ)*(+δ) + (+δ)*(−δ)
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta          # (+δ)²

    # update the underlying vector
    a[i] -= delta
    a[j] += delta


def _improve_by_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that repeatedly moves a tiny amount of mass
    from one coordinate to another while keeping the ℓ¹‑norm fixed.
    Returns the best sequence and its maximal convolution value.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)               # exact convolution
    best_a = a.copy()
    best_M = conv.max()
    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        _transfer_mass(a, conv, i, j, delta)
        cur_M = conv.max()

        if cur_M < best_M:
            best_M = cur_M
            best_a = a.copy()
        else:
            # revert
            _transfer_mass(a, conv, j, i, delta)

        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1

    return best_a, best_M


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Gentle Gaussian perturbation (as in the reference code) used as a
    final polishing step.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step sequence with a low evaluation score.
    The routine respects the global time budget (`budget_s` seconds) and
    returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    # small safety margin so the caller does not cut us off
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Choose lengths to explore (larger n gives more flexibility)
    # ------------------------------------------------------------------
    lengths = [500, 600, 800, 400, 300, 200]

    # In case a previous construction is available, keep it handy
    try:
        global height_sequence_1   # noqa: F821
        previous = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        previous = None

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)          # ℓ¹‑norm we enforce
        restarts = 3

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ------------------------------------------------------------------
            # 2️⃣  Initialise a random simplex point (or reuse a known one)
            # ------------------------------------------------------------------
            if previous is not None and previous.shape[0] == n:
                f0 = previous.copy()
            else:
                # Dirichlet‑like start: exponential draws give a sparse flavour
                f0 = rng.exponential(scale=1.0, size=n)
                f0 = f0 * (target_sum / f0.sum())

            # ------------------------------------------------------------------
            # 3️⃣  LP‑based improvement (core of the algorithm)
            # ------------------------------------------------------------------
            f_lp, M_lp = _lp_improve(f0, target_sum, deadline,
                                     max_iters=30, step=0.3)

            # ------------------------------------------------------------------
            # 4️⃣  Very cheap gradient descent on the max‑convolution entry
            # ------------------------------------------------------------------
            # (simple sub‑gradient step on the active index)
            f = f_lp.copy()
            steep = 0.5
            grad_iters = 800
            for _ in range(grad_iters):
                if time.monotonic() >= deadline:
                    break
                conv = np.convolve(f, f)
                idx = int(conv.argmax())
                # gradient of c_idx = (f * f)_idx w.r.t. f is 2 * shifted f
                grad = np.zeros_like(f)
                low = max(0, idx - (n - 1))
                high = min(idx, n - 1)
                js = np.arange(low, high + 1)
                ks = idx - js
                grad[js] = 2.0 * f[ks]

                # Take a step and project back onto the simplex
                f_new = f - steep * grad
                f_new = _project_onto_simplex(f_new, target_sum)

                new_M = np.convolve(f_new, f_new).max()
                if new_M < M_lp - 1e-12:
                    f = f_new
                    M_lp = new_M
                    steep = min(steep * 1.02, 1.0)   # enlarge a little when we improve
                else:
                    steep *= 0.7
                    if steep < 1e-6:
                        break

            # ------------------------------------------------------------------
            # 5️⃣  Record the best candidate from this restart
            # ------------------------------------------------------------------
            official = _score(f)
            if official < best_score:
                best_score = official
                best_seq = f.copy()
                print(f"[candidate] n={n} restart={r} score={best_score:.6f}")

            if best_score < 1.503:
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 6️⃣  Final polishing with hill‑climbing + Gaussian noise
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        # Hill‑climbing (mass‑transfer) – use a generous iteration budget
        refined, M_refined = _improve_by_transfer(
            best_seq.copy(),
            deadline,
            max_iters=400_000,
            step_init=0.5,
        )
        refined_score = _score(refined)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[hill‑climb] improved to {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline:
        final, final_score = _local_search(best_seq.copy(), deadline, step_initial=0.3)
        if final_score < best_score:
            best_seq = final
            best_score = final_score
            print(f"[final] polished to {best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    if best_seq is None:
        # deterministic fallback – a modest uniform vector
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    return [float(x) for x in best_seq]
