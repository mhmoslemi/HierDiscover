def _conv_matrix(f):
    n = len(f)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

    n = len(f)
    A = _conv_matrix(f)
    g = cp.Variable(n)
    lam = cp.Variable()

    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-8, reltol=1e-8, feastol=1e-8)
    if g.value is None:

    return np.maximum(g.value, 0.0), lam.value

"""
Improved search for a low‑value non‑negative step sequence.

The algorithm mixes three ideas:
1️⃣ Random Dirichlet starts for several lengths (400‑1800).
2️⃣ Fast projected gradient descent on the ℓ¹‑simplex that directly
   reduces the maximum entry of the auto‑convolution (the score is
   exactly that maximum when the ℓ¹‑norm is √(2 n)).
3️⃣ An LP refinement step (when cvxpy is available) that, for the
   current vector f, solves
        min   λ
        s.t.  (f * g)_i ≤ λ  ∀i,
              Σ g_i = √(2 n),  g_i ≥ 0 .
   The solution g replaces f and typically drops the maximal
   convolution value dramatically.
A short stochastic hill‑climbing polish finishes the run.

The whole routine respects the 1000 s time budget and returns the
best sequence found.
"""

import time
import math
import numpy as np

# optional LP solver
try:
    import cvxpy as cp
except Exception:            # pragma: no cover
    cp = None


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full auto‑convolution a*a using a power‑of‑2 FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _max_conv_and_active(a: np.ndarray):
    """Maximum convolution value, the set of (almost) active offsets,
    and the whole convolution array."""
    conv = _max_convolution_fft(a)
    max_val = np.max(conv)
    eps = max_val * 1e-9 + 1e-12
    active = np.where(conv >= max_val - eps)[0]
    return max_val, active, conv


def _gradient_submax(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximal auto‑convolution entry (summed over all
    active offsets)."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        # contribution 2*a_j to grad[i]
        grad[i] += 2.0 * a[j]
        # when i == j we added 2*a_i but only a_i should appear,
        # so subtract a_i once.
        mask = i == j
        if np.any(mask):
            grad[i[mask]] -= a[mask]
    return grad


def _compute_score_from_vector(a: np.ndarray) -> float:
    """Exact objective 2 n max(f*f) / (Σf)².  With Σf = √(2 n) it equals max."""
    n = a.shape[0]
    max_val = np.max(_max_convolution_fft(a))
    sum_val = np.sum(a)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


# ----------------------------------------------------------------------
# Projected gradient descent on the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _gradient_descent(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent that reduces max(a*a)."""
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)
    step = target_sum * 0.1
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        cur_score = _compute_score_from_vector(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
            print(f"[GD] it={it} n={len(f)} score={cur_score:.6f}")
        else:
            no_improve += 1

        if no_improve > 300:                     # early stop on stagnation
            break

        max_val, active, _ = _max_conv_and_active(f)
        grad = _gradient_submax(f, active)

        # take a step, project back onto the simplex
        f_candidate = _project_onto_simplex(f - step * grad, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)
        cand_score = _compute_score_from_vector(f_candidate)

        if cand_score < cur_score:
            f = f_candidate
            step *= 1.02
        else:
            step *= 0.5

    return best_f, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional, uses cvxpy)
# ----------------------------------------------------------------------


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build the (2n‑1)×n matrix A such that (A @ g) == f*g."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contributes f shifted by j positions
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_refine(
    f: np.ndarray,
    target_sum: float,
    max_iters: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP

        min   λ
        s.t.  (f * g) ≤ λ,
              Σ g = target_sum,
              g ≥ 0

    and replace f by the optimal g while the objective improves.
    """
    if cp is None:       # pragma: no cover
        return f, _compute_score_from_vector(f)

    best_f = f.copy()
    best_score = _compute_score_from_vector(f)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        A = _conv_matrix(best_f)

        g = cp.Variable(best_f.shape[0])
        lam = cp.Variable()
        constraints = [
            A @ g <= lam,
            cp.sum(g) == target_sum,
            g >= 0,
        ]
        prob = cp.Problem(cp.Minimize(lam), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)

        if prob.status not in ("optimal", "optimal_inaccurate"):
            print("[LP] solver failed, status:", prob.status)
            break

        g_val = np.maximum(g.value, 0.0)
        # Enforce the exact ℓ¹‑norm (tiny numerical drift may appear)
        sum_g = np.sum(g_val)
        if sum_g > 0:
            g_val = g_val * (target_sum / sum_g)

        new_score = _compute_score_from_vector(g_val)
        if new_score < best_score - 1e-9:
            best_f = g_val
            best_score = new_score
            print(f"[LP] iter={it} n={len(f)} score={new_score:.6f}")
        else:
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Stochastic polishing (Gaussian hill‑climbing)
# ----------------------------------------------------------------------


def _stochastic_refinement(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Simple Gaussian hill‑climbing that respects the ℓ¹‑norm."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial

    return best, best_score

    """

        score = _compute_score_from_vector(prior_arr)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    lengths = [400, 600, 800, 1000, 1200, 1400, 1600]

        if time.monotonic() >= deadline:

        restarts = 4
