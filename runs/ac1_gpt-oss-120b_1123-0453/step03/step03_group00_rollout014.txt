def fft_convolve(x, y):
    size = 1 << ((len(x) + len(y) - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:len(x)+len(y)-1]
    return conv

def _smooth_max_optim(a0, beta_initial=0.5, beta_factor=1.02, steps=2000, lr=0.1):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    best_score = np.inf
    best_a = a.copy()
    beta = beta_initial

    for t in range(1, steps+1):
        # compute conv
        conv = fft_convolve(a, a)
        max_val = conv.max()
        # computing current objective directly:
        # optional compute score = 2 * n * max_val / (sum^2) - but sum approx target_sum.
        score = 2.0 * len(a) * max_val / (np.sum(a) **2)  # should equal max_val if sum fixed.
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # compute softmax weights
        # to avoid overflow:
        # w = exp(beta*(conv - max_val)), then normalize.
        exp_terms = np.exp(beta * (conv - max_val))
        w = exp_terms / exp_terms.sum()
        # compute gradient
        a_rev = a[::-1]
        grad_full = fft_convolve(w, a_rev)
        grad = 2.0 * grad_full[len(a)-1 : len(a)-1 + len(a)]  # indices n-1 to n-1+n-1 inclusive
        # Adam update (descend)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # project to simplex (target sum)
        a_new = _project_onto_simplex(a_new, np.sum(a))  # maintain sum to original S as we maintain sum
        # clip to [0, 1000] for safety
        a_new = np.clip(a_new, 0.0, 1000.0)
        a = a_new
        # increase beta gradually
        beta *= beta_factor
    return best_a, best_score

"""
Propose a low‑value non‑negative step sequence.

The algorithm searches on several lengths using a *smooth‑max* (soft‑max)
approximation of the worst autocorrelation entry.  For each restart we:

1. initialise a candidate (uniform, geometric, or triangular shape);
2. minimise  F(a)=max_k (a*a)[k]  by Adam‑optimised gradient descent on the
   smoothed objective  fβ(a)=1/β·log∑_k exp(β·(a*a)[k]).
   The gradient of a single convolution entry is  ∂c_k/∂a_i = 2·a_{k‑i},
   which we compute efficiently with a single FFT‑based convolution.
3. after every step we project back onto the ℓ¹‑simplex (∑a_i = √(2n),
   a_i ≥ 0) and clip to the [0,1000] range required by ``evaluate_sequence``.
4. the best vector seen is kept; after the main loop a short stochastic
   Gaussian hill‑climbing refines it.

The whole search respects the given time budget (default 1000 s) and
returns a plain ``list[float]`` that can be fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x | x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # work on a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


# ----------------------------------------------------------------------
# Exact evaluation of the benchmark objective (scale‑invariant)
# ----------------------------------------------------------------------
def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact version of ``evaluate_sequence`` without the safety clipping."""
    n = len(f)
    conv = _fft_convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam on the soft‑max of the convolution)
# ----------------------------------------------------------------------
def _smooth_max_optimize(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Minimise max_k (a*a)[k] using a smooth‑max surrogate and Adam.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _compute_score_from_vector(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # smooth‑max parameters
    beta = 0.5               # starts very smooth
    beta_factor = 1.02       # slowly tighten towards the true max
    lr = 0.5                 # base learning rate for Adam
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # full autocorrelation
        conv = _fft_convolve(a, a)
        max_val = conv.max()
        cur_score = max_val                     # because Σa = √(2n) ⇒ ratio = max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[smooth] it={it:04d} n={n} score={cur_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 300:                # early stop on stagnation
                break

        # soft‑max weights over the convolution entries
        # stability: shift by the maximum
        exp_arg = beta * (conv - max_val)
        # avoid overflow/underflow
        exp_arg = np.clip(exp_arg, -700, 700)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max:   ∂f/∂a_i = Σ_k w_k * 2 a_{k-i}
        a_rev = a[::-1]
        grad_full = _fft_convolve(w, a_rev)                # length = 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]            # slice n‑1 … 2n‑2

        # Adam update (descent → move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # project back onto the ℓ¹‑simplex and clip to the admissible range
        a = _project_onto_simplex(a, target_sum)
        a = np.clip(a, 0.0, 1000.0)

        # tighten the soft‑max
        beta *= beta_factor

    # final exact evaluation of the best vector discovered
    final_score = _compute_score_from_vector(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Simple stochastic refinement (Gaussian perturbations)
# ----------------------------------------------------------------------
def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """Gaussian hill‑climbing while staying on the ℓ¹‑simplex."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    target_sum = np.sum(start)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        # slowly decay the perturbation size
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Public entry point ----------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # spare margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start from a previously discovered construction, if present.
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        n = len(seq)
        targ = math.sqrt(2.0 * n)
        if np.sum(seq) == 0:
            seq = np.full(n, targ / n, dtype=np.float64)
        else:
            seq = seq * (targ / np.sum(seq))
        score = _compute_score_from_vector(seq)
        best_seq, best_score = seq.copy(), score
        print(f"[init] warmed up from height_sequence_1 (n={n}) score={score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Main optimisation – smooth‑max Adam on several lengths.
    # ------------------------------------------------------------------
    # a diverse set of lengths; smaller ones are cheap, larger ones give more
    # freedom.  The total time is split roughly evenly across them.
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of restarts per length (adjustable)
        for restart in range(6):
            if time.monotonic() >= deadline:
                break

            # -----------------------------------------------------------
            # initial candidate
            # -----------------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
            else:
                mode = rng.integers(3)
                if mode == 0:
                    a0 = rng.random(n) + 0.01                     # near‑uniform random
                elif mode == 1:
                    alpha = rng.uniform(0.01, 2.0)                 # geometric decay
                    a0 = np.exp(-alpha * np.arange(n))
                else:
                    # symmetric triangular shape
                    half = (n + 1) // 2
                    inc = np.linspace(0.01, 1.0, half)
                    dec = np.linspace(1.0, 0.01, n - half)
                    a0 = np.concatenate([inc, dec])
                a0 = a0 / np.sum(a0) * target_sum

            # -----------------------------------------------------------
            # run the smooth‑max optimiser (time‑boxed)
            # -----------------------------------------------------------
            sub_deadline = min(deadline, time.monotonic() + 4.0)   # ≤4 s per run
            a_opt, score_opt = _smooth_max_optimize(a0,
                                                    target_sum,
                                                    sub_deadline,
                                                    max_iters=4000)

            if score_opt < best_score:
                best_score, best_seq = score_opt, a_opt.copy()
                print(f"[update] n={n} restart={restart} score={score_opt:.6f}")

            if best_score < 1.503:
                print("[early] target achieved")
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # 3️⃣ Final stochastic polishing (if any time remains)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: a modest uniform construction
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline,
                                                         step_initial=0.25)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # final safety clipping (evaluate_sequence also does this)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
