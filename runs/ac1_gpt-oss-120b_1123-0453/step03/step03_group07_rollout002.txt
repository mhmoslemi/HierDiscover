def simplex_projection(v, z=target): # v : array, sum constraint target
    # From https://arxiv.org/abs/1101.6081 (Duchi et al.)
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v) - z
    ind = np.arange(len(v)) + 1
    cond = sorted_v - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑Descent on a smooth approximation of the ℓ∞‑norm of the
auto‑convolution.  The algorithm works with any length up to 1000,
starts from several seeds (including the previously found
`height_sequence_1`), and repeatedly:
    1. evaluates a soft‑max (log‑sum‑exp) of the convolution,
    2. computes the gradient of this smooth objective,
    3. takes a projected gradient step onto the simplex
       (non‑negative entries with a fixed ℓ¹‑norm = √(2 n)),
    4. line‑searches a few step‑sizes and keeps any improvement,
    5. slowly increases the smoothing parameter α so that the
       objective converges to the true max‑convolution,
    6. injects small random jitter when progress stalls.

The routine respects the 1000 s budget, prints progress, and
returns the best sequence discovered (as a plain list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real convolution using FFT (fast for lengths up to a few thousand).
    Returns the full convolution of length len(a)+len(b)-1.
    """
    L = _next_pow2(len(a) + len(b) - 1)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = A * B
    conv = np.fft.irfft(C, n=L)
    return conv[: len(a) + len(b) - 1]


def _project_simplex(v: np.ndarray, total: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        { x | x_i ≥ 0,  Σ x_i = total }.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if total < 0:
        raise ValueError("total must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - total
    ind = np.arange(1, n + 1)
    # Find the rho index (last index where condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero after projection
        return np.zeros_like(v)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _dirichlet_start(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex using a Dirichlet(1,…,1)."""
    x = rng.dirichlet(np.ones(n))
    return x.astype(np.float64) * target_sum

    Search for a step‑height sequence with a low value of
    ``evaluate_sequence``.  The routine respects the given time budget
    (≈ 1000 s) and returns the best sequence found as a list of floats.
    """

    best_seq = None

        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    return [float(v) for v in best_seq]
