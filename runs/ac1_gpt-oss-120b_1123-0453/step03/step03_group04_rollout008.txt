"""
Propose a low‑score step sequence.

The algorithm works completely on the “max of the auto‑convolution”
objective (the evaluation score equals this max when the ℓ¹‑norm is
normalised to √(2·n)).  It consists of three stages:

1️⃣  **Diverse initialisations** – for a chosen length n we build a few
    hand‑crafted families (uniform, exponentials with different decay,
    power‑laws) together with a few random Dirichlet‑like vectors and,
    if available, the previously best construction.

2️⃣  **Projected sub‑gradient descent** – repeatedly locate the index k
    where the convolution attains its maximum.  The sub‑gradient of the
    ℓ∞‑objective at this point is 2·a_{k‑i} for each coefficient a_i.
    A direction that preserves the ℓ¹‑norm (by subtracting the mean) is
    taken, a step size η = η₀/√t is used and clipped so that no entry
    becomes negative.  After the update the vector is projected back onto
    the simplex (clipped and re‑scaled to the required sum).  The best
    vector seen so far is kept; the loop stops early if the target score
    1.503 is reached.

3️⃣  **Fine‑grained stochastic polishing** – a cheap swap‑move hill‑climb
    and a short Gaussian perturbation stage are applied while time
    permits, always keeping the best sequence.

All operations respect the safety checks of ``evaluate_sequence``:
non‑negative entries, optional clipping to [0,1000] (the values stay far
below 1000), and a fixed ℓ¹‑norm.  The routine returns the best sequence
found before the deadline.
"""

import time

def _normalise_to_sum(vec: np.ndarray, target: float) -> np.ndarray:
    """Clip to non‑negative and rescale so that the ℓ¹‑norm equals *target*."""
    np.clip(vec, 0.0, None, out=vec)
    total = vec.sum()
    if total > 0.0:
        vec *= target / total
    else:                                   # degenerate – fall back to uniform
        vec[:] = target / vec.size
    return vec


def _max_convolution(seq: np.ndarray) -> float:
    """Exact max entry of the full auto‑convolution."""
    return np.convolve(seq, seq).max()


def _sgd_one_step(seq: np.ndarray, target_sum: float, step: float):
    """
    Perform a single projected sub‑gradient step that tries to
    decrease the current maximum convolution entry.

    Returns (new_seq, new_max) where *new_seq* is the updated vector
    (already normalised to *target_sum*) and *new_max* its convolution max.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    cur_max = conv.max()
    k = int(conv.argmax())                     # index of the active peak

    # sub‑gradient of the max‑entry w.r.t a_i  (2·a_{k‑i} where defined)
    g = np.zeros_like(seq)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    i_range = np.arange(i_min, i_max + 1, dtype=int)
    j_range = k - i_range
    g[i_range] = 2.0 * seq[j_range]            # vectorised fill

    # direction that keeps the ℓ¹‑norm unchanged (subtract the mean)
    lam = g.mean()
    direction = -g + lam

    # guarantee non‑negativity: cap the step size if necessary
    if direction.min() < 0.0:
        # largest η so that a_i + η·direction_i ≥ 0 for all i
        η_max = np.min(seq[direction < 0.0] / -direction[direction < 0.0])
        step = min(step, 0.9 * η_max)          # a safety margin

    # simple back‑tracking line search – we accept a step that does not
    # increase the max‑convolution, otherwise halve η a few times.
    for _ in range(5):
        cand = seq + step * direction
        _normalise_to_sum(cand, target_sum)
        new_max = np.convolve(cand, cand).max()
        if new_max <= cur_max + 1e-12:        # improvement or no worsening
            return cand, new_max
        step *= 0.5
    # no improvement found – keep the original vector
    return seq, cur_max


def _stochastic_polish(seq: np.ndarray, target_sum: float, deadline: float):
    """
    Cheap hill‑climbing: random swaps of a small mass and a brief
    Gaussian perturbation.  Returns the possibly improved sequence.
    """
    rng = np.random.default_rng()
    best = seq.copy()
    best_sc = _max_convolution(best)

    n = best.size
    while time.time() < deadline:
        # ----- swap move -------------------------------------------------
        i = rng.integers(n)
        j = rng.integers(n)
        # move a small proportion of the mass from i to j
        delta = max(1e-8, best[i] * 0.02)
        if delta > best[i]:
            delta = best[i] * 0.5
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        _normalise_to_sum(cand, target_sum)
        sc = _max_convolution(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
            continue

        # ----- Gaussian tweak -------------------------------------------
        step = 0.1
        cand = best + rng.normal(scale=step, size=n)
        _normalise_to_sum(cand, target_sum)
        sc = _max_convolution(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
            continue

        # if neither move helped we break early
        break
    return best
