    if v.sum() == s and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_onto_simplex(v, s):
    # v: ndarray
    # s: target sum
    # returns projection onto {x >=0, sum x = s}
    # based on Duchi et al. (2008)
    # Assume s > 0.
    # If sum(v) == s and all v >=0, return v
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=? return zeros?
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # clip at 1000? optional after projection
    w = np.minimum(w, 1000.0)
    # if clipping reduces sum, we may need to renormalize again.
    # but given values are small, skip.
    return w

     full auto‑convolution.  The sub‑gradient at the worst index
     k* is 2·a_{k*‑i} (zero outside the support).
   • At each iteration we step a ← a – η·g, then project back onto the
     simplex {a ≥ 0, ∑a = √(2·n)} using the O(n log n) Euclidean projection.
   • The step size η decays linearly with time.
   • Occasionally a small random “jitter” or a random mass‑transfer move
     is tried to escape flat regions.

3️⃣ Final stochastic polishing – a short Gaussian hill‑climb (same
   routine as in the reference code) that may improve the best
   candidate found by the descent.

The whole search respects the supplied time budget; the majority of the
time is spent on the deterministic descent, which is far more efficient
than the simulated‑annealing loop used previously.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution and scoring (identical to evaluate_sequence)
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``seq`` with itself via real FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of 2
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Exact analogue of ``evaluate_sequence`` (no clipping/normalisation)."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Projects ``v`` onto the simplex with L1‑norm ``s``.
    Standard O(n log n) algorithm (Duchi et al. 2008).
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # already on simplex?
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho = max {j | u_j - cssv_j / j > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # enforce the cap of 1000 (practically never hit)
    w = np.minimum(w, 1000.0)
    # if clipping to 1000 broke the sum, renormalise a second time
    excess = w.sum() - s
    if excess > 1e-12:
        # simple redistribution of the excess to non‑saturated entries
        mask = w < 1000.0
        if mask.any():
            w[mask] -= excess / mask.sum()
            w = np.maximum(w, 0.0)
    return w


# ----------------------------------------------------------------------
# Interpolation utility (linear) to change length
# ----------------------------------------------------------------------


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return np.clip(new_seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Local stochastic polishing (Gaussian hill‑climb)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑normalise to the required ℓ¹‑norm
        target_sum = math.sqrt(2.0 * cand.shape[0])
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the ℓ∞‑norm of the auto‑convolution
# ----------------------------------------------------------------------


def _gradient_descent(seq: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Deterministic descent using the sub‑gradient of the maximal
    auto‑convolution entry, with projection onto the simplex.
    """
    n = seq.shape[0]
    cur = np.clip(seq, 0.0, 1000.0)
    cur = _project_onto_simplex(cur, target_sum)
    cur_score = _fast_score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    start = time.monotonic()
    total = deadline - start
    # linear schedule for the step‑fraction
    step_frac_start = 0.15
    step_frac_end = 1e-5

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = elapsed / total
        step_frac = step_frac_start * (1.0 - frac) + step_frac_end
        # 1. compute convolution and locate worst index
        conv = _conv_full_fft(cur)
        k_max = int(conv.argmax())
        # 2. sub‑gradient for that index
        g = np.zeros_like(cur)
        i_lo = max(0, k_max - (n - 1))
        i_hi = min(n - 1, k_max)
        idx = np.arange(i_lo, i_hi + 1, dtype=int)
        jdx = k_max - idx
        g[idx] = 2.0 * cur[jdx]          # sub‑gradient component
        # 3. gradient step
        eta = step_frac * target_sum
        cand = cur - eta * g
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _fast_score(cand)

        # 4. accept if we made progress
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[GD] n={n:4d} iter={int(elapsed*1000):7d} "
                      f"score={cand_score:.6f}")
        else:
            # occasional random jitter to escape plateaus
            if rng.random() < 0.02:
                jitter = rng.normal(scale=eta, size=n)
                cand2 = np.clip(cur + jitter, 0.0, 1000.0)
                cand2 = _project_onto_simplex(cand2, target_sum)
                cand2_score = _fast_score(cand2)
                if cand2_score < cur_score:
                    cur = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_seq = cand2.copy()
                        best_score = cand2_score
                        print(f"[GD‑jitter] n={n:4d} iter={int(elapsed*1000):7d} "
                              f"score={cand2_score:.6f}")
            # also try a tiny random pairwise mass transfer
            if rng.random() < 0.01:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j:
                    max_transfer = min(cur[i], eta)
                    delta = rng.random() * max_transfer
                    cand3 = cur.copy()
                    cand3[i] -= delta
                    cand3[j] += delta
                    cand3 = _project_onto_simplex(cand3, target_sum)
                    cand3_score = _fast_score(cand3)
                    if cand3_score < cur_score:
                        cur = cand3
                        cur_score = cand3_score
                        if cand3_score < best_score:
                            best_seq = cand3.copy()
                            best_score = cand3_score
                            print(f"[GD‑transfer] n={n:4d} iter={int(elapsed*1000):7d} "
                                  f"score={cand3_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # tiny safety margin so the caller can still read the result
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Warm‑start
    # ------------------------------------------------------------------
    known = globals().get("height_sequence_1", None)
    if known is not None:
        known = np.asarray(known, dtype=np.float64)
        n = known.shape[0]
        init_seq = known.copy()
    else:
        n = 500                       # default length
        init_seq = rng.random(n) + 0.05

    # normalise ℓ¹‑norm to √(2·n)
    target_sum = math.sqrt(2.0 * n)
    init_seq = np.clip(init_seq, 0.0, 1000.0)
    init_seq = _project_onto_simplex(init_seq, target_sum)

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent (major part of the budget)
    # ------------------------------------------------------------------
    gd_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.78
    best_seq, best_score = _gradient_descent(init_seq,
                                             target_sum,
                                             gd_deadline,
                                             rng)

    print(f"[global] after GD best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (Gaussian hill‑climb)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        final_seq, final_score = _local_search(best_seq, deadline, step_initial=0.4)
        if final_score < best_score:
            best_seq, best_score = final_seq, final_score
            print(f"[final] improved to {best_score:.6f}")

    # safety fallback – should never trigger
    if best_seq is None:
        best_seq = np.full(500, 1.0, dtype=np.float64)
        best_seq = _project_onto_simplex(best_seq, math.sqrt(2.0 * 500))

    # return plain Python list
    return [float(v) for v in best_seq]
