Project onto simplex function: from literature:

```
def project_onto_simplex(v, S):
    # Ensure non-negative and sum = S.
    # Compute sorted v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u > cssv / np.arange(1, len(v)+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Works for any S >0.

def gradient_descent(a0, S, deadline):
    a = a0.copy()
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_M = float(conv.max())
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute conv for current a
        conv = np.convolve(a, a)
        M = float(conv.max())
        # update best
        if M < best_M:
            best_M = M
            best_a = a.copy()
        # identify active indices
        eps = 1e-12
        active = np.where(conv >= M - eps)[0]
        # compute gradient
        g = compute_gradient(a, active)
        # step size
        step = 0.5 * (S / len(a)) / np.sqrt(iteration)  # can tune
        # try line search for up to attempts
        improved = False
        for attempt in range(10):
            a_candidate = a - step * g
            a_candidate = project_onto_simplex(a_candidate, S)
            conv_cand = np.convolve(a_candidate, a_candidate)
            M_cand = float(conv_cand.max())
            if M_cand + 1e-12 < M:
                a = a_candidate
                improved = True
                break
            else:
                step *= 0.5
        if not improved:
            # random perturbation
            perturb = 0.01 * S * rng.normal(size=len(a))
            a = a + perturb
            a = project_onto_simplex(a, S)
    return best_a, best_M
```

def project_onto_simplex(v, S):
    # based on algorithm from Duchi et al. (2008)
    if S < 0:
        raise ValueError
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent based sub‑gradient optimizer.

The algorithm searches the simplex of non‑negative vectors of a given
length n (∑a_i = √(2n)).  For each length we start from several
different seeds (global best, Dirichlet random, Gaussian‑shaped,
triangular, and tiled‑global) and then run a projected sub‑gradient
method that directly minimizes the infinity norm of the auto‑convolution
 a*a.  The sub‑gradient of max_k (a*a)[k] is obtained from the active
indices where the maximum is attained.  After each step the vector is
projected onto the simplex (preserving non‑negativity and the required
ℓ₁‑norm).  Occasional random perturbations help escape plateaus.  The
best sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _evaluate(seq: np.ndarray) -> float:
    """Wrapper around the supplied ``evaluate_sequence`` function."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    if S < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are forced to zero – return zero vector
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_gradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] with respect to a.
    For each active convolution index k we add
        2 * a_{k‑i}   for all i where the term appears.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # reversed slice of a[lo:hi+1]
        if lo == 0:
            rev = a[hi::-1]
        else:
            rev = a[hi:lo - 1:-1]
        g[lo:hi + 1] += 2.0 * rev
    return g


def _dirichlet_start(n: int, S: float, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex via a Dirichlet distribution."""
    x = rng.dirichlet(np.ones(n))
    return x.astype(np.float64) * S


def _gaussian_start(n: int, S: float) -> np.ndarray:
    """Gaussian‑shaped seed (symmetric around the centre)."""
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = n / 6.0
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a = a / a.sum() * S
    return a


def _triangular_start(n: int, S: float) -> np.ndarray:
    """Simple symmetric triangular shape."""
    if n % 2 == 0:
        half = n // 2
        left = np.arange(1, half + 1)
        right = np.arange(half, 0, -1)
    else:
        half = n // 2
        left = np.arange(1, half + 2)
        right = np.arange(half + 1, 0, -1)
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum() * S
    return a


def _repeat_global_start(global_seq: np.ndarray, n: int, S: float) -> np.ndarray:
    """Tile the global sequence to obtain a seed for a larger length."""
    base = global_seq.astype(np.float64)
    reps = int(np.ceil(n / base.size))
    a = np.tile(base, reps)[:n]
    a = a / a.sum() * S
    return a


def _gradient_descent(
    a0: np.ndarray,
    S: float,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the simplex.
    Returns the best vector encountered and its convolution maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_M = float(np.max(np.convolve(a, a)))
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = np.convolve(a, a)
        M = float(conv.max())

        if M < best_M:
            best_M = M
            best_a = a.copy()

        # active indices (those attaining the maximum)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient
        g = _compute_gradient(a, active)

        # base step size – fraction of the average coordinate
        step = 0.5 * (S / n) / math.sqrt(it)

        improved = False
        # simple back‑tracking line search
        for _ in range(5):
            a_candidate = a - step * g
            a_candidate = _project_onto_simplex(a_candidate, S)
            M_cand = float(np.max(np.convolve(a_candidate, a_candidate)))
            if M_cand < M - 1e-12:
                a = a_candidate
                improved = True
                break
            step *= 0.5

        if not improved:
            # occasional stochastic kick to escape flat regions
            if it % 20 == 0:
                pert = rng.normal(scale=0.02 * S, size=n)
                a = _project_onto_simplex(a + pert, S)

    return best_a, best_M


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative heights) with a low
    evaluation score.  The routine tries several lengths up to 2000,
    employs a handful of diverse initialisations and a projected
    sub‑gradient optimiser.  The best sequence found before the time
    budget expires is returned as a plain Python list.
    """
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # candidate lengths – longer vectors tend to admit lower maxima
    lengths = [500, 800, 1000, 1500, 2000]

    # retrieve a possible previously discovered construction
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        global_seq = np.asarray(global_seq, dtype=np.float64)
    else:
        global_seq = None

    for n in lengths:
        if time.time() >= deadline:
            break
        S = math.sqrt(2.0 * n)          # required ℓ₁‑norm

        # five different initialisation strategies
        for restart in range(5):
            if time.time() >= deadline:
                break

            # --------------------------------------------------------------
            # initialise a candidate vector
            if restart == 0 and global_seq is not None and global_seq.size == n:
                a0 = global_seq.copy()
                a0 = np.maximum(a0, 0.0)
                a0 = a0 / a0.sum() * S
                tag = "global"
            elif restart == 1:
                a0 = _dirichlet_start(n, S, rng)
                tag = "dirichlet"
            elif restart == 2:
                a0 = _gaussian_start(n, S)
                tag = "gaussian"
            elif restart == 3:
                a0 = _triangular_start(n, S)
                tag = "triangular"
            else:
                # repeat / tile the global seed (if available)
                if global_seq is not None:
                    a0 = _repeat_global_start(global_seq, n, S)
                    tag = "repeat"
                else:
                    a0 = _dirichlet_start(n, S, rng)
                    tag = "dirichlet"

            # evaluate the initial point
            cur_M = float(np.max(np.convolve(a0, a0)))
            cur_score = cur_M                # because ∑a = √(2n)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a0.copy()
                print(f"[init] n={n} restart={restart} ({tag}) score={cur_score:.6f}")

            # --------------------------------------------------------------
            # run the projected sub‑gradient optimiser
            a_opt, M_opt = _gradient_descent(a0, S, rng, deadline)

            if M_opt < best_score:
                best_score = M_opt
                best_seq = a_opt.copy()
                print(f"[gd]   n={n} restart={restart} ({tag}) score={M_opt:.6f}")

    # ----------------------------------------------------------------------
    # fallback in the very unlikely case we never found a valid vector
    if best_seq is None:
        n = 200
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n, dtype=np.float64)
        best_score = _evaluate(best_seq)

    # final sanity check via the official evaluator (adds the extra guards)
    final_score = _evaluate(best_seq)
    print(f"Best score overall: {final_score:.6f} (length {len(best_seq)})")
    return [float(v) for v in best_seq]
