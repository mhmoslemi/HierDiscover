def simplex_projection(v, s):
    """project onto the simplex {x >=0, sum x = s}."""
    # algorithm: sort descending, find rho, compute theta.
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_max_conv(seq, km):
    n = seq.shape[0]
    grad = np.zeros_like(seq)
    # compute j = km - i
    idx = np.arange(n)
    j = km - idx
    mask = (j >= 0) & (j < n)

    # adjust double count for i=j case:
    if km % 2 == 0:

        if 0 <= i_eq < n:

def _lp_improve(seq, target_sum, deadline, rng):
    cur_seq = seq.copy()
    cur_score = _fast_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    n = cur_seq.shape[0]
    # allocate time for few LP solves
    max_iter = 30
    while time.monotonic() < deadline and max_iter > 0:
        conv = _conv_full_fft(cur_seq)
        M = conv.max()
        # solve LP
        A = np.zeros((2*n-1, n))
        for i in range(n):
            A[i:i+n, i] = cur_seq[i]
        # use linprog
        c = -np.ones(n)  # maximize sum(b) => minimize -sum
        b_ub = np.full(2*n-1, M)
        # bounds >=0
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("[LP] solve failed")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= target_sum * (1 + 1e-6):
            # cannot increase sum
            break
        # scale to target_sum
        g = b * (target_sum / sum_b)
        # mix
        t = 0.3  # mixing factor
        cand = (1 - t) * cur_seq + t * g
        # clip and renorm
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0:
            cand = cand * (target_sum / s)
        cand_score = _fast_score(cand)
        if cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[LP] improved score {cand_score:.6f}")
        else:
            # maybe reduce t and try again?
            # degrade t?
            t = max(t * 0.5, 0.05)
        max_iter -= 1
    return best_seq, best_score

def _simplex_projection(v, z):
    """Project v onto the simplex {x >= 0, sum(x) = z}."""

"""
Algorithm:
1. Choose a moderate length n (default 1000). Initialize a non‑negative
   sequence with ℓ¹‑norm √(2·n).  If a previous solution exists it is
   interpolated to the chosen length.
2. Linear‑programming refinement: with the current sequence f compute
   M = max(f*f).  Solve the LP  maximise Σb_j  subject to (f*b)_k ≤ M
   and b≥0.  If the LP yields a larger ℓ¹‑norm than √(2·n) we scale the
   solution back to the required norm – this strictly decreases the
   max‑convolution value.  Repeat while time permits.
3. Projected gradient descent on the worst convolution entry.
   The gradient of the maximal entry is 2·f_{k‑i} when k is the index
   of the max entry.  A step in the opposite direction, followed by a
   projection onto the simplex {x≥0, Σx = √(2·n)}, monotonically reduces
   the objective.
4. Final stochastic polishing: small Gaussian perturbations are tried
   and accepted if they improve the fast‑score.
The whole procedure respects the supplied time budget and returns the
best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution and scoring (identical to evaluate_sequence logic)
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of seq with itself using an FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Fast analogue of evaluate_sequence (no clipping performed)."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simplex projection (ℓ¹‑norm = z, non‑negative)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """Projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.shape[0]
    # If already on the simplex we can return early
    if np.abs(v.sum() - z) < 1e-12 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    rho_arr = u - cssv / ind
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑programming refinement (step 2 of the algorithm)
# ----------------------------------------------------------------------


def _lp_refine(seq: np.ndarray,
               target_sum: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """Repeatedly solve the LP that maximises Σb under the current max‑conv bound."""
    try:
        from scipy.optimize import linprog
    except Exception as e:
        print("[LP] scipy not available, skipping LP refinement.")
        return seq, _fast_score(seq)

    n = seq.shape[0]
    cur_seq = seq.copy()
    cur_score = _fast_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        conv = _conv_full_fft(cur_seq)
        M = float(conv.max())

        # Build the (2n‑1)×n constraint matrix A such that (f*b)_k = (A b)_k.
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = cur_seq[i]

        c = -np.ones(n, dtype=np.float64)  # maximise Σb ⇒ minimise −Σb
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})

        if not res.success:
            print(f"[LP] iteration {iteration}: linprog failed – {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        # If the LP cannot increase the ℓ¹‑norm we stop.
        if sum_b <= target_sum * (1.0 + 1e-9):
            break

        # Scale the solution back to the required ℓ¹‑norm.
        new_seq = b * (target_sum / sum_b)
        new_score = _fast_score(new_seq)

        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(f"[LP] iter {iteration}: improved score {new_score:.6f}")
        else:
            # No improvement – exit early.
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected gradient descent on the worst convolution entry (step 3)
# ----------------------------------------------------------------------


def _proj_grad_descent(seq: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Gradient descent that pushes mass away from the index giving the current max‑conv."""
    n = seq.shape[0]
    cur_seq = seq.copy()
    cur_score = _fast_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    # Initial step size – a modest fraction of the total ℓ¹‑norm.
    step = max(0.01, target_sum * 0.05)
    min_step = max(1e-8, target_sum * 1e-6)

    while time.monotonic() < deadline:
        conv = _conv_full_fft(cur_seq)
        km = int(conv.argmax())

        # Gradient of the maximal entry: g_i = 2·a_{km‑i} (if index valid)
        idx = np.arange(n)
        j = km - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur_seq)
        grad[mask] = 2.0 * cur_seq[j[mask]]

        # Candidate step.
        cand = cur_seq - step * grad
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _simplex_projection(cand, target_sum)

        cand_score = _fast_score(cand)

        if cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[PGD] improved score {cand_score:.6f}")

        # adapt step size (simple back‑off)
        if cand_score >= cur_score:
            step *= 0.5
            step = max(step, min_step)

        # slow decay of step (helps to fine‑tune later)
        step *= 0.9999
        step = max(step, min_step)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stochastic local polishing (step 4)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """Gaussian hill climbing that accepts only improving moves."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)    # respect the hard bounds
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-8:
            step = step_initial

    return best, best_score

# ----------------------------------------------------------------------

    """

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        if known_seq.shape[0] != n:
            # linear interpolation to the required length
            old_n = known_seq.shape[0]
            old_x = np.linspace(0.0, 1.0, old_n)
            new_x = np.linspace(0.0, 1.0, n)
            init_seq = np.interp(new_x, old_x, known_seq)
        else:
            init_seq = known_seq.copy()
    else:
        # pure random start – small offset guarantees positivity
        init_seq = rng.random(n) + 0.1

    # enforce the hard bounds and the ℓ¹‑norm required by the problem
    init_seq = np.clip(init_seq, 0.0, 1000.0)
    cur_sum = init_seq.sum()
    if cur_sum > 0:
        init_seq = init_seq * (target_sum / cur_sum)

    # ------------------------------------------------------------------
    # Split the remaining budget among the three phases
    # ------------------------------------------------------------------
    time_left = deadline - time.monotonic()
    if time_left <= 0:
        return [float(v) for v in init_seq]

    # 30 % for LP refinement, 50 % for projected gradient, 20 % for polishing
    lp_deadline = time.monotonic() + 0.30 * time_left
    pgd_deadline = time.monotonic() + 0.50 * time_left
    final_deadline = deadline  # whatever is left

    # ------------------------------------------------------------------
    # Phase 1 – LP refinement
    # ------------------------------------------------------------------
    seq_lp, score_lp = _lp_refine(init_seq, target_sum, lp_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – Projected gradient descent
    # ------------------------------------------------------------------
    seq_pgd, score_pgd = _proj_grad_descent(seq_lp, target_sum, pgd_deadline)

    # ------------------------------------------------------------------
    # Phase 3 – Stochastic polishing
    # ------------------------------------------------------------------
    seq_final, score_final = _local_search(seq_pgd, final_deadline, step_initial=0.5)

    # Use the best of the three intermediate results (just in case a
    # later phase accidentally made things worse).

    best_score = score_final
    for cand_seq, cand_score in [(seq_lp, score_lp), (seq_pgd, score_pgd)]:

It combines a linear‑programming refinement (to increase the ℓ¹‑norm while keeping the maximal auto‑convolution unchanged), a projected gradient descent that directly reduces the worst convolution entry, and a final stochastic polishing step. ```python
"""
Algorithm:
1. Choose a moderate length n (default 1000). Initialise a non‑negative
   sequence with ℓ¹‑norm √(2·n).  If a previous solution exists it is
   interpolated to the chosen length.
2. Linear‑programming refinement: with the current sequence f compute
   M = max(f*f).  Solve the LP  maximise Σb_j  subject to (f*b)_k ≤ M
   and b≥0.  If the LP yields a larger ℓ¹‑norm than √(2·n) we scale the
   solution back to the required norm – this strictly decreases the
   max‑convolution value.  Repeat while time permits.
3. Projected gradient descent on the worst convolution entry.
   The gradient of the maximal entry is 2·f_{k‑i} when k is the index
   of the max entry.  A step in the opposite direction, followed by a
   projection onto the simplex {x≥0, Σx = √(2·n)}, monotonically reduces
   the objective.
4. Final stochastic polishing: small Gaussian perturbations are tried
   and accepted if they improve the fast‑score.
The whole procedure respects the supplied time budget and returns the
best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution and scoring (identical to evaluate_sequence logic)
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of seq with itself using an FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Fast analogue of evaluate_sequence (no clipping performed)."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simplex projection (ℓ¹‑norm = z, non‑negative)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """Projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.shape[0]
    # Early exit if already feasible
    if np.abs(v.sum() - z) < 1e-12 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    rho_arr = u - cssv / ind
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑programming refinement (step 2 of the algorithm)
# ----------------------------------------------------------------------


def _lp_refine(seq: np.ndarray,
               target_sum: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """Repeatedly solve the LP that maximises Σb under the current max‑conv bound."""
    try:
        from scipy.optimize import linprog
    except Exception as e:
        print("[LP] scipy not available, skipping LP refinement.")
        return seq, _fast_score(seq)

    n = seq.shape[0]
    cur_seq = seq.copy()
    cur_score = _fast_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        conv = _conv_full_fft(cur_seq)
        M = float(conv.max())

        # Build the (2n‑1)×n constraint matrix A such that (f*b)_k = (A b)_k.
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = cur_seq[i]

        c = -np.ones(n, dtype=np.float64)          # maximise Σb ⇒ minimise −Σb
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(0.0, deadline - time.monotonic())})

        if not res.success:
            print(f"[LP] iteration {iteration}: linprog failed – {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= target_sum * (1.0 + 1e-9):
            # No room for improvement.
            break

        # Scale back to the required ℓ¹‑norm.
        new_seq = b * (target_sum / sum_b)
        new_score = _fast_score(new_seq)

        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(f"[LP] iter {iteration}: improved score {new_score:.6f}")
        else:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected gradient descent on the worst convolution entry (step 3)
# ----------------------------------------------------------------------


def _proj_grad_descent(seq: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Gradient descent that pushes mass away from the index giving the current max‑conv."""
    n = seq.shape[0]
    cur_seq = seq.copy()
    cur_score = _fast_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    # Initial step size – modest fraction of the total ℓ¹‑norm.
    step = max(0.01, target_sum * 0.05)
    min_step = max(1e-8, target_sum * 1e-6)

    while time.monotonic() < deadline:
        conv = _conv_full_fft(cur_seq)
        km = int(conv.argmax())

        # Gradient of the maximal entry: g_i = 2·a_{km‑i} (if index valid)
        idx = np.arange(n)
        j = km - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur_seq)
        grad[mask] = 2.0 * cur_seq[j[mask]]

        cand = cur_seq - step * grad
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _simplex_projection(cand, target_sum)

        cand_score = _fast_score(cand)

        if cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[PGD] improved score {cand_score:.6f}")
        else:
            step *= 0.5
            step = max(step, min_step)

        # Gentle decay of the step size.
        step *= 0.9999
        step = max(step, min_step)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stochastic local polishing (step 4)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """Gaussian hill climbing that accepts only improving moves."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-8:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The total time budget is respected; the best sequence found before
    the deadline is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # tiny safety margin so that the caller can still read the result
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Choose a problem size – larger n allows a lower theoretical bound.
    # ------------------------------------------------------------------
    n = 1000                     # fits comfortably in memory, gives a tight lower bound
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise the sequence (warm‑start if a previous solution is present)
    # ------------------------------------------------------------------
    known_seq = globals().get("height_sequence_1", None)
    if known_seq is not None:
        known_seq = np.asarray(known_seq, dtype=np.float64)
        if known_seq.shape[0] != n:
            old_n = known_seq.shape[0]
            old_x = np.linspace(0.0, 1.0, old_n)
            new_x = np.linspace(0.0, 1.0, n)
            init_seq = np.interp(new_x, old_x, known_seq)
        else:
            init_seq = known_seq.copy()
    else:
        init_seq = rng.random(n) + 0.1   # ensure strictly positive start

    init_seq = np.clip(init_seq, 0.0, 1000.0)
    cur_sum = init_seq.sum()
    if cur_sum > 0:
        init_seq = init_seq * (target_sum / cur_sum)

    # ------------------------------------------------------------------
    # Split the remaining budget among the three phases
    # ------------------------------------------------------------------
    time_left = deadline - time.monotonic()
    if time_left <= 0:
        return [float(v) for v in init_seq]

    lp_deadline   = time.monotonic() + 0.30 * time_left
    pgd_deadline  = time.monotonic() + 0.50 * time_left
    final_deadline = deadline

    # ------------------------------------------------------------------
    # Phase 1 – Linear‑programming refinement
    # ------------------------------------------------------------------
    seq_lp, score_lp = _lp_refine(init_seq, target_sum, lp_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – Projected gradient descent
    # ------------------------------------------------------------------
    seq_pgd, score_pgd = _proj_grad_descent(seq_lp, target_sum, pgd_deadline)

    # ------------------------------------------------------------------
    # Phase 3 – Stochastic polishing
    # ------------------------------------------------------------------
    seq_final, score_final = _local_search(seq_pgd, final_deadline, step_initial=0.5)

    # Choose the best among the intermediate results (defensive)
    best_seq, best_score = seq_final, score_final
    for cand_seq, cand_score in [(seq_lp, score_lp), (seq_pgd, score_pgd)]:
        if cand_score < best_score:
            best_seq, best_score = cand_seq, cand_score

    print(f"[RESULT] best score {best_score:.6f} (n={n})")
    return [float(v) for v in best_seq]
