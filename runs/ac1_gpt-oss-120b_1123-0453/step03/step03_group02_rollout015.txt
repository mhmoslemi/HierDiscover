"""
Optimiser for the black‑box ``evaluate_sequence`` function.

The objective is

    score(f) = 2·n·max(f*f) / (sum(f)²)

with ``n = len(f)`` and ``f`` a non‑negative list.  The score is
scale‑invariant, therefore we keep the ℓ¹‑norm fixed to
``target = sqrt(2·n)`` and simply minimise the peak of the
auto‑convolution.  The algorithm consists of three stages:

1️⃣  **LP refinement**  (only for modest lengths – 200…600).  We solve the
    linear programme  

        maximise   Σ g_i
        s.t.      (f ∗ g)_k ≤ M   for all k,
                  g_i ≥ 0

    where ``M = max(f*f)``.  The solution is mixed back into the current
    vector, guaranteeing a non‑increasing peak.

2️⃣  **Projected sub‑gradient descent** on the ℓ¹‑simplex.  The sub‑gradient
    at the worst index ``k*`` is ``∂M/∂f_i = 2·f_{k*‑i}``.  After each
    step we project onto the simplex ``{x≥0, Σx = target}`` with an O(n)
    algorithm.  FFT‑based convolution supplies the worst index quickly.

3️⃣  **Stochastic polishing** – a simple Gaussian hill‑climber that
    perturbs the best vector found so far and accepts any improvement.

All phases respect the security checks performed inside ``evaluate_sequence``.
The routine stops a few hundred milliseconds before the global deadline
and returns the best sequence discovered.
"""

# ---------------------------------------------------------
# Simple ℓ¹‑simplex projection (sum = target, non‑negative)
# ---------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project ``v`` onto {x ≥ 0, Σx = target}."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:               # all entries become zero
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------
# FFT‑based convolution helpers
# ---------------------------------------------------------
def _fft_conv(f: np.ndarray) -> np.ndarray:
    """Full auto‑convolution of ``f`` using a power‑of‑two FFT."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    return conv


def _max_conv(f: np.ndarray) -> tuple[float, int]:
    """Return (maximum value, index) of the full auto‑convolution."""
    conv = _fft_conv(f)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ---------------------------------------------------------
# Linear‑programming refinement (small n)
# ---------------------------------------------------------
def _lp_refine(f: np.ndarray, iters: int, mix: float) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinements for a given vector ``f``.
    Returns the refined vector and its score.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, _score(f)  # CVX not available → nothing to do

    n = f.shape[0]
    target = math.sqrt(2.0 * n)

    A = _conv_matrix(f)                        # (2n‑1)×n matrix
    best_f = f.copy()
    best_sc = _score(f)

    for _ in range(iters):
        M = float(np.max(np.convolve(f, f)))   # current worst entry
        # Linear programme: maximise Σ g_i  s.t.  A·g ≤ M , g ≥ 0
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=5000, abstol=1e-9, reltol=1e-7)
        except Exception:
            prob.solve(solver=cp.SCS, warm_start=True,
                       max_iters=15000, verbose=False)

        if g.value is None:
            break
        g0 = np.maximum(g.value, 0.0)
        s = np.sum(g0)
        if s == 0:
            break
        g0 = g0 * (target / s)                # enforce the ℓ¹‑norm
        f_new = (1.0 - mix) * f + mix * g0
        f_new = np.clip(f_new, 0.0, 1000.0)
        sc = _score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
        f = f_new
    return best_f, best_sc


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2·n‑1) × n matrix A with (A@g)[k] = (f*g)[k]."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


# ---------------------------------------------------------
# Projected sub‑gradient descent on the ℓ¹‑simplex
# ---------------------------------------------------------
def _subgradient_descent(
    f0: np.ndarray,
    target: float,
    deadline: float,
    max_iters: int = 3000,
    eta0: float = 0.7,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient optimisation of  M(f)=max_k (f*f)_k  on the ℓ¹‑simplex.
    Returns the best vector seen and its associated max‑convolution value.
    """
    n = f0.shape[0]
    f = f0.copy()
    best_f = f.copy()
    best_M, _ = _max_conv(f)
    best_val = best_M

    eta = eta0
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        M, k = _max_conv(f)

        if M < best_val:
            best_val = M
            best_f = f.copy()

        # ----- sub‑gradient at the worst index k -----
        g = np.zeros_like(f)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        if low <= high:
            idx = np.arange(low, high + 1)
            g[idx] = 2.0 * f[k - idx]

        # ----- gradient step + projection -----
        f_next = _project_simplex(f - eta * g, target)

        M_next, _ = _max_conv(f_next)

        if M_next <= M:                         # accept improvement / keep step
            f = f_next
            eta = min(eta / decay, eta0)        # gently increase step up to η₀
        else:                                    # no improvement → shrink step
            eta *= 0.5

    return best_f, best_val


# ---------------------------------------------------------
# Stochastic local search (Gaussian perturbations)
# ---------------------------------------------------------
def _local_search(
    start: np.ndarray,
    deadline: float,
    init_step: float = 0.5,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    best = start.copy()
    best_sc = _score(best)
    step = init_step
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑normalise to the ℓ¹‑norm required by the evaluation function
        target = math.sqrt(2.0 * cand.shape[0])
        cand = _project_simplex(cand, target)
        sc = _score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
        step *= decay
        if step < 1e-7:
            step = init_step
    return best, best_sc
