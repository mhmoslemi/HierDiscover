def _spectral_flatten(seq, target_sum, max_iter, deadline, rng):
    n = len(seq)
    L = 2 * n
    seq_copy = seq.copy()
    best_local = seq_copy
    best_val = _score(seq_copy)
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        # FFT
        X = np.fft.rfft(seq_copy, n=L)
        # compute magnitude RMS
        M = np.sqrt(np.mean(np.abs(X)**2))
        # adjust magnitude
        X_new = M * np.exp(1j * np.angle(X))
        # IFFT
        seq_new_full = np.fft.irfft(X_new, n=L)
        # take first n entries, clip
        seq_new = np.clip(seq_new_full[:n], 0.0, 1000.0)
        # project to simplex
        seq_new = _project_simplex(seq_new, target_sum)
        # evaluate
        val = _score(seq_new)
        if val < best_val:
            best_val = val
            best_local = seq_new.copy()
        # acceptance step: maybe keep if better else occasional random move
        # simple: keep new regardless; maybe keep current if new not improvement to avoid oscillation.
        if val <= best_val:
            seq_copy = seq_new
        else:
            # maybe a probabilistic acceptance? or keep old.
            # we could revert to previous best_current.
            seq_copy = seq_new  # can still go; may converge anyway
    return best_local, best_val

"""
Propose a low–evaluation step sequence.

The search proceeds in three stages:

1️⃣  **Seeding** – load the previously discovered sequence (if any) and
    initialise a collection of start vectors for larger problems by
    up‑scaling, randomisation and a simple geometric decay.

2️⃣  **Spectral flattening + projected sub‑gradient descent** –
    for each chosen length `n` we first apply an alternating‑projection
    scheme that forces all Fourier coefficients to have equal magnitude.
    This tends to decorrelate the vector from its reverse and reduces the
    peak of the auto‑convolution.  The result is refined with a fast
    sub‑gradient descent on the ℓ¹‑simplex (the objective is
    `M(f)=max_k (f*f)_k`).  The algorithm adapts the step size and projects
    after every update using an O(n) simplex projection.

3️⃣  **Stochastic polishing** – a final Gaussian hill‑climbing phase
    cleans up the best vector found.

All phases respect the security checks performed inside `evaluate_sequence`.
The routine stops a fraction of a second before the global time budget
(and returns the best sequence found so far) to avoid a timeout penalty.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the ℓ¹‑simplex {x ≥ 0, Σx = target}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project `v` onto {x ≥ 0, Σx = target}.  Returns a new array."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]

    if rho_arr.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: max entry of full auto‑convolution via FFT
# ----------------------------------------------------------------------
def _fft_conv_max(f: np.ndarray):
    """
    Return (max_value, argmax_index) of the full convolution f*f.
    Uses a power‑of‑two FFT size for speed.
    """
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Wrapper around the external evaluator (used only for the final check)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Call the provided `evaluate_sequence`.  Returns a float."""
    # The harness already provides `evaluate_sequence`.
    return float(evaluate_sequence(seq.tolist()))  # type: ignore[name-defined]


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the ℓ¹‑simplex
# ----------------------------------------------------------------------
def _gradient_descent(
    f: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 3000,
    eta0: float = 0.8,
    decay: float = 0.9997,
    min_step: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for M(f)=max_k (f*f)_k.
    Returns the best vector seen and its max‑convolution value.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_M, _ = _fft_conv_max(f)

    eta = eta0
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        M, k = _fft_conv_max(f)

        if M < best_M:
            best_M = M
            best_f = f.copy()

        # sub‑gradient at the worst index k
        g = np.zeros_like(f)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        if low <= high:
            i_range = np.arange(low, high + 1, dtype=int)
            g[i_range] = 2.0 * f[k - i_range]

        # tentative step and projection
        f_candidate = _project_simplex(f - eta * g, target_sum)
        M_candidate, _ = _fft_conv_max(f_candidate)

        if M_candidate < M:                       # improvement → accept
            f = f_candidate
            eta = min(eta / decay, eta0)          # gently increase again
        else:                                      # no improvement → shrink step
            eta *= 0.5

        # avoid vanishing step size
        if eta < min_step:
            eta = eta0
        eta *= decay
    return best_f, best_M


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (Gaussian perturbations)
# ----------------------------------------------------------------------
def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.5,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fft_conv_max(best)[0]
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # renormalise to the ℓ¹‑simplex of the correct radius
        target = math.sqrt(2.0 * cand.shape[0])
        cand = _project_simplex(cand, target)

        cand_score = _fft_conv_max(cand)[0]
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        step *= decay
        if step < 1e-7:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Spectral flattening – equalise all Fourier magnitudes
# ----------------------------------------------------------------------
def _spectral_flatten(
    seq: np.ndarray,
    target_sum: float,
    max_iter: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Alternating projection:
        1. FFT → replace all magnitudes by a constant √(∑x_i²)
        2. IFFT → clip, project onto simplex
    Repeatedly reduces the peak of the auto‑convolution.
    """
    n = seq.shape[0]
    L = 2 * n                         # zero‑padded length for convolution
    cur = seq.copy()
    best = cur.copy()
    best_val = _fft_conv_max(cur)[0]

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        # occasional jitter to escape shallow plateaus
        if it % 30 == 0 and it > 0:
            cur += rng.normal(scale=0.001 * target_sum, size=n)
            cur = np.clip(cur, 0.0, 1000.0)
            cur = _project_simplex(cur, target_sum)

        # FFT of the current vector (zero‑padded to length L)
        X = np.fft.rfft(cur, n=L)

        # constant magnitude = √(∑ cur_i²)  (Parseval guarantees this is admissible)
        mag = math.sqrt(np.sum(cur * cur))

        # impose the constant magnitude, keep the original phases
        X_new = mag * np.exp(1j * np.angle(X))

        # inverse transform, keep only the first n entries
        full = np.fft.irfft(X_new, n=L)
        nxt = np.clip(full[:n], 0.0, 1000.0)
        nxt = _project_simplex(nxt, target_sum)

        val = _fft_conv_max(nxt)[0]
        if val < best_val:
            best_val = val
            best = nxt.copy()

        cur = nxt

    return best, best_val


# ----------------------------------------------------------------------
# Utility: up‑sample / interpolate an existing sequence to a larger size
# ----------------------------------------------------------------------
def _expand_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of `seq` onto `new_n` points.
    The output is not normalised – the caller should project onto the
    ℓ¹‑simplex with the appropriate target sum.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    expanded = np.interp(new_x, old_x, seq)
    return expanded


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget (seconds).  The function returns a solution
        a few hundred milliseconds before the limit.

    Returns
    -------
    list[float]
        The best sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a small safety margin (0.2 s) for the harness
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------------------
    # 0️⃣  Load the previously discovered construction (if present)
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n0 = seq.shape[0]
        target0 = math.sqrt(2.0 * n0)
        seq = seq / np.sum(seq) * target0
        best_seq = seq.copy()
        best_val = _fft_conv_max(seq)[0]
        print(f"[init] loaded height_sequence_1 (n={n0}) → max={best_val:.6f}")

    # -------------------------------------------------------------
    # 1️⃣  Main exploration over a ladder of lengths
    # -------------------------------------------------------------
    lengths = [500, 800, 1000, 1500, 2000, 2500, 3000, 4000, 5000, 6000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ---- build a few diverse starting vectors ----------------
        starts = []

        # a) stretch the current best (if any) to the new length
        if best_seq is not None:
            stretched = _expand_sequence(best_seq, n)
            stretched = _project_simplex(stretched, target)
            starts.append(stretched)

        # b) a few purely random starts
        for _ in range(2):
            rnd = rng.random(n) + 0.01
            rnd = rnd / np.sum(rnd) * target
            starts.append(rnd)

        # c) a simple geometric decay (might give a good asymmetry)
        r = 0.97
        geom = np.array([r ** i for i in range(n)], dtype=np.float64)
        geom = geom / np.sum(geom) * target
        starts.append(geom)

        # ---------------------------------------------------------
        # 2️⃣  Spectral flattening → projected sub‑gradient descent
        # ---------------------------------------------------------
        for sid, f0 in enumerate(starts):
            if time.monotonic() >= deadline:
                break

            # spectral flattening (fast, a few hundred iterations)
            sf_seq, sf_val = _spectral_flatten(
                f0,
                target,
                max_iter=200,
                deadline=deadline,
                rng=rng,
            )
            if sf_val < best_val:
                best_val = sf_val
                best_seq = sf_seq.copy()
                print(f"[sf] n={n} start={sid} → max={sf_val:.6f}")

            # gradient descent refinement (more intensive)
            gd_seq, gd_val = _gradient_descent(
                sf_seq,
                target_sum=target,
                deadline=deadline,
                max_iters=3000,
                eta0=0.8,
                decay=0.9997,
                min_step=1e-7,
            )
            if gd_val < best_val:
                best_val = gd_val
                best_seq = gd_seq.copy()
                print(f"[gd] n={n} start={sid} → max={gd_val:.6f}")

            # early exit if we already beat the target
            if best_val < 1.503:
                print("[done] target achieved during main loop")
                return [float(x) for x in best_seq.tolist()]

    # -------------------------------------------------------------
    # 3️⃣  Final polishing with stochastic local search
    # -------------------------------------------------------------
    if best_seq is None:
        # fallback: a modest uniform vector
        n = 800
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    refined_seq, refined_val = _local_search(best_seq, deadline, step_initial=0.6)
    if refined_val < best_val:
        best_seq = refined_seq
        best_val = refined_val
        print(f"[final] polished → max={best_val:.6f}")

    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq.tolist()]
