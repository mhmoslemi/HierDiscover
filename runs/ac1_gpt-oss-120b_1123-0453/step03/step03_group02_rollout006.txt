"""
Propose a low‑score step‐function sequence.

Algorithm
---------
1. **Initialisation** – start from any previously found sequence
   (`height_sequence_1`) and normalise it to the ℓ¹‑norm `√(2·n)`.
   The normalisation makes the evaluation equal to the maximal entry of
   the auto‑convolution (`max(conv)`), which is much cheaper to compute.

2. **Projected aggregated sub‑gradient descent** – for a given length `n`
   we minimise `max(conv)` on the simplex `Σ a_i = √(2·n) , a_i ≥ 0`.
   At each iteration the full auto‑convolution is computed via FFT,
   all (near) maximal indices are collected and the summed sub‑gradient
   `g_i = 2·∑_{k∈K} a_{k‑i}` (with `K` the set of maximal indices) is
   used.  A step opposite to the gradient is taken and the vector is
   projected back onto the simplex.

3. **Randomised refinements** – after the deterministic phase a cheap
   stochastic local search (pairwise mass exchanges with a simulated‑
   annealing acceptance rule) clears remaining plateaus.

4. **Length sweep** – the whole pipeline is executed for several
   lengths (500 … 3000).  The best sequence found overall is returned.

The code respects the security checks inside `evaluate_sequence`; it
only calls this function once at the very end for a final verification
and otherwise works with an internal fast evaluation based on FFT.
"""

import time
import math
import numpy as np

# ---------------------------------------------------------
# Helper: projection onto the ℓ¹‑simplex {x ≥ 0, Σx = target}
# ---------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex Σx = target, x≥0."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending order
    cssv = np.cumsum(u) - target
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]

    if rho_arr.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------
# FFT‑based full auto‑convolution (fast for n up to few‑thousands)
# ---------------------------------------------------------
def _fft_conv_full(a: np.ndarray) -> np.ndarray:
    """Return the full (2·n‑1) auto‑convolution of `a` using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m).bit_length())          # next power of two
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:m]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of `a`."""
    return float(_fft_conv_full(a).max())


# ---------------------------------------------------------
# Aggregated sub‑gradient (all indices attaining the maximal value)
# ---------------------------------------------------------
def _aggregated_gradient(a: np.ndarray, max_val: float, eps_rel: float = 1e-10) -> np.ndarray:
    """Compute the summed sub‑gradient of `max(conv)` at `a`."""
    conv = _fft_conv_full(a)
    eps = max(eps_rel * max_val, 1e-12)
    ks = np.where(conv >= max_val - eps)[0]

    n = a.shape[0]
    g = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        g[idx] += 2.0 * a[k - idx]          # sub‑gradient contribution
    return g


# ---------------------------------------------------------
# Projected aggregated sub‑gradient descent
# ---------------------------------------------------------
def _gradient_descent_agg(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 3000,
    step0: float = 0.5,
    min_step: float = 1e-7,
    eps_rel: float = 1e-10,
) -> tuple[np.ndarray, float]:
    """
    Minimise `max(conv(a))` on the simplex by aggregated sub‑gradient descent.
    Returns the best vector encountered and its maximal convolution value.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _max_conv(a)

    step = step0
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        conv = _fft_conv_full(a)
        cur_max = float(conv.max())

        # keep the best ever seen
        if cur_max < best_val:
            best_val = cur_max
            best = a.copy()

        # aggregated sub‑gradient at all (near) maximal indices
        g = _aggregated_gradient(a, cur_max, eps_rel)

        # trial move opposite to the gradient and project
        cand = _project_simplex(a - step * g, target_sum)
        cand_max = _max_conv(cand)

        if cand_max < cur_max:                     # improvement → accept
            a = cand
            step = min(step * 1.02, step0)         # gently increase step
        else:                                        # no improvement → shrink
            step *= 0.5

        if step < min_step:
            step = step0

    # final check: maybe the very last `a` is better than `best`
    final_val = _max_conv(a)
    if final_val < best_val:
        best_val = final_val
        best = a.copy()

    return best, best_val


# ---------------------------------------------------------
# Simple stochastic local search – pairwise mass exchange
# ---------------------------------------------------------
def _local_exchange_search(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 50000,
    init_temp: float = 1.0,
    cooling: float = 0.99995,
) -> tuple[np.ndarray, float]:
    """Swap small amounts of mass between two coordinates (simulated annealing)."""
    cur = start.copy()
    cur_val = _max_conv(cur)
    best = cur.copy()
    best_val = cur_val
    temp = init_temp

    n = cur.shape[0]
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if cur[i] == 0.0:
            continue

        # move a random fraction (≤10 %) of the mass at i to j
        frac = rng.random() * 0.1
        delta = cur[i] * frac

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # no need to re‑normalise – the sum stays constant

        cand_val = _max_conv(cand)

        # Metropolis acceptance
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / max(temp, 1e-12)):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()

        temp *= cooling
        if temp < 1e-8:          # restart temperature occasionally
            temp = init_temp

    return best, best_val

                a = rng.exponential(scale=1.0, size=n) + 0.1
            else:                                             # triangular bump
                peak = rng.integers(0, n)
                left = np.linspace(0.1, 1.0, peak + 1)
                right = np.linspace(1.0, 0.1, n - peak)[1:] if peak < n - 1 else np.array([])
                a = np.concatenate([left, right])
            a = a / a.sum() * target_sum
            starts.append(a)

        # ----- run deterministic optimisation for each start ----------
        for sid, a0 in enumerate(starts):
            if time.monotonic() >= deadline:
                break

            # allocate a slice of the remaining budget for this run
            sub_deadline = min(deadline,
                               time.monotonic() + (deadline - time.monotonic()) / 2)

            a_opt, score_opt = _gradient_descent_agg(
                a0,
                target_sum=target_sum,
                deadline=sub_deadline,
                max_iters=3000,
                step0=0.8,
                min_step=1e-6,
            )
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                best_n = n
                print(f"[GD] n={n} start={sid} → new best score={best_score:.6f}")

            if best_score < 1.503:
                print("[goal] reached target bound, exiting early.")
                break

        # ----- stochastic polishing on the current best of this length
        if best_seq is not None and best_seq.shape[0] == n and time.monotonic() < deadline:
            polished, polished_score = _local_exchange_search(
                best_seq,
                target_sum,
                deadline,
                rng,
                max_iters=20000,
            )
            if polished_score < best_score:
                best_score = polished_score
                best_seq = polished.copy()
                best_n = n
                print(f"[local] n={n} polished → score={best_score:.6f}")

        if best_score < 1.503:
            break

    # -----------------------------------------------------------------
    # 2️⃣  Fallback in the unlikely event nothing was discovered
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = _project_simplex(best_seq, target_sum)
        best_score = _max_conv(best_seq)

    # -----------------------------------------------------------------
    # Final verification using the official evaluator (for the harness)
    # -----------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] length={best_seq.shape[0]}  score={final_score:.6f}")

    return [float(x) for x in best_seq]
