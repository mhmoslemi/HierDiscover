"""
Propose a low‑value non‑negative step sequence.

The algorithm:
1️⃣  Pick a few candidate lengths (≈500‑1000). For each length we
    generate several simple initial shapes (uniform, triangular,
    exponential with several decay factors and, if available, the
    historic best sequence interpolated to the current size).

2️⃣  Pick the best initial shape (by the exact ``evaluate_sequence`` score)
    and improve it with a projected gradient descent that minimises a
    smooth approximation of the maximum auto‑convolution entry.
    The smooth objective is the soft‑max (log‑sum‑exp) of the convolution,
    whose gradient can be written analytically as a convolution of the
    current sequence with the soft‑max weights.  An Adam‑style update
    together with a projection onto the simplex (non‑negative entries
    with the prescribed ℓ¹‑norm) drives the sequence to lower
    max‑convolution values.

3️⃣  After the gradient phase we run a cheap stochastic hill‑climb
    (Gaussian perturbations) for final polishing.

All operations respect the time budget (default 1000 s); the routine
returns a plain ``list[float]`` that can be fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (full linear convolution) via FFT
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of ``seq`` with itself."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact evaluation (same as ``evaluate_sequence`` but faster)
# ----------------------------------------------------------------------


def _fast_score(seq: np.ndarray) -> float:
    """Exact analogue of ``evaluate_sequence`` for a non‑negative array."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Linear interpolation of a known sequence to a new length
# ----------------------------------------------------------------------


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to ``new_n`` by linear interpolation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return np.clip(new_seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Simple stochastic polishing (Gaussian hill‑climb)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """Gaussian perturbation hill‑climb until ``deadline``."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # decay / occasional reset of step size
        step *= 0.9999
        if step < 1e-8:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max of the auto‑convolution
# ----------------------------------------------------------------------


def _gradient_descent_opt(seq: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent that minimises the soft‑max
    of ``seq*seq``.  The temperature (soft‑max temperature) is linearly
    annealed from ``tau_start`` to ``tau_end``.
    """
    n = seq.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    # base learning rate – tuned a bit for the typical magnitude of f
    lr = 0.05 * target_sum / max(1.0, n)

    # momentum buffers
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)

    # soft‑max temperature schedule
    tau_start = 1.0
    tau_end = 0.001

    start_time = time.monotonic()
    total_time = max(1e-6, deadline - start_time)

    iteration = 0
    best_seq = seq.copy()
    best_score = _fast_score(seq)

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # temperature (linear annealing)
        # ------------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # ------------------------------------------------------------------
        # compute convolution and soft‑max weights
        # ------------------------------------------------------------------
        conv = _conv_full_fft(seq)                     # length 2n‑1
        # stabilise the exponentials
        conv_max = conv.max()
        w = np.exp((conv - conv_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum                               # now a probability vector

        # ------------------------------------------------------------------
        # gradient of the (smooth) max‑convolution:
        #   ∇_f L = 2 * (w * f)   where * denotes linear convolution
        # ------------------------------------------------------------------
        grad = 2.0 * np.convolve(w, seq)[:n]            # shape (n,)

        # ------------------------------------------------------------------
        # Adam update
        # ------------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        seq = seq - step

        # ------------------------------------------------------------------
        # projection onto the feasible set (non‑negative, ℓ¹ = target_sum)
        # ------------------------------------------------------------------
        seq = np.clip(seq, 0.0, 1000.0)
        cur_sum = seq.sum()
        if cur_sum > 0.0:
            seq = seq * (target_sum / cur_sum)

        # ------------------------------------------------------------------
        # occasional exact evaluation (every 200 iterations)
        # ------------------------------------------------------------------
        if iteration % 200 == 0:
            cur_score = _fast_score(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[GD] n={n:4d} iter={iteration:7d} score={cur_score:.6f} τ={tau:.5f}")

    # final exact score
    final_score = _fast_score(seq)
    if final_score < best_score:
        best_score = final_score
        best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    score.  The routine respects the given time budget (default 1000 s) and
    returns a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    # leave a tiny safety margin for the caller
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.3)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Candidate lengths – we explore a few values, larger sizes give
    # a larger search space but also more opportunities to lower the peak.
    # ------------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000]

    # if a previous solution exists, keep it handy for interpolation
    known_seq = globals().get("height_sequence_1", None)
    if known_seq is not None:
        known_seq = np.asarray(known_seq, dtype=np.float64)

    # allocate roughly 65 % of the remaining budget to the gradient phase,
    # the rest to final polishing
    remaining = deadline - time.monotonic()
    gradient_budget = remaining * 0.65
    per_len_budget = gradient_budget / max(1, len(lengths))

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Build a small pool of deterministic initial shapes
        # ------------------------------------------------------------------
        inits = []

        # 1) Uniform (all entries equal)
        uniform = np.full(n, target_sum / n, dtype=np.float64)
        inits.append(uniform)

        # 2) Linear decreasing (triangular, starts high, ends low)
        tri = np.arange(n, 0, -1, dtype=np.float64)
        tri = tri / tri.sum() * target_sum
        inits.append(tri)

        # 3) Exponential decays with several factors
        for r in (0.99, 0.97, 0.95, 0.93, 0.90, 0.85):
            exp_seq = r ** np.arange(n, dtype=np.float64)
            exp_seq = exp_seq * (target_sum / exp_seq.sum())
            inits.append(exp_seq)

        # 4) Interpolated historic best if available
        if known_seq is not None:
            if known_seq.shape[0] == n:
                cand = known_seq.copy()
            else:
                cand = _interpolate(known_seq, n)
            cand = np.clip(cand, 0.0, 1000.0)
            s = cand.sum()
            if s > 0:
                cand = cand * (target_sum / s)
            inits.append(cand)

        # ------------------------------------------------------------------
        # Choose the best deterministic starter (by exact score)
        # ------------------------------------------------------------------
        seq = None
        score = float("inf")
        for cand in inits:
            cand_score = _fast_score(cand)
            if cand_score < score:
                seq = cand.copy()
                score = cand_score

        # ------------------------------------------------------------------
        # Gradient descent optimisation (projected Adam)
        # ------------------------------------------------------------------
        gd_deadline = time.monotonic() + per_len_budget
        seq_opt, score_opt = _gradient_descent_opt(seq, target_sum, gd_deadline, rng)

        # ------------------------------------------------------------------
        # Keep the globally best candidate
        # ------------------------------------------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[global] new overall best: score={best_score:.6f} (n={n})")

        # Early exit if target already reached
        if best_score < 1.503:
            print("[global] target achieved – exiting early.")
            break

    # ------------------------------------------------------------------
    # Final stochastic polishing (Gaussian hill‑climb)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined_seq, refined_score = _local_search(best_seq, deadline, step_initial=0.5)
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"[final] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = best_seq * (math.sqrt(2.0 * n) / best_seq.sum())

    # Return a plain Python list
    return [float(v) for v in best_seq]
