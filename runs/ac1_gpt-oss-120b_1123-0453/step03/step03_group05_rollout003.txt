"""
Propose a low‑value step sequence.

Algorithm
---------
1. **Initialisation** – keep the best sequence found in previous runs
   (global ``height_sequence_1``) if it exists.

2. **Fast random sampling** – generate a few random simplex‑vectors of
   various lengths; keep the best of them as a cheap baseline.

3. **Adam‑based smooth‑max optimisation** – minimise a smooth approximation
   of the worst convolution entry using an Adam optimiser.
   * Convolution and its gradient are evaluated with FFT (O(n log n)).
   * β is increased geometrically (2 → 128) to tighten the smooth‑max.
   * After each Adam step the vector is projected onto the ℓ¹‑simplex
     (by simple scaling – sufficient because entries stay far from the 1000‑cap).

4. **Local Gaussian refinement** – a final stochastic hill‑climbing phase
   (the same routine used previously) cleans up any remaining excess.

The routine repeatedly restarts the Adam optimiser with different lengths
(`200 ≤ n ≤ 1500`) until the time budget (≈ 1000 s) is exhausted.
The best sequence encountered is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}.
    Duchi et al. (2008) implementation.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` (no type checks)."""
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = float(conv.max())
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """Simple Gaussian hill‑climbing with a fixed ℓ¹‑norm."""
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best = start_seq.copy()
    best_score = _fast_score(best)

    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        # cheap ℓ¹‑projection by scaling (keeps non‑negativity)
        s = cand.sum()
        if s > 0:
            cand = cand * (target_sum / s)
        else:
            continue
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score

# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: n + m - 1]


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Compute a smooth approximation φβ(a) of max(a∗a) and its gradient.
    Returns (phi, grad, max_conv) where ``grad`` has the same length as ``a``.
    """
    conv = _convolve_fft(a, a)               # length 2n‑1
    max_conv = conv.max()
    shifted = conv - max_conv                # stability shift
    exp_terms = np.exp(beta * shifted)
    sum_exp = exp_terms.sum()
    # soft‑max probabilities
    p = exp_terms / sum_exp
    # gradient = 2·conv(p, a) (first n entries)
    grad_full = _convolve_fft(p, a)          # length 3n‑2
    grad = 2.0 * grad_full[: a.shape[0]]
    phi = max_conv + (np.log(sum_exp) / beta)
    return phi, grad, max_conv

def _adam_optimize(n: int,
                   rng: np.random.Generator,
                   deadline: float,
                   start_vec: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max objective for a given length ``n``.
    Returns the best sequence and its (fast) score.
    """
    target_sum = math.sqrt(2.0 * n)

    # initial vector (either supplied or random)
    if start_vec is not None and start_vec.shape[0] == n:
        a = start_vec.copy()
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = rng.exponential(size=n)
            a = a / a.sum() * target_sum
    else:
        a = rng.exponential(size=n) + 0.01
        a = a / a.sum() * target_sum

    best_seq = a.copy()
    cur_conv = _convolve_fft(a, a)
    best_score = 2.0 * n * cur_conv.max() / (target_sum * target_sum)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    pow1, pow2 = 1.0, 1.0
    t_iter = 0

    # geometric schedule for the smooth‑max temperature
    beta_schedule = [2, 4, 8, 16, 32, 64, 128]

    # base learning‑rate factor (scaled by target_sum / β)
    base_lr = 0.5

    for beta in beta_schedule:
        # a modest number of inner steps; break early if out of time
        for _ in range(350):
            if time.monotonic() >= deadline:
                break

            t_iter += 1
            phi, grad, _ = _smooth_max_and_grad(a, beta)

            # Adam bias‑correction factors
            pow1 *= beta1
            pow2 *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - pow1)
            v_hat = v / (1.0 - pow2)

            # learning rate decreases with β (helps stability for large β)
            lr = (base_lr / beta) * target_sum
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # enforce constraints (non‑negative, ℓ¹‑norm, and clipping)
            a = np.clip(a, 0.0, 1000.0)
            s = a.sum()
            if s > 0:
                a = a * (target_sum / s)
            else:
                # re‑initialise if we ever collapse to zero (unlikely)
                a = rng.exponential(size=n)
                a = a / a.sum() * target_sum

            # compute cheap score from the new convolution max
            cur_conv = _convolve_fft(a, a)
            cur_max = cur_conv.max()
            cur_score = 2.0 * n * cur_max / (target_sum * target_sum)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Adam] n={n} β={beta} step={t_iter} score={cur_score:.6f}")

            # early exit if we have already broken the target bound
            if best_score < 1.503:
                return best_seq, best_score

        if time.monotonic() >= deadline:
            break

    # final Gaussian hill‑climbing refinement
    refined_seq, refined_sc = _local_search(best_seq, deadline, step_initial=0.3)
    if refined_sc < best_score:
        best_seq, best_score = refined_seq, refined_sc
        print(f"[Adam] post‑local improvement: score={best_score:.6f}")

    return best_seq, best_score

    deadline expires).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. Seed from the previous best if it exists
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seq0 = np.clip(seq0, 0.0, 1000.0)
        n0 = seq0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        if seq0.sum() > 0:
            seq0 = seq0 * (target0 / seq0.sum())
            sc0 = _fast_score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using global height_sequence_1 (n={n0}) score={sc0:.6f}")

    # --------------------------------------------------------------
    # 1. Quick random sampling (cheap baseline)
    # --------------------------------------------------------------
    for n in (200, 300, 400, 500, 600, 800):
        if time.monotonic() >= deadline:
            break
        a = rng.exponential(size=n) + 0.01
        a = a / a.sum() * math.sqrt(2.0 * n)
        sc = _fast_score(a)
        if sc < best_score:
            best_seq = a.copy()
            best_score = sc
            print(f"[rand] n={n} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2. Main optimisation loop (Adam + local refinement)
    # --------------------------------------------------------------
    # a set of candidate lengths – larger n gives a richer search space
    length_pool = [300, 400, 500, 600, 800, 1000, 1200, 1500]
    # keep trying until time runs out or we beat the target bound
    while time.monotonic() < deadline:
        # pick a length (biased towards larger sizes when time permits)
        n = int(rng.choice(length_pool))
        # optionally start from the current global best if its size matches
        start_vec = best_seq if (best_seq is not None and best_seq.shape[0] == n) else None
        seq, sc = _adam_optimize(n, rng, deadline, start_vec=start_vec)
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc
            print(f"[global] new best n={n} score={sc:.6f}")
            if best_score < 1.503:
                break

    # --------------------------------------------------------------
    # 3. Final local Gaussian refinement (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined_seq, refined_sc = _local_search(best_seq, deadline, step_initial=0.2)
        if refined_sc < best_score:
            best_seq = refined_seq
            best_score = refined_sc
            print(f"[final‑local] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Fallback deterministic uniform vector (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[final] best score={best_score:.6f} length={len(best_seq)}")
    return [float(x) for x in best_seq]
