"""
Propose a step sequence with a low max‑auto‑convolution.

The algorithm works on a fixed L1‑norm (√(2 n)) so that the
evaluation of a candidate equals the maximal entry of its
auto‑convolution.  For each chosen length n we perform several
random restarts.  Every restart runs a *projected sub‑gradient*
method that directly minimises the quantity

        M(a) = max_k (a * a)[k]

subject to a ≥ 0 and Σ a = √(2 n).  The sub‑gradient for a given
active shift k is (a shifted by k) scaled by 2; we use the
average over all shifts whose convolution value is within a tiny
ε of the current maximum.  After a gradient step the vector is
projected onto the simplex (non‑negative entries with the required
sum) using the O(n log n) algorithm of Duchi et al.  Occasionally,
if no improvement is seen for a number of iterations, a tiny random
perturbation is added to escape plateaus.

Convolutions are evaluated with an FFT‑based routine (O(n log n))
so that millions of iterations are feasible within the 1000 s
budget.  The best sequence found (across all lengths and restarts)
is returned as a plain Python list.  If a previous construction is
available via the global variable ``height_sequence_1`` it is used as
one of the initial points.
"""

import time
import math

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi, Shalev‑Shwartz, Singer & Chandra
    (ICML 2008).  Works for any s ≥ 0.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the largest ρ satisfying u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _autocorr_fft(a: np.ndarray) -> np.ndarray:
    """
    Returns the (full) auto‑convolution a * a using a real FFT.
    Length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    # Next power of two ≥ 2·n‑1 for an efficient FFT
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)
    return conv[: 2 * n - 1]


def _subgradient(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution objective.
    All shifts whose value is within eps of the maximum are taken
    and their individual gradients are averaged.
    """
    n = a.shape[0]
    M = conv.max()
    # active set (indices attaining the max up to tolerance)
    act = np.where(conv >= M - eps)[0]
    if act.size == 0:
        # should never happen, but fall back to the single maximiser
        act = [np.argmax(conv)]

    grad = np.zeros_like(a)
    idxs = np.arange(n)
    for k in act:
        # indices i for which a_i contributes to (a*a)[k] as a_i * a_{k-i}
        src = k - idxs
        mask = (src >= 0) & (src < n)
        grad[mask] += 2.0 * a[src[mask]]
    grad /= float(act.size)
    return grad

    start_time = time.time()
    # keep a small margin for the final return
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction is present, keep it handy.
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        prev = np.asarray(prev, dtype=np.float64)
    else:
        prev = None

    # Lengths to explore – larger n gives a tighter theoretical bound.
    lengths = [500, 600, 800, 1000]
    restarts_per_n = 4          # independent random starts for each length
    max_iters = 4000            # sub‑gradient steps per restart

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)          # normalisation used by the evaluator

        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            # --------------------------------------------------------------
            # Initialise a feasible vector.
            if restart == 0 and prev is not None and prev.size == n:
                a = prev.copy()
                a = np.maximum(a, 0.0)
                a = a / a.sum() * target_sum
                print(f"[init] using previous construction (n={n})")
            else:
                # Dirichlet (Gamma) draw – equivalent to a uniform point on the simplex
                a = rng.gamma(1.0, 1.0, size=n)
                a = a / a.sum() * target_sum

            # Initial score (max‑convolution; thanks to the normalisation this equals the
            # evaluator’s output)
            conv = _autocorr_fft(a)
            cur_M = conv.max()
            if cur_M < best_score:
                best_score = cur_M
                best_seq = a.copy()
                print(f"[init] n={n} restart={restart} score={cur_M:.6f}")

            # --------------------------------------------------------------
            # Main optimisation loop – projected sub‑gradient descent.
            step0 = target_sum * 0.1   # initial step‑size scale
            no_improve = 0

            for it in range(max_iters):
                if time.time() >= deadline:
                    break

                # Compute current max‑convolution and a sub‑gradient.
                conv = _autocorr_fft(a)
                M = conv.max()
                grad = _subgradient(a, conv)

                # Diminishing step size (standard sub‑gradient schedule).
                eta = step0 / math.sqrt(it + 1.0)

                # Gradient step + projection onto the simplex.
                y = a - eta * grad
                a_new = _project_simplex(y, target_sum)

                # Evaluate the candidate.
                conv_new = _autocorr_fft(a_new)
                M_new = conv_new.max()

                if M_new < M - 1e-12:
                    # Successful improvement.
                    a = a_new
                    cur_M = M_new
                    no_improve = 0
                    if M_new < best_score:
                        best_score = M_new
                        best_seq = a.copy()
                        print(f"[improve] n={n} restart={restart} it={it} score={M_new:.6f}")
                else:
                    no_improve += 1

                # If we have stalled for a while, inject a tiny random jiggle.
                if no_improve >= 30:
                    perturb = rng.normal(scale=0.02 * target_sum, size=n)
                    a = np.clip(a + perturb, 0.0, None)
                    a = a / a.sum() * target_sum
                    no_improve = 0
                    # check after the perturbation
                    conv = _autocorr_fft(a)
                    M = conv.max()
                    if M < best_score:
                        best_score = M
                        best_seq = a.copy()
                        print(f"[perturb] n={n} restart={restart} it={it} score={M:.6f}")

                # Early exit if we are already very close to the trivial lower bound.
                # The theoretical lower bound for the normalised problem is 2 n/(2 n‑1) ≈ 1.
                if M <= 1.2:
                    # Very good – stop the inner loop.
                    break

    # ------------------------------------------------------------------
    # Safety net – if nothing was found (should never happen) fall back to a flat distribution.
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Final verification with the supplied evaluator (adds the clipping guarantees).
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Best final score: {final_score:.6f} (length {len(best_seq)})")

    return [float(v) for v in best_seq]
