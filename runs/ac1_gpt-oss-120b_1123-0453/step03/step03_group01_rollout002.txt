def solve_lp(f, target_sum):
    n = len(f)
    # Build A_ub
    rows = 2*n -1
    A_ub = np.zeros((rows, n+1))
    for k in range(rows):
        # For each j from 0..n-1:
        # i = k - j
        # if 0 <= i < n => A_ub[k, j] = f[i]
        # else 0
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # j goes from lo..hi
        # i = k - j : f[i]
        # we can fill by vectorized assignment:
        # Actually easier loop.
        for j in range(lo, hi+1):
            i = k - j
            A_ub[k, j] = f[i]
        A_ub[k, -1] = -1.0  # coefficient for M: conv - M <=0 => conv - M <=0 => conv + (-1)*M <=0
    # Equality constraint sum(b) = target_sum
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    # objective: minimize M
    c = np.zeros(n+1)
    c[-1] = 1.0
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=np.full(rows, 0.0), A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        # fallback maybe return original f.
        return f.copy(), np.max(np.convolve(f,f))
    b_opt = res.x[:n]
    M_opt = res.x[-1]
    return b_opt, M_opt

"""
Propose a step‑function with a low score.

The algorithm works in three stages:

1️⃣  **LP‑based descent** – for a fixed length `n` we keep the ℓ¹‑norm
    `∑a_i = √(2n)`.  Given the current vector `f` we solve a linear
    program:

       minimise   M
       subject to  (f * b)[k] ≤ M   for all k = 0,…,2n‑2
                   b_i ≥ 0
                   ∑ b_i = √(2n)

    The LP variables are the new heights `b` and the auxiliary scalar `M`
    (the maximal convolution value).  Because the constraints are linear
    this can be solved with ``scipy.optimize.linprog``.  The optimal
    `b` has the same ℓ¹‑norm as `f`.  We replace `f` by `b` when the
    resulting `max(f*f)` decreases, otherwise we move part‑way towards `b`
    (convex combination) – this guarantees a monotone descent of the
    official score (which equals `max(f*f)` after normalisation).

2️⃣  **Hill‑climbing polishing** – after the LP phase we perform a fast
    local search that repeatedly transfers a small amount of mass from
    one index to another, updating the convolution incrementally
    (`_transfer_mass`).  This refines the solution without changing the
    ℓ¹‑norm.

3️⃣  **Gaussian perturbation** – a final gentle stochastic search
    (`_local_search`) adds normally‑distributed noise to the whole vector
    and keeps improvements.  All steps respect the non‑negativity and
    clipping rules of `evaluate_sequence`.

The routine tries several lengths (up to 800) and a few random initial
vectors, keeping the best sequence found before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used for the linear program
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog

    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Wrap the provided `evaluate_sequence` (expects a Python list)."""
    return float(evaluate_sequence(seq.tolist()))


def _solve_lp(f: np.ndarray, target_sum: float):
    """
    Solve the LP
        minimise   M
        subject to  (f * b)[k] ≤ M   for k = 0…2n‑2
                    b_i ≥ 0
                    ∑ b_i = target_sum

    Returns (b_opt, M_opt).  If the LP fails, the current vector is
    returned unchanged together with its actual max‑convolution value.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build A_ub · [b; M] ≤ 0   where the last column corresponds to –M
    # A_ub has shape (rows, n+1)
    A_ub = np.zeros((rows, n + 1), dtype=np.float64)

    # Fill the Toeplitz‑like structure row by row
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # j indices that actually appear in this convolution entry
        j = np.arange(lo, hi + 1, dtype=np.int64)
        i = k - j
        A_ub[k, j] = f[i]
        A_ub[k, -1] = -1.0  # coefficient of M

    b_ub = np.zeros(rows, dtype=np.float64)

    # Equality constraint: sum(b) = target_sum
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum], dtype=np.float64)

    # Objective: minimise M
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Solve with HiGHS (fast and robust)
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # Fallback: return the original vector (the score will stay unchanged)
        return f.copy(), np.max(np.convolve(f, f))

    b_opt = res.x[:n]
    M_opt = res.x[-1]

    # Numerical noise may produce tiny negatives – clip them
    b_opt = np.maximum(b_opt, 0.0)
    return b_opt, M_opt


def _transfer_mass(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Transfer `delta` from index i to index j (i != j, delta > 0).
    Updates the vector `a` and its self‑convolution `conv` in‑place.
    Complexity O(n).
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # Linear term: 2 * delta * a
    conv[i : i + n] -= 2.0 * delta * a
    conv[j : j + n] += 2.0 * delta * a

    # Quadratic term from the delta‑vector d (–δ at i, +δ at j)
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta

    a[i] -= delta
    a[j] += delta


def _improve_by_transfer(start_seq: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         step_init: float = 0.5):
    """
    Mass‑transfer hill climbing (the same idea as in the reference code).
    Keeps the total ℓ¹‑norm fixed and accepts only strictly improving moves.
    Returns the best vector encountered and its official score.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)
    sum_a = a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        _transfer_mass(a, conv, i, j, delta)
        cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert
            _transfer_mass(a, conv, j, i, delta)

        step *= 0.9999
        if step < 1e-5:
            step = step_init

        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray, deadline: float,
                  step_initial: float = 0.4):
    """
    Gentle Gaussian perturbation of the whole vector.
    Accepts only strictly improving candidates.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial

    return best, best_score
