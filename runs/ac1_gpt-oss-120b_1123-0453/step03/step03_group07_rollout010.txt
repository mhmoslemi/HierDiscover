"""
Propose a low‑value step sequence.

The algorithm builds on the previous LP‑based scheme but replaces the
alternating LP + convex‑mix loop with a smooth gradient‑descent
procedure that directly minimizes a smoothed version of the infinity
norm of the auto‑convolution.  For a fixed length `n` the entries are
normalised to have L1‑norm √(2 n); under this normalisation the
evaluation function equals the maximal entry of the convolution.
The optimisation proceeds as follows:

* Random restarts (Dirichlet or Gaussian‑shaped seeds).
* At each iteration compute the full auto‑convolution of the current
  vector `f`.
* Replace the hard max by a smooth “log‑sum‑exp’’ approximation with a
  temperature `τ`.  The gradient of this objective is `2·Aᵀ·w`,
  where `A` is the Toeplitz convolution matrix and `w` is the soft‑max
  weight vector over the convolution entries.
* Perform a back‑tracking line‑search, project the trial point onto the
  non‑negative simplex (preserving the prescribed sum), and accept the
  move if the true maximal convolution value decreases.
* The temperature `τ` is annealed towards zero, turning the smooth
  objective into the true max.
* The best sequence seen across all lengths and restarts is returned.

All heavy operations (convolutions, correlations) are performed with
FFT‑based linear convolution/correlation, so each iteration costs
≈ O(n log n).  The routine respects the 1000 s time budget and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of real vectors `a` and `b` using FFT.
    Returns an array of length len(a) + len(b) - 1.
    """
    n = len(a) + len(b) - 1
    N = _next_pow_two(n)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:n]
    return conv


def _corr_fft(w: np.ndarray, f: np.ndarray) -> np.ndarray:
    """
    Full linear correlation of real vectors `w` and `f` (i.e. (w⋆f)[k] = Σ_i w_i f_{i+k}).
    Returns an array of length len(w) + len(f) - 1.
    """
    m = len(w)
    n = len(f)
    N = _next_pow_two(m + n - 1)
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    corr = np.fft.irfft(W * np.conj(F), N)[: m + n - 1]
    return corr


def _gradient_f_w(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smoothed max objective:
        ∇_f ( τ·log Σ_i exp((f*f)_i / τ) )
    which equals 2·Aᵀ·w, where A is the convolution matrix.
    The implementation uses the correlation formula described in the
    module docstring.
    """
    # linear correlation w⋆f (length = 3n-2)
    corr = _corr_fft(w, f)
    n = len(f)
    # g_j = 2 * corr[n-1-j]  (j = 0 … n-1)
    g = 2.0 * corr[n - 1 :: -1][:n]
    return g


def _project_onto_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a vector onto the non‑negative simplex {y ≥ 0, Σy = target_sum}.
    Simple scaling after truncating negatives works because we never need
    the Euclidean‑closest point – any feasible point is acceptable.
    """
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s == 0.0:
        # uniform fallback
        return np.full_like(x, target_sum / x.size)
    return x * (target_sum / s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score.

    The search explores several lengths (up to 1000), performs a few
    random restarts per length, and runs a temperature‑annealed
    gradient‑descent on a smoothed max of the auto‑convolution.
    The best sequence found before the time budget expires is returned
    as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # tiny safety margin

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Lengths to try – larger lengths give a tighter bound but are slower.
    lengths = [300, 400, 500, 600, 800, 1000]

    # ``height_sequence_1`` may already contain a good construction.
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        global_seq = np.asarray(global_seq, dtype=np.float64)
    else:
        global_seq = None

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Number of random restarts – more for smaller n, fewer for large n.
        restarts = 5 if n <= 600 else 3

        for restart in range(restarts):
            if time.time() >= deadline:
                break

            # ------------------- initial point -------------------
            if restart == 0 and global_seq is not None and global_seq.size == n:
                f = global_seq.copy()
                f = np.maximum(f, 0.0)
                f = _project_onto_simplex(f, target_sum)
                print(f"[init] using global seed n={n}")
            else:
                # Mix a Dirichlet sample with an optional Gaussian bump.
                if restart % 2 == 0:
                    f = rng.dirichlet(np.ones(n)) * target_sum
                else:
                    # Gaussian‑shaped seed
                    x = np.arange(n)
                    mu = rng.uniform(0, n - 1)
                    sigma = rng.uniform(0.1 * n, 0.3 * n)
                    g = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
                    g = g / g.sum() * target_sum
                    # blend with a Dirichlet component
                    f = 0.6 * g + 0.4 * rng.dirichlet(np.ones(n)) * target_sum
                    f = _project_onto_simplex(f, target_sum)

            # ------------------- evaluate initial -----------------
            conv_f = _conv_fft(f, f)
            M = conv_f.max()
            cur_score = M  # because sum(f) == sqrt(2n)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[init] n={n} r={restart} score={cur_score:.6f}")

            # ------------------- gradient descent ----------------
            max_iters = 500
            tau = 0.5          # initial smoothing temperature
            tau_decay = 0.95   # geometric cooling
            min_tau = 1e-3
            # base step magnitude (will be normalised by ‖g‖)
            base_step = target_sum * 0.5

            for it in range(max_iters):
                if time.time() >= deadline:
                    break

                # recompute convolution & weights
                conv_f = _conv_fft(f, f)
                M = conv_f.max()
                # smoothed soft‑max weights
                max_c = conv_f.max()
                exps = np.exp((conv_f - max_c) / tau)
                w = exps / exps.sum()

                # gradient of the smooth surrogate
                g = _gradient_f_w(f, w)

                # ----- back‑tracking line search -----
                # scale step inversely to gradient norm for a sensible magnitude
                step_norm = np.linalg.norm(g) + 1e-12
                alpha = base_step / step_norm
                improved = False
                for _ in range(20):
                    f_trial = f - alpha * g
                    f_trial = _project_onto_simplex(f_trial, target_sum)
                    conv_trial = _conv_fft(f_trial, f_trial)
                    M_trial = conv_trial.max()
                    if M_trial < M - 1e-12:
                        improved = True
                        break
                    alpha *= 0.5  # shrink step

                if not improved:
                    # No improvement – tighten the smoothing or perturb.
                    tau = max(tau * 0.5, min_tau)
                    # small random jitter to jump out of plateaus
                    jitter = rng.normal(scale=0.02 * target_sum / n, size=n)
                    f = _project_onto_simplex(f + jitter, target_sum)
                    continue

                # accept the improvement
                f = f_trial
                M = M_trial
                cur_score = M
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = f.copy()
                    print(f"[grad] n={n} r={restart} it={it} τ={tau:.4f} score={cur_score:.6f}")

                # anneal temperature
                tau = max(tau * tau_decay, min_tau)

                # early stop if near the trivial lower bound (average of convolution)
                lower_bound = 2.0 * n / (2.0 * n - 1.0)
                if M <= lower_bound * 1.01:
                    print(f"[near‑lb] n={n} r={restart} it={it} score={M:.6f}")
                    break

            # -------------------------------------------------------------
            # optional local random perturbation search around the best of this restart
            f_local = best_seq.copy() if best_seq is not None else f
            for local_it in range(100):
                if time.time() >= deadline:
                    break
                perturb = rng.normal(scale=0.01 * target_sum / n, size=n)
                f_new = _project_onto_simplex(f_local + perturb, target_sum)
                M_new = _conv_fft(f_new, f_new).max()
                if M_new < best_score - 1e-12:
                    best_score = M_new
                    best_seq = f_new.copy()
                    f_local = f_new
                    print(f"[local] n={n} r={restart} it={local_it} score={M_new:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – uniform distribution if nothing better was found.
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _conv_fft(best_seq, best_seq).max()

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    # Convert to plain Python list of floats as required.
    return [float(v) for v in best_seq]
