    n = len(a)
    m = 2*n - 1
    T = np.zeros((m, n), dtype=float)
    # For each i, column i = a shifted i positions down with zero padding.

def alternating_lp(a, deadline):
    n = len(a)
    target_sum = math.sqrt(2*n)
    best_a = a
    best_score = _score(a)
    step = 0.5 # maybe mixing factor
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = np.max(conv)
        # Build Toeplitz matrix
        T = np.zeros((2*n-1, n))
        for i in range(n):
            T[i:i+n, i] = a
        # Solve LP
        b = cp.Variable(n)
        constraints = [T @ b <= M, b >= 0]
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
        except cp.error.SolverError:
            break
        if b.value is None:
            break
        b_vec = np.maximum(b.value, 0.0)  # clip small negative
        sum_b = b_vec.sum()
        if sum_b <= 0:
            break
        # Scale to target sum
        g = b_vec * (target_sum / sum_b)
        # Option 1: direct replace a = g
        a_new = (1 - step) * a + step * g
        # Ensure sum normalization (might drift due to rounding)
        a_new = np.clip(a_new, 0.0, 1000.0)
        # Renormalize to target sum exactly
        a_new *= target_sum / a_new.sum()
        # Evaluate
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"[LP n={n} iter] score={best_score:.6f}")
            # maybe reduce step or iterate
        # decide to adopt a_new for next iteration
        a = a_new
        # optionally adjust step: reduce gradually
        step *= 0.99
        if step < 0.01:
            step = 0.01
    return best_a, best_score
```

"""
Optimisation of non‑negative step sequences for the evaluation function:
    score = 2·n·max(conv(a,a)) / (sum(a)²)

The score is invariant under scaling; we therefore keep the ℓ¹‑norm fixed to
√(2 n).  The algorithm consists of three phases for each chosen length *n*:

1️⃣  **Directed mass‑transfer hill‑climbing** –  
    We repeatedly move a small amount of mass from an index that most
    contributes to the current maximal autocorrelation (the “peak” shift)
    to an index that contributes least.  The convolution is updated
    incrementally in O(n) using a hand‑crafted transfer routine, so millions
    of moves can be performed quickly.

2️⃣  **Projected sub‑gradient descent** –  
    When the hill‑climber stagnates we treat the maximal autocorrelation
    as a (convex) piecewise‑linear function.  Its sub‑gradient is the sum of
    the shifted vectors that realise the current maximum.  A step in the
    negative sub‑gradient direction is taken and the vector is projected back
    onto the simplex {x ≥ 0, ∑x = √(2 n)} using the Euclidean projection.

3️⃣  **Gaussian polishing** –  
    Finally a short stochastic Gaussian perturbation (the “_local_search”
    from the reference code) cleans up any remaining roughness.

All phases respect the non‑negativity, clipping and norm constraints
implemented inside ``evaluate_sequence``.  The routine runs for at most the
given time budget (≈ 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official score via the provided evaluate_sequence()."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index `i` to `j` (i != j, delta > 0, delta ≤ a[i]).
    The vector `a` and its convolution `conv = np.convolve(a, a)` are updated
    in‑place using the exact algebraic formula (O(n) work).
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part: 2 * a * delta
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part: d*d where d = -δ·e_i + δ·e_j
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta          # (-δ)²
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta    # (-δ)*(+δ) + (+δ)*(-δ)
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta          # (+δ)²

    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
# Phase 1 – directed mass‑transfer hill climbing
# ----------------------------------------------------------------------


def _directional_hill_climb(start_seq: np.ndarray,
                            deadline: float,
                            step_init: float = 0.5,
                            min_step: float = 1e-4,
                            decay: float = 0.9997) -> tuple[np.ndarray, float]:
    """
    Fast hill climbing that always moves mass from an index that contributes
    most to the current maximal autocorrelation to an index that contributes
    least.  Returns the best sequence encountered and its max‑convolution.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)                     # full convolution
    best_a = a.copy()
    best_M = np.max(conv)

    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline:
        # index of the current peak
        p = int(np.argmax(conv))

        # compute contribution weights w_i = a_{p-i} (zero when out of bounds)
        idx = np.arange(n)
        valid = (p - idx >= 0) & (p - idx < n)
        w = np.zeros_like(a)
        w[valid] = a[p - idx[valid]]

        # source: large w & positive mass
        src_candidates = np.where((w > 0) & (a > 1e-12))[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(w[src_candidates])]

        # target: small w (could be zero)
        j = int(np.argmin(w))

        if i == j:
            # nothing useful to do – random perturbation instead
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] < 1e-12:
                continue

        # amount to move
        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        # apply move
        _transfer_mass(a, conv, i, j, delta)
        cur_M = np.max(conv)

        if cur_M < best_M - 1e-12:
            best_M = cur_M
            best_a = a.copy()
        else:
            # revert – the move was not beneficial
            _transfer_mass(a, conv, j, i, delta)

        # decay step size, occasionally reset to avoid stagnation
        step *= decay
        if step < min_step:
            step = step_init

        it += 1

    return best_a, best_M


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_descent(start_seq: np.ndarray,
                         deadline: float,
                         eta0: float = 0.2,
                         tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method on the function M(a) = max(conv(a,a)).
    At each iteration we compute a sub‑gradient from the shifts attaining the
    maximum, take a step in the negative direction and project back onto the
    simplex {x ≥ 0, sum(x) = √(2 n)}.
    """
    a = start_seq.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # initial score
    conv = np.convolve(a, a)
    best_M = np.max(conv)
    best_a = a.copy()

    rng = np.random.default_rng()
    t = 0

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = np.max(conv)
        if M < best_M - tol:
            best_M = M
            best_a = a.copy()
        # locate all shifts attaining the maximum (within tolerance)
        max_idxs = np.where(np.abs(conv - M) <= 1e-12)[0]

        # compute sub‑gradient: g_i = 2 * Σ_{k∈max_idxs} a_{k-i}
        grad = np.zeros(n)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += 2.0 * a[k - i_range]

        # diminishing step size
        t += 1
        eta = eta0 / math.sqrt(t)

        # gradient step and projection onto the simplex
        a_next = a - eta * grad
        a_next = np.clip(a_next, 0.0, None)
        a_next = _project_onto_simplex(a_next, target_sum)

        # stop if change is negligible
        if np.linalg.norm(a_next - a) < 1e-12:
            break

        a = a_next

    return best_a, best_M

# Phase 3 – final Gaussian polishing (same as reference)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that adds Gaussian noise to the whole
    vector.  Used only as a final polishing step.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial

    return best, best_score
