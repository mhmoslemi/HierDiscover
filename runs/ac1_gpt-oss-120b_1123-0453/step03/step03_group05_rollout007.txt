"""
Multi‑restart projected gradient descent on a smooth max‑approximation of the
worst auto‑convolution entry, followed by a short stochastic local search.
The routine works for any length; it tries several lengths (up to a few
thousand) and keeps the best sequence found before the time budget expires.
All numbers are kept non‑negative and projected onto the ℓ¹‑simplex
{x≥0, Σx = √(2·n)} so the objective value 2·n·max(conv)/‖x‖₁² is
scale‑invariant.  The implementation uses FFT‑based convolutions for speed
and a moving‑average of the gradient to obtain a simple momentum effect.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σ x = s}.  Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a * b using real FFT.
    The result length is len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    c = np.fft.irfft(fa * fb, n=size)
    return c[: n + m - 1]


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence (no type checks).  Returns np.inf for
    pathological inputs.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)  # length 2n‑1
    max_b = float(conv.max())
    return float(2.0 * n * max_b / (sum_a * sum_a))


_score = _fast_score  # public alias used later


def _init_f(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random positive vector of length n, projected onto the required simplex."""
    target = math.sqrt(2.0 * n)
    f = rng.random(n) + 0.1  # avoid exact zeros
    f = f / f.sum() * target
    return f


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that respects the ℓ¹‑norm constraint.
    """
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best = start_seq.copy()
    best_score = _score(best)

    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Gradient‑descent core
# ----------------------------------------------------------------------


def _gd_descent(f: np.ndarray,
                n: int,
                target_sum: float,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max approximation of the
    worst convolution entry.  A moving‑average of the gradient provides a
    light‑weight momentum effect.
    """
    best_f = f.copy()
    best_score = _score(f)
    cur_score = best_score

    # moving‑average of the gradient (momentum)
    grad_avg = np.zeros_like(f)
    gamma = 0.9                         # momentum decay
    beta_schedule = [2, 4, 8, 16, 32, 64, 128]   # geometric continuation
    # a relatively generous step factor – scaled by target_sum each iteration
    step_factor = 0.6

    for beta in beta_schedule:
        # a modest but not too small inner loop – we let the global deadline
        # stop the process if needed.
        inner_iters = 2000
        for _ in range(inner_iters):
            if time.monotonic() >= deadline:
                return best_f, best_score

            # f * f
            conv_f = _conv_fft(f, f)                 # length 2n‑1
            max_c = conv_f.max()
            shifted = conv_f - max_c                 # shift for numerical stability
            exp_terms = np.exp(beta * shifted)       # all ≤ 1
            sum_exp = exp_terms.sum()
            if sum_exp == 0.0:
                p = np.full_like(exp_terms, 1.0 / exp_terms.size)
            else:
                p = exp_terms / sum_exp               # soft‑max distribution

            # gradient = 2 * (p * f)   (first n components)
            conv_pf = _conv_fft(p, f)                # length 3n‑2
            grad = 2.0 * conv_pf[:n]

            # update moving average
            grad_avg = gamma * grad_avg + (1.0 - gamma) * grad
            grad_norm = np.linalg.norm(grad_avg) + 1e-12
            step = step_factor * target_sum / grad_norm

            cand = f - step * grad_avg
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _project_simplex(cand, target_sum)

            cand_score = _score(cand)

            if cand_score < best_score:
                best_score = cand_score
                best_f = cand.copy()
                print(f"[GD] n={n} β={beta} score={cand_score:.6f}")
                if best_score < 1.503:
                    return best_f, best_score

            # accept monotone improvements for the current iterate
            if cand_score <= cur_score:
                f = cand
                cur_score = cand_score
            # otherwise keep the old f (simple greedy scheme)

    return best_f, best_score


def _optimize_one_restart(n: int,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a single random restart of the gradient descent for a given length.
    If a global sequence of the same length exists it is used as a warm start.
    """
    target_sum = math.sqrt(2.0 * n)

    # start from the global best if it matches the length
    if "height_sequence_1" in globals():
        seq_glob = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq_glob.shape[0] == n:
            f0 = np.clip(seq_glob, 0.0, 1000.0)
            if f0.sum() > 0:
                f0 = f0 * (target_sum / f0.sum())
                # fall back to random if the global vector happens to be all zeros
                if np.allclose(f0, 0):
                    f0 = _init_f(n, rng)
            else:
                f0 = _init_f(n, rng)
        else:
            f0 = _init_f(n, rng)
    else:
        f0 = _init_f(n, rng)

    return _gd_descent(f0, n, target_sum, deadline, rng)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the global time budget and returns a plain list of
    floats (the heights).  It repeatedly runs projected gradient descent on
    randomly initialised vectors of several lengths and finally applies a
    short stochastic local refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use the previously discovered sequence if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seq0 = np.clip(seq0, 0.0, 1000.0)
        if seq0.sum() > 0:
            seq0 = seq0 * (math.sqrt(2.0 * len(seq0)) / seq0.sum())
            sc0 = _score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] global height_sequence_1 (n={len(seq0)}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # 1.  Multi‑restart optimisation on several lengths
    # ------------------------------------------------------------------
    # A modest collection of lengths – larger lengths give a better theoretical
    # lower bound but are slightly more expensive per iteration.
    lengths = [500, 800, 1000, 1200, 1500, 1800, 2000]

    # Loop until the budget runs out; each iteration picks a length at random.
    while time.monotonic() < deadline:
        n = int(rng.choice(lengths))
        # give each restart a fair share of the remaining time (but never exceed)
        sub_deadline = deadline
        seq, sc = _optimize_one_restart(n, sub_deadline, rng)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[restart] n={n} new best score={sc:.6f}")
        # early exit if we have already beaten the target
        if best_score < 1.503:
            print("[early] target achieved – exiting.")
            break

    # ------------------------------------------------------------------
    # 2.  Final stochastic refinement (Gaussian perturbations)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined_seq, refined_sc = _local_search(best_seq, deadline, step_initial=0.4)
        if refined_sc < best_score:
            best_seq = refined_seq
            best_score = refined_sc
            print(f"[local] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _score(best_seq)

    print(f"[final] best score={best_score:.6f} length={len(best_seq)}")
    return [float(x) for x in best_seq]
