"""
Propose a low‑value step sequence.

Algorithm
---------
1. Initialise from the previous global best if it exists.
2. Repeatedly restart from a random point on the ℓ¹‑simplex
   (strictly enforcing Σ a_i = √(2·n)).
3. For each restart run a *flattening* optimisation:
   minimise the squared deviation of the auto‑convolution from a constant
   target (the average value of the convolution).  Using the exact gradient
   (derived from the quadratic form a∗a) and a projection onto the simplex
   yields very fast descent of the maximum convolution entry.
4. Occasionally run a cheap stochastic hill‑climbing refinement.
5. Keep the best sequence ever seen and return it before the time budget
   expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (identical to the previous script, kept for speed)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s} (Duchi et al.)."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` – no type checks."""
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = float(conv.max())
    # (2 * n * max_b) / (sum_a ** 2)
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _score(seq: np.ndarray) -> float:
    """Thin wrapper – keeps the public name used later."""
    return _fast_score(seq)


# ----------------------------------------------------------------------
# 1. Stochastic local search (from the original script – kept as a fallback)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best = start_seq.copy()
    best_score = _score(best)

    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# 2. Gradient descent that flattens the auto‑convolution
# ----------------------------------------------------------------------
def _flatten_optimize(seq: np.ndarray,
                      deadline: float,
                      max_iters: int = 300) -> tuple[np.ndarray, float]:
    """
    Minimise Σ_k ( (a*a)_k - avg )² with a simplex projection.
    The gradient is computed analytically as
        grad = 2 * ( (conv - avg)  *  a_rev )
    where * denotes the discrete convolution and a_rev = a[::-1].
    """
    n = seq.shape[0]
    target_sum = math.sqrt(2.0 * n)
    # target constant (average of the convolution)
    avg_const = (target_sum ** 2) / (2 * n - 1)  # = 2n/(2n-1)

    step_factor = 0.10          # controls the scale of each step
    step = target_sum * step_factor
    decay = 0.9995

    a = seq.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)           # length 2n‑1
        residual = conv - avg_const        # deviation from the constant target

        # grad = 2 * (residual ⊛ a_rev)   (⊛ = convolution)
        grad_full = np.convolve(residual, a[::-1])
        grad = 2.0 * grad_full[n - 1:2 * n - 1]   # keep the n relevant entries

        grad_norm = np.linalg.norm(grad) + 1e-12
        step_sz = step / grad_norm

        cand = a - step_sz * grad
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)

        cand_score = _score(cand)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # accept only if we improve the current sequence (simple descent)
        if cand_score <= cur_score:
            a = cand
            cur_score = cand_score

        # decay the step size a little each iteration
        step *= decay

        # early exit – we have already reached the target bound
        if best_score < 1.503:
            break

    return best, best_score


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Uses a fast “flattening” gradient descent from many random restarts,
    plus a final stochastic refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. Seed from a previously found global best (if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seq0 = np.clip(seq0, 0.0, 1000.0)
        n0 = seq0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        if seq0.sum() > 0:
            seq0 = seq0 * (target0 / seq0.sum())
            sc0 = _score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using global height_sequence_1 (n={n0}) score={sc0:.6f}")

    # --------------------------------------------------------------
    # 1. Random‑restart flatten‑gradient optimisation
    # --------------------------------------------------------------
    # a few different lengths – bigger n often gives lower scores
    lengths = [300, 400, 500, 600, 800, 1000]
    while time.monotonic() < deadline:
        n = int(rng.choice(lengths))
        target_sum = math.sqrt(2.0 * n)

        # initialise a random point on the simplex (Dirichlet‑like)
        a = rng.gamma(shape=1.0, size=n) + 1e-6
        a = a / a.sum() * target_sum

        # run the specialised flattening optimisation
        a_opt, score_opt = _flatten_optimize(a, deadline, max_iters=250)

        if score_opt < best_score:
            best_seq = a_opt.copy()
            best_score = score_opt
            print(f"[restart] n={n} score={score_opt:.6f}")

        # occasional cheap stochastic refinement on the current global best
        if time.monotonic() < deadline and best_seq is not None:
            refined, refined_sc = _local_search(best_seq, deadline,
                                                step_initial=0.4)
            if refined_sc < best_score:
                best_seq = refined
                best_score = refined_sc
                print(f"[local] improved to {best_score:.6f}")

        # early exit if we have already beaten the target bound
        if best_score < 1.503:
            print("[exit] target achieved")
            break

    # 2. Final fallback (should never happen, but guarantees a return)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)

    print(f"[final] best score={best_score:.6f} length={len(best_seq)}")
    return [float(x) for x in best_seq]
