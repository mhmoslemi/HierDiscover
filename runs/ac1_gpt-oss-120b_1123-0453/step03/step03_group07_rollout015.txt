def simplex_projection(v, target_sum):
    # reference Duchi
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    rng = np.random.default_rng(seed)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # number of restarts per length

        for r in range(restarts):
            if time.time() >= deadline:

            # initialize f
            if r == 0 and global_seq is not None and len(global_seq) == n:
                f = global_seq.copy()
                f = np.maximum(f, 0.0)

            conv = np.convolve(f, f)

"""
Simple stochastic local search for the step‑function minimisation problem.

The algorithm starts from several random (Dirichlet) restarts (and from the
global ``height_sequence_1`` if it has matching length).  For each restart it
maintains the full auto‑convolution and repeatedly moves a small amount of
mass from a “source’’ index to a “target’’ index.  The source is chosen
approximately in the direction of the sub‑gradient of the current infinity‑norm
(maximum entry of the convolution).  The move is accepted if it reduces the
maximum; otherwise a temperature‑controlled simulated‑annealing rule may accept
it.  Convolution updates are performed analytically (no FFT), so each
iteration costs O(n).  After the stochastic phase a few cheap LP‑based
refinements (the same helpers used in the previous baseline) are applied.
The best sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper to evaluate (mainly for logging)
def _evaluate(seq: np.ndarray) -> float:
    """Wrapper for the provided ``evaluate_sequence``."""
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Linear‑programming helpers (re‑used from the baseline)
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Toeplitz matrix A such that A @ g == np.convolve(f, g) (linear conv)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _lp_minimize_M(f: np.ndarray, target_sum: float):
    """Solve   minimise M   s.t.  A@g ≤ M ,  sum(g)=target_sum , g≥0."""
    try:
        import cvxpy as cp
    except Exception:               # pragma: no cover
        return None, None

    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n)
    Mvar = cp.Variable()
    constraints = [
        g >= 0,
        A @ g <= Mvar,
        cp.sum(g) == target_sum,
    ]
    prob = cp.Problem(cp.Minimize(Mvar), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:               # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:           # pragma: no cover
            return None, None

    if g.value is None or Mvar.value is None:
        return None, None

    g_val = np.maximum(g.value, 0.0)
    return g_val, float(Mvar.value)


def _optimal_mix(f: np.ndarray, g: np.ndarray,
                 Cff: np.ndarray, Cgg: np.ndarray, Cfg: np.ndarray,
                 current_M: float):
    """Exact line‑search for the best convex combination of f and g."""
    lo, hi = 0.0, 1.0
    best_M = current_M
    best_t = None

    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0

        comb1 = (1 - t1) ** 2 * Cff + 2 * t1 * (1 - t1) * Cfg + t1 ** 2 * Cgg
        M1 = np.max(comb1)

        comb2 = (1 - t2) ** 2 * Cff + 2 * t2 * (1 - t2) * Cfg + t2 ** 2 * Cgg
        M2 = np.max(comb2)

        if M1 < M2:
            hi = t2
            if M1 < best_M:
                best_M, best_t = M1, t1
        else:
            lo = t1
            if M2 < best_M:
                best_M, best_t = M2, t2

    t_opt = (lo + hi) / 2.0
    comb_opt = (1 - t_opt) ** 2 * Cff + 2 * t_opt * (1 - t_opt) * Cfg + t_opt ** 2 * Cgg
    M_opt = np.max(comb_opt)
    if M_opt < best_M:
        best_M, best_t = M_opt, t_opt

    if best_t is None or best_M >= current_M - 1e-12:
        return None, None

    new_vec = (1.0 - best_t) * f + best_t * g
    return new_vec, best_M

# Stochastic local search (SA) core
def _subgradient(f: np.ndarray, conv: np.ndarray, cur_max: float) -> np.ndarray:
    """
    Compute a sub‑gradient of the ∞‑norm of the convolution at the current
    maximal locations.  Returned array has length n.
    """
    n = f.shape[0]
    grad = np.zeros(n, dtype=np.float64)
    max_idx = np.flatnonzero(conv == cur_max)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] += 2.0 * f[k - idx]
    return grad


def _sa_step(f: np.ndarray, conv: np.ndarray, cur_max: float,
             target_sum: float, rng: np.random.Generator,
             it: int, max_iters: int):
    """
    Perform a single SA move.  Returns (new_f, new_conv, new_max, accepted, delta).
    """
    n = f.shape[0]
    # sub‑gradient gives a sensible source index
    grad = _subgradient(f, conv, cur_max)
    total = grad.sum()
    if total > 0.0:
        probs = grad / total
        i = rng.choice(n, p=probs)
    else:
        i = rng.integers(n)

    # pick a different target index uniformly
    j = i
    while j == i:
        j = rng.integers(n)

    # step size schedule (large early, shrink later)
    avg = target_sum / n
    step_factor = 2.0               # max step ≈ 2·average entry
    step_scale = avg * step_factor * (1.0 - it / max_iters)
    eps_limit = min(f[i], step_scale)
    if eps_limit <= 0:
        return f, conv, cur_max, False, 0.0

    epsilon = rng.random() * eps_limit
    if epsilon == 0.0:
        return f, conv, cur_max, False, 0.0

    # update convolution exactly
    new_conv = conv.copy()
    new_conv[i:i + n] -= 2.0 * epsilon * f
    new_conv[j:j + n] += 2.0 * epsilon * f

    new_max = new_conv.max()
    delta = new_max - cur_max
    # temperature schedule (simple linear cooling)
    T0 = 0.02
    T = T0 * (1.0 - it / max_iters)

    accept = False
    if delta < 0:
        accept = True
    elif T > 1e-12 and rng.random() < math.exp(-delta / T):
        accept = True

    if accept:
        f_new = f.copy()
        f_new[i] -= epsilon
        f_new[j] += epsilon
        return f_new, new_conv, new_max, True, delta
    else:
        return f, conv, cur_max, False, delta


def _run_sa(f: np.ndarray, target_sum: float,
            deadline: float, rng: np.random.Generator,
            max_total_iters: int = 50000):
    """
    Run the SA optimiser on a single start vector `f`.  Returns the best
    vector seen and its max‑convolution value.
    """
    n = f.shape[0]
    conv = np.convolve(f, f)
    cur_max = conv.max()
    best_f = f.copy()
    best_max = cur_max

    it = 0
    while it < max_total_iters and time.time() < deadline:
        f, conv, cur_max, accepted, _ = _sa_step(
            f, conv, cur_max, target_sum, rng, it, max_total_iters
        )
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_f = f.copy()
        it += 1
    return best_f, best_max
