        f = f0.copy()

        it = 0
        while time.monotonic() < deadline and it < 5000:

            # Evaluate

                best_seq = f.copy()

            # compute subgradient
            g = np.zeros_like(f)
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            if low <= high:
                idx = np.arange(low, high+1)
                g[idx] = 2.0 * f[k - idx]

            # momentum update
            momentum = momentum_coeff * momentum + g
            f_candidate = f - eta * momentum
            f_candidate = _project_simplex(f_candidate, target_sum)

            # compute candidate M
            M_cand, _ = _fft_conv_max(f_candidate)
            if M_cand < M:
                f = f_candidate
                eta = min(eta / decay, 0.8)
            else:
                eta *= 0.5

            # random mass transfer with some probability
            if rng.random() < 0.02:
                # choose source with > epsilon
                src = rng.choice(np.where(f > 1e-8)[0])
                dst = rng.integers(0, n)
                if src != dst:
                    delta = min(f[src], rng.random() * 0.1 * target_sum)
                    f2 = f.copy()
                    f2[src] -= delta
                    f2[dst] += delta
                    f2 = _project_simplex(f2, target_sum)
                    M2, _ = _fft_conv_max(f2)
                    if M2 < M:
                        f = f2

            # adjust eta lower bound

            if step < 1e-7:

def _softmax_mirror_descent(f, target_sum, deadline, max_iters=20000, eta0=0.5, decay=0.9995, alpha0=1.0, alpha_max=30.0, alpha_inc=0.02):
   n = len(f)
   size1 = 1 << ((2*n-1).bit_length())
   size2 = 1 << ((3*n-2).bit_length())
   best_f = f.copy()
   best_val = np.max(np.fft.irfft(np.fft.rfft(f, size1) * np.fft.rfft(f, size1), size1)[:2*n-1])
   eta = eta0
   alpha = alpha0
   it=0
   while time.monotonic() < deadline and it < max_iters:
       it+=1
       # current conv & max
       F = np.fft.rfft(f, size1)
       conv = np.fft.irfft(F*F, size1)[:2*n-1]
       M = float(conv.max())
       if M < best_val:
           best_val = M
           best_f = f.copy()
       # compute gradient via softmax
       if alpha < alpha_max:
           alpha = min(alpha_max, alpha * (1 + alpha_inc))
       # compute weights
       maxc = M  # conv.max()
       w = np.exp(alpha * (conv - maxc))
       w_sum = np.sum(w)
       if w_sum == 0:
           w = np.ones_like(w) / w.size
       else:
           w = w / w_sum
       # gradient via conv of w and reversed f
       f_rev = f[::-1]
       W = np.fft.rfft(w, size2)
       F_rev = np.fft.rfft(f_rev, size2)
       grad_full = np.fft.irfft(W * F_rev, size2)[:3*n-2]
       grad = 2.0 * grad_full[n-1:n-1+n]

       # candidate via mirror descent
       f_candidate = f * np.exp(-eta * grad)
       f_candidate = _project_simplex(f_candidate, target_sum)

       # compute candidate value
       Fc = np.fft.rfft(f_candidate, size1)
       conv_c = np.fft.irfft(Fc*Fc, size1)[:2*n-1]
       M_c = float(conv_c.max())

       if M_c < M:
           f = f_candidate
           # increase step a bit
           eta = min(eta / decay, eta0)
       else:
           # reduce step
           eta *= 0.5
           # maybe try additive step if fails? We'll fallback to random perturbation
           if eta < 1e-5:
               # random perturbation to escape
               # random delta transfer
               src = rng.integers(0, n)
               dst = rng.integers(0, n)
               if src != dst and f[src] > 0:
                   delta = min(f[src], rng.random() * 0.1 * target_sum)
                   f_pert = f.copy()
                   f_pert[src] -= delta
                   f_pert[dst] += delta
                   f_pert = _project_simplex(f_pert, target_sum)
                   f = f_pert
               eta = eta0

   return best_f, best_val

"""
search_step_sequence.py

A self‑contained optimizer that tries to minimise

    2·n·max(f*f) / (sum(f)**2)

for a non‑negative step function `f`.  Because the algorithm keeps the
ℓ¹‑norm fixed to √(2·n) the objective collapses to ``max(f*f)``.
The optimisation consists of three stages

1️⃣  **Mirror‑descent with a smooth‐max surrogate** – the worst
    auto‑convolution entry is approximated by a soft‑max (temperature
    α).  The gradient of the surrogate is obtained by a second FFT‑based
    convolution, giving a fast O(n log n) iteration that works for
    thousands of variables.

2️⃣  **Randomised escape moves** – when the multiplicative step stalls the
    current vector is perturbed by moving a small amount of mass between
    two random coordinates.

3️⃣  **Final Gaussian local search** – a cheap stochastic hill‑climbing
    phase that polishes the best vector found.

All updates respect the non‑negativity and ℓ¹‑norm constraints by a
simple projection onto the simplex.  The routine is deterministic for a
given ``seed`` and respects the 1000 s budget (a 0.2 s safety margin is
kept).  If a previous construction ``height_sequence_1`` is present it is
used as a warm start.

The function ``propose_candidate`` returns a plain ``list[float]`` ready
for ``evaluate_sequence``.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper: projection onto the non‑negative simplex of a given total sum
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project *v* onto {x≥0, Σx = target}.  Returns a new array."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - target
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full auto‑convolution (via FFT) and its maximum value
# ----------------------------------------------------------------------
def _auto_conv_max(f: np.ndarray) -> tuple[np.ndarray, float]:
    """Return (full auto‑convolution, its maximum entry)."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    m = float(conv.max())
    return conv, m


# ----------------------------------------------------------------------
# Mirror‑descent using a soft‑max surrogate for the max‑operator
# ----------------------------------------------------------------------
def _softmax_mirror_descent(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    eta0: float = 0.5,
    eta_min: float = 1e-6,
    decay: float = 0.9995,
    alpha0: float = 1.0,
    alpha_max: float = 30.0,
    alpha_inc: float = 0.02,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex with a smooth max objective.

    Parameters
    ----------
    f0          – starting vector (already normalised)
    target_sum  – the ℓ¹‑norm to keep (√(2·n))
    deadline    – absolute time when the routine must stop
    rng         – numpy random generator
    eta0        – initial step‑size for the multiplicative update
    decay       – exponential decay factor for the step‑size
    alpha0 …    – parameters for the soft‑max temperature
    """
    n = f0.shape[0]
    size_conv = 1 << ((2 * n - 1).bit_length())
    size_grad = 1 << ((3 * n - 2).bit_length())

    f = f0.copy()
    best_f = f.copy()
    # initial value
    _, best_val = _auto_conv_max(f)

    eta = eta0
    alpha = alpha0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # ----- current auto‑convolution and its max -----
        conv, cur_max = _auto_conv_max(f)

        if cur_max < best_val:
            best_val = cur_max
            best_f = f.copy()

        # ----- raise the temperature slowly (sharpen the surrogate) -----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * (1.0 + alpha_inc))

        # ----- soft‑max weights -----
        # shift by the maximum to avoid overflow
        shifted = conv - cur_max
        w = np.exp(alpha * shifted)
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth max (2 * (w * f_rev) convolution) -----
        f_rev = f[::-1]
        W = np.fft.rfft(w, size_grad)
        F_rev = np.fft.rfft(f_rev, size_grad)
        grad_full = np.fft.irfft(W * F_rev, size_grad)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # length n

        # ----- multiplicative (mirror) update -----
        f_candidate = f * np.exp(-eta * grad)
        f_candidate = _project_simplex(f_candidate, target_sum)

        # evaluate candidate
        _, cand_max = _auto_conv_max(f_candidate)

        if cand_max < cur_max:
            # improvement – accept and relax step‑size a little
            f = f_candidate
            eta = min(eta / decay, eta0)
        else:
            # no improvement – shrink step‑size
            eta = max(eta * 0.5, eta_min)
            # occasional random escape move when the step becomes tiny
            if eta <= 1e-7:
                # move a random fraction of mass from one coordinate to another
                src = rng.integers(0, n)
                dst = rng.integers(0, n)
                if src != dst and f[src] > 0.0:
                    delta = min(f[src], rng.random() * 0.05 * target_sum)
                    f_tmp = f.copy()
                    f_tmp[src] -= delta
                    f_tmp[dst] += delta
                    f_tmp = _project_simplex(f_tmp, target_sum)
                    f = f_tmp
                eta = eta0

        # exponential decay (keeps the step‑size from staying too large)
        eta *= decay

    return best_f, best_val


# ----------------------------------------------------------------------
# Simple Gaussian hill‑climbing (polishing)
# ----------------------------------------------------------------------
def _gaussian_local_search(
    start_seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    step_initial: float = 0.6,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """Stochastic local search that uses the provided ``evaluate_sequence``."""
    best = start_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    step = step_initial
    n = best.shape[0]

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to the required ℓ¹‑norm (the evaluator also does it,
        # but keeping it constant helps the optimiser)
        target = math.sqrt(2.0 * n)
        cand = _project_simplex(cand, target)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[local] improvement → score={best_score:.6f}")

        step *= decay
        if step < 1e-7:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Load any previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1️⃣  Mirror‑descent on several increasing lengths
    # ------------------------------------------------------------------
    # The lengths are chosen to give the optimizer more freedom while still

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
