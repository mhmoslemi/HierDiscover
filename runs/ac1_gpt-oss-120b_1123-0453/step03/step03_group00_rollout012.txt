def build_conv_matrix(f):
    n = len(f)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f

Propose a low‑value step sequence.

Algorithm
---------
1.  Choose several lengths n (300…1200).  For each length:
    a) initialise a non‑negative vector f with the correct ℓ¹‑norm
       (sum = √(2·n)).  Use the provided `height_sequence_1` when it

           Toeplitz matrix of f.

2. Keep the best sequence over all lengths.
3. Final stochastic Gaussian hill‑climbing polish.
4. Clip the result to the limits required by `evaluate_sequence`
   and return it as a plain list of floats.

The whole search respects the global time budget (default 1000 s).

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact value of the evaluation function (no internal clipping)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    eps = max_val * 1e-9 + 1e-12
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the worst auto‑convolution entry wrt f."""
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = f[j]
        grad[i] += 2.0 * vals
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= vals[mask]
    return grad

    no_improve = 0

    for it in range(max_iters):

            break

        cur_score = _compute_score_from_vector(f)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
            print(f"[GD] it={it} n={len(f)} score={cur_score:.6f}")
        else:
            no_improve += 1

        if no_improve > 300:
            break

        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)
        grad = _gradient_submax(f, active)

        f_candidate = _project_onto_simplex(f - step * grad, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)
        cand_score = _compute_score_from_vector(f_candidate)

        if cand_score < cur_score:
            f = f_candidate
            step *= 1.02
        else:
            step *= 0.5
    return best_f, best_score


def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """Gaussian hill‑climbing preserving the ℓ¹‑norm."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score

# Linear‑programming improvement (Frank‑Wolfe / active‑set style)
# ----------------------------------------------------------------------
has_linprog = False
try:
    from scipy.optimize import linprog
    has_linprog = True
except Exception:
    print("[LP] scipy.optimize.linprog not available – will skip LP stage.")


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the dense (2n‑1)×n matrix A with A[k, j] = f_{k‑j} (or 0)."""
    n = len(f)
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A.fill(0.0)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_improve(f0: np.ndarray,
                target_sum: float,
                deadline: float,
                max_iter: int = 25) -> tuple[np.ndarray, float]:
    """
    Perform a few Frank‑Wolfe style LP steps.
    Returns the best vector found and its score.
    """
    if not has_linprog:
        return f0, _compute_score_from_vector(f0)

    n = len(f0)
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        # Build the linear constraints A·b ≤ M
        A = _build_conv_matrix(f)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear program: maximise Σ bᵢ   ⇔   minimise −Σ bᵢ
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, 1000.0)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # something went wrong – abort LP phase
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b_vec = res.x
        sum_b = np.sum(b_vec)
        sum_f = np.sum(f)

        # If we cannot increase the ℓ¹‑norm, we are at a stationary point.
        if sum_b <= sum_f * (1 + 1e-12):
            break

        # Scale b to have the same ℓ¹‑norm as f.
        g = b_vec * (sum_f / sum_b)

        # Pre‑compute the three convolutions needed for the quadratic update.
        conv_fg = np.convolve(f, g)
        conv_gg = np.convolve(g, g)

        # Small discrete line‑search for the best mixing parameter t∈(0,1].
        best_t = 0.0
        best_Mt = M
        for t in (0.1, 0.2, 0.4, 0.6, 0.8):
            conv_new = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_gg
            Mt = np.max(conv_new)
            if Mt < best_Mt - 1e-12:
                best_Mt = Mt
                best_t = t

        # If no improvement was found, fall back to a very small step.
        if best_t == 0.0:
            t = 0.05
            conv_new = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_gg
            best_Mt = np.max(conv_new)
            best_t = t

        # Update f.
        f = (1 - best_t) * f + best_t * g
        f = np.clip(f, 0.0, 1000.0)

        cur_score = _compute_score_from_vector(f)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            print(f"[LP] n={n} iter={it} t={best_t:.3f} score={cur_score:.6f}")

        # Stop if the improvement is negligible.
        if M - best_Mt < 1e-12:
            break

    return best_f, best_score

# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from any existing global construction.
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.array(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        n0 = len(seq)
        target_sum0 = math.sqrt(2.0 * n0)
        if np.sum(seq) == 0:
            seq = np.full(n0, target_sum0 / n0, dtype=np.float64)
        else:
            seq = seq * (target_sum0 / np.sum(seq))
        score0 = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score0
        print(f"[init] using height_sequence_1, n={n0}, score={score0:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Main search over several lengths.
    # ------------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000, 1200]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # initialise a candidate vector for this length
        if isinstance(best_seq, np.ndarray) and len(best_seq) == n:
            f0 = best_seq.copy()
        elif isinstance(h1, (list, tuple, np.ndarray)) and len(h1) == n:
            f0 = np.array(h1, dtype=np.float64)
        else:
            # uniform + tiny noise
            f0 = np.full(n, target_sum / n, dtype=np.float64)
            noise = rng.normal(scale=target_sum * 0.01, size=n)
            f0 = np.clip(f0 + noise, 0.0, 1000.0)

        f0 = f0 * (target_sum / np.sum(f0))

        # ------------------------------------------------------------------
        # 1a️⃣ LP‑based improvement
        # ------------------------------------------------------------------
        sub_deadline = min(deadline, time.monotonic() + 30.0)
        f_lp, score_lp = _lp_improve(f0, target_sum, sub_deadline, max_iter=25)

        if score_lp < best_score:
            best_score = score_lp
            best_seq = f_lp.copy()
            print(f"[global] new best after LP n={n} score={score_lp:.6f}")

        # ------------------------------------------------------------------
        # 1b️⃣ Short projected‑gradient descent polish
        # ------------------------------------------------------------------
        if time.monotonic() < deadline:
            sub_deadline = min(deadline, time.monotonic() + 8.0)
            f_gd, score_gd = _gradient_descent(f_lp, target_sum,
                                               sub_deadline, max_iters=3000)
            if score_gd < best_score:
                best_score = score_gd
                best_seq = f_gd.copy()
                print(f"[global] new best after GD n={n} score={score_gd:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:
            print("[early] target achieved")
            break

    # ------------------------------------------------------------------
    # 2️⃣ Final stochastic polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline,
                                                        step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] final best score={best_score:.6f} length={len(best_seq)}")
    return [float(x) for x in best_seq]
