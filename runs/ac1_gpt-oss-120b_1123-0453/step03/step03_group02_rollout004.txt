def fft_convolve(a,b):

"""
Optimise a non‑negative step‑function (vector of heights) so that the
maximum entry of the auto‑convolution `a*a` is as small as possible while
the ℓ¹‑norm is fixed to √(2·n).  The optimisation uses

* a soft‑max re‑parameterisation a = S·softmax(v) (S = √(2·n)) to keep
  a on the simplex,
* a smoothed approximation of the max via the log‑sum‑exp with an
  increasing temperature β,
* Adam updates on the unconstrained logits v,
* several random restarts for different lengths n,
* a final cheap Gaussian hill‑climber that works directly with the
  provided `evaluate_sequence`.

The routine respects the 1000 s wall‑time limit and returns a plain
list of floats.
"""

import math
import time
import numpy as np


# ----------------------------------------------------------------------
# FFT convolution (full length) – used for both forward and gradient steps
# ----------------------------------------------------------------------
def _fft_convolve_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of a and b using FFT."""
    size = 1 << ((len(a) + len(b) - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    Fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(Fa * Fb, size)[: len(a) + len(b) - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (stable)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    v_max = np.max(v)
    exp_v = np.exp(v - v_max)
    return exp_v / np.sum(exp_v)


# ----------------------------------------------------------------------
# Gradient of the smoothed max w.r.t. a (the sequence)
# ----------------------------------------------------------------------
def _grad_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Given weight vector w (same length as the full auto‑convolution)
    compute ∇ₐ  Σ_i w_i·(a*a)_i .
    The result has length len(a).
    """
    # cross‑correlation of w with a:
    w_rev = w[::-1]
    conv = _fft_convolve_full(w_rev, a)                # length = 3·n‑2
    start = len(a) - 1                                 # centre index
    grad = 2.0 * conv[start : start + len(a)]
    return grad


# ----------------------------------------------------------------------
# Simple stochastic local search (Gaussian perturbations)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5,
                  decay: float = 0.9999) -> tuple[np.ndarray, float]:
    """
    Refine ``start_seq`` with cheap Gaussian moves.
    Returns the best vector seen (as a NumPy array) and its
    evaluate_sequence() score.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    # direct score: 2·n·max_b/(sum²) == max_b because sum = √(2·n)
    best_score = float(evaluate_sequence(best.tolist()))
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to the required ℓ¹‑norm (the evaluator also clips,
        # but keeping the norm fixed speeds up convergence)
        n = cand.shape[0]
        target = math.sqrt(2.0 * n)
        # project onto the simplex by scaling – simple because all entries ≥0
        s = np.sum(cand)
        if s > 0:
            cand = cand * (target / s)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        step *= decay
        if step < 1e-7:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value step sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Global time budget (seconds). The function stops a little before
        the limit to guarantee a return.

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # safety margin of a few hundred ms
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Load any previously discovered construction
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = seq0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        if np.sum(seq0) > 0:
            seq0 = seq0 / np.sum(seq0) * target0
            best_seq = seq0.copy()
            best_score = float(evaluate_sequence(seq0.tolist()))
            print(f"[init] loaded height_sequence_1 (n={n0}) → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Adam‑based optimisation on several lengths
    # --------------------------------------------------------------
    # lengths that give us enough freedom but still cheap to handle
    lengths = [800, 1200, 2000, 3000, 4000]
    for n in lengths:
        if time.monotonic() >= deadline:

        target_sum = math.sqrt(2.0 * n)

        # number of random restarts per length – adjusted by remaining time
        restarts = 3
        for restart in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise logits v
            if best_seq is not None and best_seq.shape[0] == n:
                # start from the current best (log‑soft‑max)
                eps = 1e-12
                s = best_seq / target_sum
                v = np.log(s + eps)
            else:
                v = rng.normal(size=n)

            beta1 = 0.9

                conv = _fft_convolve_full(a, a)

                conv_shift = conv - np.max(conv)        # for numerical stability

                if w_sum == 0.0:

                grad_a = _grad_from_weights(w, a)

                # ----- chain rule to obtain gradient w.r.t. logits v -----
                dot = np.dot(grad_a, s)                # scalar

                beta = min(beta_max, beta * beta_inc)

                if it % 500 == 0:

                # early exit if we have already beaten the target bound
                if best_score < 1.503:
                    print("[early] target 1.503 reached – exiting.")
                    return [float(x) for x in best_seq]

    # --------------------------------------------------------------
    # 2️⃣  Final polishing with a cheap stochastic hill‑climber
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:

                                               step_initial=0.5)
        if refined_score < best_score:

            best_score = refined_score
            print(f"[local] polished → score={best_score:.6f}")

    # 3️⃣  Final sanity fixes – clip & re‑normalise exactly on the simplex
    # --------------------------------------------------------------
    if best_seq is None:

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    n = best_seq.shape[0]
    target = math.sqrt(2.0 * n)
    s = np.sum(best_seq)
    if s > 0:
        best_seq = best_seq * (target / s)

    # return plain Python list as required by the harness
    return [float(x) for x in best_seq]
