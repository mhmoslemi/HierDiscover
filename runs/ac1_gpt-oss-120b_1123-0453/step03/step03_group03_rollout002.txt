def conv_matrix(f):
    n = len(f)
    rows = 2*n - 1
    A = np.zeros((rows, n))
    for j in range(n):
        start = j
        end = j + n
        A[start:end, j] = f
    return A

def find_best_t(f,g,R=1.0):
    # Use grid search of 21 points.
    best_t = 0.0
    best_M = np.max(np.convolve(f, f))
    for t in np.linspace(0,1,21):
        if t == 0: continue
        candidate = (1-t)*f + t*g
        M = np.max(np.convolve(candidate, candidate))
        if M < best_M:
            best_M=M
            best_t=t
    # optionally refine with local search around best_t
    # do ternary on interval [max(0,best_t-0.05), min(1,best_t+0.05)]
    lo = max(0.0, best_t-0.05)
    hi = min(1.0, best_t+0.05)
    for _ in range(10):
        t1 = lo + (hi-lo)/3
        t2 = hi - (hi-lo)/3
        M1 = np.max(np.convolve((1-t1)*f + t1*g, (1-t1)*f + t1*g))
        M2 = np.max(np.convolve((1-t2)*f + t2*g, (1-t2)*f + t2*g))
        if M1 < M2:
            hi = t2
            if M1 < best_M:
                best_M=M1
                best_t=t1
        else:
            lo = t1
            if M2 < best_M:
                best_M=M2
                best_t=t2
    return best_t, best_M

def greedy_local_search(seq, iterations, time_deadline):

    for it in range(iterations):
        if time.monotonic()>=deadline: break
        # compute conv and max index
        conv = np.convolve(seq, seq)
        max_idx = np.argmax(conv)
        # compute contributions per i
        i_arr = np.arange(n)
        j_arr = max_idx - i_arr
        mask = (j_arr >=0) & (j_arr < n)
        contrib = np.zeros_like(seq)
        contrib[mask]= seq[i_arr[mask]] * seq[j_arr[mask]]
        total_contrib = contrib.sum()
        if total_contrib==0: continue
        # choose src with prob proportional to contribution
        src = rng.choice(n, p=contrib/total_contrib)
        # choose dst randomly with low contribution
        low_thresh = np.percentile(contrib, 20)
        low_cand = np.where(contrib <= low_thresh)[0]
        if low_cand.size==0:
            dst = rng.integers(n)
        else:
            dst = rng.choice(low_cand)
        if src == dst: continue
        eps = rng.uniform(0, min(seq[src], 0.5*seq[src]))
        if eps <= 0: continue
        cand = seq.copy()
        cand[src] -= eps
        cand[dst] += eps
        M_new = np.max(np.convolve(cand, cand))
        if M_new < np.max(conv) - 1e-9:
            seq = cand
            # update best if improved
            M_current=M_new
            print(f"[greedy] improved M to {M_current:.6f}")
    return seq

"""
Propose a low‑score step‑function.

The algorithm proceeds in three phases for each candidate length *n*:
1. **LP refinement** –  solve   max Σ g_i   s.t. (f·g) ≤ M (where
   M = max(f*f)).  The LP is solved with scipy.optimize.linprog.
   The resulting g is rescaled to the ℓ¹‑norm √(2n) and mixed with f.
   The mixing weight *t* is optimised by a short grid + ternary search,
   which yields a better reduction of the maximal auto‑convolution peak
   than the coarse set used previously.
2. **Greedy peak reduction** –  identify the index of the current peak,
   move a small fraction of mass from a contributing index to a low‑
   contribution index, and accept the move only if the peak decreases.
3. **Stochastic polishing** –  add Gaussian perturbations with a
   slowly decaying step size; after each perturbation the vector is
   re‑normalised to the prescribed ℓ¹‑norm.  Any improvement updates the
   best solution.

All phases respect the non‑negativity and clipping constraints required
by ``evaluate_sequence``.  The routine tries a few different lengths
(500, 600, 800) and a few random restarts; the best sequence found before
the time budget expires is returned.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Optional LP back‑end
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog   # type: ignore
    _HAS_LP = True
except Exception:                           # pragma: no cover
    _HAS_LP = False

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_max(seq: np.ndarray) -> float:
    """Return max of the autocorrelation of ``seq``."""
    return float(np.max(np.convolve(seq, seq)))


def _score_from_seq(seq: np.ndarray) -> float:
    """Exact score used by the optimiser (identical to evaluate_sequence
    after normalising the ℓ¹‑norm to √(2n))."""
    n = seq.size
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    M = _conv_max(seq)
    return 2.0 * n * M / (s * s)


def _make_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix A with (A @ g) = f * g.
    Row k contains f shifted by k (zero‑padded).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        start = j
        end = j + n
        A[start:end, j] = f
    return A


def _best_mixing_t(f: np.ndarray, g: np.ndarray, init_M: float) -> tuple[float, float]:
    """
    Find a mixing weight t∈[0,1] that minimises max((1‑t)f + t g)².
    Uses a coarse grid followed by a short ternary search.
    Returns (t_best, M_best).
    """
    best_t = 0.0
    best_M = init_M

    # coarse grid
    for t in np.linspace(0.0, 1.0, 21)[1:]:
        cand = (1.0 - t) * f + t * g
        M = _conv_max(cand)
        if M < best_M:
            best_M = M
            best_t = t

    # refine around the best grid point
    lo = max(0.0, best_t - 0.05)
    hi = min(1.0, best_t + 0.05)
    for _ in range(10):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        M1 = _conv_max((1.0 - t1) * f + t1 * g)
        M2 = _conv_max((1.0 - t2) * f + t2 * g)
        if M1 < M2:
            hi = t2
            if M1 < best_M:
                best_M = M1
                best_t = t1
        else:
            lo = t1
            if M2 < best_M:
                best_M = M2
                best_t = t2
    return best_t, best_M


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that yields a low value of
    ``evaluate_sequence``.  The function respects the 1000 s time budget
    and returns the best list of floats found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # give a small safety margin for the caller
    deadline = start + max(0.0, float(budget_s) - 0.3)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to initialise a sequence of a given length
    # ------------------------------------------------------------------
    def init_sequence(n: int) -> np.ndarray:
        target = math.sqrt(2.0 * n)
        # Prefer the global construction if it matches the length
        if "height_sequence_1" in globals():
            try:
                seq0 = np.array(globals()["height_sequence_1"], dtype=float)
                if seq0.size == n:
                    seq0 = np.clip(seq0, 0.0, 1000.0)
                    seq0 = seq0 * (target / seq0.sum())
                    return seq0
            except Exception:
                pass
        # Otherwise start from a Dirichlet (roughly uniform) distribution
        seq = rng.dirichlet(np.ones(n))
        seq = seq * target
        return seq

    # ------------------------------------------------------------------
    # Lengths to try (larger n gives more flexibility but costs time)
    # ------------------------------------------------------------------
    candidate_lengths = [500, 600, 800]

    for n in candidate_lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        restarts = 4

        for restart in range(restarts):
            if time.monotonic() >= deadline:
                break

            f = init_sequence(n)
            # make sure we stay inside the allowed range
            f = np.clip(f, 0.0, 1000.0)

            cur_M = _conv_max(f)
            cur_score = cur_M                     # because sum == √(2n)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            # --------------------------------------------------------------
            # Phase 1 – LP based refinement
            # --------------------------------------------------------------
            if _HAS_LP:
                max_iter = 150 if n <= 600 else 120
                for it in range(max_iter):
                    if time.monotonic() >= deadline:
                        break

                    # recompute the worst peak
                    cur_M = _conv_max(f)

                    # Build the convolution matrix for the current f
                    A = _make_conv_matrix(f)

                    # Linear programme: maximise Σ g_i  s.t. A·g ≤ cur_M , g ≥ 0
                    c = -np.ones(n)                     # minimise –Σg_i
                    b = np.full(2 * n - 1, cur_M)

                    try:
                        res = linprog(c, A_ub=A, b_ub=b,
                                      bounds=(0, None),
                                      method='highs')
                    except Exception:
                        # fallback – give up on LP for this iteration
                        break

                    if not res.success or res.x is None:
                        break

                    g = np.maximum(res.x, 0.0)
                    sum_g = g.sum()
                    if sum_g < 1e-12:
                        break

                    # Rescale g to the fixed ℓ¹‑norm √(2n)
                    g = g * (target_sum / sum_g)

                    # Find the mixing weight that improves the peak most
                    t_opt, M_opt = _best_mixing_t(f, g, cur_M)

                    # If we cannot improve, stop the LP loop early
                    if M_opt >= cur_M - 1e-12:
                        break

                    # Apply the mixing
                    f = (1.0 - t_opt) * f + t_opt * g
                    f = np.clip(f, 0.0, 1000.0)

                    # Update the best overall record
                    if M_opt < best_score:
                        best_score = M_opt
                        best_seq = f.copy()
                        print(f"[LP] n={n} restart={restart} it={it} "
                              f"t={t_opt:.4f} M={M_opt:.6f}")

                    # Early exit if we reach the target bound
                    if best_score < 1.503:
                        print("[LP] target bound reached – exiting early")
                        return best_seq.tolist()

            # --------------------------------------------------------------
            # Phase 2 – Greedy peak reduction (mass transfer)
            # --------------------------------------------------------------
            seq = f.copy()
            n_seq = seq.size
            while time.monotonic() < deadline:
                conv = np.convolve(seq, seq)
                max_idx = int(np.argmax(conv))

                # contribution of each index to the current maximal entry
                i = np.arange(n_seq)
                j = max_idx - i
                mask = (j >= 0) & (j < n_seq)
                contrib = np.zeros_like(seq)
                contrib[mask] = seq[i[mask]] * seq[j[mask]]

                total = contrib.sum()
                if total == 0.0:
                    break

                # source index sampled proportional to its contribution
                src = rng.choice(n_seq, p=contrib / total)

                # pick a destination where the contribution is small
                low_thr = np.percentile(contrib, 25)
                low_cands = np.where(contrib <= low_thr)[0]
                if low_cands.size == 0:
                    dst = rng.integers(n_seq)
                else:
                    dst = rng.choice(low_cands)

                if src == dst:
                    continue

                # move a modest fraction of the source mass
                eps = rng.uniform(0.0, seq[src] * 0.35)
                if eps <= 0.0:
                    continue

                cand = seq.copy()
                cand[src] -= eps
                cand[dst] += eps
                cand = np.clip(cand, 0.0, 1000.0)

                M_new = _conv_max(cand)
                M_old = _conv_max(seq)
                if M_new < M_old - 1e-8:
                    seq = cand
                    if M_new < best_score:
                        best_score = M_new
                        best_seq = cand.copy()
                        print(f"[greedy] n={n} new best M={best_score:.6f}")
                    if best_score < 1.503:
                        print("[greedy] target reached")
                        return best_seq.tolist()

            # --------------------------------------------------------------
            # Phase 3 – Stochastic polishing (Gaussian perturbations)
            # --------------------------------------------------------------
            cur = best_seq.copy() if best_seq is not None else seq
            step = max(0.2, cur.mean() * 0.3)

            while time.monotonic() < deadline:
                cand = cur + rng.normal(scale=step, size=n)
                cand = np.clip(cand, 0.0, 1000.0)

                # Re‑normalise to the required ℓ¹‑norm
                s = cand.sum()
                if s == 0.0:

                cand = cand * (target_sum / s)
                cand = np.clip(cand, 0.0, 1000.0)

                M_cand = _conv_max(cand)
                if M_cand < best_score:
                    best_score = M_cand
                    best_seq = cand.copy()
                    cur = cand
                    print(f"[anneal] n={n} improved M={best_score:.6f}")

                # geometric decay of the perturbation size
                step *= 0.9996
                if step < 1e-6:
                    step = max(0.2, cur.mean() * 0.3)

            # If we already beat the target we can stop all work
            if best_score < 1.503:
                print("[overall] target reached – returning")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should never happen) return a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0, dtype=float)
        best_seq = best_seq / best_seq.sum() * math.sqrt(2.0 * n)

    return [float(x) for x in best_seq]
