"""
Propose a low‑evaluation step sequence.

The evaluation of a sequence ``a`` with ``evaluate_sequence`` reduces to

    score = 2·n·max_k (p * p)_k   where   p_i = a_i / sum(a) ,   sum(p)=1

Thus the problem is to choose a probability vector *p* whose
self‑convolution is as flat as possible.  The algorithm proceeds in three
phases:

1. **Arcsine start** – a U‑shaped distribution
   p_i ∝ 1/√(x_i(1‑x_i)) (x_i = (i+½)/n) has the property that the
   convolution of the continuous analogue is almost constant.
   The discretised version is used as a strong initial guess.

2. **Exponentiated‑gradient flattening** – at every iteration we locate the
   current maximal convolution peak(s) and take a (sub‑)gradient step
   that reduces the contribution of the indices that are responsible for
   that peak.  The step is performed multiplicatively
   (mirror‑descent on the simplex) and the vector is re‑normalised.
   A small amount of noise is added occasionally to escape plateaus.

3. **Final stochastic hill‑climb** – with the remaining time we perform
   tiny random mass transfers between two positions and keep any
   improvement.  This polishes the solution and compensates for any
   remaining discretisation error.

All operations respect the security checks of ``evaluate_sequence`` –
the returned list contains only non‑negative numbers (the sum is 1, so
the denominator never becomes too small).  The procedure uses only
``numpy`` and the provided ``evaluate_sequence`` and finishes well within
the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – fast local score (identical to evaluate_sequence when sum=1)
# ----------------------------------------------------------------------
def _local_score(p: np.ndarray) -> float:
    """Return 2·n·max(p*p) for a probability vector p (sum(p)=1)."""
    n = p.size
    conv = np.convolve(p, p)               # length 2·n‑1
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Arcsine‑type start distribution
# ----------------------------------------------------------------------
def _arcsine_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a probability vector of length n based on the continuous arcsine law."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                     # points in (0,1)
    p = 1.0 / np.sqrt(x * (1.0 - x))      # arcsine density, up‑to a constant
    p = p / p.sum()                       # normalise to a probability vector
    # tiny random jitter to break exact symmetry
    p += 1e-4 * rng.normal(size=n)
    p = np.clip(p, 0.0, None)
    p /= p.sum()
    return p

# ----------------------------------------------------------------------
# One gradient‑descent run on a given length
# ----------------------------------------------------------------------
def _run_gradient(p: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient steps that minimise max(p*p).
    Returns the best vector seen and its score.
    """
    n = p.size
    best_vec = p.copy()
    best_sc = _local_score(p)

    # step‑size schedule
    eta = 0.8
    decay = 0.9995

    while time.monotonic() < deadline:
        # convolution and current maximal value
        conv = np.convolve(p, p)
        M = conv.max()

        # early exit if we already beat the target bound (1.503)
        if best_sc < 1.503:
            break

        # indices where the maximum is attained (usually a single index)
        # a relative tolerance is safer than an absolute one
        tol = max(1e-14, M * 1e-12)
        peak_idx = np.where(conv >= M - tol)[0]

        # binary mask of the active peaks
        mask = np.zeros_like(conv)
        mask[peak_idx] = 1.0

        # gradient of the L∞‑norm (sub‑gradient):
        #   g_i = 2 * sum_{k in peaks} p_{k-i}
        # This can be computed via a convolution with the reversed mask.
        mask_rev = mask[::-1]
        L = conv.size                                   # = 2·n‑1
        rev_conv = np.convolve(p, mask_rev)            # length 3·n‑2
        g = 2.0 * rev_conv[(L - 1) - np.arange(n)]

        # exponentiated‑gradient (mirror descent on the simplex)
        mult = np.exp(-eta * g)
        p = p * mult
        # re‑normalise to stay on the probability simplex
        s = p.sum()
        if s == 0.0 or not np.isfinite(s):
            # numerical disaster – re‑initialise with a small random perturbation
            p = _arcsine_init(n, rng)
        else:
            p /= s

        # occasional stochastic perturbation to avoid getting stuck
        if rng.random() < 0.005:
            eps = 1e-4 * rng.normal(size=n)
            p = np.clip(p + eps, 0.0, None)
            p /= p.sum()

        # evaluate the new vector
        cur_sc = _local_score(p)
        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_vec = p.copy()
            print(f"[grad] n={n} score={best_sc:.6f}  eta={eta:.5f}")

        # step‑size decay
        eta *= decay
        if eta < 1e-6:
            # when the step becomes too tiny, stop the inner loop
            break

    return best_vec, best_sc

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine runs three phases (arcsine start, gradient flattening,
    stochastic hill climbing) inside the given time budget and returns
    the best list of floats found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the caller to fetch the result
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try a few different lengths; 500 is the length of the known best
    # construction, but 600 and 800 often give a finer discretisation.
    # ------------------------------------------------------------------
    lengths = [500, 600, 800]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # ---------- Phase 1 – arcsine start ----------
        p = _arcsine_init(n, rng)

        # ---------- Phase 2 – exponentiated‑gradient ----------
        p, sc = _run_gradient(p, deadline, rng)

        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
            print(f"[init] length={n}  score after gradient={sc:.6f}")

        # ---------- Phase 3 – final stochastic hill‑climb ----------
        # Use whatever time is left for a simple hill‑climbing search.
        cur = best_seq.copy()
        n = cur.size
        while time.monotonic() < deadline:
            # pick a source and a destination
            i_src = rng.integers(n)
            i_dst = rng.integers(n)
            if i_src == i_dst:
                continue
            # move at most 5 % of the source mass
            max_move = cur[i_src] * 0.05
            if max_move <= 0.0:
                continue
            move = rng.uniform(0.0, max_move)
            cand = cur.copy()
            cand[i_src] -= move
            cand[i_dst] += move
            # evaluate
            cand_sc = _local_score(cand)
            if cand_sc < best_score - 1e-12:
                best_score = cand_sc
                best_seq = cand.copy()
                cur = cand
                print(f"[hill] improved to {best_score:.6f}")
            # stop early if we have comfortably passed the target
            if best_score < 1.503:
                break

        # stop completely once the target is reached
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # ``evaluate_sequence`` will clip / normalise it again, so we do not
    # need to perform any extra scaling.
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – a uniform vector of length 500
        n = 500
        best_seq = np.full(n, 1.0 / n, dtype=float)

    return [float(x) for x in best_seq]
