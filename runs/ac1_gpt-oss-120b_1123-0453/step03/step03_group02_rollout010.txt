"""
Propose a low‑evaluation step sequence.

Algorithm
---------

* **Simplex projection** – keep the ℓ¹‑norm fixed to √(2·n) and all entries ≥0.
* **Smooth max‑flattening** – minimise a smooth approximation of
  `max_k (f*f)_k` using a soft‑max (log‑sum‑exp) surrogate.
  The gradient of the surrogate can be written as a convolution,
  which we evaluate with FFTs (O(n log n) per iteration).
  We increase the temperature parameter α progressively
  (α = 0.5 → 1 → 2 → … → 80) so the optimisation first sees a smooth
  landscape and later focuses on the true maximum.
* **Exact sub‑gradient polishing** – after the smooth phase we run a
  few thousand iterations of the classic sub‑gradient method on the
  exact max to tidy up the solution.
* **Stochastic hill‑climbing** – a final short Gaussian perturbation
  phase (optional) to escape any tiny remaining local minima.

The routine tries several vector lengths (up to a few thousand);
larger lengths give more freedom and can push the maximal
convolution close to its theoretical lower bound (≈ 1).  All
operations respect the security checks of `evaluate_sequence`,
and the function returns the best list found before the 1000 s
budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – evaluation wrapper
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Call the provided `evaluate_sequence`."""
    # The harness guarantees that evaluate_sequence is available.
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Simple ℓ¹‑simplex projection (non‑negative, fixed sum)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project `v` onto {x ≥ 0, Σ x = target}."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all will be zero
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full auto‑convolution (f*f) – returns the whole vector
# ----------------------------------------------------------------------
def _full_convolution_fft(f: np.ndarray) -> np.ndarray:
    """Full convolution of `f` with itself using a power‑of‑two FFT."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max surrogate of the max and its gradient
# ----------------------------------------------------------------------
def _softmax_grad(f: np.ndarray, alpha: float):
    """
    For a given `alpha` compute:
        conv = f * f                (full convolution)
        w    = softmax(alpha * conv)
        grad = 2 * (w  *  rev_f)_{n-1 .. 2n-2}
    where * denotes (full) convolution and rev_f = f[::-1].

    Returns (conv, grad).  Both are real 1‑d numpy arrays.
    """
    n = f.shape[0]

    # 1) full convolution f*f
    conv = _full_convolution_fft(f)

    # 2) soft‑max weights (stable)
    z = alpha * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / np.sum(exp_z)          # length 2n‑1, non‑negative, sum=1

    # 3) gradient = 2 * (w * rev_f) slice
    rev_f = f[::-1]
    size = 1 << ((len(w) + len(rev_f) - 1).bit_length())
    W = np.fft.rfft(w, size)
    R = np.fft.rfft(rev_f, size)
    wr = np.fft.irfft(W * R, size)[: len(w) + len(rev_f) - 1]
    grad = 2.0 * wr[n - 1 : n - 1 + n]   # n‑wide slice
    return conv, grad


# ----------------------------------------------------------------------
# Smooth (soft‑max) descent on a single length
# ----------------------------------------------------------------------
def _smooth_descent(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    alphas=None,
    step0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max_k (f*f)_k.

    Parameters
    ----------
    f0 : ndarray
        Starting non‑negative vector (already normalised to `target_sum`).
    target_sum : float
        Desired ℓ¹‑norm (√(2·n)).
    deadline : float
        Absolute time (monotonic) after which the routine must return.
    alphas : list[float] | None
        Sequence of temperature parameters; if None a default schedule is used.
    step0 : float
        Initial step size (will decay as 1/√t).

    Returns
    -------
    best_f : ndarray
        The best vector encountered (in terms of true max).
    best_M : float
        Its maximal auto‑convolution value.
    """
    if alphas is None:
        alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

    n = f0.shape[0]
    f = f0.copy()
    best_f = f.copy()
    best_M = np.inf
    it = 0

    for alpha in alphas:
        while time.monotonic() < deadline:
            it += 1
            conv, grad = _softmax_grad(f, alpha)

            cur_M = np.max(conv)
            if cur_M < best_M:
                best_M = cur_M
                best_f = f.copy()

            # step size decays like 1/sqrt(iter)
            eta = step0 / math.sqrt(it)

            # gradient step + projection
            f = f - eta * grad
            f = np.clip(f, 0.0, 1000.0)
            f = _project_simplex(f, target_sum)

            # if we have exhausted the inner loop without improvement for a while,
            # we simply move on to the next alpha (the outer loop will continue
            # anyway).  No explicit early‑stopping here.
    return best_f, best_M


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for polishing)
# ----------------------------------------------------------------------
def _exact_subgradient_descent(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 3000,
    eta0: float = 0.8,
    decay: float = 0.9997,
    min_step: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on `M(f)=max_k (f*f)_k`.
    Returns the best vector seen and its max‑convolution value.
    """
    n = f0.shape[0]
    f = f0.copy()
    best_f = f.copy()
    best_M, _ = _fft_conv_max(f)

    eta = eta0
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        M, k = _fft_conv_max(f)

        if M < best_M:
            best_M = M
            best_f = f.copy()

        # sub‑gradient at index k
        g = np.zeros_like(f)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        if low <= high:
            idx = np.arange(low, high + 1, dtype=int)
            g[idx] = 2.0 * f[k - idx]

        f = _project_simplex(f - eta * g, target_sum)

        # simple step‑size decay / restart
        eta = max(eta * decay, min_step)

    return best_f, best_M


# ----------------------------------------------------------------------
# Fast max‑convolution (used by the exact sub‑gradient routine)
# ----------------------------------------------------------------------
def _fft_conv_max(f: np.ndarray):
    """Return (max_value, argmax_index) of the full auto‑convolution f*f."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Small stochastic local search (optional polishing)
# ----------------------------------------------------------------------
def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.6,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to the required ℓ¹‑norm
        target = math.sqrt(2.0 * cand.shape[0])
        cand = _project_simplex(cand, target)

        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

        step *= decay
        if step < 1e-7:
            step = step_initial

    return best, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The routine explores several lengths, runs a smooth‑max descent
    (log‑sum‑exp surrogate) with an increasing temperature schedule,
    then polishes the best candidate with an exact sub‑gradient method
    and a short stochastic hill‑climbing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)  # safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0️⃣  Initialise from any known construction
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = seq.shape[0]
        target0 = math.sqrt(2.0 * n0)
        if np.sum(seq) > 0:
            seq = seq / np.sum(seq) * target0
            best_seq = seq.copy()
            best_score = _score(seq)
            print(f"[init] Loaded height_sequence_1 (n={n0}) → score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 1️⃣  Smooth‑max descent on a range of lengths
    # -----------------------------------------------------------------
    lengths = [800, 1200, 1500, 2000, 3000, 4000]   # larger n ⇒ more freedom
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # start vectors: a few random + possibly the current global best padded
        starts = []

        if best_seq is not None:
            if best_seq.shape[0] == n:
                starts.append(best_seq.copy())
            elif best_seq.shape[0] < n:
                pad = np.zeros(n, dtype=np.float64)
                pad[: best_seq.shape[0]] = best_seq
                pad = _project_simplex(pad, target_sum)
                starts.append(pad)

        # a small number of fresh random restarts
        for _ in range(3):
            f0 = rng.random(n) + 0.001
            f0 = f0 / np.sum(f0) * target_sum
            starts.append(f0)

        for start_id, f_start in enumerate(starts):
            if time.monotonic() >= deadline:
                break

            # allocate a fraction of the remaining budget to this run
            now = time.monotonic()
            remaining = deadline - now
            if remaining <= 0:
                break
            run_deadline = now + min(30.0, remaining * 0.3)   # ≤30 s or 30 % of the rest

            # ----------- smooth‑max phase -------------
            f_opt, M_opt = _smooth_descent(
                f_start,
                target_sum=target_sum,
                deadline=run_deadline,
            )
            sc = _score(f_opt)
            if sc < best_score:
                best_score = sc
                best_seq = f_opt.copy()
                print(f"[smooth] n={n} start={start_id} → score={sc:.6f} max={M_opt:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                print("[smooth] Target reached – exiting early.")
                return best_seq.tolist()

    # -----------------------------------------------------------------
    # 2️⃣  Exact sub‑gradient polishing (if we have any candidate)
    # -----------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        sub_deadline = min(deadline, time.monotonic() + max(5.0, (deadline - time.monotonic()) * 0.2))
        f_polished, M_polished = _exact_subgradient_descent(
            best_seq,
            target_sum=target_sum,
            deadline=sub_deadline,
            max_iters=4000,
        )
        sc = _score(f_polished)
        if sc < best_score:
            best_score = sc
            best_seq = f_polished.copy()
            print(f"[polish] sub‑gradient → score={sc:.6f} max={M_polished:.6f}")

    # -----------------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climbing (optional)
    # -----------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined_seq, refined_score = _local_search(
            best_seq,
            deadline,
            step_initial=0.5,
        )
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"[local] polished → score={best_score:.6f}")

    # If for any reason we never produced a vector, fall back to a uniform one.
    if best_seq is None:
        n = 800
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = _project_simplex(best_seq, math.sqrt(2.0 * n))
        best_score = _score(best_seq)
        print("[fallback] uniform vector generated.")

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
