def improve_via_lp(a, S, deadline, max_iter=200):
    best_a = a.copy()
    best_score = _score(best_a)
    n = len(a)
    t = 0.2
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        # compute M
        conv = np.convolve(a, a)
        M = conv.max()
        if M <= 0: break
        # Build matrix
        B = np.zeros((2*n-1, n), dtype=float)
        for i in range(n):
            B[i:i+n, i] = a
        # Solve LP: max sum(b) s.t. B @ b <= M
        c = -np.ones(n)  # minimize -sum(b)
        bounds = [(0, None) for _ in range(n)]
        res = scipy.optimize.linprog(c, A_ub=B, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        g = b * (S / sum_b)
        # convex combination
        a = (1 - t) * a + t * g
        # maybe renormalize sum due to numerical error:
        a = a * (S / a.sum())
        # compute new score
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # optional: reduce t
            t *= 0.8
        else:
            # maybe increase t? or reduce step
            t *= 0.9
        # early termination if best_score < target
        if best_score < 1.503:
            return best_a, best_score
    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

under the safety checks of `evaluate_sequence`.  

The algorithm consists of three stages for each candidate length *n*:

1️⃣ **Initialisation** – a few simple seeds (uniform, random, possibly the
   previously found `height_sequence_1`).  All seeds are rescaled so that
   Σ a = √(2 n).

2️⃣ **LP‑driven minimisation** – given a current vector `a`, we compute
   `M = max(conv(a,a))`.  Keeping `M` fixed we solve the linear program  

        maximise   1ᵀb
        subject to B·b ≤ M ,  b ≥ 0

   where `B` is the convolution matrix (`(B·b)_k = Σ_i a_i·b_{k‑i}`).
   The optimum `b` has the largest possible ℓ¹‑norm while still respecting
   the same convolution bound `M`.  Scaling `b` back to the required sum
   shrinks all entries and consequently reduces `M`.  We replace `a` by a
   convex combination ``a←(1‑t)·a + t·g`` (with `g` the scaled `b`) and
   repeat – this is exactly the iteration described in the problem
   statement, but here we rely on a fast LP solver (`scipy.optimize.linprog`).

   The LP loop stops when no progress is made or the time budget is
   exhausted.

3️⃣ **Fine‑grained polishing** – a cheap hill‑climbing that moves a tiny
   amount of mass from one index to another while keeping the total
   ℓ¹‑norm unchanged (implemented with an O(n) incremental convolution
   update).  After that we run a short Gaussian perturbation
   (`_local_search`) to escape shallow minima.

The whole search respects the 1000 s budget; the best vector found is
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official score (calls the provided `evaluate_sequence`)."""
    return float(evaluate_sequence(seq.tolist()))


def _cheap_score(a: np.ndarray, sum_a: float) -> float:
    """Score without the clipping performed inside `evaluate_sequence`.
    It is equivalent when the vector already satisfies the checks."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a * sum_a))


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index `i` to index `j` (i != j, delta > 0, delta <= a[i]).
    Updates `a` and its convolution `conv` in‑place (O(n)).
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    delta = min(delta, a[i])

    # linear part 2*conv(d,a)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part conv(d,d)
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta          # (-δ)²
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta    # (-δ)*(+δ) + (+δ)*(-δ)
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta          # (+δ)²

    a[i] -= delta
    a[j] += delta


def _improve_by_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing that moves a small amount of mass between two
    indices while keeping the ℓ¹‑norm fixed.  Returns the best vector seen."""
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)
    sum_a = a.sum()
    best_a = a.copy()
    best_score = _cheap_score(a, sum_a)

    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        # pick a different target index
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        _transfer_mass(a, conv, i, j, delta)
        cur_score = _cheap_score(a, sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert
            _transfer_mass(a, conv, j, i, delta)

        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """Gentle Gaussian perturbation used as a final polishing step."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial

    return best, best_score


def _lp_improve(a_start: np.ndarray,
                deadline: float,
                max_outer: int = 30,
                t_init: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Iterative LP based improvement.
    At each outer iteration we compute M = max(conv(a,a)) and solve

        maximise   Σ b_i
        s.t.       B·b ≤ M,   b ≥ 0

    where B is the convolution matrix built from the current `a`.
    The solution `b` is scaled to the required ℓ¹‑norm and mixed with the
    current vector.  The process stops when no further improvement is
    observed or the time budget expires.
    """
    a = a_start.copy()
    n = a.shape[0]
    target_sum = a.sum()                     # already = √(2·n)
    best_a = a.copy()
    best_score = _cheap_score(a, target_sum)

    t = t_init
    for outer in range(max_outer):
        if time.monotonic() >= deadline:
            break

        # convolution and current bound M
        conv = np.convolve(a, a)
        M = conv.max()

        # Build convolution matrix B (shape (2n‑1, n)):
        #   B[k, i] = a[i]  if 0 ≤ k‑i < n   else 0
        B = np.zeros((2 * n - 1, n), dtype=float)
        for i in range(n):
            B[i:i + n, i] = a

        # Linear program: maximise 1ᵀb   ⇔   minimise −1ᵀb
        c = -np.ones(n)
        bounds = [(0, None) for _ in range(n)]
        res = linprog(c,
                      A_ub=B,
                      b_ub=np.full(2 * n - 1, M),
                      bounds=bounds,
                      method='highs')

        # If LP fails or is unbounded we abort this outer loop.
        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # Scale to the exact ℓ¹‑norm required for the objective.
        g = b * (target_sum / sum_b)

        # Convex blend (ensure positivity)
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 1e-12)          # avoid exact zeros (prevent unbounded LP)
        a = a * (target_sum / a.sum())    # renormalise (numerical drift)

        # cheap score with the new vector
        cur_score = _cheap_score(a, target_sum)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # as we get closer to the optimum we shrink the mixing factor
            t = max(0.01, t * 0.8)
        else:
            # no improvement – decay the step size more aggressively
            t *= 0.9

        # early exit if we already beat the target bound
        if best_score < 1.503:
            break

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,

    """
    Search for a non‑negative step sequence with a low evaluation score.

    """
    rng = np.random.default_rng(seed)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Candidate lengths (larger n gives more flexibility)
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000]

        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)          # Σ a_i must be √(2·n)

        # ------------------------------------------------------------------
        # 2️⃣  Build a few different initial vectors for this length
        # ------------------------------------------------------------------
        starts: list[np.ndarray] = []

        #   * possibly reuse a global solution from a previous run
        try:
            init_global = np.array(height_sequence_1, dtype=float)
            if init_global.size == n:
                starts.append(init_global.copy())
        except Exception:
            pass

        #   * uniform (guaranteed feasible)
        starts.append(np.full(n, target_sum / n, dtype=float))

        #   * a few random seeds (exponential gives a sparse shape)
        for _ in range(3):
            if rng.random() < 0.5:

            else:
                vec = rng.random(n) + 0.1
            vec = np.clip(vec, 0.0, 1000.0)
            vec = vec * (target_sum / vec.sum())
            starts.append(vec)

        # ------------------------------------------------------------------
        # 3️⃣  Run the LP‑driven optimisation on each start
        # ------------------------------------------------------------------
        for start in starts:
            if time.monotonic() >= deadline:
                break

            # keep the vector strictly positive to avoid LP unboundedness
            start = np.maximum(start, 1e-12)

            improved_lp, cheap_score = _lp_improve(start, deadline,
                                                   max_outer=30,
                                                   t_init=0.2)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 5️⃣  Final Gaussian polishing (cheap, can only improve)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        refined, refined_score = _local_search(best_seq, deadline,
                                               step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final polishing] improved to {best_score:.6f}")

    # Return a plain list as required by the evaluator
    return [float(x) for x in best_seq]
