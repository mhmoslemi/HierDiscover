    a = np.full(n, target_sum / n)
    # optionally random perturb to break symmetry
    a += rng.normal(scale=target_sum / (10 * n), size=n)

    a_sum = a.sum()

    best_a = a.copy()

    # temperature schedule

    # step size scale fraction of a_i
    while time.monotonic() < deadline:
        # compute remaining fraction for schedule
        elapsed = time.monotonic() - start_time_const
        frac = (deadline - time.monotonic()) / (deadline - start_time_const)
        T = T0 * frac

        conv_candidate[2*i] += delta * delta
        conv_candidate[2*j] += delta * delta

        cand_score = 2.0 * n * cand_max / (target_sum * target_sum) # but equals cand_max

            a[j] += delta
            conv = conv_candidate
            cur_score = cand_score
            if cur_score < best_score:

"""
Propose a low‑value non‑negative step sequence.

Algorithm
---------
1.  Initialise with the historic construction `height_sequence_1` when it exists
    (normalised to ∑a_i = √(2 n)).  This gives a solid warm‑start.

2.  For several lengths n (400 … 2000) run a **simulated‑annealing** search
    on the ℓ¹‑simplex.  The state consists of the vector `a` and its full
    auto‑convolution `c = a ∗ a`.  A move transfers a small amount `δ` from one
    index i to another index j; the convolution can be updated in O(n) time by
    the exact formula derived from (a+Δ)*(a+Δ) = a∗a + 2 a∗Δ + Δ∗Δ.
    The move is accepted if it improves the maximal entry of the convolution,
    otherwise with probability exp(−Δ/ T) where the temperature T cools linearly.

3.  After the annealing phase a short **projected‑gradient descent** (the code
    from the previous state) is applied as a polish step.  This combines the
    global exploration of the annealer with the fast local optimisation of the
    gradient method.

4.  The best vector found is finally re‑normalised to the exact simplex
    ∑a_i = √(2 n) and clipped to the limits required by `evaluate_sequence`.

The search respects the 1000 s time budget; it stops early if a score below
1.503 is discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _compute_score_from_vector(f: np.ndarray) -> float:
    """
    Return the value of `evaluate_sequence` for a vector with the correct
    normalisation (∑f = √(2 n)).  Because the scaling is homogeneous the score
    equals the maximal entry of the auto‑convolution.
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = conv.max()
    # The denominator  (∑f)² equals 2 n  after the normalisation, therefore
    # the evaluate_sequence formula reduces to max_val.
    return float(max_val)


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex.
    The vector a always satisfies ∑a = √(2 n) and a_i ≥ 0.
    The state also stores its auto‑convolution `conv`.  A move transfers a small
    amount δ from index i to index j; the convolution is updated analytically.
    """
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    a = a0.copy()
    # ensure exact normalisation (floating point drift)
    a *= target_sum / a.sum()
    conv = np.convolve(a, a)                     # length 2n‑1
    cur_score = conv.max()
    best_a = a.copy()
    best_score = cur_score

    start = time.monotonic()
    T0 = 0.08                                    # initial temperature

    while time.monotonic() < deadline:
        # linear cooling
        elapsed = time.monotonic() - start
        total = deadline - start
        T = T0 * max(0.0, 1.0 - elapsed / total)

        # --- propose a move -------------------------------------------------
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1                               # guarantee i != j

        if a[i] <= 0.0:
            continue

        # move a fraction (up to 30 %) of the source mass
        max_delta = a[i]
        delta = rng.random() * max_delta * 0.30

        # ---- compute the candidate convolution ----------------------------
        # The update formula is:
        #   conv' = conv - 2 δ·shift_i(a) + 2 δ·shift_j(a) + Δ∗Δ
        # where Δ is the sparse vector having -δ at i and +δ at j.
        conv_cand = conv.copy()
        conv_cand[i:i + n] -= 2.0 * delta * a
        conv_cand[j:j + n] += 2.0 * delta * a

        # Δ∗Δ contributes +δ² at 2i and 2j, and –2δ² at i+j (i ≠ j)
        conv_cand[2 * i] += delta * delta
        conv_cand[2 * j] += delta * delta
        conv_cand[i + j] -= 2.0 * delta * delta

        cand_score = conv_cand.max()

        # ---- Metropolis acceptance -----------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability exp(−Δ/T); protect against T≈0
            if T > 1e-12 and rng.random() < math.exp(-(cand_score - cur_score) / T):
                accept = True

        if accept:
            # apply the move
            a[i] -= delta
            a[j] += delta
            conv = conv_cand
            cur_score = cand_score

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


def _gradient_descent(f0: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    The projected‑gradient routine from the previous state of the art.
    It is kept as a polishing step after the annealer.
    """
    def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
        if s < 0:
            raise ValueError("simplex sum must be non‑negative")
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        if not np.any(cond):
            return np.full_like(v, s / len(v))
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
        eps = max_val * 1e-9 + 1e-12
        return np.where(conv >= max_val - eps)[0]

    def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
        n = len(f)
        grad = np.zeros_like(f)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1)
            j = k - i
            vals = f[j]
            grad[i] += 2.0 * vals
            mask = (i == j)
            if np.any(mask):
                grad[i[mask]] -= vals[mask]
        return grad

    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score_from_vector(f)
    step = target_sum * 0.10
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        cur_score = _compute_score_from_vector(f)

        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 300:
                break

        conv = np.convolve(f, f)
        max_val = conv.max()
        active = _active_offsets(conv, max_val)
        grad = _gradient_submax(f, active)

        f_candidate = _project_onto_simplex(f - step * grad, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cand_score = _compute_score_from_vector(f_candidate)

        if cand_score < cur_score:
            f = f_candidate
            step *= 1.02
        else:
            step *= 0.5

    return best_f, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step sequence with a low `evaluate_sequence` value.
    The function respects the time budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # tiny safety margin for the final return
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣ Warm‑start from the historic construction (if any)
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        n0 = len(seq)
        target0 = math.sqrt(2.0 * n0)
        if seq.sum() == 0:
            seq = np.full(n0, target0 / n0)
        else:
            seq = seq * (target0 / seq.sum())
        score0 = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score0
        print(f"[init] height_sequence_1 length={n0} score={score0:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Simulated‑annealing phase on several lengths
    # --------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000, 1200, 1500, 2000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # allocate a slice of the remaining budget (max 20 s per length)
        length_deadline = min(deadline,
                              time.monotonic() + min(20.0,
                                                     (deadline - time.monotonic()) * 0.2))

        for restart in range(3):
            if time.monotonic() >= deadline:
                break

            # initialise the start vector
            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
            else:
                a0 = rng.random(n)
                a0 = a0 / a0.sum() * target_sum
                # a tiny random perturbation to break symmetry
                a0 += rng.normal(scale=target_sum / (10 * n), size=n)
                a0 = np.clip(a0, 0.0, None)
                a0 = a0 / a0.sum() * target_sum

            a_opt, score_opt = _simulated_annealing(a0, n, length_deadline, rng)

            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[anneal] n={n} restart={restart} score={score_opt:.6f}")

            if best_score < 1.503:
                print("[early] target achieved during annealing")
                break

        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # 2️⃣ Short projected‑gradient polishing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        gd_deadline = min(deadline, time.monotonic() + 8.0)
        best_seq, best_score = _gradient_descent(best_seq,
                                                 target_sum,
                                                 gd_deadline,
                                                 max_iters=4000)
        print(f"[gradient] polished length={n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Final normalisation / safety clipping
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: a modest uniform construction
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)

    # re‑normalise exactly to the required ℓ¹‑norm and clip
    target_sum = math.sqrt(2.0 * len(best_seq))
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    best_seq *= target_sum / best_seq.sum()

    return [float(x) for x in best_seq]
