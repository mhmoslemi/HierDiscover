def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2·n‑1) × n matrix A such that (A @ g)[k] = (f ∗ g)[k]."""
    n = f.shape[0]

        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}.
    Implementation of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]

    ind = np.arange(1, n + 1, dtype=np.float64)

        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)

    return w
