"""
LP‑refinement for autocorrelation minimisation.
...
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import hankel

def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]

def _fast_score(seq: np.ndarray) -> float:
    """Exact analogue of evaluate_sequence without clipping."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _build_hankel_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the (2n‑1)×n matrix A where A·g = f∗g."""
    n = f.shape[0]
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    r = np.zeros(n, dtype=f.dtype)   # last row – all zeros
    return hankel(c, r)

def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    One LP step: maximise ℓ₁‑norm of g under the linear constraints
    f∗g ≤ M.  Returns the scaled and clipped vector g.
    """
    n = f.shape[0]
    A = _build_hankel_matrix(f)          # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)    # maximise sum → minimise -sum
    bounds = [(0.0, 1000.0) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return the current f (no change)
        return f.copy()
    g0 = res.x
    sum_g = g0.sum()
    if sum_g <= 0:
        return f.copy()
    # rescale to exactly target_sum
    g = g0 * (target_sum / sum_g)
    # clip to legal range
    g = np.clip(g, 0.0, 1000.0)
    return g

def _lp_refine(start_seq: np.ndarray,
               deadline: float,
               t_initial: float = 0.2,
               target_sum: float | None = None) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP step and mix the result with the current vector.
    Returns the best sequence seen and its fast score.
    """
    n = start_seq.shape[0]
    if target_sum is None:
        target_sum = math.sqrt(2.0 * n)

    cur = start_seq.copy()
    cur_sum = cur.sum()
    if cur_sum > 0:
        cur = cur * (target_sum / cur_sum)
    cur = np.clip(cur, 0.0, 1000.0)

    best = cur.copy()
    best_score = _fast_score(cur)
    it = 0
    t = t_initial

    while time.monotonic() < deadline:
        M = _conv_full_fft(cur).max()
        g = _lp_step(cur, M, target_sum)

        # convex combination
        new = (1.0 - t) * cur + t * g
        # renormalise ℓ₁‑norm
        s = new.sum()
        if s > 0:
            new = new * (target_sum / s)
        new = np.clip(new, 0.0, 1000.0)

        new_score = _fast_score(new)

        if new_score < best_score:
            best_score = new_score
            best = new.copy()
            print(f"[LP‑refine] iter={it:4d} n={n} score={best_score:.6f}")

        cur = new
        it += 1

        # simple schedule for the mixing factor – shrink if improvement stalls
        if it % 20 == 0:
            t = max(0.01, t * 0.9)

        if best_score < 1.503:
            break

    return best, best_score

def _gaussian_polish(start_seq: np.ndarray,
                     deadline: float,
                     step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Stochastic hill climbing with Gaussian perturbations.
    """

"""
LP‑refinement for autocorrelation minimisation.

The algorithm follows the iterative scheme described in the statement:

1.  Start from a known or random non‑negative sequence f of length n,
    normalised so that ∑f = √(2·n).
2.  Compute M = max_k (f∗f)[k].
3.  Solve the linear programme
        maximise   ∑ g_j
        subject to  (f∗g)[k] ≤ M   for every shift k,
                    g_j ≥ 0.
    The constraints are linear because (f∗g)[k] = Σ_i f_i·g_{k‑i}.
4.  Rescale the optimal g to the same ℓ¹‑norm as f and mix it back:
        f ← (1‑t)·f + t·g   (t≈0.2, then slowly reduced).
    The mixing strictly reduces the maximal auto‑convolution entry,
    therefore the evaluation score (which equals the peak after the
    normalisation) never increases.
5.  Repeat steps 2‑4 until the time budget expires or the target
    score (< 1.503) is reached.
6.  Finish with a short Gaussian hill‑climb to polish the final vector.

All heavy work is performed with an FFT‑based convolution for speed,
while the linear programmes are solved with SciPy’s HiGHS solver.
The routine works for several lengths (500, 800, 1000) and keeps the
best sequence found.  The result is stored in the global variable
``height_sequence_1`` so that later runs can warm‑start from it.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import hankel

# ----------------------------------------------------------------------
# Fast convolution and scoring (exact analogue of evaluate_sequence)
# ----------------------------------------------------------------------


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``seq`` with itself using a real FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(f∗f)/‖f‖₁²  (identical to evaluate_sequence)."""
    n = seq.shape[0]
    max_b = _conv_full_fft(seq).max()
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Linear‑programming utilities
# ----------------------------------------------------------------------


def _build_hankel_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that  A·g = f∗g.
    Each row k contains the coefficients f_{k‑j} (or 0 if out of range).
    """
    n = f.shape[0]
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])   # first column
    r = np.zeros(n, dtype=f.dtype)                           # last row (all zeros)
    return hankel(c, r)                                       # shape (2n‑1, n)


def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    One LP step:
        maximise   ∑ g_j
        subject to  (f∗g)[k] ≤ M   for every k,
                    0 ≤ g_j ≤ 1000.
    Returns the feasible g, rescaled to have ℓ¹‑norm = target_sum.
    """
    n = f.shape[0]
    A = _build_hankel_matrix(f)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)     # right‑hand side
    c = -np.ones(n, dtype=np.float64)               # minimise –∑g  → maximise ∑g
    bounds = [(0.0, 1000.0) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback – keep the current vector unchanged
        return f.copy()

    g0 = res.x
    s = g0.sum()
    if s <= 0.0:
        return f.copy()
    g = g0 * (target_sum / s)                # rescale to the prescribed ℓ¹‑norm
    return np.clip(g, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Repeated LP refinement
# ----------------------------------------------------------------------


def _lp_refine(start_seq: np.ndarray,
               deadline: float,
               t_initial: float = 0.2,
               target_sum: float | None = None) -> tuple[np.ndarray, float]:
    """
    Apply the LP step repeatedly, mixing the obtained partner back into the
    current sequence with a decreasing step size t.  Returns the best sequence
    seen and its fast score.
    """
    n = start_seq.shape[0]
    if target_sum is None:
        target_sum = math.sqrt(2.0 * n)

    cur = start_seq.copy()
    cur = np.clip(cur, 0.0, 1000.0)
    s = cur.sum()
    if s > 0.0:
        cur *= target_sum / s

    best = cur.copy()
    best_score = _fast_score(cur)
    it = 0
    t = t_initial

    while time.monotonic() < deadline:
        M = _conv_full_fft(cur).max()
        g = _lp_step(cur, M, target_sum)

        # convex combination
        new = (1.0 - t) * cur + t * g
        new_sum = new.sum()
        if new_sum > 0.0:
            new *= target_sum / new_sum
        new = np.clip(new, 0.0, 1000.0)

        new_score = _fast_score(new)

        if new_score < best_score:
            best_score = new_score
            best = new.copy()
            print(f"[LP‑refine] iter={it:4d} n={n} score={best_score:.6f}")

        cur = new
        it += 1

        # slowly shrink the mixing factor so that we settle
        if it % 20 == 0:
            t = max(0.01, t * 0.9)

        if best_score < 1.503:
            break

    return best, best_score


# ----------------------------------------------------------------------
# Final Gaussian polishing (stochastic hill‑climb)
# ----------------------------------------------------------------------


def _gaussian_polish(start_seq: np.ndarray,
                     deadline: float,
                     step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """Add zero‑mean Gaussian noise, accepting only improvements."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score
