def project_onto_simplex(v, z):
    """Euclidean projection of vector v onto the simplex {x >= 0, sum x = z}."""
    # from Duchi et al. (2008)
    if v.sum() == z and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - z) / np.arange(1, len(v)+1))[0][-1]
    theta = (cssv[rho] - z) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative vector) with a low
evaluate_sequence() score.

Algorithm (≈1000 s budget)
--------------------------------
1. **Initialisation** – use the globally supplied ``height_sequence_1``
   if present; otherwise start from a uniform vector whose ℓ¹‑norm is
   √(2·n).  The sum is kept fixed (simplex projection) because with
   this normalisation the evaluation score equals the maximal
   auto‑convolution value M = max(f*f).

2. **Sub‑gradient descent on the peak** – while time permits:
   * Compute the auto‑convolution and the current worst index(s) k*.
   * Build a (sub‑)gradient of max‑convolution:
        ∂M/∂a_i = Σ_{k∈active} a_{k‑i}  (indices out of range are ignored).
   * Take a step opposite to the gradient, project back onto the
     simplex {x≥0, Σx = √(2·n)} and keep the step if the new peak M
     decreased, otherwise shrink the step size.
   * If no improvement is seen for many iterations, inject a small
     random perturbation to escape a local minimum.

3. **Greedy peak reduction** – after the sub‑gradient phase,
   repeatedly move a modest fraction of mass from an index that
   contributes heavily to the current peak to an index that contributes
   little, re‑project onto the simplex, and keep the move only if it
   reduces the true evaluation score.

4. **Stochastic polishing** – finally apply Gaussian perturbations
   with an exponentially decaying step size; each candidate is
   projected onto the simplex and accepted when it improves the score.

All steps respect the security checks of ``evaluate_sequence``:
non‑negative entries, clipping to ≤ 1000, and the ℓ¹‑norm is never
allowed to approach zero.  The routine prints progress and returns the
best sequence found before the time budget expires.
"""

import time
import math
import numpy as np

# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # evaluate_sequence expects a Python list of floats.
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        S = { x | x_i >= 0, Σ x_i = z }.
    Implementation of the algorithm from Duchi et al. (2008).
    """
    if z < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # If already feasible, return early
    if np.all(v >= 0) and np.isclose(v.sum(), z):
        return v.copy()

    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        tau = 0.0
    else:
        rho = rho[-1]
        tau = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - tau, 0.0)
    return w


def _subgradient_opt(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform sub‑gradient descent on the maximal auto‑convolution entry.
    Returns the best sequence seen during the run and its score.
    """
    n = seq.size
    best_seq = seq.copy()
    # With the fixed ℓ¹‑norm the evaluation score equals max(conv)
    best_score = np.max(np.convolve(seq, seq))

    # Initial step size (a fraction of the total mass)
    step = max(0.1, target_sum * 0.05)

    it = 0
    no_improve = 0
    while time.monotonic() < deadline and it < 5000:
        conv = np.convolve(seq, seq)
        M = float(conv.max())
        # The true score for this normalised vector is exactly M
        score = M

        # Update best if we have an improvement
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Early exit if the target is already met
        if best_score < 1.503:
            break

        # If we are stuck for a while, add a small random jitter
        if no_improve >= 200:
            eps = rng.normal(scale=0.01 * target_sum, size=n)
            cand = seq + eps
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _project_onto_simplex(cand, target_sum)
            seq = cand
            no_improve = 0
            it += 1
            continue

        # ------------------------------------------------------------------
        # Sub‑gradient of the peak (consider all active indices)
        # ------------------------------------------------------------------
        active = np.where(conv >= M - 1e-12)[0]
        grad = np.zeros_like(seq)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            inds = np.arange(i_start, i_end + 1)
            grad[inds] += seq[k - inds]

        if np.allclose(grad, 0.0):
            break  # nothing to improve

        # ------------------------------------------------------------------
        # Take a step opposite to the gradient and project back
        # ------------------------------------------------------------------
        cand = seq - step * grad
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_M = float(np.max(np.convolve(cand, cand)))

        if cand_M < M:
            # Accept the move and slightly increase the step size
            seq = cand
            step = min(step * 1.2, target_sum)
        else:
            # Reject – shrink the step
            step *= 0.5
            if step < 1e-4:
                step = max(step, 1e-4)
        it += 1

    return best_seq, best_score


def _greedy_peak_reduction(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    global_best_seq: np.ndarray,
    global_best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Greedy improvement: move a modest fraction of mass from a heavy‑
    contributing index to a light‑contributing one, accepting the move
    only when the evaluation score improves.
    """
    n = seq.size
    while time.monotonic() < deadline:
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))

        # contribution of each index to the current peak
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(seq)
        contrib[mask] = seq[i[mask]] * seq[j[mask]]
        total = contrib.sum()
        if total == 0.0:
            break

        # Choose a source index proportionally to its contribution
        prob = contrib / total
        i_sel = rng.choice(n, p=prob)

        # Choose a destination among indices with low contribution
        low_thresh = np.percentile(contrib, 30)
        low_cands = np.where(contrib <= low_thresh)[0]
        if low_cands.size == 0:
            j_sel = rng.integers(0, n)
        else:
            j_sel = rng.choice(low_cands)

        if i_sel == j_sel:
            continue

        # Transfer up to 30 % of the source mass
        eps = rng.uniform(0.0, 0.3 * seq[i_sel])
        if eps <= 0.0:
            continue

        cand = seq.copy()
        cand[i_sel] -= eps
        cand[j_sel] += eps
        # Preserve the ℓ¹‑norm exactly
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _score(cand)
        if cand_score < global_best_score - 1e-9:
            global_best_score = cand_score
            global_best_seq = cand.copy()
            seq = cand
            print(f"[greedy] improved score={global_best_score:.6f}")
            if global_best_score < 1.503:
                break
        # otherwise keep the current seq unchanged
    return global_best_seq, global_best_score


def _stochastic_polish(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Small Gaussian perturbations with an exponentially decaying step size.
    Accept any move that improves the true evaluation score.
    """
    cur = seq.copy()
    # Initial step roughly 20 % of the average entry
    step = 0.2 * max(1e-8, cur.mean())
    while time.monotonic() < deadline:
        cand = cur + rng.normal(scale=step, size=cur.size)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)
        if cand_score < best_score:
            best_score = cand_score
            cur = cand
            print(f"[polish] improved score={best_score:.6f}")
            if best_score < 1.503:
                break
        step *= 0.9995
        if step < 1e-6:
            step = 0.2 * max(1e-8, cur.mean()) + 1e-6
    return cur, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # Small safety margin for the caller to retrieve the result
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – prefer the provided global construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.array(globals()["height_sequence_1"], dtype=float)
            n = seq0.size
            target_sum = math.sqrt(2.0 * n)
            if seq0.sum() > 0:
                seq0 = np.clip(seq0 * (target_sum / seq0.sum()), 0.0, 1000.0)
            else:
                seq0 = np.full(n, target_sum / n)
            best_seq = seq0.copy()
            best_score = _score(best_seq)
            print(f"[init] using global height_sequence_1 (n={n}) score={best_score:.6f}")
        except Exception as exc:  # pragma: no cover
            print("[init] failed to use global height_sequence_1:", exc)
            # fallback to uniform
            n = 500
            target_sum = math.sqrt(2.0 * n)
            best_seq = np.full(n, target_sum / n)
            best_score = _score(best_seq)
            print(f"[init] fallback uniform (n={n}) score={best_score:.6f}")
    else:
        # No global available – start from a uniform vector
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _score(best_seq)
        print(f"[init] uniform start (n={n}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Sub‑gradient descent on the peak
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * best_seq.size)
    best_seq, sub_best_score = _subgradient_opt(best_seq, target_sum, deadline, rng)
    if sub_best_score < best_score:
        best_score = sub_best_score
        print(f"[phase1] sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Greedy peak reduction (local moves)
    # ------------------------------------------------------------------
    best_seq, best_score = _greedy_peak_reduction(
        best_seq, target_sum, deadline, rng, best_seq, best_score
    )

    # ------------------------------------------------------------------
    # Phase 3 – Stochastic polishing
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        best_seq, best_score = _stochastic_polish(best_seq, target_sum, deadline, rng, best_score)

    print(f"[final] best score={best_score:.6f} (n={best_seq.size})")
    # Return as plain Python list of floats
    return [float(x) for x in best_seq]

This implementation respects all constraints, runs three complementary optimisation phases within the allotted time and returns the best sequence it discovers.
