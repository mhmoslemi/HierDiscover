"""
Propose a low‑value step sequence.

The algorithm is a multi‑restart, multi‑length projected‐exponential‑gradient
descent on the objective  M = max_k (a*a)[k]  (the value returned by
``evaluate_sequence`` when the sum of the coefficients is normalised to
√(2 n)).  Each restart starts from a simple “sine‑window’’ or from the
global best sequence, performs a fast gradient update that reduces the
largest convolution entry, and falls back to a small random perturbation
when progress stalls.  The best sequence seen before the time budget
expires is returned.

Key ideas compared to the previous LP‑based approach:
* Directly target the max‑convolution value with a sub‑gradient step
  (exponential‑gradient update on the simplex) – no LP solver needed.
* Use several lengths (up to 1000) and many cheap restarts.
* Adaptive step‑size and simple back‑tracking line‑search keep the
  optimisation stable.
* Periodic Gaussian perturbations help escape plateaus.

The routine respects the security checks of ``evaluate_sequence``:
all entries are non‑negative, finite and the sum is far from zero.
"""

def _dirichlet_start(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex (Dirichlet with all parameters = 1)."""
    x = rng.dirichlet(np.ones(n))
    return x.astype(np.float64) * target_sum


def _sine_window_start(n: int, target_sum: float) -> np.ndarray:
    """Deterministic smooth start: a half‑sine window, normalised to the target sum."""
    i = np.arange(n, dtype=np.float64)
    # sin shape that is zero at the edges and positive inside
    wav = np.sin(np.pi * (i + 0.5) / n)
    wav = np.maximum(wav, 0.0)
    if wav.sum() == 0:
        wav = np.ones_like(wav)
    wav = wav / wav.sum() * target_sum
    return wav


def _conv_max_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute M = max_k (a*a)[k] and a sub‑gradient of M w.r.t. a.

    The gradient is constructed from the indices where the maximum is attained:
        grad_i = Σ_{k∈K} 2 * a_{k-i}   (indices out of range are ignored).

    Returns (M, grad) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = float(conv.max())
    # indices where the maximum occurs (within a tiny tolerance)
    max_idxs = np.where(conv >= M - 1e-14)[0]

    grad = np.zeros_like(a)
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # loop over the overlapping region
        for i in range(i_start, i_end + 1):
            j = k - i
            grad[i] += 2.0 * a[j]
    # Average if several indices share the maximum (still a valid sub‑gradient)
    if len(max_idxs) > 1:
        grad /= float(len(max_idxs))
    return M, grad


def _exp_gradient_step(a: np.ndarray, grad: np.ndarray, target_sum: float,
                       step: float) -> np.ndarray:
    """
    Perform a single exponentiated‑gradient update on the simplex.

    a_new_i = a_i * exp(-step * grad_i)
    The result is re‑normalised to have the prescribed ``target_sum``.
    """
    # keep the update strictly positive (exp cannot be zero)
    factor = np.exp(-step * grad)
    new_a = a * factor
    # renormalise to keep the L1‑norm fixed
    s = new_a.sum()
    if s == 0.0:
        # fallback – return the old vector (should never happen)
        return a.copy()
    new_a = new_a * (target_sum / s)
    return new_a

                a = _dirichlet_start(n, target_sum, rng)
                print(f"[init] Dirichlet start n={n} restart={restart}")

            # ------------------------------------------------------------------
            # Main optimisation loop – exponentiated gradient on the max‑conv.
            no_improve = 0
            for it in range(max_iters_per_restart):
                if time.time() >= deadline:
                    break

                M, grad = _conv_max_and_grad(a)

                # track global best (evaluate_sequence returns the same M for our normalisation)
                if M < best_score:
                    best_score = M
                    best_seq = a.copy()
                    print(f"[best] n={n} restart={restart} it={it} score={M:.6f}")

                # ----------- line‑search for a suitable step size ----------
                # start with a decreasing schedule, but allow back‑tracking
                step = 0.5 / math.sqrt(it + 1.0)   # initial guess
                improved = False
                for _ in range(7):
                    a_new = _exp_gradient_step(a, grad, target_sum, step)
                    M_new, _ = _conv_max_and_grad(a_new)
                    if M_new < M - 1e-12:          # strict improvement
                        a = a_new
                        improved = True
                        break
                    step *= 0.5                     # shrink step and retry

                if not improved:
                    # ----- stagnation handling -----
                    no_improve += 1
                    if no_improve >= 6:
                        # random Gaussian perturbation to escape a plateau
                        perturb = rng.normal(scale=0.02, size=n)
                        a = np.clip(a + perturb, 0.0, None)
                        a = a / a.sum() * target_sum
                        no_improve = 0
                        # recompute M after perturb (for the next iteration)
                        continue
                else:
                    no_improve = 0

                # optional early exit if we get very close to the theoretical lower bound
                # (average of the convolution = 1/(2n‑1))
                avg = 1.0 / (2 * n - 1)
                if M <= avg * 1.03:          # within 3 % of the absolute best possible
                    if M < best_score:
                        best_score = M
                        best_seq = a.copy()
                        print(f"[near‑lb] n={n} it={it} score={M:.6f}")
                    break   # stop further iterations for this restart

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    # final sanity check – run the provided evaluator once (it also masks illegal values)
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Final best score: {final_score:.6f} (length {len(best_seq)})")

    # return a plain Python list as required
    return [float(v) for v in best_seq]
