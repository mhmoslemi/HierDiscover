def lp_step(f, target_sum, t, rng):
   compute M = max conv of f
   n = len(f)
   build A matrix shape (2n-1, n)
   # fill each column i
   A = np.zeros((2*n-1, n), dtype=np.float64)
   for i in range(n):
       A[i:i+n, i] = f[i]  # because conv_k = sum_i f_i * b_{k-i}

   # c = -1 for maximizing sum b => minimize -sum b
   c = -np.ones(n)

   # bounds for b_i: (0,1000) maybe.
   bounds = [(0, None) for _ in range(n)]   # maybe None upper bound (but convertible later to 1000 clip)

   # Apply linear programming
   res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
   if not res.success:
       return f, None

   g0 = res.x
   sum_g0 = g0.sum()
   if sum_g0 == 0:
       return f, None
   g = (target_sum / sum_g0) * g0
   # Optional clipping again
   g = np.clip(g, 0, 1000)
   # Mix with f
   new_f = (1 - t) * f + t * g
   # ensure nonneg, clip
   new_f = np.clip(new_f, 0, 1000)
   # normalize sum to target_sum (to avoid drift)
   s = new_f.sum()
   if s > 0:
       new_f = new_f * (target_sum / s)
   return new_f, M

def lp_iter(seq, target_sum, t, rng):
    n=len(seq)
    m = 2*n-1
    # compute M
    conv = np.convolve(seq, seq, mode='full')
    M = conv.max()
    # build A
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = seq[i]
    # LP: maximize sum(b)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    # use high-level
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if not res.success:
        return seq, M, False
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return seq, M, False
    # mix
    g = b * (target_sum / sum_b)
    # potential clipping? but values small
    new_seq = (1 - t) * seq + t * g
    # Keep sum at target
    s = new_seq.sum()
    if s > 0:
        new_seq = new_seq * (target_sum / s)
    # optionally add small random perturbation to avoid stagnation
    # random jitter
    if rng.random() < 0.1:
        new_seq += rng.normal(scale=0.001*target_sum, size=n)
        new_seq = np.clip(new_seq, 0, None)
        s = new_seq.sum()
        if s>0:
            new_seq = new_seq * (target_sum/s)
    score = _fast_score(new_seq)
    return new_seq, score, True

"""
Heuristic optimizer for the max‑auto‑convolution problem.

1. Initialise a non‑negative sequence (random or the globally
   supplied ``height_sequence_1``) and normalise its ℓ¹‑norm to
   √(2·n).  The score of a normalised sequence equals the
   maximum entry of its auto‑convolution.

2. Iteratively improve the sequence by solving a linear program
   that finds a “partner’’ vector `g` with the largest possible
   ℓ¹‑norm under the linear constraint `f * g ≤ M` where `M`
   is the current maximal convolution value.  The partner is
   then mixed into `f` (convex combination) – this is the
   Frank–Wolfe style step described in the reference
   algorithm.

3. When the LP‑driven search stops (time limit or stagnation),
   a short stochastic hill‑climb based on Gaussian perturbations
   refines the best candidate.

The whole procedure respects the 1000 s budget, works for any
length up to a few thousand and returns a plain ``list[float]``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def _fast_score(seq: np.ndarray) -> float:
    """Score identical to ``evaluate_sequence`` for a normalised vector.

    The denominator 2·n cancels with the fixed ℓ¹‑norm √(2·n), so the
    score equals the maximal auto‑convolution entry.
    """
    n = seq.shape[0]
    max_b = np.convolve(seq, seq, mode="full").max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` (length m) to length ``new_n``."""
    m = seq.shape[0]
    if m == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, m)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, seq)


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting sequence for length ``n``.

    Preference is given to the global variable ``height_sequence_1``
    (if present) – otherwise a fresh random vector is created.
    The result is normalised to ℓ¹‑norm √(2·n).
    """
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if base.shape[0] != n:
            base = _interpolate(base, n)
        seq = base.copy()
    else:
        seq = rng.random(n) + 0.01  # ensure positivity
    # clip to the allowed interval and normalise
    seq = np.clip(seq, 0.0, 1000.0)
    target = math.sqrt(2.0 * n)
    s = seq.sum()
    if s > 0:
        seq = seq * (target / s)
    return seq

# LP‑based improvement step
# ----------------------------------------------------------------------


def _build_conv_matrix(seq: np.ndarray) -> np.ndarray:
    """Return the (2n‑1)×n matrix ``A`` with (A·b)_k = (seq * b)_k."""
    n = seq.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # each column i contributes the constant seq[i] on rows i … i+n‑1
    for i, v in enumerate(seq):
        A[i : i + n, i] = v
    return A


def _lp_improve(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Perform a greedy Frank‑Wolfe style loop using linear programming.

    The loop runs until ``deadline`` and returns the best sequence found
    together with its score.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _fast_score(best_seq)

    # mixing factor – starts fairly aggressive and is slowly reduced
    mix = 0.45

    # simple stagnation counter
    stagnation = 0

    while time.monotonic() < deadline:
        # 1) current maximal auto‑convolution value M
        conv = np.convolve(seq, seq, mode="full")
        M = conv.max()

        # 2) build linear constraints for the partner vector g
        A = _build_conv_matrix(seq)
        ub = np.full(A.shape[0], M, dtype=np.float64)

        # 3) linear program: maximise   1ᵀ·g    (i.e. minimise –1ᵀ·g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs")
        if not res.success:
            # Very unlikely – fall back to a tiny random perturbation
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = min(seq[i], target_sum * 0.01)
                seq[i] -= delta
                seq[j] += delta
                seq = np.clip(seq, 0.0, 1000.0)
                seq = seq * (target_sum / seq.sum())
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # Rescale to keep the ℓ¹‑norm fixed
        g = g * (target_sum / sum_g)

        # 4) convex combination  f_{new} = (1‑t)·f + t·g
        new_seq = (1.0 - mix) * seq + mix * g
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        # renormalise (numerical drift)
        s = new_seq.sum()
        if s > 0.0:
            new_seq = new_seq * (target_sum / s)

        new_score = _fast_score(new_seq)

        # 5) accept only improvements – this is a deterministic descent
        if new_score < best_score - 1e-12:
            best_seq = new_seq.copy()
            best_score = new_score
            stagnation = 0
            # occasionally shrink the step size to fine‑tune
            mix = max(0.02, mix * 0.97)
            # useful log
            print(
                f"[LP] n={n:4d} score={best_score:.6f}  mix={mix:.4f}  time={time.monotonic():.1f}"
            )
        else:
            stagnation += 1
            # if we are stuck, increase randomness a little
            if stagnation % 30 == 0:
                # random jitter on a few coordinates
                idx = rng.choice(n, size=min(5, n), replace=False)
                jitter = rng.normal(scale=0.005 * target_sum, size=idx.shape)
                seq[idx] = np.clip(seq[idx] + jitter, 0.0, 1000.0)
                seq = seq * (target_sum / seq.sum())
                # keep the old step size
        # always move to the candidate (even if not better) – helps
        # escaping shallow plateaus; the acceptance rule above only
        # records the best.
        seq = new_seq

        # tiny safety break
        if stagnation > 300:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final stochastic polishing
# ----------------------------------------------------------------------


def _local_search(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Gaussian hill‑climb on the original ``evaluate_sequence``."""
    rng = np.random.default_rng()
    best = seq.copy()
    best_score = _fast_score(best)
    step = 0.5  # initial standard deviation

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = max(step * 0.995, 1e-6)
        else:
            step *= 0.999
    return best, best_score

"""
Heuristic optimizer for the max‑auto‑convolution problem.

The algorithm works in three stages:

1️⃣  **Warm‑start** – a sequence of length *n* is created
    (either from a globally supplied ``height_sequence_1`` or
    randomly) and normalised so that its ℓ¹‑norm equals √(2·n).
    For a normalised vector the evaluation score is exactly the
    maximal entry of its auto‑convolution.

2️⃣  **LP‑driven Frank‑Wolfe step** – given the current sequence *f*
    we compute M = max(f∗f).  A linear program then finds a partner
    vector *g* with the largest possible ℓ¹‑norm under the linear
    constraints `f * g ≤ M`.  After rescaling *g* to the same ℓ¹‑norm
    as *f* we replace `f ← (1‑t)·f + t·g` (t shrinks gradually).
    Repeating this step drives the maximal convolution value down.
    The LP has only O(n) variables and O(2n) constraints, so it is
    fast even for n≈1000.

3️⃣  **Stochastic polishing** – a short Gaussian hill‑climb cleans
    up the best candidate found in the previous phase.

The whole routine respects the provided time budget (default 1000 s)
and returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

def _fast_score(seq: np.ndarray) -> float:
    """Score identical to ``evaluate_sequence`` for a normalised vector.

    The denominator 2·n cancels with the fixed ℓ¹‑norm √(2·n), so the
    score equals the maximal auto‑convolution entry.
    """
    n = seq.shape[0]
    max_b = np.convolve(seq, seq, mode="full").max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` (length m) to length ``new_n``."""
    m = seq.shape[0]
    if m == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, m)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, seq)


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:

    if "height_sequence_1" in globals():
