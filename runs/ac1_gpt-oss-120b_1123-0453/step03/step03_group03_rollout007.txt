def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1) × n matrix A such that (A @ g) = f * g."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes f to rows j … j+n‑1
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _score(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # evaluate_sequence expects a plain Python list
    return float(evaluate_sequence(seq.tolist()))

def _solve_lp_max_sum(f: np.ndarray, bound: float) -> np.ndarray | None:
    """Solve   maximise Σ g_i   s.t.   (f * g) ≤ bound,   g ≥ 0.
    Returns the optimal vector g (or None on failure)."""
    n = f.shape[0]
    A = _conv_matrix(f)
    b = np.full(A.shape[0], bound, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # minimise –Σ g_i
    bounds = [(0, None)] * n

    if _HAS_LINPROG:
        # SciPy's high‑level interface is fast for the sizes we use.
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                return g
        except Exception:
            pass

    # Fallback to cvxpy (slower but more robust)
    if _HAS_CVX:
        g = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= bound])
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                       abstol=1e-8, reltol=1e-6)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=20000,
                           verbose=False)
            except Exception:
                return None
        if g.value is not None:
            return np.maximum(g.value, 0.0)

    return None

def _optimal_mix_t(conv_ff: np.ndarray, conv_fg: np.ndarray,
                   conv_gg: np.ndarray) -> float:
    """Return t∈[0,1] that minimises max_k ((1‑t)f + t g)∗((1‑t)f + t g)[k].
    Uses a cheap grid + ternary refinement."""
    a = conv_ff - 2.0 * conv_fg + conv_gg        # quadratic coefficient
    b = 2.0 * (conv_fg - conv_ff)                # linear coefficient
    c = conv_ff                                   # constant term

    # coarse grid
    ts = np.linspace(0.0, 1.0, 41)
    vals = np.max(a[:, None] * ts**2 + b[:, None] * ts + c[:, None], axis=0)
    t_best = float(ts[np.argmin(vals)])

    # local ternary search around the best grid point
    left = max(0.0, t_best - 0.2)
    right = min(1.0, t_best + 0.2)
    for _ in range(15):
        t1 = left + (right - left) / 3.0
        t2 = right - (right - left) / 3.0
        v1 = np.max(a * t1 * t1 + b * t1 + c)
        v2 = np.max(a * t2 * t2 + b * t2 + c)
        if v1 < v2:
            right = t2
        else:
            left = t1
    return (left + right) / 2.0

            f = f * (1.0 + rng.uniform(-0.02, 0.02, size=n))

                M = float(conv_f.max())

                # LP to obtain a promising direction g.
                # ------------------------------------------------------
                g = _solve_lp_max_sum(f, M)
                if g is None:

                    break

                sum_g = g.sum()
                if sum_g <= 0:
                    break
                g = g * (target_sum / sum_g)        # rescale to the same ℓ¹‑norm

                # Choose the mixing coefficient t that yields the smallest new peak.
                # ------------------------------------------------------

                conv_gg = np.convolve(g, g)
                conv_fg = np.convolve(f, g)

        total = contrib.sum()
        if total == 0.0:
            break

        prob = contrib / total
        src = rng.choice(n, p=prob)

        low_thr = np.percentile(contrib, 30)
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size == 0:
            dst = rng.integers(n)
        else:
            dst = rng.choice(low_candidates)

        if src == dst:
            continue

        # Transfer a modest fraction of the source mass.
        max_transfer = 0.4 * seq[src]          # up to 40 % of the source
        if max_transfer <= 0.0:
            continue

        cand = seq.copy()
        cand[src] -= delta
        cand[dst] += delta

        cand_score = _score(cand)
        if cand_score < cur_score - 1e-9:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[greedy] improved to {best_score:.6f}")

        if best_score < 1.503:
            print("[greedy] reached target bound.")
            break

    # Phase 3 – simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        cur = best_seq.copy()
        cur_score = best_score
        # Linear cooling schedule
        total_allowed = deadline - start
        while time.monotonic() < deadline:
            elapsed = time.monotonic() - start
            # Temperature decays from 1.0 to 1e‑4
            T = 1.0 * (1e-4 / 1.0) ** (elapsed / total_allowed)

            # Propose a random transfer of mass between two positions.
            src = rng.choice(n, p=cur / cur.sum())
            dst = rng.integers(n)
            if src == dst:
                continue
            # Transfer up to 30 % of the source amount.
            max_transfer = 0.3 * cur[src]
            if max_transfer <= 0.0:
                continue
            delta = rng.uniform(0.0, max_transfer)

            cand = cur.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand = np.clip(cand, 0.0, 1000.0)

            cand_score = _score(cand)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_seq = cand.copy()
                    best_score = cand_score
                    print(f"[anneal] improved to {best_score:.6f}")
            else:
                # Metropolis acceptance.
                if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                    cur = cand
                    cur_score = cand_score

            # Optional early exit.
            if best_score < 1.503:
                print("[anneal] target reached – exiting.")
                break

    return [float(x) for x in best_seq]

        src = rng.choice(n, p=prob)

        # Destination chosen among the low‑contribution tail.
        low_thr = np.percentile(contrib, 30)
        low_candidates = np.where(contrib <= low_thr)[0]
        dst = rng.choice(low_candidates) if low_candidates.size else rng.integers(n)

        if src == dst:
            continue

        # Transfer a modest fraction of the source mass.
        max_transfer = 0.4 * seq[src]          # up to 40 % of the source
        if max_transfer <= 0.0:
            continue
        delta = rng.uniform(0.0, max_transfer)

        cand = seq.copy()
        cand[src] -= delta
        cand[dst] += delta

        cand_score = _score(cand)
        if cand_score < cur_score - 1e-9:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[greedy] improved to {best_score:.6f}")

        if best_score < 1.503:
            print("[greedy] reached target bound.")
            break

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        cur = best_seq.copy()
        cur_score = best_score
        total_allowed = deadline - start

        while time.monotonic() < deadline:
            elapsed = time.monotonic() - start
            # Temperature decays from 1.0 to 1e‑4.
            T = 1.0 * (1e-4 / 1.0) ** (elapsed / total_allowed)

            # Random transfer of mass.
            src = rng.choice(n, p=cur / cur.sum())
            dst = rng.integers(n)
            if src == dst:
                continue
            max_transfer = 0.3 * cur[src]     # up to 30 % of the source
            if max_transfer <= 0.0:
                continue
            delta = rng.uniform(0.0, max_transfer)

            cand = cur.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand = np.clip(cand, 0.0, 1000.0)

            cand_score = _score(cand)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_seq = cand.copy()
                    best_score = cand_score
                    print(f"[anneal] improved to {best_score:.6f}")
            else:
                # Metropolis acceptance.
                if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                    cur = cand
                    cur_score = cand_score

            if best_score < 1.503:
                print("[anneal] target reached – exiting.")
                break

    # Return the best sequence found (as plain Python floats).
    return [float(x) for x in best_seq]
