def _obj_and_grad(z, n, S, beta):
    # compute a
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z) * S
    # compute convolution conv via FFT
    L = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, L)
    conv = np.fft.irfft(fft_a * fft_a, L)[:2*n-1]
    # smooth max and weights
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    sum_exps = np.sum(exps)
    smooth = max_c + np.log(sum_exps) / beta
    w = exps / sum_exps
    # gradient wrt a
    # use FFT for convolution a_rev and w
    a_rev = a[::-1]
    L2 = 1 << ((3*n-2).bit_length())
    fft_a_rev = np.fft.rfft(a_rev, L2)
    fft_w = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(fft_a_rev * fft_w, L2)
    start = n-1
    g_raw = conv2[start:start+n]
    g_a = 2.0 * g_raw
    # chain rule to z
    total = np.dot(g_a, a)
    grad_z = a * g_a - a * (total / S)
    return smooth, grad_z

def _optimize_length(n, target_sum, deadline, rng, warm_seq=None, warm_val=None):
    best_seq = None
    best_val = float('inf')
    betas = [5.0, 15.0, 40.0, 80.0, 200.0]

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break
        # init z
        if warm_seq is not None and len(warm_seq) == n:
            # warm_seq may have zeros; avoid log(0)
            p = np.clip(warm_seq / target_sum, 1e-12, None)
            z0 = np.log(p)
        else:
            z0 = rng.normal(scale=0.1, size=n)
        # iterate through betas
        for beta in betas:
            if time.monotonic() >= deadline:
                break
            # objective wrapper
            def fun(z):
                val, grad = _obj_and_grad(z, n, target_sum, beta)
                return val, grad
            # call minimize
            try:
                res = minimize(fun, z0, method='L-BFGS-B', jac=True,
                               options={'maxiter': 200, 'gtol': 1e-8, 'disp': False})
            except Exception as e:
                print(f"[opt] n={n} restart={restart} beta={beta} error {e}")

            z0 = res.x  # for next beta stage
            # evaluate using evaluate_sequence occasionally
            a = _softmax_to_a(z0, target_sum)

The goal is to minimise  

\[
\Phi(a)=\frac{2n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where \((a*a)\) is the auto‑convolution of the step heights.  
The denominator is scale‑invariant, therefore we can keep the ℓ¹‑norm fixed:
\[
\sum_i a_i = S=\sqrt{2n}.
\]

The optimisation problem can be written as

* minimise the maximum entry of a quadratic form (a convex but non‑smooth function)  
* subject to the linear simplex constraints \(a_i\ge0,\;\sum a_i=S\).

The algorithm below improves on the previous implementation by

1. **Re‑parameterising** the simplex with a soft‑max – the variables `z` are unrestricted,  
   `a = S·softmax(z)`.  This guarantees non‑negativity and the ℓ¹‑norm automatically.
2. **Smooth approximation of the max** – `log‑sum‑exp` with a continuation schedule  
   \(\beta\in\{5,15,40,80,200\}\).  For small `β` the objective is smooth, for large `β`
   it is a tight approximation of the true maximum.
3. **Exact gradient** (closed form) using the convolution theorem,  
   the gradient of the smooth objective is  

   \[
   \nabla_z\Phi = a\odot g_a - a\;\frac{a^\top g_a}{S},
   \qquad
   g_a = 2\;\bigl(a\star w\bigr)_{0:n},
   \]

   where \(w\) are the soft‑max weights over the convolution entries and
   \(\star\) denotes a convolution (computed with FFTs).
4. **L‑BFGS‑B optimisation** on the smooth objectives, with several random restarts
   and a homotopy over the `β` values.
5. **A final stochastic hill‑climbing polish** (Gaussian perturbations) that
   respects the simplex via the same soft‑max projection.

All heavy operations (convolution, gradient) are performed with FFTs, giving
\(O(n\log n)\) per objective evaluation.  
The search is time‑budget aware – it stops as soon as the target bound
\(1.503\) is achieved or the global 1000 s budget expires.

"""
Proposes a low‑value non‑negative step sequence.

The optimiser works on the ℓ¹‑simplex via a soft‑max re‑parametrisation.
A smooth log‑sum‑exp surrogate of the max‑convolution is minimised with
L‑BFGS‑B, using a continuation on the smoothness parameter β.
After the deterministic phase a short stochastic hill‑climbing (Gaussian
perturbations) refines the best candidate.

All numbers are clipped to the limits required by `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """Convert unrestricted vector z to a simplex‑point a with sum=target_sum."""
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z) * target_sum
    return a


def _obj_and_grad(z: np.ndarray, n: int, S: float, beta: float):
    """
    Returns (smooth_obj, grad_z) where the smooth objective is
        (1/β)·log Σ exp(β·conv)
    and grad_z is the gradient w.r.t. the soft‑max parameters z.
    """
    # ---- a from soft‑max -------------------------------------------------
    a = _softmax_to_a(z, S)                      # shape (n,)

    # ---- auto‑convolution via FFT ---------------------------------------
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]  # length 2n‑1, real

    # ---- smooth max (log‑sum‑exp) ---------------------------------------
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))          # stable
    sum_exps = np.sum(exps)
    smooth = max_c + np.log(sum_exps) / beta      # scalar
    w = exps / sum_exps                            # softmax weights, ∑w = 1

    # ---- gradient wrt a : g_a = 2·(a_rev * w)_{n … n+n‑1} -------------
    a_rev = a[::-1]
    L2 = _next_pow2(3 * n - 2)
    fr = np.fft.rfft(a_rev, L2)
    fw = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(fr * fw, L2)            # length L2
    start = n - 1
    g_raw = conv2[start : start + n]            # length n
    g_a = 2.0 * g_raw

    # ---- chain rule to obtain grad_z ------------------------------------
    # dΦ/dz_i = a_i·g_a_i - a_i·(a·g_a)/S
    total = np.dot(g_a, a)
    grad_z = a * g_a - a * (total / S)

    return smooth, grad_z.astype(np.float64)


def _stochastic_refinement(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Gaussian hill‑climbing that keeps the sum fixed by the same soft‑max
    projection used in the main optimiser.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = evaluate_sequence(best.tolist())
    step = step_initial
    S = best.sum()

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑project onto the simplex (soft‑max)
        cand = _softmax_to_a(np.log(np.clip(cand / S, 1e-12, None)), S)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial

    return best, best_score


def _optimise_length(n: int,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> tuple[np.ndarray, float]:
    """
    Runs a few L‑BFGS‑B optimisations (different restarts) for a single length n.
    Returns the best sequence found and its exact `evaluate_sequence` score.
    """
    betas = [5.0, 15.0, 40.0, 80.0, 200.0]      # continuation schedule
    max_restarts = 4
    best_seq = None
    best_val = float("inf")

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # ---- initialise z -------------------------------------------------
        if warm_seq is not None and warm_seq.shape[0] == n:
            # soft‑max inverse (avoid log(0))
            p = np.clip(warm_seq / target_sum, 1e-12, None)
            z0 = np.log(p)
        else:
            z0 = rng.normal(scale=0.1, size=n)

        # ---- homotopy over β ---------------------------------------------
        for beta in betas:
            if time.monotonic() >= deadline:
                break

            # L‑BFGS‑B with analytic gradient
            res = minimize(fun=lambda zz: _obj_and_grad(zz, n, target_sum, beta),
                           x0=z0,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 200,
                                    'gtol': 1e-10,
                                    'disp': False})
            z0 = res.x                     # start point for next β

            # evaluate exact score (expensive, done only after each β)
            a = _softmax_to_a(z0, target_sum)
            a = np.clip(a, 0.0, 1000.0)
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[opt] n={n} restart={restart} β={beta:.0f} val={val:.6f}")

            # early exit if already under the target
            if best_val < 1.503:
                return best_seq, best_val

    # fallback: uniform distribution if nothing better was found
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())

    return best_seq, best_val


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.
    The function respects the global time budget (≈ budget_s seconds)
    and returns a plain Python list of floats ready for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        n0 = seq.size
        S0 = math.sqrt(2.0 * n0)
        if seq.sum() > 0:
            seq = seq * (S0 / seq.sum())
        else:
            seq = np.full(n0, S0 / n0, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        val = evaluate_sequence(seq.tolist())
        best_seq = seq.copy()
        best_score = val
        print(f"[init] warm‑start n={n0} val={val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Deterministic optimisation on several lengths
    # ------------------------------------------------------------------
    # a diverse set of lengths – larger sizes give a larger “search space”
    lengths = [300, 400, 500, 600, 800, 1000, 1200, 1500, 1800, 2100]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # use the current best sequence as warm‑start only if its length matches n
        warm = best_seq if (best_seq is not None and best_seq.size == n) else None
        seq_n, val_n = _optimise_length(n, target_sum, deadline, rng, warm)

        if val_n < best_score:
            best_score = val_n
            best_seq = seq_n.copy()
            print(f"[global update] n={n} val={val_n:.6f}")

        if best_score < 1.503:
            print("[early stop] target reached")
            break

    # ------------------------------------------------------------------
    # 2️⃣ Final stochastic polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_val = _stochastic_refinement(best_seq, deadline,
                                                      step_initial=0.3)
        if refined_val < best_score:
            best_seq = refined
            best_score = refined_val
            print(f"[final polish] val={refined_val:.6f}")

    # final safety clipping (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score = {best_score:.6f}  (length={best_seq.size})")
    return [float(x) for x in best_seq]
