def _gradient_of_max(seq, k):
    n = len(seq)
    g = np.zeros_like(seq)
    i_start = max(0, k - (n-1))
    i_end = min(k, n-1)
    for i in range(i_start, i_end+1):
        j = k - i
        # contribute seq[j] to derivative w.r.t seq[i]
        g[i] += seq[j]
        # Also seq[i] to derivative w.r.t seq[j] (if i != j)
        if j != i:
            g[j] += seq[i]
    return g
```

def gradient_descent(seq, deadline, rng):
    n = len(seq)
    target = math.sqrt(2.0 * n)
    step = 0.5   # initial step size scaling factor (relative)
    best_seq = seq.copy()
    best_sc = _fast_score(seq)
    while time.monotonic() < deadline:
        max_val, k_idx, conv = _max_convolution(seq)
        grad = _gradient_of_max(seq, k_idx)
        # compute norm to scale step
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            break
        # compute step size adaptively: step_factor * target / grad_norm maybe.
        # careful not to overshoot.
        new = seq - step * grad / grad_norm
        new = np.clip(new, 0, 1000)
        total = new.sum()
        if total == 0:
            new = np.full(n, target / n)
        else:
            new = new * (target / total)

        new_sc = _fast_score(new)
        if new_sc < best_sc:
            best_seq, best_sc = new.copy(), new_sc

        # Accept if improvement or else maybe keep if not too much worse? We'll accept only if improvement, else reduce step size.
        if new_sc < best_sc * 0.9999: # improvement
            seq = new
        else:
            # no improvement: reduce step size
            step *= 0.7
            if step < 1e-4:
                # add random perturbation to escape
                perturb = rng.normal(scale=0.01 * target, size=n)
                seq = seq + perturb
                seq = np.clip(seq, 0, 1000)
                seq = seq * (target / seq.sum())
                step = 0.5
        # stop criteria: step too small or time out
        if step < 1e-5:
            break
    return best_seq, best_sc

def _simulated_annealing(seq, deadline, rng):
    n = len(seq)
    target = math.sqrt(2.0*n)
    cur_seq = seq.copy()
    cur_max, cur_conv = _conv_max(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_max
    T0 = 0.01 * best_score   # initial temperature in absolute objective
    t = 0
    # Parameter
    steps = 30 # number of temperature steps
    while time.monotonic() < deadline:
        # Determine temperature schedule
        fraction = (time.monotonic() - start) / total_time
        T = (1 - fraction) * T0
        # propose random move
        i = rng.integers(n)
        if cur_seq[i] < 1e-9:
            continue
        # step size proportionally to current magnitude
        delta = cur_seq[i] * rng.uniform(0.01, 0.05)   # move 1-5% of i's mass
        j = rng.integers(n)
        if i == j:
            continue
        cand_seq = cur_seq.copy()
        cand_seq[i] -= delta
        cand_seq[j] += delta
        # clip negativity (shouldn't happen)
        if cand_seq[i] < -1e-12:
            continue
        # renormalize to maintain sum (should be unchanged due to transfer)
        # compute new max
        cand_max, _ = _conv_max(cand_seq)
        delta_obj = cand_max - cur_max
        if delta_obj < 0 or rng.random() < math.exp(-delta_obj / (T+1e-12)):
            cur_seq = cand_seq
            cur_max = cand_max
            if cur_max < best_score:
                best_seq = cur_seq.copy()
                best_score = cur_max
        # after some iterations maybe do extra random swap or random complete shuffle

def _interpolate(seq, new_n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    # ensure positivity
    new_seq = np.maximum(new_seq, 0)
    # normalize
    target = math.sqrt(2.0 * new_n)
    new_seq = new_seq * (target / new_seq.sum())
    return new_seq

def _sa_optimize(seq, deadline, rng):
    n= len(seq)
    target = math.sqrt(2.0*n)
    cur_seq = seq.copy()
    cur_max = _conv_max(cur_seq)
    best_seq = cur_seq.copy()
    best_max = cur_max
    start = time.monotonic()
    total = deadline - start
    # Temperature schedule: start high to allow escapes
    T0 = (cur_max - 1.0) * 0.5   # maybe some fraction of diff to average
    # if it's <? else default
    iter =0
    while time.monotonic() < deadline:
        # linear cooling
        elapsed = time.monotonic() - start
        frac = elapsed / total
        T = T0 * (1 - frac)
        # propose move
        i = rng.integers(n)
        if cur_seq[i] < 1e-12:
            continue
        # choose delta as random fraction of current i mass
        delta = cur_seq[i] * rng.uniform(0.01, 0.05)
        j = rng.integers(n)
        if i == j:
            continue
        cand_seq = cur_seq.copy()
        cand_seq[i] -= delta
        cand_seq[j] += delta
        cand_max = _conv_max(cand_seq)
        diff = cand_max - cur_max
        if diff < 0 or rng.random() < math.exp(-diff / (T+1e-12)):
            cur_seq, cur_max = cand_seq, cand_max
            if cur_max < best_max:
                best_seq, best_max = cur_seq.copy(), cur_max
        # occasional more global perturbation
        if iter % 10000 == 0 and iter > 0:
            # random Gaussian jitter
            pert = rng.normal(scale=0.02 * target, size=n)
            jitter_seq = cur_seq + pert

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float('inf')
    # initial from provided global
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        n_init = init.shape[0]
        if n_init > 0:
            target_init = math.sqrt(2.0 * n_init)
            init = init * (target_init / init.sum())
            sc = _fast_score(init)
            best_seq, best_score = init, sc
            print(f"[init] from height_sequence_1 n={n_init} score={sc:.6f}")
    except Exception:
        pass

    # define lengths to try
    lengths = [200, 300, 400, 500, 600, 800, 1000]
    # families and param ranges
    power_ps = np.linspace(0.2, 5.0, 20)   # p exponent
    rec_ps = np.linspace(0.2, 5.0, 20)
    quad_cs = np.linspace(0.0, 4.0, 20)
    sine_ps = np.linspace(0.2, 5.0, 20)

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        # include interpolation of current best if length differs
        if best_seq is not None and best_seq.shape[0] != n:
            cand = _interpolate_to(best_seq, n)
            sc = _fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"[interp] from length {best_seq.shape[0]} to {n} score={sc:.6f}")

        # Sweep families
        # Uniform is covered by power p=0 (if we include)
        for p in power_ps:
            a = (np.arange(1, n+1, dtype=np.float64) ** (-p) +
                 (np.arange(n, 0, -1, dtype=np.float64) ** (-p)))  # symmetric
            a = a * (target / a.sum())
            sc = _fast_score(a)
            if sc < best_score:

        for p in rec_ps:

"""
Propose a low‑value step sequence.

Algorithm Overview
------------------
1️⃣ **Parametric sweep** – generate a variety of symmetric, U‑shaped
   families (power‐law, reciprocal product, quadratic and sine‑inverse
   profiles) for several candidate lengths.  Each candidate is
   normalised so that Σ a_i = √(2·n); the evaluation reduces to the
   maximal auto‑convolution entry.  The best candidate of this cheap
   sweep is kept.

2️⃣ **Simulated‑annealing refinement** – starting from the best vector,
   a simple SA loop repeatedly transfers a small amount of mass from a
   random index to another.  Acceptance follows the Metropolis rule with
   a linear cooling schedule.  Every ~ 8000 proposals a global Gaussian
   jitter is tried.  The loop stops early once the objective drops
   below the target 1.503.

3️⃣ **Final Gaussian hill‑climb** – a short stochastic local‑search
   (Gaussian perturbations) polishes the solution.

All steps respect the security checks of ``evaluate_sequence``:
non‑negative entries, clipping to [0,1000] and exact ℓ¹‑normalisation.
The routine returns the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact objective of evaluate_sequence (max auto‑convolution)."""
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    # With Σa = √(2n) the factor simplifies to max_b
    return float(max_b)


def _conv_max(seq: np.ndarray) -> float:
    """Return only the maximal convolution entry (cheaper than full score)."""
    return np.convolve(seq, seq).max()


def _interpolate_to(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence to a different length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    target = math.sqrt(2.0 * new_n)
    new_seq = new_seq * (target / new_seq.sum())
    return new_seq


def _gradient_step(seq: np.ndarray, target: float, alpha: float) -> np.ndarray:
    """Cheap gradient‑like move that reduces the current maximal entry."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())
    g = np.zeros(n, dtype=np.float64)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    for i in range(i_start, i_end + 1):
        j = k - i
        g[i] += seq[j]
        if j != i:
            g[j] += seq[i]
    # simple descent direction:
    new = seq - alpha * g
    new = np.clip(new, 0.0, 1000.0)
    s = new.sum()
    if s == 0.0:
        new = np.full(n, target / n)
    else:
        new = new * (target / s)
    return new


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Short Gaussian hill‑climb (used for final polishing).
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)
    target = math.sqrt(2.0 * n)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target / cand.sum())
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9995
        if step < 1e-7:
            step = step_initial
    return best, best_sc


def _simulated_annealing(seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a small amount of mass from one index to another.
    The temperature decays linearly; occasional global Gaussian jitter
    helps escaping shallow minima.
    """
    n = len(seq)
    target = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_val = _conv_max(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.monotonic()
    total = max(1e-9, deadline - start)

    # initial temperature proportional to the excess over the theoretical minimum (=1)
    T0 = max(0.01, (cur_val - 1.0) * 0.5)

    it = 0
    while time.monotonic() < deadline:
        # linear cooling
        elapsed = time.monotonic() - start
        frac = elapsed / total
        T = max(1e-12, T0 * (1.0 - frac))

        # propose a transfer of a few percent of the mass at a random source index
        i = rng.integers(n)
        if cur[i] <= 1e-15:
            continue
        delta = cur[i] * rng.uniform(0.015, 0.04)   # 1.5‑4 % move
        j = rng.integers(n)
        if i == j:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = _conv_max(cand)

        diff = cand_val - cur_val
        if diff < 0 or rng.random() < math.exp(-diff / T):
            cur, cur_val = cand, cand_val
            if cur_val < best_val:
                best, best_val = cur.copy(), cur_val

        # every few thousand steps try a global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=0.02 * target, size=n)
            cand2 = cur + jitter
            cand2 = np.clip(cand2, 0.0, 1000.0)
            cand2 = cand2 * (target / cand2.sum())
            cand2_val = _conv_max(cand2)
            diff2 = cand2_val - cur_val
            if diff2 < 0 or rng.random() < math.exp(-diff2 / T):
                cur, cur_val = cand2, cand2_val
                if cur_val < best_val:
                    best, best_val = cur.copy(), cur_val

        it += 1
        if best_val < 1.503:            # early exit if target reached
            break

    return best, best_val
