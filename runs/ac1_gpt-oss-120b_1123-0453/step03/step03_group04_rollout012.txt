def hill_climb(seq, target, deadline):
    n = len(seq)
    best_seq = seq.copy()
    best_sc = max_convolution(best_seq)
    rng = np.random.default_rng()
    while time.monotonic() < deadline:
        conv = np.convolve(best_seq, best_seq)
        max_val = conv.max()
        ks = np.where(conv >= max_val - 1e-12)[0]
        # compute gradient for all ks
        grad = np.zeros(n)
        for k in ks:
            idx = np.arange(n)
            partner = k - idx
            mask = (partner >=0) & (partner < n)
            grad[mask] += 2 * best_seq[partner[mask]]
        # donors high gradient, receivers low gradient
        donor_inds = np.where(grad > grad.mean())[0]
        receiver_inds = np.where(grad < grad.mean())[0]
        if donor_inds.size == 0 or receiver_inds.size == 0:
            # fallback random
            i = rng.integers(n)
            j = rng.integers(n)
        else:
            i = rng.choice(donor_inds)
            j = rng.choice(receiver_inds)
        # transfer amount proportional to donor's mass
        delta = 0.1 * best_seq[i] * rng.random()
        if delta <= 0:
            continue
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # clip and renorm
        cand = np.clip(cand, 0.0, 1000.0)
        cand = normalize_to_sum(cand, target)
        sc = max_convolution(cand)
        if sc < best_sc:
            best_seq, best_sc = cand, sc
        # occasional gradient descent step
        if rng.random() < 0.02:
            cand2 = gradient_step(best_seq, target, alpha=0.05)
            sc2 = max_convolution(cand2)
            if sc2 < best_sc:
                best_seq, best_sc = cand2, sc2
        # maybe also occasional gaussian perturb
        if rng.random() < 0.01:
            sigma = 0.02 * best_seq.mean()
            cand3 = best_seq + rng.normal(scale=sigma, size=n)
            cand3 = np.clip(cand3,0.0,1000.0)
            cand3 = normalize_to_sum(cand3, target)
            sc3 = max_convolution(cand3)

"""
 propose_candidate – stochastic search for a non‑negative step sequence
 that minimises `evaluate_sequence`.

The algorithm works in several stages:

0️⃣  Use a previously found construction (if present) as a warm start.
1️⃣  For a collection of lengths, generate cheap parametric families:
     – uniform, power‑law, (inverse) exponential, (inverse) linear,
     – random Dirichlet vectors and sparse random vectors.
   Keep the best candidate of each length.

2️⃣  Refine each candidate with a fast hill‑climbing routine.
    * The current maximal convolution entry is identified.
    * Its gradient (w.r.t. the sequence) is `g_i = 2·a_{k‑i}`.
    * Mass is transferred from indices with large gradient to those with
      small gradient; the ℓ¹‑norm is re‑normalised to √(2·n) after every
      move.
    * Occasionally a small “gradient step’’ and a Gaussian perturbation
      are performed to escape plateaus.

3️⃣  Finally a short Gaussian‑hill‑climb (standard local search)
    polish the best sequence found.

All steps respect the safety checks of `evaluate_sequence`
(non‑negative entries, clipping to [0,1000] and exact ℓ¹‑norm).
The routine returns the best sequence before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalize_to_sum(seq: np.ndarray, target: float) -> np.ndarray:
    """Scale *seq* so that its ℓ¹‑norm equals *target*."""
    total = seq.sum()
    if total == 0.0:
        return np.full_like(seq, target / len(seq), dtype=np.float64)
    return seq * (target / total)


def _max_convolution(seq: np.ndarray) -> float:
    """Exact max entry of the auto‑convolution of *seq*."""
    return np.convolve(seq, seq).max()


def _gradient_step(seq: np.ndarray, target_sum: float, alpha: float) -> np.ndarray:
    """
    One inexpensive step that reduces the current maximal convolution entry.

    *alpha* is a small learning rate (≈0.05 works well).
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # all indices where the max is attained (usually one)
    ks = np.where(conv >= max_val - 1e-12)[0]

    grad = np.zeros(n, dtype=np.float64)
    indices = np.arange(n)
    for k in ks:
        partner = k - indices
        mask = (partner >= 0) & (partner < n)
        grad[mask] += 2.0 * seq[partner[mask]]

    # zero‑sum direction (preserves total mass before renormalisation)
    direction = -grad + grad.mean()
    new = seq + alpha * direction
    new = np.clip(new, 0.0, 1000.0)
    return _normalize_to_sum(new, target_sum)


def _initial_candidates(
    n: int, target_sum: float, rng: np.random.Generator
) -> list[np.ndarray]:
    """Return a short list of cheap, diverse starting vectors for length *n*."""
    idx = np.arange(1, n + 1, dtype=np.float64)

    cand = []

    # uniform
    cand.append(np.full(n, target_sum / n, dtype=np.float64))

    # power‑law a_i ∝ i⁻ᵖ
    for p in np.linspace(0.2, 3.0, 7):
        a = idx ** (-p)
        a = _normalize_to_sum(a, target_sum)
        cand.append(a)

    # exponential a_i ∝ exp(-α·i)
    for α in np.linspace(0.001, 0.20, 7):
        a = np.exp(-α * np.arange(n, dtype=np.float64))
        a = _normalize_to_sum(a, target_sum)
        cand.append(a)

    # inverse exponential (mass grows towards the edges)
    mid = n // 2
    for α in np.linspace(0.001, 0.20, 5):
        dist = np.abs(np.arange(n) - mid)
        a = np.exp(α * dist)
        a = _normalize_to_sum(a, target_sum)
        cand.append(a)

    # inverse linear (zero at centre, larger at edges)
    a = np.abs(np.arange(n) - mid) + 1e-3
    a = _normalize_to_sum(a, target_sum)
    cand.append(a)

    # random Dirichlet‑like vectors
    for _ in range(5):
        a = rng.random(n) + 0.1
        a = _normalize_to_sum(a, target_sum)
        cand.append(a)

    # sparse random vectors (few non‑zeros)
    for _ in range(5):
        k = max(2, int(np.sqrt(n))) + rng.integers(0, int(np.sqrt(n)) + 1)
        pos = rng.choice(n, size=min(k, n), replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[pos] = rng.random(len(pos)) + 0.1
        a = _normalize_to_sum(a, target_sum)
        cand.append(a)

    return cand


def _hill_climb(
    start_seq: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing that moves mass from high‑gradient to low‑gradient
    indices.  Returns the best sequence found before *deadline*.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_sc = _max_convolution(best_seq)
    n = len(best_seq)

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # 1) compute gradient of the (possibly several) maximal entries
        # ------------------------------------------------------------------
        conv = np.convolve(best_seq, best_seq)
        max_val = conv.max()
        ks = np.where(conv >= max_val - 1e-12)[0]

        grad = np.zeros(n, dtype=np.float64)
        idx = np.arange(n)
        for k in ks:
            partner = k - idx
            mask = (partner >= 0) & (partner < n)
            grad[mask] += 2.0 * best_seq[partner[mask]]

        # donors – large gradient, receivers – small gradient
        donor_mask = grad > grad.mean()
        recv_mask = grad < grad.mean()
        donors = np.where(donor_mask)[0]
        receivers = np.where(recv_mask)[0]

        if donors.size == 0 or receivers.size == 0:
            # fall back to completely random pair
            i = rng.integers(n)
            j = rng.integers(n)
        else:
            i = rng.choice(donors)
            j = rng.choice(receivers)

        if best_seq[i] <= 1e-12:
            continue

        # transfer a random fraction of the donor's mass
        delta = 0.1 * best_seq[i] * rng.random()
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize_to_sum(cand, target_sum)

        sc = _max_convolution(cand)
        if sc < best_sc:
            best_seq, best_sc = cand, sc

        # ------------------------------------------------------------------
        # occasional gradient‑step or Gaussian perturbation (helps escape)
        # ------------------------------------------------------------------
        if rng.random() < 0.03:
            cand2 = _gradient_step(best_seq, target_sum, alpha=0.05)
            sc2 = _max_convolution(cand2)
            if sc2 < best_sc:
                best_seq, best_sc = cand2, sc2

        if rng.random() < 0.02:
            sigma = 0.02 * best_seq.mean()
            cand3 = best_seq + rng.normal(scale=sigma, size=n)
            cand3 = np.clip(cand3, 0.0, 1000.0)
            cand3 = _normalize_to_sum(cand3, target_sum)
            sc3 = _max_convolution(cand3)
            if sc3 < best_sc:
                best_seq, best_sc = cand3, sc3

    return best_seq, best_sc


def _final_gaussian_search(
    start_seq: np.ndarray, deadline: float, step_initial: float = 0.2
) -> tuple[np.ndarray, float]:
    """
    Short Gaussian hill‑climb (small step sizes, slowly decaying).
    Returns the best sequence discovered.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _max_convolution(best)
    n = len(best)
    target = math.sqrt(2.0 * n)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize_to_sum(cand, target)
        sc = _max_convolution(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
        step *= 0.9995
        if step < 1e-8:
            step = step_initial

    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The total allowed running time is *budget_s* seconds.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)  # keep a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Warm start from a previously found construction (if any)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = _normalize_to_sum(init, target0)
                sc0 = _max_convolution(init)
                best_seq, best_score = init, sc0
                print(f"[init] n={n0} score={sc0:.6f}")
        except Exception:  # pragma: no cover
            pass

    # --------------------------------------------------------------
    # 1️⃣  Parameter sweep over several lengths
    # --------------------------------------------------------------
    lengths = [300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 1800]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # generate cheap families
        candidates = _initial_candidates(n, target_sum, rng)
        # evaluate them
        for cand in candidates:
            sc = _max_convolution(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"[cand] n={n} score={sc:.6f}")

        # pick the best among the locally generated ones for further refinement
        local_start = min(candidates, key=_max_convolution)

        # --------------------------------------------------------------
        # 2️⃣  Stochastic hill‑climbing on this length
        # --------------------------------------------------------------
        refined_seq, refined_sc = _hill_climb(local_start, target_sum, deadline)
        if refined_sc < best_score:
            best_seq, best_score = refined_seq.copy(), refined_sc
            print(f"[hill] n={n} improved score={refined_sc:.6f}")

        # also try to improve the *global* best if it already has this length
        if best_seq is not None and best_seq.shape[0] == n:
            refined_global, refined_sc2 = _hill_climb(best_seq, target_sum, deadline)
            if refined_sc2 < best_score:
                best_seq, best_score = refined_global.copy(), refined_sc2
                print(f"[global] n={n} improved score={refined_sc2:.6f}")

        # early exit if the required target is already reached
        if best_score < 1.503:
            print("[early] reached target bound, exiting.")
            break

    # --------------------------------------------------------------
    # 3️⃣  Final short Gaussian polishing
    # --------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_sc = _final_gaussian_search(best_seq, deadline, step_initial=0.2)
        if final_sc < best_score:
            best_seq, best_score = final_seq, final_sc
            print(f"[final] polished score={final_sc:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _max_convolution(best_seq)

    # Return a plain Python list (required by the outer harness)
    return [float(x) for x in best_seq]
