def _conv_fft(seq):
    n = len(seq)
    N = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(fft_seq * fft_seq, n=N)

def _smooth_max(seq, beta=10.0):
    conv = _conv_fft(seq)   # length 2n-1
    max_val = np.max(conv)
    # compute softmax
    exp_vals = np.exp(beta * (conv - max_val))  # subtract for stability
    w = exp_vals / exp_vals.sum()
    # gradient: 2 * conv(w, rev(seq))
    rev = seq[::-1]
    grad_full = np.convolve(w, rev, mode='full')
    grad = 2 * grad_full[n-1:n-1+len(seq)]
    return w, grad

def _gradient_descent(seq, target_sum, deadline):
    n = len(seq)
    step = 0.2  # initial step size
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    while time.monotonic() < deadline:
        conv = _conv_fft(seq)
        k_max = int(conv.argmax())
        # compute gradient
        grad = np.zeros_like(seq)
        if k_max < n:
            grad[:k_max+1] = seq[k_max::-1]
        else:
            offset = k_max - (n - 1)
            grad[offset:] = seq[:n - offset][::-1]
        mean_g = grad.mean()
        direction = -grad + mean_g

        cand = seq + step * direction
        # clip and renorm
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total == 0.0:
            cand = np.full(n, target_sum / n, dtype=np.float64)
        else:
            cand = cand * (target_sum / total)

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best_seq, best_sc = cand.copy(), cand_sc
            seq = cand
            # increase step mildly to accelerate
            step = min(step * 1.05, 1.0)
        else:
            # no improvement: reduce step
            step = step * 0.5
            # also try a small random perturbation to escape local minima
            if step < 1e-6:
                # random small swap as fallback
                i = np.random.randint(n)
                j = np.random.randint(n)
                delta = seq[i] * 0.05  # 5% of mass at i
                if seq[i] > delta:
                    cand = seq.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand = cand * (target_sum / cand.sum())
                    cand_sc = _fast_score(cand)
                    if cand_sc < best_sc:
                        seq = cand
                        best_seq, best_sc = cand.copy(), cand_sc
                step = 0.1
        # occasional random injection
        if np.random.rand() < 0.001:
            # random full reinitialization with some probability
            seq = np.random.random(n)
            seq = seq / seq.sum() * target_sum
        # early stop
        if best_sc < 1.503:
            break
    return best_seq, best_sc

"""
Propose a low‑value step sequence.

The algorithm works in three stages:

1️⃣  **Exhaustive parametric sweep** – for several lengths we build many
    simple families (exponential, power‑law, Hann / Hamming / Blackman,
    sine, triangular, uniform).  The families are sampled densely,
    especially the exponential decay  α≈k/n  (k=1…20) where the optimum
    for the convolution‑max problem is known to lie.  The best candidate
    from this stage becomes the global incumbent.

2️⃣  **Mass‑transport gradient descent** – starting from the incumbent we
    repeatedly identify the convolution index *k* where the current
    auto‑convolution attains its maximum.  The sub‑gradient of that
    entry with respect to the step heights is simply the reversed slice
    of the sequence involved in the convolution.  We move opposite to
    the sub‑gradient, keep the ℓ¹‑norm exactly equal to √(2·n) by a
    simple scaling, and adapt the step size (increase after a success,
    halve on a failure).  This cheap first‑order method dramatically
    reduces the maximal auto‑convolution entry.

3️⃣  **Final polishing** – a short Gaussian hill‑climb (random perturbation
    followed by renormalisation) finishes the optimisation.

All operations respect the security checks of ``evaluate_sequence``:
non‑negative entries, clipping to [0,1000] and a fixed ℓ¹‑norm.
The routine returns the best sequence discovered before the time budget
runs out (the outer harness supplies ``evaluate_sequence``).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution via FFT (full linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of `seq` with itself using FFT."""
    n = seq.shape[0]
    # smallest power‑of‑two ≥ 2·n‑1
    N = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Exact objective value (without the security wrapper)."""
    n = seq.shape[0]
    max_b = _conv_fft(seq).max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Candidate families (parametric sweep)
# ----------------------------------------------------------------------
def _candidate_families(n: int, target_sum: float):
    """Return the best vector among many hand‑crafted families."""
    best_vec = None
    best_sc = float("inf")
    # helper to normalise a raw vector
    def _norm(v):
        v = np.asarray(v, dtype=np.float64)
        v = v / v.sum() * target_sum
        return v

    # uniform (baseline)
    a = np.full(n, target_sum / n, dtype=np.float64)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    idx = np.arange(n, dtype=np.float64)

    # --- exponential decay a_i ∝ exp(-α·i) ---------------------------
    # very fine grid around the theoretically promising region α≈k/n
    alphas = np.concatenate([
        np.arange(1, 6) / (10.0 * n),          # 0.0002 … 0.001 (for n≈500)
        np.arange(1, 21) / n,                  # 0.002 … 0.04
        np.linspace(0.02, 0.20, 40),           # coarser up to 0.2
        np.linspace(0.20, 1.00, 30)            # larger decays (optional)
    ])
    alphas = np.unique(alphas)
    for α in alphas:
        a = np.exp(-α * idx)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # --- power‑law a_i ∝ (i+1)^(-p) -------------------------------
    p_vals = np.linspace(0.05, 5.0, 120)
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        a = base ** (-p)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # --- classic window functions ---------------------------------
    # Hann
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Hamming
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Blackman
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1)) \
        + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # sine window (full half‑period)
    a = np.sin(np.pi * (idx + 1) / (n + 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # triangular (peak in the middle)
    a = np.minimum(idx + 1, n - idx)
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Gradient‑descent on the active max‑convolution entry
# ----------------------------------------------------------------------
def _gradient_descent(seq: np.ndarray, target_sum: float, deadline: float):
    """Iterative sub‑gradient descent on the maximal convolution entry."""
    n = seq.shape[0]
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    # start with a moderate step size; it will be adapted automatically
    step = 0.2
    rng = np.random.default_rng()

    while time.monotonic() < deadline:
        # current full convolution (FFT is cheap for moderate n)
        conv = _conv_fft(seq)
        k = int(conv.argmax())               # index of the active entry

        # sub‑gradient of the active entry (see analysis notes)
        grad = np.zeros_like(seq)
        if k < n:                             # overlap starts at index 0
            grad[: k + 1] = seq[k :: -1]       # reversed slice
        else:                                 # overlap is shifted
            off = k - (n - 1)
            grad[off:] = seq[: n - off][::-1]

        # direction that keeps the ℓ¹‑norm unchanged (mean‑shift)
        mean_g = grad.mean()
        direction = -grad + mean_g

        # propose a new point
        cand = seq + step * direction
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total == 0.0:                      # safety fallback
            cand = np.full(n, target_sum / n, dtype=np.float64)
        else:
            cand = cand * (target_sum / total)

        cand_sc = _fast_score(cand)

        # accept if we improve; otherwise shrink the step
        if cand_sc < best_sc:
            best_seq, best_sc = cand.copy(), cand_sc
            seq = cand
            step = min(step * 1.05, 1.0)      # grow a bit after success
        else:
            step *= 0.5
            if step < 1e-6:                  # occasional random shake‑up
                i, j = rng.integers(n, size=2)
                delta = seq[i] * 0.05
                if delta > 0:
                    cand = seq.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand = cand * (target_sum / cand.sum())
                    cand_sc = _fast_score(cand)
                    if cand_sc < best_sc:
                        seq = cand
                        best_seq, best_sc = cand.copy(), cand_sc
                step = 0.2

        # early exit if we already meet the target bound
        if best_sc < 1.503:
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Tiny Gaussian hill‑climb (final polish)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray, deadline: float,
                  step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """Short stochastic refinement using isotropic Gaussian perturbations."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to the required ℓ¹‑norm
        target = math.sqrt(2.0 * n)
        cand = cand * (target / cand.sum())

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        # slowly decay the exploration radius
        step *= 0.9997
        if step < 1e-7:
            step = step_initial
    return best, best_sc
