"""
Propose a low‑value step sequence.

Algorithm
---------
1. **Initialisation** – start from the global best (`height_sequence_1`) if
   present; otherwise perform a few random restarts for several lengths.
2. **Smooth‑max gradient descent** – minimise a smooth approximation of the
   worst auto‑convolution entry using a continuation on the smoothness
   parameter β.  Nesterov momentum and an adaptive step size speed up
   convergence.
3. **Local stochastic refinement** – after the gradient phase apply a
   Gaussian hill‑climbing (projected onto the simplex) to escape shallow
   minima.
4. **Best‑of‑all** – keep the globally best vector found during the whole
   budget, stop early as soon as the target score 1.503 is reached.

All steps respect the security checks of ``evaluate_sequence`` (non‑negative,
bounded entries and a positive ℓ¹‑norm).  The routine runs for at most the
provided time budget and returns a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` (no type checks, no prints).
    Returns ``np.inf`` if the ℓ¹‑norm is too small.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = float(conv.max())
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _score(seq: np.ndarray) -> float:
    """Thin wrapper around the fast score."""
    return _fast_score(seq)


def _smoothmax_gradient(seq: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max approximation
        φβ(seq) = (1/β) log Σ_k exp(β·conv_k)
    where conv = seq ∗ seq (linear convolution).

    Returns a vector of length n.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)            # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                  # for numerical stability
    exp_terms = np.exp(beta * shifted)      # all ≤ 1
    sum_exp = exp_terms.sum()
    if sum_exp == 0.0:
        # extremely unlikely; fall back to uniform gradient
        return np.zeros_like(seq)
    p = exp_terms / sum_exp                 # softmax weights, sum to 1
    # gradient = 2 * (p ∗ seq)   (first n entries)
    grad_full = np.convolve(p, seq)         # length 3n‑2
    grad = 2.0 * grad_full[:n]
    return grad


def _gradient_descent(
    init_seq: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_schedule=None,
    inner_iters_per_beta: int = 300,
    step_factor: float = 0.5,
    momentum_coeff: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max gradient descent with Nesterov momentum.
    Returns the best vector encountered and its score.
    """
    if beta_schedule is None:
        beta_schedule = [2, 4, 8, 16, 32, 64, 128, 256]

    n = init_seq.shape[0]
    f = init_seq.copy()
    best = f.copy()
    best_score = _score(best)
    current_score = best_score

    # momentum vector, initially zero
    velocity = np.zeros_like(f)

    for beta in beta_schedule:
        # break if out of time
        if time.monotonic() >= deadline:
            break
        for _ in range(inner_iters_per_beta):
            if time.monotonic() >= deadline:
                break

            # compute gradient of smooth‑max
            grad = _smoothmax_gradient(f, beta)

            # adaptive step size (inverse to gradient norm)
            grad_norm = np.linalg.norm(grad) + 1e-12
            step = step_factor * target_sum / grad_norm

            # Nesterov‑style update
            velocity = momentum_coeff * velocity - step * grad
            cand = f + velocity

            # enforce constraints
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _project_simplex(cand, target_sum)

            cand_score = _score(cand)

            # keep the global best
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[GD] n={n} β={beta:.1f} score={cand_score:.6f}")

                # early stop if we already beat the target
                if best_score < 1.503:
                    return best, best_score

            # accept the step if it improves the current objective
            if cand_score <= current_score:
                f = cand
                current_score = cand_score
            # otherwise keep the old f (no reset of velocity – it may help later)

    return best, best_score


def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.5,
    decay: float = 0.9995,
    min_step: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing (Gaussian perturbations) while keeping
    the ℓ¹‑norm fixed to the required target (the sum of ``start_seq``).
    """
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target_sum = float(start_seq.sum())
    best = start_seq.copy()
    best_score = _score(best)

    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[local] improved to {best_score:.6f}")
            if best_score < 1.503:
                break
        step *= decay
        if step < min_step:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the global time budget and returns a plain list of
    floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)  # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. Seed from the previous best if it exists
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seq0 = np.clip(seq0, 0.0, 1000.0)
        n0 = seq0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        if seq0.sum() > 0.0:
            seq0 = seq0 * (target0 / seq0.sum())
            sc0 = _score(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using global height_sequence_1 (n={n0}) score={sc0:.6f}")

    # --------------------------------------------------------------
    # 1. Main search over several lengths
    # --------------------------------------------------------------
    # lengths are chosen to give a reasonable trade‑off between expressivity
    # and computational effort.  We interleave short and long runs.
    lengths = [400, 500, 600, 800, 1000, 1200]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of random restarts per length (more restarts for cheaper lengths)
        restarts = 4 if n <= 600 else 2

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ------------------------------------------------------
            # Random initial vector (uniformly distributed mass)
            # ------------------------------------------------------
            f0 = rng.random(n) + 0.05  # avoid exact zeros
            f0 = f0 / f0.sum() * target_sum

            # ------------------------------------------------------
            # Gradient descent phase
            # ------------------------------------------------------
            # inner iterations are scaled roughly with sqrt(n) to keep total
            # work roughly constant across different n.
            inner_iters = max(150, int(3000 / math.sqrt(n)))
            seq_gd, score_gd = _gradient_descent(
                init_seq=f0,
                target_sum=target_sum,
                deadline=deadline,
                beta_schedule=[2, 4, 8, 16, 32, 64, 128],
                inner_iters_per_beta=inner_iters,
                step_factor=0.6,
                momentum_coeff=0.85,
            )
            if score_gd < best_score:
                best_seq = seq_gd.copy()
                best_score = score_gd
                print(f"[update] n={n} restart={r} gd_score={score_gd:.6f}")

            if best_score < 1.503:
                print("[early] target reached during GD")
                return best_seq.tolist()

            # ------------------------------------------------------
            # Local stochastic refinement
            # ------------------------------------------------------
            refined_seq, refined_sc = _local_search(seq_gd, deadline, step_initial=0.4)
            if refined_sc < best_score:
                best_seq = refined_seq.copy()
                best_score = refined_sc
                print(f"[local] n={n} restart={r} refined_score={refined_sc:.6f}")

            if best_score < 1.503:
                print("[early] target reached during local search")
                return best_seq.tolist()

    # --------------------------------------------------------------
    # 2. Final fallback (in case nothing was found)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 300
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)

    print(f"[final] best score={best_score:.6f} length={len(best_seq)}")
    return [float(x) for x in best_seq]
