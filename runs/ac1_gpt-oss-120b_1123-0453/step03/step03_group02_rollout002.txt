while time... and it<max_iters:

    conv = _conv_full(f) # compute full conv array
    M = conv.max()
    # Check best
    if M < best_val:
        best_val = M
        best_f = f.copy()

    # Determine active indices
    threshold = M - max(epsilon_rel * M, 1e-12) # relative tolerance
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(f)

        high = min(k, n - 1)
        # i_range = low..high
        i_range = np.arange(low, high + 1)
        # corresponding indices of f: k - i_range
        g[i_range] += 2.0 * f[k - i_range]

    if active.size > 0:

    else:

        pass

def _conv_full(f):
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    return conv

assistantfinal**Solution Overview**

The score of a step sequence  

\[
\text{score}= \frac{2n\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is invariant under scaling of the whole vector.  
Hence we can safely **fix the ℓ¹‑norm** to  

\[
\sum_i a_i = \sqrt{2n}\;,
\]

which makes the score equal to the largest entry of the auto‑convolution
\((a*a)_k\).  The problem is now:

*find a non‑negative vector on the simplex that minimises the maximum of its
self‑convolution*.

The previous implementation used a simple sub‑gradient with the *single*
worst index.  Here we improve the optimisation in three ways

1. **Active‑set sub‑gradient** – all indices whose convolution value is
   within a small relative tolerance of the maximum are used.  The
   sub‑gradient is taken as the average of their gradients, which is a
   valid element of the sub‑differential of the max‑function.

2. **Momentum (Nesterov‑like) updates** – a velocity term
   `v = β·v – η·g` is kept, allowing the optimiser to pass shallow
   plateaux more quickly.

3. **Larger problem sizes & more restarts** – we try several lengths
   up to 5000.  For each length we launch a few random starts and also
   embed the best sequence found so far (zero‑padding + simplex projection).

After the deterministic phase a short stochastic hill‑climbing
(Gaussian perturbations) cleans the result.

All operations respect the security checks of `evaluate_sequence`;
the code never writes files, uses only NumPy (and optionally CVXPY for a
quick LP refinement on very small dimensions) and stays well within the
1000 s budget.

```python
"""
Propose a low‑score step sequence.

The algorithm consists of three stages:

1. Optional LP refinement for very small lengths (≤ 300).  This is the
   exact idea from the reference implementation.

2. Deterministic optimisation on larger lengths using an *active‑set*
   sub‑gradient of the max‑convolution together with a simple
   momentum scheme.  The vector is always projected onto the ℓ¹‑simplex
   of radius sqrt(2·n) (non‑negative, fixed sum).  Several lengths
   (500 … 5000) are explored and many random restarts are performed.

3. A short stochastic polishing phase (Gaussian perturbations) that
   further reduces the maximal convolution value.

The routine stops a few hundred milliseconds before the global deadline
so that a valid sequence is always returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional CVXPY import – used only for tiny LP refinements.
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False

# ----------------------------------------------------------------------
# Helper: projection onto the ℓ¹‑simplex {x ≥ 0, Σ x = target}.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries with sum = target."""
    if target <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full auto‑convolution (real → real).  Returns the complete
# vector of length 2·n‑1.
# ----------------------------------------------------------------------
def _conv_full(f: np.ndarray) -> np.ndarray:
    """Full auto‑convolution of a real vector using FFT."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Return the maximal entry of the auto‑convolution and its index.
# ----------------------------------------------------------------------
def _fft_conv_max(f: np.ndarray):
    conv = _conv_full(f)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Wrapper around the provided evaluation function (list → float).
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Evaluate a candidate sequence using the given evaluate_sequence."""
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Optional LP refinement (tiny dimensions only).  Returns a possibly
# improved vector and its score.
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Perform a few iterations of the LP‑based refinement described
    in the reference.  Works only when cvxpy is available."""
    if not _HAS_CVX:
        return f, _score(f)

    n = f.shape[0]
    target = math.sqrt(2.0 * n)
    # Build convolution matrix once
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]

    t_mix = 0.6
    best_f, best_sc = f, _score(f)

    for it in range(120):  # modest number of iterations
        if time.monotonic() >= deadline:
            break
        M = float(np.max(np.convolve(f, f)))
        g = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                       abstol=1e-8, reltol=1e-6)
        except Exception:  # pragma: no cover
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=8000,
                       verbose=False)

        if g.value is None:
            break
        g0 = np.maximum(g.value, 0.0)
        if np.sum(g0) < 1e-12:
            break
        g0 = _project_simplex(g0, target)
        f_new = (1.0 - t_mix) * f + t_mix * g0
        f_new = np.clip(f_new, 0.0, 1000.0)
        sc = _score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
        f = f_new
    return best_f, best_sc


# ----------------------------------------------------------------------
# Deterministic optimisation with active‑set sub‑gradient and momentum.
# ----------------------------------------------------------------------
def _momentum_descent(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 3000,
    eta0: float = 1.0,
    decay: float = 0.9994,
    beta: float = 0.9,
    tol_rel: float = 1e-4,
    min_eta: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected momentum descent on the ℓ¹‑simplex.  Returns the best vector
    seen and its maximal convolution value.
    """
    n = f0.shape[0]
    f = f0.copy()
    v = np.zeros_like(f)
    best_f = f.copy()
    best_M, _ = _fft_conv_max(f)
    best_val = best_M
    eta = eta0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        conv = _conv_full(f)
        M = float(conv.max())
        if M < best_val:
            best_val = M
            best_f = f.copy()

        # ----- active set -----
        thresh = M - max(tol_rel * M, 1e-12)
        active = np.where(conv >= thresh)[0]

        # ----- sub‑gradient (average over active constraints) -----
        g = np.zeros_like(f)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            i_range = np.arange(low, high + 1, dtype=int)
            g[i_range] += 2.0 * f[k - i_range]
        if active.size > 0:
            g /= float(active.size)

        # ----- momentum update + projection -----
        v = beta * v - eta * g
        f_candidate = _project_simplex(f + v, target_sum)

        # ----- acceptance test -----
        M_cand = float(_conv_full(f_candidate).max())
        if M_cand <= M:
            f = f_candidate
            # if we improve we allow a slight increase in step size
            eta = min(eta0, eta / decay)
        else:
            # reject – reset velocity & shrink step
            v[:] = 0.0
            eta *= 0.5

        # global decay of the learning rate
        eta *= decay
        if eta < min_eta:
            eta = eta0

    return best_f, best_val


# ----------------------------------------------------------------------
# Small stochastic polishing (Gaussian perturbations).
# ----------------------------------------------------------------------
def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.7,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """Gaussian hill‑climbing with projection onto the simplex."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial
    n = best.shape[0]
    target = math.sqrt(2.0 * n)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= decay
        if step < 1e-7:
            step = step_initial
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The search proceeds through:
    1. optional LP refinement for very small lengths,
    2. momentum‑based projected sub‑gradient descent on several lengths
       (up to 5000) with many random restarts,
    3. a final stochastic polishing phase.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Overall time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a safety margin so that we always return something
    deadline = start_time + max(0.0, float(budget_s) - 0.3)

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # Initialise from previously discovered construction if available.
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = seq0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        if np.sum(seq0) > 0:
            seq0 = _project_simplex(seq0, target0)
            best_seq = seq0.copy()
            best_score = _score(seq0)
            print(f"[init] Loaded height_sequence_1 (n={n0}) → score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 0️⃣  Small‑n LP refinement (quick win for n ≤ 300)
    # -----------------------------------------------------------------
    if _HAS_CVX and time.monotonic() < deadline:
        for n in (200, 250, 300):
            if time.monotonic() >= deadline:
                break
            target = math.sqrt(2.0 * n)
            # start from uniform distribution
            f = np.full(n, target / n, dtype=np.float64)
            f, sc = _lp_refine(f, deadline)
            if sc < best_score:
                best_score, best_seq = sc, f.copy()
                print(f"[LP‑refine] n={n} → score={sc:.6f}")

    # -----------------------------------------------------------------
    # 1️⃣  Momentum based deterministic optimisation on growing lengths.
    # -----------------------------------------------------------------
    lengths = [500, 800, 1200, 1800, 2500, 3500, 5000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ----- build a list of start vectors -----
        starts = []

        # embed previous best if possible
        if best_seq is not None:
            if best_seq.shape[0] == n:
                starts.append(best_seq.copy())
            elif best_seq.shape[0] < n:
                tmp = np.zeros(n, dtype=np.float64)
                tmp[: best_seq.shape[0]] = best_seq
                tmp = _project_simplex(tmp, target)
                starts.append(tmp)

        # a few purely random starts
        for _ in range(3):
            f0 = rng.random(n) + 0.05  # avoid exact zeros
            f0 = _project_simplex(f0, target)
            starts.append(f0)

        # ----- run optimiser for each start -----
        for sid, f_start in enumerate(starts):
            if time.monotonic() >= deadline:
                break
            # give each start a modest private deadline (half of remaining time)
            remaining = deadline - time.monotonic()
            sub_deadline = time.monotonic() + min(remaining * 0.5, 12.0)  # at most 12 s per start

            f_opt, M_opt = _momentum_descent(
                f_start,
                target_sum=target,
                deadline=sub_deadline,
                max_iters=2500,
                eta0=1.0,
                decay=0.9995,
                beta=0.85,
                tol_rel=1e-4,
                min_eta=1e-7,
            )
            sc = _score(f_opt)
            if sc < best_score:
                best_score, best_seq = sc, f_opt.copy()
                print(f"[GD] n={n} start={sid} → score={sc:.6f} (M={M_opt:.6f})")
            if best_score < 1.503:
                print("[GD] Target reached – exiting early.")
                return best_seq.tolist()

    # -----------------------------------------------------------------
    # 2️⃣  Final stochastic polishing.
    # -----------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of moderate size
        n = 600
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = _project_simplex(best_seq, math.sqrt(2.0 * n))
        best_score = _score(best_seq)

    if time.monotonic() < deadline:
        refined, refined_sc = _local_search(best_seq, deadline, step_initial=0.6)
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[local] polishing improved score to {best_score:.6f}")

    # Return a plain Python list for the harness.
    return [float(x) for x in best_seq]
```
