def annealing_search(a0, conv0, sum_a, deadline, max_iters=500_000):
    a = a0.copy()
    conv = conv0.copy()
    cur_score = 2*n*np.max(conv)/(sum_a*sum_a)
    best_a = a.copy(); best_score = cur_score
    rng = np.random.default_rng()
    temp = start_temp
    it=0
    while time.monotonic() < deadline and it < max_iters:
        # choose i (source) with weight a[i] (prefer bigger) using random choice.
        i = rng.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # target j uniformly random
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # delta fraction
        # step size maybe: log uniform: draw u in [log(min_delta), log(max_delta)], delta = exp(u)
        # but easier: delta = a[i]*rng.random()*step_factor where step_factor decays with time.
        step_factor = temp  # using temperature as fraction scaling.
        delta = a[i] * step_factor * rng.random()
        if delta <= 0:
            continue
        _transfer_mass(a, conv, i, j, delta)
        new_score = 2*n*np.max(conv)/(sum_a*sum_a)
        # accept rule
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score)/temp):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            # revert
            _transfer_mass(a, conv, j, i, delta)
        # temperature decay
        temp = max(temp * 0.99999, 1e-6)
        it += 1
    return best_a, best_score
```

    conv = conv0.copy()
    n = a.shape[0]
    cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    temp = start_temp
    step = step_scale
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # find current max index
        idx_max = int(np.argmax(conv))
        # compute contributions for each i
        # contributions = np.zeros_like(a)
        # We'll compute using vectorized: for each i, j = idx_max - i
        # mask = (j>=0) & (j<n)
        # contributions[mask] = a[mask] * a[j[mask]]
        # However we need j array.
        j = idx_max - np.arange(n)
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[j[mask]]
        # Pick source i using weighted distribution proportional to contrib (plus small epsilon)
        if contrib.sum() > 0:
            prob_i = contrib / contrib.sum()
        else:
            prob_i = a / a.sum() if a.sum() > 0 else np.full(n, 1.0/n)
        i = rng.choice(n, p=prob_i)
        if a[i] <= 1e-12:
            continue
        # target j pick via low contributions: compute inverse weights
        # We can pick uniformly from indices with low contribution:
        # compute prob_target proportional to 1/(contrib+epsilon)
        eps = 1e-12
        inv = 1.0 / (contrib + eps)
        prob_j = inv / inv.sum()
        # ensure j != i
        # we can sample until j != i
        j_target = rng.choice(n, p=prob_j)

            # try another may be uniform

                _transfer_mass(a, conv, j_target, i, delta)

        if step < 1e-5:
            step = step_scale
        temp *= 0.99995
        if temp < 1e-7:
            temp = start_temp
        it += 1

"""
Search for a low‑scoring non‑negative step sequence.

The algorithm is a multi‑start simulated‑annealing / hill‑climbing hybrid:

1.  Several diverse initial vectors are generated (uniform, exponential,
    a handcrafted “edge‑heavy’’ shape and the previously known construction
    `height_sequence_1` if available).  All vectors are normalised to the
    standard ℓ¹‑norm √(2 n); the objective is scale‑invariant.

2.  For each start a *mass‑transfer annealing* is performed.
    A move consists of transferring a small amount of mass from a source
    index to a target index while keeping the total ℓ¹‑norm constant.
    The convolution `a*a` is updated incrementally in O(n) using the
    algebraic formula from the reference code.  The cheap score
        2·n·max(conv) / (∑a)²
    is evaluated after each move.
    Moves that improve the score are always accepted; moves that worsen
    the score are accepted with probability exp(−Δ/temperature), where the
    temperature slowly decays.  The step‑size (maximum transferred fraction)
    also decays, and every 20 000 iterations the vector is symmetrised
    (`a ← (a + a[::-1]) / 2`) to encourage a flat autocorrelation.

3.  The best vector from the annealing phase is polished by a strict
    hill‑climber (`_improve_by_transfer`) that only keeps improving moves,
    followed by a gentle Gaussian local search (`_local_search`).

The routine respects the global time budget (default 1000 s) and returns
the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official score via the provided `evaluate_sequence`."""
    # evaluate_sequence expects a Python list.
    return float(evaluate_sequence(seq.tolist()))


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index `i` to index `j` (i != j, delta > 0, delta <= a[i]).
    Updates `a` and its self‑convolution `conv` in‑place (O(n) work).
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part: 2 * delta * a
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part: convolution of the delta vector with itself
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta          # (-δ)²
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta    # cross term
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta          # (+δ)²

    # update the vector
    a[i] -= delta
    a[j] += delta


def _fast_score(a_len: int, conv: np.ndarray, sum_a: float) -> float:
    """Score using the cheap formula (scale invariant)."""
    return 2.0 * a_len * np.max(conv) / (sum_a * sum_a)


def _anneal_transfer(start_a: np.ndarray,
                     start_conv: np.ndarray,
                     sum_a: float,
                     deadline: float,
                     max_iters: int = 200_000,
                     start_temp: float = 0.04,
                     step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing using mass‑transfer moves.
    Returns the best vector seen (in terms of the cheap score) and that score.
    """
    a = start_a.copy()
    conv = start_conv.copy()
    n = a.shape[0]
    cur_score = _fast_score(n, conv, sum_a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    temp = start_temp
    step = step_init
    it = 0

    # pre‑allocate a range array for contribution computation
    idx_range = np.arange(n, dtype=int)

    while time.monotonic() < deadline and it < max_iters:
        # --- pick source weighted by current mass (more mass = more likely) ---
        if a.sum() == 0:
            break
        prob_src = a / a.sum()
        i = rng.choice(n, p=prob_src)

        if a[i] <= 1e-12:
            # nothing to move, skip
            continue

        # --- pick a target (uniform) ---
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # --- amount to move --------------------------------------------------
        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        # --- apply move ------------------------------------------------------
        _transfer_mass(a, conv, i, j, delta)

        # --- evaluate cheap score --------------------------------------------
        new_score = _fast_score(n, conv, sum_a)

        # --- acceptance -------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert the transfer
            _transfer_mass(a, conv, j, i, delta)

        # --- slowly decay step size and temperature -------------------------
        step *= 0.9999
        if step < 1e-5:
            step = step_init

        temp *= 0.99995
        if temp < 1e-7:
            temp = start_temp

        # --- occasional symmetrisation to encourage a flat autocorrelation ----
        if it % 20_000 == 0 and it > 0:
            a = (a + a[::-1]) * 0.5
            conv = np.convolve(a, a)
            cur_score = _fast_score(n, conv, sum_a)

        it += 1

    return best_a, best_score


def _improve_by_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing that only keeps strictly improving moves.
    Returns the best sequence (according to the cheap score) and that score.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)
    sum_a = a.sum()
    best_a = a.copy()
    best_score = _fast_score(n, conv, sum_a)

    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        _transfer_mass(a, conv, i, j, delta)
        cur_score = _fast_score(n, conv, sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            _transfer_mass(a, conv, j, i, delta)

        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Gentle Gaussian perturbation used as a final polishing step.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score

    # 1️⃣  Candidate lengths (larger n gives more freedom but costs time)
    # ------------------------------------------------------------------
    lengths = [500, 600, 800]   # the 500‑length already gave good results

    # optional global starting point
    try:
        global_start = np.array(height_sequence_1, dtype=float)
    except Exception:
        global_start = None

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)   # ℓ¹‑norm to which we will normalise

        # ------------------------------------------------------------------
        # 2️⃣  Build a portfolio of diverse initial vectors
        # ------------------------------------------------------------------
        starts: list[np.ndarray] = []

        # uniform vector (all entries equal)
        starts.append(np.full(n, target_sum / n, dtype=float))

        # exponential random vector (sparse like shape)
        starts.append(rng.exponential(scale=1.0, size=n))

        # edge‑heavy handcrafted shape: a_i ∝ 1/√(i+1) + 1/√(n-i)
        idx = np.arange(n, dtype=float)
        edge_shape = 1.0 / np.sqrt(idx + 1.0) + 1.0 / np.sqrt(n - idx)
        starts.append(edge_shape)

        # previously discovered construction, if length matches
        if global_start is not None and global_start.shape[0] == n:
            starts.append(global_start.copy())

        # ------------------------------------------------------------------
        # 3️⃣  Optimise each start
        # ------------------------------------------------------------------
        for vec in starts:
            if time.monotonic() >= deadline:
                break

            # normalise to the required ℓ¹‑norm (scale does not affect the ratio)
            s = vec.sum()
            if s == 0.0:
                vec = np.full(n, target_sum / n, dtype=float)
            else:
                vec = vec * (target_sum / s)

            vec = np.clip(vec, 0.0, 1000.0)

            # initial convolution
            conv = np.convolve(vec, vec)

            # ---- 3a️⃣  Simulated‑annealing mass‑transfer --------------------
            a_annealed, _ = _anneal_transfer(vec, conv, target_sum, deadline,
                                             max_iters=150_000,
                                             start_temp=0.04,
                                             step_init=0.5)

            # official evaluation
            official = _score(a_annealed)
            if official < best_score:
                best_score = official
                best_seq = a_annealed.copy()
                print(f"[anneal] n={n} score={best_score:.6f}")

                if best_score < 1.503:
                    return best_seq.tolist()

            # ---- 3b️⃣  Strict hill‑climbing refinement ----------------------
            a_refined, _ = _improve_by_transfer(a_annealed, deadline,
                                                max_iters=100_000,
                                                step_init=0.3)
            official_ref = _score(a_refined)
            if official_ref < best_score:
                best_score = official_ref
                best_seq = a_refined.copy()
                print(f"[refine] n={n} score={best_score:.6f}")

                if best_score < 1.503:
                    return best_seq.tolist()

            # ---- 3c️⃣  Final Gaussian polishing ----------------------------
            if time.monotonic() < deadline:
                a_final, final_score = _local_search(best_seq, deadline,
                                                     step_initial=0.2)
                official_final = _score(a_final)
                if official_final < best_score:
                    best_score = official_final
                    best_seq = a_final.copy()
                    print(f"[final] n={n} score={best_score:.6f}")

                    if best_score < 1.503:
                        return best_seq.tolist()

    # ------------------------------------------------------------------
    # 4️⃣  Fallback – return the best we have (or a simple uniform vector)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 300
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    return [float(x) for x in best_seq]
