def _gradient_descent(a, deadline):
    n = a.shape[0]
    target_sum = 1.0
    # scaling factor for gradient (learning rate) maybe start lr=0.1, adapt.
    lr = 0.1
    gamma = 10.0
    gamma_factor = 1.02  # increase each iteration
    step_decay = 0.999
    max_iters = 200000
    it = 0
    best_a = a.copy()
    best_val = np.inf
    while time.monotonic() < deadline and it < max_iters:
        # compute conv using FFT
        L = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]
        max_conv = conv.max()
        # compute objective value (2*n * max_conv)
        val = 2.0 * n * max_conv
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # compute softmax weights
        # shift to improve numeric stability
        shift = conv.max()
        exp_term = np.exp(gamma * (conv - shift))
        w = exp_term / exp_term.sum()
        # compute gradient via convolution of reversed a and w
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        # gradient for indices 0..n-1 corresponds to grad_full[n-1:n-1+n]
        grad = 2.0 * grad_full[n-1:n-1+n]
        # descent step
        a_new = a - lr * grad
        # project onto simplex
        a = _project_onto_simplex(a_new, target_sum)
        # schedule
        lr *= step_decay
        if lr < 1e-6:
            lr = 0.1
        gamma *= gamma_factor
        if gamma > 200.0:
            gamma = 200.0
        it += 1
    # after loop, evaluate official
    return best_a, best_val
```

"""
Propose a step sequence with a low evaluation score.

Algorithm
---------
1️⃣  Initialise several candidate vectors on the probability simplex.
    – use the previously found global `height_sequence_1` if present,
    – uniform start,
    – Dirichlet (sparse) starts.

2️⃣  For each start run a *projected gradient descent* that minimises the
    smooth approximation of  max(conv(a,a)) :

        •  The convolution a*a is obtained by an FFT (O(L log L) where
           L is the next power‑of‑two ≥ 2·n‑1).

        •  A soft‑max with temperature γ turns the max‑norm into a smooth
           function.  γ is increased slowly (annealing) so the optimisation
           first moves the mass globally and then focuses on the largest
           convolution entry.

        •  The gradient of the smooth max is
                ∇ f(a)_j = 2·∑_k w_k·a_{k‑j} ,
           where w is the soft‑max weight vector.  This gradient is computed
           by a single additional FFT (convolution of the reversed vector
           with w).

        •  After each step the vector is projected back onto the simplex
           {x ≥ 0, Σx = 1}.  The projection is O(n) (the classic “water‑filling”
           algorithm).

        •  Learning‑rate and γ are scheduled (decay / anneal) and the loop
           stops when the time budget is exhausted.

3️⃣  The best vector found by the gradient phase is further polished by the
    mass‑transfer hill‑climber from the reference implementation
    (`_improve_by_transfer`).  Finally a cheap Gaussian local search
    (`_local_search`) is applied.

The whole search respects the 1000 s budget; intermediate scores are printed
so progress can be monitored.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Official score using the provided `evaluate_sequence`."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        { x ≥ 0 , Σ x = target_sum }.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Gradient‑descent with smooth max on the auto‑convolution
# ----------------------------------------------------------------------


def _grad_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr0: float = 0.05,
    gamma_start: float = 5.0,
    gamma_end: float = 150.0,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent that minimises the smooth max of ``a*a``.
    Returns the best vector encountered (on the simplex Σa = 1) and its
    *official* evaluation score.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = 1.0

    # pre‑computed FFT length
    L = _next_pow2(2 * n - 1)

    # schedules
    lr = lr0
    gamma = gamma_start
    gamma_factor = math.pow(gamma_end / gamma_start, 1.0 / max_iters)

    best_a = a.copy()
    # official score = 2·n·max(convolution) because Σa = 1
    conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[: 2 * n - 1]
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # ----- compute convolution (full) -----
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        # ----- smooth‑max weights -----
        shift = conv.max()                 # for numerical stability
        w = np.exp(gamma * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # should not happen, but guard against divide‑by‑zero
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----- gradient (2·a_rev * w) -----
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # length n

        # ----- gradient step + projection -----
        a_new = a - lr * grad
        a = _project_onto_simplex(a_new, target_sum)

        # ----- keep the best ever seen -----
        conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- schedule updates -----
        lr *= 0.9995                      # slow decay
        if lr < 1e-7:
            lr = lr0
        gamma = min(gamma_end, gamma * gamma_factor)

        it += 1

    # final official score
    official = _score(best_a)
    return best_a, official


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (same as reference, kept for polishing)
# ----------------------------------------------------------------------


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index `i` to `j` (i != j, delta > 0, delta ≤ a[i]).
    Updates ``a`` *in‑place* and also updates the full convolution ``conv``
    (also in‑place) using the exact algebraic formula (O(n)).
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    delta = min(delta, a[i])

    # linear part 2·a*δ
    conv[i : i + n] -= 2.0 * delta * a
    conv[j : j + n] += 2.0 * delta * a

    # quadratic part (δ at i, –δ at j)
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta

    # update the vector
    a[i] -= delta
    a[j] += delta


def _improve_by_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that repeatedly moves a tiny amount of mass from one index
    to another while keeping the ℓ¹‑norm fixed.  The convolution is updated
    incrementally, so each iteration costs only O(n).
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)               # exact convolution
    sum_a = a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        _transfer_mass(a, conv, i, j, delta)

        cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            _transfer_mass(a, conv, j, i, delta)   # revert

        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Gentle stochastic polishing – add Gaussian noise and keep improvements.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score
