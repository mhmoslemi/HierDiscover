"""
Propose a low‑value step sequence.

The algorithm works in three stages:

1️⃣  **Initialisation** – use the previously found construction (if any)
    and generate a few deterministic starts (uniform, “arcsine‑shaped”)
    and random starts for each chosen length.

2️⃣  **Smoothed projected gradient descent** – minimise a smooth
    approximation of  max_k (f*f)_k .
    The soft‑max parameter α is increased during the run.
    The gradient of the smooth max is computed by FFT‑based
    correlation, a momentum term is added and the iterate is
    projected onto the ℓ¹‑simplex (fixed sum = √(2·n), non‑negative).

3️⃣  **Simple stochastic polishing** – a short Gaussian‑perturbation
    local search that respects the simplex constraint.

All operations are performed with NumPy (FFT, convolutions) and respect
the security checks of `evaluate_sequence`.  The routine stops a few
hundred milliseconds before the global deadline and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – ℓ¹‑simplex projection (Σx = target, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = target}."""
    if target <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast auto‑convolution (full) using FFT
# ----------------------------------------------------------------------
def _auto_convolution(f: np.ndarray) -> np.ndarray:
    """Return the full auto‑convolution f*f (length 2·n‑1)."""
    n = f.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (soft‑max) of the auto‑convolution and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(f: np.ndarray, alpha: float):
    """
    Compute a smooth approximation of max_k (f*f)_k and its gradient.
    The smooth max is  M_α = (log Σ exp(α·c_k))/α  where c = f*f.
    Gradient = 2·correlation(w, f)  with w = softmax(α·c).
    """
    n = f.shape[0]

    # ----- convolution c = f*f -----
    c = _auto_convolution(f)                     # length 2·n‑1

    # ----- soft‑max weights -----
    max_c = np.max(c)
    # shift to avoid overflow
    shifted = np.exp(alpha * (c - max_c))
    w = shifted / shifted.sum()                  # Σ w_k = 1

    # ----- smooth max value (not strictly needed for optimisation) -----
    smooth_max = max_c + np.log(shifted.sum()) / alpha

    # ----- gradient = 2·correlation(w, f) -----
    f_rev = f[::-1]
    size = 1 << ((len(w) + len(f_rev) - 1).bit_length())
    W = np.fft.rfft(w, size)
    F_rev = np.fft.rfft(f_rev, size)
    corr_full = np.fft.irfft(W * F_rev, size)[: len(w) + len(f_rev) - 1]
    # correlation slice needed: Σ_j w_{i+j} f_j  for i = 0..n‑1
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]

    return smooth_max, grad


# ----------------------------------------------------------------------
# Fast exact evaluation (identical to evaluate_sequence but without the
# security wrappers – we stay in the safe regime of non‑negative numbers ≤1000)
# ----------------------------------------------------------------------
def _evaluate(f: np.ndarray) -> float:
    """Return the exact score 2·n·max(f*f) / (sum(f))²."""
    n = f.shape[0]
    sum_a = f.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _auto_convolution(f)
    max_b = float(conv.max())
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Deterministic “arcsine‑like” start (heavy at the edges, light in the centre)
# ----------------------------------------------------------------------
def _arcsine_start(n: int) -> np.ndarray:
    """Generate a non‑negative vector of length n proportional to
    1/sqrt(i·(n‑i+1)) (i = 1…n) – a discrete analogue of the arcsine law."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = 1.0 / np.sqrt(i * (n - i + 1.0))
    return vals

# Smoothed projected gradient descent with momentum
# ----------------------------------------------------------------------
def _optimize_sgd(

    Returns the best vector seen and its exact score.
    """

    best_score = cur_score

    max_step = target_sum * 0.5

            alpha = 0.5
        elif it < 500:
            alpha = 2.0
        elif it < 1500:
            alpha = 5.0
        else:
            alpha = 12.0

        # gradient of the smoothed max
        _, grad = _smooth_max_and_grad(f, alpha)

        # momentum update
        mom = beta * mom + (1.0 - beta) * grad

        # propose a new point and project onto the simplex
        f_cand = _project_simplex(f - step * mom, target_sum)
        cand_score = _evaluate(f_cand)

        # accept only if we improve (ensures monotonic descent)
        if cand_score < cur_score:
            f = f_cand
            cur_score = cand_score
            # modestly increase step after a success
            step = min(step * 1.04, max_step)
        else:
            # shrink step after a failure
            step = max(step * 0.7, min_step)

        # keep track of the best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best_f = f_cand.copy()
            if it % 100 == 0:
                print(f"[sgd] it={it:4d} score={best_score:.6f} step={step:.6f}")

        # early exit if we reach the target region
        if best_score < 1.503:
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (Gaussian perturbations) for polishing
# ----------------------------------------------------------------------
def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.5,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best = start_seq.copy()
    best_score = _evaluate(best)
    step = step_initial

    while time.time() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        # keep inside the admissible range
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)
        cand_score = _evaluate(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= decay
        if step < 1e-8:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the 1000 s time budget and returns the best
    sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 0️⃣  Load the previously discovered construction (if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = seq.shape[0]
        target0 = math.sqrt(2.0 * n0)
        if np.sum(seq) > 0:
            seq = seq / np.sum(seq) * target0
            sc = _evaluate(seq)
            best_seq = seq.copy()
            best_score = sc
            best_n = n0
            print(f"[init] height_sequence_1 (n={n0}) → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Choose lengths to explore (larger n gives more flexibility)
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2400, 3000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # 1️⃣a Build a collection of starting points for this length
        # ------------------------------------------------------------------
        starts = []

        # embed the current global best if compatible
        if best_seq is not None:
            if best_n == n:
                starts.append(best_seq.copy())
            elif best_n < n:
                padded = np.zeros(n, dtype=np.float64)
                padded[:best_n] = best_seq
                padded = _project_simplex(padded, target_sum)
                starts.append(padded)

        # deterministic “arcsine‑shaped” start
        arcs = _arcsine_start(n)
        arcs = arcs / arcs.sum() * target_sum
        starts.append(arcs)

        # uniform start (always feasible)
        uni = np.full(n, 1.0, dtype=np.float64)
        uni = _project_simplex(uni, target_sum)
        starts.append(uni)

        # a few random restarts
        for _ in range(2):
            rnd = rng.random(n) + 0.01
            rnd = rnd / rnd.sum() * target_sum
            starts.append(rnd)

        # ------------------------------------------------------------------
        # 1️⃣b Run the optimiser for every start (time‑sliced)
        # ------------------------------------------------------------------
        for sid, f0 in enumerate(starts):
            if time.time() >= deadline:
                break

            # allocate up to ~30 s or the remaining time (whichever is smaller)
            remaining = deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining * 0.4)

            f_opt, scr_opt = _optimize_sgd(f0, target_sum, sub_deadline)
            if scr_opt < best_score:
                best_score = scr_opt
                best_seq = f_opt.copy()
                best_n = n
                print(f"[gd] n={n} start={sid} → new best {best_score:.6f}")

            if best_score < 1.503:
                print("[early] target achieved – exiting.")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 2️⃣  Final polishing with a cheap stochastic local search
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall‑back to a simple uniform vector of moderate length
        n = 800
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = _project_simplex(best_seq, target)
        best_score = _evaluate(best_seq)

    if time.time() < deadline:
        refined, refined_score = _local_search(best_seq, deadline)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[local] polished best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list (the harness expects that)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)   # safety clip
    return [float(v) for v in best_seq]
