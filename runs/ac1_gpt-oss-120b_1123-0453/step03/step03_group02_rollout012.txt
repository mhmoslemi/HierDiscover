    max_b = conv.max()

"""
Propose a low‑evaluation step sequence.

Algorithm
---------

* Choose a few candidate lengths (200 … 1500).
* For each length initialise a random non‑negative vector on the
  ℓ¹‑simplex (sum = 1).  The exact scale does not affect the objective.
* Optimise the vector with a fast **Adam**‑style projected gradient method.
  The sub‑gradient of   M(f)=max_k (f*f)_k   is taken at the *K* worst
  convolution indices (default K=5); this yields a stronger descent
  direction than using a single worst index.
* After each update the vector is projected back onto the simplex
  (non‑negative, fixed sum) using the O(n) algorithm of
  Duchi‑Shalev‑Shwartz‑Singer‑Chandra.
* The best vector seen for the current length is kept.
* When the global time budget (default 1000 s) is almost exhausted a
  cheap stochastic hill‑climbing phase (Gaussian flips) refines the
  current global best.
* The routine returns a plain Python list of floats; the evaluation
  harness will clip values to [0,1000] and compute the official score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: project onto the ℓ¹‑simplex {x≥0, Σx = target}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = target}."""
    if target <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution (full, not circular) using FFT
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (real inputs)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence but cheap, no clipping)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return the evaluation value 2·n·max(conv)/sum²."""
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    n = seq.shape[0]
    conv = _conv_full(seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam optimiser for the nonsmooth problem (sub‑gradient at top‑K)
# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 15000,
    K: int = 5,
    lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Return (best_vector, best_score) after Adam‑type sub‑gradient descent."""
    # initialise on the simplex (sum = 1)
    x = rng.random(n) + 1e-6
    x = _project_simplex(x, 1.0)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_x = x.copy()
    best_score = _objective(x)

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # full auto‑convolution
        conv = _conv_full(x)

        # indices of the K largest entries (worst constraints)
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            # argpartition is O(n) and gives the K largest unsorted
            top_idx = np.argpartition(conv, -K)[-K:]

        # sub‑gradient = sum_{k∈top} 2·shifted_x
        g = np.zeros_like(x)
        for k in top_idx:
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            if low <= high:
                i_range = np.arange(low, high + 1)
                # a_{k-i}
                g[i_range] += 2.0 * x[k - i_range]

        # Adam update (step uses the sub‑gradient)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex (sum = 1, non‑negative)
        x_new = _project_simplex(x_new, 1.0)

        x = x_new

        # evaluate periodically
        if it % 10 == 0:
            sc = _objective(x)
            if sc < best_score:
                best_score = sc
                best_x = x.copy()
                print(f"[Adam] n={n:4d} it={it:5d} score={sc:.6f}")
            # early stop if we are already below the target
            if best_score < 1.45:
                # very good – stop the inner loop early
                break

    return best_x, best_score


# ----------------------------------------------------------------------
# Small stochastic refinement (Gaussian flips on the best vector)
# ----------------------------------------------------------------------
def _local_refine(seq: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """Gaussian perturbation hill‑climbing, keeps improvements."""
    best = seq.copy()
    best_score = _objective(best)
    step = 0.3
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)          # keep within hard caps
        cand = _project_simplex(cand, 1.0)        # preserve scale (irrelevant)
        sc = _objective(cand)
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[Refine] improved to {sc:.6f}")
        step *= 0.9995
        if step < 1e-6:
            step = 0.3
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The routine runs several Adam‑based optimisations for different
    lengths, keeps the globally best vector, and finishes with a light
    stochastic polishing phase.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 0️⃣  Warm‑start from a previously discovered construction, if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
        sc0 = _objective(seq0)
        best_seq, best_score = seq0.copy(), sc0
        print(f"[init] loaded previous sequence → score={sc0:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Adam optimisation for several lengths
    # ------------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000, 1200]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # allocate at most half the remaining time to this length
        remaining = deadline - time.monotonic()
        sub_dead = time.monotonic() + min(remaining * 0.5, 30.0)   # max 30 s per n

        # a few random restarts to diversify search
        for restart in range(3):
            if time.monotonic() >= sub_dead:
                break
            opt_vec, opt_score = _adam_optimize(
                n=n,
                deadline=sub_dead,
                rng=rng,
                max_iters=8000,
                K=7,
                lr=0.12,
            )
            if opt_score < best_score:
                best_score, best_seq = opt_score, opt_vec.copy()
                print(f"[global] new best n={n} restart={restart} score={opt_score:.6f}")

            # early exit if we already passed a very good threshold
            if best_score < 1.45:
                print("[global] target reached early.")
                break

    # ------------------------------------------------------------------
    # 2️⃣  Final stochastic refinement of the global best
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined = _local_refine(best_seq, deadline, rng)
        refined_score = _objective(refined)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final] polished best → score={best_score:.6f}")

    # Guard against the unlikely case of no candidate
    if best_seq is None:
        # simple fallback: uniform vector of length 400 (score = 2)
        n = 400
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = _project_simplex(best_seq, 1.0)

    # Return plain Python list (the evaluator will clip internally)
    return [float(v) for v in best_seq.tolist()]
