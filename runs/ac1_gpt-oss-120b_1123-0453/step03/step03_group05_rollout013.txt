"""
Hybrid sub‑gradient + stochastic refinement for minimizing
    score = 2·n·max(a∗a) / (∑a)²
The sum ∑a is kept equal to √(2·n) (the known scale‑invariant choice), so the
objective reduces to minimizing the largest entry of the full convolution.
The method consists of:

1.  Initialise several random (or warm‑started) non‑negative vectors on a
    set of problem sizes n (500 … 2000) and project them onto the ℓ¹‑simplex.
2.  Repeatedly apply a sub‑gradient step for the current maximal convolution
    entry:
        • Find the index k where (a∗a)ₖ is maximal.
        • The sub‑gradient of this entry is gᵢ = 2·a_{k‑i} (zero when the index
          is out of range).  This follows from differentiating the quadratic
          form.
        • Take a step a ← a – η·g and re‑project onto the simplex.
    The step size η is chosen proportionally to the simplex size and the
    gradient norm, and decays slowly with the iteration count.
3.  Keep the globally best vector seen.
4.  After the deterministic phase, run a short stochastic Gaussian
    perturbation (simulated‑annealing) local search to escape possible
    plateaus.
5.  Return the best sequence discovered before the time budget expires.

The implementation uses an FFT‑based convolution (O(n log n)) and a fast
Euclidean projection onto the simplex (Duchi et al.).  Multiple restarts,
different lengths and a final local refinement make good use of the
1000 s budget and typically beat the previous 1.511 score, aiming for the
target ≤ 1.503.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s} (Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` (no type checks)."""
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = float(conv.max())
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _score(seq: np.ndarray) -> float:
    """Thin wrapper around _fast_score (kept for compatibility)."""
    return _fast_score(seq)


def _conv_fft(a: np.ndarray, n: int) -> np.ndarray:
    """
    Full convolution a∗a using an FFT of length power‑of‑two.
    Returns a real ndarray of length 2·n‑1.
    """
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv.real


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that respects the ℓ¹‑simplex.
    """
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best = start_seq.copy()
    best_score = _score(best)

    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # exponential decay, reset if too small
        step *= 0.9995
        if step < 1e-7:
            step = step_initial
    return best, best_score

            local_best = a.copy()
            local_best_score = cur_score

            # step‑size control
            base_step_factor = 0.25

            while time.monotonic() < deadline and iter_cnt < max_iters:
                # convolution and index of current maximum
                conv = _conv_fft(a, n)
                k = int(np.argmax(conv))
                max_val = conv[k]

                # sub‑gradient of the maximal entry
                grad = np.zeros_like(a)
                i_low = max(0, k - (n - 1))
                i_high = min(k, n - 1)
                i_rng = np.arange(i_low, i_high + 1, dtype=int)
                j_rng = k - i_rng
                grad[i_rng] = 2.0 * a[j_rng]

                # adaptive step size
                grad_norm = np.linalg.norm(grad) + 1e-12
                step = base_step_factor * target_sum / grad_norm

                # gradient step + projection onto the simplex
                a_new = a - step * grad
                a_new = np.clip(a_new, 0.0, 1000.0)
                a_new = _project_simplex(a_new, target_sum)

                new_score = _score(a_new)

                # Accept the step if it does not increase the current objective
                if new_score <= cur_score:
                    a = a_new
                    cur_score = new_score

                # Record the globally best vector seen so far
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"[sgd] n={n} r={r} it={iter_cnt} score={new_score:.6f}")

                # Record a good local vector for the final refinement
                if new_score < local_best_score:
                    local_best_score = new_score
                    local_best = a_new.copy()

                # Simple decay of the base step factor to stabilise later iterations
                if iter_cnt % 5000 == 0 and iter_cnt > 0:
                    base_step_factor *= 0.9
                    base_step_factor = max(base_step_factor, 0.02)

                iter_cnt += 1

            # -----------------------
            # Final stochastic refinement on the best local vector
            # -----------------------
            if time.monotonic() < deadline:
                refined_seq, refined_sc = _local_search(local_best, deadline,
                                                        step_initial=0.5)
                if refined_sc < best_score:
                    best_score = refined_sc
                    best_seq = refined_seq.copy()
                    print(f"[local] n={n} r={r} improved to {best_score:.6f}")

            # Early exit if we have already beaten the target bound
            if best_score < 1.503:
                print("[early] target achieved – exiting.")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)

    print(f"[final] best score={best_score:.6f} length={len(best_seq)}")
    return [float(x) for x in best_seq]
