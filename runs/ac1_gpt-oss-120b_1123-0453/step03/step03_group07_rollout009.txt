def project_to_simplex(v, s):
    """Return the projection of v onto simplex {x >=0, sum x = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho = max i where u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    """Project v onto the simplex {x >= 0, sum x = s}."""
    # based on algorithm from Duchi et al. (2008)

    return w

"""
Propose a step‑function sequence with a low max‑auto‑convolution value.

Algorithm
---------

* For each candidate length *n* (starting from a few hundred up to a few
  thousand) we create an initial non‑negative vector whose entries follow the
  discrete arcsine law (  a_i ∝ 1/√[(i+½)(n‑½‑i)] ).  This distribution is
  known to produce a relatively flat self‑convolution.

* The vector is projected onto the simplex
  {x ≥ 0, ∑x = √(2 n)} – this matches the normalisation used by the
  ``evaluate_sequence`` function.

* Starting from that point we run a projected sub‑gradient descent:
  – compute the self‑convolution via FFT;
  – locate the indices where the convolution attains its current maximum
    (the “active” constraints);
  – the sub‑gradient of the max‑norm is the sum of the corresponding rows of
    the convolution matrix;  we evaluate it efficiently by a short loop.
  – take a step opposite to the sub‑gradient, with a diminishing step size,
    and re‑project onto the simplex.
  – if no improvement is seen for several iterations a tiny random
    perturbation is applied to escape plateaus.

* Several random Dirichlet restarts are also tried for each length.
  The best sequence seen over the whole run is returned.

The routine respects the 1000 s time budget, prints progress
information and never uses closures or file/network I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector onto the simplex {x ≥ 0, sum(x) = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")

    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)          # 1‑based indices
    # Find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _self_convolution_fft(f: np.ndarray) -> np.ndarray:
    """
    Compute the (full) self‑convolution of ``f`` using an FFT.
    Returns a real array of length 2·len(f)‑1.
    """
    n = f.shape[0]
    # size for a power‑of‑two FFT that can contain the full convolution
    m = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(f, n=m)
    conv = np.fft.irfft(F * F, n=m)
    return conv[: 2 * n - 1]


def _dirichlet_start(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex via a Dirichlet(1,…,1) distribution."""
    x = rng.dirichlet(np.ones(n))
    return x.astype(np.float64) * target_sum


def _arcsine_init(n: int, target_sum: float) -> np.ndarray:
    """
    Initialise a vector with the discrete arcsine law:
        a_i ∝ 1 / sqrt[(i+½)(n‑½‑i)] ,   i = 0,…,n‑1
    The result is projected onto the required simplex.
    """
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    w_sum = w.sum()
    f = w * (target_sum / w_sum)
    return f


def _subgradient_from_active(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the max‑convolution norm.
    For each active convolution index k we add the row of the convolution
    matrix that corresponds to k, i.e. the vector (f_{k‑i})_{i}.
    """
    n = f.shape[0]
    grad = np.zeros_like(f)
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # vector of i where the term exists
        i_range = np.arange(i_start, i_end + 1)
        grad[i_range] += f[k - i_range]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence minimising
    ``evaluate_sequence`` (equivalently minimising the infinity norm of its
    self‑convolution after normalising the sum to sqrt(2·n)).
    """
    start_time = time.time()
    # keep a tiny safety margin so we do not overrun the budget
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # Try a progression of lengths; larger n gives a lower theoretical bound.
    lengths = [256, 512, 1024, 2048, 4096]
    # Number of independent random restarts per length
    restarts_per_n = 3
    # Control of inner optimisation (more iterations for smaller n)
    base_iters = 600

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        lower_bound = 2.0 * n / (2.0 * n - 1.0)   # average value of the convolution

        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------------
            # Choose a starting point
            if restart == 0:
                f = _arcsine_init(n, target_sum)
                init_name = "arcsine"
            else:
                f = _dirichlet_start(n, target_sum, rng)
                init_name = "random"

            # Compute initial objective
            conv = _self_convolution_fft(f)
            M = float(conv.max())
            if M < best_score:
                best_score = M
                best_seq = f.copy()
                print(f"[init] n={n:4d} restart={restart} type={init_name}  M={M:.6f}")

            # ------------------------------------------------------------------
            # Projected sub‑gradient descent with occasional perturbations
            max_iters = max(50, base_iters // (n // 256 + 1))
            no_improve = 0
            for it in range(max_iters):
                if time.time() >= deadline:
                    break

                conv = _self_convolution_fft(f)
                M = float(conv.max())

                # Record best found so far
                if M < best_score:
                    best_score = M
                    best_seq = f.copy()
                    print(f"[best] n={n:4d} restart={restart} it={it:4d}  M={M:.6f}")

                # Early exit if we are essentially at the theoretical lower bound
                if M <= lower_bound * 1.005:
                    print(f"[near‑lb] n={n} restart={restart} it={it}  M={M:.6f}")
                    break

                # Indices where the convolution attains the current maximum
                eps_active = 1e-12
                active_mask = conv >= M - eps_active
                active_idx = np.nonzero(active_mask)[0]

                # Sub‑gradient of the max‑norm
                grad = _subgradient_from_active(f, active_idx)

                norm_grad = np.linalg.norm(grad)
                if norm_grad == 0.0:
                    # Degenerate case – nothing to improve
                    break

                # Diminishing step size (sqrt schedule)
                step_coeff = 0.6
                step = step_coeff * target_sum / (norm_grad * math.sqrt(it + 1))

                f_candidate = _project_onto_simplex(f - step * grad, target_sum)
                conv_cand = _self_convolution_fft(f_candidate)
                M_cand = float(conv_cand.max())

                if M_cand < M - 1e-12:
                    f = f_candidate
                    no_improve = 0
                    continue
                else:
                    no_improve += 1

                # If we have stalled for a while, inject a small random perturbation
                if no_improve >= 8:
                    perturb = rng.normal(scale=0.01 * target_sum / n, size=n)
                    f = _project_onto_simplex(f + perturb, target_sum)
                    no_improve = 0
                    continue

    # ----------------------------------------------------------------------
    # Fallback – if nothing was discovered (should never happen)
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using uniform vector")

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    # Ensure the final output complies with the evaluator's guards
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    best_seq = _project_onto_simplex(best_seq, math.sqrt(2.0 * len(best_seq)))

    return [float(v) for v in best_seq]
