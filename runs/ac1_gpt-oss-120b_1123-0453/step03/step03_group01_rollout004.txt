def _proj_gradient_descent(start_seq, deadline, max_iters=5000, lr=0.01):
    a = start_seq.copy()
    n = len(a)
    target_sum = np.sqrt(2.0*n)
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        conv = np.convolve(a, a)
        max_val = np.max(conv)
        # active indexes
        active = np.where(conv >= max_val - 1e-12)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for t in active:
            # vectorized shift
            # for i from 0 to n-1, j = t - i
            # contributions where 0 <= j < n
            # we can compute slice of a aligned
            lo = max(0, t - (n - 1))
            hi = min(n - 1, t)
            # indices i = lo..hi, j = t - i
            i_range = np.arange(lo, hi + 1)
            j_range = t - i_range
            # For those where i != j
            mask = i_range != j_range
            grad[i_range[mask]] += 2 * a[j_range[mask]]
            grad[i_range[~mask]] += a[i_range[~mask]]   # diagonal case
        # step opposite gradient
        direction = -grad
        # simple line search: step size decays
        step = lr / (1 + it * 0.001)

        # project onto simplex
        a_new = _project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
        # optionally accept if score improves
        cur_score = _score(a)
        new_score = _score(a_new)
        if new_score < cur_score:
            a = a_new
        else:
            # maybe reduce lr
            lr *= 0.9
    return a, _score(a)

"""
Propose a low‑score step sequence.

The search consists of three phases for each tried length `n`:

1️⃣  Initialise several candidates (including the global `height_sequence_1`
    if it matches the length).  All candidates are projected onto the
    simplex { x≥0, Σx = √(2n) } – the objective is scale‑invariant.

2️⃣  **Mass‑transfer hill‑climbing** – repeatedly move a small amount of
    mass from one index to another while updating the auto‑convolution
    incrementally (O(n) per move).  Only strictly improving moves are kept.

3️⃣  **Linear‑programming polishing** – given the current vector `f`,
    compute `M = max(f*f)`.  Solve the LP

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   (k = 0…2n‑2)
                    b_j ≥ 0

    which finds a vector `b` that pushes as much mass as possible without
    raising the convolution bound.  Mix `f` and the normalised `b`
    (step size `t`) and repeat a few times.

4️⃣  **Gradient‑descent fine‑tuning** on the active constraints (the
    indices attaining the current maximum), followed by a gentle
    Gaussian perturbation (`_local_search`).

The whole procedure respects the 1000 s wall‑time limit and returns the
best sequence found (as a plain Python list of non‑negative floats).
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper functions used by the optimiser
# ----------------------------------------------------------------------


def _cheap_score(a: np.ndarray) -> float:
    """Score identical to evaluate_sequence but without its safety checks."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return float(2.0 * n * np.max(conv) / (np.sum(a) ** 2))


def _score(a: np.ndarray) -> float:
    """Official score (calls the provided evaluate_sequence)."""
    return float(evaluate_sequence(a.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _transfer_mass(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """Transfer `delta` from index i to j and update the convolution in‑place."""
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part: 2 * delta * a
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part: convolution of the delta‑vector with itself
    L = 2 * n - 1
    ii = 2 * i
    if ii < L:
        conv[ii] += delta * delta
    ij = i + j
    if ij < L:
        conv[ij] -= 2.0 * delta * delta
    jj = 2 * j
    if jj < L:
        conv[jj] += delta * delta

    # update the vector
    a[i] -= delta
    a[j] += delta


def _improve_by_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing by moving a tiny amount of mass between two coordinates.
    The convolution is kept up‑to‑date incrementally.
    Returns the best vector encountered (still on the simplex) and its cheap score.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)                # exact convolution
    sum_a = np.sum(a)
    best_a = a.copy()
    best_score = _cheap_score(a)

    rng = np.random.default_rng()
    step = step_init
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        _transfer_mass(a, conv, i, j, delta)
        cur_score = _cheap_score(a)   # cheap – uses the updated conv internally

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert
            _transfer_mass(a, conv, j, i, delta)

        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Gentle Gaussian perturbation (stochastic hill‑climbing) used for final polishing.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑project to the simplex to keep the sum fixed
        cand = _project_onto_simplex(cand, np.sqrt(2.0 * len(cand)))
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score


def _solve_lp_for_vector(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j
            subject to (f * b)_k ≤ M   for all k,  b ≥ 0.
    Returns the optimal b (as a dense NumPy array) or None on failure.
    """
    n = len(f)
    L = 2 * n - 1
    # Build the convolution matrix A where (A @ b)_k = (f * b)_k
    A = np.zeros((L, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f  # each column i contributes f shifted by i
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M + 1e-9]   # tiny slack for numerical safety
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500, verbose=False)
    except Exception:
        return None
    if b.value is None:
        return None
    return np.asarray(b.value, dtype=float)


def _lp_improve(start_seq: np.ndarray,
                deadline: float,
                max_iter: int = 12,
                step: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Alternating linear‑programming improvement.
    For the current vector f compute M = max(f*f), solve the LP to obtain b,
    normalise b to the required ℓ¹‑norm and mix f←(1‑t)·f + t·b.
    """
    a = start_seq.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        M = np.max(np.convolve(a, a))
        b = _solve_lp_for_vector(a, M)
        if b is None:
            break
        sum_b = np.sum(b)
        if sum_b <= 0:
            break
        # normalise b to the same ℓ¹‑norm as a
        g = b * (target_sum / sum_b)
        # convex combination
        a = (1.0 - step) * a + step * g
        # re‑project to the simplex to eliminate numerical drift
        a = _project_onto_simplex(a, target_sum)
        step = max(step * 0.9, 0.05)   # slowly decay the mixing factor
    return a, _cheap_score(a)


def _gradient_refine(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 3000,
                     lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the active constraints
    (indices where the auto‑convolution attains its maximum).
    """
    a = start_seq.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        conv = np.convolve(a, a)
        max_val = np.max(conv)
        # active indices (allow a tiny tolerance)
        active = np.where(conv >= max_val - 1e-12)[0]
        if active.size == 0:
            break

        # sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        for t in active:
            # allowed i such that 0 ≤ i < n and 0 ≤ t‑i < n
            lo = max(0, t - (n - 1))
            hi = min(n - 1, t)
            i_vals = np.arange(lo, hi + 1)
            j_vals = t - i_vals
            mask = i_vals != j_vals
            # off‑diagonal contributions (appear twice in the sum)
            grad[i_vals[mask]] += 2.0 * a[j_vals[mask]]
            # diagonal contributions (appear once)
            grad[i_vals[~mask]] += a[i_vals[~mask]]

        # take a step opposite to the sub‑gradient
        step = lr / (1.0 + it * 0.001)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # accept if the cheap score improves
        if _cheap_score(a_new) < _cheap_score(a):
            a = a_new
        else:
            lr *= 0.9    # shrink step size if we cannot improve
    return a, _cheap_score(a)

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the given time budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    # leave a small margin for post‑processing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    target_sum = lambda n: math.sqrt(2.0 * n)

    # Lengths to experiment with – larger `n` gives more freedom but costs more time.
    lengths = [500, 800, 300, 600, 400, 200]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        t_sum = target_sum(n)

        # ------------------- initialise several starts -------------------
        starts: list[np.ndarray] = []

        # Global variable from previous runs, if applicable
        try:
            hs1 = np.array(height_sequence_1, dtype=float)
            if hs1.shape[0] == n:
                starts.append(hs1.copy())
        except Exception:
            pass

        # Uniform start
        starts.append(np.full(n, t_sum / n, dtype=float))

        # A few random starts (exponential or uniform)
        for _ in range(4):
            if rng.random() < 0.5:
                vec = rng.exponential(scale=1.0, size=n)
            else:
                vec = rng.random(n) + 0.1
            vec = vec * (t_sum / np.sum(vec))
            starts.append(vec)

        # ------------------- optimise each start -------------------
        for start in starts:
            if time.monotonic() >= deadline:
                break

            # 1️⃣  Mass‑transfer hill climbing
            cand, _ = _improve_by_transfer(start, deadline,
                                           max_iters=120_000,
                                           step_init=0.6)

            # 2️⃣  Linear‑programming polishing
            cand, _ = _lp_improve(cand, deadline,
                                  max_iter=12,
                                  step=0.5)

            # 3️⃣  Gradient‑descent fine‑tuning
            cand, _ = _gradient_refine(cand, deadline,
                                       max_iters=2500,
                                       lr=0.02)

            # 4️⃣  Final stochastic polishing
            if time.monotonic() < deadline:
                cand, _ = _local_search(cand, deadline,
                                        step_initial=0.3)

            # official evaluation (safety checks are performed inside)
            official = _score(cand)
            if official < best_score:
                best_score = official
                best_seq = cand.copy()
                print(f"[update] n={n} score={best_score:.6f}")

            # early exit if we already beat the target bound
            if best_score < 1.503:
                return best_seq.tolist()

    # fallback – should never happen but keeps the interface safe
    if best_seq is None:
        n = 300
        best_seq = np.full(n, target_sum(n) / n, dtype=float)

    return [float(x) for x in best_seq]
