def smooth_max_update(f, beta, step_size):
    conv = np.convolve(f, f)
    max_val = conv.max()
    # compute stable exponentials
    exp_vals = np.exp(beta * (conv - max_val))
    w = exp_vals / exp_vals.sum()
    # compute gradient
    grad = np.zeros_like(f)
    n = len(f)
    for k in range(2*n - 1):
        wk = w[k]
        if wk == 0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_elements = np.arange(i_start, i_end + 1)
        j = k - i_elements
        # contributions
        grad[i_elements] += wk * 2.0 * f[j]
        mask = (i_elements == j)
        if np.any(mask):
            grad[i_elements[mask]] -= wk * f[j][mask]
    # proceed with step: f_new = projection_onto_simplex(f - step_size * grad, target_sum)
    return f_new

def weighted_gradient(f, w):
    n = len(f)
    grad = np.zeros_like(f)
    for k, wk in enumerate(w):
        if wk == 0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += wk * 2.0 * f[j]
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= wk * f[j][mask]
    return grad

def smooth_max_gd(f0, target_sum, deadline, beta_start=0.5, beta_factor=1.5, beta_max=500, step_initial=0.1, max_iters=1000):

    f = f0.copy()
    best_f = f.copy()
    best_score = compute_score(f)

    step = step_initial
    beta = beta_start

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        max_val = conv.max()
        # compute softmax weights
        exp_term = np.exp(beta * (conv - max_val))
        w = exp_term / exp_term.sum()

        grad = weighted_gradient(f, w)

        f_candidate = project_onto_simplex(f - step * grad, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)
        cand_score = compute_score(f_candidate)

        # If improvement, accept and maybe increase step
        if cand_score < best_score:
            best_f, best_score = f_candidate.copy(), cand_score
            f = f_candidate
            step *= 1.02
            # optionally reduce beta if near optimum? Actually increase.
        else:
            step *= 0.5

        # update beta gradually (increase to focus on max)
        beta = min(beta * beta_factor, beta_max)

        # termination if step size tiny
        if step < 1e-12:
            break

    return best_f, best_score

"""
Simple stochastic + smoothed‐max gradient descent for the step‑function bound.

The algorithm works as follows:

*   The ℓ¹‑simplex (∑ aᵢ = √(2 n), aᵢ ≥ 0) is used – this makes the
    evaluation equal to the maximal entry of the linear convolution
    a*a, i.e. we minimise the peak of the auto‑convolution.

*   For a given length n we start from a random point on the simplex.
    A *smooth maximum* (log‑sum‑exp) with an annealing parameter β is
    used as a differentiable proxy for the true max.  The gradient of the
    proxy is a weighted sum of the gradients of all convolution offsets,
    which can be computed in O(n²) using simple NumPy slices.

*   After each gradient step the iterate is projected back onto the
    simplex (Euclidean projection).  Step size and β are adapted
    automatically; a lack of improvement triggers a step‑size shrinkage
    and a temporary reduction of β to re‑explore the landscape.

*   Several lengths (300…1000) and many random restarts are tried while
    respecting the global time budget.  The best candidate is finally
    polished with a short Gaussian hill‑climbing phase.

All numbers are clipped to the interval required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_score(v: np.ndarray) -> float:
    """Exact evaluation of the provided ``evaluate_sequence`` formula."""
    n = len(v)
    conv = np.convolve(v, v)
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _weighted_gradient(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k * (f*f)_k  w.r.t. f.
    w must have length 2·len(f)‑1 and sum(w)=1.
    """
    n = len(f)
    grad = np.zeros_like(f)
    # loop over each convolution offset
    for k in range(2 * n - 1):
        wk = w[k]
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i indices that participate in this offset
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        grad[i_start:i_end + 1] += wk * 2.0 * f[j_range]
        # correct the diagonal contribution (i == j)
        if (k & 1) == 0:          # k even → possible i=j
            i_eq = k // 2
            if i_start <= i_eq <= i_end:
                grad[i_eq] -= wk * f[i_eq]        # 2·c – c = c
    return grad


def _smooth_max_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2000,
    beta_start: float = 0.5,
    beta_factor: float = 1.5,
    beta_max: float = 256.0,
    step_initial: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smoothed max of the auto‑convolution.
    Returns the best vector seen and its score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)

    step = step_initial
    beta = beta_start
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        max_val = conv.max()

        # soft‑max weights (numerically stable)
        exp_vals = np.exp(beta * (conv - max_val))
        w_sum = exp_vals.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_vals / w_sum

        grad = _weighted_gradient(f, w)

        cand = _project_onto_simplex(f - step * grad, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best_f, best_score = cand.copy(), cand_score
            f = cand
            step *= 1.02
            no_improve = 0
        else:
            step *= 0.7
            no_improve += 1

        # anneal β – slowly focus on the true maximum
        beta = min(beta * beta_factor, beta_max)

        # if we have been stuck for a while, reduce β a bit to broaden the search
        if no_improve >= 200:
            beta = max(beta_start, beta / 2.0)
            no_improve = 0

        if step < 1e-12:
            break

    return best_f, best_score


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.2,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Small Gaussian perturbation hill‑climbing keeping the vector on the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        sigma *= 0.9995
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial
    return best, best_score
