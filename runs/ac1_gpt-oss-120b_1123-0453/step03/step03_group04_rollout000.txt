"""
Gradient‑max (soft‑max) descent for the self‑convolution sup‑norm.

The evaluation metric reduces (for a fixed ℓ¹‑norm √(2 n)) to the
maximum entry of the full convolution a*a.  We minimise this quantity
by a smooth approximation:

    φ_λ(a) = (1/λ)·log Σ_k exp(λ·c_k) ,   c = a*a

∇φ_λ can be written in closed form using a single convolution,
so each iteration costs O(n²) with n ≤ 1500 – cheap enough for the
1000 s budget.

The algorithm:
  1. initialise several candidates (previous best, uniform, random Dirichlet,
     simple power‑law families).
  2. for each candidate run projected gradient descent on φ_λ,
     slowly raising λ to focus on the true maximum.
  3. keep the best vector seen.
  4. finish with a cheap stochastic Gaussian hill‑climb (as in the
     reference solution) to polish the final answer.

All vectors are kept non‑negative, clipped to [0,1000] and re‑scaled
to the required ℓ¹‑norm after every update.  The routine returns the
best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (fast, no security wrapper)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact value of evaluate_sequence when Σseq = √(2·n)."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _softmax_gradient(seq: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate φ_λ(a) = (1/λ)·log Σ exp(λ·c_k)
    where c = a * a (full convolution).  Returns ∇φ_λ (same length as seq).
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)                     # length 2n‑1
    max_c = conv.max()
    # stable soft‑max weights
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:        # should never happen, but guard against NaNs
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # reversed sequence for the convolution that yields the gradient
    rev = seq[::-1]
    full = np.convolve(w, rev)                      # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]               # ∇φ_λ has factor 2
    return grad


def _project(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and renormalise to the required ℓ¹‑norm."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s <= 0.0:
        # fall back to a tiny uniform distribution
        seq = np.full_like(seq, target_sum / seq.size)
    else:
        seq = seq * (target_sum / s)
    return seq


def _gradient_descent(
    init: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 2500,
    lam_start: float = 2.0,
    lam_end: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate.
    Returns the best vector seen and its (exact) max‑convolution value.
    """
    a = init.copy()
    best = a.copy()
    best_score = _fast_score(a)

    n = a.size
    it = 0
    # step size is expressed as a factor of the current mean entry
    step_factor = 0.6
    no_improve = 0

    while time.monotonic() < deadline and it < max_iters:
        # current max (exact) – also used for early exit
        cur_score = _fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # stop if we have stalled too long
        if no_improve > 200:
            break

        # anneal λ – start smooth, finish close to true max
        lam = lam_start * (lam_end / lam_start) ** (it / max_iters)

        # compute gradient of the smooth surrogate
        grad = _softmax_gradient(a, lam)

        # project a small step along the negative gradient
        step = step_factor * max(a.mean(), 1e-12)
        a_new = a - step * grad
        a_new = _project(a_new, target_sum)

        # accept if we did not increase the exact max; otherwise shrink step
        new_score = _fast_score(a_new)
        if new_score <= cur_score:
            a = a_new
            # mildly increase the step factor when we make progress
            step_factor = min(1.0, step_factor * 1.01)
        else:
            step_factor *= 0.5
            # occasional random perturbation to escape a flat region
            if it % 100 == 0:
                perturb = rng.normal(scale=0.02 * target_sum / n, size=n)
                a_pert = _project(a + perturb, target_sum)
                if _fast_score(a_pert) < cur_score:
                    a = a_pert

        it += 1

        # early exit if we beat the target bound
        if best_score < 1.503:
            break

    return best, best_score


def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.2,
) -> tuple[np.ndarray, float]:
    """Short Gaussian hill‑climb (identical to the reference implementation)."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = best.size
    target = math.sqrt(2.0 * n)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target / cand.sum())
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9997
        if step < 1e-7:
            step = step_initial
    return best, best_sc

            v = rng.random(n) + 0.1
            v = v / v.sum() * target_sum
            starters.append(v)

        # d) simple power‑law families (quick sweep)
        idx = np.arange(1, n + 1, dtype=np.float64)
        for p in (0.5, 1.0, 1.5, 2.0):
            a = idx ** (-p)
            a = a / a.sum() * target_sum
            starters.append(a)

        # ------------------------------------------------------------------
        # 3️⃣  Gradient descent from each starter (time‑controlled)
        # ------------------------------------------------------------------
        for start_vec in starters:
            if time.monotonic() >= deadline:
                break

            seq_opt, sc_opt = _gradient_descent(
                start_vec,
                target_sum,
                deadline,
                rng,
                max_iters=2000,
                lam_start=2.0,
                lam_end=30.0,
            )
            if sc_opt < best_score:
                best_seq, best_score = seq_opt.copy(), sc_opt
                print(f"[gd]   n={n}  score={sc_opt:.6f}")

            if best_score < 1.503:
                # Target reached – we can stop early.
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 4️⃣  Final stochastic polishing (swap + Gaussian hill‑climb)
    # ------------------------------------------------------------------
    if best_seq is None:
        # deterministic fallback – moderate size uniform vector
        n = 400
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # cheap random swap hill‑climb
    while time.monotonic() < deadline:
        n = best_seq.size
        target_sum = math.sqrt(2.0 * n)
        avg = best_seq.mean()
        delta = max(1e-5, avg * 0.02)
        donors = np.where(best_seq > delta)[0]
        if donors.size == 0:
            break
        i = rng.choice(donors)
        j = rng.integers(n)
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_score:
            best_seq, best_score = cand, cand_sc
            print(f"[swap] improved to {best_score:.6f}")

    # short Gaussian hill‑climb
    if time.monotonic() < deadline:
        refined_seq, refined_sc = _local_search(best_seq, deadline, step_initial=0.2)
        if refined_sc < best_score:
            best_seq, best_score = refined_seq, refined_sc
            print(f"[final] improved to {best_score:.6f}")

    print(f"Best score achieved: {best_score:.6f} (length {best_seq.size})")
    return [float(x) for x in best_seq]
