def _generate_initial_candidates(n, rng):
    cand = []
    # uniform
    cand.append(np.full(n, 1.0/n))

    # monotonic power law
    for p in rng.uniform(0.2, 3.0, size=3):
        a = np.arange(1, n+1, dtype=np.float64) ** (-p)
        cand.append(a / a.sum())

    # monotonic exponential
    for alpha in rng.uniform(0.001, 0.2, size=3):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand.append(a / a.sum())

    # edge power law
    for p in rng.uniform(0.2, 3.0, size=3):
        i = np.arange(n, dtype=np.float64)
        a = ((i + 0.5) * (n - i - 0.5)) ** (-p)
        cand.append(a / a.sum())

    # edge exponential
    for alpha in rng.uniform(0.001, 0.2, size=3):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-alpha * np.minimum(i, n - 1 - i))
        cand.append(a / a.sum())

    # random Dirichlet (i.e., gamma(1))
    for _ in range(2):
        a = rng.random(n)
        cand.append(a / a.sum())

    return cand

"""
Propose a low‑value step sequence.

The search strategy is a blend of:
1.  A rich parametric sweep (monotone and edge‑peaked families, exponentials,
    power‑laws, uniform and random Dirichlet vectors).
2.  A fast projected (additive) gradient descent that uses the active
    convolution index to reduce the maximal auto‑convolution entry.
    An optional mirror‑descent (multiplicative) update is interleaved.
3.  A short stochastic polishing stage (Gaussian hill‑climb) to polish the
    best vector found.
All vectors are kept on the probability simplex (non‑negative, sum = 1);
the evaluation function is invariant under scaling, so the ℓ¹‑norm
does not affect the objective.  The routine runs for at most the given
budget (default 1000 s) and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast objective (same as evaluate_sequence but without safety checks)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return 2·n·max(convolution) / (sum(seq)^2)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Generate a handful of deterministic / random initial vectors
# ----------------------------------------------------------------------
def _initial_vectors(n: int, rng: np.random.Generator):
    """Yield several simple shapes for a given length n (already ℓ¹‑normalised)."""
    # uniform
    yield np.full(n, 1.0 / n)

    # monotone power‑law a_i ∝ (i+1)^(-p)
    for p in rng.uniform(0.2, 3.0, size=3):
        a = (np.arange(1, n + 1, dtype=np.float64)) ** (-p)
        a /= a.sum()
        yield a

    # monotone exponential a_i ∝ exp(-α·i)
    for alpha in rng.uniform(0.001, 0.20, size=3):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        a /= a.sum()
        yield a

    # edge‑peaked power‑law a_i ∝ ((i+0.5)*(n‑i‑0.5))^(-p)
    i = np.arange(n, dtype=np.float64)
    for p in rng.uniform(0.2, 3.0, size=3):
        a = ((i + 0.5) * (n - i - 0.5)) ** (-p)
        a /= a.sum()
        yield a

    # edge‑peaked exponential a_i ∝ exp(-α·min(i, n‑1‑i))
    for alpha in rng.uniform(0.001, 0.20, size=3):
        a = np.exp(-alpha * np.minimum(i, n - 1 - i))
        a /= a.sum()
        yield a

    # random Dirichlet‑like vectors
    for _ in range(2):
        a = rng.random(n) + 1e-12  # avoid exact zeros
        a /= a.sum()
        yield a


# ----------------------------------------------------------------------
# Projected additive + mirror‑descent optimiser
# ----------------------------------------------------------------------
def _optimise(seq: np.ndarray, deadline: float, rng: np.random.Generator):
    """Run a few hundred cheap gradient‑like steps; return best vector seen."""
    n = seq.shape[0]
    a = seq.copy()
    best = a.copy()
    best_sc = _fast_score(a)

    # momentum buffer for the additive update
    v = np.zeros_like(a)
    step = 0.12          # initial step size
    momentum = 0.85
    eps = 1e-14

    # main loop – stop when time runs out or when we already beat the target
    while time.monotonic() < deadline - 0.02:
        conv = np.convolve(a, a)
        k = int(conv.argmax())

        # gradient of the (unique) active convolution entry:
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = a[idx[mask]]

        # ---- additive momentum step ----
        direction = -g + g.mean()          # zero‑sum direction
        v = momentum * v + (1.0 - momentum) * direction
        a = a + step * v
        a = np.clip(a, 0.0, None)

        # ---- occasional mirror (multiplicative) update ----
        if rng.random() < 0.12:
            eta = step
            mult = np.exp(-eta * g)
            a *= mult

        # re‑normalise to stay on the simplex
        s = a.sum()
        if s < eps:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # evaluate
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

        # decrease the step a little (geometric cooling)
        step *= 0.9996
        if step < 1e-6:
            step = 0.12

        # early exit if we already beat the announced target
        if best_sc < 1.503:
            break

    return best, best_sc


# ----------------------------------------------------------------------
# Tiny Gaussian hill‑climb (purely local polish)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray, finish: float, rng: np.random.Generator,
                  step_initial: float = 0.20) -> tuple[np.ndarray, float]:
    """Gaussian perturbation + projection; runs until deadline."""
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)

    while time.monotonic() < finish:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            cand = np.full(n, 1.0 / n)
        else:
            cand /= s
        sc = _fast_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
            # reset step to keep exploring
            step = step_initial
        else:
            step *= 0.997
            if step < 1e-7:
                step = step_initial
    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence with a low evaluation score.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Seed with the previously discovered construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            vec = np.asarray(height_sequence_1, dtype=np.float64)
            if vec.size > 0:
                vec = np.abs(vec)                     # ensure non‑negative
                vec /= vec.sum()                      # normalise to sum = 1
                sc = _fast_score(vec)
                best_seq, best_score = vec.copy(), sc
                print(f"[init] height_sequence_1 (n={vec.size}) score={sc:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Sweep across a set of lengths, generate families, optimise
    # ------------------------------------------------------------------
    lengths = [3000, 2500, 2000, 1500, 1200, 1000, 800, 600, 500, 400, 300, 200]
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # generate a modest pool of starters for this length
        for cand in _initial_vectors(n, rng):
            if time.monotonic() >= deadline:
                break

            # blend with the current global best if it has the same dimension
            if best_seq is not None and best_seq.shape[0] == n:
                cand = (cand + best_seq) / 2.0
                cand /= cand.sum()

            # quick initial evaluation
            sc0 = _fast_score(cand)
            if sc0 < best_score:
                best_seq, best_score = cand.copy(), sc0
                print(f"[cand] n={n} init score={sc0:.6f}")

            # run the cheap gradient / mirror descent optimiser
            opt_vec, opt_sc = _optimise(cand, deadline, rng)
            if opt_sc < best_score:
                best_seq, best_score = opt_vec.copy(), opt_sc
                print(f"[opt]  n={n} improved to {opt_sc:.6f}")

            # early termination if target reached
            if best_score < 1.503:
                print("[target] reached required bound – exiting.")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 2️⃣  Final polishing on the best vector found
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polished, polished_sc = _local_search(best_seq, deadline, rng, step_initial=0.20)
        if polished_sc < best_score:
            best_seq, best_score = polished, polished_sc
            print(f"[final] polished improvement to {polished_sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # Return a plain Python list as required by the outer harness
    return [float(x) for x in best_seq]
