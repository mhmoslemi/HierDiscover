def _calc_conv_and_max(a):
    conv = np.convolve(a, a)
    return conv, np.max(conv)

    The algorithm combines random simulated‑annealing moves with occasional
    deterministic “mass‑rebalancing” steps based on the contribution of each
    index to the currently maximal convolution entries.  Several lengths and
    restarts are tried within the time budget; the best sequence found is
    returned.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # lengths to explore – larger n gives more degrees of freedom.
    lengths = [400, 500, 600, 800, 1000, 1200]

        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            cur_score = cur_max   # because sum = target

            # Simulated‑annealing loop.

            # We keep a moderate upper bound to guarantee termination.
            max_iter = 30000
            # Adjust max_iter to remaining time / typical cost.
            # Rough estimate: 2 × 10⁻⁴ s per iteration for n≈1000.
            remaining = deadline - time.time()
            max_iter = min(max_iter, int(remaining * 4000))

            T0 = cur_max * 0.15 + 0.001   # a tiny offset to avoid zero.
            # Step‑size schedule factor.
            step_start = 0.4 * target      # allow moves up to 40 % of total mass.
            step_end   = 0.0005 * target   # end with very small moves.

            for it in range(max_iter):
                if time.time() >= deadline:
                    break

                # Linear cooling.
                t_frac = it / max_iter
                T = T0 * (1.0 - t_frac) + 1e-9    # avoid zero
                step_factor = step_start * (1.0 - t_frac) + step_end * t_frac

                # Choose two distinct indices.
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1

                max_move = min(a[i], step_factor)
                if max_move <= 0:

                # Apply move.
                a[i] -= delta
                a[j] += delta

                new_conv = np.convolve(a, a)
                new_max = new_conv.max()
                delta_score = new_max - cur_max

                # Acceptance test.
                if delta_score < 0 or rng.random() < math.exp(-delta_score / T):
                    # Move accepted.
                    conv = new_conv
                    cur_max = new_max
                    cur_score = cur_max
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(f"[accept] n={n} restart={restart} it={it} score={cur_score:.6f}")
                else:

                    a[i] += delta
                    a[j] -= delta

                # Every few hundred iterations try a directed “balancing” move.
                if (it + 1) % 500 == 0:

                    max_val = cur_max
                    # Consider all positions whose value is within 1e‑9 of the max.
                    active = np.where(conv >= max_val - 1e-12)[0]

                    # Compute contribution of each index.

                        i_end = min(k, n - 1)
                        idx = np.arange(i_start, i_end + 1)
                        opp = k - idx
                        contrib[idx] += a[opp]

                    i_low  = int(np.argmin(contrib))

                    if i_high != i_low and a[i_high] > 0:
                        # Determine a modest transfer amount.

                        a[i_high] -= delta_bal

                        if bal_max < cur_max - 1e-12:
                            conv = bal_conv

                            cur_score = cur_max

                                best_score = cur_score
                                best_seq = a.copy()
                                print(f"[balance] n={n} restart={restart} it={it} score={cur_score:.6f}")
                        else:
                            # revert
                            a[i_high] += delta_bal
                            a[i_low]  -= delta_bal

"""
Propose a low‑value step sequence.

The search combines several ideas that differ from the previous LP‑based
scheme:

*   The current best known sequence (global ``height_sequence_1``) is used
    as a seed when its length matches the current trial length.
*   For each length we run a short simulated‑annealing / random‑walk
    optimisation on the simplex (sum of coefficients fixed to √(2 n)).
    Random moves transfer a small amount of mass from one index to another.
*   Every few hundred iterations a *deterministic balancing* step is
    performed: the index that contributes most to the presently maximal
    convolution entry loses a little mass, while the index that contributes
    the least gains that mass.  This is a cheap sub‑gradient step that
    directly reduces the ∞‑norm of the auto‑convolution.
*   The temperature and the maximal move size are linearly cooled, allowing
    coarse exploration early on and fine‑grained polishing later.
*   Several lengths (up to 1200) and a few independent restarts per length
    are tried within the allotted time budget.  The best sequence seen is
    returned as a plain list of floats.
"""

import time
import math
import numpy as np


def _conv_and_max(a: np.ndarray):
    """Return (full convolution, maximum entry)."""
    conv = np.convolve(a, a)
    return conv, float(conv.max())


def _initial_sequence(n: int, target_sum: float, rng: np.random.Generator):
    """
    Produce a starting point on the simplex.
    Prefer the globally supplied sequence if it has the same length.
    """
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)) and len(global_seq) == n:
        seq = np.asarray(global_seq, dtype=np.float64)
        seq = np.maximum(seq, 0.0)
        seq = seq / seq.sum() * target_sum
        return seq
    # otherwise a Dirichlet draw
    return rng.dirichlet(np.ones(n)) * target_sum


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step sequence that minimises evaluate_sequence().
    The function obeys the competition rules (no closures, top‑level helpers
    only) and returns a list of non‑negative floats.
    """
    start_time = time.time()
    # keep a tiny safety margin for the host environment
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more degrees of freedom.
    lengths = [400, 500, 600, 800, 1000, 1200]
    restarts_per_n = 4

    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            # ---------------------------------------------------------
            # Initialise the current vector.
            a = _initial_sequence(n, target, rng)

            conv, cur_max = _conv_and_max(a)
            cur_score = cur_max                     # sum = √(2 n) ⇒ score = max_b
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[init]  n={n:4d} restart={restart} score={cur_score:.6f}")

            # ---------------------------------------------------------
            # Simulated‑annealing loop.
            # The number of iterations is set dynamically from the remaining
            # time budget (we assume ~2 ms per iteration for n≈1000).
            remaining = deadline - time.time()
            max_iter = int(min(30000, remaining * 4000))

            # Temperature and step‑size schedules.
            T0 = cur_max * 0.15 + 1e-3               # start temperature
            step_start = 0.4 * target                # up‑to 40 % of the total mass
            step_end   = 0.0005 * target             # finish with a tiny step

            for it in range(max_iter):
                if time.time() >= deadline:
                    break

                # linear cooling
                t_frac = it / max_iter
                T = T0 * (1.0 - t_frac) + 1e-9
                step_factor = step_start * (1.0 - t_frac) + step_end * t_frac

                # --- random move -------------------------------------------------
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1

                max_move = min(a[i], step_factor)
                if max_move <= 0:
                    continue
                delta = rng.random() * max_move     # move from i to j

                a[i] -= delta
                a[j] += delta

                new_conv = np.convolve(a, a)
                new_max = float(new_conv.max())
                delta_score = new_max - cur_max

                # Metropolis acceptance
                if delta_score < 0 or rng.random() < math.exp(-delta_score / T):
                    conv = new_conv
                    cur_max = new_max
                    cur_score = cur_max
                    if cur_score < best_score:

                        best_seq = a.copy()

                else:

    if best_seq is None:

    if final_score < best_score:

    print(f"Best score overall: {best_score:.6f} (length {len(best_seq)})")
    return [float(v) for v in best_seq]
