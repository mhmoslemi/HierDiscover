import time
import math
import numpy as np
from scipy.optimize import linprog

def _score_from_vector(a: np.ndarray) -> float:
    """Return the evaluation score for a vector a (exact formula)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    sum_val = np.sum(a)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return matrix A (shape 2n‑1 × n) such that A @ b = f * b (convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contains f shifted by i positions
    for i in range(n):
        A[i:i + n, i] = f
    return A

def _lp_improve(f: np.ndarray, M: float, target_sum: float):
    """
    Solve the linear program:
        maximize   sum(b)
        subject to A @ b <= M,  b >= 0,
    where A is the convolution matrix built from f.
    Return the scaled vector g (sum = target_sum) or None if LP fails.
    """
    A = _build_conv_matrix(f)
    rows = A.shape[0]
    # objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(f.shape[0], dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None) for _ in range(f.shape[0])]

    # solve with HiGHS (fast and robust)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        # any numerical issue – abort this improvement step
        return None

    if not res.success:
        return None

    b_opt = res.x
    sum_b = np.sum(b_opt)
    if sum_b <= 1e-12:
        # degenerate solution – fall back to uniform direction
        g = np.full_like(f, target_sum / f.shape[0])
        return g

    # scale the LP solution to the required ℓ¹‑norm
    scale = target_sum / sum_b
    g = b_opt * scale

def _line_search(f: np.ndarray, g: np.ndarray, M_cur: float):
    """
    Return a new vector f' = (1‑t)f + t g having a strictly smaller peak
    auto‑convolution than M_cur, together with its new peak.
    The search tries a decreasing geometric series of t.
    If no improvement is found, (None, M_cur) is returned.
    """
    t = 0.5
    while t > 1e-6:
        cand = (1.0 - t) * f + t * g
        cand = np.clip(cand, 0.0, 1000.0)
        # sum is preserved, no need to renormalise
        M_cand = np.max(np.convolve(cand, cand))
        if M_cand < M_cur - 1e-12:
            return cand, M_cand
        t *= 0.5
    return None, M_cur

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing tuned to preserve the ℓ¹‑norm.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score

    if not res.success:
        return None

    b_opt = res.x
    sum_b = np.sum(b_opt)
    if sum_b < 1e-12:                     # degenerate solution – fallback to uniform
        return np.full_like(f, target_sum / f.shape[0])

    # Scale to the required ℓ¹‑norm.
    g = b_opt * (target_sum / sum_b)
    g = np.clip(g, 0.0, 1000.0)           # safety clipping
    return g


def _line_search(f: np.ndarray, g: np.ndarray, M_cur: float) -> tuple[np.ndarray | None, float]:
    """
    Find a convex combination (1‑t) f + t g with a strictly smaller peak.
    The search tests t = ½, ¼, ⅛, … .  Returns (new_f, new_M) or (None, M_cur).
    """
    t = 0.5
    while t > 1e-6:
        cand = (1.0 - t) * f + t * g
        cand = np.clip(cand, 0.0, 1000.0)
        M_cand = np.max(np.convolve(cand, cand))
        if M_cand < M_cur - 1e-12:
            return cand, M_cand
        t *= 0.5
    return None, M_cur


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that preserves the ℓ¹‑norm.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.

    The routine runs several LP‑based improvement loops on different
    support sizes, starts from a warm‑start if available, and finishes
    with a short stochastic polishing stage.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # reserve a small safety margin for the final polishing stage
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm start from a previously discovered construction, if present.
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq0 = np.asarray(h1, dtype=np.float64)
        n0 = len(seq0)
        target0 = math.sqrt(2.0 * n0)
        seq0 = np.clip(seq0, 0.0, 1000.0)
        if np.sum(seq0) == 0:
            seq0 = np.full(n0, target0 / n0, dtype=np.float64)
        else:
            seq0 = seq0 * (target0 / np.sum(seq0))
        sc0 = _score_from_vector(seq0)
        best_seq = seq0.copy()
        best_score = sc0
        print(f"[init] warm‑start from height_sequence_1 (n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation: LP‑based improvement on several lengths.
    # ------------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for restart in range(4):
            if time.monotonic() >= deadline:
                break

            # Initialise a candidate vector.
            if best_seq is not None and len(best_seq) == n:
                f = best_seq.copy()
            else:
                f = rng.random(n) + 0.01                # avoid exact zeros
                f = f / np.sum(f) * target_sum

            # LP‑based improvement loop.
            for it in range(30):
                if time.monotonic() >= deadline:
                    break

                conv = np.convolve(f, f)
                M = np.max(conv)
                cur_score = 2.0 * n * M / (target_sum * target_sum)

                # Record the best vector seen so far.
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = f.copy()
                    print(f"[improve] n={n} restart={restart} it={it} score={cur_score:.6f}")

                # Solve the auxiliary LP.
                g = _lp_improve(f, M, target_sum)
                if g is None:
                    # LP failed – abandon this restart.
                    break

                # Try to move towards g while guaranteeing a strict reduction.
                new_f, new_M = _line_search(f, g, M)
                if new_f is None:
                    # No admissible step size left.
                    break

                f = new_f

                # Early exit if we have already beaten the target bound.
                if cur_score < 1.503:
                    print("[early] target achieved inside LP loop")
                    return [float(x) for x in f]

    # ------------------------------------------------------------------
    # Final stochastic polishing (if time permits).
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline, step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 400
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score_from_vector(best_seq)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
