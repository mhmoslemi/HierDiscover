def simplex_projection(v, s):
    if np.sum(v) == s and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑Descent optimiser for the auto‑convolution bound.

The evaluation score (with the ℓ¹‑norm fixed to √(2·n)) is exactly the
maximum entry of the auto‑convolution `conv = f * f`.  We therefore minimise
`max(conv)` under the constraints  

    f_i ≥ 0 ,   Σ f_i = √(2·n) .

The algorithm works as follows:

1. **Initialisation** – start from the global best sequence (`height_sequence_1`)
   when it exists, otherwise from a random simplex point.
2. **Projected sub‑gradient descent** – at each iteration the indices where the
   convolution is (almost) maximal are identified; a softmax / active‑set
   weight vector `w` is built and the gradient of the smoothed max
   (`∇F = 2·correlate(w, f)`) is computed.  A step in the opposite direction
   is taken and the result is projected back onto the probability simplex
   (non‑negative, fixed ℓ¹‑norm).  The step size decays as `1/√t`.
3. **Local stochastic polishing** – after the gradient phase a few
   random “mass‑transfer” moves are tried; any improvement is kept.
4. The whole routine respects the 1000‑second wall‑clock budget and returns
   the best sequence found (as a plain list of floats).

No file or network I/O is performed; only `numpy`, `math` and `time` are used.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector `v` onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_seq(seq: np.ndarray) -> float:
    """
    Compute the same quantity that ``evaluate_sequence`` returns,
    but without the extra Python overhead.
    The sequence is assumed to be already clipped to [0, 1000].
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _active_weights(conv: np.ndarray, eps_frac: float = 0.02) -> np.ndarray:
    """
    Build a weight vector that concentrates on the (almost) active
    constraints of the max‑convolution.
    `eps_frac` determines the relative tolerance: all entries with
    `conv >= max(conv)*(1-eps_frac)` receive equal weight.
    """
    max_val = conv.max()
    # tolerance – at least the maximal entry itself is selected
    thresh = max_val * (1.0 - eps_frac)
    mask = conv >= thresh
    if not np.any(mask):
        # fall back to the unique maximum
        mask = np.zeros_like(conv, dtype=bool)
        mask[np.argmax(conv)] = True
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0.0:
        # safety fallback (should never happen)
        w = np.ones_like(conv) / conv.size
    else:
        w /= w_sum
    return w

# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **_) -> list[float]:
    """
    Search for a non‑negative step sequence with a low auto‑convolution bound.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # a little safety margin so the caller can retrieve the result
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from the global variable if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:

            n0 = seq0.size
            target0 = math.sqrt(2.0 * n0)
            if seq0.sum() > 0:
                seq0 = np.clip(seq0, 0.0, 1000.0)
                seq0 = seq0 * (target0 / seq0.sum())
            sc0 = _score_from_seq(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] using global height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception as exc:                     # pragma: no cover
            print("[init] failed to use global height_sequence_1:", exc)

    # ------------------------------------------------------------------
    # Parameter sweep over several lengths
    # ------------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000]   # longer gives more flexibility
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)            # keep the ℓ¹‑norm fixed
        # How many independent restarts for this size?
        restarts = 5 if n <= 600 else 3

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ----- initialise a feasible point ---------------------------------
            if best_seq is not None and best_seq.size == n:
                f = best_seq.copy()
            else:
                f = rng.random(n) + 0.1                         # avoid too many zeros
                f = f / f.sum() * target_sum

            # ----- projected sub‑gradient descent --------------------------------
            max_iters = 1500 if n <= 600 else 1000
            # initial step magnitude – roughly a few percent of the average entry
            step0 = target_sum * 0.3 / n

            for it in range(max_iters):
                if time.monotonic() >= deadline:
                    break

                # current auto‑convolution and its active‑set weights
                conv = np.convolve(f, f)
                w = _active_weights(conv, eps_frac=0.02)

                # gradient of the smoothed max (∇F = 2·correlate(w, f))
                grad = 2.0 * np.correlate(w, f, mode="valid")

                # step size decays like 1/√t
                step = step0 / math.sqrt(it + 1.0)

                # projected update
                f_new = f - step * grad
                f_new = _simplex_projection(f_new, target_sum)

                # evaluate – we can use the cheap closed form
                sc = _score_from_seq(f_new)

                # keep the best overall
                if sc < best_score - 1e-9:
                    best_score = sc
                    best_seq = f_new.copy()
                    print(f"[grad] n={n} rs={rs} it={it} score={sc:.6f} max_conv={conv.max():.6f}")
                    if best_score < 1.503:
                        return best_seq.tolist()

                # accept the new iterate (even if not improving) – this
                # ensures the method keeps moving; we could also keep f if
                # the update is detrimental, but the projection usually
                # prevents large deteriorations.
                f = f_new

            # ----- local stochastic polishing ------------------------------------
            # small random mass‑transfer moves
            for _ in range(200):
                if time.monotonic() >= deadline:
                    break
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                # move at most 30 % of the source mass
                eps = rng.uniform(0.0, 0.3 * f[i])
                cand = f.copy()
                cand[i] -= eps
                cand[j] += eps
                # re‑project to keep the ℓ¹‑norm exact (numerical drift possible)
                cand = _simplex_projection(cand, target_sum)
                sc = _score_from_seq(cand)
                if sc < best_score - 1e-9:
                    best_score = sc
                    best_seq = cand.copy()
                    f = cand                     # continue polishing from the improvement
                    print(f"[polish] n={n} rs={rs} eps={eps:.4f} score={sc:.6f}")
                    if best_score < 1.503:
                        return best_seq.tolist()

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _score_from_seq(best_seq)
        print("[fallback] uniform sequence used, score=", best_score)

    # final sanity check (evaluate via the provided function)
    # (kept for safety; it is cheap compared to the whole search)

        from __main__ import evaluate_sequence  # type: ignore

        # function into the global namespace.

        print(f"[final] score reported by evaluate_sequence = {final_score:.6f}")
    except Exception:  # pragma: no cover
        pass
