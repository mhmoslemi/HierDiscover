"""
Hybrid optimizer for the step‑function challenge.

The score (with sum normalized to √(2 n)) is exactly the maximum entry
of the self‑convolution a∗a.  The algorithm proceeds as follows:

1.  Random (or supplied) starts for a few different lengths.
2.  Fast mass‑transfer hill‑climbing that updates the convolution
    incrementally.
3.  Linear‑programming refinement: for the current a compute the matrix
    A with (A·g)=a∗g.  Solve   max 1ᵀg   s.t.   A·g ≤ max(a∗a), g≥0.
    Scale g to the required ℓ¹‑norm and move a towards g (step‑size
    optimisation).  Repeating this drastically reduces the ℓ∞‑norm.
4.  Projected gradient descent on the ℓ∞‑objective (sub‑gradient taken
    at the current maximal convolution indices) with projection onto the
    simplex {x≥0, ∑x=√(2 n)}.
5.  A final gentle Gaussian perturbation (_local_search) for polishing.

All steps respect the non‑negativity and clipping rules of
`evaluate_sequence`.  The routine stops before the time budget expires
and returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Evaluate a candidate using the provided evaluate_sequence."""
    return float(evaluate_sequence(seq.tolist()))


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _transfer_mass(a: np.ndarray,
                   conv: np.ndarray,
                   i: int,
                   j: int,
                   delta: float) -> None:
    """
    Transfer `delta` from index i to j (i!=j, delta>0) and update the
    convolution `conv = a*a` in‑place (O(n) work).
    """
    n = a.shape[0]
    if i == j or delta <= 0.0:
        return
    if delta > a[i]:
        delta = a[i]

    # linear part: 2 * delta * a shifted
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part: convolution of the delta‑vector with itself
    length = 2 * n - 1
    ii = 2 * i
    if ii < length:
        conv[ii] += delta * delta
    ij = i + j
    if ij < length:
        conv[ij] -= 2.0 * delta * delta
    jj = 2 * j
    if jj < length:
        conv[jj] += delta * delta

    a[i] -= delta
    a[j] += delta


def _improve_by_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer hill‑climbing (the method used in the reference code).
    Returns the best sequence seen and its (approximate) score.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)
    sum_a = a.sum()                     # invariant through transfers
    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

    rng = np.random.default_rng()
    step = step_init
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = a[i] * step * rng.random()
        if delta <= 0.0:
            continue

        _transfer_mass(a, conv, i, j, delta)
        cur_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert – the move was not beneficial
            _transfer_mass(a, conv, j, i, delta)

        step *= 0.9999
        if step < 1e-5:
            step = step_init
        it += 1

    return best_a, best_score


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.4) -> tuple[np.ndarray, float]:
    """Gentle Gaussian perturbation (the final polishing step)."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score


def _lp_refine(a: np.ndarray,
               target_sum: float,
               deadline: float,
               max_iters: int = 30) -> np.ndarray:
    """
    LP‑based refinement described in the paper:
    maximise Σg subject to a∗g ≤ max(a∗a), g≥0.
    The solution g is scaled to the required ℓ¹‑norm and mixed with a.
    """
    n = a.shape[0]
    rows = 2 * n - 1

    for _ in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        M = conv.max()

        # Build matrix A where column j is a shifted by j positions.
        A = np.zeros((rows, n))
        for j in range(n):
            A[j:j + n, j] = a

        # Linear program: maximise Σg  <=> minimise -Σg
        c = -np.ones(n)
        b_ub = np.full(rows, M)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=(0, None), method='highs')
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            break

        # Scale g so that Σg = target_sum
        scale = target_sum / sum_g
        g_scaled = g * scale

        # Try a few mixing coefficients and keep the best.
        best_candidate = a
        best_M = M
        improved = False
        for t in (0.1, 0.25, 0.5, 0.75, 1.0):
            a_new = (1 - t) * a + t * g_scaled
            # Numerical drift may slightly change the sum – re‑project.
            a_new = _project_onto_simplex(a_new, target_sum)
            conv_new = np.convolve(a_new, a_new)
            M_new = conv_new.max()
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_candidate = a_new
                improved = True

        if not improved:
            break
        a = best_candidate
    return a


def _proj_grad_descent(a: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       max_iters: int = 10_000,
                       step_init: float = 1.0) -> np.ndarray:
    """
    Projected (sub‑)gradient descent on max(a∗a) with projection onto the
    simplex {x≥0, Σx=target_sum}.  The sub‑gradient is taken at all
    indices attaining the current maximum.
    """
    n = a.shape[0]
    step = step_init
    best_a = a.copy()
    best_max = np.max(np.convolve(a, a))

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        cur_max = conv.max()
        # indices where the maximum is attained (allow a tiny tolerance)
        idx_max = np.where(conv >= cur_max - 1e-12)[0]

        # sub‑gradient of the max‑entry w.r.t. a
        grad = np.zeros_like(a)
        for k in idx_max:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            # contribution from a[k-j] for j in [lo, hi]
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * a[k - j]

        # Gradient step and projection
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        new_max = np.max(np.convolve(a_candidate, a_candidate))
        if new_max + 1e-12 < cur_max:
            a = a_candidate
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_candidate.copy()
        else:
            # no improvement – shrink step size
            step *= 0.5
            if step < 1e-8:
                break

        # optional gentle step‑size increase to avoid stagnation
        step = min(step * 1.001, step_init)

    return best_a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    # leave a tiny safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Initialise candidates (use existing global if present)
    # ------------------------------------------------------------------
    try:
        init_global = np.array(height_sequence_1, dtype=float)
    except Exception:
        init_global = None

    # Explore several lengths – larger n generally gives more flexibility.
    lengths = [300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Build a small portfolio of starts for this length.
        starts: list[np.ndarray] = []

        if init_global is not None and init_global.shape[0] == n:
            starts.append(init_global.copy())

        # uniform start (already normalized)
        starts.append(np.full(n, target_sum / n))

        # several random starts
        for _ in range(4):

            vec = vec * (target_sum / vec.sum())

        # ------------------------------------------------------------------
        # 2️⃣  Process each start with the three optimisation phases
        # ------------------------------------------------------------------

            if time.monotonic() >= deadline:

            # ---- Phase B: linear‑programming refinement --------------------
            a = _lp_refine(a, target_sum, deadline, max_iters=25)
            score_lp = _score(a)

            best_score = refined_score

    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
