def project_onto_simplex(v, s):
    """Projection onto the simplex {x >= 0, sum(x)=s}."""
    if s <= 0:
        return np.zeros_like(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    # find rho = max(i) such that u_i > cssv_i / i
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1] if np.any(cond) else 0
    tau = cssv[rho] / (rho + 1)
    w = np.maximum(v - tau, 0)
    return w

def subgrad_optimize(n, S, deadline, rng, max_iters=4000, restarts=2):
    best_seq = None
    best_val = np.inf

    for restart in range(restarts):
        if time.monotonic() >= deadline:
            break

        # initialize
        a = rng.dirichlet(np.ones(n)) * S   # random simplex point
        # ensure non-zero
        if np.allclose(a,0):
            a = np.full(n, S/n)

        # gradient descent
        step0 = S * 0.5   # initial step size
        for it in range(max_iters):
            t = time.monotonic()
            if t >= deadline:
                break

            # compute conv via FFT
            L = 1 << ((2*n - 1).bit_length())
            fa = np.fft.rfft(a, L)
            conv = np.fft.irfft(fa * fa, L)[:2*n-1]
            max_val = conv.max()
            k_max = int(conv.argmax())
            # gradient of max conv with respect to a
            idx = np.arange(n)
            j = k_max - idx

"""
Simple sub‑gradient optimisation on the ℓ¹‑simplex.

The algorithm keeps the vector on the simplex (sum = √(2 n), a_i ≥ 0)
by projecting after each gradient step.  The true evaluation
objective is

    F(a) = 2·n·max_k (a∗a)[k] / (∑ a_i)² .

For a given shift k_⋆ that maximises the convolution we have
∂max/∂a_i = 2·a_{k_⋆‑i} (if the index lies inside the vector,
otherwise 0).  Using the chain rule we obtain an explicit sub‑gradient
for F, which we descend with a decaying step‑size and periodic exact
evaluation (via the provided ``evaluate_sequence``).  Multiple random
restarts, several lengths and a final stochastic polishing are used
within the allotted time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # Works even if v already satisfies the constraints
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    tau = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - tau, 0.0)
    return w


def _conv_max_and_index(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute a∗a via FFT, return the full convolution (real),
    the maximal value and the index where it occurs.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_val = float(conv.max())
    max_idx = int(conv.argmax())
    return conv, max_val, max_idx


def _subgradient_step(a: np.ndarray, S: float) -> tuple[np.ndarray, float]:
    """
    Perform a single sub‑gradient step for the exact objective.
    Returns the updated vector and the current max‑convolution value.
    """
    n = a.shape[0]
    # convolution and its maximiser
    _, max_val, k = _conv_max_and_index(a)

    # gradient of max(conv) w.r.t. a
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]

    sum_a = a.sum()
    factor1 = 2.0 * n / (sum_a * sum_a)
    factor2 = 4.0 * n * max_val / (sum_a * sum_a * sum_a)
    grad = factor1 * g - factor2

    return grad, max_val


def _subgradient_optimize(
    n: int,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 3000,
    restarts: int = 3,
) -> tuple[np.ndarray, float]:
    """
    Run a few independent sub‑gradient descents for a fixed length n.
    Returns the best sequence found and its exact evaluation score.
    """
    best_seq = None
    best_val = float("inf")
    L = _next_pow2(2 * n - 1)  # constant for the whole run

    for restart in range(restarts):
        if time.monotonic() >= deadline:
            break

        # initialise a random point on the simplex
        a = rng.dirichlet(np.ones(n), size=1)[0] * target_sum
        step0 = target_sum * 0.5  # initial step‑size

        for it in range(max_iters):
            if time.monotonic() >= deadline:
                break

            grad, max_conv = _subgradient_step(a, target_sum)

            # decaying step‑size (square‑root schedule)
            alpha = step0 / math.sqrt(it + 1.0)
            a_new = a - alpha * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new, target_sum)

            a = a_new

            # periodic exact evaluation
            if (it % 30) == 0:
                val = evaluate_sequence(a.tolist())
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(
                        f"[sgd] n={n} restart={restart} iter={it:04d} val={val:.6f}"
                    )
                if best_val < 1.503:
                    return best_seq, best_val

        # final eval for this restart
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[sgd] n={n} restart={restart} final val={val:.6f}")

    if best_seq is None:
        # fallback: uniform distribution
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())

    return best_seq, best_val


def _stochastic_refinement(
    seq: np.ndarray, deadline: float, step_initial: float = 0.3
) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that respects the ℓ¹‑simplex.
    Keeps the total sum fixed by projecting after each perturbation.
    """
    rng = np.random.default_rng()
    best = seq.copy()
    best_score = evaluate_sequence(best.tolist())
    step = step_initial
    S = best.sum()

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, S)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-9:
            step = step_initial

    return best, best_score
