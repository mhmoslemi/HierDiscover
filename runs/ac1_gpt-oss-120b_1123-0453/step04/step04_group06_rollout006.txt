"""
Propose a low‑peak step sequence.

The optimizer works on the L∞‑norm of the auto‑convolution
under the simplex constraint Σ a_i = √(2 n), a_i ≥ 0.
It repeatedly:

* draws a random start on the simplex,
* performs a simple projected gradient descent that minimises the
  *exact* maximum of the convolution (the gradient of the active
  offset is 2 a shifted),
* adapts the step size automatically,
* restarts from the current best with a small Gaussian perturbation
  when progress stalls,
* finally refines the best candidate with a short stochastic hill‑climb.

FFT‑based convolution makes each iteration O(n log n) and lets the
search explore lengths up to a few thousand within the 1000 s budget.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_score(v: np.ndarray) -> float:
    """Exact evaluation used by the challenge (max of a*a)."""
    conv = np.convolve(v, v)
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:                     # protection in the original code
        return float("inf")
    # With the simplex normalisation the score equals max_val.
    # The factor below reproduces the exact formula of evaluate_sequence.
    n = len(v)
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Compute a*a using a power‑of‑two FFT (real‐valued, fast)."""
    n = len(a)
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()          # next power of two ≥ m
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # Small negative round‑off errors are harmless for the max operation
    # but we clamp them to zero to stay consistent with the original code.
    conv[conv < 0] = 0.0
    return conv


# ----------------------------------------------------------------------
# Core optimiser – projected gradient on the *exact* max
# ----------------------------------------------------------------------


def _gd_max_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2500,
    step_initial: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent that minimises the *maximum* entry of a*a.
    The gradient is taken only with respect to a single active convolution
    offset (the current maximum).  After each step the vector is projected back
    onto the simplex.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    best = a.copy()
    best_score = _compute_score(best)

    step = step_initial * target_sum          # scale step with the typical size
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = _conv_fft(a)
        max_val = conv.max()
        # pick one of the maximal indices – random choice helps escape ties
        max_idxs = np.flatnonzero(conv == max_val)
        k = max_idxs[0] if len(max_idxs) == 1 else rng.choice(max_idxs)

        n = len(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range

        grad = np.zeros_like(a)
        grad[i_range] = 2.0 * a[j_range]           # ∇ (a*a)_k

        cand = _project_onto_simplex(a - step * grad, target_sum)
        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best, best_score = cand.copy(), cand_score
            a = cand
            step *= 1.02
            no_improve = 0
        else:
            step *= 0.65
            no_improve += 1
            # occasional random perturbation to get out of a flat region
            if no_improve >= 200:
                noise = step_initial * target_sum * rng.normal(size=n)
                pert = best + noise
                pert = np.clip(pert, 0.0, 1000.0)
                a = _project_onto_simplex(pert, target_sum)
                step = step_initial * target_sum
                no_improve = 0

        if step < 1e-12:
            break

    return best, best_score


# ----------------------------------------------------------------------
# Small stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.2,
    max_iters: int = 8000,
) -> tuple[np.ndarray, float]:
    """Gaussian perturbation hill‑climb keeping the vector on the simplex."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _compute_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score

        sigma *= 0.9997
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **_: None) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        n = len(seq)
        target = math.sqrt(2.0 * n)
        s = seq.sum()
        if s > 0:
            seq = seq * (target / s)
        else:
            seq = np.full(n, target / n, dtype=np.float64)
        sc = _compute_score(seq)
        best_seq, best_score = seq.copy(), sc
        print(f"[init] warm‑start n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Main stochastic optimisation over several lengths
    # ------------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000, 1500, 2000]
    # Reserve a little time for the final polishing phase.
    final_phase_reserve = 30.0

    for n in lengths:
        if time.monotonic() >= deadline - final_phase_reserve:
            break

        target_sum = math.sqrt(2.0 * n)
        # Rough budget per length – proportionally larger for bigger n
        remaining = deadline - time.monotonic() - final_phase_reserve
        per_len_budget = max(5.0, remaining / (len(lengths) - lengths.index(n)))
        sub_deadline = time.monotonic() + per_len_budget

        restarts = 15 if n <= 800 else 8
        for rs in range(restarts):
            if time.monotonic() >= sub_deadline:
                break

            # ------------------------------------------------------------------
            # Initialise a random point on the simplex (or reuse a matching best)
            # ------------------------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
            else:
                a0 = rng.exponential(scale=1.0, size=n)
                a0 = a0 / a0.sum() * target_sum

            # ------------------------------------------------------------------
            # Gradient descent on the exact max of the auto‑convolution
            # ------------------------------------------------------------------
            a_opt, sc_opt = _gd_max_optimize(
                a0,
                target_sum,
                sub_deadline,
                max_iters=2500,
                step_initial=0.03,
            )

            if sc_opt < best_score:
                best_seq, best_score = a_opt.copy(), sc_opt
                print(f"[update] n={n} restart={rs} score={sc_opt:.6f}")

            # early termination if we have already crossed the target
            if best_score < 1.503:
                print("[early] target achieved – stopping search")
                deadline = time.monotonic() + 0.5
                break

    # ------------------------------------------------------------------
    # 2️⃣ Final polishing – a short hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline - 1.0:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        refined, refined_sc = _hill_climb(
            best_seq,
            target_sum,
            deadline,
            sigma_initial=0.15,
            max_iters=8000,
        )
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[refine] final score={refined_sc:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list (the harness will call evaluate_sequence)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
