def _lp_refine(f, target, deadline, rng, max_steps=3, t=0.2):
    n = f.shape[0]
    a = f.copy()
    while time.monotonic() < deadline and max_steps > 0:

            max_steps -= 1
        else:

------
0️⃣ Warm‑start from a previously found construction if available.
1️⃣ Sweep simple parametric families (power‑law, exponential, beta‑like) for many
   lengths and keep the best.
2️⃣ For each length run several random restarts of a projected Adam optimiser
   that minimises a smooth‑max surrogate of the maximum auto‑convolution.
   After the Adam run a tiny linear‑programming refinement (if cvxpy is installed)
   is performed: maximise the ℓ¹‑mass of a new vector under the current
   convolution bound, then mix it with the current iterate.
3️⃣ Final cheap stochastic mass‑swap hill‑climb.

All vectors are projected onto the feasible set (non‑negative,
ℓ¹‑norm = √(2·n)).  The implementation is fully self‑contained,
uses only NumPy (and optionally CVXPY) and respects the 2‑CPU limit.
"""

import time
import math
import numpy as np

# optional LP refinement
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return an FFT‑friendly length (NumPy >=1.20 provides this)."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for many calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fast_score(seq: np.ndarray) -> float:
    """Exact objective (max of the full convolution)."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and enforce ℓ¹‑norm = √(2·n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / total)
    return seq


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max weights over the convolution entries."""
    cmax = conv.max()
    # subtract max to avoid overflow
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    For each i: ∂/∂a_i  Σ_k w_k (a * a)_k  = 2·Σ_j w_{i+j}·a_j .
    Obtained with a single FFT‑based cross‑correlation.
    """
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)          # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]          # middle n entries
    grad = 2.0 * segment[::-1]
    return grad


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n convolution matrix C such that C @ g = f * g.
    Column i contains f shifted down by i positions.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        C[i : i + n, i] = f
    return C


def _adam_optimize_sequence(init_seq: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth‑max surrogate.
    Returns the best iterate (w.r.t. the true max‑convolution) and its score.
    """
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_local = a.copy()
    best_score = _fast_score(a)

    it = 0
    beta = 1.0                     # smoothness parameter
    base_lr = 0.05 * target / n    # base learning‑rate magnitude

    while time.monotonic() < deadline and it < max_iters:
        # current convolution and (exact) max
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # smooth‑max and gradient
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # Adam update
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        step = base_lr * (m_hat / (np.sqrt(v_hat) + eps))
        a = a - step

        # projection onto simplex
        a = np.clip(a, 0.0, 1000.0)
        tot = a.sum()
        if tot > 0.0:
            a = a * (target / tot)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.03
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # anneal β – make the surrogate sharper
        if it % 100 == 0 and beta < 500.0:
            beta *= 1.5

        it += 1

    return best_local, best_score


def _lp_refine(f: np.ndarray,
               target: float,
               deadline: float,
               rng: np.random.Generator,
               max_steps: int = 3,
               step_size: float = 0.25) -> np.ndarray:
    """
    Linear‑programming refinement (if cvxpy is available).

    Given a current vector `f`, we solve:
        maximise   sum(g_i)
        subject to  (f * g)_k ≤ M   for all k
                    g_i ≥ 0
    where M = max_k (f*f)_k.
    The solution is then mixed back with `f` (convex combination) and
    re‑normalised.  This tends to push mass away from the most
    constraining lag, often lowering the true max‑convolution.
    """
    if not _HAS_CVX:
        return f  # LP not possible – return unchanged

    a = f.copy()
    n = a.shape[0]

    while time.monotonic() < deadline and max_steps > 0:
        M = _linear_convolve(a, a).max()
        # Build the convolution matrix for the current a
        A = _conv_matrix(a)            # shape (2n‑1, n)

        # LP: maximise ℓ¹‑mass under the bound M
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break
        g_vec = np.asarray(g.value, dtype=np.float64)

        # Normalise the LP solution to the required ℓ¹‑norm
        g_norm = _normalize(g_vec)

        # Convex combination with a small step size
        a_new = (1.0 - step_size) * a + step_size * g_norm
        a_new = _normalize(a_new)

        new_M = _linear_convolve(a_new, a_new).max()
        if new_M < M - 1e-8:
            a = a_new
            max_steps -= 1
            # slowly reduce the step size to fine‑tune
            step_size = max(step_size * 0.7, 1e-3)
        else:
            # if no improvement, shrink step size aggressively
            step_size *= 0.5
            if step_size < 1e-4:
                break

    return a


class _Optimizer:
    """Container for the global best sequence and its score."""
    def __init__(self, seed: int, deadline: float):
        self.rng = np.random.default_rng(seed)
        self.deadline = deadline
        self.best_seq: np.ndarray | None = None
        self.best_score: float = float("inf")

    def update_global(self, seq: np.ndarray):
        """Update the global best if `seq` improves the objective."""
        sc = _fast_score(seq)
        if sc < self.best_score:
            self.best_score = sc
            self.best_seq = seq.copy()
            print(f"[global best] n={len(seq)} score={sc:.6f}")

    def is_time_up(self) -> bool:
        return time.monotonic() >= self.deadline


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.

    The method combines a warm‑start, a parametric sweep,
    several Adam optimiser restarts with an optional LP refinement,
    and finishes with a cheap stochastic hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    opt = _Optimizer(seed, deadline)

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from an existing construction (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            cand = _normalize(cand)
            opt.update_global(cand)
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Parameteric families (power‑law, exponential, beta‑like)
    # ------------------------------------------------------------------
    lengths = [1500, 1200, 1000, 800, 600, 500, 400, 300, 200, 150, 100]
    for n in lengths:
        if opt.is_time_up():
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(1, n + 1, dtype=np.float64)

        # power‑law a_i ∝ i^{‑p}
        for p in np.linspace(0.2, 3.0, 20):
            a = idx ** (-p)
            a = a / a.sum() * target
            a = _normalize(a)
            opt.update_global(a)

        # exponential a_i ∝ exp(-α·i)
        for alpha in np.linspace(0.001, 0.15, 20):
            a = np.exp(-alpha * np.arange(n, dtype=np.float64))
            a = a / a.sum() * target
            a = _normalize(a)
            opt.update_global(a)

        # beta‑like a_i ∝ i^{p}·(n‑i+1)^{q}
        for p in np.linspace(0.0, 3.0, 8):
            for q in np.linspace(0.0, 3.0, 8):
                a = (idx ** p) * ((n - idx + 1) ** q)
                a = a / a.sum() * target
                a = _normalize(a)
                opt.update_global(a)

    # ------------------------------------------------------------------
    # 2️⃣ Random restarts + Adam optimisation + (optional) LP refinement
    # ------------------------------------------------------------------
    for n in lengths:
        if opt.is_time_up():
            break
        target = math.sqrt(2.0 * n)

        # Choose the number of restarts adaptively – more for smaller n
        restarts = 8 if n <= 600 else 4

        for _ in range(restarts):
            if opt.is_time_up():
                break

            # Initialise – Dirichlet‑like random vector
            start = rng.gamma(1.0, size=n)
            start = start / start.sum() * target

            # Also try a “low‑high” pairing to lower the central product
            sorted_vals = np.sort(start)
            a_pair = np.empty_like(start)
            half = n // 2
            a_pair[:half] = sorted_vals[:half]          # small values left side
            a_pair[-half:] = sorted_vals[-half:][::-1]  # large values right side
            if n % 2:
                a_pair[half] = sorted_vals[half]        # middle stays as is
            start = _normalize(a_pair)

            # Adam optimiser
            seq_opt, _ = _adam_optimize_sequence(start, deadline, rng)

            # LP refinement (if cvxpy available)
            if _HAS_CVX and not opt.is_time_up():
                seq_opt = _lp_refine(seq_opt, target, deadline, rng)

            opt.update_global(seq_opt)

            # Early exit if we already beat the target bound
            if opt.best_score < 1.503:
                return opt.best_seq.tolist()

    # ------------------------------------------------------------------
    # 3️⃣ Final cheap stochastic hill‑climb on the current global best
    # ------------------------------------------------------------------
    if opt.best_seq is not None:
        while not opt.is_time_up():
            n = opt.best_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            delta = opt.best_seq[i] * 0.04
            cand = opt.best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _normalize(cand)
            opt.update_global(cand)
    else:
        # fallback – a uniform vector (unlikely to be used)
        n = 200
        opt.best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        opt.best_score = _fast_score(opt.best_seq)

    # Return a plain Python list as required by the evaluation harness
    return [float(x) for x in opt.best_seq]
