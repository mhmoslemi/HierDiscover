"""
Improved optimisation of a non‑negative step sequence.

The algorithm consists of three phases:

1️⃣  **Gradient descent on a temperature‑annealed soft‑max of the
    auto‑convolution.**  The objective `2·n·max(conv)/(sum(a)²)` is
    invariant under scaling, so we keep the ℓ¹‑norm fixed (`∑a = 1`) and
    minimise the peak of the convolution.  A smooth approximation of the
    max (soft‑max with a temperature that decays from 30 % to 1 % of the
    current peak) yields a differentiable weighted sum of the convolution
    entries.  The exact gradient of this surrogate is a cross‑correlation
    (`grad = 2·corr(a, w)`).  Adam updates with Euclidean projection onto the
    simplex keep the iterates feasible.  The diagonal term of the gradient
    is handled correctly (no artificial subtraction).

2️⃣  **Multiple random restarts & length sweep.**  Starting from the
    best sequence seen so far (or a linear interpolation of it) we run the
    Adam optimiser for several short time windows on many lengths
    (500‑2000).  The temperature schedule, a larger patience and a longer
    iteration budget give the optimiser more chance to escape plateaux.

3️⃣  **Stochastic polishing.**  The final candidate is refined by a
    simple hill‑climber that repeatedly perturbs the vector with Gaussian
    noise, optionally swaps two coordinates and always re‑projects onto the
    simplex.  After each move the sequence is symmetrised
    (`a ← (a+a[::-1])/2`) because the optimal solution is expected to be
    symmetric; this reduces the search space and improves the bound.

The function `propose_candidate` respects the global deadline (default
1000 s), prints intermediate scores and finally returns a plain list of
floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal – return the uniform point
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact score used by `evaluate_sequence` (no internal clipping)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / beta
    exp_shifted = np.exp(shifted - np.max(shifted))  # extra shift for safety
    w = exp_shifted / np.sum(exp_shifted)
    return w


def _weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. `a`.
    `weights` must have length 2·len(a)-1 and sum to 1.
    The formula is 2·corr(a, weights).
    """
    n = len(a)
    grad = np.zeros_like(a)

    # Loop over all convolution offsets – fully vectorised inside the loop.
    # For each k we add 2·w_k·a_{k-i} to grad_i for all admissible i.
    for k, wk in enumerate(weights):
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += wk * 2.0 * a[j]        # diagonal terms are correctly counted
    return grad


def _interpolate_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` while preserving sum."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    # positions in [0,1]
    old_pos = np.linspace(0.0, 1.0, old_n)
    new_pos = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_pos, old_pos, seq)
    new_seq = np.maximum(new_seq, 0.0)
    # renormalise to sum = 1
    total = np.sum(new_seq)
    if total == 0:
        new_seq[:] = 1.0 / new_n
    else:
        new_seq /= total
    return new_seq


def _adam_descent_improved(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 20000,
    patience: int = 800,
    start_temp: float = 0.30,
    end_temp: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate of the convolution peak.

    Parameters
    ----------
    a0: initial vector (already satisfies sum = target_sum)
    target_sum: the ℓ¹‑norm we keep fixed (normally 1.0)
    deadline: absolute time at which the run must stop
    max_iters: hard limit on iterations
    patience: stop early if no improvement for this many iterations
    start_temp / end_temp: geometric temperature factors (multiplied by the
        current maximal convolution value).  The temperature decays from
        ``start_temp`` to ``end_temp`` over the course of the optimisation.
    """
    a = a0.copy()
    n = len(a)
    # Adam hyper‑parameters
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # ----- compute the surrogate gradient -----
        conv = np.convolve(a, a)
        max_conv = np.max(conv)

        # geometric temperature schedule
        factor = start_temp * ((end_temp / start_temp) ** (it / max_iters))
        beta = max_conv * factor + 1e-12

        w = _softmax_weights(conv, beta)
        grad = _weighted_gradient(a, w)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        # projection onto the simplex (keeps sum exactly = target_sum)
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        # enforce symmetry – this does not change the sum
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cur_score = _compute_score_from_vector(a)
        cand_score = _compute_score_from_vector(a_candidate)

        # accept only if we improve the true objective
        if cand_score < cur_score:
            a = a_candidate
        else:
            # still keep the candidate for bookkeeping
            a = a_candidate if cand_score < best_score else a

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            print(f"[Adam] n={n:4d} it={it:5d} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score


def _stochastic_refinement(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.3,
    swap_prob: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing phase: Gaussian perturbations + occasional swaps,
    always re‑projected onto the simplex and symmetrised.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, len(best), size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = best + rng.normal(scale=step, size=best.shape)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand = (cand + cand[::-1]) * 0.5  # enforce symmetry

        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = min(step_initial, step * 1.02)  # slightly increase step on improvement
        else:
            step *= 0.995                     # decay otherwise

        if step < 1e-7:
            step = step_initial

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the given
    time budget.  Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any.
    # --------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if isinstance(warm, (list, tuple, np.ndarray)):
        seq = np.asarray(warm, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        if np.sum(seq) == 0:
            seq = np.full_like(seq, 1.0 / len(seq))
        else:
            seq = seq / np.sum(seq)          # normalise to sum = 1
        # symmetrise – the optimal shape is expected to be symmetric
        seq = (seq + seq[::-1]) * 0.5
        score = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[warm] loaded height_sequence_1, n={len(seq)} score={score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation – sweep over several lengths with multiple restarts.
    # --------------------------------------------------------------
    lengths = [500, 800, 1000, 1200, 1500, 1800, 2000]
    # Rough allocation of time: about 10 s per length, more for larger ones.
    time_per_len = 12.0

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # decide how many restarts (more for smaller n)
        restarts = 6 if n <= 1200 else 3

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # Initial vector for this run
            if best_seq is not None:
                a0 = _interpolate_sequence(best_seq, n)
            else:
                a0 = rng.random(n) + 1e-6
                a0 = a0 / np.sum(a0)

            # Ensure symmetry from the outset
            a0 = (a0 + a0[::-1]) * 0.5

            # Give each run a slice of the remaining budget
            remaining = deadline - time.monotonic()
            slice_deadline = time.monotonic() + min(time_per_len, remaining * 0.6)

            a_opt, sc_opt = _adam_descent_improved(
                a0,
                target_sum=1.0,
                deadline=slice_deadline,
                max_iters=20000,
                patience=800,
                start_temp=0.30,
                end_temp=0.01,
            )

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[update] n={n:4d} restart={rs:2d} score={sc_opt:.6f}")

            # Early exit if we already beat the target comfortably
            if best_score < 1.45:
                print("[early] satisfactory score reached")
                break

        if best_score < 1.45:
            break

    # --------------------------------------------------------------
    # Final polishing phase (if time permits).
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – a modest uniform sequence
        n = 800
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(
            best_seq,
            deadline,
            step_initial=0.25,
            swap_prob=0.25,
        )
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # final safety clipping (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best length={len(best_seq)} best score={best_score:.6f}")
    return [float(x) for x in best_seq]
