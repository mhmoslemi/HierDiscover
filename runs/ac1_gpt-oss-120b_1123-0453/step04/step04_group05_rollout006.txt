def _subgradient_descent(n,S,deadline,z0, max_iters=2000, step0=1.0):
    best_a = None; best_val = inf
    z = z0.copy()
    step = step0
    for t in range(max_iters):
        if time.monotonic() >= deadline: break
        a = _softmax_to_a(z, S)
        # compute conv and max
        L = _next_pow2(2*n -1)
        fa = np.fft.rfft(a, L)
        conv = np.fft.irfft(fa * fa, L)[:2*n-1]
        max_val = conv.max()
        # compute active mask
        mask = (conv == max_val)
        # Build w vector
        w = np.zeros_like(conv)
        count = mask.sum()
        if count == 0:
            # shouldn't happen
            w[:] = 1.0/conv.size
        else:
            w[mask] = 1.0 / count
        # compute g_a via convolution a_rev * w
        a_rev = a[::-1]
        L2 = _next_pow2(3*n - 2)
        fr = np.fft.rfft(a_rev, L2)
        fw = np.fft.rfft(w, L2)
        conv2 = np.fft.irfft(fr * fw, L2)
        start = n-1
        g_raw = conv2[start:start+n]
        g_a = 2.0 * g_raw
        total = np.dot(g_a, a)
        grad_z = a * g_a - a * (total / S)
        # update z
        # simple SGD step: z = z - step * grad_z
        z = z - step * grad_z
        # optional step decay
        step *= 0.9995
        # evaluate exact score at some interval, say every 20 iters
        if t % 20 == 0:
            seq = a
            val = evaluate_sequence(seq.tolist())
            if val < best_val:
                best_val = val
                best_a = seq.copy()
                # log
                print(f"[sgd] n={n} iter={t} val={val:.6f}")

        if step < 1e-8:
            step = step0

    return best_a if best_a is not None else a, best_val

"""
Proposes a low‑value non‑negative step sequence.

The search proceeds in three stages:

1️⃣  A series of lengths (up to a few thousand) are inspected.
    For each length a simple projected sub‑gradient descent is run on the
    simplex (via a soft‑max re‑parameterisation).  The sub‑gradient is taken
    with respect to the exact maximum of the autocorrelation (the true
    evaluation objective).  The method is very cheap – each iteration needs
    only a few FFTs – so we can afford many restarts and many iterations.

2️⃣  The best candidate found so far is refined with a short stochastic
    Gaussian hill‑climbing, again keeping the sum fixed by the same soft‑max
    projection.

3️⃣  The final sequence is clipped to the limits required by
    `evaluate_sequence` and returned.

The algorithm is deliberately different from the previous L‑BFGS‑B
implementation (it uses a true sub‑gradient rather than a smooth
log‑sum‑exp surrogate) and searches larger lengths, which is often enough
to push the bound below the target 1.503.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Convert unrestricted vector `z` to a simplex point with sum = target_sum.
    This is the standard soft‑max re‑parameterisation.
    """
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)                     # stable
    a = exp_z / np.sum(exp_z) * target_sum
    return a


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (real input)."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv


def _subgradient_descent(
    n: int,
    S: float,
    z0: np.ndarray,
    deadline: float,
    max_iters: int = 3000,
    step0: float = 0.5,
    target: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the simplex for a single length `n`.

    Returns the best sequence (as a numpy array) discovered and its
    (exact) max‑convolution value (which equals the `evaluate_sequence`
    score when the sum is S = sqrt(2 n)).
    """
    best_a = None
    best_val = float("inf")
    z = z0.copy()
    step = step0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # a from soft‑max (always non‑negative and sums to S)
        a = _softmax_to_a(z, S)

        # --- convolution and max ---
        conv = _conv_full(a)          # length (2n‑1)
        max_val = conv.max()

        # update best if improved
        if max_val < best_val:
            best_val = max_val
            best_a = a.copy()
            print(f"[sgd] n={n:4d} iter={it:4d} val={max_val:.6f}")
            if best_val < target:
                # early exit for this length
                break

        # --- build sub‑gradient of max ---
        # active set (within a tiny tolerance)
        active = np.isclose(conv, max_val, atol=1e-12)
        cnt = np.count_nonzero(active)
        if cnt == 0:                       # should not happen
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = np.where(active, 1.0 / cnt, 0.0)

        # gradient w.r.t a : g_a = 2 * (a_rev * w)_{n‑1 … n‑1+n‑1}
        a_rev = a[::-1]
        L2 = _next_pow2(3 * n - 2)
        fr = np.fft.rfft(a_rev, L2)
        fw = np.fft.rfft(w, L2)
        conv2 = np.fft.irfft(fr * fw, L2)
        start = n - 1
        g_raw = conv2[start : start + n]          # length n
        g_a = 2.0 * g_raw

        # --- chain rule for soft‑max parameters ---
        total = np.dot(g_a, a)                     # scalar = a·g_a
        grad_z = a * g_a - a * (total / S)

        # --- simple gradient step (SGD) ---
        z = z - step * grad_z

        # decay step size slowly
        step *= 0.9995
        if step < 1e-9:
            step = step0

    # fallback if no improvement was ever recorded
    if best_a is None:
        best_a = _softmax_to_a(z0, S)
        best_val = np.max(_conv_full(best_a))

    return best_a, best_val


def _stochastic_refinement(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Very short Gaussian hill‑climbing that keeps the ℓ¹‑norm fixed by
    re‑projecting with the same soft‑max parametrisation.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    S = best.sum()
    # exact score (max‑convolution) – equal to evaluate_sequence when S = √(2n)
    best_score = np.max(_conv_full(best))

    step = step_initial
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # project back onto ℓ¹‑simplex (soft‑max)
        cand = _softmax_to_a(np.log(np.clip(cand / S, 1e-12, None)), S)
        cand_score = np.max(_conv_full(cand))
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score

# Public entry point
# ------------------------------------------------------------


def propose_candidate(
    seed: int = 0,
    budget_s: float = 1000,
    **kwargs,
) -> list[float]:
    """
