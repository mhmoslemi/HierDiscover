    n = seq.shape[0]

    step = 0.5
    best_seq = seq.copy()

        exp_term = np.exp(beta * (conv - maxb))

        idx = 2*n - 2 - np.arange(n)
        grad = grad_part[idx] + seq
        # center gradient
        grad_centered = grad - grad.mean()

            # clip to non-negative
            cand = np.clip(cand, 0.0, np.inf)   # no upper bound needed, will be normalized anyway

"""
Propose a low‑value step sequence.

Algorithm
---------
1️⃣  Generate a handful of “good looking’’ candidates using a few
    parametric families (uniform, exponential, power‑law, common window
    functions) for several lengths.  Keep the best one seen so far.

2️⃣  Refine the incumbent with a **smooth‑max gradient descent**.
    * The objective is  max_i (a*a)_i  (the evaluation function reduces to
      this value once the ℓ¹‑norm is fixed to √(2·n)).
    * Instead of a hard max we minimise a smooth approximation
          f_β(a)= (1/β)·log∑_i exp(β·(a*a)_i) .
      Its gradient is
          ∇f_β(a) =  a  +  rev_conv(w_rev , a) ,
      where w_i = softmax_i(β·(a*a)_i) and rev_conv denotes the linear
      convolution of the reversed weight vector with a.
    * After a step we project back onto the simplex (non‑negative entries
      with prescribed ℓ¹‑norm) by simple scaling.
    * The smoothing parameter β is increased slowly (β←β·1.02) so that the
      optimisation first moves globally and later focuses on the true
      maximum.

3️⃣  A brief stochastic “polish’’ phase (Gaussian perturbations
    followed by renormalisation) tries to squeeze out any remaining
    improvement.

The whole search stops after the given time budget (≈ 1000 s) and
returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolutions (FFT based)
# ----------------------------------------------------------------------
def _conv_fft_self(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of `seq` with itself (FFT, O(N log N))."""
    n = seq.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # smallest power‑of‑two ≥ 2·n‑1
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


def _conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary real vectors via FFT."""
    nx = x.shape[0]
    ny = y.shape[0]
    N = 1 << ((nx + ny - 1).bit_length())
    Fx = np.fft.rfft(x, n=N)
    Fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(Fx * Fy, n=N)
    return conv[: nx + ny - 1]


# ----------------------------------------------------------------------
# Objective without the security wrapper (exact)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact evaluation of the objective (max of the auto‑convolution)."""
    n = seq.shape[0]
    max_b = _conv_fft_self(seq).max()
    # The ℓ¹‑norm is fixed to √(2·n) elsewhere, so the objective is just max_b.
    return float(max_b)


# ----------------------------------------------------------------------
# Hand‑crafted families (used for an initial sweep)
# ----------------------------------------------------------------------
def _candidate_families(n: int, target_sum: float):
    """Return the best vector among several simple families for length n."""
    best_vec = None
    best_sc = float("inf")

    def _norm(v):
        v = np.asarray(v, dtype=np.float64)
        if v.sum() == 0:
            return np.full_like(v, target_sum / len(v))
        v = v / v.sum() * target_sum
        return v

    # 1) uniform baseline
    a = np.full(n, target_sum / n, dtype=np.float64)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    idx = np.arange(n, dtype=np.float64)

    # 2) exponential decay a_i ∝ exp(-α·i)
    alphas = np.concatenate([
        np.arange(1, 6) / (10.0 * n),
        np.arange(1, 21) / n,
        np.linspace(0.02, 0.20, 40),
        np.linspace(0.20, 1.00, 30)
    ])
    alphas = np.unique(alphas)
    for α in alphas:
        a = np.exp(-α * idx)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # 3) power‑law a_i ∝ (i+1)^(-p)
    p_vals = np.linspace(0.05, 5.0, 120)
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        a = base ** (-p)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # 4) classic window functions
    # Hann
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Hamming
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Blackman
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * idx / (n - 1)))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Sine window (full half‑period)
    a = np.sin(np.pi * (idx + 1) / (n + 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Triangular (peak in the middle)
    a = np.minimum(idx + 1, n - idx)
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (core optimiser)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(start_seq: np.ndarray,
                             target_sum: float,
                             deadline: float):
    """
    Minimise max_i (a*a)_i with a smooth approximation.
    Uses a back‑tracking line‑search, increases the smoothing parameter
    β over time and projects onto the ℓ¹‑simplex after each step.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_sc = _fast_score(a)

    beta = 2.0               # smoothing factor (starts low, grows)
    step = 0.4               # initial step length
    rng = np.random.default_rng()
    # counters for occasional random shake‑up
    no_improve = 0

    while time.monotonic() < deadline:
        # --- convolution and softmax weights ---------------------------------
        conv = _conv_fft_self(a)                 # length 2n‑1
        max_b = conv.max()

        # soft‑max over the entries (stabilised)
        exp_term = np.exp(beta * (conv - max_b))
        w = exp_term / exp_term.sum()            # shape (2n‑1,)

        # --- gradient of the smooth objective --------------------------------
        # g = a  +  rev_conv(w_rev , a)
        w_rev = w[::-1]
        rev = _conv_fft_two(w_rev, a)            # length 3n‑2
        idx = 2 * n - 2 - np.arange(n)           # map to needed slice
        grad = rev[idx] + a

        # centre the gradient – this enforces the ℓ¹‑norm after the update
        grad_centered = grad - grad.mean()
        direction = -grad_centered                # descent direction

        # --- simple back‑tracking line search --------------------------------
        improved = False
        while step > 1e-8:
            cand = a + step * direction
            # enforce non‑negativity
            cand = np.clip(cand, 0.0, np.inf)
            total = cand.sum()
            if total == 0.0:                     # safety fallback
                cand = np.full(n, target_sum / n, dtype=np.float64)
            else:
                cand = cand * (target_sum / total)

            cand_sc = _fast_score(cand)
            if cand_sc < best_sc - 1e-12:
                a = cand
                best_seq, best_sc = cand.copy(), cand_sc
                step = min(step * 1.2, 2.0)      # accelerate after success
                improved = True
                no_improve = 0
                break
            else:
                step *= 0.5                       # shrink step and retry

        if not improved:
            # no progress – increase smoothing (focuses more on the true max)
            beta = min(beta * 1.02, 300.0)
            no_improve += 1

            # occasional random shake‑up to escape plateaus
            if no_improve >= 30:
                i, j = rng.integers(n, size=2)
                delta = a[i] * 0.04
                if delta > 0:
                    cand = a.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = np.clip(cand, 0.0, np.inf)
                    cand = cand * (target_sum / cand.sum())
                    cand_sc = _fast_score(cand)
                    if cand_sc < best_sc:
                        a = cand
                        best_seq, best_sc = cand.copy(), cand_sc
                no_improve = 0
                step = 0.4

        # early exit if we already beat the target bound
        if best_sc < 1.503:
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Tiny random‑perturbation polish (final step)
# ----------------------------------------------------------------------
def _local_search_polish(start_seq: np.ndarray,
                         deadline: float,
                         step_initial: float = 0.15) -> tuple[np.ndarray, float]:
    """Very short stochastic hill‑climb starting from `start_seq`."""
    n = start_seq.shape[0]
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, np.inf)
        if cand.sum() == 0.0:
            cand = np.full(n, best.sum() / n, dtype=np.float64)
        else:
            cand = cand * (best.sum() / cand.sum())
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9995
        if step < 1e-7:
            step = step_initial
    return best, best_sc
