def smooth_gradient_descent(seq, target, deadline):
    n = len(seq)
    best_seq = seq.copy()
    best_score = _fast_score(seq)   # actual max conv

    gamma = 5.0
    step = 0.1
    rng = np.random.default_rng()
    while time.monotonic() < deadline and gamma <= 30.0:
        # compute smooth value and grad
        val, grad = smooth_max_and_grad(seq, gamma)
        # direction to reduce val, sum zero constraint
        mean_grad = grad.mean()
        direction = -grad + mean_grad

        cand = seq + step * direction
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total <= 0:
            cand = np.full(n, target/n)
        else:
            cand = cand * (target / total)

        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best_seq, best_score = cand.copy(), cand_score
            seq = cand
            step = min(step*1.2, 2.0)
            # optionally increase gamma slowly
            gamma = min(gamma * 1.05, 30.0)
        else:
            step *= 0.5
            # occasionally random perturbation
            if step < 1e-5:
                i, j = rng.integers(n, size=2)
                delta = seq[i] * 0.1
                cand = seq.copy()
                cand[i] = max(0, cand[i] - delta)
                cand[j] += delta
                cand = cand * (target / cand.sum())
                cand_score = _fast_score(cand)
                if cand_score < best_score:
                    seq = cand

"""
Improved step‑function search.

The algorithm proceeds in three phases:

1️⃣  *Broad family sweep* – a handful of classical analytic families
    (exponential, power‑law, Hann, Hamming, Blackman, sine, triangular)
    are generated for a set of lengths.  Every vector is normalised to
    ℓ¹‑norm √(2·n) and the exact objective (max‑convolution) is evaluated.
    The best among all families becomes the initial incumbent.

2️⃣  *Smoothed sub‑gradient descent* – the max‑convolution is approximated
    by a smooth “log‑sum‑exp’’ function with a continuation parameter γ.
    For a given γ the gradient is the convolution of the current vector
    with the soft‑max weights.  The update direction is projected onto the
    hyper‑plane ∑aᵢ=const, a step‑size is adapted on‑the‑fly and γ is
    slowly increased (continuation).  This yields a much richer descent
    direction than the pure active‑constraint sub‑gradient used before.

3️⃣  *Polish* – a few random Gaussian perturbations (while keeping the
    ℓ¹‑norm fixed) finish the optimisation.

All operations respect the security checks of ``evaluate_sequence``:
non‑negative entries, clipping to [0,1000] and a fixed ℓ¹‑norm.
The routine stops as soon as the 1.503 target is reached or the time
budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (full linear convolution of a with itself)
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of `seq` with itself (length 2·n‑1)."""
    n = seq.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact objective value 2·n·max(conv)/(sum(seq)²).
    For the normalised vectors used throughout sum(seq)=√(2·n),
    therefore the score equals max(conv).
    """
    n = seq.shape[0]
    max_b = _conv_fft(seq).max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (used for the exhaustive sweep)
# ----------------------------------------------------------------------
def _candidate_families(n: int, target_sum: float):
    """Return the best vector among several hand‑crafted families."""
    best_vec = None
    best_sc = float("inf")

    def _norm(v):
        v = np.asarray(v, dtype=np.float64)
        if v.sum() == 0:
            return np.full_like(v, target_sum / n)
        v = v / v.sum() * target_sum
        return v

    idx = np.arange(n, dtype=np.float64)

    # uniform (baseline)
    a = np.full(n, target_sum / n, dtype=np.float64)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # exponential families a_i ∝ exp(-α·i)
    alphas = np.concatenate([
        np.arange(1, 6) / (10.0 * n),          # very fine near 0
        np.arange(1, 21) / n,                  # 1/n … 20/n
        np.linspace(0.02, 0.20, 40),
        np.linspace(0.20, 1.00, 30)
    ])
    alphas = np.unique(alphas)
    for α in alphas:
        a = np.exp(-α * idx)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # power‑law families a_i ∝ (i+1)^{-p}
    p_vals = np.linspace(0.05, 5.0, 120)
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        a = base ** (-p)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # classic window functions (all symmetric)
    # Hann
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Hamming
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Blackman
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1)))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Sine window (single half‑period)
    a = np.sin(np.pi * (idx + 1) / (n + 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Triangular (peak in the middle)
    a = np.minimum(idx + 1, n - idx)
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Smoothed max‑convolution + gradient
# ----------------------------------------------------------------------
def _conv_max_and_gradient(seq: np.ndarray, gamma: float):
    """
    Compute max(conv) and the gradient of the smoothed max (log‑sum‑exp)
    with respect to `seq`.

    Returns:
        max_b : float    – the exact maximal entry of a*a.
        grad  : np.ndarray – gradient (same length as seq).
    """
    n = seq.shape[0]

    # ---------- full convolution ----------
    conv = _conv_fft(seq)            # length 2·n‑1
    max_b = conv.max()

    # ---------- smoothed weights ----------
    # w_k ∝ exp(γ·(conv_k – max_b))   – all ≤ 1, sum to 1
    shifted = gamma * (conv - max_b)   # max entry becomes 0
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum                    # soft‑max distribution

    # ---------- gradient = a * w (cross‑correlation) ----------
    # convolution of `seq` (len n) with `w` (len 2·n‑1) → len 3·n‑2
    N = 1 << ((n + (2 * n - 1) - 1).bit_length())
    Fa = np.fft.rfft(seq, n=N)
    Fw = np.fft.rfft(w, n=N)
    grad_full = np.fft.irfft(Fa * Fw, n=N)
    start = n - 1
    grad = grad_full[start:start + n]      # slice of length n

    return max_b, grad


# ----------------------------------------------------------------------
# Continuation‑style smoothed gradient descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(seq: np.ndarray,
                            target_sum: float,
                            deadline: float):
    """
    Starting from `seq` (already ℓ¹‑normalised to `target_sum`),
    apply a continuation method:
        * low γ → smooth landscape → large steps,
        * increase γ → focus on the true max.
    Returns the best sequence found before `deadline`.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    rng = np.random.default_rng()
    # continuation parameters
    gamma = 2.0          # start with a very smooth approximation
    gamma_max = 50.0
    step = 0.5
    min_step = 1e-6
    max_iter = 20000    # safety guard

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1

        maxb, grad = _conv_max_and_gradient(best_seq, gamma)

        # direction that keeps the ℓ¹‑norm unchanged
        mean_g = grad.mean()
        direction = -grad + mean_g

        cand = best_seq + step * direction
        cand = np.clip(cand, 0.0, 1000.0)

        total = cand.sum()
        if total == 0.0:
            cand = np.full(n, target_sum / n, dtype=np.float64)
        else:
            cand = cand * (target_sum / total)

        cand_sc = _fast_score(cand)

        if cand_sc < best_sc:                     # successful move
            best_seq, best_sc = cand, cand_sc
            step = min(step * 1.1, 2.0)            # accelerate a bit
            gamma = min(gamma * 1.02, gamma_max)   # increase smoothness
        else:
            step *= 0.5
            if step < min_step:
                # small random shake‑up to escape possible plateaus
                i, j = rng.integers(n, size=2)
                delta = best_seq[i] * 0.05
                if delta > 0:
                    cand = best_seq.copy()
                    cand[i] = max(0.0, cand[i] - delta)
                    cand[j] += delta
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand = cand * (target_sum / cand.sum())
                    cand_sc = _fast_score(cand)
                    if cand_sc < best_sc:
                        best_seq, best_sc = cand, cand_sc
                step = 0.2
                gamma = max(gamma * 0.9, 2.0)       # relax a little

        # early exit if we already beat the target
        if best_sc < 1.503:
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Final stochastic polishing (Gaussian hill‑climb)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray, deadline: float,
                  step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """Very short isotropic Gaussian perturbation loop."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to the fixed ℓ¹‑norm
        target = math.sqrt(2.0 * n)
        cand = cand * (target / cand.sum())

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9997                     # cooling
        if step < 1e-7:
            step = step_initial

    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence discovered before the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a tiny safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a possibly saved incumbent
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = init * (target0 / init.sum())
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 n={n0} score={sc0:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Exhaustive parametric sweep on several lengths
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # best hand‑crafted family for this length
        fam_vec, fam_sc = _candidate_families(n, target)
        if fam_sc < best_score:
            best_seq, best_score = fam_vec.copy(), fam_sc
            print(f"[family] n={n} new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣  Smoothed gradient descent starting from the family optimum
        # ------------------------------------------------------------------
        sub_deadline = min(deadline,
                           time.monotonic() + (deadline - time.monotonic()) * 0.3)
        refined_vec, refined_sc = _smooth_gradient_descent(fam_vec,
                                                          target,
                                                          sub_deadline)
        if refined_sc < best_score:
            best_seq, best_score = refined_vec.copy(), refined_sc
            print(f"[smooth] n={n} improved to {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3️⃣  Random Dirichlet restarts (very cheap) for the same length
        # ------------------------------------------------------------------
        rnd_deadline = time.monotonic() + min(0.4,
                                              (deadline - time.monotonic()) / 2)
        while time.monotonic() < rnd_deadline:
            a = rng.random(n)
            a = a / a.sum() * target
            sc = _fast_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[rand] n={n} improved to {best_score:.6f}")

            # a quick refinement on this random point
            sub_deadline2 = time.monotonic() + min(0.2,
                                                   (deadline - time.monotonic()) / 2)
            a_ref, sc_ref = _smooth_gradient_descent(a, target, sub_deadline2)
            if sc_ref < best_score:
                best_seq, best_score = a_ref.copy(), sc_ref
                print(f"[rand+smooth] n={n} improved to {best_score:.6f}")

            # only a few random restarts per length
            if time.monotonic() > rnd_deadline:
                break

        # stop early if we have already reached the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing (Gaussian hill‑climb) on the global incumbent
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_sc = _local_search(best_seq, deadline,
                                                 step_initial=0.2)
        if polished_sc < best_score:
            best_seq, best_score = polished_seq, polished_sc
            print(f"[final] polished to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Fallback – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 400
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    # Convert to plain Python list for the harness
    return [float(v) for v in best_seq]
