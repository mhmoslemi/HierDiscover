"""
Optimisation of the auto‑convolution peak of a non‑negative step function.

The approach combines three ideas:

1️⃣  Warm‑start – use the best known construction if it exists,
    otherwise initialise with a Dirichlet sample (uniform on the simplex).

2️⃣  Adam optimiser on a smoothed (log‑sum‑exp) approximation of the
    ℓ∞‑norm of the convolution.  After each step we project back on the
    ℓ¹‑simplex (∑ a_i = √(2 n), a_i ≥ 0) and clip to the interval required
    by ``evaluate_sequence``.  The smoothing parameter β is slowly increased
    (annealed) so that optimisation first explores a smooth landscape and
    finally focuses on the true maximum.

3️⃣  A short Gaussian hill‑climbing phase that perturbs the best vector while
    staying on the simplex.  This often refines the solution found by the
    gradient method.

The whole routine respects the global time budget (default 1000 s) and
returns the best sequence found.  The code uses only NumPy and the standard
library and is fully self‑contained.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_score(v: np.ndarray) -> float:
    """Exact evaluation of the provided ``evaluate_sequence`` formula."""
    n = len(v)
    conv = np.convolve(v, v)
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights for a given β (numerically stable)."""
    max_val = conv.max()
    exp_vals = np.exp(beta * (conv - max_val))
    w_sum = exp_vals.sum()
    if w_sum == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / w_sum


def _weighted_gradient(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(f*f)_k w.r.t. f.
    w must have length 2·len(f)‑1 and Σ w = 1.
    """
    n = len(f)
    grad = np.zeros_like(f)
    # loop over each convolution offset
    for k in range(2 * n - 1):
        wk = w[k]
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        grad[i_start:i_end + 1] += wk * 2.0 * f[j_range]
        # diagonal correction (i == j)
        if (k & 1) == 0:          # k even → possible i = j
            i_eq = k // 2
            if i_start <= i_eq <= i_end:
                grad[i_eq] -= wk * f[i_eq]   # because 2·c – c = c
    return grad


def _adam_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_start: float = 0.2,
    beta_factor: float = 1.2,
    beta_max: float = 256.0,
    step_size: float = 0.5,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smoothed max of the auto‑convolution.

    Returns the best vector encountered and its exact score.
    """
    f = f0.copy()
    best = f.copy()
    best_score = _compute_score(best)

    # Adam moments
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    beta = beta_start
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        it += 1

        conv = np.convolve(f, f)
        w = _smooth_weights(conv, beta)

        grad = _weighted_gradient(f, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        cand = f - step_size * m_hat / (np.sqrt(v_hat) + eps)

        # projection + clipping (same protection as ``evaluate_sequence``)
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best, best_score = cand.copy(), cand_score
            f = cand               # keep the improvement
        else:
            # even if not improving we still move (Adam already did)
            f = cand

        # anneal the smooth‑max temperature
        beta = min(beta * beta_factor, beta_max)

        # tiny step‑size reduction if we are stagnating for a while
        if it % 200 == 0 and not np.isfinite(cand_score):
            step_size *= 0.7

    return best, best_score


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.2,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian perturbation hill‑climber that stays on the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score(cand)

        if cand_score < best_score:
            best, best_score = cand, cand_score

        sigma *= 0.9995
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial
    return best, best_score


def _resample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length ``new_n``."""
    if len(seq) == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, len(seq))
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, seq)

        restarts = 12 if n <= 1000 else 6

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise: either the current best (resampled) or a fresh Dirichlet sample
            if best_seq is not None:
                f0 = _resample_sequence(best_seq, n)
            else:
                f0 = rng.dirichlet(np.ones(n)) * target_sum

            sub_deadline = min(deadline, time.monotonic() + 12.0)

            f_opt, sc_opt = _adam_optimize(
                f0,
                target_sum,
                sub_deadline,
                beta_start=0.2,
                beta_factor=1.2,
                beta_max=256.0,
                step_size=0.5,
                max_iters=3000,
            )

            if sc_opt < best_score:
                best_seq, best_score = f_opt.copy(), sc_opt
                print(f"[update] n={n} restart={rs} score={sc_opt:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                print("[early] target attained – stopping searches")
                deadline = time.monotonic() + 0.2
                break

    # --------------------------------------------------------------
    # 3️⃣ Final polishing – short stochastic hill‑climbing.
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        refined, refined_sc = _hill_climb(best_seq, target_sum, deadline,
                                          sigma_initial=0.2, max_iters=8000)
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[refine] final score={refined_sc:.6f}")

    # final safety clip (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
