            break

        t_mix = 0.5  # could adapt

        sum_new = new_f.sum()
        if sum_new > 0:

            f = new_f

"""
Optimise a non‑negative step function for the evaluation function

    2·n·max(a*a) / (sum(a) ** 2)

The following ideas are combined:

1.  Generate a variety of initial patterns (uniform, triangular,
    Gaussian, sinusoidal, and random Dirichlet draws) on a fixed
    length `n`.  All vectors are scaled to the canonical sum
    √(2·n) – this normalisation makes the evaluation simply the
    maximal convolution entry.

2.  For each candidate run an *LP‑refinement* step:
        maximise   Σ g_j
        subject to  f * g ≤ M          (M = max(f*f))
                    g ≥ 0
    The linear program (scipy.optimize.linprog) finds a distribution
    `g` that stores as much mass as possible without raising any
    convolution entry above the current maximum `M`.  Mixing the old
    vector with `g` (convex combination) provably never increases the
    maximal convolution value, so we can iteratively decrease the
    bound.

3.  After the LP phase a short Gaussian hill‑climbing (stochastic
    local search) is applied to fine‑tune the best vector.

The whole search respects the global time budget (default 1000 s) and
returns the best sequence found.  All values are clipped to the range
required by ``evaluate_sequence`` before returning.
"""

import math
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – these are pure NumPy and do not depend on the
# external evaluation function.
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_score(v: np.ndarray) -> float:
    """Exact evaluation of the provided ``evaluate_sequence`` formula."""
    n = len(v)
    conv = np.convolve(v, v)
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.2,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """Gaussian perturbation hill‑climbing on the simplex."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        sigma *= 0.9995
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial
    return best, best_score


# ----------------------------------------------------------------------
# Pattern generators – deterministic starting points
# ----------------------------------------------------------------------


def _uniform_pattern(n: int, target_sum: float) -> np.ndarray:
    a = np.full(n, target_sum / n, dtype=np.float64)
    return a


def _triangular_pattern(n: int, target_sum: float) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 1)
        right = np.arange(half + 1, 0, -1)
    a = np.concatenate([left, right], dtype=np.float64)
    a = a * (target_sum / a.sum())
    return a


def _gaussian_pattern(n: int, target_sum: float, sigma_factor: float = 0.25) -> np.ndarray:
    x = np.arange(n)
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
    a = a * (target_sum / a.sum())
    return a


def _sinusoid_pattern(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n) + 0.5
    a = np.sin(np.pi * x / n)
    a = a * (target_sum / a.sum())
    return a


# ----------------------------------------------------------------------
# LP refinement – core of the novel optimisation idea
# ----------------------------------------------------------------------


def _lp_refine(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iter: int = 6,
    t_initial: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Iteratively improve ``f`` using the linear‑programming step described
    in the statement.  Each iteration solves

            maximise  Σ g_j
            subject to f * g ≤ M   (M = max(f*f))
                       g ≥ 0

    The solution ``g`` is mixed into ``f``; the mixing factor ``t`` is
    slowly increased when progress is observed.
    Returns the best vector seen and its evaluation score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)

    t = t_initial
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        # current peak
        conv = np.convolve(f, f)
        M = conv.max()

        n = len(f)

        # Build the Toeplitz matrix A such that (A @ g)[k] = (f * g)[k]
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i, fi in enumerate(f):
            if fi == 0.0:
                continue
            A[i : i + n, i] = fi

        # Linear program: maximise sum(g)  →  minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # LP failed – abort the refinement
            break

        g = res.x

        # Convex combination, then renormalise to the canonical sum
        new_f = (1.0 - t) * f + t * g
        s = new_f.sum()
        if s > 0.0:
            new_f = new_f * (target_sum / s)

        new_score = _compute_score(new_f)

        if new_score < best_score - 1e-12:
            best_f, best_score = new_f.copy(), new_score
            f = new_f
            # progress → be a bit more aggressive
            t = min(0.9, t * 1.05)
        else:
            # no improvement – become more conservative
            t = max(0.1, t * 0.7)
            if t <= 0.1:
                break

    return best_f, best_score
