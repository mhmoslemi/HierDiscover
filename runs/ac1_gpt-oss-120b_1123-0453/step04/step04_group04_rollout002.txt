def compute_convolution(a):
    n = len(a)
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    needed = 2*n-1
    return conv[:needed]

def max_convolution(a): return np.max(compute_convolution(a))

def gradient(a, p):
    # p same length as conv (2n-1)
    a_rev = a[::-1]
    # convolution a_rev * p
    n = len(a)
    fft_len = 1 << ((len(a_rev) + len(p) - 1).bit_length())
    fa = np.fft.rfft(a_rev, n=fft_len)
    fp = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(fa * fp, n=fft_len)
    needed = len(a_rev) + len(p) - 1
    conv = conv[:needed]
    start = n-1
    return conv[start:start+n]

Implementation details:

def project_simplex(v, s):
    assert s >= 0
    n = len(v)
    # v sorted descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # return uniform
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent optimizer for the step‑function bound.

The algorithm works with the ℓ¹‑normalisation 
    Σ a_i = √(2 n)  →  the evaluation function reduces to 
    max_k (a * a)[k] (the maximal auto‑convolution entry).

For each chosen length n we run several independent restarts.
Each restart performs a smoothed optimisation of the max‑convolution
using a soft‑max (log‑sum‑exp) surrogate with a temperature parameter α.
The gradient of the surrogate is obtained analytically in O(n log n)
via FFTs, then we take a projected gradient step onto the ℓ¹‑simplex.
The temperature α is increased during the run so that the surrogate
gradually focuses on the true maximum.

After the gradient phase a short greedy polishing (mass‑transfer hill‑climb)
is applied while time permits.  The best sequence found over all
explored lengths is returned.

All helper functions are defined at top level – no closures or lambdas – 
and the code respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, Σx = s}.
    Returns a new array with the same shape as v.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0

    if not np.any(cond):
        # corner case: all zero or s==0
        return np.full_like(v, s / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full (linear) auto‑convolution a * a.
    Returns a 1‑D array of length 2·len(a)−1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    needed = 2 * n - 1
    return conv[:needed]


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution a*a."""
    return float(np.max(_full_convolution(a)))


def _gradient_via_fft(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate with respect to a.

    For a given probability vector p (∑p = 1, length 2·n−1) the gradient is
        g_i = Σ_j a_j · p_{i+j}   (0 ≤ i < n).

    The computation is performed with a single FFT‑based convolution.
    """
    a_rev = a[::-1]
    n = a.shape[0]
    conv_len = a_rev.shape[0] + p.shape[0] - 1          # = 3n‑2
    fft_len = 1 << (conv_len.bit_length())
    fa = np.fft.rfft(a_rev, n=fft_len)
    fp = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(fa * fp, n=fft_len)
    conv = conv[:conv_len]                              # keep the needed part
    start = n - 1
    return conv[start:start + n]                        # length n


def _gradient_descent(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Smoothed projected gradient descent for a single restart.
    Returns the best sequence encountered and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best_seq = a.copy()
    best_val = _max_convolution(a)

    # hyper‑parameters
    alpha = 10.0            # start temperature for the soft‑max
    alpha_max = 2000.0
    step_base = 0.5 * target_sum / n   # typical magnitude of a coordinate

    it = 0
    while time.monotonic() < deadline:
        # current auto‑convolution and max
        conv = _full_convolution(a)
        cur_max = float(np.max(conv))

        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            # occasional progress report
            print(f"[gd] n={n} it={it} max={cur_max:.6f} α={alpha:.2f}")

        # soft‑max probabilities (stable formulation)
        shifted = conv - cur_max                     # all ≤ 0
        exp_vals = np.exp(alpha * shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp == 0.0:
            # fall back to uniform distribution
            p = np.full_like(exp_vals, 1.0 / exp_vals.size)
        else:
            p = exp_vals / sum_exp

        # gradient of the surrogate
        g = _gradient_via_fft(a, p)

        # diminishing step size
        step = step_base / math.sqrt(it + 1)

        # projected gradient step
        a_new = a - step * g
        a_new = _project_onto_simplex(a_new, target_sum)

        a = a_new
        it += 1

        # slowly raise the temperature so the surrogate focuses more on the true max
        if it % 200 == 0:
            alpha = min(alpha * 1.5, alpha_max)

    return best_seq, best_val


def _greedy_polish(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that repeatedly transfers a small amount of mass
    between two coordinates while keeping the ℓ¹‑norm fixed.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best_seq = a.copy()
    best_val = _max_convolution(a)

    while time.monotonic() < deadline:
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose a small transfer (10 % of the typical magnitude)
        delta = (rng.random() - 0.5) * 0.2 * (a[i] + a[j] + 1e-12)

        # respect non‑negativity
        if delta > a[i]:
            delta = a[i]
        if -delta > a[j]:
            delta = -a[j]
        if abs(delta) < 1e-12:
            continue

        # apply move
        a[i] -= delta
        a[j] += delta

        cur = _max_convolution(a)
        if cur < best_val - 1e-12:
            best_val = cur
            best_seq = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

    # re‑normalise (tiny numerical drift may appear)
    best_seq = _project_onto_simplex(best_seq, target_sum)
    best_val = _max_convolution(best_seq)
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  The routine explores several sequence lengths,
    runs a few gradient‑descent restarts per length, and finally applies a
    short greedy polish.  It respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣ choose a set of lengths to try
    # ------------------------------------------------------------------
    lengths = [500, 800, 1000, 1200, 1500, 1800, 2000]

    # possibly reuse a globally stored sequence as an extra candidate length
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        n0 = len(h1)
        if n0 not in lengths:
            lengths.append(n0)
    lengths = sorted(set(lengths))

    best_seq = None
    best_val = float("inf")

    # allocate roughly equal time to each length
    remaining = deadline - time.monotonic()
    per_len = max(0.2, remaining / max(1, len(lengths)))

    for idx, n in enumerate(lengths):
        if time.monotonic() >= deadline:
            break

        sub_deadline = min(deadline, time.monotonic() + per_len)
        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Warm‑start from the global sequence if it matches the length
        # ------------------------------------------------------------------
        if isinstance(h1, (list, tuple, np.ndarray)) and len(h1) == n:
            a_start = np.array(h1, dtype=np.float64)
            a_start = np.clip(a_start, 0.0, 1000.0)
            s = np.sum(a_start)
            if s > 0.0:
                a_start = a_start * (target_sum / s)
            else:
                a_start = np.full(n, target_sum / n)
            val = _max_convolution(a_start)
            if val < best_val:
                best_val = val
                best_seq = a_start.copy()
                print(f"[warm] n={n} val={val:.6f}")

        # ------------------------------------------------------------------
        # Run a few independent restarts of the gradient optimiser
        # ------------------------------------------------------------------
        restarts = 4
        for r in range(restarts):
            if time.monotonic() >= sub_deadline:
                break

            # allocate a slice of the remaining time for this restart
            now = time.monotonic()
            time_left = sub_deadline - now
            # give each restart at most a fraction of the remaining budget
            restart_deadline = now + max(0.01, time_left / max(1, restarts - r))

            # random Dirichlet‑like start
            a0 = rng.random(n) + 1e-6
            a0 = a0 / np.sum(a0) * target_sum

            seq_local, val_local = _gradient_descent(a0, restart_deadline, rng)

            if val_local < best_val:
                best_val = val_local
                best_seq = seq_local.copy()
                print(f"[gd‑best] n={n} restart={r} val={val_local:.6f}")

        # recompute remaining time and per‑length allocation
        remaining = deadline - time.monotonic()
        lengths_left = len(lengths) - (idx + 1)
        per_len = max(0.1, remaining / max(1, lengths_left))

    # ------------------------------------------------------------------
    # 2️⃣ final polishing on the global best while time permits
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polish_deadline = deadline
        seq_polished, val_polished = _greedy_polish(best_seq, polish_deadline, rng)
        if val_polished < best_val:
            best_val = val_polished
            best_seq = seq_polished.copy()
            print(f"[polish] final val={val_polished:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # optional sanity check (does not affect the returned result)
    # try:
    #     from __main__ import evaluate_sequence
    #     ev = evaluate_sequence(best_seq.tolist())
    #     print(f"[final‑eval] evaluation={ev:.6f}")
    # except Exception:
    #     pass

    return [float(x) for x in best_seq]
