"""
Smooth‑max gradient descent for the step‑function minimisation challenge.

The evaluation metric simplifies (when the ℓ¹‑norm is fixed to √(2·n))
to the maximum entry of the auto‑convolution.
We minimise a smooth approximation of the max:

    ϕ_β(a) = (1/β)·log∑_k exp(β·(a*a)_k)

and use its exact gradient, which can be written as a cross‑correlation.
The algorithm proceeds as follows:

1.  Choose a length n (we try several and keep the best).
2.  Initialise a non‑negative vector with the required ℓ¹‑sum.
3.  Repeatedly
        – compute the full auto‑convolution,
        – form the soft‑max weights,
        – obtain the gradient via a convolution,
        – take a projected‑gradient step (with a simple momentum term),
        – enforce symmetry (a_i = a_{n‑1‑i}) and project onto the simplex.
   The smoothing parameter β is increased gradually so that the iterate
   first sees a smooth landscape and finally concentrates on the true
   max‑entry.
4.  Several random restarts are performed under the global time budget.
5.  The best vector found is returned as a plain Python list.

All intermediate vectors respect the security checks of ``evaluate_sequence``:
they are non‑negative, bounded by 1000 and have a strictly positive ℓ¹‑norm.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` (without safety checks).
    Returns 2·n·max(conv)/(sum(seq)²).  For sequences with ℓ¹‑norm √(2·n)
    this is simply the maximal auto‑convolution entry.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x >= 0, sum x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero; return uniform distribution
        return np.full(n, s / n, dtype=np.float64)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Tiny numerical drift may change the sum – renormalise harmlessly
    w_sum = w.sum()
    if w_sum > 0:
        w *= s / w_sum
    else:
        w = np.full(n, s / n, dtype=np.float64)
    return w


def _full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution using FFT.  Returns a vector of length
    len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    # Next power‑of‑two for efficient FFT
    nfft = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(fa * fb, n=nfft)[:m]
    return conv

# Smooth‑max descent
# ----------------------------------------------------------------------


def _smooth_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 2000,
) -> np.ndarray:
    """
    Projected gradient descent on the smooth approximation of the max
    auto‑convolution entry.  The iterate is kept symmetric (a[i] = a[n‑1‑i])
    and projected onto the ℓ¹‑simplex of radius ``target_sum`` after each step.
    """
    a = _project_simplex(a0.astype(np.float64), target_sum)
    n = a.shape[0]
    best = a.copy()
    best_sc = _fast_score(best)

    # momentum variables
    v = np.zeros_like(a)
    mu = 0.9

    # initial step‑size (roughly 10 % of the average entry)
    init_lr = target_sum * 0.5 / n
    beta = 1.0  # smoothing parameter, will grow over time

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # ----- auto‑convolution -----
        conv = _full_convolution(a, a)                     # length 2n‑1
        max_conv = conv.max()

        # ----- smooth max (log‑sum‑exp) -----
        shift = max_conv                     # for numerical stability
        exp_vals = np.exp(beta * (conv - shift))
        sumexp = exp_vals.sum()
        w = exp_vals / sumexp                # soft‑max weights, sum to 1

        # ----- gradient (cross‑correlation of w with a) -----
        corr_full = _full_convolution(w, a[::-1])          # length 3n‑2
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]          # slice of length n

        # ----- step‑size schedule -----
        lr = init_lr / (1.0 + 0.001 * it)   # slowly decaying

        # ----- momentum update -----
        v = mu * v - lr * grad
        cand = a + v

        # ----- enforce symmetry -----
        cand = 0.5 * (cand + cand[::-1])

        # ----- projection onto feasible set -----
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)

        a = cand

        # ----- adapt smoothing parameter -----
        if it % 100 == 0:
            beta = min(beta * 1.20, 80.0)   # slowly increase, never too large

        # ----- keep the best exact solution seen so far -----
        if it % 20 == 0:
            sc = _fast_score(a)
            if sc < best_sc - 1e-12:
                best, best_sc = a.copy(), sc
                # reset momentum to avoid overshooting from a good region
                v[:] = 0.0

    return best

    target = math.sqrt(2.0 * n)
    best_seq = _project_simplex(best_seq, target)

    # Simple hill‑climb: move a tiny amount of mass from a relatively large entry
    # to a small one, accept if it improves the exact score.
    final_deadline = time.monotonic() + min(30.0, deadline - time.monotonic())
    while time.monotonic() < final_deadline:
        donors = np.where(best_seq > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best_seq[donor] * 0.08, target * 0.001)
        cand = best_seq.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _project_simplex(cand, target)
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    final_score = _fast_score(best_seq)
    print(f"[final] best score {final_score:.6f}   length {n}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
