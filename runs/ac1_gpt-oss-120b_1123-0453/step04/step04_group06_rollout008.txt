def _conv_max(a, fft_len=None):
    n = len(a)
    L = fft_len if fft_len is not None else 1
    while L < 2*n-1:
        L <<= 1
    # Use np.fft.rfft to compute real FFT
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    # clip negative tiny values due to numerical errors:
    conv = np.maximum(conv, 0.0)
    return conv, conv.max()

def _hard_max_descent(a0, target_sum, deadline, max_iters=5000):
    a = a0.copy()
    # ensure sum equals target_sum (project)
    a = _project_onto_simplex(a, target_sum)
    best_a = a.copy()
    # compute initial conv and max
    conv, cur_max = _conv_max(a)
    best_max = cur_max
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    t=0
    lr = max(0.001, target_sum / len(a) * 0.4)  # some heuristic
    stagnation = 0

    while time.monotonic() < deadline and t < max_iters:
        # gradient?
        # find indices where conv approx equal cur_max (within tolerance)
        tol = max(1e-12, cur_max * 1e-10)
        idxs = np.where(np.abs(conv - cur_max) <= tol)[0]
        # compute gradient aggregated
        grad = np.zeros_like(a)
        for k in idxs:
            j = k - np.arange(len(a))
            mask = (j >= 0) & (j < len(a))
            grad[mask] += 2 * a[j[mask]]
        grad = grad / len(idxs)
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # descent direction -grad
        a_candidate = a - step
        # project onto simplex
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        # ensure not too small or >1000 but no need.
        # compute conv and max
        conv_candidate, max_candidate = _conv_max(a_candidate)

        if max_candidate < cur_max - 1e-12:
            # accept improvement
            a = a_candidate
            conv = conv_candidate
            cur_max = max_candidate
            # enlarge step size
            lr = min(lr * 1.05, 5.0)  # avoid huge
            stagnation = 0
        else:
            # reject; shrink step size
            lr *= 0.7
            stagnation += 1

        # update best
        if cur_max < best_max - 1e-12:
            best_a = a.copy()
            best_max = cur_max
            # could reset stagnation count

        # early break if lr too small or stagnation large
        if lr < 1e-8 or stagnation >= 200:
            # maybe restart with different stochastic step or break
            break

    return best_a, best_max

"""
author:  expert‑solver
date:    2026‑11‑23

A fast non‑convex optimizer for the step‑function bound.
The code searches over several sequence lengths, using many
random restarts of a simplex‑projected Adam optimiser that
minimises the *hard* maximum of the auto‑convolution.
Only the gradient of the active convolution offsets is used,
so each iteration is O(n log n) (FFT) not O(n²).
After the descent a short Gaussian hill‑climb is performed.
All numbers are kept non‑negative and are projected onto the
ℓ¹‑simplex with Σaᵢ = √(2 n), which makes the evaluation
function equal to the peak of the convolution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x | x_i≥0, Σx_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute the linear (non‑circular) convolution a*a via FFT
    and return the whole convolution together with its maximum entry.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    # real FFT – a is real, so rfft/irfft are a bit faster
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv, float(conv.max())


def _max_gradient(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of Σ_k w_k·(a*a)_k where the weights w_k are uniform
    over the active (maximal) convolution offsets ``idxs``.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    # vectorised contribution for each active index
    for k in idxs:
        j = k - np.arange(n)                 # potential partner indices
        mask = (j >= 0) & (j < n)            # keep those inside the array
        grad[mask] += 2.0 * a[j[mask]]
    grad /= float(len(idxs))
    return grad


def _hard_max_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that only follows the gradient of the *active* (maximal)
    auto‑convolution entry.  The iterate is projected onto the ℓ¹‑simplex
    after each step.  Returns the best vector seen and its convolution peak.
    """
    a = _project_onto_simplex(a0, target_sum)
    conv, cur_max = _conv_max(a)
    best_a = a.copy()
    best_max = cur_max

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # step‑size initialised to a small fraction of a typical coordinate
    lr = max(0.001, (target_sum / a.shape[0]) * 0.4)
    stagnation = 0

    while time.monotonic() < deadline and t < max_iters:
        # indices where the convolution attains its current maximum
        tol = max(1e-12, cur_max * 1e-10)
        idxs = np.where(np.abs(conv - cur_max) <= tol)[0]
        grad = _max_gradient(a, idxs)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step                     # descend
        cand = _project_onto_simplex(cand, target_sum)

        # compute the new peak
        conv_cand, cand_max = _conv_max(cand)

        if cand_max < cur_max - 1e-12:
            # successful improvement – accept
            a = cand
            conv = conv_cand
            cur_max = cand_max
            lr = min(lr * 1.05, 5.0)        # try to accelerate a bit
            stagnation = 0
        else:
            # reject – shrink the learning rate
            lr *= 0.7
            stagnation += 1

        # keep the best point seen so far
        if cur_max < best_max - 1e-12:
            best_a = a.copy()
            best_max = cur_max
            stagnation = 0

        # early termination criteria
        if lr < 1e-8 or stagnation >= 300:
            break

    return best_a, best_max


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    sigma_initial: float = 0.3,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex.
    Small Gaussian perturbations are applied and the best point is kept.
    """
    best = start.copy()
    _, best_max = _conv_max(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        _, cand_max = _conv_max(cand)

        if cand_max < best_max:
            best = cand
            best_max = cand_max
            # keep sigma around the successful scale
            sigma = sigma_initial

        sigma *= 0.9995
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial

    return best, best_max


def _score_from_vector(a: np.ndarray) -> float:
    """Exact score = 2·n·max(conv) / (sum(a)²).  With the simplex normalisation it is just max(conv)."""
    n = a.shape[0]
    _, max_val = _conv_max(a)
    # use the official evaluator for safety (it also clips the vector)
    return evaluate_sequence([float(x) for x in a])
