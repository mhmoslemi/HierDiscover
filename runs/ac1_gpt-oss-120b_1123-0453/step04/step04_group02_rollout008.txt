def propose_candidate(seed=0, budget_s=1000):
    import needed modules; rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + float(budget_s) - 0.05

    best_seq = None; best_score = np.inf

    # maybe have length choices up to 2000 (or 2500) the optimum may be bigger.
    lengths = [200, 400, 600, 800, 1000, 1500, 2000]

    for n in lengths:
        if time.monotonic() >= deadline: break
        target = math.sqrt(2.0 * n)

        # parametric families including new ones
        cand_seq, cand_sc = _parametric_sweep(n, target, deadline)
        if cand_sc < best_score:
            best_seq, best_score = cand_seq.copy(), cand_sc

        # cheap random Dirichlet
        rnd_deadline = time.monotonic() + min(0.2, (deadline - time.monotonic()) / 5)
        while time.monotonic() < rnd_deadline:
            a = rng.random(n)
            a = a / a.sum() * target
            sc = _fast_score(a)
            if sc < best_score:

    # gradient descent - we may run multiple restarts
    # allocate a portion of time for this.
    gd_stop = time.monotonic() + max(0, (deadline - time.monotonic()) * 0.6)

    while time.monotonic() < gd_stop:
        # choose start point: maybe best_seq or new random
        if best_seq is not None and rng.random() < 0.7:
            start_seq = best_seq.copy()
        else:
            n = rng.choice(lengths)

        # run gradient descent for limited time (deadline_gd = min(gd_stop, time+some seconds)

        if gd_sc < best_score:

            print(f"[gd] improved to {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline:

def _parametric_sweep(n, target, deadline):
    best_seq = None; best_sc = np.inf
    idx = np.arange(n, dtype=np.float64)
    # baseline uniform
    a = np.full(n, target/n, dtype=np.float64)
    sc = _fast_score(a)
    if sc < best_sc: best_seq,best_sc = a.copy(), sc

    # exponential
    # lambda values from 0.5 to 20 (step 0.5)
    lambdas = np.concatenate((np.arange(0.5, 5.5, 0.5), np.arange(5,21,2), np.arange(21,101,10)))
    for lam in lambdas:
        α = lam / n
        a = np.exp(-α * idx)
        a = a / a.sum() * target
        sc = _fast_score(a)
        if sc < best_sc: best_seq,best_sc = a.copy(), sc

    # power law
    p_vals = np.linspace(0.05,5,120)
    base = np.arange(1, n+1, dtype=np.float64)
    for p in p_vals:
        a = base ** (-p)
        a = a / a.sum() * target
        sc = _fast_score(a)
        if sc < best_sc: best_seq,best_sc = a.copy(), sc

    # windows: Hann, Hamming, Blackman, sine, triangular (already)
    # Add to list similarly as before; we can reuse code but fine.

    # Gaussian
    # sigma factor values
    sigma_factors = np.linspace(0.1, 0.5, 17)  # from 0.1 n to 0.5 n
    mu = (n - 1) / 2.0
    for sf in sigma_factors:
        sigma = sf * n
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        a = a / a.sum() * target
        sc = _fast_score(a)
        if sc < best_sc: best_seq,best_sc = a.copy(), sc

    # Beta distribution
    p_vals = [0.5, 1, 2, 3, 5]
    q_vals = [0.5, 1, 2, 3, 5]
    x = (idx + 0.5) / n
    for p in p_vals:
        for q in q_vals:

"""
Low‑level optimiser for the non‑negative step‑function score.

The algorithm consists of three phases:

1️⃣  **Parametric sweep** – for a set of lengths we generate many
    hand‑crafted families (exponential, power‑law, Gaussian,
    Beta, logistic, Hann / Hamming / Blackman, sine, triangular).
    The families are sampled densely (especially the decay parameters)
    and the best candidate from all families is kept as a global incumbent.

2️⃣  **Multi‑peak sub‑gradient descent** – starting from the incumbent we
    repeatedly locate the *top K* convolution entries (instead of a single
    maximum).  The sub‑gradient of each active entry is the reversed slice
    of the sequence involved in the convolution; we average these gradients,
    make the step orthogonal to the ℓ¹‑direction (to keep the sum exactly
    √(2·n)), adapt the step size and occasionally perform a random “shake‑up”.
    This cheap first‑order method drives down the maximal auto‑convolution
    value much faster than a single‑peak descent.

3️⃣  **Stochastic polishing** – a short Gaussian hill‑climb (random
    perturbations + renormalisation) cleans up the remaining noise.

All operations respect the security wrapper of ``evaluate_sequence``:
non‑negative entries, clipping to [0, 1000] and a fixed ℓ¹‑norm.
The routine returns the best sequence discovered before the time budget
runs out (the outer harness supplies ``evaluate_sequence``).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (full linear) via FFT
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of `seq` with itself using FFT."""
    n = seq.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # smallest power‑of‑2 ≥ 2·n‑1
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Exact objective (without the security wrapper)."""
    n = seq.shape[0]
    max_b = _conv_fft(seq).max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Normalisation helper (ℓ¹‑norm = target)
# ----------------------------------------------------------------------
def _normalise(v: np.ndarray, target: float) -> np.ndarray:
    """Scale v so that its ℓ¹‑norm equals target (after clipping)."""
    v = np.clip(v, 0.0, 1000.0)
    tot = v.sum()
    if tot == 0.0:
        return np.full_like(v, target / v.shape[0])
    return v * (target / tot)


# ----------------------------------------------------------------------
# Candidate families (parametric sweep)
# ----------------------------------------------------------------------
def _candidate_families(n: int, target: float, deadline: float):
    """Return the best vector among many hand‑crafted families.

    The function respects the global deadline – it aborts early if time runs out.
    """
    best_vec = None
    best_sc = float("inf")
    idx = np.arange(n, dtype=np.float64)

    # ------- helper ----------------------------------------------------
    def _try(v):
        nonlocal best_vec, best_sc
        if time.monotonic() >= deadline:
            raise RuntimeError  # abort sweep
        v = _normalise(v, target)
        sc = _fast_score(v)
        if sc < best_sc:
            best_sc, best_vec = sc, v.copy()
    # ------------------------------------------------------------------

    # Uniform (baseline)
    _try(np.full(n, target / n, dtype=np.float64))

    # ---------- Exponential decay a_i ∝ exp(-α·i) --------------------
    # λ = α·n  varies roughly from 0.5 to 100
    lambdas = np.concatenate((
        np.arange(0.5, 5.5, 0.5),
        np.arange(5, 21, 2),
        np.arange(20, 101, 10),
    ))
    for lam in lambdas:
        α = lam / n
        _try(np.exp(-α * idx))

    # ---------- Power‑law a_i ∝ (i+1)^(-p) ------------------------
    for p in np.linspace(0.05, 5.0, 120):
        _try((idx + 1) ** (-p))

    # ---------- Classic windows --------------------------------------
    # Hann
    _try(0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1))))
    # Hamming
    _try(0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1)))
    # Blackman
    _try(0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1)))
    # Sine window (full half‑period)
    _try(np.sin(np.pi * (idx + 1) / (n + 1)))
    # Triangular
    _try(np.minimum(idx + 1, n - idx))

    # ---------- Gaussian window (centered) -------------------------
    mu = (n - 1) / 2.0
    for sf in np.linspace(0.10, 0.50, 17):          # σ = sf·n
        σ = sf * n
        _try(np.exp(-0.5 * ((idx - mu) / σ) ** 2))

    # ---------- Beta‑distribution shaped sequences -----------------
    # a_i ∝ x^{p-1} (1‑x)^{q-1},   x = (i+0.5)/n   (p,q>0)
    x = (idx + 0.5) / n
    p_vals = [0.5, 1, 2, 3, 5]
    q_vals = [0.5, 1, 2, 3, 5]
    for p in p_vals:
        for q in q_vals:
            a = (x ** (p - 1)) * ((1 - x) ** (q - 1))
            _try(a)

    # ---------- Logistic (sigmoid) shapes -------------------------
    # a_i = 1 / (1 + exp(β·(i‑µ)))   with β>0, µ∈[0,n‑1]
    for β in np.linspace(1.0, 10.0, 10):          # steepness
        for µ in np.linspace(0, n - 1, 5):
            _try(1.0 / (1.0 + np.exp(β * (idx - µ))))

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Multi‑peak sub‑gradient descent
# ----------------------------------------------------------------------
def _gradient_descent_multi(seq: np.ndarray, target: float, deadline: float):
    """Iterative descent using the averaged sub‑gradient of the top‑K conv entries."""
    n = seq.shape[0]
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    step = 0.2
    rng = np.random.default_rng()
    K = 3                # number of top entries to average

    while time.monotonic() < deadline:
        conv = _conv_fft(seq)
        # indices of the K largest entries
        if K >= conv.size:
            top = np.arange(conv.size)
        else:
            top = np.argpartition(-conv, range(K))[:K]

        # accumulate sub‑gradient of each active entry
        grad = np.zeros_like(seq)
        for k in top:
            if k < n:                     # overlap starts at 0
                grad[: k + 1] += seq[k :: -1]
            else:
                off = k - (n - 1)
                grad[off:] += seq[: n - off][::-1]
        grad = grad / len(top)

        # make the step orthogonal to the ℓ¹‑direction (keeps the sum unchanged)
        mean_g = grad.mean()
        direction = -grad + mean_g

        cand = seq + step * direction
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalise(cand, target)

        cand_sc = _fast_score(cand)

        if cand_sc < best_sc - 1e-12:          # strict improvement
            best_seq, best_sc = cand.copy(), cand_sc
            seq = cand
            step = min(step * 1.07, 1.0)       # gently increase step on success
        else:
            step *= 0.5
            # occasional random shake‑up to escape plateaus
            if step < 1e-6:
                i, j = rng.integers(n, size=2)
                δ = seq[i] * 0.05
                if δ > 0:
                    cand = seq.copy()
                    cand[i] -= δ
                    cand[j] += δ
                    cand = _normalise(cand, target)
                    cand_sc = _fast_score(cand)
                    if cand_sc < best_sc:
                        seq = cand
                        best_seq, best_sc = cand.copy(), cand_sc
                step = 0.2

        if best_sc < 1.503:       # early exit if target already met
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Tiny Gaussian hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray, deadline: float,
                  step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """Short stochastic refinement using isotropic Gaussian perturbations."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = _normalise(cand, np.sqrt(2.0 * n))
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9997
        if step < 1e-7:
            step = step_initial
    return best, best_sc

    best_seq = None
