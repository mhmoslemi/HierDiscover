def _adam_optimize(init_seq, deadline, rng):
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    best_seq = a.copy()
    best_score = _fast_score(a)

    # Adam hyper params
    lr = 0.1 # base learning rate; may tune
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = 1.0
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # compute conv
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
        # compute smooth weights and gradient
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # jitter: occasionally
        if it % 250 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0:
                delta = a[i] * 0.02
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # anneal beta
        if it % 200 == 0 and beta < 500.0:
            beta *= 1.5

        # optionally adjust lr decay
        if it % 500 == 0:
            lr *= 0.9

        it += 1

    return best_seq, best_score
```

def _adam_optimize(init_seq, deadline, rng):
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    best_seq = a.copy()
    best_score = _max_autocorr(a)
    
    # hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # step size scaling
    base_lr = target / (5.0 * n)  # adjust
    lr = base_lr
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = 1.0
    it = 0
    max_iters = 8000  # optional, high
    while time.monotonic() < deadline and it < max_iters:
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # jitter
        if it % 500 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0:
                delta = a[i] * 0.03
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # beta schedule
        if it % 300 == 0 and beta < 500.0:
            beta *= 1.4

        # step size decay
        if it % 1000 == 0:
            lr *= 0.7

        it += 1

    return best_seq, best_score
```

def _gd_optimize(init_seq, deadline, rng):
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0*n)
    best_a = a.copy()
    best_score = _max_autocorr(a)

    beta = 1.0
    step_base = target / n * 5.0
    momentum = np.zeros_like(a)
    momentum_coeff = 0.9

    max_iters = 5000
    it=0
    while time.monotonic() < deadline and it < max_iters:
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # momentum update
        momentum = momentum_coeff * momentum + grad
        step = step_base / (1.0 + it * 0.001)  # slower decay
        a = a - step * momentum

        # projection
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # jitter occasionally
        if it % 300 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0:
                delta = a[i] * 0.02
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # anneal beta
        if it % 200 == 0 and beta < 2000:
            beta *= 1.6

        it += 1

    return best_a, best_score

"""
Improved step‑function search.

The algorithm consists of three phases:

0️⃣  Warm‑start from a previously discovered construction (if any).

1️⃣  A large parametric sweep.  For many lengths we generate
    families of candidates (pure power‑law, pure exponential,
    combined power‑law·exponential, sinusoidal, triangular and
    random mixtures).  After normalising the ℓ¹‑norm to √(2 n) we
    keep the best sequence seen so far.

2️⃣  Refined optimisation with a smooth‑max surrogate.
    For the most promising lengths we run several random restarts
    of a momentum‑accelerated gradient descent on the
    differentiable soft‑max of the autocorrelation.
    Gradients are obtained by a single FFT‑accelerated
    convolution, the iterate is projected onto the feasible set
    (non‑negative, ℓ¹‑norm = √(2 n)).  The smoothness parameter β
    is annealed to make the surrogate approach the true maximum.
    Occasionally a small random mass transfer (“jitter”) is added
    to escape plateaus.

3️⃣  A cheap random‑mass‑swap hill‑climb is performed while time
    permits to polish the global best.

The whole search respects the 1000 s time budget and returns a
plain Python list ready for ``evaluate_sequence``.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_fast_len(n: int) -> int:
    """Return a fast FFT length (fallback to power‑of‑2)."""
    try:
        return np.fft.next_fast_len(n)  # type: ignore[attr-defined]
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for many calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _normalize(seq: np.ndarray) -> np.ndarray:
    """
    Clip to [0,1000] and enforce ℓ¹‑norm = √(2·n).
    The function works in‑place on a copy.
    """
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total > 0.0:
        seq = seq * (target / total)
    else:  # degenerate case – fill uniformly
        seq = np.full(n, target / n, dtype=np.float64)
    return seq


def _max_autocorr(seq: np.ndarray) -> float:
    """Exact max of the autocorrelation (i.e. the score after normalisation)."""
    conv = _linear_convolve(seq, seq)
    return float(conv.max())


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights over the convolution entries.
    Returns a probability vector that concentrates on the current maximal entries.
    """
    cmax = conv.max()
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    For an entry a_i the derivative is 2·∑_j w_{i+j}·a_j.
    It can be obtained by a single cross‑correlation via FFT.
    """
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)          # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]          # central n entries
    grad = 2.0 * segment[::-1]
    return grad


def _interpolate(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    if new_len == seq.shape[0]:
        return seq.copy()
    old_x = np.arange(seq.shape[0], dtype=np.float64)
    new_x = np.linspace(0, seq.shape[0] - 1, new_len, dtype=np.float64)
    return np.interp(new_x, old_x, seq)


def _gd_optimize(
    init_seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max gradient descent with momentum.
    Returns the best sequence encountered and its true max‑autocorrelation.
    """
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    best_seq = a.copy()
    best_score = _max_autocorr(a)

    # hyper‑parameters
    beta = 1.0
    step_base = (target / n) * 5.0          # similar magnitude as in the reference code
    momentum = np.zeros_like(a)
    mom_coeff = 0.9

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # ----- current evaluation (exact) -----
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()

        # ----- smooth‑max surrogate and gradient -----
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # ----- momentum step -----
        momentum = mom_coeff * momentum + grad
        step = step_base / (1.0 + it * 0.001)      # very slow decay
        a = a - step * momentum

        # ----- projection onto feasible set -----
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # ----- occasional stochastic jitter -----
        if it % 300 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.03
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # ----- anneal β (sharpen the surrogate) -----
        if it % 200 == 0 and beta < 3000.0:
            beta *= 1.55

        it += 1

    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the allocated time budget and returns the best
    sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the global best (no closure – inline)
    # ------------------------------------------------------------------
    # (the code appears multiple times; keep it short in the loop)

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously found construction (if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            cand = _normalize(cand)
            sc = _max_autocorr(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[global best] warm‑start n={len(cand)} score={sc:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Parametric sweep – generate many cheap candidates
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500, 1800, 2000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(n, dtype=np.float64)

        # ---- power‑law families ----
        for p in np.linspace(0.05, 4.5, 25):
            a = (idx + 1) ** (-p)
            a = _normalize(a)
            sc = _max_autocorr(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[global best] n={n} pow p={p:.3f} score={sc:.6f}")

        # ---- exponential families ----
        for alpha in np.linspace(0.0005, 0.12, 25):
            a = np.exp(-alpha * idx)
            a = _normalize(a)
            sc = _max_autocorr(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[global best] n={n} exp α={alpha:.4f} score={sc:.6f}")

        # ---- combined power‑law * exponential ----
        for _ in range(20):
            p = rng.uniform(0.0, 5.0)
            alpha = rng.uniform(0.0, 0.15)
            a = ((idx + 1) ** (-p)) * np.exp(-alpha * idx)
            a = _normalize(a)
            sc = _max_autocorr(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[global best] n={n} comb p={p:.2f} α={alpha:.4f} score={sc:.6f}")

        # ---- sinusoidal families ----
        for power in np.linspace(0.5, 3.0, 12):
            a = np.sin(np.pi * (idx + 0.5) / n) ** power
            a = _normalize(a)
            sc = _max_autocorr(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[global best] n={n} sin pow={power:.2f} score={sc:.6f}")

        # ---- triangular / tent shape ----
        a = np.minimum(idx + 1, n - idx)                       # linear increase then decrease
        a = _normalize(a)
        sc = _max_autocorr(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[global best] n={n} tent score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Refined gradient‑descent optimisation on promising lengths
    # ------------------------------------------------------------------
    # We pick the top‑k lengths (by the best score observed) for intensive work.
    # If we have never found a candidate yet, fall back to the smallest length.
    if best_seq is None:
        candidate_lengths = [200]
    else:
        # gather the scores we observed per length (rough estimate using best_seq length)
        candidate_lengths = [best_seq.shape[0]]
        # also add a few neighbouring lengths to explore scaling effects
        for delta in (-100, 100, -200, 200):
            cand_len = best_seq.shape[0] + delta
            if 150 <= cand_len <= 2500:
                candidate_lengths.append(cand_len)
        # ensure we also try a couple of longer lengths
        candidate_lengths.extend([1800, 2000])

    # deduplicate while preserving order
    seen = set()
    final_lengths = []
    for l in candidate_lengths:
        if l not in seen:
            seen.add(l)
            final_lengths.append(l)

    for n in final_lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # number of random restarts – more for smaller n (cheaper)
        restarts = 4 if n <= 800 else 2

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # random Dirichlet‑like start
            start = rng.gamma(1.0, size=n)
            start = start / start.sum() * target

            # blend with an up‑sampled/global best when dimensions differ
            if best_seq is not None:
                if best_seq.shape[0] != n:
                    up = _interpolate(best_seq, n)
                    start = (start + up) * 0.5
                else:
                    start = (start + best_seq) * 0.5
                start = _normalize(start)

            seq_opt, sc_opt = _gd_optimize(start, deadline, rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = seq_opt.copy()
                print(f"[global best] refined n={n} restart={r} score={sc_opt:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                print("[early exit] target reached")
                return [float(x) for x in best_seq.tolist()]

    # ------------------------------------------------------------------
    # 3️⃣ Final cheap hill‑climb while time permits
    # ------------------------------------------------------------------
    if best_seq is not None:
        while time.monotonic() < deadline:
            n = best_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            delta = best_seq[i] * 0.02
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _normalize(cand)
            sc = _max_autocorr(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"[global best] hill‑climb n={n} score={sc:.6f}")
            if best_score < 1.503:
                break
    else:
        # fallback – uniform vector of modest length
        n = 300
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best_score = _max_autocorr(best_seq)
        print(f"[fallback] using uniform n={n} score={best_score:.6f}")

    # Return a plain Python list as required by the outer harness
    return [float(x) for x in best_seq.tolist()]
