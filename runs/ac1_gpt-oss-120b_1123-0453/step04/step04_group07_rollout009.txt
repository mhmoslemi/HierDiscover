def conv_max(seq):
    # compute convolution via FFT to get length 2n-1, avoid O(n^2)
    N = 2*n -1
    m = 1 << (N-1).bit_length()  # next power of two
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)[:N]
    return np.max(conv)

def golden_section_search(func, lo, hi, tol=1e-5, max_iter=60):
    φ = (np.sqrt(5) - 1) / 2
    x1 = hi - φ * (hi - lo)
    x2 = lo + φ * (hi - lo)
    f1 = func(x1)
    f2 = func(x2)
    for _ in range(max_iter):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - φ * (hi - lo)
            f1 = func(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + φ * (hi - lo)
            f2 = func(x2)
        if hi - lo < tol:
            break
    # choose best among x1, x2
    if f1 < f2:
        return x1, f1
    else:
        return x2, f2

def golden_section(f, a, b, tol=1e-5, max_iter=30):
    phi = (np.sqrt(5) - 1) / 2
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if time.monotonic() >= deadline:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = f(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = f(d)
        if abs(b - a) < tol:
            break
    if fc < fd:
        return c, fc
    else:
        return d, fd

The objective of `evaluate_sequence` is  

\[
\text{score}= \frac{2n\;\max (f*f)}{(\sum f)^2}\; .
\]

If we normalise the sum to \(\sum f=\sqrt{2n}\) the denominator becomes
\(2n\) and the score is simply the **maximum entry of the auto‑convolution**
\(b_k=(f*f)_k\).  
Thus we have to find a non‑negative vector `f` (with a fixed ℓ¹‑norm) that
makes the largest convolution entry as small as possible.

A one‑parameter family that works extremely well is an **exponential
profile**

\[
f_i = c\,r^{\,i},\qquad i=0,\dots ,n-1,\qquad0<r<1 .
\]

For a fixed length \(n\) the normalising constant is  

\[
c =\frac{\sqrt{2n}\;(1-r)}{1-r^{\,n}} .
\]

The maximum of the convolution can be written in closed form, but it is
easier (and still very cheap) to evaluate it numerically.  
For a given `n` we minimise the score over the single variable
\(x=p\,n\) where \(p=1-r\).  
The optimal value of \(x\) is independent of `n` (≈ 1.256), so the same
construction works for any length.

The optimiser used here is a **golden‑section search** on the scalar
parameter `x`.  After the best exponential candidate has been found we
run a short **simulated‑annealing** polish to possibly improve it further.
The whole routine respects the global time budget (default 1000 s) and
returns the best sequence discovered.

---  

```python
"""
Propose a low‑score non‑negative step sequence.

The algorithm searches the 1‑parameter family  a_i = c·r**i   (exponential
decay).  For each tried length n we minimise the score over r using a
golden‑section line‑search on the transformed variable x = n·(1‑r).  The
resulting sequence is normalised so that sum(a)=sqrt(2·n), making the
objective equal to the maximal auto‑convolution entry.
Finally a short simulated‑annealing hill‑climber refines the best
candidate.  The routine stops before the supplied time budget (default
1000 s) and returns a plain Python list ready for evaluate_sequence().
"""

import math
import time
import numpy as np


# ----------------------------------------------------------------------
# Basic utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(seq: np.ndarray) -> float:
    """Maximum entry of the full linear convolution seq * seq."""
    # Using FFT is faster for lengths > ~300, but np.convolve is fine for
    # the moderate sizes we use.  Switch automatically.
    n = len(seq)
    if n > 500:
        m = 1 << ((2 * n - 1).bit_length())
        fft = np.fft.rfft(seq, m)
        conv = np.fft.irfft(fft * fft, m)[: 2 * n - 1]
    else:
        conv = np.convolve(seq, seq)
    return float(np.max(conv))


def _score(seq: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` for a sequence already normalised
    to sum = sqrt(2·n).  In that case the denominator 2·n cancels and the
    score equals the maximal auto‑convolution entry.
    """
    n = len(seq)
    max_conv = _max_convolution(seq)
    # sum is sqrt(2·n) by construction, but we keep the formula for safety
    sum_val = np.sum(seq)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_val * sum_val))


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where the convolution attains (almost) the maximal value."""
    eps = max_val * 1e-12 + 1e-14
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the worst auto‑convolution entry w.r.t. f."""
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        vals = f[j]
        grad[i] += vals
        grad[j] += f[i]
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= f[i[mask]]
    return grad


# ----------------------------------------------------------------------
# Simple simulated‑annealing (polish step)
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.05 * target_sum / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-8:
            T = init_temp
        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score


# ----------------------------------------------------------------------
# Golden‑section search for the optimal exponential parameter
# ----------------------------------------------------------------------


def _golden_section(func, a, b, tol=1e-4, max_iter=40, deadline=None):
    """Univariate minimisation (maximise if func returns larger numbers)."""
    phi = (math.sqrt(5) - 1) / 2  # 0.618...
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)

    it = 0
    while it < max_iter and (deadline is None or time.monotonic() < deadline):
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
        if abs(b - a) < tol:
            break
        it += 1

    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence with a low score for `evaluate_sequence`.
    The routine tries an exponential family (a_i = c·r^i) for several
    lengths, optimising the decay r with a golden‑section line‑search.
    The best sequence is optionally polished with a tiny simulated‑annealing
    run.  All work stops before the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # finish a little early to avoid hard timeout
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Try exponential constructions for several lengths
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1500, 2000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # function that returns the score for a given scalar x = n·(1‑r)
        def score_for_x(x):
            if x <= 0.0:
                return float("inf")
            p = x / n               # p = 1‑r
            r = 1.0 - p
            if r <= 0.0:
                return float("inf")
            # unnormalised exponential vector
            vec = np.power(r, np.arange(n, dtype=np.float64))
            s = np.sum(vec)
            if s == 0.0:
                return float("inf")
            vec = vec * (target_sum / s)   # enforce Σa_i = sqrt(2n)
            return _score(vec)

        # Search interval for x.  Theory predicts optimum around 1.2–1.5
        lo, hi = 0.01, 5.0
        hi = min(hi, n * 0.999)   # 1‑r cannot exceed 1

        x_opt, x_score = _golden_section(
            score_for_x,
            lo,
            hi,
            tol=1e-4,
            max_iter=35,
            deadline=deadline,
        )

        # Build the corresponding sequence
        p_opt = x_opt / n
        r_opt = 1.0 - p_opt
        seq = np.power(r_opt, np.arange(n, dtype=np.float64))
        seq = seq * (target_sum / np.sum(seq))

        seq_score = _score(seq)

        if seq_score < best_score:
            best_score = seq_score
            best_seq = seq.copy()
            best_n = n
            print(
                f"[exp] n={n:4d}  x_opt={x_opt:.5f}  r_opt={r_opt:.6f}  score={seq_score:.6f}"
            )

        # early exit if we are already well below the target
        if best_score < 1.0:
            print("[early] reached score < 1.0, stop searching")
            break

    # ------------------------------------------------------------------
    # 2️⃣  Optional polishing with simulated annealing
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _simulated_annealing(best_seq, deadline,
                                                     init_temp=0.2,
                                                     decay=0.9997)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final‑anneal] improved score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] using uniform construction")

    # enforce clipping as required by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
```
