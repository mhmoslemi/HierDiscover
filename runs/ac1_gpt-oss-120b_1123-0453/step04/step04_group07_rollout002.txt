"""
Optimiser for the non‑negative step‑function bound.

Strategy
--------
1. **Exploratory Phase** – generate a diverse set of candidate sequences
   (parametric shapes, Dirichlet samples, uniform) for several lengths.
   Each candidate is scaled to the required ℓ¹‑norm  √(2 n) and its exact
   objective value (the same as `evaluate_sequence`) is computed.
   The best candidate is kept.

2. **Refinement Phase** – run a long projected Adam optimisation starting
   from the best candidate found above.  Adam is applied on the ℓ¹‑simplex
   (the mass is always projected back to the required sum) and uses the
   sub‑gradient of the maximal auto‑convolution entry.

3. **Polishing Phase** – a temperature‑controlled simulated‑annealing
   hill‑climber moves a small amount of mass between two coordinates while
   preserving the ℓ¹‑norm; uphill moves are occasionally accepted.

The routine respects a user‑supplied time budget (default 1000 s) and
returns the best list of floats discovered before the deadline.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # work on a copy
    v = v.copy()
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # return uniform point
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence)
# ----------------------------------------------------------------------
def _score(f: np.ndarray) -> float:
    """Return the value of the evaluation function for f."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


# ----------------------------------------------------------------------
# Helper for sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where the convolution attains (almost) the maximal value."""
    eps = max_val * 1e-12 + 1e-14
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (f*f)[k] w.r.t. f.
    Summed over all active shifts.
    """
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        # derivative w.r.t. f[i] is f[j]
        grad[i] += f[j]
        # derivative w.r.t. f[j] is f[i]
        grad[j] += f[i]
        # diagonal terms (i == j) have been added twice; correct:
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= f[i[mask]]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser on the ℓ¹‑simplex
# ----------------------------------------------------------------------
def _adam_descent(f0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Minimise the worst auto‑convolution entry with projected Adam.
    The iterate is always projected onto the simplex Σ a_i = target_sum,
    a_i ≥ 0.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    it = 0
    no_improve = 0

    while it < max_iters and time.monotonic() < deadline:
        it += 1

        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)
        grad_max = _gradient_submax(f, active)

        # scale for the objective 2·n·max/Σ²
        sum_f = np.sum(f)
        grad_obj = (2.0 * len(f) / (sum_f * sum_f)) * grad_max

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = _project_onto_simplex(f_candidate, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cur_score = _score(f)
        cand_score = _score(f_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_f = f_candidate.copy()
            no_improve = 0
            print(f"[Adam] it={it:5d} n={len(f)} score={cand_score:.6f}")

        if cand_score < cur_score:
            f = f_candidate
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.7
                no_improve = 0

        if no_improve > 800:          # early stop
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Simulated annealing hill‑climber (mass‑preserving)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled hill‑climber.
    One move transfers a small amount of mass from one index to another
    while keeping the ℓ¹‑norm exactly.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    # typical step size (5 % of the average mass)
    step_scale = 0.05 * target_sum / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        if it % 256 == 0:                     # occasionally re‑project
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:7d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-8:
            T = init_temp

        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.
    The routine obeys the given time budget and returns a plain Python
    list suitable for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # safety margin

    best_seq = None
    best_score = float("inf")

    def _maybe_update(seq: np.ndarray, sc: float):
        nonlocal best_seq, best_score
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Update] n={len(seq)} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Quick exploratory search over several lengths
    # ------------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # uniform benchmark
        seq = np.full(n, target_sum / n)
        _maybe_update(seq, _score(seq))

        # ----- parametric families ------------------------------------
        i = np.arange(n)
        centre = (n - 1) / 2.0
        distances = np.minimum(i, n - 1 - i)

        # exponential decay from centre (|i‑centre|)
        for c in np.logspace(-5, 1, num=30):
            if time.monotonic() >= deadline:
                break
            vals = np.exp(-c * np.abs(i - centre))
            if vals.sum() == 0:
                continue
            seq = vals * (target_sum / vals.sum())
            _maybe_update(seq, _score(seq))

        # Gaussian‑like (quadratic exponent)
        for c in np.logspace(-5, 1, num=30):
            if time.monotonic() >= deadline:
                break
            vals = np.exp(-c * ((i - centre) ** 2))
            if vals.sum() == 0:
                continue
            seq = vals * (target_sum / vals.sum())
            _maybe_update(seq, _score(seq))

        # U‑shaped (large at edges, small in centre)
        for c in np.logspace(-5, 1, num=30):
            if time.monotonic() >= deadline:
                break
            vals = np.exp(-c * distances)
            if vals.sum() == 0:
                continue
            seq = vals * (target_sum / vals.sum())
            _maybe_update(seq, _score(seq))

        # triangular tent with controllable half‑width
        for half_w in np.linspace(0.05 * n, n, num=20):
            if time.monotonic() >= deadline:
                break
            d = np.abs(i - centre)
            vals = np.maximum(0.0, 1.0 - d / half_w)
            if vals.sum() == 0:
                continue
            seq = vals * (target_sum / vals.sum())
            _maybe_update(seq, _score(seq))

        # ----- random Dirichlet draws ---------------------------------
        alphas = [0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0]
        draws_per_alpha = 30
        for alpha in alphas:
            if time.monotonic() >= deadline:
                break
            for _ in range(draws_per_alpha):
                if time.monotonic() >= deadline:
                    break
                vals = rng.gamma(shape=alpha, scale=1.0, size=n)
                if vals.sum() == 0:
                    continue
                seq = vals * (target_sum / vals.sum())
                _maybe_update(seq, _score(seq))

    # ------------------------------------------------------------------
    # 2️⃣  Long Adam refinement on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15.0:          # allocate a few seconds for refinement
            n = len(best_seq)
            target_sum = math.sqrt(2.0 * n)
            # decide a generous iteration budget based on remaining time
            max_iters = int(2e5 * (remaining / 300.0))
            max_iters = max(2000, max_iters)
            sub_deadline = time.monotonic() + remaining - 2.0
            f_opt, sc_opt = _adam_descent(best_seq,
                                          target_sum,
                                          sub_deadline,
                                          max_iters=max_iters)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = f_opt.copy()
                print(f"[Adam‑refine] n={n} score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final simulated‑annealing polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_sc = _simulated_annealing(best_seq, deadline,
                                                  init_temp=0.3,
                                                  decay=0.9997)
        if refined_sc < best_score:
            best_score = refined_sc
            best_seq = refined.copy()
            print(f"[Anneal‑final] score={refined_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[Fallback] used uniform construction")

    # Clip to the hard bounds of evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[Result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
