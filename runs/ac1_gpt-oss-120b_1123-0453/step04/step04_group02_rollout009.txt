def _project_simplex(v, target_sum):
    """Euclidean projection of vector v onto the simplex {x >=0, sum = target_sum}."""
    # Using algorithm from Duchi et al.
    # If sum(v) == target_sum and v >=0, just return v.
    # else compute.
    # But to be safe for numeric issues.

Pseudo:

    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(n) + 1
    # find rho = max{i | u_i > cssv_i / i}
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.2)

    best_seq = None
    best_score = np.inf

    # length list
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500, 1800]  # maybe more up to 2000.

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # generate param families
        cand_vec, cand_sc = _candidate_families(n, target)
        if cand_sc < best_score:
            best_seq, best_score = cand_vec.copy(), cand_sc
            print(f"[family] n={n} score={best_score:.6f}")

        # cheap random Dirichlet
        # we might sample maybe upto certain time
        r_dead = time.monotonic() + min(0.5, (deadline - time.monotonic()) / 2)
        while time.monotonic() < r_dead:
            a = rng.random(n)
            a = a / a.sum() * target
            sc = _fast_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[rand] n={n} improved to {best_score:.6f}")

        # If we have a candidate for this n, apply smooth optimizer
        if best_seq is not None and best_seq.shape[0] == n:
            seq_opt, sc_opt = _smooth_optimize(best_seq, target, deadline)
            if sc_opt < best_score:
                best_seq, best_score = seq_opt, sc_opt
                print(f"[smooth] n={n} refined to {best_score:.6f}")

    # After all lengths, maybe further polishing with local search (as earlier)
    if best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_sc = _local_search(best_seq, deadline, step_initial=0.1)
        if polished_sc < best_score:
            best_seq, best_score = polished_seq, polished_sc
            print(f"[final] polished to {best_score:.6f}")

    # fallback
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
        best_score = _fast_score(best_seq)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    # convert to list of floats.
    return [float(v) for v in best_seq]

"""
A fast, multi‑stage optimizer for the “min‑max auto‑convolution” problem.

Stages
-------
1️⃣  Parametric sweep  – tries many classic windows (exponential,
    power‑law, Hann, Hamming, Blackman, sine, triangular, Gaussian)
    and also DPSS tapers (if SciPy is available).  The best of these
    becomes the global incumbent.

2️⃣  Smooth‑max gradient descent – replaces the crude sub‑gradient on the
    active convolution entry with a differentiable approximation of the
    ‑max‑operator (log‑sum‑exp).  The gradient of this smooth objective
    is obtained analytically via a second FFT, giving a cheap, global
    descent direction.  After each step the iterate is projected onto
    the simplex {x≥0, Σx = √(2·n)} (the ℓ¹‑norm used by the original code).

3️⃣  Final stochastic polishing – a short Gaussian hill‑climb (identical
    to the one used previously) cleans up any remaining slack.

The whole routine respects the security checks of ``evaluate_sequence``,
runs for at most the supplied time budget (‑200 ms safety margin) and
returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution (full) using FFT
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of `seq` with itself using FFT."""
    n = seq.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (max‐convolution) – identical to evaluate_sequence
# when the ℓ¹‑norm is √(2·n)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score (max‑convolution) for a normalised sequence."""
    n = seq.shape[0]
    max_b = _conv_fft(seq).max()
    # sum(seq) == √(2·n) → the factor 2·n / sum² cancels out.
    return float(max_b)


# ----------------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x≥0, Σx = target}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with given sum."""
    v = np.asarray(v, dtype=np.float64)
    # Fast path
    if v.min() >= 0 and abs(v.sum() - target) < 1e-12:
        return v.copy()

    # Duchi et al. algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return the zero vector
        return np.maximum(v, 0.0)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) plus analytical gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(seq: np.ndarray, beta: float):
    """Smooth approximation of max(conv) and its gradient w.r.t. `seq`."""
    n = len(seq)
    conv = _conv_fft(seq)                      # length 2n‑1
    maxc = conv.max()
    # log‑sum‑exp in a numerically stable way
    exp_vals = np.exp(beta * (conv - maxc))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp                     # soft‑max weights, sum = 1

    # Gradient of the smooth max:
    #   ∂/∂a_k   Σ_i w_i * ∂conv_i/∂a_k = 2 Σ_i w_i a_{i‑k}
    a_rev = seq[::-1]
    N = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a_rev, n=N)
    conv_full = np.fft.irfft(W * A, n=N)[: len(w) + n - 1]
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]   # slice of length n

    # smooth value = maxc + log(sum_exp)/beta
    smooth_val = maxc + np.log(sum_exp) / beta
    return smooth_val, grad


# ----------------------------------------------------------------------
# Parametric families – exhaustive sweep for a given length
# ----------------------------------------------------------------------
def _candidate_families(n: int, target: float):
    """Return the best sequence among many hand‑crafted families."""
    best_vec = None
    best_sc = float("inf")
    idx = np.arange(n, dtype=np.float64)

    def _norm(v):
        v = np.asarray(v, dtype=np.float64)
        if v.sum() == 0:
            return np.full_like(v, target / n)
        return v / v.sum() * target

    # Uniform (baseline)
    a = np.full(n, target / n, dtype=np.float64)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # ----- exponential decay ------------------------------------------------
    alphas = np.concatenate([
        np.linspace(0.0005, 0.005, 25),
        np.linspace(0.005, 0.05, 40),
        np.linspace(0.05, 0.3, 30),
        np.linspace(0.3, 1.0, 20)
    ])
    alphas = np.unique(alphas)
    for alpha in alphas:
        a = np.exp(-alpha * idx)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # ----- power‑law --------------------------------------------------------
    p_vals = np.linspace(0.05, 5.0, 120)
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        a = base ** (-p)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # ----- classic windows --------------------------------------------------
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))               # Hann
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))               # Hamming
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1)))                # Blackman
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    a = np.sin(np.pi * (idx + 1) / (n + 1))                             # Sine window
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    a = np.minimum(idx + 1, n - idx)                                   # Triangular
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # ----- Gaussian (left‑handed, decreasing) -------------------------------
    sigmas = np.linspace(0.2, 0.8, 12) * n
    for sigma in sigmas:
        a = np.exp(- (idx ** 2) / (2.0 * sigma ** 2))
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # ----- DPSS tapers (if SciPy is present) --------------------------------
    try:
        from scipy.signal import windows
        for NW in (2, 2.5, 3, 3.5, 4):
            a = windows.dpss(n, NW, Kmax=1, sym=False)
            a = _norm(a)
            sc = _fast_score(a)
            if sc < best_sc:
                best_sc, best_vec = sc, a.copy()
    except Exception:
        pass  # SciPy unavailable – silently ignore

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Smooth‑max gradient descent with simplex projection
# ----------------------------------------------------------------------
def _smooth_optimize(seq_init: np.ndarray, target_sum: float, deadline: float):
    """
    Perform a few phases of smooth‑max descent.
    Returns the best sequence encountered and its (exact) score.
    """
    seq = seq_init.copy()
    best_seq = seq.copy()
    best_sc = _fast_score(seq)
    rng = np.random.default_rng()
    # A schedule of β – start smooth, finish tight
    betas = [5.0, 12.0, 30.0, 60.0, 120.0]

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        step = target_sum * 0.1          # heuristic initial step size

        while time.monotonic() < deadline:
            smooth_val, grad = _smooth_obj_and_grad(seq, beta)

            # descent direction (negative gradient)
            direction = -grad

            # propose a new point and project onto the ℓ¹‑simplex
            cand = seq + step * direction
            cand = _project_simplex(cand, target_sum)

            cand_sc = _fast_score(cand)

            if cand_sc < best_sc - 1e-9:
                # improvement – keep it and become slightly bolder
                best_seq, best_sc = cand.copy(), cand_sc
                seq = cand
                step = min(step * 1.15, target_sum)   # grow slowly
                continue
            else:
                # no improvement – shrink step
                step *= 0.5
                if step < 1e-12:
                    # occasional random shake‑up to escape flat regions
                    noise = rng.normal(scale=target_sum * 0.02, size=seq.shape)
                    cand = _project_simplex(seq + noise, target_sum)
                    cand_sc = _fast_score(cand)
                    if cand_sc < best_sc:
                        seq = cand
                        best_seq, best_sc = cand.copy(), cand_sc
                    step = target_sum * 0.1

            # early exit if we already beat the target bound
            if best_sc < 1.503:
                return best_seq, best_sc

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Tiny Gaussian hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray, deadline: float,
                  step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """Very short stochastic refinement using isotropic Gaussian perturbations."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑normalise to the required ℓ¹‑norm
        target = math.sqrt(2.0 * n)
        cand = cand * (target / cand.sum())
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        # slowly shrink the exploration radius
        step *= 0.9997
        if step < 1e-8:
            step = step_initial
    return best, best_sc
