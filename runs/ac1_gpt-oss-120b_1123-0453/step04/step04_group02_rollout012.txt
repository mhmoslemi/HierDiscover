def subgradient(seq):
    conv = _conv_fft(seq)
    max_val = conv.max()
    eps = max_val * 0.02  # consider indices within 2% of max.
    active = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(seq)
    n = len(seq)
    for k in active:
        # For each k, reversed slice as earlier
        if k < n:
            grad[: k+1] += seq[k::-1]   # reversed slice
        else:
            off = k - (n - 1)
            grad[off:] += seq[:n-off][::-1]
    # The gradient of max is subgradient, but we sum contributions; direction should be negative of this sum.
    # Ensure sum constraint: projection onto hyperplane sum=0.
    grad -= grad.mean()
    return grad

"""
Optimizing a non‑negative step function for the evaluation metric

The objective is
    2·n·max(conv(a,a)) / (sum(a))²
where `conv(a,a)` is the *full* linear convolution of the sequence with itself.
The optimum is invariant under scaling of `a`, therefore we keep the
ℓ¹‑norm fixed to √(2·n) (which makes the objective equal to the maximal
convolution entry).  The search proceeds in three stages:

1️⃣ **Broad sampling** – for a variety of lengths (up to 5000) we generate
   many candidate families (exponential, power‑law, Gaussian, cosine‑power,
   binomial‑coefficients) together with cheap Dirichlet draws.  The best few
   candidates are retained.

2️⃣ **Momentum sub‑gradient descent** – starting from each retained candidate
   we repeatedly identify the convolution index where the maximum is attained,
   build a sub‑gradient (the reversed‑slice of the sequence involved in that
   convolution), apply a momentum update while keeping the ℓ¹‑norm exactly
   constant, and adapt the step size.  Random “shake‑ups’’ are performed when
   the step becomes too small.

3️⃣ **Final stochastic polishing** – a short isotropic Gaussian hill‑climb
   finishes the optimisation.

All operations respect the security checks performed by ``evaluate_sequence``:
non‑negative entries, clipping to [0,1000] and a fixed ℓ¹‑norm.
The function returns the best sequence discovered before the time budget
expires (the harness provides ``evaluate_sequence`` for the final score).
"""

import time
import math
import numpy as np
from scipy.special import gammaln

# ----------------------------------------------------------------------
# Fast convolution via FFT (full linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``seq`` with itself using FFT."""
    n = seq.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Exact objective value without the security wrapper."""
    n = seq.shape[0]
    max_b = _conv_fft(seq).max()
    sum_a = seq.sum()
    # The security wrapper rejects sums <0.01 – we discard such degenerate cases.
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Helper: normalise a raw vector to the required ℓ¹‑norm √(2·n)
# ----------------------------------------------------------------------
def _normalise(v: np.ndarray, target_sum: float) -> np.ndarray:
    v = np.asarray(v, dtype=np.float64)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, target_sum / v.size)
    return v * (target_sum / total)


# ----------------------------------------------------------------------
# Parametric families (used both for systematic sweep and random draws)
# ----------------------------------------------------------------------
def _family_uniform(n: int, target: float) -> np.ndarray:
    return np.full(n, target / n, dtype=np.float64)


def _family_exponential(n: int, target: float, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * idx)
    return _normalise(a, target)


def _family_powerlaw(n: int, target: float, p: float) -> np.ndarray:
    base = np.arange(1, n + 1, dtype=np.float64)
    a = base ** (-p)
    return _normalise(a, target)


def _family_gaussian(n: int, target: float, sigma: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    ctr = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx - ctr) / sigma) ** 2)
    return _normalise(a, target)


def _family_cosine_power(n: int, target: float, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))   # Hann shape
    a = w ** power
    return _normalise(a, target)


def _family_binomial(n: int, target: float) -> np.ndarray:
    # p_i ∝ C(n‑1,i)  (simple symmetric binomial distribution)
    k = np.arange(n, dtype=np.float64)
    logp = gammaln(n) - gammaln(k + 1.0) - gammaln(n - k)
    logp -= np.max(logp)                     # stability
    p = np.exp(logp)
    return _normalise(p, target)


# ----------------------------------------------------------------------
# Random candidate generation for a given length
# ----------------------------------------------------------------------
def _random_candidate(rng: np.random.Generator, n: int, target: float) -> np.ndarray:
    # Dirichlet‑like draw (Gamma(1,1) i.i.d. → uniform on the simplex)
    a = rng.random(n)
    return _normalise(a, target)


# ----------------------------------------------------------------------
# Gradient descent on the active maximal convolution entry (with momentum)
# ----------------------------------------------------------------------
def _gradient_descent(seq: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      momentum_beta: float = 0.85,
                      epsilon_rel: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent with momentum.
    The sub‑gradient is built from all convolution entries that lie
    within ``epsilon_rel`` relative distance of the current maximum.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    # momentum vector
    m_vec = np.zeros_like(seq)
    step = 0.2
    rng = np.random.default_rng()

    while time.monotonic() < deadline:
        conv = _conv_fft(seq)
        max_val = conv.max()
        # all indices that are “almost maximal”
        thresh = max_val - epsilon_rel * max_val
        active = np.where(conv >= thresh)[0]

        # build (averaged) sub‑gradient of the active entries
        grad = np.zeros_like(seq)
        for k in active:
            if k < n:                                 # overlap starts at 0
                grad[: k + 1] += seq[k::-1]
            else:
                off = k - (n - 1)
                grad[off:] += seq[: n - off][::-1]

        grad /= max(1, active.size)       # average over the active set
        # projection that keeps the ℓ¹‑norm unchanged (remove mean component)
        grad -= grad.mean()

        # momentum update
        m_vec = momentum_beta * m_vec - (1 - momentum_beta) * grad
        cand = seq + step * m_vec
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total == 0.0:
            cand = np.full_like(seq, target_sum / n)
        else:
            cand = cand * (target_sum / total)

        cand_sc = _fast_score(cand)

        if cand_sc < best_sc:
            best_sc, best_seq = cand_sc, cand.copy()
            seq = cand
            step = min(step * 1.05, 2.0)        # gentle growth after success
            # reset momentum to avoid drift
            m_vec.fill(0.0)
        else:
            step *= 0.5
            if step < 1e-7:
                # random shake‑up – move a small fraction from one index to another
                i, j = rng.integers(n, size=2)
                delta = seq[i] * 0.05
                if delta > 0:
                    cand = seq.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand = cand * (target_sum / cand.sum())
                    cand_sc = _fast_score(cand)
                    if cand_sc < best_sc:
                        best_sc, best_seq = cand_sc, cand.copy()
                        seq = cand
                step = 0.2
                m_vec.fill(0.0)

        # early exit if we already beat the target bound
        if best_sc < 1.503:
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Tiny Gaussian hill‑climb (final polish)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """Isotropic Gaussian perturbations with renormalisation."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        target = math.sqrt(2.0 * n)
        cand = cand * (target / cand.sum())
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9997
        if step < 1e-7:
            step = step_initial
    return best, best_sc

        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[family] n={n} uniform new best {best_score:.6f}")

        # Exponential – sweep a few α values
        for α in np.linspace(0.001, 5.0 / n, 15):
            vec = _family_exponential(n, target, α)
            sc = _fast_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[exp] n={n} α={α:.6f} new best {best_score:.6f}")

        # Power‑law
        for p in np.linspace(0.2, 3.0, 15):
            vec = _family_powerlaw(n, target, p)
            sc = _fast_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[pow] n={n} p={p:.3f} new best {best_score:.6f}")

        # Gaussian widths
        for sigma in np.linspace(n / 10.0, n / 2.0, 12):
            vec = _family_gaussian(n, target, sigma)
            sc = _fast_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[gauss] n={n} σ={sigma:.1f} new best {best_score:.6f}")

        # Cosine‑power
        for pw in np.linspace(0.5, 3.0, 12):
            vec = _family_cosine_power(n, target, pw)
            sc = _fast_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[cos] n={n} power={pw:.2f} new best {best_score:.6f}")

        # Binomial shape
        vec = _family_binomial(n, target)
        sc = _fast_score(vec)
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[binom] n={n} new best {best_score:.6f}")

        # ---- cheap random Dirichlet samples (time‑bounded) ------------------------
        rnd_deadline = time.monotonic() + min(0.3,
                                              (deadline - time.monotonic()) / 4)
        while time.monotonic() < rnd_deadline:
            vec = _random_candidate(rng, n, target)
            sc = _fast_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[rand] n={n} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Keep the best few candidates for a more expensive local optimisation
    # ------------------------------------------------------------------
    # store top_k candidates (including the global best found so far)
    if best_seq is not None:
        top_candidates.append((best_score, best_seq.copy()))
    # note: additional good candidates could be gathered during the sweep;
    # for simplicity we keep only the global incumbent.

    # ------------------------------------------------------------------
    # 3️⃣  Momentum sub‑gradient descent on the retained candidates
    # ------------------------------------------------------------------
    for rank, (sc, cand) in enumerate(sorted(top_candidates, key=lambda x: x[0])):
        if time.monotonic() >= deadline:
            break
        n = cand.shape[0]
        target = math.sqrt(2.0 * n)
        print(f"[grad‑start] candidate #{rank+1} n={n} score={sc:.6f}")
        refined_seq, refined_sc = _gradient_descent(cand, target, deadline)
        if refined_sc < best_score:
            best_seq, best_score = refined_seq.copy(), refined_sc
            print(f"[grad] refined candidate #{rank+1} to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final stochastic polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_sc = _local_search(best_seq, deadline,
                                                  step_initial=0.2)
        if polished_sc < best_score:
            best_seq, best_score = polished_seq, polished_sc
            print(f"[final] polished to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    return [float(v) for v in best_seq]
