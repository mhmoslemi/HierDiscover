def adam_optimize(z0, n, S, beta_schedule, steps, learning_rate, max_time, deadline):
    m = np.zeros_like(z0)
    v = np.zeros_like(z0)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = None
    best_val = inf
    for step in range(steps):
        if time.monotonic() >= deadline:
            break
        beta = beta_schedule(min(step // schedule_len, len(beta_schedule)-1))
        smooth, grad = _obj_and_grad(z0, n, S, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        z0 = z0 - learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        # maybe shift to keep scale stable?
        if (step+1) % eval_every == 0:
            a = _softmax_to_a(z0, S)
            a = np.clip(a, 0.0, 1000.0)
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[adam] n={n} step={step} val={val:.6f}")
            if best_val < target:
                break
    return best_a, best_val

"""
Proposes a low‑value non‑negative step sequence.

The optimiser works directly on the ℓ¹‑simplex via a soft‑max
parameterisation.  A smooth log‑sum‑exp surrogate of the
max‑convolution is minimised with a hand‑rolled Adam optimiser,
while the “smoothness” (β) is increased in a continuation
schedule.  Multiple random restarts and interpolation warm‑starts
from the best incumbent are used.  Finally a short Gaussian
hill‑climbing polish refines the best candidate.

The code respects a global time budget (≈ budget_s seconds) and
returns a plain Python list ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Convert unconstrained logits `z` into a simplex point `a`
    (non‑negative, sum = target_sum) via the soft‑max trick.
    """
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z) * target_sum
    return a


def _obj_and_grad(z: np.ndarray, n: int, S: float, beta: float):
    """
    Smooth surrogate of the max‑convolution and its gradient
    w.r.t. the soft‑max logits `z`.

    Returns
    -------
    smooth : float
        (1/β)·log Σ exp(β·conv)  (smooth max)
    grad_z : np.ndarray (shape (n,))
        Gradient of `smooth` w.r.t. `z`.
    """
    # a from soft‑max (shape (n,))
    a = _softmax_to_a(z, S)

    # ---- self‑convolution via FFT ---------------------------------
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]   # length 2n‑1, real

    # ---- smooth max (log‑sum‑exp) ---------------------------------
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))          # stable
    sum_exps = np.sum(exps)
    smooth = max_c + np.log(sum_exps) / beta
    w = exps / sum_exps                            # softmax weights, Σw = 1

    # ---- gradient w.r.t. a : g_a = 2·(a_rev * w)_{n … n+n‑1} -------
    a_rev = a[::-1]
    L2 = _next_pow2(3 * n - 2)
    fr = np.fft.rfft(a_rev, L2)
    fw = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(fr * fw, L2)             # length L2
    start = n - 1
    g_raw = conv2[start : start + n]             # length n
    g_a = 2.0 * g_raw

    # ---- chain rule to obtain grad_z -------------------------------
    # dΦ/dz_i = a_i·g_a_i - a_i·(a·g_a)/S
    total = np.dot(g_a, a)
    grad_z = a * g_a - a * (total / S)

    return float(smooth), grad_z.astype(np.float64)


def _adam_optimize(z0: np.ndarray,
                   n: int,
                   S: float,
                   beta_schedule,
                   inner_iters: int,
                   lr: float,
                   deadline: float):
    """
    Simple Adam optimiser for the smooth surrogate.
    Returns the best a‑vector seen (as ndarray) and its
    exact `evaluate_sequence` value.
    """
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = None
    best_val = float("inf")
    t = 0

    # evaluate once at the start (occasionally we may never hit eval step)
    a_start = _softmax_to_a(z, S)
    a_start = np.clip(a_start, 0.0, 1000.0)
    val_start = evaluate_sequence(a_start.tolist())
    if val_start < best_val:
        best_val = val_start
        best_a = a_start.copy()

    # main loop
    max_outer = len(beta_schedule) * inner_iters
    while t < max_outer and time.monotonic() < deadline:
        beta = beta_schedule[min(t // inner_iters, len(beta_schedule) - 1)]
        smooth, grad = _obj_and_grad(z, n, S, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation
        if (t + 1) % max(1, inner_iters // 2) == 0:
            a = _softmax_to_a(z, S)
            a = np.clip(a, 0.0, 1000.0)
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[adam] n={n} step={t} β={beta:.0f} val={val:.6f}")
            if best_val < 1.503:
                break

        t += 1

    return best_a, best_val


def _interpolate_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Resize a sequence by linear interpolation (preserves shape)."""
    old_len = seq.size
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _optimise_length_adam(n: int,
                          target_sum: float,
                          deadline: float,
                          rng: np.random.Generator,
                          warm_seq: np.ndarray | None):
    """
    Runs a few Adam optimisations (different restarts) for length `n`.
    Returns the best sequence found and its exact `evaluate_sequence` score.
    """
    max_restarts = 3
    best_seq = None
    best_val = float("inf")

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        if warm_seq is not None and warm_seq.size == n and restart == 0:
            # use warm start (log‑softmax inverse, safe from zeros)
            p = np.clip(warm_seq / target_sum, 1e-12, None)
            z0 = np.log(p)
        else:
            # random gaussian logits
            z0 = rng.normal(scale=0.1, size=n)

        # continuation on β (increasing smoothness)
        beta_sched = [5.0, 15.0, 40.0, 80.0, 200.0, 500.0, 1000.0]
        inner = 200                     # iterations per β‑stage
        lr = 0.2                         # learning‑rate for Adam

        seq, val = _adam_optimize(z0, n, target_sum,
                                  beta_sched, inner, lr, deadline)

        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[optim] n={n} restart={restart} val={val:.6f}")

        if best_val < 1.503:
            break

    # fallback – uniform distribution
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())

    return best_seq, best_val


def _stochastic_refinement(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing (keeps the ℓ¹‑constraint via soft‑max).
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = evaluate_sequence(best.tolist())
    step = step_initial
    S = best.sum()

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑project onto the simplex (identical to the soft‑max map)
        p = np.clip(cand / S, 1e-12, None)
        cand = _softmax_to_a(np.log(p), S)

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # exponential cooling of step size
        step *= 0.9995
        if step < 1e-8:
            step = step_initial

    return best, best_score

        best_score = evaluate_sequence(best_seq.tolist())

    print(f"[result] best score = {best_score:.6f} (length={best_seq.size})")
    return [float(x) for x in best_seq]
