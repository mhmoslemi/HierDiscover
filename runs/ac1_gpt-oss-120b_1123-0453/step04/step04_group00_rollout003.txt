* The best sequence (lowest true max‑convolution) seen during the run
  is kept globally and returned at the end of the allotted time.
* Simple random jitter and a few random restarts are added to escape
  plateaus.

The implementation uses only NumPy and SciPy (HiGHS linear‑programming
solver) and respects the 1000 s budget.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------
_EPS = 1e-12      # avoid exact zeros (LP would become unbounded)


def _target_l1_norm(n: int) -> float:
    """Desired ℓ¹‑norm for a sequence of length n."""
    return math.sqrt(2.0 * n)


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip, enforce a tiny positive floor and rescale to the target ℓ¹‑norm."""
    n = seq.shape[0]
    seq = np.clip(seq, 0.0, 1000.0)
    seq = np.maximum(seq, _EPS)
    total = seq.sum()
    target = _target_l1_norm(n)
    if total <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / total)
    return seq


def _conv_max(seq: np.ndarray) -> float:
    """Exact max‑value of the linear convolution seq*seq."""
    return float(np.convolve(seq, seq).max())


def _build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix A such that (A @ b) = a * b (linear convolution).
    Entry A[i, j] = a[j] if j ≤ i ≤ j+n‑1 else 0.
    """
    n = a.shape[0]
    rows = np.arange(2 * n - 1)[:, None]          # (2n‑1, 1)
    cols = np.arange(n)[None, :]                  # (1, n)
    mask = (rows >= cols) & (rows <= cols + n - 1)  # (2n‑1, n) bool
    A = np.where(mask, a[None, :], 0.0)            # broadcasting a over rows
    return A


def _solve_lp_max_sum(A: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ b_j   s.t.   A @ b ≤ M·1 ,  b ≥ 0.
    Returns the optimal b (as a dense vector) or None on failure.
    """
    n = A.shape[1]
    # linprog minimises c·x, so use c = –1 (maximise sum)
    c = -np.ones(n, dtype=np.float64)
    ub = np.full(A.shape[0], M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x


def _lp_optimize(initial: np.ndarray,
                 deadline: float,
                 rng: np.random.Generator,
                 max_iters: int = 80) -> tuple[np.ndarray, float]:
    """
    Alternating LP optimisation.
    Returns the best sequence encountered and its exact max‑convolution.
    """
    a = _normalize(initial.copy())
    n = a.shape[0]
    target = _target_l1_norm(n)

    best_seq = a.copy()
    best_score = _conv_max(a)

    # mixing factor τ – start relatively large and decay slowly
    tau = 0.5
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        M = _conv_max(a)

        # ---- linear programme ----
        A = _build_convolution_matrix(a)
        b = _solve_lp_max_sum(A, M)
        if b is None:
            # fallback: break out of the loop
            break

        # Scale b to the required ℓ¹‑norm
        b_sum = b.sum()
        if b_sum <= 0.0:
            b = np.full_like(b, target / n)
        else:
            b = b * (target / b_sum)

        # ---- mixing step ----
        a = (1.0 - tau) * a + tau * b
        a = _normalize(a)

        # occasional jitter to avoid stagnation
        if it % 15 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > _EPS:
                delta = a[i] * rng.uniform(0.01, 0.04)
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # decay τ gently (helps convergence)
        tau = max(0.05, tau * 0.97)

        cur_score = _conv_max(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation value.
    The approach combines cheap parametric seeds and an LP‑driven
    alternating optimisation (see module docstring).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to record a new global best (no closure – handled inline)
    # ------------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray) -> None:
        nonlocal best_seq, best_score
        sc = _conv_max(candidate)
        if sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"[global best] n={candidate.shape[0]} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – medium sizes tend to give the best trade‑off.
    # ------------------------------------------------------------------
    lengths = [800, 600, 500, 400, 300, 200, 150, 120, 100]

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from any previously discovered construction.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            cand = _normalize(cand)
            _maybe_update(cand)
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Sweep a few simple parametric families and improve each with LP.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = _target_l1_norm(n)
        idx = np.arange(1, n + 1, dtype=np.float64)

        # ----- power‑law seeds (a_i ∝ i^{‑p}) -----
        for p in np.linspace(0.5, 3.0, 9):
            if time.monotonic() >= deadline:
                break
            a = idx ** (-p)
            a = _normalize(a)
            seq_opt, _ = _lp_optimize(a, deadline, rng, max_iters=60)
            _maybe_update(seq_opt)

        # ----- exponential seeds (a_i ∝ exp(-α·i)) -----
        for alpha in np.linspace(0.001, 0.12, 9):
            if time.monotonic() >= deadline:
                break
            a = np.exp(-alpha * np.arange(n, dtype=np.float64))
            a = _normalize(a)
            seq_opt, _ = _lp_optimize(a, deadline, rng, max_iters=60)
            _maybe_update(seq_opt)

    # ------------------------------------------------------------------
    # 2️⃣ Random restarts – each run gets its own LP optimisation.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = _target_l1_norm(n)

        # Number of independent restarts (more for smaller n)
        restarts = 4 if n <= 500 else 2
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break
            # Dirichlet‑like random start
            start = rng.gamma(1.0, size=n)
            start = _normalize(start)

            # Blend with the current global best if dimensions match
            if best_seq is not None and best_seq.shape[0] == n:
                start = (start + best_seq) / 2.0
                start = _normalize(start)

            seq_opt, _ = _lp_optimize(start, deadline, rng, max_iters=80)
            _maybe_update(seq_opt)

            # early exit if we have already beaten the target value
            if best_score < 1.503:
                print("[early exit] target achieved")
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # 3️⃣ Final cheap hill‑climb – tiny random mass swaps.
    # ------------------------------------------------------------------
    if best_seq is not None:
        while time.monotonic() < deadline:
            n = best_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            delta = best_seq[i] * rng.uniform(0.01, 0.04)
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _normalize(cand)
            _maybe_update(cand)

    # ------------------------------------------------------------------
    # Fallback – a simple uniform vector if nothing else succeeded.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        best_seq = np.full(n, _target_l1_norm(n) / n, dtype=np.float64)
        best_score = _conv_max(best_seq)

    # Return as a plain Python list (required by the harness)
    return [float(x) for x in best_seq]
