"""
Propose a low‑score step sequence.

Algorithm
---------
1️⃣  Warm‑start with any previously known construction.
2️⃣  Sweep a few simple parametric families (power‑law, exponential,
    Gaussian‑like) for several lengths, keep the best.
3️⃣  For each length run a small number of *LP‑based* improvement steps.
    For a fixed sequence `f` we solve
        max   Σ b_j
        s.t.  b_j ≥ 0,
              (f * b)_k ≤ M   for all k,
    where M = max(f*f).  The LP gives a vector `b` with the largest
    ℓ¹‑norm that does not increase the worst auto‑convolution value.
    After scaling `b` to the required ℓ¹‑norm we mix it with `f`
    (convex combination) and repeat.  This often reduces the peak
    convolution markedly.
4️⃣  Perform a few random‑restart runs with the same LP scheme.
5️⃣  Finish with a cheap random‑mass‑swap hill‑climb.

All vectors are projected onto the feasible set
(non‑negative, entries ≤1000, ℓ¹‑norm = √(2n)).
The best sequence found before the time budget expires is returned
as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and enforce ℓ¹‑norm = √(2·n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / total)
    return seq


def _conv_max(seq: np.ndarray) -> float:
    """Return max entry of the linear self‑convolution of `seq`."""
    # For the moderate sizes we work with we can use np.convolve directly.
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _lp_improve(
    seq: np.ndarray,
    target: float,
    mix: float,
    iters: int,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Perform `iters` LP‑based improvement steps on `seq`.

    Parameters
    ----------
    seq : np.ndarray
        Starting non‑negative vector (already ℓ¹‑normalised).
    target : float
        Desired ℓ¹‑norm (√(2·n)).
    mix : float
        Convex‑combination weight t ∈ (0,1) used for f←(1‑t)f+t·g.
    iters : int
        Maximum number of LP iterations.
    deadline : float
        Time limit (monotonic time) after which we abort early.
    rng : np.random.Generator
        Random generator (used only for occasional jitter).

    Returns
    -------
    np.ndarray
        The improved sequence (still ℓ¹‑normalised).
    """
    a = seq.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1

    # Pre‑allocate the constraint matrix; we will overwrite its columns each iteration.
    A_ub = np.zeros((conv_len, n), dtype=np.float64)

    for it in range(iters):
        if time.monotonic() >= deadline:
            break

        # Current worst auto‑convolution value.
        M = _conv_max(a)

        # Build A_ub where row k, column j contains a_{k‑j} when valid, else 0.
        # This is equivalent to placing the vector a on the diagonal of each column.
        # Using slicing is much faster than a double loop.
        A_ub.fill(0.0)
        for j in range(n):
            A_ub[j : j + n, j] = a

        # Solve the LP: maximise Σ b_j  ⇔ minimise -Σ b_j
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(conv_len, M, dtype=np.float64)

        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=(0, None),
                method="highs",
                options={"presolve": True},
            )
        except Exception:  # pragma: no cover
            # If the solver crashes, give up on further LP steps.
            break

        if not res.success:
            # No feasible improvement – stop early.
            break

        b = res.x
        sum_b = b.sum()
        if sum_b > 0.0:
            g = b * (target / sum_b)
        else:
            g = np.full(n, target / n, dtype=np.float64)

        # Convex combination with the current iterate.
        a = (1.0 - mix) * a + mix * g

        # Clip to the allowed box and renormalise (clipping can disturb the ℓ¹‑norm).
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # Occasionally inject a tiny random jitter to escape tiny plateaus.
        if it % 3 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.01
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # Enforce symmetry – the objective is invariant under reversal and symmetry
        # often helps to keep the peak small.
        a = (a + a[::-1]) * 0.5

    return a


def _random_restart_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random non‑negative ℓ¹‑normalised vector of length n."""
    start = rng.gamma(1.0, size=n)  # Gamma(1,1) == Exp(1)
    target = math.sqrt(2.0 * n)
    if start.sum() > 0.0:
        start = start * (target / start.sum())
    else:
        start = np.full(n, target / n, dtype=np.float64)
    return start

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.

    The routine combines a warm‑start, a fast parametric sweep,
    several LP‑based refinement passes and a final cheap random‑mass‑swap
    hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None          # type: np.ndarray | None
    best_score = float("inf")  # evaluation value (max convolution after normalisation)

    # ------------------------------------------------------------------
    # Helper to update the global best
    # ------------------------------------------------------------------
    def _update_global(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = _conv_max(seq)  # because we keep the ℓ¹‑norm = √(2·n)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global best] n={len(seq)}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from any existing construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            cand = _normalize(cand)
            _update_global(cand)
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Parametric sweep (simple families)
    # ------------------------------------------------------------------
    lengths = [1200, 1000, 800, 600, 500, 400, 300, 200, 150, 100]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(1, n + 1, dtype=np.float64)

        # power‑law a_i ∝ i^{‑p}
        for p in np.linspace(0.3, 3.0, 12):
            a = idx ** (-p)
            a = a / a.sum() * target
            a = _normalize(a)
            _update_global(a)

        # exponential a_i ∝ exp(-α·i)
        for alpha in np.linspace(0.001, 0.12, 12):
            a = np.exp(-alpha * np.arange(n, dtype=np.float64))
            a = a / a.sum() * target
            a = _normalize(a)
            _update_global(a)

        # Gaussian‑like centred shape (symmetric)
        sigma_vals = np.linspace(0.1 * n, 0.4 * n, 5)
        centre = (n - 1) / 2.0
        for sigma in sigma_vals:
            a = np.exp(-0.5 * ((idx - 1 - centre) / sigma) ** 2)
            a = a / a.sum() * target
            a = _normalize(a)
            _update_global(a)

    # ------------------------------------------------------------------
    # 2️⃣ LP‑based refinement on the current global best for each length
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # If we already have a best of this length, try to improve it.
        if best_seq is not None and best_seq.shape[0] == n:
            seq = best_seq.copy()
        else:
            # otherwise start from a random Dirichlet vector
            seq = _random_restart_sequence(n, rng)

        # Do a modest number of LP steps (fast, cheap) – 8 iterations works well.
        seq_ref = _lp_improve(seq, target, mix=0.45, iters=8, deadline=deadline, rng=rng)
        _update_global(seq_ref)

    # ------------------------------------------------------------------
    # 3️⃣ Random restarts + deeper LP refinement
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # number of restarts scaled with length (shorter sequences are cheaper)
        restarts = 6 if n <= 600 else 3

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            start = _random_restart_sequence(n, rng)

            # Blend with global best of same dimension if available (helps convergence).
            if best_seq is not None and best_seq.shape[0] == n:
                start = (start + best_seq) * 0.5
                start = _normalize(start)

            # Deeper LP improvement (15 iterations, a slightly larger mixing weight).
            seq_imp = _lp_improve(
                start, target, mix=0.5, iters=15, deadline=deadline, rng=rng
            )
            _update_global(seq_imp)

            # Early exit if we already beat the target bound.
            if best_score < 1.503:
                print("[early exit] target achieved")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 4️⃣ Final cheap random‑mass‑swap hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None:
        while time.monotonic() < deadline:
            n = best_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            delta = best_seq[i] * 0.03
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _normalize(cand)
            _update_global(cand)
    else:
        # Fallback: uniform vector of moderate length.
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best_score = _conv_max(best_seq)

    # Return a plain Python list as required by the outer harness.
    return [float(x) for x in best_seq]
