def _subgradient_descent(a0, target_sum, deadline, max_iters=5000, patience=500):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _compute_score_from_vector(a)
    no_improve = 0
    for it in range(1, max_iters+1):
        if time.monotonic() >= deadline:
            break
        conv = np.convolve(a, a)
        max_val = np.max(conv)
        # active indices
        # tolerance: small relative to max_val
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]
        # compute gradient as subgradient of max
        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1)
            j = k - i
            grad[i] += 2.0 * a[j]
        # step size schedule
        eta = 0.5 / np.sqrt(it)   # 0.5 can adjust
        a_candidate = a - eta * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        cand_score = _compute_score_from_vector(a_candidate)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            print(f"[subgrad] n={n} it={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break
        a = a_candidate
    return best_a, best_score

    idx = np.arange(n)
    if symmetric:

    a = 1.0 / (dist.astype(float) ** p)
    return a

    g_i = 2·a_{k*‑i} (zero if the index is out of range).
    A diminishing step‑size and Euclidean projection onto the simplex
    (Σ a_i = 1, a_i ≥ 0) keep the iterates feasible.

3️⃣  A short Adam‑based refinement (smooth log‑sum‑exp surrogate) and a
    stochastic polishing phase finish the search.

The routine respects the global time budget (≈1000 s) and returns the
best sequence it has found.  All numbers are non‑negative, the caller
will later feed the result to ``evaluate_sequence`` which applies the
same clipping again.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import cvxpy as cp  # optional, not used in the current pipeline

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact value of the external ``evaluate_sequence`` (no clipping)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / beta
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    return w


def _weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """Gradient of Σ_k w_k·conv_k(a,a) w.r.t. `a`."""
    n = len(a)
    grad = np.zeros_like(a)
    for k, wk in enumerate(weights):
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = a[j]
        grad[i] += wk * 2.0 * vals
        # adjust diagonal terms that were added twice
        mask = i == j
        if np.any(mask):
            grad[i[mask]] -= wk * vals[mask]
    return grad


def _adam_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 20000,
    patience: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation of a smoothed max‑convolution.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = len(a)
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        beta = max_conv * 0.07 + 1e-12
        w = _softmax_weights(conv, beta)
        grad = _weighted_gradient(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cur_score = _compute_score_from_vector(a)
        cand_score = _compute_score_from_vector(a_candidate)

        if cand_score < cur_score:
            a = a_candidate

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            # occasional progress report
            print(f"[Adam] n={n} it={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score


def _stochastic_refinement(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.3,
    swap_prob: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Gaussian hill‑climbing + occasional random swaps, with simplex projection.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, len(best), size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = best + rng.normal(scale=step, size=best.shape)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = min(step_initial, step * 1.02)
        else:
            step *= 0.995

        if step < 1e-6:
            step = step_initial

    return best, best_score


def _generate_power_law(n: int, p: float, symmetric: bool = False) -> np.ndarray:
    """
    Return a vector proportional to (distance)^(-p), normalised to sum 1.
    If ``symmetric`` is True the distance is the distance to the nearest edge,
    producing a shape that is large at both ends and small in the middle.
    """
    idx = np.arange(n, dtype=float)
    if symmetric:
        dist = np.minimum(idx, n - 1 - idx) + 1.0
    else:
        dist = idx + 1.0
    a = 1.0 / np.power(dist, p)
    a = np.clip(a, 0.0, 1000.0)
    a /= np.sum(a)
    return a


def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 20000,
    patience: int = 800,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a) = max_k Σ_{i+j=k} a_i a_j .
    The gradient for an active index k is 2·a_{k‑i} (zero outside the range).
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        max_val = np.max(conv)
        # active set – all indices within a tiny tolerance of the max
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1)
            j = k - i
            grad[i] += 2.0 * a[j]          # sub‑gradient of the max‑term

        # diminishing step size
        eta = 0.5 / math.sqrt(it)

        a_candidate = a - eta * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cand_score = _compute_score_from_vector(a_candidate)
        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            print(f"[SubGrad] n={n} it={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

        a = a_candidate

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a tiny safety margin for the external runner
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously found construction (if any)
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        if np.sum(seq) == 0:
            seq = np.full_like(seq, 1.0 / len(seq))
        else:
            seq /= np.sum(seq)           # normalise to sum = 1
        score = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[warm] loaded height_sequence_1, n={len(seq)} score={score:.6f}")

    # --------------------------------------------------------------
    # Parameter sweep over a few simple families (inverse‑power laws)
    # --------------------------------------------------------------
    lengths = [200, 400, 600, 800, 1000, 1200, 1500, 1800, 2000, 2500]
    exponents = np.linspace(0.2, 2.0, 19)   # 0.2, 0.3, …, 2.0
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = 1.0

        # try a few deterministic families first
        for sym in (False, True):
            for p in exponents:
                if time.monotonic() >= deadline:
                    break
                a0 = _generate_power_law(n, p, symmetric=sym)
                sc = _compute_score_from_vector(a0)
                if sc < best_score:
                    best_score = sc
                    best_seq = a0.copy()
                    print(f"[grid] n={n:4d} sym={sym} p={p:4.2f} score={sc:.6f}")

        # ----------------------------------------------------------
        # Sub‑gradient refinement of the current best of this length
        # ----------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            # allocate a modest time slice for this length
            slice_deadline = min(deadline, time.monotonic() + 12.0)
            a_opt, sc_opt = _subgradient_descent(best_seq.copy(), target_sum,
                                                 slice_deadline, max_iters=20000,
                                                 patience=800)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[sgd]   n={n:4d} refined score={sc_opt:.6f}")

            # a short Adam polishing – sometimes rescues the sub‑gradient result
            if time.monotonic() < deadline:
                adam_deadline = min(deadline, time.monotonic() + 8.0)
                a_opt2, sc_opt2 = _adam_descent(a_opt.copy(), target_sum,
                                                adam_deadline,
                                                max_iters=15000, patience=300)
                if sc_opt2 < best_score:
                    best_score = sc_opt2
                    best_seq = a_opt2.copy()
                    print(f"[adam]  n={n:4d} refined score={sc_opt2:.6f}")

        # early exit if we already beat the target comfortably
        if best_score < 1.45:
            print("[early] reached a satisfactory bound")
            break

    # --------------------------------------------------------------
    # Final stochastic polishing (if any time left)
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – a modest uniform vector
        n = 800
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline,
                                                         step_initial=0.3)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # final safety clipping (the external evaluator does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best length={len(best_seq)} score={best_score:.6f}")
    return [float(x) for x in best_seq]
