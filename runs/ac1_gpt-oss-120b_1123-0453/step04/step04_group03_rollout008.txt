def max_convolution(a):
    # compute linear convolution via FFT
    n = len(a)
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)
    return np.max(conv[:2*n-1])

def _adam_descent_dynamic(a0, target_sum, deadline, max_iters=50000, patience=800):
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr = 0.05
    beta1=0.9
    beta2=0.999
    eps=1e-8
    # schedule params
    beta_factor_start=0.2
    beta_factor_end=0.01
    factor_decay = 0.9999  # each iteration
    cur_factor = beta_factor_start
    no_improve=0
    for it in range(1, max_iters+1):
        if time.monotonic() >= deadline:
            break
        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        # update cur_factor toward end
        cur_factor = max(beta_factor_end, cur_factor * factor_decay)
        beta = max_conv * cur_factor + 1e-12
        w = _softmax_weights(conv, beta)
        grad = _weighted_gradient(a, w)
        # adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1-beta1**it)
        v_hat = v / (1-beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        cur_score = _score(a)
        cand_score = _score(a_candidate)
        if cand_score < cur_score:
            a = a_candidate
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            print(f"[AdamDyn] n={len(a)} iter={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break
    return best_a, best_score

def _simulated_annealing(start, deadline, init_temp=0.1, cooling=0.9995):
    best = start.copy()
    best_score = _score(best)
    cur = best.copy()
    cur_score = best_score
    temp = init_temp
    rng = np.random.default_rng()
    while time.monotonic() < deadline:
        # propose move
        n = len(cur)
        cand = cur.copy()
        # choose move type
        if rng.random() < 0.3:
            # swap two values
            i,j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer delta between two positions
            i,j = rng.integers(0, n, size=2)
            # sample delta in range [-cand[i], cand[j]] ensuring nonnegativity after move
            delta = rng.normal(scale=0.05)  # adjust
            # but ensure after move nonnegative: adjust
            delta = np.clip(delta, -cand[i], cand[j])
            cand[i] -= delta
            cand[j] += delta
        # maybe also small Gaussian jitter for many entries:
        if rng.random() < 0.2:
            # add to many entries but preserve sum via projection
            cand += rng.normal(scale=0.01, size=n)
        # projection onto simplex to correct sum (target sum = sum(best) perhaps 1)
        target = np.sum(start)
        cand = _project_onto_simplex(cand, target)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        # acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[SA] n={len(start)} new best {best_score:.6f}")
        # cooling
        temp *= cooling
        # if temp too low, break
        if temp < 1e-6:
            temp = init_temp
    return best, best_score
