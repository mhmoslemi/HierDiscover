def _conv_linear(a, b):
    L = 1 << ((len(a)+len(b)-1).bit_length())
    A = np.fft.rfft(a, L)

   f = f0.copy()
   best_f = f.copy()
   best_score = _score(f)
   m = np.zeros_like(f)
   v = np.zeros_like(f)
   beta1=0.9; beta2=0.999; eps=1e-8
   it=0
   epsilon = epsilon_start
   no_improve = 0
   while it < max_iters and time.monotonic() < deadline:
       it+=1
       conv = _conv_linear(f, f)   # shape 2n-1
       max_val = np.max(conv)
       cur_score = max_val   # since sum fixed
       # compute weights
       max_c = max_val
       exps = np.exp((conv - max_c) / epsilon)
       weights = exps / np.sum(exps)
       # gradient
       grad = _gradient_from_softmax(f, weights)
       
       # add factor for objective scaling if needed: factor = 2*len(f) / (target_sum**2)
       # But constant factor doesn't affect direction; we maybe multiply by factor to get comparable magnitude to Adam step.

       # Adam update
       m = beta1 * m + (1 - beta1) * grad
       v = beta2 * v + (1 - beta2) * (grad**2)
       m_hat = m / (1 - beta1**it)
       v_hat = v / (1 - beta2**it)
       step = lr * m_hat / (np.sqrt(v_hat) + eps)
       f_candidate = f - step
       # projection onto simplex
       f_candidate = _project_onto_simplex(f_candidate, target_sum)
       f_candidate = np.clip(f_candidate, 0.0, 1000.0)

       # Compute candidate score
       conv_cand = _conv_linear(f_candidate, f_candidate)
       cand_score = np.max(conv_cand)

       if cand_score < best_score - 1e-12:
           best_score = cand_score

"""
Search for a step‑function coefficients minimizing `evaluate_sequence`.
The core is a smooth approximation of the maximal auto‑convolution
(max‑entry) using a soft‑max (log‑sum‑exp) surrogate.  An Adam
optimizer on the ℓ¹‑simplex (sum fixed) drives the surrogate down.
The temperature (smoothness) ε is annealed, turning the objective
gradually into the true max, which helps escaping poor local minima.
A short simulated‑annealing polishing phase finishes the search.
"""
import time
import math
import numpy as np

# ---------- Helper utilities ----------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s}.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # shift by max for numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # edge case – return a uniform distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_linear(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) convolution of two real vectors using FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    n = len(a) + len(b) - 1
    L = 1 << (n - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    conv = np.fft.irfft(A * B, L)
    return conv[:n]

def _max_convolution(f: np.ndarray) -> tuple[np.ndarray, float]:
    """Return (conv, max(conv))."""
    conv = _conv_linear(f, f)
    max_val = np.max(conv)
    return conv, max_val

def _gradient_from_softmax(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max objective.
    w must be a probability vector of length 2·len(f)-1.
    """
    rev_f = f[::-1]
    conv_rev = _conv_linear(rev_f, w)          # length 3·n‑2
    n = len(f)
    # slice n entries centred at index n‑1
    grad = 2.0 * conv_rev[n-1 : n-1 + n]
    return grad

def _score_from_max(max_val: float) -> float:
    """Score assuming the ℓ¹‑norm has been normalised to √(2·n)."""
    return float(max_val)

def _raw_score(f: np.ndarray) -> float:
    """Exact evaluation according to the supplied `evaluate_sequence`."""
    n = len(f)
    conv = _conv_linear(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))

# ---------- Smooth Adam optimiser ----------
def _smooth_adam(f0: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 max_iters: int = 8000,
                 lr: float = 0.12,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8,
                 epsilon_start: float = 0.5,
                 epsilon_decay: float = 0.999,
                 epsilon_min: float = 0.02,
                 early_stop: int = 1500) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising a soft‑max surrogate of the maximal
    auto‑convolution entry.  The surrogate smoothness ε is annealed.
    Returns the best vector seen and its (true) maximal convolution value.
    """
    f = f0.copy()
    n = len(f)

    # keep the best discovered point
    best_f = f.copy()
    best_score = _raw_score(f)

    # Adam state
    m = np.zeros_like(f)
    v = np.zeros_like(f)

    it = 0
    no_improve = 0
    epsilon = epsilon_start

    while it < max_iters and time.monotonic() < deadline:
        it += 1

        # full auto‑convolution (used for both objective and soft‑max)
        conv, max_val = _max_convolution(f)

        # soft‑max weights (stable formulation)
        shifted = conv - max_val               # max subtraction for stability
        weights = np.exp(shifted / epsilon)
        weights_sum = np.sum(weights)
        if weights_sum == 0.0:
            # numerical fallback – fall back to pure max‑gradient
            active = np.isclose(conv, max_val)
            grad = np.zeros_like(f)
            for k, act in enumerate(active):
                if act:
                    i_start = max(0, k - (n - 1))
                    i_end = min(k, n - 1)
                    i = np.arange(i_start, i_end + 1, dtype=np.int64)
                    j = k - i
                    grad[i] += f[j]
                    grad[j] += f[i]
                    mask = (i == j)
                    if np.any(mask):
                        grad[i[mask]] -= f[i[mask]]
        else:
            w = weights / weights_sum           # probability simplex
            grad = _gradient_from_softmax(f, w)

        # Adam step (minimisation → move against gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = _project_onto_simplex(f_candidate, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cand_score = _raw_score(f_candidate)

        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            best_f = f_candidate.copy()
            no_improve = 0
            print(f"[SmoothAdam] n={n} it={it:5d} eps={epsilon:.4f} score={cand_score:.6f}")
        else:
            no_improve += 1

        # Accept the candidate if it improves the true max‑convolution;
        # otherwise keep the old point (Adam already adapts the step size).
        cur_score = _raw_score(f)
        if cand_score < cur_score:
            f = f_candidate

        # Anneal the soft‑max temperature
        epsilon = max(epsilon_min, epsilon * epsilon_decay)

        # Simple adaptive learning‑rate reduction
        if no_improve > 600:
            lr *= 0.7
            no_improve = 0

        if no_improve > early_stop:
            break

    return best_f, best_score

# ---------- Simulated‑annealing polishing ----------
def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Tiny hill‑climber that moves a random amount of mass from one index
    to another while preserving the ℓ¹‑norm.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _raw_score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.05 * target_sum / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _raw_score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-8:
            T = init_temp

        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score

# ---------- Public entry point ----------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.
    The routine respects the given time budget and returns a plain
    Python list suitable for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # ------ warm‑start -------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64).copy()
        seq = np.clip(seq, 0.0, 1000.0)
        n0 = len(seq)
        target_sum0 = math.sqrt(2.0 * n0)
        if np.sum(seq) == 0.0:
            seq = np.full(n0, target_sum0 / n0)
        else:
            seq = seq * (target_sum0 / np.sum(seq))
        s = _raw_score(seq)
        best_seq = seq.copy()
        best_score = s
        print(f"[init] warm‑start from height_sequence_1, n={n0}, score={s:.6f}")

    # ------ lengths to explore -----------------------------------------
    lengths = [400, 500, 600, 800, 1000, 1200, 1500, 1800]
    # optionally add a few larger lengths if time permits
    if time.monotonic() < deadline - 20:
        lengths.extend([2200, 2500])

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of random restarts – a little more for the most promising n
        restarts = 12 if n <= 1000 else 8
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ----- initial point ---------------------------------------
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
                f0 += rng.normal(scale=0.001 * target_sum, size=n)
                f0 = _project_onto_simplex(np.clip(f0, 0.0, None), target_sum)
            else:
                f0 = rng.random(n)
                f0 = _project_onto_simplex(f0, target_sum)

            # ----- smooth Adam optimisation ----------------------------
            sub_deadline = min(deadline, time.monotonic() + 7.0)
            f_opt, score_opt = _smooth_adam(f0, target_sum, sub_deadline)

            if score_opt < best_score:
                best_score = score_opt
                best_seq = f_opt.copy()
                print(f"[update] n={n} restart={r} score={score_opt:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503 - 1e-6:
                print("[early] target reached")
                return [float(x) for x in best_seq]

    # ------ final polishing ---------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _simulated_annealing(best_seq, deadline,
                                                     init_temp=0.2,
                                                     decay=0.9997)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final] annealed score={refined_score:.6f}")

    # ------ fallback ----------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _raw_score(best_seq)
        print("[fallback] using uniform construction")

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]

        cur_score = _raw_score(f)
        if cand_score < cur_score:
            f = f_candidate

        # Anneal soft‑max temperature.
        epsilon = max(epsilon_min, epsilon * epsilon_decay)

        # Simple adaptive learning‑rate schedule.
        if no_improve > 600:
            lr *= 0.7
            no_improve = 0

        if no_improve > early_stop:
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (same as in the previous baseline)
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Hill‑climber that moves a random amount of mass between two indices
    while preserving the ℓ¹‑norm.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _raw_score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.05 * target_sum / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _raw_score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-8:
            T = init_temp

        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.
    Respects the given time budget and returns a plain Python list ready for
    `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start from any previously saved construction.
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64).copy()
        seq = np.clip(seq, 0.0, 1000.0)
        n0 = len(seq)
        target_sum0 = math.sqrt(2.0 * n0)
        if np.sum(seq) == 0.0:
            seq = np.full(n0, target_sum0 / n0)
        else:
            seq = seq * (target_sum0 / np.sum(seq))
        s = _raw_score(seq)
        best_seq = seq.copy()
        best_score = s
        print(f"[init] warm‑start from height_sequence_1, n={n0}, score={s:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Main optimisation – smooth‑max Adam on several lengths.
    # ------------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000, 1200, 1500, 1800]
    # Add a couple of larger lengths if time permits.
    if time.monotonic() + 30 < deadline:
        lengths.extend([2200, 2500])

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        restarts = 12 if n <= 1000 else 8
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ----- initialise a starting point -------------------------
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
                f0 += rng.normal(scale=0.001 * target_sum, size=n)
                f0 = _project_onto_simplex(np.clip(f0, 0.0, None), target_sum)
            else:
                f0 = rng.random(n)
                f0 = _project_onto_simplex(f0, target_sum)

            # ----- smooth‑max Adam optimisation -----------------------
            sub_deadline = min(deadline, time.monotonic() + 7.0)
            f_opt, score_opt = _smooth_adam(f0,
                                            target_sum,
                                            sub_deadline)

            if score_opt < best_score:
                best_score = score_opt
                best_seq = f_opt.copy()
                print(f"[update] n={n} restart={r} score={score_opt:.6f}")

            # Early exit if we already beat the target.
            if best_score < 1.503 - 1e-6:
                print("[early] target reached")
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # 3️⃣ Final polishing with simulated annealing (if time remains).
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _simulated_annealing(best_seq,
                                                     deadline,
                                                     init_temp=0.2,
                                                     decay=0.9997)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final] annealed score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Fallback construction (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _raw_score(best_seq)
        print("[fallback] using uniform construction")

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
