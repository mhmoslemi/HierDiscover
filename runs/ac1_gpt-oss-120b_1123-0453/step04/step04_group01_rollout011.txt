def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)

def _simplex_projection(v, target):
    """Project v onto nonnegative simplex of total sum target."""
    # algorithm from Duchi 2008
    n = len(v)
    if v.sum() == target and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

        final_refine_deadline = deadline - 0.1  # ensure small leftover

def _simplex_projection(v, target):
    """Project v onto the simplex {x >= 0, sum x = target}."""
    if target < 0:
        raise ValueError
    n = v.shape[0]
    # If all values are already non-negative and sum equals target, return.
    # else compute.
    # Following algorithm from "Efficient Projections onto the L1-Ball for Learning in High Dimensions"
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # due to numerical issues, adjust sum:
    w_sum = w.sum()
    if not np.isclose(w_sum, target):
        # adjust tiny differences by scaling
        if w_sum > 0:
            w = w * (target / w_sum)
        else:
            w = np.full_like(v, target / n)
    return w

def _swap_hill_climb(seq, target, deadline, rng):
    best = seq.copy()
    best_score = _fast_score(best)
    n = best.shape[0]

    while time.time() < deadline:
        # randomly pick donor index weighted by a >0, and receiver index.
        donor_candidates = np.where(best > 1e-12)[0]
        if donor_candidates.size == 0:
            break
        donor = int(rng.choice(donor_candidates))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue

        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _simplex_projection(cand, target)
        cand_score = _fast_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

Improved step‑function minimiser.

Approach
--------
1️⃣ **Broad sweep** – generate many analytic families (uniform, ramps,
   power‑laws, exponentials, logistic, Beta‑distributions, sine‑powers,
   random Dirichlet draws) for several lengths (up to 2000).  Every
   candidate is ℓ¹‑normalised to √(2·n) and scored by the exact

   are kept for further work.

2️⃣ **Intensive refinement** – each of the top candidates is refined by a
   fast projected‑gradient / mass‑transfer heuristic.  At every step the
   active convolution index *k* (where the auto‑convolution attains its
   maximum) is identified; the gradient of the active entry
   (∂bₖ/∂aᵢ = 2·a_{k‑i}) is used to take a descent step that respects
   the ℓ¹‑norm and non‑negativity (projection onto the simplex).  Random
   perturbations, occasional smoothing and cheap Dirichlet restarts help
   to escape shallow local minima.

3️⃣ **Polishing** – a cheap swap‑hill‑climb followed by a short
   Gaussian‑perturbation loop (simulated‑annealing style) finishes the
   search.

All helper functions are top‑level, the code respects the security checks
of ``evaluate_sequence`` and runs well within the 1000 s time budget.

import time

# ----------------------------------------------------------------------
# Utility functions

def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``vec`` so that its ℓ¹‑norm equals ``target_sum``."""
    s = vec.sum()
    if s <= 0.0:
        return np.full_like(vec, target_sum / vec.shape[0], dtype=np.float64)
    return vec * (target_sum / s)


def _project_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x ≥ 0, Σ x = target}.
    """
    if target < 0:
        raise ValueError("target must be non‑negative")
    n = v.shape[0]
    # Duchi et al. (2008) – O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – fallback to uniform vector
        theta = 0.0
    else:
        rho = ind[cond].max()
        theta = cssv[cond][rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical guard – re‑scale to hit the exact target sum
    w_sum = w.sum()
    if w_sum > 0:
        w *= target / w_sum
    else:
        w = np.full_like(v, target / n, dtype=np.float64)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Auto‑convolution of ``a`` using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` without the safety wrapper.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))

def _beta_family(n: int, a: float, b: float, target_sum: float) -> np.ndarray:

    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    w = np.maximum(w, 1e-14)
    return _normalize(w, target_sum)

def _refine_sequence(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 30000,
) -> np.ndarray:
    """
    Heuristic optimiser that reduces the maximal entry of the auto‑convolution.
    Uses a projected gradient step on the active entry, occasional random
    perturbations and smoothing.
    """
    n = seq.shape[0]
    best = _project_to_simplex(seq, target_sum)
    best_sc = _fast_score(best)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(best)
        k = int(conv.argmax())

        # --- gradient of the active convolution entry ---
        grad = np.zeros(n, dtype=np.float64)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idxs] = 2.0 * best[k - idxs]  # ∂b_k/∂a_i = 2·a_{k-i}

        # descent direction that preserves the ℓ¹‑norm (sum zero)
        direction = -grad + grad.mean()

        # feasible step size (stay non‑negative)
        negative = direction < 0
        if np.any(negative):
            max_alpha = np.min(best[negative] / -direction[negative])
        else:
            max_alpha = 1.0
        if max_alpha <= 0:
            max_alpha = 1.0

        # take a modest step
        step = 0.5 * max_alpha
        cand = best + step * direction
        cand = _project_to_simplex(cand, target_sum)

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc - 1e-12:
            best, best_sc = cand, cand_sc
            continue

        # ------------------------------------------------------------------
        # If gradient step failed, try a cheap mass‑transfer move
        # ------------------------------------------------------------------
        contributions = best[idxs] * best[k - idxs]
        donor = int(idxs[np.argmax(contributions)])

        # try to find a receiver outside the active support (doesn't increase b_k)
        outside = np.setdiff1d(np.arange(n, dtype=np.int64), idxs, assume_unique=True)
        if outside.size > 0:
            receiver = int(outside[np.argmin(best[outside])])
        else:
            # fallback – pick the index with smallest partner value inside the support
            receiver = int(idxs[np.argmin(best[k - idxs])])

        if donor == receiver:
            receiver = int(rng.integers(n))
            if receiver == donor:
                continue

        max_move = min(best[donor] * 0.4, target_sum * 0.005)
        if max_move <= 0:
            continue
        eps = max_move
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _project_to_simplex(cand, target_sum)

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc - 1e-12:
            best, best_sc = cand, cand_sc
            continue

        # ------------------------------------------------------------------
        # Occasional stochastic perturbations (simulated‑annealing flavour)
        # ------------------------------------------------------------------
        if rng.random() < 0.015:
            step_size = target_sum * 0.02
            pert = rng.normal(scale=step_size, size=n)
            cand = best + pert
            cand = _project_to_simplex(cand, target_sum)
            cand_sc = _fast_score(cand)
            if cand_sc < best_sc:
                best, best_sc = cand, cand_sc
                continue

        # ------------------------------------------------------------------
        # Smoothing to erase tiny spikes
        # ------------------------------------------------------------------
        if rng.random() < 0.012:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(best, kernel, mode="same")
            smooth = _project_to_simplex(smooth, target_sum)
            smooth_sc = _fast_score(smooth)
            if smooth_sc < best_sc:
                best, best_sc = smooth, smooth_sc
                continue

        # ------------------------------------------------------------------
        # Rare Dirichlet restart
        # ------------------------------------------------------------------
        if rng.random() < 0.005:
            cand = rng.dirichlet(np.ones(n)) * target_sum
            cand_sc = _fast_score(cand)
            if cand_sc < best_sc:
                best, best_sc = cand, cand_sc
                continue

    return best


# ----------------------------------------------------------------------
# Final polishing helpers
# ----------------------------------------------------------------------


def _swap_hill_climb(
    seq: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator
) -> np.ndarray:
    """Very cheap hill‑climb that swaps a small mass between two indices."""
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    while time.time() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target_sum * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _project_to_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
    return best


def _gaussian_local_search(
    seq: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator
) -> np.ndarray:
    """Short Gaussian perturbation loop (annealing‑style)."""
    best = seq.copy()
    best_sc = _fast_score(best)
    step = target_sum * 0.03

    while time.time() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape[0])
        cand = _project_to_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
            step *= 0.9
        else:
            step *= 0.985
            if step < 1e-7:
                step = target_sum * 0.03
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  The routine stops a little before the limit.
    kwargs : dict
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # Use a previously discovered construction as a (possibly) good start.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = _project_to_simplex(init, target0)
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 0️⃣  Sweep a range of lengths and keep the K best raw candidates.
    # ------------------------------------------------------------------
    lengths = [2000, 1500, 1200, 1000, 800, 600, 500, 400, 300, 200]
    top_k = 5
    top_candidates = []  # list of (seq, score)

    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # generate a modest fleet of diverse start vectors
        candidates = _generate_candidates_extended(n, target, rng)

        for cand in candidates:
            if time.time() >= deadline:
                break
            sc = _fast_score(cand)
            if len(top_candidates) < top_k:
                top_candidates.append((cand, sc))
                top_candidates.sort(key=lambda x: x[1])
            else:
                if sc < top_candidates[-1][1]:
                    top_candidates[-1] = (cand, sc)
                    top_candidates.sort(key=lambda x: x[1])

            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"[sweep] n={n} new global best {sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Refine the K best candidates with the intensive optimiser.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and top_candidates:
        # allocate roughly equal time to each refinement run
        per_candidate = remaining / len(top_candidates)
        new_top = []
        for seq, sc in top_candidates:
            n = seq.shape[0]
            target = math.sqrt(2.0 * n)
            ref_deadline = time.time() + per_candidate
            refined = _refine_sequence(seq, target, ref_deadline, rng)
            refined_sc = _fast_score(refined)
            if refined_sc < best_score:
                best_seq, best_score = refined.copy(), refined_sc
                print(f"[refine] n={n} improved global best to {refined_sc:.6f}")
            new_top.append((refined, refined_sc))
        # keep the best of the refined set for final polishing
        top_candidates = sorted(new_top, key=lambda x: x[1])[:top_k]
        best_seq, best_score = top_candidates[0]

    # ------------------------------------------------------------------
    # 2️⃣  Final polishing on the best vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback uniform vector of moderate length
        n = 300
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    n = best_seq.shape[0]
    target = math.sqrt(2.0 * n)
    best_seq = _project_to_simplex(best_seq, target)

    # cheap swap‑hill‑climb
    best_seq = _swap_hill_climb(best_seq, target, deadline, rng)

    # short Gaussian hill‑climb (if time permits)
    if time.time() < deadline:
        best_seq = _gaussian_local_search(best_seq, target, deadline, rng)

    final_score = _fast_score(best_seq)
    print(f"[final] best score {final_score:.6f}   length {best_seq.shape[0]}")

    # return a plain Python list – evaluate_sequence will re‑check the constraints
    return [float(x) for x in best_seq]
