    # compute correlation of seq and w: grad_i = Σ_j seq_j * w_{i + j}
    # compute using FFT:
    n = len(seq)
    m = len(w)
    L = n + m - 1
    N = 1 << ((L - 1).bit_length() + 1)  # maybe next power >= L. Or use np.fft.rfft with N.
    Fa = np.fft.rfft(seq, n=N)
    Fw = np.fft.rfft(w, n=N)
    # correlation: ifft(conj(Fa) * Fw)
    corr = np.fft.irfft(Fa.conj() * Fw, n=N)
    # slice
    start = m - 1
    grad = corr[start:start+n]
    return grad

"""
Propose a low‑value step sequence.

Algorithm
---------
The objective after the mandatory ℓ¹–normalisation is the maximal entry of the
self‑convolution of the step heights.  We therefore minimise that peak.

1. **Parametric families** – a quick sweep over several classic shapes
   (uniform, exponential, power‑law, Hann, …) provides a solid baseline.

2. **Simulated annealing** – starting from the best family candidate we
   repeatedly move a small amount of mass from one index to another
   (preserving the ℓ¹‑norm).  New states are accepted with a Metropolis
   probability that slowly cools, allowing the search to escape shallow
   local minima while focussing on reducing the maximal convolution entry.

3. **Projected sub‑gradient descent** – after annealing we run a few
   iterations of the gradient‑descent method used in the previous
   solution.  The direction is the sub‑gradient of the active (maximal)
   convolution entry, projected onto the hyper‑plane ∑aᵢ=√(2 n).

4. **Final stochastic polish** – a short isotropic Gaussian hill‑climb
   finishes the optimisation.

All steps respect the security checks of ``evaluate_sequence``:
non‑negative, clipped to [0,1000] and renormalised to the required ℓ¹‑norm.
The routine returns the best sequence found before the 1000 s budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (full linear) via FFT
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of `seq` with itself using a power‑of‑two FFT."""
    n = seq.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # smallest power of 2 ≥ 2n‑1
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Normalised objective (exact, not the security‑wrapped version)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact objective value for a normalised sequence."""
    n = seq.shape[0]
    conv = _conv_fft(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)      # = max_b when ∑a = √(2n)


def _normalise(seq: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1000] and rescale to the required ℓ¹‑norm."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total == 0.0:
        return np.full_like(seq, target / seq.size)
    return seq * (target / total)


# ----------------------------------------------------------------------
# Hand‑crafted parametric families
# ----------------------------------------------------------------------
def _candidate_families(n: int, target_sum: float) -> tuple[np.ndarray, float]:
    """Return the best vector among several hand‑crafted families."""
    best_vec = None
    best_sc = float("inf")

    def _norm(v):
        v = np.asarray(v, dtype=np.float64)
        v = v / v.sum() * target_sum
        return v

    idx = np.arange(n, dtype=np.float64)

    # uniform
    a = np.full(n, target_sum / n, dtype=np.float64)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # exponential families – dense sweep of alpha values
    alphas = np.concatenate((
        np.arange(1, 6) / (10.0 * n),          # very small
        np.arange(1, 21) / n,                 # 1/n … 20/n
        np.linspace(0.02, 0.20, 40),
        np.linspace(0.20, 1.00, 30)
    ))
    alphas = np.unique(alphas)
    for α in alphas:
        a = np.exp(-α * idx)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # power‑law a_i ∝ (i+1)^(-p)
    p_vals = np.linspace(0.05, 5.0, 120)
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        a = base ** (-p)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # classic window functions
    for name, win in {
        "hann": 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1))),
        "hamming": 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1)),
        "blackman": (0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
                     + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1))),
        "sine": np.sin(np.pi * (idx + 1) / (n + 1)),
        "triangular": np.minimum(idx + 1, n - idx)
    }.items():
        a = _norm(win)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Simulated annealing using mass‑transfer moves
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing for a fixed length."""
    n = seq.shape[0]
    cur = seq.copy()
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    # initial temperature proportional to the current objective
    T0 = max(0.1, cur_score)

    while time.monotonic() < deadline:
        # geometric cooling
        T = T0 * (0.9999 ** it)

        # pick a donor with positive mass and a random receiver
        i = rng.integers(n)
        while cur[i] <= 0.0:
            i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        # transfer a random fraction of the donor's mass
        eps = cur[i] * rng.random() * 0.4   # up to 40 % of the donor
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        # renormalisation is not required – total mass unchanged
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _fast_score(cand)
        delta = cand_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
                print(f"[anneal] new best {best_score:.6f}  (iter {it})")
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the active max‑convolution entry
# ----------------------------------------------------------------------
def _gradient_descent(seq: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """A few iterations of the sub‑gradient method used in the reference code."""
    n = seq.shape[0]
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    step = 0.2
    rng = np.random.default_rng()

    while time.monotonic() < deadline:
        conv = _conv_fft(seq)
        k = int(conv.argmax())

        grad = np.zeros_like(seq)
        if k < n:
            grad[: k + 1] = seq[k :: -1]
        else:
            off = k - (n - 1)
            grad[off:] = seq[: n - off][::-1]

        # Projection onto the hyperplane Σa = target_sum  (mean‑shift)
        direction = -grad + grad.mean()
        cand = seq + step * direction
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target_sum / cand.sum())

        cand_sc = _fast_score(cand)

        if cand_sc < best_sc:
            best_seq, best_sc = cand.copy(), cand_sc
            seq = cand
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.5
            if step < 1e-6:               # occasional random shake‑up
                i, j = rng.integers(n, size=2)
                delta = seq[i] * 0.05
                if delta > 0:
                    cand = seq.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand = cand * (target_sum / cand.sum())
                    cand_sc = _fast_score(cand)
                    if cand_sc < best_sc:
                        seq = cand
                        best_seq, best_sc = cand.copy(), cand_sc
                step = 0.2

        if best_sc < 1.1:                # early stop when close to theoretical bound
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Final short Gaussian hill‑climb
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray, deadline: float, step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """Isotropic Gaussian perturbations with decreasing radius."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (math.sqrt(2.0 * n) / cand.sum())
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9997
        if step < 1e-7:
            step = step_initial
    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Use the incumbent if the notebook already supplies one
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                tgt0 = math.sqrt(2.0 * n0)
                init = _normalise(init, tgt0)
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 n={n0} score={sc0:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Explore several lengths, keep a solid baseline from families
    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 2000, 3000, 5000, 8000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # 1a – parametric families
        cand_vec, cand_sc = _candidate_families(n, target)
        if cand_sc < best_score:
            best_seq, best_score = cand_vec.copy(), cand_sc
            print(f"[family] n={n} new best {best_score:.6f}")

        # 1b – a handful of random Dirichlet starts
        rnd_until = time.monotonic() + min(0.6, (deadline - time.monotonic()) / 4)
        while time.monotonic() < rnd_until:
            a = rng.random(n)
            a = _normalise(a, target)
            sc = _fast_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[rand] n={n} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Simulated annealing from the current incumbent
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        anneal_deadline = time.monotonic() + min(30.0, (deadline - time.monotonic()) / 2)
        annealed_seq, annealed_sc = _anneal(best_seq, target, anneal_deadline)
        if annealed_sc < best_score:
            best_seq, best_score = annealed_seq, annealed_sc
            print(f"[anneal] refined to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Sub‑gradient descent refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        grad_deadline = time.monotonic() + min(15.0, (deadline - time.monotonic()) / 3)
        refined_seq, refined_sc = _gradient_descent(best_seq, target, grad_deadline)
        if refined_sc < best_score:
            best_seq, best_score = refined_seq, refined_sc
            print(f"[grad] refined to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final Gaussian polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_sc = _local_search(best_seq, deadline, step_initial=0.2)
        if polished_sc < best_score:
            best_seq, best_score = polished_seq, polished_sc
            print(f"[final] polished to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list as required by the harness
    return [float(v) for v in best_seq]
