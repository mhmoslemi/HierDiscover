"""
Propose a low‑value non‑negative step sequence for `evaluate_sequence`.

The algorithm works in three phases:

1️⃣ Warm‑start from an existing construction (if the global variable
   `height_sequence_1` exists).  The sequence is re‑normalised to the
   simplex Σ a_i = √(2 n).

2️⃣ Projected Adam optimisation on the ℓ¹‑simplex, but instead of the
   raw max‑convolution we minimise a *smooth* approximation
   L_α(f) = (1/α)·log ∑ exp(α·(f*f)).  The gradient of this loss can be
   written in closed form using a single correlation, which we compute
   efficiently with `np.convolve`.  α is increased during the run so
   that the optimisation starts smooth (good gradients) and ends
   sharp (close to the true max).

3️⃣ A temperature‑controlled simulated‑annealing polish that moves
   small amounts of mass between coordinates while preserving the
   ℓ¹‑norm exactly.  Up‑hill moves are accepted with the Metropolis
   probability, allowing the search to escape shallow minima.

All phases respect the global time budget (default 1000 s) and the best
sequence found is returned as a plain Python list of floats.
"""

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # edge case – return a uniform distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(f: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` for a given candidate.
    The clipping performed inside `evaluate_sequence` is irrelevant for the
    values we generate (the sum is kept at √(2 n) and all entries stay far
    below 1000).
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))

    """
    Returns a smooth approximation of the max‑convolution objective and its
    gradient.

    L_α(f) = max(conv) + (1/α)·log Σ_k exp(α·(conv_k - max(conv))).
    The gradient is
        ∇L = 2 * ( w ⋆ f ),   where w_k = softmax_α(conv_k)
    and "⋆" denotes the (non‑flipped) correlation.
    """
    conv = np.convolve(f, f)               # length 2n‑1
    max_conv = np.max(conv)

    # stable soft‑max over the convolution entries
    shift = alpha * (conv - max_conv)      # shift ≤ 0, max shift = 0
    exp_shift = np.exp(shift)
    Z = np.sum(exp_shift)
    w = exp_shift / Z                      # length 2n‑1, sums to 1

    # smooth loss value
    loss = max_conv + math.log(Z) / alpha

    # gradient = 2 * correlation(w, f)
    # correlation can be obtained by convolving w with the reversed f
    corr = np.convolve(w, f[::-1])         # length 3n‑2
    n = len(f)
    grad = 2.0 * corr[n - 1 :: -1][:n]     # take the first n entries reversed

    return loss, grad


# ----------------------------------------------------------------------
# Projected Adam optimisation on the simplex (smooth loss)
# ----------------------------------------------------------------------


def _adam_smooth(f0: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 max_iters: int = 4000,
                 alpha_start: float = 5.0,
                 alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Projected Adam descent that minimises a smoothed version of the
    max‑auto‑convolution.  The iterate is always projected onto the
    ℓ¹‑simplex Σ a_i = target_sum, a_i ≥ 0.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)

    m = np.zeros_like(f)          # first moment
    v = np.zeros_like(f)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05                # base learning rate

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # linearly increase α from start to end
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (max_iters - 1)

        loss, grad = _smooth_objective(f, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = np.clip(f_candidate, 0.0, None)
        f_candidate = _project_onto_simplex(f_candidate, target_sum)

        cur_score = _score(f)
        cand_score = _score(f_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_f = f_candidate.copy()
            print(f"[Adam] it={it:4d} n={len(f)} α={alpha:5.2f} score={cand_score:.6f}")

        # accept the candidate if it improves the *true* objective
        if cand_score < cur_score:
            f = f_candidate

    return best_f, best_score


# ----------------------------------------------------------------------
# Simulated Annealing polishing
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled hill‑climber that preserves the ℓ¹‑norm
    exactly (mass is transferred between two coordinates).
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.05 * target_sum / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-8:
            T = init_temp

        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.

    The routine respects the given time budget and returns a plain
    Python list suitable for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start from a previously found construction, if any.
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64).copy()
        n = len(seq)
        target_sum = math.sqrt(2.0 * n)
        if np.sum(seq) == 0.0:
            seq = np.full(n, target_sum / n)
        else:
            seq = seq * (target_sum / np.sum(seq))
        seq = np.clip(seq, 0.0, 1000.0)
        s = _score(seq)
        best_seq = seq.copy()
        best_score = s
        print(f"[init] warm‑start from height_sequence_1, n={n}, score={s:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Main optimisation – Adam on several lengths and restarts.
    # --------------------------------------------------------------
    # lengths are chosen to give a good trade‑off between expressiveness
    # and speed.  We start with modest sizes and climb up to ~2500.
    lengths = list(range(300, 2501, 200))  # 300, 500, …, 2400
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of random restarts – more restarts for smaller n (cheaper)
        restarts = 12 if n <= 800 else 6

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a starting point
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
                f0 += rng.normal(scale=0.001 * target_sum, size=n)
                f0 = np.clip(f0, 0.0, None)
                f0 = _project_onto_simplex(f0, target_sum)
            else:
                # Dirichlet‑like random start (exponential draws then project)
                f0 = rng.exponential(scale=1.0, size=n)
                f0 = _project_onto_simplex(f0, target_sum)

            # allocate a few seconds for each descent run
            sub_deadline = min(deadline, time.monotonic() + 9.0)
            f_opt, score_opt = _adam_smooth(
                f0,
                target_sum,
                sub_deadline,
                max_iters=3500,
                alpha_start=4.0,
                alpha_end=80.0,
            )

            if score_opt < best_score:
                best_score = score_opt
                best_seq = f_opt.copy()
                print(f"[update] n={n} restart={r} score={score_opt:.6f}")

            # early exit if we already beat the target threshold
            if best_score < 1.503:
                print("[early] target reached during Adam phase")
                break
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # 3️⃣ Final polishing with simulated annealing (if time permits).
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _simulated_annealing(
            best_seq,
            deadline,
            init_temp=0.2,
            decay=0.9997,
        )
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final] annealed score={refined_score:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never be needed) – a simple uniform construction.
    # --------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] using uniform construction")

    # Clip once more for safety before returning.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
