    An Adam optimiser with Euclidean projection onto the ℓ¹‑simplex
    (Σ a_i = √(2 n),  a_i ≥ 0) follows this smoothed landscape while the
    temperature is slowly lowered, turning the surrogate into the true
    max.  The best iterate is retained.

3️⃣  **Final polishing** – a simple simulated‑annealing hill‑climber
    (mass moves between two coordinates while preserving the ℓ¹‑norm)
    searches locally around the best Adam point.

All steps respect the global time budget (default 1000 s).  The function
returns a plain Python list ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
#  Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Scoring helpers (exact copy of evaluate_sequence without the
#  clipping that is irrelevant for our search)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = len(a)
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    sum_val = np.sum(a)
    if sum_val < 0.01:
        return float("inf")
    return 2.0 * n * max_val / (sum_val * sum_val)


# ----------------------------------------------------------------------
#  Soft‑max utilities for the smoothed max‑convolution
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Return a probability vector proportional to exp(conv/temperature)."""
    if temperature <= 0:
        # exact max – put equal mass on all tying indices
        w = np.zeros_like(conv)
        m = np.max(conv)
        mask = conv == m
        w[mask] = 1.0 / mask.sum()
        return w
    # stable soft‑max
    scaled = conv / temperature
    m = np.max(scaled)
    ex = np.exp(scaled - m)
    return ex / ex.sum()


def _gradient_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k   (w is a probability vector of length 2n‑1).
    The formula is  ∂/∂a_i = 2·Σ_j a_j·w_{i+j} – a_i·w_{2i}.
    """
    n = len(a)
    grad = np.empty_like(a)
    # Σ_j a_j·w_{i+j}  for each i
    for i in range(n):
        grad[i] = np.dot(a, w[i:i + n])
    # correct the diagonal double‑counting
    grad = 2.0 * grad - a * w[2 * np.arange(n)]
    return grad


# ----------------------------------------------------------------------
#  Smoothed Adam optimiser (projects onto the ℓ¹‑simplex after each step)
# ----------------------------------------------------------------------
def _adam_softmax(f0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_iters: int = 3000,
                  init_temp: float = 1.0,
                  temp_decay: float = 0.995,
                  lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Minimise max(a*a) by optimising the temperature‑controlled soft‑max.
    The iterate is always projected onto the simplex Σ a_i = target_sum,
    a_i ≥ 0.
    """
    f = _project_onto_simplex(f0, target_sum)
    best_f = f.copy()
    best_sc = _score(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    while it < max_iters and time.monotonic() < deadline:
        it += 1

        conv = np.convolve(f, f)
        T = max(init_temp * (temp_decay ** it), 1e-9)
        w = _softmax_weights(conv, T)

        # gradient of the weighted sum (the denominator Σ a_i is constant)
        grad = _gradient_weighted(f, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = f - step
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        sc_cand = _score(cand)
        if sc_cand < best_sc:
            best_sc = sc_cand
            best_f = cand.copy()
            print(f"[Adam‑soft] it={it:4d} n={len(f)} temp={T:.3e} score={sc_cand:.6f}")

        # accept the candidate only if it improves the raw objective
        if sc_cand < _score(f):
            f = cand

    return best_f, best_sc


# ----------------------------------------------------------------------
#  Simple simulated annealing hill‑climber (identical to the earlier version)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9997) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.05 * target_sum / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T = max(T * decay, 1e-9)
        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score


# ----------------------------------------------------------------------
#  Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low value of
    `evaluate_sequence`.  The routine respects the given time budget and
    returns a plain Python list ready for evaluation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1️⃣  Warm‑start from a known construction (if present)
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64).copy()
        seq = np.clip(seq, 0.0, 1000.0)
        n = len(seq)
        target = math.sqrt(2.0 * n)
        if np.sum(seq) == 0.0:
            seq = np.full(n, target / n)
        else:
            seq = seq * (target / np.sum(seq))
        sc = _score(seq)
        best_seq = seq.copy()
        best_score = sc
        print(f"[init] warm‑start from height_sequence_1, n={n}, score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Deterministic Gaussian sweep (fast global exploration)
    # --------------------------------------------------------------
    lengths = [400, 600, 800, 1000, 1200, 1500, 1800, 2500]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        mu = (n - 1) / 2.0
        # width parameter α = σ / n explored on a modest grid
        for alpha in np.linspace(0.08, 0.45, 12):
            if time.monotonic() >= deadline:
                break
            sigma = max(1.0, alpha * n)
            idx = np.arange(n)
            raw = np.exp(-((idx - mu) ** 2) / (2.0 * sigma ** 2))
            raw_sum = np.sum(raw)
            if raw_sum == 0.0:
                continue
            seq = raw * (target_sum / raw_sum)
            sc = _score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[gauss] n={n} α={alpha:.3f} score={sc:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                print("[early] Gaussian sweep reached target")
                return [float(x) for x in best_seq]

    # --------------------------------------------------------------
    # 3️⃣  Smoothed Adam refinement on the current best length
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        # re‑scale precisely to the simplex (tiny numerical drift may have crept in)
        best_seq = best_seq * (target_sum / np.sum(best_seq))
        sub_deadline = min(deadline, time.monotonic() + 0.6 * (deadline - time.monotonic()))
        f_opt, sc_opt = _adam_softmax(best_seq,
                                      target_sum,
                                      sub_deadline,
                                      max_iters=5000,
                                      init_temp=2.0,
                                      temp_decay=0.995,
                                      lr=0.04)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = f_opt.copy()
            print(f"[refine] Adam‑soft final score={sc_opt:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing polishing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_sc = _simulated_annealing(best_seq, deadline,
                                                  init_temp=0.2,
                                                  decay=0.9997)
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[final] annealed score={refined_sc:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] using uniform construction")

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
