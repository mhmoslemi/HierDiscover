"""
candidate_proposer.py

Search for a low‑value step sequence.
-------------------------------------------------
Algorithm outline
0️⃣  Warm‑start from any existing global variable name
    ``height_sequence_1`` (if present).

1️⃣  Exhaustive parametric sweep:
    * uniform,
    * power‑law a_i ∝ (i+1)^{-p},
    * exponential a_i ∝ exp(-α·i),
    * half‑sine a_i ∝ sin(π·(i+0.5)/n),
    * linear ramp a_i ∝ i+1,
    * their reversed versions.
    All candidates are projected onto the feasible ℓ¹‑sphere
    (non‑negative, Σa_i = √(2·n)).

2️⃣  For each length we launch a few **simulated‑annealing**
    restarts.  Each step moves a small fraction of mass from one
    index to another, keeps the total mass fixed, and accepts the
    move according to a temperature schedule.  The cost function
    is the exact max‑auto‑convolution ``max(a∗a)`` (computed by a
    fast FFT‑based convolution).  The current best is recorded
    globally.

3️⃣  A final **smooth‑max gradient descent** (the method used in the
    previous baseline) is applied to the overall best vector to
    polish it further.

The search respects the 1000 s time budget, prints intermediate
progress, and finally returns the best vector as a plain Python
list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return an FFT‑friendly length (fallback to power‑of‑2)."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for many calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _conv_max(seq: np.ndarray) -> float:
    """Exact value of max(seq∗seq)."""
    conv = _linear_convolve(seq, seq)
    return float(conv.max())


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Force 0 ≤ a_i ≤ 1000 and ℓ¹‑norm = √(2·n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / total)
    return seq


def _fast_score(seq: np.ndarray) -> float:
    """Convenient wrapper used for quick comparisons (exact max)."""
    return _conv_max(seq)


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights for a surrogate of max (numerically stable)."""
    cmax = conv.max()
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Derivative w.r.t. a_i is 2·∑_j w_{i+j}·a_j.
    Computed via one FFT‑based correlation.
    """
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)          # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]           # middle n entries
    grad = 2.0 * segment[::-1]
    return grad


def _optimize_sequence(
    init_seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max gradient descent with an annealing schedule for β.
    Returns the best vector seen and its true max‑auto‑convolution.
    """
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    best_local = a.copy()
    best_score = _fast_score(a)

    it = 0
    beta = 1.0                      # start with a very smooth surrogate
    base_step = (target / n) * 5.0  # crude step size

    while time.monotonic() < deadline and it < max_iters:
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # --- smooth‑max surrogate & gradient ---
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # --- simple annealed step ---
        step = base_step / (1.0 + it * 0.02)
        a = a - step * grad

        # --- projection onto feasible set ---
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # occasional jitter to escape plateaux
        if it % 200 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.03
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # β‑annealing
        if it % 100 == 0 and beta < 250.0:
            beta *= 1.45

        it += 1

    return best_local, best_score


def _simulated_annealing(
    init_seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the exact max‑auto‑convolution.
    Moves a small chunk of mass from one index to another.
    The total ℓ¹‑mass stays constant, therefore the iterate always
    stays feasible (non‑negative, Σ = √(2·n)).
    """
    a = init_seq.copy()
    n = a.shape[0]
    cur_score = _conv_max(a)

    best_seq = a.copy()
    best_score = cur_score

    # initial temperature – a fraction of the current cost
    T0 = cur_score * 0.25 + 1e-12

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # temperature schedule (geometric cooling)
        T = T0 * (0.995 ** it)

        # pick two distinct positions
        i, j = rng.choice(n, size=2, replace=False)

        if a[i] <= 0.0:
            it += 1
            continue

        # amount to move: up to 5 % of the source entry
        delta = a[i] * rng.uniform(0.0, 0.05)

        # propose the new vector (mass transfer keeps the ℓ¹‑norm unchanged)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # safety – guarantee non‑negativity (already ensured by delta)
        if cand[i] < 0.0:
            it += 1
            continue

        cand_score = _conv_max(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / (T + 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()

        it += 1

    return best_seq, best_score


def _update_global(
    seq: np.ndarray,
    best_seq: np.ndarray | None,
    best_score: float,
) -> tuple[np.ndarray | None, float]:
    """
    Compare *seq* (already normalized) with the current global best.
    Returns the possibly updated best sequence and its score.
    """
    sc = _fast_score(seq)
    if sc < best_score:
        print(f"[global best] n={len(seq)} score={sc:.6f}")
        return seq.copy(), sc
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 0,
    budget_s: float = 1000,
    **kwargs,
) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.
    The routine combines:
      • a parametric sweep over several families,
      • a few simulated‑annealing runs,
      • a final smooth‑max gradient descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from an externally supplied construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            cand = _normalize(cand)
            best_seq, best_score = _update_global(cand, best_seq, best_score)
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Exhaustive parametric sweep
    # ------------------------------------------------------------------
    lengths = [2500, 2000, 1500, 1200, 1000, 800, 600, 500, 400, 300, 200, 150, 100]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(1, n + 1, dtype=np.float64)

        # uniform (baseline)
        uniform = np.full(n, target / n, dtype=np.float64)
        uniform = _normalize(uniform)
        best_seq, best_score = _update_global(uniform, best_seq, best_score)

        # power‑law a_i ∝ i^{‑p}
        for p in np.linspace(0.2, 3.0, 15):
            a = idx ** (-p)
            a = a / a.sum() * target
            a = _normalize(a)
            best_seq, best_score = _update_global(a, best_seq, best_score)

        # exponential a_i ∝ exp(-α i)
        for alpha in np.linspace(0.001, 0.2, 15):
            a = np.exp(-alpha * np.arange(n, dtype=np.float64))
            a = a / a.sum() * target
            a = _normalize(a)
            best_seq, best_score = _update_global(a, best_seq, best_score)

        # half‑sine shape
        a = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
        a = a / a.sum() * target
        a = _normalize(a)
        best_seq, best_score = _update_global(a, best_seq, best_score)

        # linear ramp (increasing) and its reverse
        ramp = np.arange(1, n + 1, dtype=np.float64)
        ramp = ramp / ramp.sum() * target
        ramp = _normalize(ramp)
        best_seq, best_score = _update_global(ramp, best_seq, best_score)

        rev_ramp = ramp[::-1]
        best_seq, best_score = _update_global(rev_ramp, best_seq, best_score)

    # ------------------------------------------------------------------
    # 2️⃣ Simulated‑annealing refinement (multiple restarts per length)
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        restarts = 5 if n <= 800 else 2

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # start from a Dirichlet‑like random vector
            start = rng.gamma(1.0, size=n)
            start = start / start.sum() * target

            # if we already have a good vector of same length, blend it
            if best_seq is not None and best_seq.shape[0] == n:
                start = (start + best_seq) * 0.5
                start = _normalize(start)

            cand_seq, cand_score = _simulated_annealing(start, deadline, rng)
            best_seq, best_score = _update_global(cand_seq, best_seq, best_score)

            # early exit if we have already crossed an ambitious threshold
            if best_score < 1.45:   # arbitrary safety target
                print("[early exit] very good score reached")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 3️⃣ Final polish with smooth‑max gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None:
        polished, polished_score = _optimize_sequence(best_seq, deadline, rng)
        best_seq, best_score = _update_global(polished, best_seq, best_score)

    # ------------------------------------------------------------------
    # 4️⃣ Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # Return a plain Python list
    return [float(x) for x in best_seq]
