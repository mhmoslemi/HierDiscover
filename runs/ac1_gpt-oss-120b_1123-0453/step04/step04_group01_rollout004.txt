Implement:

def simplex_projection(v, z):
    # project onto {x >= 0, sum(x) = z}
    # from https://doi.org/10.1007/11564071_22, algorithm O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Subgradient optimiser for the step‑function evaluation challenge.

The algorithm works directly on the ℓ¹‑normalised vector (sum = √(2·n)).
It repeatedly:
   1. evaluates the full auto‑convolution via an FFT,
   2. finds the index of the maximal entry,
   3. takes a sub‑gradient step that reduces that entry,
   4. projects the result back onto the simplex {x ≥ 0, Σx = √(2·n)}.

Multiple random restarts (including a greedy Sidon‑set seed) are run for
different lengths.  The best vector found before the time budget expires
is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x ≥ 0, Σx = z})
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of sum z."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpinán (2013)
    # Complexity O(n log n) due to the sort.
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    # Find the rho index: largest i where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are projected to zero, return zeros
        return np.zeros_like(v)
    rho = ind[cond][-1]  # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution (full) using FFT – fast for repeated calls.
# ----------------------------------------------------------------------


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the full (linear) convolution of a with itself using FFT.

    Parameters
    ----------
    a : np.ndarray
        Input vector (real, length n).
    fft_len : int
        Length of the FFT (power of two ≥ 2·n−1).

    Returns
    -------
    np.ndarray
        Full convolution of length 2·n−1.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    # Keep only the first 2·n‑1 entries
    return conv[: 2 * a.shape[0] - 1]


def _prepare_fft_len(n: int) -> int:
    """Return a power‑of‑two length suitable for convolution of size n."""
    size = 2 * n - 1
    # Next power of two
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Search helpers
# ----------------------------------------------------------------------


def _initial_sidon_vector(n: int, target_sum: float) -> np.ndarray:
    """
    Construct a sparse vector whose support is a greedy Sidon set.
    All non‑zero entries have equal mass, so the ℓ¹‑norm equals target_sum.
    """
    # Greedy construction of a Sidon set within {0,…,n‑1}
    max_size = int(math.sqrt(2 * n))  # theoretical upper bound
    positions = []
    sums = set()
    for candidate in range(n):
        conflict = False
        for p in positions:
            if (candidate + p) in sums:
                conflict = True
                break
        if conflict:
            continue
        # also avoid duplicate self‑sum
        if candidate * 2 in sums:
            continue
        # accept
        positions.append(candidate)
        # update the set of sums (unordered)
        for p in positions:
            sums.add(candidate + p)
        sums.add(candidate * 2)
        if len(positions) >= max_size:
            break
    if not positions:
        # fallback to uniform
        return np.full(n, target_sum / n, dtype=np.float64)
    m = len(positions)
    mass = target_sum / m
    vec = np.zeros(n, dtype=np.float64)
    vec[positions] = mass
    return vec


def _subgradient_optimize(
    init_vec: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 5000,
) -> np.ndarray:
    """
    Perform a simple projected sub‑gradient descent on the max‑convolution
    objective.  Returns the best vector encountered during the run.
    """
    n = init_vec.shape[0]
    a = _project_simplex(init_vec.copy(), target_sum)
    best = a.copy()
    best_score = _fast_score(a)
    # Pre‑compute FFT length for this n
    fft_len = _prepare_fft_len(n)

    # Step‑size schedule parameters
    step_initial = target_sum * 0.4  # fairly large starting step
    # Diminishing factor
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # full auto‑convolution
        conv = _full_convolution_fft(a, fft_len)
        # max entry and its index
        idx_max = int(conv.argmax())
        cur_score = float(conv[idx_max])
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score

        # sub‑gradient of the active entry
        # gradient_i = 2 * a_{k-i} if k-i inside range else 0
        i_start = max(0, idx_max - (n - 1))
        i_end = min(idx_max, n - 1)
        idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx_range] = 2.0 * a[idx_max - idx_range]

        # Diminishing step size (simple 1/√t schedule)
        step = step_initial / math.sqrt(it + 1)

        # Try a few step‑size candidates to guarantee descent
        improved = False
        for factor in (1.0, 0.5, 0.25):
            cand = a - step * factor * grad
            cand = _project_simplex(cand, target_sum)
            conv_cand = _full_convolution_fft(cand, fft_len)
            cand_score = float(conv_cand.max())
            if cand_score < cur_score - 1e-12:
                a = cand
                improved = True
                break
        if not improved:
            # no improvement – perform a tiny random perturbation
            eps = target_sum * 0.001
            cand = a + rng.normal(scale=eps, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _project_simplex(cand, target_sum)
            a = cand

    return best


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` but without the safety wrapper.
    For a vector already ℓ¹‑normalised this equals the maximal auto‑convolution entry.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


def _run_for_length(
    n: int,
    time_budget: float,
    rng: np.random.Generator,
    global_best_seq: np.ndarray | None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a vector of length ``n`` within the supplied time budget.
    Returns the best vector found and its score.
    """
    target = math.sqrt(2.0 * n)
    deadline = time.monotonic() + time_budget
    best_seq = None
    best_score = float("inf")

    # 1) seed vectors: uniform, Sidon, a few Dirichlet draws, perhaps the global best if length matches
    seeds = []

    # uniform
    seeds.append(np.full(n, target / n, dtype=np.float64))

    # Sidon set
    seeds.append(_initial_sidon_vector(n, target))

    # a handful of random Dirichlet vectors
    for _ in range(3):
        seeds.append(rng.dirichlet(np.ones(n)) * target)

    # if we have a global best of the same length, use it as a seed
    if global_best_seq is not None and global_best_seq.shape[0] == n:
        seeds.append(global_best_seq.copy())

    # optimisation loop for each seed
    for start in seeds:
        if time.monotonic() >= deadline:
            break
        # allocate a slice of remaining time for this seed
        remaining = deadline - time.monotonic()
        # give each seed at most 15 s or the remaining proportion
        sub_deadline = time.monotonic() + min(15.0, remaining * 0.4)
        cand = _subgradient_optimize(start, target, sub_deadline, rng)
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[n={n}] seed improved to {sc:.6f}")

    # A final short random‑restart hill‑climb (swap‑based) can polish the result
    if best_seq is not None and time.monotonic() < deadline:
        # simple swap hill climb
        best_seq = _swap_hill_climb(best_seq, target, deadline, rng)
        best_score = _fast_score(best_seq)

    return best_seq, best_score


def _swap_hill_climb(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Very cheap hill‑climb that swaps a tiny amount of mass between two positions.
    """
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]
    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target_sum * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _project_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
    return best

# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises the
    evaluation score.  The function respects the 1000‑second budget,
    uses several lengths, a sub‑gradient optimiser, and a final polishing
    stage.  The best vector found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # Lengths to explore – larger n gives more flexibility.
    lengths = [1000, 800, 600, 500, 400, 300, 200]

    # allocate time roughly equally, but give a larger share to the biggest length
    remaining_lengths = len(lengths)
    while time.monotonic() < deadline and lengths:
        n = lengths.pop(0)
        # simple proportional allocation: remaining_time / remaining_lengths
        time_left = deadline - time.monotonic()
        if time_left <= 0:
            break
        # give a minimum of 5 s per length, otherwise split evenly
        alloc = max(5.0, time_left / (len(lengths) + 1))
        # run optimisation for this length
        cand_seq, cand_sc = _run_for_length(n, alloc, rng, best_seq)
        if cand_seq is not None and cand_sc < best_score:
            best_seq, best_score = cand_seq.copy(), cand_sc
            print(f"[global] new best length {n} score {cand_sc:.6f}")

    # If nothing was found (unlikely), fall back to a uniform vector of moderate size.
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # Final polishing – a short Gaussian perturbation if time permits
    if time.monotonic() < deadline:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        # Gaussian perturbation with diminishing step size
        step = target * 0.02
        while time.monotonic() < deadline:
            cand = best_seq + rng.normal(scale=step, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _project_simplex(cand, target)
            cand_sc = _fast_score(cand)
            if cand_sc < best_score:
                best_seq, best_score = cand, cand_sc
                step *= 0.9
            else:
                step *= 0.99
                if step < 1e-6:
                    step = target * 0.02

    final_sc = _fast_score(best_seq)
    print(f"[final] best score {final_sc:.6f}   length {best_seq.shape[0]}")
    # Return as a plain list of floats
    return [float(x) for x in best_seq]

            cand = _project_simplex(cand, target_sum)
            cand_score = _fast_score(cand)
            if cand_score < cur_score - 1e-12:
                a = cand
                improved = True
                break
        if not improved:
            # fallback: tiny random jitter
            eps = target_sum * 0.001
            cand = a + rng.normal(scale=eps, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _project_simplex(cand, target_sum)
            a = cand

    return best


def _swap_hill_climb(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """Cheap hill‑climb that moves a small amount of mass between two positions."""
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target_sum * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _project_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
    return best


def _run_for_length(
    n: int,
    time_budget: float,
    rng: np.random.Generator,
    global_best_seq: np.ndarray | None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a vector of length ``n`` within ``time_budget`` seconds.
    Returns the best vector found and its evaluation score.
    """
    target = math.sqrt(2.0 * n)
    deadline = time.monotonic() + time_budget

    best_seq = None
    best_score = float("inf")

    # ---- seed vectors -------------------------------------------------
    seeds: list[np.ndarray] = []

    # uniform
    seeds.append(np.full(n, target / n, dtype=np.float64))

    # greedy Sidon set
    seeds.append(_initial_sidon_vector(n, target))

    # a few random Dirichlet draws
    for _ in range(3):
        seeds.append(rng.dirichlet(np.ones(n)) * target)

    # previous global best of the same length (if any)
    if global_best_seq is not None and global_best_seq.shape[0] == n:
        seeds.append(global_best_seq.copy())

    # explicitly reuse the previously discovered construction if it exists
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(height_sequence_1, dtype=np.float64)
            if old.shape[0] == n:
                seeds.append(_project_simplex(old, target))
        except Exception:
            pass

    # ---- optimisation for each seed ----------------------------------
    for start in seeds:
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        sub_deadline = time.monotonic() + min(15.0, remaining * 0.4)
        cand = _subgradient_optimize(start, target, sub_deadline, rng)
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[n={n}] improved to {sc:.6f}")

    # final cheap polishing with a swap‑based hill‑climb
    if best_seq is not None and time.monotonic() < deadline:
        best_seq = _swap_hill_climb(best_seq, target, deadline, rng)
        best_score = _fast_score(best_seq)

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the supplied time budget, explores several lengths,
    runs a projected sub‑gradient optimiser with random restarts and finishes
    with a short Gaussian polishing stage.  The best sequence found is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    # safety margin so we don't hit the exact timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # Lengths to try – larger n gives more freedom, but we also keep some
    # medium sizes in case they turn out to be easier for the optimiser.
    lengths = [1000, 800, 600, 500, 400, 300, 200]

    # simple proportional time allocation (larger n gets a slightly larger share)
    while time.monotonic() < deadline and lengths:
        n = lengths.pop(0)
        time_left = deadline - time.monotonic()
        if time_left <= 0:
            break
        # guarantee at least a few seconds per length
        allocated = max(5.0, time_left / (len(lengths) + 1))
        cand_seq, cand_sc = _run_for_length(n, allocated, rng, best_seq)
        if cand_seq is not None and cand_sc < best_score:
            best_seq, best_score = cand_seq, cand_sc
            print(f"[global] new best length {n} score {cand_sc:.6f}")

    # Fallback – should never happen, but keep a safe default.
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # Final Gaussian polishing (annealing‑style) if time permits
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        step = target * 0.02
        while time.monotonic() < deadline:
            cand = best_seq + rng.normal(scale=step, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _project_simplex(cand, target)
            cand_sc = _fast_score(cand)
            if cand_sc < best_score:
                best_seq, best_score = cand, cand_sc
                step *= 0.9
            else:
                step *= 0.99
                if step < 1e-6:
                    step = target * 0.02

    final_sc = _fast_score(best_seq)
    print(f"[final] best score {final_sc:.6f}   length {best_seq.shape[0]}")
    return [float(x) for x in best_seq]
