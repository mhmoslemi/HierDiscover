    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


# ----------------------------------------------------------------------
# Core helpers
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` after ℓ¹‑normalisation."""
    return float(np.convolve(seq, seq).max())


def _project(seq: np.ndarray, target: float) -> np.ndarray:
    """Clamp to [0,1000] and rescale to have ℓ¹‑norm = target."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total > 0.0:
        seq = seq * (target / total)
    else:
        seq = np.full_like(seq, target / seq.shape[0], dtype=np.float64)
    return seq


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max weights over the convolution entries."""
    cmax = conv.max()
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the smooth‑max surrogate (cross‑correlation)."""
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)          # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]          # middle n entries
    grad = 2.0 * segment[::-1]
    return grad


def _subgradient_exact(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the true max at index k."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _smooth_max_optimize(a: np.ndarray,
                         target: float,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 1200,
                         beta_start: float = 1.0,
                         beta_max: float = 400.0) -> tuple[np.ndarray, float]:
    """Smooth‑max gradient descent with momentum."""
    n = a.shape[0]
    best_a = a.copy()
    best_val = _fast_score(a)
    mom = np.zeros_like(a)
    mu = 0.9
    base_step = target * 5.0 / n
    beta = beta_start
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        step = base_step / (1.0 + it * 0.01)
        mom = mu * mom - step * grad
        a = a + mom
        a = _project(a, target)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.04
                a[i] -= delta
                a[j] += delta
                a = _project(a, target)

        # anneal β
        if it % 100 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.45)

        it += 1

    return best_a, best_val


def _exact_max_optimize(a: np.ndarray,
                        target: float,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iters: int = 1800) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with momentum."""
    n = a.shape[0]
    best_a = a.copy()
    best_val = _fast_score(a)
    mom = np.zeros_like(a)
    mu = 0.9
    base_step = target * 3.0 / n
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()

        k = int(np.argmax(conv))
        grad = _subgradient_exact(a, k)

        step = base_step / (1.0 + it * 0.01)
        mom = mu * mom - step * grad
        a = a + mom
        a = _project(a, target)

        # jitter every few iterations
        if it % 200 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project(a, target)

        it += 1

    return best_a, best_val


def _optimize_two_phase(start_seq: np.ndarray,
                        target: float,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Smooth‑max phase followed by exact‑max refinement."""
    seq1, val1 = _smooth_max_optimize(start_seq, target, deadline, rng)
    if time.monotonic() < deadline:
        seq2, val2 = _exact_max_optimize(seq1, target, deadline, rng)
        if val2 < val1:
            return seq2, val2
    return seq1, val1


# ----------------------------------------------------------------------
# Initial‑vector factories
# ----------------------------------------------------------------------
def _random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.gamma(1.0, size=n)


def _powerlaw_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 4.5)
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _exponential_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.001, 0.6)
    return np.exp(-alpha * np.arange(n, dtype=np.float64))


def _piecewise_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    split = rng.integers(1, n - 1)
    high = rng.uniform(0.2, 1.0)
    low = rng.uniform(0.0, high)
    v = np.full(n, low, dtype=np.float64)
    v[:split] = high
    v += rng.normal(scale=0.01, size=n)
    return np.clip(v, 0.0, None)


def _generate_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a random family and produce a candidate."""
    choice = rng.integers(0, 4)
    if choice == 0:
        v = _random_initial(n, rng)
    elif choice == 1:
        v = _powerlaw_initial(n, rng)
    elif choice == 2:
        v = _exponential_initial(n, rng)
    else:
        v = _piecewise_initial(n, rng)
    return v


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.

    The search combines a parametric sweep, random restarts
    refined by a two‑phase (smooth + exact) gradient descent,
    and a final cheap hill‑climb.  It respects the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.3)

    best_seq = None
    best_score = float("inf")

    def _update_global(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = _fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global best] n={len(seq)} score={sc:.6f}")

    # 0️⃣ Warm‑start from a previously stored construction, if available.
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            n = cand.shape[0]
            cand = _project(cand, math.sqrt(2.0 * n))
            _update_global(cand)
        except Exception:  # pragma: no cover
            pass

    # 1️⃣ Parametric sweep (power‑law & exponential families)
    lengths = [500, 600, 700, 800, 900, 1000, 1200, 1500]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(1, n + 1, dtype=np.float64)

        for p in np.linspace(0.3, 4.0, 25):
            a = idx ** (-p)
            a = _project(a, target)
            _update_global(a)

        for alpha in np.linspace(0.001, 0.5, 25):
            a = np.exp(-alpha * np.arange(n, dtype=np.float64))
            a = _project(a, target)
            _update_global(a)

    # 2️⃣ Random restarts + two‑phase optimisation
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        restarts = 8 if n <= 800 else 4
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            start = _generate_initial(n, rng)
            start = _project(start, target)

            # blend with the best of the same length, if we have one
            if best_seq is not None and best_seq.shape[0] == n:
                start = _project((start + best_seq) / 2.0, target)

            seq_opt, sc_opt = _optimize_two_phase(start, target, deadline, rng)
            _update_global(seq_opt)

            if best_score < 1.503:          # target already met
                print("[early exit] target reached")
                return best_seq.tolist()

    # 3️⃣ Final cheap hill‑climb (random mass swaps)
    if best_seq is not None:
        while time.monotonic() < deadline:
            n = best_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            delta = best_seq[i] * 0.025
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project(cand, math.sqrt(2.0 * n))
            _update_global(cand)
    else:
        # fallback – uniform vector
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    return [float(x) for x in best_seq]
