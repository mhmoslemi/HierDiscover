def _spectral_flatten(seq, target_sum, deadline, rng, max_iters=300):
    n = seq.shape[0]
    # choose FFT length M = next power of two >= 2*n-1
    M = 1 << ((2*n-1).bit_length())
    best = _normalize(seq, target_sum)
    best_score = _fast_score(best)
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        # pad to M
        pad = np.zeros(M, dtype=np.float64)
        pad[:n] = best
        A = np.fft.rfft(pad)
        mag_sq = np.abs(A) ** 2
        target_mag = np.sqrt(mag_sq.mean())
        phases = np.angle(A)
        # keep DC phase zero
        # build new spectrum with constant magnitude
        A_new = target_mag * np.exp(1j * phases)
        # ensure the conjugate symmetry automatically handled by rfft
        new_pad = np.fft.irfft(A_new, n=M)
        new_seq = new_pad[:n]
        # enforce nonnegativity and bound 1000
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        # renormalize
        new_seq = _normalize(new_seq, target_sum)
        new_score = _fast_score(new_seq)
        # Keep improvement
        if new_score < best_score - 1e-12:
            best, best_score = new_seq, new_score
        else:
            # random jitter to escape local minima sometimes
            if rng.random() < 0.05:
                # perturb
                eps = rng.normal(scale=0.01*target_sum, size=n)
                cand = best + eps
                cand = np.clip(cand, 0.0, 1000.0)
                cand = _normalize(cand, target_sum)
                cand_score = _fast_score(cand)
                if cand_score < best_score:
                    best, best_score = cand, cand_score
    return best

def _refine_sequence_multi(seq, target_sum, deadline, rng, max_iters=2000):
    n = seq.shape[0]
    best = _normalize(seq, target_sum)
    best_score = _fast_score(best)
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        conv = np.convolve(best, best)
        max_val = conv.max()
        # identify top active indices within 0.5% of max
        tol = 1e-4 * max_val  # maybe relative tolerance
        active = np.where(conv >= max_val - tol)[0]
        # compute gradient g = sum_{k in active} partner vector
        grad = np.zeros_like(best)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # partner indices
            # For each i in idxs: partner = best[k-i]
            idx = np.arange(i_start, i_end+1, dtype=np.int64)
            grad[idx] += best[k - idx]
        # donor index high gradient, receiver low gradient
        donor = int(np.argmax(grad))
        receiver = int(np.argmin(grad))
        if donor == receiver:
            # random
            donor = int(rng.integers(n))
            receiver = int(rng.integers(n))
            if donor == receiver:
                continue
        # compute epsilon
        eps = min(best[donor] * 0.2, target_sum * 0.001)
        if eps <= 0:
            continue
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_score:
            best, best_score = cand, cand_sc
            continue
        # else try projected gradient similarly...
        # Keep same as original fallback

        # Compute direction: -grad + mean(grad) (keeping sum fixed)
        direction = -grad + grad.mean()
        # compute step bound
        neg = direction < 0
        if np.any(neg):
            max_alpha = np.min(best[neg] / -direction[neg])
        else:
            max_alpha = 1.0
        # step size
        alpha = min(0.05 * max_alpha, 0.02 * target_sum / n)
        cand2 = best + alpha * direction
        cand2 = np.clip(cand2, 0.0, 1000.0)
        cand2 = _normalize(cand2, target_sum)
        cand2_sc = _fast_score(cand2)
        if cand2_sc < best_score:
            best, best_score = cand2, cand2_sc
            continue

        # occasional random Dirichlet injection
        if rng.random() < 0.01:
            cand3 = rng.dirichlet(np.ones(n)) * target_sum
            cand3_sc = _fast_score(cand3)
            if cand3_sc < best_score:
                best, best_score = cand3, cand3_sc
    return best

"""
Step‑function minimisation – spectral‑flattening + focused mass‑transfer.

The routine creates many candidate shapes (uniform, exponential,
Beta‑family, power‑law, Dirichlet draws).  Each candidate is first
“flattened” by an alternating‑projection scheme that enforces a flat
magnitude spectrum (hence a near‑constant auto‑convolution).  After the
spectral stage a cheap sub‑gradient / mass‑transfer refinement reduces
the active convolution entry, followed by a short stochastic polish.
All operations respect the security checks of ``evaluate_sequence``.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``vec`` to have ℓ¹‑norm exactly ``target_sum`` (preserving non‑negativity)."""
    s = vec.sum()
    if s <= 0.0:
        n = vec.shape[0]
        return np.full(n, target_sum / n, dtype=np.float64)
    return vec * (target_sum / s)


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` without the safety wrapper.
    For sequences normalised to √(2·n) this equals the maximal auto‑convolution entry.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Families of analytic starting shapes
# ----------------------------------------------------------------------


def _beta_family(n: int, a: float, b: float, target_sum: float) -> np.ndarray:
    """Beta‑distribution shape a_i ∝ x^{a‑1} (1‑x)^{b‑1},  x∈(0,1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    w = np.maximum(w, 1e-12)
    w = w / w.sum() * target_sum
    return w


# ----------------------------------------------------------------------
# Spectral‑flattening (alternating‑projection) step
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _spectral_flatten(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 200,
) -> np.ndarray:
    """
    Repeatedly enforce a flat magnitude spectrum (|FFT| constant) while
    keeping the vector non‑negative and ℓ¹‑normalised.
    """
    n = seq.shape[0]
    M = _next_pow2(2 * n - 1)               # length used for the FFT (zero‑padded)
    best = _normalize(seq, target_sum)
    best_sc = _fast_score(best)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # zero‑pad, FFT, flatten magnitude
        pad = np.zeros(M, dtype=np.float64)
        pad[:n] = best
        A = np.fft.rfft(pad)                # complex spectrum (real input)
        mag_sq = np.abs(A) ** 2
        const_mag = np.sqrt(mag_sq.mean())  # target magnitude (same power)
        phases = np.angle(A)
        A_new = const_mag * np.exp(1j * phases)

        # inverse FFT, keep first n entries
        new_pad = np.fft.irfft(A_new, n=M)
        new_seq = new_pad[:n]

        # enforce non‑negativity, clamp to the allowed range, renormalise
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        new_seq = _normalize(new_seq, target_sum)

        new_sc = _fast_score(new_seq)
        if new_sc < best_sc - 1e-12:
            best, best_sc = new_seq, new_sc
            continue

        # occasional random jitter to escape tiny plateaus
        if rng.random() < 0.03:
            eps = rng.normal(scale=0.02 * target_sum, size=n)
            cand = best + eps
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _normalize(cand, target_sum)
            cand_sc = _fast_score(cand)
            if cand_sc < best_sc:
                best, best_sc = cand, cand_sc

    return best


# ----------------------------------------------------------------------
# Focused refinement – mass transfer + sub‑gradient step
# ----------------------------------------------------------------------


def _refine_sequence_multi(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 5000,
) -> np.ndarray:
    """
    Heuristic that reduces the maximal auto‑convolution entry.
    It repeatedly identifies a (small) set of active convolution indices,
    builds a sub‑gradient (sum of partner values) and moves a little
    amount of mass from the largest‑gradient position to the smallest.
    """
    n = seq.shape[0]
    best = _normalize(seq, target_sum)
    best_sc = _fast_score(best)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(best, best)
        max_val = conv.max()
        # indices whose value is close to the maximum (relative tolerance)
        tol = max_val * 1e-4
        active = np.where(conv >= max_val - tol)[0]

        # build sub‑gradient: sum of partner entries over all active ks
        grad = np.zeros_like(best)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            grad[idx] += best[k - idx]

        # donor = index with largest gradient, receiver = smallest
        donor = int(np.argmax(grad))
        receiver = int(np.argmin(grad))
        if donor == receiver:
            donor = int(rng.integers(n))
            receiver = int(rng.integers(n))
            if donor == receiver:
                continue

        # propose a tiny mass transfer
        eps = min(best[donor] * 0.2, target_sum * 0.001)
        if eps <= 0.0:
            continue
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc - 1e-12:
            best, best_sc = cand, cand_sc
            continue

        # fallback – projected gradient step that keeps the ℓ¹‑norm fixed
        direction = -grad + grad.mean()
        neg = direction < 0
        if np.any(neg):
            max_alpha = np.min(best[neg] / -direction[neg])
        else:
            max_alpha = 1.0
        alpha = min(0.07 * max_alpha, 0.03 * target_sum / n)
        cand = best + alpha * direction
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc - 1e-12:
            best, best_sc = cand, cand_sc
            continue

        # occasional random Dirichlet injection (helps escape deep basins)
        if rng.random() < 0.01:
            cand = rng.dirichlet(np.ones(n)) * target_sum
            cand_sc = _fast_score(cand)
            if cand_sc < best_sc:
                best, best_sc = cand, cand_sc

    return best


# ----------------------------------------------------------------------
# Cheap stochastic polishing
# ----------------------------------------------------------------------


def _swap_hill_climb(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """Very cheap hill‑climb that swaps a tiny delta between two indices."""
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target_sum * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
    return best


def _gaussian_local_search(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """Short Gaussian perturbation loop (annealing‑style)."""
    best = seq.copy()
    best_sc = _fast_score(best)
    step = target_sum * 0.02

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape[0])
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
            step *= 0.9
        else:
            step *= 0.99
            if step < 1e-6:
                step = target_sum * 0.02
    return best

# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The strategy mixes analytic families, an FFT‑based spectral‑flattening
    stage, a focussed mass‑transfer refinement, and final stochastic polishing.
    """
    rng = np.random.default_rng(seed)
    # stop a little before the official deadline to avoid a hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.3)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Seed from a previously discovered construction (if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = _normalize(init, target0)
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad sweep over several lengths (larger n → more flexibility)
    # ------------------------------------------------------------------
    lengths = [2000, 1500, 1200, 1000, 800, 600, 500, 400, 300, 200]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # --------------------
        # generate start vectors
        # --------------------
        candidates = []

        # uniform
        candidates.append(np.full(n, target / n, dtype=np.float64))

        # arcsine / Beta(½,½)
        candidates.append(_beta_family(n, 0.5, 0.5, target))

        # exponential tail
        alpha = 0.04
        exp_vec = np.exp(-alpha * np.arange(n, dtype=np.float64))
        candidates.append(_normalize(exp_vec, target))

        # power‑law decay
        idx = np.arange(1, n + 1, dtype=np.float64)
        pow_vec = idx ** -1.2
        candidates.append(_normalize(pow_vec, target))

        # a few random Dirichlet draws (already normalised)
        for _ in range(3):
            candidates.append(rng.dirichlet(np.ones(n)) * target)

        # -------------------------------------------------
        # evaluate the raw fleet and keep the best so far
        # -------------------------------------------------
        for cand in candidates:
            if time.monotonic() >= deadline:
                break
            sc = _fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"[cand] n={n} new best {sc:.6f}")

        # -------------------------------------------------
        # 2️⃣  Spectral flattening stage
        # -------------------------------------------------
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break
        # reserve at most 20 s or a fraction of the remaining budget for this length
        flatten_deadline = time.monotonic() + min(20.0, remaining * 0.5)
        for cand in candidates:
            if time.monotonic() >= flatten_deadline:
                break
            seq = _spectral_flatten(cand, target, flatten_deadline, rng, max_iters=120)
            sc = _fast_score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[spectral] n={n} improved to {sc:.6f}")

        # -------------------------------------------------
        # 3️⃣  Focused refinement (mass‑transfer + sub‑gradient)
        # -------------------------------------------------
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break
        refine_deadline = time.monotonic() + min(30.0, remaining * 0.5)

        # start from the current global best of the *same* length,
        # otherwise fall back to a fresh Dirichlet draw
        if best_seq is not None and best_seq.shape[0] == n:
            start = best_seq.copy()
        else:
            start = rng.dirichlet(np.ones(n)) * target

        refined = _refine_sequence_multi(start, target, refine_deadline, rng)
        sc_ref = _fast_score(refined)
        if sc_ref < best_score:
            best_seq, best_score = refined.copy(), sc_ref
            print(f"[refine] n={n} improved to {sc_ref:.6f}")

        # -----------------------------------------------------------------
        # a couple of cheap random restarts (occasionally opens new basins)
        # -----------------------------------------------------------------
        for restart in range(2):
            if time.monotonic() >= deadline:
                break
            start = rng.dirichlet(np.ones(n)) * target
            sub_deadline = time.monotonic() + min(12.0, deadline - time.monotonic())
            refined = _refine_sequence_multi(start, target, sub_deadline, rng)
            sc = _fast_score(refined)
            if sc < best_score:
                best_seq, best_score = refined.copy(), sc
                print(f"[restart] n={n} run {restart} new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing on the globally best vector
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a modest uniform vector (should never happen)
        n = 300
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # enforce exact ℓ¹‑norm (tiny drift might have accumulated)
    n = best_seq.shape[0]
    target = math.sqrt(2.0 * n)
    best_seq = _normalize(best_seq, target)

    # cheap swap‑based hill‑climb
    best_seq = _swap_hill_climb(best_seq, target, deadline, rng)

    # final short Gaussian perturbation stage (if time permits)
    if time.monotonic() < deadline:
        best_seq = _gaussian_local_search(best_seq, target, deadline, rng)

    final_score = _fast_score(best_seq)
    print(f"[final] best score {final_score:.6f}   length {best_seq.shape[0]}")

    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
