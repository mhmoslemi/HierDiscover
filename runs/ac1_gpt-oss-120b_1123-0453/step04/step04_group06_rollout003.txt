def conv_max(a):
    n = a.shape[0]
    size = 1 << (2*n-1).bit_length() # next power of two > 2n-1 for speed
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
    return conv.max(), conv

"""
subgradient_optimize.py

Search for a low‑value step‑function (non‑negative sequence) that minimises

    2·n·max(a∗a) / (sum(a)²)

The search uses the following ideas:

*   The objective is scale‑invariant, so we keep the ℓ¹‑norm fixed to
    √(2·n) (the “simplex” ∑ a_i = √(2n), a_i ≥ 0).  With this normalisation
    the objective equals the maximum entry of the *linear* auto‑convolution
    of the sequence.

*   The exact sub‑gradient of the maximal convolution entry is easy:
    if k is a maximising index then ∂/∂a_i (a∗a)_k = 2·a_{k‑i}
    (zero when k‑i is out of range).  When several indices attain the
    maximum we average the corresponding gradients.

*   Projected sub‑gradient descent on the simplex (Euclidean projection)
    is performed with a simple decreasing step‑size schedule.  When progress
    stalls we (a) shrink the step, (b) occasionally add a small random
    perturbation to escape flat regions, and (c) restart from new random
    points.

*   Convolutions are computed by FFT (O(n log n)) which makes thousands of
    iterations cheap even for n≈1500.

*   After the main descent we run a short Gaussian hill‑climbing phase to
    polish the best sequence.

The whole procedure respects the supplied time budget (default 1000 s) and
returns a plain Python list of non‑negative floats that can be fed to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    Reference: Duchi et al., “Efficient Projections onto the ℓ1‑Ball”.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # Work on a copy to avoid modifying the input
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be equal
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _linear_conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using FFT (real → real).  Returns an
    array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _eval_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a given sequence a.
    The sum is guaranteed to be positive (otherwise ∞ is returned by
    ``evaluate_sequence`` – we avoid that situation here).
    """
    n = a.shape[0]
    conv = _linear_conv_fft(a)
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_a * sum_a))


def _subgradient_of_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Compute the (averaged) sub‑gradient of the maximal convolution entry.
    For each k ∈ max_idxs the gradient component i is 2·a_{k‑i} when
    0 ≤ k‑i < n, else 0.  The result is the average over all active k.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        grad[i] += 2.0 * a[k - i]
    grad /= max_idxs.size
    return grad


def _subgradient_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 3000,
    step_initial: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max(a*a) (scale‑invariant).
    Returns the best vector seen and its evaluation score.
    """
    rng = np.random.default_rng()
    a = _project_onto_simplex(np.clip(a0, 0.0, 1000.0), target_sum)

    best = a.copy()
    best_score = _eval_score(best)

    step = step_initial
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = _linear_conv_fft(a)
        max_val = conv.max()
        max_idxs = np.where(conv >= max_val - 1e-12)[0]
        grad = _subgradient_of_max(a, max_idxs)

        cand = a - step * grad
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _eval_score(cand)

        if cand_score + 1e-12 < best_score:
            best, best_score = cand.copy(), cand_score
            a = cand
            step = min(step * 1.02, 2.0)
            no_improve = 0
        else:
            step *= 0.7
            no_improve += 1

            # Escape plateaus after many non‑improving steps
            if no_improve >= 150:
                # small random jitter – keep us on the simplex
                jitter = rng.normal(scale=0.05, size=a.shape)
                cand2 = a + jitter
                cand2 = np.clip(cand2, 0.0, 1000.0)
                cand2 = _project_onto_simplex(cand2, target_sum)
                cand2_score = _eval_score(cand2)
                if cand2_score < best_score:
                    a = cand2
                    best, best_score = cand2.copy(), cand2_score
                no_improve = 0
                step = step_initial

        if step < 1e-12:
            break

    return best, best_score


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.2,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian perturbation hill‑climbing that stays on the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _eval_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _eval_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        sigma *= 0.9995
        max_iters -= 1
        if sigma < 1e- nine:
            sigma = sigma_initial
    return best, best_score

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:

                break

    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:

        target_sum = math.sqrt(2.0 * n)
        refined_seq, refined_score = _hill_climb(
            best_seq,
            target_sum,
            deadline,
            sigma_initial=0.2,
            max_iters=8000,
        )
        if refined_score < best_score:
            best_seq, best_score = refined_seq, refined_score
            print(f"[refine] final score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (clipped to the interval required by
    # evaluate_sequence).  The caller will handle the final evaluation.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Should never happen – return a minimal valid sequence
        best_seq = np.full(1, 1.0, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
