def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0,  Σ x_i = s}.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Edge case – return a uniform distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(f: np.ndarray) -> float:

def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where the convolution attains (almost) the maximal value."""
    eps = max_val * 1e-12 + 1e-14
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the worst auto‑convolution entry w.r.t. `f`.
    The gradient is summed over all active offsets.
    """
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        vals = f[j]
        # each term contributes f[i] * f[j]; derivative w.r.t. f[i] is f[j]
        grad[i] += vals
        # and symmetrically w.r.t. f[j]
        grad[j] += f[i]
        # The diagonal i == j has been added twice; correct it:
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= f[i[mask]]
    return grad

def _adam_descent(f0: np.ndarray,

def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Simple temperature‑controlled hill‑climber.
    One move consists of transferring a small amount of mass from one
    coordinate to another (exact L¹‑norm is preserved).
    """
    rng = np.random.default_rng()

        # negative.

        cand[j] += delta

        cand = np.clip(cand, 0.0, 1000.0)

        # already preserved; nevertheless a small numerical drift may
        # appear – re‑project onto the simplex once in a while.

        step_scale *= 0.9999

    return best, best_score
