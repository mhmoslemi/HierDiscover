def _simulated_annealing(start, deadline, init_temp=0.1, decay=0.9995):
    rng = np.random.default_rng()
    a = start.copy()
    best = a.copy()
    best_score = objective(a)
    T = init_temp
    while time.monotonic() < deadline:
        # propose new a: perturbation
        cand = a + rng.normal(scale=T, size=a.shape)
        cand = np.clip(cand, 0, 1000)
        cand = _project_onto_simplex(cand, target_sum=1.0)
        score = objective(cand)
        if score < best_score or rng.random() < np.exp(-(score - best_score)/max(T,1e-12)):
            a = cand
            if score < best_score:
                best = cand
                best_score = score
        T *= decay
        if T < 1e-6:
            T = init_temp
    return best, best_score

We search for a non‑negative step sequence `a = (a₀,…,a_{n‑1})` that minimises  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}\;,
\]

where `*` denotes the discrete (linear) convolution.  
Because the value is scale‑invariant we may enforce `∑a_i = 1` and simply
minimise `2·n·max(conv)`.  

The programme consists of two phases:

1. **Fast Adam optimisation**  
   *The objective is smoothed by a soft‑max over the convolution entries.*  
   For a given length `n` we run several random restarts.  
   Convolutions and the weighted gradient are evaluated **with FFTs**
   ( `O(n log n)` ) – this allows us to try much larger `n` (up to 5000)
   and many more iterations than the reference implementation.

2. **Simulated‑annealing polishing**  
   Starting from the best Adam result we perform a long stochastic
   hill‑climbing run: Gaussian perturbations are accepted with a
   Metropolis rule while the vector is projected back onto the simplex.
   The temperature is slowly decreased, giving a fine‑grained local
   search that often improves the Adam solution.

All operations respect the constraints (non‑negative, `∑a_i = 1`,
clip at `1000`).  The best sequence found before the time budget

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = len(a) + len(b) - 1
    nfft = _next_pow2(n)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Convolution of a vector with itself (full length)."""
    return _conv_fft(a, a)


def _objective_from_vector(a: np.ndarray) -> float:
    """Exact value of the evaluation function (ignoring global clipping)."""
    n = len(a)
    conv = _conv_self_fft(a)
    max_conv = np.max(conv)
    # sum(a) is forced to 1 in our search, but keep the formula general
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    max_c = np.max(conv)
    # shift prevents overflow/underflow, clip extremes for safety
    shifted = (conv - max_c) / beta
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    return exp_shifted / np.sum(exp_shifted)


def _weighted_gradient_fft(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    The formula reduces to a single convolution:
        g = 2 * (rev(weights) * a)[n‑1 : 2n‑1][::-1]
    where * denotes linear convolution.
    """
    n = len(a)
    # reverse the weight vector
    w_rev = weights[::-1]
    # convolution of w_rev (length 2n‑1) with a (length n)
    conv = _conv_fft(w_rev, a)               # length 3n‑2
    # slice n‑1 … 2n‑2 (inclusive) and reverse
    grad_slice = conv[n - 1 : 2 * n - 1][::-1]
    return 2.0 * grad_slice

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_descent_fft(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 50000,
    patience: int = 1200,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max smoothed max‑convolution.
    Returns the best vector seen during the run and its exact score.
    """
    a = a0.copy()
    n = len(a)

    # Adam hyper‑parameters (tuned for this problem)
    lr = 0.07
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective_from_vector(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # ----- smoothed max ------------------------------------------------
        conv = _conv_self_fft(a)
        max_conv = np.max(conv)

        # temperature proportional to current maximum (keeps scale‑free)
        beta = max_conv * 0.06 + 1e-12
        w = _softmax_weights(conv, beta)

        # ----- gradient ----------------------------------------------------
        grad = _weighted_gradient_fft(a, w)

        # ----- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cur_score = _objective_from_vector(a)
        cand_score = _objective_from_vector(a_candidate)

        # accept if it improves the exact objective
        if cand_score < cur_score:
            a = a_candidate

        # keep global best
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            print(f"[Adam] n={n} it={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score


def _simulated_annealing(
    start: np.ndarray,
    deadline: float,
    init_temp: float = 0.15,
    decay: float = 0.9996,
    swap_prob: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑style annealing.
    Proposes Gaussian perturbations (occasionally a random swap) while
    staying on the simplex.  The temperature is geometrically decayed.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _objective_from_vector(cur)
    best = cur.copy()
    best_score = cur_score
    T = init_temp
    n = len(cur)
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = cur + rng.normal(scale=T, size=n)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _objective_from_vector(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                # if we improve we allow a tiny temperature boost
                T = min(init_temp, T * 1.02)

        # geometric cooling
        T *= decay
        if T < 1e-7:
            T = init_temp

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the given time
    budget.  Returns a plain Python list of floats ready for evaluation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the evaluation wrapper
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        if np.sum(seq) == 0:
            seq = np.full_like(seq, 1.0 / len(seq))
        else:
            seq = seq / np.sum(seq)          # normalise to ℓ¹‑norm = 1
        score = _objective_from_vector(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[warm] loaded height_sequence_1, n={len(seq)} score={score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation – Adam descent on a range of lengths
    # --------------------------------------------------------------
    # Larger n gives a theoretical lower bound 2·n/(2·n‑1) → 1,
    # but also makes the problem harder.  We try a generous spectrum.
    lengths = [800, 1200, 1600, 2000, 2500, 3000, 3500, 4000, 4500, 5000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = 1.0

        # number of random restarts (more for smaller n)
        restarts = 8 if n <= 2000 else 5

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a candidate
            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
            else:
                # Dirichlet with concentration 0.7 gives slightly spiky starts
                a0 = rng.dirichlet([0.7] * n).astype(np.float64)

            sub_deadline = min(deadline, time.monotonic() + 12.0)  # ~12 s per run
            a_opt, sc_opt = _adam_descent_fft(a0, target_sum, sub_deadline,
                                              max_iters=50000, patience=1500)

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[update] n={n} restart={rs} score={sc_opt:.6f}")

            # early exit once we are comfortably below the target 1.503
            if best_score < 1.48:
                print("[early] good enough – skipping further restarts")
                break
        if best_score < 1.48:
            break

    # --------------------------------------------------------------
    # Final polishing with simulated annealing (if time permits)
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: modest uniform distribution of size 1000
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective_from_vector(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _simulated_annealing(best_seq, deadline,
                                                      init_temp=0.18,
                                                      decay=0.9995)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # final safety clipping (evaluate_sequence repeats the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best length={len(best_seq)} final score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
