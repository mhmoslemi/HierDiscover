"""
Hybrid optimisation for the step‑function bound.

The optimiser works in three stages:

1️⃣  **Warm‑start** – if a previously discovered sequence is available under the
    name ``height_sequence_1`` it is normalised and taken as the initial
    candidate.

2️⃣  **Linear‑programming refinement** – given a current vector ``f`` (∑f=1)
    we solve the LP

          minimise   M
          subject to  (f * b)[k] ≤ M   for all k
                      Σ b_i = 1,   b_i ≥ 0

    which finds the sequence ``b`` that has the smallest possible maximal
    cross‑convolution with ``f``.  Mixing ``f`` and ``b`` (with a gradually
    shrinking mixing factor) strictly reduces the quantity
    ``2·n·max(conv(f,f))/‖f‖₁²`` in practice.

3️⃣  **Fine‑grained Adam descent + stochastic polishing** – after a few LP
    cycles a short projected Adam optimisation (soft‑max smoothing of the
    max‑convolution) is applied, followed by a tiny stochastic hill‑climbing
    phase.  The best vector seen over the whole budget is finally returned.

All steps respect the global time budget (default 1000 s) and keep the
ℓ¹‑norm exactly 1, which makes the objective invariant to scaling.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(a: np.ndarray) -> float:
    """Exact value of the evaluation function (no internal clipping)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    sum_val = np.sum(a)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / beta
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    return w


def _weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. ``a``.
    ``weights`` must have length 2·len(a)‑1 and sum to 1.
    """
    n = len(a)
    grad = np.zeros_like(a)
    for k, wk in enumerate(weights):
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = a[j]
        grad[i] += wk * 2.0 * vals
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= wk * vals[mask]
    return grad


def _adam_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 20000,
    patience: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max approximation of the max‑convolution.
    """
    a = a0.copy()
    n = len(a)

    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        beta = max_conv * 0.07 + 1e-12           # smoothing proportional to the current max
        w = _softmax_weights(conv, beta)
        grad = _weighted_gradient(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cur_score = _compute_score(a)
        cand_score = _compute_score(a_candidate)

        if cand_score < cur_score:
            a = a_candidate

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            # occasional progress print
            if it % 200 == 0:
                print(f"[Adam] n={n} it={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score


def _stochastic_refinement(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.3,
    swap_prob: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Tiny hill‑climbing phase: Gaussian jitters + occasional swaps,
    always re‑projected onto the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, len(best), size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = best + rng.normal(scale=step, size=best.shape)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _compute_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = min(step_initial, step * 1.02)
        else:
            step *= 0.995
        if step < 1e-6:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming based refinement
# ----------------------------------------------------------------------


def _lp_optimize_cross(f: np.ndarray) -> tuple[np.ndarray | None, float | None]:
    """
    Solve the LP:
        minimise   M
        s.t.       (f * b)[k] ≤ M   for all k
                   Σ b_i = 1,   b_i ≥ 0

    Returns the optimal ``b`` (normalised to sum = 1) and the minimal ``M``.
    If the solver fails, ``(None, None)`` is returned.
    """
    n = len(f)
    rows = 2 * n - 1

    # Build A_ub : each row k contains the coefficients f_{k-j} for b_j
    A = np.zeros((rows, n + 1), dtype=np.float64)   # extra column for -M
    for j in range(n):
        A[j:j + n, j] = f
    A[:, n] = -1.0                       # coefficient of the auxiliary variable M

    b_ub = np.zeros(rows, dtype=np.float64)   # right‑hand side of ≤ constraints

    # Equality: Σ b_i = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0, None)] * n + [(0, None)]   # b_i ≥ 0, M ≥ 0
    c = np.concatenate([np.zeros(n), np.array([1.0])])   # minimise M

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None, None

    x = res.x
    b = x[:n]
    M = x[-1]

    # Re‑normalise b (numerical drift)
    sb = b.sum()
    if sb > 0:
        b = b / sb
    else:
        b = np.full_like(b, 1.0 / n)

    return b, M


def _lp_refinement(
    f: np.ndarray,
    deadline: float,
    max_cycles: int = 10,
    mix_factor: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Repeated LP refinements: each cycle finds a vector ``b`` with a
    strictly smaller cross‑convolution bound and mixes it into ``f``.
    """
    best = f.copy()
    best_score = _compute_score(best)

    for cycle in range(max_cycles):
        if time.monotonic() >= deadline:
            break
        b, M = _lp_optimize_cross(best)
        if b is None:
            break

        # Mix the new direction – the factor shrinks after successful steps.
        new = (1.0 - mix_factor) * best + mix_factor * b
        new = np.clip(new, 0.0, 1000.0)
        new_sum = new.sum()
        if new_sum > 0:
            new = new / new_sum

        new_score = _compute_score(new)
        if new_score + 1e-12 < best_score:
            best = new
            best_score = new_score
            mix_factor = max(0.1, mix_factor * 0.9)   # finer later
        else:
            mix_factor *= 0.5
            if mix_factor < 1e-3:
                break

    return best, best_score

# Public entry point required by the evaluation harness

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within ``budget_s`` seconds.
    Returns a plain ``list[float]`` that will be fed to ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction if present.
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        if seq.sum() == 0.0:
            seq = np.full_like(seq, 1.0 / len(seq))
        else:
            seq = seq / seq.sum()
        sc = _compute_score(seq)
        best_seq = seq.copy()
        best_score = sc
        print(f"[warm] loaded height_sequence_1 n={len(seq)} score={sc:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop – try several lengths & random restarts.
    # --------------------------------------------------------------
    # Lengths where a dense LP (O(n²) memory) is still cheap.
    lengths = [500, 600, 800, 1000, 1200]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = 1.0
        # a few random starts per length
        restarts = 4 if n <= 800 else 2
        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            if best_seq is not None and best_seq.shape[0] == n:
                f = best_seq.copy()
            else:
                # Dirichlet‑like start (concentration 0.5 → sparser vectors)
                f = rng.gamma(0.5, size=n)
                f = f / f.sum()

            # -------- LP refinement --------
            sub_deadline = min(deadline, time.monotonic() + 8.0)
            f, sc = _lp_refinement(f, sub_deadline, max_cycles=8, mix_factor=0.6)

            # -------- Short Adam fine‑tune (optional) --------
            if time.monotonic() < deadline:
                sub_deadline = min(deadline, time.monotonic() + 4.0)
                f_adam, sc_adam = _adam_descent(
                    f, target_sum, sub_deadline, max_iters=8000, patience=800
                )
                if sc_adam < sc:
                    f, sc = f_adam, sc_adam

            if sc < best_score:
                best_score = sc
                best_seq = f.copy()
                print(f"[update] n={n} restart={rs} score={sc:.6f}")

            # early exit if we already beat the target bound
            if best_score <= 1.503:
                print("[goal] target bound reached")
                break
        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # Final stochastic polishing (if time permits).
    # --------------------------------------------------------------
    if best_seq is None:
        # Very fallback – uniform vector of moderate size
        n = 800
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    if time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline, step_initial=0.2)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # safety clipping (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return [float(x) for x in best_seq]
