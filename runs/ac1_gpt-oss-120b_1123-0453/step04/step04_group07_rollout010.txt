    monotone step reduces the objective and quickly drives the maximum
    convolution towards a small plateau.

2️⃣  **Projected Adam** –  The LP phase is followed by a short Adam descent
    on the simplex (Σ a_i = const, a_i ≥ 0) that minimises the worst
    convolution entry.  The sub‑gradient is the sum of the gradients of all
    active (maximising) shifts.

3️⃣  **Simulated annealing** –  Finally a temperature‑controlled hill‑climber
    moves a small amount of mass between two coordinates while keeping the
    L¹‑norm exact, to escape shallow minima and polish the best candidate.

All three stages respect the global time budget (default 1000 s) and keep
the entries inside the interval [0, 1000] required by `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s}.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(f: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` (the clipping performed inside that
    function is irrelevant for the values we generate).
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _max_convolution(f: np.ndarray) -> float:
    """Return the maximum entry of the linear auto‑convolution of f."""
    return np.max(np.convolve(f, f))


# ----------------------------------------------------------------------
# Linear‑programming step (stage 1)
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise   Σ b_j
        s.t.      b_j ≥ 0
                  (f * b)[k] ≤ M   (for all k)
    and return the feasible vector b (renormalised to the same L¹‑mass as f).
    Returns None if the LP fails.
    """
    n = len(f)
    # Build A_ub where row k corresponds to (f * b)[k] ≤ M.
    # (f * b)[k] = Σ_i f_i * b_{k-i}; coefficient for b_j is f_{k-j}
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=np.float64)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over indices of b that appear in this shift
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        row[idx] = f[k - idx]          # f_{k-idx}
        rows.append(row)
    A = np.vstack(rows)                # shape (2n‑1, n)

    # maximise Σ b  ⇔ minimise –Σ b
    c = -np.ones(n, dtype=np.float64)

    # Upper bound for each shift
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Bounds for variables (non‑negative)
    bounds = [(0.0, None)] * n

    # Call the high‑performance HiGHS interior‑point solver
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        return None
    b = res.x
    sum_f = np.sum(f)
    sum_b = np.sum(b)
    if sum_b == 0:
        return None
    # Renormalise to keep the L¹‑mass unchanged (scale invariance of the objective)
    b = b * (sum_f / sum_b)
    return b


def _lp_optimize(f0: np.ndarray,
                 t: float,
                 max_iters: int,
                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based iterative improvement starting from f0.
    Returns the best sequence found and its score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        it += 1
        M = _max_convolution(f)
        b = _lp_step(f, M)
        if b is None:
            break
        # convex combination
        f_candidate = (1.0 - t) * f + t * b
        # keep it feasible w.r.t. the bounds used in `evaluate_sequence`
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cur_score = _score(f)
        cand_score = _score(f_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_f = f_candidate.copy()
            print(f"[LP] it={it:3d} n={len(f)} score={cand_score:.6f}")

        # accept if improves the raw objective
        if cand_score < cur_score:
            f = f_candidate
        else:
            # if we stop improving for a while, shrink the mixing factor
            if it % 10 == 0:
                t *= 0.9
                t = max(t, 0.01)

        # early stop when improvement stalls
        if it % 20 == 0 and best_score < 1.55:
            # a fairly good bound already – move on to fine tuning
            break
    return best_f, best_score


# ----------------------------------------------------------------------
# Projected Adam optimisation (stage 2)
# ----------------------------------------------------------------------


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where the convolution attains (almost) the maximal value."""
    eps = max_val * 1e-12 + 1e-14
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the worst auto‑convolution entry w.r.t. `f`.
    The gradient is summed over all active offsets.
    """
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        # contribution of term a_i * a_j ; d/da_i = a_j
        grad[i] += f[j]
        grad[j] += f[i]
        # diagonal term counted twice – correct
        dup = i == j
        if np.any(dup):
            grad[i[dup]] -= f[i[dup]]
    return grad


def _adam_descent(f0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises the peak auto‑convolution.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.04
    it = 0
    no_improve = 0

    while it < max_iters and time.monotonic() < deadline:
        it += 1
        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)

        grad_max = _gradient_submax(f, active)

        sum_f = np.sum(f)
        grad_obj = (2.0 * len(f) / (sum_f * sum_f)) * grad_max

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = _project_onto_simplex(f_candidate, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cand_score = _score(f_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_f = f_candidate.copy()
            no_improve = 0
            print(f"[Adam] it={it:4d} n={len(f)} score={cand_score:.6f}")

        # accept if improves the raw objective
        cur_score = _score(f)
        if cand_score < cur_score:
            f = f_candidate
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.7
                no_improve = 0

        if no_improve > 800:     # early stopping
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Simulated annealing polishing (stage 3)
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.25,
                         decay: float = 0.9994) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled hill‑climber that moves a small amount of mass
    between two coordinates while keeping the L¹‑norm exact.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    total = np.sum(start)

    T = init_temp
    step_scale = 0.04 * total / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        if it % 256 == 0:                          # occasional re‑projection
            cand = _project_onto_simplex(cand, total)

        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-8:
            T = init_temp

        step_scale *= 0.9999
        if step_scale < 1e-6 * total:
            step_scale = 0.04 * total / n

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence under the
    1000 s time budget.  The routine runs three stages:
    LP‑based polishing → Adam descent → Simulated annealing.
    The best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ If a warm‑start is available use it; otherwise start from uniform.
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64).copy()
        seq = np.clip(seq, 0.0, 1000.0)
        n0 = len(seq)
        if np.sum(seq) == 0.0:
            seq = np.full(n0, 1.0 / n0)
        best_seq = seq
        best_score = _score(seq)
        print(f"[init] warm‑start from height_sequence_1, n={n0}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ LP‑based iterative improvement on a few different lengths.
    # ------------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000]          # moderate sizes – stay fast
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = 1.0                     # we keep the L¹‑mass = 1 (scale‑invariant)

        # initialise a starting point for this length
        if best_seq is not None and len(best_seq) == n:
            f0 = best_seq.copy()
            # add a tiny jitter to escape plateaus
            f0 += rng.normal(scale=0.001, size=n)
            f0 = np.clip(f0, 0.0, None)
        else:
            f0 = rng.random(n)
        f0 = _project_onto_simplex(f0, target_sum)

        # allocate a few seconds for the LP‑phase for this length
        phase_deadline = min(deadline, time.monotonic() + 12.0)

        f_lp, score_lp = _lp_optimize(f0, t=0.25, max_iters=30, deadline=phase_deadline)
        if score_lp < best_score:
            best_score = score_lp
            best_seq = f_lp.copy()
            print(f"[LP‑best] n={n} score={score_lp:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣ Short Adam descent (optional, only if time permits)
        # ------------------------------------------------------------------
        if time.monotonic() < deadline:
            sub_deadline = min(deadline, time.monotonic() + 6.0)
            f_adam, score_adam = _adam_descent(
                f_lp, target_sum, sub_deadline, max_iters=2500
            )
            if score_adam < best_score:
                best_score = score_adam
                best_seq = f_adam.copy()
                print(f"[Adam‑best] n={n} score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final simulated‑annealing polishing (if any time left).
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _simulated_annealing(best_seq, deadline)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final‑anneal] score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Fallback – simple uniform sequence (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] uniform construction used")

    # Clip once more for safety and return as plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
