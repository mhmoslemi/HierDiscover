def project_onto_simplex(v, S):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, S / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    conv = np.convolve(f, f)
    max_val = conv.max()

def _project_onto_simplex(v, s):
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

"""
Hybrid optimizer for the “step‑function bound” problem.

The algorithm works in two stages:

1️⃣  **Low‑dimensional parametric search** – for several lengths we
    generate candidate sequences from a handful of simple parametric
    families (exponential, symmetric exponential, linear, triangular).
    For each family a single scalar (e.g. the decay factor) is swept,
    the sequence is scaled to the required ℓ¹‑norm and the maximum
    entry of the auto‑convolution is evaluated with an FFT (O(n log n)).
    The best among all families and lengths is kept.

2️⃣  **Projected sub‑gradient polishing** – starting from the best
    candidate we run a long projected sub‑gradient descent on the
    objective `max_k (a*a)[k]` (the ℓ∞‑norm of the auto‑convolution).
    The sub‑gradient is computed analytically from the active
    convolution entries, the iterate is projected onto the ℓ¹‑simplex
    `∑a_i = √(2 n)` and clipped to the allowed range [0,1000].
    The step size decays automatically; the loop runs until the
    allocated time budget (≈ 1000 s) expires.

Both stages respect the security checks of ``evaluate_sequence``:
all entries are non‑negative, bounded by 1000 and the ℓ¹‑norm is
kept constant, so the final sequence can be handed directly to the
scoring routine.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return the maximum entry of a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    needed = 2 * len(a) - 1
    return float(conv[:needed].max())


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto {x≥0, ∑x = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgrad(x: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max_k (x*x)[k]``.
    For each active convolution index k we add contributions
    a_i ← a_i + a_j and a_j ← a_j + a_i (i+j = k).
    """
    n = len(x)
    conv = np.convolve(x, x)
    max_val = float(conv.max())
    eps = max_val * 1e-12 + 1e-12
    active = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(x)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        grad[i] += x[j]
        grad[j] += x[i]
    if len(active) > 0:
        grad /= float(len(active))
    return grad

        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)

    # final safety clamp (required by evaluate_sequence)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return [float(v) for v in best_seq]
