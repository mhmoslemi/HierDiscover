def build_constraint_matrix(f):
    n = len(f)
    K = 2*n - 1
    # Use each shift k: row k holds f reversed? Actually row k (0-index) corresponds to conv index k = i + j.
    # For variable g_j, contribution to conv index k is f_{k - j} if 0 <= k - j < n else 0.
    # So A[k, j] = f_{k - j] if 0 <= k - j < n else 0.
    # So we can fill using vectorized operations.
    # Use outer addition of indices: K indices (0..K-1), n indices (0..n-1)
    K = 2*n - 1
    # Use broadcasting: idx_k[:, None] - idx_j[None, :] => shape (K,n).
    # Then mask valid indices: valid = (ind >=0) & (ind < n)
    # Then A = np.where(valid, f[ind], 0)
    # This is O(K n) memory/time each call; for n=500 -> 1000*500=500k entries, fine.
    idx_k = np.arange(K)[:, None]
    idx_j = np.arange(n)[None, :]
    ind = idx_k - idx_j
    valid = (ind >= 0) & (ind < n)
    A = np.where(valid, f[ind], 0.0)
    return A

from scipy.optimize import linprog

def solve_lp_max_sum(f, M):
    n = len(f)
    K = 2*n - 1
    A = build_constraint_matrix(f)
    b = np.full(K, M)
    c = -np.ones(n)   # minimize -sum = maximize sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # Might be infeasible due to numeric precision; return f unchanged
        return None
    g = res.x
    sum_g = g.sum()
    return g, sum_g

def _build_constraint_matrix(f):
    n = len(f)
    K = 2*n - 1
    # indices arrays:
    idx_k = np.arange(K)[:, None]  # shape (K,1)
    idx_j = np.arange(n)[None, :]  # shape (1,n)
    ind = idx_k - idx_j
    # mask valid rows
    mask = (ind >= 0) & (ind < n)
    # use broadcasting to fill A
    A = np.zeros((K, n), dtype=np.float64)
    A[mask] = f[ind[mask]]
    return A
```

def _simulated_annealing(start, deadline, init_temp=0.2, decay=0.9995):
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score
    n = len(start)
    target_sum = 1.0 # sum constraint
    T = init_temp
    step_scale = 0.01 / n  # typical delta
    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonneg, ensure sum close to 1 (floating errors)
        # maybe reproject occasionally
        if it % 256 == 0:
            cand = _project_onto_simplex(cand, 1.0)
        cand_score = _score(cand)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
        T *= decay
        if T < 1e-12:
            T = init_temp
        step_scale *= 0.9999
        if step_scale < 1e-12:
            step_scale = 0.01 / n
    return best, best_score

"""
Hybrid Linear‑Program + Simulated‑Annealing optimizer for the
non‑negative step sequence problem.

* The sequence is considered a probability vector (∑a_i = 1); the
  evaluation score is   2·n·max((a*a)ₖ),  which is scale‑invariant.

* Given a current distribution «f», we solve a linear program
  (LP) that maximises the total mass Σg_i under the constraints
        (f * g)ₖ ≤ M      for every shift k,
  where M = max(f * f).  If the LP can increase the total mass,
  the scaled solution ĝ = g / Σg is mixed back into f.  This
  step provably never raises the peak autocorrelation and often
  reduces it.

* After the LP phase we run a lightweight temperature‑controlled
  simulated‑annealing hill‑climber that moves a small amount of mass
  between two positions while preserving the ℓ¹‑norm.  The annealer
  refines the solution locally and helps to escape shallow plateaus.

* The routine respects a global time budget (default 1000 s) and
  returns the best sequence discovered as a plain Python list.
"""
import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """
    Exact value of the provided ``evaluate_sequence`` (without the
    internal clipping, which is irrelevant for the ranges we generate).
    The vector a is assumed to be the raw step heights (any positive scale
    works because the score is scale‑invariant).
    """
    n = len(a)
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    sum_val = np.sum(a)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2·n‑1, n)) such that
        (f * g)_k = (A @ g)[k] .
    Row k, column j contains f_{k‑j} when the index is valid,
    otherwise zero.
    """
    n = f.shape[0]
    K = 2 * n - 1
    idx_k = np.arange(K)[:, None]          # (K,1)
    idx_j = np.arange(n)[None, :]          # (1,n)
    ind = idx_k - idx_j                     # (K,n)
    mask = (ind >= 0) & (ind < n)           # valid positions
    A = np.zeros((K, n), dtype=np.float64)
    A[mask] = f[ind[mask]]
    return A


# ----------------------------------------------------------------------
# Linear‑program improvement step
# ----------------------------------------------------------------------


def _lp_improve(f0: np.ndarray,
                max_iters: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Starting from a probability vector f0 (∑f = 1, f ≥ 0) perform up to
    ``max_iters`` LP‑based improvement steps.  Each step solves

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k,
                    g_i ≥ 0,

    where M = max(f * f).  If the LP can increase the total mass,
    the scaled solution ĝ = g / Σg is mixed back into f.
    The function returns the best distribution found and its score.
    """
    f = f0.copy()
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    it = 0
    # initial maximal autocorrelation
    conv = np.convolve(f, f)
    M = np.max(conv)

    while it < max_iters and time.monotonic() < deadline:
        it += 1

        # ----- build LP -----
        A = _build_constraint_matrix(f)           # (2n‑1) × n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # minimise –Σg  ⇔ maximise Σg

        lp_res = linprog(c,
                         A_ub=A,
                         b_ub=b_ub,
                         bounds=[(0.0, None)] * n,
                         method='highs',
                         options={'presolve': True, 'disp': False})

        if not lp_res.success:
            # numerical trouble – abort the LP phase
            break

        g = lp_res.x
        sum_g = g.sum()

        # No room for improvement if the LP cannot increase total mass
        if sum_g <= 1.0 + 1e-12:
            break

        # Normalise the LP solution to a probability vector
        g_hat = g / sum_g

        # Try a few mixing coefficients and keep the best candidate
        best_local = None
        best_local_score = float('inf')
        for t in (0.1, 0.2, 0.4):
            cand = (1.0 - t) * f + t * g_hat
            cand_score = _score(cand)
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local = cand

        # Update the current iterate
        f = best_local
        conv = np.convolve(f, f)
        M = np.max(conv)

        # Record the best overall vector
        if best_local_score < best_score - 1e-12:
            best_score = best_local_score
            best_f = f.copy()
            print(f"[LP] it={it:3d} n={n} score={best_local_score:.6f}")

        # Optional early exit if we already passed the target
        if best_score < 1.503:
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9997) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled hill‑climber that preserves the ℓ¹‑norm
    exactly (mass is transferred from one index to another).  The move
    size slowly contracts while the temperature slowly cools.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = 1.0                     # we keep the vector on the probability simplex
    T = init_temp
    step_scale = 0.01 / n                # typical transferred mass

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # occasional re‑projection to the simplex (prevents drift)
        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-12:
            T = init_temp
        step_scale *= 0.9999
        if step_scale < 1e-12:
            step_scale = 0.01 / n

    return best, best_score

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step sequence.
    The routine first tries to warm‑start from any existing global
    variable named ``height_sequence_*``.  It then refines the candidate
    with the LP‑based improvement loop and finishes with a short
    simulated‑annealing polishing phase, all while respecting the
    global time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a tiny safety margin for the caller
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start from any previously discovered construction.
    # --------------------------------------------------------------
    for var_name in ["height_sequence_1", "height_sequence_2",
                     "height_sequence_3"]:
        seq = globals().get(var_name)
        if isinstance(seq, (list, tuple, np.ndarray)):
            seq_arr = np.asarray(seq, dtype=np.float64)
            seq_arr = np.clip(seq_arr, 0.0, 1000.0)
            if seq_arr.sum() == 0.0:
                continue
            # work with the probability version (sum = 1)
            prob = seq_arr / seq_arr.sum()
            sc = _score(prob)
            if sc < best_score:
                best_score = sc
                best_seq = prob.copy()
                print(f"[init] warm‑start from {var_name}, n={len(prob)}, "
                      f"score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Fallback – uniform distribution if nothing else is available.
    # --------------------------------------------------------------
    if best_seq is None:
        n0 = 500
        best_seq = np.full(n0, 1.0 / n0, dtype=np.float64)
        best_score = _score(best_seq)
        print(f"[fallback] uniform start, n={n0}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Main optimisation loop (LP → annealing) on the current length.
    # --------------------------------------------------------------
    n = len(best_seq)
    # allocate most of the remaining budget to the LP phase, keep a
    # small reserve for the final annealing.
    lp_deadline = min(deadline, time.monotonic() + 0.75 * (deadline - time.monotonic()))
    best_seq, best_score = _lp_improve(best_seq,
                                       max_iters=500,
                                       deadline=lp_deadline)

    # --------------------------------------------------------------
    # 4️⃣ Final polishing with simulated annealing (if time permits).
    # --------------------------------------------------------------
    if time.monotonic() < deadline:
        refined_seq, refined_score = _simulated_annealing(best_seq,
                                                         deadline,
                                                         init_temp=0.2,
                                                         decay=0.9997)
        if refined_score < best_score:
            best_seq, best_score = refined_seq, refined_score
            print(f"[final] annealed score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣ Return a plain Python list (any positive scaling works).
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)          # safety clamp
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
