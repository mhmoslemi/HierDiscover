from scipy.linalg import toeplitz
from scipy.optimize import linprog

def _conv_matrix(f):
    n = len(f)
    c = np.concatenate([f, np.zeros(n-1, dtype=f.dtype)])
    r = np.concatenate([f[:1], np.zeros(n-1, dtype=f.dtype)])
    return toeplitz(c, r)

def _lp_optimize(f_initial, max_iter, deadline):
    n = len(f_initial)
    S = math.sqrt(2.0 * n)
    f = f_initial.copy()
    best_f = f.copy()
    best_M = np.convolve(f, f).max()
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        # compute bound
        M = np.convolve(f, f).max()
        # build matrix
        A = _conv_matrix(f)
        # prepare LP
        c = -np.ones(n)
        # bounds
        bounds = [(0.0, 1000.0) for _ in range(n)]
        b_ub = np.full(2*n - 1, M, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})
        if not res.success:
            break
        g = res.x
        if g.sum() <= 1e-12:
            break
        # scale g to have ℓ1 = S
        g = g * (S / g.sum())
        # combine
        t = 0.5
        new_f = (1 - t) * f + t * g
        # enforce nonnegativity and renormalize
        new_f[new_f < 0] = 0.0
        total = new_f.sum()
        if total > 0:
            new_f = new_f * (S / total)
        else:
            new_f = np.full(n, S / n, dtype=np.float64)
        new_M = np.convolve(new_f, new_f).max()
        # accept if improved
        if new_M < best_M:
            best_M = new_M
            best_f = new_f.copy()
            f = new_f
        else:
            # try pure g
            g_M = np.convolve(g, g).max()
            if g_M < best_M:
                best_M = g_M
                best_f = g.copy()
                f = g
            else:
                # no improvement -> break
                break
    return best_f, best_M

def _lp_refine(init_seq, deadline, rng, max_iter=8):
    f = init_seq.copy()
    n = len(f)
    S = math.sqrt(2.0 * n)
    best_f = f.copy()
    best_M = _fast_score(f)
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        # compute M from current f
        M = _fast_score(f)
        # build convolution matrix
        A = _conv_matrix(f)  # shape (2n-1, n)
        # linear program
        c = -np.ones(n)
        bounds = [(0.0, 1000.0) for _ in range(n)]
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        if g.sum() <= 1e-12:
            break
        # normalize g to sum S
        g = g * (S / g.sum())
        # combine via mixing factor
        t = 0.5
        new_f = (1 - t) * f + t * g
        # re-normalize to sum S (to correct numerical errors)
        new_f = _normalize(new_f)
        new_M = _fast_score(new_f)
        if new_M < best_M - 1e-6:
            best_M = new_M
            best_f = new_f.copy()
            f = new_f
        else:
            # try using g directly
            g_M = _fast_score(g)
            if g_M < best_M -1e-6:
                best_M = g_M
                best_f = g.copy()
                f = g
            else:
                # no progress -> break
                break
    return best_f, best_M

"""
Propose a low‑value step sequence for the given evaluation function.

The algorithm works in three phases:

0️⃣ *Warm‑start*: use an existing global construction if it is present.
1️⃣ *Fast parametric sweep*: generate a large set of power‑law and exponential
   candidates for several lengths and keep the best (objective = max(f∗f)
   after ℓ¹‑normalisation, which equals the true score).
2️⃣ *Linear‑programming refinement*: for the current best length run a small
   alternating scheme.  Given a sequence f we solve a linear programme that
   maximises ∑g_j subject to the linear constraints   (f∗g)_k ≤ M,
   where M = max(f∗f).  The solution g is scaled to the required ℓ¹‑norm,
   mixed with f and accepted if the maximal auto‑convolution decreases.
   A few iterations of this LP‑mix are cheap and often lower the bound
   significantly (it is a different optimisation strategy from the previous
   gradient‑descent code).

Finally a cheap random mass‑swap hill‑climb is performed until the time
budget expires.  The best sequence found is returned as a plain list of
floats.
"""

import time
import math
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_fast_len(n: int) -> int:
    """Return a good length for FFTs (fallback to power of two)."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _fast_score(seq: np.ndarray) -> float:
    """Exact objective after ℓ¹‑normalisation: max(f∗f)."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and enforce ℓ¹‑norm = √(2·n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / total)
    return seq


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix A such that A @ g = f ∗ g (full convolution).
    """
    n = f.shape[0]
    # first column = f followed by zeros, first row = [f0, 0, …, 0]
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return toeplitz(c, r)


def _lp_refine(init_seq: np.ndarray,
               deadline: float,
               max_iter: int = 8) -> tuple[np.ndarray, float]:
    """
    LP‑based refinement for a fixed length.
    Starting from ``init_seq`` we repeatedly:
      * compute the current bound M = max(f∗f)
      * solve a linear programme that maximises the ℓ¹‑mass of g under
        the linear constraints (f∗g)_k ≤ M
      * scale g to the required ℓ¹‑norm and mix it with f.
    The best sequence encountered (lowest max‑auto‑convolution) is returned.
    """
    f = init_seq.copy()
    n = f.shape[0]
    target = math.sqrt(2.0 * n)

    best_f = f.copy()
    best_M = _fast_score(f)

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        # current bound
        M = _fast_score(f)

        # build the convolution matrix for the linear constraints
        A = _conv_matrix(f)                # shape (2n‑1, n)

        # linear programme: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        # bounds 0 ≤ g_i ≤ 1000  (upper bound keeps the LP bounded)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        # right‑hand side of the inequalities (same bound for every row)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={"presolve": True})

        if not res.success:
            # Something went wrong – stop the refinement for this length.
            break

        g = res.x
        if g.sum() <= 1e-12:
            break

        # renormalise g to the required ℓ¹‑norm
        g = g * (target / g.sum())

        # mix the new candidate with the current one (convex combination)
        t = 0.5
        new_f = (1.0 - t) * f + t * g
        new_f = _normalize(new_f)

        new_M = _fast_score(new_f)

        # accept if improvement, otherwise try g alone
        if new_M < best_M - 1e-8:
            best_M = new_M
            best_f = new_f.copy()
            f = new_f
        else:
            g_M = _fast_score(g)
            if g_M < best_M - 1e-8:
                best_M = g_M
                best_f = g.copy()
                f = g
            else:
                # no progress – exit early
                break

    return best_f, best_M


def _update_global(seq: np.ndarray,
                   best_seq_ref: list,
                   best_score_ref: list) -> None:
    """Compare *seq* with the current global best and keep it if better."""
    sc = _fast_score(seq)
    if sc < best_score_ref[0]:
        best_score_ref[0] = sc
        best_seq_ref[0] = seq.copy()
        print(f"[global best] n={len(seq)} score={sc:.6f}")

            start_seq = _normalize(start_seq)

        # several restarts of the LP refinement
        restarts = 4
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break
            refined_seq, refined_score = _lp_refine(start_seq, deadline, max_iter=8)
            _update_global(refined_seq, best_seq, best_score)

            # for the next restart use the refined sequence as a new start
            start_seq = refined_seq

    # ------------------------------------------------------------------
    # 3️⃣  Final cheap hill‑climb (random mass swaps)
    # ------------------------------------------------------------------
    if best_seq[0] is not None:
        while time.monotonic() < deadline:
            seq = best_seq[0]
            n = seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            delta = seq[i] * 0.03
            cand = seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _normalize(cand)
            _update_global(cand, best_seq, best_score)
    else:
        # fallback – uniform vector of moderate length
        n = 200
        uniform = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best_seq[0] = uniform
        best_score[0] = _fast_score(uniform)

    # Return a plain Python list as required by the outer harness
    return [float(x) for x in best_seq[0]]
