def solve_lp(a, M):
    n = len(a)
    # Build matrix A_ub shape (2n-1, n)
    # for k in range(2n-1):
    #   for j in range(n):
    #       i = k - j
    #       if 0 <= i < n: A_ub[k, j] = a[i]
    # else 0
    #
    # This may be done with convolution: For each column j, a shifted.
    # easiest: Use loops.
    #
    # variables: g_j, bounds (0, None)
    # objective: maximize sum(g) => in linprog we minimize -sum(g)
    obj = np.full(n, -1.0)  # minimize -sum => maximize sum
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # i indexes a, j = k - i
        # So for each i, set A_ub[k, j] = a[i]
        for i in range(i_start, i_end+1):
            j = k - i
            A_ub[k, j] = a[i]
    b_ub = np.full(2*n-1, M)
    res = linprog(c=obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None
    g = res.x
    return g

The optimizer works on a *scale‑invariant* formulation: we keep the ℓ¹‑norm of the step heights equal to 1 (any positive scaling yields the same evaluation value).  
For a given length `n` we perform three stages  

1. **Random warm‑starts** – generate a few initial non‑negative vectors (uniform, Dirichlet, or the previous best).  
2. **Projected sub‑gradient descent** – minimise the worst auto‑convolution entry using the exact sub‑gradient of the max‑function and a projection onto the ℓ¹‑simplex.  
3. **LP‑refinement** – with the current vector `f` we solve a linear programme  

```
maximise    Σ g_j                (total ℓ¹‑mass)
subject to  (f * g)_k ≤ M   for every k = 0 … 2n‑2
            g_j ≥ 0
```

where `M = max_k (f * f)_k`.  
The LP gives a direction `g` that can carry more mass without increasing the current worst convolution value.  
We mix the old and the new vector (`f ← (1‑t)·f + t·g`) with a modest step size `t` and repeat a few times.  
All steps are projected back onto the simplex to keep the sum exactly 1 and the entries non‑negative.  

The whole search is time‑bounded (default 1000 s).  Several lengths are tried (including the length of the previous best solution) and the best vector found is returned.

```python
"""
Hybrid optimizer for the step‑function bound.

* Random warm‑starts (uniform / Dirichlet).
* Projected sub‑gradient descent on the max‑convolution.
* Linear‑programming refinement (maximize ℓ¹‑mass under the current
  worst‑convolution bound) and convex combination update.

All vectors are projected onto the ℓ¹‑simplex (sum = 1, non‑negative),
because the evaluation function is scale‑invariant.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution a*a."""
    return float(np.max(np.convolve(a, a)))


def _objective(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence (scale is irrelevant)."""
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    return float(2.0 * len(a) * _conv_max(a) / (s * s))


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the ℓ¹‑simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)_k.
    For each active k (where the max is attained) we add the derivative
    of the corresponding convolution entry: ∂/∂a_j (Σ_i a_i a_{k‑i})
    = 2·a_{k‑j} (zero if the index is out of range).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    eps = max_val * 1e-9 + 1e-12
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        # derivative contribution 2·a_j
        grad[i] += 2.0 * a[j]
    return grad


def _solve_lp_max_mass(f: np.ndarray, M: float):
    """
    Solve the linear programme
        maximise   Σ g_j
        subject to (f * g)_k ≤ M   for all k
                     g_j ≥ 0
    Returns the optimal g (as a NumPy array) or None if the solver fails.
    """
    try:
        from scipy.optimize import linprog
    except Exception:          # scipy not available – skip the LP step
        return None

    n = f.shape[0]
    m = 2 * n - 1

    # Build the constraint matrix A_ub where A_ub[k, j] = f_{k‑j}
    # (i.e. column j is f shifted by j positions)
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f  # place the whole vector f into rows j … j+n‑1

    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # minimise –Σg  ⇔ maximise Σg

    # “highs” is fast and reliable for dense problems of this size
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if res.success:
        return res.x
    return None


def _gradient_descent(a: np.ndarray, steps: int, step0: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on max‑convolution.
    Keeps Σa = 1 throughout.
    """
    target = np.sum(a)                     # should be 1.0
    for it in range(steps):
        g = _subgradient_max_conv(a)
        # step size decays geometrically
        step = step0 * (0.95 ** it)
        a = _project_onto_simplex(a - step * g, target)
    return a

# ----------------------------------------------------------------------

    an LP‑based mass‑maximisation step, and returns the best vector found.
    """

    # 2️⃣ main loop over lengths
    # --------------------------------------------------------------
    for n in length_candidates:
        if time.monotonic() >= deadline:

        # a few independent restarts for each length
        restarts = 4
        for rs in range(restarts):
            if time.monotonic() >= deadline:

            if isinstance(h1, (list, tuple, np.ndarray)) and len(h1) == n:
```
