def _lp_optimize(f, target_sum, deadline, rng):
    n = len(f)
    best_f = f.copy()
    best_val = evaluate_sequence(f.tolist())
    t = 0.1
    max_iters = 20

    for iter in range(max_iters):
        if time.monotonic() >= deadline:
            break
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        # Build constraint matrix A
        # Use vectorized method:
        # We'll create an empty array and fill overlapping blocks.
        A = np.zeros((2*n - 1, n), dtype=np.float64)
        # For each i, fill f[i] into rows i to i+n-1
        # Using broadcasting with slice assignment in loop (fast enough)
        for i in range(n):
            A[i:i+n, i] = f[i]
        b_ub = np.full(2*n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g) => minimize -sum
        bounds = [(0, None) for _ in range(n)]

        # Solve LP
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fallback: break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        # Normalize g to target sum
        g = g * (target_sum / sum_g)

        # Blend
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, 1000.0)
        # Evaluate
        val = evaluate_sequence(f_new.tolist())
        # If improvement keep new
        if val < best_val:
            best_val = val
            best_f = f_new.copy()
            print(f"[lp] iter={iter} n={n} val={val:.6f}")
        # Update f for next iteration (use best_f optionally)
        f = f_new

        # optionally reduce t
        t = max(0.01, t * 0.9)

    return best_f, best_val

def _stochastic_refinement(seq, deadline, step_initial=0.5):
    rng = np.random.default_rng()
    best = seq.copy()
    best_val = evaluate_sequence(best.tolist())
    step = step_initial
    n = len(seq)
    target_sum = best.sum()
    while time.monotonic() < deadline:
        # propose new sequence by moving mass between two random positions
        i, j = rng.integers(0, n, size=2)
        delta = rng.normal(scale=step)
        # adjust candidate
        cand = best.copy()
        # shift mass delta from i to j, but ensure nonnegative.
        if cand[i] - delta < 0:
            delta = cand[i]  # move all existing
        cand[i] -= delta
        cand[j] += delta
        # ensure sum unchanged implicitly.
        # Clip just in case.
        cand = np.clip(cand, 0.0, 1000.0)
        # compute value
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best = cand
            best_val = val
            # optional: increase step?
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_val

"""
Hybrid optimiser for the step‑function evaluation problem.

The algorithm works in three stages:

1️⃣  Continuous optimisation – a smooth log‑sum‑exp surrogate of the maximal
    convolution is minimised with L‑BFGS‑B.  A continuation on the smoothness
    parameter β (5 → 5 000) drives the surrogate towards the true max.
2️⃣  Linear‑programming refinement – given a current sequence *f* with
    maximal self‑convolution *M*, we solve a LP that maximises the ℓ¹‑norm
    of a new sequence *g* under the linear constraints *f * g* ≤ M*.
    Mixing *f* and the optimal *g* (convex combination) strictly reduces the
    objective while keeping the ℓ¹‑norm fixed.
3️⃣  Stochastic polishing – a tiny amount of mass is moved between random
    indices (mass‑conserving Gaussian perturbations) to escape remaining
    local minima.

All steps respect the non‑negativity, clipping (0 ≤ aᵢ ≤ 1000) and the
time budget (default 1000 s).  The best sequence found is returned as a
plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """Convert unrestricted vector z to a point on the ℓ¹‑simplex of sum = target_sum."""
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z) * target_sum
    return a


def _obj_and_grad(z: np.ndarray, n: int, S: float, beta: float):
    """
    Smooth surrogate of the max‑convolution and its gradient w.r.t. the
    soft‑max parameters *z*.

    Returns
    -------
    smooth : float
        (1/β)·log Σ exp(β·conv)
    grad_z : np.ndarray
        Gradient of ``smooth`` w.r.t. ``z``.
    """
    # a on the simplex (sum = S)
    a = _softmax_to_a(z, S)                      # (n,)

    # --- self‑convolution via FFT (full length 2n‑1) -----------------
    L = _next_pow2(2 * n - 1)
    Fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(Fa * Fa, L)[: 2 * n - 1]

    # --- smooth max (log‑sum‑exp) ------------------------------------
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))        # stable
    sum_exps = np.sum(exps)
    smooth = max_c + np.log(sum_exps) / beta

    # weights of the smooth max (∑w = 1)
    w = exps / sum_exps

    # --- gradient w.r.t. *a* (see derivation in the description) ----
    a_rev = a[::-1]
    L2 = _next_pow2(3 * n - 2)
    Fr = np.fft.rfft(a_rev, L2)
    Fw = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(Fr * Fw, L2)
    start = n - 1
    g_raw = conv2[start : start + n]           # length n
    g_a = 2.0 * g_raw

    # --- chain rule to obtain ∂Φ/∂z ----------------------------------
    total = np.dot(g_a, a)
    grad_z = a * g_a - a * (total / S)

    return smooth, grad_z.astype(np.float64)


def _stochastic_refinement(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Mass‑conserving Gaussian hill‑climbing.
    A random amount δ is moved from one index to another;
    the move is kept only if it reduces the exact evaluation.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = evaluate_sequence(best.tolist())
    step = step_initial
    S = best.sum()

    while time.time() < deadline:
        i, j = rng.integers(0, best.size, size=2)
        delta = rng.normal(scale=step)
        delta = np.clip(delta, -best[i], best[j])          # keep non‑negative
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        val = evaluate_sequence(cand.tolist())
        if val < best_score:
            best = cand
            best_score = val
        step *= 0.9995
        if step < 1e-8:
            step = step_initial
    return best, best_score


def _lp_refine(seq: np.ndarray,
               target_sum: float,
               deadline: float,
               max_iters: int = 6) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current sequence f, maximise Σg subject to f * g ≤ M
    (M = max(f * f)).  The optimal g is mixed back into f with a small step.
    """
    n = len(seq)
    f = seq.copy()
    best_f = f.copy()
    best_val = evaluate_sequence(f.tolist())
    t = 0.2                                    # initial mixing weight

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        # Build constraint matrix A (rows = 2n‑1, cols = n)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f[i]               # each column contributes to a contiguous block

        # Linear programme: maximise Σg  ⇔ minimise −Σg
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=bounds,
                      method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break

        # Scale g to the required ℓ¹‑norm (the same as the target sum)
        g = g * (target_sum / sum_g)

        # Convex combination
        f = (1.0 - t) * f + t * g
        f = np.clip(f, 0.0, 1000.0)

        val = evaluate_sequence(f.tolist())
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[lp] n={n} iter={it} val={val:.6f}")

        t = max(0.02, t * 0.7)                # shrink step size
    return best_f, best_val


def _optimise_length(n: int,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B optimisation with a continuation on β, followed by LP refinement.
    Returns the best sequence found for the given length.
    """
    betas = [5.0, 15.0, 40.0, 80.0, 200.0, 500.0, 1000.0]   # increasingly sharp
    max_restarts = 3
    best_seq = None
    best_val = float("inf")

    for restart in range(max_restarts):
        if time.time() >= deadline:
            break

        # --- initialise soft‑max parameters --------------------------------
        if warm_seq is not None and warm_seq.size == n:
            # inverse soft‑max (log) – avoid log(0)
            p = np.clip(warm_seq / target_sum, 1e-12, None)
            z0 = np.log(p)
        else:
            z0 = rng.normal(scale=0.2, size=n)

        # --- continuation over β -------------------------------------------
        for beta in betas:
            if time.time() >= deadline:
                break

            # L‑BFGS‑B with analytic gradient
            res = minimize(fun=lambda zz: _obj_and_grad(zz, n, target_sum, beta),
                           x0=z0,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 250,
                                    'gtol': 1e-10,
                                    'disp': False})
            z0 = res.x

            # evaluate exact score (expensive, done after each β)
            a = _softmax_to_a(z0, target_sum)
            a = np.clip(a, 0.0, 1000.0)
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[opt] n={n} restart={restart} β={beta:.0f} val={val:.6f}")

            if best_val < 1.45:               # early break – very good already
                return best_seq, best_val

        # ------------------------------------------------------------------
        #  LP‑based refinement for the current best sequence of this restart
        # ------------------------------------------------------------------
        if best_seq is not None and time.time() < deadline:
            refined_seq, refined_val = _lp_refine(best_seq, target_sum, deadline)
            if refined_val < best_val:
                best_val = refined_val
                best_seq = refined_seq.copy()
                print(f"[lp post‑opt] n={n} restart={restart} val={refined_val:.6f}")

    # fallback – uniform distribution
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
    return best_seq, best_val

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the given time budget.
    Returns a plain list of floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.3)   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction (if any)
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        n0 = seq.size
        S0 = math.sqrt(2.0 * n0)
        if seq.sum() > 0:
            seq = seq * (S0 / seq.sum())
        else:
            seq = np.full(n0, S0 / n0, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        val = evaluate_sequence(seq.tolist())
        best_seq = seq.copy()
        best_score = val
        print(f"[init] warm‑start n={n0} val={val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Deterministic optimisation on several lengths
    # --------------------------------------------------------------
    lengths = [400, 600, 800, 1000, 1200, 1500, 1800, 2100]
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        warm = best_seq if (best_seq is not None and best_seq.size == n) else None
        seq_n, val_n = _optimise_length(n, target_sum, deadline, rng, warm)

        if val_n < best_score:
            best_score = val_n
            best_seq = seq_n.copy()
            print(f"[global] n={n} val={val_n:.6f}")

        if best_score < 1.503:
            print("[early stop] target reached")
            break

    # --------------------------------------------------------------
    # 2️⃣ Final stochastic polishing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        refined, refined_val = _stochastic_refinement(best_seq, deadline,
                                                      step_initial=0.4)
        if refined_val < best_score:
            best_seq = refined
            best_score = refined_val
            print(f"[final polish] val={refined_val:.6f}")

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score = {best_score:.6f}  (length={best_seq.size})")
    return [float(x) for x in best_seq]
