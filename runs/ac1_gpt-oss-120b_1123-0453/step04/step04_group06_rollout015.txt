def _lp_mix_optimize(f0, target_sum, deadline, max_outer=5, tau=0.2):
    import cvxpy as cp
    n = len(f0)
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)
    # Pre-allocate matrix A once per outer iteration because f changes
    for outer in range(max_outer):
        if time.monotonic() >= deadline:
            break
        # Compute convolution f*f
        conv = np.convolve(f, f)
        M = conv.max()
        # Build matrix A (2n-1 x n)
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = f  # vectorized broadcasting
        # Solve LP
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception as e:
            # If solver fails, break
            print(f"[lp] solver error: {e}")
            break
        if g.value is None:
            print("[lp] no solution")
            break
        g0 = np.maximum(g.value, 0.0)
        sum_g = g0.sum()
        if sum_g == 0:
            break
        scale = target_sum / sum_g
        g_scaled = g0 * scale
        # Mix
        f = (1 - tau) * f + tau * g_scaled
        # Project onto simplex
        f = _project_onto_simplex(f, target_sum)
        f = np.clip(f, 0.0, 1000.0)
        sc = _compute_score(f)
        if sc < best_score:
            best_f, best_score = f.copy(), sc
        # optionally decay tau
        tau *= 0.9
    return best_f, best_score

The algorithm mixes three ideas:

1.  Projected sub‑gradient descent on the true max (no smoothing).
    The gradient of the active convolution offset k is
    ∂(f*f)_k / ∂f_i = 2·f_{k‑i} (0 ≤ i ≤ n‑1, 0 ≤ k‑i < n).

    scaled to the required ℓ¹‑norm and mixed with f.

3.  A final short Gaussian hill‑climbing phase.

Many random restarts with different initial shapes (uniform,

The best sequence found is returned as a plain Python list.

    if s < 0:
        raise ValueError("target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_score(v: np.ndarray) -> float:
    """Exact score = max convolution (the factor 2 n / sum² cancels)."""
    conv = np.convolve(v, v)
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:
        return float("inf")
    return float(max_val)


# ----------------------------------------------------------------------
# 1️⃣  Sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def _max_subgradient(a: np.ndarray):
    """Return (grad, index_of_max, max_value) for the active convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    grad[idx] = 2.0 * a[k - idx]          # d/d a_i of (f*f)_k
    return grad, k, conv[k]


def _subgrad_optimize(
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 3000,
    step_init: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max_i (f*f)_i."""
    x = x0.copy()
    best = x.copy()
    best_score = _compute_score(x)

    step = step_init
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        grad, _, _ = _max_subgradient(x)
        cand = _project_onto_simplex(x - step * grad, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best, best_score = cand.copy(), cand_score
            x = cand
            step *= 1.02                     # a little acceleration
        else:
            step *= 0.70                     # backtrack
            # no move – keep x as is

        if step < 1e-14:
            break

    return best, best_score


# ----------------------------------------------------------------------
# 2️⃣  Alternating LP refinement (the “g‑step” from the statement)
# ----------------------------------------------------------------------


def _lp_mix_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    outer_iters: int = 5,
    tau_start: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    For a fixed f, solve   max Σ g   s.t. f∗g ≤ M, g≥0,
    scale g to the ℓ¹‑norm and mix f←(1‑τ)f+τ g.
    Repeating a few times usually shrinks the max‑convolution.
    """
    if not _HAS_CVX:
        # cvxpy not available – fall back to a no‑op
        return f0.copy(), _compute_score(f0)

    n = len(f0)
    f = f0.copy()
    best = f.copy()
    best_score = _compute_score(f)

    tau = tau_start
    for outer in range(outer_iters):
        if time.monotonic() >= deadline:
            break

        # current bound M
        conv = np.convolve(f, f)
        M = float(conv.max())

        # build the (2n‑1)×n matrix A with A[k,i] = f_{k‑i} (or 0)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i : i + n, i] = f  # slice assignment, broadcasting f[0:n]

        # linear programme
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception as e:                     # pragma: no cover
            print(f"[LP] solver failed: {e}")
            break

        if g.value is None:
            # infeasible – abort this refinement
            break

        g0 = np.maximum(g.value, 0.0)
        sum_g = float(g0.sum())
        if sum_g <= 0.0:
            break

        # scale to the required ℓ¹‑norm
        scale = target_sum / sum_g
        g_scaled = g0 * scale

        # convex mix
        f = (1.0 - tau) * f + tau * g_scaled
        f = _project_onto_simplex(f, target_sum)
        f = np.clip(f, 0.0, 1000.0)

        score = _compute_score(f)
        if score < best_score:
            best, best_score = f.copy(), score

        # slowly shrink the mixing weight
        tau *= 0.8

    return best, best_score


# ----------------------------------------------------------------------
# 3️⃣  Fine‑grained Gaussian hill climbing
# ----------------------------------------------------------------------


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.2,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """Gaussian perturbations with projection back onto the simplex."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        sigma *= 0.9995                     # slowly cool
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step sequence within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Optional warm‑start from a previously discovered sequence
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        n0 = len(seq)
        target0 = math.sqrt(2.0 * n0)
        seq = _project_onto_simplex(seq, target0)
        sc = _compute_score(seq)
        best_seq, best_score = seq.copy(), sc
        print(f"[init] warm‑start n={n0} score={sc:.6f}")

    # --------------------------------------------------------------
    # Main search: random restarts + sub‑gradient + LP mixing
    # --------------------------------------------------------------
    # We concentrate on a few lengths that usually give good trade‑offs.
    lengths = [300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        # number of restarts grows with n (larger space, more time)
        restarts = 12 if n <= 600 else 8

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ---- initial vector -------------------------------------------------
            # a few different families to diversify the search space
            init_kind = rng.choice(["uniform", "exp", "power"])
            if init_kind == "uniform":
                f0 = np.full(n, target_sum / n, dtype=np.float64)
            elif init_kind == "exp":
                f0 = rng.exponential(scale=1.0, size=n)
                f0 = f0 / f0.sum() * target_sum
            else:  # power‑law
                power = rng.uniform(0.5, 5.0)
                raw = rng.random(n) ** power
                f0 = raw / raw.sum() * target_sum

            # ---- sub‑gradient descent (focuses on the worst offset) ----------
            sub_deadline = min(deadline, time.monotonic() + 12.0)
            f_opt, sc_opt = _subgrad_optimize(f0, target_sum, sub_deadline,
                                              max_iters=2500, step_init=0.8)

            # ---- LP refinement (alternating “f‑g” step) ------------------------
            if _HAS_CVX:
                lp_deadline = min(deadline, time.monotonic() + 8.0)
                f_opt, sc_opt = _lp_mix_optimize(f_opt, target_sum, lp_deadline,
                                                 outer_iters=4, tau_start=0.3)

            # keep the best we have seen so far
            if sc_opt < best_score:
                best_seq, best_score = f_opt.copy(), sc_opt
                print(f"[update] n={n} restart={rs} kind={init_kind} score={sc_opt:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                print("[early] target reached")
                deadline = time.monotonic() + 0.3   # tiny reserve
                break

    # --------------------------------------------------------------
    # Final polishing – short stochastic hill climbing
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        refined, refined_sc = _hill_climb(best_seq, target_sum, deadline,
                                          sigma_initial=0.2, max_iters=6000)
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[refine] final score={refined_sc:.6f}")

    # final safety clip (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
