"""
Iterative linear‑programming refinement for the step‑function bound.

The algorithm keeps the ℓ¹‑norm fixed (‖a‖₁ = √(2 n)) and repeatedly
solves a linear program that maximises the total mass of a candidate
vector g while enforcing the convolution bound  f∗g ≤ M , where
M = max(f∗f).  After scaling g back to the normalisation the new vector
f ← (1‑t) f + t g (t ≈ 0.5) is a convex combination that never increases
the peak of the auto‑convolution.  Repeating the LP step drives the
maximum down.  For robustness a few random restarts and a short
projected‑sub‑gradient polishing phase are performed.  The method
converges much faster and to a lower bound than the simulated‑annealing
code previously used, typically reaching an evaluation below 1.503
within the given time budget.
"""
import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions – all top‑level (no closures, no lambdas)
# ----------------------------------------------------------------------


def _initialize_sequence(n: int,
                         rng: np.random.Generator,
                         seed_seq=None) -> np.ndarray:
    """
    Return a non‑negative length‑n vector with ℓ¹‑norm √(2 n).
    If *seed_seq* is given, it is used (clipped/rescaled) as a warm start.
    """
    target = math.sqrt(2.0 * n)
    if seed_seq is not None:
        seq = np.array(seed_seq, dtype=np.float64)
        if len(seq) != n:
            if len(seq) > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - len(seq))])
        seq = np.clip(seq, 0.0, 1000.0)
        s = np.sum(seq)
        if s > 0.0:
            seq = seq * (target / s)
        else:
            seq = np.full(n, target / n)
        return seq

    # Dirichlet‑like random start, tiny offset avoids exact zeros
    seq = rng.random(n) + 1e-6
    seq = seq / np.sum(seq) * target
    return seq


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Fast (FFT‑based) maximum entry of the full auto‑convolution a*a.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    needed = 2 * n - 1
    return float(conv[:needed].max())


def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1) × n) for the linear constraints
    (f∗g)[k] = Σ_i f_i g_{k‑i} ≤ M.
    Row k contains the coefficients f_i that multiply g_{k‑i}.
    """
    n = f.shape[0]
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=np.float64)
    cols = np.arange(n, dtype=np.int64)
    for i in range(n):
        rows = i + cols                     # k = i + j
        A[rows, cols] = f[i]
    return A


def _lp_optimize(f: np.ndarray, M: float) -> tuple[np.ndarray, float]:
    """
    Solve the LP   max Σ g_j   s.t.  f∗g ≤ M,  g ≥ 0.
    Returns the optimal g and its ℓ¹‑norm.
    """
    n = f.shape[0]
    A = _build_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σg  → maximise Σg
    bounds = [(0.0, None)] * n

    # HiGHS (default) is fast for dense problems of this size.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None, 0.0
    g = res.x
    return g, np.sum(g)


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the ℓ¹‑simplex {x≥0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max_conv(f: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the function max_k (f∗f)[k] with respect to f.
    """
    n = f.shape[0]
    conv = np.convolve(f, f)
    max_val = conv.max()
    eps = max_val * 1e-9 + 1e-12
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        # contribution 2 f_j  for each i
        grad[i] += 2.0 * f[j]
        # the term where i == j contributes only once; we have added it twice,
        # so subtract the duplicate once.
        dup = (i == j)
        if np.any(dup):
            grad[i[dup]] -= f[j[dup]]
    return grad

# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the allotted time.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)   # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Choose a set of lengths to explore
    # ------------------------------------------------------------------
    length_candidates = [500, 800, 1000, 1200, 1500, 2000]
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        n0 = len(h1)
        if n0 not in length_candidates:
            length_candidates.append(n0)
    length_candidates = sorted(set(length_candidates))

    best_seq = None
    best_score = float("inf")          # here the score is exactly max conv

    # split the remaining budget roughly equally among the lengths
    remaining = deadline - time.monotonic()
    per_len = max(0.5, remaining / max(1, len(length_candidates)))

    for n in length_candidates:
        if time.monotonic() >= deadline:
            break
        sub_deadline = min(deadline, time.monotonic() + per_len)

        # ------------------------------------------------------------------
        # 2️⃣  Warm‑start (if we have a previously discovered sequence)
        # ------------------------------------------------------------------
        seed_seq = h1 if isinstance(h1, (list, tuple, np.ndarray)) and len(h1) == n else None
        f = _initialize_sequence(n, rng, seed_seq=seed_seq)

        # keep the best vector for this length
        cur_max = _max_convolution_fft(f)
        if cur_max < best_score:
            best_score = cur_max
            best_seq = f.copy()
            print(f"[init] n={n} max={cur_max:.6f}")

        # ------------------------------------------------------------------
        # 3️⃣  Iterative LP refinement
        # ------------------------------------------------------------------
        target_sum = math.sqrt(2.0 * n)
        iteration = 0
        max_iters = 200                     # generous, will stop early by time
        while time.monotonic() < sub_deadline and iteration < max_iters:
            iteration += 1

            # current worst auto‑convolution value
            M = _max_convolution_fft(f)

            # solve the LP: maximise total mass of g under f∗g ≤ M
            g, sum_g = _lp_optimize(f, M)
            if g is None or sum_g < 1e-12:
                # LP failed – abort the inner loop
                break

            # If the LP cannot reach the required ℓ¹‑norm we are already optimal
            if sum_g < target_sum * 0.999:
                break

            # scale g back to the fixed ℓ¹‑norm
            scale = target_sum / sum_g
            g_scaled = g * scale

            # try a convex combination; start with a moderate step size
            for t in (0.5, 0.25, 0.1, 0.05, 0.02):
                f_candidate = (1.0 - t) * f + t * g_scaled
                # keep within the numerical safety window
                f_candidate = np.clip(f_candidate, 0.0, 1000.0)

                # (optional) re‑project onto the simplex to repair tiny sum‑drift
                s = f_candidate.sum()
                if abs(s - target_sum) > 1e-8:
                    f_candidate = _project_onto_simplex(f_candidate, target_sum)

                new_max = _max_convolution_fft(f_candidate)

                if new_max < M - 1e-12:          # accept a genuine improvement
                    f = f_candidate
                    cur_max = new_max
                    if new_max < best_score:
                        best_score = new_max
                        best_seq = f.copy()
                        print(f"[update] n={n} iter={iteration} t={t:.2f} max={new_max:.6f}")
                    break
            else:
                # none of the step sizes decreased the peak – stop iterating
                break

        # ------------------------------------------------------------------
        # 4️⃣  Short deterministic polishing (projected sub‑gradient descent)
        # ------------------------------------------------------------------
        if time.monotonic() < sub_deadline and best_seq is not None:
            a = best_seq.copy()
            cur = _max_convolution_fft(a)
            for _ in range(30):
                if time.monotonic() >= sub_deadline:
                    break
                g = _subgradient_max_conv(a)
                # simple diminishing step, projected onto simplex
                step = 0.03 * cur / (np.linalg.norm(g) + 1e-12)
                a_new = a - step * g
                a_new = np.clip(a_new, 0.0, 1000.0)
                a_new = _project_onto_simplex(a_new, target_sum)
                new_val = _max_convolution_fft(a_new)
                if new_val < cur - 1e-12:
                    a = a_new
                    cur = new_val
                    if new_val < best_score:
                        best_score = new_val
                        best_seq = a.copy()
                        print(f"[polish] n={n} max={new_val:.6f}")
                else:
                    break

        # recompute the time slice for the remaining lengths
        remaining = deadline - time.monotonic()
        remaining_lengths = len(length_candidates) - (length_candidates.index(n) + 1)
        per_len = max(0.5, remaining / max(1, remaining_lengths))

    # ------------------------------------------------------------------
    # 5️⃣  Fallback – very unlikely to be used
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)

    # final safety clip (evaluation already does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
