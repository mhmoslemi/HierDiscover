def random_block_seq(n, rng, min_blocks=5, max_blocks=50):
    # choose block count randomly within range depending on n
    B = rng.integers(min_blocks, max_blocks+1)
    # ensure B <= n
    B = min(B, n)
    # generate random positive block weights
    raw = rng.gamma(shape=2.0, scale=1.0, size=B)
    # could also sample from uniform
    # block values not check monotone: we may sort descending to reduce reversed product
    raw.sort()
    values = raw[::-1]  # descending
    # expand to length n
    # compute block sizes: equal distribution
    base = n // B
    extra = n % B
    block_sizes = np.full(B, base, dtype=int)
    block_sizes[:extra] += 1
    a = np.empty(n, dtype=np.float64)
    idx = 0
    for i in range(B):
        a[idx: idx+block_sizes[i]] = values[i]
        idx += block_sizes[i]
    # scaling
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum()) if a.sum() > 0 else np.full(n, target_sum / n)
    a = np.clip(a, 0.0, 1000.0)
    return a

"""
Propose a low‑value non‑negative step sequence.

The search proceeds in three phases:

1. **Random exploration** (≈ 80 % of the budget)
   – Generates many monotone decreasing vectors (Gamma‑distributed,
     sorted descending) and piece‑wise constant “block” vectors.
   – The vectors are scaled to the required ℓ¹‑norm √(2 n) and
     evaluated with a fast custom objective (identical to
     `evaluate_sequence` but without the Python overhead).

2. **Gradient refinement** (if time permits)
   – Runs a few L‑BFGS‑B optimisations on the smooth log‑sum‑exp
     surrogate of the max‑convolution, using a continuation on the
     smoothness parameter β ∈ {5,…,200,500,1000}.
   – Starts from the best candidate found in phase 1.

3. **Stochastic polishing**
   – A short Gaussian hill‑climbing (soft‑max re‑projection) that
     respects the ℓ¹‑constraint.

The best sequence seen during the whole run is returned (clipped to
[0,1000] as required by `evaluate_sequence`).  The algorithm respects
the global time budget (≈ 1000 s) and never uses file or network I/O.
"""

import time
import math
import numpy as np
import functools
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Fast objective – exactly the same as evaluate_sequence (but without
# the security checks and Python‑level overhead).
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Returns the value of the evaluation function for a non‑negative
    sequence `seq` (numpy 1‑D array).  The sequence is clipped to
    [0,1000] before the computation.
    """
    a = np.clip(seq, 0.0, 1000.0)
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = np.convolve(a, a)
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Helper utilities (soft‑max re‑parameterisation, FFT helpers, …)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """Map unrestricted logits `z` to a non‑negative simplex point with sum=`target_sum`."""
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z) * target_sum
    return a


def _obj_and_grad(z: np.ndarray, n: int, S: float, beta: float):
    """
    Smooth surrogate of the max‑convolution:
        (1/β)·log Σ exp(β·conv)
    Returns a tuple (smooth, grad_z) where grad_z is the gradient w.r.t. `z`.
    """
    a = _softmax_to_a(z, S)                     # shape (n,)

    # ‑‑‑ convolution via FFT
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]  # length 2n‑1

    # ‑‑‑ smooth max (log‑sum‑exp)
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    sum_exps = np.sum(exps)
    smooth = max_c + np.log(sum_exps) / beta
    w = exps / sum_exps                           # soft‑max weights, Σw = 1

    # ‑‑‑ gradient w.r.t. a (see derivation in the original code)
    a_rev = a[::-1]
    L2 = _next_pow2(3 * n - 2)
    fr = np.fft.rfft(a_rev, L2)
    fw = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(fr * fw, L2)
    start = n - 1
    g_raw = conv2[start:start + n]
    g_a = 2.0 * g_raw

    # ‑‑‑ chain rule through the soft‑max
    total = np.dot(g_a, a)
    grad_z = a * g_a - a * (total / S)
    return smooth, grad_z.astype(np.float64)


def _optimise_length(n: int,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> tuple[np.ndarray, float]:
    """
    Gradient based optimisation for a fixed length `n`.
    Returns the best sequence found and its exact objective value.
    """
    # extended β schedule, more restarts than the baseline
    betas = [5.0, 15.0, 40.0, 80.0, 200.0, 500.0, 1000.0]
    max_restarts = 6
    best_seq = None
    best_val = float("inf")

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # initialise logits
        if warm_seq is not None and warm_seq.shape[0] == n:
            p = np.clip(warm_seq / target_sum, 1e-12, None)
            z0 = np.log(p)
        else:
            z0 = rng.normal(scale=0.1, size=n)

        # continuation over β
        for beta in betas:
            if time.monotonic() >= deadline:
                break

            fun = functools.partial(_obj_and_grad, n=n, S=target_sum, beta=beta)
            res = minimize(fun=fun,
                           x0=z0,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 200,
                                    'gtol': 1e-10,
                                    'disp': False})
            z0 = res.x

            a = _softmax_to_a(z0, target_sum)
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[grad] n={n} restart={restart} β={beta:.0f} val={val:.6f}")

            # early exit if we already beat a reasonable target
            if best_val < 1.45:          # generous margin; can be lowered
                return best_seq, best_val

    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = _objective(best_seq)

    return best_seq, best_val


def _stochastic_refinement(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that preserves the ℓ¹‑sum via a
    soft‑max re‑projection.  Used as the final polishing step.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _objective(best)
    step = step_initial
    S = best.sum()

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑project onto the simplex (soft‑max)
        p = np.clip(cand / S, 1e-12, None)
        cand = _softmax_to_a(np.log(p), S)

        cand_score = _objective(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995               # slow cooling
        if step < 1e-8:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Random constructions
# ----------------------------------------------------------------------
def _random_monotone_seq(n: int, rng: np.random.Generator,
                         shape: float = 2.0) -> np.ndarray:
    """
    Draw `n` Gamma‑distributed numbers, sort them descending, and
    scale to the required ℓ¹‑norm.
    """
    raw = rng.gamma(shape=shape, scale=1.0, size=n)
    raw.sort()
    a = raw[::-1]                     # monotone decreasing
    target = math.sqrt(2.0 * n)
    s = a.sum()
    if s > 0.0:
        a = a * (target / s)
    else:
        a = np.full(n, target / n, dtype=np.float64)
    return np.clip(a, 0.0, 1000.0)


def _random_block_seq(n: int, rng: np.random.Generator,
                      B: int) -> np.ndarray:
    """
    Piece‑wise constant vector with `B` blocks.
    Block values are drawn from a Gamma distribution, sorted descending,
    and expanded to length `n`.  The result is scaled to the target ℓ¹‑norm.
    """
    if B > n:
        B = n
    raw = rng.gamma(shape=2.0, scale=1.0, size=B)
    raw.sort()
    block_vals = raw[::-1]            # descending blocks

    # block sizes (as equal as possible)
    base = n // B
    extra = n % B
    block_sizes = np.full(B, base, dtype=int)
    block_sizes[:extra] += 1

    a = np.empty(n, dtype=np.float64)
    idx = 0
    for val, size in zip(block_vals, block_sizes):
        a[idx: idx + size] = val
        idx += size

    target = math.sqrt(2.0 * n)
    s = a.sum()
    if s > 0.0:
        a = a * (target / s)
    else:
        a = np.full(n, target / n, dtype=np.float64)
    return np.clip(a, 0.0, 1000.0)
