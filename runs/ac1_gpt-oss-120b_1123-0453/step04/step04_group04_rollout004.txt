            else:
                seq = np.concatenate([seq, np.zeros(n - len(seq))])
        seq = np.clip(seq, 0.0, 1000.0)
        s = np.sum(seq)
        if s == 0.0:
            seq = np.full(n, target / n)
        else:
            seq = seq * (target / s)
        return seq

    if rng is None:
        rng = np.random.default_rng()
    seq = rng.random(n) + 1e-3          # avoid exact zeros initially
    seq = seq / np.sum(seq) * target
    return seq


def _max_auto_convolution(a: np.ndarray) -> float:
    """Return max entry of a*a (plain O(n²) convolution – fine for n≈1500)."""
    conv = np.convolve(a, a)
    return float(conv.max())


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A that satisfies (f * g)[k] = (A g)[k].

    Column j of A is f shifted by j positions:
        A[k, j] = f[k‑j]  if 0 ≤ k‑j < n, else 0.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # fill column‑wise: each column j occupies rows j … j+n‑1
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_improve(f: np.ndarray, bound_M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_j
        s.t.       g_j ≥ 0
                   (f * g)[k] ≤ bound_M   for all k
    and return the solution scaled to `target_sum`.  Returns None on failure.
    """
    n = f.size
    A = _build_conv_matrix(f)               # (2n‑1) × n
    b = np.full(2 * n - 1, bound_M)

    # linprog minimises; use -1 as objective to maximise Σ g_j
    c = -np.ones(n, dtype=np.float64)

    # bounds for each variable: g_j ≥ 0, no explicit upper bound
    bounds = [(0.0, None)] * n

    # solve – “highs” is fast and works for moderate sizes (n≈500‑1500)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return None
    # scale to the required ℓ¹‑norm (kept constant through the whole run)
    return g * (target_sum / s)


def _simulated_annealing(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very similar to the baseline annealer, but deliberately kept as a
    separate function so that we can invoke it after the LP phase.
    Returns the best vector seen and its max‑convolution value.
    """
    n = a.size
    target_sum = np.sum(a)               # invariant during the run
    fft_len = 1 << ((2 * n - 1).bit_length())
    cur_max = _max_auto_convolution(a)

    best_seq = a.copy()
    best_val = cur_max

    rng = np.random.default_rng()
    start = time.monotonic()
    total_time = max(1e-12, deadline - start)

    # initial temperature – a fraction of the current max
    init_temp = max(cur_max * 0.5, 1e-4)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = elapsed / total_time
        # exponential cooling
        T = init_temp * (0.0001 ** frac)

        # step size proportional to average entry
        avg = target_sum / n
        step = avg * (T / init_temp) * 3.0

        # pick distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose a transfer of mass
        delta = rng.normal(scale=step)
        delta = min(delta, a[i])           # cannot go negative at i
        delta = max(delta, -a[j])          # cannot go negative at j
        if abs(delta) < 1e-12:
            continue

        # apply the move
        a[i] -= delta
        a[j] += delta

        new_max = _max_auto_convolution(a)

        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        elif T > 0.0:
            prob = math.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()
        else:
            a[i] += delta
            a[j] -= delta

    return best_seq, best_val

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a coefficient vector that minimises the evaluation function.
    Mixes a linear‑programming guided refinement with simulated annealing.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.05)   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Choose a handful of lengths to explore.
    # ------------------------------------------------------------------
    length_candidates = [250, 300, 400, 500, 600, 800, 1000]
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        n0 = len(h1)
        if n0 not in length_candidates:
            length_candidates.append(n0)
    length_candidates = sorted(set(length_candidates))

    best_seq = None
    best_score = float("inf")

    # allocate time roughly equally among lengths (dynamic)
    for n in length_candidates:
        if time.monotonic() >= deadline:
            break

        # Remaining time and per‑length budget
        remaining = deadline - time.monotonic()
        # leave a little slack for the polishing phase
        per_len = max(0.3, remaining / (len(length_candidates) - length_candidates.index(n)))

        sub_deadline = time.monotonic() + per_len

        # --------------------------------------------------------------
        # Initialise a sequence (warm‑start if possible)
        # --------------------------------------------------------------
        seed_seq = h1 if isinstance(h1, (list, tuple, np.ndarray)) and len(h1) == n else None
        a = _initialize_sequence(n, seed_seq=seed_seq, rng=rng)

        # Short annealing warm‑up (≈10 % of the allocated time)
        warm_deadline = min(sub_deadline, time.monotonic() + per_len * 0.1)
        a, _ = _simulated_annealing(a, warm_deadline)

        # --------------------------------------------------------------
        # LP‑guided refinement loop
        # --------------------------------------------------------------
        # We keep the ℓ¹‑norm constant – this also makes the evaluation
        # invariant under scaling.
        target_sum = np.sum(a)

        while time.monotonic() < sub_deadline:
            cur_max = _max_auto_convolution(a)

            # Solve the LP to obtain a direction g
            g = _lp_improve(a, cur_max, target_sum)
            if g is None:
                break

            # Try convex combinations with decreasing step size until
            # an improvement is observed.
            improved = False
            for t in [0.5, 0.25, 0.125, 0.0625, 0.03125]:
                a_candidate = (1.0 - t) * a + t * g
                # evaluate – use the supplied checker directly
                score = evaluate_sequence(a_candidate.tolist())
                if score < best_score - 1e-12:
                    best_score = score
                    best_seq = a_candidate.copy()
                    a = a_candidate
                    improved = True
                    print(f"[LP] n={n} t={t:.3f} score={best_score:.6f}")
                    break
            if not improved:
                # No improvement with any step size → stop LP phase for this n
                break

        # --------------------------------------------------------------
        # Optional polishing with a longer simulated‑annealing run
        # --------------------------------------------------------------
        if time.monotonic() < deadline:
            polish_deadline = deadline
            a_polish, _ = _simulated_annealing(a.copy(), polish_deadline)
            score_polish = evaluate_sequence(a_polish.tolist())
            if score_polish < best_score - 1e-12:
                best_score = score_polish
                best_seq = a_polish.copy()
                print(f"[polish] n={n} final score={best_score:.6f}")

        # Update remaining time for the next length
        # (the loop recomputes `remaining` automatically)

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = _initialize_sequence(n, rng=rng)

    # Clip to the limits required by `evaluate_sequence`
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Store the best found sequence globally for possible warm‑starts in later runs
    globals()["height_sequence_1"] = best_seq.tolist()

    return [float(x) for x in best_seq]
