def solve_lp_for_g(f, M, target_sum):
    n = f.shape[0]
    # Build A_ub matrix only once per f? Since f changes each iteration.
    # Build with loops: A_ub = np.zeros((2*n-1, n))
    for i in range(n):
        A_ub[i:i+n, i] = f
    # b_ub is M * np.ones(2*n-1)
    res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b == 0:
        return None
    scale = target_sum / sum_b
    g = b * scale
    return g

def propose_candidate(seed=0, budget_s=1000, **kwargs):

    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 0.5)

    best_seq = None
    best_score = float('inf')
    lengths = [800, 600, 500, 400, 300, 200]   # maybe up to 800 to improve

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # generate a few starting points
        starts = []
        # uniform
        starts.append(np.full(n, target_sum / n))
        # if global best-known with same n, use it
        if best_seq is not None and best_seq.shape[0] == n:
            starts.append(best_seq.copy())
        # some random dirichlet draws
        for _ in range(2):
            starts.append(rng.dirichlet(np.ones(n)) * target_sum)
        # maybe triangular
        x = np.arange(n)
        tri = 1 - np.abs(x - (n-1)/2) / ((n-1)/2)
        tri[tri<0]=0
        starts.append(_normalize(tri * target_sum, target_sum))

        for f0 in starts:
            if time.monotonic() >= deadline:
                break
            f = _normalize(f0, target_sum)  # ensure sum correct
            # iterative improvement
            it = 0
            while time.monotonic() < deadline and it < 30:
                it += 1
                conv_f = np.convolve(f, f)
                M = conv_f.max()
                # solve LP for g
                n_local = n
                # build A_ub
                A = np.zeros((2*n_local - 1, n_local), dtype=np.float64)
                # assign each column i

                    A[i:i+n_local, i] = f

                res = linprog(c=-np.ones(n_local), A_ub=A, b_ub=b_ub,
                              bounds=[(0, None)]*n_local, method='highs')
                if not res.success:
                    break
                b_opt = res.x
                sum_b = b_opt.sum()
                if sum_b <= 0:
                    break
                scale = target_sum / sum_b
                g = b_opt * scale
                # small convex combo

                # enforce positivity, clip and renorm

    # final polishing using earlier local methods

"""
Iterative LP‑based improvement + local polishing.

The evaluation metric for a sequence a with ℓ¹‑norm √(2·n) is simply
max_k (a⋆a)_k, i.e. the largest entry of the auto‑convolution.
The algorithm proceeds in two phases:

1️⃣  For each chosen length n we start from a few sensible seeds
    (uniform, triangular, a previously discovered best, random Dirichlet).
    Given a current vector f we compute M = max(f⋆f).  We then solve the
    linear program

        maximise   Σ b_i
        subject to  f⋆b ≤ M   (component‑wise)
                    b_i ≥ 0

    The LP yields a vector b whose ℓ¹‑norm can be larger than that of f.
    After scaling b to the required ℓ¹‑norm we take a convex combination
    f ← (1‑t)·f + t·b (t≈0.2) – this provably never raises M.
    Repeating the LP → blend step (≈20 iterations) drives M down.

2️⃣  When the time budget is almost exhausted we hand the best vector to the
    fast hill‑climb / Gaussian polishing routine from the previous
    implementation.  The result is returned as a plain Python list.

All numbers are kept non‑negative, clipped to [0,1000] and re‑normalised to
√(2·n) after each modification, thus satisfying the security checks of
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# optional import – will be None if SciPy is unavailable
try:
    from scipy.optimize import linprog  # type: ignore
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False


def _normalize(vec: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ¹‑norm exactly *target*."""
    s = vec.sum()
    if s <= 0.0:
        return np.full_like(vec, target / vec.size)
    return vec * (target / s)


def _fast_score(seq: np.ndarray) -> float:
    """Exact analogue of ``evaluate_sequence`` (no safety checks)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A where A @ b = np.convolve(f, b).
    Shape of A is (2·n‑1, n).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column i contains f shifted down by i positions
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve the LP   maximise Σ b_i   s.t.  f⋆b ≤ M, b ≥ 0.
    Return the scaled vector g (ℓ¹‑norm = target_sum) or None on failure.
    """
    if not _HAS_SCIPY:
        return None

    n = f.shape[0]
    A = _build_conv_matrix(f)                 # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σb_i  → minimise –Σb_i
    c = -np.ones(n, dtype=np.float64)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
    except Exception:  # pragma: no cover
        return None

    if not res.success:
        return None

    b = res.x
    s = b.sum()
    if s <= 0.0:
        return None
    scale = target_sum / s
    g = b * scale
    return g


def _refine_sequence(seq: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 5000) -> np.ndarray:
    """
    Small local optimiser (mass‑transfer + projected gradient) – the same
    routine used in the previous state‑of‑the‑art code.  It is kept as a
    polishing stage after the LP phase.
    """
    n = seq.shape[0]
    best = _normalize(seq, target_sum)
    best_sc = _fast_score(best)

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        conv = np.convolve(best, best)
        max_val = conv.max()
        k = int(conv.argmax())

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1, dtype=np.int64)

        partners = best[k - idxs]
        contributions = best[idxs] * partners
        donor = idxs[np.argmax(contributions)]

        outside = np.setdiff1d(np.arange(n, dtype=np.int64), idxs, assume_unique=True)
        if outside.size > 0:
            receiver = int(outside[np.argmin(best[outside])])
        else:
            receiver = int(idxs[np.argmin(partners)])

        if donor == receiver:
            receiver = int(rng.integers(n))
            if receiver == donor:
                continue

        max_move = min(best[donor] * 0.4, target_sum * 0.002)
        if max_move <= 0.0:
            continue

        partner_donor = best[k - donor]
        if 0 <= k - receiver < n:
            partner_receiver = best[k - receiver]
        else:
            partner_receiver = 0.0

        if partner_donor > partner_receiver:
            eps = max_move
            cand = best.copy()
            cand[donor] -= eps
            cand[receiver] += eps
        else:
            i = int(rng.choice(np.where(best > 0)[0]))
            j = int(rng.integers(n))
            eps = min(best[i] * 0.2, target_sum * 0.001)
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc - 1e-12:
            best, best_sc = cand, cand_sc
            continue

        # projected gradient fallback
        grad = np.zeros(n, dtype=np.float64)
        for i in range(i_start, i_end + 1):
            j = k - i
            grad[i] = best[j]
        direction = -grad + grad.mean()
        neg = direction < 0
        if np.any(neg):
            max_alpha = np.min(best[neg] / -direction[neg])
        else:
            max_alpha = 1.0
        alpha = min(0.08 * max_alpha, 0.04 * target_sum / n)
        cand = best + alpha * direction
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc - 1e-12:
            best, best_sc = cand, cand_sc
            continue

        if rng.random() < 0.04:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(best, kernel, mode="same")
            smooth = np.clip(smooth, 0.0, 1000.0)
            smooth = _normalize(smooth, target_sum)
            smooth_sc = _fast_score(smooth)
            if smooth_sc < best_sc:
                best, best_sc = smooth, smooth_sc
                continue

        if rng.random() < 0.01:
            cand = rng.dirichlet(np.ones(n)) * target_sum
            cand_sc = _fast_score(cand)
            if cand_sc < best_sc:
                best, best_sc = cand, cand_sc
                continue

    return best


def _swap_hill_climb(seq: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     rng: np.random.Generator) -> np.ndarray:
    """Very cheap hill‑climb that swaps a small mass between two indices."""
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target_sum * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
    return best


def _gaussian_local_search(seq: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           rng: np.random.Generator) -> np.ndarray:
    """Short annealing‑style Gaussian perturbation loop."""
    best = seq.copy()
    best_sc = _fast_score(best)
    step = target_sum * 0.02

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape[0])
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
            step *= 0.9
        else:
            step *= 0.99
            if step < 1e-6:
                step = target_sum * 0.02
    return best


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low max‑auto‑convolution value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the routine stops a little before the limit).
    kwargs : dict
        Ignored – kept for compatibility with the outer harness.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.35)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to initialise a vector of length n from several families.
    # ------------------------------------------------------------------
    def _initial_vectors(n: int, target: float) -> list[np.ndarray]:
        vecs = []

        # uniform
        vecs.append(np.full(n, target / n, dtype=np.float64))

        # triangular (peak in the centre)
        x = np.arange(n, dtype=np.float64)
        tri = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
        tri[tri < 0] = 0.0
        vecs.append(_normalize(tri, target))

        # a few random Dirichlet draws
        for _ in range(3):
            vecs.append(rng.dirichlet(np.ones(n)) * target)

        # if a previously discovered best of same length exists, use it
        if "height_sequence_1" in globals():
            try:
                old = np.asarray(height_sequence_1, dtype=np.float64)
                if old.shape[0] == n:
                    vecs.append(_normalize(old, target))
            except Exception:
                pass

        return vecs

    # ------------------------------------------------------------------
    # Main loop over a selection of lengths (larger n gives more flexibility)
    # ------------------------------------------------------------------
    lengths = [1000, 800, 600, 500, 400, 300, 200]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # generate a modest set of starting vectors
        starts = _initial_vectors(n, target_sum)

        for start in starts:
            if time.monotonic() >= deadline:
                break

            f = _normalize(start, target_sum)

            # ----- LP‑based improvement loop -----
            it = 0
            max_lp_iters = 25
            while time.monotonic() < deadline and it < max_lp_iters:
                it += 1
                conv_f = np.convolve(f, f)
                M = conv_f.max()

                g = _lp_step(f, M, target_sum)
                if g is None:
                    break

                # convex combination – the step size t is fixed (≈0.2)
                t = 0.2
                f_new = (1.0 - t) * f + t * g
                f_new = np.clip(f_new, 0.0, 1000.0)
                f_new = _normalize(f_new, target_sum)

                # check for genuine improvement
                new_M = np.convolve(f_new, f_new).max()
                if new_M < M - 1e-8:
                    f = f_new
                else:
                    # no progress – exit the LP phase for this start
                    break

            # ----- score and global update -----
            sc = _fast_score(f)   # with our normalisation this equals max conv
            if sc < best_score - 1e-12:
                best_seq, best_score = f.copy(), sc
                print(f"[n={n}] new best score {sc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (cheap hill‑climb + short Gaussian annealing)
    # ------------------------------------------------------------------
    if best_seq is None:
        # very fallback – uniform of a modest size
        n = 300
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    n = best_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)
    best_seq = _normalize(best_seq, target_sum)

    # a few iterations of the earlier fast local optimiser
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        best_seq = _refine_sequence(best_seq, target_sum,
                                    deadline, rng, max_iters=3000)

    # tiny swap‑based hill‑climb
    if time.monotonic() < deadline:
        best_seq = _swap_hill_climb(best_seq, target_sum, deadline, rng)

    # short Gaussian perturbation if time permits
    if time.monotonic() < deadline:
        best_seq = _gaussian_local_search(best_seq, target_sum, deadline, rng)

    final_score = _fast_score(best_seq)
    print(f"[final] best score {final_score:.6f}   length {best_seq.shape[0]}")
    return [float(x) for x in best_seq]
