        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:

def _normalize(seq: np.ndarray) -> np.ndarray:

# Sub‑gradient of the max‑convolution objective
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the convex function  max_k (a*a)_k.
    Returns a sub‑gradient based on the currently active indices.
    """
    max_val = conv.max()
    # tolerance to collect numerically tied maximal entries
    eps = max_val * 1e-12 + 1e-15
    w = (conv >= max_val - eps).astype(np.float64)
    w_sum = w.sum()
    if w_sum > 0.0:
        w = w / w_sum                     # normalise – step size becomes scale‑free
    # gradient = 2 * correlation of a with w (see derivation in analysis)
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)     # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]      # middle n entries
    grad = 2.0 * segment[::-1]
    return grad


# ----------------------------------------------------------------------
# Optimisation core (projected sub‑gradient descent)
# ----------------------------------------------------------------------
def _optimize_subgradient(start: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.
    Returns the best vector seen and its raw max‑convolution value.
    """
    a = start.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    best_vec = a.copy()
    best_val = _fast_score(a)

    it = 0
    # initial step size – roughly a few percent of the average entry
    step0 = target / (30.0 * n)

    while time.time() < deadline and it < 4000:
        conv = _linear_convolve(a, a)
        cur_max = conv.max()

        if cur_max < best_val:
            best_val = cur_max
            best_vec = a.copy()

        grad = _active_set_gradient(a, conv)

        # simple diminishing step
        step = step0 / (1.0 + it * 0.001)

        a = a - step * grad

        # projection onto the feasible set
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # occasional stochastic jitter to escape plateaus
        if it % 250 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.02
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        it += 1

    return best_vec, best_val

# Random‑mass‑swap refinement (cheap hill‑climb)
# ----------------------------------------------------------------------
def _refine_by_swaps(start: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that moves a small fraction of mass between two
    coordinates whenever the move reduces the max‑convolution value.
    """
    a = start.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_val = _fast_score(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = a[i] * 0.05
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cur_val = _fast_score(cand)
        if cur_val < best_val:
            best_val = cur_val
            best_vec = cand
            a = cand
    return best_vec, best_val
