"""
LP-based alternating refinement for minimizing peak auto‑convolution.

The algorithm:
1. Fix the length n and the L1‑norm S = sqrt(2n) (the objective is scale‑invariant).
2. Initialise a random non‑negative vector f on the simplex Σ f_i = S.
3. Repeat:
   a. Compute M = max(f * f) (full convolution).
   b. Solve a linear program: maximise Σ g_i subject to (f * g)_k ≤ M for all k,
      g_i ≥ 0.  For fixed f the constraints are linear in g.
   c. Rescale g to have Σ g_i = S and blend f←(1‑t)f + t g
      (t = 0.5, later reduced if no improvement).
   d. Project the blend back onto the simplex.
   e. Stop when M does not improve.
4. Keep the best vector over several random restarts and several lengths.
5. Finally run a short stochastic hill‑climb to polish the result.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # shift v so that we can apply the classic algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1) × n matrix A such that (f*g) = A @ g."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j receives a copy of f shifted by j rows
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _compute_score(v: np.ndarray) -> float:
    n = v.shape[0]
    conv = np.convolve(v, v)
    max_val = conv.max()
    sum_val = v.sum()
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))

def _refine_by_lp(f0: np.ndarray, target_sum: float, deadline: float,
                  max_outer: int = 30, t_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based alternating refinement described above.
    Returns the best vector seen and its evaluation score.
    """
    n = f0.shape[0]
    best = f0.copy()
    best_score = _compute_score(best)
    f = f0.copy()
    t = t_initial

    for outer in range(max_outer):
        if time.monotonic() >= deadline:
            break

        # 1) current peak of the auto‑convolution
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        # 2) solve LP for g
        A = _conv_matrix(f)                     # shape (2n‑1, n)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)       # maximise Σ g  → minimise −Σ g
        # Use method='highs' – it respects bounds and is fast for medium size LP.
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})
        if not res.success:
            # If LP fails (rare), abort this run.
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            # fallback – keep current f
            break
        # 3) rescale to the required L1‑norm
        g = g * (target_sum / sum_g)

        # 4) blend f and g
        f_candidate = (1.0 - t) * f + t * g
        f_candidate = _project_onto_simplex(f_candidate, target_sum)

        cand_score = _compute_score(f_candidate)

        if cand_score + 1e-12 < best_score:
            best, best_score = f_candidate.copy(), cand_score
            f = f_candidate
            # if we made progress we can keep the step size
            # and possibly increase it modestly
            t = min(0.9, t * 1.05)
        else:
            # No improvement – shrink the mixing factor
            t *= 0.5
            if t < 1e-4:
                # Step size too small, stop early
                break

    return best, best_score

def _hill_climb(start: np.ndarray, target_sum: float, deadline: float,
                sigma_initial: float = 0.1, max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian perturbation hill‑climbing while staying on the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        sigma *= 0.999
        max_iters -= 1
        if sigma < 1e-8:
            sigma = sigma_initial
    return best, best_score

    """

    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
