```
def _exponential_sequence(n, alpha):
    r = np.exp(-alpha)
    # compute geometric series
    # a_i = C * r**i

    return C * (r ** np.arange(n))
```

```
def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)
    best_seq = None
    best_score = float('inf')
    lengths = [500, 800, 1200, 1600, 2000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # candidate generation
        cand_list = []
        # Dirichlet candidates
        for conc in [0.3, 0.7, 1.5, 3.0]:
            v = rng.gamma(shape=conc, size=n)
            if np.sum(v) == 0:
                v = np.full(n, 1.0)
            v = v / np.sum(v) * target_sum
            cand_list.append(v)
        # exponential candidates
        for alpha in np.linspace(0.001, 0.01, 9):
            v = _exponential_sequence(n, alpha)
            cand_list.append(v)
        # random uniform
        v = rng.random(n)
        v = v / np.sum(v) * target_sum
        cand_list.append(v)

        # For each candidate, run adam descent
        for idx, f0 in enumerate(cand_list):
            if time.monotonic() >= deadline:
                break
            # quick run: assign subdeadline
            sub_deadline = min(deadline, time.monotonic() + 6.0)
            f_opt, s_opt = _adam_descent(f0, target_sum, sub_deadline,
                                         max_iters=5000)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = f_opt.copy()
                print(f"[best] n={n} init={idx} score={s_opt:.6f}")

            if best_score < 1.4: # early exit
                print("[early] reached target")
                return [float(x) for x in best_seq]
```

Candidate generator for evaluate_sequence.

Strategy:
1. Sample a diverse set of initial step sequences (Dirichlet random,
   geometric/exponential families, plain uniform).
2. For each candidate run a projected Adam optimizer that minimizes the
   worst auto‑convolution entry (the same objective as the evaluation
   function).  The optimizer works on the ℓ¹‑simplex (fixed ℓ¹‑norm) so
   the objective coincides with the reported score.
3. Keep the best solution found across all lengths and restarts.
4. Finish with a temperature‑controlled simulated‑annealing hill‑climber
   that moves mass between distant indices while preserving the total
   sum, allowing occasional uphill moves to escape local minima.
The routine respects the global time budget (default 1000 s) and prints
progress information.
"""

import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x | x_i ≥ 0, Σx_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(f: np.ndarray) -> float:
    """Exact value of evaluate_sequence (clipping inside that function is harmless)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where convolution attains (almost) the maximal value."""
    eps = max_val * 1e-12 + 1e-14
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the worst auto‑convolution entry w.r.t. f."""
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        vals = f[j]
        grad[i] += vals
        grad[j] += f[i]
    return grad


def _adam_descent(f0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the maximal entry of the auto‑convolution."""
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.08
    it = 0
    no_improve = 0

    while it < max_iters and time.monotonic() < deadline:
        it += 1

        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)

        grad_max = _gradient_submax(f, active)

        sum_f = np.sum(f)
        grad_obj = (2.0 * len(f) / (sum_f * sum_f)) * grad_max

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = _project_onto_simplex(f_candidate, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

            no_improve = 0

        else:
            no_improve += 1

            f = f_candidate
        else:
            if no_improve > 500:
                lr *= 0.7
                no_improve = 0

        if no_improve > 1000:
            break

def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.3,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """Neighbourhood search that preserves the ℓ¹‑norm."""

    cur = start.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score
    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.08 * target_sum / n

    while time.monotonic() < deadline:

        if i == j:

        if max_move <= 0:

def _exponential_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Return a geometric sequence a_i = C * exp(-alpha * i) whose ℓ¹‑norm
    equals sqrt(2n).  The constant C is chosen analytically.
    """
    target = math.sqrt(2.0 * n)
    if alpha <= 0.0:
        # fall back to uniform
        return np.full(n, target / n, dtype=np.float64)

    r = math.exp(-alpha)
    # Avoid division by zero when r is very close to 1
    if abs(1.0 - r) < 1e-12:
        return np.full(n, target / n, dtype=np.float64)

    geom_sum = (1.0 - r ** n) / (1.0 - r)
    C = target / geom_sum
    idx = np.arange(n, dtype=np.float64)
    return C * (r ** idx)

        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] using uniform construction")

    # Clip once more for safety with respect to `evaluate_sequence`.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
