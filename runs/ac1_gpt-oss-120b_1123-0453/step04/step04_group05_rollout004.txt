def hill_climb(initial_seq, deadline, rng):
    cur_seq = initial_seq.copy()
    n = cur_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)
    cur_seq = cur_seq * (target_sum / cur_seq.sum())  # ensure sum
    cur_val = compute_max_conv(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    start = time.monotonic()
    total = deadline - start
    # parameters
    max_no_improve = 5000
    no_improve_counter = 0
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        fraction = elapsed / total
        temp = (1.0 - fraction) * 0.5 + fraction * 1e-5 # start 0.5, end small
        # choose move
        i = rng.integers(n)
        # ensure a_i > 0
        if cur_seq[i] <= 1e-12:
            continue
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # move fraction
        move_fraction = rng.random() * max(0.01, 0.2 * (1 - fraction))
        epsilon = cur_seq[i] * move_fraction
        if epsilon <= 0:
            continue
        # Propose new seq
        cand = cur_seq.copy()
        cand[i] -= epsilon
        cand[j] += epsilon
        # ensure non-negative (should hold)
        cand_val = compute_max_conv(cand)
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / temp):
            cur_seq = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                no_improve_counter = 0
                # optionally print
                print(f"[SA] new best max_conv={best_val:.6f} n={n}")
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1
        # restart if stuck
        if no_improve_counter >= max_no_improve and (deadline - time.monotonic()) > 5.0:
            # random restart with Dirichlet
            alpha = 0.3
            cand_dir = rng.gamma(alpha, size=n)
            cand_dir = cand_dir / cand_dir.sum() * target_sum
            cur_seq = cand_dir
            cur_val = compute_max_conv(cur_seq)
            no_improve_counter = 0
    return best_seq, best_val

    n = cur_seq.shape[0]

        elapsed = now - start
        frac = elapsed / total
        # temperature schedule
        temp = max(0.5 * (1 - frac), 0.001) # simple linear decreasing
        # move fraction
        move_frac = max(0.001, 0.2 * (1 - frac))
        # compute convolution
        conv = compute_conv(cur_seq)
        max_val = float(conv.max())

        if rng.random() < 0.7 and high >= low:

            src = rng.integers(low, high+1)

def final_polish(seq, deadline, rng):
    best_seq = seq.copy()
    best_val = compute_max_conv(best_seq)
    step = seq.mean() * 0.1  # magnitude

"""
Simulated‑annealing hill‑climber on the ℓ¹‑simplex.

The evaluation function is homogeneous: after the mandatory normalisation
∑aᵢ = √(2 n) the score is exactly the maximum entry of the (full) convolution
a*a.  The algorithm therefore works directly on the max‑convolution value.

* For a chosen length n we start from a Dirichlet‑type random vector and
  normalise it to the required ℓ¹‑norm.
* A fast SA loop moves a small amount of mass ε from a source index to a
  target index while the total sum stays fixed.  The temperature and the
  step‑size decay linearly with elapsed time.
* The source index is often chosen from the set of indices that actively
  contribute to the current maximal convolution (the “active set”), which
  focuses the search where it matters most.
* The best sequence found for each length is kept, and the globally best
  sequence across several candidate lengths is returned.
* If time permits a short Gaussian polishing step refines the final answer.

The code respects the global time budget (≈ budget_s seconds) and returns a
plain Python list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """
    Compute the full (2 n‑1) convolution of ``seq`` with itself using FFT.
    Returns a 1‑D array of length 2 n‑1.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(seq, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv


def _max_convolution(seq: np.ndarray) -> tuple[float, int, np.ndarray]:
    """
    Return (max_value, argmax_index, whole_convolution) for a sequence.
    The max value is the objective for a normalised sequence (∑a = √(2 n)).
    """
    conv = _full_convolution(seq)
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx, conv


def _initial_dirichlet(n: int, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """
    Draw a random point on the ℓ¹‑simplex using a Gamma (Dirichlet) distribution
    with concentration α = 0.3 (spiky points).  The result is scaled to
    ``target_sum``.
    """
    alpha = 0.3
    x = rng.gamma(alpha, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0, dtype=np.float64)
    x = x * (target_sum / x.sum())
    return x


def _embed_sequence(src: np.ndarray, n: int) -> np.ndarray:
    """
    Embed ``src`` (length m) into a zero‑padded array of length n (n≥m) centred.
    If n < m the source is truncated.
    """
    m = src.shape[0]
    if n == m:
        return src.copy()
    if n < m:
        return src[:n].copy()
    out = np.zeros(n, dtype=np.float64)
    start = (n - m) // 2
    out[start:start + m] = src
    return out


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fast SA on the simplex.
    Returns (best_sequence, best_max_convolution).
    """
    cur = start_seq.copy()
    n = cur.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # ensure exact ℓ¹‑norm (small numeric drift may accumulate)
    cur = cur * (target_sum / cur.sum())

    cur_max, cur_idx, cur_conv = _max_convolution(cur)
    best_seq = cur.copy()
    best_val = cur_max

    start_time = time.monotonic()
    total_time = deadline - start_time
    # counters for stagnation handling
    no_improve = 0
    RESTART_AFTER = 8000            # restart if no improvement this many steps

    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        elapsed = now - start_time
        fraction = elapsed / total_time

        # --- temperature and step‑size schedules (linear decay) ---
        temperature = max(0.5 * (1.0 - fraction), 1e-4)
        move_frac = max(0.005, 0.18 * (1.0 - fraction))

        # --- pick a source index -------------------------------------------------
        # With high probability pick from the “active set” that contributes
        # to the current maximal convolution entry.
        low = max(0, cur_idx - (n - 1))
        high = min(n - 1, cur_idx)
        if (high >= low) and rng.random() < 0.7:
            src = rng.integers(low, high + 1)
        else:
            # weighted by current magnitude (more interesting)
            probs = cur / cur.sum()
            src = rng.choice(n, p=probs)

        if cur[src] <= 1e-14:
            continue

        # --- pick a target index (any index different from src) -------------------
        tgt = rng.integers(n - 1)
        if tgt >= src:
            tgt += 1

        # --- propose move ---------------------------------------------------------
        epsilon = cur[src] * move_frac * rng.random()
        if epsilon <= 0.0:
            continue
        cand = cur.copy()
        cand[src] -= epsilon
        cand[tgt] += epsilon

        # (still non‑negative by construction, also ℓ¹‑norm is preserved)

        # --- evaluate candidate ----------------------------------------------------
        cand_max, cand_idx, _ = _max_convolution(cand)
        delta = cand_max - cur_max

        # --- Metropolis acceptance -------------------------------------------------
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / temperature):
                accept = True

        if accept:
            cur = cand
            cur_max = cand_max
            cur_idx = cand_idx
            # keep best
            if cur_max < best_val:
                best_val = cur_max
                best_seq = cur.copy()
                no_improve = 0
                print(f"[SA] new best max={best_val:.6f} (n={n})")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # --- restart if stuck ------------------------------------------------------
        if no_improve >= RESTART_AFTER and (deadline - time.monotonic()) > 5.0:
            cur = _initial_dirichlet(n, rng, target_sum)
            cur_max, cur_idx, _ = _max_convolution(cur)
            no_improve = 0

    return best_seq, best_val


def _gaussian_polish(start_seq: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    A short Gaussian hill‑climbing phase that keeps the ℓ¹‑norm fixed via
    re‑normalisation.  Useful for final fine tuning.
    """
    cur = start_seq.copy()
    n = cur.shape[0]
    target_sum = math.sqrt(2.0 * n)

    cur_val, _, _ = _max_convolution(cur)
    best_seq = cur.copy()
    best_val = cur_val
    step = max(cur.mean(), 1e-2) * 0.3

    while time.monotonic() < deadline:
        cand = cur + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, None)
        # re‑normalise to the exact ℓ¹‑norm
        s = cand.sum()
        if s <= 0.0:
            continue
        cand = cand * (target_sum / s)
        cand_val, _, _ = _max_convolution(cand)
        if cand_val < best_val:
            best_seq = cand.copy()
            best_val = cand_val
            cur = cand
            cur_val = cand_val
            step *= 0.95
            print(f"[polish] improved to {best_val:.6f}")
        else:
            # gradually shrink step if nothing improves
            step *= 0.9995
            if step < 1e-6:
                step = max(cur.mean(), 1e-2) * 0.3
    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.

    The search proceeds over several candidate lengths (300 … 2500).  For each
    length a directed simulated‑annealing loop is run; the best sequence across
    all lengths is kept.  A final Gaussian polishing step refines the winner if
    time permits.  The function returns a plain list of floats ready for
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a thin safety margin for the outer framework
    deadline = start_time + max(0.0, float(budget_s) - 0.15)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction (if any)
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq0 = np.asarray(h1, dtype=np.float64)
        n0 = seq0.size
        target0 = math.sqrt(2.0 * n0)
        if seq0.sum() > 0:
            seq0 = seq0 * (target0 / seq0.sum())
        else:
            seq0 = np.full(n0, target0 / n0, dtype=np.float64)
        val0 = evaluate_sequence(seq0.tolist())
        best_seq = seq0.copy()
        best_val = float(val0)
        print(f"[init] warm‑start from length {n0}, score={best_val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Explore several lengths with SA
    # --------------------------------------------------------------
    # A modest but diverse pool of lengths (larger lengths give more freedom)
    candidate_lengths = [300, 500, 800, 1000, 1500, 1800, 2100, 2500]

    for n in candidate_lengths:
        if time.monotonic() >= deadline - 5.0:   # leave a few seconds for polishing
            break

        target_sum = math.sqrt(2.0 * n)

        # initialise a starting point
        if best_seq is not None and best_seq.shape[0] == n:
            start_seq = best_seq.copy()
        else:
            # try to embed the current global best (if any) otherwise random
            if best_seq is not None:
                start_seq = _embed_sequence(best_seq, n)
                # after embedding we must re‑normalise the ℓ¹‑norm
                start_seq = start_seq * (target_sum / start_seq.sum())
            else:
                start_seq = _initial_dirichlet(n, rng, target_sum)

        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.monotonic()
        if remaining <= 5.0:
            break
        # at most 30 s per length, but give more to larger n if time permits
        slice_deadline = time.monotonic() + min(30.0, remaining * 0.5)

        seq, val = _simulated_annealing(start_seq, slice_deadline, rng)

        # keep the globally best sequence
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[global] new best (n={n}) max={best_val:.6f}")

        # early exit if we already beat the target (1.503)
        if best_val < 1.503:
            print("[stop] target threshold reached")
            break

    # --------------------------------------------------------------
    # 2️⃣ Final polishing (Gaussian hill‑climbing) if time left
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline - 0.5:
        refined_seq, refined_val = _gaussian_polish(best_seq, deadline, rng)
        if refined_val < best_val:
            best_seq = refined_seq
            best_val = refined_val
            print(f"[final] polished max={best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Safety clipping and final exact evaluation
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final exact score (the framework will also compute it, but we print it)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[result] best score = {final_score:.6f}  (length={best_seq.size})")

    return [float(x) for x in best_seq]
