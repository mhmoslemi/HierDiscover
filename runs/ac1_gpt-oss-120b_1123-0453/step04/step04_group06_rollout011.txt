"""
Modified search for a low‑value step sequence.

The evaluation target is
    score = 2·n·max(conv(f,f)) / (sum(f))² .
For a fixed ℓ¹‑norm (∑f = √(2n)) this equals the peak of the
auto‑convolution, i.e. we minimise  max_k (f*f)_k.

Algorithm
---------

1.  Warm‑start from a previously stored construction if available.
2.  For several lengths (300 … 1000) perform a number of random restarts.
    Each restart runs an Adam optimiser on a *smooth‑max* surrogate:
    w_k ∝ exp(β·conv_k) and the gradient of the weighted sum
    ∑_k w_k (f*f)_k is computed analytically.
    After every step the iterate is projected back onto the ℓ¹‑simplex
    (∑ = √(2 n), f_i ≥ 0) and clipped to the limits required by
    ``evaluate_sequence``.
    The temperature β is slowly increased so that the optimisation
    focuses on the true max.
3.  When the time budget is almost exhausted a short stochastic hill‑climb
    (Gaussian perturbations + simplex projection) is used to polish the
    best candidate.
4.  The best sequence found is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_score(v: np.ndarray) -> float:
    """Exact value of the provided ``evaluate_sequence`` expression."""
    n = len(v)
    conv = np.convolve(v, v)
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _weighted_gradient(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(f*f)_k w.r.t. f (f ≥ 0).
    w must have length 2·len(f)‑1 and Σw = 1.
    """
    n = len(f)
    grad = np.zeros_like(f)
    # each convolution offset contributes a Toeplitz‑shaped slice
    for k in range(2 * n - 1):
        wk = w[k]
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        grad[i_start:i_end + 1] += wk * 2.0 * f[j_range]
        # diagonal correction (the term i=j appears only once in the sum)
        if (k & 1) == 0:            # k even → possible i = j
            i_eq = k // 2
            if i_start <= i_eq <= i_end:
                grad[i_eq] -= wk * f[i_eq]   # 2·c – c = c
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 3000,
    beta_start: float = 0.5,
    beta_factor: float = 1.04,
    beta_max: float = 1024.0,
    step_initial: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth‑max proxy for the peak of the auto‑convolution.
    Returns the best vector seen and its exact score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)

    # Adam state
    m = np.zeros_like(f)          # first moment
    v = np.zeros_like(f)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = step_initial
    beta = beta_start
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        max_val = conv.max()

        # smooth‑max weights (stable log‑sum‑exp)
        exp_vals = np.exp(beta * (conv - max_val))
        w_sum = exp_vals.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_vals / w_sum

        grad = _weighted_gradient(f, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        update = step * m_hat / (np.sqrt(v_hat) + eps)
        cand = f - update
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best_f, best_score = cand.copy(), cand_score
            step *= 1.02
            no_improve = 0
        else:
            step *= 0.94
            no_improve += 1

        # progress the iterate regardless of improvement
        f = cand

        # anneal the smoothing parameter β
        beta = min(beta * beta_factor, beta_max)

        # if we have been stuck for a while, temporarily cool down β
        if no_improve >= 250:
            beta = max(beta_start, beta / 2.0)
            no_improve = 0

        # stop if step becomes tiny
        if step < 1e-12:
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# final stochastic polishing
# ----------------------------------------------------------------------


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.2,
    max_iters: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing on the simplex.
    Keeps the best point seen; accepts only improvements.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score(cand)

        if cand_score < best_score:
            best, best_score = cand, cand_score
        # exponential decay of the perturbation magnitude
        sigma *= 0.9997
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial
    return best, best_score
