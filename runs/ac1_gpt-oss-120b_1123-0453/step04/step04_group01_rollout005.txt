"""
Gradient‑softmax descent for the step‑function minimisation challenge.

The evaluation function is
    F(a) = 2·n·max_k (a * a)[k] / (∑ a_i)².

For a fixed length n we keep the ℓ¹–norm exactly √(2·n); then
    F(a) = max_k (a * a)[k].

We minimise the maximum entry of the auto‑convolution.
Instead of moving mass only according to the *active* index,
we smooth the max by a soft‑max (log‑sum‑exp) with a temperature β.
The gradient of the smoothed objective is a weighted sum of the
gradients of the individual convolution entries, which can be computed
by a single extra convolution.  A projected (Euclidean) gradient step
onto the simplex {a ≥ 0, ∑ a = √(2n)} yields a new feasible vector.
β is increased slowly so the algorithm first sees a smooth landscape
and then focuses on the true maximum.

The search proceeds as follows

0.  If the notebook already defines a good construction (`height_sequence_1`)
    it is used as a seed.
1.  For several lengths (1000 … 200) a modest set of parametric families
    (uniform, triangular, power‑law, exponential, Beta, sine‑powered,
    random Dirichlet) is generated.
2.  Each seed is refined by the soft‑max gradient descent for a few
    seconds; the best vector found overall is remembered.
3.  A handful of pure random restarts are processed in the same way.
4.  The final best vector is polished with cheap hill‑climbing moves
    (swap‑based mass transfer and a short Gaussian perturbation).

All operations respect the security checks of ``evaluate_sequence``:
entries stay non‑negative, below 1000, and the ℓ¹‑norm is exactly
√(2·n).  The routine stops a little before the global deadline and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``vec`` so that its ℓ¹‑norm equals ``target_sum`` (preserving sign)."""
    s = vec.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        n = vec.shape[0]
        return np.full(n, target_sum / n, dtype=np.float64)
    return vec * (target_sum / s)


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` without the safety wrapper.
    For a sequence normalised to √(2·n) this equals max auto‑convolution entry.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        S = {x | x_i ≥ 0, ∑ x_i = target_sum }.
    Implements the algorithm of Wang & Carreira‑Perpinyà (2013).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_family(n: int, a: float, b: float, target_sum: float) -> np.ndarray:
    """
    Generate a discretised Beta‑distribution shape.
    a_i ∝ x^{a‑1} (1‑x)^{b‑1}  over x∈(0,1).
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    w = np.maximum(w, 1e-12)      # avoid exact zeros
    w = w / w.sum() * target_sum
    return w


def _generate_candidates(
    n: int, target_sum: float, rng: np.random.Generator
) -> list[np.ndarray]:
    """
    Produce a modest fleet of plausible start vectors for length ``n``.
    """
    cand = []

    # uniform
    cand.append(np.full(n, target_sum / n, dtype=np.float64))

    # triangular (peak in the centre)
    x = np.arange(n, dtype=np.float64)
    tri = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
    tri[tri < 0] = 0.0
    cand.append(_normalize(tri, target_sum))

    # power‑law a_i ∝ i⁻ᵖ   (i starts at 1)
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.1, 3.0, 12):
        a = idx ** (-p)
        cand.append(_normalize(a, target_sum))

    # exponential a_i ∝ e^{-α·i}
    for alpha in np.linspace(0.001, 0.20, 8):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand.append(_normalize(a, target_sum))

    # beta family with random parameters
    for _ in range(8):
        a_param = rng.uniform(0.1, 5.0)
        b_param = rng.uniform(0.1, 5.0)
        cand.append(_beta_family(n, a_param, b_param, target_sum))

    # arcsine (Beta(½,½))
    cand.append(_beta_family(n, 0.5, 0.5, target_sum))

    # sine‑powered shapes
    for p in np.linspace(0.5, 3.0, 6):
        a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1))) ** p
        cand.append(_normalize(a, target_sum))

    # a few random Dirichlet draws
    for _ in range(4):
        a = rng.dirichlet(np.ones(n)) * target_sum
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Soft‑max gradient utilities
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the softened objective
        g(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
    evaluated at ``a``.
    Returns a vector of length len(a).
    """
    conv = np.convolve(a, a)                 # length 2·n‑1
    max_c = conv.max()
    # weights ∈ (0,1]; the maximal entry gets weight 1
    w = np.exp(beta * (conv - max_c))

    # grad_i = 2 * Σ_j w_{i+j} * a_j   (see derivation in the analysis)
    rev = a[::-1]
    full = np.convolve(w, rev, mode="full")
    start = len(a) - 1
    grad = 2.0 * full[start : start + len(a)]
    return grad


def _optimize_sequence(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 10_000,
) -> np.ndarray:
    """
    Heuristic optimisation based on a soft‑max approximation of the
    max‑convolution.  Performs projected gradient steps on the
    ℓ¹‑simplex with an increasing inverse temperature β.
    """
    a = _normalize(seq, target_sum)
    n = a.shape[0]

    # keep the best vector seen so far (might be different from the current a)
    best = a.copy()
    best_sc = _fast_score(best)

    # parameters for the schedule
    beta = 1.0
    beta_max = 30.0                         # focus more on the true max later
    # initial step size – roughly a small fraction of the average entry
    step = target_sum * 0.2 / n

    # previous score (for simple acceptance rule)
    prev_sc = best_sc

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        grad = _softmax_gradient(a, beta)

        # projected gradient step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        sc = _fast_score(cand)

        # keep global best
        if sc < best_sc:
            best = cand.copy()
            best_sc = sc
            # a small positive print helps debugging without overwhelming the log
            print(f"[gd] iter {it:4d} new best {sc:.6f}")

        # simple Metropolis‑style acceptance – accept if improves, otherwise
        # keep the current point (occasionally allow uphill moves)
        if sc < prev_sc:
            a = cand
            prev_sc = sc
        else:
            # with a tiny probability we still move to avoid being stuck
            if rng.random() < 0.001:
                a = cand
                prev_sc = sc

        # adapt step size
        if sc < prev_sc:
            step = min(step * 1.08, target_sum * 0.6 / n)
        else:
            step = max(step * 0.73, target_sum * 1e-5 / n)

        # increase β slowly
        if beta < beta_max:
            beta = min(beta * 1.018, beta_max)

    return best


# ----------------------------------------------------------------------
# Very cheap hill‑climbing polishers (same as in the previous runner)
# ----------------------------------------------------------------------


def _swap_hill_climb(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """Swap a small amount of mass between two coordinates."""
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target_sum * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _project_onto_simplex(cand, target_sum)

        sc = _fast_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
    return best


def _gaussian_local_search(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """Short annealing‑style Gaussian perturbation."""
    best = seq.copy()
    best_sc = _fast_score(best)
    step = target_sum * 0.02

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape[0])
        cand = _project_onto_simplex(cand, target_sum)

        sc = _fast_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
            step *= 0.92
        else:
            step *= 0.99
            if step < 1e-6:
                step = target_sum * 0.02
    return best


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed (reproducible runs).
    budget_s : float
        Time budget in seconds.  The routine stops a little before the limit.
    kwargs : dict
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)
    # Give ourselves a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.4)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Initialise from a previously discovered construction, if any.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = _normalize(init, target0)
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception:  # pragma: no cover
            pass

    # --------------------------------------------------------------
    # 1️⃣  Broad sweep over a few lengths
    # --------------------------------------------------------------
    lengths = [1000, 800, 600, 500, 400, 300, 200]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # -----------------------------------------------------------------
        # a) generate a pool of plausible seeds for this length
        # -----------------------------------------------------------------
        pool = _generate_candidates(n, target, rng)

        # -----------------------------------------------------------------
        # b) optimise each seed (time‑budget aware)
        # -----------------------------------------------------------------
        for seed_vec in pool:
            if time.monotonic() >= deadline:
                break
            # allocate a modest slice of the remaining budget
            remaining = deadline - time.monotonic()
            sub_deadline = time.monotonic() + min(6.0, remaining * 0.15)
            refined = _optimize_sequence(seed_vec, target, sub_deadline, rng)
            sc = _fast_score(refined)
            if sc < best_score:
                best_seq, best_score = refined.copy(), sc
                print(f"[sweep] n={n} new best {sc:.6f}")

        # -----------------------------------------------------------------
        # c) a few pure random restarts (might discover new basins)
        # -----------------------------------------------------------------
        for restart in range(2):
            if time.monotonic() >= deadline:
                break
            start = rng.dirichlet(np.ones(n)) * target
            sub_deadline = time.monotonic() + min(5.0, deadline - time.monotonic())
            refined = _optimize_sequence(start, target, sub_deadline, rng)
            sc = _fast_score(refined)
            if sc < best_score:
                best_seq, best_score = refined.copy(), sc
                print(f"[restart] n={n} run {restart} new best {sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Final polish on the globally best vector
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: a modest uniform vector of length 200
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # ensure exact ℓ¹‑norm (tiny drift may have occurred)
    n = best_seq.shape[0]
    target = math.sqrt(2.0 * n)
    best_seq = _project_onto_simplex(best_seq, target)

    # cheap swap‑based hill‑climb
    best_seq = _swap_hill_climb(best_seq, target, deadline, rng)

    # final short Gaussian perturbation if time permits
    if time.monotonic() < deadline:
        best_seq = _gaussian_local_search(best_seq, target, deadline, rng)

    final_score = _fast_score(best_seq)
    print(f"[final] best score {final_score:.6f}   length {best_seq.shape[0]}")

    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
