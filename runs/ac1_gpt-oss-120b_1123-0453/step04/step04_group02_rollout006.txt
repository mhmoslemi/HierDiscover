def project_onto_simplex(v, S):
    # v is ndarray.
    # ensure nonnegative after projection.
    if v.sum() == S and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - S) / np.arange(1, len(v)+1))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

This is standard.

        conv = _conv_fft(a) # full length 2n-1

"""
Smoothed‑gradient descent for the max‑auto‑convolution problem.

The evaluation function reduces to the maximal entry of the linear
convolution of the (non‑negative) step heights after they have been
scaled to ℓ¹‑norm √(2·n).  We minimise this quantity by a
first‑order method that works on a smooth approximation of the max:
    Lβ(a) = (1/β)·log Σk exp(β·(a*a)[k]).
Its gradient can be computed efficiently with two FFT‑based convolutions.
For each length we start from a good parametric family (exponential,
power‑law, Hann …) and also from a few random Dirichlet draws,
run the smoothed gradient descent with adaptive step‑size and a
gradually increasing smoothing parameter β, and finally perform a short
random hill‑climb.  The best sequence encountered before the time budget
expires (≈1000 s) is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolutions (FFT based)
# ----------------------------------------------------------------------
def _full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    N = 1 << ((n + m - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)
    return conv[: n + m - 1]


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution a*a."""
    return _full_convolve_fft(a, a)


# ----------------------------------------------------------------------
# Exact objective (max of the auto‑convolution)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Return max_k (a*a)[k] ."""
    return _conv_self_fft(a).max()


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = S}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = S}."""
    # Numerical safety: if the sum is already S and all entries non‑negative,
    # simply return a copy.
    if v.sum() == S and np.all(v >= 0):
        return v.copy()
    # Standard O(n log n) algorithm.
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - S) / np.arange(1, len(v) + 1))[0]
    if len(rho) == 0:                        # all entries will be zero
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - S) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for a quick good start)
# ----------------------------------------------------------------------
def _candidate_families(n: int, target_sum: float):
    """Return the best vector among a collection of hand‑crafted families."""
    best_vec = None
    best_sc = float("inf")
    idx = np.arange(n, dtype=np.float64)

    def _norm(v):
        v = np.asarray(v, dtype=np.float64)
        if v.sum() == 0:
            return np.full_like(v, target_sum / n)
        return v * (target_sum / v.sum())

    # Uniform baseline
    a = np.full(n, target_sum / n, dtype=np.float64)
    best_vec, best_sc = a, _max_conv(a)

    # Exponential families a_i ∝ exp(-α·i)
    alphas = np.concatenate([
        np.arange(1, 6) / (10.0 * n),          # very fine near 0
        np.arange(1, 21) / n,                  # α≈k/n, k=1…20
        np.linspace(0.02, 0.20, 40),           # medium region
        np.linspace(0.20, 1.00, 30)            # larger decays
    ])
    alphas = np.unique(alphas)
    for α in alphas:
        a = np.exp(-α * idx)
        a = _norm(a)
        sc = _max_conv(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # Power‑law families a_i ∝ (i+1)^(-p)
    p_vals = np.linspace(0.05, 5.0, 120)
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        a = base ** (-p)
        a = _norm(a)
        sc = _max_conv(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # Classic windows
    def _try(a):
        nonlocal best_vec, best_sc
        a = _norm(a)
        sc = _max_conv(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))          # Hann
    _try(a)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))          # Hamming
    _try(a)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1)))            # Blackman
    _try(a)
    a = np.sin(np.pi * (idx + 1) / (n + 1))                        # Sine window
    _try(a)
    a = np.minimum(idx + 1, n - idx)                               # Triangular
    _try(a)

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Smoothed gradient descent on log‑sum‑exp approximation of max
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float):
    """
    Minimise max_k (a*a)[k] using a smooth approximation.
    Returns the best sequence found and its (exact) max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _max_conv(a)

    # Parameters for the optimisation
    step = 0.01                     # initial step length
    beta = 5.0                      # smoothing parameter
    beta_factor = 1.02
    beta_max = 200.0
    rng = np.random.default_rng()
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # ----- compute current convolution & softmax weights -----
        conv = _conv_self_fft(a)                     # length 2n‑1
        max_c = conv.max()
        # soft‑max weights (log‑sum‑exp gradient)
        exp_term = np.exp(beta * (conv - max_c))     # safe shift
        w = exp_term / exp_term.sum()                # shape (2n‑1,)

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]                              # needed for convolution
        rev_conv = _full_convolve_fft(a, w_rev)      # length 3n‑2
        grad = 2.0 * rev_conv[2 * n - 2 - np.arange(n)]

        # ----- attempt a projected step -----
        improved = False
        cur_step = step
        while cur_step > 1e-9:
            cand = a - cur_step * grad
            # projection onto the feasible set (non‑negative, sum = target)
            cand = np.clip(cand, 0.0, 1000.0)
            total = cand.sum()
            if total == 0.0:
                cand = np.full(n, target_sum / n, dtype=np.float64)
            else:
                cand = cand * (target_sum / total)

            cand_val = _max_conv(cand)
            if cand_val < best_val - 1e-12:
                # strict improvement
                a = cand
                best_seq, best_val = cand.copy(), cand_val
                print(f"[gd] iter {it} new best {best_val:.6f}")
                improved = True
                # enlarge the step a little after a success
                step = min(step * 1.1, 1.0)
                break
            elif cand_val <= max_c + 1e-12:
                # non‑worsening move – accept to keep making progress
                a = cand
                improved = True
                step = min(step * 1.05, 1.0)
                break
            else:
                cur_step *= 0.5        # backtrack

        if not improved:
            # step was too aggressive – shrink globally
            step *= 0.5
            if step < 1e-7:
                # occasional random shake‑up to escape flat regions
                i, j = rng.integers(n, size=2)
                delta = a[i] * 0.05
                if delta > 0:
                    cand = a.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand = cand * (target_sum / cand.sum())
                    cand_val = _max_conv(cand)
                    if cand_val < best_val:
                        a = cand
                        best_seq, best_val = cand.copy(), cand_val
                        print(f"[shake] iter {it} improved {best_val:.6f}")
                step = 0.01

        # increase smoothing gradually
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # early exit if we have already beaten the target bound
        if best_val < 1.503:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny stochastic polishing (random local moves)
# ----------------------------------------------------------------------
def _polish_local(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Final short random hill‑climb."""
    a = seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    best_val = _max_conv(a)
    rng = np.random.default_rng()
    step = 0.1

    while time.monotonic() < deadline:
        i, j = rng.integers(n, size=2)
        delta = a[i] * 0.03
        if delta == 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target / cand.sum())
        cand_val = _max_conv(cand)
        if cand_val < best_val:
            a, best_val = cand, cand_val
            # gradually shrink the perturbation radius
            step = max(step * 0.995, 1e-4)
        else:
            step *= 0.999
    return a, best_val

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # keep a small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Use the previously discovered construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = tmp.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                tmp = tmp * (target0 / tmp.sum())
                sc0 = _max_conv(tmp)
                best_seq, best_score = tmp, sc0
                print(f"[init] height_sequence_1 n={n0} score={sc0:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # Lengths to explore (moderate sizes give a good trade‑off)
    # --------------------------------------------------------------
    lengths = [300, 500, 800, 1200]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ----- 1️⃣  start from a good parametric family -----
        fam_vec, fam_sc = _candidate_families(n, target)
        if fam_sc < best_score:
            best_seq, best_score = fam_vec.copy(), fam_sc
            print(f"[family] n={n} new best {best_score:.6f}")

        # ----- 2️⃣  run smoothed gradient descent on that family -----
        sub_dead = deadline - 0.1
        seq, sc = _smooth_gradient_descent(fam_vec, target, sub_dead)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[gd] n={n} improved to {best_score:.6f}")

        # ----- 3️⃣  a few random Dirichlet restarts for robustness -----
        for _ in range(4):
            if time.monotonic() >= deadline:
                break
            a = rng.random(n)
            a = a / a.sum() * target
            seq, sc = _smooth_gradient_descent(a, target, deadline)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[rand] n={n} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Final stochastic polishing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polished, sc_polish = _polish_local(best_seq, deadline)
        if sc_polish < best_score:
            best_seq, best_score = polished, sc_polish
            print(f"[final] polished to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _max_conv(best_seq)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    return [float(v) for v in best_seq]
