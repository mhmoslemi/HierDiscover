def _exponential_descent(a, target_sum, deadline, iters=2000, eta=0.5):
    best_a = a.copy()
    best_score = _score(a)
    for t in range(1, iters+1):
        if time.monotonic() >= deadline: break
        # compute conv and max
        conv = np.convolve(a, a)
        max_val = np.max(conv)
        # compute active set
        active = _active_offsets(conv, max_val)
        # compute subgradient of max (grad_max)
        grad_max = _gradient_submax(a, active)
        # compute factor
        factor = 2.0 * len(a) / (np.sum(a) ** 2)
        grad = factor * grad_max
        # multiplicative update
        eta_t = eta / np.sqrt(t)  # diminishing
        a = a * np.exp(-eta_t * grad)
        # projection onto simplex
        a = _project_onto_simplex(a, target_sum)
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _hill_climb(a, target_sum, deadline):
    n = len(a)
    conv = np.convolve(a, a)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (target_sum**2)  # if sum equals target_sum

    best = a.copy()
    best_score = cur_score

    T = 0.2
    decay = 0.9995
    step_scale = target_sum * 0.05 / n
    it = 0
    rng = np.random.default_rng()
    while time.monotonic() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if i == j:
            continue
        max_move = a[i]
        if max_move <= 0:
            continue
        delta = rng.random() * min(step_scale, max_move)
        # update conv in place
        # store copies to revert if needed
        conv_i_slice = slice(i, i+n)
        conv_j_slice = slice(j, j+n)
        # subtract contributions from i using current a (before change)
        conv[conv_i_slice] -= delta * a
        conv[conv_j_slice] += delta * a

        new_max = conv.max()
        # accept criteria: if new_max < cur_max else metropolis
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            cur_score = 2.0 * n * cur_max / (target_sum**2)
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score
                print(f"[HC] it={it} score={cur_score:.6f}")
        else:
            # revert conv changes
            conv[conv_i_slice] += delta * a
            conv[conv_j_slice] -= delta * a

        # temperature schedule
        T *= decay
        if T < 1e-8:
            T = 0.2

        # shrink step scale
        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:

            cur_score = 2.0 * n * cur_max / (target_sum**2)

                best_score = cur_score

"""
search.py – an alternative optimisation for `evaluate_sequence`.

Strategy
--------
1. Warm‑start from the provided `height_sequence_1` (if any).
2. For a collection of problem sizes `n` we run many *restarts* of a
   simple exponential‑gradient descent on the ℓ¹‑simplex.  The descent
   uses the true sub‑gradient of the maximal auto‑convolution entry and
   an adaptive multiplicative update (mirror descent).  The iterate is
   projected back onto the simplex after each step.
3. The best candidate from the previous phase is refined by a fast
   hill‑climber that moves a small amount of mass from one index to
   another while updating the convolution incrementally (O(n) per move).
   An annealing schedule occasionally accepts uphill moves.
4. A lightweight parametric sweep (geometric sequences) is performed
   in the remaining time – any improvement replaces the current best.
5. The final best sequence is returned as a plain Python list.

All helper functions are defined at top level; no closures or lambdas
are used.  The code respects the supplied time budget (default 1000 s)
and prints progress messages useful for debugging.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Exact objective value (identical to evaluate_sequence)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    sum_val = np.sum(a)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where the convolution attains (almost) the maximal value."""
    eps = max_val * 1e-12 + 1e-14
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the worst auto‑convolution entry w.r.t. a.
    Returned gradient corresponds to the derivative of `max_k (a*a)_k`.
    """
    n = len(a)
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
        mask = (i == j)
        if np.any(mask):
            g[i[mask]] -= a[i[mask]]
    return g


# ----------------------------------------------------------------------
# Exponential‑gradient descent on the simplex
# ----------------------------------------------------------------------


def _exp_descent(a0: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent style optimisation that minimises the max auto‑convolution.
    The iterate stays on the ℓ¹‑simplex Σa = target_sum, a ≥ 0.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)

    n = len(a)
    factor_const = 2.0 * n / (target_sum * target_sum)

    it = 0
    while it < max_iters and time.monotonic() < deadline:
        it += 1
        conv = np.convolve(a, a)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)

        # sub‑gradient of the maximal convolution entry
        gmax = _gradient_submax(a, active)

        # full gradient of the objective (the constant factor can be omitted)
        grad = factor_const * gmax

        # diminishing step size for the multiplicative update
        eta = 0.5 / math.sqrt(it)

        # multiplicative (mirror) update, keep a tiny epsilon to stay positive
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-12)

        # project back onto the simplex (preserves total mass)
        a = _project_onto_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[EXP] it={it:4d} n={n} score={cur_val:.6f}")

        # early exit if score is already very good
        if best_val < 1.45:
            break

    return best, best_val


# ----------------------------------------------------------------------
# Hill‑climbing with incremental convolution updates
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                target_sum: float,
                deadline: float,
                init_temp: float = 0.3,
                decay: float = 0.9997) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climber.
    A move transfers a small amount of mass from i → j; the convolution
    is updated incrementally (O(n)).  The total ℓ¹‑mass stays constant.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = len(a)

    conv = np.convolve(a, a)
    cur_max = np.max(conv)
    cur_score = 2.0 * n * cur_max / (target_sum * target_sum)

    best = a.copy()
    best_score = cur_score

    T = init_temp
    step_scale = 0.05 * target_sum / n   # typical transferred mass

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue

        delta = rng.random() * min(step_scale, a[i])
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        conv[i:i + n] -= delta * a
        conv[j:j + n] += delta * a

        new_max = conv.max()
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            cur_score = 2.0 * n * cur_max / (target_sum * target_sum)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[HC ] it={it:7d} n={n} score={cur_score:.6f}")
        else:
            # revert the tentative change
            conv[i:i + n] += delta * a
            conv[j:j + n] -= delta * a

        # temperature decay
        T *= decay
        if T < 1e-9:
            T = init_temp

        # slowly shrink the typical move size
        step_scale *= 0.9999
        if step_scale < 1e-8 * target_sum:
            step_scale = 0.05 * target_sum / n

        # occasional re‑projection to keep numerical drift in check
        if it % 1024 == 0:
            a = _project_onto_simplex(a, target_sum)
            conv = np.convolve(a, a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max / (target_sum * target_sum)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best, best_score


# ----------------------------------------------------------------------
# Simple parametric family (geometric) – useful for a quick sweep
# ----------------------------------------------------------------------


def _geometric_candidate(n: int, r: float, target_sum: float) -> np.ndarray:
    """Return a geometric sequence a_i = r**i, projected onto the simplex."""
    a = np.array([r ** i for i in range(n)], dtype=np.float64)
    a = _project_onto_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.
    The routine respects the time budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously found construction (if present)
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64).copy()
        seq = np.clip(seq, 0.0, 1000.0)
        n = len(seq)
        target = math.sqrt(2.0 * n)
        if seq.sum() == 0.0:
            seq = np.full(n, target / n)
        else:
            seq = seq * (target / np.sum(seq))
        s = _score(seq)
        best_seq = seq.copy()
        best_score = s
        print(f"[init] warm‑start length={n}, score={s:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Exponential‑gradient phase on many lengths / restarts
    # ------------------------------------------------------------------
    lengths = [256, 300, 400, 500, 600, 800, 1024, 1200, 1500, 1800, 2048]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        restarts = 10 if n <= 1000 else 6
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a start point
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
                f0 += rng.normal(scale=0.001 * target_sum, size=n)
                f0 = np.clip(f0, 0.0, None)
                f0 = _project_onto_simplex(f0, target_sum)
            else:
                f0 = rng.random(n)
                f0 = _project_onto_simplex(f0, target_sum)

            sub_deadline = min(deadline, time.monotonic() + 6.0)
            f_opt, sc_opt = _exp_descent(f0, target_sum, sub_deadline,
                                         max_iters=3000)

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = f_opt.copy()
                print(f"[exp] n={n} restart={r} score={sc_opt:.6f}")

            # early exit if we already beat the target bound
            if best_score < 1.45:
                print("[early] target reached during exponential descent")
                break
        if best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Geometric‑family sweep for extra diversity
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        for n in [500, 800, 1200, 1600]:
            if time.monotonic() >= deadline:
                break
            target_sum = math.sqrt(2.0 * n)
            # sweep r in a narrow high‑range – values very close to 1
            for r in np.linspace(0.985, 0.9995, 30):
                a = _geometric_candidate(n, r, target_sum)
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[geom] n={n} r={r:.4f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Hill‑climbing (simulated annealing) polish on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        refined, refined_score = _hill_climb(best_seq, target_sum, deadline,
                                            init_temp=0.3, decay=0.9995)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final] hill‑climb improved score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] using uniform construction")

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(v) for v in best_seq]
