def _project_onto_simplex(v, target):
    # algorithm from Duchi et al.
    # v is 1D numpy array.
    if v.sum() == target and np.all(v >= 0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _adam_optimize(start_seq, deadline, rng):
    n = start_seq.shape[0]
    target = math.sqrt(2.0 * n)
    a = start_seq.copy()
    # Ensure within bounds and normalized
    a = np.clip(a, 0, 1000)
    total = a.sum()
    if total > 0:
        a = a * (target/total)
    else:
        a = np.full(n, target/n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = target / n * 0.5  # heuristic step
    beta = 1.0
    best_seq = a.copy()
    best_score = _conv_max(a)
    it = 0
    max_iters = 5000
    while time.monotonic() < deadline and it < max_iters:
        conv = _linear_convolve(a, a)  # length 2n-1
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
        # smooth approx
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto nonnegative and simplex
        a = np.clip(a, 0.0, 1000.0)
        # project onto simplex with target sum
        a = _project_onto_simplex(a, target)
        # schedule
        if it % 100 == 0 and beta < 400:
            beta *= 1.5
        # optional step decay
        if it % 200 == 0:
            step *= 0.95
        it += 1
    return best_seq, best_score

def _anneal_refine(seq, deadline, rng):
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    cur = seq.copy()
    cur_score = _conv_max(cur)
    best_seq = cur.copy()
    best_score = cur_score
    # temperature schedule
    T = cur_score * 0.1 + 1e-4
    decay = 0.9995
    max_iters = 1000000  # high but break by time
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # propose moving mass
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0:
            it += 1
            continue
        delta = cur[i] * rng.uniform(0.01, 0.2)  # move 1-20% of mass at i
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # ensure bound 1000
        if new[j] > 1000:
            # clip excess back to i if possible
            excess = new[j] - 1000
            new[j] = 1000
            new[i] += excess
            if new[i] > 1000:
                it += 1
                continue
        # new is feasible (nonnegative, sum unchanged)
        new_score = _conv_max(new)
        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()
        # temperature decay
        T *= decay
        # occasionally boost temperature to escape plateaus
        if it % 5000 == 0:
            T = max(T, cur_score * 0.05)
        it += 1
    return best_seq, best_score

"""
Propose a low‑value step sequence.

The search proceeds in three stages:

0️⃣ Warm‑start from a previously found construction if it exists.
1️⃣ Sweep several parametric families (power‑law, exponential, sinusoid,
   Gaussian) for many lengths and keep the best.
2️⃣ For each length run several random restarts of an Adam‑style projected
   gradient descent that minimizes a smooth‑max surrogate of the
   convolution peak.  The surrogate is gradually sharpened (β ↑) and the
   iterate is projected onto the ℓ¹‑simplex (∑a = √(2 n), a ≥ 0).
3️⃣ Finally a cheap simulated‑annealing “mass‑swap’’ refinement is applied
   to the globally best vector.

All vectors are kept feasible (non‑negative, ≤1000, correct ℓ¹‑norm), and
the best sequence encountered before the time budget expires is returned.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (no closures)
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return a FFT‑friendly length (fallback to a power of two)."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for many calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Exact maximum of the self‑convolution of ``a``."""
    n = a.shape[0]
    size = _next_fast_len(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max weights for the smooth‑max surrogate."""
    cmax = conv.max()
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    For each i   ∂/∂a_i  (∑_k w_k (a*a)_k) = 2·∑_j w_{i+j}·a_j .
    Obtained efficiently by one FFT‑based convolution.
    """
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)          # length = 3·n‑2
    n = a.shape[0]
    seg = corr[n - 1 : 2 * n - 1]               # central n entries
    grad = 2.0 * seg[::-1]
    return grad


def _project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x ≥ 0,  Σ x = target}.
    Implementation of Duchi–Shalev‑Shwartz–Singer–Chandra (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # If already feasible, return after possible clipping to 1000
    v = np.clip(v, 0.0, 1000.0)
    if abs(v.sum() - target) < 1e-12 and np.all(v >= 0):
        return v

    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - target
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = cssv[-1] / len(v)
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    # final clip to the problem's hard upper bound
    w = np.clip(w, 0.0, 1000.0)
    # re‑normalise in case clipping to 1000 changed the sum
    s = w.sum()
    if s > 0:
        w *= target / s
    else:
        w[:] = target / len(v)
    return w


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the smooth‑max surrogate.
    Returns the best vector seen and its exact max‑convolution value.
    """
    n = start_seq.shape[0]
    target = math.sqrt(2.0 * n)

    a = _project_onto_simplex(start_seq, target)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    step = target / n * 0.5          # heuristic base step size
    beta = 1.0                       # soft‑max sharpness

    best_local = a.copy()
    best_score = _conv_max(a)

    it = 0
    max_iters = 5000

    while time.monotonic() < deadline and it < max_iters:
        # convolution and exact max (for book‑keeping)
        conv = _linear_convolve(a, a)          # length 2·n‑1
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # smooth‑max & gradient
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the feasible set
        a = _project_onto_simplex(a, target)

        # schedules
        if it % 100 == 0 and beta < 400.0:
            beta *= 1.5
        if it % 200 == 0:
            step *= 0.95

        it += 1

    return best_local, best_score


def _anneal_refine(seq: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing mass‑swap refinement.
    The ℓ¹‑norm is preserved, positivity is kept by construction.
    """
    n = seq.shape[0]
    cur = seq.copy()
    cur_score = _conv_max(cur)

    best_seq = cur.copy()
    best_score = cur_score

    # temperature starts as a modest fraction of the current value
    T = max(cur_score * 0.1, 1e-4)
    decay = 0.9995
    max_iter = 2_000_000   # broken by the time limit

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0.0:
            it += 1
            continue

        # transfer a random fraction (1 %–20 %) of the mass at i to j
        delta = cur[i] * rng.uniform(0.01, 0.20)
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # respect the hard upper bound 1000
        if new[j] > 1000.0:
            excess = new[j] - 1000.0
            new[j] = 1000.0
            new[i] += excess
            if new[i] > 1000.0 or new[i] < 0.0:
                it += 1
                continue

        new_score = _conv_max(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        # temperature decay
        T *= decay
        if it % 5000 == 0:
            T = max(T, cur_score * 0.05)

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.
    The routine combines a warm‑start, an extensive parametric sweep,
    several Adam‑based gradient descents and a final annealing phase.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ---------- 0️⃣ Warm‑start from a previous construction ----------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = cand.shape[0]
            target0 = math.sqrt(2.0 * n0)
            cand = _project_onto_simplex(cand, target0)
            sc = _conv_max(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[warm start] n={n0} score={sc:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ---------- 1️⃣ Parametric sweep ----------
    lengths = [2000, 1500, 1200, 1000, 800, 600, 500,
               400, 300, 200, 150, 100]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(n, dtype=np.float64)

        # Power‑law a_i ∝ (i+1)^{-p}
        for p in np.linspace(0.2, 3.0, 12):
            a = (idx + 1) ** (-p)
            a = a / a.sum() * target
            a = _project_onto_simplex(a, target)
            sc = _conv_max(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[param] power‑law n={n} p={p:.2f} score={sc:.6f}")

        # Exponential a_i ∝ exp(-α·i)
        for alpha in np.linspace(0.001, 0.12, 12):
            a = np.exp(-alpha * idx)
            a = a / a.sum() * target
            a = _project_onto_simplex(a, target)
            sc = _conv_max(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[param] exponential n={n} α={alpha:.3f} score={sc:.6f}")

        # Sinusoidal (shifted to stay non‑negative)
        for phase in np.linspace(0, math.pi, 4):
            a = np.sin(math.pi * (idx + 1) / (n + 1) + phase) + 1.0
            a = np.clip(a, 0, None)
            a = a / a.sum() * target
            a = _project_onto_simplex(a, target)
            sc = _conv_max(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[param] sinusoid n={n} phase={phase:.2f} score={sc:.6f}")

        # Gaussian‑like families
        centre = (n - 1) / 2.0
        for sigma_rel in np.linspace(0.05, 0.40, 8):
            sigma = sigma_rel * n
            a = np.exp(-((idx - centre) ** 2) / (2 * sigma ** 2))
            a = a / a.sum() * target
            a = _project_onto_simplex(a, target)
            sc = _conv_max(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[param] gaussian n={n} σ={sigma_rel:.2f} score={sc:.6f}")

    # ---------- 2️⃣ Adam‑based gradient descent ----------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # number of random restarts (more for cheaper lengths)
        if n <= 600:
            restarts = 8
        elif n <= 1200:
            restarts = 4
        else:
            restarts = 2

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # Dirichlet‑like random init
            start = rng.gamma(1.0, size=n)
            start = start / start.sum() * target
            start = _project_onto_simplex(start, target)

            # optionally blend with current global best when dimensions match
            if best_seq is not None and best_seq.shape[0] == n:
                start = (start + best_seq) * 0.5
                start = _project_onto_simplex(start, target)

            seq_opt, sc_opt = _adam_optimize(start, deadline, rng)

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = seq_opt.copy()
                print(f"[adam] n={n} restart={r} score={sc_opt:.6f}")

            if best_score < 1.503:          # target already achieved
                print("[early exit] target reached")
                return [float(x) for x in best_seq.tolist()]

    # ---------- 3️⃣ Final annealing refinement ----------
    if best_seq is not None:
        seq_ref, sc_ref = _anneal_refine(best_seq, deadline, rng)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = seq_ref.copy()
            print(f"[anneal] refined score={sc_ref:.6f}")

    # Fallback – uniform vector if nothing was ever found
    if best_seq is None:
        n = 300
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _conv_max(best_seq)
        print(f"[fallback] n={n} score={best_score:.6f}")

    # Return a plain Python list as required by the outer harness
    return [float(x) for x in best_seq.tolist()]
