"""
Projected Exponential‑Gradient (EG) optimizer with a smooth
maximum approximation.  For a fixed length n we keep the ℓ¹‑norm
‖a‖₁ = √(2 n) (the usual scaling for the bound) and minimise the
largest entry of the auto‑convolution a∗a.  The nondifferentiable
max is replaced by a smooth log‑sum‑exp with a continuation schedule
for the smoothing parameter λ.  In each iteration we
    • compute the full convolution by FFT,
    • form the softmax weights pₖ ∝ exp(λ·(bₖ−max b)),
    • obtain the gradient ∇f(a)=2·a∗p (derived analytically),
    • perform an exponentiated‑gradient update a_i←a_i·exp(−η·∇_i) and
      renormalise onto the ℓ¹‑simplex.
The step‑size η decays slowly; λ is doubled every few hundred
iterations, driving the smooth objective towards the true max.
For several candidate lengths we run this optimizer (with a few
random restarts) inside the allotted time, keep the best sequence,
and finish with a brief simulated‑annealing polish (the previous

# ----------------------------------------------------------------------
# Helper FFT convolution utilities
# ----------------------------------------------------------------------
def _fft_full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y using an FFT of power‑of‑two length."""
    lx, ly = x.shape[0], y.shape[0]
    L = 1 << ((lx + ly - 1).bit_length())   # smallest power of two ≥ lx+ly‑1
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[:lx + ly - 1]


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a (full convolution)."""
    conv = _fft_full_convolve(a, a)
    return float(np.max(conv))


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    return _fft_full_convolve(a, a)


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex (sum = s, non‑negative)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        # safety for pure zeros
        a_candidate = np.maximum(a_candidate, 1e-12)
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        # accept if the true max does not increase (simple greedy descent)
        cand_max = _max_convolution(a_candidate)
        if cand_max <= cur_max:
            a = a_candidate
            cur_max = cand_max
        else:
            # if not accepted, shrink the step size a little
            eta *= 0.7
            # do not replace a (stay where we are)
            # continue with the same iteration counter

        # decay step size gradually
        eta = max(eta * decay, 1e-6)

        # increase λ according to schedule
        if (it + 1) % lam_inc_every == 0:
            lam = min(lam * 1.5, lam_max)

        it += 1

    # final evaluation of the best point found
    best_val = _max_convolution(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (the previous method) – used only for polishing
# ----------------------------------------------------------------------
def _fft_max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Fast max‑convolution used by the SA routine."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    needed = 2 * len(a) - 1
    return float(np.max(conv[:needed]))


def _initialize_sequence(
    n: int,
    seed_seq=None,
    rng: np.random.Generator | None = None,
) -> np.ndarray:
    """Create a non‑negative start with ℓ¹‑norm √(2 n)."""
    target = math.sqrt(2.0 * n)
    if seed_seq is not None:
        seq = np.array(seed_seq, dtype=np.float64)
        if seq.size != n:
            if seq.size > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - seq.size)])
        seq = np.clip(seq, 0.0, 1000.0)
        s = np.sum(seq)
        if s == 0.0:
            seq = np.full(n, target / n)
        else:
            seq = seq * (target / s)
        return seq

    if rng is None:
        rng = np.random.default_rng()
    seq = rng.random(n) + 1e-3
    seq = seq / np.sum(seq) * target
    return seq


def _simulated_annealing(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Fallback SA (identical to the previous implementation)."""
    n = len(a)
    target_sum = np.sum(a)            # already = √(2 n) in our usage
    fft_len = 1 << ((2 * n - 1).bit_length())
    cur_max = _fft_max_convolution(a, fft_len)

    best_seq = a.copy()
    best_val = cur_max

    rng = np.random.default_rng()
    start = time.monotonic()
    total_time = max(1e-12, deadline - start)

    init_temp = max(cur_max * 0.5, 1e-4)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = elapsed / total_time
        T = init_temp * (0.0001 ** frac)

        avg = target_sum / n
        step = avg * (T / init_temp) * 3.0

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = rng.normal(scale=step)
        delta = min(delta, a[i])
        delta = max(delta, -a[j])
        if abs(delta) < 1e-12:
            continue

        a[i] -= delta
        a[j] += delta
        new_max = _fft_max_convolution(a, fft_len)

        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 1e-12:
                prob = math.exp(-(new_max - cur_max) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()
        else:
            a[i] += delta
            a[j] -= delta

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 0,
    budget_s: float = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation value.
    Uses exponential‑gradient descent with a smooth‑max continuation,
    followed by a short simulated‑annealing polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.05)   # tiny safety margin

    # ------------------------------------------------------------------
    # 1️⃣ choose candidate lengths (prioritise longer sequences)
    # ------------------------------------------------------------------
    length_candidates = [400, 600, 800, 1000, 1200, 1500, 1800, 2000]
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        n0 = len(h1)
        if n0 not in length_candidates:
            length_candidates.append(n0)
    length_candidates = sorted(set(length_candidates))

    best_seq = None
    best_score = float("inf")

    remaining_lengths = len(length_candidates)
    while time.monotonic() < deadline and remaining_lengths:
        # allocate a fair slice of the remaining budget to the next length
        now = time.monotonic()
        time_left = deadline - now
        per_len = max(0.3, time_left / remaining_lengths)   # at least a few tenths of a sec
        n = length_candidates[-remaining_lengths]          # iterate in order

        sub_deadline = min(deadline, now + per_len)

        # -------- warm‑start from global seed if possible ----------
        seed_seq = None
        if isinstance(h1, (list, tuple, np.ndarray)) and len(h1) == n:
            seed_seq = h1

        # -------- multiple random restarts within the time slice -------
        local_best_a = None
        local_best_val = float("inf")
        while time.monotonic() < sub_deadline:
            a0 = _initialize_sequence(n, seed_seq=seed_seq, rng=rng)
            # run EG optimiser; give it the remaining slice of the sub‑deadline
            run_deadline = min(sub_deadline, time.monotonic() + 0.8 * (sub_deadline - time.monotonic()))
            a_opt, val_opt = _eg_optimize(a0, run_deadline, max_iters=2000)
            if val_opt < local_best_val:
                local_best_val = val_opt
                local_best_a = a_opt.copy()
            # after the first restart reuse the seed_seq flag only once
            seed_seq = None

        # -------------------------------------------------------------
        if local_best_a is not None:
            # compute exact evaluation (max‑convolution) for reporting
            exact_val = _max_convolution(local_best_a)
            if exact_val < best_score:
                best_score = exact_val
                best_seq = local_best_a.copy()
                print(f"[update] n={n} max_conv={exact_val:.6f}")

        # move to next length
        remaining_lengths -= 1

    # ------------------------------------------------------------------
    # 2️⃣ optional short polishing with simulated annealing
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        # allocate the tiny remainder to polishing
        polish_deadline = deadline
        a_start = best_seq.copy()
        a_polished, val_polished = _simulated_annealing(a_start, polish_deadline)
        if val_polished < best_score:
            best_score = val_polished
            best_seq = a_polished.copy()
            print(f"[polish] improved max_conv={val_polished:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)

    # Clip to the allowed range just in case
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return plain Python list
    return [float(x) for x in best_seq]
```

This implementation replaces the pure simulated‑annealing loop with a deterministic exponential‑gradient descent that directly targets the (smoothed) maximum of the auto‑convolution while honouring the ℓ¹‑norm constraint.  
The continuation on the smoothing parameter λ gradually sharpens the objective, allowing the optimizer to first move in a smooth landscape and then focus on the true ‖f∗f‖_∞.  
A brief simulated‑annealing polish is kept as a safety net to escape any remaining shallow local minima.
