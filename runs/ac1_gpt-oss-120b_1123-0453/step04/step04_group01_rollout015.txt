def simplex_proj(v, s):
    # Project v onto {x: x >=0, sum x = s}. Use algorithm of sorting v.
    u = np.sort(v)[::-1]
    sv = np.cumsum(u) - s
    rho = np.where(u - sv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = sv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v, s):
    if s <= 0:
        return np.zeros_like(v)
    # ensure nonnegative after projection.
    # use sorted v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative, return zeros?
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    best_sc = _fast_score(best)

        # compute gradient
        n = len(best)
        grad = np.zeros_like(best)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # for i from i_start to i_end: grad[i] = 2 * best[k - i]

        grad[idxs] = 2.0 * best[k - idxs]

        # step size schedule:

        step0 = target_sum * 0.1
        step = step0 / np.sqrt(t + 1)

        cand = best - step * grad
        # project onto simplex:
        cand = project_onto_simplex(cand, target_sum)
        # ensure <=1000
        cand = np.minimum(cand, 1000.0)

            if cand_sc < best_sc:

    return best

"""
Step‑function minimiser.

The evaluation score of a non‑negative vector `a` (with L1‑norm
√(2·n)) equals the maximal entry of its auto‑convolution
`b = a * a`.  The goal is therefore to produce a vector whose
autocorrelation is as flat as possible.

This implementation explores a variety of parametric families
(Uniform, exponential, power‑law, Beta‑shapes, random Dirichlet)
as seeds, then refines each seed with a **projected sub‑gradient
descent** that directly reduces the currently maximal convolution
entry.  The descent step size is aggressive at the beginning and
decays over time; after each step the vector is projected onto the
non‑negative simplex of the required L1‑norm.  A few random
re‑starts and a cheap hill‑climb / Gaussian polishing finish the
search.

The routine runs for the allotted time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale `vec` to have ℓ¹‑norm exactly `target_sum` (preserving non‑negativity)."""
    s = vec.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        n = vec.shape[0]
        return np.full(n, target_sum / n, dtype=np.float64)
    return vec * (target_sum / s)


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be zero
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n‑1) using FFT for speed.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of `evaluate_sequence` for a correctly normalised
    sequence: returns the maximal auto‑convolution entry.
    """
    conv = _conv_full(seq)
    max_b = conv.max()
    # when `seq` is ℓ¹‑normalised to √(2·n) the denominator collapses to 1,
    # so the score equals the max convolution value.
    return float(max_b)


def _beta_family(n: int, a: float, b: float, target_sum: float) -> np.ndarray:
    """Beta‑distribution shape a_i ∝ x^{a‑1} (1‑x)^{b‑1},  x∈(0,1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    w = np.maximum(w, 1e-12)
    w = w / w.sum() * target_sum
    return w


def _generate_initials(n: int, target_sum: float, rng: np.random.Generator):
    """Return a modest collection of plausible starting vectors for length n."""
    cand = []

    # uniform
    cand.append(np.full(n, target_sum / n, dtype=np.float64))

    # exponential families (log‑spaced alphas)
    for alpha in np.logspace(-4, -0.1, 12):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand.append(_normalize(a, target_sum))

    # power‑law families
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.1, 3.0, 12):
        a = idx ** (-p)
        cand.append(_normalize(a, target_sum))

    # random Beta shapes
    for _ in range(8):
        a = _beta_family(n, rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0), target_sum)
        cand.append(a)

    # a few Dirichlet draws
    for _ in range(8):
        a = rng.dirichlet(np.ones(n)) * target_sum
        cand.append(a)

    return cand

def _flatten_max(seq: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 rng: np.random.Generator,
                 max_iters: int = 5000) -> np.ndarray:
    """
    Projected sub‑gradient descent that repeatedly reduces the current
    maximal entry of the auto‑convolution.  The step size scales like
    0.5 / (1+iteration) to be aggressive early and safe later.
    """
    a = _normalize(seq, target_sum)
    best = a.copy()
    best_sc = _fast_score(best)
    n = a.shape[0]

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        conv = _conv_full(a)
        k = int(conv.argmax())
        M = conv[k]                      # current maximal convolution value

        # gradient of the active entry  (∂b_k/∂a_i = 2·a_{k-i} when defined)
        idx = np.arange(n)
        mask = (k - idx >= 0) & (k - idx < n)
        g = np.zeros_like(a)
        g[mask] = a[k - idx[mask]]         # 1·a_{k-i}, the factor 2 will be absorbed in step size

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            # dead‑end, restart from random Dirichlet
            a = rng.dirichlet(np.ones(n)) * target_sum
            continue

        # step‑size schedule – aggressive early, then decays
        lam = max(0.02, 0.5 / (1.0 + it))
        alpha = lam * M / (norm2 + 1e-12)

        cand = a - alpha * g
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc

        # always continue the descent (even if not improved)
        a = cand

        # occasional random shake to escape plateaus
        if rng.random() < 0.003:
            a = rng.dirichlet(np.ones(n)) * target_sum

    return best


def _swap_hill_climb(seq: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator) -> np.ndarray:
    """Very cheap hill‑climb that swaps a tiny amount of mass between two indices."""
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target_sum * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
    return best


def _gaussian_local_search(seq: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           rng: np.random.Generator) -> np.ndarray:
    """Short annealing‑style Gaussian perturbation loop."""
    best = seq.copy()
    best_sc = _fast_score(best)
    step = target_sum * 0.02

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape[0])
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
            step *= 0.9
        else:
            step *= 0.99
            if step < 1e-7:
                step = target_sum * 0.02
    return best

        # ----------------------------------------------------------
        #   generate a modest fleet of seed vectors
        # ----------------------------------------------------------
        seeds = _generate_initials(n, target, rng)

        # quick evaluation of the fleet, keep the best few
        scored = [(s, _fast_score(s)) for s in seeds]
        scored.sort(key=lambda x: x[1])
        top_seeds = [scored[i][0] for i in range(min(4, len(scored)))]

        for s in top_seeds:
            sc = _fast_score(s)
            if sc < best_score:
                best_seq, best_score = s.copy(), sc
                print(f"[seed] n={n} new best {sc:.6f}")

        # ----------------------------------------------------------
        #   Focused refinement with projected sub‑gradient descent
        # ----------------------------------------------------------
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break
        # allocate at most 25 s per length (or whatever is left)
        sub_deadline = time.monotonic() + min(25.0, remaining * 0.5)

        # start from the current global best of this length if we have it
        if best_seq is not None and best_seq.shape[0] == n:
            start = best_seq.copy()
        else:
            start = rng.dirichlet(np.ones(n)) * target

        refined = _flatten_max(start, target, sub_deadline, rng)
        refined_sc = _fast_score(refined)
        if refined_sc < best_score:
            best_seq, best_score = refined.copy(), refined_sc
            print(f"[refine] n={n} improved to {refined_sc:.6f}")

        # a couple of cheap random restarts
        for r in range(2):
            if time.monotonic() >= deadline:
                break
            rs = rng.dirichlet(np.ones(n)) * target
            sub_deadline = time.monotonic() + min(12.0, deadline - time.monotonic())
            refined = _flatten_max(rs, target, sub_deadline, rng)
            refined_sc = _fast_score(refined)
            if refined_sc < best_score:
                best_seq, best_score = refined.copy(), refined_sc
                print(f"[restart] n={n} run {r} new best {refined_sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Final polishing on the globally best vector.
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: a modest uniform vector (n=300)
        n = 300
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    # ensure exact ℓ¹ normalisation
    n = best_seq.shape[0]
    target = math.sqrt(2.0 * n)
    best_seq = _project_onto_simplex(best_seq, target)

    # cheap swap‑based hill climb
    best_seq = _swap_hill_climb(best_seq, target, deadline, rng)

    # short Gaussian hill climb if time permits
    if time.monotonic() < deadline:
        best_seq = _gaussian_local_search(best_seq, target, deadline, rng)

    final_score = _fast_score(best_seq)
    print(f"[final] best score {final_score:.6f}   length {best_seq.shape[0]}")
    return [float(x) for x in best_seq]
