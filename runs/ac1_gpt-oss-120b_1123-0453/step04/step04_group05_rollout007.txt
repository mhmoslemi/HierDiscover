"""
Propose a low‑value step sequence.

Algorithm
---------
* Work on the ℓ¹‑simplex (sum of heights = 1) – the evaluation is scale‑free.
* For each length n we try a diverse collection of initial points
      – uniform, several deterministic “U‑shaped’’ power‑law patterns,
        Dirichlet draws with different concentration parameters.
* Every initial point is refined by a continuation of a smooth‑max
  surrogate:
        fβ(a) = (1/β)·log Σ_k exp(β·(a*a)_k)          (β ↑)
  The surrogate is differentiable; we minimise it with L‑BFGS‑B
  (analytic gradient, FFT‑based convolutions).  The β‑schedule
  [5, 15, 40, 80, 200, 400, 800] drives the optimiser close to the
  true max‑convolution.
* After each β‑stage the exact `evaluate_sequence` value is computed.
  The best sequence seen so far (over all lengths, restarts and
  initialisations) is kept.
* When the global time budget allows a short stochastic polishing
  (Gaussian perturbations with projection onto the simplex) is applied.
* The whole search respects the 1000 s budget; at any moment the best
  feasible sequence found is returned.

The method differs from the reference code by:
    – an enriched pool of initial shapes (U‑shapes, Dirichlet samples);
    – a deeper β‑continuation (up to 800);
    – more L‑BFGS iterations per stage;
    – a fast FFT‑based smooth‑max gradient.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Simple helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """Convert unrestricted vector `z` to a simplex point of sum `target_sum`."""
    # this is numerically stable soft‑max
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z) * target_sum
    return a


def _obj_and_grad(z: np.ndarray, n: int, S: float, beta: float):
    """
    Smooth surrogate of max‑convolution.
    Returns (smooth_value, grad_z) where
        smooth = (1/β)·log Σ_k exp(β·conv_k)
    and grad_z is the gradient w.r.t. the soft‑max parameters `z`.
    """
    a = _softmax_to_a(z, S)                     # shape (n,)

    # ----- auto‑convolution via FFT ------------------------------------
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]   # length 2n‑1, real

    # ----- smooth max (log‑sum‑exp) ------------------------------------
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))          # stable
    sum_exps = np.sum(exps)
    smooth = max_c + np.log(sum_exps) / beta

    w = exps / sum_exps                           # soft‑max over the k‑indices

    # ----- gradient w.r.t. a (see derivation in the docstring) -------
    a_rev = a[::-1]
    L2 = _next_pow2(3 * n - 2)
    fr = np.fft.rfft(a_rev, L2)
    fw = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(fr * fw, L2)            # length ≥ 3n‑2
    start = n - 1
    g_raw = conv2[start:start + n]               # length n
    g_a = 2.0 * g_raw

    # ----- chain rule for the soft‑max parameters ---------------------
    total = np.dot(g_a, a)
    grad_z = a * g_a - a * (total / S)            # ∂Φ/∂z_i
    return smooth, grad_z.astype(np.float64)


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = target_sum}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(seq: np.ndarray) -> float:
    """Fast computation of max_{k} (seq * seq)_k using FFT."""
    n = len(seq)
    L = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, L)
    conv = np.fft.irfft(f * f, L)[: 2 * n - 1]
    return float(np.max(conv))


def _exact_score(seq: np.ndarray) -> float:
    """Exact `evaluate_sequence` value (fast local copy)."""
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    max_b = _max_convolution(arr)
    return float(2 * arr.size * max_b / (s * s))

# Local optimisation for a fixed length
# ----------------------------------------------------------------------


def _optimise_one_restart(z0: np.ndarray,
                          n: int,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Run continuation on β using L‑BFGS‑B, starting from `z0`.
    Returns the best concrete sequence (as a numpy array) and its exact score.
    """
    betas = [5.0, 15.0, 40.0, 80.0, 200.0, 400.0, 800.0]
    best_seq = None
    best_val = float("inf")
    zz = z0.copy()

    for beta in betas:
        if time.monotonic() >= deadline:
            break

        res = minimize(fun=lambda zz_: _obj_and_grad(zz_, n, target_sum, beta),
                       x0=zz,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 400,
                                'gtol': 1e-12,
                                'disp': False})
        zz = res.x
        a = _softmax_to_a(zz, target_sum)
        a = np.clip(a, 0.0, 1000.0)
        val = _exact_score(a)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[β={beta:4.0f}] n={n}  val={val:.6f}")

        # early exit if we are already below the global target
        if best_val < 1.40:                 # generous safety margin
            break

    return best_seq, best_val


def _optimise_length(n: int,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length `n` using several restarts.
    Returns the best sequence and its exact evaluate_sequence score.
    """
    max_restarts = 5
    best_seq = None
    best_val = float("inf")

    # ── list of deterministic warm‑starts (U‑shapes, uniform)  ────────
    deterministic = []
    # uniform
    deterministic.append(np.full(n, target_sum / n, dtype=np.float64))

    # power‑law “U‑shapes’’ (symmetric, high at ends, low in the middle)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        idx = np.arange(n, dtype=np.float64)
        w = (idx + 1.0) ** (-p) + (n - idx) ** (-p)
        deterministic.append(w)

    # ── convert them to soft‑max parameters (z) ────────────────────────
    initial_z_list = []

    for seq in deterministic:
        seq = np.maximum(seq, 0.0)
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq = seq * (target_sum / seq_sum)
        p = np.clip(seq / target_sum, 1e-12, None)
        z = np.log(p)                      # any shift works
        initial_z_list.append(z)

    # include one random Dirichlet start per restart (handled later)

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # ----- pick initial point ----------------------------------------
        if warm_seq is not None and warm_seq.shape[0] == n and restart == 0:
            # use supplied warm sequence for the very first restart
            p = np.clip(warm_seq / target_sum, 1e-12, None)
            z0 = np.log(p)
        elif restart < len(initial_z_list):
            z0 = initial_z_list[restart]
        else:
            # random Dirichlet draw – concentration varies with restart
            alpha = max(0.2, 5.0 - restart)   # 5,4,3,… make it sparser later
            b = rng.dirichlet(np.full(n, alpha))
            b = b * target_sum
            p = np.clip(b / target_sum, 1e-12, None)
            z0 = np.log(p)

        seq, val = _optimise_one_restart(z0, n, target_sum, deadline)

        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[global] n={n}  restart={restart}  best={val:.6f}")

        # stop early if already very good
        if best_val < 1.40:
            break

    # fallback – if nothing succeeded, return uniform
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = _exact_score(best_seq)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic polishing (Gaussian hill‑climbing)
# ----------------------------------------------------------------------


def _stochastic_refinement(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that respects the simplex.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _exact_score(best)
    step = step_initial
    target_sum = best.sum()

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        # project back onto the simplex
        cand = _project_simplex(cand, target_sum)
        cand_score = _exact_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-7:
            step = step_initial

    return best, best_score

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low `evaluate_sequence`
    value.  The routine respects the supplied time budget (≈ budget_s s)
    and returns a plain Python list ready for the evaluator.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        if seq.size > 0:
            seq = np.clip(seq, 0.0, 1000.0)
            seq_sum = seq.sum()
            if seq_sum > 0:
                seq = seq * (1.0 / seq_sum)          # normalise to sum=1
            else:
                seq = np.full_like(seq, 1.0 / seq.size)
            val = _exact_score(seq)
            best_seq = seq.copy()
            best_score = val
            print(f"[init] warm‑start length={seq.size}  val={val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Deterministic optimisation on several lengths
    # ------------------------------------------------------------------
    # A spread of lengths – longer sequences give a larger search space.
    lengths = [400, 600, 800, 1000, 1300, 1600, 2000, 2500, 3000, 3500]
    target_sum = 1.0                     # we keep the ℓ¹‑norm = 1 (scale‑free)

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # use the current best sequence as warm‑start only when sizes match
        warm = best_seq if (best_seq is not None and best_seq.size == n) else None

        seq_n, val_n = _optimise_length(n, target_sum, deadline, rng, warm)

        if val_n < best_score:
            best_score = val_n
            best_seq = seq_n.copy()
            print(f"[global update] n={n}  val={val_n:.6f}")

        # early exit if the target is already reached
        if best_score < 1.503:
            print("[early stop] target reached")
            break

    # ------------------------------------------------------------------
    # 2️⃣ Final stochastic polishing (time permitting)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_val = _stochastic_refinement(best_seq, deadline,
                                                      step_initial=0.2)
        if refined_val < best_score:
            best_seq = refined
            best_score = refined_val
            print(f"[final polish] val={refined_val:.6f}")

    # final clipping for safety (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score = {best_score:.6f}  (length={best_seq.size})")
    return [float(x) for x in best_seq]
