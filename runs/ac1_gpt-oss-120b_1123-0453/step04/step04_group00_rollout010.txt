"""
A fast multi‑stage search for a non‑negative step sequence that minimises

    2 n·max(a∗a) / (∑a)²

(the value returned by ``evaluate_sequence``).  The main ideas are

* **Good analytic seeds** – an “arcsine” shape is known to have an almost
  flat self‑convolution (continuous arcsine density → uniform convolution).
  We generate a family of such seeds (varying the singularity parameter
  and a power‑law exponent) together with simple power‑law / exponential
  families.

* **Smooth‑max gradient descent** – the non‑differentiable max is replaced
  by a soft‑max (log‑sum‑exp) with an annealed temperature β.  The gradient
  of the surrogate is obtained by a single FFT‑based convolution, and each
  iterate is projected onto the feasible set (non‑negative, ℓ¹‑norm
  √(2 n)).  A simple momentum term and an AdaGrad‑style step‑size
  adaptation improve robustness.

* **Random‑restart hill‑climbing** – after the gradient phase we perform a
  cheap “mass‑swap’’ hill‑climb that moves a small fraction of the mass
  between random positions; the best improvement is kept.

The routine works for any budget (seconds) and automatically cuts
off when the deadline is reached.  It returns a plain ``list[float]``,
ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return a good FFT length (power‑of‑2 fallback if unavailable)."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for many calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fast_score(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a correctly normalised seq."""
    # Full linear convolution – cheap for the moderate sizes we use.
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and enforce ℓ¹‑norm = √(2·n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / total)
    return seq


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) weights over convolution entries."""
    cmax = conv.max()
    # Stabilise the exponentials.
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    For a_i the derivative is 2·∑_j w_{i+j}·a_j.
    Implemented as a cross‑correlation via a single FFT convolution.
    """
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)          # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]          # middle n entries
    grad = 2.0 * segment[::-1]                 # flip back
    return grad


def _optimize_sequence(init_seq: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       max_iters: int = 2500) -> tuple[np.ndarray, float]:
    """
    Smooth‑max gradient descent with simple momentum and an annealing β.
    Returns the best vector seen and its exact max‑convolution value.
    """
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    # Momentum / AdaGrad style variables
    velocity = np.zeros_like(a)
    eps = 1e-12

    best_local = a.copy()
    best_score = _fast_score(a)

    beta = 1.0                # start with a very smooth surrogate
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        # ----- evaluate current point -----
        conv = _linear_convolve(a, a)          # length 2·n‑1
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # ----- smooth‑max surrogate -----
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # ----- simple Adam‑like update -----
        step_size = 0.5 / (1.0 + 0.001 * it)   # decreasing step magnitude
        velocity = 0.9 * velocity + 0.1 * grad
        a = a - step_size * velocity

        # ----- projection onto feasible set -----
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # ----- occasional jitter to escape plateaus -----
        if it % 250 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.04
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # ----- anneal β (sharpen the surrogate) -----
        if it % 100 == 0 and beta < 500.0:
            beta *= 1.6

        it += 1

    return best_local, best_score


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------
def _arcsine_seed(n: int, power: float = 0.5, shift: float = 0.5) -> np.ndarray:
    """
    Discrete approximation of the arcsine density on [0, n‑1].
    a_i ∝ ((i+shift)*(n‑1‑i+shift))**(-power).
    For power≈0.5 the continuous convolution is uniform.
    """
    idx = np.arange(n, dtype=np.float64)
    denom = (idx + shift) * (n - 1 - idx + shift)
    # avoid division by zero – shift guarantees positivity
    w = np.power(denom, -power)
    return _normalize(w)


def _gaussian_seed(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian (truncated) seed."""
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    return _normalize(w)


def _powerlaw_seed(n: int, p: float) -> np.ndarray:
    """Simple power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = np.power(idx, -p)
    return _normalize(w)


def _exponential_seed(n: int, alpha: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-alpha·i)."""
    idx = np.arange(n, dtype=np.float64)
    w = np.exp(-alpha * idx)
    return _normalize(w)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.

    The routine consists of
        0️⃣ warm‑start from a previously found global best (if any),
        1️⃣ a broad parametric sweep (arcsine, Gaussian, power‑law, exponential),
        2️⃣ several random restarts of a smooth‑max gradient descent,
        3️⃣ a final cheap hill‑climb that swaps small mass fractions.
    The best sequence seen before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)  # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the global best
    # ------------------------------------------------------------------
    def _update_global(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = _fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global best] n={len(seq)} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a known construction (if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            cand = _normalize(cand)
            _update_global(cand)
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Parametric sweep – a rich family of analytic seeds
    # ------------------------------------------------------------------
    lengths = [2000, 1500, 1200, 1000, 800, 600, 500, 400, 300, 200, 150, 100]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ---- arcsine‑type seeds (vary power and shift) ----
        for power in np.linspace(0.35, 0.75, 8):
            for shift in np.linspace(0.1, 2.0, 6):
                a = _arcsine_seed(n, power=power, shift=shift)
                _update_global(a)

        # ---- Gaussian seeds (vary σ) ----
        for sigma in np.linspace(0.08, 0.30, 10):
            a = _gaussian_seed(n, sigma=sigma)
            _update_global(a)

        # ---- power‑law seeds ----
        for p in np.linspace(0.2, 3.0, 12):
            a = _powerlaw_seed(n, p=p)
            _update_global(a)

        # ---- exponential seeds ----
        for alpha in np.linspace(0.001, 0.18, 12):
            a = _exponential_seed(n, alpha=alpha)
            _update_global(a)

    # ------------------------------------------------------------------
    # 2️⃣ Random restarts + smooth‑max gradient descent
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # More restarts for smaller problems (they are cheap)
        restarts = 12 if n <= 500 else 5

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # Dirichlet‑like random start (Gamma(1,1) = Exp(1))
            start = rng.gamma(1.0, size=n)
            start = start / start.sum() * target

            # Blend with the current best of the same length (if any)
            if best_seq is not None and best_seq.shape[0] == n:
                start = (start + best_seq) * 0.5
                start = _normalize(start)

            seq_opt, sc_opt = _optimize_sequence(start, deadline, rng)
            _update_global(seq_opt)

            # Early exit if we have already beaten the target
            if best_score < 1.503:
                print("[early exit] target reached")
                return [float(x) for x in best_seq.tolist()]

    # ------------------------------------------------------------------
    # 3️⃣ Final cheap refinement – random mass‑swap hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None:
        while time.monotonic() < deadline:
            n = best_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            # Move a small percentage of mass from i to j
            delta = best_seq[i] * 0.025
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _normalize(cand)
            _update_global(cand)

            # Occasionally perform a three‑point “rotation’’ of mass
            if rng.random() < 0.02:
                i, j, k = rng.choice(n, size=3, replace=False)
                eps = best_seq[i] * 0.015
                cand = best_seq.copy()
                cand[i] -= eps
                cand[j] += eps * 0.6
                cand[k] += eps * 0.4
                cand = _normalize(cand)
                _update_global(cand)
    else:
        # Fallback – a simple uniform vector (never used in practice)
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # Return a plain Python list as required by the outer harness
    return [float(x) for x in best_seq.tolist()]
