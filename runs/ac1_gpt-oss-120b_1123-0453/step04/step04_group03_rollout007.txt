def lp_refine(f, target_sum, max_iters, deadline):
    best_f = f.copy()
    best_score = compute_score(best_f)
    for it in range(max_iters):
        if time.monotonic() >= deadline: break
        n = len(f)
        # compute M
        conv = np.convolve(f, f)
        M = np.max(conv)
        # Build conv matrix
        A = build_convolution_matrix(f) # shape (2n-1,n)
        # Solve LP: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        # Use linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fallback: break
            break
        g = res.x
        sum_g = np.sum(g)
        if sum_g <= 0: break
        # Rescale to target_sum
        g = g * (target_sum / sum_g)
        # Blend
        t = 0.2  # may adjust adaptively: if M decreased significantly, increase t? but okay.
        f_new = (1 - t) * f + t * g
        # Ensure non-negativity, clip
        f_new = np.clip(f_new, 0.0, 1000.0)
        # Might want to reproject onto simplex to enforce sum
        # but we already scaled sum so sum = target_sum after blending? Actually sum(f_new) = (1-t)*target_sum + t*target_sum = target_sum.
        # So sum maintained.
        # Compute score
        cur_score = compute_score(f_new)
        if cur_score < best_score:
            best_f = f_new.copy()
            best_score = cur_score
            f = f_new
            # optionally reduce t if we see fluctuations?
        else:
            # if no improvement, maybe break
            # Could also try decreasing t and continue
            # but simple: break if no improvement for certain iterations
            break
    return best_f, best_score

"""
Improved search for a low‑value non‑negative step sequence.

Algorithm overview
-------------------
1️⃣  **Simplex‑initialisation** – for a set of candidate lengths we draw a few
    random points on the ℓ¹‑simplex (∑aᵢ = 1).  Lengths up to a few thousand are
    tried; the best intermediate result is kept.

2️⃣  **Projected Adam descent with annealed smoothing** – we minimise a smooth
    surrogate of the maximum auto‑convolution:
        F(a) = log‑sum‑exp( conv(a,a) / β ).
    The smoothing factor β is gradually reduced, which drives the iterates
    towards a true minimiser of max(conv).

3️⃣  **Linear‑programming refinement** – for a current vector `f` we solve
    a linear program
        maximise   ∑ gᵢ
        subject to  (f * g)ₖ ≤ M   for all convolution indices k,
                    gᵢ ≥ 0,
    where `M = max(conv(f,f))`.  The optimal `g` has the largest possible ℓ¹‑norm
    under the convolution bound.  After normalising `g` to the same ℓ¹‑norm as `f`
    we blend:  f ← (1‑t)·f + t·g   (t≈0.2).  Repeating the LP step shrinks the
    peak of the auto‑convolution.

4️⃣  **Stochastic polishing** – a short hill‑climbing phase with Gaussian
    perturbations and occasional pairwise swaps finishes the search.

All steps respect the global time budget (default 1000 s).  The routine returns
the best sequence found as a plain ``list[float]``; the external
``evaluate_sequence`` routine will apply the required clipping.
"""

import time
import math

import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz
from scipy.sparse import csr_matrix

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact value of the evaluation function (no clipping performed)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Numerically stable soft‑max over the convolution entries."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / beta
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    return w


def _weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σₖ wₖ·convₖ(a,a) w.r.t. ``a``.
    ``weights`` must have length 2·len(a)‑1 and sum to 1.
    """
    n = len(a)
    grad = np.zeros_like(a)
    for k, wk in enumerate(weights):
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += wk * 2.0 * a[j]
        # diagonal entries counted twice; correct once
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= wk * a[mask]
    return grad


def _adam_descent(a0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_iters: int = 15000,
                  patience: int = 300) -> tuple[np.ndarray, float]:
    """
    Projected Adam descent on a smoothed surrogate of ``max(conv)``.
    Returns the best vector seen and its exact evaluation score.
    """
    a = a0.copy()
    n = len(a)

    # Adam hyper‑parameters
    lr = 0.07
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    no_improve = 0

    # decreasing schedule for the smoothing factor
    beta_factor = 0.10

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        max_conv = np.max(conv)

        # anneal smoothing: start fairly smooth and shrink slowly
        beta = max_conv * beta_factor
        beta_factor = max(0.001, beta_factor * 0.9995)   # geometric decay

        w = _softmax_weights(conv, beta)
        grad = _weighted_gradient(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cur_score = _compute_score_from_vector(a)
        cand_score = _compute_score_from_vector(a_candidate)

        # accept improvement
        if cand_score < cur_score:
            a = a_candidate

        # keep the globally best vector
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            print(f"[Adam] n={n} it={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2·n‑1) × n convolution matrix C such that C·g = f * g.
    Returned as a CSR sparse matrix.
    """
    n = len(f)
    # First column: f followed by n‑1 zeros
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # First row: f₀ then zeros
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    dense = toeplitz(col, row)          # shape (2n‑1, n)
    return csr_matrix(dense)


def _lp_refine(f: np.ndarray,
               target_sum: float,
               max_inner_iters: int,
               deadline: float,
               blend_factor: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Alternating linear‑programming refinement.
    At each inner iteration we solve
        maximise   ∑ gᵢ
        subject to (f * g)ₖ ≤ M   ∀k,
                    gᵢ ≥ 0,
    where M = max(f*f).  The optimal g is renormalised to the same ℓ¹‑norm as f
    and then blended with f.
    Returns the best vector encountered and its exact score.
    """
    best_f = f.copy()
    best_score = _compute_score_from_vector(best_f)

    for it in range(1, max_inner_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv_ff = np.convolve(f, f)
        M = np.max(conv_ff)

        # Build the convolution matrix for the current f
        A = _build_conv_matrix(f)                     # (2n‑1) × n, sparse
        b = np.full(A.shape[0], M + 1e-12)            # small epsilon for feasibility

        # Linear programme: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(len(f), dtype=float)
        bounds = [(0.0, None) for _ in range(len(f))]

        # Solve with HiGHS (fast interior‑point method for sparse LP)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # If the LP fails, abort the refinement for this length.
            print(f"[LP] iteration {it} failed (status: {res.message})")
            break

        g = res.x
        sum_g = np.sum(g)
        if sum_g <= 0:
            break

        # Renormalise g so that it has the same ℓ¹‑norm as f
        g = g * (target_sum / sum_g)

        # Blend a little towards g
        f_new = (1.0 - blend_factor) * f + blend_factor * g
        f_new = np.clip(f_new, 0.0, 1000.0)          # keep in admissible range

        score_new = _compute_score_from_vector(f_new)
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new                                   # continue from the improved point
            # gradually tighten the blend as we progress
            blend_factor = max(0.05, blend_factor * 0.97)
            print(f"[LP] n={len(f)} it={it} score={score_new:.6f}")
        else:
            # No improvement – we stop early.
            break

    return best_f, best_score


def _stochastic_refinement(start: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.25,
                           swap_prob: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: Gaussian perturbations + occasional swaps,
    always projecting back onto the simplex of the original ℓ¹‑norm.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, len(best), size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = best + rng.normal(scale=step, size=best.shape)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = min(step_initial, step * 1.02)
        else:
            step *= 0.995
        if step < 1e-8:
            step = step_initial

    return best, best_score


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence`` under a fixed time budget.
    The function returns a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.15)   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction (if any).
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        if np.sum(seq) == 0:
            seq = np.full_like(seq, 1.0 / len(seq))
        else:
            seq = seq / np.sum(seq)          # normalize to ℓ¹‑norm = 1
        score = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[warm] loaded height_sequence_1, n={len(seq)} score={score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Main exploration – several lengths, random restarts, Adam + LP.
    # ------------------------------------------------------------------
    # Lengths increasing geometrically; larger n give a better theoretical bound.
    lengths = [500, 800, 1200, 1600, 2000]          # can be adjusted on the fly
    target_sum = 1.0                                 # we keep the ℓ¹‑norm = 1

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # Determine how many restarts we can afford for this length.
        max_restarts = 6 if n <= 1200 else 4
        for rs in range(max_restarts):
            if time.monotonic() >= deadline:
                break

            # Initialise a candidate for this length
            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
            else:
                a0 = rng.random(n) + 1e-6          # avoid exact zeros
                a0 = _project_onto_simplex(a0, target_sum)

            # Allocate a sub‑deadline (roughly 12 s per run; adjust if needed)
            sub_deadline = min(deadline, time.monotonic() + 12.0)

            # ---- Adam phase (smooth surrogate) ----
            a_adam, score_adam = _adam_descent(a0, target_sum,
                                               sub_deadline,
                                               max_iters=12000,
                                               patience=250)

            # ---- Linear‑programming refinement ----
            # We give the LP phase a dedicated slice of the remaining budget.
            lp_deadline = min(deadline, time.monotonic() + 8.0)
            a_lp, score_lp = _lp_refine(a_adam, target_sum,
                                        max_inner_iters=12,
                                        deadline=lp_deadline,
                                        blend_factor=0.20)

            # Update global best if improvement is observed
            if score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"[update] n={n} restart={rs} score={score_lp:.6f}")

            # Early exit when we have crossed a comfortable margin
            if best_score < 1.45:
                print("[early] satisfactory bound reached")
                break
        if best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Final stochastic polishing (if time permits).
    # ------------------------------------------------------------------
    if best_seq is None:
        # completely fallback – a modest uniform vector of length 800
        n = 800
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    if time.monotonic() < deadline:
        polish_deadline = deadline
        refined, refined_score = _stochastic_refinement(best_seq,
                                                        polish_deadline,
                                                        step_initial=0.25)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # Return a plain Python list – evaluate_sequence will be called later.
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
