"""
Optimisation of a non‑negative step function for the ``evaluate_sequence`` challenge.

The objective (after normalising the ℓ¹‑norm to √(2·n)) equals the
maximum entry of the auto‑convolution.  This implementation

* sweeps several lengths (up to 1500) and generates a rich pool of
  parametric seed vectors (uniform, triangular, power‑law, exponential,
  Beta‑shapes, quadratic, Gaussian‑bell, random Dirichlet).
* keeps the few best seeds for each length and refines them with an
  aggressive projected‑gradient / mass‑transfer routine that uses a
  back‑tracking line‑search – this yields larger, monotone improvement
  steps compared with the previous fixed‑step scheme.
* occasionally applies smoothing, random perturbations and short random
  restarts to escape shallow plateaus.
* finishes with cheap hill‑climbing swaps and a brief Gaussian‑annealing
  polish.

All vectors are kept non‑negative, clipped to 1000 and projected onto the
simplex of the required ℓ¹‑norm after each modification.
The routine respects the 1000 s time budget, returning the best sequence
found before the deadline.
"""

import time
import math

# ----------------------------------------------------------------------
# Simple utility helpers
# ----------------------------------------------------------------------


def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``vec`` to have exact ℓ¹‑norm ``target_sum`` (preserves non‑negativity)."""
    s = vec.sum()
    if s <= 0.0:
        # fall back to uniform distribution
        n = vec.shape[0]
        return np.full(n, target_sum / n, dtype=np.float64)
    return vec * (target_sum / s)


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` without the safety wrapper.
    For sequences normalised to sum = √(2·n) this equals the maximal entry
    of the auto‑convolution.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)

# Families of candidate vectors

def _beta_family(n: int, a: float, b: float, target_sum: float) -> np.ndarray:
    """Beta distribution shape a_i ∝ x^{a‑1}(1‑x)^{b‑1},  x∈(0,1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    w = np.maximum(w, 1e-12)
    w = w / w.sum() * target_sum
    return w


def _quadratic_family(n: int, target_sum: float) -> np.ndarray:
    """Concave quadratic shape a_i ∝ (i+1)*(n‑i)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) * (n - i)
    return _normalize(w, target_sum)


def _gaussian_bell(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Discrete Gaussian bell centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w = np.maximum(w, 1e-12)
    return _normalize(w, target_sum)


def _generate_candidates(
    n: int, target_sum: float, rng: np.random.Generator
) -> list[np.ndarray]:
    """Create a diverse set of plausible starting vectors for length ``n``."""
    cand = []

    # 1. Uniform
    cand.append(np.full(n, target_sum / n, dtype=np.float64))

    # 2. Simple triangular (peak centre)
    x = np.arange(n, dtype=np.float64)
    tri = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
    tri[tri < 0] = 0.0
    cand.append(_normalize(tri, target_sum))

    # 3. Quadratic (parabolic) shape
    cand.append(_quadratic_family(n, target_sum))

    # 4. Power‑law a_i ∝ i^{-p}
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.1, 4.0, 15):
        a = idx ** (-p)
        cand.append(_normalize(a, target_sum))

    # 5. Exponential a_i ∝ e^{-α·i}
    for alpha in np.linspace(0.001, 0.30, 12):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand.append(_normalize(a, target_sum))

    # 6. Beta family (including arcsine and a wide range of (a,b))
    for _ in range(10):
        a_param = rng.uniform(0.05, 5.0)
        b_param = rng.uniform(0.05, 5.0)
        cand.append(_beta_family(n, a_param, b_param, target_sum))

    # 7. Arcsine (Beta(½,½))
    cand.append(_beta_family(n, 0.5, 0.5, target_sum))

    # 8. Sine‑powered shapes  a_i ∝ sin^p(π·i/(n+1))
    for p in np.linspace(0.5, 4.0, 12):
        a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1))) ** p
        cand.append(_normalize(a, target_sum))

    # 9. Gaussian bells with varying sigma
    for sigma in np.linspace(0.1 * n, 0.4 * n, 5):
        cand.append(_gaussian_bell(n, sigma, target_sum))

    # 10. A few random Dirichlet draws (already ℓ¹‑normalised)
    for _ in range(8):
        a = rng.dirichlet(np.ones(n)) * target_sum
        cand.append(a)

    return cand

def _swap_hill_climb(seq: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """Cheap hill‑climb that swaps a tiny mass between two indices."""
    a = seq.copy()
    best_sc = _fast_score(a)
    n = a.shape[0]

    while time.monotonic() < deadline:
        donors = np.where(a > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(a[donor] * 0.05, target_sum * 0.0009)
        cand = a.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            a = cand
            best_sc = cand_sc
    return a


def _gaussian_local_search(seq: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """Short annealing‑style Gaussian perturbation loop."""
    a = seq.copy()
    best_sc = _fast_score(a)
    step = target_sum * 0.02

    while time.monotonic() < deadline:
        cand = a + rng.normal(scale=step, size=a.shape[0])
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            a = cand
            best_sc = cand_sc

        else:

            if step < 1e-6:

    return a

    final_score = _fast_score(best_seq)
    print(f"[final] best score {final_score:.6f}   length {best_seq.shape[0]}")

    # return as plain Python list
    return [float(x) for x in best_seq]
