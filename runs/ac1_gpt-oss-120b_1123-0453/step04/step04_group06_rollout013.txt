def _adam_optimize(f0, target_sum, deadline, max_iters=3000, beta_start=1.0, beta_factor=1.2, beta_max=256.0, lr_initial=0.5):
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta = beta_start

    for it in range(1, max_iters+1):
        if time.monotonic() >= deadline:
            break
        # compute conv and weights
        conv = np.convolve(f, f)
        max_val = conv.max()
        # compute smoothmax weights
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        grad = _weighted_gradient(f, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr_initial * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = _project_onto_simplex(f - step, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cand_score = _compute_score(f_candidate)
        if cand_score < best_score:
            best_f, best_score = f_candidate.copy(), cand_score
        # Accept or reject (maybe always accept)
        # Use simple acceptance: always move to candidate (like gradient descent)
        f = f_candidate

        # adjust beta
        beta = min(beta_max, beta * beta_factor)
        # optional: adjust lr_initial if no improvement
        # We'll keep constant

    return best_f, best_score

def _subgradient_optimize(f0, target_sum, deadline, max_iters=2000, step_initial=0.5):
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)
    step = step_initial
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        conv = np.convolve(f, f)
        max_val = conv.max()
        # active indices within tolerance
        tol = max_val * 1e-12 + 1e-12
        active = np.where(conv >= max_val - tol)[0]
        # uniform weights over active set
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        grad = _weighted_gradient(f, w)
        cand = _project_onto_simplex(f - step * grad, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best_f, best_score = cand.copy(), cand_score
            f = cand
            step *= 1.02
            no_improve = 0
        else:
            step *= 0.7
            no_improve += 1

        if no_improve >= 200:
            # random restart maybe
            step = step_initial
            no_improve = 0

        if step < 1e-12:
            break

    return best_f, best_score
```

"""
Optimise the step‑function bound within the given time budget.

The search consists of three phases:

1️⃣   **Initialisation** – several deterministic or random constructions
     (uniform, V‑shape, edge‑heavy) are generated for a variety of lengths.
     All vectors are scaled to the ℓ¹‑simplex Σ a_i = √(2 n) and clipped
     to the interval required by ``evaluate_sequence``.

2️⃣   **Projected optimisation**
     *   *Sub‑gradient descent* on the exact ℓ∞‑objective (the maximum
         convolution entry).  The gradient is obtained by summing the
         true gradients of the active (max‑attaining) convolution shifts.
     *   *Adam‑style optimisation* on a smooth‑max surrogate (log‑sum‑exp)
         of the convolution entries.  Both methods keep iterates on the
         simplex via Euclidean projection.
     The best result of the two optimisers is kept.

3️⃣   **Final polishing** – a short stochastic hill‑climbing phase
     (Gaussian perturbations) stays on the simplex and improves the
     best candidate found.

All heavy loops are written with NumPy so that a few thousand iterations
fit easily into the 1000 s budget.  Progress and intermediate scores are
printed for diagnostics.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_score(v: np.ndarray) -> float:
    """Exact value of the evaluation formula (no clipping)."""
    n = len(v)
    conv = np.convolve(v, v)
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _weighted_gradient(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(f*f)_k w.r.t. f.
    w must have length 2·len(f)‑1 and sum to 1.
    """
    n = len(f)
    grad = np.zeros_like(f)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        grad[i_start:i_end + 1] += wk * 2.0 * f[j_range]
        # diagonal correction (already handled by factor 2)
    return grad

def _subgradient_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2000,
    step_initial: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true ℓ∞‑objective.
    Active shifts (those attaining the current maximum) receive
    uniform weight, producing a valid sub‑gradient.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)
    step = step_initial
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        max_val = conv.max()
        # active set (within a tiny tolerance)
        tol = max_val * 1e-12 + 1e-12
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size

        grad = _weighted_gradient(f, w)

        cand = _project_onto_simplex(f - step * grad, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best_f, best_score = cand.copy(), cand_score
            f = cand
            step *= 1.02
            no_improve = 0
        else:
            step *= 0.7
            no_improve += 1

        if no_improve >= 200:
            # shake the iterator a little
            step = step_initial
            no_improve = 0

        if step < 1e-12:
            break

    return best_f, best_score


def _adam_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2000,
    beta_start: float = 0.5,
    beta_factor: float = 1.5,
    beta_max: float = 256.0,
    step_initial: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation on a smoothed‑max surrogate.
    The surrogate is the log‑sum‑exp of the convolution entries with a
    temperature parameter β that is increased steadily.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)

    beta = beta_start
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step = step_initial

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        max_val = conv.max()

        # soft‑max weights (stable)
        exp_vals = np.exp(beta * (conv - max_val))
        w_sum = exp_vals.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_vals / w_sum

        grad = _weighted_gradient(f, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        delta = step * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_onto_simplex(f - delta, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best_f, best_score = cand.copy(), cand_score
            f = cand
            step *= 1.02
        else:
            step *= 0.7
            # keep the old iterate (no “reject” step here)

        # anneal β
        beta = min(beta_max, beta * beta_factor)

        if step < 1e-12:
            break

    return best_f, best_score


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.2,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing on the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        sigma *= 0.9995
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial
    return best, best_score

def _init_random(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like random start."""
    a = rng.random(n) + 1e-8
    a = a / a.sum() * target_sum
    return a


def _init_v_shape(n: int, target_sum: float) -> np.ndarray:
    """V‑shaped (large at the ends, small in the centre)."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.abs(idx - center) + 1e-6
    a = a / a.sum() * target_sum
    return a


def _init_edge_heavy(n: int, target_sum: float) -> np.ndarray:
    """Edge‑heavy shape: a_i = max(i+1, n‑i)."""
    a = np.maximum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(np.float64)
    a = a / a.sum() * target_sum
    return a

                    f"score={sc_opt:.6f}"
                )

            # early stopping if we beat the target bound
            if best_score < 1.503:
                print("[early] target achieved")
                deadline = time.monotonic() + 0.2  # tiny buffer
                break

    # --------------------------------------------------------------
    # 2️⃣ Final polishing – stochastic hill‑climbing
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        refined, refined_sc = _hill_climb(
            best_seq,
            target_sum,
            deadline,
            sigma_initial=0.1,
            max_iters=8000,
        )
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[refine] final score={refined_sc:.6f}")

    # ------------------------------------------------------------------
    # Safety clip (evaluate_sequence does the same) and return.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a uniform sequence of modest length
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
