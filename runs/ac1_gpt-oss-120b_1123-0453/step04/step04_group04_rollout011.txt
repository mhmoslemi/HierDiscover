def _adam_step(a, grad, m, v, t, lr, beta1, beta2, eps):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)

    a_new = a - step
    return a_new, m, v

def compute_gradient(a, sigma, fft_len):
    # compute conv_self for softmax
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*len(a)-1]
    maxb = float(conv.max())
    # compute softmax weights
    # to avoid overflow: compute exp((conv - maxb)/sigma)
    shifted = (conv - maxb) / sigma
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0/w.size)
    else:
        w /= w_sum
    # compute gradient
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    cross = np.fft.irfft(fw * fa_rev, n=fft_len)
    grad = 2.0 * cross[len(a)-1 : len(a)-1 + len(a)]
    return grad, maxb
```

"""Propose a step‑function sequence minimizing the max auto‑convolution.

Algorithm:
1. For several candidate lengths (including the known good length) run a
   projected‑Adam optimisation that directly minimises a smooth
   approximation of the max entry of the auto‑convolution.
   The smoothness is controlled by a temperature‑like parameter σ that
   is gradually reduced (annealed) so that the optimisation first follows
   a soft‑max of the whole convolution and later focuses on the true
   worst entry.
2. Each optimisation starts from a random feasible point (or the warm
   start `height_sequence_1` when the length matches) and respects the
   ℓ¹‑norm constraint Σ aᵢ = √(2 n) by projecting on the simplex after
   every update.
3. The best sequence seen across all lengths and restarts is returned.
"""

import time
import math
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(n) + 1
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # all zero, return uniform distribution
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   verbose: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the maximal auto‑convolution entry.
    Returns the best sequence seen and its maximal convolution value.
    """
    n = a0.shape[0]
    a = a0.copy()
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0
    # learning rate – proportional to the ℓ¹‑norm
    lr = target_sum * 0.2   # empirical
    # FFT length sufficient for a*a and w*a_rev convolutions
    fft_len = 1 << ((3 * n - 2).bit_length())
    best_a = a.copy()
    best_val = float('inf')
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        # ----- convolution a*a -------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        maxb = float(conv_full.max())

        # keep the best ever seen
        if maxb < best_val:
            best_val = maxb
            best_a = a.copy()
            if verbose:
                print(f"[adam] it={iteration} maxb={maxb:.6f}")

        # ----- temperature schedule (σ) ---------------------------------------
        # start with a relatively smooth approximation and shrink it gradually
        # σ = maxb * max(σ_min, 0.5 ** (iteration / 200))
        sigma_factor = max(0.02, 0.5 ** (iteration / 200.0))
        sigma = maxb * sigma_factor + 1e-12   # avoid zero

        # ----- soft‑max weights w ---------------------------------------------
        shifted = (conv_full - maxb) / sigma   # ≤ 0
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----------------------------
        # g_i = 2 * Σ_j w_{i+j} a_j   (cross‑correlation of w and a)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        cross = np.fft.irfft(fw * fa_rev, n=fft_len)
        grad = 2.0 * cross[n - 1 : n - 1 + n]

        # ----- Adam update ----------------------------------------------------
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # projection on the simplex (keeps non‑negativity and Σa = target_sum)
        a = _project_onto_simplex(a, target_sum)

    return best_a, best_val

def _adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    verbose: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the maximal auto‑convolution entry.
    Returns a tuple (best_sequence, best_value) where best_value is the
    maximal entry of a*a (which equals the evaluation score because
    Σa = √(2 n)).
    """
    n = a0.shape[0]
    a = a0.copy()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0
    lr = target_sum * 0.2          # learning‑rate proportional to the ℓ¹‑norm

    # FFT length sufficient for both a*a (2n‑1) and w*a_rev (3n‑2)
    fft_len = 1 << ((3 * n - 2).bit_length())

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- a*a ---------------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        maxb = float(conv_full.max())

        # store best
        if maxb < best_val:
            best_val = maxb
            best_a = a.copy()
            if verbose:
                print(f"[adam] it={it} maxb={maxb:.6f}")

        # ----- σ schedule (smooth‑max temperature) -------------------------
        # start relatively smooth and shrink it exponentially
        sigma_factor = max(0.02, 0.5 ** (it / 200.0))
        sigma = maxb * sigma_factor + 1e-12  # avoid zero

        # ----- soft‑max weights ---------------------------------------------
        shifted = (conv_full - maxb) / sigma   # ≤ 0
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -------------------------
        # g_i = 2 * Σ_j w_{i+j} a_j   (cross‑correlation of w and a)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        cross = np.fft.irfft(fw * fa_rev, n=fft_len)
        grad = 2.0 * cross[n - 1 : n - 1 + n]

        # ----- Adam update -------------------------------------------------
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # Projection back onto the ℓ¹‑simplex (keeps non‑negativity and sum)
        a = _project_onto_simplex(a, target_sum)

    return best_a, best_val


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)   # tiny safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Candidate lengths (include the known construction if present)
    # ------------------------------------------------------------------
    length_candidates = [500, 800, 1000, 1200, 1500, 1800]
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        n0 = len(h1)
        if n0 not in length_candidates:
            length_candidates.append(n0)
    length_candidates = sorted(set(length_candidates))

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2️⃣ Allocate roughly equal time to each length
    # ------------------------------------------------------------------
    total_lengths = len(length_candidates)
    for idx, n in enumerate(length_candidates):
        if time.monotonic() >= deadline:
            break

        remaining = deadline - time.monotonic()
        # remaining lengths (including current)
        remaining_lengths = total_lengths - idx
        per_len = remaining / max(1, remaining_lengths)

        sub_deadline = time.monotonic() + per_len

        target_sum = math.sqrt(2.0 * n)

        # ----------------------------------------------------------------
        # 3️⃣ Random restarts (warm‑start when possible)
        # ----------------------------------------------------------------
        restarts = 4
        for r in range(restarts):
            if time.monotonic() >= sub_deadline:
                break

            if (
                r == 0
                and isinstance(h1, (list, tuple, np.ndarray))
                and len(h1) == n
            ):
                a0 = np.array(h1, dtype=np.float64)
                a0 = np.clip(a0, 0.0, 1000.0)
                s = a0.sum()
                if s == 0.0:
                    a0 = np.full(n, target_sum / n)
                else:
                    a0 = a0 * (target_sum / s)
            else:
                # Dirichlet‑like random start
                rnd = rng.random(n) + 1e-6
                a0 = rnd / rnd.sum() * target_sum

            # ----------------------------------------------------------------
            # 4️⃣ Optimise with Adam
            # ----------------------------------------------------------------
            seq, val = _adam_optimize(a0, target_sum, sub_deadline, verbose=False)

            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[update] n={n} restart={r} score={val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Fallback in the unlikely case nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)

    # Ensure final sequence respects the limits of evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # (optional) report the final score using the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] best_score={best_score:.6f}, evaluate_sequence={final_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
