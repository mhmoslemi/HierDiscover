def _anneal_opt(a0, deadline, max_iters=None):
    a = a0.copy()
    cur_score = _compute_score_from_vector(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = len(a)
    # compute schedule
    start_time = time.monotonic()
    it = 0
    while time.monotonic() < deadline:
        # maybe set schedule
        remaining = deadline - time.monotonic()
        # compute temperature based on proportion of time elapsed
        t_frac = (time.monotonic() - start_time) / (deadline - start_time)
        T = max(0.001, (1 - t_frac) * 0.5)  # start 0.5, decay to near 0
        # propose move
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # delta as proportion of a[i]
        delta = rng.random() * min(a[i], 0.1)  # up to 10% of donor
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonneg < 1000
        # No need to clip as delta ensures nonneg and sum unchanged.
        # Compute score
        score_new = _compute_score_from_vector(a_new)
        if score_new < cur_score or rng.random() < math.exp((cur_score - score_new) / T):
            a = a_new
            cur_score = score_new
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[SA] n={n} it={it} best={best_score:.6f}")
        it += 1
        # optionally break after many iter if no improvement.
        if it >= max_iters:
            break
    return best_a, best_score

"""
Hybrid annealing + Adam refinement for the step‑function bound.

The search works in two phases:

1️⃣  **Simulated‑Annealing (SA)** on the probability simplex.
    The current vector `a` (∑a = 1, a≥0) is represented as a soft‑max
    of a real vector `z`.  At each iteration a Gaussian perturbation is
    applied to `z`, a new candidate `a'` is obtained by the soft‑max,
    and the exact objective `2·n·max(conv(a',a'))` is evaluated.
    An annealed temperature schedule allows uphill moves, helping the
    search escape local minima.  The best candidate found by SA is kept.

2️⃣  **Adam‑style projected gradient descent** that smooths the max
    using a soft‑max over the convolution entries.  This polish stage
    quickly drives the solution to a (near) stationary point while
    preserving the ℓ¹‑norm.

The algorithm iterates over several sequence lengths (up to 3000) and
multiple random restarts, respecting the overall 1000 s budget.
The best sequence found is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all are top‑level, no closures or lambdas.
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # uniform vector if all entries are projected to zero
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact score used by ``evaluate_sequence`` (no internal clipping)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / beta
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    return w


def _weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. `a`.
    `weights` must have length 2·len(a)‑1 and sum to 1.
    """
    n = len(a)
    grad = np.zeros_like(a)
    for k, wk in enumerate(weights):
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = a[j]
        grad[i] += wk * 2.0 * vals
        # correct the double‑count on the diagonal (i == j)
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= wk * vals[mask]
    return grad


def _adam_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 20000,
    patience: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation of the smoothed max‑convolution.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = len(a)
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        # smoothing parameter – a few percent of the current maximal value
        beta = max_conv * 0.07 + 1e-12
        w = _softmax_weights(conv, beta)
        grad = _weighted_gradient(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cur_score = _compute_score_from_vector(a)
        cand_score = _compute_score_from_vector(a_candidate)

        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            print(f"[Adam] n={n} it={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score


def _simulated_annealing(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple temperature‑controlled annealing on the simplex.
    The state is represented by a log‑parameter vector `z` so that
    a = softmax(z) automatically satisfies ∑a = 1 and a≥0.
    """
    rng = np.random.default_rng()
    n = len(a0)
    # initialise log‑parameterisation (add a tiny epsilon for safety)
    epsilon = 1e-12
    z = np.log(a0 + epsilon)

    cur_score = _compute_score_from_vector(a0)
    best_a = a0.copy()
    best_score = cur_score
    start = time.monotonic()

    # schedule parameters (tuned empirically)
    sigma_start, sigma_end = 0.8, 0.01   # magnitude of Gaussian steps
    T_start, T_end = 0.6, 0.0005        # temperature for Metropolis accept

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        sigma = sigma_start * (1 - frac) + sigma_end * frac
        T = T_start * (1 - frac) + T_end * frac

        # propose a new log‑vector
        z_new = z + rng.normal(scale=sigma, size=n)

        # soft‑max to obtain a candidate on the simplex
        max_z = np.max(z_new)
        exp_z = np.exp(z_new - max_z)          # stability
        a_new = exp_z / np.sum(exp_z)

        new_score = _compute_score_from_vector(a_new)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / max(T, 1e-12)):
            z, cur_score = z_new, new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
                print(f"[SA] n={n} elapsed={elapsed:.1f}s best={best_score:.6f}")

    return best_a, best_score
