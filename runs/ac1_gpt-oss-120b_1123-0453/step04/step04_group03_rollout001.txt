"""
Propose a low‑value non‑negative step sequence.

The algorithm is a gradient‑based optimisation of the *smoothed* max of the
auto‑convolution, but it works in the *softmax* (log‑sum‑exp) domain and
uses the fast Fourier transform for O(n log n) convolutions.  The
parameterisation

    a_i = S * exp(x_i) / Σ_j exp(x_j)

automatically satisfies the non‑negativity and ℓ¹‑norm constraints,
so no explicit projection is required.  Adam is applied to the
parameters `x`; the gradient of the smooth objective is obtained via a
single FFT‑based convolution per iteration.  After the main Adam phase
a cheap stochastic refinement (Gaussian perturbations + simplex
projection) is performed when time permits.

The search explores several lengths (up to 5000) and a few random
re‑starts, keeping the best sequence seen.  All computation respects the
overall time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _fft_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using a power‑of‑two FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    n = a.size + b.size - 1
    # next power of two for efficient FFT
    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    y = np.exp(x - mx)
    s = np.sum(y)
    return y / s


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max over the convolution entries with temperature `beta`.
    Returns a vector of the same length that sums to 1.
    """
    maxc = np.max(conv)
    shifted = (conv - maxc) / beta
    exp_shifted = np.exp(shifted)
    return exp_shifted / np.sum(exp_shifted)


def _weighted_gradient_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k * (a*a)_k w.r.t. `a`.
    Both `a` (length n) and `w` (length 2n‑1) are real.
    """
    n = a.size
    # reverse of a
    a_rev = a[::-1]

    # conv_rev = w * a_rev  (full)
    conv_rev_full = _fft_convolution(w, a_rev)
    # slice that corresponds to index i (see derivation)
    start = n - 1
    conv_rev = conv_rev_full[start:start + n]

    # slice_sum_i = Σ_{k=i}^{i+n-1} w_k
    ones = np.ones(n, dtype=a.dtype)
    conv_one_full = _fft_convolution(w, ones)
    slice_sum = conv_one_full[start:start + n]

    # gradient_i = conv_rev_i + a_i * slice_sum_i
    return conv_rev + a * slice_sum


def _compute_score_from_vector_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the score when the ℓ¹‑norm is 1:
        score = 2 * n * max(conv(a,a))
    This avoids the O(n²) convolution used by the reference routine.
    """
    n = a.size
    conv = _fft_convolution(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def _adam_optimize_x(
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 20000,
    patience: int = 800,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑space `x` (so that a = softmax(x)*target_sum).
    Returns the best sequence `a` (already summed to `target_sum`) and its
    exact evaluation (`_compute_score_from_vector`).
    """
    x = x0.copy()
    n = x.size
    # Adam hyper‑parameters
    lr = 0.08
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # initial point
    a = _softmax(x) * target_sum
    best_a = a.copy()
    best_score = _compute_score_from_vector_fast(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # ----- forward pass -----
        a = _softmax(x) * target_sum                     # ℓ¹‑norm = target_sum
        conv = _fft_convolution(a, a)                    # length 2n‑1
        max_conv = np.max(conv)

        # ----- smoothing (soft‑max) -----
        beta = max_conv * 0.07 + 1e-12
        w = _softmax_weights(conv, beta)                 # length 2n‑1, Σw = 1

        # ----- gradient w.r.t. a -----
        g_a = _weighted_gradient_fft(a, w)

        # ----- chain rule to obtain gradient w.r.t. x -----
        # a = target_sum * softmax(x)  →  da/dx_i = a_i * (δ_ij - a_j/target_sum)
        s = a / target_sum                               # this is the soft‑max vector
        dot = np.dot(g_a, s)                             # scalar
        g_x = s * (g_a - dot)                            # shape (n,)

        # ----- Adam update (gradient descent) -----
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # ----- progress tracking -----
        cur_score = 2.0 * n * max_conv                     # exact score for sum=target_sum
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam] n={n:4d} it={it:5d} score={cur_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    # final exact evaluation (uses the reference O(n²) routine only once)
    final_score = _compute_score_from_vector_fast(best_a)
    return best_a, final_score


def _stochastic_refinement(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.3,
    swap_prob: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing around `start`.  The ℓ¹‑norm is kept fixed by a
    Euclidean projection onto the simplex.
    """
    rng = np.random.default_rng()
    n = start.size
    target_sum = np.sum(start)
    best = start.copy()
    best_score = _compute_score_from_vector_fast(best)
    step = step_initial

    # helper for simplex projection (no external dependencies)
    def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
        if s < 0:
            raise ValueError("negative simplex sum")
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        if not np.any(cond):
            return np.full_like(v, s / len(v))
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = best + rng.normal(scale=step, size=n)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_simplex(cand, target_sum)

        cand_score = _compute_score_from_vector_fast(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = min(step_initial, step * 1.02)
        else:
            step *= 0.995

        if step < 1e-6:
            step = step_initial

    return best, best_score

        score = _compute_score_from_vector_fast(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[warm] loaded height_sequence_1, n={len(seq)} score={score:.6f}")

    # --------------------------------------------------------------

        for rs in range(restarts):

            # Initialise parameters x (log‑space).  Use the current best if it matches n.
            if best_seq is not None and best_seq.size == n:
                # convert to log‑space, then add tiny noise

            else:

            )

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[update] n={n} restart={rs} score={sc_opt:.6f}")

            # Early exit if we already beat the target comfortably
            if best_score < 1.45:
                print("[early] satisfactory score reached")
                break

        if best_score < 1.45:
            break

    # --------------------------------------------------------------
    # 2️⃣ Final stochastic polishing (if time permits).
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform sequence of moderate length

        if refined_score < best_score:

            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # Ensure the values respect the clipping performed by `evaluate_sequence`.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # --------------------------------------------------------------
    # Return a plain list – the external `evaluate_sequence` will be called later.
    # --------------------------------------------------------------
    return [float(v) for v in best_seq]
