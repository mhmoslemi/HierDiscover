    a = a0.copy()
    n = len(a)
    # Adam hyperparameters
    lr = 0.04
    bet1, bet2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    no_improve = 0
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute conv via FFT
        conv = _conv_fft(a)
        max_conv = conv.max()
        # compute softmax weights with decreasing beta (optional schedule)
        beta = max_conv * beta_factor + 1e-12
        # compute w via stable softmax
        shifted = conv - max_conv

        w = np.exp(shifted / beta)

"""
Propose a low‑value non‑negative step sequence.

The algorithm builds on the previous Adam‑based optimisation but adds
several enhancements that make it more effective within the 1000 s budget:

* **FFT‑based convolutions** – exact auto‑convolution and the weighted
  gradient are obtained with O(n log n) work (instead of O(n²)).  This
  permits the use of longer sequences (up to a few‑thousand entries) and
  many optimisation steps without exceeding the time limit.

* **Soft‑max smoothing with a decreasing temperature** – the objective
  `max(conv)` is approximated by `log‑sum‑exp(conv/β)`.  The temperature
  `β` is set to a small fraction of the current maximal value and is
  gradually reduced, so the optimisation first moves quickly in a smooth
  landscape and later focuses on the true maximum.

* **Diverse initialisations** – uniform, Dirichlet‑sampled and
  sparse‑vector seeds (≈√n non‑zeros) are tried.  Sparse starts tend to
  spread pairwise sums and often achieve a lower peak in the
  auto‑convolution.

* **Adam with projection onto the ℓ¹‑simplex** – the ℓ¹‑norm is kept at
  a fixed value (1.0) throughout the optimisation, guaranteeing scale
  invariance of the original evaluation function.

* **Stochastic polishing** – after the main Adam phase a cheap hill‑climbing
  routine performs Gaussian perturbations and occasional pair swaps,
  always re‑projecting onto the simplex.  This frequently yields a final
  improvement of a few‑per‑cent on the objective.

All helpers are defined at top‑level; the entry point required by the
evaluation framework is `propose_candidate`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # Work on a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – distribute uniformly
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolutions and gradient (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Exact auto‑convolution a*a using real FFT.
    Returns a vector of length 2·len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def _weighted_gradient_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of L(a)=Σ_k w_k·(a*a)_k w.r.t. a.
    ``w`` must have length 2·len(a)-1 and sum to 1.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow_two(n + w.shape[0] - 1)  # ≥ 3·n‑2
    fa_rev = np.fft.rfft(a_rev, n=L)
    fw = np.fft.rfft(w, n=L)
    grad_full = np.fft.irfft(fa_rev * fw, n=L)
    # slice indices n‑1 … n‑1+n‑1 (inclusive)
    start = n - 1
    end = start + n
    grad = 2.0 * grad_full[start:end]
    return grad


# ----------------------------------------------------------------------
# Exact evaluation (identical to the external helper)
# ----------------------------------------------------------------------


def _compute_score_from_vector(f: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a given vector (no internal clipping).
    """
    n = f.shape[0]
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


# ----------------------------------------------------------------------
# Adam optimiser (FFT version)
# ----------------------------------------------------------------------


def _adam_descent_fft(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 15000,
    patience: int = 500,
    beta_factor: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises the smoothed max‑convolution.
    Returns the best vector discovered and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    lr = 0.04
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # ---- auto‑convolution -------------------------------------------------
        conv = _conv_fft(a)
        max_conv = np.max(conv)

        # ---- soft‑max weights -------------------------------------------------
        beta = max_conv * beta_factor + 1e-12
        shifted = conv - max_conv               # ≤ 0
        exp_shifted = np.exp(shifted / beta)    # range (0,1]
        w = exp_shifted / np.sum(exp_shifted)   # sum = 1

        # ---- gradient (FFT) --------------------------------------------------
        grad = _weighted_gradient_fft(a, w)

        # ---- Adam update -----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cur_score = _compute_score_from_vector(a)
        cand_score = _compute_score_from_vector(a_candidate)

        # accept improvement immediately (helps escape flat regions)
        if cand_score < cur_score:
            a = a_candidate

        # record global best
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            print(f"[Adam] n={n:4d} it={it:5d} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score

def _stochastic_refinement(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.3,
    swap_prob: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbations + occasional pairwise swaps.
    The ℓ¹‑norm is preserved by simplex projection after each move.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, len(best), size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = best + rng.normal(scale=step, size=best.shape)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = min(step_initial, step * 1.02)   # grow a bit on success
        else:
            step *= 0.995                           # slowly decay

        if step < 1e-6:
            step = step_initial

    return best, best_score

            # early exit if we already beat the target comfortably
            if best_score < 1.45:
                print("[early] satisfactory score reached")
                break

        if best_score < 1.45:
            break

    # --------------------------------------------------------------
    # Fallback – a modest uniform sequence (should never be used)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 800
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    # --------------------------------------------------------------
    # Final stochastic polishing (time‑permitting)
    # --------------------------------------------------------------
    if time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline, step_initial=0.2)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # Return a plain Python list – the external evaluator will clip again.
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
