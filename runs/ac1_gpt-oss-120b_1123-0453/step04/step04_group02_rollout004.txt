"""
propose_candidate – Search for a non‑negative step sequence with a low
evaluate_sequence score.

The method consists of three phases.

1️⃣  **Initialisation** – we try a handful of handcrafted families
    (uniform, exponential, Hann, etc.) for several lengths and keep the
    best one as a starting point.

2️⃣  **Smooth‑max gradient descent** – the objective contains a
    “max” over the auto‑convolution entries.  We replace the max by a
    smooth approximation
        Sτ(a)= (1/τ)·log∑k exp(τ·(a*a)[k])
    (τ controls the smoothness).  The gradient of Sτ can be written
    analytically and evaluated with two FFT‑based convolutions:
        ∇Sτ = 2·(w * a)   where   w_k ∝ exp(τ·(a*a)[k]).
    At each iteration we move opposite to the gradient, project onto
    the hyperplane Σa = √(2·n) (the problem is scale‑invariant) and
    enforce non‑negativity.  The step size and the smoothing parameter
    τ are adapted on‑the‑fly.  Several random restarts are performed for
    each length.

3️⃣  **Final stochastic polish** – a short isotropic Gaussian hill‑climb
    (random perturbations + re‑normalisation) is applied to the best
    candidate found.

All operations respect the validation performed inside
``evaluate_sequence`` – entries stay in [0,1000] and the ℓ¹‑norm stays
> 0.01, so the returned list is always accepted.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast (full) linear convolution using FFT
# ----------------------------------------------------------------------
def _conv_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors a and b (length n+m‑1)."""
    n, m = a.shape[0], b.shape[0]
    N = 1 << ((n + m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(A * B, n=N)
    return conv[: n + m - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself."""
    return _conv_general(a, a)


# ----------------------------------------------------------------------
# Exact objective (same formula as evaluate_sequence, but without the
# security wrapper – it is faster and gives identical values for valid
# inputs).
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _conv_self(a).max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

# ----------------------------------------------------------------------
def _candidate_families(n: int, target_sum: float):
    """Return the best vector among several simple parametric families."""
    best_vec = None
    best_sc = float("inf")
    idx = np.arange(n, dtype=np.float64)

    def _norm(v):
        v = np.asarray(v, dtype=np.float64)
        return v * (target_sum / v.sum())

    # uniform
    a = np.full(n, target_sum / n, dtype=np.float64)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # exponential decays
    alphas = np.concatenate([
        np.arange(1, 6) / (10.0 * n),
        np.arange(1, 21) / n,
        np.linspace(0.02, 0.20, 40),
        np.linspace(0.20, 1.00, 30),
    ])
    alphas = np.unique(alphas)
    for α in alphas:
        a = np.exp(-α * idx)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # power‑law
    p_vals = np.linspace(0.05, 5.0, 120)
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        a = base ** (-p)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # Hann window
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # Hamming window
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # Blackman window
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * idx / (n - 1)))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # sine window
    a = np.sin(np.pi * (idx + 1) / (n + 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # triangular
    a = np.minimum(idx + 1, n - idx)
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient (via FFT)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, tau: float):
    """
    Returns (smooth_max, gradient) where
        smooth_max = (1/τ)·log∑ₖ exp(τ·(a*a)[k])
    and gradient = 2·(w * a)[:n] with wₖ ∝ exp(τ·(a*a)[k]).
    """
    n = a.shape[0]
    conv = _conv_self(a)                 # length 2n‑1
    max_val = conv.max()
    # stable soft‑max weights
    shifted = conv - max_val
    exps = np.exp(tau * shifted)
    Z = exps.sum()
    w = exps / Z                         # sum_k w_k = 1
    smooth = max_val + math.log(Z) / tau

    # gradient = 2 * (w * a) (linear convolution, then take first n entries)
    grad_full = _conv_general(w, a)      # length (2n‑1)+n‑1 = 3n‑2
    grad = 2.0 * grad_full[:n]
    return smooth, conv, w, grad


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (with projection onto Σa = target_sum)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float):
    """
    Projected gradient descent on the smooth‑max of the auto‑convolution.
    Returns the best vector encountered and its exact score.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _fast_score(a)

    # hyper‑parameters
    step = 0.15          # initial step size
    tau = 0.8            # temperature for the smooth max
    rng = np.random.default_rng()

    while time.monotonic() < deadline:
        smooth, conv, w, grad = _smooth_max_and_grad(a, tau)

        # direction that keeps the ℓ¹‑norm unchanged (projection)
        direction = -grad + grad.mean()

        cand = a + step * direction
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise the ℓ¹‑norm to the prescribed target
        total = cand.sum()
        if total == 0.0:
            cand = np.full_like(cand, target_sum / n)
        else:
            cand = cand * (target_sum / total)

        cand_score = _fast_score(cand)

        # simple accept‑if‑better rule; otherwise shrink the step
        if cand_score < best_score:
            best_a, best_score = cand.copy(), cand_score
            a = cand
            step = min(step * 1.10, 2.0)          # grow a little
            tau = min(tau * 1.05, 25.0)           # sharpen the max
        else:
            step *= 0.55
            if step < 1e-6:                       # occasional random shake‑up
                i, j = rng.integers(n, size=2)
                delta = a[i] * 0.07
                if delta > 0:
                    cand2 = a.copy()
                    cand2[i] = max(cand2[i] - delta, 0.0)
                    cand2[j] = min(cand2[j] + delta, 1000.0)
                    cand2 = cand2 * (target_sum / cand2.sum())
                    sc2 = _fast_score(cand2)
                    if sc2 < best_score:
                        a = cand2
                        best_a, best_score = cand2.copy(), sc2
                step = 0.15
                tau = max(tau * 0.90, 0.5)

        # early exit if we already hit the theoretical lower bound
        if best_score < 1.02:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Small stochastic polishing (Gaussian hill‑climb)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, deadline: float):
    """Isotropic Gaussian perturbations with re‑normalisation."""
    n = a0.shape[0]
    best = a0.copy()
    best_sc = _fast_score(best)
    rng = np.random.default_rng()
    step = 0.2

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        tot = cand.sum()
        if tot == 0.0:
            continue
        cand = cand * (best.sum() / tot)   # keep the same ℓ¹‑norm
        sc = _fast_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
        step *= 0.9997
        if step < 1e-7:
            step = 0.2
    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence found before the time budget runs out.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for bookkeeping prints etc.
    deadline = start + max(0.0, float(budget_s) - 0.15)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Use a previously discovered sequence if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = init * (target0 / init.sum())
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 n={n0} score={sc0:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Scan a few lengths – larger lengths lower the theoretical bound
    # ------------------------------------------------------------------
    lengths = [400, 600, 800, 1000, 1200]   # feel free to adjust
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # 1a️⃣  Hand‑crafted families (quick, cheap)
        # ------------------------------------------------------------------
        fam_vec, fam_sc = _candidate_families(n, target)
        if fam_sc < best_score:
            best_seq, best_score = fam_vec.copy(), fam_sc
            print(f"[family] n={n}  score={fam_sc:.6f}")

        # ------------------------------------------------------------------
        # 1b️⃣  Several random Dirichlet seeds (very cheap)
        # ------------------------------------------------------------------
        for _ in range(6):
            a = rng.random(n)
            a = a * (target / a.sum())
            sc = _fast_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[rand]   n={n}  score={sc:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣  Smooth‑max gradient descent from each promising seed
        # ------------------------------------------------------------------
        seeds = []
        if best_seq is not None and best_seq.shape[0] == n:
            seeds.append(best_seq.copy())
        # add a few fresh random seeds (to diversify)
        for _ in range(3):
            a = rng.random(n)
            a = a * (target / a.sum())
            seeds.append(a)

        for init in seeds:
            if time.monotonic() >= deadline:
                break
            refined, refined_sc = _smooth_gradient_descent(init, target, deadline)
            if refined_sc < best_score:
                best_seq, best_score = refined.copy(), refined_sc
                print(f"[grad]   n={n}  improved to {refined_sc:.6f}")

        # ------------------------------------------------------------------
        # 3️⃣  Final polish (only if we still have time)
        # ------------------------------------------------------------------
        if best_seq is not None and time.monotonic() < deadline:
            polished, polished_sc = _hill_climb(best_seq, deadline)
            if polished_sc < best_score:
                best_seq, best_score = polished, polished_sc
                print(f"[final]  n={n}  polished to {polished_sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be required)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    # convert to plain Python list as required by the harness
    return [float(v) for v in best_seq]
