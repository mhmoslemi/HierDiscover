def _project_onto_simplex(v, z=1.0):
    """Project vector v onto the simplex {x >=0, sum x = z}."""
    # Implementation based on:
    #   Wang & Carreira-Perpinán 2013.
    # similar to Duchi et al.
    if v.ndim != 1:
        raise ValueError
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Improved search for a low‑peak autocorrelation (step‑function) sequence.

Algorithm
---------
* Work on several lengths (1000 … 200).  Larger n gives more freedom.
* For each length generate cheap analytic seeds (uniform, triangular,
  arcsine, aggressive Beta(α,β) with α,β∈[0.1,0.5] and a few Dirichlet draws).
* Refine each seed by **projected exponentiated‑gradient descent** on the
  simplex {a≥0, Σa = √(2n)}.
  – At each iteration the active convolution index k (where the auto‑
    convolution attains its maximum) is identified.
  – The sub‑gradient of the active entry is g_i = 2·a_{k‑i} (zero outside the
    support).  An exponentiated update a_i←a_i·exp(−η·g_i) reduces the
    contribution of the “worst’’ indices while staying non‑negative.
  – After the multiplicative step the vector is symmetrised
    (a←(a+a[::-1])/2) and projected back onto the simplex.
  – The learning rate η decays geometrically.
* Keep the best vector over all lengths and restarts.
* A final cheap hill‑climb (pairwise mass swaps + a short Gaussian
  perturbation) polishes the solution.

The whole routine respects the security checks of ``evaluate_sequence`` –
entries stay non‑negative, bounded by 1000 and the ℓ¹‑norm is fixed to
√(2·n).  The function stops just before the allotted time limit and returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``vec`` to have ℓ¹‑norm exactly ``target_sum`` (preserving non‑negativity)."""
    s = vec.sum()
    if s <= 0.0:
        # fall back to uniform distribution
        n = vec.shape[0]
        return np.full(n, target_sum / n, dtype=np.float64)
    return vec * (target_sum / s)


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` without the safety wrapper.
    For a vector with Σa = √(2·n) the returned value equals max(a*a).
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


def _project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of a vector ``v`` onto the simplex
    { x ≥ 0, Σ x = z }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    # make a copy to avoid modifying the original
    v = v.astype(np.float64, copy=True)
    u = np.sort(v)[::-1]                      # descending sort
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / len(v)
    else:
        rho = ind[cond][-1] - 1                # zero‑based index of the last positive entry
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry a_i = a_{n-1-i}."""
    return (v + v[::-1]) * 0.5


def _init_beta_shape(n: int, a: float, b: float, target: float) -> np.ndarray:
    """Beta‑distribution shape a_i ∝ x^{a‑1}(1‑x)^{b‑1}, x∈(0,1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    w = np.maximum(w, 1e-12)
    w = w / w.sum() * target
    return w


def _generate_initial_vectors(n: int, target: float, rng: np.random.Generator):
    """Return a modest list of plausible start vectors for length ``n``."""
    cand = []

    # uniform
    cand.append(np.full(n, target / n, dtype=np.float64))

    # simple symmetric triangular
    x = np.arange(n, dtype=np.float64)
    tri = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
    tri[tri < 0] = 0.0
    cand.append(_normalize(tri, target))

    # arcsine (Beta(½,½))
    cand.append(_init_beta_shape(n, 0.5, 0.5, target))

    # a few aggressive Beta shapes (α,β ∈ [0.1,0.5] tend to concentrate mass at ends)
    for _ in range(4):
        a_param = rng.uniform(0.1, 0.5)
        b_param = rng.uniform(0.1, 0.5)
        cand.append(_init_beta_shape(n, a_param, b_param, target))

    # random Dirichlet draws (already on the simplex)
    for _ in range(3):
        cand.append(rng.dirichlet(np.ones(n)) * target)

    return cand


# ----------------------------------------------------------------------
# Core optimiser – projected exponentiated‑gradient descent
# ----------------------------------------------------------------------


def _exp_grad_descent(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 15000,
) -> np.ndarray:
    """
    Refine ``start`` by projected exponentiated‑gradient descent on the
    ℓ∞‑norm of the auto‑convolution.
    """
    n = start.shape[0]
    # initialise on the simplex
    cur = _normalize(start, target_sum)
    # keep a copy of the best vector seen so far
    best = cur.copy()
    best_sc = _fast_score(best)

    # initial learning rate (η) – a value around 0.8 works well
    eta0 = 0.8
    decay = 0.998

    eps_tolerance = 1e-12

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # compute auto‑convolution and locate the (one of) maximal entry
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        # all indices that are (almost) maximal
        act = np.where(conv >= max_val - eps_tolerance)[0]
        k = int(rng.choice(act))

        # gradient of the active entry (sub‑gradient of max)
        grad = np.zeros(n, dtype=np.float64)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idxs] = 2.0 * cur[k - idxs]  # ∂b_k/∂a_i = 2*a_{k-i}

        # exponentiated update – pushes down components with large gradient
        eta = eta0 * (decay ** it)
        cur_exp = cur * np.exp(-eta * grad)

        # enforce symmetry – the optimal shape is typically symmetric
        cur_exp = _symmetrize(cur_exp)

        # project back onto the simplex Σa = target_sum, a≥0
        cur = _project_onto_simplex(cur_exp, target_sum)

        # keep the best vector
        sc = _fast_score(cur)
        if sc < best_sc:
            best, best_sc = cur.copy(), sc

        # occasional random jitter to escape plateaus
        if (it + 1) % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.02, size=n)
            cur = np.clip(cur + jitter, 0.0, None)
            cur = _project_onto_simplex(cur, target_sum)

    return best


# ----------------------------------------------------------------------
# Light‑weight polishing primitives
# ----------------------------------------------------------------------


def _swap_hill_climb(seq: np.ndarray, target_sum: float, deadline: float,
                    rng: np.random.Generator) -> np.ndarray:
    """Very cheap pair‑wise mass‑swap hill‑climb."""
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target_sum * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _project_onto_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
    return best


def _gaussian_local_search(seq: np.ndarray, target_sum: float, deadline: float,
                           rng: np.random.Generator) -> np.ndarray:
    """Short annealing‑style Gaussian perturbation loop."""
    best = seq.copy()
    best_sc = _fast_score(best)
    step = target_sum * 0.02

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape[0])
        cand = _project_onto_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
            step *= 0.9
        else:
            step *= 0.99
            if step < 1e-7:
                step = target_sum * 0.02
    return best

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    kwargs : dict
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.3)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ – start from a known construction if present.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = _normalize(init, target0)
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣ – sweep over several lengths, allocating time proportionally.
    # ------------------------------------------------------------------
    lengths = [1000, 800, 600, 500, 400, 300, 200]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # generate a modest set of seeds for this length
        # --------------------------------------------------------------
        seeds = _generate_initial_vectors(n, target, rng)

        # maybe reuse the global best if it already has the same length
        if best_seq is not None and best_seq.shape[0] == n:
            seeds.append(best_seq.copy())

        # --------------------------------------------------------------
        # allocate a slice of the remaining budget for this length
        # --------------------------------------------------------------
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break
        # we will spend roughly half of the remaining budget on this length,
        # the rest will be kept for later lengths / polishing.
        length_budget = min(30.0, remaining * 0.5)
        # split the budget among the seeds (at least a few seconds each)
        per_seed_budget = max(3.0, length_budget / max(1, len(seeds)))

        for idx, start_vec in enumerate(seeds):
            if time.monotonic() >= deadline:
                break
            sub_deadline = time.monotonic() + per_seed_budget
            refined = _exp_grad_descent(start_vec, target, sub_deadline, rng)
            refined_sc = _fast_score(refined)
            if refined_sc < best_score:
                best_seq, best_score = refined.copy(), refined_sc
                print(f"[refine] n={n} seed={idx} improved to {refined_sc:.6f}")

            # Occasionally launch a tiny random restart if we have time left
            if rng.random() < 0.08 and time.monotonic() + 5.0 < deadline:
                restart_vec = rng.dirichlet(np.ones(n)) * target
                sub_deadline2 = time.monotonic() + 5.0
                cand = _exp_grad_descent(restart_vec, target, sub_deadline2, rng)
                cand_sc = _fast_score(cand)
                if cand_sc < best_score:
                    best_seq, best_score = cand.copy(), cand_sc
                    print(f"[restart] n={n} extra run new best {cand_sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ – final polishing on the globally best vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: modest uniform vector
        n = 300
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # enforce exact ℓ¹‑norm and symmetry before polishing
    n = best_seq.shape[0]
    target = math.sqrt(2.0 * n)
    best_seq = _project_onto_simplex(_symmetrize(best_seq), target)

    # cheap swap‑based hill‑climb
    best_seq = _swap_hill_climb(best_seq, target, deadline, rng)

    # short Gaussian perturbation stage (if time permits)
    if time.monotonic() < deadline:
        best_seq = _gaussian_local_search(best_seq, target, deadline, rng)

    final_score = _fast_score(best_seq)
    print(f"[final] best score {final_score:.6f}   length {best_seq.shape[0]}")

    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
