def _project_onto_simplex(v, s):
    """Euclidean projection of vector v onto simplex {x >= 0, sum x = s}."""
    # algorithm from Condat 2016
    if s < 0:
        raise ValueError('s must be non-negative')
    n = len(v)
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All zeros
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    return w

"""
Propose a low‑value step sequence.

The search combines three ideas:

1. **Gradient descent on the ℓ¹‑simplex** – the objective is the
   maximal entry of the auto‑convolution.  For a current vector `a`
   we compute a sub‑gradient (the sum of the active convolutions) and
   take a step opposite to it, projecting back onto the simplex
   `{a≥0, Σa = √(2 n)}` with the Euclidean projection.

2. **Multiple random restarts** – for each tested length we start from
   several simple seeds (warm‑start, Dirichlet random, sparse spikes,
   linear ramp, uniform).  This diversifies the search and reduces the
   chance of being trapped in a local minimum.

3. **A short simulated‑annealing polishing phase** – after the deterministic
   descent we run a few hundred annealing moves to possibly escape the
   basin of attraction and squeeze a little more out of the best vector.

The algorithm respects the 1000 s budget, prints progress, and finally
returns the best sequence found (as a plain Python list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full auto‑convolution a*a using an FFT of length ``fft_len``.
    Returns the first (2·len(a)‑1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    needed = 2 * len(a) - 1
    return conv[:needed]


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the auto‑convolution of ``a``."""
    conv = _full_convolution(a, fft_len)
    return float(conv.max())


def _subgradient_and_max(a: np.ndarray, fft_len: int):
    """
    Compute a sub‑gradient of the maximal convolution entry and the value.
    The sub‑gradient is the sum of gradients of all (near) active constraints.
    """
    n = len(a)
    conv = _full_convolution(a, fft_len)
    max_val = conv.max()
    eps = max_val * 1e-9 + 1e-12
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        grad[i_range] += a[j_range]          # derivative of a_i a_{k-i}
    return grad, max_val


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x | x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = len(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that minimizes
    ``evaluate_sequence`` within the allotted time.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.05)  # tiny safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Choose a set of lengths to explore
    # ------------------------------------------------------------------
    length_candidates = [500, 800, 1000, 1200, 1500, 1800, 2000]
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        length_candidates.append(len(h1))
    length_candidates = sorted(set(length_candidates))

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 2️⃣  Main loop over lengths
    # ------------------------------------------------------------------
    for n in length_candidates:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)               # we keep Σa = √(2 n)
        fft_len = _next_pow2(2 * n - 1)                # FFT length for this n

        # how many random restarts for this length? 5 is a good trade‑off.
        max_restarts = 5

        for restart in range(max_restarts):
            if time.monotonic() >= deadline:
                break

            # --------------------------------------------------------------
            # 2.1️⃣  Initialise a starting vector
            # --------------------------------------------------------------
            if restart == 0 and isinstance(h1, (list, tuple, np.ndarray)) and len(h1) == n:
                a = np.array(h1, dtype=np.float64)
                a = np.clip(a, 0.0, 1000.0)
                s = a.sum()
                a = a * (target_sum / s) if s > 0 else np.full(n, target_sum / n)
            elif restart == 1:
                # Dirichlet‑like random start
                a = rng.random(n) + 1e-6
                a = a / a.sum() * target_sum
            elif restart == 2:
                # Sparse “spike” start – roughly √(2 n) equal spikes
                k = max(2, int(math.sqrt(2 * n)))
                a = np.zeros(n)
                pos = rng.choice(n, size=k, replace=False)
                w = target_sum / k
                a[pos] = w
                # small jitter to break symmetry
                a[pos] += rng.random(k) * 0.01 * w
                a = a * (target_sum / a.sum())
            elif restart == 3:
                # Linear decreasing ramp
                a = np.linspace(1.0, 0.0, n)
                a = a / a.sum() * target_sum
            else:
                # Uniform start
                a = np.full(n, target_sum / n)

            # --------------------------------------------------------------
            # 2.2️⃣  Deterministic projected sub‑gradient descent
            # --------------------------------------------------------------
            # start value
            cur_val = _max_convolution(a, fft_len)
            best_local_seq = a.copy()
            best_local_val = cur_val

            init_step = target_sum * 0.2          # initial step magnitude
            max_iters = 2000                     # enough iterations for convergence

            for it in range(max_iters):
                if time.monotonic() >= deadline:
                    break

                grad, cur_val = _subgradient_and_max(a, fft_len)

                # diminishing step size
                step = init_step / math.sqrt(it + 1)

                # projected move
                cand = _project_onto_simplex(a - step * grad, target_sum)
                cand = np.clip(cand, 0.0, 1000.0)

                cand_val = _max_convolution(cand, fft_len)

                # accept if we improve (no uphill moves in this phase)
                if cand_val <= cur_val:
                    a = cand
                    cur_val = cand_val
                    if cand_val < best_local_val - 1e-12:
                        best_local_val = cand_val
                        best_local_seq = a.copy()
                else:
                    # occasional random perturbation to shake out of flats
                    if rng.random() < 0.01:
                        i = rng.integers(0, n)
                        j = rng.integers(0, n - 1)
                        if j >= i:
                            j += 1
                        delta = rng.normal(scale=step)
                        delta = max(min(delta, a[i]), -a[j])
                        a[i] -= delta
                        a[j] += delta
                        a = _project_onto_simplex(a, target_sum)

            # --------------------------------------------------------------
            # 2.3️⃣  Keep the best vector found for this length
            # --------------------------------------------------------------
            if best_local_val < best_val:
                best_val = best_local_val
                best_seq = best_local_seq.copy()
                print(f"[global update] n={n} restart={restart} val={best_val:.6f}")

            # --------------------------------------------------------------
            # 2.4️⃣  Brief simulated‑annealing polishing (optional)
            # --------------------------------------------------------------
            if time.monotonic() < deadline and best_seq is not None and len(best_seq) == n:
                a = best_seq.copy()
                cur_val = _max_convolution(a, fft_len)
                # annealing schedule
                T0 = cur_val * 0.5 + 1e-4
                anneal_iters = 2000
                for t in range(anneal_iters):
                    if time.monotonic() >= deadline:
                        break
                    frac = t / anneal_iters
                    T = T0 * (0.001 ** frac)
                    step = target_sum * (T / T0) * 1.5
                    i = rng.integers(0, n)
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    delta = rng.normal(scale=step)
                    delta = min(delta, a[i])
                    delta = max(delta, -a[j])
                    if abs(delta) < 1e-12:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_val = _max_convolution(a, fft_len)
                    accept = False
                    if new_val < cur_val - 1e-12:
                        accept = True
                    else:
                        if T > 1e-12:
                            prob = math.exp(-(new_val - cur_val) / T)
                            if rng.random() < prob:
                                accept = True
                    if accept:
                        cur_val = new_val
                        if cur_val < best_val:
                            best_val = cur_val
                            best_seq = a.copy()
                    else:
                        # revert
                        a[i] += delta
                        a[j] -= delta

    # ------------------------------------------------------------------
    # 3️⃣  Fallback in the unlikely case nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)

    # ------------------------------------------------------------------
    # 4️⃣  Return a Python list, clipped to the allowed range
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
