    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)

        # ensure we have enough mass at i

"""
Improved step‑function search.

The algorithm proceeds in three phases:

0️⃣  If a previous construction is available (global variable
    ``height_sequence_1``) it is used as a warm‑start.

1️⃣  For each candidate length we optimise a *single‑parameter exponential*
    family a_i ∝ exp(‑γ i/n).  The optimal γ is found by a cheap
    golden‑section search on the true objective
    ``max(conv(a,a))``.  The best exponential‑shaped sequence over a
    handful of lengths is kept as a global best.

2️⃣  The global best is refined by a projected gradient descent on a
    smooth‑max surrogate (soft‑max of the convolution entries).  After
    the gradient stage a very cheap O(n) “mass‑swap” hill‑climb is run,
    which updates the convolution incrementally and accepts only moves
    that lower the true maximum.

All steps stop as soon as the overall time budget (``budget_s``) is
exhausted.  The final sequence is returned as a plain ``list[float]``
ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real input, speed‑optimal)
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return a length optimal for FFT (fallback to power‑of‑2)."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact objective value: max_{i} (seq*seq)[i] after normalisation."""
    return float(np.convolve(seq, seq).max())


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and enforce ℓ¹‑norm = √(2·n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / total)
    return seq


# ----------------------------------------------------------------------
# Exponential family optimisation (golden‑section on γ)
# ----------------------------------------------------------------------
def _exp_seq(n: int, gamma: float, target: float) -> np.ndarray:
    """Return the normalised exponential sequence for a given γ."""
    alpha = gamma / n
    a = np.exp(-alpha * np.arange(n, dtype=np.float64))
    a = a / a.sum() * target
    return _normalize(a)


def _optimize_exponential(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Find the best γ for a_i ∝ exp(-γ i/n) by golden‑section search."""
    target = math.sqrt(2.0 * n)
    # search interval for γ (empirically the optimum lies near ≈2)
    lo, hi = 0.5, 5.0
    phi = (np.sqrt(5) - 1) / 2  # ≈0.618
    best_seq = None
    best_score = float("inf")

    # initial evaluations (avoid the case lo=hi)
    while time.monotonic() < deadline and hi - lo > 1e-4:
        g1 = lo + (1 - phi) * (hi - lo)
        g2 = lo + phi * (hi - lo)

        seq1 = _exp_seq(n, g1, target)
        seq2 = _exp_seq(n, g2, target)

        sc1 = _fast_score(seq1)
        sc2 = _fast_score(seq2)

        if sc1 < best_score:
            best_score, best_seq = sc1, seq1
        if sc2 < best_score:
            best_score, best_seq = sc2, seq2

        # shrink the interval
        if sc1 < sc2:
            hi = g2
        else:
            lo = g1

    # final check at the centre of the last interval
    gmid = (lo + hi) / 2.0
    seq_mid = _exp_seq(n, gmid, target)
    sc_mid = _fast_score(seq_mid)
    if sc_mid < best_score:
        best_score, best_seq = sc_mid, seq_mid

    return best_seq, best_score

# Power‑law family optimisation (optional, same scheme as exponential)
# ----------------------------------------------------------------------
def _power_seq(n: int, p: float, target: float) -> np.ndarray:
    """Return the normalised power‑law sequence a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a = a / a.sum() * target
    return _normalize(a)


def _optimize_powerlaw(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Golden‑section search on the exponent p ∈ [0.1, 5.0]."""
    target = math.sqrt(2.0 * n)
    lo, hi = 0.1, 5.0
    phi = (np.sqrt(5) - 1) / 2
    best_seq = None
    best_score = float("inf")

    while time.monotonic() < deadline and hi - lo > 1e-4:
        p1 = lo + (1 - phi) * (hi - lo)
        p2 = lo + phi * (hi - lo)

        seq1 = _power_seq(n, p1, target)
        seq2 = _power_seq(n, p2, target)

        sc1 = _fast_score(seq1)
        sc2 = _fast_score(seq2)

        if sc1 < best_score:
            best_score, best_seq = sc1, seq1
        if sc2 < best_score:
            best_score, best_seq = sc2, seq2

        if sc1 < sc2:
            hi = p2
        else:
            lo = p1

    # centre evaluation
    pmid = (lo + hi) / 2.0
    seq_mid = _power_seq(n, pmid, target)
    sc_mid = _fast_score(seq_mid)
    if sc_mid < best_score:
        best_score, best_seq = sc_mid, seq_mid

    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (same idea as the reference code)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights approximating the max of the convolution."""
    cmax = conv.max()
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the smooth‑max surrogate (2·a ⋆ w)."""
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)        # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]        # the middle n entries
    return 2.0 * segment[::-1]               # reverse to obtain the proper order


def _smoothmax_optimize(init_seq: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the smooth‑max surrogate."""
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    best_local = a.copy()
    best_score = _fast_score(a)

    it = 0
    beta = 1.0
    base_step = (target / n) * 8.0      # relatively aggressive start

    while time.monotonic() < deadline and it < 6000:
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        step = base_step / (1.0 + it * 0.015)
        a = a - step * grad

        # projection onto the feasible set (non‑negative, ℓ¹‑norm)
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # occasional jitter to break symmetry
        if it % 250 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.025
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # anneal β (make the surrogate sharper)
        if it % 120 == 0 and beta < 300.0:
            beta *= 1.5

        it += 1

    final_score = _fast_score(best_local)
    return best_local, final_score


# ----------------------------------------------------------------------
# Very cheap O(n) random‑mass‑swap hill‑climb
# ----------------------------------------------------------------------
def _hillclimb(seq: np.ndarray,
               conv: np.ndarray,
               cur_max: float,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Local search that moves a small amount of mass between two positions."""
    n = seq.shape[0]
    cur_seq = seq.copy()
    cur_conv = conv.copy()
    best_seq = cur_seq.copy()
    best_val = cur_max

    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] <= 1e-12:
            continue

        # propose a transfer that is a few percent of the source mass
        delta = cur_seq[i] * rng.uniform(0.015, 0.045)

        # respect the upper bound 1000
        if cur_seq[j] + delta > 1000.0:
            delta = 1000.0 - cur_seq[j]
            if delta <= 0.0:
                continue

        # update convolution incrementally (O(n))
        new_conv = cur_conv.copy()
        new_conv[i:i + n] -= 2.0 * delta * cur_seq
        new_conv[j:j + n] += 2.0 * delta * cur_seq

        idx_2i = 2 * i
        idx_2j = 2 * j
        idx_ij = i + j
        new_conv[idx_2i] += delta * delta
        new_conv[idx_2j] += delta * delta
        new_conv[idx_ij] -= 2.0 * delta * delta

        new_max = new_conv.max()
        if new_max < cur_max - 1e-12:
            # accept
            cur_seq[i] -= delta
            cur_seq[j] += delta
            cur_conv = new_conv
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.

    The method combines:
      * a fine golden‑section search over exponential (and optional power‑law)
        families for several lengths,
      * a projected smooth‑max gradient descent from the best family member,
      * a final cheap O(n) random-mass-swap hill‑climb.

    The global best is returned as a plain list ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    def _update_global(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = _fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global best] n={len(seq)} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            cand = _normalize(cand)
            _update_global(cand)
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Parametric families (exponential + power‑law) over a range of n
    # ------------------------------------------------------------------
    lengths = [500, 800, 1000, 1200, 1500, 1800, 2000, 2500]
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # – exponential family –
        seq_exp, sc_exp = _optimize_exponential(n, deadline)
        if seq_exp is not None:
            _update_global(seq_exp)

        # – power‑law family (optional, cheap) –
        seq_pow, sc_pow = _optimize_powerlaw(n, deadline)
        if seq_pow is not None:
            _update_global(seq_pow)

    # ------------------------------------------------------------------
    # If nothing succeeded yet, fall back to a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # 2️⃣ Gradient‑descent refinement (run for the remaining budget)
    # ------------------------------------------------------------------
    gd_deadline = deadline - 0.3   # keep a short margin for the final hill‑climb
    if time.monotonic() < gd_deadline:
        seq_gd, sc_gd = _smoothmax_optimize(best_seq.copy(), gd_deadline, rng)
        _update_global(seq_gd)

    # ------------------------------------------------------------------
    # 3️⃣ Final cheap hill‑climb (uses the current global best)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        cur_conv = np.convolve(best_seq, best_seq)
        cur_max = float(cur_conv.max())
        seq_hc, sc_hc = _hillclimb(best_seq.copy(),
                                   cur_conv.copy(),
                                   cur_max,
                                   deadline,
                                   rng)
        _update_global(seq_hc)

    # ------------------------------------------------------------------
    # Return the best sequence discovered as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
