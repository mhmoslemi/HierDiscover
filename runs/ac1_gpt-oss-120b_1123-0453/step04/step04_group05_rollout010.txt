"""
search for a low‑value non‑negative step sequence.

Algorithm
---------
1️⃣  For a collection of lengths `n` we create several random start
    vectors on the ℓ¹‑simplex (sum = √(2·n)).
2️⃣  A fast *greedy local search* moves a small amount of mass from an
    index that contributes most to the current peak of the auto‑convolution
    to an index that contributes least.  The convolution is kept up‑to‑date
    analytically, so each iteration costs O(n) and many thousands of moves
    can be performed in a short time.
3️⃣  The best candidate found is polished with a smooth‑max L‑BFGS‑B
    optimisation (soft‑max parametrisation) and a short stochastic
    hill‑climber (Gaussian perturbations) – the same refinement used in the
    reference implementation.
4️⃣  The whole process respects a global time budget (≈ budget_s seconds)
    and finally returns the best sequence as a plain ``list[float]`` ready
    for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """Convert unrestricted vector z to a simplex‑point a with sum=target_sum."""
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z) * target_sum
    return a


def _obj_and_grad(z: np.ndarray, n: int, S: float, beta: float):
    """
    Smooth surrogate of the max‑convolution (log‑sum‑exp) and its gradient.
    Returns (smooth_obj, grad_z) where ``smooth_obj`` approximates the
    max value of the auto‑convolution of the softmax‑parametrised vector.
    """
    a = _softmax_to_a(z, S)                      # shape (n,)

    # auto‑convolution via FFT
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]  # length 2n‑1, real

    # smooth max (log‑sum‑exp)
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))          # stable
    sum_exps = np.sum(exps)
    smooth = max_c + np.log(sum_exps) / beta
    w = exps / sum_exps                            # soft‑max weights, ∑w = 1

    # gradient w.r.t. a : g_a = 2·(a_rev * w)_{n … n+n‑1}
    a_rev = a[::-1]
    L2 = _next_pow2(3 * n - 2)
    fr = np.fft.rfft(a_rev, L2)
    fw = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(fr * fw, L2)            # length L2
    start = n - 1
    g_raw = conv2[start : start + n]            # length n
    g_a = 2.0 * g_raw

    # chain rule (soft‑max) → gradient w.r.t. z
    total = np.dot(g_a, a)
    grad_z = a * g_a - a * (total / S)

    return smooth, grad_z.astype(np.float64)


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a*a (full discrete convolution) using a FFT."""
    n = a.shape[0]
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    Fa = np.fft.rfft(a, Lfft)
    conv = np.fft.irfft(Fa * Fa, Lfft)[:L]
    return conv


def _local_search(a_start: np.ndarray,
                  deadline: float,
                  max_iter: int = 20000,
                  step_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer search that monotonically decreases the peak of
    the auto‑convolution while keeping the ℓ¹‑norm fixed.
    Returns the best vector found and its peak value.
    """
    n = a_start.size
    L = 2 * n - 1
    a = a_start.copy()
    conv = _conv_self(a)
    best_a = a.copy()
    best_val = conv.max()
    rng = np.random.default_rng()
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        idx = int(np.argmax(conv))
        cur_max = conv[idx]

        # indices that actually contribute to conv[idx]
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        idx_range = np.arange(lo, hi + 1)
        partner = idx - idx_range
        contrib = a[idx_range] * a[partner]

        i_high = idx_range[np.argmax(contrib)]
        i_low = idx_range[np.argmin(contrib)]

        # safety – if the chosen donor is empty pick a random one
        if a[i_high] <= 1e-12:
            i_high = rng.integers(lo, hi + 1)
            i_low = rng.integers(lo, hi + 1)
            if i_high == i_low:
                continue

        delta_max = min(a[i_high], 1000.0 - a[i_low])
        if delta_max <= 0.0:
            continue

        delta = delta_max * step_factor * rng.random()
        if delta <= 0.0:
            continue

        # construct the linear change on the convolution vector
        diff = np.zeros(L, dtype=a.dtype)
        diff[i_low:i_low + n] = a
        diff[i_high:i_high + n] -= a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()

        # accept only non‑increasing moves
        if new_max <= cur_max * (1.0 + 1e-12):
            a[i_high] -= delta
            a[i_low] += delta
            conv = conv_new
            cur_max = new_max
            if cur_max < best_val - 1e-12:
                best_val = cur_max
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # try a smaller step
            delta2 = delta_max * step_factor * 0.1 * rng.random()
            if delta2 > 0.0:
                conv_new2 = conv + 2.0 * delta2 * diff
                if conv_new2.max() <= cur_max * (1.0 + 1e-12):
                    a[i_high] -= delta2
                    a[i_low] += delta2
                    conv = conv_new2
                    cur_max = conv.max()
                    no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1

        # occasional random jitter to escape flat regions
        if rng.random() < 0.001:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta_r = min(a[i], 1000.0 - a[j]) * 0.05 * rng.random()
                if delta_r > 0.0:
                    diff_r = np.zeros(L, dtype=a.dtype)
                    diff_r[j:j + n] = a
                    diff_r[i:i + n] -= a
                    conv = conv + 2.0 * delta_r * diff_r
                    a[i] -= delta_r
                    a[j] += delta_r
                    no_improve = 0

        # stop if no improvement for many steps
        if no_improve > 5000:
            break

    return best_a, best_val


def _stochastic_refinement(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """Short Gaussian hill‑climber that respects the ℓ¹‑sum."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = evaluate_sequence(best.tolist())
    step = step_initial
    S = best.sum()

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑project onto the simplex (soft‑max)
        cand = _softmax_to_a(np.log(np.clip(cand / S, 1e-12, None)), S)
        cand_score = evaluate_sequence(cand.tolist())

        if cand_score < best_score:
            best = cand
            best_score = cand_score

        step *= 0.9995
        if step < 1e-8:
            step = step_initial

    return best, best_score


def _refine_with_lbfgs(a_start: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Smooth‑max L‑BFGS‑B optimisation started from ``a_start``.
    Returns the best vector and its exact ``evaluate_sequence`` value.
    """
    n = a_start.size
    target_sum = math.sqrt(2.0 * n)

    # initialise soft‑max parameters from the supplied vector
    p = np.clip(a_start / target_sum, 1e-12, None)
    z0 = np.log(p)

    betas = [5.0, 15.0, 40.0, 80.0, 200.0, 500.0]
    best_a = a_start.copy()
    best_val = evaluate_sequence(best_a.tolist())

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        res = minimize(fun=lambda zz: _obj_and_grad(zz, n, target_sum, beta),
                       x0=z0,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 400,
                                'gtol': 1e-10,
                                'disp': False})
        z0 = res.x
        a = _softmax_to_a(z0, target_sum)
        a = np.clip(a, 0.0, 1000.0)
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[lbfgs] n={n} beta={beta:.0f} val={val:.6f}")

        if best_val < 1.503:
            break

    return best_a, best_val

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the supplied
    time budget (seconds).  The function returns a plain ``list[float]`` that
    can be fed directly to ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a tiny safety margin for the platform
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # optional warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    warm0 = globals().get("height_sequence_1")
    if isinstance(warm0, (list, tuple, np.ndarray)):
        w = np.asarray(warm0, dtype=np.float64)
        n0 = w.size
        S0 = math.sqrt(2.0 * n0)
        if w.sum() > 0:
            w = w * (S0 / w.sum())
        else:
            w = np.full(n0, S0 / n0, dtype=np.float64)
        w = np.clip(w, 0.0, 1000.0)
        val0 = evaluate_sequence(w.tolist())
        best_seq = w.copy()
        best_score = val0
        print(f"[init] warm‑start length={n0} value={val0:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  explore several lengths with fast greedy local search
    # ------------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000, 1200, 1500, 1800, 2100,
               2500, 3000]  # a diverse set; larger n gives more freedom
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        S_target = math.sqrt(2.0 * n)
        # decide how many restarts based on the length (smaller n → cheaper)
        restarts = 5 if n <= 1200 else (3 if n <= 2100 else 2)
        # allocate a rough per‑restart budget (at least 8 s)
        remaining = deadline - time.monotonic()
        per_restart = max(8.0, remaining / ((len(lengths) - lengths.index(n)) * restarts))

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ---------- initial vector ----------
            if warm0 is not None and isinstance(warm0, (list, tuple, np.ndarray)):
                # if the warm vector has the right size, reuse it
                if np.asarray(warm0).size == n:
                    a0 = np.asarray(warm0, dtype=np.float64)
                    a0 = a0 * (S_target / a0.sum())
                else:
                    # otherwise start from a deterministic sine shape (often a good seed)
                    theta = np.linspace(0, np.pi, n + 2)[1:-1]
                    a0 = np.sin(theta)
                    a0 = a0 * (S_target / a0.sum())
            else:
                # pure random Dirichlet start
                a0 = rng.exponential(scale=1.0, size=n)
                a0 = a0 * (S_target / a0.sum())

            # ---------- greedy local search ----------
            ls_deadline = min(deadline, time.monotonic() + per_restart)
            seq_opt, val_opt = _local_search(a0, ls_deadline)

            if val_opt < best_score:
                best_score = val_opt
                best_seq = seq_opt.copy()
                print(f"[local] n={n} restart={r} max_conv={val_opt:.6f}")

            if best_score < 1.503:
                print("[early stop] target reached during local search")
                break

        # ---------- optional L‑BFGS polishing ----------
        if best_seq is not None and best_seq.size == n and time.monotonic() < deadline:
            seq_ref, val_ref = _refine_with_lbfgs(best_seq, deadline, rng)
            if val_ref < best_score:
                best_score = val_ref
                best_seq = seq_ref.copy()
                print(f"[refine] n={n} value={val_ref:.6f}")

            if best_score < 1.503:
                print("[early stop] target reached after refinement")
                break

    # ------------------------------------------------------------------
    # 2️⃣  final stochastic polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_val = _stochastic_refinement(best_seq, deadline,
                                                      step_initial=0.2)
        if refined_val < best_score:
            best_seq = refined
            best_score = refined_val
            print(f"[final polish] value={refined_val:.6f}")

    # final safety clipping (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best value = {best_score:.6f}  (length={best_seq.size})")
    return [float(x) for x in best_seq]
