   import cvxpy as cp

   def solve_lp_for_g(f, M):
        n = len(f)
        g = cp.Variable(n, nonneg=True)
        # Build convolution constraints:
        # conv_k = sum_i f[i] * g[k-i] (with indices)
        # We can use Toeplitz: For each k (0 .. 2n-2), define a linear expression.
        # Represent as sum_{i= max(0, k-n+1)}^{min(k, n-1)} f[i] * g[k-i]
        constraints = []
        for k in range(2*n-1):
            # Determine indices i where both f[i] and g-index (k-i) are within bounds.
            i_start = max(0, k - (n-1))
            i_end = min(k, n-1)
            # sum_i f[i] * g[k-i]
            # Equivalent to sum_j f[j] * g[k-j]; we can implement using cp.sum(cp.multiply(...))
            # Let's accumulate using list comprehension: cp.sum(cp.multiply(f[i_start:i_end+1], g[k-i_start:k-i_end-1:-1]))

            # Let's compute weights: w = f[i_start:i_end+1], variables: g_sub = g[(k - i_end):(k - i_start+1)]? Actually let's map.
            # For simplicity compute linear expression using sum of (f[i] * g[k-i]) via cp.sum.
            expr = 0

def lp_step(f, M):
    n = len(f)
    T = np.zeros((2*n-1, n))
    for i in range(n):
        T[i:i+n, i] = f[i]
    g = cp.Variable(n)
    constraints = [g >= 0, T @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)

"""
Propose a low‑value non‑negative step sequence.

Algorithm overview
-------------------
1.  **Normalization** – every candidate is scaled to have ℓ¹‑norm
    √(2·n).  When this holds the evaluation function is simply the
    maximal entry of the auto‑convolution.

2.  **LP‑driven Frank‑Wolfe refinement** –  given a current sequence `f`
    (‖f‖₁ = √(2·n)) we compute `M = max(conv(f,f))`.  Solving the linear
    program

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k
                     g_i ≥ 0

    yields a direction `g` that concentrates mass where the entries of
    `f` are small.  After scaling `g` to the required ℓ¹‑norm we move
    partially towards `g` (step size chosen by a tiny line‑search).  The
    maximal auto‑convolution strictly decreases and the process is
    repeated until no improvement or the time budget expires.

3.  **Stochastic polishing** – a short isotropic Gaussian perturbation
    loop (always re‑normalising to the ℓ¹‑norm) finishes the optimisation.

The method works for moderate lengths (≈300–1500) and easily fits the
1000 s budget.  It improves on the previous gradient‑descent code by
using the global LP direction rather than a single sub‑gradient.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np
import cvxpy as cp


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _normalize(vec: np.ndarray, target: float) -> np.ndarray:
    """Scale ``vec`` so that its ℓ¹‑norm equals ``target``."""
    total = vec.sum()
    if total == 0.0:
        return np.full_like(vec, target / vec.size, dtype=np.float64)
    return vec * (target / total)


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``seq`` with itself using an FFT."""
    n = seq.size
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


def _max_conv(seq: np.ndarray) -> float:
    """Return max entry of the auto‑convolution of ``seq``."""
    return _conv_fft(seq).max()


def _toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the (2·n‑1)×n Toeplitz matrix ``T`` such that ``T @ g`` equals
    the full linear convolution ``f * g``.
    """
    n = f.size
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = f[i]
    return T


# ----------------------------------------------------------------------
# LP step (maximise ℓ¹ norm under convolution bound)
# ----------------------------------------------------------------------
def _lp_optimal_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   subject to   (f*g)_k ≤ M,  g_i ≥ 0.
    Returns the optimal vector ``g`` (may be far from the required ℓ¹ norm)
    or ``None`` if the solver fails.
    """
    n = f.size
    T = _toeplitz_matrix(f)                     # (2n‑1) × n
    g = cp.Variable(n, nonneg=True)

    constraints = [T @ g <= M]                  # element‑wise ≤ works in CVXPY
    problem = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        problem.solve(solver=cp.ECOS, warm_start=True, verbose=False,
                      max_iters=5000, abstol=1e-9, reltol=1e-9)
    except cp.error.SolverError:
        return None

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)


# ----------------------------------------------------------------------
# Frank‑Wolfe style refinement using the LP direction
# ----------------------------------------------------------------------
def _refine_via_lp(start_seq: np.ndarray,
                   deadline: float,
                   max_iters: int = 20) -> tuple[np.ndarray, float]:
    """
    Iteratively improve ``start_seq`` (already ℓ¹‑normalised) by
    repeatedly solving the LP above and moving towards the solution.
    Returns the best sequence found and its maximal auto‑convolution.
    """
    n = start_seq.size
    target = math.sqrt(2.0 * n)

    seq = start_seq.copy()
    best_seq = seq.copy()
    best_val = _max_conv(seq)

    # a modest initial step size; it will be adapted on the fly
    step = 0.4
    while time.monotonic() < deadline and max_iters > 0:
        max_iters -= 1
        M = _max_conv(seq)
        g0 = _lp_optimal_g(seq, M)
        if g0 is None:
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # scale to the required ℓ¹‑norm
        g = g0 * (target / sum_g0)

        # line‑search on the convex combination (1‑α)·seq + α·g
        improved = False
        trial_step = step
        for _ in range(6):                     # at most six back‑tracks
            if time.monotonic() >= deadline:
                break
            cand = (1.0 - trial_step) * seq + trial_step * g
            cand = _normalize(cand, target)
            cand_val = _max_conv(cand)

            if cand_val + 1e-9 < best_val:      # strict improvement
                seq = cand
                best_seq = cand.copy()
                best_val = cand_val
                improved = True
                step = min(trial_step * 1.2, 1.0)   # enlarge for next iteration
                break
            else:
                trial_step *= 0.5               # shrink and retry

        if not improved:
            # no feasible improvement → stop the loop
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final stochastic polishing (Gaussian hill‑climb)
# ----------------------------------------------------------------------
def _gaussian_polish(start_seq: np.ndarray,
                     deadline: float,
                     init_step: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Small isotropic Gaussian perturbation loop that always re‑normalises
    to the required ℓ¹‑norm.
    """
    n = start_seq.size
    target = math.sqrt(2.0 * n)
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_val = _max_conv(best)

    step = init_step
    while time.monotonic() < deadline:
        perturb = rng.normal(scale=step, size=n)
        cand = best + perturb
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target)
        cand_val = _max_conv(cand)

        if cand_val < best_val:
            best, best_val = cand, cand_val
            step *= 1.03                     # slowly enlarge after success
        else:
            step *= 0.99                     # otherwise decay

        if step < 1e-7:
            step = init_step

    return best, best_val

# ----------------------------------------------------------------------

    incumbent = globals().get("height_sequence_1")

                inc = _normalize(inc, target0)
                sc0 = _max_conv(inc)
                best_seq, best_score = inc.copy(), sc0
                print(f"[init] height_sequence_1 n={inc.size} score={sc0:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Scan a handful of lengths, initialise, LP‑refine
    # ------------------------------------------------------------------
    # Lengths are chosen to give a good trade‑off between flexibility and speed

        # Try two different starts for each length:

        for start_type in (0, 1):
            if time.monotonic() >= deadline:

            if start_type == 0:
                alpha = 2.0 / n
                a = np.exp(-alpha * np.arange(n, dtype=np.float64))
                a = _normalize(a, target)

            else:

                a = _normalize(a, target)

            if sc0 < best_score:

                print(f"[start-{label}] n={n} new best {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:

        target = math.sqrt(2.0 * n)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    # The harness expects a plain Python list of floats
