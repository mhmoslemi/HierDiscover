def build_constraint_matrix(f, active_shifts):
    n = len(f)
    m = len(active_shifts)
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_shifts):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # j = k - i
        # For each i in range, we have coefficient f[i] for variable g[j]
        # Equivalent: for j in [i_start..i_end]?
        # Use i = i_start..i_end ; j = k - i
        for i in range(i_start, i_end + 1):
            j = k - i
            # coefficient f[i] for variable g[j]
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))
    return A

"""
Propose a step‑function sequence with a low peak of the self‑convolution.
The algorithm runs three phases within the time budget:

1️⃣ Warm‑start from a previously discovered sequence if it exists.
2️⃣ Iterative linear‑programming “tight‑envelope” descent:
   – For the current vector f (∑f=S) compute M = max(f*f).
   – Solve the LP   maximise 1ᵀg   subject to   f*g ≤ M , g ≥ 0 .
   – Rescale g to the correct ℓ¹‑norm and mix it with f.
   – Repeat while the bound improves.
3️⃣ Polish the best candidate with a short Adam optimisation and a
   temperature‑controlled simulated‑annealing hill‑climber.

All vectors are always projected onto the simplex {x≥0, Σx=S} to keep the
normalisation exact.  The routine stops early if the target bound
(≈1.503) is reached.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(f: np.ndarray) -> float:
    """Exact objective 2·n·max(f*f)/(Σf)² (scale‑invariant)."""
    n = f.shape[0]
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _active_shifts(conv: np.ndarray, M: float, tol: float = 1e-8) -> np.ndarray:
    """Indices where the convolution is (almost) maximal."""
    eps = max(tol * M, 1e-12)
    return np.where(conv >= M - eps)[0]


def _build_constraint_matrix(f: np.ndarray, shifts: np.ndarray) -> sparse.csr_matrix:
    """
    Build a sparse matrix A where    (A @ g)[k] = (f * g)[shifts[k]].
    Only the rows for the supplied shifts are created.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(shifts):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])
    return sparse.csr_matrix((data, (rows, cols)), shape=(len(shifts), n))


# ----------------------------------------------------------------------
# Adam optimiser on the ℓ¹‑simplex (used for polishing)
# ----------------------------------------------------------------------


def _adam_descent(f0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the peak of the self‑convolution."""
    f = f0.copy()
    best_f = f.copy()
    best_sc = _score(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    no_imp = 0

    it = 0
    while it < max_iters and time.monotonic() < deadline:
        it += 1
        conv = np.convolve(f, f)
        M = np.max(conv)
        act = _active_shifts(conv, M, tol=1e-9)

        # sub‑gradient of the maximal convolution entry
        grad_max = np.zeros_like(f)
        n = len(f)
        for k in act:
            i0 = max(0, k - (n - 1))
            i1 = min(k, n - 1)
            i = np.arange(i0, i1 + 1, dtype=int)
            j = k - i
            grad_max[i] += f[j]
            grad_max[j] += f[i]
            mask = (i == j)
            grad_max[i[mask]] -= f[i[mask]]

        # full gradient of the objective (2·n·M / S²) w.r.t. f
        S = np.sum(f)
        grad = (2.0 * n / (S * S)) * grad_max

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_cand = f - step
        f_cand = _project_onto_simplex(f_cand, target_sum)
        f_cand = np.clip(f_cand, 0.0, 1000.0)

        cand_sc = _score(f_cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_f = f_cand.copy()
            no_imp = 0
            print(f"[Adam] it={it:4d} n={n} score={cand_sc:.6f}")
        else:
            no_imp += 1

        if cand_sc < _score(f):
            f = f_cand
        else:
            if no_imp > 250:
                lr *= 0.7
                no_imp = 0

        if no_imp > 800:
            break

    return best_f, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (used for polishing)
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.25,
                         decay: float = 0.9996) -> tuple[np.ndarray, float]:
    """Mass‑transfer neighbourhood search keeping the ℓ¹‑norm fixed."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_sc = _score(cur)

    best = cur.copy()
    best_sc = cur_sc

    n = len(start)
    S = np.sum(start)

    T = init_temp
    step_scale = 0.05 * S / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        if it % 256 == 0:
            cand = _project_onto_simplex(cand, S)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                print(f"[Anneal] it={it:6d} score={cand_sc:.6f}")
        else:
            prob = math.exp(-(cand_sc - cur_sc) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        T *= decay
        if T < 1e-9:
            T = init_temp
        step_scale *= 0.9999
        if step_scale < 1e-7 * S:
            step_scale = 0.05 * S / n

    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises
    evaluate_sequence(...).  The method combines an LP‑based descent,
    a few Adam steps and a finishing simulated‑annealing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start from a known construction, if present.
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64).copy()
        n0 = len(seq)
        S0 = math.sqrt(2.0 * n0)
        seq = np.clip(seq, 0.0, 1000.0)
        seq = _project_onto_simplex(seq, S0)
        sc = _score(seq)
        best_seq = seq.copy()
        best_score = sc
        print(f"[init] warm‑start n={n0} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ LP‑descent over several lengths.
    # --------------------------------------------------------------
    lengths = [500, 600, 800, 1000, 1200]          # try a few sizes
    target = 1.503                                   # early‑exit target

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        S = math.sqrt(2.0 * n)

        # initialise f
        if best_seq is not None and len(best_seq) == n:
            f = best_seq.copy()
            # tiny random jitter to avoid exact stagnation
            f += rng.normal(scale=0.001 * S, size=n)
            f = np.clip(f, 0.0, None)
            f = _project_onto_simplex(f, S)
        else:
            f = rng.random(n)
            f = _project_onto_simplex(f, S)

        # ----------------------------------------------------------
        # LP descent loop
        # ----------------------------------------------------------
        lp_iters = 0
        while time.monotonic() < deadline and lp_iters < 30:
            lp_iters += 1
            conv = np.convolve(f, f)
            M = np.max(conv)

            # keep the best ever observed
            if M < best_score:
                best_score = M
                best_seq = f.copy()
                print(f"[LP] n={n} iter={lp_iters} M={M:.6f}")
                if best_score < target:
                    print("[early] target reached during LP phase")
                    return [float(x) for x in best_seq]

            # define the active constraints (those attaining the maximum)
            act = _active_shifts(conv, M, tol=1e-7)

            # construct constraint matrix A (f * g ≤ M)
            A = _build_constraint_matrix(f, act)
            b = np.full(A.shape[0], M)

            # linear program: maximise 1ᵀg  → minimise -1ᵀg
            c = -np.ones(n)
            lp_res = linprog(c, A_ub=A, b_ub=b,
                             bounds=[(0, None)] * n,
                             method='highs',
                             options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})

            if not lp_res.success:
                # fallback to a uniform distribution if LP fails
                g = np.full(n, S / n)
            else:
                g = lp_res.x

            sum_g = np.sum(g)
            if sum_g <= 1e-12:
                g = np.full(n, S / n)
                sum_g = S

            # rescale to the correct ℓ¹‑norm
            g = g * (S / sum_g)

            # mix with the current iterate
            t = 0.5
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0.0, 1000.0)
            f_new = _project_onto_simplex(f_new, S)
            f = f_new

        # ----------------------------------------------------------
        # 3️⃣ Short Adam polishing (if time permits)
        # ----------------------------------------------------------
        if time.monotonic() < deadline:
            sub_deadline = min(deadline, time.monotonic() + 6.0)
            f_opt, sc_opt = _adam_descent(f, S, sub_deadline, max_iters=1500)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = f_opt.copy()
                print(f"[Adam] n={n} post‑opt score={sc_opt:.6f}")
                if best_score < target:
                    print("[early] target reached after Adam")
                    return [float(x) for x in best_seq]

    # --------------------------------------------------------------
    # 4️⃣ Final simulated annealing polishing (if time allows)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_sc = _simulated_annealing(best_seq, deadline,
                                                  init_temp=0.25,
                                                  decay=0.9997)
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[final] annealed score={refined_sc:.6f}")

    # --------------------------------------------------------------
    # 5️⃣ Fallback (uniform) – should never be needed.
    # --------------------------------------------------------------
    if best_seq is None:
        n = 500
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = _score(best_seq)
        print("[fallback] using uniform construction")

    # final safety clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best_score={best_score:.6f} length={len(best_seq)}")
    return [float(x) for x in best_seq]
