"""
search for a low‑value step sequence (minimises evaluate_sequence)

The algorithm consists of three layers:

1️⃣  Warm‑start – either from the previously discovered
    ``height_sequence_1`` or from a uniform distribution.

2️⃣  *Active‑set LP iteration* –  the exact step described in the
    problem statement.  For a fixed sequence `f` we solve a linear
    program

        maximise   Σ b_j
        subject to b_j ≥ 0
                   (f * b)_k ≤ M   for all shifts k,

    where `M = max_k (f*f)_k`.  The solution `b` is scaled to the
    required ℓ¹‑norm and mixed with `f` (small convex step).  This
    provably never increases the maximal auto‑convolution value, and
    typically drives it down rapidly.

3️⃣  A short Adam + simulated‑annealing polish (the same code that
    worked before) to fine‑tune the final candidate.

The whole routine respects the global time budget (default 1000 s) and
returns the best list of floats found before the deadline.
"""

import time
import math
import numpy as np

# optionally import the LP solver – SciPy is available in the execution
# environment.  If it cannot be imported we silently fall back to the
# previous Adam‑annealing pipeline.
try:
    from scipy.optimize import linprog
    _HAVE_LP = True
except Exception:  # pragma: no cover
    _HAVE_LP = False


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s }.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_sequence(seq: np.ndarray) -> float:
    """
    For a sequence whose ℓ¹‑norm is √(2·n) the objective equals the
    maximal entry of its auto‑convolution.
    """
    conv = np.convolve(seq, seq)
    return float(np.max(conv))


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that
        (A @ b)[k] = (f * b)[k]
    for any non‑negative vector b.
    """
    n = f.size
    rows = np.arange(n)[:, None] + np.arange(n)  # shape (n, n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # each column j receives the vector f shifted by j positions
    A[rows.ravel(), np.arange(n).repeat(n)] = np.tile(f, n)
    return A


# ----------------------------------------------------------------------
# Layer 2 – active‑set LP descent
# ----------------------------------------------------------------------


def _lp_descent(f0: np.ndarray,
                target_sum: float,
                deadline: float,
                max_outer_iters: int = 30,
                inner_step: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based active‑set iteration.
    Returns (best_sequence, best_score).
    """
    n = f0.size
    f = f0.copy()
    best_f = f.copy()
    best_score = _score_from_sequence(f)

    step = inner_step

    for outer in range(max_outer_iters):
        if time.monotonic() >= deadline:
            break

        # current maximal auto‑convolution value
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # construct the linear constraints A @ b ≤ M
        A = _build_convolution_matrix(f)

        # linear programme: maximise Σ b_j   ⇔ minimise −Σ b_j
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # solve
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=(0, None),
                      method='highs',
                      options={'disp': False})

        if not res.success:
            # LP failed – give up this run
            break

        b = res.x
        sum_b = float(b.sum())
        if sum_b <= 0.0:
            break

        # scale to the required ℓ¹‑norm
        g = b * (target_sum / sum_b)

        # convex combination (step size may shrink if we stall)
        f_new = (1.0 - step) * f + step * g

        # evaluate
        new_score = _score_from_sequence(f_new)

        # keep the overall best
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            print(f"[LP] n={n} outer={outer:2d} step={step:.3f} "
                  f"score={new_score:.6f}")

        # if the new max did not improve, shrink the step
        if new_score >= best_score:
            step *= 0.7
            if step < 1e-4:
                # step became tiny → stop early
                break
        else:
            # otherwise keep a healthy step size
            step = max(step, inner_step)

        f = f_new

        # convergence test – stop when improvement is negligible
        if abs(M - new_score) < 1e-7:
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Layer 3 – Adam + simulated annealing (same as the reference code)
# ----------------------------------------------------------------------


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    eps = max_val * 1e-12 + 1e-14
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    n = f.size
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        grad[i] += f[j]
        grad[j] += f[i]
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= f[i[mask]]
    return grad


def _adam_descent(f0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises the peak auto‑convolution.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _score_from_sequence(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05
    no_improve = 0

    it = 0
    while it < max_iters and time.monotonic() < deadline:
        it += 1

        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)

        grad_max = _gradient_submax(f, active)

        sum_f = np.sum(f)
        grad_obj = (2.0 * len(f) / (sum_f * sum_f)) * grad_max

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = _project_onto_simplex(f_candidate, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cand_score = _score_from_sequence(f_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_f = f_candidate.copy()
            no_improve = 0
            print(f"[Adam] it={it:4d} n={len(f)} score={cand_score:.6f}")
        else:
            no_improve += 1

        if cand_score < _score_from_sequence(f):
            f = f_candidate
        else:
            if no_improve > 200:
                lr *= 0.7
                no_improve = 0

        if no_improve > 800:
            break

    return best_f, best_score


def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9997) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score_from_sequence(cur)

    best = cur.copy()
    best_score = cur_score

    n = start.size
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.05 * target_sum / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score_from_sequence(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-8:
            T = init_temp

        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.
    The routine respects the given time budget and returns a plain
    Python list suitable for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")
    TARGET = 1.503  # known target to beat

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64).copy()
        seq = np.clip(seq, 0.0, 1000.0)
        n = seq.size
        target_sum = math.sqrt(2.0 * n)
        if np.sum(seq) == 0.0:
            seq = np.full(n, target_sum / n)
        else:
            seq = seq * (target_sum / np.sum(seq))
        best_seq = seq
        best_score = _score_from_sequence(seq)
        print(f"[init] warm‑start from height_sequence_1, n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ LP‑driven optimisation over several lengths & restarts
    # --------------------------------------------------------------
    lengths = [400, 500, 600, 800, 1000, 1200, 1500, 1800]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of random restarts per length
        restarts = 8 if n > 1000 else 12

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a candidate
            if best_seq is not None and best_seq.size == n:
                f0 = best_seq.copy()
                f0 += rng.normal(scale=0.001 * target_sum, size=n)
                f0 = np.clip(f0, 0.0, None)
                f0 = _project_onto_simplex(f0, target_sum)
            else:
                f0 = rng.random(n)
                f0 = _project_onto_simplex(f0, target_sum)

            # --------------------------------------------------
            # LP descent (if we have SciPy)
            # --------------------------------------------------
            if _HAVE_LP:
                sub_deadline = min(deadline, time.monotonic() + 8.0)
                f_lp, score_lp = _lp_descent(f0,
                                             target_sum,
                                             sub_deadline,
                                             max_outer_iters=25,
                                             inner_step=0.25)
                if score_lp < best_score:
                    best_score = score_lp
                    best_seq = f_lp.copy()
                    print(f"[update] LP n={n} restart={r} score={score_lp:.6f}")

                if best_score <= TARGET:
                    print("[early] target reached during LP phase")
                    return [float(x) for x in best_seq]

            # --------------------------------------------------
            # Adam refinement (fallback / extra polishing)
            # --------------------------------------------------
            sub_deadline = min(deadline, time.monotonic() + 6.0)
            f_adam, score_adam = _adam_descent(f0,
                                               target_sum,
                                               sub_deadline,
                                               max_iters=3000)
            if score_adam < best_score:
                best_score = score_adam
                best_seq = f_adam.copy()
                print(f"[update] Adam n={n} restart={r} score={score_adam:.6f}")

            if best_score <= TARGET:
                print("[early] target reached during Adam phase")
                return [float(x) for x in best_seq]

    # --------------------------------------------------------------
    # 3️⃣ Final polishing by simulated annealing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _simulated_annealing(best_seq,
                                                     deadline,
                                                     init_temp=0.2,
                                                     decay=0.9997)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final] annealed score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Fallback – uniform construction
    # --------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score_from_sequence(best_seq)
        print("[fallback] using uniform construction")

    # final safety clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
