def proj_simplex(v, target):
    # v is np.ndarray
    # ensure non-negative after projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

That's the classic Euclidean projection onto simplex.

"""
Propose a low‑value step sequence.

Algorithm
---------

1️⃣  *Family sweep* – For a small set of lengths we generate many hand‑crafted
    families (exponential, power‑law, common windows, random Dirichlet)
    and keep the best one as the current incumbent.

2️⃣  *Smooth‑max Adam optimisation* –  Instead of moving only against the
    single active convolution index we minimise a smooth approximation of the
    max‑convolution:
        f(a) = (1/β)·log Σ_i exp(β·(a*a)_i) .
    Its gradient can be written in closed form using two convolutions, so each
    iteration costs only a few FFTs.  We run an Adam optimiser with a
    projection onto the simplex {a≥0, Σa = √(2n)} to respect the security
    checks.  The temperature β is increased during the run so that the
    optimisation steadily focuses on the true maximum.

3️⃣  *Final polishing* – A short stochastic Gaussian hill‑climb (still
    projected onto the simplex) finishes the search.

The routine respects the 1000 s budget, prints intermediate scores and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Utility: projection onto the probability simplex (scaled to target)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target}."""
    v = np.asarray(v, dtype=np.float64)
    if target <= 0:
        return np.zeros_like(v)
    # standard O(n log n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are (almost) zero – return uniform distribution
        return np.full_like(v, target / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Fast convolutions (FFT based)
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of `seq` with itself (length 2·n‑1)."""
    n = seq.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors."""
    n = a.shape[0]
    m = b.shape[0]
    N = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# 3.  Exact objective (max of the auto‑convolution)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a correctly normalised seq."""
    n = seq.shape[0]
    max_b = _conv_fft(seq).max()
    # With Σa = √(2n) the expression simplifies to max_b.
    return float(max_b)


# ----------------------------------------------------------------------
# 4.  Gradient of the smooth‑max (soft‑max) of the auto‑convolution
# ----------------------------------------------------------------------
def _grad_softmax_max(seq: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/β)·log Σ exp(β·(seq*seq)_i)."""
    n = seq.shape[0]

    # 1) compute the auto‑convolution
    conv = _conv_fft(seq)                     # length 2n‑1

    # 2) soft‑max probabilities (stable)
    max_c = conv.max()
    exp_shift = np.exp(beta * (conv - max_c))
    p = exp_shift / exp_shift.sum()           # length 2n‑1, sums to 1

    # 3) two auxiliary convolutions needed for the gradient
    #    g2_j = Σ_i p_i * a_{j-i}  = (p * a)[j]   (first n entries)
    conv_pa = _conv_full_fft(p, seq)          # length 3n‑2
    g2 = conv_pa[:n]

    #    g1_j = Σ_i p_i * a_{i-j} = (p * a_rev)[j+n‑1]   (central n entries)
    a_rev = seq[::-1]
    conv_pa_rev = _conv_full_fft(p, a_rev)    # length 3n‑2
    g1 = conv_pa_rev[n - 1 : n - 1 + n]

    # gradient = g1 + g2
    grad = g1 + g2
    return grad


# ----------------------------------------------------------------------
# 5.  Adam optimiser on the smooth‑max objective
# ----------------------------------------------------------------------
def _adam_optimize(init_seq: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   start_beta: float = 5.0,
                   max_beta: float = 200.0,
                   beta_inc: float = 1.2) -> tuple[np.ndarray, float]:
    """Projected Adam optimisation of the smooth‑max objective."""
    a = init_seq.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5                      # base learning rate
    beta = start_beta
    t = 0

    best_seq = a.copy()
    best_sc = _fast_score(a)

    while time.monotonic() < deadline:
        t += 1
        # gradient of the smooth‑max
        grad = _grad_softmax_max(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected gradient step
        a_new = a - step
        a_new = _proj_simplex(a_new, target_sum)

        # evaluate
        cur_sc = _fast_score(a_new)

        # keep the best ever seen
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = a_new.copy()
            # encourage larger steps after a success
            lr = min(lr * 1.05, 5.0)
        else:
            # be a bit more cautious after a non‑improvement
            lr = max(lr * 0.90, 0.01)

        a = a_new

        # slowly raise the temperature β
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * beta_inc, max_beta)

        # early exit if we already beat the target bound
        if best_sc < 1.503:
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
# 6.  Tiny stochastic hill‑climb (final polish)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """Gaussian perturbation + projection, cheap final polishing."""
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial
    n = len(best)

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = _proj_simplex(cand, best.sum())
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        step *= 0.9997
        if step < 1e-7:
            step = step_initial
    return best, best_sc


# ----------------------------------------------------------------------
# 7.  Parametric families (quick exhaustive sweep)
# ----------------------------------------------------------------------
def _candidate_families(n: int, target_sum: float) -> tuple[np.ndarray, float]:
    """Return the best vector among several hand‑crafted families."""
    best_vec = None
    best_sc = float("inf")

    def _norm(v):
        v = np.asarray(v, dtype=np.float64)
        v = v / v.sum() * target_sum
        return v

    # uniform baseline
    a = np.full(n, target_sum / n, dtype=np.float64)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    idx = np.arange(n, dtype=np.float64)

    # ------------------ exponential decay ------------------------------
    alphas = np.concatenate([
        np.arange(1, 6) / (10.0 * n),
        np.arange(1, 21) / n,
        np.linspace(0.02, 0.20, 40),
        np.linspace(0.20, 1.00, 30)
    ])
    alphas = np.unique(alphas)
    for α in alphas:
        a = np.exp(-α * idx)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # ------------------ power‑law ------------------------------------
    p_vals = np.linspace(0.05, 5.0, 120)
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        a = base ** (-p)
        a = _norm(a)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # ------------------ classic windows -------------------------------
    # Hann
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Hamming
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # Blackman
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1)) \
        + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # sine window
    a = np.sin(np.pi * (idx + 1) / (n + 1))
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()
    # triangular
    a = np.minimum(idx + 1, n - idx)
    a = _norm(a)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


# ----------------------------------------------------------------------
# 8.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for final clean‑up steps
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Optional global initialisation (previous best if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = _proj_simplex(init, target0)
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] previous n={n0} score={sc0:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Sweep over several lengths, keep the best family candidate
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500, 1800, 2000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ----- exhaustive parametric families -----
        cand_vec, cand_sc = _candidate_families(n, target)
        if cand_sc < best_score:
            best_seq, best_score = cand_vec.copy(), cand_sc
            print(f"[family] n={n} new best {best_score:.6f}")

        # ----- cheap random Dirichlet samples (time‑boxed) -----
        rnd_deadline = time.monotonic() + min(0.4,
                                             (deadline - time.monotonic()) * 0.2)
        while time.monotonic() < rnd_deadline:
            a = rng.random(n)
            a = _proj_simplex(a, target)
            sc = _fast_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[rand] n={n} improved to {best_score:.6f}")

        # ----- local Adam optimisation for the *current* best of this n -----
        # (only if we still have time and the incumbent has the same length)
        if best_seq is not None and best_seq.shape[0] == n:
            sub_deadline = deadline - 0.05   # leave a little head‑room
            if time.monotonic() < sub_deadline:
                opt_seq, opt_sc = _adam_optimize(best_seq,
                                                 target,
                                                 sub_deadline)
                if opt_sc < best_score:
                    best_seq, best_score = opt_seq.copy(), opt_sc
                    print(f"[adam] n={n} improved to {best_score:.6f}")

        # early exit if we have already crossed the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Final polishing (short stochastic hill‑climb)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_sc = _local_search(best_seq,
                                                  deadline,
                                                  step_initial=0.2)
        if polished_sc < best_score:
            best_seq, best_score = polished_seq, polished_sc
            print(f"[final] polished to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    return [float(v) for v in best_seq]
```

This implementation replaces the simple sub‑gradient descent of the previous approach by a *smooth‑max* Adam optimisation that simultaneously reduces all near‑max convolution entries, uses a proper Euclidean projection onto the required simplex, and adds a short stochastic polishing phase. It searches several lengths (up to 2000) and explores a variety of parametric families; the most promising candidate for each length is refined with the Adam optimiser. The algorithm stays well within the 1000 s budget, prints progress, and finally returns the best sequence it discovered.
