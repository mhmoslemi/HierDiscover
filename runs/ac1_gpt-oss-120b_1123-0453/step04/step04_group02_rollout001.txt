"""
Propose a low‑value step sequence.

The algorithm works in three stages:

1️⃣  **Broad sweep of parametric families** – for several lengths we
    generate many simple families (exponential, power‑law, Hann,
    Hamming, Blackman, sine, triangular) and keep the best one.
    A few cheap Dirichlet samples are also tried.

2️⃣  **Improved sub‑gradient descent** – starting from the incumbent we
    repeatedly locate the *active* entries of the auto‑convolution
    (the largest ones, a small top‑K set).  For each active index we
    compute the exact gradient of that convolution entry
    (including both a_i and a_{k‑i} contributions).  The average of
    those gradients is used as a descent direction; the step size is
    adapted automatically and occasional random “shakes’’ are added
    when progress stalls.

3️⃣  **Final stochastic polishing** – a short Gaussian hill‑climb
    (random perturbations + renormalisation) finishes the optimisation.

All operations respect the security checks of ``evaluate_sequence``:
non‑negative entries, clipping to [0,1000] and exact ℓ¹‑norm

# ----------------------------------------------------------------------
# Fast full linear convolution using FFT (real‑to‑real)
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of seq with itself using FFT.
    Returns a 1‑D array of length 2·len(seq)‑1.
    """
    n = seq.shape[0]
    # smallest power of two ≥ 2·n‑1
    N = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]

# Exact objective (without the security wrapper)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation value for a sequence already satisfying the sum‑norm."""
    n = seq.shape[0]
    max_b = _conv_fft(seq).max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    # Because we normally keep sum_a = sqrt(2·n), the factor 2·n / sum_a² = 1.
    # Keeping the full expression makes the function robust for any scaling.
    return 2.0 * n * max_b / (sum_a ** 2)

def _candidate_families(n: int, target_sum: float):

    best_vec = None
    best_sc = float("inf")

        v = v / v.sum() * target_sum
        return v

    idx = np.arange(n, dtype=np.float64)

    # ---- Uniform (baseline) -----------------------------------------
    a = np.full(n, target_sum / n, dtype=np.float64)
    sc = _fast_score(a)
    if sc < best_sc:

        np.linspace(0.02, 0.20, 40),          # coarser up to 0.2

    alphas = np.unique(alphas)
    for α in alphas:
        a = np.exp(-α * idx)
        a = _normalise(a)
        sc = _fast_score(a)
        if sc < best_sc:

# ----------------------------------------------------------------------
# Sub‑gradient for the (smoothed) maximum of the auto‑convolution
# ----------------------------------------------------------------------
def _active_subgradient(seq: np.ndarray, conv: np.ndarray, top_k: int = 3) -> np.ndarray:
    """
    Compute the average gradient of the top‑k convolution entries.
    The exact gradient of conv[k] w.r.t. a_i is a_{k-i} + a_i (where defined).
    """
    n = seq.shape[0]
    # select top‑k indices (unsorted, cheap)
    if top_k >= conv.shape[0]:
        idxs = np.arange(conv.shape[0])
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(seq)
    for k in idxs:
        if k < n:
            rev = seq[k::-1]               # a_{k‑i}
            grad[: k + 1] += rev + seq[: k + 1]
        else:
            off = k - (n - 1)               # first overlapping index
            length = n - off
            rev = seq[:length][::-1]
            grad[off:] += rev + seq[off:]
    grad /= len(idxs)
    return grad

# Sub‑gradient descent with adaptive step and occasional shaking
# ----------------------------------------------------------------------
def _gradient_descent(seq: np.ndarray, target_sum: float, deadline: float,
                     top_k: int = 3) -> tuple[np.ndarray, float]:
    """
    Refine a sequence by (averaged) sub‑gradient descent on the maximal
    auto‑convolution entry.  Returns the best sequence found and its score.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    step = max(target_sum * 0.001, 0.02)   # modest initial step
    rng = np.random.default_rng()
    no_improve = 0

    while time.monotonic() < deadline:
        conv = _conv_fft(seq)
        grad = _active_subgradient(seq, conv, top_k=top_k)

        # move opposite to the (averaged) gradient
        cand = seq - step * grad
        cand = np.clip(cand, 0.0, 1000.0)

        total = cand.sum()
        if total == 0.0:
            cand = np.full(n, target_sum / n, dtype=np.float64)
        else:
            cand = cand * (target_sum / total)

        cand_sc = _fast_score(cand)

        if cand_sc < best_sc:
            best_seq, best_sc = cand.copy(), cand_sc
            seq = cand
            step = min(step * 1.12, target_sum * 0.5)   # modest growth
            no_improve = 0
        else:
            step *= 0.55
            no_improve += 1

        # occasional random shake‑up if stuck
        if no_improve >= 30:
            i, j = rng.integers(n, size=2)
            delta = seq[i] * 0.07
            if delta > 0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = np.clip(cand, 0.0, 1000.0)
                cand = cand * (target_sum / cand.sum())
                cand_sc = _fast_score(cand)
                if cand_sc < best_sc:
                    seq = cand
                    best_seq, best_sc = cand.copy(), cand_sc
            step = max(step, target_sum * 0.001)
            no_improve = 0

        # early termination if we already beat the target bound
        if best_sc < 1.503:
            break

    return best_seq, best_sc

# Short stochastic polishing (Gaussian hill‑climb)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray, deadline: float,
                  step_initial: float = 0.2) -> tuple[np.ndarray, float]:
    """Stochastic refinement with isotropic Gaussian perturbations."""
    n = len(start_seq)
    target = math.sqrt(2.0 * n)
    best = start_seq.copy()
    best_sc = _fast_score(best)
    rng = np.random.default_rng()
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target / cand.sum())
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        # slowly decay the exploration radius
        step *= 0.9995
        if step < 1e-7:
            step = step_initial

    return best, best_sc

        # ----- parametric families -------------------------------------------------
        cand_vec, cand_sc = _candidate_families(n, target)
        if cand_sc < best_score:
            best_seq, best_score = cand_vec.copy(), cand_sc
            print(f"[family] n={n} new best {best_score:.6f}")

        # ----- cheap random Dirichlet samples (time‑boxed) -------------------------
        rnd_deadline = time.monotonic() + min(0.6,
                                             (deadline - time.monotonic()) / 2)
        while time.monotonic() < rnd_deadline:
            a = rng.random(n)
            a = a / a.sum() * target
            sc = _fast_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[rand]  n={n} improved {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Gradient‑descent refinement on the incumbent
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        refined_seq, refined_sc = _gradient_descent(best_seq, target,
                                                    deadline, top_k=4)
        if refined_sc < best_score:
            best_seq, best_score = refined_seq, refined_sc
            print(f"[grad]  refined to {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final stochastic polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        polished_seq, polished_sc = _local_search(best_seq, deadline,
                                                  step_initial=0.15)
        if polished_sc < best_score:
            best_seq, best_score = polished_seq, polished_sc
            print(f"[final] polished to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    # Convert to plain Python list of floats as required by the harness.
    return [float(v) for v in best_seq]
