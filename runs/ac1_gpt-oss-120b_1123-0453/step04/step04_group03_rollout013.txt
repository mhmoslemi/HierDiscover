        # compute dot product of a reversed shifting
        # a[i] * g[j] where i+j=k
        # Equivalent to sum_{i=max(0, k-(n-1))}^{min(k, n-1)} a_i * g_{k-i}
        # Build vector of coefficients for g: of length n, with coefficient a_i where index j=k-i within bounds.
        # Use for each k create vector coeff of length n with zeros except at indices j where i valid.

        i_end = min(k, n - 1)
        # indices i: i_start..i_end
        for i in range(i_start, i_end + 1):
            j = k - i
            coeff[j] = a[i]  # coefficient for g_j
        constraints.append(coeff @ g <= M)
    # objective: maximize sum(g)
    obj = cp.Maximize(cp.sum(g))

    prob.solve(solver=cp.GLPK, warm_start=True)

        return None

        return g.value

def _intensive_refinement(start, deadline):
    n = len(start)
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    cur = best.copy()
    cur_score = best_score
    step = 0.01  # movement proportion of current values

    while time.monotonic() < deadline:
        # choose move type
        if rng.random() < 0.5:
            # transfer mass from i to j
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_transfer = cur[i]
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer * 0.5  # up to half of source mass
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # Gaussian perturbation of all entries
            cand = cur + rng.normal(scale=step, size=n)
        # enforce nonnegativity and project onto simplex
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, 1.0)  # sum =1

        cand_score = _compute_score_from_vector(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                # adapt step up
                step = min(0.1, step * 1.02)
        else:
            # adapt step down
            step *= 0.995
            if step < 1e-5:
                step = 0.01
    return best, best_score

The search consists of three stages:

     max‑convolution.  The learning‑rate, smoothing factor and patience

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact score used by `evaluate_sequence` (no internal clipping)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / beta
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    return w


def _weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σₖ wₖ·(a*a)ₖ w.r.t. ``a``.
    ``weights`` must have length 2·len(a)‑1 and sum to 1.
    """
    n = len(a)
    grad = np.zeros_like(a)
    # each convolution offset contributes a rank‑1 matrix a[j]·e_i
    for k, wk in enumerate(weights):
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += wk * a[j] * 2.0
        # the diagonal (i==j) has been added twice; keep a single copy
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= wk * a[i[mask]]
    return grad


def _adam_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 100_000,
    patience: int = 2_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the soft‑max surrogate of max‑convolution.
    Returns the best vector seen (projected onto the simplex) and its exact score.
    """
    a = a0.copy()
    n = len(a)

    # Adam hyper‑parameters – a little more aggressive than the reference.
    lr = 0.10
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        max_conv = np.max(conv)

        # a relatively sharp smoothing – 3 % of the current maximum.
        beta = max_conv * 0.03 + 1e-12
        w = _softmax_weights(conv, beta)
        grad = _weighted_gradient(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cur_score = _compute_score_from_vector(a)
        cand_score = _compute_score_from_vector(a_candidate)

        if cand_score < cur_score:
            a = a_candidate

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            no_improve = 0
            print(f"[Adam] n={n} it={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score


def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Direct (projected) sub‑gradient descent on the true max‑convolution.
    Uses a decreasing step size  ηₜ = 0.5 / √t .
    """
    a = a0.copy()
    n = len(a)

    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        max_val = np.max(conv)
        k = np.argmax(conv)

        # sub‑gradient of the max entry (no smoothing)
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_idx = np.arange(i_start, i_end + 1)
        grad[i_idx] = a[k - i_idx]     # ∂/∂a_i (a_i a_{k-i}) = a_{k-i}

        # step size decays as 1/√t
        step = 0.5 / math.sqrt(it)

        a_candidate = a - step * grad
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        cand_score = _compute_score_from_vector(a_candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()

        # Accept the candidate if it improves the actual objective;
        # otherwise keep the current point (standard sub‑gradient rule).
        cur_score = _compute_score_from_vector(a)
        if cand_score < cur_score:
            a = a_candidate

        # optional early stopping
        if it % 5_000 == 0 and best_score < 1.45:
            break

    return best_a, best_score


def _intensive_refinement(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing based on random mass transfers.
    Keeps the ℓ¹‑norm fixed via simplex projection.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    cur = best.copy()
    cur_score = best_score
    step = step_initial                # typical size of transferred mass (as absolute amount)

    while time.monotonic() < deadline:
        if rng.random() < 0.15:
            # Gaussian perturbation of all coordinates
            cand = cur + rng.normal(scale=step, size=cur.shape)
        else:
            # Transfer a random amount from i → j
            i, j = rng.integers(0, len(cur), size=2)
            if i == j:
                continue
            max_transfer = cur[i]
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer * 0.5   # at most 50 % of the source mass
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, 1.0)          # we keep Σaᵢ = 1 throughout

        cand_score = _compute_score_from_vector(cand)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            cur = cand.copy()
            cur_score = cand_score
            # on improvement we gently increase the step size
            step = min(step_initial, step * 1.02)
        else:
            # otherwise we slowly shrink the step size
            step *= 0.995
            # restart from the best point occasionally
            if rng.random() < 0.001:
                cur = best.copy()
                cur_score = best_score
            if step < 1e-6:
                step = step_initial

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any).
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        if np.sum(seq) == 0:
            seq = np.full_like(seq, 1.0 / len(seq))
        else:
            seq = seq / np.sum(seq)          # normalise to Σ=1
        score = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[warm] loaded height_sequence_1, n={len(seq)} score={score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Main optimisation – Adam + sub‑gradient, several lengths/restarts.
    # --------------------------------------------------------------
    lengths = [500, 800, 1000, 1200, 1500, 1800, 2000, 2500, 3000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = 1.0

        # decide how many re‑starts we can afford for this length
        restarts = 4 if n <= 1200 else 2

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a candidate
            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
            else:
                a0 = rng.random(n) + 1e-6
                a0 = a0 / np.sum(a0) * target_sum

            # ------------------------------------------------------------------
            # (a) Adam descent – give it ~12 s (or less if time runs out)
            # ------------------------------------------------------------------
            sub_deadline = min(deadline, time.monotonic() + 12.0)
            a_adam, sc_adam = _adam_descent(
                a0, target_sum, sub_deadline, max_iters=200_000, patience=3_000
            )

            # ------------------------------------------------------------------
            # (b) Sub‑gradient refinement – use the remaining slice of this run
            # ------------------------------------------------------------------
            remaining = min(deadline, time.monotonic() + 8.0)
            a_sub, sc_sub = _subgradient_descent(
                a_adam, target_sum, remaining, max_iters=200_000
            )

            # pick the best of the two
            if sc_sub < sc_adam:
                a_opt, sc_opt = a_sub, sc_sub
            else:
                a_opt, sc_opt = a_adam, sc_adam

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[update] n={n} restart={rs} score={sc_opt:.6f}")

            # early exit if we have already crossed a comfortable margin
            if best_score < 1.45:
                print("[early] satisfactory score reached")
                break
        if best_score < 1.45:
            break

    # --------------------------------------------------------------
    # 2️⃣ Final intensive polishing (if time permits).
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of length 800
        n = 800
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _intensive_refinement(best_seq, deadline,
                                                       step_initial=0.02)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # final safety clipping (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # --------------------------------------------------------------
    # Return a plain Python list – the external evaluation script will
    # invoke ``evaluate_sequence`` on it afterwards.
    # --------------------------------------------------------------
    return [float(x) for x in best_seq]
