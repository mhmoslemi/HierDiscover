def sa_optimize(seq, target_sum, deadline):
    best = seq.copy()
    best_score = fast_score(best)
    cur = seq.copy()
    cur_score = best_score
    n = len(seq)
    T0 = 0.01
    # compute schedule
    # we can use geometric cooling: T = T0 * (0.999**iteration)
    step_scale = target_sum * 0.01  # typical delta magnitude
    rng = np.random.default_rng()
    iter = 0
    while time.monotonic() < deadline:
        # propose move 
        i, j = rng.integers(n, size=2)
        if i == j: continue
        # choose delta from a normal distribution centered at 0
        delta = rng.normal(scale=step_scale)
        # ensure we don't go negative
        if cur[i] - delta < 0:
            delta = cur[i]  # move all
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # enforce clipping amplitude 1000
        # if any exceed 1000, clip and scale? but large values seldom
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        # renormalize to target sum
        total = new_seq.sum()
        if total == 0.0:
            new_seq = np.full(n, target_sum/n)
        else:
            new_seq = new_seq * (target_sum / total)
        new_score = fast_score(new_seq)
        # accept 
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score)/T):
            cur = new_seq
            cur_score = new_score
            if new_score < best_score:
                best = new_seq.copy()
                best_score = new_score
        # update temperature
        iter += 1
        T = T0 * (0.9995**iter) # decay
        # optionally adjust step scale based on acceptance ratio, etc.
        if iter % 1000 == 0:
            # adapt step_scale
            step_scale = max(step_scale * 0.99, target_sum * 0.0005)
    return best, best_score
```

def _simulated_annealing(seq: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator):
    n = len(seq)
    best = seq.copy()
    best_sc = _fast_score(best)
    cur = seq.copy()
    cur_sc = best_sc

    # Temperature schedule: start high then exponential decay.
    T0 = 0.01
    # step fraction distribution: we will move fraction f*cur[i]; but we can also add absolute scale.
    it = 0
    while time.monotonic() < deadline:
        # propose move
        i = rng.integers(n)
        if cur[i] == 0.0:
            continue
        j = rng.integers(n)
        # ensure i!=j
        if i == j:
            continue
        # max amount we can transfer without exceeding 1000 at j
        max_transfer = min(cur[i], 1000.0 - cur[j])
        if max_transfer <= 0:
            continue
        # choose transferred amount: sample fractional factor
        frac = rng.random()
        delta = max_transfer * frac
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum unchanged, but ensure nonnegative after due to rounding
        # no need to clip beyond 1000 as we limited delta
        cand_sc = _fast_score(cand)
        # temperature
        T = T0 * (0.9995 ** it)
        # acceptance
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / max(T, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            # update best
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        it += 1
        # optionally break if time near deadline
        if it % 5000 == 0 and time.monotonic() >= deadline - 0.1:
            break
    return best, best_sc
```

"""
Propose a step sequence with a low convolution‐peak score.

Strategy
--------
1. **Broad parametric sweep** – for several lengths we evaluate many
   hand‑crafted families (exponential, power‑law, classic windows,
   Gaussian).  For the exponential and power families we run a cheap
   one‑dimensional scalar optimisation (SciPy bounded Brent) to fine‑tune
   the decay exponent.

2. **Simulated annealing** – starting from the overall best candidate we
   repeatedly move a random amount of mass from one index to another.
   Moves are accepted if they improve the objective or with a
   temperature‑dependent probability.  The temperature decays
   geometrically; occasional global Gaussian perturbations help the
   search escape plateaus.

3. **Projected gradient descent (max‑entry)** – a short refinement that
   follows the sub‑gradient of the currently maximal convolution entry,
   keeping the ℓ¹‑norm fixed.

All operations respect the security checks of ``evaluate_sequence``:
non‑negative entries, clipping to [0,1000] and a fixed ℓ¹‑norm
√(2·n).  The routine stops as soon as the time budget is exhausted and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via FFT (exact, not circular)
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of `seq` with itself using FFT."""
    n = seq.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two ≥ 2n‑1
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Exact objective (without the wrapper).  seq must already satisfy the sum."""
    n = seq.shape[0]
    max_b = _conv_fft(seq).max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Families of candidate vectors (with optional SciPy refinement)
# ----------------------------------------------------------------------
def _candidate_families_with_refine(n: int, target_sum: float, rng: np.random.Generator):
    """Return the best vector among many parametric families plus refined search."""
    best_vec = None
    best_sc = float("inf")

    def _normalise(v):
        v = np.asarray(v, dtype=np.float64)
        v = v / v.sum() * target_sum
        return v

    idx = np.arange(n, dtype=np.float64)

    # ---- uniform -------------------------------------------------------
    a = np.full(n, target_sum / n, dtype=np.float64)
    sc = _fast_score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # ---- classic windows -----------------------------------------------
    def _try(v):
        nonlocal best_sc, best_vec
        v = _normalise(v)
        sc = _fast_score(v)
        if sc < best_sc:
            best_sc, best_vec = sc, v.copy()
        return v, sc

    # Hann
    _try(0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1))))
    # Hamming
    _try(0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1)))
    # Blackman
    _try(0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
          + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1)))
    # Sine (half‑period)
    _try(np.sin(np.pi * (idx + 1) / (n + 1)))
    # Triangular (peak in the middle)
    _try(np.minimum(idx + 1, n - idx))

    # ---- Gaussian windows -----------------------------------------------
    # centre at (n‑1)/2
    centre = (n - 1) / 2.0
    for sigma_factor in np.linspace(0.08, 0.40, 13):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        _try(a)

    # ---- exponential decay ---------------------------------------------
    # coarse grid first
    alphas = np.concatenate([
        np.arange(1, 6) / (10.0 * n),           # tiny alphas
        np.arange(1, 21) / n,                  # 1/n … 20/n
        np.linspace(0.02, 0.20, 40),           # up to 0.2
        np.linspace(0.20, 1.00, 30)            # larger decays
    ])
    alphas = np.unique(alphas)
    for α in alphas:
        a = np.exp(-α * idx)
        _try(a)

    # ---- power‑law ------------------------------------------------------
    for p in np.linspace(0.05, 5.0, 120):
        base = np.arange(1, n + 1, dtype=np.float64)
        a = base ** (-p)
        _try(a)

    # ---- combined exponential‑polynomial --------------------------------
    # a_i ∝ exp(-α i) * (i+1)^p
    for α in np.linspace(0.5 / n, 5.0 / n, 15):
        for p in np.linspace(0.0, 2.0, 11):
            a = np.exp(-α * idx) * ((idx + 1) ** p)
            _try(a)

    # ------------------------------------------------------------------
    # Refine exponential shape with a 1‑D optimisation (if SciPy is present)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import minimize_scalar

        def _score_alpha(alpha):
            a = np.exp(-alpha * idx)
            a = _normalise(a)
            return _fast_score(a)

        # search in a reasonably wide interval
        res = minimize_scalar(_score_alpha,
                              bounds=(1e-6, 5.0),
                              method='bounded',
                              options={'xatol': 1e-7, 'maxiter': 60})
        if res.success:
            a = np.exp(-res.x * idx)
            a = _normalise(a)
            sc = _fast_score(a)
            if sc < best_sc:
                best_sc, best_vec = sc, a.copy()
                print(f"[refineExp] n={n} α={res.x:.6f} score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Refine power‑law shape similarly
    # ------------------------------------------------------------------
    try:
        def _score_power(p):
            base = np.arange(1, n + 1, dtype=np.float64)
            a = base ** (-p)
            a = _normalise(a)
            return _fast_score(a)

        res = minimize_scalar(_score_power,
                              bounds=(0.01, 10.0),
                              method='bounded',
                              options={'xatol': 1e-5, 'maxiter': 60})
        if res.success:
            a = np.arange(1, n + 1, dtype=np.float64) ** (-res.x)
            a = _normalise(a)
            sc = _fast_score(a)
            if sc < best_sc:
                best_sc, best_vec = sc, a.copy()
                print(f"[refinePow] n={n} p={res.x:.6f} score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # A few cheap random Dirichlet samples (helps to discover exotic shapes)
    # ------------------------------------------------------------------
    for _ in range(30):
        a = rng.random(n)
        a = a / a.sum() * target_sum
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Simulated annealing – global stochastic refinement
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         rng: np.random.Generator,
                         max_steps: int = 10_000_000):
    """Simple SA that moves mass between two bins while preserving ℓ¹‑norm."""
    n = start_seq.shape[0]
    best = start_seq.copy()
    best_sc = _fast_score(best)

    cur = start_seq.copy()
    cur_sc = best_sc

    # temperature schedule
    T0 = 0.015
    # safe guard for division by zero in acceptance probability
    eps_T = 1e-12

    it = 0
    while time.monotonic() < deadline and it < max_steps:
        # --- propose a local mass transfer ------------------------------
        i = rng.integers(n)
        if cur[i] == 0.0:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue

        # how much can we move without violating the 1000‑clip?
        max_transfer = min(cur[i], 1000.0 - cur[j])
        if max_transfer <= 0.0:
            it += 1
            continue
        # random fraction (biased to small moves)
        delta = max_transfer * (rng.random() ** 2)

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum preserved, non‑negativity ensured by construction, clipping by construction

        cand_sc = _fast_score(cand)

        # temperature for this iteration
        T = T0 * (0.9997 ** it)

        # Metropolis acceptance
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / max(T, eps_T)):
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc

        # occasional global Gaussian perturbation
        if it % 4000 == 0 and it > 0:
            step_sigma = target_sum * 0.02
            pert = best + rng.normal(scale=step_sigma, size=n)
            pert = np.clip(pert, 0.0, 1000.0)
            pert = pert * (target_sum / pert.sum())
            pert_sc = _fast_score(pert)
            if pert_sc < cur_sc:
                cur, cur_sc = pert, pert_sc
                if pert_sc < best_sc:
                    best, best_sc = pert.copy(), pert_sc

        # early exit if we already beat the target bound
        if best_sc < 1.503:
            break

        it += 1

    return best, best_sc


# ----------------------------------------------------------------------
# Gradient descent on the active maximal convolution entry (as in the
# original baseline).  A few tweaks improve stability.
# ----------------------------------------------------------------------
def _gradient_descent(seq: np.ndarray,
                     target_sum: float,
                     deadline: float):
    """Projected sub‑gradient descent using the active max‑convolution entry."""
    n = seq.shape[0]
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    step = 0.2
    rng = np.random.default_rng()

    while time.monotonic() < deadline:
        conv = _conv_fft(seq)
        k = int(conv.argmax())                     # index of maximal entry

        # sub‑gradient of that entry:
        grad = np.zeros_like(seq)
        if k < n:                                   # overlap starts at 0
            grad[: k + 1] = seq[k :: -1]
        else:                                       # shifted part
            off = k - (n - 1)
            grad[off:] = seq[: n - off][::-1]

        # keep ℓ¹‑norm unchanged: subtract the mean component
        direction = -grad + grad.mean()

        cand = seq + step * direction
        # clip to allowed interval
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to the required ℓ¹‑norm
        total = cand.sum()
        if total == 0.0:
            cand = np.full(n, target_sum / n, dtype=np.float64)
        else:
            cand = cand * (target_sum / total)

        cand_sc = _fast_score(cand)

        # accept if strictly better, otherwise shrink step
        if cand_sc < best_sc:
            best_seq, best_sc = cand.copy(), cand_sc
            seq = cand
            step = min(step * 1.05, 1.0)          # accelerate after success
        else:
            step *= 0.5
            # occasional random shake‑up when step becomes tiny
            if step < 1e-6:
                i, j = rng.integers(n, size=2)
                delta = seq[i] * 0.04
                if delta > 0:
                    cand2 = seq.copy()
                    cand2[i] -= delta
                    cand2[j] += delta
                    cand2 = np.clip(cand2, 0.0, 1000.0)
                    cand2 = cand2 * (target_sum / cand2.sum())
                    cand2_sc = _fast_score(cand2)
                    if cand2_sc < best_sc:
                        seq = cand2
                        best_seq, best_sc = cand2.copy(), cand2_sc
                step = 0.2

        if best_sc < 1.503:
            break

    return best_seq, best_sc

# Main entry point for the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.25)   # keep a tiny margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Use a pre‑computed incumbent if the global variable exists
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = init * (target0 / init.sum())
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] height_sequence_1 n={n0} score={sc0:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1️⃣  Exhaustive parametric sweep (fast)
    # --------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500, 1800, 2000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        cand_vec, cand_sc = _candidate_families_with_refine(n, target, rng)
        if cand_sc < best_score:
            best_seq, best_score = cand_vec.copy(), cand_sc
            print(f"[family] n={n} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Global simulated‑annealing refinement (most time is spent here)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        best_seq, best_score = _simulated_annealing(best_seq, target, deadline, rng)
        print(f"[SA] best score after SA {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final projected gradient descent polish
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        best_seq, best_score = _gradient_descent(best_seq, target, deadline)
        print(f"[grad] final refined score {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never trigger)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    # Convert to plain Python list as required by the harness
    return [float(v) for v in best_seq]
