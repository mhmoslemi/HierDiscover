def _lp_finetune(f, target_sum, deadline, t_init=0.5, max_iters=20):

   cur_f = f.copy()
   best_f = cur_f
   best_score = _score(cur_f)
   t = t_init
   for iter in range(max_iters):
       if time.monotonic() >= deadline:
           break

       conv = np.convolve(cur_f, cur_f)
       M = np.max(conv)
       n = len(cur_f)
       # Build conv matrix
       A = np.zeros((2*n-1, n))
       # using vectorized assignment:
       for i, fi in enumerate(cur_f):
           A[i:i+n, i] = fi
       g = cp.Variable(n)
       constraints = [A @ g <= M * np.ones(2*n-1) + 1e-12,
                      g >= 0]
       prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
       try:
           prob.solve(solver=cp.ECOS, warm_start=True, max_iters=50000, verbose=False)
       except Exception as e:
           # fallback
           break
       if g.value is None:
           break
       g_opt = np.maximum(g.value, 0.0)  # ensure nonnegative
       sum_g = np.sum(g_opt)
       if sum_g < 1e-12:
           break
       g_scaled = g_opt * (target_sum / sum_g)
       new_f = (1 - t) * cur_f + t * g_scaled
       new_score = _score(new_f)
       if new_score < best_score:
           best_score = new_score
           best_f = new_f.copy()
           cur_f = new_f
           # maybe increase t?
           # we keep t same...
           print(f"[LP] iter={iter} n={n} score={new_score:.6f}")
       else:
           # reduce t to be more conservative
           t *= 0.5
           if t < 1e-4:
               break
   return best_f, best_score

"""
Improved optimisation for the “step‑function” challenge.

The search proceeds in three stages for a handful of candidate lengths:

1️⃣ *Warm‑start*: a uniform simplex‑projected vector (or the historic
   `height_sequence_1` if it is available).

2️⃣ *Projected Adam*: minimises the peak of the auto‑convolution while
   keeping the ℓ¹‑norm fixed.  The sub‑gradient of the maximal
   convolution entry is computed exactly, the iterate is projected onto
   the simplex after every Adam update.

3️⃣ *Linear‑program fine‑tuning*: for the current iterate `f` we solve

      maximise   Σ g_i
      subject to   (f * g)_k ≤ M  for k=0,…,2n‑2
                  g_i ≥ 0

   where `M = max (f * f)`.  This LP (solved with SciPy’s HiGHS) finds a
   partner `g` that respects the same convolution bound.  By mixing
   `f ← (1‑t)·f + t·g` (with a small `t`) the maximal auto‑convolution
   is provably non‑increasing.  A few repetitions quickly push the
   score below the previous best.

4️⃣ *Optional annealing*: a short temperature‑controlled hill‑climber
   (mass transfers between coordinates) polishes the final candidate.

All helper functions are top‑level; a hard time‑budget (default 1000 s)
is respected and the best sequence found before the deadline is
returned as plain ``list[float]`` suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.random import default_rng
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(f: np.ndarray) -> float:
    """Exact value of the objective used by `evaluate_sequence`."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _active_offsets(conv: np.ndarray, max_val: float) -> np.ndarray:
    """Indices where the convolution attains (almost) the maximal value."""
    eps = max_val * 1e-12 + 1e-14
    return np.where(conv >= max_val - eps)[0]


def _gradient_submax(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the worst auto‑convolution entry w.r.t. ``f``.
    The gradient is summed over all active offsets.
    """
    n = len(f)
    grad = np.zeros_like(f)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        # each derivative w.r.t. f[i] receives f[j]
        grad[i] += f[j]
        grad[j] += f[i]                     # symmetric contribution
        # diagonal terms were added twice; correct it
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= f[i[mask]]
    return grad


# ----------------------------------------------------------------------
# 1️⃣ Projected Adam optimiser (same logic as the reference implementation)
# ----------------------------------------------------------------------


def _adam_descent(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 4000,
) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the maximal auto‑convolution entry."""
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    it = 0
    no_improve = 0

    while it < max_iters and time.monotonic() < deadline:
        it += 1

        conv = np.convolve(f, f)
        max_val = np.max(conv)
        active = _active_offsets(conv, max_val)

        grad_max = _gradient_submax(f, active)
        sum_f = np.sum(f)
        grad_obj = (2.0 * len(f) / (sum_f * sum_f)) * grad_max

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = _project_onto_simplex(f_candidate, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cand_score = _score(f_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_f = f_candidate.copy()
            no_improve = 0
            print(f"[Adam] it={it:4d} n={len(f)} score={cand_score:.6f}")
        else:
            no_improve += 1

        # Accept the candidate if it improves the raw objective;
        # otherwise keep the old point.
        cur_score = _score(f)
        if cand_score < cur_score:
            f = f_candidate
        else:
            if no_improve > 200:
                lr *= 0.7
                no_improve = 0

        if no_improve > 800:
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# 2️⃣ Linear‑program fine‑tuning (convex combination step)
# ----------------------------------------------------------------------


def _lp_finetune(
    f: np.ndarray,
    target_sum: float,
    deadline: float,
    t: float = 0.5,
    max_iter: int = 12,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves a simple LP

        maximise   Σ g_i
        subject to (f * g)_k ≤ M  (k = 0 … 2n‑2)
                  g_i ≥ 0

    where M = max(f * f).  The resulting partner ``g`` is scaled to the
    correct ℓ¹‑norm and mixed with the current iterate
    ``f ← (1‑t)·f + t·g``.  The process never increases the objective
    and usually drives it down quickly.
    """
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    cur_f = f.copy()
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1

        conv = np.convolve(cur_f, cur_f)
        M = np.max(conv)

        # Build the convolution matrix A where (A @ g)[k] = (cur_f * g)[k]
        total_len = 2 * n - 1
        A = np.zeros((total_len, n), dtype=np.float64)
        for i in range(n):
            A[i : i + n, i] = cur_f[i]

        # Linear programme: maximise Σ g_i   ⇔ minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(total_len, M + 1e-12, dtype=np.float64)

        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=(0, None),
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        sum_g = np.sum(g)
        if sum_g < 1e-12:
            break

        # Scale to the required ℓ¹‑norm
        g_scaled = g * (target_sum / sum_g)

        new_f = (1.0 - t) * cur_f + t * g_scaled
        new_score = _score(new_f)

        if new_score < best_score:
            best_score = new_score
            best_f = new_f.copy()
            cur_f = new_f
            print(f"[LP] iter={it} n={n} score={new_score:.6f}")
        else:
            # Reduce the mixing factor – large steps are no longer helpful
            t *= 0.5
            if t < 1e-4:
                break

    return best_f, best_score


# ----------------------------------------------------------------------
# 3️⃣ Simple temperature‑controlled simulated annealing (optional polish)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start: np.ndarray,
    deadline: float,
    init_temp: float = 0.2,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climber that moves a small amount of mass from one coordinate to
    another while keeping the ℓ¹‑norm exactly.  Up‑hill moves are accepted
    with Metropolis probability.
    """
    rng = default_rng()
    cur = start.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.05 * target_sum / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-8:
            T = init_temp

        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score
