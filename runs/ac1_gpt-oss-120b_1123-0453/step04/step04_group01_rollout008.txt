def conv_fft(x):
    L = 1 << ((2*n-1).bit_length())
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[:2*n-1]

import time, math, numpy as np

def _normalize(vec, target_sum):
    s = vec.sum()
    if s <= 0:
        return np.full_like(vec, target_sum / vec.shape[0])
    return vec * (target_sum / s)

def _fast_score(seq):
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a**2)

def _project_simplex(v):
    """Projection of v onto the probability simplex {x: x_i>=0, sum x=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the rho where u_i - (cssv_i / i) > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries are zero or negative
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(p):
    n = p.shape[0]
    L = _next_pow2(2 * n - 1)
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[:2 * n - 1]
    return conv

def _corr_fft(p, w):
    # correlation: corr_i = sum_j p_j * w_{i + j}
    n = p.shape[0]
    L = _next_pow2(3 * n - 2)
    fp = np.fft.rfft(p, n=L)
    fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(fp * np.conj(fw), n=L)[:3 * n - 2]
    start = len(w) - 1  # zero shift
    grad = 2.0 * corr[start:start + n]
    return grad

    target = math.sqrt(2.0 * n)
    p = np.copy(p0)
    # ensure simplex
    p = _project_simplex(p)
    best_p = np.copy(p)
    best_sc = _fast_score(p * target)
    # Adam states
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    beta_factor = 1.5
    iter_cnt = 0

def _swap_hill_climb(seq, target, deadline, rng):

        cand = _normalize(cand, target)
        sc = _fast_score(cand)

    return best

"""
Search for a low‑value step‑function.

The algorithm works with the probability vector p (ℓ¹‑norm = 1) and
optimises a smooth approximation of the max‑convolution entry
using Adam‑style updates.  The soft‑max temperature β is increased
periodically so the optimisation first sees a smoothed objective and
then focuses on the true maximum.  After the main optimisation a cheap
hill‑climb and a short Gaussian perturbation polish the result.

All sequences are projected onto the probability simplex, then scaled
to the required ℓ¹‑norm √(2·n) before being returned.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``vec`` so that its ℓ¹‑norm equals ``target_sum``."""
    s = vec.sum()
    if s <= 0.0:
        return np.full_like(vec, target_sum / vec.shape[0])
    return vec * (target_sum / s)


def _fast_score(seq: np.ndarray) -> float:
    """Exact analogue of ``evaluate_sequence`` (no safety wrapper)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    # Find the largest index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT‑based convolution / correlation (real‑valued)
# ----------------------------------------------------------------------


def _conv_fft(p: np.ndarray) -> np.ndarray:
    """Linear convolution p * p using FFT (returns length 2·n‑1)."""
    n = p.shape[0]
    L = _next_pow2(2 * n - 1)
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    return conv


def _corr_fft(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation needed for the gradient:
    grad_i = 2 Σ_j p_j * w_{i+j},   i = 0 … n‑1.
    """
    n = p.shape[0]
    L = _next_pow2(3 * n - 2)                     # length for correlation
    fp = np.fft.rfft(p, n=L)
    fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(fp * np.conj(fw), n=L)[: 3 * n - 2]
    start = len(w) - 1                            # zero‑shift index
    grad = 2.0 * corr[start : start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max objective
# ----------------------------------------------------------------------


def _adam_optimize(p0: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator):
    """
    Minimise the maximal auto‑convolution entry for length ``n``.
    Work with the probability vector p (ℓ¹‑norm = 1) and a smooth‑max
    surrogate controlled by a temperature β that is increased during
    optimisation.  Adam updates are projected back onto the simplex.
    """
    target = math.sqrt(2.0 * n)          # final ℓ¹‑norm for the step heights
    p = _project_simplex(p0.astype(np.float64))

    best_p = p.copy()
    best_sc = _fast_score(p * target)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    # smooth‑max temperature schedule
    beta_soft = 2.0
    beta_factor = 1.5

    it = 0
    max_iters = 20000
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # (p * p) – linear convolution
        conv = _conv_fft(p)

        # soft‑max weights w_k = exp(β·conv_k) / Σ exp(β·conv)
        tmp = beta_soft * conv
        max_tmp = tmp.max()
        exp_tmp = np.exp(tmp - max_tmp)          # safe shift
        w = exp_tmp / exp_tmp.sum()

        # gradient of the smooth‑max objective
        grad = _corr_fft(p, w)

        # Adam update (note: we *subtract* the gradient because we minimise)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step

        # keep p on the probability simplex
        p = _project_simplex(p)

        # increase the temperature every few iterations
        if it % 300 == 0:
            beta_soft = min(beta_soft * beta_factor, 1e5)

        # periodic exact evaluation (cheap for n≤1000)
        if it % 50 == 0:
            sc = _fast_score(p * target)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_p = p.copy()
                lr = max(lr * 0.9, 0.01)          # slow down a bit after improvement
                print(f"[adam n={n} it={it}] new best {sc:.6f}")

    return best_p, best_sc


# ----------------------------------------------------------------------
# Cheap post‑processing (swap hill‑climb + Gaussian perturbation)
# ----------------------------------------------------------------------


def _swap_hill_climb(seq: np.ndarray,
                    target: float,
                    deadline: float,
                    rng: np.random.Generator) -> np.ndarray:
    """Swap a tiny mass between two indices if it improves the score."""
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]
    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target)
        sc = _fast_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
    return best


def _gaussian_local_search(seq: np.ndarray,
                           target: float,
                           deadline: float,
                           rng: np.random.Generator) -> np.ndarray:
    """Short annealing‑style Gaussian perturbation loop."""
    best = seq.copy()
    best_sc = _fast_score(best)
    step = target * 0.02
    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape[0])
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand, target)
        sc = _fast_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
            step *= 0.9
        else:
            step *= 0.99
            if step < 1e-6:
                step = target * 0.02
    return best
