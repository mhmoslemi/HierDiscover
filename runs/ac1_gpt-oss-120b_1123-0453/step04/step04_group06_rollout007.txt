        cand = _project_onto_simplex(cand, target_sum)

        if cand_score < best_score:

            f = cand
        else:
            # Still move to cand to allow exploration
            f = cand
        beta = min(beta * beta_factor, beta_max)
    return best_f, best_score

"""
search_step_function.py

Search for a non‑negative step‑function (a list of heights) that minimizes
the evaluation function

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the discrete linear convolution.  The search runs inside a
hard 1000 s time budget (the outer harness supplies ``budget_s``) and returns
the best sequence it has found.

Algorithm
---------

* **Normalization** –  for a length ``n`` we keep the ℓ¹‑norm fixed to
  ``sqrt(2·n)``.  With this scaling the score equals simply ``max(conv)``.

* **Adam‑based smoothed‑max gradient descent** –  the true max is approximated
  by a soft‑max (log‑sum‑exp) with a temperature parameter ``β`` that is
  annealed from a small to a large value.  The gradient of the weighted sum
  of convolution entries is computed analytically (vectorised, no Python loops)
  and an Adam optimizer is used to obtain an adaptive step direction.
  After each step the vector is projected back onto the simplex
  ``{a_i ≥ 0, Σ a_i = sqrt(2·n)}``.

* **Multi‑restart & length sweep** –  we try several lengths
  (300 … 1800) and for each length a handful of random restarts.
  A previously discovered construction (``height_sequence_1``) is used as a warm
  start when its length matches the current ``n``.

* **Stochastic hill‑climbing polish** –  after the gradient phase a short
  Gaussian perturbation hill‑climb (projection onto the simplex after each
  perturbation) is applied to the current best candidate.

* **Time‑aware loop** –  every sub‑procedure is given a local deadline so
  that the outer 1000 s limit is never exceeded.

The combination of a robust optimiser (Adam), annealing of the smooth‑max
parameter and a final stochastic polish provides a better exploration of the
non‑convex landscape than plain gradient descent and is able to improve the
previous bound (≈ 1.511) within the allowed time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are equal
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(v: np.ndarray) -> float:
    """Exact score used by ``evaluate_sequence`` (no clipping)."""
    n = len(v)
    conv = np.convolve(v, v)
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _weighted_gradient(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(f*f)_k  w.r.t. f.
    ``w`` has length 2·len(f)‑1 and Σ w = 1.
    The implementation follows the O(n²) vector‑ised scheme used in the
    reference code.
    """
    n = len(f)
    grad = np.zeros_like(f)

    # Loop over convolution offsets – the heavy work is done by NumPy slicing.
    for k in range(2 * n - 1):
        wk = w[k]
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end   = min(k, n - 1)
        # indices i that participate in offset k
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        # each term contributes 2·wk·f[j] to the gradient of f[i]
        grad[i_start:i_end + 1] += wk * 2.0 * f[j_range]

        # correct the diagonal contribution (i == j) – it should be counted once
        if (k & 1) == 0:          # k even → possible i=j
            i_eq = k // 2
            if i_start <= i_eq <= i_end:
                grad[i_eq] -= wk * f[i_eq]        # 2·c – c = c
    return grad


def _adam_smooth_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 1500,
    beta_start: float = 0.5,
    beta_factor: float = 1.5,
    beta_max: float = 256.0,
    adam_lr: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Smoothed‑max minimisation using the Adam optimiser.
    Returns the best vector seen together with its exact score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)

    m = np.zeros_like(f)  # first moment
    v = np.zeros_like(f)  # second moment
    beta = beta_start
    t = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        max_val = conv.max()

        # soft‑max weights – numerically stable
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        # gradient of the weighted sum of convolution entries
        grad = _weighted_gradient(f, w)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        cand = f - step
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _compute_score(cand)

        if cand_score < best_score:
            best_f, best_score = cand.copy(), cand_score

        # always move to the candidate – Adam is tolerant of non‑improving steps
        f = cand

        # anneal the soft‑max temperature
        beta = min(beta * beta_factor, beta_max)

    return best_f, best_score


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.15,
    max_iters: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that stays on the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score(cand)

        if cand_score < best_score:
            best, best_score = cand, cand_score

        sigma *= 0.9995          # slowly shrink the perturbation radius
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step sequence within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a tiny safety margin for the outer harness
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if isinstance(warm, (list, tuple, np.ndarray)):
        warm_arr = np.asarray(warm, dtype=np.float64)
        warm_arr = np.clip(warm_arr, 0.0, 1000.0)
        n_warm = len(warm_arr)
        target_w = math.sqrt(2.0 * n_warm)
        s = warm_arr.sum()
        if s > 0:
            warm_arr = warm_arr * (target_w / s)
        else:
            warm_arr = np.full(n_warm, target_w / n_warm, dtype=np.float64)
        warm_score = _compute_score(warm_arr)
        best_seq, best_score = warm_arr.copy(), warm_score
        print(f"[init‑warm] n={n_warm} score={warm_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Length sweep + multi‑restart Adam optimisation
    # ------------------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000, 1200, 1500, 1800]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Number of restarts is larger for smaller n (they are cheap)
        restarts = 12 if n <= 600 else 6

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a point on the simplex
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
            else:
                # Dirichlet‑like random start (very small offset to avoid exact zeros)
                f0 = rng.random(n) + 1e-12
                f0 = f0 / f0.sum() * target_sum

            # give each optimisation a local deadline (≈ 12 s or the remaining time)
            sub_deadline = min(deadline, time.monotonic() + 12.0)

            opt_seq, opt_score = _adam_smooth_optimize(
                f0,
                target_sum,
                sub_deadline,
                max_iters=2000,
                beta_start=0.5,
                beta_factor=1.5,
                beta_max=256.0,
                adam_lr=0.02,
            )

            if opt_score < best_score:
                best_seq, best_score = opt_seq.copy(), opt_score
                print(f"[update] n={n} restart={rs} score={opt_score:.6f}")

            # early exit if we already beat the target bound
            if best_score < 1.503:
                print("[early‑exit] target achieved")
                deadline = time.monotonic() + 0.2   # tiny buffer
                break

    # ------------------------------------------------------------------
    # 2️⃣ Final stochastic polish
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n_final = len(best_seq)
        target_sum = math.sqrt(2.0 * n_final)
        refined_seq, refined_score = _hill_climb(
            best_seq,
            target_sum,
            deadline,
            sigma_initial=0.15,
            max_iters=8000,
        )
        if refined_score < best_score:
            best_seq, best_score = refined_seq, refined_score
            print(f"[refine] final score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall‑back: a uniform sequence of length 300 (guaranteed feasible)
        n_fb = 300
        target_fb = math.sqrt(2.0 * n_fb)
        best_seq = np.full(n_fb, target_fb / n_fb, dtype=np.float64)
        best_score = _compute_score(best_seq)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f} length={len(best_seq)}")
    return [float(x) for x in best_seq]
