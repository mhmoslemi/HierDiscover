    u = np.sort(v)[::-1]

    w = np.maximum(v - theta, 0)
    return w

def _simulated_annealing(a, target, deadline, rng):
    n = len(a)
    cur = a.copy()
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score
    # initial temperature: maybe (cur_score - 1) * 0.5, or relative
    T0 = (cur_score - 1.0) if cur_score > 1 else 0.1
    T = T0
    iter = 0
    while time.monotonic() < deadline and iter < max_iters:
        iter += 1
        # random move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if cur[i] <= 0:
            continue
        max_move = min(cur[i] * 0.3, target * 0.05)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # possible to overshoot boundaries but they are nonnegative
        cand_score = _fast_score(cand)
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        # cooling
        T *= 0.99995
        # periodic large step: every 5000 iter, perform random Dirichlet restart if not improving
        if iter % 5000 == 0:
            if rng.random() < 0.3:
                # new random start
                cand = rng.dirichlet(np.ones(n)) * target
                cand_score = _fast_score(cand)
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                cur = cand.copy()
                cur_score = cand_score
```

def _project_simplex(v, target):
    """Euclidean projection of vector v onto the simplex {x >= 0, sum x = target}."""
    # Following Duchi et al. 2008
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # enforce sum exactly target (may be off due to numerical)
    w_sum = w.sum()
    if w_sum > 0:
        w *= target / w_sum
    else:
        w = np.full_like(v, target / n)
    return w

"""
search for a low‑max‑auto‑convolution step sequence.

The objective of ``evaluate_sequence`` (after the ℓ¹‑normalisation
∑a_i = √(2 n)) equals the largest entry of the auto‑convolution
a * a.  We therefore minimise max(a * a) under a_i ≥ 0 and a fixed sum.

The algorithm mixes three ideas:

1️⃣  **Broad parametric start** – generate several analytic families
    (uniform, triangular, power‑law, exponential, Beta‑shapes,
    sine‑powered) and a few Dirichlet draws, each ℓ¹‑normalised.

2️⃣  **Global search** – a simple simulated‑annealing loop that repeatedly
    moves a random amount of mass from one index to another, accepting
    improvements or uphill moves with a temperature that cools
    geometrically.  The convolution is recomputed exactly (fast enough
    for n ≤ 1000) and the sum is kept exactly constant.

3️⃣  **Local refinement** – projected gradient descent on a smooth
    surrogate ``∑ (conv_k)^p`` with an ever‑increasing exponent *p*
    (p→∞ approximates the max).  The gradient is obtained via a single
    convolution, the step is projected onto the simplex {x≥0, ∑x=√(2 n)}.
    This quickly pushes the mass away from the most‑active convolution
    index.

After the main optimisation a cheap hill‑climb (pairwise swaps) and a
short Gaussian perturbation polish the final vector.

All operations respect the security checks of ``evaluate_sequence``:
non‑negative entries, values ≤ 1000 and the ℓ¹‑norm is exactly
√(2 n).  The routine returns the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``vec`` to have ℓ¹‑norm exactly ``target_sum`` (preserving non‑negativity)."""
    s = vec.sum()
    if s <= 0.0:
        # fall back to uniform distribution
        n = vec.shape[0]
        return np.full(n, target_sum / n, dtype=np.float64)
    return vec * (target_sum / s)


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` but without the safety wrapper.
    Returns 2·n·max(conv)/(sum(seq)²).  For ℓ¹‑normalised sequences the factor
    equals the maximal auto‑convolution entry.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


def _project_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    { x ≥ 0,  Σ x = target } (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - target
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # final renormalisation (numerical safety)
    w_sum = w.sum()
    if w_sum > 0:
        w *= target / w_sum
    else:
        w = np.full_like(v, target / n)
    return w

# Candidate generation
# ----------------------------------------------------------------------


def _beta_family(n: int, a: float, b: float, target_sum: float) -> np.ndarray:
    """Beta‑distribution shape a_i ∝ x^{a‑1} (1‑x)^{b‑1},  x∈(0,1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    w = np.maximum(w, 1e-12)
    w = w / w.sum() * target_sum
    return w


def _generate_initial_candidates(n: int, target_sum: float, rng: np.random.Generator):
    """Return a list of plausible start vectors for length ``n``."""
    cand = []

    # uniform
    cand.append(np.full(n, target_sum / n, dtype=np.float64))

    # simple symmetric triangle (peak in the centre)
    x = np.arange(n, dtype=np.float64)
    tri = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
    tri[tri < 0] = 0.0
    cand.append(_normalize(tri, target_sum))

    # power‑law a_i ∝ i⁻ᵖ
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.2, 3.0, 15):
        a = idx ** (-p)
        cand.append(_normalize(a, target_sum))

    # exponential a_i ∝ e^{-α·i}
    for alpha in np.linspace(0.001, 0.25, 12):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand.append(_normalize(a, target_sum))

    # Beta family, wide range of parameters (including a,b<1)
    for _ in range(10):
        a_param = rng.uniform(0.1, 5.0)
        b_param = rng.uniform(0.1, 5.0)
        cand.append(_beta_family(n, a_param, b_param, target_sum))

    # explicit arcsine shape (Beta(½,½))
    cand.append(_beta_family(n, 0.5, 0.5, target_sum))

    # sine‑powered shapes a_i ∝ sin^p(π·i/(n+1))
    for p in np.linspace(0.5, 4.0, 12):
        a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1))) ** p
        cand.append(_normalize(a, target_sum))

    # a few random Dirichlet draws
    for _ in range(8):
        a = rng.dirichlet(np.ones(n)) * target_sum
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Simulated annealing (global search)
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iter: int = 2_000_000) -> np.ndarray:
    """
    Simple simulated‑annealing that moves a random amount of mass from one index
    to another.  The total ℓ¹‑mass stays exactly ``target_sum``.
    """
    n = start.shape[0]
    cur = start.copy()
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    # initial temperature – a fraction of the excess over the theoretical lower bound (≈1)
    T = max(0.01, (cur_score - 1.0) * 0.3)

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1

        # pick donor / receiver
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or cur[i] <= 0.0:
            continue

        # amount to move – keep it a modest fraction of donor mass
        max_move = min(cur[i] * 0.4, target_sum * 0.02)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = _fast_score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_score - cur_score) / (T + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # geometric cooling
        T *= 0.99995

        # occasional random restart to escape deep traps
        if it % 25000 == 0 and rng.random() < 0.1:
            cur = rng.dirichlet(np.ones(n)) * target_sum
            cur_score = _fast_score(cur)

    return best


# ----------------------------------------------------------------------
# Projected gradient descent on a smooth Lp surrogate (local refinement)
# ----------------------------------------------------------------------


def _gradient_descent_lp(start: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         rng: np.random.Generator,
                         p_start: int = 4,
                         p_end: int = 24,
                         step0: float = 0.15) -> np.ndarray:
    """
    Minimise Σ (conv_k)^p with a growing exponent p (p → ∞ approximates max).
    Each iteration takes a projected gradient step onto the simplex.
    """
    n = start.shape[0]
    cur = start.copy()
    best = cur.copy()
    best_score = _fast_score(cur)

    p = p_start
    step = step0 * target_sum / n          # base step size

    while time.monotonic() < deadline:
        conv = np.convolve(cur, cur)                # length 2n‑1
        # surrogate objective (not used directly, only for acceptance)
        obj = np.sum(conv ** p)

        # gradient of Σ conv^p :  g = 2 * (cur_rev * (p * conv^{p‑1}))
        w = p * (conv ** (p - 1))
        # convolution of reversed cur with w, then take the middle n entries
        grad = 2.0 * np.convolve(cur[::-1], w)[n - 1: n - 1 + n]

        # projected step
        cand = cur - step * grad
        cand = _project_simplex(cand, target_sum)

        cand_score = _fast_score(cand)

        # accept if the true max‑convolution improves, otherwise keep the better of the two
        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score
        if cand_score <= _fast_score(cur):
            cur, step = cand, step * 1.02          # increase step a little on success
        else:
            step *= 0.90                            # otherwise shrink step

        # gradually increase p (tighten the surrogate)
        if p < p_end:
            p = min(p_end, p + 1)

        # safety break if step becomes too tiny
        if step < 1e-9:
            break

    return best


# ----------------------------------------------------------------------
# Final cheap polishing (swap‑hill‑climb + Gaussian perturbation)
# ----------------------------------------------------------------------


def _swap_hill_climb(seq: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator) -> np.ndarray:
    """Very cheap hill‑climb that swaps a small delta between two indices."""
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.05, target_sum * 0.001)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _project_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
    return best


def _gaussian_local_search(seq: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           rng: np.random.Generator) -> np.ndarray:
    """Short Gaussian perturbation loop (annealing‑style)."""
    best = seq.copy()
    best_sc = _fast_score(best)
    step = target_sum * 0.02

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape[0])
        cand = _project_simplex(cand, target_sum)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
            step *= 0.9
        else:
            step *= 0.99
            if step < 1e-7:
                step = target_sum * 0.02
    return best

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  The routine stops a little before the limit

    kwargs : dict

    Returns

    list[float]
        The best sequence found before the deadline.
    """

    best_seq = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

            n0 = init.shape[0]

                init = _normalize(init, target0)
                sc0 = _fast_score(init)

                print(f"[init] height_sequence_1 (n={n0}) score={sc0:.6f}")
        except Exception:

    # ------------------------------------------------------------------
    # Explore a range of lengths – larger n gives more freedom.
    # ------------------------------------------------------------------
    lengths = [800, 600, 500, 400, 300, 200, 100]

    for n in lengths:

        target = math.sqrt(2.0 * n)

        # ---- generate a modest fleet of starting vectors -----------------
        candidates = _generate_initial_candidates(n, target, rng)

        for cand in candidates:
            if time.monotonic() >= deadline:
                break
            sc = _fast_score(cand)
            if sc < best_score:

        # ---- global simulated annealing ---------------------------------
        remaining = deadline - time.monotonic()
        if remaining <= 0:

        # allocate roughly half of the remaining budget for this length

    # ------------------------------------------------------------------
    n = best_seq.shape[0]

    best_seq = _swap_hill_climb(best_seq, target, deadline, rng)
    if time.monotonic() < deadline:

    final_score = _fast_score(best_seq)
    print(f"[final] best score {final_score:.6f}   length {best_seq.shape[0]}")

    return [float(x) for x in best_seq]
