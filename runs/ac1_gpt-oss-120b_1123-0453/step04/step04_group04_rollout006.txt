import time, math, numpy as np, scipy.optimize

def _fft_max_convolution(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    needed = 2 * n - 1
    return float(np.max(conv[:needed]))

def _project_simplex(v, s):
    """Euclidean projection of v onto the simplex {x>=0, sum x = s}."""
    # Reference: Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgradient(a, conv, max_val, eps=1e-12):
    """Return a sub‑gradient of max(conv) at point a."""
    n = a.shape[0]
    active = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        # Contribution 2*a[j] to each i
        grad[i] += 2.0 * a[j]
    return grad

def _lp_step(f, M, target_sum):
    """Solve the linear program max 1ᵀg  s.t. f*g ≤ M, g ≥ 0 and return a scaled g."""
    n = f.shape[0]
    rows = 2 * n - 1
    # construct convolution matrix of f (each column is f shifted)
    A = np.zeros((rows, n), dtype=np.float64)
    for col in range(n):
        A[col:col + n, col] = f
    b = np.full(rows, M, dtype=np.float64)
    # linear program: maximize sum(g)  <=> minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                bounds=[(0, None)] * n,
                                method='highs')
    if not res.success:
        return None
    g = res.x
    g_sum = g.sum()
    if g_sum <= 0:
        return None
    # scale to the required ℓ¹‑norm (the problem statement normalises to √(2n))
    scale = target_sum / g_sum
    g *= scale
    return g

def _polish_subgradient(a, target_sum, deadline):
    """Run many sub‑gradient steps with projection onto the ℓ¹‑simplex."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    best = a.copy()
    best_obj = 2.0 * n * _fft_max_convolution(a, fft_len) / (target_sum ** 2)
    start = time.monotonic()
    it = 0
    avg = target_sum / n
    while time.monotonic() < deadline:
        it += 1
        # compute convolution and its max
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2 * n - 1]
        max_val = conv.max()
        # sub‑gradient of the max entry
        g = _subgradient(a, conv, max_val)
        # diminishing step size
        step = avg * 0.5 / math.sqrt(it)
        a_new = a - step * g
        a_new = _project_simplex(a_new, target_sum)
        a_new = np.clip(a_new, 0.0, 1000.0)
        # evaluate new max
        fa_new = np.fft.rfft(a_new, n=fft_len)
        conv_new = np.fft.irfft(fa_new * fa_new, n=fft_len)[:2 * n - 1]
        obj_new = 2.0 * n * conv_new.max() / (target_sum ** 2)
        # accept if improvement
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best = a_new.copy()
        # simple greedy acceptance
        if obj_new <= 2.0 * n * max_val / (target_sum ** 2):
            a = a_new
        else:
            # occasional random restart to avoid stagnation
            if np.random.rand() < 0.001:
                a = best.copy()
        # early exit if we already beat the target 1.503 enough
        if best_obj < 1.48:
            break
    return best, best_obj

def _improve_sequence(a, target_sum, deadline):
    """Apply a few LP‑steps followed by a long sub‑gradient polish."""
    n = a.shape[0]
    # keep a copy of the current best
    best_seq = a.copy()
    best_obj = 2.0 * n * _fft_max_convolution(a) / (target_sum ** 2)
    # how many LP iterations we can afford?
    lp_iters = 5
    t = 0.15   # mixing parameter
    for _ in range(lp_iters):
        if time.monotonic() >= deadline:
            break
        M = _fft_max_convolution(best_seq)
        g = _lp_step(best_seq, M, target_sum)
        if g is None:
            break
        # convex combination
        new_seq = (1.0 - t) * best_seq + t * g
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        # evaluate
        new_obj = 2.0 * n * _fft_max_convolution(new_seq) / (target_sum ** 2)
        if new_obj < best_obj - 1e-12:
            best_seq = new_seq
            best_obj = new_obj
        else:
            # if not improving, shrink t
            t *= 0.5
            if t < 1e-4:
                break
    # final polishing with many sub‑gradient steps
    remaining = deadline - time.monotonic()
    if remaining > 0.1:

"""
A two‑phase optimiser for the step‑function bound.

1️⃣  “LP boost” – a few linear‑programming steps that imitate the
    classical f‑g alternation: with the current sequence f we solve
    max ∑g  s.t.  f∗g ≤ M (M = max f∗f) and g ≥ 0, then mix
    g back into f.  This quickly pushes mass away from the currently
    active convolution peaks.

2️⃣  Projected sub‑gradient polishing – many cheap gradient steps that
    minimise the maximal entry of f∗f while keeping the ℓ¹‑norm fixed
    (projection onto the simplex).  The step size decays with the
    iteration count, guaranteeing convergence.

The routine works for several candidate lengths, respects the global
time budget and always returns a non‑negative list that can be fed to
``evaluate_sequence``.
"""

import time
import math

import numpy as np
import scipy.optimize


def _fft_max_convolution(a: np.ndarray, fft_len: int | None = None) -> float:
    """Maximum entry of the auto‑convolution a∗a computed by FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    needed = 2 * n - 1
    return float(np.max(conv[:needed]))


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, ∑x = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(a: np.ndarray, conv: np.ndarray, max_val: float,
                 eps: float = 1e-12) -> np.ndarray:
    """Sub‑gradient of the max‑entry of a∗a."""
    n = a.shape[0]
    active = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]            # derivative of a_i a_j + a_j a_i
    return grad


def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve   max ∑g   s.t.   f∗g ≤ M ,  g ≥ 0 ,
    then scale the solution to have ℓ¹‑norm = target_sum.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix of f (each column is a shifted copy of f)
    A = np.zeros((rows, n), dtype=np.float64)
    for col in range(n):
        A[col:col + n, col] = f

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  ↔ minimise -∑g

    res = scipy.optimize.linprog(
        c, A_ub=A, b_ub=b,
        bounds=[(0, None)] * n,
        method='highs',
    )
    if not res.success:
        return None
    g = res.x
    g_sum = g.sum()
    if g_sum <= 0.0:
        return None
    scale = target_sum / g_sum
    g *= scale
    return g


def _polish_subgradient(a: np.ndarray, target_sum: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Long projected sub‑gradient descent."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    best = a.copy()
    best_obj = 2.0 * n * _fft_max_convolution(a, fft_len) / (target_sum ** 2)

    start = time.monotonic()
    it = 0
    avg = target_sum / n

    while time.monotonic() < deadline:
        it += 1
        # current convolution and its maximum
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2 * n - 1]
        cur_max = conv.max()

        # sub‑gradient of the max entry
        g = _subgradient(a, conv, cur_max)

        # diminishing step size
        step = avg * 0.5 / math.sqrt(it)

        a_new = a - step * g
        a_new = _project_onto_simplex(a_new, target_sum)
        a_new = np.clip(a_new, 0.0, 1000.0)

        # evaluate new max
        fa_new = np.fft.rfft(a_new, n=fft_len)
        conv_new = np.fft.irfft(fa_new * fa_new, n=fft_len)[:2 * n - 1]
        new_obj = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best = a_new.copy()

        # greedy acceptance
        if new_obj <= 2.0 * n * cur_max / (target_sum ** 2):
            a = a_new
        else:
            # occasional random jump to combat stagnation
            if np.random.rand() < 0.001:
                a = best.copy()

        # early exit once we are comfortably below the target threshold
        if best_obj < 1.48:
            break

    return best, best_obj


def _improve_sequence(a0: np.ndarray, target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Run a few LP‑boost steps followed by intensive sub‑gradient polishing.
    Returns the improved sequence and its maximal auto‑convolution entry.
    """
    n = a0.shape[0]
    best_seq = a0.copy()
    best_obj = 2.0 * n * _fft_max_convolution(best_seq) / (target_sum ** 2)

    # ----------- LP boost -------------------------------------------------
    t = 0.15                # mixing weight
    for _ in range(5):     # a small fixed number of LP iterations
        if time.monotonic() >= deadline:
            break
        M = _fft_max_convolution(best_seq)
        g = _lp_step(best_seq, M, target_sum)
        if g is None:
            break
        new_seq = (1.0 - t) * best_seq + t * g
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        new_obj = 2.0 * n * _fft_max_convolution(new_seq) / (target_sum ** 2)

        if new_obj < best_obj - 1e-12:
            best_seq = new_seq
            best_obj = new_obj
        else:
            t *= 0.5        # shrink the mixing factor if no progress
            if t < 1e-4:
                break

    # ----------- sub‑gradient polishing ----------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 0.1:
        best_seq, best_obj = _polish_subgradient(best_seq, target_sum, deadline)

    return best_seq, best_obj
