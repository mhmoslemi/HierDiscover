The code below implements a *smooth‑max* optimisation of the peak auto‑convolution.
The idea is to replace the non‑differentiable max by a differentiable
log‑sum‑exp (controlled by a temperature `α`).  
For a fixed ℓ¹‑norm the objective reduces to the maximum convolution entry,
so minimising the smooth‑max drives the whole convolution towards a flat
distribution.

* **Gradient** – The gradient of the smooth‑max can be written in closed form.
  If `w` are the soft‑max weights of the convolution, then  

  \[
  ∇\;{\rm smooth\_max}(f)=f+g,\qquad
  g_i=\sum_k w_k\,f_{k-i}
  \]

  The term `g` is obtained with a single convolution (`w * reverse(f)`).

* **Optimization loop** – An Adam‐style optimiser is used on the ℓ¹‑simplex.
  After every update the iterate is projected onto the simplex
  `Σ a_i = √(2n) , a_i ≥ 0`.  The temperature `α` is increased slowly
  (annealing) so that the algorithm first smoothes the landscape and then
  concentrates on the true maximum.

* **Search strategy** –  
  1. Warm‑start from the previously discovered construction `height_sequence_1`
     (if available).  
  2. Run the smooth‑max Adam optimiser for several sequence lengths
     (`n = 800 … 3000`) and a handful of random restarts.  
  3. Polishing by the simulated‑annealing hill‑climber from the previous
     solution (kept unchanged).  

All heavy work is done with FFT‑based convolutions (`O(n log n)`) so many
iterations fit comfortably into the 1000 s budget.

    2·n·max(conv(a,a)) / (sum(a)²)

optimisation on the ℓ¹‑simplex.  The temperature (α) is increased during the
run, letting the optimiser first smooth the landscape and later focus on the
true peak.  After the gradient phase a short simulated‑annealing polishing
step is applied.
"""

import time

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x | x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real‑valued inputs)."""
    size = _next_pow2(len(a) + len(b) - 1)
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    C = np.fft.irfft(A * B, size)
    return C[: len(a) + len(b) - 1]


def _score(f: np.ndarray) -> float:
    """
    Exact objective of `evaluate_sequence` (clipping inside that function is
    irrelevant for the values we generate).  The sum is assumed to be
    √(2·n), therefore the return value equals max(conv(f,f)).
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient
# ----------------------------------------------------------------------


def _smooth_max_and_grad(f: np.ndarray, alpha: float):
    """
    Returns
        smooth = (1/α)·log Σ_k exp(α·conv_k)
        grad   = f + g,   where g_i = Σ_k w_k·f_{k-i}
    with w_k = soft‑max weights derived from the convolution.
    """
    n = len(f)

    # 1️⃣ auto‑convolution
    conv = _fft_convolve(f, f)                     # length 2n‑1
    max_c = np.max(conv)

    # 2️⃣ soft‑max of the convolution (stable)
    shifted = alpha * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp                     # Σ w = 1
    smooth = max_c + np.log(sum_exp) / alpha

    # 3️⃣ g = (w * reverse(f))_{i+n‑1}
    f_rev = f[::-1]
    conv_w_rev = _fft_convolve(w, f_rev)          # length 3n‑2
    g = conv_w_rev[n - 1 : n - 1 + n]              # slice of size n

    grad = f + g
    return smooth, grad


# ----------------------------------------------------------------------
# Adam optimiser that minimises the smooth‑max (projected onto simplex)
# ----------------------------------------------------------------------


def _smooth_adam(f0: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the ℓ¹‑simplex for the smooth‑max objective.
    The temperature α is annealed (multiplied by 1.05 each iteration) up to
    a safe ceiling.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.15                         # base learning rate

    alpha = 0.5
    max_alpha = 250.0

    it = 0
    while it < max_iters and time.monotonic() < deadline:
        it += 1

        smooth, grad = _smooth_max_and_grad(f, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = _project_onto_simplex(f_candidate, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cur_score = _score(f)
        cand_score = _score(f_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_f = f_candidate.copy()
            print(f"[SmoothAdam] it={it:4d} n={len(f)} α={alpha:.1f} score={cand_score:.6f}")

        # accept if improves the true objective (not just smooth)
        if cand_score < cur_score:
            f = f_candidate

        # anneal temperature
        alpha = min(alpha * 1.05, max_alpha)

    return best_f, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (re‑used from the previous solution)
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.05 * target_sum / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-8:
            T = init_temp
        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Returns a list of non‑negative heights that (hopefully) improves the
    current best upper bound.  The routine respects the given time budget
    (default 1000 s) by allocating time to several sequence lengths,
    performing a smooth‑max Adam descent and a final annealing pass.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction (if any)
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64).copy()
        n0 = len(seq)
        target_sum0 = math.sqrt(2.0 * n0)
        seq = _project_onto_simplex(seq, target_sum0)
        s0 = _score(seq)
        best_seq = seq.copy()
        best_score = s0
        print(f"[init] warm‑start from height_sequence_1, n={n0}, score={s0:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Main optimisation – smooth‑max Adam on several lengths
    # --------------------------------------------------------------
    lengths = [800, 1200, 1800, 2400, 3000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        restarts = 6

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a starting point
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
                f0 += rng.normal(scale=0.01 * target_sum, size=n)
                f0 = np.clip(f0, 0.0, None)
                f0 = _project_onto_simplex(f0, target_sum)
            else:
                f0 = rng.random(n)
                f0 = _project_onto_simplex(f0, target_sum)

            # give each Adam run a few seconds (adjustable)
            sub_deadline = min(deadline, time.monotonic() + 12.0)
            f_opt, score_opt = _smooth_adam(f0,
                                            target_sum,
                                            sub_deadline,
                                            max_iters=5000)

            if score_opt < best_score:
                best_score = score_opt
                best_seq = f_opt.copy()
                print(f"[update] n={n} restart={r} smooth_adam score={score_opt:.6f}")

                if best_score < 1.503:
                    print("[early] target reached during Adam phase")
                    break
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # 2️⃣ Final polishing with simulated annealing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _simulated_annealing(best_seq,
                                                     deadline,
                                                     init_temp=0.3,
                                                     decay=0.9997)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final] annealed score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] uniform construction used")

    # clip once more for safety (evaluate_sequence applies the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
