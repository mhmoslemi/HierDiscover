"""
Improved step‑function search via alternating linear‑programming updates.

Algorithm
---------
1.  Pick a length `n` and initialise a non‑negative vector `f` with
    sum = √(2 n) (either a warm start, a random Dirichlet draw or the
    uniform distribution).

2.  Repeatedly improve `f`:
    a.  Compute the current maximal auto‑convolution value `M`.
    b.  Solve the linear programme

            maximise   Σ g_i
            subject to  Σ_j f_j · g_{k‑j} ≤ M   for all k = 0 … 2n‑2
                        g_i ≥ 0

       (the constraints are the entries of the convolution `f*g`).
       The LP is a simple dense problem (≤ 2 000 variables, ≤ 4 000
       constraints) and is solved with SciPy’s HiGHS backend.

    c.  Scale the LP solution `g₀` to the required ℓ¹‑norm:
            g = (√(2 n) / Σ g₀) · g₀
       This may violate the bound `M` but is harmless for the next step.

    d.  If `g` already yields a smaller evaluation score than `f`,
        replace `f ← g`.  Otherwise try a few convex combinations
        `f_new = (1‑t)·f + t·g` with `t ∈ {0.2,0.4,0.6,0.8}` and keep the
        best improving candidate.

    e.  Stop the inner loop after a fixed number of iterations,
        when the time budget expires, or when the score falls below the
        target `1.503`.

3.  Run the whole procedure for several lengths (500 … 2000) and a few
    random restarts.  Keep the best sequence found overall.

4.  Finally perform a short stochastic hill‑climb (Gaussian perturbation
    with soft‑max re‑projection) to polish the candidate.

The method is fundamentally different from the previous smooth‑max / L‑BFGS
approach and exploits the exact LP formulation of the “increase the
ℓ¹‑norm while respecting the current convolution bound” step described in
the problem statement.

The function `propose_candidate` conforms to the required signature and
returns a list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense (2n‑1) × n matrix A such that A @ g = np.convolve(f, g).
    For each column j, the constant value f[j] appears on rows j … j+n‑1.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill column‑wise – this is cheap for n up to a few thousand.
    for j in range(n):
        A[j:j + n, j] = f[j]
    return A


def _lp_optimize_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        maximise Σ g_i
        s.t.   (f * g)_k ≤ M   for all convolution indices k
               g_i ≥ 0
    Returns the optimal vector `g` (not yet scaled to the target ℓ¹‑norm).
    """
    n = f.shape[0]
    A = _convolution_matrix(f)                # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises, so use -c to maximise Σ g_i.
    c = -np.ones(n, dtype=np.float64)

    # HiGHS is fast and handles dense problems well.
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 5.0})
    if not res.success:
        # In the unlikely event of failure, fall back to the current f.
        return f.copy()
    return res.x


def _improve_sequence_via_lp(f: np.ndarray,
                             target_sum: float,
                             max_iters: int,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Apply the alternating LP improvement described in the module docstring.
    Returns the best sequence encountered and its exact `evaluate_sequence`
    value.
    """
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    n = f.shape[0]

    # Mixing parameters – a small set of trial step sizes.
    trial_ts = (0.2, 0.4, 0.6, 0.8)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # Current bound from self‑convolution.
        conv_self = np.convolve(best_f, best_f)
        M = conv_self.max()

        # Solve the LP to maximise ℓ¹‑norm under the bound.
        g0 = _lp_optimize_sum(best_f, M)

        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break  # degenerate, abort this run

        # Rescale to the required ℓ¹‑norm.
        scale = target_sum / sum_g0
        g = g0 * scale

        # Ensure we stay inside the hard caps imposed by `evaluate_sequence`.
        g = np.clip(g, 0.0, 1000.0)

        # Direct replacement test.
        score_g = evaluate_sequence(g.tolist())
        if score_g < best_score:
            best_f = g
            best_score = score_g
            # print progress
            print(f"[LP] iter={it} replace score={score_g:.6f}")
            if best_score < 1.503:
                break
            continue  # try again from the improved point

        # Otherwise try convex combinations.
        improved = False
        for t in trial_ts:
            f_new = (1.0 - t) * best_f + t * g
            # Re‑normalise the ℓ¹‑norm (important because convex combo may drift).
            f_new = f_new * (target_sum / f_new.sum())
            f_new = np.clip(f_new, 0.0, 1000.0)

            score_new = evaluate_sequence(f_new.tolist())
            if score_new < best_score:
                best_f = f_new
                best_score = score_new
                improved = True
                print(f"[LP] iter={it} t={t:.1f} new score={score_new:.6f}")
                if best_score < 1.503:
                    break
        if best_score < 1.503:
            break
        if not improved:
            # No improvement – stop early to avoid wasted LP solves.
            break

    return best_f, best_score


def _stochastic_refinement(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3,
                           target_sum: float = None) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climb that preserves the ℓ¹‑norm via a soft‑max
    style re‑projection (same technique used by the previous solution).
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = evaluate_sequence(best.tolist())
    step = step_initial
    S = target_sum if target_sum is not None else best.sum()

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # Re‑project onto the simplex with the same total sum S.
        # Use the soft‑max trick: p_i = exp(log(c_i/S)) normalized.
        p = np.clip(cand / S, 1e-12, None)
        cand = np.exp(np.log(p))           # essentially identity, but keeps positive
        cand = cand / cand.sum() * S
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        step *= 0.9995
        if step < 1e-8:
            step = step_initial

    return best, best_score

    """

    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.3)   # tiny safety margin

    best_seq = None

    # ------------------------------------------------------------------

    print(f"[result] best score = {best_score:.6f} (length={best_seq.size})")
    # Return a plain Python list as required by the evaluator.
    return [float(x) for x in best_seq]
