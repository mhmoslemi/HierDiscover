def _adam_optimize_sequence(init_seq, deadline, rng, max_iters=5000):
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    best_local = a.copy()
    best_score = _fast_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta = 1.0  # for smooth max
    for it in range(1, max_iters+1):
        if time.monotonic() >= deadline: break
        # compute conv and max
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        # weights
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = 0.1 * (m_hat / (np.sqrt(v_hat) + eps))
        a = a - step
        # projection
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n)
        # anneal beta
        if it % 200 == 0 and beta < 400:
            beta *= 1.2
        # jitter occasional
        if it % 500 == 0:
            i,j = rng.choice(n, size=2, replace=False)
            if a[i] > 0:
                delta = a[i] * 0.02
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)
    return best_local, best_score

"""
Search for a low‑value non‑negative step sequence.

The routine consists of three stages:

1️⃣  Exhaustive parametric sweep over simple families (exponential,
    power‑law and a product of both).  The sweep is dense around the
    theoretically optimal decay rate α≈2/n, which is known to give a
    very low max‑auto‑convolution.

2️⃣  Refinement of the current best candidate by a projected Adam‑style
    gradient descent on a smooth‑max surrogate of the max‑convolution.
    The projection enforces non‑negativity and the ℓ¹‑norm required by
    ``evaluate_sequence`` (∑a = √(2 n)).

3️⃣  A cheap random mass‑swap hill‑climb finishes the search.

All helpers are top‑level, no closures or lambda expressions are used.
The function respects the 1000 s time budget and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution helpers (FFT‐accelerated)
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return a size suitable for fast FFT convolution."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for many calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fast_score(seq: np.ndarray) -> float:
    """Exact value of evaluate_sequence after ℓ¹‑normalisation."""
    conv = np.convolve(seq, seq)          # 2 n‑1 points, cheap for our n
    return float(conv.max())


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and rescale to ℓ¹‑norm = √(2·n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / total)
    return seq


# ----------------------------------------------------------------------
# Smooth‑max surrogate utilities
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) weights for a convolution vector."""
    cmax = conv.max()
    # stability trick
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    For a_i the derivative is 2·∑_j w_{i+j}·a_j.
    This is a cross‑correlation and can be obtained by one FFT convolution.
    """
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)                     # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]                     # middle n entries
    grad = 2.0 * segment[::-1]
    return grad


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------
def _exp_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-alpha·i)."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * idx)
    return a


def _power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a


def _mixed_family(n: int, alpha: float, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-alpha·i)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p) * np.exp(-alpha * (idx - 1))
    return a


# ----------------------------------------------------------------------
# Adam‑style optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize_sequence(init_seq: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            max_iters: int = 3000) -> tuple[np.ndarray, float]:
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    best_local = a.copy()
    best_score = _fast_score(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    beta = 1.0                     # smooth‑max temperature
    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # smooth‑max and gradient
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # Adam update (learning rate tuned for our scale)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto feasible set
        a = np.clip(a, 0.0, 1000.0)
        tot = a.sum()
        if tot > 0.0:
            a = a * (target / tot)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # anneal smoothness (make surrogate sharper)
        if it % 200 == 0 and beta < 400.0:
            beta *= 1.3

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.02
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

    return best_local, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.
    The search proceeds through a dense parametric sweep (exponential,
    power‑law, mixed), Adam‑based refinement and a final cheap hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the global best (pure function – no closure)
    # ------------------------------------------------------------------
    def _update_global(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = _fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global best] n={len(seq)} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction, if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            cand = _normalize(cand)
            _update_global(cand)
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Dense parametric sweep
    # ------------------------------------------------------------------
    lengths = [500, 600, 800, 1000, 1200, 1500]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ---- exponential family (focus around α≈2/n) ----
        alpha_opt = 2.0 / n
        alphas = np.unique(
            np.concatenate((
                np.linspace(alpha_opt / 5, alpha_opt * 5, 60),
                np.linspace(0.001, 0.25, 30)
            ))
        )
        alphas = alphas[(alphas > 0) & (alphas < 10)]

        for alpha in alphas:
            if time.monotonic() >= deadline:
                break
            a = _exp_family(n, alpha)
            a = _normalize(a)
            _update_global(a)

        # ---- power‑law family ----
        ps = np.linspace(0.05, 5.0, 40)
        for p in ps:
            if time.monotonic() >= deadline:
                break
            a = _power_family(n, p)
            a = _normalize(a)
            _update_global(a)

        # ---- mixed (exponential × power‑law) ----
        for _ in range(30):
            if time.monotonic() >= deadline:
                break
            # sample parameters on log‑linear scales around the guessed optimum
            alpha = np.exp(rng.uniform(np.log(alpha_opt / 10), np.log(alpha_opt * 10)))
            p = rng.uniform(0.0, 5.0)
            a = _mixed_family(n, alpha, p)
            a = _normalize(a)
            _update_global(a)

    # ------------------------------------------------------------------
    # 2️⃣  Refine the current best with Adam optimiser (several restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n_best = best_seq.shape[0]
        for restart in range(5):
            if time.monotonic() >= deadline:
                break
            # small random perturbation of the current best
            start = best_seq.copy()
            i, j = rng.choice(n_best, size=2, replace=False)
            if start[i] > 0.0:
                delta = start[i] * 0.03
                start[i] -= delta
                start[j] += delta
                start = _normalize(start)

            seq_opt, _ = _adam_optimize_sequence(start, deadline, rng,
                                                 max_iters=2500)
            _update_global(seq_opt)

            if best_score < 1.503:
                print("[early exit] target reached during refinement")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 3️⃣  Final cheap random‑mass‑swap hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None:
        while time.monotonic() < deadline:
            n = best_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            delta = best_seq[i] * 0.02
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _normalize(cand)
            _update_global(cand)

            if best_score < 1.503:
                print("[early exit] target reached in final hill‑climb")
                break
    else:
        # fallback: uniform vector (should never happen)
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # --------------------------------------------------------------
    # Return a plain Python list as required by the outer harness
    # --------------------------------------------------------------
    return [float(x) for x in best_seq]
