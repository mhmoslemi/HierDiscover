def fft_convolve(a,b):
   L = 1 << ((len(a)+len(b)-2).bit_length())
   A = np.fft.rfft(a, n=L)
   B = np.fft.rfft(b, n=L)
   C = A * B
   conv = np.fft.irfft(C, n=L)[:len(a)+len(b)-1]
   return conv

def upsample(seq, new_n):
    if len(seq) == new_n:
        return seq.copy()
    old_x = np.linspace(0, 1, len(seq))
    new_x = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq

def _upsample(seq, new_n):
    old_n = len(seq)

    return np.interp(new_x, old_x, seq)

"""
Optimisation of a non‑negative step function (a_0,…,a_{n‑1}) to minimise

    2·n·max(a*a) / (∑a)^2 .

Because the objective is scale‑invariant we can fix the ℓ¹‑norm
∑a = √(2 n).  The problem is then “ minimise the peak of the
auto‑convolution ”.

The search proceeds as follows

1.  Initialise from the previously‑found construction (global variable
    ``height_sequence_1``).  If it exists we keep its shape, otherwise a
    uniform start is used.

2.  A schedule of increasing lengths is explored
    (500, 800, 1000, 1200, 1500, 1800, 2000).  For a new length the best
    sequence found so far is up‑sampled by linear interpolation and
    renormalised to the required ℓ¹‑norm.

3.  For each length many short “restarts’’ are performed.
    Each restart runs an Adam optimiser on a smooth‑max surrogate of the
    true max‑norm:
       •  The auto‑convolution is evaluated with a real FFT
          (O(n log n)).
       •  A log‑sum‑exp (soft‑max) with an annealing parameter β turns
          the max into a differentiable weighted sum.
       •  The weighted gradient is obtained analytically and also via
          FFT (convolution with the weight vector).
       •  After the Adam step the iterate is projected onto the simplex
          {a_i≥0, Σa_i = √(2 n)} and forced to be symmetric
          (a_i = a_{n‑1‑i}) – symmetry is known to lower the peak.
       •  β is slowly increased (up to 1024) so the optimisation focuses
          more and more on the true maximum.

4.  The best candidate over all lengths is finally refined by a cheap
    Gaussian hill‑climbing phase that respects the simplex constraints.

All helper functions are top‑level, no lambdas or closures are used.
The routine respects a global time budget (default 1000 s) and prints
progress information.  The returned list contains only non‑negative
floats, ready for ``evaluate_sequence``.

import time
import math

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact score (identical to evaluate_sequence when the sum is fixed)
# ----------------------------------------------------------------------
def _compute_score(v: np.ndarray) -> float:
    """Return 2·n·max(v*v)/(∑v)² – the exact objective."""
    n = len(v)
    conv = np.convolve(v, v)          # length 2·n‑1
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


# ----------------------------------------------------------------------
# Fast convolution helpers (real FFT, power‑of‑two lengths)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


def _auto_convolve_fft(a: np.ndarray) -> np.ndarray:
    """Return a*a (length 2·n‑1) using real FFT."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    # tiny negative values may appear because of rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


def _weighted_gradient_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    w must have length 2·n‑1 and Σw = 1.
    Returns a vector of length n.
    """
    n = len(a)
    w_rev = w[::-1]
    L = _next_pow2(3 * n - 2)               # length needed for a * w_rev
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)
    conv_full = np.fft.irfft(A * W, n=L)[: 3 * n - 2]
    # entries n‑1 … 2·n‑2 (inclusive) constitute the needed segment
    segment = conv_full[n - 1 : 2 * n - 1]   # length n
    grad = 2.0 * segment[::-1]              # reverse to obtain Σ_j a_j w_{i+j}
    return grad


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) weights
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Return softmax weights w_k = exp(β·(conv_k−max(conv))) / Σ…
    Guarantees Σ w = 1 and numerical stability.
    """
    m = np.max(conv)
    exp_vals = np.exp(beta * (conv - m))
    s = np.sum(exp_vals)
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max objective
# ----------------------------------------------------------------------
def _adam_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2000,
    beta_start: float = 0.5,
    beta_factor: float = 1.6,
    beta_max: float = 1024.0,
    lr: float = 0.5,
    betas: tuple[float, float] = (0.9, 0.999),
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(a*a).
    Returns the best vector seen and its exact score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)

    beta = beta_start
    no_improve = 0
    t = 0

    while time.monotonic() < deadline and t < max_iters:
        t += 1

        # auto‑convolution and smooth‑max weights
        conv = _auto_convolve_fft(f)
        w = _softmax_weights(conv, beta)

        # weighted gradient (exact, via FFT)
        grad = _weighted_gradient_fft(f, w)

        # Adam update
        m = betas[0] * m + (1.0 - betas[0]) * grad
        v = betas[1] * v + (1.0 - betas[1]) * (grad * grad)

        m_hat = m / (1.0 - betas[0] ** t)
        v_hat = v / (1.0 - betas[1] ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = f - step
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        # enforce exact symmetry (helps to keep the peak low)
        cand = (cand + cand[::-1]) * 0.5

        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best_f, best_score = cand.copy(), cand_score
            f = cand
            lr *= 1.02
            no_improve = 0
        else:
            f = cand
            lr *= 0.7
            no_improve += 1

        # anneal β towards the true max
        beta = min(beta * beta_factor, beta_max)

        if no_improve >= 200:
            # broaden the search a little
            beta = max(beta_start, beta / 2.0)
            no_improve = 0

        if lr < 1e-12:
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Simple up‑sampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Small Gaussian hill‑climbing on the simplex
# ----------------------------------------------------------------------
def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.15,
    max_iters: int = 8000,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand = (cand + cand[::-1]) * 0.5  # keep symmetry
        cand_score = _compute_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        sigma *= 0.9995
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial
    return best, best_score


# ----------------------------------------------------------------------
# Public entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low value of
    evaluate_sequence(...).  The routine respects a total time budget
    (default 1000 s) and returns a plain Python list of non‑negative
    floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # small safety margin for the outer harness
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start from the previously saved construction, if any.
    # ------------------------------------------------------------------
    init_seq = None
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        init_seq = np.array(h1, dtype=np.float64)
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        n0 = len(init_seq)
        target0 = math.sqrt(2.0 * n0)
        s0 = init_seq.sum()
        if s0 > 0:
            init_seq = init_seq * (target0 / s0)
    else:
        # fallback: uniform 500‑step start
        n0 = 500
        init_seq = np.full(n0, 1.0, dtype=np.float64)
        init_seq = init_seq * (math.sqrt(2.0 * n0) / n0)

    # ensure symmetry from the beginning
    init_seq = (init_seq + init_seq[::-1]) * 0.5

    best_seq = init_seq.copy()
    best_score = _compute_score(best_seq)
    print(f"[init] length={len(best_seq)} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Main multi‑length search
    # ------------------------------------------------------------------
    lengths = [500, 800, 1000, 1200, 1500, 1800, 2000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # up‑sample current best to the new length
        base_seq = _upsample(best_seq, n) if len(best_seq) != n else best_seq.copy()
        base_seq = np.clip(base_seq, 0.0, 1000.0)
        s = base_seq.sum()
        if s > 0:
            base_seq = base_seq * (target_sum / s)
        else:
            base_seq = np.full(n, target_sum / n, dtype=np.float64)

        # enforce symmetry before the restarts
        base_seq = (base_seq + base_seq[::-1]) * 0.5

        # crude budget split: larger n gets fewer restarts
        max_restarts = 8 if n <= 1000 else 4

        for rs in range(max_restarts):
            if time.monotonic() >= deadline:
                break

            if rs == 0:
                f0 = base_seq.copy()
            else:
                f0 = rng.random(n) + 1e-8
                f0 = f0 / f0.sum() * target_sum
                f0 = (f0 + f0[::-1]) * 0.5

            sub_deadline = min(deadline, time.monotonic() + 12.0)

            f_opt, sc_opt = _adam_optimize(
                f0,
                target_sum,
                sub_deadline,
                max_iters=2500,
                beta_start=0.5,
                beta_factor=1.6,
                beta_max=1024.0,
                lr=0.6,
                betas=(0.9, 0.999),
                eps=1e-8,
            )

            if sc_opt < best_score:
                best_seq, best_score = f_opt.copy(), sc_opt
                print(f"[update] n={n} restart={rs} score={sc_opt:.6f}")

                if best_score < 1.503:
                    print("[early] target achieved – exiting early")
                    deadline = time.monotonic() + 0.2
                    break

        # early stop if target already met
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing with a cheap stochastic hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        refined, refined_sc = _hill_climb(
            best_seq,
            target_sum,
            deadline,
            sigma_initial=0.15,
            max_iters=8000,
        )
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[refine] final score={refined_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Return the best found sequence
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
