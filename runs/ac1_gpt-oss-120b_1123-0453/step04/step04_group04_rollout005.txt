def _smooth_max_and_grad(a, alpha):
    n = a.shape[0]
    # conv
    fft_len = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    # softmax weights
    shifted = conv - max_val  # negative numbers
    # any potential underflow? but fine.
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0/len(w))
    else:
        w /= w_sum
    # compute gradient via correlation
    w_rev = w[::-1]
    # conv length for correlation: n + m - 1
    conv_len_full = 1 << ((n + (2*n-1) - 1).bit_length())
    fa2 = np.fft.rfft(a, n=conv_len_full)
    fw = np.fft.rfft(w_rev, n=conv_len_full)
    corr_full = np.fft.irfft(fa2 * fw, n=conv_len_full)

    grad = 2.0 * corr_full[start - np.arange(n)]

"""
Projected‑gradient optimizer with a smooth maximum surrogate.

The method keeps the ℓ¹‑norm fixed (‖a‖₁ = √(2 n)) and repeatedly
minimises a smooth approximation of maxₖ (a*a)[k] using
a soft‑max weighting (parameter α).  The gradient of the surrogate
is obtained analytically via a correlation (implemented with FFTs).
After each gradient step the vector is projected onto the simplex
to enforce non‑negativity and the fixed L¹‑norm.
A schedule of increasing α refines the solution towards the true
ℓ∞‑objective.  Several sequence lengths and random restarts are
explored within the time budget; the best sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_needed(min_len: int) -> int:
    """Return the next power‑of‑two length ≥ min_len (for real FFT)."""
    return 1 << ((min_len - 1).bit_length())


def _fft_max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution a*a using an FFT."""
    n = a.shape[0]
    fft_len = _fft_len_needed(2 * n - 1)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv[: 2 * n - 1]
    return float(conv.max())


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, ∑x = s}.
    Returned array has the same shape as v.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components will be equal
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_and_grad(a: np.ndarray, alpha: float):
    """
    For a given vector a (non‑negative, fixed ℓ¹‑norm) and smoothing
    parameter α > 0, compute

        m = maxₖ (a*a)[k]                     (the true maximum)
        wₖ = exp(α·((a*a)[k] – m)) / Σₖ ...

    and the gradient of the smooth maximum:
        ∇ = 2·corr(a, w)   (correlation, not convolution).

    Returns (m, ∇) where ∇ is a NumPy array of the same shape as a.
    """
    n = a.shape[0]
    # ----- true auto‑convolution (needed for the max) -----
    fft_len = _fft_len_needed(2 * n - 1)
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv_full[: 2 * n - 1]          # length 2n‑1
    m_val = conv.max()

    # ----- soft‑max weights -----
    # shift by the maximum for numerical stability
    shifted = conv - m_val                # ≤ 0
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:                      # extreme under‑flow → uniform
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # ----- correlation a ⋆ w (equivalently a * w_rev) -----
    w_rev = w[::-1]                       # reverse the weight vector
    corr_len = _fft_len_needed(n + w_rev.shape[0] - 1)
    fa_corr = np.fft.rfft(a, n=corr_len)
    fw_corr = np.fft.rfft(w_rev, n=corr_len)
    corr_full = np.fft.irfft(fa_corr * fw_corr, n=corr_len)

    start = w.shape[0] - 1                # = 2n‑2
    idx = start - np.arange(n)            # slice indices for the correlation
    grad = 2.0 * corr_full[idx]           # gradient of the smooth max

    return float(m_val), grad


def _optimize_sequence(a_init: np.ndarray, deadline: float):
    """
    Projected‑gradient descent on the smooth‑max surrogate.
    Returns (best_sequence, best_score) where score = maxₖ (a*a)[k].
    """
    n = a_init.shape[0]
    target_sum = a_init.sum()                     # = √(2 n) by construction
    a = a_init.copy()
    best_seq = a.copy()
    best_val = _fft_max_convolution(a)

    avg = target_sum / n
    step = avg * 0.5                               # initial step‑size

    # increasing schedule for the smoothing parameter α
    alphas = [0.4, 0.8, 1.6, 3.2, 6.4, 12.8, 25.6]

    # iterate while we have time left
    while time.monotonic() < deadline:
        for alpha in alphas:
            # limit iterations per α to keep the whole run within the deadline
            it = 0
            max_iters = 200
            while time.monotonic() < deadline and it < max_iters:
                it += 1
                # current smooth‑max value and its gradient
                cur_max, grad = _smooth_max_and_grad(a, alpha)

                # gradient step + projection onto the ℓ¹‑simplex
                a_candidate = a - step * grad
                a_candidate = _project_onto_simplex(a_candidate, target_sum)
                # values are naturally ≤ 1000 because target_sum ≈ √(2 n) ≪ 1000
                a_candidate = np.clip(a_candidate, 0.0, 1000.0)

                # true objective (maximum auto‑convolution entry)
                val_candidate = _fft_max_convolution(a_candidate)

                # keep the best ever seen
                if val_candidate + 1e-12 < best_val:
                    best_val = val_candidate
                    best_seq = a_candidate.copy()
                    print(
                        f"[improve] n={n} α={alpha:.2f} it={it} "
                        f"val={val_candidate:.6f}"
                    )

                # accept only if we made progress on the real objective
                if val_candidate < cur_max - 1e-12:
                    a = a_candidate
                else:
                    # no improvement → shrink step size
                    step *= 0.7

                # slowly decay the step size (helps convergence)
                step *= 0.995
                if step < 1e-12:
                    return best_seq, best_val
    return best_seq, best_val


def _initialize_sequence(n: int, rng: np.random.Generator, target: float,
                         warm: np.ndarray | None = None) -> np.ndarray:
    """
    Create a starting non‑negative sequence of length n with ℓ¹‑norm = target.
    If *warm* is provided and has the correct length, it is rescaled;
    otherwise a random exponential vector is used.
    """
    if warm is not None and warm.shape[0] == n:
        seq = warm.astype(np.float64)
        s = seq.sum()
        if s > 0:
            seq = seq * (target / s)
        else:
            seq = np.full(n, target / n)
        return seq

    # exponential draw gives a fairly sparse start (good for exploration)
    raw = rng.exponential(scale=1.0, size=n) + 1e-6
    seq = raw * (target / raw.sum())
    return seq
