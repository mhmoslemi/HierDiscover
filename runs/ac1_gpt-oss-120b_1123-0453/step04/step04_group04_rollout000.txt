"""Propose a low‑value non‑negative step sequence.

The algorithm explores several lengths.  For each length it runs many
random‑restart simulated‑annealing searches with a fixed ℓ¹‑norm
(‖a‖₁ = √(2 n)).  The annealer moves mass between two coordinates while
keeping the sum fixed and accepts uphill moves according to a
temperature schedule.  After the allocated time the best sequence is
polished by a projected sub‑gradient descent on the maximal
auto‑convolution entry.  The best sequence seen across all lengths is
returned.
"""

import time
import math
import numpy as np

def _fft_max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max entry of a*a using FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    needed = 2 * len(a) - 1
    return float(np.max(conv[:needed]))

def _initialize_sequence(n: int, seed_seq=None, rng: np.random.Generator = None) -> np.ndarray:
    """Create a length‑n non‑negative vector with ℓ¹‑norm √(2 n)."""
    target = math.sqrt(2.0 * n)
    if seed_seq is not None:
        seq = np.array(seed_seq, dtype=np.float64)
        if len(seq) != n:
            if len(seq) > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - len(seq))])
        seq = np.clip(seq, 0.0, 1000.0)
        s = np.sum(seq)
        if s == 0.0:
            seq = np.full(n, target / n)
        else:
            seq = seq * (target / s)
        return seq
    if rng is None:
        rng = np.random.default_rng()
    seq = rng.random(n) + 1e-6      # ensure strictly positive
    seq = seq / np.sum(seq) * target
    return seq

def _simulated_annealing(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Transfer mass between two coordinates while keeping the ℓ¹ norm fixed."""
    n = len(a)
    target_sum = np.sum(a)   # = √(2 n)
    avg = target_sum / n
    fft_len = 1 << ((2 * n - 1).bit_length())
    cur_max = _fft_max_convolution(a, fft_len)

    best_seq = a.copy()
    best_val = cur_max

    rng = np.random.default_rng()
    start = time.monotonic()
    total_time = max(1e-12, deadline - start)

    # initial temperature
    init_temp = max(cur_max * 0.5, 1e-4)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = elapsed / total_time
        # exponential cooling: final temperature ≈ 1e‑4·init_temp
        T = init_temp * (0.0001 ** frac)

        # step size proportional to temperature
        step = avg * (T / init_temp) * 3.0   # ~3·avg at start

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose a transfer amount
        delta = rng.normal(scale=step)
        # clamp to keep both entries non‑negative
        if delta > a[i]:
            delta = a[i]
        if -delta > a[j]:
            delta = -a[j]
        if abs(delta) < 1e-15:
            continue

        a[i] -= delta
        a[j] += delta

        new_max = _fft_max_convolution(a, fft_len)

        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 1e-12:
                prob = math.exp(-(new_max - cur_max) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()
        else:
            # revert move
            a[i] += delta
            a[j] -= delta

    return best_seq, best_val

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgradient_max_convolution(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximal entry of a*a."""
    n = len(a)
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-9 + 1e-12
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += a[j]
        grad[j] += a[i]
    return grad

def _polish_gradient_descent(a: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on max(a*a) with decreasing step size."""
    n = len(a)
    target_sum = np.sum(a)   # should be √(2 n)
    fft_len = 1 << ((2 * n - 1).bit_length())
    cur_max = _fft_max_convolution(a, fft_len)

    start = time.monotonic()
    while time.monotonic() < deadline:
        # compute sub‑gradient at current point
        g = _subgradient_max_convolution(a)
        norm = np.linalg.norm(g)
        if norm < 1e-14:
            break

        # simple backtracking line search
        # initial step = average mass * 0.5
        step = 0.5 * (target_sum / n)
        improved = False
        while step > 1e-12 * (target_sum / n):
            a_candidate = a - (step / (norm + 1e-12)) * g
            a_candidate = _project_onto_simplex(a_candidate, target_sum)
            a_candidate = np.clip(a_candidate, 0.0, 1000.0)
            new_max = _fft_max_convolution(a_candidate, fft_len)
            if new_max < cur_max - 1e-12:
                a = a_candidate
                cur_max = new_max
                improved = True
                break
            step *= 0.5
        if not improved:
            # no improvement achievable with current gradient → break
            break
    return a

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the time budget.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.05)   # safety margin

    # ----------------------------------------------------------------------
    # 1️⃣ Choose sequence lengths to explore.
    # ----------------------------------------------------------------------
    length_candidates = [300, 400, 500, 600, 800, 1000, 1200, 1500]
    # incorporate previously found sequence if available
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        n0 = len(h1)
        if n0 not in length_candidates:
            length_candidates.append(n0)
    length_candidates = sorted(set(length_candidates))

    # Allocate time: 85 % for exploration (annealing), 15 % for polishing
    total_time = deadline - start_time
    explore_time = total_time * 0.85
    polish_time = total_time - explore_time

    # Give more budget to lengths that historically work well.
    # Heuristic weights (larger length → more flexibility).
    weights = {300: 1, 400: 1, 500: 3, 600: 1, 800: 2, 1000: 1, 1200: 1, 1500: 1}
    total_weight = sum(weights.get(n, 1) for n in length_candidates)

    best_seq = None
    best_val = float("inf")
    best_len = None

    # ----------------------------------------------------------------------
    # 2️⃣ Exploration phase – simulated annealing with many restarts.
    # ----------------------------------------------------------------------
    time_spent = time.monotonic() - start_time
    for n in length_candidates:
        # time allocated for this length
        alloc = explore_time * (weights.get(n, 1) / total_weight)
        sub_deadline = min(deadline, time.monotonic() + alloc)

            seed_seq = best_seq if best_len == n else None
            a0 = _initialize_sequence(n, seed_seq=seed_seq, rng=rng)

            # Give the annealer the remainder of the sub‑deadline (or a hard cap)
            inner_deadline = min(sub_deadline, deadline)
            best_local, val_local = _simulated_annealing(a0, inner_deadline)

            if val_local < best_val:
                best_val = val_local
                best_seq = best_local.copy()
                best_len = n
                print(f"[global update] n={n} value={val_local:.6f}")

            # exit early if we are close to the overall deadline
            if time.monotonic() >= deadline:
                break

    # ----------------------------------------------------------------------
    # 3️⃣ Polishing phase – projected sub‑gradient descent.
    # ----------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()

        polish_deadline = time.monotonic() + min(polish_time, remaining)
        best_seq = _polish_gradient_descent(best_seq, polish_deadline)
        best_val = _fft_max_convolution(best_seq,
                                        1 << ((2 * len(best_seq) - 1).bit_length()))
        print(f"[polish] n={len(best_seq)} value={best_val:.6f}")

    # ----------------------------------------------------------------------
    # 4️⃣ Final normalisation and safety checks.
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform sequence of modest length
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
    else:
        n = len(best_seq)
        target = math.sqrt(2.0 * n)
        cur_sum = best_seq.sum()
        if cur_sum > 0:
            best_seq = best_seq * (target / cur_sum)
        else:
            best_seq = np.full(n, target / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

        best_seq = _polish_gradient_descent(best_seq, polish_deadline)
        best_val = _fft_max_convolution(
            best_seq,
            1 << ((2 * len(best_seq) - 1).bit_length())
        )
        print(f"[polish] n={len(best_seq)} value={best_val:.6f}")

    # ----------------------------------------------------------------------
    # 4️⃣ Final normalisation and safety checks
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback: simple uniform vector of moderate length
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
    else:
        n = len(best_seq)
        target = math.sqrt(2.0 * n)
        cur_sum = best_seq.sum()
        if cur_sum > 0:
            best_seq = best_seq * (target / cur_sum)
        else:
            best_seq = np.full(n, target / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
