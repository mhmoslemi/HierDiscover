"""
Optimise step sequence via multi‑scale soft‑max L‑BFGS‑B + up‑sampling.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """Map unrestricted vector z to a non‑negative vector a on the ℓ¹‑simplex
    with sum = target_sum using the soft‑max parametrisation."""
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z) * target_sum
    return a

def _obj_and_grad(z: np.ndarray, n: int, S: float, beta: float):
    """
    Smooth approximation of max_{k} (a * a)[k] using log‑sum‑exp.
    Returns (smooth_obj, grad_z) where grad_z = ∂smooth/∂z .
    """
    a = _softmax_to_a(z, S)                     # shape (n,)

    # convolution a*a via FFT
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]

    # log‑sum‑exp smooth max
    m = np.max(conv)
    exps = np.exp(beta * (conv - m))
    sum_exps = np.sum(exps)
    smooth = m + np.log(sum_exps) / beta
    w = exps / sum_exps                         # weights, sum to 1

    # gradient wrt a : g_a = 2 * (a_rev * w)_{n-1 … n-1+n-1}
    a_rev = a[::-1]
    L2 = _next_pow2(3 * n - 2)
    fr = np.fft.rfft(a_rev, L2)
    fw = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(fr * fw, L2)          # length L2
    start = n - 1
    g_raw = conv2[start:start + n]
    g_a = 2.0 * g_raw

    # chain rule: dΦ/dz_i = a_i * (g_a_i - (a·g_a)/S)
    total = np.dot(g_a, a)
    grad_z = a * (g_a - total / S)
    return float(smooth), grad_z.astype(np.float64)

def _fun_and_jac(z, n, S, beta):
    """Wrapper for scipy.optimize that returns (f, grad)."""
    return _obj_and_grad(z, n, S, beta)

def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability distribution onto a new length."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    # work with probabilities
    S_old = np.sum(seq)
    if S_old == 0:
        prob = np.full(old_n, 1.0 / old_n)
    else:
        prob = seq / S_old
    # node positions
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    prob_new = np.interp(new_x, old_x, prob)
    # ensure sum = 1 (renormalise)
    prob_new = prob_new / np.sum(prob_new)
    S_new = math.sqrt(2.0 * new_n)
    a_new = prob_new * S_new
    # clip to the limits required by evaluate_sequence
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

def _optimise_length(n: int,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length `n` with sum `target_sum`.
    Returns best sequence and its exact evaluate_sequence score.
    """
    # schedule of smoothness parameters β (larger -> tighter to real max)
    betas = [5.0, 15.0, 40.0, 80.0, 200.0, 500.0, 1000.0]
    max_restarts = 6
    best_seq = None
    best_val = float("inf")

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # initialise z
        if warm_seq is not None and warm_seq.shape[0] == n:
            # warm start from known sequence (soft‑max inverse)
            p = np.clip(warm_seq / target_sum, 1e-12, None)
            z0 = np.log(p)
        else:
            # random normal start
            z0 = rng.normal(scale=0.2, size=n)

        # perform continuation over β
        for beta in betas:
            if time.monotonic() >= deadline:
                break

            res = minimize(fun=_fun_and_jac,
                           args=(n, target_sum, beta),
                           x0=z0,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300,
                                    'gtol': 1e-9,
                                    'disp': False})
            z0 = res.x

            # evaluate exact objective (which is the true max convolution)
            a = _softmax_to_a(z0, target_sum)
            a = np.clip(a, 0.0, 1000.0)
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[opt] n={n} restart={restart} β={beta:.0f} val={val:.6f}")

            if best_val < 1.48:   # far better than target, stop early
                return best_seq, best_val

    # fallback uniform distribution if nothing else
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] n={n} uniform val={best_val:.6f}")

    return best_seq, best_val

def _stochastic_refinement(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.4,
                           decay: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that respects the ℓ¹‑simplex.
    The perturbation is projected back using the soft‑max trick.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = evaluate_sequence(best.tolist())
    step = step_initial
    S = best.sum()

    while time.monotonic() < deadline:
        # propose Gaussian move in the space of the underlying logits
        # we work directly on the vector, then re‑project to simplex.
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # project onto simplex via soft‑max with the same target sum
        cand = _softmax_to_a(np.log(np.clip(cand / S, 1e-12, None)), S)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= decay
        if step < 1e-8:
            step = step_initial

    return best, best_score

"""
Optimise a non‑negative step sequence.

The algorithm builds a prototype at a moderate size, up‑samples it to a
handful of larger lengths and refines each candidate with a smooth‑max
L‑BFGS‑B continuation (soft‑max re‑parametrisation of the ℓ¹‑simplex).
A short stochastic hill‑climb finishes the search.  The implementation
avoids all lambda expressions, respects the time budget, and prints
progress information.
"""
import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no lambdas, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """Map unrestricted vector z to a non‑negative vector a on the ℓ¹‑simplex.

    The result satisfies a ≥ 0 and sum(a) = target_sum.
    """
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z) * target_sum
    return a


def _obj_and_grad(z: np.ndarray, n: int, S: float, beta: float):
    """
    Smooth approximation of max_k (a * a)[k] using log‑sum‑exp.

    Returns (smooth objective, gradient w.r.t. the soft‑max parameters z).
    """
    a = _softmax_to_a(z, S)                     # shape (n,)

    # a * a via FFT (real‑to‑real convolution)
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]

    # smooth max = (1/β) log Σ exp(β·conv)
    m = np.max(conv)
    exps = np.exp(beta * (conv - m))
    sum_exps = np.sum(exps)
    smooth = m + np.log(sum_exps) / beta
    w = exps / sum_exps                         # soft‑max weights, Σ w = 1

    # gradient w.r.t. a : g_a = 2·(a_rev * w)_{n‑1 … n‑1+n‑1}
    a_rev = a[::-1]
    L2 = _next_pow2(3 * n - 2)
    fr = np.fft.rfft(a_rev, L2)
    fw = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(fr * fw, L2)          # length L2
    start = n - 1
    g_raw = conv2[start:start + n]            # length n
    g_a = 2.0 * g_raw

    # chain rule: ∂Φ/∂z_i = a_i·(g_a_i - (a·g_a)/S)
    total = np.dot(g_a, a)
    grad_z = a * (g_a - total / S)
    return float(smooth), grad_z.astype(np.float64)


def _fun_and_jac(z, n, S, beta):
    """Wrapper for scipy.optimize: returns (objective, gradient)."""
    return _obj_and_grad(z, n, S, beta)


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability distribution onto a new length.

    The input `seq` is assumed to have non‑negative entries.
    The returned array has sum = sqrt(2·new_n) and respects the clipping limits.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()

    sum_old = np.sum(seq)
    if sum_old == 0.0:
        prob = np.full(old_n, 1.0 / old_n)
    else:
        prob = seq / sum_old

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    prob_new = np.interp(new_x, old_x, prob)
    prob_new /= np.sum(prob_new)                     # renormalise
    S_new = math.sqrt(2.0 * new_n)
    a_new = prob_new * S_new
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new


def _optimise_length(n: int,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length `n` with fixed ℓ¹‑norm `target_sum`.

    Returns the best sequence found and its exact score from `evaluate_sequence`.
    """
    betas = [5.0, 15.0, 40.0, 80.0, 200.0, 500.0, 1000.0]   # continuation schedule
    max_restarts = 6
    best_seq = None
    best_val = float("inf")

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # Initialise soft‑max parameters z
        if warm_seq is not None and warm_seq.shape[0] == n:
            # Inverse soft‑max (log‑probabilities)
            p = np.clip(warm_seq / target_sum, 1e-12, None)
            z0 = np.log(p)
        else:
            z0 = rng.normal(scale=0.2, size=n)

        # Continuation over β
        for beta in betas:
            if time.monotonic() >= deadline:
                break

            res = minimize(fun=_fun_and_jac,
                           args=(n, target_sum, beta),
                           x0=z0,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300,
                                    'gtol': 1e-9,
                                    'disp': False})
            z0 = res.x

            # Exact evaluation (true max of the convolution)
            a = _softmax_to_a(z0, target_sum)
            a = np.clip(a, 0.0, 1000.0)
            val = evaluate_sequence(a.tolist())
            if val < best_val:

                best_seq = a.copy()
                print(f"[opt] n={n} restart={restart} β={int(beta)} val={val:.6f}")

                return best_seq, best_val
