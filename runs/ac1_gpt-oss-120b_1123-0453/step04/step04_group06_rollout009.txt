Pseudo:

def _adam_optimize(f0, target_sum, deadline, max_iters=5000, learning_rate=0.1, beta1=0.9, beta2=0.999, epsilon=1e-8, beta_start=0.5, beta_factor=1.2, beta_max=256):

    f = f0.copy()
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    best_f, best_score = f.copy(), compute_score(f)
    step = 0
    for it in range(1, max_iters+1):
        if time.monotonic() >= deadline: break
        conv = np.convolve(f, f)
        max_val = conv.max()
        # compute weights
        exp_vals = np.exp(beta_start * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # weighted gradient with convolution method
        # compute convolution w * f_rev using FFT or np.convolve
        f_rev = f[::-1]
        g_full = np.convolve(w, f_rev)
        # slice
        g = g_full[len(f)-1:len(f)-1+len(f)]
        diag = w[2*np.arange(len(f))]
        grad = 2 * g - diag * f
        # Adam updates
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        update = learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        cand = f - update
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0) # enforce bounds
        cand_score = compute_score(cand)
        if cand_score < best_score - 1e-12:
            best_f, best_score = cand.copy(), cand_score
        f = cand
        # anneal beta
        beta_start = min(beta_max, beta_start * beta_factor)
        # optional reduce learning_rate if no improvement for many steps
    return best_f, best_score

    t = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        max_val = conv.max()
        # weights
        # to avoid overflow: subtract max_val
        w = np.exp(beta * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / conv.size)
        else:
            w /= w_sum

        grad = _weighted_gradient(f, w)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        cand = f - step
        # project onto simplex
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = compute_score(cand)
        if cand_score < best_score - 1e-12:
            best_f, best_score = cand.copy(), cand_score
        f = cand

        # anneal beta
        beta = min(beta_max, beta * beta_factor)
        # optional: if no improvement for many steps reduce lr
        # we could keep simple.

    return best_f, best_score

```python
"""
Optimiser for the step‑function peak bound.
The algorithm is a three‑stage heuristic:

1.  Many random (and Gaussian) restarts are optimised by Adam on a
    smooth‑max approximation of the convolution peak.
2.  The best candidate of each restart is refined by a sub‑gradient
    descent that works on the true maximal convolution entry.
3.  A short stochastic hill‑climb finishes the search.

The optimiser respects the ℓ¹‑simplex (∑aᵢ = √(2 n), aᵢ ≥ 0) used by the
evaluation function, projects onto the simplex after each update and
clips to the interval required by ``evaluate_sequence``.  Several
problem sizes (300 … 1200) are tried while staying inside the global
time budget; the best sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be equal
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a given sequence."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_val = np.max(conv)
    sum_val = np.sum(seq)
    if sum_val < 0.01:
        return float("inf")
    # This is the same as the original formula, but we keep the full expression
    # to avoid any accidental mismatch with the official evaluator.
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _weighted_gradient(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(f*f)_k w.r.t. f.
    Parameters
    ----------
    f : np.ndarray, shape (n,)
        Current vector (non‑negative, sum fixed).
    w : np.ndarray, shape (2·n‑1,)
        Non‑negative weights that sum to one.
    Returns
    -------
    grad : np.ndarray, shape (n,)
        ∂/∂f Σ_k w_k·(f*f)_k.
    """
    n = len(f)
    # convolution of w with the reversed f
    g_full = np.convolve(w, f[::-1])
    # The part that corresponds to indices 0…n‑1 is the slice [n‑1:n‑1+n]
    start = n - 1
    g = g_full[start:start + n]
    # diagonal correction (i == j) appears when k is even and i = k/2
    diag = w[2 * np.arange(n)]
    grad = 2.0 * g - diag * f
    return grad


def _adam_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 1500,
    learning_rate: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
    beta_start: float = 0.5,
    beta_factor: float = 1.5,
    beta_max: float = 256.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smoothed‑max proxy of the auto‑convolution peak.

    The objective at iteration t is   φ_t(f) = Σ_k w_k·(f*f)_k,
    where w_k = softmax(β_t·conv_k).  As β_t grows the proxy approaches the
    true maximum.
    """
    f = f0.copy()
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    best_f = f.copy()
    best_score = _compute_score(f)
    step = 0
    beta = beta_start

    for it in range(1, max_iters + 1):

            break

        conv = np.convolve(f, f)
        max_val = conv.max()
        # stable soft‑max
        exp_vals = np.exp(beta * (conv - max_val))
        w_sum = exp_vals.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_vals / w_sum

        # gradient of the smooth‑max objective
        grad = _weighted_gradient(f, w)

        # Adam update
        step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        update = learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = f - update
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _compute_score(cand)
        if cand_score + 1e-12 < best_score:
            best_f, best_score = cand.copy(), cand_score

        f = cand

        # anneal the smooth‑max temperature
        beta = min(beta_max, beta * beta_factor)

    return best_f, best_score


def _max_subgradient_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 500,
    step_start: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum of the autocorrelation.
    Only the entry that attains the current maximum is used to build the
    gradient (i.e. a piece‑wise linear approximation of the max‑function).
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)
    step = step_start
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        k = int(conv.argmax())          # index of the maximal entry

        n = len(f)
        grad = np.zeros_like(f)

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        grad[i_range] += 2.0 * f[j_range]

        if (k & 1) == 0:               # k even → possible diagonal term
            i_eq = k // 2
            if i_start <= i_eq <= i_end:
                grad[i_eq] -= f[i_eq]

        # Normalise the direction to avoid dependence on the gradient norm.
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        direction = grad / norm

        cand = f - step * direction
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _compute_score(cand)
        if cand_score + 1e-12 < best_score:

            f = cand
            step *= 1.02
            no_improve = 0
        else:
            step *= 0.70
            no_improve += 1

        if step < 1e-12:
            break
        if no_improve >= 200:
            step = step_start
            no_improve = 0

    return best_f, best_score


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.2,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:

    improvement.
    """

    sigma = sigma_initial

        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        sigma *= 0.9995
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial
    return best, best_score

Optimiser for the step‑function peak bound.

The algorithm consists of three stages:

1️⃣  Many restarts (random, Gaussian‑shaped, or up‑sampled warm‑starts)

    peak.  The smoothness parameter β is annealed, turning the proxy

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto {x | xᵢ ≥ 0, Σxᵢ = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_score(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a given sequence."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_val = np.max(conv)
    sum_val = np.sum(seq)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _weighted_gradient(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σₖ wₖ·(f∗f)ₖ with respect to ``f``.
    ``w`` must have length 2·len(f)-1 and Σwₖ = 1.
    """
    n = len(f)
    # (w * reversed(f))  → indices n‑1 … n‑1+n‑1 give the needed sums

    g = g_full[start:start + n]
    diag = w[2 * np.arange(n)]                 # contribution when i=j
    return 2.0 * g - diag * f

                if sc_opt2 < sc_opt:
                    f_opt, sc_opt = f_opt2, sc_opt2

            if sc_opt < best_score - 1e-12:
                best_seq, best_score = f_opt.copy(), sc_opt
                print(f"[update] n={n} restart={rs} score={sc_opt:.6f}")

            # Early stop if we have breached the target bound.
            if best_score < 1.503:
                print("[early] target reached – terminating search early")
                deadline = time.monotonic() + 0.2   # tiny buffer for polishing
                break

    # --------------------------------------------------------------
    # 2️⃣ Final polishing – short hill‑climb on the best candidate.
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        refined, refined_sc = _hill_climb(
            best_seq,
            target_sum,
            deadline,
            sigma_initial=0.15,
            max_iters=5000,
        )
        if refined_sc < best_score - 1e-12:
            best_seq, best_score = refined, refined_sc
            print(f"[refine] final score={refined_sc:.6f}")

    # ------------------------------------------------------------------
    # Return the result (already clipped to the evaluator's limits).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
