"""
Proposes a low‑value non‑negative step sequence.

Algorithm
---------
* Work on the ℓ¹‑simplex via a soft‑max re‑parameterisation so that the
  sum of the coefficients is fixed to √(2 n).  For a given length n the
  objective reduces to minimising the infinity‑norm of the auto‑convolution
  (the value returned by `evaluate_sequence` is exactly that norm).

* For each length n we run several restarts of a smooth approximation of
  the max‑convolution:
    - The smooth surrogate is the log‑sum‑exp of the convolution with a
      temperature β (β grows during a continuation schedule, ending at 1000
      for a very tight approximation).
    - Gradients of the surrogate w.r.t. the soft‑max parameters are computed
      analytically by means of FFT‑based convolutions.
    - L‑BFGS‑B (from SciPy) is used for fast quasi‑Newton optimisation.

* The best sequence found for a length is used (by linear interpolation)
  as a warm‑start for the next, larger length.  This lets the optimiser
  inherit useful structure while exploring larger search spaces.

* After the deterministic phase a short stochastic hill‑climbing phase
  (Gaussian perturbations followed by a soft‑max re‑projection) further
  refines the best candidate.

* The routine respects a global time budget (≈ budget_s seconds) and
  returns the best sequence discovered before the deadline.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Convert an unrestricted vector z to a point a on the simplex
    with sum = target_sum using the soft‑max map.
    """
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.exp(z - max_z).sum() * target_sum
    return a


def _obj_and_grad(z: np.ndarray, n: int, S: float, beta: float):
    """
    Smooth surrogate of max(a*a) and its gradient w.r.t. the soft‑max
    parameters z.

    Returns (smooth_value, grad_z) where
        smooth_value = (1/β)·log Σ exp(β·conv)
    and grad_z is the gradient of this value w.r.t. z.
    """
    # a from soft‑max
    a = _softmax_to_a(z, S)               # shape (n,)

    # ---- auto‑convolution a*a via FFT ---------------------------------
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]   # length 2n‑1, real

    # ---- smooth max (log‑sum‑exp) ------------------------------------
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))          # stable
    sum_exps = np.sum(exps)
    smooth = max_c + np.log(sum_exps) / beta

    # soft‑max weights for the convolution entries
    w = exps / sum_exps                            # ∑ w = 1

    # ---- gradient wrt a : g_a = 2·(a_rev * w)_{n … n+n‑1} ----------
    a_rev = a[::-1]
    L2 = _next_pow2(3 * n - 2)
    fr = np.fft.rfft(a_rev, L2)
    fw = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(fr * fw, L2)              # length L2
    start = n - 1
    g_raw = conv2[start : start + n]              # length n
    g_a = 2.0 * g_raw

    # ---- chain rule to obtain grad_z ---------------------------------
    # dΦ/dz_i = a_i·g_a_i - a_i·(a·g_a)/S
    total = np.dot(g_a, a)
    grad_z = a * g_a - a * (total / S)

    return smooth, grad_z.astype(np.float64)


def _obj_and_grad_wrapper(z, n, S, beta):
    """
    Wrapper required by scipy.optimize.minimize (passes extra args).
    """
    return _obj_and_grad(z, n, S, beta)


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linearly interpolate a sequence to a new length.
    The returned array has the same total sum as the input (up to numerical error).
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    # map points to [0, 1)
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq


def _optimise_length(n: int,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n using the smooth surrogate & L‑BFGS‑B.
    Returns the best sequence found (as a plain NumPy array) and its exact
    `evaluate_sequence` score.
    """
    # Continuation schedule – larger β gives a tighter approximation of max.
    betas = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 200.0, 500.0, 1000.0]
    max_restarts = 8
    best_seq = None
    best_val = float("inf")

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # initialise z (soft‑max pre‑image)
        if warm_seq is not None and warm_seq.shape[0] == n:
            p = np.clip(warm_seq / target_sum, 1e-12, None)
            z0 = np.log(p)
        else:
            z0 = rng.normal(scale=0.1, size=n)

        # run the β‑homotopy
        for beta in betas:
            if time.monotonic() >= deadline:
                break

            res = minimize(_obj_and_grad_wrapper,
                           z0,
                           args=(n, target_sum, beta),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 400,
                                    'gtol': 1e-10,
                                    'disp': False})
            z0 = res.x

            # evaluate the exact (non‑smoothed) objective
            a = _softmax_to_a(z0, target_sum)
            a = np.clip(a, 0.0, 1000.0)
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[opt] n={n} restart={restart} β={beta:.0f} val={val:.6f}")

            # occasional random jitter to escape shallow local minima
            if rng.random() < 0.07:
                z0 = z0 + rng.normal(scale=0.02, size=n)

            # early exit if we already beat the target
            if best_val < 1.503:
                return best_seq, best_val

    # fallback – uniform distribution (should never happen)
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())

    return best_seq, best_val


def _stochastic_refinement(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that respects the ℓ¹‑simplex
    by re‑projecting with the same soft‑max map used in the deterministic phase.
    Returns the refined sequence and its exact score.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = evaluate_sequence(best.tolist())
    step = step_initial
    S = best.sum()

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        # re‑project onto the simplex using soft‑max
        cand = _softmax_to_a(np.log(np.clip(cand / S, 1e-12, None)), S)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        step *= 0.9995
        if step < 1e-8:
            step = step_initial

    return best, best_score


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.
    The function respects the global time budget (≈ budget_s seconds)
    and returns a plain Python list of floats ready for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the final print / return
    deadline = start_time + max(0.0, float(budget_s) - 0.1)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        n0 = seq.size
        if n0 > 0 and seq.sum() > 0.0:
            S0 = math.sqrt(2.0 * n0)
            seq = seq * (S0 / seq.sum())
        else:
            # fallback to uniform
            seq = np.full(n0, math.sqrt(2.0 * n0) / n0, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        val = evaluate_sequence(seq.tolist())
        best_seq = seq.copy()
        best_score = val
        print(f"[init] warm‑start n={n0} val={val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Deterministic optimisation on a range of lengths
    # ------------------------------------------------------------------
    # We start with moderate sizes and gradually increase them.
    lengths = [400, 600, 800, 1200, 1800, 2500, 3500, 5000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # Build a warm‑start of the right size (interpolation + renormalisation)
        if best_seq is not None:
            warm = _resize_sequence(best_seq, n)
            # renormalise to the required ℓ¹‑norm
            warm = warm * (target_sum / warm.sum())
        else:
            warm = None

        seq_n, val_n = _optimise_length(n, target_sum, deadline, rng, warm)

        if val_n < best_score:
            best_score = val_n
            best_seq = seq_n.copy()
            print(f"[global update] n={n} val={val_n:.6f}")

        # Early exit if we beat the target bound
        if best_score < 1.503:
            print("[early stop] target reached")
            break

    # ------------------------------------------------------------------
    # 2️⃣ Final stochastic polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_val = _stochastic_refinement(best_seq, deadline,
                                                      step_initial=0.3)
        if refined_val < best_score:
            best_seq = refined
            best_score = refined_val
            print(f"[final polish] val={refined_val:.6f}")

    # final safety clipping (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score = {best_score:.6f}  (length={best_seq.size})")
    return [float(x) for x in best_seq]
