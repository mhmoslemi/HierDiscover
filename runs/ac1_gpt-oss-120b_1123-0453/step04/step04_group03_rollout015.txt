def _conv_and_active(a):
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    # treat entries within a tiny relative tolerance as active
    eps = 1e-12
    active = np.where(conv >= max_val - eps)[0]
    return conv, max_val, active

def _gradient_from_active(a, active):
    n = len(a)
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += a[j]
    grad /= max(1, len(active))
    return grad

def _subgradient_optimize(a0, target_sum, deadline,
                          max_iters=6000, lr0=0.2,
                          patience=400):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    cur_score = best_score
    no_improve = 0
    lr = lr0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv, max_val, active = _conv_and_active(a)
        grad = _gradient_from_active(a, active)

        # decay learning rate
        lr = lr0 / (1.0 + it * 1e-4)

        a_candidate = a - lr * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        # occasionally add a tiny random jitter to aid escape
        if np.random.rand() < 0.02:
            a_candidate += np.random.normal(scale=lr*0.5, size=n)
            a_candidate = _project_onto_simplex(a_candidate, target_sum)

        cand_score = _compute_score_from_vector(a_candidate)

        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            no_improve = 0
        else:
            # acceptance with small temperature to avoid stagnation
            temp = max(0.01, 0.1 * (1.0 - it / max_iters))
            if np.random.rand() < np.exp(-(cand_score - cur_score) / temp):
                a = a_candidate
                cur_score = cand_score
                no_improve = 0
            else:
                no_improve += 1

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[subgrad] n={n} it={it} score={cand_score:.6f}")

        if no_improve >= patience:
            break

    return best_a, best_score

def _random_local_search(start, deadline,
                         step_initial=0.2,
                         swap_prob=0.15):
    """Simple hill‑climbing with Gaussian moves and swaps."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, len(best), size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = best + rng.normal(scale=step, size=best.shape)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:

"""
Sub‑gradient optimiser for the min‑max auto‑convolution problem.

The evaluation metric is  2 n·max(conv(a,a)) / (∑a)² .
Because the objective is homogeneous we can fix  ∑a = 1  and simply
minimise  max(conv(a,a)).  The optimiser works on the exact ℓ∞‑norm
(there is no smoothing) and performs projected sub‑gradient descent on
the simplex, followed by a short stochastic polishing phase.
Several sequence lengths and random restarts are explored within the
time budget.  The routine returns a plain ``list[float]`` which will be
plugged into the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform point
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact score used by ``evaluate_sequence`` (no internal clipping)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _conv_and_active(a: np.ndarray):
    """Return the full convolution, its maximal value and the (near)‑max indices."""
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    # treat values within a tiny absolute tolerance as active (helps when several k tie)
    eps = 1e-12
    active = np.where(conv >= max_val - eps)[0]
    return conv, max_val, active


def _gradient_from_active(a: np.ndarray, active):
    """
    Sub‑gradient of  max_k conv_k(a,a)  at the current point.
    For every active index k we add the vector (a_{k‑i}) and finally average.
    """
    n = len(a)
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += a[j]
    # average over the active set to stay inside the sub‑gradient set
    grad /= max(1, len(active))
    return grad


def _subgradient_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 8000,
    lr0: float = 0.3,
    patience: int = 600,
):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = len(a)

    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)

    cur_score = best_score
    no_improve = 0
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv, max_val, active = _conv_and_active(a)
        grad = _gradient_from_active(a, active)

        # simple inverse‑time decay for the learning rate
        lr = lr0 / (1.0 + it * 1e-4)

        # basic greedy step
        a_candidate = a - lr * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        # occasional random jitter to help escape plateaux
        if rng.random() < 0.02:
            a_candidate += rng.normal(scale=lr * 0.5, size=n)
            a_candidate = _project_onto_simplex(a_candidate, target_sum)

        cand_score = _compute_score_from_vector(a_candidate)

        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            no_improve = 0
        else:
            # small probability to accept a worsening move (simulated‑annealing flavour)
            temp = max(0.01, 0.1 * (1.0 - it / max_iters))
            if rng.random() < np.exp(-(cand_score - cur_score) / temp):
                a = a_candidate
                cur_score = cand_score
                no_improve = 0
            else:
                no_improve += 1

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[subgrad] n={n} it={it} score={cand_score:.6f}")

        if no_improve >= patience:
            break

    return best_a, best_score


def _random_local_search(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.25,
    swap_prob: float = 0.15,
):
    """
    Simple hill‑climbing phase: alternating Gaussian perturbations with
    occasional pairwise swaps, always re‑projecting onto the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, len(best), size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = best + rng.normal(scale=step, size=best.shape)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = min(step_initial, step * 1.05)
        else:
            step *= 0.995
        if step < 1e-6:
            step = step_initial

    return best, best_score

# ----------------------------------------------------------------------
