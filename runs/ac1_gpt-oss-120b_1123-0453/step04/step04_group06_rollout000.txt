def _adam_optimize(f0, target_sum, deadline, max_iters=2000, beta_start=0.5, beta_factor=1.5, beta_max=256.0, step_initial=0.5):
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta = beta_start
    step = step_initial
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        # compute conv
        conv = np.convolve(f, f)
        # weights for smoothed max
        max_val = conv.max()
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        grad = _weighted_gradient(f, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        # candidate step
        cand = f - step * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _compute_score(cand)

        if cand_score < best_score:
            best_f, best_score = cand.copy(), cand_score
            f = cand
            step *= 1.02
        else:
            # keep f unchanged
            step *= 0.7

        # anneal beta
        beta = min(beta * beta_factor, beta_max)

        # optional: restart m,v if step too small etc.
        if step < 1e-12:
            break

    return best_f, best_score

def _simulated_annealing(start, target_sum, deadline, init_temp=0.1, min_temp=1e-4, decay=0.995, max_iter=20000, step_start=0.5):
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    temp = init_temp
    step = step_start
    it = 0
    while time.monotonic() < deadline and it < max_iter:
        # propose a move
        i,j = rng.choice(len(cur), size=2, replace=False)
        # max delta is limited by current value at i and step
        max_delta = min(step, cur[i])
        if max_delta <= 0:
            it += 1
            continue
        delta = rng.random() * max_delta
        cand = cur.copy()
        cand[i] -= delta
        cand[j] = min(1000.0, cand[j] + delta)
        cand_score = _compute_score(cand)
        # Accept if better, else with probability exp((cur_score - cand_score)/temp)
        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / max(temp, 1e-12)):
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
        # update schedule
        temp = max(min_temp, temp * decay)
        step = max(1e-5, step * 0.9999)
        it += 1
    return best, best_score

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # Fast O(n log n) projection (Duchi et al. 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(v: np.ndarray) -> float:
    """Exact evaluation of the supplied ``evaluate_sequence`` formula."""
    n = len(v)
    conv = np.convolve(v, v)
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _weighted_gradient(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(f∗f)_k w.r.t. f.
    w must have length 2·len(f)−1 and Σw = 1.
    """
    n = len(f)
    grad = np.zeros_like(f)
    # loop over each convolution offset
    for k in range(2 * n - 1):
        wk = w[k]
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        # each pair (i,j) contributes wk·2·f[j] to ∂/∂f[i]
        grad[i_start:i_end + 1] += wk * 2.0 * f[j_range]
        # correct diagonal term when i == j (i.e. k even and i=j=k/2)
        if (k & 1) == 0:
            i_eq = k // 2
            if i_start <= i_eq <= i_end:
                # contribution 2·f[i] – f[i] = f[i]
                grad[i_eq] -= wk * f[i_eq]
    return grad

# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2000,
    beta_start: float = 0.5,
    beta_factor: float = 1.5,
    beta_max: float = 256.0,
    step_initial: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smooth approximation of the max
    of the auto‑convolution.  Returns the best vector and its exact
    score.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)

    # Adam state
    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    step = step_initial
    beta = beta_start
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        max_val = conv.max()

        # soft‑max weights (stable)
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        grad = _weighted_gradient(f, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        cand = f - step * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best_f, best_score = cand.copy(), cand_score
            f = cand
            step *= 1.02
            no_improve = 0
        else:
            step *= 0.7
            no_improve += 1

        # anneal smoothing parameter β
        beta = min(beta * beta_factor, beta_max)

        # if stuck too long, relax β a little to broaden search
        if no_improve >= 200:
            beta = max(beta_start, beta / 2.0)
            no_improve = 0

        if step < 1e-12:
            break

    return best_f, best_score


def _simulated_annealing(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    init_temp: float = 0.1,
    min_temp: float = 1e-4,
    decay: float = 0.995,
    max_iters: int = 20000,
    step_start: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Simple SA/hill‑climb on the simplex.  Moves a tiny amount of mass
    from one index to another while preserving the total ℓ¹‑norm.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    temp = init_temp
    step = step_start

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # propose a transfer of mass from i → j
        i, j = rng.choice(len(cur), size=2, replace=False)
        max_delta = min(step, cur[i])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _compute_score(cand)

        # Metropolis acceptance
        improve = cand_score < cur_score
        if improve or rng.random() < math.exp((cur_score - cand_score) / max(temp, 1e-12)):
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score

        # schedule updates
        temp = max(min_temp, temp * decay)
        step = max(1e-6, step * 0.9999)

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step sequence within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # give a small safety margin for the host harness
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣ Warm‑start from a known construction if available.
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        n = len(seq)
        s_target = math.sqrt(2.0 * n)
        cur_sum = seq.sum()
        if cur_sum > 0:
            seq = seq * (s_target / cur_sum)
        else:
            seq = np.full(n, s_target / n, dtype=np.float64)
        sc = _compute_score(seq)
        best_seq, best_score = seq.copy(), sc
        print(f"[init] warm‑start n={n} score={sc:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Global optimisation – Adam on smoothed max.
    # --------------------------------------------------------------
    # Consider a spread of lengths; larger n often yields better bounds.
    lengths = [300, 400, 500, 600, 800, 1000, 1200, 1500]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # number of random restarts; reduce for very large n to stay within budget
        restarts = 12 if n <= 800 else 6
        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a feasible point on the simplex
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
            else:
                f0 = rng.random(n) + 1e-8
                f0 = f0 / f0.sum() * target_sum

            # give each restart a small time slice (≈10 s)
            sub_deadline = min(deadline, time.monotonic() + 10.0)

            f_opt, sc_opt = _adam_optimize(
                f0,
                target_sum,
                sub_deadline,
                max_iters=2000,
                beta_start=0.5,
                beta_factor=1.5,
                beta_max=256.0,
                step_initial=0.5,
            )

            if sc_opt < best_score:
                best_seq, best_score = f_opt.copy(), sc_opt
                print(f"[adam] n={n} restart={rs} score={sc_opt:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                print("[early] target reached during Adam phase")
                deadline = time.monotonic() + 0.2  # tiny buffer
                break

    # --------------------------------------------------------------
    # 2️⃣ Local refinement – simulated annealing.
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        refined, refined_sc = _simulated_annealing(
            best_seq,
            target_sum,
            deadline,
            init_temp=0.1,
            min_temp=1e-4,
            decay=0.995,
            max_iters=20000,
            step_start=0.2,
        )
        if refined_sc < best_score:
            best_seq, best_score = refined, refined_sc
            print(f"[sa] final refined score={refined_sc:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Fallback (should never happen) – uniform distribution.
    # --------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best_score = _compute_score(best_seq)

    # Final safety clipping (evaluate_sequence does the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
