"""
Fast non‑negative step‑function optimizer.

The algorithm improves on the previous Adam + annealing approach by

* computing convolutions with FFT (O(n log n) instead of O(n²)),
* using a richer sub‑gradient that averages over all “almost‑maximal”
  convolution entries (instead of a single active offset),
* fixing the missing factor‑2 in the exact gradient of a convolution
  entry,
* a simple learning‑rate schedule and early‑stopping criteria,
* a final simulated‑annealing polish (the same routine as before).

The optimiser works on the ℓ¹‑simplex Σ aᵢ = √(2 n) (the scaling that makes the
objective equal to the maximal auto‑convolution entry).  It is run for a set
of problem sizes and a few random restarts; the best candidate found before
the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution helpers (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear auto‑convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(fa * fa, n=n_fft)
    return conv[: 2 * n - 1]


def _gradient_weighted_full(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a) w.r.t. a (exact, factor 2 included).
    a : length n  ≥ 1
    w : length 2·n‑1 (weights for each convolution offset)
    Returns gradient of shape n.
    """
    n = a.shape[0]
    # padded length for (a) * (w_rev) convolution
    n_fft = 1 << ((3 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=n_fft)
    fw_rev = np.fft.rfft(w[::-1], n=n_fft)          # reverse w
    conv = np.fft.irfft(fa * fw_rev, n=n_fft)       # real result
    # slice indices n‑1 … 2n‑2 (inclusive) – length n
    start = n - 1
    slice_seg = conv[start : start + n]
    # reverse order and multiply by 2 (each product appears twice)
    grad = 2.0 * slice_seg[::-1]
    return grad


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------


def _score(f: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` (no clipping needed for our domain)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


# ----------------------------------------------------------------------
# Adam optimizer with richer active‑set gradient
# ----------------------------------------------------------------------


def _adam_active_descent(f0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimizing the maximal auto‑convolution entry.
    The gradient is the average over all offsets whose value lies within a
    shrinking tolerance of the current maximum.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _score(f)

    m = np.zeros_like(f)
    v = np.zeros_like(f)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.07                               # base learning rate
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = _conv_self(f)
        max_val = np.max(conv)

        # ----- active set -------------------------------------------------
        # tolerance shrinks from a few percent to 1e‑6 of the peak
        tol = max(max_val * max(0.01, 1.0 / (it + 10)), 1e-12)
        active = np.where(conv >= max_val - tol)[0]

        # uniform weighting over the active offsets
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size

        # ----- gradient --------------------------------------------------
        grad = _gradient_weighted_full(f, w)

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_candidate = f - step
        f_candidate = _project_onto_simplex(f_candidate, target_sum)
        f_candidate = np.clip(f_candidate, 0.0, 1000.0)

        cur_score = _score(f)
        cand_score = _score(f_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_f = f_candidate.copy()
            no_improve = 0
            print(f"[Adam] it={it:4d} n={len(f)} score={cand_score:.6f}")
        else:
            no_improve += 1

        # accept if the *raw* maximum con­volution decreases
        max_cand = np.max(_conv_self(f_candidate))
        if max_cand < max_val:
            f = f_candidate
        else:
            # occasional jitter to escape plateaus
            if no_improve % 300 == 0 and no_improve > 0:
                rng = np.random.default_rng()
                f += rng.normal(scale=0.001 * target_sum, size=len(f))
                f = _project_onto_simplex(f, target_sum)

        if no_improve > 800:                 # early exit
            break

    return best_f, best_score

# Simulated‑annealing polishing (unchanged)
# ----------------------------------------------------------------------


def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.2,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    n = len(start)
    target_sum = np.sum(start)

    T = init_temp
    step_scale = 0.05 * target_sum / n

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        max_move = min(cur[i], step_scale)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        if it % 256 == 0:
            cand = _project_onto_simplex(cand, target_sum)

        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[Anneal] it={it:6d} score={cand_score:.6f}")
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        T *= decay
        if T < 1e-8:
            T = init_temp

        step_scale *= 0.9999
        if step_scale < 1e-6 * target_sum:
            step_scale = 0.05 * target_sum / n

    return best, best_score

def _alternating_initial(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a deterministic “large‑at‑the‑ends” start:
    random Dirichlet mass, sorted descending, then placed
    alternately at the leftmost and rightmost still‑free position.
    """
    vals = rng.random(n)
    vals = vals / np.sum(vals) * target_sum
    desc = np.sort(vals)[::-1]                     # descending

    pos = np.empty(n, dtype=int)
    left, right = 0, n - 1
    for k in range(n):
        if k % 2 == 0:
            pos[k] = left
            left += 1
        else:
            pos[k] = right
            right -= 1

    a = np.empty(n, dtype=float)
    a[pos] = desc
    return a

        # number of restarts per length
        restarts = 6 if n <= 2000 else 4
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise
            if best_seq is not None and len(best_seq) == n:
                f0 = best_seq.copy()
                f0 += rng.normal(scale=0.001 * target_sum, size=n)
                f0 = np.clip(f0, 0.0, None)
                f0 = _project_onto_simplex(f0, target_sum)
            else:
                f0 = _alternating_initial(n, target_sum, rng)

            # allocate a few seconds for each descent run
            sub_deadline = min(deadline, time.monotonic() + 12.0)
            f_opt, score_opt = _adam_active_descent(f0,
                                                    target_sum,
                                                    sub_deadline,
                                                    max_iters=2500)

            if score_opt < best_score:
                best_score = score_opt
                best_seq = f_opt.copy()
                print(f"[update] n={n} restart={r} score={score_opt:.6f}")

            # early stop if we already beat the target bound
            if best_score < 1.503:
                print("[early] target achieved during Adam phase")
                return [float(x) for x in best_seq]

    # --------------------------------------------------------------
    # Final polishing with simulated annealing (if time permits).
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_score = _simulated_annealing(best_seq, deadline,
                                                     init_temp=0.2,
                                                     decay=0.9997)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[final] annealed score={refined_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform construction (should never trigger).
    # --------------------------------------------------------------
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] using uniform construction")

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
