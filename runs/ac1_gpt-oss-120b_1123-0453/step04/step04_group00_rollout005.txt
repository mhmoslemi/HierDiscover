"""
Algorithm:
- Perform a systematic search over simple parametric families (exponential,
  power‑law) for several lengths.  For each family we analytically normalise
  the L1‑norm to √(2 n) and keep the best candidate.
- Starting from the best candidate for each length, run a sub‑gradient descent
  on the exact objective  max(f∗f).  The sub‑gradient is obtained from the
  convolution entries that attain the maximum.  The iterate is projected onto
  the feasible simplex (non‑negative, L1 = √(2 n)) using the O(n)
  Duchi‑et‑al. algorithm.
- The descent uses an adaptive step size and occasional random jitter to
  escape plateaus.
- After the gradient phase a cheap random‑mass‑swap hill‑climb is performed
  until the time budget expires.
The function returns the best sequence (as a plain list) found before the
deadline.
"""

import time
import math
import numpy as np

def _next_fast_len(n: int) -> int:
    """Return a good FFT length, falling back to power‑of‑2."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))

def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for repeated calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv

def _fast_score(seq: np.ndarray) -> float:
    """Exact value of evaluate_sequence (max of linear convolution)."""
    conv = np.convolve(seq, seq)
    return float(conv.max())

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum x = s} (Duchi et al. 2008)."""
    # Work on a copy to avoid side effects.
    v = v.copy()
    # Ensure non‑negative (the algorithm works with any values,
    # the final max ensures feasibility).
    if v.sum() == 0.0:
        # Simple uniform distribution if the vector is all zeros.
        return np.full_like(v, s / v.shape[0])
    # Sort v in descending order.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # Find the largest rho satisfying the condition.
    rho_arr = np.where(u > cssv / np.arange(1, v.shape[0] + 1))[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise while keeping non‑negativity.
    total = w.sum()
    if total > 0:
        w *= s / total
    else:
        w = np.full_like(v, s / v.shape[0])
    return w

def _gradient_of_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of f(a)=max(a∗a) at the current point.
    The gradient is the average of the gradients of all active constraints.
    """
    max_val = conv.max()
    # Choose a tolerance relative to the magnitude of the maximum.
    eps = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - eps)[0]
    w = np.zeros_like(conv)
    if active.size == 0:
        # Should not happen, but fall back to uniform weighting.
        w[:] = 1.0 / conv.shape[0]
    else:
        w[active] = 1.0 / float(active.size)
    # The gradient of sum_k w_k * (a∗a)_k is 2 * (a ⋆ w_rev).
    # We reuse the helper that implements exactly this.
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)          # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]          # the middle n entries
    grad = 2.0 * segment[::-1]                # reverse back
    return grad

def _subgradient_descent(init_a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(a∗a) with projection onto the ℓ¹‑simplex.
    Returns the best vector encountered and its exact max‑convolution.
    """
    a = init_a.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _fast_score(a)

    # Initial step size: a few times the average entry.
    step = 5.0 * target / n
    it = 0
    while time.monotonic() < deadline and it < max_iter:
        conv = _linear_convolve(a, a)                # length 2·n‑1
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()

        # Sub‑gradient for the active max entries.
        grad = _gradient_of_max(a, conv)

        # Simple step‑size decay.
        cur_step = step / (1.0 + 0.001 * it)

        # Gradient move + projection.
        a_new = a - cur_step * grad
        a_new = _project_onto_simplex(a_new, target)

        # Occasionally inject a small random jitter to avoid stagnation.
        if it % 250 == 0 and it > 0:
            i, j = rng.choice(n, size=2, replace=False)
            delta = a_new[i] * 0.02
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_onto_simplex(a_new, target)

        a = a_new
        it += 1

    return best_a, best_val

def _search_parametric(n: int, target: float) -> tuple[np.ndarray, float]:
    """
    Exhaustively sweep a few low‑dimensional parametric families for length n.
    Returns the best vector found and its max‑convolution.
    """
    best_seq = None
    best_score = float('inf')
    # Power‑law: a_i ∝ (i+1)^{-p}
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.2, 5.0, 30):
        a = idx ** (-p)
        a = a / a.sum() * target
        a = _project_onto_simplex(a, target)          # ensures feasibility
        score = _fast_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # Exponential: a_i ∝ r^{i}
    for r in np.linspace(0.5, 0.9995, 35):
        a = r ** np.arange(n, dtype=np.float64)
        a = a / a.sum() * target
        a = _project_onto_simplex(a, target)
        score = _fast_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # Gamma‑like: a_i ∝ (i+1)^k * exp(-λ i)
    for k in [0.0, 0.5, 1.0, 2.0]:
        for lam in np.linspace(0.001, 0.10, 20):
            a = ((idx) ** k) * np.exp(-lam * np.arange(n, dtype=np.float64))
            a = a / a.sum() * target
            a = _project_onto_simplex(a, target)
            score = _fast_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()

    return best_seq, best_score

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low max‑auto‑convolution score.

    The strategy mixes a parametric sweep, sub‑gradient descent, and a final
    random‑mass‑swap hill‑climb, all respecting the ℓ¹‑normalisation required
    by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float('inf')

    # Helper to update global best and print progress.
    def _update_global(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = _fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global best] n={len(seq)} score={sc:.6f}")

    # 0️⃣ Warm‑start from any previously discovered construction.
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            cand = _project_onto_simplex(cand, math.sqrt(2.0 * cand.shape[0]))
            _update_global(cand)
        except Exception:  # pragma: no cover
            pass

    # Lengths to explore – modest variety balanced against time budget.
    lengths = [200, 250, 300, 350, 400, 450, 500, 550, 600]

    # 1️⃣ Parametric sweep.
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        seq, sc = _search_parametric(n, target)
        if seq is not None:
            _update_global(seq)

    # 2️⃣ Sub‑gradient descent from promising starts.
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        # Decide how many random restarts for this length.
        restarts = 4 if n <= 500 else 2
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break
            # Initial vector: either random Dirichlet or blend with global best.
            start = rng.gamma(1.0, size=n)
            start = start / start.sum() * target
            if best_seq is not None and best_seq.shape[0] == n:
                start = (start + best_seq) / 2.0
                start = _project_onto_simplex(start, target)

            # Run descent.
            opt_seq, opt_val = _subgradient_descent(start, deadline, rng,
                                                    max_iter=6000)
            _update_global(opt_seq)

            # Early exit if we already beat the target goal.
            if best_score < 1.503:
                print("[early exit] target achieved")
                return best_seq.tolist()

    # 3️⃣ Final cheap random‑mass‑swap hill‑climb.
    if best_seq is not None:
        while time.monotonic() < deadline:
            n = best_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            delta = best_seq[i] * 0.02
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_onto_simplex(cand, math.sqrt(2.0 * n))
            _update_global(cand)
    else:
        # Fallback – uniform distribution of size 300.
        n = 300
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # Return as plain Python list.
    return [float(x) for x in best_seq]

It also adds extra random restarts, adaptive step‑size decay and a final cheap mass‑swap hill‑climb. All helpers are defined at top‑level as required.

```python
"""
Algorithm:
- Sweep simple parametric families (exponential, power‑law, gamma‑like)
  for several lengths, keeping the best candidate.
- From each promising candidate run a sub‑gradient descent on the exact
  objective max(a∗a).  The sub‑gradient is obtained from the convolution
  entries that attain the maximum.  After each step the iterate is projected
  onto the feasible simplex {x≥0, sum(x)=√(2 n)} using the O(n) Duchi‑et‑al.
  routine.
- The descent uses an adaptive step size and occasional random jitter to
  escape plateaus.
- When the time budget is almost exhausted a cheap random‑mass‑swap
  hill‑climb is performed.
The best sequence found before the deadline is returned as a plain Python
list, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return a good FFT length, falling back to a power of two."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (efficient for repeated calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fast_score(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` (max of linear convolution)."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector onto the simplex {x≥0, sum(x)=s}
    (Duchi, Shalev‑Shwartz, Singer, Chandra 2008).
    """
    v = v.copy()
    if v.sum() == 0.0:
        return np.full_like(v, s / v.shape[0])

    # Sort descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u > cssv / np.arange(1, v.shape[0] + 1))[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)

    # Renormalise for numerical safety.
    total = w.sum()
    if total > 0.0:
        w *= s / total
    else:
        w = np.full_like(v, s / v.shape[0])
    return w


def _gradient_of_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of f(a)=max(a∗a) at the current point.
    The gradient is the average of the gradients of all active constraints.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - eps)[0]

    w = np.zeros_like(conv)
    if active.size == 0:  # fallback – should never happen
        w[:] = 1.0 / conv.shape[0]
    else:
        w[active] = 1.0 / float(active.size)

    # Gradient = 2·a ⋆ rev(w)  (cross‑correlation).
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)          # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]           # middle n entries
    grad = 2.0 * segment[::-1]                  # reverse back
    return grad


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _subgradient_descent(init_a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iter: int = 6000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(a∗a) with projection onto the ℓ¹‑simplex.
    Returns the best vector encountered and its exact max‑convolution.
    """
    a = init_a.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _fast_score(a)

    # Initial step size – a few times the average entry.
    step0 = 5.0 * target / n
    it = 0
    while time.monotonic() < deadline and it < max_iter:
        conv = _linear_convolve(a, a)        # length 2·n‑1
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()

        grad = _gradient_of_max(a, conv)

        # Decaying step size.
        step = step0 / (1.0 + 0.001 * it)

        a_new = a - step * grad
        a_new = _project_onto_simplex(a_new, target)

        # Small random jitter every few hundred iterations.
        if it % 250 == 0 and it > 0:
            i, j = rng.choice(n, size=2, replace=False)
            delta = a_new[i] * 0.02
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_onto_simplex(a_new, target)

        a = a_new
        it += 1

    return best_a, best_val


def _search_parametric(n: int, target: float) -> tuple[np.ndarray, float]:
    """
    Exhaustive sweep over a few low‑dimensional families for a fixed length.
    Returns the best sequence found together with its max‑convolution value.
    """
    best_seq = None
    best_score = float('inf')
    idx = np.arange(1, n + 1, dtype=np.float64)

    # 1) Power‑law a_i ∝ i^{‑p}
    for p in np.linspace(0.2, 5.0, 30):
        a = idx ** (-p)
        a = a / a.sum() * target
        a = _project_onto_simplex(a, target)
        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # 2) Exponential a_i ∝ r^{i}
    for r in np.linspace(0.5, 0.9995, 35):
        a = r ** np.arange(n, dtype=np.float64)
        a = a / a.sum() * target
        a = _project_onto_simplex(a, target)
        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # 3) Gamma‑like a_i ∝ (i+1)^k·exp(-λ i)
    for k in (0.0, 0.5, 1.0, 2.0):
        for lam in np.linspace(0.001, 0.10, 20):
            a = (idx ** k) * np.exp(-lam * np.arange(n, dtype=np.float64))
            a = a / a.sum() * target
            a = _project_onto_simplex(a, target)
            sc = _fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low max‑auto‑convolution score.

    The search combines a parametric sweep, several sub‑gradient runs with
    random restarts, and a final cheap hill‑climb.  It respects the L1
    normalisation required by ``evaluate_sequence`` and aborts gracefully
    when the time budget is exhausted.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float('inf')

    def _update_global(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = _fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global best] n={len(seq)} score={sc:.6f}")

    # 0️⃣ Warm‑start from a previously discovered construction, if any.
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            target0 = math.sqrt(2.0 * cand.shape[0])
            cand = _project_onto_simplex(cand, target0)
            _update_global(cand)
        except Exception:  # pragma: no cover
            pass

    # Lengths to explore – a compromise between diversity and speed.
    lengths = [200, 250, 300, 350, 400, 450, 500, 550, 600]

    # 1️⃣ Parametric sweep.
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        seq, sc = _search_parametric(n, target)
        if seq is not None:
            _update_global(seq)

    # 2️⃣ Sub‑gradient descent with several restarts per length.
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        restarts = 4 if n <= 500 else 2
        for _ in range(restarts):
            if time.monotonic() >= deadline:
                break
            start = rng.gamma(1.0, size=n)
            start = start / start.sum() * target

            # Blend with the current global best if dimensions match.
            if best_seq is not None and best_seq.shape[0] == n:
                start = (start + best_seq) / 2.0
                start = _project_onto_simplex(start, target)

            opt_seq, _ = _subgradient_descent(start, deadline, rng)
            _update_global(opt_seq)

            if best_score < 1.503:
                print("[early exit] target achieved")
                return best_seq.tolist()

    # 3️⃣ Final cheap random‑mass‑swap hill‑climb (very fast).
    if best_seq is not None:
        while time.monotonic() < deadline:
            n = best_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)
            delta = best_seq[i] * 0.02
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_onto_simplex(cand, math.sqrt(2.0 * n))
            _update_global(cand)
    else:
        # Fallback – uniform vector of moderate size.
        n = 300
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    # Return a plain Python list as required by the outer harness.
    return [float(x) for x in best_seq]
```
