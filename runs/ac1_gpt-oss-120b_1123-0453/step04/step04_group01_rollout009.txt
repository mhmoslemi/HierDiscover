def _simplex_projection(v, z=1.0):
    # ensure sum(v) = z and v >= 0
    n = len(v)
    u = np.sort(v)[::-1]  # sort descending
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _projected_grad_descent(a, target_sum, deadline, rng, max_iter=2000):
    n = a.shape[0]
    best = a.copy()
    best_score = _fast_score(best)
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        conv = _convolution(a)  # full conv
        max_val = conv.max()
        k = int(conv.argmax())
        # gradient for active conv
        grad = np.zeros(n, dtype=np.float64)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idxs] = 2.0 * a[k - idxs]
        # step size
        pos = grad > 0
        if np.any(pos):
            max_alpha = np.min(a[pos] / grad[pos])
            alpha = 0.5 * max_alpha
        else:
            alpha = 0.0
        # add some random noise to avoid stagnation
        alpha = alpha * (0.5 + 0.5 * rng.random())
        a_new = a - alpha * grad
        # project onto simplex
        a_new = _simplex_projection(a_new, target_sum)
        # evaluate
        new_score = _fast_score(a_new)
        if new_score < best_score:
            best = a_new
            best_score = new_score
        # maybe accept slight increase with some probability to escape local minima
        if new_score < _fast_score(a) or rng.random() < 0.001:
            a = a_new
        else:
            # keep a unchanged
            pass
    return best

"""
search_candidate_v2.py

A revised search routine for the step‑function minimisation challenge.
The main ideas that differ from the previous implementation are

* **Projected sub‑gradient descent** – the active convolution entry is
  identified, its exact sub‑gradient (2·a_{k‑i}) is computed and a step
  is taken in the negative direction.  After the step the iterate is
  *projected onto the ℓ¹‑simplex* (non‑negative, exact ℓ¹‑norm).  This
  respects the constraints without the need for ad‑hoc “mean‑subtraction”.

* **Progressive up‑scaling** – start with a modest length, find a good
  shape, linearly interpolate it to a larger length and continue the
  descent.  Larger n gives more freedom and usually yields a smaller
  objective.

* **Richer parametric fleet** – beyond the families used before, the
  sweep now includes very steep Beta‑shapes, mixed‑Beta mixtures,
  sine‑powered, logarithmic and extra Dirichlet draws.

* **Time‑aware orchestration** – the total budget (≈1000 s) is split
  between a broad sweep, focussed descent on the most promising
  candidates and a final polishing stage (swap hill‑climb + short
  Gaussian annealing).

All helper functions are top‑level, there are no closures or lambdas,
and the code respects the security checks of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al.) – projects onto {x ≥ 0, ∑x = z}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of sum z."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.shape[0]
    # work on a copy to avoid side effects
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform distribution
        return np.full(n, z / n, dtype=np.float64)
    rho = np.nonzero(cond)[0][-1]            # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Normalisation helper – scales a non‑negative vector to exact ℓ¹‑norm
# ----------------------------------------------------------------------
def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale `vec` so that its ℓ¹‑norm equals `target_sum` (preserving non‑negativity)."""
    s = vec.sum()
    if s <= 0.0:
        # fallback – uniform distribution
        n = vec.shape[0]
        return np.full(n, target_sum / n, dtype=np.float64)
    return vec * (target_sum / s)


# ----------------------------------------------------------------------
# Fast score – exact analogue of `evaluate_sequence` without safety checks.
# For a vector already normalised to √(2·n) the value equals the maximal auto‑convolution entry.
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Full convolution via FFT (more efficient for repeated calls)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2n‑1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Parametric families for the sweep
# ----------------------------------------------------------------------
def _beta_family(n: int, a: float, b: float, target_sum: float) -> np.ndarray:
    """Discrete Beta density on {0,…,n‑1} with shape parameters a,b."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    w = np.maximum(w, 1e-14)                 # avoid exact zeros
    w = w / w.sum() * target_sum
    return w


def _logarithmic_family(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ -log(x) with x∈(0,1)."""
    x = (np.arange(1, n + 1, dtype=np.float64)) / (n + 1)
    w = -np.log(x)
    w = w / w.sum() * target_sum
    return w


def _mixed_beta(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two Beta densities with a random mixing weight."""
    a1, b1 = rng.uniform(0.1, 5.0, size=2)
    a2, b2 = rng.uniform(0.1, 5.0, size=2)
    w1 = _beta_family(n, a1, b1, 1.0)          # unnormalised for now
    w2 = _beta_family(n, a2, b2, 1.0)
    lam = rng.random()
    w = lam * w1 + (1.0 - lam) * w2
    w = w / w.sum() * target_sum
    return w


def _generate_candidates(n: int, target_sum: float, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a diverse fleet of start vectors for length `n`."""
    cand = []

    # 0. uniform
    cand.append(np.full(n, target_sum / n, dtype=np.float64))

    # 1. simple triangular (peak in centre)
    x = np.arange(n, dtype=np.float64)
    tri = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
    tri[tri < 0] = 0.0
    cand.append(_normalize(tri, target_sum))

    # 2. power‑law a_i ∝ i⁻ᵖ
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.05, 4.0, 20):
        a = idx ** (-p)
        cand.append(_normalize(a, target_sum))

    # 3. exponential decay a_i ∝ e^{-α·i}
    for alpha in np.linspace(0.0005, 0.25, 20):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand.append(_normalize(a, target_sum))

    # 4. beta family (wide grid, including very sharp shapes)
    for a_par in np.concatenate([np.linspace(0.05, 2.0, 8), np.linspace(2.0, 10.0, 6)]):
        for b_par in np.concatenate([np.linspace(0.05, 2.0, 8), np.linspace(2.0, 10.0, 6)]):
            cand.append(_beta_family(n, a_par, b_par, target_sum))

    # 5. arcsine (Beta(½,½))
    cand.append(_beta_family(n, 0.5, 0.5, target_sum))

    # 6. sine‑powered shapes a_i ∝ sin^p(π·i/(n+1))
    for p in np.linspace(0.5, 4.0, 10):
        a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1))) ** p
        cand.append(_normalize(a, target_sum))

    # 7. logarithmic family
    cand.append(_logarithmic_family(n, target_sum))

    # 8. mixtures of two betas
    for _ in range(5):
        cand.append(_mixed_beta(n, target_sum, rng))

    # 9. random Dirichlet draws
    for _ in range(6):
        a = rng.dirichlet(np.ones(n)) * target_sum
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the active convolution entry
# ----------------------------------------------------------------------
def _projected_grad_descent(
    init: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 3000,
) -> np.ndarray:
    """
    Reduce `max(conv(a,a))` by moving opposite to the exact sub‑gradient of the
    active entry and projecting onto the ℓ¹‑simplex after each step.
    """
    n = init.shape[0]
    a = _normalize(init, target_sum)
    best = a.copy()
    best_sc = _fast_score(best)

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1

        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient of the active entry (2·a_{k‑i} where defined)
        grad = np.zeros(n, dtype=np.float64)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idxs] = 2.0 * a[k - idxs]

        # maximal feasible step that keeps the iterate non‑negative
        pos = grad > 0
        if np.any(pos):
            max_alpha = np.min(a[pos] / grad[pos])
            # step size slowly decays with the iteration count
            alpha = 0.6 * max_alpha * (1.0 / (1.0 + it * 1e-4))
        else:
            alpha = 0.0

        # occasional random jitter to escape flat regions
        if rng.random() < 0.02:
            alpha *= 0.5 + 0.5 * rng.random()

        a_candidate = a - alpha * grad
        a_candidate = _simplex_projection(a_candidate, target_sum)

        sc = _fast_score(a_candidate)
        if sc < best_sc:
            best, best_sc = a_candidate.copy(), sc

        # accept if we made progress; otherwise keep the old point
        if sc < _fast_score(a):
            a = a_candidate

        # very cheap occasional smoothing (only if it helps)
        if rng.random() < 0.01:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode="same")
            smooth = _simplex_projection(smooth, target_sum)
            sc_s = _fast_score(smooth)
            if sc_s < best_sc:
                best, best_sc = smooth.copy(), sc_s
                a = smooth

    return best


# ----------------------------------------------------------------------
# Tiny hill‑climb for polishing (swap small mass)
# ----------------------------------------------------------------------
def _swap_hill_climb(seq: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator) -> np.ndarray:
    best = seq.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    while time.monotonic() < deadline:
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor = int(rng.choice(donors))
        receiver = int(rng.integers(n))
        if donor == receiver:
            continue
        eps = min(best[donor] * 0.06, target_sum * 0.0015)
        cand = best.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = _simplex_projection(cand, target_sum)
        sc = _fast_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
    return best


# ----------------------------------------------------------------------
# Short Gaussian perturbation (annealing style)
# ----------------------------------------------------------------------
def _gaussian_local_search(seq: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator) -> np.ndarray:
    best = seq.copy()
    best_sc = _fast_score(best)
    step = target_sum * 0.015

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape[0])
        cand = _simplex_projection(cand, target_sum)
        sc = _fast_score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
            step *= 0.9
        else:
            step *= 0.98
            if step < 1e-7:
                step = target_sum * 0.015
    return best
