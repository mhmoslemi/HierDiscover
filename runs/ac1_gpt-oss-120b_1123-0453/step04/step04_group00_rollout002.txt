def _smooth_max_descent(init_seq, deadline, rng, max_iters=3000):
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    best = a.copy()
    best_score = _fast_score(a)

    beta = 1.0
    beta_max = 500.0
    step_base = target / n * 4.0
    momentum = np.zeros_like(a)
    beta_inc = 1.2

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best = a.copy()
        # compute smooth weights
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)
        # momentum update
        momentum = 0.9 * momentum - step_base / (1 + it*0.01) * grad
        a = a + momentum
        # projection
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n)
        # occasional jitter
        if it % 250 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0:
                delta = a[i] * 0.05
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)
        # beta schedule
        if cur_max > best_score * 0.99 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)
    return best, best_score

"""
Improved step‑function search.

The algorithm works in three phases:

0️⃣  Warm‑start from the previously found construction, if any.
1️⃣  Exhaustive parametric sweep over several families (power‑law,
    exponential, combined power‑law·exponential) for a range of
    lengths.  Every candidate is ℓ¹‑normalised to √(2 n); the
    max‑auto‑convolution (which equals the true evaluate_sequence
    value after this normalisation) is recorded.
2️⃣  For the most promising lengths we run many *projected sub‑gradient*
    descents on a smooth‑max surrogate of the max‑convolution.
    The optimiser uses a momentum term, an annealed “sharpness”
    parameter β and occasional random jitter to escape plateaus.
    The number of iterations is much larger than in the reference
    implementation.
3️⃣  A cheap random‑swap hill‑climb finishes the search while time
    permits.

All helper functions are top‑level (no closures).  The code respects
the 1000 s budget, stays within the ℓ¹‑sphere, and returns a plain
Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global mutable state for the current best sequence
# ----------------------------------------------------------------------
_global_state = {"seq": None, "score": float("inf")}


# ----------------------------------------------------------------------
# Basic utilities (FFT‑accelerated convolution, fast scoring, normalisation)
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return a length suitable for fast FFT (fallback to power‑of‑2)."""
    try:
        # NumPy ≥ 1.20 provides next_fast_len
        return np.fft.next_fast_len(n)
    except Exception:                                   # pragma: no cover
        return 2 ** int(np.ceil(math.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT – fast for many repeated calls."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fast_score(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` after ℓ¹‑normalisation."""
    # After normalisation sum(seq)=√(2·n) ⇒ evaluate_sequence == max(conv)
    return float(np.convolve(seq, seq).max())


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and enforce ℓ¹‑norm = √(2·n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    tot = seq.sum()
    if tot <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / tot)
    return seq


# ----------------------------------------------------------------------
# Smooth‑max surrogate helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max weights for a convolution vector."""
    cmax = conv.max()
    # shift for numerical stability
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    ∂/∂a_i  Σ_j w_j (a⋆a)_j = 2·Σ_j w_{i+j} a_j.
    Implemented via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)            # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1: 2 * n - 1]             # central n entries
    grad = 2.0 * segment[::-1]
    return grad


# ----------------------------------------------------------------------
# Global‑state helper
# ----------------------------------------------------------------------
def _update_global(seq: np.ndarray) -> None:
    """Update the module‑wide best sequence if the candidate improves."""
    global _global_state
    sc = _fast_score(seq)
    if sc < _global_state["score"]:
        _global_state["score"] = sc
        _global_state["seq"] = seq.copy()
        print(f"[global best] n={len(seq)} score={sc:.6f}")


# ----------------------------------------------------------------------
# Optimisation core: smooth‑max gradient descent with momentum & jitter
# ----------------------------------------------------------------------
def _smooth_max_descent(
    init_seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the smooth‑max surrogate."""
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    best_local = a.copy()
    best_score = _fast_score(a)

    # hyper‑parameters
    beta = 1.0
    beta_max = 600.0
    beta_inc = 1.30
    base_step = (target / n) * 4.0
    momentum = np.zeros_like(a)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = _linear_convolve(a, a)               # 2·n‑1 entries
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # smooth‑max and gradient
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # momentum update (Nesterov‑like)
        step = base_step / (1.0 + it * 0.01)
        momentum = 0.9 * momentum - step * grad
        a = a + momentum

        # projection onto feasible set (non‑negative, ℓ¹ = √(2·n))
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # occasional random jitter to avoid stagnation
        if it % 250 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.04
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # anneal β – make the surrogate sharper when progress stalls
        if cur_max > best_score * 0.995 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

    return best_local, best_score


# ----------------------------------------------------------------------
# Random‑mass‑swap hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _random_swap_hillclimb(seq: np.ndarray, deadline: float, rng: np.random.Generator):
    """Very cheap local search: move a small fraction of mass between two indices."""
    a = seq.copy()
    n = a.shape[0]
    best_score = _fast_score(a)

    while time.monotonic() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # move at most 5 % of the smaller entry
        delta = min(a[i], a[j]) * (rng.random() * 0.05)
        if rng.random() < 0.5:
            # i → j
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # j → i
            cand = a.copy()
            cand[j] -= delta
            cand[i] += delta

        cand = _normalize(cand)
        sc = _fast_score(cand)
        if sc < best_score:
            best_score = sc
            a = cand
            _update_global(a)

    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.

    The search consists of:
      * warm‑start from the existing construction (if present),
      * a dense parametric sweep over several families,
      * many restarts of a smooth‑max gradient descent (with momentum,
        annealed β and jitter),
      * a final cheap random‑swap hill‑climb.
    The best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 0️⃣  Warm‑start from a previously found construction (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=np.float64)
            cand = _normalize(cand)
            _update_global(cand)
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep
    # ------------------------------------------------------------------
    # Lengths explored (smaller lengths are cheaper, larger ones give more freedom)
    lengths = [1500, 1200, 1000, 800, 600, 500, 400, 300, 200, 150, 100]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based for power‑law

        # ----- pure power‑law -------------------------------------------------
        for p in np.linspace(0.2, 5.0, 30):
            a = idx ** (-p)
            a = a / a.sum() * target
            a = _normalize(a)
            _update_global(a)

        # ----- pure exponential -----------------------------------------------
        for alpha in np.linspace(0.001, 0.2, 30):
            a = np.exp(-alpha * np.arange(n, dtype=np.float64))
            a = a / a.sum() * target
            a = _normalize(a)
            _update_global(a)

        # ----- combined power‑law * exponential --------------------------------
        for p in np.linspace(0.2, 3.0, 12):
            for alpha in np.linspace(0.001, 0.15, 12):
                a = (idx ** (-p)) * np.exp(-alpha * np.arange(n, dtype=np.float64))
                a = a / a.sum() * target
                a = _normalize(a)
                _update_global(a)

    # ------------------------------------------------------------------
    # 2️⃣  Random restarts + smooth‑max gradient descent
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # number of restarts – more for smaller n (cheaper runs)
        restarts = 8 if n <= 600 else 4

        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # Dirichlet‑like random start
            start = rng.gamma(1.0, size=n)
            start = start / start.sum() * math.sqrt(2.0 * n)
            start = _normalize(start)

            # blend with current global best when dimensions match
            if _global_state["seq"] is not None and _global_state["seq"].shape[0] == n:
                start = (start + _global_state["seq"]) * 0.5
                start = _normalize(start)

            seq_opt, _ = _smooth_max_descent(start, deadline, rng)
            _update_global(seq_opt)

            # early exit if we already beat the target bound
            if _global_state["score"] < 1.503:
                print("[early exit] target achieved")
                return [_global_state["seq"][i].item() for i in range(_global_state["seq"].shape[0])]

    # ------------------------------------------------------------------
    # 3️⃣  Final cheap refinement – random‑mass‑swap hill‑climb
    # ------------------------------------------------------------------
    if _global_state["seq"] is not None:
        _random_swap_hillclimb(_global_state["seq"], deadline, rng)
    else:
        # fallback – a simple uniform vector (should never happen)
        n = 200
        _global_state["seq"] = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        _global_state["score"] = _fast_score(_global_state["seq"])

    # Return a plain Python list as required by the outer harness
    final_seq = _global_state["seq"]
    return [float(x) for x in final_seq]
