def solve_lp(f, M, target_sum):
    n = len(f)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    b = np.full(L, M, dtype=np.float64)

"""
Propose a non‑negative step‐function sequence that minimizes the evaluation
function defined in the challenge.

The algorithm proceeds in three stages for one or more candidate lengths:

1️⃣  Randomised warm starts (Dirichlet, uniform or from the previous best).
    Each start is refined by a fast simulated‑annealing walk which preserves the
    ℓ¹‑norm (∑a_i = √(2 n)).  The walk transfers a small amount of mass between
    two coordinates while accepting uphill moves according to an exponential
    cooling schedule.

2️⃣  After the annealer stops, a linear‑programming “boost” is performed.
    For the current vector f we maximise ∑g_i under the linear constraints
    (f*g)_k ≤ M (where M = max(f*f)).  The solution g has a larger ℓ¹‑norm,
    is rescaled to the target norm and mixed with f (t=0.5).  This step is
    guaranteed not to increase the maximal auto‑convolution and often lowers it.

3️⃣  A short projected sub‑gradient descent polishes the best vector found.

All work respects the supplied time budget (default 1000 s).  The routine
returns a plain list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _fft_max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a*a using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    needed = 2 * len(a) - 1
    return float(np.max(conv[:needed]))


def _initialize_sequence(
    n: int, seed_seq=None, rng: np.random.Generator | None = None
) -> np.ndarray:
    """Create a non‑negative start of length n with ℓ¹‑norm √(2 n)."""
    target = math.sqrt(2.0 * n)
    if seed_seq is not None:
        seq = np.array(seed_seq, dtype=np.float64)
        if len(seq) != n:
            if len(seq) > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - len(seq))])
        seq = np.clip(seq, 0.0, 1000.0)
        s = np.sum(seq)
        if s == 0.0:
            seq = np.full(n, target / n)
        else:
            seq = seq * (target / s)
        return seq

    if rng is None:
        rng = np.random.default_rng()
    # Dirichlet‑like start: Gamma draws give a richer variety than pure uniform
    gamma = rng.gamma(0.7, 1.0, size=n)  # shape <1 ⇒ sparser vectors
    seq = gamma / np.sum(gamma) * target
    return seq


def _simulated_annealing(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Transfer mass between two coordinates while keeping the ℓ¹‑norm fixed."""
    n = len(a)
    target_sum = np.sum(a)               # = √(2 n)
    fft_len = 1 << ((2 * n - 1).bit_length())
    cur_max = _fft_max_convolution(a, fft_len)

    best_seq = a.copy()
    best_val = cur_max

    rng = np.random.default_rng()
    start = time.monotonic()
    total_time = max(1e-12, deadline - start)

    init_temp = max(cur_max * 0.5, 1e-4)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = elapsed / total_time
        # exponential cooling
        T = init_temp * (0.0001 ** frac)

        # step size shrinks with temperature
        avg = target_sum / n
        step = avg * (T / init_temp) * 3.0

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = rng.normal(scale=step)
        # keep within bounds
        if delta > a[i]:
            delta = a[i]
        if -delta > a[j]:
            delta = -a[j]
        if abs(delta) < 1e-12:
            continue

        a[i] -= delta
        a[j] += delta

        new_max = _fft_max_convolution(a, fft_len)

        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 1e-12:
                prob = math.exp(-(new_max - cur_max) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()
        else:
            a[i] += delta
            a[j] -= delta

    return best_seq, best_val


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build the (2n‑1)×n matrix A with A[k,i] = f[k‑i] when in range, else 0."""
    n = len(f)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Linear programme:
        maximise   ∑g_i
        subject to  (f * g)_k ≤ M  for all k,
                    g_i ≥ 0.
    Returns the optimal g (may be all zeros if the LP fails).
    """
    n = len(f)
    L = 2 * n - 1
    A = _conv_matrix(f)          # shape (L, n)
    b = np.full(L, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximise sum ⇒ minimise −sum
    bounds = [(0.0, None) for _ in range(n)]

    # use the HiGHS solver – fast and available in recent SciPy builds
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    # fallback: zero vector (won’t improve the bound)
    return np.zeros_like(f)


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the ℓ¹‑simplex {x≥0, ∑x = s}.
    Implementation follows the algorithm by Duchi et al. (2008).
    """
    if np.sum(v) == s and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max_conv(f: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the map f ↦ max_k (f*f)_k.
    For each active index (where the convolution attains its maximum)
    we add the gradient of that linear piece.
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    eps = max_val * 1e-9 + 1e-12
    active = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(f)

    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += 2.0 * f[j]
        # correct the double‑count on the diagonal
        diag = (i == j)
        if np.any(diag):
            grad[i[diag]] -= f[j[diag]]
    return grad


def _polish_subgradient(a: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent for a few iterations (fast polishing)."""
    n = len(a)
    target_sum = np.sum(a)
    fft_len = 1 << ((2 * n - 1).bit_length())
    cur = _fft_max_convolution(a, fft_len)

    while time.monotonic() < deadline:
        g = _subgradient_max_conv(a)
        step = target_sum * 0.05
        a_candidate = _project_onto_simplex(a - step * g, target_sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        nxt = _fft_max_convolution(a_candidate, fft_len)
        if nxt < cur - 1e-12:
            a = a_candidate
            cur = nxt
        else:
            # no improvement – break early
            break
    return a

    remaining_total = deadline - time.monotonic()
    per_len_budget = max(5.0, remaining_total / max(1, len(length_candidates)))

    for n in length_candidates:
        if time.monotonic() >= deadline:
            break

        # dedicated time slice for this length
        sub_deadline = min(deadline,
                           time.monotonic() + per_len_budget)

        target_sum = math.sqrt(2.0 * n)
        fft_len = 1 << ((2 * n - 1).bit_length())

        # ----------------------------------------------------------------
        # 2️⃣  Run several independent annealing runs inside the slice
        # ----------------------------------------------------------------
        while time.monotonic() < sub_deadline - 5.0:   # keep a few seconds for LP & polish
            # small time budget for a single annealing run (≈15–25 s)
            run_time = min(20.0, sub_deadline - time.monotonic() - 4.0)
            run_deadline = time.monotonic() + run_time

            # pick an initialisation method
            method = rng.choice(["warm", "dirichlet", "uniform"])
            if method == "warm" and h1 is not None and len(h1) == n:
                a0 = _initialize_sequence(n, seed_seq=h1, rng=rng)
            elif method == "dirichlet":
                # Dirichlet with a random concentration parameter
                alpha = rng.choice([0.2, 0.5, 1.0, 2.0, 5.0])
                g = rng.gamma(alpha, 1.0, size=n)
                a0 = g / np.sum(g) * target_sum
            else:
                a0 = _initialize_sequence(n, rng=rng)

            # simulated annealing
            seq_local, max_local = _simulated_annealing(a0.copy(), run_deadline)

            # update global best if improvement
            if max_local < best_score:
                best_score = max_local
                best_seq = seq_local.copy()
                print(f"[update] n={n} SA score={max_local:.6f}")

            # ----------------------------------------------------------------
            # 3️⃣  Linear‑programming boost (if time permits)
            # ----------------------------------------------------------------
            if time.monotonic() + 3.0 < sub_deadline:
                M = max_local
                g_opt = _solve_lp_max_sum(seq_local, M)
                sum_g = np.sum(g_opt)
                if sum_g > target_sum * 1.01:               # worthwhile improvement
                    scale = target_sum / sum_g
                    g_scaled = g_opt * scale
                    # convex combination (t=0.5 is a safe step)
                    t_mix = 0.5
                    f_new = (1 - t_mix) * seq_local + t_mix * g_scaled
                    f_new = _project_onto_simplex(f_new, target_sum)
                    f_new = np.clip(f_new, 0.0, 1000.0)
                    new_max = _fft_max_convolution(f_new, fft_len)
                    if new_max < best_score:
                        best_score = new_max
                        best_seq = f_new.copy()
                        print(f"[LP boost] n={n} new score={new_max:.6f}")

        # --------------------------------------------------------------------
        # 4️⃣  Short deterministic polish for the best sequence of this length
        # --------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            polish_deadline = min(deadline, time.monotonic() + 5.0)
            polished = _polish_subgradient(best_seq.copy(), polish_deadline)
            polished_max = _fft_max_convolution(polished, fft_len)
            if polished_max < best_score:
                best_score = polished_max
                best_seq = polished.copy()
                print(f"[polish] n={n} polish score={polished_max:.6f}")

        # recompute remaining allocation for remaining lengths
        remaining_total = deadline - time.monotonic()
        lengths_left = len(length_candidates) - (length_candidates.index(n) + 1)
        if lengths_left > 0:
            per_len_budget = max(5.0, remaining_total / lengths_left)

    # --------------------------------------------------------------------
    # 5️⃣  Final global polish (if any time left)
    # --------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline - 0.5:
        final_deadline = deadline
        best_seq = _polish_subgradient(best_seq, final_deadline)
        # recompute exact score using the official evaluator (just for safety)
        # NOTE: evaluate_sequence is already imported in the execution environment.
        final_score = evaluate_sequence(best_seq.tolist())
        if final_score < best_score:
            best_score = final_score
            print(f"[final polish] score={final_score:.6f}")

    # --------------------------------------------------------------------
    # 6️⃣  Fallback – uniform vector if nothing was found
    # --------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # ensure at least the minimal ℓ¹‑norm required by the evaluator
    if np.sum(best_seq) < 0.01:
        best_seq[0] = 0.01

    return [float(x) for x in best_seq]
