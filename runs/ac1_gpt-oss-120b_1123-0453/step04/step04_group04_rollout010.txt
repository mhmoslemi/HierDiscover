def _project_onto_simplex(v, s):
    if s < 0:
        raise
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

"""
Gradient–softmax optimizer for the step‑function bound.

The algorithm searches across several sequence lengths. For each length it
performs a few random restarts.  In each restart it minimises a smooth
approximation of  maxᵢ (a*a)[i]  by using a soft‑max (log‑sum‑exp) with a
temperature β and projected gradient descent onto the ℓ¹‑simplex
‖a‖₁ = √(2 n).  The gradient of the smoothed objective is a linear
correlation that can be evaluated with a single FFT, so many iterations
fit easily into the 1000 s budget.  After the main phase a few
geometric‑shape candidates are tried as cheap fall‑backs.

The returned sequence is a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be zero – return the uniform point
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _initialize_sequence(
    n: int, seed_seq=None, rng: np.random.Generator | None = None
) -> np.ndarray:
    """Create a non‑negative length‑n vector with ℓ¹‑norm √(2 n)."""
    target = math.sqrt(2.0 * n)
    if seed_seq is not None:
        seq = np.array(seed_seq, dtype=np.float64)
        if seq.size != n:
            if seq.size > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - seq.size)])
        seq = np.clip(seq, 0.0, 1000.0)
        s = seq.sum()
        if s == 0.0:
            seq = np.full(n, target / n)
        else:
            seq = seq * (target / s)
        return seq

    if rng is None:
        rng = np.random.default_rng()
    # start from a random point on the simplex
    seq = rng.random(n) + 1e-6
    seq = seq / seq.sum() * target
    return seq


def _fft_max_convolution(a: np.ndarray, fft_len: int | None = None) -> float:
    """Return max entry of the full auto‑convolution a*a using an FFT."""
    n = a.size
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
    return float(conv.max())

# Main search routine
# ----------------------------------------------------------------------

        sub_deadline = min(deadline, time.monotonic() + per_len)

        target_sum = math.sqrt(2.0 * n)

                # a) compute full convolution (FFT)
                # -------------------------------------------------
                Fa = np.fft.rfft(a, n=fft_len_conv)
                conv = np.fft.irfft(Fa * Fa, n=fft_len_conv)[: 2 * n - 1]

                # -------------------------------------------------
                # b) soft‑max probabilities over convolution entries
                # -------------------------------------------------
                mx = conv.max()
                # stability: subtract the max before exponentiation
                exp_vals = np.exp(beta * (conv - mx))
                p = exp_vals / exp_vals.sum()

                # -------------------------------------------------
                # c) gradient of the smoothed max (correlation)
                # -------------------------------------------------
                Fp = np.fft.rfft(p, n=fft_len_corr)
                Fa_corr = np.fft.rfft(a, n=fft_len_corr)
                corr_full = np.fft.irfft(np.conj(Fp) * Fa_corr, n=fft_len_corr)
                # linear correlation length = 3n‑2; gradient needed for indices 0…n‑1:
                start_idx = n - 1
                grad = 2.0 * corr_full[start_idx : start_idx + n]

                # -------------------------------------------------
                # d) projected gradient step
                # -------------------------------------------------
                step = step_base / (1.0 + it * 1e-3)  # slowly decaying step
                a_candidate = a - step * grad
                a_candidate = _project_onto_simplex(a_candidate, target_sum)
                a_candidate = np.clip(a_candidate, 0.0, 1000.0)

                # evaluate new max
                new_max = _fft_max_convolution(a_candidate, fft_len_conv)

                if new_max < cur_max - 1e-12:
                    # accepted improvement
                    a = a_candidate
                    cur_max = new_max
                    if cur_max < best_val:
                        best_val = cur_max
                        best_seq = a.copy()
                        print(f"[update] n={n} restart={r} iter={it} val={best_val:.6f}")
                    # keep the current temperature
                else:
                    # no improvement – make the soft‑max sharper a bit
                    if beta_idx < len(beta_schedule) - 1:
                        beta_idx += 1
                        beta = beta_schedule[beta_idx]
                    else:
                        # occasional random jitter to escape flat regions
                        if it % 50 == 0:
                            noise = rng.normal(scale=step * 0.3, size=n)
                            a = _project_onto_simplex(a + noise, target_sum)
                            a = np.clip(a, 0.0, 1000.0)
                            cur_max = _fft_max_convolution(a, fft_len_conv)

        # --------------------------------------------------------------
        # 3️⃣  Try a few deterministic geometric families (cheap heuristics)
        # --------------------------------------------------------------
        if time.monotonic() < sub_deadline:
            # geometric progression a_i = r^i (scaled to the required ℓ¹ norm)
            for r in np.linspace(0.2, 0.99, 9):
                base = np.power(r, np.arange(n, dtype=np.float64))
                scale = target_sum / base.sum()
                a_geo = base * scale
                val_geo = _fft_max_convolution(a_geo, fft_len_conv)
                if val_geo < best_val:
                    best_val = val_geo
                    best_seq = a_geo.copy()
                    print(f"[geo] n={n} r={r:.2f} val={best_val:.6f}")

        # --------------------------------------------------------------
        # 4️⃣  Update remaining budget and per‑length allocation
        # --------------------------------------------------------------
        remaining = deadline - time.monotonic()
        left_lengths = len(length_candidates) - (length_candidates.index(n) + 1)
        per_len = max(0.5, remaining / max(1, left_lengths))

    # ------------------------------------------------------------------
    # 5️⃣  Fallback in the extremely unlikely case no sequence was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # optional final report (uses the provided evaluate_sequence)
    try:
        final_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"[final] best score = {final_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
