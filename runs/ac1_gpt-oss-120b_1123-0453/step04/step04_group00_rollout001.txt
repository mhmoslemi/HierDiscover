def interleave(values):
    result = []
    def rec(vals):
        if not vals:
            return
        mid = len(vals)//2
        result.append(vals[mid])
        rec(vals[:mid])
        rec(vals[mid+1:])
    rec(values)
    return np.array(result)

"""
Improved search for low‑value step sequences.

The algorithm consists of three phases:

0️⃣  Warm‑start from any previously discovered sequence (``height_sequence_1``).

1️⃣  Broad parametric sweep (uniform, power‑law, exponential, two‑level shapes)
    over a set of lengths (up to 2000).  Each candidate is normalised so that
    ``∑ a_i = sqrt(2·n)`` – under this normalisation the evaluation score equals
    the maximal auto‑convolution value.

2️⃣  For the most promising lengths we launch many random restarts of a
    projected **momentum gradient descent** that minimises a smooth‑max
    approximation of the convolution peak.  The smoothing parameter ``β`` is
    annealed from ≈1 to a few thousand, the step size decays, and occasional
    jitter avoids stagnation.  The iterate is always projected onto the
    feasible set (non‑negative, bounded, fixed ℓ¹‑norm).

3️⃣  A cheap stochastic hill‑climb (mass‑swap moves) refines the current global
    best until the time budget expires.

All heavy work uses FFT‑accelerated convolutions, and the code respects the
1000 s time limit by checking the deadline frequently.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return a fast length for FFT (fallback to power‑of‑2)."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:      # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (efficient for many calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _max_convolution(seq: np.ndarray) -> float:
    """Exact maximal entry of the linear auto‑convolution of ``seq``."""
    conv = _linear_convolve(seq, seq)      # length 2·n‑1
    return float(conv.max())


def _normalize(seq: np.ndarray) -> np.ndarray:
    """
    Clip to [0, 1000] and enforce ℓ¹‑norm = sqrt(2·n).
    The function works in‑place‑free and returns a new array.
    """
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total > 0.0:
        seq = seq * (target / total)
    else:
        seq = np.full(n, target / n, dtype=np.float64)
    return seq


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max (log‑sum‑exp) weights for the entries of ``conv``.
    ``beta`` controls the sharpness (β→∞ recovers the true max).
    """
    cmax = conv.max()
    # subtract the max to keep numbers in a stable range
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    If ``w`` contains the (soft) max‑weights of the convolution,
    the gradient w.r.t. ``a_i`` is 2·∑_j w_{i+j}·a_j.
    This cross‑correlation is obtained with a single FFT convolution.
    """
    w_rev = w[::-1]                                 # reverse the weight vector
    corr = _linear_convolve(a, w_rev)               # length = n + (2n‑1) – 1 = 3n‑2
    n = a.shape[0]
    # extract the central n entries (indices n‑1 … 2·n‑2) and reverse
    segment = corr[n - 1 : 2 * n - 1]                # shape (n,)
    grad = 2.0 * segment[::-1]
    return grad


def _optimize_sequence(
    init_seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    *,
    max_iter: int = 4000,
    beta_start: float = 1.0,
    beta_mult: float = 1.7,
    beta_max: float = 3000.0,
    step_factor: float = 2.5,
    step_decay: float = 0.001,
    jitter_every: int = 200,
    momentum: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with momentum on the smooth‑max surrogate.

    Returns the best vector seen during the run and its true maximal
    auto‑convolution value.
    """
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    # one‑shot normalisation (just in case)
    a = _normalize(a)

    best_local = a.copy()
    best_score = _max_convolution(a)

    beta = beta_start
    base_step = step_factor * target / n          # ~ step_factor·average entry
    velocity = np.zeros_like(a)

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        # ---- current objective ----
        conv = _linear_convolve(a, a)            # length 2·n‑1
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # ---- smooth‑max weights and gradient ----
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # ---- momentum step ----
        step = base_step / (1.0 + it * step_decay)
        velocity = momentum * velocity - step * grad
        a = a + velocity

        # ---- projection onto feasible set ----
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # ---- occasional jitter to escape plateaux ----
        if it % jitter_every == 0 and it > 0:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * 0.03
                a[i] -= delta
                a[j] += delta
                a = _normalize(a)

        # ---- anneal the smoothness parameter ----
        if it % 100 == 0 and beta < beta_max:
            beta = min(beta * beta_mult, beta_max)

        it += 1

    return best_local, best_score


def _spread_permute(values: np.ndarray) -> np.ndarray:
    """
    Re‑order ``values`` so that the largest entries are placed as far apart as
    possible (alternating ends).  The input is taken as a 1‑D array; the
    returned array has the same multiset of values.
    """
    n = values.shape[0]
    # sort descending – the biggest values will be placed first
    sorted_vals = np.sort(values)[::-1]
    # generate the "alternating ends" index pattern: 0, n‑1, 1, n‑2, …
    idx = np.empty(n, dtype=int)
    lo, hi, pos = 0, n - 1, 0
    while lo <= hi:
        idx[pos] = lo
        pos += 1
        lo += 1
        if lo <= hi:
            idx[pos] = hi
            pos += 1
            hi -= 1
    new_seq = np.empty_like(values)
    new_seq[idx] = sorted_vals
    return new_seq

def propose_candidate(

    Search for a non‑negative step sequence with a low evaluation score.

    The routine combines a warm‑start, an extensive parametric sweep,
    many random restarts of a momentum‑based smooth‑max gradient descent,
    and a final stochastic hill‑climb.  The best sequence found before the
    time budget expires is returned as a plain Python list.
    """

    best_seq = None          # type: np.ndarray | None
    best_score = float("inf")

        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global best] n={len(seq)} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously saved construction (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:

            cand = _normalize(cand)
            _maybe_update(cand)
        except Exception:      # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Broad parametric sweep
    # ------------------------------------------------------------------
    lengths = [2000, 1500, 1200, 1000, 800, 600, 500, 400, 300, 200, 150, 100]
    power_ps = np.linspace(0.2, 5.0, 30)          # exponents for power‑law
    alphas = np.linspace(0.001, 0.30, 30)        # decay rates for exponential
    plateau_fracs = [0.05, 0.10, 0.20, 0.30]      # fraction of indices for a flat top

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # uniform baseline
        a_uni = np.full(n, 1.0, dtype=np.float64)
        a_uni = _normalize(a_uni)
        _maybe_update(a_uni)

        idx = np.arange(1, n + 1, dtype=np.float64)

        # power‑law family
        for p in power_ps:
            a = idx ** (-p)
            a = _normalize(a)

            # also try a spread permutation of the same multiset
            a_spread = _spread_permute(a)
            a_spread = _normalize(a_spread)
            _maybe_update(a_spread)

        # exponential family

            a = _normalize(a)
            _maybe_update(a)
            a_spread = _spread_permute(a)
            a_spread = _normalize(a_spread)
            _maybe_update(a_spread)

        # two‑level (plateau + exponential tail)
        for frac in plateau_fracs:
            k = max(1, int(frac * n))
            tail_len = n - k
            if tail_len <= 0:
                continue
            tail = np.exp(-0.05 * np.arange(tail_len, dtype=np.float64))
            a = np.concatenate([np.full(k, 1.0), tail])
            a = _normalize(a)
            _maybe_update(a)
            a_spread = _spread_permute(a)
            a_spread = _normalize(a_spread)
            _maybe_update(a_spread)

    # ------------------------------------------------------------------
    # 2️⃣ Momentum smooth‑max gradient descent with many random restarts
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # decide how many restarts based on size (more cheap runs for small n)
        if n <= 600:

            max_iter = 8000
        else:
            restarts = 4

            if best_score <= 1.503:

                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # 3️⃣ Final cheap stochastic hill‑climb (mass swaps)
    # ------------------------------------------------------------------
    if best_seq is not None:
        cur_seq = best_seq.copy()
        cur_score = best_score
        while time.monotonic() < deadline:
            n = cur_seq.shape[0]
            i, j = rng.choice(n, size=2, replace=False)

            delta = cur_seq[i] * 0.025                     # move 2.5 % of the source mass
            cand = cur_seq.copy()
            cand[i] -= delta

            # otherwise discard

    else:
        # Fallback – uniform vector of moderate length

        cur_seq = _normalize(cur_seq)
        best_seq = cur_seq
        best_score = _max_convolution(cur_seq)

    # ------------------------------------------------------------------
    # Return the best sequence discovered as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
