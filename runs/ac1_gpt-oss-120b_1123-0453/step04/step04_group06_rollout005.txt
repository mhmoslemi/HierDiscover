from scipy.optimize import linprog

def lp_max_sum(f, M):
    n = len(f)
    # Build constraint matrix A_ub (size (2n-1, n))
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f
    # bounds
    bounds = [(0, None)] * n
    # objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    # Solve
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x
```

    n = len(f)
    # Build conv matrix A.
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):

    # Bounds: g_i >=0, M >=0

                  bounds=bounds, method='highs')

    g = res.x[:n]
    M_opt = res.x[-1]

A hybrid optimizer for the step‑function peak‑convolution problem.

The evaluation of a sequence a (after the required normalisation
∑a = √(2 n)) equals the maximal entry of the auto‑convolution a*a.
We therefore minimise that peak.

The algorithm proceeds in three stages for each chosen length n:

1.  **LP‑based descent** – given a current vector f we solve a tiny linear
    programme that finds a vector g (∑g = √(2 n), g≥0) with the smallest

    peak.

3.  **Gaussian hill‑climbing** – a short stochastic refinement on the
    simplex for the final polish.

global time budget (≈ 1000 s).  The best sequence found is returned as a

# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # edge case: all zero after projection
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Evaluation helpers
# ------------------------------------------------------------
def _compute_score(v: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a vector v (∑v = √(2 n))."""
    conv = np.convolve(v, v)
    max_val = np.max(conv)
    sum_val = np.sum(v)
    if sum_val < 0.01:
        return float("inf")
    n = len(v)
    return float(2.0 * n * max_val / (sum_val * sum_val))

def _max_conv(v: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution (faster than full score)."""
    return np.max(np.convolve(v, v))

# ------------------------------------------------------------
# Gradient‑descent on a smooth maximum (re‑use from the reference)
# ------------------------------------------------------------
def _weighted_gradient(f: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(f*f)_k w.r.t. f.
    `w` must have length 2·len(f)-1 and Σ w = 1.
    """
    n = len(f)
    grad = np.zeros_like(f)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        grad[i_start:i_end + 1] += wk * 2.0 * f[j_range]
        if (k & 1) == 0:               # even → possible i=j
            i_eq = k // 2
            if i_start <= i_eq <= i_end:
                grad[i_eq] -= wk * f[i_eq]
    return grad


def _smooth_max_optimize(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2000,
    beta_start: float = 0.5,
    beta_factor: float = 1.5,
    beta_max: float = 256.0,
    step_initial: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a log‑sum‑exp (smooth‑max) proxy for the
    peak of the auto‑convolution.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _compute_score(f)

    step = step_initial
    beta = beta_start
    no_improve = 0

    for _ in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(f, f)
        max_val = conv.max()

        # soft‑max weights (stable formulation)
        exp_vals = np.exp(beta * (conv - max_val))
        w_sum = exp_vals.sum()
        w = exp_vals / w_sum if w_sum != 0.0 else np.full_like(conv, 1.0 / conv.size)

        grad = _weighted_gradient(f, w)

        cand = _project_onto_simplex(f - step * grad, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _compute_score(cand)

        if cand_score + 1e-12 < best_score:
            best_f, best_score = cand.copy(), cand_score
            f = cand
            step *= 1.02
            no_improve = 0
        else:
            step *= 0.7
            no_improve += 1

        beta = min(beta * beta_factor, beta_max)

        if no_improve >= 200:
            beta = max(beta_start, beta / 2.0)
            no_improve = 0

        if step < 1e-12:
            break

    return best_f, best_score


def _hill_climb(
    start: np.ndarray,
    target_sum: float,
    deadline: float,
    sigma_initial: float = 0.2,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian perturbation hill‑climbing on the simplex.
    """
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score(best)
    sigma = sigma_initial

    while time.monotonic() < deadline and max_iters > 0:
        cand = best + sigma * rng.normal(size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _compute_score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        sigma *= 0.9995
        max_iters -= 1
        if sigma < 1e-9:
            sigma = sigma_initial
    return best, best_score

# ------------------------------------------------------------
# Linear‑programming step (minimise the bound M)
# ------------------------------------------------------------
def _solve_min_M_lp(f: np.ndarray, target_sum: float):
    """
    Solve the LP:
        minimise M
        s.t.   (f * g)_k ≤ M   for all k
               Σ g_i = target_sum
               g_i ≥ 0 , M ≥ 0
    Returns (g, M) on success, otherwise (None, None).
    """
    n = len(f)
    # Build convolution matrix A where (A @ g)_k = (f * g)_k
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f

    # Augment with the M variable (last column = -1)
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1), dtype=np.float64)])
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)

    # Equality: sum(g) = target_sum
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([target_sum], dtype=np.float64)

    # Bounds: g_i ≥ 0, M ≥ 0
    bounds = [(0, None)] * n + [(0, None)]

    # Objective: minimise M  →  c = [0,…,0,1]
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None, None
    g = res.x[:n]
    M_opt = res.x[-1]
    return g, M_opt

# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low peak of its auto‑convolution.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a small margin for the outer harness
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------
    # Warm‑start from a previously discovered construction
    # --------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        n0 = len(seq)
        target0 = math.sqrt(2.0 * n0)
        if seq.sum() > 0:
            seq = seq * (target0 / seq.sum())
        else:
            seq = np.full(n0, target0 / n0, dtype=np.float64)
        sc = _compute_score(seq)
        best_seq, best_score = seq.copy(), sc
        print(f"[init] warm‑start n={n0} score={sc:.6f}")

    # --------------------------------------------------------
    # Lengths to explore – we try a handful spread across the range.
    # --------------------------------------------------------
    lengths = [300, 400, 500, 600, 800, 1000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # number of random restarts (more for smaller n)
        restarts = 8 if n <= 600 else 5
        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # initialise a random feasible point on the simplex
            if best_seq is not None and len(best_seq) == n:
                f = best_seq.copy()
            else:
                f = rng.random(n) + 1e-8
                f = f / f.sum() * target_sum

            # ----------------------------------------------------
            # LP‑based descent phase
            # ----------------------------------------------------
            t = 0.5                       # mixing coefficient
            max_lp_iters = 70
            for it in range(max_lp_iters):
                if time.monotonic() >= deadline:
                    break

                cur_M = _max_conv(f)

                # solve the LP that yields the best improvement direction
                g, M_opt = _solve_min_M_lp(f, target_sum)
                if g is None:
                    break

                # try to mix f and g; adapt t if needed
                improved = False
                for _ in range(5):
                    f_new = (1.0 - t) * f + t * g
                    # re‑project onto the simplex to combat numerical drift
                    f_new = _project_onto_simplex(f_new, target_sum)
                    f_new = np.clip(f_new, 0.0, 1000.0)

                    new_M = _max_conv(f_new)
                    if new_M < cur_M - 1e-12:
                        f = f_new
                        cur_M = new_M
                        improved = True
                        # modestly increase t for the next iteration
                        t = min(t * 1.2, 0.9)
                        break
                    else:
                        t *= 0.5        # back‑off

                if not improved:
                    # cannot make progress – exit the LP loop early
                    break

                # update global best
                if cur_M < best_score:
                    best_score = cur_M
                    best_seq = f.copy()
                    print(f"[update] n={n} rs={rs} it={it} score={cur_M:.6f}")

                if best_score < 1.503:
                    print("[early] target achieved, breaking")
                    deadline = time.monotonic() + 0.2
                    break

            # ----------------------------------------------------
            # Short smooth‑max gradient refinement
            # ----------------------------------------------------
            if time.monotonic() < deadline:
                sub_dead = min(deadline, time.monotonic() + 6.0)
                f_ref, sc_ref = _smooth_max_optimize(
                    f,
                    target_sum,
                    sub_dead,
                    max_iters=600,
                    beta_start=0.5,
                    beta_factor=1.3,
                    beta_max=128.0,
                    step_initial=0.3,
                )
                if sc_ref < best_score:
                    best_score = sc_ref
                    best_seq = f_ref.copy()
                    print(f"[gd] n={n} rs={rs} gd_score={sc_ref:.6f}")

            # ----------------------------------------------------
            # Final stochastic hill‑climbing (if time permits)
            # ----------------------------------------------------
            if time.monotonic() < deadline and best_seq is not None:
                refined, refined_sc = _hill_climb(best_seq, target_sum,
                                                 deadline,
                                                 sigma_initial=0.15,
                                                 max_iters=3000)
                if refined_sc < best_score:
                    best_score = refined_sc
                    best_seq = refined.copy()
                    print(f"[hill] n={n} refined_score={refined_sc:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                deadline = time.monotonic() + 0.2
                break

    # ------------------------------------------------------------
    # Fallback – should never happen, but ensures a valid return.
    # ------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _compute_score(best_seq)

    # Final safety clamp (evaluate_sequence will do the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[result] best score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
