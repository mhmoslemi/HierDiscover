    for it in range(max_iter):

        conv = np.convolve(a, a)
        M = np.max(conv)
        # Solve LP
        n = len(a)
        max_f = np.max(a)
        if max_f == 0:

        ub = M / max_f
        # Build A

        # For simplicity build dense.
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = a
        # bounds list
        bounds = [(0.0, ub) for _ in range(n)]
        # solve
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            break
        b = res.x
        sum_b = np.sum(b)

        # optionally reduce step

    return a

"""
Improved step‑function search for the evaluation function.

The optimizer combines three complementary phases inside the given
time budget (default 1000 s).

1️⃣  **Randomised warm‑starts** – each trial starts from a Dirichlet
    sample with a randomly chosen concentration (α∈{0.3,0.6,1.0,2.0})
    and is projected onto the unit simplex (∑aᵢ = 1).

2️⃣  **Projected Adam on a smooth max** – we minimise a smooth surrogate
    of the true objective

        F(a) = log‑sum‑exp( conv(a,a) / β ),

    where β≈0.1·max(conv).  Adam yields fast adaptive steps; after each
    update the vector is projected back onto the simplex and clipped to
    the permissible range [0, 1000].

3️⃣  **Linear‑programming refinement** – given the current vector f,
    we solve the LP

        maximise   1ᵀ b
        subject to  f * b ≤ M      (pointwise)
                    b ≥ 0
                    b ≤ M / max(f),

    where M = max(conv(f,f)).  The optimal b (renormalised) is mixed
    into f with a decreasing mixing factor.  This step directly attacks
    the maximal convolution entry and often yields a noticeable drop
    in the exact score.

4️⃣  **Stochastic polishing** – a simple Gaussian‑perturbation /
    random‑swap hill‑climber finishes the search.

All phases respect the overall deadline; the best sequence encountered
is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _exact_score(a: np.ndarray) -> float:
    """Score returned by `evaluate_sequence` (no internal clipping)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()
    sum_val = a.sum()
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


# ----------------------------------------------------------------------
# Smoothed Adam descent
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the entries of `conv` with temperature `beta`."""
    max_c = conv.max()
    shifted = (conv - max_c) / beta
    exp_shifted = np.exp(shifted - np.max(shifted))  # extra shift for safety
    w = exp_shifted / exp_shifted.sum()
    return w


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σₖ wₖ·convₖ(a,a) w.r.t. `a`."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    # each convolution index contributes a rank‑1 term
    for k, wk in enumerate(w):
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += wk * 2.0 * a[j]
        # diagonal i==j was added twice – keep a single copy
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= wk * a[mask]
    return grad


def _adam_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iters: int = 30000,
    patience: int = 1000,
) -> tuple[np.ndarray, float]:
    """Projected Adam minimising a smooth approximation of the max‑conv."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _exact_score(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        max_conv = conv.max()
        # a temperature that is a few‑percent of the current peak
        beta = max_conv * 0.12 + 1e-12
        w = _softmax_weights(conv, beta)
        grad = _weighted_gradient(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        cur_score = _exact_score(a)
        cand_score = _exact_score(cand)

        # accept improvement
        if cand_score < cur_score:
            a = cand

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            no_improve = 0
            print(f"[Adam] n={n:4d} it={it:5d} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return matrix A such that (A @ b)[k] = (f * b)[k]."""
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f
    return A


def _lp_refine(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iter: int = 12,
    step_init: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    LP based improvement of `a`.
    At each iteration we maximise Σbⱼ under the constraint f∗b ≤ M,
    then mix the optimiser back into `a`.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _exact_score(a)

    step = step_init
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        max_f = a.max()
        if max_f == 0:
            break

        # bound each variable to keep the LP bounded
        ub = M / max_f
        bounds = [(0.0, ub) for _ in range(n)]

        A = _build_conv_matrix(a)
        b_ub = np.full(2 * n - 1, M)

        res = linprog(
            c=-np.ones(n),  # maximise sum(b)  → minimise -sum(b)
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # LP failed – abort further refinements
            break

        b_vec = res.x
        sumb = b_vec.sum()
        if sumb < 1e-12:
            break

        g = b_vec / sumb
        # convex combination – step shrinks geometrically
        a = (1.0 - step) * a + step * g
        a = _project_onto_simplex(a, target_sum)
        a = np.clip(a, 0.0, 1000.0)

        cur_score = _exact_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[LP]  n={n:4d} it={it+1:2d} step={step:.3f} score={cur_score:.6f}")

        # slowly reduce the mixing factor
        step = max(step * 0.85, 0.02)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic polishing (Gaussian hill‑climbing + swaps)
# ----------------------------------------------------------------------


def _stochastic_refinement(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.4,
    swap_prob: float = 0.22,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _exact_score(best)
    step = step_initial
    target_sum = best.sum()

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, len(best), size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = best + rng.normal(scale=step, size=best.shape)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _exact_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = min(step_initial, step * 1.03)
        else:
            step *= 0.996

        if step < 1e-7:
            step = step_initial

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **_kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the given time
    budget and return it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from any previously discovered construction.
    # --------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        if seq.sum() == 0:
            seq = np.full_like(seq, 1.0 / len(seq))
        else:
            seq = seq / seq.sum()
        sc = _exact_score(seq)
        best_seq = seq.copy()
        best_score = sc
        print(f"[warm] loaded height_sequence_1, n={len(seq)} score={sc:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop – several lengths, restarts, Adam + LP.
    # --------------------------------------------------------------

    # lengths that give a reasonable trade‑off between quality and speed
    lengths = [400, 600, 800, 1000, 1200, 1500, 1800, 2000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = 1.0  # we work on the unit simplex

        # number of independent restarts for this length
        restarts = 6 if n <= 1200 else 4

        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ---- initialise candidate -------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
            else:
                # random Dirichlet – pick a concentration (α) at random
                alpha = rng.choice([0.3, 0.6, 1.0, 2.0])
                a0 = rng.dirichlet(alpha * np.ones(n))
                a0 = np.clip(a0, 1e-12, None)  # avoid exact zeros for LP stability
                a0 = a0 / a0.sum() * target_sum

            # ---- allocate a slice of the remaining budget for Adam --------------------
            sub_deadline = min(deadline, time.monotonic() + 12.0)  # ~12 s per Adam run
            a_opt, sc_opt = _adam_descent(
                a0,
                target_sum,
                sub_deadline,
                lr=0.06,
                max_iters=25000,
                patience=800,
            )

            # ---- LP refinement (if time permits) ------------------------------------
            if time.monotonic() < deadline:
                lp_deadline = min(deadline, time.monotonic() + 6.0)  # ~6 s for LP
                a_opt, sc_lp = _lp_refine(a_opt, target_sum, lp_deadline)
                if sc_lp < sc_opt:
                    sc_opt = sc_lp

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[update] n={n} restart={rs} score={sc_opt:.6f}")

            # early exit if we have already crossed a comfortable threshold
            if best_score < 1.45:
                print("[early] achieved satisfactory score")
                break
        if best_score < 1.45:
            break

    # --------------------------------------------------------------
    # Final polishing (Gaussian + swaps) – if we still have time.
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution of moderate length
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _exact_score(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline, step_initial=0.35)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # final safety clipping (evaluate_sequence performs the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best_n={len(best_seq)} best_score={best_score:.6f}")
    return [float(v) for v in best_seq]
