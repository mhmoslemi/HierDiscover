"""
A fast, multi‑restart optimiser for the “max‑convolution” score.

The algorithm works on the simplex via a log‑sum‑exp smooth approximation
of the max‑convolution (the same idea as in the reference code) but
extends it with:

* a larger and more aggressive continuation schedule (β up to 1000);
* more random restarts per length;
* a progressive length schedule reaching up to 5000 entries;
* warm‑starts obtained by linear interpolation when the length changes;
* a final short CMA‑ES‑style stochastic polish (Gaussian perturbations).

The whole search respects the global time budget (default 1000 s) and
returns the best sequence found in plain ``list[float]`` format.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_to_a(z: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Convert an unrestricted vector ``z`` to a point ``a`` on the
    ℓ¹‑simplex with prescribed sum ``target_sum`` using a soft‑max style
    re‑parameterisation.
    """
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z) * target_sum
    return a


def _obj_and_grad(z: np.ndarray, n: int, S: float, beta: float):
    """
    Smooth surrogate of the max‑convolution.
    Returns a tuple ``(smooth, grad_z)`` where ``smooth`` is
        (1/β)·log Σ exp(β·conv)
    and ``grad_z`` the gradient w.r.t. the soft‑max parameters ``z``.
    """
    a = _softmax_to_a(z, S)                     # shape (n,)

    # ----- auto‑convolution via FFT (real) -----
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]      # length 2n‑1, real

    # ----- smooth max (log‑sum‑exp) -----
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))            # stable exp
    sum_exps = np.sum(exps)
    smooth = max_c + np.log(sum_exps) / beta
    w = exps / sum_exps                               # soft‑max weights, Σw = 1

    # ----- gradient w.r.t. a : g_a = 2·(a_rev ⋆ w)_{n‑1 … n‑1+n‑1} -----
    a_rev = a[::-1]
    L2 = _next_pow2(3 * n - 2)
    fr = np.fft.rfft(a_rev, L2)
    fw = np.fft.rfft(w, L2)
    conv2 = np.fft.irfft(fr * fw, L2)                # length L2 (real)
    start = n - 1
    g_raw = conv2[start: start + n]                 # length n
    g_a = 2.0 * g_raw

    # ----- chain rule back to z (soft‑max) -----
    # dΦ/dz_i = a_i·g_a_i - a_i·(a·g_a)/S
    total = np.dot(g_a, a)
    grad_z = a * g_a - a * (total / S)

    return smooth, grad_z.astype(np.float64)


def _stochastic_refinement(start_seq: np.ndarray,
                           deadline: float,
                           step_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian hill‑climbing that respects the ℓ¹‑simplex.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = evaluate_sequence(best.tolist())
    step = step_initial
    S = best.sum()

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑project onto the simplex (same soft‑max‑based projection)
        # we reuse the same total S to keep the scale constant
        cand = _softmax_to_a(np.log(np.clip(cand / S, 1e-12, None)), S)

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # slowly shrink the step size – occasional resets keep search alive
        step *= 0.9995
        if step < 1e-9:
            step = step_initial

    return best, best_score


def _interpolate_to_length(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length ``new_n``."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    return np.interp(x_new, x_old, seq)


def _optimise_length(n: int,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> tuple[np.ndarray, float]:
    """
    Run several L‑BFGS‑B optimisations on a single length ``n``.
    Returns the best sequence and its exact ``evaluate_sequence`` score.
    """
    # a richer continuation schedule – larger β values improve the max‑approximation
    betas = [5.0, 15.0, 40.0, 80.0, 200.0, 500.0, 1000.0]
    max_restarts = 8
    best_seq = None
    best_val = float("inf")

    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # ----- initialise z -------------------------------------------------
        if warm_seq is not None and warm_seq.shape[0] == n:
            # soft‑max inverse (avoid log(0))
            p = np.clip(warm_seq / target_sum, 1e-12, None)
            z0 = np.log(p)
        else:
            # random normal start
            z0 = rng.normal(scale=0.1, size=n)

        # ----- continuation over β -----------------------------------------
        for beta in betas:
            if time.monotonic() >= deadline:
                break

            res = minimize(fun=lambda zz: _obj_and_grad(zz, n, target_sum, beta),
                           x0=z0,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 250,
                                    'gtol': 1e-12,
                                    'disp': False})

            z0 = res.x                     # warm start next β

            # evaluate the exact (non‑smooth) score
            a = _softmax_to_a(z0, target_sum)
            a = np.clip(a, 0.0, 1000.0)
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[opt] n={n} restart={restart} β={beta:.0f} val={val:.6f}")

            # early exit if we have already crossed the target
            if best_val < 1.503:
                return best_seq, best_val

    # fallback – uniform distribution (never worse than 2)
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())

    return best_seq, best_val


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence.
    The function respects the global time budget (≈ budget_s seconds)
    and returns a plain Python list of floats ready for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    h1 = globals().get("height_sequence_1")
    if isinstance(h1, (list, tuple, np.ndarray)):
        seq = np.asarray(h1, dtype=np.float64)
        n0 = seq.size
        S0 = math.sqrt(2.0 * n0)
        if seq.sum() > 0.0:
            seq = seq * (S0 / seq.sum())
        else:
            seq = np.full(n0, S0 / n0, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        val = evaluate_sequence(seq.tolist())
        best_seq = seq.copy()
        best_score = val
        print(f"[init] warm‑start n={n0} val={val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Deterministic optimisation over a set of lengths
    # ------------------------------------------------------------------
    # Choose a mix of moderate and large lengths – larger n gives a richer search space.
    lengths = [500, 800, 1200, 1800, 2500, 3500, 5000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)   # keep the same normalisation convention

        # warm‑start: either reuse the current best (interpolated) or nothing
        warm = None
        if best_seq is not None:
            warm = _interpolate_to_length(best_seq, n)

        seq_n, val_n = _optimise_length(n, target_sum, deadline, rng, warm)

        if val_n < best_score:
            best_score = val_n
            best_seq = seq_n.copy()
            print(f"[global update] n={n} val={val_n:.6f}")

        if best_score < 1.503:
            print("[early stop] target reached")
            break

    # ------------------------------------------------------------------
    # 2️⃣ Final stochastic polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refined, refined_val = _stochastic_refinement(best_seq, deadline,
                                                      step_initial=0.3)
        if refined_val < best_score:
            best_seq = refined
            best_score = refined_val
            print(f"[final polish] val={refined_val:.6f}")

    # final clipping (evaluate_sequence does the same, but we keep it safe)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[result] best score = {best_score:.6f}  (length={best_seq.size})")
    return [float(x) for x in best_seq]
