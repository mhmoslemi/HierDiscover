"""
Propose a low‑value non‑negative step sequence.

The search combines three ideas that differ from the previous
Adam‑based approach:

1️⃣  **Linear‑programming “pair‑wise” update**.
    For a fixed length `n` we keep the ℓ¹‑norm of the vector equal to 1.
    Given a current sequence `f`, we compute `M = max(conv(f,f))` and solve
    the linear program

        maximise   Σ g_i
        subject to conv(f,g) ≤ M   (one inequality per convolution index)
                   g_i ≥ 0

    The LP pushes mass into coordinates where `f` is small, thereby
    flattening the auto‑convolution.  The obtained `g` is re‑scaled to the
    required ℓ¹‑norm and mixed with `f` (convex combination).  Repeating
    this step steadily reduces `max(conv(f,f))`.

2️⃣  **Projected Adam refinement** – after a handful of LP‑updates we run
    a short Adam optimiser (identical to the one used before) to fine‑tune
    the sequence on the smooth `log‑sum‑exp` approximation of the max.

3️⃣  **Stochastic polishing** – a final hill‑climbing phase with Gaussian
    perturbations and occasional swaps, always re‑projected onto the simplex.

All steps respect the global time budget (default 1000 s).  The algorithm
explores several lengths (up to 2000) and a few random restarts per length.
The best sequence found is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper utilities (same as in the reference implementation)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_vector(f: np.ndarray) -> float:
    """Exact objective used by `evaluate_sequence` (scale‑invariant)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    sum_val = np.sum(f)
    if sum_val < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (sum_val * sum_val))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / beta
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    return w


def _weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """Gradient of Σ w_k·conv_k(a,a) w.r.t. `a`."""
    n = len(a)
    grad = np.zeros_like(a)
    for k, wk in enumerate(weights):
        if wk == 0.0:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        vals = a[j]
        grad[i] += wk * 2.0 * vals
        mask = (i == j)
        if np.any(mask):
            grad[i[mask]] -= wk * vals[mask]
    return grad


def _adam_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 20000,
    patience: int = 400,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth max‑convolution objective."""
    a = a0.copy()
    n = len(a)
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score_from_vector(best_a)
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        beta = max_conv * 0.07 + 1e-12
        w = _softmax_weights(conv, beta)
        grad = _weighted_gradient(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = _project_onto_simplex(cand, target_sum)
        cand = np.clip(cand, 0.0, 1000.0)

        cur_score = _compute_score_from_vector(a)
        cand_score = _compute_score_from_vector(cand)

        if cand_score < cur_score:
            a = cand

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            no_improve = 0
            print(f"[Adam] n={n} it={it} score={cand_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_a, best_score


def _stochastic_refinement(
    start: np.ndarray,
    deadline: float,
    step_initial: float = 0.3,
    swap_prob: float = 0.2,
) -> tuple[np.ndarray, float]:
    """Gaussian hill‑climbing + random swaps with simplex projection."""
    rng = np.random.default_rng()
    best = start.copy()
    best_score = _compute_score_from_vector(best)
    step = step_initial
    target_sum = np.sum(start)

    while time.monotonic() < deadline:
        if rng.random() < swap_prob:
            i, j = rng.integers(0, len(best), size=2)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = best + rng.normal(scale=step, size=best.shape)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        cand_score = _compute_score_from_vector(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = min(step_initial, step * 1.02)
        else:
            step *= 0.995

        if step < 1e-6:
            step = step_initial

    return best, best_score


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1)×n matrix `A` such that `A @ g == conv(f,g)`."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # column j contains `f` shifted down by `j` rows
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_g_given_f(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    Linear programme:
        maximise Σ g_i
        s.t.   conv(f,g) ≤ M  (one inequality per convolution index)
               g_i ≥ 0
    Returns the optimal `g` re‑scaled to have ℓ¹‑norm = target_sum.
    """
    n = len(f)
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimise -Σ g_i  ⇔ maximise Σ g_i
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})

    if not res.success:
        # fallback: uniform distribution
        g = np.full(n, target_sum / n, dtype=np.float64)
    else:
        g = res.x

    sum_g = np.sum(g)
    if sum_g == 0.0:
        g = np.full(n, target_sum / n, dtype=np.float64)
    else:
        g = g * (target_sum / sum_g)

    return g

# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value non‑negative step sequence within the time budget.
    Returns the sequence as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from any existing global variable (if present)
    # --------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if isinstance(warm, (list, tuple, np.ndarray)):
        seq = np.asarray(warm, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        if np.sum(seq) == 0.0:
            seq = np.full_like(seq, 1.0 / len(seq))
        else:
            seq = seq / np.sum(seq)          # normalise to sum = 1
        score = _compute_score_from_vector(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[warm] loaded height_sequence_1, n={len(seq)} score={score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop over several lengths
    # --------------------------------------------------------------
    lengths = [600, 800, 1000, 1200, 1500, 1800, 2000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = 1.0               # ℓ¹‑norm fixed to 1 (scale‑invariant)

        restarts = 3 if n <= 1200 else 2
        for rs in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ----- initialise a candidate -----
            if best_seq is not None and len(best_seq) == n:
                a = best_seq.copy()
            else:
                a = rng.random(n) + 1e-8
                a = a / np.sum(a) * target_sum

            # ----- Alternate LP updates + mixing -----
            lp_iters = 12                     # enough to see monotone improvement
            for it in range(lp_iters):
                if time.monotonic() >= deadline:
                    break
                conv_a = np.convolve(a, a)
                M = np.max(conv_a)

                # solve the LP for the companion vector g
                g = _solve_g_given_f(a, M, target_sum)

                # convex mixing (t controls step size)
                t = 0.2
                a_new = (1.0 - t) * a + t * g
                a_new = np.clip(a_new, 0.0, 1000.0)
                a_new = _project_onto_simplex(a_new, target_sum)

                cur_score = _compute_score_from_vector(a_new)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_new.copy()
                    print(f"[LP] n={n} rs={rs} it={it} score={cur_score:.6f}")

                # stop early if the update is negligible
                if np.linalg.norm(a_new - a) < 1e-12:
                    break
                a = a_new

            # ----- short Adam refinement (optional) -----
            sub_deadline = min(deadline, time.monotonic() + 8.0)
            a_refined, sc_refined = _adam_descent(a, target_sum, sub_deadline,
                                                  max_iters=8000, patience=400)
            if sc_refined < best_score:
                best_score = sc_refined
                best_seq = a_refined.copy()
                print(f"[Adam] n={n} rs={rs} refined score={sc_refined:.6f}")

            # early exit if we already beat a generous target
            if best_score < 1.45:
                print("[early] desirable bound reached")
                break
        if best_score < 1.45:
            break

    # --------------------------------------------------------------
    # Final stochastic polishing (if time permits)
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution of modest length
        n = 800
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_score_from_vector(best_seq)

    if time.monotonic() < deadline:
        refined, refined_score = _stochastic_refinement(best_seq, deadline,
                                                         step_initial=0.25)
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score
            print(f"[final] refined score={refined_score:.6f}")

    # safety clipping (evaluate_sequence performs the same)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # --------------------------------------------------------------
    # Return as a plain Python list
    # --------------------------------------------------------------
    return [float(x) for x in best_seq]
