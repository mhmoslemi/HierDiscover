def _adam_optimize(init_seq, deadline, rng):
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0*n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    # hyperparameters
    lr = 0.1 * (target/n)  # step size initial relative to average
    beta = 1.0
    best_seq = a.copy()
    best_score = _fast_score(a)
    it = 1

    while time.time() < deadline and it <= 2000:
        # compute conv & smooth weights
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        # update best
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
        # gradient
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        # projection
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n)
        # anneal beta gradually
        if it % 150 == 0 and beta < 5000:
            beta = min(beta * 1.6, 5000)
        # optionally jitter occasionally
        if it % 300 == 0:
            # small random jitter: add Gaussian noise then renorm
            noise = rng.normal(scale=0.02, size=n)
            a = a + noise
            a = np.clip(a, 0.0, 1000.0)
            a = _normalize(a)
        it += 1

    return best_seq, best_score

def _random_swap_refine(seq, deadline, rng):
    a = seq.copy()
    n = a.shape[0]
    cur_score = _fast_score(a)
    best_seq = a.copy()
    best_score = cur_score
    # temperature schedule based on current best
    T0 = cur_score * 0.1 + 0.01
    it = 0
    while time.time() < deadline:
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] == 0:
            continue
        delta = a[i] * rng.uniform(0.01, 0.07)  # move between 1% and 7%
        # ensure not exceed 1000 after adding to j
        if a[j] + delta > 1000.0:
            delta = 1000.0 - a[j]
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # optional re-normalization (preserve sum)
        # sum unchanged; but may be small negative due to rounding: ensure nonnegative
        new_a = np.clip(new_a, 0.0, 1000.0)
        # renorm to target sum
        total = new_a.sum()
        target = math.sqrt(2.0*n)
        new_a = new_a * (target / total) if total>0 else np.full(n, target/n)
        new_score = _fast_score(new_a)
        # acceptance
        if new_score < cur_score:
            accept = True
        else:
            # temperature decay
            T = T0 * (0.9995**it)
            prob = math.exp(-(new_score - cur_score) / (T + 1e-12))
            accept = rng.random() < prob
        if accept:
            a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_a.copy()
                print(f"[refine best] n={n} score={best_score:.6f}")
        it += 1
    return best_seq, best_score

"""
Search for a low‑value step sequence.

The algorithm consists of three phases:

0️⃣ Warm‑start from a previously discovered construction if available.

1️⃣ Exhaustive parametric sweep over several families (symmetric power‑law,
   symmetric exponential, arcsine‑like and sine‑power) for a range of
   lengths.  The candidates are normalised so that the ℓ¹‑norm equals
   √(2 n); in this regime the evaluation score coincides with the maximum
   auto‑convolution value.

2️⃣ For the most promising candidates a fast Adam‑style projected gradient
   descent is applied.  The objective is the smooth‑max surrogate of the
   auto‑convolution; the surrogate’s temperature β is annealed to make the
   gradient approach the true sub‑gradient of the max.  After each step the
   iterate is projected onto the feasible set (non‑negative, clipped to
   1000, and ℓ¹‑normalised).

3️⃣ A cheap random‑mass‑swap simulated‑annealing refinement is executed on
   the best sequence found so far until the time budget expires.

All operations are FFT‑accelerated, allowing us to explore lengths up to a
few thousand within the 1000 s budget.  The best sequence is returned as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global container for the best solution discovered so far
# ----------------------------------------------------------------------
_global_best = {"seq": None, "score": float("inf")}


# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_fast_len(n: int) -> int:
    """Return a good FFT length (fallback to power‑of‑2)."""
    try:
        return np.fft.next_fast_len(n)
    except Exception:  # pragma: no cover
        return 2 ** int(np.ceil(np.log2(n)))


def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for repeated calls)."""
    n = x.shape[0] + y.shape[0] - 1
    size = _next_fast_len(n)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and enforce ℓ¹‑norm = √(2·n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total <= 0.0:
        seq = np.full(n, target / n, dtype=np.float64)
    else:
        seq = seq * (target / total)
    return seq


def _fast_score(seq: np.ndarray) -> float:
    """
    Exact score after normalisation.
    For a normalised vector the evaluation equals max(a*a).
    """
    conv = _linear_convolve(seq, seq)
    return float(conv.max())


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights for a given β (stable version)."""
    cmax = conv.max()
    exp_term = np.exp(beta * (conv - cmax))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_i w_i·(a*a)_i .
    Derivative = 2·corr(a, w_rev).
    """
    w_rev = w[::-1]
    corr = _linear_convolve(a, w_rev)          # length = 3·n‑2
    n = a.shape[0]
    segment = corr[n - 1 : 2 * n - 1]          # middle n entries
    return 2.0 * segment[::-1]


# ----------------------------------------------------------------------
# Global update helper
# ----------------------------------------------------------------------
def _maybe_update_best(seq: np.ndarray) -> None:
    """Update the global best if `seq` improves the score."""
    sc = _fast_score(seq)
    if sc < _global_best["score"]:
        _global_best["score"] = sc
        _global_best["seq"] = seq.copy()
        print(f"[global best] n={len(seq)} score={sc:.6f}")


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smooth‑max objective
# ----------------------------------------------------------------------
def _adam_optimize(init_seq: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation.
    Returns the best vector encountered and its true max‑auto‑convolution.
    """
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # hyper‑parameters
    lr = 0.1 * (target / n)            # initial learning rate
    beta = 1.0                         # start with a smooth surrogate
    best_seq = a.copy()
    best_score = _fast_score(a)

    it = 1
    while time.time() < deadline and it <= 3000:
        conv = _linear_convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()

        # smooth‑max surrogate
        w = _smooth_weights(conv, beta)
        grad = _gradient_smooth(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto feasible set
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0.0:
            a = a * (target / total)
        else:
            a = np.full(n, target / n, dtype=np.float64)

        # anneal β (sharpen the surrogate)
        if it % 150 == 0 and beta < 5000.0:
            beta = min(beta * 1.6, 5000.0)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            noise = rng.normal(scale=0.02, size=n)
            a = a + noise
            a = _normalize(a)

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement based on random mass swaps
# ----------------------------------------------------------------------
def _random_swap_refine(seq: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb / simulated‑annealing that repeatedly transfers
    a small fraction of mass from one coordinate to another.
    """
    a = seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    cur_score = _fast_score(a)
    best_seq = a.copy()
    best_score = cur_score

    # initial temperature proportional to the current score
    T0 = max(0.01, cur_score * 0.08)

    it = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 0.0:
            continue
        # transfer a small random fraction of a[i]
        delta = a[i] * rng.uniform(0.01, 0.07)
        # keep the upper bound 1000
        if a[j] + delta > 1000.0:
            delta = 1000.0 - a[j]
            if delta <= 0.0:
                continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # renormalise (sum is preserved, but clipping may change it)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalize(cand)

        cand_score = _fast_score(cand)

        # acceptance rule
        if cand_score < cur_score:
            accept = True
        else:
            T = T0 * (0.9995 ** it)
            prob = math.exp(-(cand_score - cur_score) / (T + 1e-12))
            accept = rng.random() < prob

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[refine] n={n} score={best_score:.6f}")

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step sequence with a low evaluation score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from an existing construction, if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.array(height_sequence_1, dtype=np.float64)
            tmp = _normalize(tmp)
            _maybe_update_best(tmp)
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Broad parametric sweep (symmetric families)
    # ------------------------------------------------------------------
    lengths = [2500, 2000, 1500, 1200, 1000, 800,
               600, 500, 400, 300, 200, 150, 100]

    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(1, n + 1, dtype=np.float64)

        # ---- symmetric power‑law a_i ∝ i^{-p} + (n+1-i)^{-p} ----
        for p in np.linspace(0.2, 5.0, 25):
            if time.time() >= deadline:
                break
            w = (idx ** (-p)) + ((n + 1 - idx) ** (-p))
            seq = w * (target / w.sum())
            seq = _normalize(seq)
            _maybe_update_best(seq)

        # ---- symmetric exponential a_i ∝ exp(-α·dist_to_edge) ----
        dist = np.minimum(idx - 1, n - idx)  # distance from nearest edge, 0‑based
        for alpha in np.linspace(0.0005, 0.20, 25):
            if time.time() >= deadline:
                break
            w = np.exp(-alpha * dist)
            seq = w * (target / w.sum())
            seq = _normalize(seq)
            _maybe_update_best(seq)

        # ---- arcsine‑like shape a_i ∝ 1/√(i·(n+1‑i)) ----
        if time.time() < deadline:
            w = 1.0 / np.sqrt(idx * (n + 1 - idx))
            seq = w * (target / w.sum())
            seq = _normalize(seq)
            _maybe_update_best(seq)

        # ---- sine‑power shape a_i ∝ sin(π·i/(n+1))^p (mid‑peaked) ----
        for p in np.linspace(1.0, 6.0, 12):
            if time.time() >= deadline:
                break
            w = np.sin(np.pi * idx / (n + 1)) ** p
            seq = w * (target / w.sum())
            seq = _normalize(seq)
            _maybe_update_best(seq)

    # ------------------------------------------------------------------
    # 2️⃣ Projected Adam optimisation from several random restarts
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        # number of restarts – fewer for very large n
        restarts = 6 if n <= 800 else 3

        for r in range(restarts):
            if time.time() >= deadline:
                break
            # Dirichlet‑like random start
            start = rng.gamma(1.0, size=n)
            start = start / start.sum() * target

            # Blend with the current global best if dimensions match
            if _global_best["seq"] is not None and _global_best["seq"].shape[0] == n:
                start = (start + _global_best["seq"]) / 2.0
                start = _normalize(start)

            opt_seq, _ = _adam_optimize(start, deadline, rng)
            _maybe_update_best(opt_seq)

            # early exit once the target is already reached
            if _global_best["score"] <= 1.503:
                print("[early exit] target achieved")
                return [_global_best["seq"][i].item() for i in range(_global_best["seq"].shape[0])]

    # ------------------------------------------------------------------
    # 3️⃣ Final cheap refinement – random mass swaps (simulated annealing)
    # ------------------------------------------------------------------
    if _global_best["seq"] is not None:
        refined_seq, _ = _random_swap_refine(_global_best["seq"], deadline, rng)
        _maybe_update_best(refined_seq)
    else:
        # Fallback: a simple uniform vector (should never happen)
        n = 200
        target = math.sqrt(2.0 * n)
        _global_best["seq"] = np.full(n, target / n, dtype=np.float64)
        _global_best["score"] = _fast_score(_global_best["seq"])

    # Return a plain Python list as required by the outer harness
    best_array = _global_best["seq"]
    return [float(x) for x in best_array]
