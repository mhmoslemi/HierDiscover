def propose_candidate(seed=0, budget_s=1000, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.2)

    best_seq = None; best_val = np.inf

    # 1. explore lengths and families
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500, 1800, 2200, 2600, 3000, 3500, 4000]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # families: uniform, exponential, power-law, Hann, Hamming, Blackman, Sine, Triangular, Kaiser, DPSS
        # evaluate each and keep best.
        # We'll create a list of candidate vectors and evaluate.

        # Uniform
        a = np.full(n, target / n, dtype=np.float64)
        val = _fast_score(a)  # using internal fast method
        if val < best_val:
            best_seq, best_val = a.copy(), val

        # Exponential family
        # Use α = c / n for c in 0.5 to 20 step 0.25 (like 80 points)
        cs = np.concatenate([np.linspace(0.5, 5.0, 100), np.linspace(5.0, 20.0, 80)])
        for c in cs:
            α = c / n
            idx = np.arange(n, dtype=np.float64)
            a = np.exp(-α * idx)
            a = (a / a.sum()) * target
            val = _fast_score(a)
            if val < best_val:
                best_seq, best_val = a.copy(), val

        # Power-law family a_i ∝ (i+1)^(-p)
        p_vals = np.linspace(0.05, 5.0, 80)
        base = np.arange(1, n+1, dtype=np.float64)
        for p in p_vals:
            a = base ** (-p)
            a = (a / a.sum()) * target
            val = _fast_score(a)
            if val < best_val:
                best_seq, best_val = a.copy(), val

        # Hann and others
        idx = np.arange(n, dtype=np.float64)
        # Hann
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
        a = (a / a.sum()) * target
        val = _fast_score(a)
        if val < best_val:
            best_seq, best_val = a.copy(), val
        # Hamming
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
        a = (a / a.sum()) * target
        val = _fast_score(a)
        if val < best_val:
            best_seq, best_val = a.copy(), val
        # Blackman
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1))
        a = (a / a.sum()) * target
        val = _fast_score(a)
        if val < best_val:
            best_seq, best_val = a.copy(), val

        # Sine window
        a = np.sin(np.pi * (idx + 1) / (n + 1))
        a = (a / a.sum()) * target
        val = _fast_score(a)
        if val < best_val:
            best_seq, best_val = a.copy(), val

        # Triangular
        a = np.minimum(idx + 1, n - idx)
        a = (a / a.sum()) * target
        val = _fast_score(a)
        if val < best_val:
            best_seq, best_val = a.copy(), val

        # Kaiser window if scipy available
        try:
            from scipy.signal.windows import kaiser
            betas = [4, 6, 8, 10, 12, 14]
            for beta in betas:
                a = kaiser(n, beta, sym=False)  # sym=False for non-symmetric maybe?
                a = (a / a.sum()) * target
                val = _fast_score(a)
                if val < best_val:
                    best_seq, best_val = a.copy(), val
        except Exception:
            pass

        # DPSS windows
        try:
            from scipy.signal.windows import dpss
            NW_vals = [2, 3, 4, 5, 6]
            for NW in NW_vals:
                a = dpss(n, NW, Kmax=1, sym=False)[0]
                a = np.abs(a)  # ensure non-negative (DPSS is real, can be negative near edges)
                a = (a / a.sum()) * target
                val = _fast_score(a)
                if val < best_val:
                    best_seq, best_val = a.copy(), val
        except Exception:
            pass

        # random Dirichlet samples random trials for this n
        # allocate some time slice
        rnd_deadline = time.monotonic() + min(0.2, (deadline - time.monotonic()) / 5)
        while time.monotonic() < rnd_deadline:
            a = rng.random(n)
            a = a / a.sum() * target
            val = _fast_score(a)
            if val < best_val:
                best_seq, best_val = a.copy(), val

    # 2. Gradient descent refinement
    if best_seq is not None:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        # refine function defined earlier
        best_seq, best_val = _gradient_descent_improved(best_seq, target, deadline, best_val)

    # 3. Simulated annealing (optional) as final polishing
    if best_seq is not None and time.monotonic() < deadline:
        best_seq, best_val = _annealing_refine(best_seq, deadline)

    print(f"[final] best value {best_val:.6f}, length {len(best_seq)}")
    return [float(v) for v in best_seq]

def _grad_for_k(seq, k):
    n = len(seq)
    grad = np.zeros_like(seq)
    if k < n:
        # j from 0 to k inclusive
        # grad[j] = 2 * seq[k - j]
        grad[:k+1] = 2.0 * seq[k::-1]
    else:
        off = k - (n - 1)
        # j from off to n-1
        # grad[j] = 2 * seq[(n - 1) - (j - off)] = 2 * seq[n - 1 - (j - off)]
        # Equivalent: grad[off:] = 2 * seq[(n-1):(off-1):-1]
        # Use slice: seq[n-1:off-1:-1] fine.
        grad[off:] = 2.0 * seq[n-1:off-1:-1]
    return grad

def _gradient_descent_improved(seq, target, deadline, current_best_val):
    n = len(seq)
    best_seq = seq.copy()
    best_val = current_best_val
    step = 0.1  # start moderate
    rng = np.random.default_rng()

    while time.monotonic() < deadline:
        conv = _conv_fft(seq)
        max_val = conv.max()
        # use tolerance factor
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(seq)
        for k in active:
            grad += _grad_for_k(seq, k)
        # Remove component aligned with ones vector to keep sum unchanged
        # Direction should have zero sum: sum(direction)=0
        # The mean of grad is average contribution per element.
        # We can project onto subspace sum=0 by subtracting grad.mean()
        direction = -grad + grad.mean()
        # If direction norm zero, break.
        if np.linalg.norm(direction) < 1e-12:
            break
        cand = seq + step * direction
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total > 0.0:
            cand = cand * (target / total)
        else:
            cand = np.full_like(seq, target / n)
        cand_val = _fast_score(cand)
        if cand_val < best_val:
            best_seq, best_val = cand.copy(), cand_val
            seq = cand
            step = min(step * 1.2, 1.0)
        else:
            step *= 0.5
            # Random reinitialization of direction occasionally
            if step < 1e-6:
                # small random perturb
                i, j = rng.integers(n, size=2)
                delta = seq[i] * 0.05
                if delta > 0:
                    cand = seq.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand = cand * (target / cand.sum())
                    cand_val = _fast_score(cand)
                    if cand_val < best_val:
                        seq = cand
                        best_seq, best_val = cand.copy(), cand_val
                step = 0.1
        # early exit if best_val below known threshold
        if best_val < 1.503:
            break
    return best_seq, best_val

Now annealing refine:

def _annealing_refine(seq, deadline):
    n = len(seq)
    target = math.sqrt(2.0 * n)
    best_seq = seq.copy()
    best_val = _fast_score(seq)
    rng = np.random.default_rng()
    T0 = 0.1
    T = T0
    decay = 0.9995
    while time.monotonic() < deadline:
        # propose small random perturbation on two indices (preserving sum)
        i, j = rng.integers(n, size=2)
        # choose delta fraction up to 5% of element i
        max_delta = min(best_seq[i], target*0.05)  # limit moving
        delta = rng.random() * max_delta
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target / cand.sum())
        cand_val = _fast_score(cand)
        # acceptance
        if cand_val < best_val or rng.random() < math.exp(-(cand_val - best_val) / T):
            best_seq, best_val = cand, cand_val
        T *= decay
        if T < 1e-6:
            T = T0
    return best_seq, best_val

"""
Propose a low‑value step sequence.

Algorithm
---------
1️⃣  **Broad parametric sweep** – for a range of lengths (200 … 5000)
    we generate many hand‑crafted families:
        * uniform,
        * exponential  a_i ∝ exp(-α·i),
        * exponential × polynomial  a_i ∝ (i+1)^p·exp(-α·i),
        * power‑law a_i ∝ (i+1)^{-p},
        * classic windows (Hann, Hamming, Blackman, sine, triangular),
        * Kaiser windows (β = 4…14) and DPSS windows (NW = 2…6) when SciPy is
          available,
        * cheap Dirichlet random samples.
    The best vector of all families becomes the global incumbent.

2️⃣  **Improved sub‑gradient descent** – starting from the incumbent we
    repeatedly locate *all* convolution entries that are (almost) maximal.
    The exact sub‑gradient of the maximum of the auto‑convolution is the sum
    of the “reversed‑slice’’ gradients of the active indices (factor 2).
    After removing the component parallel to the all‑ones vector we move
    opposite to this direction, re‑normalise the ℓ¹‑norm to √(2·n) and
    adapt the step size (increase after success, halve after failure).
    The method quickly drives the maximum auto‑correlation down.

3️⃣  **Simulated‑annealing polishing** – a short annealing phase performs
    random pair‑wise mass transfers (preserving the ℓ¹‑norm) and accepts
    uphill moves with a temperature schedule.  This helps escape any
    plateaus left by the deterministic descent.

All operations respect the security checks of ``evaluate_sequence``:
non‑negative entries, clipping to [0,1000] and a fixed ℓ¹‑norm.  The
function returns the best sequence found before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution via FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (no security wrapper)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_fft(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    # after normalisation sum_a = sqrt(2·n) → the factor cancels,
    # but we keep the formula for safety.
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Normalise a raw vector to the required ℓ¹‑norm (≤1000 per entry)
# ----------------------------------------------------------------------
def _normalise(v: np.ndarray, target_sum: float) -> np.ndarray:
    v = np.maximum(v, 0.0)               # protect against negatives
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, target_sum / v.size)
    v = v * (target_sum / total)          # ℓ¹‑norm = target_sum
    return np.clip(v, 0.0, 1000.0)         # respect the hard cap


# ----------------------------------------------------------------------
# Sub‑gradient of a single convolution entry (factor 2 is exact)
# ----------------------------------------------------------------------
def _grad_for_k(seq: np.ndarray, k: int) -> np.ndarray:
    n = len(seq)
    g = np.zeros_like(seq)
    if k < n:
        # overlapping part grows from the left
        g[: k + 1] = 2.0 * seq[k :: -1]
    else:
        off = k - (n - 1)
        # overlapping part shrinks from the right
        g[off:] = 2.0 * seq[n - 1 : off - 1 : -1]
    return g


# ----------------------------------------------------------------------
# Sub‑gradient descent that respects the ℓ¹‑norm
# ----------------------------------------------------------------------
def _gradient_descent(seq: np.ndarray, target_sum: float, deadline: float,
                     best_sofar: float) -> tuple[np.ndarray, float]:
    n = seq.shape[0]
    best_seq = seq.copy()
    best_val = best_sofar
    step = 0.2
    rng = np.random.default_rng()

    while time.monotonic() < deadline:
        conv = _conv_fft(seq)
        max_val = conv.max()
        # all indices with value essentially equal to the maximum
        eps = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(seq)
        for k in active:
            grad += _grad_for_k(seq, k)

        # project gradient onto the hyper‑plane Σ a_i = const
        direction = -grad + grad.mean()     # ∑ direction = 0

        if np.linalg.norm(direction) < 1e-12:
            break

        cand = seq + step * direction
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalise(cand, target_sum)

        cand_val = _fast_score(cand)

        if cand_val < best_val:
            best_seq, best_val = cand.copy(), cand_val
            seq = cand
            step = min(step * 1.2, 1.0)
        else:
            step *= 0.5
            if step < 1e-7:                     # occasional shake‑up
                i, j = rng.integers(n, size=2)
                delta = seq[i] * 0.05
                if delta > 0.0:
                    cand = seq.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand = _normalise(cand, target_sum)
                    cand_val = _fast_score(cand)
                    if cand_val < best_val:
                        seq = cand
                        best_seq, best_val = cand.copy(), cand_val
                step = 0.2

        if best_val < 1.503:                    # target reached – stop early
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Short simulated‑annealing refinement
# ----------------------------------------------------------------------
def _annealing_refine(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    n = len(seq)
    target_sum = math.sqrt(2.0 * n)
    best_seq = seq.copy()
    best_val = _fast_score(best_seq)
    rng = np.random.default_rng()
    T = 0.2
    decay = 0.9996

    while time.monotonic() < deadline:
        i, j = rng.integers(n, size=2)
        # move a small fraction of mass from i to j
        max_move = min(best_seq[i], target_sum * 0.07)
        delta = rng.random() * max_move
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalise(cand, target_sum)
        cand_val = _fast_score(cand)

        # Metropolis acceptance
        if cand_val < best_val or rng.random() < math.exp(-(cand_val - best_val) / T):
            best_seq, best_val = cand, cand_val

        T *= decay
        if T < 1e-6:
            T = 0.2

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Initialise from a possibly pre‑computed candidate
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = init.shape[0]
            if n0 > 0:
                target0 = math.sqrt(2.0 * n0)
                init = _normalise(init, target0)
                sc0 = _fast_score(init)
                best_seq, best_val = init, sc0
                print(f"[init] height_sequence_1 n={n0} score={sc0:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1️⃣  Exhaustive parametric sweep (multiple lengths)
    # --------------------------------------------------------------
    # lengths – we go up to a few thousand items; larger n often yields
    # a lower maximal auto‑correlation.
    lengths = [200, 300, 400, 500, 600, 800, 1000,
               1200, 1500, 1800, 2200, 2600, 3000, 3500, 4000, 4500, 5000]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(n, dtype=np.float64)

        # ---- uniform ------------------------------------------------
        a = np.full(n, target / n, dtype=np.float64)
        sc = _fast_score(a)
        if sc < best_val:
            best_seq, best_val = a.copy(), sc
            print(f"[family] uniform n={n} → {sc:.6f}")

        # ---- exponential --------------------------------------------
        # α = c / n, c in a dense grid (0.5 … 20)
        cs = np.concatenate([np.linspace(0.5, 5.0, 80),
                             np.linspace(5.0, 20.0, 80)])
        for c in cs:
            if time.monotonic() >= deadline:
                break
            α = c / n
            a = np.exp(-α * idx)
            a = _normalise(a, target)
            sc = _fast_score(a)
            if sc < best_val:
                best_seq, best_val = a.copy(), sc
                print(f"[family] exp(c={c:.2f}) n={n} → {sc:.6f}")

        # ---- exponential × polynomial --------------------------------
        # a_i = (i+1)^p * exp(-α i)
        p_vals = np.linspace(0.0, 2.5, 6)          # 0 … 2.5
        for p in p_vals:
            if time.monotonic() >= deadline:
                break
            for c in cs:
                if time.monotonic() >= deadline:
                    break
                α = c / n
                a = ((idx + 1) ** p) * np.exp(-α * idx)
                a = _normalise(a, target)
                sc = _fast_score(a)
                if sc < best_val:
                    best_seq, best_val = a.copy(), sc
                    print(f"[family] exp·poly p={p:.1f} c={c:.2f} n={n} → {sc:.6f}")

        # ---- power‑law -----------------------------------------------
        p_vals = np.linspace(0.05, 5.0, 80)
        base = np.arange(1, n + 1, dtype=np.float64)
        for p in p_vals:
            if time.monotonic() >= deadline:
                break
            a = base ** (-p)
            a = _normalise(a, target)
            sc = _fast_score(a)
            if sc < best_val:
                best_seq, best_val = a.copy(), sc
                print(f"[family] power p={p:.2f} n={n} → {sc:.6f}")

        # ---- classic windows -----------------------------------------
        # Hann
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
        a = _normalise(a, target)
        sc = _fast_score(a)
        if sc < best_val:
            best_seq, best_val = a.copy(), sc
            print(f"[family] Hann n={n} → {sc:.6f}")

        # Hamming
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
        a = _normalise(a, target)
        sc = _fast_score(a)
        if sc < best_val:
            best_seq, best_val = a.copy(), sc
            print(f"[family] Hamming n={n} → {sc:.6f}")

        # Blackman
        a = (0.42 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
             + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1)))
        a = _normalise(a, target)
        sc = _fast_score(a)
        if sc < best_val:
            best_seq, best_val = a.copy(), sc
            print(f"[family] Blackman n={n} → {sc:.6f}")

        # Sine (half‑period)
        a = np.sin(np.pi * (idx + 1) / (n + 1))
        a = _normalise(a, target)
        sc = _fast_score(a)
        if sc < best_val:
            best_seq, best_val = a.copy(), sc
            print(f"[family] Sine n={n} → {sc:.6f}")

        # Triangular (peak in the middle)
        a = np.minimum(idx + 1, n - idx)
        a = _normalise(a, target)
        sc = _fast_score(a)
        if sc < best_val:
            best_seq, best_val = a.copy(), sc
            print(f"[family] Triangle n={n} → {sc:.6f}")

        # ---- Kaiser windows (if SciPy is present) --------------------
        try:
            from scipy.signal.windows import kaiser
            betas = [4, 6, 8, 10, 12, 14]
            for beta in betas:
                a = kaiser(n, beta, sym=False)
                a = _normalise(a, target)
                sc = _fast_score(a)
                if sc < best_val:
                    best_seq, best_val = a.copy(), sc
                    print(f"[family] Kaiser β={beta} n={n} → {sc:.6f}")
        except Exception:
            pass

        # ---- DPSS windows (if SciPy is present) ----------------------
        try:
            from scipy.signal.windows import dpss
            NW_vals = [2, 3, 4, 5, 6]
            for NW in NW_vals:
                a = dpss(n, NW, Kmax=1, sym=False)[0]
                a = np.maximum(a, 0.0)          # enforce non‑negativity
                a = _normalise(a, target)
                sc = _fast_score(a)
                if sc < best_val:
                    best_seq, best_val = a.copy(), sc
                    print(f"[family] DPSS NW={NW} n={n} → {sc:.6f}")
        except Exception:
            pass

        # ---- cheap random Dirichlet samples -------------------------
        # spend a tiny slice of the remaining budget on each length
        rem = deadline - time.monotonic()
        if rem <= 0.0:
            break
        rnd_deadline = time.monotonic() + min(0.3, rem / 4.0)
        while time.monotonic() < rnd_deadline:
            a = rng.random(n)
            a = _normalise(a, target)
            sc = _fast_score(a)
            if sc < best_val:
                best_seq, best_val = a.copy(), sc
                print(f"[rand] n={n} → {sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Refine with sub‑gradient descent
    # --------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        best_seq, best_val = _gradient_descent(best_seq, target, deadline, best_val)
        print(f"[grad] refined → {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final annealing polish
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        best_seq, best_val = _annealing_refine(best_seq, deadline)
        print(f"[anneal] polished → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    print(f"[result] best score = {best_val:.6f} (length={len(best_seq)})")
    return [float(v) for v in best_seq]
