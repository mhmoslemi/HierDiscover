def _conv_fft_general(a, b):
    n = len(a)
    m = len(b)
    N = 1 << ((n + m - 1).bit_length())
    Fa = np.fft.rfft(a, n=N)
    Fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(Fa * Fb, n=N)
    return conv[: n + m - 1]

    n = len(seq)
    conv = _conv_fft(seq)  # length 2n-1
    max_val = conv.max()

    # w = np.exp(beta*(conv - max_val))

        grad = np.zeros_like(seq)
        k = int(conv.argmax())
        if k < n:
            grad[: k+1] = seq[k::-1] # reversed slice
        else:
            off = k - (n-1)
            grad[off:] = seq[: n-off][::-1]
        return grad
    p = w / w_sum

def _smooth_gradient_descent(seq, target, deadline):
    n = len(seq)
    best_seq = seq.copy()
    best_sc = _fast_score(seq)
    step = 0.5
    beta = 4.0
    rng = np.random.default_rng()
    while time.monotonic() < deadline:
        conv = _conv_fft(seq)
        maxc = conv.max()
        # compute softmax weights
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0 or np.isinf(w_sum):
            # fallback: take gradient of active max entry
            grad = np.zeros_like(seq)
            k = int(conv.argmax())
            if k < n:
                grad[:k+1] = seq[k::-1]
            else:
                off = k - (n-1)
                grad[off:] = seq[:n-off][::-1]
        else:
            p = w / w_sum
            # term1 & term2 using conv general
            term_full = _conv_fft_general(p, seq)
            start = n-1
            term1 = term_full[start:start+n]
            rev = p[::-1]
            term_full_rev = _conv_fft_general(rev, seq)
            term2 = term_full_rev[start:start+n]
            grad = term1 + term2

        direction = -grad
        # propose new point
        cand = seq + step * direction
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_to_simplex(cand, target)
        cand_sc = _fast_score(cand)

        if cand_sc < best_sc:
            best_seq, best_sc = cand.copy(), cand_sc
            seq = cand
            step = min(step * 1.2, 2.0)
            beta = min(beta * 1.05, 30.0)
        else:
            step *= 0.5
            # occasional random perturbation
            if step < 1e-6:
                i, j = rng.integers(n, size=2)
                delta = seq[i] * 0.1
                if delta > 0:
                    cand = seq.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = _project_to_simplex(cand, target)
                    cand_sc = _fast_score(cand)
                    if cand_sc < best_sc:
                        seq = cand
                        best_seq, best_sc = cand.copy(), cand_sc
                step = 0.5
        # early exit if threshold reached
        if best_sc < 1.502:
            break
    return best_seq, best_sc

def _conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    na = a.shape[0]
    nb = b.shape[0]
    N = 1 << ((na + nb - 1).bit_length())
    Fa = np.fft.rfft(a, n=N)
    Fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(Fa * Fb, n=N)
    return conv[: na + nb - 1]
```

"""
Propose a low‑value step sequence.

Algorithm
---------
1.  Initialise from the optional global `height_sequence_1`.
2.  Exhaustive parametric sweep on several lengths:
    * 1‑D golden‑section optimisation of the exponential family
      a_i ∝ exp(-α·i)  with α = k/n, k∈[0.2, 6] – this fine‑tunes the
      theoretically optimal region α≈k/n.
    * 1‑D golden‑section optimisation of the power‑law family
      a_i ∝ (i+1)^(-p)  with p∈[0.05, 5].
    * Short random Dirichlet sampling for diversity.
3.  Smoothed gradient descent.
    The objective 2 n·max(conv)/‖a‖_1² is invariant under scaling, so we
    keep the ℓ¹‑norm fixed to √(2·n).  At each step we minimise a
    smooth approximation of the max via a soft‑max (log‑sum‑exp) with
    temperature β.  The gradient of the smoothed max is obtained at
    O(N log N) cost by two FFT‑based convolutions.
    An adaptive step size and a gradual increase of β guide the
    optimisation to a low‑peak convolution.
4.  Final Gaussian hill‑climb (small random perturbations) for polishing.

The whole routine respects the security checks of `evaluate_sequence`,
uses only NumPy (no external solvers) and stops as soon as the time
budget is exhausted, returning the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of `seq` with itself (FFT, O(n log n))."""
    n = seq.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(F * F, n=N)
    return conv[: 2 * n - 1]


def _conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary 1‑D arrays using FFT."""
    na = a.shape[0]
    nb = b.shape[0]
    N = 1 << ((na + nb - 1).bit_length())
    Fa = np.fft.rfft(a, n=N)
    Fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(Fa * Fb, n=N)
    return conv[: na + nb - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Exact objective value (same formula as `evaluate_sequence`)."""
    n = seq.shape[0]
    max_b = _conv_fft(seq).max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


def _project_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Clamp to non‑negative and renormalise to a given ℓ¹‑norm."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, target / v.size)
    return v * (target / s)


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution entry
# ----------------------------------------------------------------------
def _smooth_gradient(seq: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the log‑sum‑exp (soft‑max) of the auto‑convolution."""
    n = seq.shape[0]
    conv = _conv_fft(seq)                     # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (numerically safe)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        # fall back to the true sub‑gradient of the active entry
        grad = np.zeros_like(seq)
        k = int(conv.argmax())
        if k < n:
            grad[: k + 1] = seq[k :: -1]
        else:
            off = k - (n - 1)
            grad[off:] = seq[: n - off][::-1]
        return grad

    p = w / w_sum  # normalized soft‑max (probability vector)

    # term1 = (p * a) convolution, centred at index n‑1
    full1 = _conv_fft_general(p, seq)                # length 3n‑2
    start = n - 1
    term1 = full1[start:start + n]

    # term2 = (reversed p * a) convolution
    full2 = _conv_fft_general(p[::-1], seq)
    term2 = full2[start:start + n]

    return term1 + term2


def _smooth_gradient_descent(
    seq: np.ndarray,
    target: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the smoothed max‑convolution."""
    n = seq.shape[0]
    best_seq = seq.copy()
    best_sc = _fast_score(seq)

    step = 0.5
    beta = 4.0
    rng = np.random.default_rng()

    while time.monotonic() < deadline:
        # gradient of the smoothed objective
        grad = _smooth_gradient(seq, beta)

        # move opposite to the gradient
        cand = seq - step * grad
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_to_simplex(cand, target)

        cand_sc = _fast_score(cand)

        if cand_sc < best_sc:
            best_seq, best_sc = cand.copy(), cand_sc
            seq = cand
            step = min(step * 1.2, 2.0)      # slowly enlarge step on success
            beta = min(beta * 1.05, 30.0)    # sharpen the soft‑max
        else:
            step *= 0.5
            # occasional random shake‑up when step becomes tiny
            if step < 1e-6:
                i, j = rng.integers(n, size=2)
                delta = seq[i] * 0.1
                if delta > 0.0:
                    cand = seq.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = _project_to_simplex(cand, target)
                    cand_sc = _fast_score(cand)
                    if cand_sc < best_sc:
                        seq = cand
                        best_seq, best_sc = cand.copy(), cand_sc
                step = 0.5

        # early exit if we already beat the target bound
        if best_sc < 1.502:
            break

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Small stochastic polishing (Gaussian perturbations)
# ----------------------------------------------------------------------
def _local_search(
    start_seq: np.ndarray,
    deadline: float,
    step_initial: float = 0.2,
) -> tuple[np.ndarray, float]:
    """Isotropic Gaussian hill‑climb around `start_seq`."""
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    target = math.sqrt(2.0 * n)

    best = start_seq.copy()
    best_sc = _fast_score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_to_simplex(cand, target)

        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc

        step *= 0.9997
        if step < 1e-7:
            step = step_initial

    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 0,
    budget_s: float = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats (the best sequence found before
    the deadline).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Initialise from a possible pre‑computed candidate
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.size > 0:
                n0 = init.shape[0]
                target0 = math.sqrt(2.0 * n0)
                init = _project_to_simplex(init, target0)
                sc0 = _fast_score(init)
                best_seq, best_score = init, sc0
                print(f"[init] n={n0} score={sc0:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1️⃣  Exhaustive parametric sweep (fine 1‑D optimisation)
    # --------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500, 2000, 2500]

    phi = (math.sqrt(5) - 1) / 2  # golden‑section constant

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        idx = np.arange(n, dtype=np.float64)

        # ---- exponential family (α = k/n) ----
        # k is searched on [0.2, 6.0] – the known good region is around 1.5
        left, right = 0.2, 6.0
        best_exp_seq = None
        best_exp_sc = float("inf")

        for _ in range(30):
            if time.monotonic() >= deadline:
                break
            k1 = left + (right - left) * phi
            k2 = right - (right - left) * phi

            a1 = np.exp(- (k1 / n) * idx)
            a1 = _project_to_simplex(a1, target)
            sc1 = _fast_score(a1)

            a2 = np.exp(- (k2 / n) * idx)
            a2 = _project_to_simplex(a2, target)
            sc2 = _fast_score(a2)

            if sc1 < sc2:
                right = k2
                if sc1 < best_exp_sc:
                    best_exp_sc = sc1
                    best_exp_seq = a1.copy()
            else:
                left = k1
                if sc2 < best_exp_sc:
                    best_exp_sc = sc2
                    best_exp_seq = a2.copy()

        if best_exp_seq is not None and best_exp_sc < best_score:
            best_seq, best_score = best_exp_seq, best_exp_sc
            print(f"[exp] n={n} score={best_exp_sc:.6f}")

        # ---- power‑law family (a_i ∝ (i+1)^{-p}) ----
        left, right = 0.05, 5.0
        base = np.arange(1, n + 1, dtype=np.float64)
        best_pow_seq = None
        best_pow_sc = float("inf")

        for _ in range(25):
            if time.monotonic() >= deadline:
                break
            p1 = left + (right - left) * phi
            p2 = right - (right - left) * phi

            a1 = base ** (-p1)
            a1 = _project_to_simplex(a1, target)
            sc1 = _fast_score(a1)

            a2 = base ** (-p2)
            a2 = _project_to_simplex(a2, target)
            sc2 = _fast_score(a2)

            if sc1 < sc2:
                right = p2
                if sc1 < best_pow_sc:
                    best_pow_sc = sc1
                    best_pow_seq = a1.copy()
            else:
                left = p1
                if sc2 < best_pow_sc:
                    best_pow_sc = sc2
                    best_pow_seq = a2.copy()

        if best_pow_seq is not None and best_pow_sc < best_score:
            best_seq, best_score = best_pow_seq, best_pow_sc
            print(f"[pow] n={n} score={best_pow_sc:.6f}")

        # ---- cheap random Dirichlet samples for a short burst ----
        rnd_deadline = time.monotonic() + min(0.3,
                                             (deadline - time.monotonic()) * 0.5)
        while time.monotonic() < rnd_deadline:
            a = rng.random(n)
            a = _project_to_simplex(a, target)
            sc = _fast_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[rand] n={n} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smoothed gradient‑descent refinement on the incumbent
    # --------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        refined_seq, refined_sc = _smooth_gradient_descent(best_seq, target,
                                                           deadline)

            best_seq, best_score = refined_seq, refined_sc
            print(f"[smooth] refined to {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final stochastic polishing
    # --------------------------------------------------------------

        polished_seq, polished_sc = _local_search(best_seq, deadline,
                                                  step_initial=0.2)
        if polished_sc < best_score:
            best_seq, best_score = polished_seq, polished_sc
            print(f"[final] polished to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    print(f"[result] best score = {best_score:.6f} (length={len(best_seq)})")
    return [float(v) for v in best_seq]
