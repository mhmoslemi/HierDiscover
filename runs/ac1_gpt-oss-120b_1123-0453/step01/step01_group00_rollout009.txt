"""
Propose a low‑value step sequence.

The algorithm works with a fixed ℓ¹‑norm (√(2·n)).  For a given length n it
repeatedly solves a linear programme

    maximise   Σ g_i
    subject to  (f * g)_k ≤ M   for every k,
                g_i ≥ 0,

where M = max(f*f) and (f * g) denotes the *full* convolution.
If the LP can produce a larger ℓ¹‑mass than the current one, the solution
is rescaled to the target sum and mixed with the current vector.
Because the mixing is convex, the worst auto‑convolution entry never
increases while the ℓ¹‑mass stays constant, so M (and hence the scorer)
monotonically decreases.

The routine tries several lengths (200 … 800).  For each length it runs
until the time budget expires, keeping the best sequence seen.  A final
random‑perturbation phase refines the best candidate.

If SciPy is available the dense LP is solved with `linprog`; otherwise it
falls back to CVXPY (ECOS).  When neither solver works a simple random walk
is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Detect available LP back‑ends
try:
    from scipy.optimize import linprog          # type: ignore
    _HAS_SCIPY = True
except Exception:                               # pragma: no cover
    _HAS_SCIPY = False

if not _HAS_SCIPY:
    try:
        import cvxpy as cp                      # type: ignore
        _HAS_CVX = True
    except Exception:                           # pragma: no cover
        _HAS_CVX = False
else:
    _HAS_CVX = False


# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Wrapper for the provided `evaluate_sequence`."""
    # evaluate_sequence expects a list of Python floats
    return float(evaluate_sequence(seq.tolist()))   # type: ignore


# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n convolution matrix A such that (A @ g)[k] = (f * g)[k].
    The rows correspond to all full‑convolution offsets.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # Fill row k : for each i where 0 ≤ i < n and 0 ≤ k‑i < n set A[k,i] = f[i]
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start … i_end (inclusive)
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    return A


# ----------------------------------------------------------------------
def _lp_max_sum(f: np.ndarray, M: float):
    """
    Solve  maximise Σ g_i   s.t.  (f * g)_k ≤ M, g ≥ 0 .
    Returns a result‑like object with attributes ``success`` and ``x``.
    """
    n = f.shape[0]
    A = _conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # minimise –Σg_i  → maximise Σg_i
    bounds = [(0, None)] * n

    if _HAS_SCIPY:
        # SciPy's high‑level LP solver (dual‑simplex + interior‑point)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        # Normalise the interface to what we need
        class _Res:
            pass
        out = _Res()
        out.success = (res.status == 0)
        out.x = res.x if out.success else None
        return out

    if _HAS_CVX:
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        # ECOS is fast for dense LPs; we use a generous iteration limit.
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000)

        class _Res:
            pass
        out = _Res()
        out.success = prob.status == cp.OPTIMAL
        out.x = g.value if out.success else None
        return out

    # No solver – signal failure
    class _Fail:
        pass
    out = _Fail()
    out.success = False
    out.x = None
    return out


# ----------------------------------------------------------------------
def _improve_sequence(start_f: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      max_iters: int = 300) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based improvement loop for one fixed length.
    Returns the best sequence found and its score.
    """
    f = start_f.copy()
    n = f.shape[0]
    best_seq = f.copy()
    best_sc = _score(f)
    it = 0

    while it < max_iters and time.monotonic() < deadline:
        # current worst auto‑convolution entry
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        # solve LP
        res = _lp_max_sum(f, M)
        if not getattr(res, "success", False) or res.x is None:
            # LP failed – abort this length
            break

        g = np.maximum(res.x, 0.0)
        sum_g = float(np.sum(g))
        if sum_g <= 1e-12:
            break

        # Rescale to keep the ℓ¹‑norm fixed.
        scale = target_sum / sum_g
        g_scaled = g * scale

        # Convex combination (t = 0.5 works well in practice)
        t = 0.5
        f_new = (1.0 - t) * f + t * g_scaled
        f_new = np.clip(f_new, 0.0, 1000.0)

        # Evaluate
        sc = _score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_seq = f_new.copy()
            print(f"[improve] n={n:<4} it={it:<3} score={sc:.6f}  M={M:.6f}")

            # Early exit if we already beat the target
            if best_sc < 1.503:
                break

        f = f_new
        it += 1

    return best_seq, best_sc

def propose_candidate(seed: int = 42, budget_s: float = 1100,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the given evaluation function.
    The routine tries several lengths (200‑800), runs the LP‑based refinement
    for as many iterations as time permits, and finishes with a short random
    perturbation phase.
    """
    rng = np.random.default_rng(seed)
    # leave a tiny safety margin for the runner that calls us
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – order from small to large; more time → larger n
    lengths = [200, 300, 400, 500, 600, 800]

    # If a previously discovered construction is in the global namespace,
    # try to reuse it (it has length 200 in the reference run).
    try:
        global height_sequence_1
        prev = np.array(height_sequence_1, dtype=np.float64)
        if prev.size not in lengths:
            lengths = [prev.size] + lengths
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Initialise a feasible starting vector with positive entries.
        if prev is not None and prev.size == n:
            f0 = prev.copy()
            f0 = f0 / np.sum(f0) * target_sum
        else:
            # Random positive values (avoid exact zeros)
            f0 = rng.random(n) + 0.1
            f0 = f0 / np.sum(f0) * target_sum

        # Run the LP‑based improvement loop.
        seq, sc = _improve_sequence(f0, target_sum, deadline, max_iters=400)

        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[global] New best after length {n}: score = {sc:.6f}")

        # Stop early if the target is already reached.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final short random‑perturbation refinement (if time is left).
    if best_seq is not None and time.monotonic() < deadline:
        f = best_seq.copy()
        n = f.shape[0]
        sc = best_score
        while time.monotonic() < deadline:
            # Perturb a random subset (≈5 % of entries)
            idx = rng.integers(0, n, size=max(1, int(0.05 * n)))
            delta = rng.normal(scale=0.05, size=idx.shape)
            cand = f.copy()
            cand[idx] += delta
            cand = np.clip(cand, 0.0, 1000.0)

            cand_sc = _score(cand)
            if cand_sc < sc:
                f = cand
                sc = cand_sc
                if sc < best_score:
                    best_score = sc
                    best_seq = f.copy()
                    print(f"[refine] improved to {sc:.6f}")
                if best_score < 1.503:
                    break

    # ------------------------------------------------------------------
    # Fallback – deterministic uniform vector (should never be used).
    if best_seq is None:
        n_fallback = 256
        best_seq = np.full(n_fallback, math.sqrt(2 * n_fallback) / n_fallback,
                           dtype=np.float64)

    # Ensure a plain list of Python floats is returned.
    return [float(x) for x in best_seq]
