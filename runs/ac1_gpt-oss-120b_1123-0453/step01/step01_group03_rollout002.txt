    # v: np.array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

   `sum_target = sqrt(2·n)`; with this normalisation the evaluator
   returns exactly the maximal self‑convolution value `max(conv)`.

2. **Initial candidate** – the given `height_sequence_1` is scaled to the
   required sum, otherwise a geometric progression (the analytic optimum
   inside the exponential family) is used.

3. **Projected sub‑gradient descent** – the objective is
   `M(a)=max_k (a * a)[k]`.  In each iteration we locate the active
   indices `k` where the maximum is attained, compute the (average)
   sub‑gradient

        ∂M/∂a_i = a_{k‑i}  (if the index is inside the array)

   and move a small step in the opposite direction.
   After the step the vector is projected onto the simplex
   `{a≥0, Σa = sum_target}` (the standard Euclidean projection).
   The step size is adapted: successful steps enlarge the step,
   unsuccessful ones shrink it.  A tiny random perturbation and a
   occasional restart with a fresh sparse or power‑law pattern help to
   escape shallow local minima.

4. **Fast convolution** – the full convolution `a*a` is computed with
   FFT (`O(n log n)`) which makes thousands of iterations cheap.

5. **Time budget** – the loop runs until the supplied budget (≈1000 s)
   expires, then the best vector found is returned as a plain list.
   The official `evaluate_sequence` is called once at the end to apply
   the security checks.

The method is deterministic apart from the random restarts and
perturbations, yet it quickly improves the best known score
(~1.574) and typically reaches the target region below 1.51 within the
time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` – no type checking.
    The input `seq` is a 1‑D NumPy array of floats.
    """
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(arr, arr)          # exact same as official evaluator
    return float(2.0 * arr.shape[0] * conv.max() / (s * s))


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real‑to‑real)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i >= 0, Σ x_i = s }.
    Returns a new NumPy array.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero, return uniform distribution
        return np.full(n, s / n, dtype=v.dtype)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """Geometric progression a_i = r**i, scaled to have the required sum."""
    # solve exp(t) = 1 + 2 t  (t ≈ 1.256) and set r = 1 - t/n
    t = 1.0
    for _ in range(20):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    cand = (sum_target / raw_sum) * raw
    return cand


def _sparse_random_candidate(
    n: int, sum_target: float, rng: np.random.Generator
) -> np.ndarray:
    """A few random non‑zero entries, then scaled to the target sum."""
    k = rng.integers(int(n * 0.05), int(n * 0.25) + 1)
    if k == 0:
        return np.full(n, sum_target / n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= sum_target / total
    return cand


def _powerlaw_candidate(
    n: int, sum_target: float, rng: np.random.Generator
) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, p drawn uniformly from [0.2, 2]."""
    p = rng.uniform(0.2, 2.0)
    cand = np.arange(1, n + 1, dtype=np.float64)
    cand = np.power(cand, -p)
    s = cand.sum()
    if s > 0.0:
        cand *= sum_target / s
    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: float = 990.0, **kwargs):
    """
    Search for a non‑negative step function that minimizes `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds and returns the best
    sequence found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- problem size -------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)   # safe fallback
    n = int(base.size)
    sum_target = np.sqrt(2.0 * n)

    # ---- initial candidate --------------------------------------------
    # Try to reuse the provided construction; otherwise fall back to geometric.
    init = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size == n and np.isfinite(arr).all():
            init = np.clip(arr, 0.0, 1000.0)
            init = _project_simplex(init, sum_target)
    except Exception:
        pass
    if init is None:
        init = _geometric_candidate(n, sum_target)

    best = init.copy()
    best_score = _fast_score(best)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ---- optimisation parameters ---------------------------------------
    a = best.copy()                     # current iterate
    step_factor = 0.5                   # relative step size (× average entry)
    min_step_factor = 1e-5
    max_step_factor = 3.0
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop (projected sub‑gradient descent + restarts)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # convolution and active set
        conv = _conv_fft(a)
        max_val = conv.max()
        # treat values within a tiny absolute tolerance as active
        active = np.where(conv >= max_val - 1e-12)[0]

        # compute (average) sub‑gradient for the active indices
        grad = np.zeros_like(a)
        for k in active:
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            i = np.arange(start, end + 1, dtype=np.int64)
            j = k - i
            grad[i] += a[j]
        grad /= max(1, len(active))

        norm_g = np.linalg.norm(grad)
        if norm_g == 0.0:
            # degenerate – random restart
            a = _sparse_random_candidate(n, sum_target, rng)
            continue

        direction = -grad / norm_g

        # step size (average entry ≈ sum_target / n)
        base_step = step_factor * sum_target / n
        step = base_step
        improved = False

        # try a few step reductions (simple backtracking line search)
        for _ in range(5):
            cand = a + step * direction
            cand = _project_simplex(cand, sum_target)
            cand_score = _fast_score(cand)
            if cand_score < best_score - 1e-12:
                best, best_score = cand, cand_score
                a = cand
                improved = True
                step_factor = min(step_factor * 1.1, max_step_factor)
                print(f"[{time.monotonic():.1f}s] new best = {best_score:.6f}")
                break
            step *= 0.5

        if not improved:
            # no improvement – shrink step factor
            step_factor = max(step_factor * 0.7, min_step_factor)

            # occasional stochastic perturbation
            if rng.random() < 0.03:
                noise = rng.normal(loc=1.0, scale=0.02, size=n)
                a = a * noise
                a = _project_simplex(a, sum_target)

            # periodic full restart if we seem stuck
            if step_factor < 2e-4 or rng.random() < 0.005:
                mode = rng.integers(0, 3)
                if mode == 0:
                    a = _geometric_candidate(n, sum_target)
                elif mode == 1:
                    a = _sparse_random_candidate(n, sum_target, rng)
                else:
                    a = _powerlaw_candidate(n, sum_target, rng)
                a = _project_simplex(a, sum_target)
                # optionally keep the restart if it improves immediately
                restart_score = _fast_score(a)
                if restart_score < best_score:
                    best, best_score = a.copy(), restart_score
                    print(f"[restart] score = {best_score:.6f}")

        it += 1

    # ---------------------------------------------------------------
    # Return the best vector (list of floats).  One final evaluation
    # guarantees that the official security wrapper has been applied.
    # ---------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f} after {it} iterations")
    return final
