"""
Gradient‑descent optimiser for the step‑function attack.

We minimise   score = 2·n·max_k (a*a)[k] / (∑a)²
which, after normalising the ℓ¹‑norm to √(2n), is simply the
maximum self‑convolution value.  The algorithm works on a
single‑parameter sequence a∈ℝⁿ₊ and repeatedly performs a
soft‑max‑based gradient step on the loss

    Lβ(a) = (1/β)·log ∑_k exp(β·(a*a)[k])

which approaches the ℓ∞‑norm as β→∞.  The gradient can be written
as a convolution and is evaluated with FFTs (O(n log n) per step).
A handful of heuristics – learning‑rate decay, β‑schedule,
occasional Gaussian noise and symmetric re‑normalisation – allow the
method to escape local minima and quickly improve on the previous
best score (≈1.574).  The routine runs until the supplied time budget
expires and returns the best sequence discovered.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the official one but without checks)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the official score for a valid (already clipped) sequence."""
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Utility: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Candidate generators (used only for the initial guess)
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """Geometric progression a_i = r**i, scaled to the required ℓ¹‑norm."""
    # Solve e^t = 1 + 2t   (≈1.256)   – this is the classical start.
    t = 1.0
    for _ in range(20):
        f = math.exp(t) - 1.0 - 2.0 * t
        fp = math.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    cand = (sum_target / raw_sum) * raw
    return cand


def _sparse_random_candidate(n: int, sum_target: float, rng: np.random.Generator) -> np.ndarray:
    """Random sparse vector (5–25 % non‑zero) scaled to the required norm."""
    k = rng.integers(max(1, int(n * 0.05)), max(2, int(n * 0.25) + 1))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= (sum_target / total)
    return cand

    # Pre‑allocate arrays to avoid repeated allocations
    a_rev = np.empty_like(a)

    # ------------------------------------------------------------------
    # Hyper‑parameters (will be annealed)
    # ------------------------------------------------------------------
    lr = 0.2            # initial learning rate
    beta = 0.5          # soft‑max temperature (small → smooth)
    min_lr = 1e-4
    lr_decay = 0.9995   # multiplicative decay per iteration
    beta_max = 12.0
    beta_growth = 1.001

    best = a.copy()
    best_score = _fast_score(best)

    start = time.monotonic()
    it = 0

    print(f"[init] n={n}, sum_target={sum_target:.3f}, start score={best_score:.6f}")

    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) self‑convolution a*a via FFT (full length 2n‑1)
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(a_fft * a_fft, fft_len)
        b = conv_full[: 2 * n - 1]                    # (a*a)

        max_b = b.max()
        cur_score = float(2.0 * n * max_b / (sum_target * sum_target))

        # --------------------------------------------------------------
        # 2) Keep the best solution
        # --------------------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            elapsed = time.monotonic() - start
            print(f"[{elapsed:5.1f}s] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) Soft‑max weights for the loss Lβ = (1/β)·log∑exp(β·b)
        # --------------------------------------------------------------
        # Stabilise the exponent by subtracting max_b
        w = np.exp(beta * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 4) Gradient of Lβ w.r.t. a:
        #    ∇_a Lβ = 2 * (w ⋆ a_rev)  (⋆ = full convolution)
        # --------------------------------------------------------------
        a_rev[:] = a[::-1]
        w_fft = np.fft.rfft(w, fft_len)
        rev_fft = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(w_fft * rev_fft, fft_len)
        # Extract the central n entries (indices n‑1 … n‑1+n‑1)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # --------------------------------------------------------------
        # 5) Gradient step, projection to the feasible set
        # --------------------------------------------------------------
        a -= lr * grad
        # Enforce non‑negativity and the evaluator's clipping range
        a = np.clip(a, 0.0, 1000.0)

        # Renormalise the ℓ¹‑norm exactly
        s = a.sum()
        if s < 1e-12:                     # numerical safeguard
            # if everything collapsed, re‑initialise a tiny random vector
            a = rng.random(n) + 1e-3
            s = a.sum()
        a *= (sum_target / s)

        # --------------------------------------------------------------
        # 6) Annealing / occasional random jitter
        # --------------------------------------------------------------
        lr = max(min_lr, lr * lr_decay)
        if beta < beta_max:
            beta *= beta_growth

        # Occasionally inject a small Gaussian perturbation to escape plateaus
        if rng.random() < 0.01:
            jitter = rng.normal(scale=0.01 * a.mean(), size=n)
            a += jitter
            a = np.clip(a, 0.0, 1000.0)
            a *= (sum_target / a.sum())

        # --------------------------------------------------------------
        # 7) Simple restart if we have been stagnant for a while
        # --------------------------------------------------------------
        if it % 5000 == 0:
            # Keep the current best but try a fresh random start as a backup
            a = _sparse_random_candidate(n, sum_target, rng)

    # ------------------------------------------------------------------
    # Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    final_score = float(evaluate_sequence(final))
    total_time = time.monotonic() - start
    print(f"[done] total time = {total_time:.1f}s, final score = {final_score:.6f}")
    return final
