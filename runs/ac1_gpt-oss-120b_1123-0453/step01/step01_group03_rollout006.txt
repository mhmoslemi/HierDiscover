def _project_onto_simplex(v, s):
    # v is numpy array
    # Implementation for Euclidean projection onto simplex.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - s) / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # all zeros?
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑autocorrelation (low max‑self‑convolution) step function.

Algorithm
---------
* Normalise every candidate to the L1‑norm `sqrt(2·n)`.  With this
  normalisation the official score equals the maximal self‑convolution
  value `max(a*a)`, which is scale‑invariant.

* Initialise the search with a diverse set of analytically motivated
  families:
    – geometric progression,
    – its reverse,
    – sinusoidal, triangular,
    – power‑law, Gaussian,
    – inverse‑square‑root (theoretical ∼ π/2 optimum),
    – sparse random patterns.

* Main loop (until the time budget expires) repeatedly applies a
  random “move’’ chosen from:
    0. block‑wise multiplicative perturbation,
    1. single‑index scaling,
    2. redistribution between two indices,
    3. fresh sparse random candidate,
    4. fresh power‑law candidate,
    5. fresh Gaussian candidate,
    6. fresh inverse‑sqrt candidate,
    7. a few steps of smooth‑max gradient descent starting from the
       current best,
    8. optional symmetrisation (`a ← (a+a[::-1])/2`).

  After each mutation the fast evaluator (identical to the official
  `evaluate_sequence`) is called; improvements replace the current best.

* When the time runs out the best vector is converted to a plain Python
  list, fed once through the official `evaluate_sequence` (to trigger the
  security checks) and returned.

The combination of strong analytical seeds, aggressive random moves,
and a cheap projected‑gradient descent on a smooth approximation of the
`max` objective typically pushes the score well below the previous best
(~1.57) and often reaches the target region < 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast objective (identical to the official evaluator, without checks)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the value computed by `evaluate_sequence` for a valid seq."""
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(arr, arr)
    return float(2.0 * arr.shape[0] * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (L1‑norm = `s`)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u - (cssv - s) / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators (all return a NumPy array of length n, sum = sum_target)
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """a_i = r**i, normalised."""
    # Solve e^t = 1 + 2t (t≈1.256) and set r = 1‑t/n.
    t = 1.0
    for _ in range(20):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    r = max(0.0, 1.0 - t / n)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    return raw * (sum_target / raw_sum)


def _sin_candidate(n: int, sum_target: float) -> np.ndarray:
    """a_i = sin(π·(i+1)/(n+1)), scaled."""
    x = np.arange(1, n + 1, dtype=np.float64)
    raw = np.sin(np.pi * x / (n + 1))
    raw_sum = raw.sum()
    return raw * (sum_target / raw_sum)


def _triangular_candidate(n: int, sum_target: float) -> np.ndarray:
    """Symmetric triangular shape (1,2,…,⌈n/2⌉,…,2,1)."""
    left = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[-2::-1]
    else:
        right = left[::-1]
    raw = np.concatenate([left, right])
    raw_sum = raw.sum()
    return raw * (sum_target / raw_sum)


def _powerlaw_candidate(n: int, sum_target: float, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}, scaled."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    raw = np.power(idx, -p)
    raw_sum = raw.sum()
    return raw * (sum_target / raw_sum)


def _gaussian_candidate(n: int, sum_target: float, sigma: float) -> np.ndarray:
    """Discretised Gaussian centred in the middle, scaled."""
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    raw = np.exp(-0.5 * (x / sigma) ** 2)
    raw_sum = raw.sum()
    return raw * (sum_target / raw_sum)


def _sparse_random_candidate(n: int, sum_target: float, rng: np.random.Generator) -> np.ndarray:
    """Random sparse vector (5‑25 % non‑zero entries), scaled."""
    k = rng.integers(int(n * 0.05), int(n * 0.25) + 1)
    idx = rng.choice(n, size=max(k, 1), replace=False)
    vals = rng.exponential(scale=1.0, size=idx.size)
    raw = np.zeros(n, dtype=np.float64)
    raw[idx] = vals
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.full(n, sum_target / n, dtype=np.float64)
        raw_sum = sum_target
    return raw * (sum_target / raw_sum)


def _inverse_sqrt_candidate(n: int, sum_target: float, delta: float, power: float = 0.5) -> np.ndarray:
    """a_i = (i+delta)^{-power}, scaled."""
    idx = np.arange(n, dtype=np.float64) + delta
    raw = np.power(idx, -power)
    raw_sum = raw.sum()
    return raw * (sum_target / raw_sum)


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (projected onto the L1‑simplex)
# ----------------------------------------------------------------------
def _gradient_descent_step(a: np.ndarray, sum_target: float,
                          lr: float, alpha: float) -> np.ndarray:
    """
    One gradient‑descent step on the smooth approximation
        F(a) = logsumexp(α·conv(a))/α
    The gradient is projected onto the simplex to keep the L1‑norm fixed.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    mx = conv.max()
    # smooth weights (stable exp)
    expargs = alpha * (conv - mx)
    # subtract max to avoid overflow
    expargs -= expargs.max()
    w = np.exp(expargs)
    w /= w.sum()
    # gradient: 2 * (w * a_rev) convolution, then slice the central part
    rev_a = a[::-1]
    grad_full = np.convolve(w, rev_a)
    grad = 2.0 * grad_full[n - 1: n - 1 + n]
    # step & projection
    a_new = a - lr * grad
    a_new = _project_onto_simplex(a_new, sum_target)
    # clipping to the legal interval
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function minimizing `evaluate_sequence`.
    Returns the best sequence (list of float) found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)
    # A tiny safety margin – the caller may give a budget a little too large.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Determine problem size (fallback to 256 if the global is missing)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)
    n = int(base.size)
    sum_target = np.sqrt(2.0 * n)      # the normalisation used in the paper

    # ------------------------------------------------------------------
    # Initialise a diverse pool of candidates
    # ------------------------------------------------------------------
    best = None
    best_score = float("inf")

    def _maybe_update(cand: np.ndarray, label: str = ""):
        nonlocal best, best_score
        cand = np.clip(cand, 0.0, 1000.0)
        s = _fast_score(cand)
        if s < best_score:
            best, best_score = cand, s
            elapsed = time.monotonic() - (deadline - budget_s + 2.0)
            print(f"[{elapsed:5.1f}s] new best ({label}) = {best_score:.6f}")

    # deterministic family seeds
    _maybe_update(_geometric_candidate(n, sum_target), "geo")
    _maybe_update(_geometric_candidate(n, sum_target)[::-1].copy(), "rev_geo")
    _maybe_update(_sin_candidate(n, sum_target), "sin")
    _maybe_update(_triangular_candidate(n, sum_target), "tri")
    # a few power‑law seeds
    for p in (0.3, 0.5, 0.8, 1.2, 2.0):
        _maybe_update(_powerlaw_candidate(n, sum_target, p), f"pow_{p:.1f}")
    # Gaussian seeds (different sigmas)
    for sigma in (n * 0.05, n * 0.12, n * 0.25):
        _maybe_update(_gaussian_candidate(n, sum_target, sigma), f"gauss_{sigma:.0f}")
    # inverse‑sqrt seeds
    for delta in (0.1, 0.5, 1.0, 2.0, 4.0):
        _maybe_update(_inverse_sqrt_candidate(n, sum_target, delta), f"isqrt_δ{delta}")

    # a random sparse starter – often helps escape bad basins
    _maybe_update(_sparse_random_candidate(n, sum_target, rng), "sparse0")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        mode = rng.integers(0, 9)   # nine different move families

        # ----------------------------------------------------------------
        # 0 – block multiplicative perturbation (as in the reference code)
        # ----------------------------------------------------------------
        if mode == 0:
            cand = best.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.12))
            cand[start:start + width] *= factor
            if rng.random() < 0.2:
                cand = 0.5 * (cand + cand[::-1])

        # ----------------------------------------------------------------
        # 1 – single‑index scaling
        # ----------------------------------------------------------------
        elif mode == 1:
            cand = best.copy()
            i = rng.integers(n)
            factor = np.exp(rng.normal(0.0, 0.09))
            cand[i] *= factor

        # ----------------------------------------------------------------
        # 2 – redistribution between two indices
        # ----------------------------------------------------------------
        elif mode == 2:
            cand = best.copy()
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * 0.4
            delta = eps * min(cand[i], cand[j])
            if cand[i] > cand[j]:
                cand[i] -= delta
                cand[j] += delta
            else:
                cand[j] -= delta
                cand[i] += delta

        # ----------------------------------------------------------------
        # 3 – brand‑new sparse random pattern
        # ----------------------------------------------------------------
        elif mode == 3:
            cand = _sparse_random_candidate(n, sum_target, rng)

        # ----------------------------------------------------------------
        # 4 – fresh power‑law candidate (random exponent)
        # ----------------------------------------------------------------
        elif mode == 4:
            p = rng.uniform(0.2, 2.5)
            cand = _powerlaw_candidate(n, sum_target, p)

        # ----------------------------------------------------------------
        # 5 – fresh Gaussian candidate (random width)
        # ----------------------------------------------------------------
        elif mode == 5:
            sigma = rng.uniform(n * 0.04, n * 0.30)
            cand = _gaussian_candidate(n, sum_target, sigma)

        # ----------------------------------------------------------------
        # 6 – fresh inverse‑sqrt / power‑law family (random delta & power)
        # ----------------------------------------------------------------
        elif mode == 6:
            delta = rng.uniform(0.0, 4.0)
            power = rng.uniform(0.3, 1.0)   # 0.5 corresponds to the classic 1/√
            cand = _inverse_sqrt_candidate(n, sum_target, delta, power)

        # ----------------------------------------------------------------
        # 7 – a few steps of projected gradient descent from the current best
        # ----------------------------------------------------------------
        elif mode == 7:
            cand = best.copy()
            # hyper‑parameters are drawn randomly to encourage exploration
            steps = rng.integers(3, 7)               # 3‑6 inner steps
            lr = rng.uniform(0.02, 0.25)            # learning rate
            alpha = rng.uniform(5.0, 30.0)          # smooth‑max sharpness
            for _ in range(steps):
                cand = _gradient_descent_step(cand, sum_target, lr, alpha)
                # optionally decay learning rate a little
                lr *= 0.9
            # No extra projection – _gradient_descent_step already projects

        # ----------------------------------------------------------------
        # 8 – forced symmetrisation (helps for many shapes)
        # ----------------------------------------------------------------
        else:  # mode == 8
            cand = 0.5 * (best + best[::-1])

        # Evaluate the mutated candidate
        cand = np.clip(cand, 0.0, 1000.0)
        cur_score = _fast_score(cand)

        if cur_score < best_score:
            best, best_score = cand, cur_score
            elapsed = time.monotonic() - (deadline - budget_s + 2.0)
            print(f"[{elapsed:5.1f}s] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (plain Python list) after a final official check
    # ------------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f}")
    return final
