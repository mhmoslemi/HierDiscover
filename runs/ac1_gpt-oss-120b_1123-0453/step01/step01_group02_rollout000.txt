def fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    # compute convolution via FFT (zero pad to length 2*n)
    size = 1 << (2*n -1).bit_length()
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n-1]
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_conv / (sum_a * sum_a)

"""Propose a near‑optimal step function for the given evaluation metric.

The algorithm works in three stages:

1. **Parametric grid search** –  a_i ≈ (i+α)^{-p} is evaluated on a
   fine grid of (α,p) around the known good region (α≈0.4, p≈0.5).  The
   best shape from this stage is used as a seed.

2. **Gradient‑descent style refinement** –  while respecting the
   scale‑invariance of the score we repeatedly
   * compute the convolution,
   * locate the index k where the convolution attains its maximum,
   * decrease every entry a_i proportionally to a_{k‑i} (the exact
     derivative of the maximal convolution entry) and then renormalise
     to keep the total sum constant.  The step size decays exponentially.

3. **Occasional stochastic perturbations** – every few hundred iterations
   a random block is scaled by a log‑normal factor, optionally symmetrised
   and re‑normalised.  If the perturbation improves the score it is kept.

All operations are carried out with NumPy (FFT based convolution) for
speed; the official ``evaluate_sequence`` function is called only once at
the end to produce the final, verified score.  The routine stops after
the allotted time (≈ 950 s) and returns the best sequence found.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas – as required by the rules)
# ----------------------------------------------------------------------


def fast_score(seq: np.ndarray) -> float:
    """
    Compute the same objective as ``evaluate_sequence`` but using an FFT
    based convolution for speed.  The input is assumed to be a
    1‑dimensional NumPy array of non‑negative floats; values are clipped
    to the allowed interval [0, 1000].
    """
    a = np.clip(seq, 0.0, 1000.0)
    n = a.shape[0]
    # length for FFT – next power of two ≥ 2*n‑1
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return math.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def normalize_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale ``seq`` so that its sum becomes ``target_sum``.
    """
    cur = seq.sum()
    if cur == 0.0:
        return seq
    return seq * (target_sum / cur)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Return a cheap O(n) non‑negative perturbation of ``base``.
    A random contiguous block is multiplied by a log‑normal factor and,
    with probability 0.35, the vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    # reasonable block width (between 0.5 % and 8 % of the vector)
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal factor
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    ``evaluate_sequence``.  The implementation follows the three‑stage
    strategy described in the module docstring.
    """

    # Determine problem size
    # ------------------------------------------------------------------

    idx = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)  # reserve ~20 s for finalisation
