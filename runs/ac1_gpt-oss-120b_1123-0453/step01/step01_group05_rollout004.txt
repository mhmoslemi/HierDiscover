    # similar to https://github.com/inducer/pymc3/blob/master/pymc3/math.py

def _block_perturb(seq, rng):
    n = len(seq)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, n - width + 1))
    candidate = seq.copy()
    candidate[start:start+width] *= np.exp(rng.normal(0.0, 0.12))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    candidate = np.clip(candidate, 0.0, 1000.0)

"""
Greedy LP‑based refinement of the classic “arcsine” step shape.

The method starts from the analytically motivated sequence
    a_i ∝ 1/√(i+1)   (scaled so that Σa = √(2 n)).
For a fixed length n it repeatedly

1.   Computes M = max(a ∗ a) (the quantity that `evaluate_sequence`
     returns when Σa = √(2 n)).
2.   Solves the linear programme
        maximise    Σ b_i
        subject to  a ∗ b ≤ M ,   b_i ≥ 0 .
    The feasible region is a simplex cut by the convolution bound; the
    optimum usually has a larger ℓ₁‑norm than a while respecting the same M.
3.   Scales the solution back to Σb = √(2 n) and mixes it with the current
    sequence:
        a ← (1‑t)·a + t·b   with a small t (0.2 … 0.8).
    The mixture strictly reduces the maximal self‑convolution unless a
    stationary point has been reached.

The convolution a∗b is encoded as a sparse Toeplitz matrix, so each LP
solves a problem of size (2 n‑1)×n with n variables.  After a few LP
iterations (or when the time budget is nearly exhausted) a cheap
block‑perturbation hill‑climber is run for a final fine‑tuning,
exactly as in the reference implementation.

The function returns the best sequence found within the allotted time.
"""

import math
import time

import numpy as np
import scipy.sparse as sp
import cvxpy as cp


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = ind[cond][-1]            # largest index where condition holds
        theta = cssv[cond][-1] / rho   # corresponding threshold
    w = np.maximum(v - theta, 0.0)
    return w


def _make_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Return the (2n‑1)×n sparse Toeplitz matrix that realises a∗b."""
    n = a.shape[0]
    # Each column i contributes a[i] on rows i … i+n‑1
    rows = np.concatenate([np.arange(i, i + n, dtype=np.int32) for i in range(n)])
    cols = np.concatenate([np.full(n, i, dtype=np.int32) for i in range(n)])
    data = np.repeat(a, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _block_perturb(seq: np.ndarray, rng: np.random.Generator,
                   target_sum: float) -> np.ndarray:
    """Cheap random block perturbation (used for a final fine‑tuning)."""
    n = seq.size
    # block width is a modest fraction of n
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    cand = seq.copy()
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    # re‑normalise to the required ℓ₁‑norm
    s = cand.sum()
    if s > 0:
        cand *= target_sum / s
    return cand


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that (hopefully)
    improves the current best bound.  The algorithm follows the LP‑refinement
    described in the module docstring and falls back to a cheap stochastic
    local search when the remaining time is small.
    """
    # ------------------------------------------------------------------
    # 0) Global settings
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # 1) Choose the length n.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        if n == 0:
            raise Exception
        # Limit n for the LP stage (memory / speed).  We keep the original
        # length for the final fine‑tuning.
        n_lp = min(n, 2000)
    except Exception:
        n = 2000
        n_lp = n

    # ------------------------------------------------------------------
    # 2) Initialise with the arcsine (1/√i) shape, scaled to Σa = √(2 n_lp).
    # ------------------------------------------------------------------
    idx = np.arange(1, n_lp + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(idx)
    target_sum = math.sqrt(2.0 * n_lp)
    a = a / a.sum() * target_sum               # ℓ₁‑norm = √(2 n)

    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n_lp}, initial score={best_score:.6f}")

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 3) LP‑driven refinement loop (time‑bounded, max 8 iterations)
    # ------------------------------------------------------------------
    max_lp_iters = 8
    lp_iter = 0
    while lp_iter < max_lp_iters and time.monotonic() < deadline:
        # compute current bound M = max(a∗a)
        conv_aa = np.convolve(a, a)
        M = np.max(conv_aa)

        # build the convolution matrix for the current a
        A = _make_convolution_matrix(a)

        # Linear programme: maximise Σb  s.t. A·b ≤ M , b ≥ 0
        b = cp.Variable(n_lp, nonneg=True)
        constraints = [A @ b <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        try:
            prob.solve(solver=cp.ECOS,
                       warm_start=False,
                       max_iters=2000,
                       abstol=1e-7,
                       reltol=1e-7,
                       feastol=1e-7,
                       verbose=False)
        except Exception as exc:
            print(f"[lp] solver exception: {exc}")
            break

        if b.value is None:
            print("[lp] no solution returned – aborting LP phase")
            break

        b_val = np.maximum(np.array(b.value).flatten(), 0.0)  # guard tiny negatives
        sum_b = b_val.sum()
        if sum_b <= 1e-12:
            print("[lp] sum of b is (almost) zero – aborting")
            break

        # scale b back to the required ℓ₁‑norm
        scale = target_sum / sum_b
        g = b_val * scale

        # try a few mixing parameters and keep the best improvement
        improved = False
        for t in (0.2, 0.5, 0.8):
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, 1000.0)
            # re‑normalise (clipping may have altered the sum)
            s = cand.sum()
            if s == 0:
                continue
            cand *= target_sum / s

            cand_score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(cand_score) and cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand.copy()                     # continue from the improvement
                improved = True
                print(f"[improve] LP iter {lp_iter}, t={t:.2f}, score={best_score:.6f}")
                break

        if not improved:
            # No improvement – perform a small random jitter to escape a plateau
            jitter = np.exp(rng.normal(0.0, 0.02, size=n_lp))
            a = a * jitter
            a = np.clip(a, 0.0, 1000.0)
            a *= target_sum / a.sum()
        lp_iter += 1

    # ------------------------------------------------------------------
    # 4) Final cheap stochastic hill‑climbing (if time permits)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = _block_perturb(best_seq, rng, target_sum)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[improve] block perturbation, score={best_score:.6f}")

    print(f"[final] best score {best_score:.6f} after "
          f"{time.monotonic() - start_time:.2f}s")
    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
