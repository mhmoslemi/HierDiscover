"""
Propose a low‑value step sequence.

The algorithm is a two‑phase search:

1. **LP‑based refinement (main phase)**
   * For several problem sizes `n` we keep a vector `f` whose ℓ¹‑norm is
     √(2 n) (this is the normalisation used by the scorer).
   * In each iteration we compute `M = max(f*f)`.  With `f` fixed the
     convolution `f*g` is linear in a new vector `g`.  We therefore solve
     a linear program that **maximises the ℓ¹‑mass** of `g` subject to the
     bound `f*g ≤ M`.  After the LP we rescale `g` to the target ℓ¹‑norm
     and move a fraction `t` towards it: `f ← (1‑t)·f + t·g`.
   * Repeating this step monotonically reduces `M` and therefore the
     evaluation score `2·n·M / (‖f‖₁)²`.  We run many iterations (up to
     several dozens) for each `n` and keep the best sequence seen.

2. **Simple local search (fallback / polishing)**
   * Starting from the best LP result we repeatedly pick two positions,
     transfer a small amount of mass from one to the other (keeping the
     total sum unchanged) and keep the change if the evaluation improves.
   * This cheap stochastic walk can sometimes push the score a little
     further after the LP has stagnated.

The whole routine respects the 1000 s time budget,
uses CVXPY/ECOS when available and falls back to a pure random walk if
the LP solver cannot be imported.  All returned values are non‑negative,
clipped to the allowed interval `[0, 1000]`, and are returned as a plain
list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global random generator – the seed is set in `propose_candidate`.
_rng = None

# ----------------------------------------------------------------------
def conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the dense convolution matrix A such that (A @ g) == f * g."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill by the definition of linear convolution.
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over indices of f that participate in output k
        for i in range(i_start, i_end + 1):
            j = k - i          # column in g
            A[k, j] = f[i]
    return A


# ----------------------------------------------------------------------
def _score(seq: list[float]) -> float:
    """Thin wrapper around the provided `evaluate_sequence`."""
    # The evaluator already applies all required checks and clipping.
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
def _lp_refine(
    n: int,
    target_sum: float,
    deadline: float,
    mix: float = 0.25,
    max_iters: int = 80,
) -> tuple[np.ndarray | None, float]:
    """
    Run the LP‑based improvement for a given length `n`.

    Returns (best_sequence, best_score) or (None, inf) if nothing
    succeeded.
    """
    global _rng
    best_seq = None
    best_score = float("inf")

    # Initialise a random feasible vector.
    f = _rng.random(n) + 0.1                # avoid exact zeros
    f = f / np.sum(f) * target_sum

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # Current worst auto‑convolution entry.
        M = float(np.max(np.convolve(f, f)))

        # Build the linear convolution matrix (fixed f, linear in g).
        A = conv_matrix(f)

        # --------------------------------------------------------------
        # Solve   max   Σ g_i   s.t.   A·g ≤ M,   g ≥ 0
        # --------------------------------------------------------------
        try:
            import cvxpy as cp
        except Exception:                     # pragma: no cover
            # No LP solver – abort this size.
            break

        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:                     # pragma: no cover
            try:
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
            except Exception:                 # pragma: no cover
                break

        if g.value is None:
            break  # LP failed – give up on this length.

        g0 = np.maximum(g.value, 0.0)         # safeguard against tiny negatives
        s_g = np.sum(g0)
        if s_g < 1e-12:
            break

        # Rescale so that ℓ¹‑norm matches the target (the score is scale‑invariant,
        # but keeping the normalisation makes the LP bound M meaningful).
        g_rescaled = g0 * (target_sum / s_g)

        # Convex‑combination update.
        f_new = (1.0 - mix) * f + mix * g_rescaled
        f_new = np.clip(f_new, 0.0, 1000.0)

        sc = _score(f_new.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = f_new.copy()
            print(f"[LP] n={n:3d} it={it:2d} score={sc:.6f}")

        # Early exit if we have already beaten a very good bound.
        if best_score < 1.48:
            break

        f = f_new

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    A cheap stochastic walk that preserves the total ℓ¹‑mass.
    It swaps a tiny amount of mass between two randomly chosen indices.
    """
    best = seq.copy()
    best_score = _score(best.tolist())
    n = best.shape[0]

    while time.monotonic() < deadline:
        i, j = _rng.integers(0, n, size=2)
        if i == j:
            continue
        # Propose a small transfer; scale by the magnitude of the source.
        delta = _rng.normal(loc=0.0, scale=0.02) * max(1e-6, best[i])
        if delta == 0.0:
            continue
        cand_i = best[i] + delta
        cand_j = best[j] - delta
        if cand_i < 0.0 or cand_j < 0.0:
            continue
        cand = best.copy()
        cand[i] = cand_i
        cand[j] = cand_j
        sc = _score(cand.tolist())
        if sc < best_score:
            best_score = sc
            best = cand
            # keep the improvement visible
            print(f"[local] score={sc:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    The search runs a multi‑size LP‑based refinement (sizes 200, 300, 350,
    400) and finishes with a lightweight random local search.  The routine
    respects the 1000 s overall time limit (a small safety margin is kept).
    """
    global _rng
    _rng = np.random.default_rng(seed)

    # Give ourselves a tiny safety margin before the external timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) LP‑based refinement on several lengths.
    # ------------------------------------------------------------------
    # The benchmark that produced 1.518 used n=200.  We try a few larger
    # lengths; the problem size stays modest enough for ECOS to solve quickly.
    for n in (200, 250, 300, 350, 400):
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        best_seq, best_score = _lp_refine(
            n=n,
            target_sum=target_sum,
            deadline=deadline,
            mix=0.25,
            max_iters=80,
        )
        if best_seq is not None and best_score < overall_best_score:
            overall_best_score = best_score
            overall_best_seq = best_seq.copy()
            print(f"[global] new best after size {n}: {best_score:.6f}")

        # If we already reached a very good bound we can stop early.
        if overall_best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # 2) If the LP stage failed entirely (e.g. CVXPY missing) fall back
    #    to a pure random walk.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        print("[fallback] LP unavailable – start random walk")
        n = 200
        seq = np.full(n, 1.0, dtype=np.float64)
        overall_best_seq = seq.copy()
        overall_best_score = _score(seq.tolist())

    # ------------------------------------------------------------------
    # 3) Light‑weight local polishing (only a few seconds left).
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        polished_seq, polished_score = _local_search(overall_best_seq, deadline)
        if polished_score < overall_best_score:
            overall_best_score = polished_score
            overall_best_seq = polished_seq.copy()
            print(f"[global] after polishing: {polished_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety: return a plain Python list of floats.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # Deterministic safe fallback – a constant sequence of length 256.
        overall_best_seq = np.ones(256, dtype=np.float64)

    # Clip once more to be absolutely sure we follow the validator’s rules.
    overall_best_seq = np.clip(overall_best_seq, 0.0, 1000.0)

    return [float(x) for x in overall_best_seq]
