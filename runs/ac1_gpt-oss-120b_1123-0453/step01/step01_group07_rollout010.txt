"""
Propose a candidate step‑function sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a combination of:
* a dense grid search over power‑law kernels (exponent & offset)
* a smoothed‑max gradient descent on the simplex (exponentiated update)
* occasional block perturbations and random restarts for exploration.

All operations are O(n log n) via FFT, so the routine comfortably fits
the 1000 s budget even for n≈6000–8000.
"""

import time
import numpy as np


def generate_power_sequence(n: int, exponent: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Power‑law kernel a_i = (i + offset + 1)^(-exponent)."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)


def next_pow2(x: int) -> int:
    """Small utility: the next power of two >= x."""
    return 1 << (x - 1).bit_length()


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative floats that (heuristically) minimise
    ``evaluate_sequence``.  The function runs a fast grid‑search,
    followed by a smoothed‑max gradient descent on the probability
    simplex (exponentiated gradient), sprinkled with random block
    perturbations and occasional restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    # safety margin – leave a couple of seconds for clean‑up
    deadline = t_start + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Choose a problem size.
    # If a previous construction exists we reuse its length,
    # otherwise we fall back to a reasonably large default.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            raise ValueError("degenerate previous sequence")
    except Exception:
        n = 5800                # a length that gave the best known result
        a = generate_power_sequence(n, exponent=0.5, offset=0.0)

    # Normalise to a probability distribution (scale does not matter for the score)
    a = a / a.sum()

    best_seq = a.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Dense grid search over power‑law parameters.
    # ------------------------------------------------------------------
    exponent_grid = np.arange(0.45, 0.56, 0.005)         # 0.45 … 0.55
    offset_grid   = np.arange(0.0, 2.01, 0.20)          # 0 … 2 in steps of 0.2
    size_conv = next_pow2(2 * n - 1)                    # FFT size for a*a

    for exp in exponent_grid:
        for off in offset_grid:
            cand = generate_power_sequence(n, exponent=exp, offset=off)
            cand = cand / cand.sum()
            # convolution via FFT (fast, O(n log n))
            A = np.fft.rfft(cand, n=size_conv)
            conv = np.fft.irfft(A * A, n=size_conv)[: 2 * n - 1]
            score = 2.0 * n * conv.max()               # sum(cand)=1 → denominator =1
            if score < best_score:
                best_score = float(score)
                best_seq = cand.copy()
                a = cand.copy()
                print(f"grid exp={exp:.3f} off={off:.2f} → score={score:.6f}")
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Smoothed‑max (log‑sum‑exp) gradient descent on the simplex.
    # ------------------------------------------------------------------
    # Pre‑compute FFT size sufficient for both convolution and correlation.
    size_full = next_pow2(3 * n)            # ≥ n + (2n‑1) – 1
    beta = 0.5                              # starts with a soft max
    beta_max = 200.0
    beta_inc = 0.5
    eta0 = 0.20                             # base step for exponentiated update

    it_without_improve = 0
    last_best = best_score

    while time.monotonic() < deadline:
        it = it_without_improve + 1

        # ---- convolution a*a -------------------------------------------------
        A_fft = np.fft.rfft(a, n=size_full)
        conv_full = np.fft.irfft(A_fft * A_fft, n=size_full)
        conv = conv_full[: 2 * n - 1]                     # length 2n‑1
        max_val = conv.max()
        max_idx = int(np.argmax(conv))

        # ---- current score ----------------------------------------------------
        cur_score = 2.0 * n * max_val                     # sum(a)=1
        if cur_score < best_score:
            best_score = float(cur_score)
            best_seq = a.copy()
            it_without_improve = 0
            last_best = best_score
            print(f"iter {it_without_improve:5d} new best {best_score:.6f}")
        else:
            it_without_improve += 1

        # ---- increase smoothing (beta) ----------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # ---- smooth‑max weights ------------------------------------------------
        # w_i ∝ exp(beta * (conv_i - max_val)), tiny epsilon keeps all entries >0
        eps = 1e-12
        w = np.exp(beta * (conv - max_val))
        w = (w + eps) / (w.sum() + eps * w.size)

        # ---- gradient of the smoothed max (2 * a ⋆ w) -------------------------
        w_rev = w[::-1]                                   # reverse for correlation
        W_fft = np.fft.rfft(w_rev, n=size_full)

        offset = (2 * n - 2)                              # position of zero lag

        # ---- exponentiated‑gradient update (stays on the simplex) ----------
        eta = eta0 / np.sqrt(it)                         # decreasing step

        if a_sum > 0:

        else:                                            # degenerate – re‑initialise
            a = np.abs(rng.normal(size=n))
            a = a / a.sum()

        # ---- occasional random block perturbation -----------------------------
        if rng.random() < 0.04:                           # ~4 % of iterations

            factor = float(np.exp(rng.normal(0.0, 0.12)))  # ~12 % log‑normal change
            a[start : start + width] *= factor
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            print(f"  block perturb at iter {it}")

        # ---- very occasional random restart ----------------------------------
        if it_without_improve > 5000 and rng.random() < 0.15:
            exp = rng.uniform(0.44, 0.56)
            off = rng.uniform(0.0, 2.5)
            a = generate_power_sequence(n, exponent=exp, offset=off)
            a = a / a.sum()
            it_without_improve = 0
            print(f"  restart with exp={exp:.3f} off={off:.2f}")

        # safety break in case something goes wrong
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found (plain Python list of floats).
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
