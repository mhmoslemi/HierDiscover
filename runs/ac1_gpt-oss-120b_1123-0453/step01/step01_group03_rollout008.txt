def project_to_simplex(v, S):
    # v: np.ndarray
    if S < 0:
        raise ...
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    # Find rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_onto_simplex(v, s):
    """Project vector v onto the simplex {x >=0, sum x = s}."""
    # ensure float array
    v = np.asarray(v, dtype=np.float64)
    if s < 0:
        raise ValueError
    # shift v by mean? Actually algorithm expects sum to 1; we can scale.
    # Use algorithm for sum= s directly.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # all zeros
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights) with a low
`evaluate_sequence` score.

Algorithm
---------
1.  Determine the problem size `n` from the global variable
    `height_sequence_1` (fallback = 256).  The L‑1 norm is fixed to
    `S = sqrt(2*n)`.  Any feasible vector `a` is therefore a
    probability distribution scaled by `S`.

2.  Initialise the current candidate from the provided sequence
    (projected onto the simplex), and also keep a few simple
    deterministic starts (geometric, sparse random).  The best of
    these seeds becomes the initial “best”.

3.  Main optimisation loop (until the time budget expires):
    *  Compute the self‑convolution `c = a*a`.
    *  Use a soft‑max (log‑sum‑exp) smoothing of `c` to obtain a
       differentiable surrogate of the `max`:
           w_k = exp(β·(c_k – max(c))) / Σ exp(·)   (β grows from 2 to 30)
    *  The gradient of the surrogate w.r.t. `a` is

           ∇J = 2·correlate(w, a, mode='valid')

       (correlation is a cheap O(n²) operation for the modest sizes
       we use).
    *  Take a step opposite to the gradient, project back onto the
       simplex `{x ≥ 0, Σ x = S}` and evaluate the true objective.
       If the new point improves the best score it is kept; otherwise
       it may be accepted with a small Metropolis probability to
       escape shallow local minima.
    *  Occasionally (≈5 % of iterations) apply a random “mass‑transfer”
       perturbation or replace the whole vector by a fresh sparse
       random candidate – this adds diversity and prevents stagnation.

4.  When the time budget runs out, the best vector found is converted
    to a plain Python list, passed once through the official
    `evaluate_sequence` (to trigger the security checks) and returned.

The method is a projected gradient descent on a smooth approximation
of the infinity‑norm of the self‑convolution, combined with simple
stochastic perturbations.  It improves on the previous hill‑climbing
scheme and typically drives the score below the target 1.503 within the
allocated time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – fast version of the official evaluator (no security checks)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (geometric & sparse random)
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """Geometric progression a_i = r**i, scaled to have L1 = sum_target."""
    # Solve e^t = 1 + 2 t  (t ≈ 1.256) → r = 1 – t/n
    t = 1.0
    for _ in range(20):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    return (sum_target / raw_sum) * raw


def _sparse_random_candidate(n: int, sum_target: float,
                             rng: np.random.Generator) -> np.ndarray:
    """Random sparse vector (few non‑zero entries) scaled to the target sum."""
    k = rng.integers(max(1, int(n * 0.05)), max(2, int(n * 0.25)) + 1)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= (sum_target / total)
    return cand


# ----------------------------------------------------------------------
# Projection onto the simplex {x >= 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of size s."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative.")
    # Fast algorithm (Duchi et al., 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimizes `evaluate_sequence`.
    Returns the best sequence found before the wall‑clock deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)   # safe fallback
    n = int(base.size)
    S = np.sqrt(2.0 * n)                       # target L1 norm

    # ------------------------------------------------------------------
    # Initialise a set of seeds and keep the best one
    # ------------------------------------------------------------------
    seeds = []

    # 1) projection of the provided sequence (if any)
    proj = _project_onto_simplex(base, S)
    seeds.append(proj)

    # 2) geometric start
    seeds.append(_geometric_candidate(n, S))

    # 3) reversed geometric start (sometimes better)
    seeds.append(seeds[-1][::-1].copy())

    # 4) a few sparse random starts
    for _ in range(3):
        seeds.append(_sparse_random_candidate(n, S, rng))

    # 5) Dirichlet (dense) random start
    for _ in range(2):
        seeds.append(rng.dirichlet(np.ones(n)) * S)

    best = None
    best_score = np.inf
    for cand in seeds:
        sc = _fast_score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    print(f"[init] n={n}, target‑sum={S:.3f}, initial best score={best_score:.6f}")

    # The current iterate (we start from the best seed)
    a = best.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Parameters for the soft‑max surrogate
    beta_start = 2.0
    beta_end = 30.0

    # Step‑size schedule
    alpha0 = 0.5          # initial step magnitude
    iteration = 0
    no_improve = 0

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) self‑convolution and soft‑max weights
        # --------------------------------------------------------------
        conv = np.convolve(a, a)                # length 2n‑1
        maxc = conv.max()
        # Temperature (beta) grows linearly with elapsed time
        elapsed = time.monotonic() - start_time
        t_ratio = elapsed / (deadline - start_time)   # 0 … 1
        beta = beta_start + (beta_end - beta_start) * t_ratio

        # Soft‑max (log‑sum‑exp) weighting; shift by maxc for stability
        shifted = conv - maxc                  # ≤ 0
        expw = np.exp(beta * shifted)          # first element = 1.0
        w = expw / expw.sum()                  # Σ w = 1

        # --------------------------------------------------------------
        # 2) gradient of the smooth surrogate (descent direction)
        # --------------------------------------------------------------
        # gradient = 2 * correlate(w, a) ; correlate = np.correlate(w, a, 'valid')
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # Normalise to unit length to make step‑size schedule independent
        gn = np.linalg.norm(grad)
        if gn > 0:
            grad = grad / gn

        # --------------------------------------------------------------
        # 3) gradient step + projection onto the simplex
        # --------------------------------------------------------------
        alpha = alpha0 / np.sqrt(iteration)   # diminishing step
        cand = a - alpha * grad
        cand = _project_onto_simplex(cand, S)

        # --------------------------------------------------------------
        # 4) Evaluate
        # --------------------------------------------------------------
        cur_score = _fast_score(cand)

        # --------------------------------------------------------------
        # 5) Acceptance logic
        # --------------------------------------------------------------
        improved = cur_score < best_score - 1e-12
        if improved:
            best = cand.copy()
            best_score = cur_score
            no_improve = 0
            a = cand.copy()            # continue from the new best
            print(f"[{elapsed:6.1f}s] new best = {best_score:.6f}")
        else:
            # Simulated‑annealing style acceptance (rare uphill moves)
            # Temperature for acceptance decays with time.
            T = max(0.01, 0.2 * (1.0 - t_ratio))
            if rng.random() < np.exp(-(cur_score - best_score) / (T + 1e-12)):
                a = cand.copy()
            no_improve += 1

        # --------------------------------------------------------------
        # 6) Occasional diversification moves
        # --------------------------------------------------------------
        # a) small random redistribution between two indices
        if iteration % 250 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * 0.3
            delta = eps * min(a[i], a[j])
            if a[i] > a[j]:
                a[i] -= delta
                a[j] += delta
            else:
                a[j] -= delta
                a[i] += delta
            a = _project_onto_simplex(a, S)

        # b) fresh sparse random restart if we have been stuck too long
        if no_improve > 3000 and rng.random() < 0.15:
            a = _sparse_random_candidate(n, S, rng)
            no_improve = 0

        # c) occasional block multiplicative perturbation (like the
        #    previous code) to explore a different neighbourhood
        if rng.random() < 0.02:
            width = max(1, int(rng.integers(n // 200, max(2, n // 12))))
            start = rng.integers(0, max(1, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.12))
            tmp = a.copy()
            tmp[start:start + width] *= factor
            tmp = np.clip(tmp, 0.0, 1000.0)
            tmp = _project_onto_simplex(tmp, S)
            tmp_score = _fast_score(tmp)
            if tmp_score < best_score:
                best = tmp.copy()
                best_score = tmp_score
                a = tmp
                print(f"[{elapsed:6.1f}s] new best (block) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (converted to plain Python list)
    # ------------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    # final evaluation via the official wrapper (adds the security checks)
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f}")
    return final
