"""
Iterative linear‑programming flattening of the self‑convolution.

The algorithm keeps the L¹‑norm of the step function fixed.
At each iteration we:
  1. Compute the current worst self‑convolution value M = max(a*a).
  2. Build the linear map A such that A @ g = a * g (ordinary convolution).
  3. Solve the LP  maximise Σ gᵢ   s.t.  A @ g ≤ M  and g ≥ 0.
     The optimum g has a larger total mass than a while still respecting the
     same peak M, therefore mixing a toward g reduces the peak.
  4. Scale g back to the original L¹‑norm, mix a ← (1‑t)·a + t·g and repeat.
We adapt the mixing weight t when progress stalls and fall back to a cheap
random block perturbation if the LP fails.  The best sequence seen is
returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Construct the dense matrix A (size (2n‑1)×n) such that
        A @ g == np.convolve(a, g)   (valid for 0‑based indexing).

    For each column i, the column is a shifted copy of `a`:
        A[i : i + n, i] = a
    All other entries are zero.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = a
    return A


def _fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the evaluation score 2·n·max(a*a)/(sum a)² using FFT,
    which is much faster than the official checker.
    """
    n = a.shape[0]
    a_hat = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(a_hat * a_hat, fft_len)[: 2 * n - 1]
    M = conv.max()
    s = a.sum()
    return 2.0 * n * M / (s * s)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Very cheap stochastic move used only when the LP cannot make progress.
    It mimics the block‑perturbation from the reference code.
    """
    cand = a.copy()
    n = a.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    above and runs for at most ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Choose the problem size and an initial sequence.
    # --------------------------------------------------------------
    try:
        # If a previous construction exists we reuse (or down‑sample) it.
        base = np.asarray(height_sequence_1, dtype=np.float64)  # pylint: disable=undefined-variable
        n0 = base.size
        if n0 > 2500:

            n = 2000

    except Exception:
        # No previous sequence – start from the analytically motivated arcsine shape.

    # Normalise the L¹‑norm (the absolute scale does not influence the score).

    a = base.copy()

        fast_sc = _fast_score(a_new, fft_len)

            else:

        else:

            no_improve = 0

    # 4) Return the best sequence discovered.
    # --------------------------------------------------------------
    total_time = time.monotonic() - (deadline - budget_s + 5.0)
    print(f"[final] best score {best_score:.6f} after {total_time:.2f} s")
    return [float(v) for v in best_seq]
