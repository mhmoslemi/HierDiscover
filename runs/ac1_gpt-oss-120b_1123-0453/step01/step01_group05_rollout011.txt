"""
propose_candidate
-----------------

Generate a step‑function (a non‑negative list of heights) that tries to minimise

    2·n·max(a*a) / (sum(a)²)

where ``a*a`` denotes the full linear self‑convolution of the list.
The algorithm works in three stages:

1. **Initialisation** – use the classic arcsine shape (a_i ∝ 1/√i) which is already
   close to optimal, or fall back to a random Dirichlet draw if a previous
   solution is available.

2. **Iterative smoothing** – repeatedly flatten the convolution by a
   projected‑gradient (mirror‑descent) step.
   The loss used is the L² distance between the self‑convolution and the
   perfectly flat target (which is 1/(2n‑1) for each entry).
   The gradient of this loss can be written as a (cross‑)convolution,
   so each iteration only needs a few FFTs (O(n log n)).

3. **Tracking the best** – the official ``evaluate_sequence`` function is called
   only occasionally (to keep the overhead low).  The best sequence found
   during the time budget is returned.

The whole optimiser is pure NumPy, uses only top‑level helper functions
(no closures, no lambdas) and respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _linear_convolve_fft(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution of ``a`` and ``b`` using FFT.
    ``size`` must be ≥ len(a) + len(b) - 1.
    Returns the full convolution (length = len(a)+len(b)-1).
    """
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Produce a list of non‑negative heights that (hopefully) improves the
    evaluation score.  The function respects the contest rules:
    * only top‑level helpers are used,
    * no filesystem/network I/O,
    * ``evaluate_sequence`` is called directly (it is assumed to be imported).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose a problem size (prefer re‑using a previous construction)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        if n == 0:
            raise Exception
    except Exception:
        n = 5000                               # a modest size that still gives many degrees of freedom

    # ------------------------------------------------------------------
    # 2) Initialise the probability vector p (sum(p) == 1)
    # ------------------------------------------------------------------
    # 2a) Prefer the provided “old” sequence, but fall back to the arcsine shape.
    p = None
    try:
        old = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if old.shape[0] == n and np.isfinite(old).all():
            p = old.clip(min=0.0)
    except Exception:
        pass

    if p is None:
        # arcsine shape a_i ∝ 1/√(i+1)
        idx = np.arange(1, n + 1, dtype=np.float64)
        p = 1.0 / np.sqrt(idx)
    # Normalise to a proper probability distribution (L1‑norm = 1)
    p_sum = p.sum()
    if p_sum <= 0 or not np.isfinite(p_sum):
        p = np.full(n, 1.0 / n)
    else:
        p = p / p_sum

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT sizes that will be reused
    # ------------------------------------------------------------------
    # convolution p * p needs length 2n‑1
    size_conv = _next_pow2(2 * n - 1)
    # cross‑correlation for the gradient needs length (2n‑1)+n‑1 = 3n‑2
    size_corr = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 4) Optimisation loop – mirror‑descent on the L2 distance to the flat target
    # ------------------------------------------------------------------
    avg_target = 1.0 / (2 * n - 1)               # the uniform value each entry would have if the convolution were flat
    best_seq = p.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start = time.monotonic()
    deadline = start + float(budget_s)
    # hyper‑parameters
    eta0 = 0.5                                   # base step size (used after normalising by max|grad|)
    eval_interval = 0.3                          # seconds between official evaluations
    next_eval = start + eval_interval
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # ----- (a) self‑convolution a*a --------------------------------
        p_fft = np.fft.rfft(p, size_conv)
        conv = np.fft.irfft(p_fft * p_fft, size_conv)[: 2 * n - 1]

        # ----- (b) gradient of the L2 loss -----------------------------
        # w = conv - avg_target  (can be negative)
        w = conv - avg_target

        # cross‑correlation w ⋆ p   =>  grad = 4 * (w ⋆ p)
        p_rev = p[::-1]
        w_fft = np.fft.rfft(w, size_corr)
        p_rev_fft = np.fft.rfft(p_rev, size_corr)
        corr = np.fft.irfft(w_fft * p_rev_fft, size_corr)[: 3 * n - 2]
        grad = 4.0 * corr[n - 1 : n - 1 + n]      # length n

        # ----- (c) mirrored‑gradient step (exponentiated update) -----
        maxg = np.max(np.abs(grad))
        if maxg > 0:
            eta = eta0 / maxg                      # guarantees |eta*grad| ≤ eta0
        else:
            eta = 0.0

        # exponentiated‑gradient (always stays non‑negative)
        p = p * np.exp(-eta * grad)

        # ----- (d) re‑normalise to stay on the simplex -----------------
        s = p.sum()
        if s <= 0 or not np.isfinite(s):
            # a numerical safeguard – fall back to a tiny uniform perturbation
            p = np.full(n, 1.0 / n)
        else:
            p = p / s

        # ----- (e) occasional official evaluation -----------------------
        now = time.monotonic()
        if now >= next_eval:
            next_eval = now + eval_interval
            try:
                score = float(evaluate_sequence(p.tolist()))
                if score < best_score:
                    best_score = score
                    best_seq = p.copy()
                    print(f"[improve] iter={iteration}, score={best_score:.6f}, t={now - start:.2f}s")
            except Exception:
                # any unexpected error – just continue
                pass

        # ----- (f) a tiny random jitter to avoid stagnation ------------
        # (performed every 1000 iterations)
        if iteration % 1000 == 0:
            jitter = np.exp(0.001 * rng.normal(size=n))  # multiplicative log‑normal noise
            p = p * jitter
            p = p / p.sum()  # re‑normalise

    # ------------------------------------------------------------------
    # 5) Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"[final] best score {best_score:.6f} after {time.monotonic() - start:.2f}s")
    return [float(x) for x in best_seq]
