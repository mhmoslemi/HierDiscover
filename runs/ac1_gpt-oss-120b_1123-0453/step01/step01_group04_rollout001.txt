Now need projection onto simplex. Implementation as per algorithm:

def project_onto_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v, s=1.0):
    # v is 1D np.ndarray
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v)+1))
    # Find last index where rho_arr > 0
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        # all zero?
        theta = cssv[-1]/len(v)
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def correlation(a, w):
    # both real arrays
    N = 1 << ((len(a) + len(w) - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    W = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(A * np.conj(W), n=N)
    # result length = len(a) + len(w) - 1
    return corr

Propose a low‑value step‑function for the given evaluation routine.

Algorithm
---------
* Work on the probability simplex (the sequence is normalised to sum 1,
  because the objective is invariant under scaling).
* Use a smooth approximation of the max‑convolution:
      w_k = exp(β·(c_k – max(c)))   (β grows during the run)
  where c = a * a is the self‑convolution.
* The (sub‑)gradient of the smooth max is
      g = 2·corr(a , w)            (cross‑correlation of a with the weight vector)
  which can be obtained efficiently with FFT‑based convolution.
* Perform a projected gradient step
      a ← a – η·g
  followed by:
      – optional symmetrisation,
      – projection onto the simplex (non‑negative entries, sum = 1).
* Occasionally apply a random block perturbation, and restart from a fresh
  power‑law when the search has stalled.
* All heavy operations (convolution, correlation) use FFTs, making it cheap
  enough to run many thousands of iterations within the 1000 s budget.
* The best sequence found (in the original scale – scaling does not affect the
  objective) is returned as a plain list of floats.
"""

import time

# Helper functions (top‑level – no closures)

def project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x ≥ 0, ∑x = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond].max() - 1  # zero‑based index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real‑FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[: 2 * n - 1]
    return conv


def correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w) = ∑_j a_j·w_{j+k} for all lags k,
    returned as a real array of length len(a)+len(w)-1.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(A * np.conj(W), n=size)[: n + m - 1]
    return corr


def make_power_law(n: int, alpha: float) -> np.ndarray:
    """Return a non‑negative power‑law sequence a_i = (i+1)^{-α} (i=0..n‑1)."""
    idx = np.arange(n, dtype=np.float64)
    a = np.power(idx + 1.0, -alpha)
    return np.clip(a, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce one O(n)‑memory non‑negative perturbation of ``base``.
    It multiplies a randomly chosen contiguous block by exp(N(0,σ²))
    and, with 35 % probability, mirrors the whole vector.
    """
    cand = base.copy()
    n = cand.size
    # block width roughly between n/200 and n/12
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
