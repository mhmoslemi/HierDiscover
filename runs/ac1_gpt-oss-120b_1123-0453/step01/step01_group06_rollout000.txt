"""
Hybrid stochastic & guided‑gradient optimizer for the step‑function score.

The objective is
    score = 2·n·max(conv(a,a)) / (sum(a))²   (a_i ≥ 0)

The algorithm keeps a single candidate `cur` and a best‑ever `best`.
It alternates two families of moves:

1. **Random exploration** – block‑wise scaling, pairwise swap,
   Gaussian noise, small smoothing.  These moves explore the space
   quickly and provide diversity.

2. **Targeted flattening** – the convolution index `k` with the
   current maximal value is identified; mass is moved from the pair
   of indices that contributes most to that peak to a location that
   contributes almost nothing.  This is a cheap sub‑gradient step that
   directly reduces the worst convolution value.

A simple simulated‑annealing acceptance rule (`exp(-(new‑old)/T)`)
allows occasional uphill steps, helping the search escape local minima.
Only candidates that improve the exact `evaluate_sequence` value are
promoted to `best`.  The fast FFT‑based estimator (`_score_estimate`)
filters the vast majority of proposals, keeping the runtime well
within the 1000 s budget.

The code works for any initial sequence (the global `height_sequence_1`
is used when present) and returns the best list of heights found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of seq with itself using FFT (real input)."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the self‑convolution."""
    return float(_conv_full(seq).max())


def _score_estimate(seq: np.ndarray) -> float:
    """Fast proxy for evaluate_sequence (uses FFT, no clipping)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Random perturbations (the “exploration” moves)
# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative scaling of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # draw a log‑scale factor; sigma itself is random for extra diversity
    sigma = float(0.05 + 0.20 * rng.random())
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:stop] *= factor
    # optional mirroring
    if rng.random() < 0.30:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from i → j."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)  # ≤20 %
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add small zero‑mean Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)


def _smooth_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Light smoothing via a 3‑point moving average (preserves total mass)."""
    # keep the array length, treat boundaries with half‑weight
    kernel = np.array([0.25, 0.5, 0.25])
    smooth = np.convolve(seq, kernel, mode='same')
    return np.clip(smooth, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Targeted gradient‑like move (the “flatten” move)
# ----------------------------------------------------------------------
def _flatten_targeted(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the contribution of the currently maximal convolution index.
    Moves mass from the most offending pair to a location that does not
    contribute (or contributes very little) to that same index.
    """
    cand = seq.copy()
    n = cand.size
    # full convolution to locate the worst index
    conv = _conv_full(cand)
    k = int(conv.argmax())

    # indices that actually appear in conv[k]
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k - idx

    # contribution of each pair (i, partner[i])
    contrib = cand[idx] * cand[partner]

    # source index = one side of the pair with the largest product
    src_local = int(idx[np.argmax(contrib)])
    src_partner = int(partner[np.argmax(contrib)])

    # choose a destination far from the offending pair.
    # Prefer indices whose partner value for shift k is tiny.
    forbidden = {src_local, src_partner}
    all_idx = np.arange(n)
    dst_candidates = np.setdiff1d(all_idx, list(forbidden))
    if dst_candidates.size == 0:
        return cand

    # compute partner values for the destination candidates (if they were
    # to be used in the same shift k they would pair with k-dst)
    dst_partner = k - dst_candidates
    # keep only those with a valid partner index
    mask = (dst_partner >= 0) & (dst_partner < n)
    safe_dst = dst_candidates[mask] if mask.any() else dst_candidates
    dst = int(rng.choice(safe_dst))

    # move a modest fraction of the source mass
    frac = float(0.05 + 0.25 * rng.random())   # 5 %–30 %
    delta = cand[src_local] * frac
    cand[src_local] = max(0.0, cand[src_local] - delta)
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Stochastic‑gradient hybrid optimizer.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  The caller will subtract a small safety
        margin; we stop a few seconds early to guarantee a return.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for finalisation/printing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---------------------------------------------------------------
    # 0️⃣ Initialise – use the global warm‑start if present
    # ---------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_seq = None

    if base_seq is None or base_seq.ndim != 1 or base_seq.size == 0:
        # fallback: modest length, roughly 4000 items (same order as before)
        n_fallback = 4800
        base_seq = np.ones(n_fallback, dtype=np.float64)

    # ensure validity
    base_seq = np.clip(base_seq, 0.0, 1000.0)

    # Normalise to a convenient total sum (the exact value is irrelevant)
    target_sum = float(base_seq.sum())
    if target_sum < 0.01:
        target_sum = 1.0
        base_seq = np.ones_like(base_seq)

    # ---------------------------------------------------------------
    # 1️⃣ Warm‑start with a few analytic families (quick scans)
    # ---------------------------------------------------------------
    n = base_seq.size
    best_seq = base_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] start score = {best_score:.6f}")

    # try power‑law families a_i = (i+1)^{-p}
    for p in np.linspace(0.30, 0.70, 9):
        idx = np.arange(n, dtype=np.float64) + 1.0
        cand = np.power(idx, -p)
        cand = np.clip(cand, 0.0, 1000.0)
        # keep the same total mass as the current best
        cand *= (target_sum / cand.sum())
        est = _score_estimate(cand)
        if est < best_score:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_seq = cand
                best_score = exact
                print(f"[warm‑start] p={p:.3f} → score {best_score:.6f}")

    # ---------------------------------------------------------------
    # 2️⃣ Main optimisation loop
    # ---------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_est = _score_estimate(cur_seq)

    # simulated‑annealing schedule
    T0 = best_score * 0.05  # initial temperature (5 % of current score)
    it = 0
    report_interval = 2000

    move_funcs = [
        _make_block_perturbation,
        _perturb_swap,
        _perturb_noise,
        _smooth_move,
        _flatten_targeted,
    ]

    while time.monotonic() < deadline:
        # linear cooling
        elapsed = time.monotonic() + (budget_s - deadline) - (budget_s - 5.0)
        T = max(0.0, T0 * (1.0 - (time.monotonic() - (deadline - budget_s + 5.0)) / (budget_s - 5.0)))
        # pick a move
        move = rng.choice(move_funcs)
        candidate = move(cur_seq, rng)

        # re‑scale to keep the total sum (scaling does not affect the true score)
        s = float(candidate.sum())
        if s < 0.01:
            continue
        candidate *= (target_sum / s)

        est = _score_estimate(candidate)

        # acceptance test (simulated annealing)
        accept = False
        if est <= cur_est:
            accept = True
        else:
            # probability to accept a worse move
            if T > 0.0:
                prob = math.exp(-(est - cur_est) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_seq = candidate
            cur_est = est

            # check against global best (exact evaluation only when promising)
            if est < best_score - 1e-12:
                exact = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best_seq = candidate.copy()
                    best_score = exact
                    print(f"[{it:07d}] NEW BEST  score={best_score:.6f}")

        it += 1
        if it % report_interval == 0:
            print(f"[{it:07d}] cur_est={cur_est:.6f}  best={best_score:.6f}")

    # ---------------------------------------------------------------
    # 3️⃣ Return the best sequence found
    # ---------------------------------------------------------------
    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    return [float(x) for x in best_seq]
