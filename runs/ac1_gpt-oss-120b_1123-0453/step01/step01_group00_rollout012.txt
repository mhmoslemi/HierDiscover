def fast_score(arr):
    s = np.sum(arr)
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    maxb = np.max(conv)
    return 2.0 * len(arr) * maxb / (s * s)

def conv_matrix(f):
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # For i from i_start to i_end, column = k - i
        # set entry A[k, k-i] = f[i]
        # Equivalent: for i range...
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A

def lp_improve(f, target_sum, t_grid, deadline):
    import cvxpy as cp
    n = len(f)
    best_f = f.copy()
    best_sc = fast_score(best_f)
    # Continue iteration until time runs out or stall
    while time.monotonic() < deadline:
        # Compute current max convolution value M
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        # Build A
        A = conv_matrix(f)
        # Define LP variable
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000)
            except Exception:
                break
        if g.value is None:
            break
        g0 = np.maximum(g.value, 0.0)
        sum_g = np.sum(g0)
        if sum_g <= 1e-12:
            break
        # Rescale to target sum
        g_rescaled = g0 * (target_sum / sum_g)
        # Evaluate candidates for each t
        cur_sc = fast_score(f)
        improved = False
        best_local_f = f
        best_local_sc = cur_sc
        best_t = 0.0
        for t in t_grid:
            f_cand = (1.0 - t) * f + t * g_rescaled
            # Clip (precision)
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            sc = fast_score(f_cand)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local_f = f_cand

"""
Propose a low‑value step sequence using an extended LP refinement loop
followed by a simple hill‑climbing phase.

The algorithm works with a fixed ℓ¹‑norm (∑a_i = √(2·n)).  For such a
normalisation the evaluation metric equals the worst auto‑convolution
entry M = max (a*a).  An LP maximises the ℓ¹‑mass of a candidate g under
the linear constraints (f*g)_k ≤ M, guaranteeing that the cross‑terms
between the current sequence f and the new one g stay bounded.
After solving the LP we line‑search over several mixing factors t∈[0,1]
to obtain the best convex combination (1‑t)·f + t·g.
Repeating the step many times monotonically reduces M.
When the time budget is almost exhausted a cheap random‑swap hill‑climb
finetunes the best sequence found.
"""

import time
import math
import numpy as np


def fast_score(arr: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    s = np.sum(arr)
    if s < 0.01:
        return float("inf")
    conv = np.convolve(arr, arr)
    maxb = np.max(conv)
    return 2.0 * len(arr) * maxb / (s * s)


def conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build the (2n‑1)×n convolution matrix A with (A @ g)[k] = (f*g)[k]."""
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def lp_improve(f: np.ndarray,
               target_sum: float,
               t_grid,
               deadline: float,
               max_iter: int = 200) -> tuple[np.ndarray, float]:
    """Repeated LP‑based improvement of a sequence f."""
    try:
        import cvxpy as cp
    except Exception as e:          # pragma: no cover
        print("[LP] cvxpy not available:", e)
        return f, fast_score(f)

    n = len(f)
    best_f = f.copy()
    best_sc = fast_score(best_f)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current worst auto‑convolution value
        M = float(np.max(np.convolve(f, f)))

        # linear constraints (f * g)_k ≤ M
        A = conv_matrix(f)

        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:  # fallback to another solver
            try:
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
            except Exception:
                break

        if g.value is None:
            break

        g0 = np.maximum(g.value, 0.0)
        sum_g = np.sum(g0)
        if sum_g <= 1e-12:
            break

        # Rescale g so that it has the same ℓ¹‑norm as f (target_sum)
        g_rescaled = g0 * (target_sum / sum_g)

        # Test a few mixing factors
        cur_sc = fast_score(f)
        best_local_f = f
        best_local_sc = cur_sc
        best_t = 0.0

        for t in t_grid:
            cand = (1.0 - t) * f + t * g_rescaled
            cand = np.clip(cand, 0.0, 1000.0)
            sc = fast_score(cand)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local_f = cand
                best_t = t

        # keep the best mix for the next iteration
        if best_t > 0.0:
            f = best_local_f

        # global best bookkeeping
        if best_local_sc < best_sc:
            best_sc = best_local_sc
            best_f = best_local_f.copy()
            print(f"[LP] iter={it:3d} best={best_sc:.6f} t={best_t:.2f}")

    return best_f, best_sc


def hill_climb(start_f: np.ndarray,
               best_f: np.ndarray,
               best_sc: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """Simple random‑swap hill climbing that preserves the ℓ¹‑norm."""
    n = len(start_f)
    rng = np.random.default_rng()
    cur_f = start_f.copy()
    cur_sc = fast_score(cur_f)

    # initial step size (a few percent of the average entry)
    step = max(0.01, np.mean(cur_f) * 0.5)
    no_improve = 0

    while time.time() < deadline and no_improve < 2000:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if cur_f[i] < 1e-12:
            continue

        max_delta = min(cur_f[i], step, 1000.0 - cur_f[j])
        if max_delta <= 0.0:
            continue

        delta = rng.random() * max_delta
        cand = cur_f.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        sc = fast_score(cand)
        if sc < cur_sc - 1e-9:
            cur_f = cand
            cur_sc = sc
            no_improve = 0
            # update the global best if we beat it
            if sc < best_sc:
                best_sc = sc
                best_f = cand.copy()
                print(f"[HC] new best score {best_sc:.6f}")
            # slowly increase step size after a success
            step = min(step * 1.05, 10.0)
        else:
            no_improve += 1
            # shrink step size slowly during failures
            step = max(step * 0.995, 1e-6)

    return best_f, best_sc

    """

    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, float(budget_s) - 5.0)   # keep a small safety margin

    # ------------------------------------------------------------------
    # Starting point – use the previously found construction if available
    init = globals().get("height_sequence_1")

    n = len(f)

    target_sum = math.sqrt(2.0 * n)
