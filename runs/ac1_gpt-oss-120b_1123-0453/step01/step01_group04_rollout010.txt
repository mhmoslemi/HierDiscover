def project_simplex(v, z=1):
    # v: numpy array, project onto {x >=0, sum x = z}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑value step‑function using a smooth‑max gradient descent.

Key ideas
---------

1.  The objective of ``evaluate_sequence`` is
        2·n·max(conv(a,a)) / (∑a)² .
    For a fixed length ``n`` the denominator is maximised when we keep the
    ℓ₁‑norm of the vector at the normalising value √(2·n).  Hence the problem
    reduces to *minimising the maximum entry of the self‑convolution* while
    keeping the sum fixed.

2.  The maximum of a vector can be approximated by a smooth “soft‑max”:
        s(x) = log(∑ exp(τ·x)) / τ .
    Its gradient w.r.t. a component aᵢ is
        ∂s/∂aᵢ = 2·∑ⱼ aⱼ·w_{i+j} ,
    where wₖ ∝ exp(τ·(conv(a,a)ₖ−max(conv))) are non‑negative weights that
    concentrate on the indices that currently attain (or are close to) the
    maximum.  This gives a cheap, dense gradient that points away from the
    current peaks.

3.  After taking a gradient step we project the vector onto the simplex
    `{x ≥ 0, ∑x = √(2·n)}` – a standard Euclidean projection that preserves the
    required normalisation while keeping all entries non‑negative.

4.  The temperature τ is slowly increased (focuses more on the true max) and
    the learning rate is adapted: increase when a step succeeds, decrease when
    it does not.  Random block perturbations and occasional restarts from a
    fresh power‑law keep the search from getting stuck.

The whole routine respects the supplied time‑budget, periodically prints
progress, and finally returns the best sequence found (as a plain ``list`` of
floats).  All numbers are clipped to the range required by ``evaluate_sequence``.
"""

import time

def make_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law a[i] = (i+1)⁻ᵅ, clipped to the allowed range."""
    idx = np.arange(n, dtype=np.float64)
    a = np.power(idx + 1.0, -alpha)
    return np.clip(a, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a multiplicative log‑normal perturbation to a random contiguous block
    and (with 35 % probability) symmetrise the whole vector.
    """
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        { x | x_i ≥ 0,  Σ x_i = z }.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)  # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = ind[cond].max()
        theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a, target_sum)
            last_improve = time.monotonic()
            tau = 0.5
            lr = 0.05
            restart_score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(restart_score) and restart_score < best_score:
                best_score = restart_score
                best_seq = a.copy()
                print(f"[{time.monotonic():.1f}] restart improved to {best_score:.6f}")

        # ---- 7. early exit if the target is already met ---------------
        if best_score < 1.503:
            print(f"[{time.monotonic():.1f}] target bound reached – stopping early.")
            break

    # final sanity clipping (evaluate_sequence also clips)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[{time.monotonic():.1f}] search finished – best score = {best_score:.6f} "
          f"after {iteration} iterations.")
    return [float(x) for x in best_seq]
