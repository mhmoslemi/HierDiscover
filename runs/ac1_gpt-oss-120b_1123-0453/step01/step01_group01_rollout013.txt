def adam_optimize(a, step, iterations, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    m = zeros_like(a)
    v = zeros_like(a)
    best = a.copy(); best_score = eval(a)
    for t in range(1, iterations+1):
        conv = self_convolution(a)
        s_max = np.argmax(conv)
        # compute subgradient
        idx = np.arange(n)
        mate = s_max - idx
        mask = (mate>=0)&(mate<n)
        g = np.zeros_like(a)
        g[mask] = 2*a[mate[mask]]
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = proj_simplex(a)
        # optionally add small noise
        if t % 100 == 0:
            a += rng.normal(scale=1e-3, size=n)
            a = proj_simplex(a)
        score = eval(a)
        if score < best_score:
            best = a.copy()
            best_score = score
    return best, best_score

"""
Propose a low‑score step‑function.

The objective is   score = 2 * n * max(conv(a,a)) / (sum(a)**2)
with a_i ≥ 0.  Because the score is invariant under scaling,
we keep the vector on the probability simplex (sum = 1).

Strategy
--------
1.  Determine the problem size `n`.  If a previous construction
    (`height_sequence_1`) is available we re‑use its length,
    otherwise we fall back to n = 6000.

2.  Initialise several families of candidate distributions:
      * The classical “arcsine’’ (central binomial) distribution.
      * A parametric Beta‑family   a_i ∝ x_i^{α‑1}(1‑x_i)^{β‑1},
        where x_i = (i+0.5)/n.  We sweep α,β in a modest grid
        (≈100 points) and also mix each shape with a tiny uniform
        background (ε∈{0,0.05,…,0.2}).  All candidates are
        normalised to sum = 1 and the best score is retained.

3.  Refine the best candidate with a simple projected multiplicative
    gradient descent (sub‑gradient of the maximal convolution entry).
    The update is
        a ← a * exp(‑η * g)   (g_i = 2·a_{s_max‑i})
    followed by an L1‑projection onto the simplex.
    Adam‑style moments are *not* needed – the multiplicative rule
    already respects the non‑negativity constraint.
    Small isotropic noise and occasional block‑perturbations (as in
    the reference code) are injected to escape shallow minima.

4.  The loop runs until the time budget (≈ 1000 s) expires.
    The best sequence found is returned as a plain list of floats.

The algorithm is entirely self‑contained, uses only NumPy, and
does not rely on any external files or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0, sum w_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself via FFT."""
    n = x.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv


def evaluate_score(a: np.ndarray) -> float:
    """
    Fast internal evaluation of the objective for a vector already
    normalised to sum = 1.  The true evaluator (`evaluate_sequence`)
    is called only at the very end; here we use the same formula.
    """
    n = a.shape[0]
    max_conv = np.max(self_convolution(a))
    return 2.0 * n * max_conv


def arcsine_sequence(n: int) -> np.ndarray:
    """
    Return the classical “arcsine’’ distribution of length n,
    i.e. a_i ∝ C(2i,i)/4^i, normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑distribution on {0,…,n‑1}:
        a_i ∝ x_i^{α‑1} (1‑x_i)^{β‑1},
    where x_i = (i+0.5)/n  (so 0 < x_i < 1).
    The result is normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # work in log‑space to avoid overflow for α,β < 1
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    # guard against nan/inf caused by extreme parameters
    shape[~np.isfinite(shape)] = 0.0
    total = shape.sum()
    if total == 0.0:
        shape[:] = 1.0 / n
    else:
        shape /= total
    return shape


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) non‑negative perturbation used for diversification.
    A random contiguous block is multiplied by exp(N(0,σ²)).
    Optionally the whole vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    # block width ≈ [n/200 , n/12]
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

        idx = np.arange(n)
        mate = s_max - idx
        mask = (mate >= 0) & (mate < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[mate[mask]]

        # 3) multiplicative update (always stays non‑negative)
        a = a * np.exp(-lr * g)

        # 4) projection back onto the simplex (keeps sum = 1)
        a = proj_simplex(a)

        # 5) small isotropic noise every few iterations (helps escape plateaus)
        if it % 30 == 0:
            a += rng.normal(scale=0.001, size=n)
            a = proj_simplex(a)

        # 6) evaluate current candidate
        cur_score = evaluate_score(a)

        # 7) keep the best solution seen so far
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stagnation = 0
            # optionally shrink learning rate a little after a good step
            lr = max(lr * 0.97, 0.02)
        else:
            stagnation += 1
            # if we have been stuck for a while, restart from the best seen
            if stagnation > 2000:
                a = best.copy()
                stagnation = 0
                lr = 0.5

        # 8) occasional block perturbation for extra exploration
        if it % 500 == 0 and it > 0:
            pert = make_block_perturbation(best, rng)
            pert = pert / pert.sum()
            pert_score = evaluate_score(pert)
            if pert_score < best_score:
                best = pert.copy()
                best_score = pert_score
                a = pert.copy()
                stagnation = 0
                lr = 0.5

        # 9) book‑keeping
        it += 1
        if it % 2000 == 0:
            print(
                f"Iter {it:6d} | best_score = {best_score:.6f} | lr = {lr:.4f}"
            )

    # ------------------------------------------------------------------
    # Return the best sequence found (plain Python floats)
    # ------------------------------------------------------------------
    print(f"Finished after {it} iterations, best_score = {best_score:.6f}")
    return [float(x) for x in best]
