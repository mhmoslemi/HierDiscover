"""
Propose a step‑function (non‑negative heights) with a low
evaluate_sequence() score.

Approach
--------
* Work with the L1‑norm fixed to sqrt(2·n); then the score equals the
  maximum entry of the self‑convolution (max_b).
* Initialise from the best known construction (height_sequence_1) or,
  if unavailable, from the analytic geometric candidate.
* minimise the smoothed maximum ϕ_ε(a)=ε·log∑exp((a*a)[t]/ε) by
  projected (momentum) gradient descent.
  – The gradient of ϕ_ε is a correlation of the soft‑max weights with a.
  – Correlations and convolutions are evaluated by FFT for speed.
* After each step the vector is projected onto the simplex
  {x≥0,∑x = sqrt(2·n)}.
* Learning‑rate, smoothing ε and momentum decay gradually.
* Keep the best vector seen; occasional restarts from the current best
  with random jitter avoid stagnation.
* The routine respects the supplied time budget (≈1000 s) and finally
  returns a plain Python list; the official evaluate_sequence() is called
  once for the final verification.

# ----------------------------------------------------------------------
# Helper: fast convolution using FFT (linear convolution, no wrap‑around)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of 1‑D real arrays via FFT."""
    n = x.size + y.size - 1
    N = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    res = np.fft.irfft(X * Y, N)
    return res[:n]


# ----------------------------------------------------------------------
# Helper: fast version of evaluate_sequence (no type checks)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Exact same value as evaluate_sequence for a valid non‑negative vector."""
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = _fft_convolve(arr, arr)
    maxb = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x≥0, sum(x)=z}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project vector v onto the probability‑simplex scaled by z."""
    if z <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform distribution
        return np.full_like(v, z / len(v))
    rho = ind[cond][-1] - 1            # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic geometric candidate (fallback)
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """a_i = r**i, scaled so that ∑a_i = sum_target (r≈1‑t/n)."""
    # solve e^t = 1 + 2t  →  t≈1.256
    t = 1.0
    for _ in range(20):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    cand = (sum_target / raw_sum) * raw
    return cand

        eps = max(1e-4, eps * eps_decay)

        # ----- occasional random perturbation to escape flat regions -----
        if rng.random() < 0.005:
            jitter = rng.normal(scale=avg * 0.05, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a = _project_simplex(a, sum_target)
            no_improve = 0

        # ----- restart if we have not improved for a while -----
        if no_improve >= max_no_improve:
            a = best + rng.normal(scale=avg * 0.1, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a, sum_target)
            momentum.fill(0.0)
            lr = avg * 0.5
            eps = 0.5
            no_improve = 0
            print(f"[{time.monotonic() - start_time:5.1f}s] restart")

    # --------------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # --------------------------------------------------------------
    final = [float(v) for v in best.tolist()]
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f}")
    return final
