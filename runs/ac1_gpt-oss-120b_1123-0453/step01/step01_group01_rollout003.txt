"""
Propose a candidate step‑function using a hybrid
Simulated‑Annealing + projected sub‑gradient descent.

The algorithm starts from the provided `height_sequence_1`
(if any) or from the analytically optimal “arcsine”
distribution.  It continuously perturbs the current
distribution by several types of moves (local Gaussian
noise, block scaling, mass transfer between two indices,
or a small projected sub‑gradient step).  Each proposal
is projected onto the probability simplex (non‑negative,
sum = 1) and evaluated via the cheap FFT‑based formula

    score = 2·n·max( a * a ),

where `*` denotes linear convolution.  The Metropolis
acceptance rule with a slowly decreasing temperature
allows the search to escape local minima.  The best
sequence seen so far is kept and finally returned.

The whole loop stops after the allotted budget (≈ 1000 s)
so the function always returns a valid list of floats.
"""

import time
import numpy as np

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_sequence(n: int) -> np.ndarray:
    """Return the arcsine distribution of length n (sum = 1)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a

def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself via FFT."""
    n = x.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv

def evaluate_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    conv = self_convolution(a)
    return float(2.0 * a.shape[0] * np.max(conv))

def subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient descent step on the exact
    max‑convolution objective.
    """
    n = a.shape[0]
    conv = self_convolution(a)
    s = int(conv.argmax())
    idx = np.arange(n)
    mate = s - idx
    mask = (mate >= 0) & (mate < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[mate[mask]]
    cand = a - step * g
    # small isotropic jitter to avoid stagnation
    cand += np.random.default_rng().normal(scale=1e-4, size=n)
    cand = np.maximum(cand, 0.0)
    cand = proj_simplex(cand)
    return cand

    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself, via FFT."""
    n = x.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv


def evaluate_score(a: np.ndarray) -> float:
    """Return the objective 2·n·max(conv(a,a)) for a probability vector."""
    conv = self_convolution(a)
    return float(2.0 * a.shape[0] * np.max(conv))


def subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Perform one projected sub‑gradient descent step for the exact
    max‑convolution objective.
    """
    n = a.shape[0]
    conv = self_convolution(a)
    s = int(conv.argmax())
    idx = np.arange(n)
    mate = s - idx
    mask = (mate >= 0) & (mate < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[mate[mask]]
    cand = a - step * g
    # tiny isotropic jitter to avoid getting stuck
    cand += np.random.default_rng().normal(scale=1e-4, size=n)
    cand = np.maximum(cand, 0.0)
    cand = proj_simplex(cand)
    return cand


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The routine runs for roughly `budget_s` seconds, using a hybrid
    simulated‑annealing / projected sub‑gradient descent strategy,
    and finally returns a list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size (length of the step function)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        # fallback: a power‑of‑two size makes the FFT cheap
        n = 8192
        base = None

    # ------------------------------------------------------------------
    # Initialise the current solution (simplex, sum = 1)
    # ------------------------------------------------------------------
    cur = arcsine_sequence(n)                # arcsine start (already on simplex)
    cur_score = evaluate_score(cur)
    best = cur.copy()
    best_score = cur_score

    if base is not None:
        tmp = np.clip(base, 0.0, np.inf)
        s = tmp.sum()
        if s < 1e-12:
            tmp = np.ones_like(tmp)
            s = tmp.sum()
        tmp /= s
        tmp_score = evaluate_score(tmp)
        if tmp_score < cur_score:
            cur = tmp.copy()
            cur_score = tmp_score
        if tmp_score < best_score:
            best = tmp.copy()
            best_score = tmp_score

    # ------------------------------------------------------------------
    # Simulated annealing schedule
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # a few seconds safety margin
    it = 0
    T0 = 0.02          # initial temperature
    T_min = 1e-6       # final temperature
    step0 = 0.20       # initial sub‑gradient step size

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # Choose a perturbation type (5 alternatives)
        # --------------------------------------------------------------
        move_type = rng.integers(0, 5)
        cand = cur.copy()

        if move_type == 0:
            # ----- block scaling (multiplicative bump) -----
            w_low = max(1, n // 200)
            w_high = max(2, n // 12)
            width = int(rng.integers(w_low, w_high + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.18)))
            cand[start:start + width] *= factor

        elif move_type == 1:
            # ----- transfer a fraction of mass between two indices -----
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and cand[i] > 0.0:
                frac = rng.uniform(0.02, 0.30)      # up to 30 % of mass at i
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta

        elif move_type == 2:
            # ----- small isotropic Gaussian noise on the whole vector -----
            cand += rng.normal(scale=0.008, size=n)

        elif move_type == 3:
            # ----- a short projected sub‑gradient descent step ----------
            elapsed = time.monotonic() - start_time
            frac_time = min(1.0, elapsed / (deadline - start_time))
            step = step0 * (1.0 - frac_time) + 1e-4 * frac_time
            cand = subgradient_step(cur, step)

        else:
            # ----- randomised restart to arcsine + jitter ---------------
            cand = arcsine_sequence(n)
            cand += rng.normal(scale=0.04, size=n)

        # --------------------------------------------------------------
        # Project onto the simplex (non‑negative, sum = 1)
        # --------------------------------------------------------------
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s <= 1e-12:
            cand = np.full(n, 1.0 / n)
        else:
            cand /= s      # normalisation (equivalent to projection)

        # --------------------------------------------------------------
        # Evaluate the candidate
        # --------------------------------------------------------------
        cand_score = evaluate_score(cand)

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        if cand_score < cur_score:
            accept = True
        else:
            elapsed = time.monotonic() - start_time
            frac_time = min(1.0, elapsed / (deadline - start_time))
            T = (1.0 - frac_time) * T0 + frac_time * T_min
            prob = np.exp(-(cand_score - cur_score) / T) if T > 0 else 0.0
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score

        # --------------------------------------------------------------
        # Keep the globally best solution
        # --------------------------------------------------------------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            print(f"Iter {it:7d} – new best score: {best_score:.6f}")

        # --------------------------------------------------------------
        # Periodic progress report
        # --------------------------------------------------------------
        if it % 20000 == 0 and it > 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.1f}s] iter {it:7d} – cur {cur_score:.6f} – best {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (plain Python floats)
    # ------------------------------------------------------------------
    print(f"Finished after {it} iterations, best score = {best_score:.6f}")
    return [float(x) for x in best]
