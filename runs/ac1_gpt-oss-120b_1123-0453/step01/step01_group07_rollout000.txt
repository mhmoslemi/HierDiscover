def gradient_step(seq, step, rng):
    n = seq.size
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # tolerance for indices tied
    tol = max_val * 1e-12
    idxs = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(seq)
    for k in idxs:
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        i = np.arange(i_start, i_end+1)
        j = k - i
        grad[i] += 2.0 * seq[j]
    # compute direction
    mean_grad = grad.mean()
    direction = -grad + mean_grad
    # step size may be scaled by a factor from rng?
    new_seq = seq + step * direction
    return new_seq
```

def block_perturb(seq, rng):
    n = seq.size
    lo = max(1, n // 300)  # narrower blocks for fine tuning
    hi = max(2, n // 30)
    width = int(rng.integers(lo, hi+1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    new_seq = seq.copy()
    new_seq[start:start+width] *= factor
    # optional symmetrization
    if rng.random() < 0.07:
        new_seq = 0.5 * (new_seq + new_seq[::-1])
    return new_seq
```

"""
Gradient‑descent + stochastic local search for minimizing ``evaluate_sequence``.

The objective is 2·n·max(conv(a,a)) after normalising the coefficients
to sum to one (the original score is scale‑invariant).  The algorithm
maintains a non‑negative vector on the simplex and repeatedly:

1. Takes a descent step in the direction that reduces the peak of the
   autocorrelation (negative sub‑gradient of the max‑convolution).
2. Occasionally applies a multiplicative perturbation to a random block.
3. Occasionally restarts from a fresh power‑law seed.
4. Occasionally forces symmetry (a ↔ a[::-1]).

All candidates are projected back onto the simplex and the best
sequence seen is returned.  The procedure runs until the time budget
(~1000 s) is exhausted.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def generate_power_sequence(n: int, exponent: float = 0.5,
                           offset: float = 0.0) -> np.ndarray:
    """Return a non‑negative power‑law sequence a_i = (i+offset+1)^(-exponent)."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    n = seq.size
    # Block size adapts to n; we allow a fairly wide range.
    lo = max(1, n // 300)
    hi = max(2, n // 30)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.07)))   # ≈7 % log‑normal change
    new = seq.copy()
    new[start:stop] *= factor
    # Occasionally enforce symmetry (helps flatten the convolution)
    if rng.random() < 0.07:
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)


def gradient_step(seq: np.ndarray, step: float,
                  rng: np.random.Generator) -> np.ndarray:
    """
    Perform one sub‑gradient descent step aimed at lowering the peak of
    the autocorrelation.  The sequence is assumed to be normalised
    (sum == 1).  Returns a *new* (unnormalised) sequence.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # Indices attaining (or almost attaining) the maximum.
    tol = max_val * 1e-12
    max_idxs = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(seq)
    # For each peak index k we add 2*a_{k-i} to grad[i] where defined.
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += 2.0 * seq[j]

    # Project gradient onto the tangent plane of the simplex:
    mean_grad = grad.mean()
    direction = -grad + mean_grad

    # Add a tiny stochastic component to avoid deterministic cycles.
    direction += rng.normal(0.0, 0.001, size=seq.shape)

    new_seq = seq + step * direction
    return np.clip(new_seq, 0.0, 1000.0)


def _score_from_seq(seq: np.ndarray) -> float:
    """
    Compute the exact score used by ``evaluate_sequence`` for a
    normalised (sum == 1) non‑negative sequence.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())

# Main entry point
# ----------------------------------------------------------------------

    Return a list of non‑negative heights that (heuristically) minimises
    ``evaluate_sequence`` within the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # Leave a small safety margin for the harness.
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine the length of the sequence.
    # ------------------------------------------------------------------
    try:

    try:

    except Exception:

    init = np.clip(init, 0.0, 1000.0)
    s = init.sum()
    if s <= 0.0:
        init = np.ones(n, dtype=np.float64) / n
    else:
        init = init / s                      # normalise to sum == 1

    best_seq = init.copy()
    best_score = _score_from_seq(best_seq)
    print(f"[init] length={n}, score={best_score:.9f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    step_size = 0.25                # initial gradient step magnitude

    # ------------------------------------------------------------------
    print(f"[final] best score {best_score:.9f} after {time.monotonic() - start_time:.2f}s")
    return [float(x) for x in best_seq]
