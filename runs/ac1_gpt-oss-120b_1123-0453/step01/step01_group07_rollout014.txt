def fast_score(seq):
    n = len(seq)
    a = np.asarray(seq, dtype=np.float64)
    # Clip not required maybe but ensure non-negative and within bounds.
    a = np.clip(a,0,1000)
    # compute convolution via FFT
    size = 2*n - 1
    # Next power of two for efficiency
    m = 1 << (size - 1).bit_length()
    FA = np.fft.rfft(a, m)
    conv = np.fft.irfft(FA * FA, m)[:size]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a**2)

"""
Improved search for a low‑score step‑function.

* A fast surrogate of ``evaluate_sequence`` is built on an FFT based
  convolution.  It yields the exact same result (up to rounding) but
  is ~100× faster for n≈5k, allowing many more trials inside the time
  budget.

* The search space consists of a truncated power‑law kernel
  a_i = (i+offset+1)^(-exponent) with an optional zero padding on both
  ends.  For exponent≈0.5 this kernel gives the theoretical bound
  π/2≈1.571; by adding a modest padding we can lower the convolution peak
  slightly below π and therefore improve the bound.

* Phase 1 – a coarse grid over exponent, offset and padding – quickly
  finds a good starting vector.

* Phase 2 – hill‑climbing with three mutation operators:
      – block_perturb : multiplicative change of a random contiguous block,
      – global_noise  : tiny log‑normal noise on the whole vector,
      – targeted_perturb : reduces a sub‑block that participates in the
        current maximal convolution entry.
  A simple simulated‑annealing acceptance rule lets the search escape
  shallow local minima.

* Random restarts occasionally inject a fresh power‑law seed.

The algorithm runs until the supplied time budget expires (default ≈ 1000 s)
and returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast surrogate of evaluate_sequence (FFT based convolution)
# ----------------------------------------------------------------------
def _fast_score_and_conv(seq: np.ndarray):
    """Return (score, convolution array) using the same rules as evaluate_sequence."""
    # Clip as the official evaluator does
    seq = np.clip(seq.astype(np.float64), 0.0, 1000.0)
    n = seq.size
    # length of full convolution
    length = 2 * n - 1
    # next power of two – FFT works fastest on these sizes
    m = 1 << ((length - 1).bit_length())
    # real‑FFT → element‑wise product → inverse FFT
    f = np.fft.rfft(seq, m)
    conv = np.fft.irfft(f * f, m)[:length]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf, conv
    score = 2.0 * n * max_b / (sum_a ** 2)
    return float(score), conv


def _fast_score(seq: np.ndarray) -> float:
    """Very cheap score (no convolution array)."""
    seq = np.clip(seq.astype(np.float64), 0.0, 1000.0)
    n = seq.size
    length = 2 * n - 1
    m = 1 << ((length - 1).bit_length())
    f = np.fft.rfft(seq, m)
    conv = np.fft.irfft(f * f, m)[:length]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Generic building blocks
# ----------------------------------------------------------------------
def generate_power_sequence(n: int, exponent: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i = (i + offset + 1) ** (-exponent), clipped to the evaluator limits."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)


def generate_tapered_sequence(
    n: int, exponent: float, offset: float, pad: int
) -> np.ndarray:
    """
    Return a sequence where the first and last ``pad`` entries are zero
    and the centre follows a power‑law kernel.
    """
    interior = n - 2 * pad
    if interior <= 0:
        # fallback – all zeros (invalid for the evaluator)
        return np.zeros(n, dtype=np.float64)
    core = generate_power_sequence(interior, exponent=exponent, offset=offset)
    return np.concatenate([np.zeros(pad), core, np.zeros(pad)]).astype(np.float64)


# ----------------------------------------------------------------------
# Mutation operators
# ----------------------------------------------------------------------
def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous sub‑block."""
    n = seq.size
    lo = max(1, n // 300)          # at least 1 element
    hi = max(2, n // 6)            # up to ~1/6 of the vector
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))   # ≈ 7 % log‑normal factor
    out = seq.copy()
    out[start : start + width] *= factor
    # occasional symmetrisation (helps a little for the almost‑symmetric kernel)
    if rng.random() < 0.1:
        out = 0.5 * (out + out[::-1])
    return np.clip(out, 0.0, 1000.0)


def global_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Very small log‑normal noise applied to the whole vector."""
    noise = rng.normal(0.0, 0.02, size=seq.shape)
    factor = np.exp(noise)
    return np.clip(seq * factor, 0.0, 1000.0)


def targeted_perturb(seq: np.ndarray, conv: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce a sub‑block that participates in the current maximal convolution entry.
    The block is chosen inside the index interval that contributes to the max.
    """
    n = seq.size
    k = int(conv.argmax())                     # shift giving the current max
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)                      # inclusive bound
    length = i_end - i_start + 1
    if length <= 0:
        return seq.copy()

    # pick a sub‑block inside the contributing interval
    max_sub = max(1, length // 3)               # keep it reasonably small
    sub_width = int(rng.integers(1, max_sub + 1))
    sub_start = int(rng.integers(i_start, i_end - sub_width + 2))
    factor = float(np.exp(rng.normal(-0.05, 0.02)))  # slight reduction (≈ 5 %)
    out = seq.copy()
    out[sub_start : sub_start + sub_width] *= factor

    # Occasionally sprinkle a tiny global noise so that the move is not purely
    # deterministic – this helps to escape plateaus.
    if rng.random() < 0.2:
        out = global_noise(out, rng)

    return np.clip(out, 0.0, 1000.0)

    start_time = time.monotonic()
    # keep a small safety margin for the host process
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

        if prev.shape[0] == n:

            best_score = s

    except Exception:
        pass

    # a simple analytic seed (no padding)
    seed_seq = generate_power_sequence(n, exponent=0.5, offset=0.0)
    s = _fast_score(seed_seq)
    if s < best_score:

    # ------------------------------------------------------------------
    exp_grid = np.arange(0.48, 0.531, 0.005)          # 0.48 … 0.53
    off_grid = np.arange(0.0, 2.01, 0.1)              # 0.0 … 2.0

                if time.monotonic() >= deadline:
                    break
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – hill‑climbing with three mutation types and simulated annealing
    # ------------------------------------------------------------------
    if best_seq is None:
