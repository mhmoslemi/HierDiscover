def _conv(seq1, seq2):
    n1 = seq1.size
    n2 = seq2.size
    size = _next_pow_two(n1 + n2 - 1)
    fa = np.fft.rfft(seq1, n=size)
    fb = np.fft.rfft(seq2, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: n1 + n2 - 1]

        subject to Σ_i f_i·g_{k‑i} ≤ M   for all k
                    g_j ≥ 0

   where `M = max(conv(f,f))`.  The LP delivers a new direction `g`
   that does not increase the current maximum convolution.  `g` is rescaled
   to the fixed normalisation Σ a_j = √(2 n) (which makes the objective
   equal to the maximal self‑convolution) and mixed into `f` with a
   diminishing step size `t = 2/(iteration+2)`.  The process repeats until the
   time budget expires, always keeping the best exact sequence found.

If ``cvxpy`` is unavailable the optimiser falls back to the original
stochastic algorithm.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT helpers (top‑level – no closures)
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv(seq1: np.ndarray, seq2: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for n≈few‑thousand)."""
    n1, n2 = int(seq1.size), int(seq2.size)
    size = _next_pow_two(n1 + n2 - 1)
    fa = np.fft.rfft(seq1, n=size)
    fb = np.fft.rfft(seq2, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: n1 + n2 - 1]


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    conv = _conv(seq, seq)
    return float(conv.max())


def _score_estimate(seq: np.ndarray) -> float:
    """Fast estimator of the official metric (uses FFT)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Perturbation moves – identical to the reference implementation
# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)


def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = seq.copy()
    n = cand.size
    conv = _conv(cand, cand)
    k_max = int(conv.argmax())
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]
    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand
    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), np.array(list(forbidden), dtype=int))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))
    delta = cand[src] * float(0.05 + 0.15 * rng.random())
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


_MOVE_FUNCS = [
    _make_block_perturbation,
    _perturb_swap,
    _perturb_noise,
    _flatten_move,
]


# ----------------------------------------------------------------------
# Frank‑Wolfe – LP refinement (requires CVXPY)
# ----------------------------------------------------------------------
def _build_lp_constraints(active_idxs, f, M):
    """
    Produce a list of CVXPY constraints for the active set.
    Each constraint encodes   Σ_i f_i·g_{k‑i} ≤ M .
    """
    import cvxpy as cp

    n = f.size
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_idxs:
        coeff = np.zeros(n, dtype=np.float64)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        idx = np.arange(j_start, j_end + 1, dtype=int)
        coeff[idx] = f[k - idx]                # f_i where i=j, partner=k‑j
        constraints.append(cp.sum(cp.multiply(coeff, g)) <= M)
    return g, constraints


def _solve_lp(active_set, f, M, warm_g=None):
    """
    Solve the LP  maximise Σ g_j  subject to  f∗g ≤ M on the active set.
    Returns the optimal g (as a NumPy array) or None on failure.
    """
    import cvxpy as cp

    n = f.size
    # build optimisation variable
    g_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_set:
        coeff = np.zeros(n, dtype=np.float64)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        idx = np.arange(j_start, j_end + 1, dtype=int)
        coeff[idx] = f[k - idx]
        constraints.append(cp.sum(cp.multiply(coeff, g_var)) <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        # warm‑start improves speed once we have a previous solution
        if warm_g is not None:
            g_var.value = warm_g
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    except Exception as exc:
        print("[LP] solver exception:", exc)
        return None

    if prob.status != cp.OPTIMAL:
        # non‑optimal can happen for ill‑conditioned active sets
        return None
    g_val = g_var.value
    if g_val is None:
        return None
    return np.array(g_val, dtype=np.float64)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine combines a cheap stochastic
    phase with a Frank‑Wolfe LP refinement.  It respects the time budget
    (leaving a few seconds margin) and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the caller usually subtracts a safety margin).

    Returns
    -------
    list[float]
        The best coefficient sequence discovered.
    """
    import numpy as np
    import time
    import math

    # ------------------------------------------------------------------
    # 0️⃣  Setup / warm‑start
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)  # safety margin

    # obtain a starting sequence – the provided global is preferred
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
    except Exception:
        base_seq = np.ones(256, dtype=np.float64)

    # basic sanity checks
    if base_seq.ndim != 1 or base_seq.size == 0 or not np.isfinite(base_seq).all():
        base_seq = np.ones(256, dtype=np.float64)

    base_seq = np.clip(base_seq, 0.0, 1000.0)

    n = int(base_seq.size)
    target_sum = math.sqrt(2.0 * n)  # fixed normalisation, makes objective = max(conv)

    # normalise to the target sum
    if base_seq.sum() == 0:
        base_seq = np.ones(n, dtype=np.float64)
    base_seq = base_seq * (target_sum / base_seq.sum())

    # initial evaluation
    best_seq = base_seq.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = float("inf")
    print(f"[init] baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Analytic warm‑start (power‑law shapes)
    # ------------------------------------------------------------------
    for p in np.linspace(0.30, 0.60, 7):
        idx = np.arange(1, n + 1, dtype=np.float64)
        cand = np.power(idx, -p)
        cand = cand * (target_sum / cand.sum())
        est = _score_estimate(cand)
        if est < best_score:
            exact = float(evaluate_sequence(cand.tolist()))
            if exact < best_score:
                best_seq = cand.copy()
                best_score = exact
                print(f"[warm‑start] p={p:.3f} → new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Fast stochastic local search (first half of budget)
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    it = 0
    report_interval = 5000
    # allocate roughly half of remaining time to stochastic phase
    stoch_deadline = time.monotonic() + 0.45 * (deadline - time.monotonic())

    while time.monotonic() < stoch_deadline:
        move = rng.choice(_MOVE_FUNCS)
        cand = move(cur_seq, rng)
        cand = cand * (target_sum / cand.sum())
        # quick upper bound – cheap because of FFT
        est = _score_estimate(cand)
        if est < best_score:
            exact = float(evaluate_sequence(cand.tolist()))
            if exact < best_score:
                best_seq = cand.copy()
                cur_seq = cand.copy()
                best_score = exact
                print(f"[stoch {it:07d}] improvement → {best_score:.6f}")
        it += 1
        if it % report_interval == 0:
            print(f"[stoch {it:07d}] current best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Frank‑Wolfe LP refinement (remaining time)
    # ------------------------------------------------------------------
    # Try to import cvxpy – if unavailable we fall back to stochastic only.
    try:
        import cvxpy as cp  # noqa: F401
        cvxpy_available = True
    except Exception:
        cvxpy_available = False
        print("[info] CVXPY not available – skipping LP refinement.")

    f = best_seq.copy()
    lp_iter = 0
    warm_g = None  # store previous LP solution for warm‑start

    while cvxpy_available and time.monotonic() < deadline:
        # current maximal self‑convolution value
        conv_f_f = _conv(f, f)
        M = float(conv_f_f.max())
        # initialise active set – indices where conv is within 0.01 % of the max
        eps = max(1e-12, 1e-4 * M)
        active = set(np.where(conv_f_f >= M - eps)[0])

        # ---------- solve LP, expanding active set when violated ----------
        while True:
            g = _solve_lp(active, f, M, warm_g)
            if g is None:
                # LP failed – abort refinement
                break
            # check all constraints (full convolution)
            conv_f_g = _conv(f, g)
            viol = np.where(conv_f_g > M + 1e-9)[0]
            new = [k for k in viol if k not in active]
            if not new or len(active) > 800:  # safeguard against runaway size
                warm_g = g  # keep for next iteration
                break
            active.update(new)

        if g is None:
            # If LP never succeeded, perform a cheap random move as a fallback.
            cand = _make_block_perturbation(f, rng)
            cand = cand * (target_sum / cand.sum())
            exact = float(evaluate_sequence(cand.tolist()))
            if exact < best_score:
                best_seq = cand.copy()
                f = cand.copy()
                best_score = exact
                print(f"[fallback] improvement → {best_score:.6f}")
            continue

        # Scale g to the fixed normalisation
        sum_g = float(g.sum())
        if sum_g < 1e-12:
            # degenerate solution – skip this iteration
            continue
        g = g * (target_sum / sum_g)

        # Frank‑Wolfe step size t = 2/(k+2)  (k starts at 0)
        t = min(1.0, 2.0 / (lp_iter + 2.0))
        f_new = (1.0 - t) * f + t * g
        # re‑normalise (tiny numeric drift)
        f_new = f_new * (target_sum / f_new.sum())

        # evaluate the new candidate exactly
        exact = float(evaluate_sequence(f_new.tolist()))
        if exact < best_score:
            best_seq = f_new.copy()
            best_score = exact
            print(f"[lp {lp_iter:04d}] improvement → {best_score:.6f}")
        else:
            # still move forward – the algorithm is monotone in the bound
            pass
        f = f_new.copy()
        lp_iter += 1
        if lp_iter % 10 == 0:
            print(f"[lp {lp_iter:04d}] current best = {best_score:.6f}")

    print(f"Finished – stochastic iters: {it}, LP iters: {lp_iter}, best score = {best_score:.6f}")
    # Return as plain Python list of floats
    return [float(x) for x in best_seq]
