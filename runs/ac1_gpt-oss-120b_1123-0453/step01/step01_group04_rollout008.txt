"""
Gradient‑descent optimizer for the step‑function bound.

The function `propose_candidate` builds a symmetric non‑negative sequence
`a` (sum is fixed to 1, scaling does not change the objective).  The
objective is

    F(a) = 2·n·max_k (a*a)[k]            (sum(a)=1)

so minimizing the maximal value of the self‑convolution.
A smooth approximation of the max is used:

    L(a) = (1/β)·log Σ_k exp(β·c_k) ,   c = a*a

and the gradient ∇L is obtained analytically as a correlation of the
soft‑max weights with `a`.  The update rule is a simple momentum SGD
with projection onto the probability simplex (non‑negative, sum = 1)
and an explicit symmetrisation step (aᵢ = a_{n‑1‑i}) which is known to be
optimal for this symmetric problem.

All convolutions are performed by FFT (real‑valued) which makes each
iteration O(n log n).  The algorithm runs until the allocated time budget
(≈1000 s) elapses, keeping the best sequence found.  If the bound drops
below the target 1.503 the search stops early.

The returned sequence is a list of Python floats; it is already clipped
to the allowed range [0, 1000] by the `evaluate_sequence` routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of a and b using a pre‑allocated FFT length L."""
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L).real
    # length of the linear convolution is len(a)+len(b)-1
    return conv[: a.size + b.size - 1]


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Numerically stable soft‑max with temperature 1/β."""
    shift = np.max(x)
    ex = np.exp(beta * (x - shift))
    s = np.sum(ex)
    if s == 0:
        # fallback – uniform weights
        return np.full_like(x, 1.0 / x.size)
    return ex / s


def _arcsine_shape(n: int) -> np.ndarray:
    """Continuous arcsine density sampled on a grid, normalised to sum‑1."""
    i = np.arange(n, dtype=np.float64)
    # (i+0.5)*(n-i-0.5) mimics (x·(1‑x)) on a discrete grid
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / a_sum
    return a


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is all zeros – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety – renormalise once more
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------

        cand_obj = 2.0 * n * np.max(conv_cand)

        # ----- accept if strictly better --------------------------------
        if cand_obj < best_obj:
            best_seq = cand
            best_obj = cand_obj
            last_improve = time.monotonic()
            velocity[:] = 0.0                # reset momentum after improvement
        else:
            # keep the current best_seq, continue moving
            pass

        # ----- occasional random jitter when stuck --------------------
        if time.monotonic() - last_improve > 20.0:
            # small Gaussian perturbation + re‑normalisation
            noise = 0.02 * rng.standard_normal(n)
            cand = best_seq + noise
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand)
            cand = 0.5 * (cand + cand[::-1])
            # recompute cheap objective
            fa_cand = np.fft.rfft(cand, n=L)
            conv_full_cand = np.fft.irfft(fa_cand * fa_cand, n=L).real
            cand_obj = 2.0 * n * np.max(conv_full_cand[: 2 * n - 1])
            if cand_obj < best_obj:
                best_seq = cand
                best_obj = cand_obj
                last_improve = time.monotonic()
                velocity[:] = 0.0
                # reset some hyper‑parameters
                lr = 0.1
                beta = 2.0

        # ----- learning‑rate decay -------------------------------------
        lr = max(lr * lr_decay, lr_min)

        # ----- periodic status report ---------------------------------
        if iter_cnt % 200 == 0:
            elapsed = time.monotonic() - (deadline - budget_s)
            print(f"[{elapsed:.1f}s] it={iter_cnt} best_obj={best_obj:.6f} "
                  f"β={beta:.2f} lr={lr:.5f}")

    # ------------------------------------------------------------------
    # Final clean‑up: ensure the sequence respects the evaluator's clips
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Convert to a plain Python list (the harness expects list[float])
    return [float(x) for x in best_seq]
