Hybrid local search with periodic linear programming improvement.

We start from existing best sequence (height_sequence_1) or arcsine.
We iteratively apply cheap stochastic perturbations to escape local minima.
Every few seconds we solve a linear program that maximises the L1 norm
subject to the current max‑convolution bound.  Mixing the LP solution
with the current vector strictly reduces the bound if possible.
The loop stops after the time budget and returns the best sequence found.
"""

def propose_candidate(seed=123, budget_s=950, **kwargs):
    import time, numpy as np
    import math
    from scipy.optimize import linprog
    import scipy.sparse as sp

    # Determine length
    try:
        n = len(height_sequence_1)
        base = np.array(height_sequence_1, dtype=np.float64)
        # enforce non‑negative and clip
        base = np.clip(base, 0.0, 1000.0)
        base = base / np.sum(base)   # normalise L1=1
    except Exception:
        n = 2000
        idx = np.arange(1, n+1, dtype=np.float64)
        base = 1.0/np.sqrt(idx)
        base = base / np.sum(base)

    def fast_score(a):
        nloc = a.shape[0]
        size = 1 << ((2*nloc-1).bit_length())
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa*fa, size)[:2*nloc-1]
        maxb = conv.max()
        s = a.sum()
        return 2.0 * nloc * maxb / (s*s)

    # initial
    cur = base.copy()
    cur_score = fast_score(cur)
    best_seq = cur.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, score={best_score:.6f}")

                cand[i] -= delta
                cand[j] += delta

"""
Hybrid search for a low‑value step‑function sequence.

The algorithm works in two phases:

1.  Start from a non‑negative “arcsine’’ shape (a_i ∝ 1/√(i+1))
    – this is already close to the continuous optimum
    (theoretical limit 2·n·maxconv/(∑a)² → π/2 ≈ 1.5708).

2.  Repeatedly apply cheap stochastic moves (block jitter,
    mass‑transfer, smoothing, mirroring) while keeping the best
    sequence seen.  Every few seconds a small linear‑programming
    sub‑problem is solved:

        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f * g)_k ≤ M   (k = 0…2n‑2)

    where f is the current sequence and M = max(f*f).  The LP
    returns a “more spread’’ vector g; a convex combination
    (1‑t)·f + t·g (renormalised to L1‑norm = 1) is taken as a
    new candidate.  Mixing with the LP solution provably never
    increases the bound and usually reduces it.

The whole search is bounded by a user‑supplied time budget
(default 950 s, leaving a small safety margin).  Progress is
printed and the best sequence found is returned as a plain
Python list of floats.
"""

import math
import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# cheap perturbations – all operate on a *copy* of the current vector
# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal jitter on a random block of `base`."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (always at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring – helps to keep the shape roughly symmetric
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _make_transfer_perturbation(base: np.ndarray,
                                rng: np.random.Generator) -> np.ndarray:
    """Move a small fraction of mass from one index to another."""
    cand = base.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j or cand[i] == 0.0:
        return cand
    delta = 0.02 * cand[i]                # 2 % of the source entry
    cand[i] -= delta
    cand[j] += delta
    return cand


def _make_smooth_perturbation(base: np.ndarray,
                              rng: np.random.Generator) -> np.ndarray:
    """Replace a fraction of the vector by a short moving average."""
    cand = base.copy()
    n = cand.size
    win = int(rng.integers(3, 9))        # window length 3‑8
    kernel = np.ones(win) / win
    smooth = np.convolve(cand, kernel, mode='same')
    cand = 0.7 * cand + 0.3 * smooth
    return cand


def _make_mirror_perturbation(base: np.ndarray) -> np.ndarray:
    """Average the sequence with its reversed copy (enforces symmetry)."""
    return 0.5 * (base + base[::-1])


# ----------------------------------------------------------------------
# fast evaluation (identical to evaluate_sequence for valid inputs)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/(sum(a)²) using FFT (O(n log n))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(a.sum())
    # the denominator is never zero in our search (we always renormalise)
    return 2.0 * n * max_b / (s * s)

# main entry point
# ----------------------------------------------------------------------

                     budget_s: float = 950.0,
                     **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that (hopefully) improves
    on the current best bound.  The function respects the contest rules
    (no I/O, top‑level helpers only) and prints a few progress lines.
    """
    # ------------------------------------------------------------------
    # 1) determine problem size and initialise a decent starting point
    # ------------------------------------------------------------------
    try:
        # use the globally supplied sequence if it exists
        raw = np.array(height_sequence_1, dtype=np.float64)   # pylint: disable=undefined-variable
        if raw.size == 0:

        raw = np.clip(raw, 0.0, 1000.0)
        # Optional: cap the length to keep the LP tractable
        max_len = 2500

            raw = raw[:max_len]
        a = raw / raw.sum()                     # L1‑normalise (scale is irrelevant)
        n = a.size
    except Exception:

        n = 2000
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / np.sqrt(idx)
        a = a / a.sum()
    rng = np.random.default_rng(seed)

    # (the pattern does NOT depend on the actual values of the vector)
    rows_pat = np.add.outer(np.arange(n, dtype=np.int64),

    cols_pat = np.repeat(np.arange(n, dtype=np.int64), n)

    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    next_lp = time.monotonic() + lp_interval
    t_mix = 0.5                        # mixing weight with LP solution
    temperature = best_score * 0.02   # tiny “annealing’’ temperature
    decay = 0.9995

    while time.monotonic() < deadline:

        else:

        # but keeping the L1‑norm = 1 avoids overflow in the LP.
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total == 0.0:

        cand = cand / total

        cand_score = _fast_score(cand)

        # ---------- accept if we improve the *global* best ----------
        if cand_score < best_score - 1e-12:
            official = float(evaluate_sequence(cand.tolist()))
            if official < best_score:
                best_seq = cand.copy()
                best_score = official
                cur = cand.copy()
                cur_score = cand_score
                temperature = best_score * 0.02
                print(f"[improve] new best {best_score:.6f}  t={time.monotonic():.2f}")
                continue

        # ---------- simulated‑annealing style acceptance ----------
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
        else:
            prob = math.exp(-(cand_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score
                temperature *= decay

        # ---------- periodic LP refinement ----------
        now = time.monotonic()
        if now >= next_lp:
            next_lp = now + lp_interval

            # current bound M = max(conv(cur,cur))
            size = 1 << ((2 * n - 1).bit_length())
            fa = np.fft.rfft(cur, size)
            conv_cur = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
            M = float(conv_cur.max())

            # build the sparse matrix A such that (cur * g)_k = (A @ g)_k
            data = np.tile(cur, n)                # repeats of the coefficient vector
            A = sp.csr_matrix((data, (rows_pat, cols_pat)),
                              shape=(2 * n - 1, n))

            # Linear programme:
            #   minimise   - Σ g_i          (i.e. maximise Σ g_i)
            #   subject to A @ g ≤ M, 0 ≤ g_i ≤ 1000
            c_vec = -np.ones(n)
            b_vec = np.full(2 * n - 1, M)
            bounds = [(0.0, 1000.0) for _ in range(n)]

            # give the solver a modest time budget (a fraction of lp_interval)
            try:
                res = linprog(c_vec,
                              A_ub=A,
                              b_ub=b_vec,
                              bounds=bounds,
                              method='highs',
                              options={'time_limit': int(lp_interval * 0.8)})
                if res.success:
                    g = res.x
                    sum_g = g.sum()
                    # if the LP managed to increase the L1‑mass we can mix
                    if sum_g > 1.0 + 1e-8:
                        g_norm = g / sum_g                # L1‑norm = 1
                        cand_lp = (1.0 - t_mix) * cur + t_mix * g_norm
                        cand_lp = np.clip(cand_lp, 0.0, 1000.0)
                        cand_lp = cand_lp / cand_lp.sum()
                        cand_lp_score = _fast_score(cand_lp)
                        if cand_lp_score < best_score - 1e-12:
                            official_lp = float(evaluate_sequence(cand_lp.tolist()))
                            if official_lp < best_score:
                                best_seq = cand_lp.copy()
                                best_score = official_lp
                                cur = cand_lp.copy()
                                cur_score = cand_lp_score
                                temperature = best_score * 0.02
                                print(f"[lp improve] score {best_score:.6f}  t={now:.2f}")
                # else the LP did not find a useful direction – we simply continue
            except Exception as exc:   # pragma: no cover (defensive)
                print("[lp error]", exc)

    # ------------------------------------------------------------------
    # 4) final report
    # ------------------------------------------------------------------
    print(f"[final] best score {best_score:.6f} after {time.monotonic():.2f} s")
    return [float(x) for x in best_seq]
