Gradient‑based simulated‑annealing optimizer for the step‑function bound.

The algorithm works on a fixed‑size sequence a (length n).  At each

1. compute the full self‑convolution a*a by FFT,
2. replace the non‑smooth max‑operator by a soft‑max with a temperature T,

3. obtain an analytical gradient of the surrogate (again using FFT),
4. take a projected gradient step, enforce non‑negativity, symmetry and the
   prescribed L1‑norm ( √(2 n) ), and finally clip to the security limits
   [0, 1000],
5. accept the new point if the true evaluation improves, otherwise accept
   it with a Metropolis probability that anneals with the same temperature T,
6. inject occasional random block perturbations and restart from a fresh
   power‑law when progress stalls.

    n = len(x) + len(y) - 1
    size = _next_pow_two(n)

    fy = rfft(y, size)
    conv = irfft(fx * fy, size)[:n]

    # Normalise to the L1‑norm used by the evaluator ( √(2 n) ).
    target_sum = math.sqrt(2.0 * n)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0
        s = a.sum()
    a = a * (target_sum / s)

    # --------------------------------------------------------------
    # Hyper‑parameters (tuned empirically)
    # --------------------------------------------------------------
    lr = 0.8               # learning rate
    min_lr = 1e-4
    lr_decay = 0.9999

    T = 0.4                # soft‑max temperature
    min_T = 1e-3
    T_decay = 0.9995

    best_seq = a.copy()
    # direct evaluation (the routine does the same clipping internally)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    iter_cnt = 0
    last_improve = time.monotonic()

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        iter_cnt += 1

        # ---- full self‑convolution (FFT) -------------------------
        conv = _conv_fft(a, a)                     # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)   # equals evaluate_sequence when sum = √(2n)

        # ---- keep the global best --------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.monotonic()
            # tighten a bit after a win
            lr = max(min_lr, lr * 0.99)
            T = max(min_T, T * 0.99)

        # ---- early exit if target reached -------------------------
        if best_score < 1.503:
            print(f"[{time.monotonic():.1f}] target reached after {iter_cnt} iters (best={best_score:.6f})")
            break

        # ---- soft‑max approximation of the max --------------------
        shift = max_conv
        # exponent may under‑/overflow – clamp
        exp_arg = (conv - shift) / T
        exp_arg = np.clip(exp_arg, -700, 0)       # np.exp(-700) ≈ 5e‑305
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        B = np.dot(w, conv)                       # weighted average ≈ smoothed peak
        factor = 1.0 + (conv - B) / T
        w_factor = w * factor

        # ---- gradient of the smoothed objective (FFT) ----------
        conv2 = _conv_fft(w_factor, a)             # length 3n‑2
        grad_part = 2.0 * conv2[n - 1 : n - 1 + n]  # slice belonging to indices 0…n‑1

        S = a.sum()
        grad = (2.0 * n / (S * S)) * grad_part - (4.0 * n * B) / (S * S * S)

        # ---- projected gradient step ------------------------------
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)

        # optional symmetry (helps to keep the large peak central)
        a_new = 0.5 * (a_new + a_new[::-1])

        # renormalise to the prescribed L1‑norm
        s_new = a_new.sum()
        if s_new == 0.0:          # extremely unlikely – fallback to current point
            a_new = a.copy()
            s_new = a_new.sum()
        a_new = a_new * (target_sum / s_new)

        # enforce the security clip
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        # ---- accept / reject (Metropolis) ------------------------
        new_score = float(evaluate_sequence(a_new.tolist()))
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # simulated‑annealing acceptance
            prob = math.exp((cur_score - new_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
        else:
            # occasional pull towards the best‑known point
            if rng.random() < 0.01:
                a = 0.5 * (a + best_seq)

        # ---- occasional random block perturbation -----------------
        if rng.random() < 0.02:
            cand = make_block_perturbation(a, rng)
            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                last_improve = time.monotonic()
                print(f"[{time.monotonic():.1f}] found better through block perturbation ({best_score:.6f})")

        # ---- stagnation handling – restart ------------------------
        if time.monotonic() - last_improve > 30.0:
            alpha = rng.uniform(0.45, 0.55)
            a = make_power_law(n, alpha)
            a = a * (target_sum / a.sum())
            lr = 0.8
            T = 0.4
            last_improve = time.monotonic()
            print(f"[{time.monotonic():.1f}] restart with power‑law α={alpha:.4f}")

        # ---- decay hyper‑parameters --------------------------------
        lr = max(min_lr, lr * lr_decay)
        T = max(min_T, T * T_decay)

        # ---- progress report ---------------------------------------
        if iter_cnt % 200 == 0:
            print(f"[{time.monotonic():.1f}] iter={iter_cnt} cur={cur_score:.6f} best={best_score:.6f} lr={lr:.5f} T={T:.5f}")

    # final sanity clip (evaluate_sequence already clips)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Search finished – best score = {best_score:.6f} after {iter_cnt} iterations.")
    return [float(v) for v in best_seq]
