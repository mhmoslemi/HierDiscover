            if sc < best_score:

                best_seq = seq.copy()
    # Stage 0.2: search small p variations around 0.5

                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()

    max_iters = 20000

        conv = np.convolve(current, current)

"""
Propose a near‑optimal step‑function for the evaluation metric.

The algorithm works in three stages:

1. **Analytic family search** – a fine 2‑D grid search over the family
   a_i = (i+α)^‑p (p≈0.5) to obtain a high‑quality starting vector.
2. **Aggressive flattening** – a custom hill‑climbing loop that moves
   mass from indices that contribute most to the current maximal
   convolution entry to several low‑contributing indices.  Contributions
   are recomputed each iteration, and the move is accepted only if the
   score improves.
3. **Noise & random block perturbations** – occasional jitter and
   block‑multiplicative perturbations help escape plateaus.  The best
   sequence found before the time budget expires is returned.

All operations respect the evaluator’s constraints (non‑negative,
clipped to [0,1000]) and run within the provided 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for internal use.
    It clips the sequence to the allowed range, computes the full
    auto‑convolution via ``np.convolve`` and returns the objective.
    """
    a = np.clip(seq, 0.0, 1000.0)
    n = a.size
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    s = np.sum(a)
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def make_block_perturbation(base: np.ndarray,
                            rng: np.random.Generator) -> np.ndarray:
    """
    Produce a cheap O(n) non‑negative perturbation.
    A random contiguous block is multiplied by a log‑normal factor,
    optionally mirrored, and finally clipped to the allowed interval.
    """
    cand = base.copy()
    n = cand.size
    if n == 0:
        return cand
    # block width between 0.5 % and 15 % of the vector length
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative log‑normal perturbation
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (helps to escape plateaus)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _flatten_step(seq: np.ndarray,
                  rng: np.random.Generator,
                  max_iters: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing that repeatedly moves a small fraction of
    mass from a few high‑contributing indices (for the current maximal
    convolution entry) to several low‑contributing ones.
    Returns the best sequence found and its score.
    """
    n = seq.size
    i_arr = np.arange(n, dtype=np.int64)
    best_seq = seq.copy()
    best_sc = _score(best_seq)

    # parameters controlling how many donors / receivers we move per step
    K = 5            # number of donors and receivers
    frac_remove = 0.07  # fraction of each donor's mass to relocate

    while time.monotonic() < deadline and max_iters > 0:
        conv = np.convolve(seq, seq)
        k_max = int(np.argmax(conv))

        # contribution of each entry to the maximal convolution entry
        j = k_max - i_arr
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(seq)
        contrib[mask] = seq[i_arr[mask]] * seq[j[mask]]

        # pick donors (largest contributions) and receivers (smallest)
        donor_idx = np.argpartition(contrib, -K)[-K:]
        recv_idx = np.argpartition(contrib, K)[:K]

        # compute total mass to be moved
        removal = np.zeros_like(seq)
        total_removed = 0.0
        for d in donor_idx:
            r = seq[d] * frac_remove
            if r > 0.0:
                removal[d] = r
                total_removed += r

        if total_removed == 0.0:
            # fallback to a cheap random block move
            cand = make_block_perturbation(seq, rng)
        else:
            share = total_removed / K
            cand = seq - removal
            cand[recv_idx] += share
            cand = np.clip(cand, 0.0, 1000.0)

        sc = _score(cand)

        # keep improvement
        if sc < best_sc:
            best_sc = sc
            best_seq = cand.copy()
            seq = cand
            # occasional status output
            if max_iters % 1000 == 0:
                print(f"  flatten step improved: score={best_sc:.6f}")
        else:
            # occasionally accept a non‑improving move to escape stalls
            if rng.random() < 0.001:
                seq = cand
        max_iters -= 1

    return best_seq, best_sc

# Main entry point required by the grader

def propose_candidate(seed: int = 42,
                      budget_s: float = 990.0,
                      **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) minimizes
    ``evaluate_sequence``.  The implementation follows the three‑stage
    strategy described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a small safety margin for the final return
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine a reasonable length n.  Use the provided sequence if it
    # looks sane, otherwise fall back to a modest size.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        n = 6000                       # a length that gives a good trade‑off
    idx = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1) Fine grid search over the analytic family a_i = (i+α)^‑p
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # coarse sweep for α with the classic exponent p = 0.5
    alphas = np.linspace(0.01, 2.0, 200)        # fine enough for the target
    p_fixed = 0.5
    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        cand = (idx + alpha) ** (-p_fixed)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # local refinement around the best α found
    if best_seq is not None:
        # find the α that produced the current best (cheap re‑lookup)
        best_alpha = alphas[np.argmin([_score((idx + a) ** (-p_fixed)) for a in alphas])]
        fine_alphas = np.linspace(max(0.0, best_alpha - 0.05),
                                   best_alpha + 0.05, 101)
        for alpha in fine_alphas:
            if time.monotonic() >= deadline:
                break
            cand = (idx + alpha) ** (-p_fixed)
            sc = _score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    # a small exploration of the exponent p around 0.5
    if time.monotonic() < deadline:
        p_vals = np.linspace(0.45, 0.55, 41)
        for p in p_vals:
            for alpha in np.linspace(best_alpha - 0.02,
                                     best_alpha + 0.02, 41):
                if time.monotonic() >= deadline:
                    break
                cand = (idx + alpha) ** (-p)
                sc = _score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()

    if best_seq is None:               # safety fallback (should never happen)
        best_seq = np.ones(n, dtype=np.float64)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 2) Aggressive flattening (batch hill‑climbing)
    # ------------------------------------------------------------------
    max_iters_flatten = 20000
    best_seq, best_score = _flatten_step(best_seq, rng,
                                         max_iters_flatten, deadline)

    # ------------------------------------------------------------------
    # 3) Basin‑hopping style random jitter + occasional block moves
    # ------------------------------------------------------------------
    # small zero‑mean Gaussian noise applied to the best vector
    jitter_iters = 5000
    for _ in range(jitter_iters):
        if time.monotonic() >= deadline:
            break
        noise = rng.normal(0.0, 0.02, size=n)          # modest amplitude
        noise -= noise.mean()                         # keep total sum roughly stable
        cand = best_seq + noise
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"  jitter improved: score={best_score:.6f}")

        # occasional larger block perturbation
        if rng.random() < 0.02:
            block_cand = make_block_perturbation(best_seq, rng)
            scb = _score(block_cand)
            if scb < best_score:
                best_score = scb
                best_seq = block_cand.copy()
                print(f"  block perturbation improved: score={best_score:.6f}")

    # a final sanity symmetrisation (often reduces the peak a bit)
    sym = 0.5 * (best_seq + best_seq[::-1])
    sym_sc = _score(sym)
    if sym_sc < best_score:
        best_score = sym_sc
        best_seq = sym

    print(f"Finished search – best score achieved: {best_score:.6f}")

    # Return a plain Python list as required by the grader.
    return [float(x) for x in best_seq]
