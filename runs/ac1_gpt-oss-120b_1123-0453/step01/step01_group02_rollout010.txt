"""
Propose a near‑optimal step‑function for the evaluation metric.

Algorithm
---------
1) Initialise from the supplied `height_sequence_1` (or a modest default).
2) Perform a two‑stage global search:
   * a coarse grid over the classical power family a_i = (i+α)^‑p.
   * a large random search over an extended 4‑parameter family
        a_i = (i+α)^‑p * (1 + λ·(i/(n‑1))^η) ,
     where λ (tail‑boost) and η (shape of the boost) may be zero.
   The random search explores the parameter space densely inside the
   remaining time budget and keeps the best candidate.
3) Refine the best candidate by a deterministic hill‑climbing step:
   repeatedly identify the index `k_max` at which the auto‑convolution
   attains its maximum.  Move a small fraction of mass from the element
   that contributes most to that peak to the element that contributes
   least.  Accept the move only if the score improves; occasionally
   accept a random block perturbation to escape plateaus.
4) Return the best sequence encountered as a plain Python list.

The whole procedure respects the 1000 s time limit, prints progress,
and uses only NumPy (no external solvers).
"""

import numpy as np
import time
import math

# The evaluator is provided by the execution environment.
# from __main__ import evaluate_sequence   # (already available)


def _make_block_perturbation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``seq``.
    The perturbation multiplies a random contiguous block by a log‑normal factor
    and optionally symmetrises the vector.  Result is clipped to [0, 1000].
    """
    cand = seq.copy()
    n = cand.size
    # choose a block width: from 0.5% to 15% of the vector length
    wmin = max(1, n // 200)
    wmax = max(2, n // 12)
    width = int(rng.integers(wmin, wmax + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative log‑normal perturbation
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring – helps to escape symmetric plateaus
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _generate_sequence(p: float, alpha: float, lam: float, eta: float,
                       idx: np.ndarray, n: int) -> np.ndarray:
    """Generate a candidate sequence for the given parameters."""
    base = (idx + alpha) ** (-p)
    if lam != 0.0:
        factor = 1.0 + lam * (idx / (n - 1)) ** eta
        seq = base * factor
    else:
        seq = base
    return np.clip(seq, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Generate a step‑function that minimises ``evaluate_sequence``.

    The routine combines a refined grid/random search over a 4‑parameter
    family and a deterministic hill‑climbing that moves mass from high‑
    to low‑contributing indices of the current maximal convolution entry.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine the working length n.
    # ----------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0 or not np.isfinite(base_seq).all():
            raise ValueError
    except Exception:
        # fallback to a modest size if the supplied sequence is unsuitable
        base_seq = np.ones(256, dtype=np.float64)

    n = base_seq.size
    idx = np.arange(n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 1) Coarse grid search over the pure power family (α, p)
    # ----------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # reserve a few seconds

    print(f"[INIT] Working length n = {n}")

    p_grid = np.linspace(0.35, 0.65, 61)          # coarse p
    alpha_grid = np.linspace(0.01, 5.0, 101)     # coarse α

    for p in p_grid:
        for alpha in alpha_grid:
            if time.monotonic() >= deadline:
                break
            cand = (idx + alpha) ** (-p)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:

                best_seq = cand.copy()
                print(f"[GRID] p={p:.4f} α={alpha:.4f} → score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2) Randomised search over the extended 4‑parameter family
    # ----------------------------------------------------------------------
    # Parameter ranges – chosen generously but bounded.
    p_low, p_high = max(0.01, best_seq is not None and (best_seq[0] and 0.4) or 0.4), 0.6
    # We'll centre the random search around the best pure‑power parameters.
    # If we never found a pure‑power candidate (should not happen) fall back.
    if best_seq is None:

    else:

        half = min(500, n // 2)
        x = np.log(idx[:half] + 1e-12)
        y = np.log(best_seq[:half] + 1e-12)
        # fit y = -p * x + const   (ignore offset α for the centre)
        p_center = -np.polyfit(x, y, 1)[0]

    # Ranges for the random search
    p_range = (0.45, 0.55)
    alpha_range = (0.01, 2.0)

    eta_range = (0.1, 3.0)          # shape of the boost

        alpha = float(rng.uniform(alpha_range[0], alpha_range[1]))

    print(f"[RANDOM DONE] evaluated {iter_count} candidates – best score so far: {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 3) Deterministic hill‑climbing refinement
    # ----------------------------------------------------------------------
    if best_seq is None:
        # safe fallback – should never happen
        best_seq = np.ones(n, dtype=np.float64)

    current = best_seq.copy()
    cur_score = best_score
    improvement_counter = 0
    max_iters = 200_000                     # hard cap; will be cut by time

    while time.monotonic() < deadline and max_iters > 0:

        k_max = int(np.argmax(conv))

        # contribution of each element to the peak at k_max
        i = np.arange(n, dtype=np.int64)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(current)
        contrib[mask] = current[i[mask]] * current[j[mask]]

        donor = int(np.argmax(contrib))
        receiver = int(np.argmin(contrib))

        # avoid degenerate moves
        if donor == receiver or current[donor] <= 0.0:
            candidate = _make_block_perturbation(current, rng)
        else:
            # transfer a small, adaptive fraction of donor's mass
            fraction = 0.05 + 0.10 * rng.random()   # 5–15%
            delta = float(min(current[donor] * fraction, current[donor] * 0.20))
            cand = current.copy()
            cand[donor] -= delta
            cand[receiver] += delta
            candidate = np.clip(cand, 0.0, 1000.0)

        # evaluate
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < cur_score:
            current = candidate
            cur_score = score
            improvement_counter += 1
            if score < best_score:
                best_score = score
                best_seq = candidate.copy()
                print(f"[CLIMB] iter={max_iters:6d} improved to {best_score:.6f}")
        else:
            # with a tiny probability accept a worsening move to escape local minima
            if rng.random() < 0.001:
                current = candidate
                cur_score = score

        max_iters -= 1

    print(f"[FINISH] best score attained: {best_score:.6f}")
    # Return as plain Python list (required by the grader)
    return [float(x) for x in best_seq]
