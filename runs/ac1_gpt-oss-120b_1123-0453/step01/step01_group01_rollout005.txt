"""
Smoothed‑gradient descent with annealed soft‑max and occasional block perturbations.

The score to minimise is 2·n·max(conv(a,a)) / (sum(a)**2).  Because the score
is scale‑invariant we keep the vector on the probability simplex
(∑ a_i = 1, a_i ≥ 0).  At each iteration we

* compute the full self‑convolution via a single FFT,
* build a soft‑max over the convolution values (temperature τ) – this is a
  smooth approximation of the max,
* obtain a gradient g_i = 2·∑_k w_k a_{k‑i} (cross‑correlation of a with the
  soft‑max weights) again with FFTs,
* take a projected gradient step with a decaying step‑size η,
* add a tiny isotropic Gaussian noise,
* occasionally inject a random block perturbation to escape shallow minima,
* keep the best sequence ever seen.

The algorithm starts from the analytically optimal “arcsine” distribution or,
if the global variable `height_sequence_1` is present, from that sequence
(normalised to sum 1).  It stops after the allotted time budget and returns
the best sequence found.
"""

import time
import numpy as np


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_sequence(n: int) -> np.ndarray:
    """
    Return the “arcsine’’ sequence of length n:
        a_i = C(2i,i) / 4^i , then normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) non‑negative perturbation used as a diversification step.
    A random contiguous block is multiplied by a log‑normal factor;
    occasionally the whole vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and ~8 % of the length
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:      # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    Optimise the step‑function bound using a smoothed‑max gradient descent.
    Returns a list of non‑negative heights (the best sequence found before the
    time budget expires).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine problem size (use the provided sequence if it looks usable)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
        use_base = True
    except Exception:
        base = None
        n = 6000               # a comfortable default
        use_base = False

    # ------------------------------------------------------------------
    # Initialise the current solution: best of arcsine / provided sequence
    # ------------------------------------------------------------------
    a_arc = arcsine_sequence(n)
    cur = a_arc.copy()
    cur_score = float(evaluate_sequence(cur.tolist()))

    if use_base:
        tmp = np.maximum(base, 0.0)
        s = tmp.sum()
        if s == 0.0:
            tmp = np.ones_like(tmp)
            s = tmp.sum()
        tmp /= s
        tmp_score = float(evaluate_sequence(tmp.tolist()))
        if tmp_score < cur_score:
            cur = tmp.copy()
            cur_score = tmp_score

    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # FFT configuration (single size used for both convolution and correlation)
    # ------------------------------------------------------------------
    size = 1 << ((3 * n - 1).bit_length())   # next power of two ≥ 3n

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    eta = 0.35                     # initial step‑size
    decay_eta = 0.9999
    tau_factor = 0.20              # initial τ = τ_factor * max(conv)
    decay_tau = 0.99996
    temp0 = best_score * 0.03      # initial temperature for occasional uphill moves
    decay_temp = 0.99995
    no_improve = 0
    max_no_improve = 20000        # trigger a gentle restart

    it = 0
    while time.monotonic() < deadline:
        # ---------- FFT of the current vector ----------
        A = np.fft.rfft(cur, n=size)

        # ---------- self‑convolution (full) ----------
        conv_full = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

        # ---------- soft‑max weights ----------
        max_val = conv_full.max()
        cur_tau = max(max_val * tau_factor, 1e-12)   # avoid division by zero
        # shifted values are ≤ 0, so exp stays ≤ 1

                if rng.random() < prob:
                    cur = cand
                    cur_score = cand_score
            no_improve += 1

        # ---------- record global best ----------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            no_improve = 0
            # tiny reset of step‑size when a big improvement is found
            eta = min(0.5, eta / 0.9)

        # ---------- occasional large random block perturbation ----------
        if it % 2000 == 0 and it > 0:
            pert = make_block_perturbation(best, rng)
            pert = pert / np.sum(pert)          # re‑normalise to simplex
            pert_score = float(evaluate_sequence(pert.tolist()))
            if pert_score < best_score:
                best = pert.copy()
                best_score = pert_score
                cur = pert.copy()
                cur_score = pert_score
                no_improve = 0

        # ---------- restart if stagnating ----------
        if no_improve > max_no_improve:
            # start from the best found so far plus a tiny random jitter
            cur = best + rng.normal(scale=0.001, size=n)
            cur = proj_simplex(cur)
            cur_score = float(evaluate_sequence(cur.tolist()))
            no_improve = 0

        # ---------- decay hyper‑parameters ----------
        eta *= decay_eta
        tau_factor = max(0.01, tau_factor * decay_tau)
        it += 1

        # ---------- progress report ----------
        if it % 5000 == 0:
            print(
                f"Iter {it:6d} | best_score = {best_score:.6f} | "
                f"cur_score = {cur_score:.6f} | eta = {eta:.5f}"
            )

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    print(f"Finished after {it} iterations, best score = {best_score:.6f}")
    return [float(x) for x in best]
