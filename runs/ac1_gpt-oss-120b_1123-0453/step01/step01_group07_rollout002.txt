def convolve_fft(a):
    n = a.shape[0]
    size = 1 << (2*n-1).bit_length()
    # Next power of two
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)

"""
Propose a near‑optimal non‑negative step‑function.

The algorithm works in three stages:

1. **Randomised power‑law search** –  
   a large number of sequences a_i = (i+offset+1)^(-exp) are generated
   for several lengths.  The objective
       2·n·max(a*a) / (sum a)^2
   is evaluated with a fast FFT based convolution (avoids the expensive
   np.convolve used in the official evaluator).  The best candidate so far
   is kept.

2. **Projected gradient descent** –  
   Starting from the best candidate, we repeatedly
   * compute the full convolution,
   * identify the index k where the convolution attains its maximum,
   * form the sub‑gradient of the objective with respect to a_i
        g_i = a_{k-i} – 2·max_b / sum_a   (out‑of‑range positions give 0‑term),
   * take a small step opposite to the gradient, clip to non‑negative
     and renormalise the vector to keep the total sum constant (the
     objective is scale‑invariant).
   The step size is adapted: it is increased after a successful move
   and reduced after a failure.  Occasionally a random block perturbation
   is applied to escape shallow local minima.

3. **Final verification** –  
   The official ``evaluate_sequence`` is called once on the best
   discovered vector and the result is printed.  The vector is returned
   as a plain Python list.

The whole procedure respects a user supplied time budget (default ≈ 1000 s)
and uses only NumPy and the standard library, so it runs on the
restricted execution environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def generate_power_sequence(n: int, exponent: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """
    a_i = (i + offset + 1) ** (-exponent), clipped to the allowed range.
    """
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolve_fft(seq: np.ndarray) -> np.ndarray:
    """
    Fast (real‑valued) convolution of ``seq`` with itself using FFT.
    Returns the full convolution (length 2·len(seq)-1).
    """
    n = seq.size
    m = 2 * n - 1
    size = _next_pow2(m)
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:m]
    return conv


def fast_score(seq: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(conv) / (sum(seq)^2) using FFT.
    Mirrors the behaviour of the supplied ``evaluate_sequence`` but
    is much faster and suitable for the inner optimisation loop.
    """
    n = seq.size
    conv = convolve_fft(seq)
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative perturbation of a randomly chosen contiguous block.
    The block width is a modest fraction of the length.
    """
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.06))          # ≈6 % log‑normal change
    new = seq.copy()
    new[start:stop] *= factor
    if rng.random() < 0.1:                         # occasional symmetrisation
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)


def gradient_step(seq: np.ndarray, target_sum: float, eta: float) -> np.ndarray:
    """
    One projected gradient‑descent step.

    * conv = seq * seq (full convolution, FFT based)
    * kmax = index of the maximal convolution entry
    * g_i = a_{kmax-i} – 2·max_b / sum_a   (zero when the partner index is out of range)
    * new_seq = seq – η·g, projected onto the non‑negative orthant
      and renormalised so that its sum equals ``target_sum`` (scale‑invariant).
    """
    n = seq.size
    conv = convolve_fft(seq)
    max_b = float(conv.max())
    kmax = int(conv.argmax())
    sum_a = float(seq.sum())
    offset = 2.0 * max_b / sum_a

    # Build the gradient vector
    idx = np.arange(n)
    partner = kmax - idx
    valid = (partner >= 0) & (partner < n)
    grad = np.empty_like(seq)
    grad.fill(-offset)
    grad[valid] = seq[partner[valid]] - offset

    # Gradient descent update
    new_seq = seq - eta * grad
    new_seq = np.clip(new_seq, 0.0, None)

    # Renormalise to keep the total sum constant (scale does not matter)
    s = new_seq.sum()
    if s > 0.0:
        new_seq *= target_sum / s
    return new_seq


def random_search(best_seq: np.ndarray,
                  best_score: float,
                  length_options: list[int],
                  start_time: float,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Massive random sampling of power‑law sequences.
    Uses the fast FFT based score and keeps the best discovered vector.
    """
    while time.monotonic() < deadline:
        n = int(rng.choice(length_options))
        exponent = rng.uniform(0.45, 0.55)
        offset = rng.uniform(0.0, 10.0)
        cand = generate_power_sequence(n, exponent, offset)
        score = fast_score(cand)
        if score < best_score:
            best_seq = cand
            best_score = score
            elapsed = time.monotonic() - start_time
            print(f"[random] n={n} exp={exponent:.4f} off={offset:.2f} → score={score:.6f} (t={elapsed:.1f}s)")
    return best_seq, best_score


def refine_sequence(seq: np.ndarray,
                    target_sum: float,
                    start_time: float,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with occasional block perturbations.
    The step size ``eta`` is adaptively increased after a successful move
    and halved after a failed one.
    """
    best_seq = seq.copy()
    best_score = fast_score(best_seq)
    eta = 0.05 * target_sum                # initial step magnitude
    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        it += 1
        cand = gradient_step(best_seq, target_sum, eta)
        cand_score = fast_score(cand)

        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            eta *= 1.02                     # slightly accelerate
            last_improve = it
            if it % 10 == 0:
                elapsed = time.monotonic() - start_time
                print(f"[grad] iter={it}  best={best_score:.6f}  eta={eta:.6f} (t={elapsed:.1f}s)")
        else:
            eta *= 0.5                      # back‑off
            # If step size becomes tiny, re‑initialise a modest value
            if eta < 1e-8:
                eta = 0.05 * target_sum

        # Occasionally try a stochastic block perturbation to jump out of a plateau
        if rng.random() < 0.02:
            cand2 = block_perturb(best_seq, rng)
            score2 = fast_score(cand2)
            if score2 < best_score:
                best_seq = cand2
                best_score = score2
                eta *= 1.01
                last_improve = it
                elapsed = time.monotonic() - start_time
                print(f"[block] iter={it}  new best={best_score:.6f} (t={elapsed:.1f}s)")

        # If we have gone a long time without improvement, restart from a fresh
        # power‑law seed (helps avoid permanent stagnation).
        if it - last_improve > 300 and rng.random() < 0.8:
            n = best_seq.size
            exp = rng.uniform(0.45, 0.55)
            off = rng.uniform(0.0, 10.0)
            cand3 = generate_power_sequence(n, exp, off)
            cand3 = block_perturb(cand3, rng)
            score3 = fast_score(cand3)
            if score3 < best_score:
                best_seq = cand3
                best_score = score3
                eta = 0.05 * target_sum
                last_improve = it
                elapsed = time.monotonic() - start_time
                print(f"[restart] iter={it}  score={best_score:.6f} (t={elapsed:.1f}s)")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low value of the provided
    ``evaluate_sequence`` objective.  The function respects the given
    time budget (default ≈ 1000 s) and returns the best vector found as a
    plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # Keep a small safety margin to ensure we return before the judge kills us
    deadline = start_time + max(0.0, float(budget_s) - 1.5)

    # ------------------------------------------------------------------
    # 1) Determine a reasonable starting length (use previous best if present)
    # ------------------------------------------------------------------
    try:
        n0 = int(len(height_sequence_1))
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        start_score = fast_score(init_seq)
        best_seq = init_seq
        best_score = start_score
        print(f"[init] Using existing height_sequence_1 (n={n0}) → score={best_score:.6f}")
    except Exception:
        n0 = 5000
        init_seq = generate_power_sequence(n0, exponent=0.5, offset=0.0)
        best_seq = init_seq
        best_score = fast_score(init_seq)
        print(f"[init] No previous sequence – start with power law (n={n0}) → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Randomised power‑law search (≈15 % of total budget)
    # ------------------------------------------------------------------
    length_options = [
        max(2000, int(n0 * 0.8)),
        n0,
        min(12000, int(n0 * 1.2)),
        4000,
        6000,
        8000,
        10000,
    ]
    length_options = sorted(set(int(x) for x in length_options if x > 0))
    deadline_random = start_time + 0.15 * (deadline - start_time)
    best_seq, best_score = random_search(best_seq, best_score,
                                         length_options, start_time,
                                         deadline_random, rng)

    # ------------------------------------------------------------------
    # 3) Projected gradient descent (use remaining time)
    # ------------------------------------------------------------------
    target_sum = float(best_seq.sum())
    best_seq, best_score = refine_sequence(best_seq, target_sum,
                                           start_time, deadline, rng)

    # ------------------------------------------------------------------
    # 4) Final official evaluation (once) and return
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] Official evaluation score = {final_score:.6f}")

    # Ensure the output conforms to the required type (list of plain floats)
    return [float(x) for x in best_seq]
