Standard algorithm:

def project_onto_simplex(v):
    """Project vector v onto probability simplex (sum = 1, v >= 0) returning projected vector."""
    # convert to numpy array
    u = np.sort(v)[::-1]  # sort descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑value step‑function using a smooth‑max gradient descent.
The algorithm

1. Builds a symmetric “arcsine” start‑point:
      a_i = 1 / sqrt( (i+½)·(n‑½‑i) )
   and rescales it so that Σ a_i = √(2·n) (the value for which the
   evaluation routine returns the plain max‑convolution).

2. Repeatedly performs a projected gradient step that tries to flatten
   the whole convolution vector:
   * Full convolution c = a * a is obtained with an FFT (O(n log n)).
   * A smooth weight w_k = exp(β·(c_k–max(c))) concentrates on the
     presently largest entries, but still gives a small influence to
     the others.  w is normalised to a probability distribution.
   * The gradient of the smoothed objective is
         ∇ = 2·(w ⋆ a_rev)   where a_rev = a[::-1].
     This is also computed by an FFT.
   * a is moved a short step opposite to the gradient,
     then projected onto the probability simplex (Σ=1, a_i≥0) and
     finally rescaled to Σ a_i = √(2·n).

3. Every few thousand iterations a random block perturbation is applied
   to escape possible plateaus.  The step size and the smoothing
   parameter β are slowly adapted: they grow when progress is made
   and shrink otherwise.

4. The best sequence seen so far is kept and returned.  Progress
   information is printed regularly.  The whole search respects the
   supplied time‑budget (default 1000 s).

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def make_arcsine_shape(n: int) -> np.ndarray:
    """
    Return a symmetric arcsine‑like distribution:
        a_i = 1 / sqrt( (i+0.5) * (n-0.5-i) )
    This reproduces the continuous arcsine density on a discrete grid.
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, Σ x_i = 1}
    Implementation follows Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("project_onto_simplex expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the full (linear) convolution of x and y using FFTs.
    L must be a power of two with L >= len(x) + len(y) - 1.
    Returns a real array of length len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    Z = X * Y
    conv = np.fft.irfft(Z, n=L)
    return conv[: len(x) + len(y) - 1]

    """

    # ------------------------------------------------------------------

        if iter_counter % 500 == 0:
