"""
Simulated annealing refinement of a step function.
...
"""

import time
import math
import numpy as np

def _initial_sequence():
    """Return an initial normalized non‑negative sequence."""
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 200
        seq = np.full(n, 1.0, dtype=np.float64)
    # make sure all entries are finite and non‑negative
    seq = np.where(np.isnan(seq) | np.isinf(seq), 0.0, seq)
    seq = np.clip(seq, 0.0, 1000.0)
    # normalize ℓ¹‑norm to √(2 n) – any scaling leaves the score unchanged,
    # but fixing it avoids drift caused by floating‑point errors.
    target = math.sqrt(2.0 * len(seq))
    seq = seq / np.sum(seq) * target
    return seq

def _evaluate(seq):
    """Wrapper for the provided scorer."""
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed=42, budget_s=1100):
    """
    Simulated‑annealing based search for a low‑value step sequence.
    The algorithm starts from the previous best construction (if available)
    and repeatedly moves a small amount of mass from a “large‑gradient”
    index to a random index while keeping the ℓ¹‑norm fixed.
    Moves are accepted with a Metropolis criterion that cools down
    over the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Initialise
    f = _initial_sequence()
    n = len(f)
    conv = np.convolve(f, f)                # full convolution, length 2n‑1
    cur_max = conv.max()
    best_f = f.copy()
    best_max = cur_max

    # Temperature schedule (linear in log‑space)
    T_start = 1.0
    T_end = 1e-4
    total_time = deadline - time.monotonic()
    start_time = time.monotonic()
    iteration = 0

    while time.monotonic() < deadline:
        # ------------------- temperature -------------------
        elapsed = time.monotonic() - start_time
        # exponential cooling
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        # ------------------- choose move -------------------
        # gradient at the current worst auto‑convolution entry
        k_max = int(np.argmax(conv))
        # compute gradient g_i = 2 * f_{k_max-i} on the feasible index range
        start_i = max(0, k_max - (n - 1))
        end_i = min(k_max, n - 1)
        idx = np.arange(start_i, end_i + 1, dtype=np.int64)
        grad = np.zeros(n, dtype=np.float64)
        grad[idx] = 2.0 * f[k_max - idx]

        # probability proportional to gradient (add epsilon to avoid zeros)
        prob = grad + 1e-12
        prob /= prob.sum()
        i = rng.choice(n, p=prob)

        # choose a destination index uniformly (different from i)
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n - 1)

        # amount of mass to move – a fraction of the donor entry,
        # the fraction shrinks slowly as time passes.
        max_delta = f[i]
        if max_delta <= 1e-12:
            continue
        # fraction ∈ [0.01, 0.2] decreasing linearly
        frac_max = 0.2 * (1.0 - elapsed / total_time) + 0.01
        delta = rng.random() * max_delta * frac_max
        if delta <= 0.0:
            continue

        # ------------------- apply move (in‑place) -------------------
        # keep a copy so we can roll back if the move is rejected
        conv_old = conv.copy()
        f_i_old = f[i]
        f_j_old = f[j]

        # linear part of the update
        conv[i:i + n] -= 2.0 * delta * f
        conv[j:j + n] += 2.0 * delta * f

        # quadratic correction
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        s = i + j
        if s < 2 * n - 1:
            conv[s] -= 2.0 * delta * delta

        # update the sequence itself
        f[i] -= delta
        f[j] += delta

        new_max = conv.max()

        # ------------------- Metropolis acceptance -------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # probability to accept a worsening move
            prob_accept = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob_accept:

"""
Simulated‑annealing refinement of a step function.

The scorer
    2·n·max(f*f) / (sum(f)**2)
is homogeneous: scaling the vector does not change the score.
Hence the algorithm works with a fixed ℓ¹‑norm (√(2·n)) which makes the
objective simply the worst entry of the auto‑convolution.

The method starts either from the previously discovered best construction
(`height_sequence_1` when it exists) or from a uniform vector.
At each iteration a small amount of mass is moved from an index that
contributes heavily to the current worst convolution entry (chosen
proportionally to the gradient 2·f_{k_max‑i}) to a random destination index.
The move is accepted with a Metropolis rule; the temperature cools
exponentially until the time budget (≈ 1000 s) expires.
The best sequence seen during the run is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: initialise a feasible sequence (ℓ¹‑norm = √(2·n))
def _initial_sequence() -> np.ndarray:
    """Return a non‑negative sequence with ℓ¹‑norm √(2·n)."""
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=np.float64)
    else:
        # fallback: uniform vector of length 200
        seq = np.full(200, 1.0, dtype=np.float64)

    # safety – clip to the interval accepted by the scorer
    seq = np.where(np.isnan(seq) | np.isinf(seq), 0.0, seq)
    seq = np.clip(seq, 0.0, 1000.0)

    # normalise to the standard ℓ¹‑norm (any scaling leaves the score unchanged)
    target = math.sqrt(2.0 * len(seq))
    seq = seq / np.sum(seq) * target
    return seq


# ----------------------------------------------------------------------
# Wrapper for the supplied scorer (already imported in the execution env)
def _evaluate(seq: list[float]) -> float:
    """Evaluate a candidate sequence using the provided evaluate_sequence."""
    return float(evaluate_sequence(seq))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """
    Simulated‑annealing search for a low‑value step sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget (seconds).  The routine stops a couple of
        seconds before the limit so that the caller can still retrieve the
        result.

    Returns
    -------
    list[float]
        The best sequence found (plain Python floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Initial data
    f = _initial_sequence()                     # length n, ℓ¹‑norm = √(2·n)
    n = len(f)
    conv = np.convolve(f, f)                    # full convolution, length 2·n‑1
    cur_max = conv.max()
    best_f = f.copy()
    best_max = cur_max

    # ------------------------------------------------------------------
    # Temperature schedule (exponential cooling)
    T_start = 1.0
    T_end = 1e-4
    total_time = deadline - time.monotonic()
    start_time = time.monotonic()
    it = 0
    # counters for occasional restarts (optional)
    last_improve_it = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        # exponential cooling
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        # ---------- pick a donor index (large gradient) ----------
        k_max = int(np.argmax(conv))                 # worst convolution index
        # gradient g_i = 2·f_{k_max-i} on the feasible range
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad = np.zeros(n, dtype=np.float64)
        grad[idx] = 2.0 * f[k_max - idx]

        # probability proportional to gradient (add tiny constant for stability)
        prob = grad + 1e-12
        prob /= prob.sum()
        i = rng.choice(n, p=prob)

        # ---------- pick a destination index ----------
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n - 1)

        # ---------- amount of mass to move ----------
        max_delta = f[i]
        if max_delta <= 1e-12:
            continue
        # fraction of the donor mass; shrinks as we approach the deadline
        frac = 0.2 * (1.0 - elapsed / total_time) + 0.01
        delta = rng.random() * max_delta * frac
        if delta <= 0.0:
            continue

        # ---------- apply the move (in‑place, with rollback ability) ----------
        conv_old = conv.copy()
        fi_old = f[i]
        fj_old = f[j]

        # linear part: -2·δ·f_{k-i}  and +2·δ·f_{k-j}
        conv[i:i + n] -= 2.0 * delta * f
        conv[j:j + n] += 2.0 * delta * f

        # quadratic correction (self‑terms)
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        s = i + j
        if s < 2 * n - 1:
            conv[s] -= 2.0 * delta * delta

        # update the sequence itself
        f[i] -= delta
        f[j] += delta

        new_max = conv.max()

        # ---------- Metropolis acceptance ----------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # probability to accept a worsening move
            prob_up = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob_up:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f.copy()
                last_improve_it = it
                # progress report
                eval_val = 2.0 * n * best_max / (np.sum(best_f) ** 2)
                print(
                    f"[SA] it={it:06d} eval={eval_val:.6f} max={best_max:.6f} T={T:.4e}"
                )
        else:
            # roll back
            conv[:] = conv_old
            f[i] = fi_old
            f[j] = fj_old

        # optional cheap restart if we have been stagnant for a long while
        if it - last_improve_it > 200_000 and elapsed > 0.5 * total_time:
            # re‑initialise from a fresh random vector (keeps the best ever)
            f = _initial_sequence()
            conv = np.convolve(f, f)
            cur_max = conv.max()
            last_improve_it = it

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence encountered (plain Python list of floats)
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_f.tolist()]
    final_score = _evaluate(best_seq)
    print("[final] best evaluation =", final_score)
    return best_seq
