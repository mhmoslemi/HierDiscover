def perturb(seq, rng):
    n = seq.size
    cand = seq.copy()
    # randomly choose between three types
    choice = rng.random()
    if choice < 0.5:
        # block scaling
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = rng.integers(0, max(1, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.12))
        cand[start:start+width] *= factor
    elif choice < 0.8:
        # mass transfer between two indices
        i, j = rng.choice(n, size=2, replace=False)
        # amount to move: up to min(cand[i], alpha) where alpha small
        max_move = min(cand[i], 0.1)  # ensure nonnegativity bound
        delta = rng.random() * max_move
        cand[i] -= delta
        cand[j] += delta
    else:
        # add Gaussian noise globally
        cand += rng.normal(scale=0.001, size=n)
    # No clipping, will project later.
    return cand

"""
Simulated‑annealing + gradient‑informed proposals for the max‑convolution
score.

The algorithm works on a probability simplex (non‑negative entries,
sum = 1 – the score is scale invariant).  It starts from the analytically
good “arcsine’’ distribution (or the provided historic solution) and
performs a long‑running simulated‑annealing loop.  At each iteration a
candidate is generated either by a random block‑scale perturbation,
a tiny mass transfer between two positions, or a gradient step that
reduces the currently largest convolution entries (we use the top‑k
entries instead of only the very maximum).  The candidate is projected
onto the simplex and accepted with the usual Metropolis probability.
The temperature is slowly cooled; occasional restarts keep the search
alive.  After the annealing phase a short deterministic projected‑gradient
refinement is applied to polish the best vector found.

The whole routine respects the 1000 s budget and returns a list of
floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_sequence(n: int) -> np.ndarray:
    """Discrete arcsine (central binomial) sequence of length *n*, normalised."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def self_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself via real FFT."""
    n = x.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv


def block_perturbation(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block of *a* by a log‑normal factor."""
    n = a.size
    # block width between ~0.5 % and ~8 % of the vector
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    b = a.copy()
    b[start:start + width] *= factor
    # occasional mirroring (helps symmetry)
    if rng.random() < 0.35:
        b = 0.5 * (b + b[::-1])
    return b


def mass_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small amount of mass from one coordinate to another."""
    n = a.size
    i, j = rng.choice(n, size=2, replace=False)
    # move at most 5 % of the source bin (or whatever is available)
    max_move = min(a[i], 0.05)
    delta = rng.random() * max_move
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def gradient_step(a: np.ndarray, top_k: int, step: float) -> np.ndarray:
    """
    Compute a gradient step that simultaneously reduces the *top_k*
    maximal convolution entries.  The raw gradient is summed over the
    selected indices and then normalised before stepping.
    """
    n = a.size
    conv = self_convolution(a)
    # indices of the *top_k* largest entries (may contain duplicates)
    idxs = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    for s in idxs:
        lo = max(0, s - (n - 1))
        hi = min(s, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx_range] += 2.0 * a[s - idx_range]
    norm = np.linalg.norm(grad) + 1e-12
    direction = -step * (grad / norm)
    return a + direction


def evaluate_on_simplex(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a vector already on the simplex.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = self_convolution(a)
    return float(2.0 * n * np.max(conv))

# Main entry point required by the outer harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    current best bound.  The search combines simulated‑annealing,
    gradient‑guided proposals (using the *top_k* largest convolution entries)
    and occasional random restarts.  The routine respects the supplied time
    budget and prints progress messages.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------------
    # Choose a working length (use the historic vector if present)
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        # no historic data – pick a reasonably large size
        n = 8192
        base = None

    # --------------------------------------------------------------
    # Assemble a few sensible starting points
    # --------------------------------------------------------------
    starts = []

    # 1. arcsine (the analytically good baseline)
    starts.append(arcsine_sequence(n))

    # 2. symmetric version of arcsine (enforces a mirror structure)
    a_sym = (starts[0] + starts[0][::-1]) / 2.0
    a_sym = proj_simplex(a_sym)
    starts.append(a_sym)

    # 3. uniform distribution (worst, but occasionally a good escape)
    starts.append(proj_simplex(np.ones(n, dtype=np.float64)))

    # 4. if a historic sequence is available, use its normalised version
    if base is not None:
        tmp = np.maximum(base, 0.0)
        if tmp.sum() == 0.0:
            tmp = np.ones_like(tmp)
        tmp /= tmp.sum()
        starts.append(tmp)

    # --------------------------------------------------------------
    # Global best tracking
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for s in starts:
        sc = evaluate_on_simplex(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print(f"Initial best score = {best_score:.6f} (length {n})")

    # --------------------------------------------------------------
    # Simulated‑annealing parameters
    # --------------------------------------------------------------
    T = 0.15 * best_score          # initial temperature (fraction of score)
    cooling = 0.9995               # geometric cooling
    top_k = 5                      # number of active convolution entries
    step_sz = 0.05                 # size of deterministic gradient steps
    iteration = 0
    last_report = time.monotonic()
    report_interval = 8.0          # seconds

    # the current state used for proposals; we start from the best so far
    cur = best_seq.copy()
    cur_score = best_score
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ------- generate a candidate --------------------------------
        r = rng.random()
        if r < 0.45:
            # block perturbation (large‑scale, multiplicative)
            cand = block_perturbation(cur, rng)
        elif r < 0.80:
            # small mass transfer (local)
            cand = mass_transfer(cur, rng)
        else:
            # deterministic gradient step using the *top_k* entries
            cand = gradient_step(cur, top_k=top_k, step=step_sz)

        # small isotropic noise every few dozen steps
        if iteration % 30 == 0:
            cand += rng.normal(scale=0.001, size=n)

        # enforce constraints
        cand = proj_simplex(cand)

        # ------- evaluate candidate ----------------------------------
        cand_score = evaluate_on_simplex(cand)

        # ------- Metropolis acceptance --------------------------------
        delta = cand_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ------- temperature / step adaptation -----------------------
        T *= cooling
        # keep a tiny step size floor to avoid stagnation
        step_sz = max(step_sz * 0.9999, 1e-6)

        # ------- occasional restart if stagnating --------------------
        if no_improve > 3000:
            # restart from a fresh arcsine vector, possibly perturbed
            cur = arcsine_sequence(n) + rng.normal(scale=0.002, size=n)
            cur = proj_simplex(cur)
            cur_score = evaluate_on_simplex(cur)
            no_improve = 0
            T = 0.15 * best_score   # reset temperature a little

        # ------- progress report ------------------------------------
        if time.monotonic() - last_report > report_interval:
            print(
                f"Iter {iteration:7d} | best = {best_score:.6f} | cur = {cur_score:.6f} "
                f"| T = {T:.6f} | step = {step_sz:.6f}"
            )
            last_report = time.monotonic()

    # --------------------------------------------------------------
    # Short deterministic refinement (projected‑gradient on the max entry)
    # --------------------------------------------------------------
    a = best_seq.copy()
    for _ in range(2000):
        conv = self_convolution(a)
        s_max = int(conv.argmax())
        # gradient of the *single* maximal entry
        lo = max(0, s_max - (n - 1))
        hi = min(s_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx_range] = 2.0 * a[s_max - idx_range]

        # tiny step in the opposite direction
        a_new = a - 0.02 * g / (np.linalg.norm(g) + 1e-12)
        a_new = proj_simplex(a_new)

        new_score = evaluate_on_simplex(a_new)
        if new_score < best_score:
            best_seq = a_new.copy()
            best_score = new_score
            a = a_new
        else:
            # no improvement – shrink the step size a bit
            a = a_new

    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # Return plain Python floats as required by the outer harness
    return [float(x) for x in best_seq]
