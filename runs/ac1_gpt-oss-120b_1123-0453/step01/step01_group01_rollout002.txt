"""
Propose a better step‑function sequence.

The algorithm works in three stages:

1. **Initialisation** – take the previous best sequence (if available)
   and the classic “arcsine” distribution.  Both are normalised to
   sum = 1, which makes the evaluation score simply
   `2·n·max(conv(a,a))`.

2. **Coarse parametric search** – generate families of Beta‑like
   shapes  a_i ∝ x_i**p·(1‑x_i)**q   (with x_i=(i+½)/n) for a grid of
   exponents p,q.  The best shape on this grid becomes the new
   starting point.

3. **Refinement** – run a projected (simplex) sub‑gradient descent on
   the maximal convolution entry, with a small amount of stochastic
   noise and occasional block‑perturbations to escape plateaus.
   The loop stops when the allotted wall‑clock time (≈ 1000 s) is
   exhausted.

The sequence returned is a list of non‑negative floats; scaling does
not affect the score because the evaluation function is homogeneous.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_sequence(n: int) -> np.ndarray:
    """Return the “arcsine’’ sequence of length n (normalised to sum = 1)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def self_convolution(v: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself via FFT."""
    n = v.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a cheap O(n) non‑negative perturbation of *base*.
    A random contiguous block is scaled by a log‑normal factor;
    optionally the whole vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    # choose a block width between 0.5 % and ~8 % of the length
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # keep everything in the allowed numeric range
    return np.clip(cand, 0.0, 1000.0)


def beta_like_sequence(n: int, p: float, q: float) -> np.ndarray:
    """
    Generate a Beta‑type shape a_i ∝ x_i**p·(1‑x_i)**q,
    where x_i = (i+½)/n.  The result is normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)
    x = (idx + 0.5) / n                     # strictly in (0,1)
    # safe log‑space computation
    log_a = p * np.log(x) + q * np.log1p(-x)
    # avoid overflow / underflow
    log_a = np.clip(log_a, -700, 700)
    a = np.exp(log_a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.ones_like(a) / n
    else:
        a /= s
    return a


def score_from_sequence(v: np.ndarray) -> float:
    """
    Compute the evaluation score for a sequence that is already normalised
    to sum = 1.  This mirrors `evaluate_sequence` but avoids the extra Python
    overhead.
    """
    n = v.shape[0]
    conv = self_convolution(v)
    return 2.0 * n * np.max(conv)

            print(f"Iter {it:6d}  improved to {best_score:.6f}")
        else:
            no_improve += 1
            # If stuck for a while, restart from a random Beta shape
            if no_improve > 1800:
                p_rand = rng.uniform(-0.9, 0.6)
                q_rand = rng.uniform(-0.9, 0.6)
                a = beta_like_sequence(n, p_rand, q_rand)
                a = proj_simplex(a)
                no_improve = 0
                print(f"Iter {it:6d}  restart with random (p,q)=({p_rand:.3f},{q_rand:.3f})")

        # ----- occasional block‑perturbation for diversification -----
        if it % 450 == 0 and it > 0:
            pert = make_block_perturbation(best, rng)
            pert = pert / np.sum(pert)
            pert_score = score_from_sequence(pert)
            if pert_score < best_score:
                best = pert.copy()
                best_score = pert_score
                a = pert.copy()
                no_improve = 0
                print(f"Iter {it:6d}  block perturb improved to {best_score:.6f}")

        # ----- decay step size and iterate -----
        step *= decay
        it += 1

        # ----- periodic progress report -----
        if it % 1000 == 0:
            print(f"Iter {it:6d} | best_score={best_score:.6f} | step={step:.6f}")

    print(f"Finished after {it} iterations, best_score={best_score:.6f}")
    # Return as plain Python floats (the evaluation routine expects a list)
    return [float(x) for x in best]
