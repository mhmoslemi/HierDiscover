def convolution_matrix(f):
    n = f.shape[0]
    m = 2*n - 1
    # allocate A
    A = np.empty((m, n), dtype=np.float64)
    # We'll fill all with zeros then assign.
    A[:] = 0.0
    for col in range(n):
        # number of rows where this column contributes
        length = min(n, m - col)
        A[col:col+length, col] = f[:length]
    return A

"""
Propose a near‑optimal step function for the metric

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages.

1️⃣  **Initial shape** – a power‑law family  a_i ≈ (i+α)⁻½  (the
    analytically flat convolution) is searched on a modest length
    (≤ 1800) and normalised to the canonical sum `√(2 n)`.  This gives a
    starting score close to π/2 ≈ 1.571.

2️⃣  **Linear‑programming refinement** – with the current vector `f`
    we formulate the linear programme

        maximise   Σ g_i
        subject to conv(f,g) ≤ M   (M = max(conv(f,f)))
                  g_i ≥ 0

    where `conv(f,g)` is linear in `g`.  Solving the LP yields a vector
    `g` with a (typically) much larger ℓ¹‑norm while still respecting the
    same convolution bound `M`.  After rescaling `g` to the target sum
    `√(2 n)`, we blend `f` and `g` via a convex combination
    `h = (1−t)·f + t·g` (t∈[0,1]).  For several candidate `t` values the
    new maximum convolution `M' = max(conv(h,h))` is measured and the
    best blend is kept.  The blend never increases the score and often
    yields a noticeable reduction.

3️⃣  **Stochastic hill‑climbing** – when the LP can no longer improve
    the score we fall back to cheap random block perturbations
    (multiplicative log‑normal changes on a contiguous block) and keep
    any improvement.  This phase also serves as a safety net if the LP
    becomes infeasible.

The whole search is time‑boxed (default 950 s) and stops early if no
progress is made.  The returned sequence consists of non‑negative
floats and respects the evaluator’s clipping limits.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``base``.
    A random contiguous block is multiplied by a log‑normal factor and
    optionally symmetrised.  Values are clipped to the interval [0,1000].
    """
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and ~5 % of n
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def convolution_matrix(f: np.ndarray) -> np.ndarray:
    """Build the (2n‑1) × n matrix A such that A·g = conv(f,g) (full convolution)."""
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # column i contributes f[0:len] on rows i … i+len‑1
    for col in range(n):
        length = min(n, m - col)            # how many rows this column touches
        A[col:col + length, col] = f[:length]
    return A


def evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence (expects a list)."""
    return float(evaluate_sequence(seq.tolist()))
