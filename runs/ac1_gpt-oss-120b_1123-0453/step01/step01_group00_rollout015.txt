"""
Propose a low‑value step sequence.

The algorithm is a refined version of the LP‑based scheme that was
used to obtain a score of 1.518.  For each chosen length n we
  • start from a random non‑negative vector with the prescribed
    ℓ¹‑norm √(2 n),
  • repeatedly solve the linear program
        maximise   Σ g_i
        subject to (f∗g)ₖ ≤ M   for all k   (M = max(f∗f)),
        g_i ≥ 0
    (this is exactly the LP described in the statement),
  • rescale the LP solution to the target ℓ¹‑norm,
  • replace f by that new vector if it strictly improves the maximum
    auto‑convolution value M, otherwise fall back to a small convex
    combination (mixing factor t) and keep trying.
  • stop when the bound cannot be improved any further or the time
    budget expires.

If CVXPY is not available we fall back to a simple random‑walk search.
The best sequence seen over all lengths and random restarts is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Small helper utilities (no closures, no lambdas)
def _evaluate(seq: np.ndarray) -> float:
    """Return the scorer value for a numpy 1‑D array."""
    # evaluate_sequence is already imported in the execution environment
    return float(evaluate_sequence(seq.tolist()))


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) == np.convolve(f, g).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # fill by sliding the vector f over all possible offsets
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over positions of f that appear in output k
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _lp_step(f: np.ndarray, M: float, target_sum: float):
    """
    Solve the LP that maximizes Σ g_i under (f∗g)ₖ ≤ M.
    Returns a feasible g (scaled to the target ℓ¹‑norm) or None if the
    solver fails.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None

    n = f.shape[0]
    A = _conv_matrix(f)

    # define the LP
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # solve – ECOS is generally fast for dense problems
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)
        except Exception:  # pragma: no cover
            return None

    if g.value is None:
        return None

    g_val = np.maximum(g.value, 0.0)  # guard against tiny negative artefacts
    s = np.sum(g_val)
    if s < 1e-12:
        return None
    # rescale to the required ℓ¹‑norm
    g_scaled = g_val * (target_sum / s)
    return g_scaled


def _random_restart(f_len: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative vector of length f_len whose sum equals
    √(2·f_len).  The distribution is uniform on the simplex.
    """
    target = math.sqrt(2.0 * f_len)
    x = rng.random(f_len) + 0.1                # avoid exact zeros
    x = x / np.sum(x) * target
    return x


def _simple_fallback(budget: float, seed: int):
    """
    Very basic random walk used when CVXPY is unavailable or LP fails.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget
    n = 200
    target = math.sqrt(2.0 * n)
    seq = np.full(n, target / n, dtype=np.float64)
    best_seq = seq.copy()
    best_score = _evaluate(seq)

    while time.time() < deadline:
        cand = seq + rng.normal(scale=0.05, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / np.sum(cand) * target
        sc = _evaluate(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            seq = cand
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score.

    The routine tries a few values of n (200…600) and, for each,
    performs several random restarts.  In every restart a linear program
    (as described in the problem statement) is solved repeatedly;
    the current sequence is replaced whenever the maximal
    auto‑convolution value strictly decreases.  A small convex
    combination fallback is used when the direct replacement would
    increase the bound.  The best sequence found before the time budget
    expires is returned as a plain Python list of floats.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)  # keep a margin

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # try a range of lengths – larger n allow a tighter theoretical bound
    lengths = [200, 300, 400, 500, 600]
    # number of independent restarts per length (reduced if we run out of time)
    restarts_per_n = 4

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            # ----- initialise a random feasible vector -----
            f = _random_restart(n, rng)
            cur_score = _evaluate(f)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[init] n={n} restart={restart} score={cur_score:.6f}")

            # ----- iterative LP improvement -----
            # maximum number of LP steps before we give up on this restart
            max_lp_iters = 200
            # mixing factor used when the pure LP step does not improve the bound
            t_mix = 0.2

            for it in range(max_lp_iters):
                if time.time() >= deadline:
                    break

                # current maximal auto‑convolution entry
                conv = np.convolve(f, f)
                M = float(conv.max())

                # solve the linear program; fall back if solver is unavailable
                g_scaled = _lp_step(f, M, target_sum)
                if g_scaled is None:
                    # LP failed – abort this restart
                    break

                # compute the bound resulting from the LP solution
                conv_new = np.convolve(g_scaled, g_scaled)
                M_new = float(conv_new.max())

                # ----- decide how to update the current vector -----
                improved = M_new < M - 1e-12
                if improved:
                    f = g_scaled
                    cur_score = float(2.0 * n * M_new / (target_sum ** 2))
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                        print(f"[LP] n={n} restart={restart} it={it} score={cur_score:.6f}")
                else:
                    # try a convex combination (1‑t)·f + t·g_scaled
                    f_candidate = (1.0 - t_mix) * f + t_mix * g_scaled
                    conv_cand = np.convolve(f_candidate, f_candidate)
                    M_cand = float(conv_cand.max())
                    if M_cand < M:
                        f = f_candidate
                        cur_score = float(2.0 * n * M_cand / (target_sum ** 2))
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = f.copy()
                            print(f"[mix] n={n} restart={restart} it={it} score={cur_score:.6f}")
                        # when mixing helps we can be slightly more aggressive
                        t_mix = min(0.5, t_mix * 1.1)
                    else:
                        # mixing did not help – become more conservative
                        t_mix = max(0.05, t_mix * 0.5)
                        # if t_mix becomes tiny we stop trying further LP steps
                        if t_mix < 0.05:
                            break

                # early exit if we are essentially at the theoretical lower bound
                lower_bound = 2.0 * n / (2.0 * n - 1.0)   # = average of the convolution
                if M_new <= lower_bound * 1.01:
                    best_score = M_new
                    best_seq = f.copy()
                    print(f"[nearly optimal] n={n} score={best_score:.6f}")
                    break

    # ------------------------------------------------------------------
    # If all LP attempts failed (or CVXPY was not present), fall back to a
    # very cheap random walk that still respects the security checks.
    if best_seq is None:
        fallback_budget = max(0.0, deadline - time.time())
        best_seq, best_score = _simple_fallback(fallback_budget, seed)

    # final safety: convert to plain list of floats
    return [float(x) for x in best_seq]
