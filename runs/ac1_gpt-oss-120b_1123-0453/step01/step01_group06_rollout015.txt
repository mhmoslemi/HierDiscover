import numpy as np
import time

def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _conv_max(seq: np.ndarray) -> float:
    """Fast FFT based max convolution of seq with itself."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)
    return float(conv[: 2 * n - 1].max())

def _score_estimate(seq: np.ndarray) -> float:
    """Fast estimate of the evaluation function (exact for the given objective)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)

def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """
    Fast FFT‑based computation of max(conv(seq, seq)).
    Returns a Python float.
    """
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score_estimate(seq: np.ndarray) -> float:
    """Exact value of the objective (fast – uses FFT)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """

    Occasionally mirrors the whole vector to keep it roughly symmetric.
