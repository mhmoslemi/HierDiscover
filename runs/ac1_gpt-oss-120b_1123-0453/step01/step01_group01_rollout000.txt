def gradient_corr(a, w):
    n = a.size
    # w length = 2n-1
    # compute conv = np.convolve(a, w[::-1])
    size = 1
    while size < len(a) + len(w) -1:
        size <<= 1
    A = np.fft.rfft(a, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)
    conv = np.fft.irfft(A * W_rev, n=size)
    # conv length size; the valid part for correlation is indices (len(w)-1) to (len(w)-1+n-1) = (2n-2) to (3n-3)
    start = len(w)-1
    g = 2.0 * conv[start:start+n]
    return g

def gradient_via_fft(a, w):
    # a length n, w length 2n-1
    n = a.shape[0]
    # correlation: g_j = 2 * sum_i a_i w_{j+i}
    # compute full convolution of a and w_rev
    w_rev = w[::-1]
    size = 1
    conv_len = n + len(w_rev) - 1
    while size < conv_len:
        size <<= 1
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * W, n=size)
    start = len(w_rev) - 1  # which is len(w)-1
    g = 2.0 * conv[start:start+n]
    return g

"""
Deterministic‐annealing optimizer for the max‑convolution score.

The objective (with sum(a)=1) is 2·n·max(conv(a,a)).  To minimise it we
approximate the non‑smooth “max’’ by a soft‑max with a temperature β.
For a given β the smoothed objective is

    Jβ(a) = Σk w_k·c_k ,   where   w_k = exp(β·c_k)/Σℓ exp(β·c_ℓ)
    and c = a * a  (full linear convolution).

∇Jβ(a) can be written in closed form:
    (∇Jβ)_j = 2· Σ_i a_i·w_{j+i}    (cross‑correlation of a and w).

The gradient is obtained with a single FFT of a (for the convolution)
and a second FFT of the reversed weight vector.  An Adam optimiser
(adaptive moments) is used together with a projection onto the
probability simplex.  After each update we symmetrise the vector
(a_i = a_{n‑1‑i}) because the best known constructions are symmetric.

The temperature β is increased slowly (deterministic annealing) so that
the optimisation first moves in a smooth landscape and then focuses on
the true max‑convolution term.  Occasionally a random block‑perturbation
is applied to the current best solution to escape shallow minima.

The function returns the best sequence found within the allocated time.
"""

import time
import numpy as np


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Uses the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_sequence(n: int) -> np.ndarray:
    """Central binomial (arcsine) distribution of length n, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) perturbation: pick a random contiguous block and scale it.
    The result is clipped to the allowed range and re‑normalised.
    """
    n = base.size
    cand = base.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring (helps keep symmetry)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    (hopefully) minimises evaluate_sequence().
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Determine problem size and initialise a starting point.
    # ----------------------------------------------------------------------

        n = int(base.size)
        base = np.maximum(base, 0.0)
        s = base.sum()
        if s == 0.0:

        else:
            base = base / s
    except Exception:
        # fallback: arcsine distribution of a reasonable length
        n = 6000
        base = arcsine_sequence(n)

    # ----------------------------------------------------------------------
    # Pre‑compute FFT size (next power of two ≥ 3·n) – needed for
    #   * a*a   (full convolution length 2n‑1)
    #   * a⋆w   (correlation length up to 3n‑2)
    # ----------------------------------------------------------------------
    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1

    # ----------------------------------------------------------------------
    # Optimiser state
    # ----------------------------------------------------------------------
    a = base.copy()                     # current iterate (on simplex)
    best = a.copy()                     # best solution seen so far
    best_score = float('inf')
    m = np.zeros_like(a)                # Adam 1st moment
    v = np.zeros_like(a)                # Adam 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.05                        # base learning rate
    iter_cnt = 0
    t_adam = 0

    # temperature for soft‑max (starts low → smooth, then grows)
    beta_temp = 1.0
    beta_inc = 1.001                    # exponential increase per iteration
    beta_max = 50.0

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        iter_cnt += 1

        # ---------- convolution a*a ----------
        A_fft = np.fft.rfft(a, n=fft_size)
        conv_full = np.fft.irfft(A_fft * A_fft, n=fft_size)
        conv = conv_full[: 2 * n - 1]          # full linear convolution
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ---------- keep the best ----------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{iter_cnt:6d}] New best score = {best_score:.6f}")

            # early exit if we already hit the target
            if best_score <= 1.503:

        # ---------- soft‑max weights ----------
        # w_k = exp(beta·(c_k−max)) / Σℓ exp(beta·(c_ℓ−max))
        # shifting by max avoids overflow
        exp_vec = np.exp(beta_temp * (conv - max_conv))
        w = exp_vec / exp_vec.sum()           # length 2n‑1, non‑negative

        # ---------- gradient = 2·(a ⋆ w) ----------
        w_rev = w[::-1]
        W_rev_fft = np.fft.rfft(w_rev, n=fft_size)
        corr_full = np.fft.irfft(A_fft * W_rev_fft, n=fft_size)
        start = len(w_rev) - 1                 # = 2n‑2
        grad = 2.0 * corr_full[start : start + n]

        # ---------- Adam update ----------
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        a = a - step0 * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- enforce symmetry ----------
        a = 0.5 * (a + a[::-1])

        # ---------- projection onto simplex ----------
        a = proj_simplex(a)

        # ---------- anneal the temperature ----------
        beta_temp = min(beta_temp * beta_inc, beta_max)

        # ---------- occasional random block perturbation ----------
        if iter_cnt % 1000 == 0:
            pert = best.copy()
            pert = make_block_perturbation(pert, rng)
            pert = proj_simplex(pert)
            # evaluate perturbed candidate
            A_fft_p = np.fft.rfft(pert, n=fft_size)
            conv_p_full = np.fft.irfft(A_fft_p * A_fft_p, n=fft_size)
            conv_p = conv_p_full[: 2 * n - 1]
            score_p = 2.0 * n * conv_p.max()
            if score_p < best_score:
                best_score = score_p
                best = pert.copy()
                a = pert.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[{iter_cnt:6d}] Perturbation improved to {best_score:.6f}")

        # ---------- occasional restart if stuck ----------
        if iter_cnt % 5000 == 0 and cur_score - best_score > 1e-5:
            a = best.copy()
            m.fill(0.0)
            v.fill(0.0)

    print(f"Finished after {iter_cnt} iterations, best score = {best_score:.6f}")
    # Return as plain Python floats (required by the outer harness)
    return [float(x) for x in best]
