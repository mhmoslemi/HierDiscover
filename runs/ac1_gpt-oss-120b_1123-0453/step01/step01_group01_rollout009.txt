def propose_candidate(seed=43, budget_s=1100, **kwargs):
    import time, numpy as np, math, sys, random

    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 5.)

    # Determine n
    base_given = None
    n = None
    try:
        base_given = np.asarray(height_sequence_1, dtype=np.float64)
        if base_given.ndim == 1 and base_given.size > 0 and np.isfinite(base_given).all():
            n = int(base_given.size)
    except Exception:
        pass
    if n is None:
        n = 4000

    # Helper: arcsine_sequence
    def arcsine_sequence_local(N):
        a = np.empty(N, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, N):
            a[i] = a[i-1] * (2*i - 1) / (2.0 * i)
        a /= a.sum()
        return a

    # init best using arcsine
    best_a = arcsine_sequence_local(n)
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial arcsine score {best_score:.6f}")

    # also consider given base if present
    if base_given is not None:
        tmp = base_given.copy()
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s == 0:
            tmp = np.ones_like(tmp) / tmp.size
        else:
            tmp = tmp / s
        s_score = evaluate_sequence(tmp.tolist())
        if s_score < best_score:
            best_a = tmp.copy()
            best_score = s_score
            print(f"Using provided base, score {best_score:.6f}")

    # Adam optimizer state
    rng = np.random.default_rng(seed)

    # Parameterization: logits u such that a = softmax(u)
    u = np.log(best_a + 1e-12)  # small epsilon to avoid log0

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    no_improve = 0
    # schedule for lambda (softmax temperature)
    lam_start = 5.0
    lam_end = 50.0

    # function for FFT convolution
    def conv_self_fft(x):
        nlen = x.size
        size = 2 * nlen - 1
        # rfft length = size//2 + 1
        f = np.fft.rfft(x, n=size)
        conv = np.fft.irfft(f * f, n=size)
        return conv

    # weighted conv (a_rev * w)
    def conv_weighted_fft(a_rev, w):
        # compute full convolution length = len(a_rev) + len(w) -1 = n + (2n-1) -1 = 3n-2
        N = 3 * n - 2
        size2 = 1 << (N - 1).bit_length()  # next power of two
        Fa = np.fft.rfft(a_rev, n=size2)
        Fw = np.fft.rfft(w, n=size2)
        conv = np.fft.irfft(Fa * Fw, n=size2)
        # slice central part
        start = n - 1
        return conv[start:start + n]

    while time.monotonic() < deadline:
        # compute a
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)  # subtract max for stability
        a = exp_u / exp_u.sum()
        # compute self-convolution
        conv = conv_self_fft(a)
        # compute lambda value based on progress
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (deadline - start))
        lam = lam_start + (lam_end - lam_start) * progress

        # softmax weights
        max_c = conv.max()
        # shift to avoid overflow
        w = np.exp(lam * (conv - max_c))
        w = w / w.sum()

        # gradient w.r.t a
        a_rev = a[::-1]
        g_a = 2.0 * conv_weighted_fft(a_rev, w)

        # gradient w.r.t u (softmax)
        dot = np.dot(a, g_a)
        g_u = a * (g_a - dot)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        step = 0.05 / (np.sqrt(v_hat) + eps)
        u -= step * m_hat

        # Optionally add small isotropic noise occasionally
        if it % 50 == 0:

        # Evaluate occasionally (e.g., every 200 iterations)
        if it % 200 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
                print(f"Iter {it:6d} | new best {best_score:.6f}")
            else:
                no_improve += 1
                if no_improve > 5000:

                    if rng.random() < 0.4:
                        a0 = arcsine_sequence_local(n)
                        u = np.log(a0 + 1e-12)
                        print(f"Restarting from arcsine at iter {it}")
                    else:
                        dir_sample = rng.gamma(1.0, size=n)  # exponential
                        dir_sample /= dir_sample.sum()
                        u = np.log(dir_sample + 1e-12)
                        print(f"Restarting from random Dirichlet at iter {it}")
                    m.fill(0.0)
                    v.fill(0.0)
                    no_improve = 0

        # block perturbation every 1500 iterations
        if it % 1500 == 0:
            # create a copy of best and perturb a block multiplicatively
            pert = best_a.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 10)))
            start_idx = int(rng.integers(0, max(1, n - width + 1)))

            pert[start_idx:start_idx+width] *= factor

            pert = np.clip(pert, 0.0, 1000.0)

            if pert_score < best_score:
                best_score = pert_score
                best_a = pert.copy()
                # re‑initialize u to this perturbed solution
                u = np.log(best_a + 1e-12)
                m.fill(0.0)
                v.fill(0.0)
                print(f"Iter {it:6d} | block perturb improved to {best_score:.6f}")

    print(f"Finished after {it} iterations, best score = {best_score:.6f}")
    return [float(x) for x in best_a]

"""
Simple yet powerful optimizer for the “max‑convolution” score.

The score to minimise is
    2 * n * max(conv(a, a)) / (sum(a)**2)

Key ideas
---------
* Work on the simplex by re‑parameterising the sequence with logits
  `u` → `a = softmax(u)`.  This automatically guarantees non‑negativity
  and ∑a = 1, i.e. the normalisation used by the evaluation routine.
* Replace the non‑smooth `max` by a temperature‑controlled soft‑max:
      w_k ∝ exp( λ·(conv_k – max(conv)) )
  With a small λ the gradient is smooth (helps exploration); as the
  optimisation proceeds λ is increased so the direction converges to the
  true sub‑gradient of the maximal entry.
* Use the Adam optimiser on the logits – it provides adaptive step sizes
  and momentum without any hand‑tuned decay schedule.
* Periodic “block perturbations’’ (multiplicative random stretch of a short
  segment) and occasional random restarts (arcsine or Dirichlet) help
  escape plateaus.
* All heavy operations (self‑convolution, weighted convolution needed for
  the gradient) are performed with FFTs, giving O(n log n) per iteration.
  With a modest length (≈ 4 000) the optimiser can run several hundred
  thousand iterations within the 1 000 s budget.

The function `propose_candidate` returns the best sequence found when the
time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def arcsine_sequence(n: int) -> np.ndarray:
    """Arcsine distribution (central binomial coefficients), normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def softmax_logits(u: np.ndarray) -> np.ndarray:
    """Convert logits to a probability vector on the simplex (stable)."""
    m = np.max(u)
    e = np.exp(u - m)
    return e / e.sum()


def self_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (size = 2·len(a)‑1)."""
    n = a.size
    size = 2 * n - 1
    f = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv


def gradient_wrt_a_fft(a_rev: np.ndarray, w: np.ndarray, n: int, size_fft: int) -> np.ndarray:
    """
    Compute  g = 2·(a_rev * w)  (restricted to the indices that correspond
    to the gradient ∂/∂a_i of the soft‑max weighted sum of convolutions).
    """
    Fa = np.fft.rfft(a_rev, n=size_fft)
    Fw = np.fft.rfft(w, n=size_fft)
    full = np.fft.irfft(Fa * Fw, n=size_fft)        # real result
    start = n - 1                                   # pick the central n entries
    return 2.0 * full[start:start + n]

# Main optimiser
# ----------------------------------------------------------------------
