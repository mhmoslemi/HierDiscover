import numpy as np
import time
import math

def _conv_self(a):
    """Return full self‑convolution of a (length 2·n‑1)."""
    n = a.size
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)
    return conv[: 2 * n - 1]

def _project_to_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    # find the last index where u_i - cssv_i/ i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries negative, return uniform simplex point
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_from_weights(a, w):
    """Gradient ∂/∂a Σ_k w_k (a∗a)_k  = 2·a∗w (cross‑correlation)."""
    n = a.size
    L = w.size                      # = 2·n‑1
    size = int(2 ** np.ceil(np.log2(n + L - 1)))   # >= 3·n‑2
    # convolution of a and reversed w
    Fa = np.fft.rfft(a, size)
    Fw = np.fft.rfft(w[::-1], size)
    conv_full = np.fft.irfft(Fa * Fw, size)
    # extract the slice that corresponds to indices 0…n‑1
    grad = 2.0 * conv_full[L - 1 - np.arange(n)]
    return grad

def _make_block_perturbation(base, rng):
    """Cheap O(n) multiplicative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # choose a block width roughly between 0.5 % and 10 % of n
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # keep values inside the allowed range
    np.clip(cand, 0.0, 1000.0, out=cand)
    # renormalise to the simplex (sum = 1)
    cand = _project_to_simplex(cand, 1.0)
    return cand

    projected sub‑gradient step (using a soft max or a variance term)
    and occasionally injects a random block perturbation to escape
    plateaux.  The best sequence seen during the run is returned.

    # -------------------------------------------------------------

            raise Exception

"""
Gradient‑based search for a low‑peak self‑convolution step‑function.

The algorithm proceeds as follows

1.  Pick a length `n`.  If a previous construction exists we reuse its
    length, otherwise a modest default (≈3000) is used.
2.  Initialise with the “arcsine’’ shape   a_i ∝ 1/√(i+1)   and
    normalise the L1‑norm to 1 (the objective is scale‑invariant).
3.  Repeatedly improve the current vector `cur` while staying in the
    feasible simplex {x≥0, Σx=1}:

    *   With high probability a *projected sub‑gradient* step is taken.
        The gradient is built from a weight vector `w` that either
        (a) concentrates on the convolution indices attaining the
            maximum (sub‑gradient of the “max’’ term) or
        (b) is the centred deviation `b‑mean(b)` (gradient of a
            variance‑like surrogate).  The two options are mixed
        randomly.
    *   Occasionally a cheap random *block perturbation* (multiplicative
        log‑normal jitter on a randomly chosen block) is applied in
        order to escape plateaus.
    *   After each move we project onto the simplex (non‑negative,
        fixed L1‑sum) and recompute the cheap objective
        `score = 2·n·max(conv(cur,cur))`.  The official evaluator
        `evaluate_sequence` is called only when the cheap score beats
        the best‑so‑far value.

4.  The best sequence observed during the run is returned.

The whole search respects the 1000 s budget, uses only NumPy and the
provided `evaluate_sequence`, and prints progress messages.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))          # power‑of‑2 for speed
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)
    return conv[: 2 * n - 1]


def _project_to_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex
    {x ≥ 0, Σ x = s}.
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be negative – return uniform point
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a∗a)_k w.r.t. `a`.
    It equals 2·a ⋆ w (cross‑correlation), which we compute via FFT.
    """
    n = a.size
    L = w.size                           # = 2·n‑1
    size = int(2 ** np.ceil(np.log2(n + L - 1)))   # ≥ 3·n‑2
    # correlation is convolution of a with reversed w
    Fa = np.fft.rfft(a, size)
    Fw = np.fft.rfft(w[::-1], size)
    conv_full = np.fft.irfft(Fa * Fw, size)
    grad = 2.0 * conv_full[L - 1 - np.arange(n)]
    return grad


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Very cheap multiplicative perturbation of ``base``.
    A random block (≈0.5 % … 8 % of the vector) is jittered in log‑space;
    with some probability the vector is mirrored to preserve symmetry.
    The result is projected back onto the simplex Σx=1.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # 0.5 % of n
    hi = max(2, n // 12)           # ≈8 % of n
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start : start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    # project to the simplex (sum = 1)
    cand = _project_to_simplex(cand, 1.0)
    return cand

    cur = cur.astype(np.float64)

    rng = np.random.default_rng(seed)

    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    b_cur = _conv_self(cur)

        cur_score = nxt_score
        it += 1

    # 4) final report
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {it} iters, runtime={time.monotonic()-start_time:.2f}s")
    return [float(x) for x in best_seq]
