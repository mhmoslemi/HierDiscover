import time
import numpy as np
from scipy.optimize import linprog

def fast_evaluate(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    total = float(seq.sum())
    if total < 0.01:
        return np.inf
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(F * F, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (total * total)

def generate_power_sequence(n, exponent=0.5, offset=0.0):
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)

def block_perturb(seq, rng):
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.07)))  # ~7% typical
    new = seq.copy()
    new[start:start+width] *= factor
    # occasional symmetrisation
    if rng.random() < 0.05:
        new = 0.5 * (new + new[::-1])
    np.clip(new, 0.0, 1000.0, out=new)
    return new

def mass_transfer(seq, rng, max_fraction=0.5):
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j or seq[i] <= 0:
        return seq
    max_delta = seq[i] * max_fraction
    delta = float(rng.random() * max_delta)
    new = seq.copy()
    new[i] -= delta
    new[j] += delta
    # keep sum constant, nonnegativity already assured
    np.clip(new, 0.0, 1000.0, out=new)
    return new

def active_set_lp(f, M, active_idx):
    """
    Solve   max sum(g)   s.t.  f * g <= M on the active positions.
    Returns the optimal g (numpy array) or None if infeasible.
    """
    n = f.size
    m = len(active_idx)
    # Build A_ub and b_ub
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        # coefficients for g_j: f_{k-j} if valid
        # j runs from 0..n-1, i = k - j
        # we set A[row, j] = f[i] if 0 <= i < n else 0
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for j = k - i, i ranges as above
        # Fill slice:
        # j = k - i => i = k - j
        # For i from i_start..i_end:
        #    j = k - i
        # This maps decreasing j.
        # We'll fill using loop over i range.
        # Given n moderate, simple loop ok.
        for i in range(i_start, i_end + 1):
            j = k - i
            A[row, j] = f[i]
    b = np.full(m, M, dtype=np.float64)
    # objective: maximize sum(g)  <=> minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    # Use high-performance solver
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    except Exception:
        return None
    if res.success:
        return res.x
    return None

"""
Propose a high‑quality step‑function (non‑negative sequence) for the
given evaluator.

The algorithm is a hybrid of simulated‑annealing moves and an
active‑set linear‑programming refinement:

* Start from a power‑law kernel (≈ (i+1)^‑0.5) – the classic analytic
  construction whose convolution is almost constant.
* Keep the total sum fixed (scale‑invariant objective) and repeatedly
  apply cheap stochastic moves:
    - multiplicative perturbation of a random contiguous block,
    - transfer of a random amount from one index to another,
    - tiny global log‑normal noise.
* Accept a move if it improves the current best score, otherwise
  accept with a temperature‑controlled probability (simulated‑annealing).
* Every few iterations a small linear program is solved:
  for the current sequence f and its current peak M = max(f * f),
  maximise ∑ g_j subject to f * g ≤ M on the most critical convolution
  positions (those where f * f is near its maximum).  The optimal g
  is mixed back into f (convex combination) to obtain a lower peak.
* Very occasional random restarts with fresh power‑law parameters are
  performed to avoid local minima.
* The routine runs until the supplied time budget (≈ 1000 s) expires and
  returns the best sequence found as a plain Python list of floats.

Only NumPy and SciPy are required; the code respects the evaluator’s
security checks (clipping, non‑negativity, sum‑threshold) and works for
any sequence length.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _fast_evaluate(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` using FFT convolution."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = float(seq.sum())
    if total < 0.01:
        return np.inf
    n = seq.size
    # size = next power of two for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(F * F, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (total * total)


def _generate_power_sequence(n: int, exponent: float = 0.5,
                             offset: float = 0.0) -> np.ndarray:
    """Return a_i = (i+offset+1)^(-exponent) (non‑negative, clipped)."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)


def _block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.07)))   # ≈7 % typical change
    new = seq.copy()
    new[start:start + width] *= factor
    # occasional symmetrisation (helps the convolution shape)
    if rng.random() < 0.05:
        new = 0.5 * (new + new[::-1])
    np.clip(new, 0.0, 1000.0, out=new)
    return new


def _mass_transfer(seq: np.ndarray, rng: np.random.Generator,
                   max_fraction: float = 0.5) -> np.ndarray:
    """Move a random amount from one index to another, preserving the total sum."""
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j or seq[i] <= 0.0:
        return seq
    delta = float(rng.random() * min(seq[i] * max_fraction, seq[i]))
    new = seq.copy()
    new[i] -= delta
    new[j] += delta
    np.clip(new, 0.0, 1000.0, out=new)
    return new


def _active_set_lp(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve   max sum(g)   s.t.   f * g ≤ M   on the positions ``active_idx``.
    Returns the optimal g (numpy array) or ``None`` if the LP fails.
    """
    n = f.size
    m = active_idx.size
    # Build a dense matrix A (m × n) – each row corresponds to one convolution constraint.
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        # coefficient for g_j is f_{k‑j} when the index is inside the window.
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # fill the non‑zero slice in one go (vectorised over i)
        i_vals = np.arange(i_min, i_max + 1, dtype=np.int64)
        j_vals = k - i_vals
        A[row, j_vals] = f[i_vals]
    b = np.full(m, M, dtype=np.float64)
    # maximise ∑ g_j  <=>  minimise -∑ g_j
    c = -np.ones(n, dtype=np.float64)

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    except Exception:          # pragma: no cover – safety net
        return None
    if res.success:
        return res.x
    return None

                best_seq = f.copy()
                best_score = restart_score
                elapsed = time.monotonic() - start_time
                print(f"Random restart improved to {best_score:.8f} at time {elapsed:.1f}s")

        iteration += 1

    # --------------------------------------------------------------
    # Return the best sequence discovered (plain Python list).
    # --------------------------------------------------------------
    return [float(x) for x in best_seq]
