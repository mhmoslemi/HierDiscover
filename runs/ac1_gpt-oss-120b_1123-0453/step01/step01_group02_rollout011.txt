"""
Propose a near‑optimal non‑negative step‑function for the given
evaluation metric.

Algorithm
---------
1.  **Initial guess** – Grid‑search a two‑parameter family
    a_i = (i + α)^{-p} (p≈0.5) to obtain a good starting vector.
    The search is performed in two stages: a coarse sweep over a wide
    range, then a refined local sweep around the best parameters.

2.  **Aggressive hill‑climbing** – Repeatedly move a fraction of the
    mass from the entry that contributes most to the *current* maximal
    auto‑convolution value to an entry that contributes the least
    (practically the smallest element of the vector).  The step size
    (fraction of donor mass) is chosen randomly in [0.2, 0.5] so that
    the search makes large, yet still safe, moves.  Every few hundred
    iterations a random block perturbation (multiplicative log‑normal
    factor on a contiguous segment) or a tiny smoothing operation is
    performed to escape plateaus.

3.  **Stopping** – The loop runs until the allocated time budget
    (≈1000 s) is exhausted or no improvement has been observed for a
    long stretch.  The best sequence found is returned as a plain
    Python list of floats.

All numbers are kept inside the interval [0, 1000] and are non‑negative,
exactly matching the constraints of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def fast_score(seq: np.ndarray) -> float:
    """Return the objective 2·n·max(conv)/sum² for a non‑negative vector."""
    n = seq.size
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def power_law(p: float, alpha: float, n: int) -> np.ndarray:
    """Generate a_i = (i+α)^{-p} for i = 0 … n‑1."""
    idx = np.arange(n, dtype=np.float64)
    return (idx + alpha) ** (-p)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and 15 % of the vector
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # log‑normal factor
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring (helps symmetry)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def smoothing_step(vec: np.ndarray) -> np.ndarray:
    """A cheap three‑point averaging that preserves the total sum."""
    n = vec.size
    if n < 3:
        return vec
    new = np.empty_like(vec)
    new[0] = 0.5 * vec[0] + 0.5 * vec[1]
    new[-1] = 0.5 * vec[-2] + 0.5 * vec[-1]
    new[1:-1] = 0.25 * vec[:-2] + 0.5 * vec[1:-1] + 0.25 * vec[2:]
    return np.clip(new, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise the
    supplied `evaluate_sequence` function.  The implementation follows the
    strategy described in the module docstring.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size (n).  Fall back to a moderate size if the
    # global `height_sequence_1` is unavailable or malformed.
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        n = 6000                     # a safe default that still runs fast
    # --------------------------------------------------------------
    # 1) Coarse + fine grid search over (p, α) for the power‑law family.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # reserve a few seconds

    idx = np.arange(n, dtype=np.float64)

    # ---- coarse phase -------------------------------------------------
    p_grid_coarse = np.linspace(0.48, 0.54, 13)      # 13 values, step ≈0.005
    alpha_grid_coarse = np.linspace(0.05, 2.0, 20)  # 20 values, step ≈0.1

    for p in p_grid_coarse:
        for alpha in alpha_grid_coarse:
            if time.monotonic() >= deadline:
                break
            cand = (idx + alpha) ** (-p)
            score = fast_score(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"coarse p={p:.3f} α={alpha:.3f} → score={best_score:.6f}")

        if time.monotonic() >= deadline:
            break

    # ---- fine phase ----------------------------------------------------
    if best_seq is not None:
        # extract region around the best parameters
        # (we reuse the last successful p and α)
        p0 = p
        alpha0 = alpha
    else:
        # fallback defaults
        p0, alpha0 = 0.5, 0.4

    p_fine = np.linspace(max(0.40, p0 - 0.04), min(0.60, p0 + 0.04), 9)
    alpha_fine = np.linspace(max(0.01, alpha0 - 0.2), alpha0 + 0.2, 41)

    for p in p_fine:
        for alpha in alpha_fine:
            if time.monotonic() >= deadline:
                break
            cand = (idx + alpha) ** (-p)
            score = fast_score(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"fine   p={p:.3f} α={alpha:.3f} → score={best_score:.6f}")

        if time.monotonic() >= deadline:
            break

    # If, for any reason, the search failed, start from a simple plateau.
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) * (np.sqrt(2.0 * n) / n)

    # --------------------------------------------------------------
    # 2) Aggressive hill‑climbing refinement.
    # --------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    best_seq = cur.copy()
    best_score = cur_score

    max_iters = 25000                # safety cap
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline and iter_count < max_iters:
        iter_count += 1

        # compute convolution once (used both for score & for donor choice)
        conv = np.convolve(cur, cur)
        k_max = int(np.argmax(conv))
        max_conv = float(conv[k_max])
        sum_cur = float(cur.sum())

        # Early‑exit check – if we already beat the target, we may stop
        if best_score < 1.503:
            break

        # ------------------------------------------------------------------
        # Identify donor (largest contribution to the peak) and receiver.
        # ------------------------------------------------------------------
        # contribution of index i to (a*a)_k_max : a[i] * a[k_max-i] (if in range)
        i = np.arange(n, dtype=np.int64)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i[mask]] * cur[j[mask]]

        donor = int(np.argmax(contrib))          # most harmful entry
        # the safest receiver is simply the smallest element (far‑tail)
        receiver = int(np.argmin(cur))

        # If donor == receiver (degenerate case) fall back to block perturbation
        if donor == receiver or cur[donor] <= 0.0:
            candidate = block_perturbation(cur, rng)
        else:
            # move a *significant* fraction of donor's mass to receiver
            fraction = float(rng.uniform(0.2, 0.5))          # 20‑% … 50‑%
            delta = min(cur[donor] * fraction, cur[donor])
            candidate = cur.copy()
            candidate[donor] -= delta
            candidate[receiver] += delta
            candidate = np.clip(candidate, 0.0, 1000.0)

        # Occasionally (≈1 % of the time) perform an extra smoothing step.
        if rng.random() < 0.01:
            candidate = smoothing_step(candidate)

        # Evaluate the proposal (fast version – identical to evaluate_sequence)
        cand_score = fast_score(candidate)

        if cand_score < cur_score:                 # improvement – keep it
            cur = candidate
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_seq = candidate.copy()
                print(f"Iter {iter_count:5d}  improved → score={best_score:.6f}")
        else:
            no_improve += 1
            # every few thousand unproductive steps try a random jump
            if no_improve >= 2000 and rng.random() < 0.1:
                cur = block_perturbation(cur, rng)
                cur_score = fast_score(cur)
                no_improve = 0

        # safety break – if we have been stagnant for a long time, quit
        if no_improve >= 5000:
            break

    print(f"Search finished after {iter_count} iterations – best score={best_score:.6f}")

    # Return plain Python list as required by the external grader.
    return [float(v) for v in best_seq]
