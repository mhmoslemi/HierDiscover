"""
Hybrid optimiser for the non‑negative step‑function problem.

The algorithm mixes two complementary ideas:

1️⃣ LP‑guided redistribution – given the current sequence `f`,
   we solve a linear program that finds a non‑negative sequence `g`
   of maximal L₁‑norm while respecting the convolution bound
   `f * g ≤ M` (with `M = max(f*f)`).  Mixing a small amount of `g`
   into `f` provably does not increase the worst convolution value.
   The LP is solved with SciPy’s HiGHS backend using a sparse Toeplitz
   convolution matrix (one constant diagonal per offset).

2️⃣ Stochastic local moves – the usual block‑wise scaling, mass swap,
   Gaussian noise, “flatten” move (targeting the current worst
   convolution index) and an “edge‑push” move that transfers mass from
   the central region to the ends.  Each candidate is first screened
   by a fast FFT‑based estimator; only promising candidates are sent
   to the authoritative ``evaluate_sequence`` function.

The search runs until the allotted time budget expires (a few seconds
are reserved as safety margin).  The best sequence found is returned.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Smallest power of two not smaller than ``x``."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Return the maximal entry of the linear convolution ``seq*seq``."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(conv[: 2 * n - 1].max())


def _score_estimate(seq: np.ndarray) -> float:
    """Fast estimator of the evaluation function (uses FFT)."""
    s = float(seq.sum())
    if s < 0.01:                     # guard against division by zero
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)

# Stochastic move primitives
# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= np.exp(rng.normal(0.0, 0.12))
    if rng.random() < 0.35:           # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one position to another."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)   # up to 20 % of source
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add small zero‑mean Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)


def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Reduce the contribution of the currently worst convolution index."""
    cand = seq.copy()
    n = cand.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())

    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]

    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand

    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))

    delta = cand[src] * float(0.05 + 0.15 * rng.random())   # 5‑%‑20 %
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


def _push_to_edges(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a fraction of mass from the centre to the ends."""
    cand = seq.copy()
    n = cand.size
    w = max(1, n // 20)                     # width of the central window
    centre = n // 2
    start_c = max(0, centre - w // 2)
    end_c = min(n, start_c + w)
    central_idx = np.arange(start_c, end_c, dtype=int)

    # pick a source proportional to its current mass
    probs = cand[central_idx]
    if probs.sum() == 0.0:
        return cand
    src = int(rng.choice(central_idx, p=probs / probs.sum()))

    # destination – either left or right edge block
    if rng.random() < 0.5:
        dst_block = np.arange(0, w, dtype=int)
    else:
        dst_block = np.arange(n - w, n, dtype=int)
    dst_block = np.setdiff1d(dst_block, [src])
    if dst_block.size == 0:
        return cand
    dst = int(rng.choice(dst_block))

    frac = float(rng.random() * 0.3 + 0.1)   # 10 %‑40 %
    delta = cand[src] * frac
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (LP‑guided redistribution)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Given a non‑negative sequence ``f`` (sum fixed) and a bound ``M``,
    solve the LP

        max   Σ_j g_j
        s.t.  (f * g)_k ≤ M   for all k
              g_j ≥ 0

    Returns the optimal ``g`` or ``None`` if the solver fails.
    """
    n = f.size
    # Sparse Toeplitz matrix A where A_{k,j} = f_{k-j} (or 0 if out of range)
    offsets = np.arange(n)                     # 0 … n‑1
    A = scipy.sparse.diags(f, offsets, shape=(2 * n - 1, n), format='csc')

    c = -np.ones(n)                            # minimise –Σ g_j  ⇔ maximise Σ g_j
    b = np.full(2 * n - 1, M)                  # RHS of the convolution constraints
    bounds = [(0.0, 1000.0) for _ in range(n)]

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs')
    if res.success:
        return res.x
    return None

    lp_calls = 0
    max_lp_calls = 30
    lp_probability = 0.05   # chance to try an LP step each iteration

    while time.monotonic() < deadline:
        it += 1

        # ---------- try an LP‑guided redistribution ----------
        cand = None
        if lp_calls < max_lp_calls and rng.random() < lp_probability:
            M = _conv_max(best_seq)                     # current worst convolution
            g = _solve_lp_max_sum(best_seq, M)
            if g is not None:
                sum_g = g.sum()
                if sum_g > 1e-12:
                    # rescale g to have the same L₁‑mass as the current sequence
                    g_scaled = g * (best_seq.sum() / sum_g)

                    # try a handful of mixing coefficients
                    for t in (0.5, 0.25, 0.1, 0.05):
                        tmp = (1.0 - t) * best_seq + t * g_scaled
                        tmp = np.clip(tmp, 0.0, 1000.0)
                        # renormalise (numerical drift after clipping)
                        tmp = tmp * (best_seq.sum() / tmp.sum())
                        est = _score_estimate(tmp)
                        if est < best_score - 1e-12:
                            exact = float(evaluate_sequence(tmp.tolist()))
                            if exact < best_score:
                                best_seq = tmp
                                best_score = exact
                                print(f"[{it}] LP step t={t:.2f} → {best_score:.6f}")
                                break
                lp_calls += 1

        # ---------- otherwise perform a cheap random move ----------
        if cand is None:
            move = rng.choice(move_funcs)
            cand = move(best_seq, rng)

            # keep the total sum (scaling does not influence the objective)
            s = cand.sum()
            if s < 0.01:
                continue
            cand = cand * (best_seq.sum() / s)

            est = _score_estimate(cand)
            if est < best_score - 1e-12:
                exact = float(evaluate_sequence(cand.tolist()))
                if exact < best_score:
                    best_seq = cand
                    best_score = exact
                    print(f"[{it}] improvement → {best_score:.6f}")

        # periodic reporting
        if it % report_interval == 0:
            print(f"[{it}] current best = {best_score:.6f}")

    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    return [float(x) for x in best_seq]
