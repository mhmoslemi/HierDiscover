"""
Propose a low‑value step function.

The algorithm has two stages.

*Stage 1 – LP‑style update*  
Given the current sequence f (with Σf = √(2 n)), compute the current
peak M = max(f∗f).  An inexpensive greedy algorithm solves the linear
program

    max   Σg_j
    s.t.  f∗g ≤ M ,    g_j ≥ 0

by allocating mass to the indices where f is smallest while respecting
the convolution bound.  The resulting g is scaled to the required
ℓ¹‑norm and mixed with f (convex combination).  This step usually
reduces the peak dramatically.

*Stage 2 – fine‑grained donor/receiver hill climbing*  
After the LP‑style updates the sequence is refined with many tiny
mass‑transfer moves: a “donor’’ index that contributes most to the
current maximal convolution entry donates a small fraction of its mass
to a “receiver’’ that contributes the least.  The convolution vector is
updated in O(n) using slice arithmetic, the new peak is evaluated, and
the move is accepted if the peak (and therefore the objective) improves.
Occasional random block perturbations and restarts prevent stagnation.

Both stages respect all security checks of ``evaluate_sequence``:
non‑negative entries, clipping to [0, 1000] and preservation of the
ℓ¹‑norm (which does not affect the objective because the evaluation
function is scale‑invariant).

The routine runs for up to the supplied time budget (default 1000 s)
and returns the best sequence found.
"""

import time

# Helper utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """ℓ¹‑norm that makes the objective equal to the maximal convolution."""
    return np.sqrt(2.0 * n)


def _make_power_law(n: int, alpha: float, shift: float = 0.5) -> np.ndarray:
    """Symmetric decreasing power‑law a_i = (i+shift)^‑α, then ℓ¹‑normalised."""
    a = np.power(np.arange(n, dtype=np.float64) + shift, -alpha)
    a = np.clip(a, 0.0, 1000.0)
    a_sum = a.sum()
    if a_sum > 0.0:
        a *= (_target_sum(n) / a_sum)
    return a


def _block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    # block width in [n/200, n/12]
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirror to keep near‑symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _greedy_lp_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Greedy construction of a feasible vector g solving
        max Σ g_j   subject to   f∗g ≤ M , g ≥ 0.
    The algorithm processes indices of f in ascending order,
    allocating as much mass as possible to each position while
    respecting the remaining slack of the convolution constraints.
    """
    n = f.size
    slack = np.full(2 * n - 1, M, dtype=np.float64)   # M - (f∗g)  , initially M
    g = np.zeros(n, dtype=np.float64)

    # mask for positive entries of f (they bound the step size)
    pos_mask = f > 1e-12
    f_pos = f[pos_mask]

    # processing order: smallest f first
    order = np.argsort(f)

    for j in order:
        # slice of slack that will be affected when we increase g_j
        sl = slack[j : j + n]                 # length n
        # potential step size limited by each active constraint
        potentials = sl[pos_mask] / f_pos     # element‑wise division
        if potentials.size == 0:
            delta = 0.0
        else:
            delta = float(potentials.min())
        if delta <= 0.0:
            continue
        g[j] = delta
        # update all affected slacks
        slack[j : j + n] -= f * delta
        # numerical safety
        slack[j : j + n] = np.maximum(slack[j : j + n], 0.0)

    return g


def _conv_max(seq: np.ndarray) -> tuple[np.ndarray, float, int]:
    """Return (full convolution, peak value, index of the peak)."""
    conv = np.convolve(seq, seq)
    idx = int(conv.argmax())
    return conv, float(conv[idx]), idx

def propose_candidate(seed: int = 123, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best bound.  The routine respects the time budget (default
    1000 seconds) and may be called repeatedly by the judging harness.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine the working length.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))  # supplied by the platform
        if n < 2:
            raise Exception
    except Exception:
        n = 4000  # safe default

    target_sum = _target_sum(n)

    # ------------------------------------------------------------------
    # Initialise the sequence – try a few power‑law exponents and keep
    # the best one (including the supplied construction if available).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # a few random power‑law seeds
    for _ in range(5):
        alpha = rng.uniform(0.45, 0.55)
        cand = _make_power_law(n, alpha)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq = cand
            best_score = sc

    # try the historic construction if it exists
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n:
            parent = np.clip(parent, 0.0, 1000.0)
            parent = parent * (target_sum / parent.sum())
            psc = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(psc) and psc < best_score:
                best_seq = parent.copy()
                best_score = psc
    except Exception:
        pass

    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Phase 1 – LP‑style updates (a handful of heavy iterations).
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    lp_iters = 0
    while time.monotonic() < deadline and lp_iters < 12:
        lp_iters += 1
        conv, M, _ = _conv_max(seq)
        # construct g by greedy LP solution
        g = _greedy_lp_g(seq, M)
        if g.sum() == 0.0:
            break
        # scale g to the prescribed ℓ¹‑norm
        g = g * (target_sum / g.sum())
        # convex combination – start with a modest step size
        t = 0.2 if lp_iters % 2 else 0.1
        cand = (1.0 - t) * seq + t * g
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target_sum / cand.sum())
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            seq = cand
            best_seq = cand.copy()
            best_score = cand_score
            print(f"[LP {lp_iters}] improved score = {best_score:.6f}")

        # occasional random block perturbation inside the LP phase
        if rng.random() < 0.15:
            pert = _block_perturbation(seq, rng)
            pert = pert * (target_sum / pert.sum())
            pscore = float(evaluate_sequence(pert.tolist()))
            if np.isfinite(pscore) and pscore < best_score:
                seq = pert
                best_seq = pert.copy()
                best_score = pscore
                print(f"[LP {lp_iters}] block perturb improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – high‑frequency donor/receiver hill climbing.
    # ------------------------------------------------------------------
    step_frac = 0.12                 # maximal fraction of donor mass moved
    last_improve = time.monotonic()
    iter_counter = 0

    while time.monotonic() < deadline:
        iter_counter += 1

        conv, max_val, k_max = _conv_max(seq)

        # contributions of each index to the current maximal entry
        idx = np.arange(n, dtype=np.int64)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(seq)
        contrib[mask] = seq[idx[mask]] * seq[partner[mask]]

        donor = int(np.argmax(contrib))
        receiver = int(np.argmin(contrib))

        if donor == receiver:
            donor = rng.integers(0, n)
            receiver = rng.integers(0, n)
            while receiver == donor:
                receiver = rng.integers(0, n)

        delta = min(seq[donor], step_frac * seq[donor])
        if delta <= 0.0:
            cand = _block_perturbation(seq, rng)
        else:
            cand = seq.copy()
            cand[donor] -= delta
            cand[receiver] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)

        # re‑normalise to keep the ℓ¹‑norm fixed (scale‑invariance ensures
        # the objective does not change, but it keeps numbers well‑behaved)
        s = cand.sum()
        if s > 0.0:
            cand = cand * (target_sum / s)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            seq = cand
            best_seq = cand.copy()
            best_score = cand_score
            last_improve = time.monotonic()
            step_frac = min(0.30, step_frac * 1.03)
        else:
            step_frac = max(0.001, step_frac * 0.991)

        # occasional random perturbations (escape from plateaus)
        if rng.random() < 0.02:
            cand2 = _block_perturbation(seq, rng)
            cand2 = cand2 * (target_sum / cand2.sum())
            s2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(s2) and s2 < best_score:
                seq = cand2
                best_seq = cand2.copy()
                best_score = s2
                last_improve = time.monotonic()

        # restart if we have been idle for a while
        if time.monotonic() - last_improve > 35.0:
            # fresh power‑law with a new exponent
            alpha = rng.uniform(0.45, 0.55)
            seq = _make_power_law(n, alpha)
            last_improve = time.monotonic()
            step_frac = 0.12
            print(f"[{time.monotonic():.1f}] restart – new α = {alpha:.4f}")

        # occasional status print
        if iter_counter % 250 == 0:
            print(f"[{time.monotonic():.1f}] it={iter_counter}  best_score={best_score:.6f}")

        # early stop if we have already passed the target bound
        if best_score < 1.503:
            print(f"Target bound reached – stopping early (score={best_score:.6f}).")
            break

    # final sanity clip (the evaluator also clips)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Search finished – best score = {best_score:.6f} after {iter_counter} iterations.")
    return [float(x) for x in best_seq.tolist()]
