"""
Propose a step‑function that (heuristically) minimises the evaluation
function ``evaluate_sequence``.

Algorithm
---------

1. **Initial population** – generate a few power‑law candidates
   a_i = (i+offset+1)^(-p) with random exponent, offset and length.
   Keep the best among them and also keep the known global
   ``height_sequence_1`` (if it exists).

2. **Gradient‑style descent on the maximal convolution value**.
   At each iteration:
      * compute the autocorrelation b = a*a,
      * locate the index (or indices) where b attains its maximum,
      * for every position i accumulate a gradient equal to the sum of
        a_{k-i} over all maximal‑index k (these are the terms that
        contribute to the current peak),
      * move the sequence a a little *away* from that gradient
        (a ← a – η·grad) preserving non‑negativity,
      * optionally add a tiny global log‑normal noise,
      * trim trailing values that are essentially zero (they only increase
        the length factor n in the score),
      * accept the new point if it improves the score; otherwise keep it
        as the current point (so the search can continue from a non‑optimal
        neighbourhood).

   The learning rate η decays linearly with the remaining time.

3. **Escapes & restarts** – with a small probability the current point is
   perturbed by the block‑perturbation operator from the original code.
   Occasionally a completely new random power‑law seed is generated.

4. **Return the best sequence** found before the time budget expires.

The method needs only O(n) memory, uses only NumPy and the supplied
``evaluate_sequence`` and runs entirely within the given time budget.
"""

def generate_power_sequence(n: int,
                            exponent: float = 0.5,
                            offset: float = 0.0) -> np.ndarray:
    """Return a non‑negative power‑law sequence a_i = (i+offset+1)^(-exponent)."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.06)))   # ~6 % log‑normal change
    new = seq.copy()
    new[start:stop] *= factor
    if rng.random() < 0.1:                     # occasional symmetrisation
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)


def global_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Very small multiplicative log‑normal noise applied to the whole vector."""
    noise = rng.normal(0.0, 0.02, size=seq.shape)
    factor = np.exp(noise)
    new = seq * factor
    return np.clip(new, 0.0, 1000.0)

    except Exception:
        best_seq = generate_power_sequence(5000, exponent=0.5, offset=0.0)

    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"initial score (baseline) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) quick random exploration of the 3‑dimensional parameter space
    #    (length, exponent, offset).  We reserve roughly the first 5 % of
    #    the time for this phase.
    # ------------------------------------------------------------------
    exploration_deadline = start_time + 0.05 * (deadline - start_time)
    while time.monotonic() < exploration_deadline:
        n = int(rng.integers(2000, 6000))
        exp = float(rng.uniform(0.35, 0.65))
        off = float(rng.uniform(0.0, 15.0))
        seq = generate_power_sequence(n, exponent=exp, offset=off)

        # trim trailing values that are effectively zero – this reduces n
        keep = np.where(seq > 1e-9)[0]
        if keep.size:
            seq = seq[:keep[-1] + 1]

        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = seq
            best_score = score
            print(f"explore better → n={seq.size}, exp={exp:.4f}, off={off:.2f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) gradient‑style descent on the maximal convolution value.
    # ------------------------------------------------------------------
    current_seq = best_seq.copy()
    total_time = deadline - start_time

    while time.monotonic() < deadline:
        # Compute autocorrelation and locate its maximal entries.
        conv = np.convolve(current_seq, current_seq)
        max_b = conv.max()
        # Numerical tolerance – treat values within 1e‑12 relative of the max
        # as active.
        tol = max_b * 1e-12
        active_idx = np.where(conv >= max_b - tol)[0]

        # --------------------------------------------------------------
        # Gradient of the max‑convolution w.r.t. each a_i.
        # For each active index k we add a_{k-i} to grad_i whenever 0≤i<n
        # and 0≤k-i<n.
        # --------------------------------------------------------------
        n = current_seq.size
        grad = np.zeros_like(current_seq)

        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # slice of a that contributes to this k
            # a_slice[j] = current_seq[k - (i_start + j)]   for j = 0..(i_end-i_start)
            idxs = np.arange(i_start, i_end + 1)
            grad[idxs] += current_seq[k - idxs]

        # --------------------------------------------------------------
        # Update rule – move *away* from the gradient.
        # Learning rate decays linearly with the remaining time.
        # --------------------------------------------------------------
        remaining = deadline - time.monotonic()
        lr = 0.12 * (remaining / total_time)   # start ≈0.12, ends near 0
        new_seq = current_seq - lr * grad
        # keep non‑negative numbers and upper bound 1000 (as required by the
        # evaluator).
        new_seq = np.clip(new_seq, 0.0, 1000.0)

        # Occasionally add a very small global perturbation to avoid
        # stagnation.
        if rng.random() < 0.10:
            new_seq = global_noise(new_seq, rng)

        # Trim trailing near‑zero entries – they only increase the length
        # factor n in the score.
        nz = np.where(new_seq > 1e-8)[0]
        if nz.size:
            new_seq = new_seq[:nz[-1] + 1]
        else:
            # safeguard against a completely zero vector
            new_seq = np.array([1.0], dtype=np.float64)

        # --------------------------------------------------------------
        # Evaluate the new candidate.
        # --------------------------------------------------------------
        score = float(evaluate_sequence(new_seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = new_seq
            best_score = score
            elapsed = time.monotonic() - start_time
            print(f"improvement {best_score:.6f} at t={elapsed:.1f}s (n={new_seq.size})")

        # Accept the new point as the current state even if it is not
        # better – this is a simple “always move” policy that lets the
        # gradient explore the landscape.
        current_seq = new_seq

        # --------------------------------------------------------------
        # Occasional escapes:
        #   – block perturbation (large local change)
        #   – full random restart with a fresh power‑law seed
        # --------------------------------------------------------------
        if rng.random() < 0.015:
            cand = block_perturb(current_seq, rng)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_seq = cand
                best_score = sc
                print(f"block‑perturb win → {sc:.6f}")

        if rng.random() < 0.008:
            # fresh random power‑law; keep the best of a few variants
            exp = float(rng.uniform(0.45, 0.55))
            off = float(rng.uniform(0.0, 20.0))
            nl = int(rng.integers(2500, 6000))
            cand = generate_power_sequence(nl, exponent=exp, offset=off)
            cand = block_perturb(cand, rng)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_seq = cand
                best_score = sc
                print(f"restart win → {sc:.6f}")

    # ------------------------------------------------------------------
    # Return the best candidate found (plain Python list as required).
    # ------------------------------------------------------------------
    print(f"final best score = {best_score:.6f}, length = {best_seq.size}")
    return [float(x) for x in best_seq]
```

This implementation combines a fast random seeding phase, a gradient‑descent style optimisation targeting the peak of the autocorrelation, occasional large perturbations and restarts, and trimming of trailing zeroes to keep the length factor `n` small. It respects the 1000 s time budget, uses only NumPy and the provided `evaluate_sequence`, and returns a plain list of non‑negative floats.
