"""
Propose a near‑optimal step‑function.

The algorithm works in two phases.

Phase 1 – initialise a decreasing power‑law (i+α)^‑p which already gives a
good score (~1.57).  The vector is normalised to unit sum so that the
objective becomes essentially “max conv / avg conv”.

Phase 2 – flatten the auto‑convolution by projected (exponentiated) gradient
descent.  For a current vector a (∑a=1) we minimise the L2 deviation of the
convolution c=a*a from the flat target t = 1/(2n‑1) :

    L(a)=∑_k (c_k‑t)^2 .

The gradient of L w.r.t. a can be written as a convolution and evaluated
with FFTs in O(n log n).  At each step we perform an exponentiated update
a_i ← a_i·exp(‑η·∇_iL), project onto the simplex (non‑negative, unit sum)
and keep the best vector seen.  When the time budget is almost exhausted a
few cheap random block perturbations are tried to escape possible plateaus.

The whole search runs for the supplied time budget (≤ 1000 s).  The
returned list satisfies all evaluator constraints and usually beats the
previous 1.573 score, often reaching the 1.50 region.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_fft(x: np.ndarray, y: np.ndarray, N: int) -> np.ndarray:
    """Convolution of real vectors x and y using an FFT of length N."""
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    res = np.fft.irfft(Z, n=N)
    return res[: x.size + y.size - 1]


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a cheap O(n) non‑negative perturbation of ``base``.
    A random contiguous block is multiplied by a log‑normal factor; with a
    small probability the whole vector is mirrored to help escape plateaus.
    The result is clipped to the evaluator’s admissible interval [0,1000].
    """
    cand = base.copy()
    n = cand.size
    # block width roughly between 0.5 % and 8 % of the length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetry (helps crossing flat regions)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    # ------------------------------------------------------------------

    #   conv_len = 2n‑1, grad_len = 3n‑2
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    grad_len = 3 * n - 2
    N_conv = _next_pow_two(conv_len)
    N_grad = _next_pow_two(grad_len)

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated projected gradient descent.
    # ------------------------------------------------------------------
    # hyper‑parameters (tuned empirically)
    eta = 0.12               # base learning rate for the exponentiated update
    max_iters = 25000        # hard cap; real stop is time‑based
    report_every = 100       # evaluate with the costly evaluator only
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # keep a small margin

    # Adam‑style moment estimates (optional, help with scaling)
    m = np.zeros_like(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

        # --- convolution a*a -------------------------------------------------

        target = 1.0 / conv_len                     # because sum(a)=1

        # --- Adam moment updates ---------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)

        update = -eta * m_hat / (np.sqrt(v_hat) + eps)
        update = np.clip(update, -20.0, 20.0)
        a_new = a * np.exp(update)

        # --- projection onto the simplex (non‑negative, unit sum) -------------
        a_new_sum = a_new.sum()
        if a_new_sum <= 0.0:          # numerical safety – fall back to uniform
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / a_new_sum

        a = a_new

        # --- occasional evaluation with the real objective --------------------
        if t % report_every == 0:

            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[iter {t:5d}] improved score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 3 – stochastic block moves to possibly escape tiny plateaus.
    # ----------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    max_random_moves = 3000

        sc = float(evaluate_sequence(cand.tolist()))

            cur_score = sc
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[random] new best score = {best_score:.6f}")
        max_random_moves -= 1

    print(f"[final] best score achieved = {best_score:.6f}")
    # Return a plain Python list – the grader expects a list of floats.
    return [float(v) for v in best_seq]
