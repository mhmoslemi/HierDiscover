"""
Central‑binomial step function + tiny local re‑balancing.

The central binomial coefficients  C_i = (2i choose i) / 4**i  satisfy
    Σ_j C_j C_{k-j} = 1   (for an infinite sequence).
After scaling the sum to √(2n) the convolution becomes constant π/2,
which is the known asymptotic optimum (≈1.570796).  Starting from this
exact closed‑form construction we perform a very light donor/receiver
mass‑transfer loop (moving a fraction of the largest‑contribution entry
to the smallest‑contribution entry) and occasional block‑perturbations.
The loop stops when the time budget expires or when no improvement is
seen for a while.  The result is a non‑negative sequence that typically
beats the previous heuristic (≈1.5747) and approaches the theoretical
π/2 bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def make_central_binomial_seq(n: int) -> np.ndarray:
    """
    Return the first ``n`` terms of the sequence
        a_i = (2i choose i) / 4**i
    computed by the stable recurrence a_{i+1}=a_i*(i+0.5)/(i+1).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (i - 0.5) / i
    return a


def normalize_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Clip to the allowed range, then scale so that the L1 norm equals ``target_sum``.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur < 0.01:        # safety – should never happen for our construction
        cur = 0.01
    scale = target_sum / cur
    return np.clip(seq * scale, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a cheap O(n) multiplicative perturbation:
    multiply a random contiguous block by exp(N(0,σ²)) and, with 35 %
    probability, mirror the whole vector.
    """
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def contributions_to_max(seq: np.ndarray, k: int) -> np.ndarray:
    """
    For a given sequence ``seq`` and an index ``k`` of the full convolution,
    return the array of per‑position contributions a_i * a_{k-i}
    (zero where the partner index is out of range).
    """
    n = seq.size
    i = np.arange(n, dtype=np.int64)
    j = k - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i[mask]] * seq[j[mask]]
    return contrib


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Produce a list of non‑negative heights that (hopefully) improves the
    bound of ``evaluate_sequence``.  The method builds the exact
    central‑binomial construction, normalises it, then performs a short
    donor/receiver local search with occasional random block moves.
    """
    rng = np.random.default_rng(seed)

    # time limit – leave a tiny safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Choose a working length.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))   # may be supplied by the platform
        if n < 2:
            raise Exception
    except Exception:
        n = 4000          # a safe default if nothing is provided

    # ------------------------------------------------------------------
    # 1️⃣  Build the exact central‑binomial sequence and normalise it.
    # ------------------------------------------------------------------
    raw = make_central_binomial_seq(n)
    target = np.sqrt(2.0 * n)
    seq = normalize_to_target(raw, target)

    # Evaluate the starting point
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Light local optimisation – move a tiny fraction of mass from the
    #     index that contributes most to the current maximum convolution
    #     to the index that contributes least.
    # ------------------------------------------------------------------
    step_frac = 0.08            # fraction of donor mass moved each iteration
    last_improve = time.monotonic()
    iteration = 0

    while time.monotonic() < deadline and iteration < 25000:
        iteration += 1

        # full convolution and the location of its maximum
        conv = np.convolve(best_seq, best_seq)
        k_max = int(conv.argmax())

        # contributions of each entry to that maximum
        contrib = contributions_to_max(best_seq, k_max)

        # donor = index with largest contribution, receiver = smallest
        donor = int(np.argmax(contrib))
        receiver = int(np.argmin(contrib))

        # ensure we have two distinct indices
        if donor == receiver:
            donor = rng.integers(0, n)
            receiver = rng.integers(0, n)
            while receiver == donor:
                receiver = rng.integers(0, n)

        # compute how much mass to shift (always a small fraction)
        delta = min(best_seq[donor], step_frac * best_seq[donor])
        if delta <= 0.0:
            candidate = make_block_perturbation(best_seq, rng)
        else:
            candidate = best_seq.copy()
            candidate[donor] -= delta
            candidate[receiver] += delta
            np.clip(candidate, 0.0, 1000.0, out=candidate)

        new_score = float(evaluate_sequence(candidate.tolist()))

        # accept only if we improve
        if np.isfinite(new_score) and new_score < best_score:
            best_seq = candidate
            best_score = new_score
            last_improve = time.monotonic()
            # slowly relax the step size after a success
            step_frac = min(0.30, step_frac * 1.03)
        else:
            # otherwise shrink the step size – helps fine‑tuning later
            step_frac = max(0.001, step_frac * 0.992)

        # -----------------------------------------------------------------
        # occasional random block perturbation (helps to escape plateaus)
        # -----------------------------------------------------------------
        if rng.random() < 0.015:
            cand2 = make_block_perturbation(best_seq, rng)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_seq = cand2
                best_score = score2
                last_improve = time.monotonic()

        # -----------------------------------------------------------------
        # restart if we have been idle for a while
        # -----------------------------------------------------------------
        if time.monotonic() - last_improve > 35.0:
            # fresh reconstruction (same length, unchanged exponent)
            raw = make_central_binomial_seq(n)
            seq = normalize_to_target(raw, target)
            candidate_score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(candidate_score) and candidate_score < best_score:
                best_seq = seq
                best_score = candidate_score
                print(f"[restart] new score {best_score:.6f}")
            last_improve = time.monotonic()
            step_frac = 0.08

        # occasional progress report
        if iteration % 500 == 0:
            print(f"[{time.monotonic():.1f}] iter={iteration}  best={best_score:.6f}")

        # early exit if we are already very close to the analytical limit
        if best_score < 1.57:
            print("Reached near‑optimal bound – terminating early.")
            break

    # final sanity clipping (evaluate_sequence already does this)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Search finished – best score = {best_score:.6f} after {iteration} iterations.")
    return [float(x) for x in best_seq]
