It starts from a power‑law seed, optionally refines the exponent/offset by a tiny grid search, then runs a **soft‑max‑gradient descent** that directly drives down the peak of the auto‑convolution while keeping the total mass fixed.  
 sporadic random block perturbations are injected to avoid getting trapped in a flat region, and the overall best sequence found before the time budget expires is returned.

```python
"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1. Choose the length n (taken from the provided `height_sequence_1`
   if it exists, otherwise 5000).

2. Initialise a non‑negative sequence a_i = (i + offset + 1)^(-exponent)
   (power‑law).  A small grid of exponents/offsets is scanned and the
   best seed (lowest score) is kept.

3. Gradient descent on a smooth surrogate of the max‑convolution:
   * Let c = a * a be the full auto‑convolution.
   * Approximate max(c) by a soft‑max:
         w_k = exp(β·(c_k−max(c)))   (β grows slowly)
         w ← w / sum(w)
   * Gradient of ϕ(a)=∑_k w_k·c_k is
         ∇ϕ = 2·Hankel(w)·a,
     implemented efficiently by a reversed convolution.
   * Update a multiplicatively to keep positivity:
         a ← a * exp(−η·∇ϕ),   then renormalise a to total mass = 1.
   * β is increased each iteration (sharpening the surrogate),
     η is scaled to the largest gradient component.

4. Every few hundred iterations a random block perturbation (identical to the
   one used in the earlier hill‑climber) is applied to help escape flat
   plateaus.

5. The best sequence (lowest true `evaluate_sequence` value) seen during the
   run is returned as a plain Python list.

The implementation uses only NumPy (and optionally SciPy for a fallback
LP‑based refinement, which is not needed for the main search).  All
operations are O(n log n) per iteration and fit comfortably within the
1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def generate_power_sequence(n: int, exponent: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Return a non‑negative power‑law sequence a_i = (i+offset+1)^(-exponent)."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    # The evaluator clamps to [0,1000]; our values are far below that.
    return seq


def evaluate_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective without the overhead of the provided
    `evaluate_sequence`.  Returns the same value (up to the final clipping
    performed by the evaluator) for non‑negative, sub‑1000 sequences.
    """
    n = seq.size
    # Full linear convolution (same result as np.convolve(seq, seq))
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:  # keep behaviour identical to the guard in evaluate_sequence
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block (as in the old code)."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.06)))  # ~6 % log‑normal change
    new = seq.copy()
    new[start:stop] *= factor
    if rng.random() < 0.1:  # occasional symmetrisation
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main proposal function
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (heuristically) minimises
    `evaluate_sequence`.  The routine runs for up to approx. `budget_s`
    seconds and prints progress information.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # Leave a small safety margin for the host environment
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))  # pylint: disable=undefined-variable
    except Exception:
        n = 5000

    # ------------------------------------------------------------------
    # 1) Small grid search over exponent / offset to obtain a good seed
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    exponent_grid = np.linspace(0.48, 0.52, 9)   # finer than the old grid
    offset_grid   = np.linspace(0.0, 2.0, 9)

    for exp in exponent_grid:
        for off in offset_grid:
            cand = generate_power_sequence(n, exponent=exp, offset=off)
            score = evaluate_score(cand)
            if score < best_score:
                best_seq = cand.astype(np.float64)
                best_score = score
                elapsed = time.monotonic() - start_time
                print(f"[seed] exponent={exp:.5f}, offset={off:.3f} → score={score:.6f} (t={elapsed:.1f}s)")
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    # Fall back to a uniform vector if for any reason the grid failed
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = evaluate_score(best_seq)

    # Normalise to unit sum – the objective is scale‑invariant
    best_seq = best_seq / best_seq.sum()

    # ------------------------------------------------------------------
    # 2) Soft‑max gradient descent on the max‑convolution
    # ------------------------------------------------------------------
    beta = 1.0          # start of soft‑max temperature
    beta_max = 250.0
    beta_factor = 1.001
    lr_base = 0.02      # base learning‑rate factor

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # Full auto‑convolution (fast enough for n≈5k)
        conv = np.convolve(best_seq, best_seq)
        max_conv = conv.max()

        # ----- soft‑max weights focusing on the peak -----
        # stability: subtract the max before exponentiation
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:   # extremely unlikely, guard against division by zero
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # ----- gradient of the soft‑max surrogate -----
        # Using the identity ∇_a Σ_k w_k (a*a)_k = 2·Hankel(w)·a
        # where the Hankel matrix multiplication can be performed by a reversed convolution.
        w_rev = w[::-1]
        rev_conv = np.convolve(w_rev, best_seq)          # length = (2n-1) + n - 1
        L = conv.size                                    # = 2n-1
        # Grab the n entries corresponding to indices 0 … n‑1
        grad = 2.0 * rev_conv[L - 1 - np.arange(n)]

        # ----- adaptive step size -----
        # Scale step so that the largest component changes by at most ~lr_base
        max_abs = np.max(np.abs(grad)) + 1e-12
        eta = lr_base / max_abs

        # Multiplicative update (keeps positivity automatically)
        new_seq = best_seq * np.exp(-eta * grad)

        # Renormalise to unit sum
        new_seq_sum = new_seq.sum()
        if new_seq_sum == 0.0:        # pathological, fall back to previous sequence
            new_seq = best_seq.copy()
        else:
            new_seq = new_seq / new_seq_sum

        # Occasionally inject a random block perturbation (helps escape plateaus)
        if iteration % 200 == 0 and rng.random() < 0.15:
            new_seq = block_perturb(new_seq, rng)

        # Compute (exact) evaluation score using the fast helper.
        cur_score = evaluate_score(new_seq)

        # Accept if we improve the true objective
        if cur_score < best_score:
            best_seq = new_seq
            best_score = cur_score
            elapsed = time.monotonic() - start_time
            print(f"[iter {iteration:5d}] new best {best_score:.6f}  (t={elapsed:.1f}s, β={beta:.2f})")

        # Update soft‑max temperature (sharpen) – never exceed β_max
        beta = min(beta * beta_factor, beta_max)

        # Safety break if progress stalls for a long time
        if iteration % 500 == 0 and (time.monotonic() - start_time) > (budget_s * 0.9):
            # we have already spent most of the budget; stop early.
            break

    # ------------------------------------------------------------------
    # 3) Final polishing with a few classic hill‑climbing steps
    # ------------------------------------------------------------------
    # Use the same cheap block perturbations as the old baseline for a short
    # finishing pass (typically a few seconds).
    hp_deadline = deadline - 0.5   # leave a little margin
    while time.monotonic() < hp_deadline:
        cand = block_perturb(best_seq, rng)
        cand_score = evaluate_score(cand)
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            elapsed = time.monotonic() - start_time
            print(f"[hill] improvement {best_score:.6f} (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # Return a plain Python list (the evaluator expects a list of floats)
    # ------------------------------------------------------------------
    final_seq = [float(x) for x in best_seq]
    print(f"Finished. Best score = {best_score:.6f}, length = {len(final_seq)}")
    return final_seq
```
