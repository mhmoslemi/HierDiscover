def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    import time, numpy as np, math, random

    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # margin

    # Determine n from existing height_sequence_1 or fallback
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = base.shape[0]
    except Exception:
        n = 5000
        base = None

    # Initial sequence: arcsine distribution
    a = arcsine_sequence(n)

    # if a base sequence given, incorporate it by averaging
    if base is not None:
        # ensure sum =1
        base = np.maximum(base, 0.0)
        if base.sum() == 0:
            base = np.ones_like(base)
        base = base / base.sum()
        # blend a and base a little
        a = 0.5 * a + 0.5 * base
        a = proj_simplex(a)

    cur = a.copy()
    cur_score = float(2.0 * n * np.max(self_convolution(cur)))  # sum=1
    best = cur.copy()
    best_score = cur_score

    # Parameters
    T = 0.001  # temperature start (adjust based on score)
    T_factor = 0.999  # cooling per iteration
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # Decide move
        if rng.random() < 0.7:
            # swap move
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]  # swap
            # sum unchanged, nonneg satisfied
        else:
            # block scaling move
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = rng.integers(0, max(1, n - width + 1))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.08)))  # moderate scale
            cand = cur.copy()
            cand[start:stop] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand / cand.sum()  # re-normalize to simplex

        # evaluate
        cand_score = float(2.0 * n * np.max(self_convolution(cand)))

        # acceptance
        delta = cand_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"Iter {it:6d} | new best {best_score:.6f}")
                no_improve = 0
        else:
            no_improve += 1

        # cooling
        T *= T_factor
        it += 1

        # occasional reset to best to avoid drift
        if it % 2000 == 0:
            cur = best.copy()
            cur_score = best_score
            T = max(T, 1e-6)  # ensure temperature not too low

        # occasional larger random re-perturbation if stuck
        if no_improve > 5000:
            # random perturb around best
            cand = best.copy()
            # apply random jitter across whole vector
            cand += rng.normal(scale=0.005, size=n)
            cand = proj_simplex(cand)
            cur = cand
            cur_score = float(2.0 * n * np.max(self_convolution(cur)))
            no_improve = 0

    print(f"Finished after {it} iterations, best score = {best_score:.6f}")
    return [float(x) for x in best]

"""
Simulated‑annealing + permutation search for the minimisation of
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm proceeds in two stages.
1.   A large number of random permutations of the analytically known
     “arcsine’’ distribution are examined; because the objective only
     depends on the multiset of coefficients, a favourable ordering can
     considerably lower the peak of the self‑convolution.
2.   The best ordering found is refined by a projected gradient step on a
     smooth approximation of the max‑operator (log‑sum‑exp).  The step is
     performed on the probability simplex (non‑negative, sum = 1) using a
     fast FFT based convolution to obtain the gradient.
Both stages respect the 1000 s time limit; intermediate progress is printed.
"""
import time
import math
import numpy as np

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (v >= 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_sequence(n: int) -> np.ndarray:
    """Discrete arcsine distribution, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a

def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself using FFT."""
    n = x.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) minimise the score."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------------
    # Determine length n and a starting multiset of coefficients.
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
        # normalise to a probability vector (the scale does not affect the score)
        base = np.maximum(base, 0.0)
        if base.sum() == 0.0:
            base = np.ones_like(base)
        base = base / base.sum()
        multiset = base.copy()
    except Exception:
        # fallback length – moderate size that runs many iterations fast
        n = 5000
        multiset = arcsine_sequence(n)

    # ----------------------------------------------------------------------
    # Helper to compute the score for a probability vector (sum = 1)
    # ----------------------------------------------------------------------
    def score_of(p):
        conv = self_convolution(p)
        return float(2.0 * n * np.max(conv))

    # ----------------------------------------------------------------------
    # Stage 1 – exhaustive random permutation search.
    # ----------------------------------------------------------------------
    best = multiset.copy()
    best_score = score_of(best)
    perm_iter = 0
    half_deadline = time.monotonic() + (deadline - time.monotonic()) / 2.0
    while time.monotonic() < half_deadline:
        cand = rng.permuted(multiset)
        cand_score = score_of(cand)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            print(f"Perm {perm_iter:6d} | new best = {best_score:.6f}")
        perm_iter += 1

    # ----------------------------------------------------------------------
    # Stage 2 – gradient‑based refinement on a smooth approximation of max.
    # ----------------------------------------------------------------------
    a = best.copy()            # current working point
    cur_score = best_score
    step = 0.07                # initial step size
    step_decay = 0.9999
    beta = 0.5                 # smoothness of the max‑approximation
    beta_inc = (30.0 - beta) / max(1, int((deadline - time.monotonic()) / 0.1))
    it = 0
    while time.monotonic() < deadline:
        # 1) convolution and smooth weights
        conv = self_convolution(a)
        max_val = conv.max()
        # log‑sum‑exp weights (more diffuse for small beta)
        w = np.exp(beta * (conv - max_val))
        w /= w.sum()

        # 2) gradient of the smooth max (via cross‑correlation)
        #    g_j = 2 * sum_i w_i * a_{i-j}  (with indices out of range ignored)

        fw = np.fft.rfft(w_rev, n=size)
        rev_conv = np.fft.irfft(fa * fw, n=size)   # linear convolution
        # extract the needed slice: g_j = 2 * rev_conv[2n-2 - j]
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * rev_conv[idx]

        # 3) gradient step + projection onto simplex
        cand = a - step * g

        cand_score = score_of(cand)

            best_score = cand_score

        # 6) annealing of parameters

        beta = min(beta + beta_inc, 30.0)
        it += 1

        # occasional random jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.002, size=n)
            a = proj_simplex(a + jitter)
            cur_score = score_of(a)

    print(f"Finished after {it} gradient steps (plus {perm_iter} permutations). "
          f"Best score = {best_score:.6f}")
    # Convert to plain Python floats as required by the evaluation harness
    return [float(x) for x in best]
