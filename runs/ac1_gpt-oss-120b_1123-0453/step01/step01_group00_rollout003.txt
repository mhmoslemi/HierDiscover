    function.  The algorithm works in two phases:

    1.  An LP‑based iterative improvement (the same idea as the reference
        code) is run for a few different lengths, starting from random

            max   Σ g_i

        ℓ¹‑norm equal to √(2 n) and mixed with *f*.  Repeating this step

Propose a low‑value step sequence.

The search runs in two stages.

1️⃣  LP‑based improvement (the same principle as the reference code):
    - Keep the ℓ¹‑norm fixed to √(2·n) so that the evaluation reduces to
      the worst entry of the auto‑convolution.
    - For a current vector f compute M = max(f * f).
    - Solve the linear program
          maximise   Σ g_i
          subject to (f * g)_k ≤ M   for every shift k,
                     g_i ≥ 0 .
      The solution g is rescaled to the same ℓ¹‑norm and mixed with f:
          f ← (1‑t)·f + t·g .
      Repeating the step strictly drives M (hence the score) down.
    - The procedure is executed for several lengths (200 … 600) and a few
      random restarts.  The best vector ever seen is stored.

2️⃣  Stochastic hill‑climbing refinement:
    - When the time budget is almost exhausted a simple local‑search
      perturbs the best vector with Gaussian noise, accepts improvements
      and slowly reduces the perturbation size.

Both stages respect the security checks of `evaluate_sequence`
(clipping to [0, 1000] and forcing non‑negativity).  The function returns
the best sequence (plain Python list) discovered before the deadline.
"""

def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2·n‑1) × n matrix A such that (A @ g)[k] = (f * g)[k]."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _score(seq: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence (expects a Python list)."""
    # evaluate_sequence performs its own clipping, but we keep the sequence as a
    # list of Python floats to be safe.
    return float(evaluate_sequence(seq.tolist()))


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  step_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing starting from `start_seq`.
    Random Gaussian perturbations are accepted if they improve the score.
    The step size decays slowly; the routine stops when `deadline` is reached.
    """
    rng = np.random.default_rng()
    best = start_seq.copy()
    best_score = _score(best)
    step = step_initial

    while time.monotonic() < deadline:
        cand = best + rng.normal(scale=step, size=best.shape)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # exponential decay of the step size
        step *= 0.9999
        if step < 1e-7:
            step = step_initial
    return best, best_score

                break

            # Initialise a random positive vector with the required ℓ¹‑norm.
            f = rng.random(n) + 0.1
            f = f / np.sum(f) * target_sum

            # Mixing weight: start relatively aggressive.
            t = 0.5

            # At most this many LP iterations – we abort earlier if time runs out.
            for it in range(120):
                if time.monotonic() >= deadline:
                    break

                # Current worst value of the auto‑convolution.
                M = float(np.max(np.convolve(f, f)))

                # Build the convolution matrix for the constraints (f * g) ≤ M.
                A = _conv_matrix(f)

                if not _HAS_CVX:
                    # No linear‑programming backend – quit the LP phase.
                    break

                # Linear program: maximise Σ g_i  subject to A·g ≤ M, g ≥ 0.
                g = cp.Variable(n, nonneg=True)
                constraints = [A @ g <= M]
                prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

                # ECOS is fast for dense problems; fall back to SCS on failure.
                try:
                    prob.solve(solver=cp.ECOS, warm_start=True,
                               max_iters=5000, abstol=1e-8, reltol=1e-6)
                except Exception:  # pragma: no cover
                    prob.solve(solver=cp.SCS, warm_start=True,
                               max_iters=15000, verbose=False)

                if g.value is None:
                    # LP failed – abort this restart.
                    break

                g0 = np.maximum(g.value, 0.0)
                sum_g = np.sum(g0)
                if sum_g < 1e-12:
                    break

                # Rescale so that ℓ¹‑norm matches the target.
                g_rescaled = g0 * (target_sum / sum_g)

                # Convex combination keeps the ℓ¹‑norm exactly at target_sum.
                f_new = (1.0 - t) * f + t * g_rescaled
                f_new = np.clip(f_new, 0.0, 1000.0)

                # Evaluate the new candidate.
                sc = _score(f_new)
                if sc < best_score:
                    best_score = sc
                    best_seq = f_new.copy()
                    print(f"[LP] n={n} restart={restart} it={it} score={sc:.6f} M={M:.6f}")

                # Early exit if we have reached the target bound.
                if best_score < 1.503:
                    print("[LP] Desired target achieved – exiting early.")
                    return best_seq.tolist()

                # Adjust the mixing weight if progress stalls.
                if it % 15 == 0 and it > 0:
                    recent = _score(f_new)
                    if recent > best_score * 1.001:
                        t = max(0.1, t * 0.8)

                f = f_new

    # --------------------------------------------------------------
    # Phase 2 – stochastic refinement (fallback / final polishing)
    # --------------------------------------------------------------
    if best_seq is None:
        # Deterministic safe fallback – a uniform vector of moderate length.
        n = 200
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_score = _score(best_seq)

    # Use whatever time is left for a simple hill climbing.
    if time.monotonic() < deadline:
        refined_seq, refined_score = _local_search(best_seq, deadline,
                                                   step_initial=0.5)
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"[local] improved to {best_score:.6f}")

    # Return a plain Python list of floats as required.
    return [float(x) for x in best_seq]
