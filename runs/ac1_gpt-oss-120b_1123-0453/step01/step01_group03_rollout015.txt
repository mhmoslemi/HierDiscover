def project_to_simplex(v, s):
    # v is 1d array, s >0
    # returns w such that w >=0, sum w = s, and w is closest in Euclidean distance.
    # Implementation from Duchi et al. 2008.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

assistantfinal**Solution Overview**

The score that `evaluate_sequence` returns is, after the standard normalisation  
`Σ aᵢ = √(2 n)`, exactly the largest value of the self‑convolution `a*a`.  
Hence we have to find a non‑negative vector whose *peak* of `a*a` is as
small as possible.

The best known analytical construction is a geometric progression,
derived from the equation `eᵗ = 1+2t`.  It gives a score ≈ 1.57 for the
current test size (≈ 4500).  To go below the target 1.503 we explore more
flexible families:

* **Geometric** – one parameter `r`.
* **Power‑law** – `aᵢ ∝ (i+1)^{‑p}`.
* **Shifted power‑law** – `aᵢ ∝ (i+α)^{‑p}` (the extra shift `α>0`
  reduces the first few large entries and lowers the convolution peak).
* **Stretched exponential** – `aᵢ ∝ exp(‑(i/γ)ᵝ)`.
* **Sparse random** – a few large spikes.

For each candidate we scale it to the required ℓ₁‑norm,
compute the maximal convolution with a fast NumPy routine and keep the
best vector seen so far.

A small **gradient step** is also applied occasionally: the gradient of the
maximal convolution entry is `gᵢ = aᵢ + a_{k‑i}` (with `k` the index of the
current peak).  A short move in the negative gradient direction,
followed by projection onto the simplex (non‑negative, fixed sum),
often reduces the peak further.

The search runs until the time budget (≈ 1000 s) expires, trying several
different sequence lengths (a fraction of the supplied `height_sequence_1`
and a few larger multiples) and a mixture of deterministic optimisation
(golden‑section for the one‑parameter families) and stochastic sampling.

The final sequence is returned as a plain Python list; a final call to the
official `evaluate_sequence` guarantees that the wrapper security checks are
applied.

"""
propose_candidate
=================

Search for a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  The score after the standard ℓ₁‑normalisation
(`Σ a = sqrt(2·n)`) is exactly the peak of the self‑convolution.
The algorithm explores several parametric families (geometric,
power‑law, shifted power‑law, stretched exponential, sparse random),
optimises their parameters (golden‑section for 1‑D families,
random search for 2‑D families) and performs occasional gradient‑descent
updates.  The best sequence found within the time budget is returned.

# ----------------------------------------------------------------------
# Fast evaluator (identical to the official one but without checks)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Peak of the self‑convolution after clipping to [0,1000] and
    normalisation (the clipping never fires for the vectors we build)."""
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(arr, arr)
    return float(conv.max())

# Simple helpers

def _project_to_simplex(v: np.ndarray, total: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = total}."""
    if total <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - total
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _normalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to the legal interval and rescale to the required ℓ₁‑norm."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s > 0.0:
        seq = seq * (target_sum / s)
    else:
        seq = np.full_like(seq, target_sum / len(seq))
    return seq

# Parametric families

def _geometric_seq(r: float, n: int, target_sum: float) -> np.ndarray:
    """aᵢ = r**i, scaled to Σa = target_sum."""
    idx = np.arange(n, dtype=np.float64)
    a = np.power(r, idx, dtype=np.float64)
    # safety for r≈0 (all zeros)
    if a.sum() == 0.0:
        a = np.ones(n, dtype=np.float64)
    a = a * (target_sum / a.sum())
    return a


def _powerlaw_seq(p: float, n: int, target_sum: float) -> np.ndarray:
    """aᵢ ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(idx, -p, dtype=np.float64)
    a = a * (target_sum / a.sum())
    return a


def _offset_powerlaw_seq(offset: float, p: float,
                         n: int, target_sum: float) -> np.ndarray:
    """aᵢ ∝ (i + offset)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + offset
    a = np.power(idx, -p, dtype=np.float64)
    a = a * (target_sum / a.sum())
    return a


def _stretched_exp_seq(gamma: float, beta: float,
                       n: int, target_sum: float) -> np.ndarray:
    """aᵢ ∝ exp( - (i / gamma)^{beta} )."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-np.power(idx / gamma, beta, dtype=np.float64))
    a = a * (target_sum / a.sum())
    return a


def _sparse_random_candidate(n: int, target_sum: float,
                             rng: np.random.Generator) -> np.ndarray:
    """A few non‑zero entries drawn from an exponential law."""
    k = rng.integers(max(1, int(n * 0.05)), max(2, int(n * 0.25) + 1))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a = a * (target_sum / a.sum())
    return a

# One‑dimensional optimisation (golden‑section)
# ----------------------------------------------------------------------
def _golden_section(func, lo: float, hi: float, iters: int = 30):
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    for _ in range(iters):
        c = lo + (1 - phi) * (hi - lo)
        d = lo + phi * (hi - lo)
        if func(c) < func(d):
            hi = d
        else:
            lo = c
    return (lo + hi) / 2


def _optimise_geometric(n: int, target_sum: float):
    """Find the best r for the geometric family."""
    def f(r):
        return _fast_score(_geometric_seq(r, n, target_sum))
    # search interval (0, 1).  0.999999 gives almost the uniform case.
    r_opt = _golden_section(f, 0.0, 0.999999, iters=35)
    seq = _geometric_seq(r_opt, n, target_sum)
    return seq, _fast_score(seq)


def _optimise_offset_powerlaw(n: int, target_sum: float, rng,
                              time_budget: float = 0.4):
    """Random search for (offset, p) around the 1/√i shape."""
    deadline = time.monotonic() + time_budget
    best_seq = None
    best_score = float("inf")
    # keep the current best parameters for a tiny local refinement
    best_offset = 0.0
    best_p = 0.5
    while time.monotonic() < deadline:
        # sample roughly around the known good region
        offset = rng.exponential(scale=1.0) + 0.1  # >0
        p = rng.uniform(0.3, 0.7)
        seq = _offset_powerlaw_seq(offset, p, n, target_sum)
        score = _fast_score(seq)
        if score < best_score:

            best_offset, best_p = offset, p
    # a few focused refinements

        offset = max(0.0, best_offset + rng.normal(0.0, 0.2))
        p = max(0.01, best_p + rng.normal(0.0, 0.05))
        seq = _offset_powerlaw_seq(offset, p, n, target_sum)
        score = _fast_score(seq)
        if score < best_score:
            best_seq, best_score = seq, score
            best_offset, best_p = offset, p
    return best_seq, best_score


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)_k at the maximal shift."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(conv.argmax())
    # gradient: g[i] = a[i] + a[k-i]  (if k-i inside the range)
    grad = a.copy()
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    j = k - idx
    grad[idx] += a[j]
    return grad

            offset = rng.exponential(scale=0.5) + 0.05
            p = rng.uniform(0.3, 0.7)
            cand = _offset_powerlaw_seq(offset, p, n, target_sum)

        elif mode == 3:                    # stretched exponential
            beta = rng.uniform(0.5, 2.0)
            gamma = rng.uniform(0.5 * n, 2.0 * n)
            cand = _stretched_exponential_seq(gamma, beta, n, target_sum)

        elif mode == 4:                    # fresh sparse random pattern
            cand = _sparse_random_candidate(n, target_sum, rng)

        elif mode == 5 and best_seq is not None and best_seq.shape[0] == n:
            # small multiplicative block perturbation of the current best
            cand = best_seq.copy()
            width = rng.integers(1, max(2, n // 50))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.09))
            cand[start:start + width] *= factor

        else:                              # random redistribution of mass
            cand = _sparse_random_candidate(n, target_sum, rng)

        cand = _normalise(cand, target_sum)
        cur_score = _fast_score(cand)

        if cur_score < best_score:
            best_seq, best_score = cand, cur_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:5.1f}s] new best (n={n}) = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Occasionally perform a tiny gradient‑descent step on the current best
        # ------------------------------------------------------------------
        if best_seq is not None and rng.random() < 0.07:
            grad = _max_conv_gradient(best_seq)
            step = 0.04 * rng.random()   # a small learning rate
            new_seq = best_seq - step * grad
            new_seq = _normalise(new_seq, target_sum)
            new_score = _fast_score(new_seq)
            if new_score < best_score:
                best_seq, best_score = new_seq, new_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:5.1f}s] gradient step improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector as a plain Python list.
    # ------------------------------------------------------------------
    final = [float(v) for v in best_seq.tolist()]
    # final check with the official evaluator (applies the security wrapper)
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f}")
    return final
