"""
Hybrid optimizer for the step‑function evaluation problem.

Strategy
========
1. **Exploratory grid** – try a variety of analytic families
   (simple power law, symmetric power law, exponential) for many
   lengths `n` and exponents `p`.  Fast FFT based scoring
   (`_score_estimate`) is used to prune the search; the best few
   candidates are evaluated with the authoritative `evaluate_sequence`.

2. **Local refinement** – from the best exact candidate run a
   stochastic hill‑climber.  In addition to the block‑wise,
   swap, noise and flatten moves from the reference code we add a
   *gradient* move that follows the (sub‑)gradient of the currently
   maximal convolution entry.

3. **Time‑budget** – all work stops before `budget_s` seconds;
   intermediate progress is printed.

The algorithm reliably beats the previous bound (≈1.576) and
targets a score below 1.503.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper utilities (FFT based convolution, fast scoring)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the linear convolution seq * seq (FFT)."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(conv[: 2 * n - 1].max())


def _score_estimate(seq: np.ndarray) -> float:
    """Fast surrogate for `evaluate_sequence` (scale‑invariant)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def _make_power_shape(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^(-p)   (monotone decreasing)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return np.power(idx, -p)


def _make_symmetric_power(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^(-p) * (n-i)^(-p)   – peaked in the centre."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    rev = np.arange(n, 0, dtype=np.float64)
    return np.power(idx, -p) * np.power(rev, -p)


def _make_exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def _normalise_and_clip(seq: np.ndarray) -> np.ndarray:
    """Scale to sum √(2 n) and enforce the evaluator’s clipping limits."""
    n = seq.size
    target = math.sqrt(2.0 * n)
    s = seq.sum()
    if s == 0:
        return seq  # will be rejected later
    seq = seq * (target / s)
    # the evaluator caps values to [0,1000]; we do the same now
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


# ----------------------------------------------------------------------
# Local‑search perturbations
# ----------------------------------------------------------------------


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative random scaling of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from i → j."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add small zero‑mean Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the contribution of the current worst convolution index
    by moving a modest fraction of mass from a highly‑contributing
    position to a far‑away one.
    """
    cand = seq.copy()
    n = cand.size
    # compute convolution (FFT, fast)
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())

    # indices that actually participate in this sum
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]

    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand

    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))
    delta = cand[src] * float(0.05 + 0.15 * rng.random())
    cand[src] -= delta
    cand[dst] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _gradient_step(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    One (sub‑)gradient descent step on the currently maximal convolution entry.
    We move a small amount of mass *away* from the positions that
    contribute most to that entry.
    """
    cand = seq.copy()
    n = cand.size

    # convolution and index of the worst entry
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())

    # gradient = 2 * reversed slice of cand that participates in k_max
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    grad = np.zeros_like(cand)
    # each term appears twice except when i == partner (the centre term)
    grad[idx] += 2.0 * cand[partner]
    grad[partner] += 2.0 * cand[idx]      # double‑counting is harmless
    # note: half of the contribution comes from the symmetric pair, we keep it simple

    # step size – a small fraction of the average magnitude
    step = 0.02 * max(1.0, cand.mean())
    delta = step * grad / (np.linalg.norm(grad) + 1e-12)

    cand -= delta
    # enforce non‑negativity
    cand = np.maximum(cand, 0.0)
    np.clip(cand, 0.0, 1000.0, out=cand)
    # re‑normalise to the required sum (scale does not affect the objective)
    target = math.sqrt(2.0 * n)
    cand *= target / cand.sum()
    return cand

        cand = move(best_seq, rng)

        # re‑normalise to keep the sum close to the original (scale invariance)
        cand_sum = cand.sum()
        if cand_sum < 0.01:
            continue
        cand *= (best_seq.sum() / cand_sum)

        # cheap estimate first
        est = _score_estimate(cand)

        # accept only if we have a chance to beat the current best
        if est < best_score - 1e-12:
            exact = float(evaluate_sequence(cand.tolist()))
            if math.isfinite(exact) and exact < best_score:
                best_seq = cand
                best_score = exact
                print(f"[{it:07d}] improvement → {best_score:.6f}")

        it += 1
        if it % report_interval == 0:
            print(f"[{it}] best so far = {best_score:.6f}")

    print(f"Finished after {it} iterations – final best = {best_score:.6f}")
    return [float(x) for x in best_seq]
