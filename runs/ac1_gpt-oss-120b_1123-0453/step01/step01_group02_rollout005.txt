"""
Hybrid optimizer for the step‑function problem.

The algorithm works in three stages:

1. **Coarse analytic search** –  a 2‑parameter family
       a_i = (i+α)^(-p)   (p≈0.5)   is examined on a fine grid.
   The vector is forced to be symmetric (a_i = a_{n‑1‑i}) because a
   symmetric shape concentrates mass at the two ends and lowers the
   maximal auto‑convolution.  The best candidate from this stage seeds
   the local optimiser.

2. **Fast convolution** –  convolution is performed with an FFT
   (O(n log n)) to keep the inner loop cheap even for several‑thousand
   entries.

3. **Aggressive mass‑redistribution hill‑climbing** –  at each step
   the index `k_max` where the convolution attains its maximum is located.
   The contribution of every entry to that peak is
       c_i = a_i * a_{k_max-i}   (zero when the partner index is out of range).
   The algorithm moves a fraction of the mass from the *donor* (largest
   contribution) to the *receiver* (smallest contribution).  Several
   fractions (1.0, 0.5, 0.25, …) are tried and the first improvement is
   kept.  After each successful move the whole vector is re‑symmetrised
   (which preserves the total sum) and clipped to the allowed range.
   Occasionally a random block‑perturbation is tried to escape plateaus.
   The loop runs until the time budget expires.

The routine returns a plain ``list[float]`` that can be fed directly to
``evaluate_sequence``.  It respects all security checks of the evaluator
and never produces negative or overly large entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Return full (linear) convolution of *seq* with itself using FFT."""
    n = seq.size
    # size = next power of two >= 2*n-1  (fast FFT)
    size = 1 << ((2 * n - 1).bit_length())
    # real‑FFT is sufficient because the data are real
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    return conv[: 2 * n - 1]


def contributions_at_peak(seq: np.ndarray, k: int) -> np.ndarray:
    """Contribution of each entry of *seq* to convolution value at index *k*."""
    n = seq.size
    i = np.arange(n, dtype=np.int64)
    j = k - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i[mask]] * seq[j[mask]]
    return contrib


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap non‑negative perturbation of ``base`` – multiply a random block."""
    cand = base.copy()
    n = cand.size
    # block width between n//200 and n//12 (always at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring to keep symmetry candidates
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def evaluate(seq: np.ndarray) -> float:
    """Thin wrapper converting a NumPy array to a Python list for the grader."""
    return float(evaluate_sequence(seq.tolist()))

# Main entry point required by the framework
# ----------------------------------------------------------------------
