"""
Hybrid exponential‑family search for the step‑function evaluation problem.

The objective
    score = 2·n·max(conv(a,a)) / (sum(a)²)
has a simple analytic lower bound for sequences of the form
    a_i = C·r^i   (0<r<1) .
For such a geometric decay the convolution at offset k is
    (k+1)·C²·r^k   (0≤k≤n‑1).

If we write r = 1‑ε and choose ε ≈ factor / n, then
    max(conv) ≈ n·exp(‑factor)      and
    sum(a)   ≈ n                   (when factor≪1)
so the score is roughly 2·exp(‑factor).  Taking factor≈0.30 gives
a score ≈1.48, already beating the target 1.503.

The optimiser therefore explores:
    * a grid of (length n, factor) pairs,
    * a random refinement stage,
    * a final light stochastic local search (block scaling, swaps,
      tiny Gaussian noise) to escape the grid discretisation.

All heavy work is done with a fast FFT‑based estimator
(_score_estimate); the authoritative evaluate_sequence() is invoked
only when a candidate looks promising.

The function `propose_candidate` conforms to the required signature
and returns the best sequence found before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Return max of the linear convolution seq * seq (FFT implementation)."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # keep only the valid linear‑convolution part
    return float(conv[: 2 * n - 1].max())


def _score_estimate(seq: np.ndarray) -> float:
    """Fast estimate of the evaluation score (same formula as evaluate_sequence)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


def exponential_sequence(n: int, factor: float, scale: float = 1000.0) -> np.ndarray:
    """
    Geometric sequence a_i = scale·(1‑ε)^i  with ε = factor / n  (0<factor≤n).
    Returns a NumPy array of length n clipped to [0,1000].
    """
    if n <= 0:
        raise ValueError("Length must be positive")
    # ε = factor / n, ensure ε<1 so that r>0
    eps = max(min(factor / n, 0.999999), 1e-12)
    r = 1.0 - eps
    # Guard against r≤0 (should not happen)
    if r <= 0.0:
        seq = np.zeros(n, dtype=np.float64)
    else:
        # use log to avoid overflow for large n
        log_r = math.log(r)
        idx = np.arange(n, dtype=np.float64)
        seq = np.exp(idx * log_r) * scale
    # clipping as required by evaluate_sequence
    return np.clip(seq, 0.0, 1000.0)


def _perturb_block_scale(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block (as in the reference)."""
    cand = seq.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional mirroring helps symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of mass from i to j."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add low‑amplitude Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the caller provides a small safety margin).

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Warm‑start: use the provided construction if it exists
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0:
            raise ValueError
    except Exception:
        base_seq = np.ones(256, dtype=np.float64)

    base_seq = np.clip(base_seq, 0.0, 1000.0)
    best_seq = base_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 1️⃣  Grid search over exponential families
    # ------------------------------------------------------------------
    # Lengths we will try – they are modest enough for the FFT estimator
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000]
    # Factors roughly correspond to epsilon = factor / n ; we scan 0.2…0.6
    factor_grid = np.linspace(0.15, 0.6, 15)

    print("[grid] starting grid search")
    for n in lengths:
        for factor in factor_grid:
            if time.monotonic() >= deadline:
                break
            seq = exponential_sequence(n, factor, scale=1000.0)
            est = _score_estimate(seq)
            if est < best_score * 0.999:          # cheap filter
                exact = float(evaluate_sequence(seq.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best_seq = seq
                    best_score = exact
                    print(f"[grid] n={n:5d} factor={factor:.3f} → new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Randomised refinement of (n, factor)
    # ------------------------------------------------------------------
    print("[refine] start random refinement")
    while time.monotonic() < deadline:
        # Sample length uniformly in a reasonable interval
        n = int(rng.integers(1500, 12000))
        # Sample factor log‑uniformly between 0.05 and 1.0
        log_f = rng.uniform(np.log(0.05), np.log(1.0))
        factor = math.exp(log_f)
        seq = exponential_sequence(n, factor, scale=1000.0)
        est = _score_estimate(seq)
        if est < best_score * 0.997:
            exact = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_seq = seq
                best_score = exact
                print(f"[refine] n={n:5d} factor={factor:.4f} → new best {best_score:.6f}")

        # occasional tiny perturbation of the current best
        if rng.random() < 0.12:
            cand = _perturb_block_scale(best_seq, rng)
            cand = _perturb_swap(cand, rng)
            cand = _perturb_noise(cand, rng)

            # keep the total sum (scaling does not affect the exact score)
            s = float(cand.sum())
            if s >= 0.01:
                cand *= (best_seq.sum() / s)
                est = _score_estimate(cand)
                if est < best_score - 1e-12:
                    exact = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(exact) and exact < best_score:
                        best_seq = cand
                        best_score = exact
                        print(f"[local] improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final tidy‑up – ensure the returned list is plain Python floats
    # ------------------------------------------------------------------
    print(f"Finished search – best score = {best_score:.6f} (length={best_seq.size})")
    return [float(x) for x in best_seq]
