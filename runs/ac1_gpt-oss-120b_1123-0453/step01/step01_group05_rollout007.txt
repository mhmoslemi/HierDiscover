def compute_score(a: np.ndarray) -> float:
    n = a.shape[0]; sum_a = a.sum()
    if sum_a < 1e-12: return np.inf
    # compute conv via FFT
    size = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    max_b = conv.max()
    return float(2 * n * max_b / (sum_a**2))

"""
Propose a candidate step‑function sequence.

The algorithm searches over simple power‑law shapes
    a_i ∝ (i+δ)^{-α}   (i = 1,…,n)
which have a known closed–form for the asymptotic value of the
objective 2·n·max(conv)/sum².  For α=½ the ratio equals π/2≈1.571,
while for larger α the ratio drops below the target 1.503.
We enumerate a modest grid of (α,δ) pairs, evaluate each
candidate via a fast FFT‑based convolution, keep the best one,
and then perform a short stochastic “block‑perturbation” hill‑climb
to polish the result.  All numbers are non‑negative, the sequence
is renormalised to unit L¹‑norm (the objective is scale‑invariant),
and the final sequence is returned as a plain list of Python floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: fast evaluation of the objective (no security checks)
# ------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq))/∑seq² using FFT (seq must be 1‑D)."""
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return np.inf
    # convolution length 2*n-1 – pad to next power of two for speed
    size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ------------------------------------------------------------
# Helper: cheap block perturbation (used for final polishing)
# ------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator,
                             sigma: float = 0.08) -> np.ndarray:
    """Multiplicative log‑normal jitter on a random contiguous block."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12  (as in the reference code)
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # log‑normal multiplicative factor
    cand[start:stop] *= float(np.exp(rng.normal(0.0, sigma)))
    # occasional symmetric averaging (helps keep the shape tidy)
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the global bounds required by evaluate_sequence
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
