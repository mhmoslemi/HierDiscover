Define `_solve_lp_max_sum`:

```
def _solve_lp_max_sum(f, M):
    n = f.shape[0]
    # Build csr matrix A_ub
    data = []
    rows = []
    cols = []
    for i, fi in enumerate(f):
        # column i, rows from i to i+n-1 inclusive (size n)
        start = i
        end = i + n
        # rows indices: k = start .. end-1
        rows.extend(range(start, end))
        cols.extend([i]*n)
        data.extend([fi]*n)
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n), dtype=np.float64)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

"""
Propose a step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
* Choose a manageable length *n* (capped at 1500) and fix the L1‑norm
  to ``sqrt(2·n)`` – with this normalisation the objective equals the
  maximum value of the self‑convolution.
* Initialise the current vector with a few deterministic candidates
  (geometric, reversed, sparse random) and keep the best.
* Repeatedly improve the vector until the time budget expires.
  The improvement step is chosen randomly among:
    1. **LP step** – maximise ``∑ g_i`` under the linear constraints
       ``(f*g)(k) ≤ M`` where ``M = max(f*f)``.  The problem is a
       linear programme with ``n`` variables and ``2·n‑1`` constraints.
       The solution ``g`` is scaled to the required L1‑norm and blended
       with ``f`` (mixing factor *t*).  The LP is solved with SciPy’s
       HiGHS interior‑point solver.
    2. Block multiplicative perturbation.
    3. Single‑index scaling.
    4. Mass transfer between two indices.
    5. Fresh sparse random pattern.
  After every mutation the sequence is re‑normalised.
* The fast scorer ``_fast_score`` (identical to the official evaluator
  except for the type checks) is used for cheap evaluation.
* Whenever a new best score is found it is printed.  If the score drops
  below the target (1.503) the search stops early.
* The best vector found is finally evaluated with the official
  ``evaluate_sequence`` and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Fast scorer – identical to the official evaluator for valid inputs.
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the score 2·n·max(conv(seq,seq))/sum(seq)^2 (clipped)."""
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Simple candidate generators (used for initialisation).
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """Geometric progression a_i = r**i, scaled to have L1=sum_target."""
    # Solve e^t = 1 + 2 t   (t≈1.256)   →  r = 1 - t/n
    t = 1.0
    for _ in range(20):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    cand = (sum_target / raw_sum) * raw
    return cand


def _sparse_random_candidate(n: int, sum_target: float,
                             rng: np.random.Generator) -> np.ndarray:
    """Create a random sparse vector (few non‑zero entries) and scale it."""
    k = rng.integers(int(n * 0.05), int(n * 0.25) + 1)
    if k == 0:
        return np.full(n, sum_target / n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= (sum_target / total)
    return cand


# ----------------------------------------------------------------------
# Helper for the linear‑programming step.
# ----------------------------------------------------------------------
_LP_PATTERNS = {}  # cache of (rows, cols, shape) for each n


def _get_lp_pattern(n: int):
    """Return (row_idx, col_idx, shape) describing the full convolution matrix."""
    pat = _LP_PATTERNS.get(n)
    if pat is None:
        # Each column i contributes to rows i ... i+n‑1 with coefficient f_i.
        rows = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
        cols = np.repeat(np.arange(n), n)
        shape = (2 * n - 1, n)
        _LP_PATTERNS[n] = (rows, cols, shape)
        pat = _LP_PATTERNS[n]
    return pat


def _max_self_convolution(arr: np.ndarray) -> float:
    """Compute max_k (arr * arr)[k] using an FFT (O(n log n))."""
    n = arr.shape[0]
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))
    # real‑FFT is sufficient because the data are real.
    F = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(F * F, n=size)
    return float(conv[: 2 * n - 1].max())


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max  sum(g)
    s.t.    f * g ≤ M   (full convolution)
            g ≥ 0
    Returns the optimal g (may be all zeros if the LP fails).
    """
    n = f.shape[0]
    rows, cols, shape = _get_lp_pattern(n)
    # Build A_ub where each column i has n entries equal to f_i.
    data = np.repeat(f, n)
    A_ub = csr_matrix((data, (rows, cols)), shape=shape, dtype=np.float64)
    b_ub = np.full(shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -sum(g) == maximise sum(g)
    bounds = [(0, None)] * n
    # HiGHS solver is fast and accepts sparse matrices.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None

    # ------------------------------------------------------------------
    best = _geometric_candidate(n, sum_target)
    best_score = _fast_score(best)

    rev = best[::-1].copy()
    rev_score = _fast_score(rev)
    if rev_score < best_score:
        best, best_score = rev, rev_score

    sparse = _sparse_random_candidate(n, sum_target, rng)
    sp_score = _fast_score(sparse)
    if sp_score < best_score:
        best, best_score = sparse, sp_score

    print(f"[init] n={n}, target‑sum={sum_target:.3f}, best score={best_score:.6f}")

    # The current working vector (may be worse than the best seen so far)
    cur = best.copy()
    t = 0.20            # blending factor for the LP step
    it = 0
    TARGET = 1.503      # stop early if we beat this bound

    while time.monotonic() < deadline:
        it += 1
        mode = rng.integers(0, 5)   # 0: LP, 1: block, 2: single, 3: transfer, 4: fresh sparse

        if mode == 0:
            # ----- LP improvement -------------------------------------------------
            M = _max_self_convolution(cur)
            g0 = _solve_lp_max_sum(cur, M)
            if g0 is not None and g0.sum() > 0.0:
                # Scale g0 to the required L1‑norm
                scale = sum_target / g0.sum()
                g = g0 * scale
                cand = (1.0 - t) * cur + t * g
            else:
                # fall back to a tiny random jitter if LP fails
                cand = cur + rng.normal(scale=0.001 * sum_target, size=n)

        elif mode == 1:
            # ----- multiplicative block -------------------------------------------
            cand = cur.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.12))
            cand[start:start + width] *= factor

        elif mode == 2:
            # ----- single‑index scaling --------------------------------------------
            cand = cur.copy()
            i = rng.integers(n)
            factor = np.exp(rng.normal(0.0, 0.09))
            cand[i] *= factor

        elif mode == 3:
            # ----- transfer mass between two positions -------------------------------
            cand = cur.copy()
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * 0.4
            delta = eps * min(cand[i], cand[j])
            if cand[i] > cand[j]:
                cand[i] -= delta
                cand[j] += delta
            else:
                cand[j] -= delta
                cand[i] += delta

        else:
            # ----- fresh sparse random candidate ------------------------------------
            cand = _sparse_random_candidate(n, sum_target, rng)

        # ----- normalise L1‑norm (protect against drift) -------------------------
        total = cand.sum()
        if total > 0.0:
            cand = cand * (sum_target / total)
        else:
            cand = np.full(n, sum_target / n, dtype=np.float64)

        # occasional tiny jitter to keep the search stochastic
        if rng.random() < 0.05:
            cand += rng.normal(scale=0.001 * sum_target, size=n)

        cand = np.clip(cand, 0.0, 1000.0)

        cur_score = _fast_score(cand)

        if cur_score < best_score:
            best, best_score = cand.copy(), cur_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:5.1f}s] it {it:5d} new best = {best_score:.6f}")
            if best_score < TARGET:
                # We have beaten the target – stop early.
                break

        # In any case, continue the stochastic walk from the new candidate.
        cur = cand

        # Slowly decay the LP blending factor to sharpen the final shape.
        if it % 100 == 0:
            t = max(0.02, t * 0.95)

    # ------------------------------------------------------------------
    # Return the best sequence (converted to plain Python list).
    # ------------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    final_score = evaluate_sequence(final)    # official wrapper (security checks)
    print(f"[done] final score = {final_score:.6f}")
    return final
