"""
Propose a candidate step‑function sequence.

Algorithm
---------
* Initialise a collection of seeds by a fine grid over the two‑parameter
  family a_i = (i+offset+1)^(-p).  This family is known to give a near‑constant
  self‑convolution and therefore a good starting bound.
* Keep the best seed (including the optional global ``height_sequence_1``)
  as the current solution.
* Repeatedly improve the current solution until the time budget expires.
  Three families of moves are used:
    1. **Gradient step** – an explicit (projected) gradient of the evaluation
       function is computed from the index where the convolution attains its
       maximum.  Several step‑sizes are tried and the best improving one is
       kept.
    2. **Block perturbation** – a random contiguous block is multiplied by a
       log‑normal factor (≈ 6 %).  This is the same move as in the reference
       implementation and efficiently explores the neighbourhood.
    3. **Random parametric shape** – a fresh power‑law seed is drawn and an
       optional Gaussian modulation (bump or dip) is applied.  This creates
       qualitatively different shapes and helps escape local minima.
* After each successful move the solution is optionally symmetrised
  (averaged with its reverse) because symmetry tends to flatten the
  convolution.
* A very simple simulated‑annealing acceptance rule is employed:
  non‑improving moves are accepted with a small probability that decays
  linearly with the remaining time, encouraging exploration early on.
* The best sequence found during the whole run is returned as a plain Python
  list of floats.

The method respects the 1000 s budget, uses only NumPy (no external files),
and works for any length – it defaults to the length of the supplied
``height_sequence_1`` if present, otherwise 5000.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------

def generate_power_sequence(n: int, exponent: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Return a non‑negative sequence a_i = (i+offset+1)^(-exponent)."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.06)))      # ~6 % log‑normal change
    new = seq.copy()
    new[start:stop] *= factor
    # Occasional symmetrisation inside the block move
    if rng.random() < 0.1:
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)


def global_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small multiplicative log‑normal noise applied to the whole vector."""
    noise = rng.normal(0.0, 0.02, size=seq.shape)
    factor = np.exp(noise)
    new = seq * factor
    return np.clip(new, 0.0, 1000.0)


def symmetrize(seq: np.ndarray) -> np.ndarray:
    """Enforce (soft) symmetry by averaging with the reversed sequence."""
    return np.clip((seq + seq[::-1]) * 0.5, 0.0, 1000.0)


def random_param_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a fresh candidate from a random power‑law possibly modulated by a
    Gaussian bump/dip.
    """
    # Random power‑law seed
    exponent = rng.uniform(0.42, 0.60)
    offset = rng.uniform(0.0, 2.5)
    base = generate_power_sequence(n, exponent=exponent, offset=offset)

    # Optional Gaussian modulation (amplitude may be negative → dip)
    if rng.random() < 0.6:
        A = rng.uniform(-0.6, 0.6)                               # amplitude
        mu = rng.uniform(0, n - 1)                               # centre
        sigma = rng.uniform(0.02, 0.30) * n                       # width
        i = np.arange(n, dtype=np.float64)
        mod = 1.0 + A * np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
        base = base * mod

    # Occasionally enforce symmetry (helps flatten convolution)
    if rng.random() < 0.25:
        base = symmetrize(base)

    return np.clip(base, 0.0, 1000.0)


def gradient_step(seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Perform a single gradient descent step on ``seq``.
    The gradient of the evaluation function is computed analytically from the
    index where the self‑convolution attains its maximum.
    Several step‑sizes are tried and the best improving candidate is returned.
    """
    n = seq.size
    # Full convolution and its maximum
    conv = np.convolve(seq, seq)
    max_idx = int(np.argmax(conv))
    max_c = conv[max_idx]
    S = np.sum(seq)

    # Shifted copy a_{max_idx-i} (zero where out of range)
    idx = max_idx - np.arange(n)
    a_shift = np.where((idx >= 0) & (idx < n), seq[idx], 0.0)

    # Gradient of the evaluation function (derived analytically)
    factor = (2.0 * n) / (S * S)
    grad = factor * (seq + a_shift) - (4.0 * n * max_c) / (S ** 3)

    # Try a handful of step sizes
    best_cand = seq
    best_score = float('inf')
    for step in (0.08, 0.18, 0.35, 0.55, 0.80):
        cand = seq - step * grad
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_cand = cand
            best_score = score

    return best_cand, best_score


def accept_worse(delta: float, time_frac: float) -> bool:
    """
    Probabilistic acceptance of a worsening move.
    ``time_frac`` is the fraction of remaining time (0 → out of time,
    1 → just started).  Early on we are more permissive.
    """
    # Scale the temperature so that early acceptance is noticeable but dies out
    temperature = 0.03 * time_frac          # adjust the 0.03 constant if needed
    if temperature <= 0:
        return False
    prob = math.exp(-delta / temperature)
    return prob > np.random.random()

# ----------------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a sequence (list of floats) that (heuristically) minimises
    ``evaluate_sequence`` within the given time budget.

    The algorithm combines:
        * fine grid initialisation on a power‑law family,
        * a projected gradient descent step,
        * block perturbations,
        * random parametric re‑initialisation,
        * occasional symmetrisation,
        * a simple simulated‑annealing acceptance rule.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # Reserve a few seconds for finalisation
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Determine the sequence length
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))           # use previous best if available
    except Exception:
        n = 5000

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    exponent_vals = np.linspace(0.45, 0.55, 13)   # 13 points → fine resolution
    offset_vals   = np.linspace(0.0, 2.0, 9)     # 9 points

    for p in exponent_vals:

            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best_seq = seq.copy()
                best_score = score
                print(f"[grid] p={p:.4f}, off={off:.2f} → {score:.6f}")
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Try the provided global variable as a seed (if it exists)
    # ------------------------------------------------------------------
    try:
        seq0 = np.array(height_sequence_1, dtype=np.float64)
        score0 = float(evaluate_sequence(seq0.tolist()))
        if np.isfinite(score0) and score0 < best_score:
            best_seq = seq0.copy()
            best_score = score0
            print(f"[seed] height_sequence_1 → {score0:.6f}")
    except Exception:
        pass

    # Final fallback (should never happen)
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using all‑ones seed")

    # Current solution (may temporarily be worse than the best)
    cur_seq = best_seq.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        time_fraction = max(0.0, (deadline - now) / (deadline - start_time))  # 1 → early

        # Choose a move type
        move_type = rng.random()
        if move_type < 0.48:
            # Gradient descent step
            cand_seq, cand_score = gradient_step(cur_seq)
        elif move_type < 0.78:
            # Block perturbation
            cand_seq = block_perturb(cur_seq, rng)
            cand_score = float(evaluate_sequence(cand_seq.tolist()))
        elif move_type < 0.93:
            # Global multiplicative noise (tiny)
            cand_seq = global_noise(cur_seq, rng)
            cand_score = float(evaluate_sequence(cand_seq.tolist()))
        else:
            # Random parametric re‑initialisation (with optional bump)
            cand_seq = random_param_sequence(n, rng)
            cand_score = float(evaluate_sequence(cand_seq.tolist()))

        # ------------------------------------------------------------------
        # Acceptance logic (simulated annealing style)
        # ------------------------------------------------------------------
        if np.isfinite(cand_score):
            if cand_score < cur_score:
                cur_seq, cur_score = cand_seq, cand_score
            else:
                # Occasionally accept a worsening move to escape plateaus
                if accept_worse(cand_score - cur_score, time_fraction):
                    cur_seq, cur_score = cand_seq, cand_score

            # Track a global best
            if cand_score < best_score:
                best_seq, best_score = cand_seq.copy(), cand_score
                elapsed = now - start_time
                print(f"[new best] {best_score:.6f} after {elapsed:.1f}s")

        # ------------------------------------------------------------------
        # Periodic soft symmetrisation (helps flatten the convolution)
        # ------------------------------------------------------------------
        if rng.random() < 0.015:       # about 1.5 % of iterations
            sym = symmetrize(cur_seq)
            sym_score = float(evaluate_sequence(sym.tolist()))
            if np.isfinite(sym_score) and sym_score < cur_score:
                cur_seq, cur_score = sym, sym_score
                if sym_score < best_score:
                    best_seq, best_score = sym.copy(), sym_score
                    elapsed = time.monotonic() - start_time
                    print(f"[sym best] {best_score:.6f} after {elapsed:.1f}s")

        # Occasionally force a fresh random restart (helps avoid deep local minima)
        if rng.random() < 0.002:

            fresh_score = float(evaluate_sequence(fresh.tolist()))
            if np.isfinite(fresh_score):
                if fresh_score < cur_score:
                    cur_seq, cur_score = fresh, fresh_score
                if fresh_score < best_score:
                    best_seq, best_score = fresh.copy(), fresh_score
                    elapsed = time.monotonic() - start_time
                    print(f"[restart] {best_score:.6f} after {elapsed:.1f}s")

    # ------------------------------------------------------------------
    # Final clean‑up: ensure the result respects the evaluator's clipping
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return as plain Python list (exact type expected by the grader)
    return [float(x) for x in best_seq]
