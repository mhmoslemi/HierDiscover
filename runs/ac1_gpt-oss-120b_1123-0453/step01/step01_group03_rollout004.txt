import time
import numpy as np

def _fast_score(seq):
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * maxb / (s * s))

def _geometric_candidate(n, sum_target):
    # Solve e^t = 1 + 2t (approx solution) as earlier
    t = 1.0
    for _ in range(30):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = r ** idx
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    cand = raw * (sum_target / raw_sum)
    return cand

def _sparse_random_candidate(n, sum_target, rng):
    k = rng.integers(int(max(1, n * 0.05)), int(max(2, n * 0.25)) + 1)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= sum_target / total
    else:
        cand = np.full(n, sum_target / n, dtype=np.float64)
    return cand

    size = 1 << ((2 * n - 1).bit_length())
    lr = 0.05               # initial learning rate
    alpha_start = 0.5
    alpha_end = 5.0
    it = 0
    # reserve a small time slice for the second phase
    phase1_deadline = deadline - 120.0   # keep ~2 min for hill climbing

    while time.monotonic() < phase1_deadline:
        it += 1
        a = best
        # FFT of the current vector
        A = np.fft.rfft(a, n=size)
        # full self‑convolution (real)
        conv = np.fft.irfft(A * A, n=size)[:2 * n - 1]

        # smooth max via soft‑max (temperature alpha)
        cur_alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / 8000.0)
        mx = conv.max()
        # stabilise the exponentials
        exp_term = np.exp(cur_alpha * (conv - mx))
        probs = exp_term / exp_term.sum()          # size = 2n‑1, sums to 1

        # gradient of the smoothed objective:
        # ∇ = 2 * (a ⋆ probs)  (cross‑correlation)
        probs_rev = probs[::-1]
        B = np.fft.rfft(probs_rev, n=size)
        grad_full = np.fft.ifft(A * B, n=size).real   # length = size
        L = 2 * n - 1
        grad = 2.0 * grad_full[(L - 1) - np.arange(n)]

        # projected gradient step
        cand = a - lr * grad
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total > 0.0:
            cand *= (sum_target / total)

        # evaluate candidate (max of its self‑convolution)
        A_cand = np.fft.rfft(cand, n=size)
        conv_cand = np.fft.irfft(A_cand * A_cand, n=size)[:2 * n - 1]
        cand_score = float(conv_cand.max())   # identical to evaluate_sequence after normalisation

        # acceptance
        if cand_score < best_score:
            best, best_score = cand, cand_score
            lr = min(lr * 1.02, 0.5)            # slowly enlarge step if we are making progress
        else:
            lr = max(lr * 0.97, 1e-6)           # otherwise shrink

        # occasional random restarts
        if rng.random() < 0.001:
            cand = _sparse_random_candidate(n, sum_target, rng)
            sc = _fast_score(cand)
            if sc < best_score:
                best, best_score = cand, sc

        # periodic progress report
        if it % 500 == 0:
            elapsed = time.monotonic() - (deadline - budget_s + 5.0)
            print(f"[{elapsed:5.1f}s][grad] iter={it}, best={best_score:.6f}, lr={lr:.4f}")

    # ----------------------------------------------------------------------
    # Phase 2 – cheap hill‑climbing refinements (same ideas as the reference)
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        mode = rng.integers(0, 5)

        cand = best.copy()
        if mode == 0:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.08))
            cand[start:start + width] *= factor
            if rng.random() < 0.2:
                cand = 0.5 * (cand + cand[::-1])
        elif mode == 1:
            i = rng.integers(n)
            factor = np.exp(rng.normal(0.0, 0.07))
            cand[i] *= factor
        elif mode == 2:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * 0.3
            delta = eps * min(cand[i], cand[j])
            if cand[i] > cand[j]:
                cand[i] -= delta
                cand[j] += delta
            else:
                cand[j] -= delta
                cand[i] += delta
        elif mode == 3:
            cand = _sparse_random_candidate(n, sum_target, rng)
        else:
            # power‑law candidate
            p = rng.uniform(0.2, 2.5)
            idx = np.arange(1, n + 1, dtype=np.float64)
            cand = np.power(idx, -p)
            cand *= sum_target / cand.sum()

        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total > 0.0:
            cand *= (sum_target / total)

        cur_score = _fast_score(cand)
        if cur_score < best_score:
            best, best_score = cand, cur_score
            elapsed = time.monotonic() - (deadline - budget_s + 5.0)
            print(f"[{elapsed:5.1f}s] hill new best = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return result
    # ----------------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f}")
    return final

"""
Propose a step–function (list of non‑negative heights) that minimises
`evaluate_sequence`.  The search runs in two stages:

1.  **Smooth gradient descent** – we minimise a soft‑max approximation of the
    self‑convolution’s supremum.  The gradient is obtained analytically
    (cross‑correlation) and evaluated with FFTs; the step is projected onto
    the non‑negative simplex (sum fixed to √(2 n)).  The temperature of the
    soft‑max is annealed from ~0.5 up to ~5.0.

2.  **Hill‑climbing refinement** – cheap multiplicative and redistribution
    mutations (the same ideas as the reference code) are applied for the
    remaining time budget.

Both phases respect the 1000 s budget and print intermediate progress.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation (identical to the official evaluator for valid inputs)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """Geometric progression a_i = r**i, scaled to have L1 = sum_target."""
    # Solve e^t = 1 + 2 t  (t ≈ 1.256) and set δ = t / n, r = 1 – δ.
    t = 1.0
    for _ in range(30):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = r ** idx
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    cand = raw * (sum_target / raw_sum)
    return cand


def _sparse_random_candidate(n: int, sum_target: float,
                             rng: np.random.Generator) -> np.ndarray:
    """Random sparse vector (few non‑zero entries) scaled to the required sum."""
    # sparsity between 5 % and 25 % of positions
    k = rng.integers(int(max(1, n * 0.05)), int(max(2, n * 0.25)) + 1)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= (sum_target / total)
    else:
        cand = np.full(n, sum_target / n, dtype=np.float64)
    return cand
