def compute_conv(a):
    b = np.convolve(a, a)
    return b

def max_convolution_info(a):
    b = np.convolve(a, a)
    M = b.max()
    idx = np.flatnonzero(np.abs(b - M) < 1e-12)  # indices with near max
    shifts = idx - (len(a)-1)
    return M, shifts

"""
Simple hybrid optimiser for the step‑function bound.

The optimiser works in three stages:

1.  **Warm start** – a handful of random power‑law seeds are drawn,
    normalised to the standard L1‑norm √(2 n) and the best one is kept.
2.  **Projected sub‑gradient descent** – the index where the convolution
    attains its maximum is identified; the sub‑gradient of that entry
    (which is linear in the current vector) is computed and a step is
    taken in a direction that keeps the total sum (to first order)
    constant.  The step size adapts to the current magnitude of the
    direction and the iterates are projected back onto the feasible box
    [0, 1000] and re‑scaled.
3.  **Exploratory perturbations** – occasional block‑wise multiplicative
    perturbations (or tiny global log‑normal noise) are applied to help
    escape plateaus.  Very rare random restarts with fresh power‑law
    parameters are also performed.

All heavy work (convolution, gradient) is done with NumPy/FFT, which
comfortably fits into the 1000 s budget even for n≈5 000.  The routine
returns the best sequence discovered as a plain ``list[float]``.
"""

import time
from typing import List

import numpy as np


def generate_power_sequence(
    n: int, exponent: float = 0.5, offset: float = 0.0
) -> np.ndarray:
    """a_i = (i + offset + 1)^{-exponent}, clipped to the allowed range."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.06)))  # ~6 % log‑normal change
    new = seq.copy()
    new[start:stop] *= factor
    if rng.random() < 0.1:  # occasional symmetrisation
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)


def global_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Tiny multiplicative log‑normal noise applied to the whole vector."""
    noise = rng.normal(0.0, 0.02, size=seq.shape)
    factor = np.exp(noise)
    new = seq * factor
    return np.clip(new, 0.0, 1000.0)


def convolve_self_fft(seq: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (full length 2 n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    # next power of two >= m
    size = 1 << ((m - 1).bit_length())
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[:m]
    return conv


def compute_score(seq: np.ndarray, n: int) -> float:
    """
    Fast version of ``evaluate_sequence`` – expects ``seq`` already
    clipped to [0, 1000].  Returns 2 n·max(conv)/(sum(seq)²) or np.inf
    if the sum is too small.
    """
    total = seq.sum()
    if total < 0.01:
        return np.inf
    conv = convolve_self_fft(seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (total ** 2))


def subgradient_max_shift(a: np.ndarray, shift: int) -> np.ndarray:
    """
    Sub‑gradient of (a*a)[shift] w.r.t. ``a``.
    ``shift`` is expressed in the full‑convolution index system:
    shift = k - (n‑1) where k is the position of the maximum.
    """
    n = a.size
    g = np.zeros_like(a)
    if shift == 0:
        g += 2.0 * a
    elif shift > 0:
        g[: n - shift] += a[shift:]
        g[shift:] += a[: n - shift]
    else:  # shift < 0
        s = -shift
        g[: n - s] += a[s:]
        g[s:] += a[: n - s]
    return g
