def _project_onto_simplex(v, s=1.0):
    # shift v to descending sort
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s

        theta = 0
    else:

        theta = cssv[rho] / (rho + 1)

"""
Propose a candidate step‑function sequence.

The algorithm consists of three stages:

1.  **Parametric warm‑start** – we try a handful of simple
    power‑law families  a_i ∝ (i+offset)^(-p)  (including the
    classic arcsine shape p=0.5) and keep the best shape seen.
    The search is cheap because the score can be evaluated with a
    single FFT.

2.  **Projected gradient descent on the max‑convolution** – the score
    2·n·max(b)/(∑a)² is invariant under scaling, therefore we fix
    ∑a=1 and minimise the quantity  max(b).  For a given
    max‑index k* the sub‑gradient of max(b) is simply the reversed
    vector shifted by k*.  We move a small step against this
    gradient, then project back onto the probability simplex
    (non‑negative entries with unit sum).  This operation flattens
    the convolution and usually improves the bound.

3.  **Stochastic block perturbations** – occasional random
    multiplicative jitter on a contiguous block (as in the reference
    code) helps to escape shallow plateaux.  Every new candidate is
    evaluated quickly with an FFT; only promising candidates are
    checked with the official ``evaluate_sequence`` function.

All helper functions are top‑level, there are no closures or lambdas,
and the whole search respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex:
        { x | x_i >= 0, sum(x) = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(a: np.ndarray, n: int, fft_size: int) -> float:
    """
    Compute the evaluation score using the same formula as
    ``evaluate_sequence`` while assuming the sequence is already
    clipped to the admissible range and has sum>0.
    """
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]
    max_b = np.max(conv)
    s = a.sum()
    return 2.0 * n * max_b / (s * s)


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) non‑negative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _gradient_step(a: np.ndarray,
                   max_idx: int,
                   step: float) -> np.ndarray:
    """
    Perform one projected gradient step that reduces the current
    maximal convolution entry ``max_idx``.
    """
    n = a.size
    rev = a[::-1]
    offset = n - 1 - max_idx
    grad = np.zeros_like(a)
    if offset >= 0:
        grad[: n - offset] = 2.0 * rev[offset:]
    else:
        grad[-offset :] = 2.0 * rev[: n + offset]
    # move against the gradient, then project onto the simplex
    cand = a - step * grad
    return _project_onto_simplex(cand)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 950.0,
                     **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improve
    the evaluation bound.  The routine respects the 1000 s limit,
    keeps the best sequence seen so far and prints progress.
    """
    # ------------------------------------------------------------------
    # 1) Problem size (use the length of a previously found solution if
    #    available, otherwise fall back to a modest default).
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # pylint: disable=undefined-variable
        if n == 0:
            raise Exception
    except Exception:
        n = 2000

    # FFT size (next power of two) – constant for the whole run.
    fft_size = int(2 ** np.ceil(np.log2(2 * n - 1)))

    # ------------------------------------------------------------------
    # 2) Warm‑start: simple power‑law families.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Helper to evaluate a candidate quickly and, when promising,
    # confirm with the official checker.
    def try_candidate(cand: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_score
        # quick check that sum is not degenerate
        if cand.sum() < 1e-12:
            return
        cand_score = _fast_score(cand, n, fft_size)
        if cand_score < best_score - 1e-6:
            official = float(evaluate_sequence(cand.tolist()))
            if official < best_score:
                best_seq = cand.copy()
                best_score = official
                print(f"[{tag}] new best score {best_score:.6f}")

    # Systematic grid over exponent p and offset o.
    exponents = np.linspace(0.34, 0.66, 7)          # around 0.5
    offsets = np.linspace(0.1, 2.0, 10)            # away from zero
    for p in exponents:
        for o in offsets:
            idx = np.arange(o, o + n, 1.0)
            shape = 1.0 / np.power(idx, p)
            shape /= shape.sum()
            try_candidate(shape, tag=f"grid(p={p:.3f},o={o:.2f})")

    # A few random mixtures of arcsine and uniform for extra diversity.
    uniform = np.ones(n) / n
    arcsine = 1.0 / np.sqrt(np.arange(1, n + 1))
    arcsine /= arcsine.sum()
    for _ in range(15):
        lam = rng.random()
        shape = lam * arcsine + (1 - lam) * uniform
        shape = _project_onto_simplex(shape)   # ensure exact sum=1
        try_candidate(shape, tag=f"mix(lam={lam:.2f})")

    # Fallback – if nothing succeeded (should not happen) start from pure arcsine.
    if best_seq is None:
        base = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
        base /= base.sum()
        best_seq = base.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[fallback] initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Main optimisation loop (gradient + stochastic perturbations).
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    iteration = 0
    # Initial step size for the gradient descent (will decay linearly)
    step0 = 0.07

    while time.monotonic() < deadline:
        iteration += 1

        # ----- occasional gradient step --------------------------------
        if iteration % 41 == 0:                                   # about once per 40 iters
            # compute the current max‑convolution index
            fa = np.fft.rfft(best_seq, fft_size)
            conv = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            # linear decay of the step size
            elapsed = time.monotonic() - start_time
            frac = elapsed / (deadline - start_time)
            step = step0 * (1.0 - frac)

            cand = _gradient_step(best_seq, kmax, step)
            try_candidate(cand, tag="grad")
            continue

        # ----- stochastic block perturbation ---------------------------
        cand = _make_block_perturbation(best_seq, rng)
        # keep the sequence on the simplex (unit sum)
        cand = _project_onto_simplex(cand)

        # quick evaluation, full check only if improvement is plausible
        cand_score = _fast_score(cand, n, fft_size)
        if cand_score < best_score - 1e-6:
            official = float(evaluate_sequence(cand.tolist()))
            if official < best_score:
                best_seq = cand
                best_score = official
                print(f"[perturb] iter {iteration} score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"[final] best score {best_score:.6f} after "
          f"{time.monotonic() - start_time:.2f}s, length={n}")
    return [float(x) for x in best_seq]
