"""
Propose a near‑optimal step‑function for the given evaluation metric.

Algorithm
---------
1. **Problem size** – Use the length of the supplied ``height_sequence_1``.
   All candidates are built with this length so that the score is comparable.

2. **Analytic families** – Search a few low‑dimensional families that are known
   to give almost flat self‑convolutions:
   * a_i = (i+α)^‑p                               (simple power‑law)
   * a_i = ((i+α)*(n‑1‑i+α))^‑p                   (symmetric “product’’ law)
   * a_i = (i+α)^‑p + (n‑1‑i+α)^‑p                (symmetrised sum)
   A coarse grid on p∈[0.40,0.60] and α∈[0,5] is examined; the best vector
   becomes the starting point for the next phase.

3. **Projected gradient descent on a smooth proxy of the max‑convolution** –
   The true objective 2·n·max(conv)/sum² is invariant under scaling, so we
   fix Σa_i = 1 and minimise  max(conv).  The non‑smooth max is replaced by
   the smooth approximation
        f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k) .
   Its gradient can be written in closed form and evaluated with a single
   FFT‑based convolution.  At every iteration we
   * compute the gradient,
   * take a small step,
   * project the result onto the probability simplex (non‑negative, sum = 1).

   The smoothness parameter λ is increased slowly (1 → 20) while the step
   size is decreased (η ∝ 1/√t).  Periodically the exact score is evaluated
   with ``evaluate_sequence`` and the best vector is remembered.

4. **Fine‑grained hill climbing** – Starting from the best vector obtained
   by the gradient method, we repeatedly move a tiny amount of mass from the
   index that contributes most to the current maximal convolution entry to the
   index that contributes least.  Occasional random block perturbations are
   tried to escape plateaus.

The whole search runs until the allocated time budget expires (default 1000 s)
and returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level – no closures or lambdas)
# ----------------------------------------------------------------------


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block of *base* by a log‑normal factor and
    optionally mirror the vector.  The result is clipped to the allowed range.
    """
    cand = base.copy()
    n = cand.size
    # choose a block width (roughly 0.5 %–8 % of the vector)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetry
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: enforce exact sum 1
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the (full) convolution of *a* and *b* using a power‑of‑two FFT."""
    n = a.size + b.size - 1
    # next power of two for speed
    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    C = A * B
    conv = np.fft.irfft(C, size)[:n]
    return conv

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The function runs for at most *budget_s* seconds.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the working length n
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)       # fallback length
    n = base.size
    idx = np.arange(n, dtype=np.float64)

    # --------------------------------------------------------------
    # 1) Grid search over a few low‑dimensional families
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- family A : (i+α)^‑p ------------------------------------
    p_vals = np.linspace(0.40, 0.60, 41)
    alphas = np.linspace(0.0, 5.0, 51)
    for p in p_vals:
        if time.monotonic() >= deadline:
            break
        for alpha in alphas:
            cand = (idx + alpha) ** (-p)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"A grid p={p:.3f} α={alpha:.3f} → score={best_score:.6f}")

    # ---- family B : ((i+α)*(n‑1‑i+α))^‑p -------------------------
    for p in p_vals:
        if time.monotonic() >= deadline:
            break
        for alpha in alphas:
            cand = ((idx + alpha) * (n - 1 - idx + alpha)) ** (-p)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"B grid p={p:.3f} α={alpha:.3f} → score={best_score:.6f}")

    # ---- family C : (i+α)^‑p + (n‑1‑i+α)^‑p ----------------------
    for p in p_vals:
        if time.monotonic() >= deadline:
            break
        for alpha in alphas:
            cand = (idx + alpha) ** (-p) + (n - 1 - idx + alpha) ** (-p)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"C grid p={p:.3f} α={alpha:.3f} → score={best_score:.6f}")

    # Safety – if nothing succeeded fall back to a uniform vector
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("Fallback to uniform sequence, score:", best_score)

    # --------------------------------------------------------------
    # 2) Projected gradient descent on a smooth approximation of max(conv)
    # --------------------------------------------------------------
    # Work with a normalized vector (sum = 1) – scaling does not affect the score
    a = best_seq.copy()
    a = a / np.sum(a)            # now Σa_i = 1

    # Parameters of the gradient loop
    max_iter = 3000
    step0 = 0.3                  # initial step size
    lam = 1.0                    # smooth‑max parameter (will grow)
    lam_max = 20.0
    lam_inc = 1.05               # multiplicative increase per iteration

    # Helper slice indices for the gradient extraction (see analysis)
    slice_start = n - 1
    slice_end = slice_start + n

    # Keep track of the best vector seen during the gradient phase
    best_grad_seq = a.copy()
    best_grad_score = best_score

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        # Full self‑convolution
        conv = fft_convolve(a, a)                     # length = 2n‑1
        max_c = conv.max()
        # Smooth max:   f_λ = (1/λ) log Σ exp(λ·c_k)
        # Use a numerically stable shift by max_c.
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            break
        w /= w_sum                                   # weight vector, Σw = 1

        # Gradient of the smooth objective:
        #   ∇f = 2 * (w * a_rev)   where * denotes convolution
        a_rev = a[::-1]
        grad_full = fft_convolve(w, a_rev)           # length = 3n‑2
        grad = 2.0 * grad_full[slice_start:slice_end]  # shape (n,)

        # Decaying step size
        eta = step0 / np.sqrt(it + 1.0)

        # Gradient step + projection onto the simplex Σx=1, x≥0
        a_next = project_simplex(a - eta * grad)

        a = a_next

        # Occasionally evaluate the true objective (every 5 iterations)
        if it % 5 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_grad_score:
                best_grad_score = score
                best_grad_seq = a.copy()
                print(f"Iter {it:4d}: improved score to {best_grad_score:.6f}")

        # Update smooth‑max parameters
        lam = min(lam * lam_inc, lam_max)
        it += 1

    # --------------------------------------------------------------
    # 3) Fine‑grained hill climbing (mass transfer between donor/receiver)
    # --------------------------------------------------------------
    current = best_grad_seq.copy()
    cur_score = best_grad_score
    max_hc_iters = 20000

    while time.monotonic() < deadline and max_hc_iters > 0:
        # compute convolution and locate the peak index
        conv = fft_convolve(current, current)
        k_max = int(np.argmax(conv))

        # contribution of each entry to that peak: a_i * a_{k_max-i}
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(current)
        contrib[mask] = current[i[mask]] * current[j[mask]]

        donor = int(np.argmax(contrib))
        receiver = int(np.argmin(contrib))

        if donor == receiver or current[donor] <= 0.0:
            candidate = make_block_perturbation(current, rng)
        else:
            # transfer a small, adaptive fraction of the donor's mass
            delta = float(rng.uniform(0.0, min(0.5, 0.1) * current[donor]))
            if delta < 1e-12:
                delta = current[donor] * 0.01
            candidate = current.copy()
            candidate[donor] -= delta
            candidate[receiver] += delta
            candidate = np.clip(candidate, 0.0, 1000.0)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < cur_score:
            current = candidate
            cur_score = score
            if score < best_score:
                best_score = score
                best_seq = candidate.copy()
                print(f"Hill‑climb improved score to {best_score:.6f}")
        else:
            # occasionally accept a neutral move to avoid stagnation
            if rng.random() < 0.001:
                current = candidate
                cur_score = score

        max_hc_iters -= 1

    print(f"Search finished – best score achieved: {best_score:.6f}")
    # Convert to plain python list (the grader expects this)
    return [float(x) for x in best_seq]
