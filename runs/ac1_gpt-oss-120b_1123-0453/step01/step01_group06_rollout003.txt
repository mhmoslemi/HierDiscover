```
def _gradient(seq):
    n = seq.size
    # compute convolution via FFT
    size = _next_pow_two(2*n-1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    kmax = int(conv.argmax())
    max_val = conv[kmax]
    # compute gradient g for M
    g = np.zeros_like(seq)
    i_min = max(0, kmax-(n-1))
    i_max = min(n-1, kmax)
    # loop over i in [i_min,i_max]
    # conj: partner = kmax - i
    for i in range(i_min, i_max+1):
        j = kmax - i
        if i == j:
            g[i] += seq[j]
        else:
            g[i] += seq[j] * 2  # as we argued (two contributions)
    # gradient of O:
    S = seq.sum()
    if S < 1e-12:
        return np.zeros_like(seq)  # not used
    Ograd = (2 * n / (S * S)) * (g - (2 * max_val / S) * np.ones_like(seq))
```

The algorithm is a hybrid optimiser that combines

* a **warm‑start** over simple analytic families (power‑law, geometric, …),
* a **stochastic local search** with several cheap perturbations,
* a **gradient‑step** that directly follows the (sub‑)gradient of the true objective,
* a **simulated‑annealing** acceptance rule so the search can escape local minima,

```python
"""Hybrid stochastic optimiser for the step‑function evaluation problem.

The optimiser consists of:
1. Warm‑start over a few analytic families (power‑law, geometric).
2. Main loop (simulated‑annealing) with several cheap moves:
   * block multiplicative scaling,
   * pairwise mass exchange,
   * small Gaussian noise,
   * “flatten” move that removes mass from the most‑offending convolution term,
   * a deterministic gradient‑step (sub‑gradient of the objective),
   * optional monotonic‑shape enforcement,
   * occasional random‑reset to a fresh power‑law shape.
Only when the fast FFT estimate is better than the current best the
authoritative ``evaluate_sequence`` function is called.
The loop runs until the supplied time budget is exhausted.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Fast max of the linear convolution seq * seq (FFT based)."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(conv[: 2 * n - 1].max())


def _conv_max_and_index(seq: np.ndarray):
    """Return (max value, argmax) of the convolution."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv[k]), k


def _score_estimate(seq: np.ndarray) -> float:
    """Fast estimate of the true objective (identical to the exact formula)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Perturbation moves
# ----------------------------------------------------------------------


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative scaling of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring to keep some symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from i → j (sum stays unchanged)."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)  # ≤ 20 % of source mass
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add small zero‑mean Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)


def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Reduce the contribution of the currently most‑violated convolution index."""
    cand = seq.copy()
    n = cand.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())

    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]

    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand

    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))

    delta = cand[src] * float(0.05 + 0.15 * rng.random())   # move 5‑%‑20 %
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


def _gradient_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """One sub‑gradient step for the true objective."""
    n = seq.size
    # full convolution to obtain the maximal index
    M, k_max = _conv_max_and_index(seq)

    # indices that contribute to the maximal entry
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    factor = np.where(idx == partner, 1.0, 2.0)            # 1× on the diagonal, 2× otherwise
    grad_M = np.zeros_like(seq)
    grad_M[idx] = factor * seq[partner]                  # ∂M/∂a_i

    S = seq.sum()
    if S < 1e-12:
        return seq.copy()

    # gradient of the whole objective  O = 2 n M / S²
    grad_O = (2.0 * n / (S * S)) * (grad_M - (2.0 * M / S) * np.ones_like(seq))

    # fixed modest step size (tuned empirically)
    lr = 0.01
    cand = seq - lr * grad_O
    cand = np.clip(cand, 0.0, 1000.0)

    # renormalise the sum (scaling does not change the objective)
    s_new = cand.sum()
    if s_new > 0.0:
        cand *= (S / s_new)
    return cand


def _monotonic_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force a non‑increasing shape (optional smoothing)."""
    cand = np.maximum.accumulate(seq[::-1])[::-1]   # enforce a_i ≥ a_{i+1}
    # tiny random jitter to avoid stagnation
    if rng.random() < 0.1:
        cand += rng.normal(0.0, 0.01 * cand.mean(), size=cand.shape)
    cand = np.clip(cand, 0.0, 1000.0)
    s = cand.sum()
    if s > 0.0:
        cand *= (seq.sum() / s)
    return cand


def _random_reset(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Replace the current sequence by a fresh power‑law shape."""
    n = seq.size
    p = rng.uniform(0.2, 0.8)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cand = idx ** (-p)
    cand *= rng.uniform(0.8, 1.2, size=n)          # random amplitude wiggle
    cand = np.clip(cand, 0.0, 1000.0)
    # keep the original total mass
    s0 = seq.sum()
    cand *= (s0 / cand.sum())
    return cand

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a step‑function with minimal ``evaluate_sequence`` value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the caller will subtract a tiny safety margin).

    Returns
    -------
    list[float]
        The best sequence discovered before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a small safety margin for the caller
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start – analytic families
    # ------------------------------------------------------------------
    # Use the provided construction if available, otherwise fall back to a uniform vector.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_seq = np.ones(2000, dtype=np.float64)

    if base_seq.ndim != 1 or base_seq.size == 0 or not np.isfinite(base_seq).all():
        base_seq = np.ones(2000, dtype=np.float64)

    base_seq = np.clip(base_seq, 0.0, 1000.0)
    n = base_seq.size

    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # power‑law family a_i = (i+1)^(-p)
    for p in np.linspace(0.30, 0.65, 15):
        idx = np.arange(1, n + 1, dtype=np.float64)
        cand = np.power(idx, -p)
        cand = np.clip(cand, 0.0, 1000.0)
        est = _score_estimate(cand)
        if est < best_score:
            exact = float(evaluate_sequence(cand.tolist()))
            if exact < best_score:
                best = cand
                best_score = exact
                print(f"[warm] p={p:.3f} → best {best_score:.6f}")

    # geometric family a_i = r^i
    for r in np.linspace(0.90, 0.999, 12):
        cand = r ** np.arange(n, dtype=np.float64)
        cand = np.clip(cand, 0.0, 1000.0)
        est = _score_estimate(cand)
        if est < best_score:
            exact = float(evaluate_sequence(cand.tolist()))
            if exact < best_score:
                best = cand
                best_score = exact
                print(f"[warm] r={r:.4f} → best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Main stochastic optimiser (simulated annealing)
    # ------------------------------------------------------------------
    current = best.copy()
    current_score = _score_estimate(current)

    # move pool – the heavy moves (flatten, gradient) appear less often
    move_pool = (
        [_make_block_perturbation] * 5
        + [_perturb_swap] * 4
        + [_perturb_noise] * 2
        + [_flatten_move] * 1
        + [_gradient_move] * 2
        + [_monotonic_move] * 1
        + [_random_reset] * 1
    )
    move_pool = np.array(move_pool)  # numpy allows fast random choice

    it = 0
    report_interval = 2000
    total_time = deadline - start_time

    while time.monotonic() < deadline:
        # linearly decaying temperature based on remaining time
        elapsed = time.monotonic() - start_time
        remaining = max(0.0, deadline - time.monotonic())
        temperature = max(1e-5, remaining / total_time)   # in [0,1]

        # pick a move
        move = rng.choice(move_pool)
        candidate = move(current, rng)

        # keep the total mass (scaling does not affect the objective)
        s = candidate.sum()
        if s < 0.01:
            continue
        candidate *= (current.sum() / s)

        est = _score_estimate(candidate)

        # SA acceptance
        if est < current_score or rng.random() < math.exp(-(est - current_score) / temperature):
            current = candidate
            current_score = est

        # update the global best (exact evaluation only if promising)
        if est < best_score - 1e-12:
            exact = float(evaluate_sequence(candidate.tolist()))
            if exact < best_score:
                best = candidate.copy()
                best_score = exact
                print(f"[{it:07d}] NEW BEST → {best_score:.6f}")

        it += 1
        if it % report_interval == 0:
            print(f"[{it:07d}] best={best_score:.6f}  cur={current_score:.6f}")

    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    return [float(x) for x in best]
```
