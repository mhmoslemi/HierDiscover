def project_simplex(v, s=1.0):
    """Projects v onto the simplex {x >= 0, sum x = s}."""
    # v: numpy array
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # find rho = largest i where condition true
    rho = ind[cond][-1]  # if cond non-empty
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v, s=1.0):
    """Projection of vector v onto simplex {x >= 0, sum x = s}."""
    # Code from Duchi et al. (2008)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent search for a low‑value step‑function.

The evaluation function is

    obj(a) = 2·n·max(a*a) / (sum(a)²)      (n = len(a))

It is scale‑invariant, so we keep the sequence normalised to
sum(a)=1 and minimise   max(a*a).

For a fixed sequence a the quantity max(a*a) is the ℓ∞‑norm of the
autoconvolution b = a*a.  Its sub‑gradient at the index k where b
is maximal (or at a softmax over the top indices) is
g_i = 2·a_{k‑i}.  Using a smooth version of the max (softmax
weights w_k ∝ exp(β·(b_k‑max_b))) gives a well‑behaved gradient.
We perform a projected gradient step onto the probability simplex
{a_i≥0, Σa_i=1} and enforce symmetry a_i = a_{n‑1‑i} after each
step.  The step size is adapted, occasional random block
perturbations are used for escaping plateaus and a simple restart
strategy is applied when no improvement is seen for a while.

The routine runs for the allotted time (default 1000 s) and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top‑level – no closures)
# ----------------------------------------------------------------------


def make_power_law(n: int, alpha: float) -> np.ndarray:
    """Power‑law a_i = (i+1)^{-α} (i=0..n‑1)."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -alpha)
    return a


def init_arcsine(n: int) -> np.ndarray:
    """Arcsine‑like start: a_i = 1/√(i+1) normalised to sum=1."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    a /= a.sum()
    return a


def project_to_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Project vector v onto the simplex {x ≥ 0, Σx = s}.
    Implementation of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        tau = 0.0
    else:
        rho = ind[cond][-1] - 1  # zero‑based index
        tau = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - tau, 0.0)
    return w


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Small random multiplicative perturbation of a contiguous block.
    Mirrors the whole vector with 35 % probability (keeps it roughly symmetric).
    """
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def compute_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute a softmax‑weighted gradient of max(a*a).

    Returns g where g_i = 2·Σ_k w_k·a_{k‑i}
    with w_k = exp(β·(b_k‑max_b)) / Σ_j exp(...).
    """
    n = a.shape[0]
    # full convolution
    b = np.convolve(a, a)
    max_b = b.max()

    # soft‑max weights (stable)
    shifted = b - max_b                # ≤ 0
    w_raw = np.exp(beta * shifted)     # max weight = 1
    w_sum = w_raw.sum()
    if w_sum == 0:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = w_raw / w_sum

    # gradient = 2 * (w * a_rev) convolved, slice the central n entries
    a_rev = a[::-1]
    conv = np.convolve(w, a_rev)                     # length 3n‑2
    g = 2.0 * conv[n - 1:n - 1 + n]                  # take n entries
    return g, max_b

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The algorithm uses projected gradient descent with a soft‑max
    approximation of the ℓ∞‑norm of the autocorrelation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine working length.  Use the supplied construction if present.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))          # supplied by the platform
        if n < 2:
            raise Exception
    except Exception:
        n = 5000  # a safe default size

    # ------------------------------------------------------------------
    # Initial candidates – keep the best of a few power‑law seeds.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # arcsine start (the analytic construction)
    cand = init_arcsine(n)
    score = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(score):
        best_seq = cand.copy()
        best_score = score

    # a few random power‑law exponents around 0.5
    for _ in range(5):
        alpha = rng.uniform(0.44, 0.58)
        a = make_power_law(n, alpha)
        a /= a.sum()                     # keep sum = 1 for easy handling
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq = a.copy()
            best_score = sc

    # try the historic construction if it exists
    try:
        hist = np.asarray(height_sequence_1, dtype=np.float64)
        if hist.ndim == 1 and hist.size > 0 and np.isfinite(hist).all():
            hist = np.clip(hist, 0.0, 1000.0)
            hist /= hist.sum()
            sc = float(evaluate_sequence(hist.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_seq = hist.copy()
                best_score = sc
    except Exception:
        pass

    if best_seq is None:                     # ultra‑safe fallback
        best_seq = np.ones(n, dtype=np.float64) / n

    # ------------------------------------------------------------------
    # Parameters for the projected gradient descent
    # ------------------------------------------------------------------
    a = best_seq.copy()                      # current iterate
    lr = 0.12                                 # initial step size
    beta = 5.0                                # soft‑max sharpness
    min_lr = 1e-5
    max_lr = 0.30
    iter_cnt = 0
    last_improve = time.monotonic()
    perturb_prob = 0.015                     # occasional random block perturb

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iter_cnt += 1

        # Compute gradient and the current maximal convolution value
        g, cur_max_b = compute_gradient(a, beta)

        # Gradient step (move opposite to the gradient of the max)
        cand = a - lr * g

        # Keep the sequence non‑negative and enforce symmetry
        cand = np.maximum(cand, 0.0)
        cand = 0.5 * (cand + cand[::-1])

        # Project onto the simplex Σa_i = 1
        cand = project_to_simplex(cand, s=1.0)

        # Small random block perturbation to escape flat regions
        if rng.random() < perturb_prob:
            cand = make_block_perturbation(cand, rng)
            cand = project_to_simplex(cand, s=1.0)

        # Evaluate the true objective (cheaper than recomputing max_b)
        score = float(evaluate_sequence(cand.tolist()))

        # Keep the best ever‑seen sequence
        if np.isfinite(score) and score < best_score:
            best_seq = cand.copy()
            best_score = score
            last_improve = time.monotonic()
            print(f"[{time.monotonic():.2f}] iter={iter_cnt}  new best = {best_score:.6f}")

            # on success enlarge the step size a little
            lr = min(max_lr, lr * 1.03)

        else:
            # shrink step size when nothing improves
            lr = max(min_lr, lr * 0.995)

        # Accept the candidate for the *inner* descent if it reduces the
        # surrogate (the max‑convolution) – this keeps the optimisation
        # moving even when the exact evaluate_sequence does not improve
        # every single step.
        cand_max_b = np.convolve(cand, cand).max()
        if cand_max_b < cur_max_b:
            a = cand
        else:
            # occasional acceptance to add stochasticity
            if rng.random() < 0.003:
                a = cand

        # Restart if we have been idle for a while
        if time.monotonic() - last_improve > 30.0:
            alpha = rng.uniform(0.44, 0.58)
            a = make_power_law(n, alpha)
            a /= a.sum()
            a = 0.5 * (a + a[::-1])          # keep it roughly symmetric
            lr = 0.12
            last_improve = time.monotonic()
            print(f"[{time.monotonic():.2f}] iter={iter_cnt}  restart with α={alpha:.4f}")

        # Periodic status output
        if iter_cnt % 500 == 0:
            print(f"[{time.monotonic():.2f}] iter={iter_cnt}  cur_max={cur_max_b:.9f}  best_score={best_score:.6f}")

        # Early exit if the target bound is reached
        if best_score < 1.503:
            print("Target bound achieved – exiting early.")
            break

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the
    # evaluator – this is merely cosmetic as evaluate_sequence does it
    # internally as well).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Search finished – best_score = {best_score:.6f} after {iter_cnt} iterations.")
    return [float(x) for x in best_seq]
