"""
Propose a (near‑optimal) step‑function for the provided evaluation metric.

Algorithm
---------
1. Determine the working length `n` from the global variable
   `height_sequence_1` (fallback to 1024).  The target total sum is
   `sqrt(2*n)`, because the score is invariant under a global scaling
   of the sequence (except for clipping at 1000 which never occurs for
   our amplitudes).

2. **Global exploration** – sample a large number of three‑parameter
   families

        a_i = (i + α)^{-p} · exp(-β·i)          (i = 0 … n‑1)

   with
        p ∈ [0.35, 0.80]    α ∈ [0.01, 10]    β ∈ [0, 5·10⁻³]

   Each candidate is re‑scaled to the prescribed sum and evaluated with
   an inexpensive FFT based convolution.  The best parameters are kept.

3. **Local refinement** – starting from the best parameters a simple
   simulated‑annealing / hill‑climbing loop is run until the time budget
   expires.

   * Move type A – *parameter perturbation* – add Gaussian noise to
     (p, α, β) and rebuild the sequence.
   * Move type B – *mass‑transfer* – compute the convolution, locate the
     index `k_max` achieving the maximum, compute the contribution of each
     entry to that peak, then move a small fraction of mass from the
     largest contributor to the smallest.  The total sum is kept constant.

   Improvements are accepted greedily; a tiny temperature schedule allows
   occasional uphill steps.

4. The best sequence discovered (as a plain Python list of floats) is
   returned.  All values are non‑negative and far below the 1000‑clip
   limit, so the built‑in security checks of `evaluate_sequence` never
   intervene.
"""
import time
import math
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def max_convolution_fft(a: np.ndarray) -> Tuple[np.ndarray, float, int]:
    """
    Compute the full convolution a*a using a real FFT, return the
    convolution array, its maximum value and the index of that maximum.
    """
    n = a.shape[0]
    size = 2 * n - 1
    # real FFT of the sequence (zero‑padded to `size`)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return conv, max_val, max_idx


def score_from_seq(a: np.ndarray) -> float:
    """
    Compute the evaluation score for a non‑negative sequence `a`.
    This reproduces the logic of `evaluate_sequence` (without the
    clipping to [0,1000] – our candidates never hit those limits).
    """
    n = a.shape[0]
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    _, max_val, _ = max_convolution_fft(a)
    return 2.0 * n * max_val / (s * s)


def generate_family(p: float, alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Produce the raw (un‑scaled) sequence belonging to the three‑parameter
    family a_i = (i+α)^{-p} * exp(-β·i).
    """
    idx = np.arange(n, dtype=np.float64)
    # (i+α)^{-p} part – protect against α=0 by ensuring α>0
    base = (idx + alpha) ** (-p)
    if beta != 0.0:
        base *= np.exp(-beta * idx)
    return base


def normalise_sum(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale `a` so that its sum equals `target_sum` (preserving non‑negativity)."""
    current = float(np.sum(a))
    if current == 0:
        return a
    scale = target_sum / current
    a = a * scale
    # safety clipping – values stay far below the evaluator's 1000‑limit
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def contribution_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given convolution index `k` (where the maximum is attained),
    return a vector `c` where c[i] = a[i]·a[k-i] if the product exists,
    otherwise 0.
    """
    n = a.shape[0]
    i = np.arange(n, dtype=np.int64)
    j = k - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[i[mask]] * a[j[mask]]
    return contrib


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> List[float]:
    """
    Generate a step‑function that (hopefully) minimizes the provided
    evaluation metric.  The function respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # leave a tiny safety margin

    # --------------------------------------------------------------
    # 1️⃣ Determine `n` and target sum.
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        n = 1024                      # modest fallback length
    target_sum = math.sqrt(2.0 * n)

    print(f"[setup] Working length n = {n}, target sum = {target_sum:.4f}")

    # --------------------------------------------------------------
    # 2️⃣ Global random exploration of the (p,α,β) family.
    # --------------------------------------------------------------
    best_score = float("inf")
    best_seq = None
    best_params = None

    # Fixed number of random samples – we stop early if we run out of time.
    MAX_RANDOM_SAMPLES = 8000
    for it in range(MAX_RANDOM_SAMPLES):
        if time.monotonic() >= deadline:
            break

        # Sample parameters from generous ranges.
        p = rng.uniform(0.35, 0.80)          # exponent
        alpha = rng.uniform(0.01, 10.0)      # offset
        beta = rng.uniform(0.0, 5e-3)        # exponential tail factor

        cand = generate_family(p, alpha, beta, n)
        cand = normalise_sum(cand, target_sum)
        sc = score_from_seq(cand)

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (p, alpha, beta)
            print(f"[explore] iter {it:04d}: p={p:.4f} α={alpha:.4f} β={beta:.6f} → score={sc:.6f}")

    if best_seq is None:
        # Extremely unlikely – fall back to uniform.
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = score_from_seq(best_seq)
        best_params = (0.5, 1.0, 0.0)
        print("[fallback] using uniform sequence")

    # --------------------------------------------------------------
    # 3️⃣ Local refinement – simulated annealing / hill‑climbing.
    # --------------------------------------------------------------
    # State variables
    cur_seq = best_seq.copy()
    cur_score = best_score
    cur_params = list(best_params)   # mutable list [p, α, β]

    # Simple temperature schedule: start with a modest temperature and
    # linearly decay to a tiny floor.
    T_initial = 1e-2 * cur_score
    T_min = 1e-6

    # Counters for occasional random jumps
    no_improve_counter = 0
    MAX_NO_IMPROVE = 2000

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        elapsed = time.monotonic() - start_time
        frac = elapsed / (budget_s - 2.0)
        # temperature linearly decays
        T = max(T_min, T_initial * (1.0 - frac))

        # Randomly decide which move to apply
        if rng.random() < 0.55:
            # ── Move type A: perturb parameters (p,α,β) ─────────────────────
            dp = rng.normal(0.0, 0.02 * (1.0 - frac) + 0.001)
            dα = rng.normal(0.0, 0.10 * (1.0 - frac) + 0.001)
            dβ = rng.normal(0.0, 1e-4 * (1.0 - frac) + 1e-6)

            new_p = max(0.10, min(2.0, cur_params[0] + dp))
            new_alpha = max(1e-5, cur_params[1] + dα)
            new_beta = max(0.0, cur_params[2] + dβ)

            cand = generate_family(new_p, new_alpha, new_beta, n)
            cand = normalise_sum(cand, target_sum)
            cand_score = score_from_seq(cand)

            accept = False
            if cand_score < cur_score:
                accept = True
            else:
                # probabilistic uphill move
                if rng.random() < math.exp((cur_score - cand_score) / T):
                    accept = True

            if accept:
                cur_seq = cand
                cur_score = cand_score
                cur_params = [new_p, new_alpha, new_beta]
                no_improve_counter = 0
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    best_params = (new_p, new_alpha, new_beta)
                    print(f"[refine] iter {iteration:05d} (param) → score={cand_score:.6f}")

        else:
            # ── Move type B: mass transfer based on the dominant convolution index ─
            # Compute convolution and locate the peak
            _, _, k_max = max_convolution_fft(cur_seq)

            # Compute contribution of each entry to that peak
            contrib = contribution_vector(cur_seq, k_max)

            donor = int(np.argmax(contrib))
            receiver = int(np.argmin(contrib))

            # Guard against degenerate case (all contributions equal)
            if donor == receiver:
                # fallback: random block perturbation (log‑normal factor)
                width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = min(n, start + width)
                cand = cur_seq.copy()
                cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
                if rng.random() < 0.35:
                    cand = 0.5 * (cand + cand[::-1])
                cand = np.clip(cand, 0.0, 1000.0)
                cand = normalise_sum(cand, target_sum)
            else:
                available = float(cur_seq[donor])
                if available <= 0.0:
                    continue  # nothing to move – skip this iteration
                delta = rng.uniform(0.0, min(0.5, 0.10) * available)
                # Ensure a minimal non‑zero transfer
                if delta < 1e-12:
                    delta = available * 0.01
                cand = cur_seq.copy()
                cand[donor] -= delta
                cand[receiver] += delta
                cand = np.clip(cand, 0.0, 1000.0)
                cand = normalise_sum(cand, target_sum)

            cand_score = score_from_seq(cand)

            accept = False
            if cand_score < cur_score:
                accept = True
            else:
                if rng.random() < math.exp((cur_score - cand_score) / T):
                    accept = True

            if accept:
                cur_seq = cand
                cur_score = cand_score
                no_improve_counter = 0
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    print(f"[refine] iter {iteration:05d} (mass) → score={cand_score:.6f}")
            else:
                no_improve_counter += 1

        # Occasionally restart from a fresh random family if stuck
        if no_improve_counter >= MAX_NO_IMPROVE and time.monotonic() < deadline:
            # new random seed for parameters
            p = rng.uniform(0.35, 0.80)
            alpha = rng.uniform(0.01, 10.0)
            beta = rng.uniform(0.0, 5e-3)
            cur_seq = generate_family(p, alpha, beta, n)
            cur_seq = normalise_sum(cur_seq, target_sum)
            cur_score = score_from_seq(cur_seq)
            cur_params = [p, alpha, beta]
            no_improve_counter = 0
            print(f"[restart] iter {iteration:05d} → new random params, score={cur_score:.6f}")

    print(f"[finished] best score achieved: {best_score:.6f}")
    # Convert to plain Python list as required by the grader.
    return [float(x) for x in best_seq]
