def compute_weights(conv, beta):
    max_c = np.max(conv)
    # to avoid underflow, shift.
    exp_vals = np.exp(beta * (conv - max_c))
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0:
        # all underflow -> uniform
        w = np.full_like(conv, 1.0/conv.size)
    else:
        w = exp_vals / sum_exp
    return w

"""
Propose a near‑optimal step function for the metric
    score = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines three ideas:

1. **Analytic seeds** – a power‑law shape (i+α)^‑½, a symmetric bell‑shape
   and the previous best vector are taken as starting points.

2. **Gradient descent on a smoothed “max‑conv”** – we use a soft‑max
   weighting of all convolution entries (β controls the temperature) and
   compute the gradient of the weighted sum using FFT‑based convolutions.
   The update is multiplicative (a ← a·exp(‑η·grad)) and the vector is
   re‑normalised to keep the sum = 1 (the score is scale‑invariant).

3. **Final fine‑grained hill climbing** – after the gradient phase we
   repeatedly move a small amount of mass from the entry that contributes
   most to the current maximal convolution entry to the entry that
   contributes least.  This mirrors the simple heuristic that gave the
   previous best result but works on the already‑refined vectors.

The whole search runs for up to ~990 s (leaving a safety margin).  All
numbers are kept non‑negative and far below the clipping bound 1000, so
the evaluator’s clipping does not affect the result.
"""

import time
import numpy as np


def make_block_perturbation(base, rng):
    """
    Cheap O(n) non‑negative perturbation: multiply a random contiguous block
    by a log‑normal factor and optionally symmetrise the whole vector.
    """
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def fft_convolve(x, y):
    """
    Convolution via real FFT.  Returns the full linear convolution
    (length len(x)+len(y)-1) as a float64 NumPy array.
    """
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()                 # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:n]
    return conv


def compute_weights(conv, beta):
    """
    Soft‑max weights for the convolution entries.
    w_k = exp(beta * (conv_k - max(conv))) normalized to sum to 1.
    """
    max_c = np.max(conv)
    # shift by max to avoid overflow; all exponents ≤ 0
    exp_vals = np.exp(beta * (conv - max_c))
    s = np.sum(exp_vals)
    if s == 0.0:
        # extreme underflow – fall back to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def gradient_weighted_conv(a, w):
    """
    Gradient of Σ_k w_k * (a*a)_k with respect to a.
    The formula reduces to 2 * (a_rev * w)[n‑1 : n‑1+n],
    where a_rev = a[::-1] and * is convolution.
    """
    grad_full = fft_convolve(a[::-1], w)          # length 3n‑2
    n = a.shape[0]
    return 2.0 * grad_full[n - 1:n - 1 + n]


def optimize(start_vec, deadline, name):
    """
    Run the soft‑max gradient descent on a single start vector.
    Returns the best vector found and its score.
    """
    a = np.asarray(start_vec, dtype=np.float64).copy()
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= s                                        # normalise to sum = 1

    cur_score = float(evaluate_sequence(a.tolist()))
    best_a = a.copy()
    best_score = cur_score
    print(f"[{name}] initial score = {cur_score:.6f}")

    # temperature schedule (larger β → more focus on the current maximum)
    betas = [1_000.0, 5_000.0, 20_000.0, 50_000.0]
    eta0 = 200.0                                     # multiplicative step size

    for beta in betas:
        eta = eta0
        it = 0
        while time.monotonic() < deadline:
            # 1) convolution of the current vector with itself
            conv = fft_convolve(a, a)                # length 2n‑1

            # 2) soft‑max weights
            w = compute_weights(conv, beta)

            # 3) gradient of the weighted sum
            grad = gradient_weighted_conv(a, w)

            # 4) zero‑mean the gradient (preserves the total sum)
            grad_centered = grad - np.mean(grad)

            # 5) multiplicative update, then renormalise
            factor = np.exp(-eta * grad_centered)
            new = a * factor
            new_sum = new.sum()
            if new_sum == 0.0:                        # fallback to uniform
                new = np.full_like(a, 1.0 / a.size)
                new_sum = 1.0
            new /= new_sum

            # 6) evaluate
            new_score = float(evaluate_sequence(new.tolist()))
            if np.isfinite(new_score) and new_score < best_score:
                best_score = new_score
                best_a = new.copy()
                print(f"[{name}] β={beta:.0f} it={it}  improved → {best_score:.6f}")

            # 7) acceptance decision (simple greedy with tiny random jumps)
            if new_score <= cur_score:
                a = new
                cur_score = new_score
            else:
                # occasional random jump to jump out of plateaus
                if np.random.rand() < 0.001:
                    a = new
                    cur_score = new_score

            # decay the step size a little
            eta *= 0.9995
            it += 1
            if it >= 5_000:          # safety cap per temperature
                break

    return best_a, best_score
