```
def _project_simplex(v, s):
    """Project vector v onto simplex {x: x>=0, sum x = s}."""
    # using method from Wang & Carreira-Perpinan
    if s < 0:
        raise ValueError
    n = v.shape[0]
    # Shift by mean? Standard algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Search for a non‑negative step sequence with minimal
`evaluate_sequence` value.

The search works in two phases.

Phase 1 – Projected gradient descent
------------------------------------
Starting from a reasonable baseline (the provided `height_sequence_1`
or a simple analytic family) we repeatedly
* compute the self‑convolution,
* locate the indices where the convolution reaches its maximum,
* compute the sub‑gradient of the maximal entry,
* take a small step opposite to the gradient (additive or multiplicative),
* project the result onto the probability simplex Σa_i = sqrt(2·n), a_i ≥ 0.

A tiny line‑search over a handful of step sizes guarantees monotone
improvement.  This phase quickly drives the maximal auto‑correlation
down from the hill‑climbing baseline.

Phase 2 – Light hill‑climbing refinements
-----------------------------------------
When the gradient phase stalls we fall back to cheap stochastic moves
(block‑wise scaling, single‑index scaling, mass redistribution,
random sparse re‑initialisation and power‑law families) exactly as in the
reference implementation, keeping any improvement.

All candidates are clipped to the range [0, 1000] required by the
official evaluator.  The best sequence found before the time budget
expires is returned as a plain Python list; a final call to
`evaluate_sequence` guarantees that the security wrapper is exercised.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast version of `evaluate_sequence` (no type checks)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Compute the same value as `evaluate_sequence` for a valid array."""
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx_i = s, x_i ≥ 0."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)                # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1] - 1               # last true index (0‑based)
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _compute_gradient(a: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(a*a) w.r.t. a (non‑negative)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()
    # tolerance to catch floating‑point ties
    max_idxs = np.flatnonzero(conv >= M - 1e-12)

    grad = np.zeros_like(a)
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        grad[i] += a[j]

        # diagonal contribution (i == j) must be counted twice
        if (k & 1) == 0:                 # k even
            i_mid = k // 2
            if i_start <= i_mid <= i_end:
                grad[i_mid] += a[i_mid]  # extra copy for the second term
    return grad


# ----------------------------------------------------------------------
# Candidate generators (used for initialisation and occasional restarts)
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """Geometric progression a_i = r**i, scaled to the required sum."""
    # Solve exp(t) = 1 + 2t for t (≈1.256) and set r = 1 - t/n.
    t = 1.0
    for _ in range(15):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    return (sum_target / raw_sum) * raw


def _sparse_random_candidate(n: int, sum_target: float,
                             rng: np.random.Generator) -> np.ndarray:
    """Create a random sparse vector (few non‑zero entries) and scale it."""
    k = rng.integers(max(2, int(n * 0.05)), max(3, int(n * 0.25)) + 1)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= (sum_target / total)
    return cand


def _powerlaw_candidate(n: int, sum_target: float,
                        rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} with random exponent p."""
    p = rng.uniform(0.2, 2.0)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cand = np.power(idx, -p)
    s = cand.sum()
    if s > 0.0:
        cand *= (sum_target / s)
    return cand

    # Phase 2 – Hill‑climbing refinements (same spirit as the reference)
    # ------------------------------------------------------------------
    hill_deadline = deadline
    while time.monotonic() < hill_deadline:
        mode = rng.integers(0, 6)

        if mode == 0:
            # block multiplicative perturbation
            cand = best.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.12))
            cand[start:start + width] *= factor
            if rng.random() < 0.2:
                cand = 0.5 * (cand + cand[::-1])

        elif mode == 1:
            # single index scaling
            cand = best.copy()
            i = rng.integers(n)
            factor = np.exp(rng.normal(0.0, 0.09))
            cand[i] *= factor

        elif mode == 2:
            # move a fraction of mass between two indices
            cand = best.copy()
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * 0.4
            delta = eps * min(cand[i], cand[j])
            if cand[i] > cand[j]:
                cand[i] -= delta
                cand[j] += delta
            else:
                cand[j] -= delta
                cand[i] += delta

        elif mode == 3:
            # fresh sparse random candidate
            cand = _sparse_random_candidate(n, sum_target, rng)

        elif mode == 4:
            # power‑law candidate
            cand = _powerlaw_candidate(n, sum_target, rng)

        else:   # mode == 5
            # small random redistribution over the whole vector
            cand = best.copy()
            mask = rng.random(n) < 0.05
            if mask.any():
                transfer = cand[mask] * rng.uniform(0.0, 0.3, size=mask.sum())
                cand[mask] -= transfer
                cand[~mask] += transfer.sum() / (~mask).sum()
        # clip to legal interval (the fast scorer does the same)
        cand = np.clip(cand, 0.0, 1000.0)

        cur_score = _fast_score(cand)
        if cur_score < best_score:
            best, best_score = cand.copy(), cur_score
            elapsed = time.monotonic() - (deadline - budget_s + 4.0)
            print(f"[{elapsed:5.1f}s] hill improvement → {best_score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list (official evaluator will run)
    # ------------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f}")
    return final
