def fast_score(seq):
    arr = np.asarray(seq, dtype=np.float64)
    n = arr.shape[0]
    if arr.sum() < 0.01:
        return np.inf
    # compute linear convolution via FFT with zero padding length 2*n-1
    size = 2 * n - 1
    f = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(f * f, n=size)
    max_b = conv.max()
    return 2.0 * n * max_b / (arr.sum() ** 2)

"""
Propose a step‑function that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Analytic seeding** – a fine grid search over the two‑parameter family
   a_i = (i+α)^‑p (p≈0.5) to obtain a very good initial vector.

2. **Multi‑peak hill‑climbing** – repeatedly
   * compute the full auto‑convolution,
   * look at the *k* largest convolution entries (the “hot spots”),
   * accumulate for every index i how much it contributes to those hot spots,
   * move a modest amount of mass from the worst (largest contributor) to the best
     (smallest contributor).  This flattens the convolution and lowers the
     maximal value.

   The move size adapts dynamically and a tiny probability of accepting a
   non‑improving move (simulated‑annealing style) helps to escape plateaus.

3. **Random‑restart polishing** – when the time budget allows, the current
   best vector is perturbed (log‑normal block scaling, optional mirroring) and
   hill‑climbing is started again.  The overall best vector seen during the run
   is finally returned.

All vectors are kept non‑negative and clipped to the interval [0,1000] as
required by ``evaluate_sequence``.  The heavy lifting (convolution and score)
is performed with FFT‑based convolution for speed; the final score is obtained
through the provided ``evaluate_sequence`` function.
"""

import numpy as np
import time
from typing import List

# ----------------------------------------------------------------------
# Helper: fast evaluation of the same objective that ``evaluate_sequence``
# computes, but without the safety checks.  This function is used for the
# internal search because it is much faster than calling ``evaluate_sequence``
# many times.
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the objective value for a non‑negative sequence ``seq``."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    n = seq.size
    size = 2 * n - 1                     # length of linear convolution
    # FFT‑based linear convolution (real input ⇒ use rfft/irfft)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)
    max_b = conv.max()
    return 2.0 * n * max_b / (total * total)


# ----------------------------------------------------------------------
# Helper: cheap non‑negative perturbation of a base vector.
# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a cheap O(n) perturbation:
      * pick a random contiguous block,
      * multiply it by a log‑normal factor,
      * optionally mirror the whole vector,
      * clip to [0,1000].
    """
    cand = base.copy()
    n = cand.size
    # block width: between 0.5 % and 15 % of the length
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main entry point required by the grader.
# ----------------------------------------------------------------------

    """
    Search for a near‑optimal step‑function under the 1000 s time budget.

    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # Determine working length n (use the globally supplied sequence if it
    # exists and is well‑formed, otherwise fall back to a modest size).
    # ------------------------------------------------------------------
    try:

        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():

    n = int(base.size)

    # ------------------------------------------------------------------
    # 1) Analytic grid search over a_i = (i+α)^-p
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    best_seq = None
    best_score = float("inf")
    # coarse grid – p fixed to 0.5, α in a wide range
    alphas = np.linspace(0.01, 4.0, 81)       # step ≈0.05
    p_fixed = 0.5
    for alpha in alphas:
        cand = (idx + alpha) ** (-p_fixed)
        sc = _fast_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # fine grid around the best α, also vary p slightly
    if best_seq is not None:
        alpha_center = alphas[np.argmin([_fast_score((idx + a) ** (-p_fixed)) for a in alphas])]
        p_vals = np.linspace(0.45, 0.55, 41)

                                 alpha_center + 0.3, 61)

                sc = _fast_score(cand)

    # Ensure we have a finite starting vector

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
