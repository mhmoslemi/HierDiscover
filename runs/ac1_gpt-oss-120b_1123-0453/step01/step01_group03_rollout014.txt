def _proj_simplex(v, target_sum):
    """
    Project a vector onto the simplex {x >= 0, sum x = target_sum}.
    """
    # Using algorithm from Duchi et al.
    n = v.shape[0]
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    # Find rho
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Edge cases: if target_sum is zero, projection is zero vector.

"""
Frank‑Wolfe style optimisation for the self‑convolution bound.
-----------------------------------------------------------------
We treat the problem

    minimise   max_k (a * a)[k]          (a_i ≥ 0, Σ a_i = √(2n))

which is scale‑invariant, so we keep the L1‑norm fixed to
√(2 n) throughout the search.  Starting from a few analytically
motivated seeds we iteratively improve the vector using the
following steps (the idea comes from the paper cited in the
prompt):

1.  Compute the current self‑convolution and its maximum M.
2.  Select a small “active set’’ of indices where the convolution
    is close to M (these constraints are the ones that are tight).
3.  Solve the linear programme

        maximise   Σ g_i
        subject to 0 ≤ g_i
                    (a * g)[k] ≤ M   for k in the active set

   (the constraints are linear in g).  This gives a direction g that
   would increase the feasible L1‑mass without breaking the active
   bounds.
4.  Rescale g to the required L1‑norm and move a small step towards
   it:   a ← (1‑t) a + t g   (t is a decreasing step‑size).
5.  Keep the best vector seen so far.  Occasionally we inject a fresh
   random sparse candidate or apply a tiny multiplicative jitter to
   avoid getting stuck.

The linear programme has only a handful of rows (≤ 30) even for
n≈ 5000, so each iteration is cheap and we can afford many of them
within the 1000 s budget.  The routine returns the best sequence as a
plain Python list; the outer framework will then call the official
`evaluate_sequence` on it.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast evaluator (identical to the official one but without security checks)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Simple seed generators
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """a_i = r**i, r ≈ 1‑t/n with t solving e^t = 1+2t (t≈1.256)."""
    # solve e^t = 1 + 2 t (Newton)
    t = 1.0
    for _ in range(30):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    cand = (sum_target / raw_sum) * raw
    return cand


def _sparse_random_candidate(n: int, sum_target: float,
                             rng: np.random.Generator) -> np.ndarray:
    """Randomly pick k≈5‑25 % positions and fill them with Exp(1) values."""
    k = rng.integers(max(1, int(n * 0.05)), max(2, int(n * 0.25) + 1))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= (sum_target / total)
    return cand


def _powerlaw_candidate(n: int, sum_target: float,
                        rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 3.0)
    cand = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    s = cand.sum()
    if s > 0.0:
        cand *= (sum_target / s)
    return cand


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = target}
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex with sum=target."""
    if target <= 0.0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Build the linear‑programming matrix for a given active set
# ----------------------------------------------------------------------
def _build_ub_matrix(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    For each k in 'active' construct the row representing
        (f * g)[k] = Σ_i f_i * g_{k-i}  ≤  M .
    The row has shape (n,) with entry j = f_{k-j} if 0≤k-j<n else 0.
    """
    n = f.shape[0]
    K = active.size
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(active):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1, dtype=np.intp)
        A[row, j_range] = f[k - j_range]
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine runs until the supplied time budget
    expires and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)   # safe fallback

    n = int(base.size)
    sum_target = np.sqrt(2.0 * n)               # normalisation used in the paper

    # ------------------------------------------------------------------
    # Initialise a few deterministic/random seeds
    # ------------------------------------------------------------------
    seeds = [
        _geometric_candidate(n, sum_target),
        _geometric_candidate(n, sum_target)[::-1],
        _powerlaw_candidate(n, sum_target, rng),
        _sparse_random_candidate(n, sum_target, rng)
    ]

    # keep the best of the initial seeds
    best = min(seeds, key=_fast_score)
    best_score = _fast_score(best)

    print(f"[init] n={n}, target_sum={sum_target:.3f}, best init score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main Frank‑Wolfe style optimisation loop
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 3.0)
    total_time = deadline - start

    # current working vector (starts from the best seed)
    cur = best.copy()
    cur_score = best_score

    # hyper‑parameters
    t0 = 0.4                # initial step size towards the LP direction
    eps0 = 0.35             # initial fraction of total mass moved per LP step
    max_active = 30         # maximal number of active constraints
    min_active = 5          # minimal number even for very flat conv

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start
        # ------------------------------------------------------------------
        # 1.  Compute current self‑convolution and its maximum M
        # ------------------------------------------------------------------
        conv = np.convolve(cur, cur)
        M = conv.max()
        cur_score = _fast_score(cur)

        # ------------------------------------------------------------------
        # 2.  Determine active constraints (indices where convolution ≈ M)
        # ------------------------------------------------------------------
        # tolerance shrinks as we progress, ensuring a few active rows.
        tol = 1e-3 + 0.4 * (1.0 - elapsed / total_time)   # from 0.7 → 0.001
        active_idx = np.where(conv >= M * (1.0 - tol))[0]
        if active_idx.size > max_active:
            # keep only the largest values
            order = np.argsort(conv[active_idx])[::-1][:max_active]
            active_idx = active_idx[order]
        elif active_idx.size < min_active:
            # guarantee at least a few rows
            order = np.argsort(conv)[::-1][:min_active]
            active_idx = order

        # ------------------------------------------------------------------
        # 3.  Solve the linear programme maximise Σ g_i  s.t. (cur * g)[k] ≤ M
        # ------------------------------------------------------------------
        A_ub = _build_ub_matrix(cur, active_idx)
        b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(g) → minimise -sum(g)

        # Linprog with HiGHS (fast for many variables, few rows)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            # fallback: small random jitter to escape a possible degenerate state
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            cur = np.clip(cur * jitter, 0.0, 1000.0)
            cur = _proj_simplex(cur, sum_target)
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            # nothing useful found – inject a random sparse pattern
            cur = _sparse_random_candidate(n, sum_target, rng)
            continue

        # ------------------------------------------------------------------
        # 4.  Scale g to the normalised L1‑norm and move towards it
        # ------------------------------------------------------------------
        g = g * (sum_target / sum_g)

        # step size schedule (decreases linearly)
        t = max(0.05, t0 * (1.0 - elapsed / total_time))
        cand = (1.0 - t) * cur + t * g

        # ensure legal interval before projection (clip to [0,1000])
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _proj_simplex(cand, sum_target)

        cand_score = _fast_score(cand)

        # ------------------------------------------------------------------
        # 5.  Keep the best ever seen; accept the new point if it improves
        # ------------------------------------------------------------------
        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score
            print(f"[{elapsed:5.1f}s] new best = {best_score:.6f}")

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
        else:
            # simulated‑annealing style acceptance to escape plateaus
            temp = max(1e-4, 0.1 * (deadline - time.time()) / total_time)
            if rng.random() < np.exp(-(cand_score - cur_score) / temp):
                cur, cur_score = cand, cand_score

        # ------------------------------------------------------------------
        # 6.  Occasionally inject diversity (random sparse restart or jitter)
        # ------------------------------------------------------------------
        if rng.random() < 0.02:
            # tiny multiplicative jitter
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            cur = np.clip(cur * jitter, 0.0, 1000.0)
            cur = _proj_simplex(cur, sum_target)

        if rng.random() < 0.005:
            # full random restart (helps when stuck)
            cur = _sparse_random_candidate(n, sum_target, rng)

    # ------------------------------------------------------------------
    # Return the best vector as plain Python list
    # ------------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    # One final call to the official evaluator (adds security clipping)
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f}")
    return final
