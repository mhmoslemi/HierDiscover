def compute_gradient(seq: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    # compute r = conv(seq, seq) using next_pow_two(2*n-1).
    conv_full = np.fft.irfft(np.fft.rfft(seq, size_conv) * np.fft.rfft(seq, size_conv), n=size_conv)[:2*n-1]
    # compute softmax weights
    max_r = conv_full.max()
    exps = np.exp(alpha * (conv_full - max_r))
    w = exps / exps.sum()
    # compute gradient using correlation
    size_corr = _next_pow_two(len(seq) + len(w) - 1)  # 3n-2 approx
    A = np.fft.rfft(seq, n=size_corr)
    W = np.fft.rfft(w, n=size_corr)
    # correlation via conj of W
    corr_complex = np.fft.irfft(A * np.conj(W), n=size_corr)
    # get linear correlation part length = n + len(w) -1
    corr_linear = corr_complex[:len(seq) + len(w) - 1]
    # gradient_i = 2 * corr_linear[i + len(w) - 1] (shift i)
    offset = len(w) - 1
    grad = 2.0 * corr_linear[offset:offset+len(seq)]
    # approximate objective value smoothed: smoothed_max = (np.log(np.sum(np.exp(alpha * (conv_full - max_r)))) + max_r) / alpha
    # or compute directly as max_r + log(sum_exp)/alpha.
    softmax_max = max_r + np.log(np.sum(np.exp(alpha * (conv_full - max_r)))) / alpha
    return softmax_max, grad

def _smoothed_max_and_gradient(seq: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    n = seq.size
    # Compute convolution r
    conv_len = 2*n - 1
    size_conv = _next_pow_two(conv_len)
    # Compute full convolution: use real FFT (rfft)
    fa = np.fft.rfft(seq, n=size_conv)
    conv = np.fft.irfft(fa * fa, n=size_conv)[:conv_len]
    # Softmax weights
    max_r = conv.max()
    # shifting for numeric stability
    exp_vals = np.exp(alpha * (conv - max_r))
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        w = np.full_like(conv, 1.0/conv_len)
    else:
        w = exp_vals / sum_exp
    # Compute smoothed max value = max_r + log(sum_exp)/alpha

"""
Gradient‑based stochastic optimizer for the step‑function evaluation problem.

The core idea:
* Approximate the nonsmooth objective  f(a)=2·n·max(a*a)/(∑a)²  by a smooth “soft‑max”
  of the convolution values:  m_α(a)= (1/α)·log∑exp(α·(a*a)).
* The gradient of m_α is a simple correlation between the current sequence and
  the soft‑max weights (computable with FFT in O(n log n)).
* We run an Adam‑style update on the sequence, keep it non‑negative,
  renormalise to unit ℓ₁‑norm (scaling does not change the true objective) and
  occasionally try cheap random perturbations.
* A fast FFT‑based upper bound is used to decide whether to call the authoritative
  `evaluate_sequence`; only promising candidates are evaluated exactly.

The routine respects the 1000 s budget, prints progress and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = seq.size
    conv_len = 2 * n - 1
    size = _next_pow_two(conv_len)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:conv_len]
    return conv


def _score_estimate(seq: np.ndarray) -> float:
    """Fast evaluation of the official objective using FFT (no security checks)."""
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_full(seq).max()
    return 2.0 * n * max_b / (s * s)


def _softmax_grad(seq: np.ndarray, alpha: float):
    """
    Compute a smooth approximation of max(a*a) and its gradient.

    Returns
    -------
    m : float
        The smooth max value  m_α = max_r + log(∑exp(α·(r-max_r)))/α .
    grad : np.ndarray, shape (n,)
        Gradient of m_α with respect to `seq` (the partial derivative
        ∂m_α/∂a_i).  The gradient already includes the factor 2 coming from
        the fact that a*a contains each product a_i·a_j twice (i≠j).
    """
    n = seq.size
    # --- convolution r = a*a ------------------------------------------
    r = _conv_full(seq)                     # length = 2n‑1
    r_max = r.max()
    # --- soft‑max weights --------------------------------------------
    # stability: subtract r_max before exponentiation
    exp_vals = np.exp(alpha * (r - r_max))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp                  # weights, sum to 1
    # smooth max
    m = r_max + np.log(sum_exp) / alpha

    # --- gradient = 2 * correlation(a, w) ---------------------------
    # linear correlation y[i] = Σ_j a_j * w[i + j] for i=0,…,n‑1
    # correlation can be obtained as a convolution with the reversed w.
    w_rev = w[::-1]
    size = _next_pow_two(n + w.size - 1)    # enough for linear convolution
    A = np.fft.rfft(seq, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * W, n=size)[: n + w.size - 1]

    offset = w.size - 1                     # position where shift 0 sits
    start = offset - n + 1
    # grad_i = 2 * conv[offset - i]  → 2 * reversed slice
    grad = 2.0 * conv[start: offset + 1][::-1]   # length n
    return float(m), grad


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray,
                  rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)   # ≤20 % of source mass
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray,
                   rng: np.random.Generator) -> np.ndarray:
    """Add small Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)


def _flatten_move(seq: np.ndarray,
                  rng: np.random.Generator) -> np.ndarray:
    """Reduce the contribution of the currently worst convolution index."""
    cand = seq.copy()
    n = cand.size
    r = _conv_full(cand)
    k_max = int(r.argmax())

    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]
    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand

    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))

    delta = cand[src] * float(0.05 + 0.15 * rng.random())  # 5‑%‑20 %
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Gradient‑based optimiser for the step‑function problem.

    Parameters
    ----------
    seed : int
        Random seed (deterministic runs).
    budget_s : float
        Time budget in seconds (the caller subtracts a small safety margin).

    Returns
    -------
    list[float]
        The best non‑negative sequence found before the budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1️⃣ Initialise – use the supplied construction if available.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        base = np.ones(3000, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    if base.sum() < 0.01:                     # safeguard against pathological input
        base = np.ones_like(base)

    # Normalise to unit ℓ₁‑norm (scale‑invariant for the true objective)
    base = base / base.sum()
    n = base.size

    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length={n}, initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(best_seq)
    v = np.zeros_like(best_seq)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5                         # base learning rate

    # Soft‑max temperature (α).  Start soft → become sharper.
    alpha = 1.0

    it = 0
    report_every = 5000

    # occasional stochastic moves (same set as the previous implementation)
    move_functions = [_make_block_perturbation,
                      _perturb_swap,
                      _perturb_noise,
                      _flatten_move]

    # ------------------------------------------------------------------
    # 3️⃣ Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- gradient step -------------------------------------------------
        smoothed_max, grad = _softmax_grad(best_seq, alpha)   # O(n log n)
        # scaling factor for the true objective (sum = 1 ⇒ C = 2·n)
        C = 2.0 * n
        full_grad = C * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * full_grad
        v = beta2 * v + (1.0 - beta2) * (full_grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = best_seq - step
        cand = np.maximum(cand, 0.0)

        # renormalise to unit sum (preserves scale‑invariance)
        s = cand.sum()
        if s >= 0.01:
            cand = cand / s
        else:
            cand = best_seq.copy()

        # cheap bound – reject obviously worse candidates
        est = _score_estimate(cand)
        if est < best_score - 1e-12:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_seq = cand
                best_score = exact
                print(f"[{it:07d}] gradient improvement → {best_score:.6f}")

        # ---- occasional random perturbations (exploration) -----------------
        if rng.random() < 0.03:            # ~3 % of iterations
            move = rng.choice(move_functions)
            cand2 = move(best_seq, rng)
            cand2 = cand2 / cand2.sum()
            est2 = _score_estimate(cand2)
            if est2 < best_score:
                exact2 = float(evaluate_sequence(cand2.tolist()))
                if np.isfinite(exact2) and exact2 < best_score:
                    best_seq = cand2
                    best_score = exact2
                    print(f"[{it:07d}] random move improvement → {best_score:.6f}")

        # ---- temperature schedule -------------------------------------------
        if it % 1000 == 0 and alpha < 20.0:
            alpha = min(20.0, alpha * 1.07)   # slowly sharpen the soft‑max

        it += 1
        if it % report_every == 0:
            approx_obj = C * smoothed_max / (best_seq.sum() ** 2)
            print(f"[{it:07d}] approx={approx_obj:.6f}  best={best_score:.6f}")

    print(f"Finished after {it} iterations – best score = {best_score:.6f}")

    # Return as plain Python list (the caller may rescale if they wish)
    return [float(x) for x in best_seq]
