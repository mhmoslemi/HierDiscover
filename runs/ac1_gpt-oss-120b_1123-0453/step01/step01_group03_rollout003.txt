"""
Propose a step‑function with a low self‑convolution peak.

The evaluation for a normalised vector (sum = sqrt(2·n)) is exactly the
maximum entry of the full self‑convolution.  We therefore minimise
`max(conv(a,a))`.

Algorithm
---------
1. Initialise `a` from the provided `height_sequence_1` (or a geometric
   start) and normalise its ℓ₁‑norm to `sqrt(2·n)`.
2. Repeatedly perform a *projected gradient descent* on a smooth surrogate
   of the max‑function:
     * `p` is a soft‑max over the convolution values:
       `p_i ∝ exp(λ·(conv_i – max(conv)))`.  As λ grows the soft‑max
       concentrates on the current peak.
     * The gradient of the smooth objective w.r.t. `a` is
       `g = (a * rev(p))[2·n‑2‑k] + a`  (derived from the derivative of the
       convolution) and is obtained with two FFT‑based convolutions.
     * A step `a ← a – η·g` is taken, followed by projection onto the
       feasible set (non‑negative, ℓ₁‑norm fixed).  The vector is kept
       symmetric (`a ← (a + reverse(a))/2`) because the problem is
       invariant under reversal.
3. λ is increased slowly (continuation) and the step size η is scaled
   with `sum_target / ‖g‖` to keep updates moderate.
4. If no improvement of the best value is observed for a long stretch,
   a fresh sparse random vector is mixed in to escape possible plateaux.
5. The loop runs until the time budget expires; the best vector ever seen
   is returned as a plain list.  A final call to the official
   `evaluate_sequence` guarantees that the security checks are applied.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast score (identical to the official evaluator for valid inputs)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative sequence."""
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:                      # not a realistic situation here
        return float('inf')
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """Geometric progression a_i = r**i, scaled to have ℓ₁ = sum_target."""
    # Solve e^t = 1 + 2t (t ≈ 1.256); then r = 1 - t/n.
    t = 1.0
    for _ in range(20):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    cand = (sum_target / raw_sum) * raw
    return cand


def _sparse_random_candidate(n: int, sum_target: float,
                             rng: np.random.Generator) -> np.ndarray:
    """Random sparse vector (5‑25 % non‑zero entries) normalised to sum_target."""
    k = rng.integers(int(n * 0.05), int(n * 0.25) + 1)
    if k <= 0:
        return np.full(n, sum_target / n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= (sum_target / total)
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine runs for roughly `budget_s` seconds
    (leaving a small buffer) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimensions and normalisation
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = None

    if base is None:
        # fall back to a deterministic geometric start
        n = 256
        a = _geometric_candidate(n, np.sqrt(2.0 * n))
    else:
        n = int(base.size)
        a = base.astype(np.float64)

    sum_target = np.sqrt(2.0 * n)          # the normalisation used in the paper

    # Normalise the initial vector
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.0:
        a *= (sum_target / s)
    else:
        a = np.full(n, sum_target / n, dtype=np.float64)

    # Enforce symmetry (the problem is reversible)
    a = (a + a[::-1]) * 0.5

    # ------------------------------------------------------------------
    # Pre‑computed constants
    # ------------------------------------------------------------------
    L = _next_pow2(3 * n)                 # FFT length (covers all needed convolutions)
    idx_n = np.arange(n, dtype=np.int64)  # used for indexing the gradient

    # ------------------------------------------------------------------
    # Initial best solution
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = _fast_score(best)
    print(f"[init] n={n}, target‑sum={sum_target:.3f}, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Gradient‑descent parameters
    # ------------------------------------------------------------------
    lam = 0.5          # initial smoothing parameter (soft‑max temperature)
    lam_max = 12.0     # where we almost recover the true max
    lam_inc = 0.0005   # increase per iteration
    lr_factor = 0.7    # scales the raw step (tuned empirically)
    last_improve_it = 0
    symmetrise = True  # keep the vector symmetric after each update

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # tiny safety margin

    it = 0
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1️⃣  Compute the full self‑convolution of the current vector.
        # --------------------------------------------------------------
        fft_a = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
        max_b = conv_full.max()
        cur_score = max_b                     # because the normalisation makes the score = max_b
        if cur_score < best_score - 1e-12:
            best = a.copy()
            best_score = cur_score
            last_improve_it = it
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:5.1f}s] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2️⃣  Build a smooth approximation of the max via a soft‑max.
        # --------------------------------------------------------------
        # shift by the maximum for numerical stability
        shifted = conv_full - max_b
        exp_term = np.exp(lam * shifted)
        p = exp_term / exp_term.sum()         # shape (2n‑1,), sums to 1

        # --------------------------------------------------------------
        # 3️⃣  Gradient of the smooth objective.
        # --------------------------------------------------------------
        p_rev = p[::-1]                       # reverse to obtain the needed convolution
        fft_p_rev = np.fft.rfft(p_rev, n=L)
        conv_rev = np.fft.irfft(fft_a * fft_p_rev, n=L)[: 3 * n - 2]
        # g_first[k] = Σ_j a_j * p_{j+k}  (k = 0 … n‑1)
        g_first = conv_rev[(2 * n - 2) - idx_n]
        g = g_first + a                        # the extra “+a” comes from the derivative of a_k·a_k

        # --------------------------------------------------------------
        # 4️⃣  Take a projected gradient step.
        # --------------------------------------------------------------
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        step = lr_factor * sum_target / (norm_g + 1e-12)

        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)         # projection onto the non‑negative orthant
        s_new = a_new.sum()
        if s_new > 0.0:
            a_new *= (sum_target / s_new)      # re‑normalise the ℓ₁‑norm
        else:                                 # degenerated case (should not happen)
            a_new = a.copy()

        if symmetrise:
            a_new = (a_new + a_new[::-1]) * 0.5

        a = a_new

        # --------------------------------------------------------------
        # 5️⃣  Schedule the smoothing parameter.
        # --------------------------------------------------------------
        lam = min(lam_max, lam + lam_inc)

        # --------------------------------------------------------------
        # 6️⃣  Occasionally inject a little random noise to escape plateaux.
        # --------------------------------------------------------------
        if it % 3000 == 0 and it > 0:
            # if we have been stuck for a while, blend in a fresh sparse candidate
            if it - last_improve_it > 15000:
                fresh = _sparse_random_candidate(n, sum_target, rng)
                a = 0.5 * (a + fresh)           # gentle mixing
                a = (a + a[::-1]) * 0.5 if symmetrise else a

        # --------------------------------------------------------------
        # 7️⃣  Progress report (every few seconds)
        # --------------------------------------------------------------
        if it % 500 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:5.1f}s] iter={it}, λ={lam:.3f}, cur={cur_score:.6f}, best={best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    # a final official evaluation (security checks, clipping, etc.)
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f} after {it} iterations")
    return final
