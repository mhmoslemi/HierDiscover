def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Search for a low evaluation step sequence.
    Uses simulated annealing / hill climbing on fixed‑length vectors.
    Starts from the best known sequence if available, otherwise from a random one.
    Adjusts two entries at a time to keep the ℓ¹ norm = sqrt(2·n) invariant.
    """
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    best_seq = None
    best_score = math.inf
    # define helper to evaluate fast
    def score(seq):
        # ensure list
        return float(evaluate_sequence(seq.tolist()))
    # define helper to renorm (not needed if we keep sum constant)
    
    # candidate dimensions to try
    lengths = [200, 300, 400, 500]  # may adapt based on time
    # optionally also 150 or 250 for speed
    # for each length, run annealing for some iterations
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # Prepare starting vector
        if 'height_sequence_1' in globals():
            # If that sequence has correct length, use it.
            hs = globals()['height_sequence_1']
            if len(hs) == n:
                cur = np.array(hs, dtype=float)
                cur = np.clip(cur, 0.0, 1000.0)
                cur_sum = cur.sum()
                target = math.sqrt(2.0 * n)
                # renormalize to target sum (scaling invariant)
                if cur_sum > 0:
                    cur = cur * (target / cur_sum)
                else:
                    cur = np.full(n, target / n)
            else:
                # random start with correct sum
                cur = rng.random(n)
                cur = cur / cur.sum() * math.sqrt(2.0 * n)
        else:
            cur = rng.random(n)
            cur = cur / cur.sum() * math.sqrt(2.0 * n)
        # compute initial score
        cur_score = score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"init n={n} score={cur_score:.6f}")
        # parameters for annealing
        temp0 = 1.0
        temp_final = 1e-4
        max_iter = 200000  # large upper bound, will exit by time
        # schedule: exponential decay
        iter_done = 0
        while time.monotonic() < deadline and iter_done < max_iter:
            # temperature based on progress
            progress = iter_done / max_iter
            temp = temp0 * (temp_final / temp0) ** progress
            # pick two indices
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # choose delta up to a fraction of current value at i
            max_delta = cur[i] * 0.5  # move at most half of source
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            # apply move
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            # compute score
            new_score = score(new_seq)
            # acceptance
            if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / temp):
                cur = new_seq
                cur_score = new_score
                # update global best
                if new_score < best_score:
                    best_score = new_score
                    best_seq = cur.copy()
                    print(f"[n={n}] iter={iter_done} new best {best_score:.6f}")
            iter_done += 1
        print(f"Finished n={n} iters={iter_done} best_score={best_score:.6f}")
    # final safety
    if best_seq is None:
        # fallback uniform
        n = 200
        best_seq = np.full(n, math.sqrt(2.0*n)/n)
        best_score = score(best_seq)
        print("fallback uniform score", best_score)
    # clip and convert
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

def peak_contributions(seq, k):
    # compute contributions only for i where both term valid.
    # left = max(0, k-(n-1))
    # right = min(n-1, k)
    left = max(0, k-(n-1))
    right = min(n-1, k)
    i_range = np.arange(left, right+1)
    contributions = seq[i_range] * seq[k - i_range]

import time, math, numpy as np

def max_conv(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using FFT (fast)."""
    n = seq.shape[0]
    # next power of two >= 2*n-1
    m = 1 << ((2*n - 1).bit_length())
    # Fourier transform
    fa = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    # only first 2n-1 entries are valid
    return float(conv[:2*n-1].max())

def full_conv(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of seq with itself, using FFT."""
    n = seq.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    return conv[:2*n-1]

def move_random(seq: np.ndarray, rng: np.random.Generator, max_delta_factor: float = 0.5) -> np.ndarray:
    """Transfer a fraction of mass from a random source index to a random target index."""
    n = seq.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    src = i
    dst = j
    if seq[src] <= 0.0:
        return seq.copy()
    max_delta = seq[src] * max_delta_factor
    if max_delta <= 0.0:
        return seq.copy()
    delta = rng.random() * max_delta
    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    # Guard against tiny negative due to floating error
    if new_seq[src] < 0.0:
        new_seq[dst] += new_seq[src]   # add back the negative part
        new_seq[src] = 0.0
    return new_seq

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine performs a simulated‑annealing / hill‑climbing loop on several
    vector lengths.  Mass is moved between two coordinates while the ℓ¹‑norm
    (set to √(2·n)) stays invariant, because the score only depends on the
    maximum auto‑convolution entry when the sum is normalised.
    A small fraction of moves are *targeted*: they look at the current
    convolution peak and shift mass away from indices that contribute most to
    that peak.  The best sequence found before the time budget expires is
    returned as a plain Python list of floats.
    """
    # initialisation
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = math.inf

    # try to start from the globally supplied sequence if it exists
    if 'height_sequence_1' in globals():
        seq0 = np.array(globals()['height_sequence_1'], dtype=float)
        seq0 = np.clip(seq0, 0.0, 1000.0)
        # normalise to the required ℓ¹‑norm for its length
        n0 = seq0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        cur_sum = seq0.sum()
        if cur_sum > 0:
            seq0 = seq0 * (target0 / cur_sum)
        sc0 = max_conv(seq0)
        best_seq = seq0.copy()
        best_score = sc0
        print(f"[init] using global height_sequence_1, n={n0}, score={sc0:.6f}")

    # list of lengths to explore – larger lengths give a larger search space
    lengths = [200, 300, 400, 500, 600, 800]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # initialise a working sequence
        if best_seq is not None and best_seq.shape[0] == n:
            cur = best_seq.copy()
            # re‑scale to exact target sum (scaling does not affect the score)
            s = cur.sum()
            if s > 0:
                cur = cur * (target_sum / s)
        else:
            cur = rng.random(n)
            cur = cur / cur.sum() * target_sum

        cur_score = max_conv(cur)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[init] n={n}, score={cur_score:.6f}")

        # iteration budget – a few hundred thousand moves for each length,
        # limited by remaining wall‑clock time.
        max_iters = int(150_000 * (n / 200))
        iter_done = 0
        while time.time() < deadline and iter_done < max_iters:
            # temperature – linear cooling from 1.0 to 1e‑4
            temp = 1.0 - (iter_done / max_iters) * (1.0 - 1e-4)

            # decide whether to perform a targeted move (30 % of the time)
            if rng.random() < 0.30:
                # compute the full convolution once – needed for the peak index
                conv_full = full_conv(cur)
                k = int(np.argmax(conv_full))

                # indices that actually contribute to this peak
                left = max(0, k - (n - 1))
                right = min(n - 1, k)
                idxs = np.arange(left, right + 1)
                contrib = cur[idxs] * cur[k - idxs]
                total = contrib.sum()
                if total > 0:
                    probs = contrib / total
                    i = rng.choice(idxs, p=probs)
                else:
                    i = rng.integers(left, right + 1)

                # choose a destination different from the source
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)

                max_delta = cur[i] * 0.5
                if max_delta > 0:
                    delta = rng.random() * max_delta
                    new_seq = cur.copy()
                    new_seq[i] -= delta
                    new_seq[j] += delta
                else:

                if cand_score < best_score:

    if best_seq is None:
        # deterministic fallback – uniform vector of length 200
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    # clip to the allowed range and return as plain Python list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # double‑check using the official scorer (mainly for debugging)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final best score (official) = {final_score:.6f}, length = {len(best_seq)}")
    return [float(x) for x in best_seq]

"""
Simulated‑annealing / hill‑climbing search for a low‑value step sequence.

The evaluation metric reduces (when the ℓ¹‑norm is fixed to √(2·n))
to the maximum entry of the auto‑convolution.
The algorithm therefore works directly with that quantity.

For a given length n we:
  * start from a random feasible vector (or from the globally supplied
    `height_sequence_1` if it has the same length);
  * repeatedly move a small amount of mass from one coordinate to another
    (the ℓ¹‑norm stays invariant);
  * with probability 0.3 the move is *targeted*: we look at the current
    convolution peak, pick a contributor proportional to its product
    and shift a fraction of it to a random other index;
  * accept the move if it improves the peak or with a Metropolis
    probability that cools linearly from 1.0 to 1e‑4;
  * keep the best sequence ever seen.

Time is split evenly among a list of candidate lengths
[200, 300, 400, 500, 600, 800]; the search stops as soon as the overall
budget expires.  The final sequence is clipped to the allowed interval
[0, 1000] and returned as a plain Python list.
"""

import time
import math
import numpy as np


def max_conv(seq: np.ndarray) -> float:
    """Maximum entry of seq * seq using an FFT‑based convolution."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fa = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    return float(conv[: 2 * n - 1].max())


def full_conv(seq: np.ndarray) -> np.ndarray:
    """Full linear auto‑convolution of seq (length 2n‑1) via FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    return conv[: 2 * n - 1]


def move_random(seq: np.ndarray, rng: np.random.Generator,
                max_delta_factor: float = 0.5) -> np.ndarray:
    """Transfer a fraction of mass from a random source index to a random target."""
    n = seq.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    src, dst = i, j
    if seq[src] <= 0.0:
        return seq.copy()
    max_delta = seq[src] * max_delta_factor
    if max_delta <= 0.0:
        return seq.copy()
    delta = rng.random() * max_delta
    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    # protect against tiny negatives caused by rounding
    if new_seq[src] < 0.0:
        new_seq[dst] += new_seq[src]
        new_seq[src] = 0.0
    return new_seq


def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.
    The function respects the 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    # leave a small margin for the final sanity check
    global_deadline = time.time() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # start from the globally provided sequence if it exists
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=float)
        seq0 = np.clip(seq0, 0.0, 1000.0)
        n0 = seq0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        s = seq0.sum()
        if s > 0.0:
            seq0 = seq0 * (target0 / s)
        sc0 = max_conv(seq0)
        best_seq = seq0.copy()
        best_score = sc0
        print(f"[init] global height_sequence_1 used, n={n0}, score={sc0:.6f}")

    # ------------------------------------------------------------------
    # lengths we will explore – larger n gives more freedom but costs more time
    lengths = [200, 300, 400, 500, 600, 800]
    total_lengths = len(lengths)

    for idx, n in enumerate(lengths):
        if time.time() >= global_deadline:
            break

        # split the remaining time roughly evenly among the still‑to‑process lengths
        remaining_lengths = total_lengths - idx
        time_left = global_deadline - time.time()
        per_len_deadline = time.time() + 0.9 * time_left / remaining_lengths

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # initialise a feasible vector for this length
        if best_seq is not None and best_seq.shape[0] == n:
            cur = best_seq.copy()
            s = cur.sum()
            if s > 0.0:
                cur = cur * (target_sum / s)
        else:
            cur = rng.random(n)
            cur = cur / cur.sum() * target_sum

        cur_score = max_conv(cur)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[init] n={n}, score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # main annealing loop for this length
        it = 0
        while time.time() < per_len_deadline:
            # linear cooling from 1.0 down to 1e‑4
            temp = 1.0 - (it / 1_000_000) * (1.0 - 1e-4)   # 1 M is a soft iteration cap
            # ------------------------------------------------------------------
            # 30 % of the time we do a *targeted* move
            if rng.random() < 0.30:
                conv_full = full_conv(cur)
                k = int(np.argmax(conv_full))

                # indices that actively contribute to the peak at k
                left = max(0, k - (n - 1))
                right = min(n - 1, k)
                idxs = np.arange(left, right + 1)
                contrib = cur[idxs] * cur[k - idxs]
                total = contrib.sum()
                if total > 0.0:
                    probs = contrib / total
                    i = rng.choice(idxs, p=probs)
                else:
                    i = rng.integers(left, right + 1)

                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)

                max_delta = cur[i] * 0.5
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    new_seq = cur.copy()
                    new_seq[i] -= delta
                    new_seq[j] += delta
                else:
                    new_seq = move_random(cur, rng)
            else:
                new_seq = move_random(cur, rng)

            cand_score = max_conv(new_seq)

            # Metropolis acceptance
            if cand_score < cur_score or rng.random() < math.exp(
                -(cand_score - cur_score) / (temp + 1e-12)
            ):
                cur = new_seq
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cur.copy()
                    print(f"[n={n}] it={it} new best {best_score:.6f}")

            it += 1

        print(f"[n={n}] finished after {it} iterations, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback if somehow nothing was produced
    if best_seq is None:
        n = 200
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    # final clip and official evaluation (useful for debugging)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final official score = {final_score:.6f}, length = {len(best_seq)}")

    # return a plain Python list of floats
    return [float(x) for x in best_seq]
