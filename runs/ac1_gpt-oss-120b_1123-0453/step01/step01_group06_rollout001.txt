def compute_conv(a):
    n = len(a)
    size = next_pow_two(2*n-1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

def compute_soft_weights(conv, alpha):
    # compute stable softmax
    maxc = conv.max()
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    return p

def compute_gradient(a, p):
    # a_rev = a[::-1]
    n = len(a)
    # p length M = 2n-1
    size = _next_pow_two(len(p) + n - 1)
    fp = np.fft.rfft(p, n=size)
    far = np.fft.rfft(a[::-1], n=size)
    conv_full = np.fft.irfft(fp * far, n=size)[:len(p) + n -1]
    # gradient slice:
    grad = 2.0 * conv_full[n-1:n-1 + n]
    return grad

def gradient_step(a, lr, alpha):
    conv = compute_conv(a)
    p = compute_soft_weights(conv, alpha)
    grad = compute_gradient(a, p)

    # we want to descend: a_new = a - lr * grad
    a_new = a - lr * grad

    # enforce nonnegativity
    a_new = np.maximum(a_new, 0.0)

"""
Hybrid gradient‑plus‑stochastic optimiser for the step‑function problem.

The algorithm keeps a single incumbent sequence (initialised from the
provided ``height_sequence_1``) and repeatedly improves it by two
complementary mechanisms:

1. **Soft‑max gradient step** – a differentiable surrogate of the
   L∞‑norm of the self‑convolution is constructed by a soft‑max
   weighting of the convolution values.  The gradient of this surrogate
   is obtained analytically (via FFT) and a small step in the negative
   direction is taken.  The temperature (soft‑max “α”) is increased
   gradually so that the optimisation first smoothes the landscape and
   later focuses on the true maximum.

2. **Exploratory perturbations** – block‑wise scaling, mass swaps,
   Gaussian noise and the specialised “flatten” move from the reference
   implementation are applied with a 50 % probability to introduce
   diversity and to escape shallow minima.

A cheap FFT based estimator of the objective (`_score_estimate`) is used
to filter candidates; only promising ones are evaluated with the
authoritative (and slower) `evaluate_sequence`.  The loop runs until the
time budget expires (the caller supplies a small safety margin).  The
best sequence found is returned as a plain list of floats.

All numbers are kept in the interval [0, 1000] and the total sum is
maintained (scale‑invariant) to avoid the ``sum_a < 0.01`` guard in the
objective.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – no closures, top‑level only
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv(seq: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (exact up to floating error)."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return float(_conv(seq).max())


def _score_estimate(seq: np.ndarray) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` – identical to the true
    objective when the sum is non‑zero (the function is scale invariant).
    """
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Soft‑max based gradient utilities
# ----------------------------------------------------------------------


def _soft_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Stable soft‑max weights for a vector of convolution values.
    The weights sum to one and concentrate on large entries when
    ``alpha`` is large.
    """
    maxc = conv.max()
    # subtract maxc for numerical stability
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        # fallback – uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp


def _gradient(seq: np.ndarray, alpha: float):
    """
    Gradient of the soft‑max weighted sum of the self‑convolution.
    Returns the gradient vector (length n), the raw convolution (length 2n‑1)
    and the soft‑max weight vector (length 2n‑1).
    """
    # full convolution of the current sequence
    conv = _conv(seq)                     # shape (2n‑1,)
    # soft‑max weights (probability distribution over the indices)
    w = _soft_weights(conv, alpha)        # shape (2n‑1,)

    # gradient = 2 * (w * a) where * denotes cross‑correlation.
    rev = seq[::-1]                       # reversed sequence
    size = _next_pow_two(w.size + rev.size - 1)
    fw = np.fft.rfft(w, n=size)
    fr = np.fft.rfft(rev, n=size)
    corr = np.fft.irfft(fw * fr, n=size)[: w.size + rev.size - 1]

    n = seq.size
    # gradient_i corresponds to slice n‑1 … n‑1+n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad, conv, w


def _gradient_step(seq: np.ndarray, lr: float, alpha: float) -> np.ndarray:
    """
    Perform one gradient descent step on ``seq``.
    The step is projected onto the non‑negative orthant and re‑scaled to
    keep the sum unchanged (the objective is homogeneous).
    """
    grad, _, _ = _gradient(seq, alpha)

    cand = seq - lr * grad
    cand = np.clip(cand, 0.0, 1000.0)

    # re‑normalise to the original sum (scale‑invariant)
    s = cand.sum()
    if s < 1e-12:                     # degenerate – fall back to the incumbent
        cand = seq.copy()
        s = cand.sum()
    cand *= (seq.sum() / s)
    return cand


# ----------------------------------------------------------------------
# Randomised perturbation moves (taken from the reference code)
# ----------------------------------------------------------------------


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:            # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)   # ≤20 % of source mass
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise to a modest sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)


def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the contribution to the currently worst convolution index.
    The idea is to move a modest amount of mass away from the pair
    responsible for the peak.
    """
    cand = seq.copy()
    n = cand.size
    # compute full convolution (FFT) and locate the maximal index
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())

    # indices that actually contribute to this sum
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]

    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand

    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))

    delta = cand[src] * float(0.05 + 0.15 * rng.random())   # 5‑%‑20 %
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Replace the sequence by its average with its reversal."""
    return np.clip(0.5 * (seq + seq[::-1]), 0.0, 1000.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the caller subtracts a small safety margin).

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    rng = np.random.default_rng(seed)

    # ---- initial sequence -------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # provided globally
    except Exception:
        base_seq = np.ones(1024, dtype=np.float64)

    base_seq = np.clip(base_seq, 0.0, 1000.0)

    # Normalise to sum = 1 (scale‑invariant, avoids the “sum_a < 0.01” guard)
    if base_seq.sum() == 0.0:
        base_seq[:] = 1.0
    base_seq = base_seq / base_seq.sum()

    n = base_seq.size
    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    best_est = _score_estimate(best)

    # ---- hyper‑parameters -------------------------------------------------
    # learning‑rate schedule (starts larger, decays)
    lr_start = 0.08
    lr_end = 0.001

    # soft‑max temperature schedule
    alpha_start = 5.0
    alpha_end = 300.0

    # move set for stochastic exploration
    move_fns = [
        _make_block_perturbation,
        _perturb_swap,
        _perturb_noise,
        _flatten_move,
        _symmetrize,
    ]

    # ---- main optimisation loop ------------------------------------------
    it = 0
    report_interval = 5000
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    while time.monotonic() < deadline:
        # fraction of the total budget already consumed
        elapsed = (time.monotonic() + float(budget_s) - deadline)  # approx elapsed
        f = min(1.0, elapsed / float(budget_s))

        # linearly interpolated parameters
        lr = lr_start * (1.0 - f) + lr_end * f
        alpha = alpha_start * (1.0 - f) + alpha_end * f

        # 50 % of the time use a gradient step, otherwise a random move
        if rng.random() < 0.5:
            cand = _gradient_step(best, lr, alpha)
        else:
            cand = rng.choice(move_fns)(best, rng)

        # keep the sum fixed (scale invariant)
        s = cand.sum()
        if s < 0.01:          # safety – revert to incumbent if degenerate
            cand = best.copy()
            s = cand.sum()
        cand *= (best.sum() / s)   # re‑scale to the incumbent sum ( = 1)

        # cheap fast estimate – decides whether a full evaluation is worth it
        est = _score_estimate(cand)

        if est < best_est - 1e-12:          # plausible improvement
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = cand
                best_score = exact
                best_est = est
                print(f"[{it:07d}] new best exact {best_score:.6f} (est {best_est:.6f})")

        it += 1
        if it % report_interval == 0:
            print(f"[{it:07d}] current best exact {best_score:.6f}, est {best_est:.6f}")

    print(f"Finished after {it} iterations – best exact score = {best_score:.6f}")
    # convert back to plain Python list
    return [float(x) for x in best]
