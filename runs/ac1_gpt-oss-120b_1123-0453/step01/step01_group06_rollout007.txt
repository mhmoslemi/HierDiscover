```python
import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow_two(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _conv_max(seq: np.ndarray) -> float:
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(conv[: 2 * n - 1].max())

def _score_estimate(seq: np.ndarray) -> float:
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)

def _solve_lp_max_sum(f: np.ndarray, M: float, row_idx: np.ndarray, col_idx: np.ndarray) -> np.ndarray | None:
    n = f.size
    # data for sparse matrix: repeat each f_i n times
    data = np.repeat(f, n)
    rows = 2 * n - 1
    A = csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))
    # Upper bound vector
    b = np.full(rows, M, dtype=np.float64)
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    # Use HiGHS solver
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start:start+width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    """
    Alternating LP‑based flattening optimizer.
    Starts from a power‑law shape, repeatedly solves a linear program
    max(sum(g)) under the current convolution bound and mixes the solution
    with the current sequence (convex combination).  Random block
    perturbations are occasionally applied to escape stagnation.
    The algorithm runs until the time budget expires and returns the best
    sequence encountered.
    """

"""
Hybrid LP‑based flattening optimizer for the step‑function problem.

The optimizer works in three phases:

1. **Warm‑start** – a short sweep over power‑law shapes a_i = (i+1)^{-p}
   (p∈[0.30,0.60]) normalised to the standard sum √(2 n).  The best
   shape according to the exact ``evaluate_sequence`` is kept.

2. **LP‑driven flattening** – given the current sequence f we compute the
   current autocorrelation bound M = max(conv(f,f)).  We then solve a
   linear programme

        maximise    Σ g_j
        subject to  g ≥ 0
                    conv(f, g) ≤ M   (for every convolution index)

   The solution g puts mass where f is already small, i.e. it “flattens”
   the profile without increasing the current worst convolution value.
   After scaling g to have the same L¹‑norm as f we replace

        f ← (1‑t)·f + t·g

   with a moderate step size t (≈0.2).  The bound is recomputed and the
   exact objective is queried only when the cheap FFT bound predicts an
   improvement.  This loop runs until the time budget expires.

3. **Exploration moves** – when the LP step does not yield a better
   bound we occasionally apply a cheap block‑wise multiplicative
   perturbation (the “make‑block‑perturbation” move) to escape possible
   local minima.  Any genuine improvement is accepted.

The whole procedure respects the 1000 s budget, uses only NumPy,
SciPy (HiGHS LP solver) and never performs I/O.  The best sequence
found is returned as a plain Python list of floats.
"""

import math
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT length)."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Fast (FFT) computation of max(conv(seq, seq))."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(conv[: 2 * n - 1].max())


def _score_estimate(seq: np.ndarray) -> float:
    """Fast surrogate for the exact objective (uses FFT)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


def _solve_lp_max_sum(f: np.ndarray,
                      M: float,
                      row_idx: np.ndarray,
                      col_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   subject to   g ≥ 0 , conv(f,g) ≤ M.
    Returns the optimal g (numpy array) or None on failure.
    """
    n = f.size
    # Build the sparse matrix A for constraints A·g ≤ M.
    # Each column i contains the value f_i on rows i … i+n‑1.
    data = np.repeat(f, n)                   # length n*n
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    # Upper‑bound vector (same M for every row)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective – minimise –Σ g, i.e. maximise Σ g.
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # HiGHS is the default high‑performance LP solver in SciPy ≥1.6.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block (exploration)."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
