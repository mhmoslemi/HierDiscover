"""
Propose a candidate step‑function sequence.

The algorithm starts from the analytically optimal “arcsine’’ distribution,
then repeatedly solves a linear programme that maximises the ℓ1‑mass of a
new vector `g` while keeping the mixed convolution `f * g` bounded by the
current maximal self‑convolution `M = max(f*f)`.  After solving the LP the
solution is rescaled to the original ℓ1‑mass, which strictly reduces the
upper bound `M`.  A small convex combination with the previous iterate
stabilises the progress.  The loop stops when the time budget is exhausted
or no improvement is obtained.

The construction runs entirely in memory, uses only NumPy, SciPy and the
provided `evaluate_sequence` function and finishes well inside the 1000 s
budget.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def arcsine_sequence(n: int) -> np.ndarray:
    """Return the discrete arcsine distribution of length ``n`` (sum = 1)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself (via real FFT)."""
    n = x.shape[0]
    size = 2 * n - 1
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)
    return conv


def build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Build the sparse matrix A such that   A @ g = f * g   (full convolution).
    ``A`` has shape (2*n-1, n) and entry A[k, j] = f[k-j] whenever 0≤k-j<n,
    otherwise 0.
    """
    n = f.shape[0]
    # each column j corresponds to a diagonal offset = j with the vector f.
    # `sp.diags` copies the data for each diagonal, giving a matrix with n²
    # non‑zero entries (still manageable for n≈2000).
    offsets = np.arange(n, dtype=int)          # 0 … n‑1
    data = [f] * n                             # list of n copies of the same array
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A

    Return a list of non‑negative heights that (hopefully) improves the
    evaluation score.  The routine respects the 1000 s overall time limit.

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Choose problem size (length of the step function)
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
        init_from_base = True
    except Exception:
        # fallback – a size that keeps the LP cheap while still giving a good bound
        n = 2000
        base = None
        init_from_base = False

    # ----------------------------------------------------------------------
    # Starting point: arcsine or the supplied sequence (normalised to sum=1)
    # ----------------------------------------------------------------------
    f = arcsine_sequence(n)

    if init_from_base:
        tmp = base.copy()
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s == 0.0:
            tmp = np.ones_like(tmp)
            s = tmp.sum()
        tmp /= s                     # normalise to sum = 1  (scale does not matter)
        # keep the better of the two
        score_arc = evaluate_sequence(f.tolist())
        score_tmp = evaluate_sequence(tmp.tolist())
        if score_tmp < score_arc:
            f = tmp.copy()
            best_score = score_tmp
        else:
            best_score = score_arc
    else:

    best = f.copy()
    S = f.sum()                     # ℓ1‑mass (always 1 after the normalisation)

    print(f"Initial length n = {n}, initial score = {best_score:.6f}")

    # ----------------------------------------------------------------------

        conv_ff = self_convolution(f)

        A = build_conv_matrix(f)

        # maximise sum(g)   ⇔   minimise -sum(g)

                break

    # ----------------------------------------------------------------------
    # Final polishing – a few cheap sub‑gradient steps (optional)
    # ----------------------------------------------------------------------
    # The sub‑gradient method is cheap and can shave off the last fractions.
    def proj_simplex(v: np.ndarray) -> np.ndarray:
        """Project onto the probability simplex (sum = 1, v ≥ 0)."""
        n = v.shape[0]
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

    step = 0.02
    decay = 0.9995
    for it in range(3000):
        if time.monotonic() >= deadline:
            break
        conv = self_convolution(best)
        M = conv.max()
        smax = int(conv.argmax())

        # sub‑gradient using *all* active indices (within a tiny tolerance)
        eps = 1e-12
        active = np.where(conv >= M - eps)[0]
        grad = np.zeros_like(best)
        idx = np.arange(n)
        for s in active:
            mate = s - idx
            mask = (mate >= 0) & (mate < n)
            grad[mask] += 2.0 * best[mate[mask]]

        cand = best - step * grad
        # occasional isotropic noise to shake out of flat regions
        if it % 200 == 0:
            cand += rng.normal(scale=0.001, size=n)

        cand = proj_simplex(cand)          # keep sum = 1 and non‑negative
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step = 0.02                    # revive step size after improvement
        else:
            step *= decay

        if it % 500 == 0:
            print(f"Sub‑grad iter {it:04d} | best_score = {best_score:.6f}")

    print(f"Finished search – best score = {best_score:.6f}")
    # Return plain Python floats (the wrapper will handle any required clipping)
    return [float(v) for v in best]
