def reduce_overlap(seq, rng):
    n = len(seq)
    b = np.convolve(seq, seq)
    L = int(np.argmax(b))
    # compute overlapping indices
    start = max(0, L - (n-1))
    end = min(n-1, L)
    # overlapped region indices start..end inclusive
    # non-overlapped region indices: complement
    overlapped = np.arange(start, end+1)
    nonoverlapped = np.setdiff1d(np.arange(n), overlapped)
    if nonoverlapped.size == 0:
        # fallback random perturb
        return block_perturb(seq, rng)
    # move a fraction of values from overlapped to nonoverlapped
    # choose fraction of total mass to move: e.g., 0.05 * sum overlapped
    total_overlap = seq[overlapped].sum()
    if total_overlap == 0:
        return block_perturb(seq, rng)
    move_amount = rng.random() * 0.03 * total_overlap # up to 3% of overlapped mass
    # choose a source index from overlapped weighted by value
    src = rng.choice(overlapped, p=seq[overlapped]/total_overlap)
    # reduce source
    reduction = min(move_amount, seq[src])
    seq[src] -= reduction
    # distribute reduction equally or weighted randomly to nonoverlapped
    dst = rng.choice(nonoverlapped, size=2, replace=False) # send to two random indices
    seq[dst] += reduction / len(dst)
    return seq

import numpy as np, time, random

def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    start_t = time.monotonic()
    deadline = start_t + max(0.0, float(budget_s) - 2.0)

    try:
        n = len(height_sequence_1)
    except NameError:
        n = 6000

    # Helper functions
    def sym_power_seq(n, exponent, offset):
        idx = np.arange(n, dtype=np.float64) + offset + 1.0
        seq = idx ** (-exponent)
        seq = (seq + seq[::-1]) / 2.0
        seq = np.clip(seq, 0.0, 1000.0)
        return seq

    def score(seq):
        # use evaluate_sequence directly
        return float(evaluate_sequence(seq.tolist()))

    # Keep best
    best_seq = None
    best_score = np.inf
    best_exp = None
    best_off = None

    # ---- Coarse grid
    exps = np.arange(0.45, 0.56, 0.01)
    offs = np.arange(0.0, 2.01, 0.2)

    for p in exps:
        for off in offs:
            seq = sym_power_seq(n, p, off)
            sc = score(seq)
            if sc < best_score:
                best_seq = seq.copy()
                best_score = sc
                best_exp, best_off = p, off
                print(f"grid coarse p={p:.3f} off={off:.2f} → {sc:.6f}")
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    # ---- Refined grid around current best
    if best_seq is not None:
        p_center = best_exp
        off_center = best_off
    else:
        p_center = 0.5
        off_center = 0.0

    # refined search
    for _ in range(5):
        # shrink step sizes
        p_range = 0.02
        off_range = 0.5
        exps = np.arange(p_center - p_range, p_center + p_range + 1e-9, 0.002)
        offs = np.arange(off_center - off_range, off_center + off_range + 1e-9, 0.05)

        for p in exps:
            if p <= 0: continue
            for off in offs:
                if off < 0: continue
                seq = sym_power_seq(n, p, off)
                sc = score(seq)
                if sc < best_score:
                    best_seq = seq.copy()
                    best_score = sc
                    best_exp, best_off = p, off
                    print(f"grid fine p={p:.3f} off={off:.2f} → {sc:.6f}")
                if time.monotonic() >= deadline:
                    break
            if time.monotonic() >= deadline:
                break
        # update centers for next refinement
        p_center = best_exp
        off_center = best_off
        if time.monotonic() >= deadline:
            break

    # ---- Parameter local random walk (simulated annealing)
    temperature_start = 0.1
    temperature_end = 1e-5

    while time.monotonic() < deadline:
        # Annealing temperature based on remaining time fraction
        elapsed = time.monotonic() - start_t
        frac = 1.0 - (elapsed / (deadline - start_t))
        T = temperature_end + (temperature_start - temperature_end) * frac

        # Propose new exponents
        p_new = best_exp + rng.normal(0.0, 0.005) * T/0.1   # scale noise
        off_new = best_off + rng.normal(0.0, 0.1) * T/0.1

        # Clip
        if p_new <= 0: p_new = 1e-6
        if off_new < 0: off_new = 0.0

        seq = sym_power_seq(n, p_new, off_new)
        sc = score(seq)

        # Accept if better or with probability exp(-(sc-best_score)/T)
        if sc < best_score or rng.random() < math.exp(-(sc - best_score) / max(T, 1e-12)):
            best_seq = seq.copy()
            best_score = sc
            best_exp, best_off = p_new, off_new
            print(f"anneal new best p={p_new:.4f} off={off_new:.4f} → {sc:.6f}")

        # Occasionally perform a block perturb on best_seq to escape local minima
        if rng.random() < 0.02:
            cand = block_perturb(best_seq, rng)  # reuse earlier defined function
            cand = (cand + cand[::-1]) / 2.0  # enforce symmetry
            sc2 = score(cand)
            if sc2 < best_score:
                best_seq = cand.copy()
                best_score = sc2
                print(f"block improve → {sc2:.6f}")

        # Additional occasional random restart
        if rng.random() < 0.005:
            p_rand = rng.uniform(0.45, 0.55)
            off_rand = rng.uniform(0.0, 2.0)
            seq = sym_power_seq(n, p_rand, off_rand)
            sc = score(seq)
            if sc < best_score:
                best_seq = seq.copy()
                best_score = sc
                best_exp, best_off = p_rand, off_rand
                print(f"restart → {sc:.6f}")

    # Ensure nonnegativity and clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Return as list of floats
    return [float(x) for x in best_seq.tolist()]

def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S < 0.01:
        return np.inf
    n = a.shape[0]
    # convolution via FFT
    size = 2*n - 1
    # pad to next power of two for efficiency
    nfft = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A)[:size]
    max_b = conv.max()
    return float(2 * n * max_b / (S * S))
```

"""
Propose the best non‑negative step‑function heights for the given evaluation.
The algorithm is deliberately different from the previous hill‑climber:

1.  A *symmetrised* power‑law kernel a_i = (i+offset+1)^(-p) is used as a
    parametric family that yields an almost constant self‑convolution.
2.  A dense grid search (exponent, offset) is performed with a fast FFT‑based
    evaluation to locate a very good starting point.
3.  The parameters are then refined by a simple simulated‑annealing walk.
4.  Finally a few block‑perturbations (multiplicative, local) are tried to
    escape possible plateaus.
5.  The best sequence found before the time budget expires is returned as a
    plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def sym_power_sequence(n: int, exponent: float, offset: float) -> np.ndarray:
    """
    Produce a symmetric power‑law sequence:
        a_i = (i+offset+1)^(-exponent)
    and symmetrise it by averaging with its reverse.  Values are clipped to the
    legal range [0, 1000].
    """
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    seq = (seq + seq[::-1]) / 2.0                # enforce mirror symmetry
    return np.clip(seq, 0.0, 1000.0)


def fast_score(seq: np.ndarray) -> float:
    """
    Compute the same score as ``evaluate_sequence`` but using an FFT based
    convolution (much faster for the many calls made during the search).
    """
    a = np.clip(seq, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.shape[0]
    size = 2 * n - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A)[:size]            # linear convolution
    m = conv.max()
    return float(2.0 * n * m / (s * s))


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative perturbation of a random contiguous block of the sequence.
    """
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.06)))   # ~6 % log‑normal change
    new = seq.copy()
    new[start:start + width] *= factor
    # occasional symmetrisation to keep the shape balanced
    if rng.random() < 0.1:
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The routine respects the time budget (≈1000 s) and returns the best
    candidate found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a safety margin so the caller does not kill us early
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Length of the problem (use a known construction if available)
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))          # type: ignore[name-defined]
    except Exception:
        n = 6000                                 # a comfortable default

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    best_exp = 0.5
    best_off = 0.0

    # ------------------------------------------------------------------
    # 1) Dense grid search over the (exponent, offset) space
    # ------------------------------------------------------------------
    exp_grid = np.arange(0.44, 0.57, 0.002)      # 0.44 … 0.56 in steps of 0.002
    off_grid = np.arange(0.0, 2.01, 0.02)       # 0 … 2.0 in steps of 0.02

    for exp in exp_grid:
        for off in off_grid:
            seq = sym_power_sequence(n, float(exp), float(off))
            sc = fast_score(seq)
            if sc < best_score:
                best_seq = seq.copy()
                best_score = sc
                best_exp, best_off = float(exp), float(off)
                print(f"grid   exp={exp:.3f} off={off:.2f} → score={sc:.6f}")
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Local parameter refinement (simulated annealing)
    # ------------------------------------------------------------------
    # temperature schedule: start relatively warm, cool to almost zero
    T_start = 0.1
    T_end = 5e-6

    while time.monotonic() < deadline:
        # linear cooling based on remaining time
        elapsed = time.monotonic() - start_time
        frac = 1.0 - (elapsed / (deadline - start_time))
        T = T_end + (T_start - T_end) * max(0.0, min(1.0, frac))

        # propose a new (exp,off) pair
        p_new = best_exp + rng.normal(0.0, 0.003) * (T / T_start)
        o_new = best_off + rng.normal(0.0, 0.05) * (T / T_start)

        # keep parameters inside the feasible region
        if p_new <= 0.0:
            p_new = 1e-6
        if o_new < 0.0:
            o_new = 0.0

        cand = sym_power_sequence(n, float(p_new), float(o_new))
        sc = fast_score(cand)

        # accept if better, otherwise with a Metropolis probability
        if sc < best_score or rng.random() < math.exp(-(sc - best_score) / max(T, 1e-12)):
            best_seq = cand.copy()
            best_score = sc
            best_exp, best_off = float(p_new), float(o_new)
            print(f"anneal exp={p_new:.5f} off={o_new:.4f} → score={sc:.6f}")

        # ----------------------------------------------------------------
        # 3) Occasional block perturbations of the current best sequence
        # ----------------------------------------------------------------
        if rng.random() < 0.03:
            pert = block_perturb(best_seq, rng)
            pert = (pert + pert[::-1]) / 2.0               # keep symmetry
            sc2 = fast_score(pert)
            if sc2 < best_score:
                best_seq = pert.copy()
                best_score = sc2
                print(f"block  improved → score={sc2:.6f}")

        # ----------------------------------------------------------------
        # 4) Rare random restarts to explore distant regions
        # ----------------------------------------------------------------
        if rng.random() < 0.002:
            p_r = rng.uniform(0.44, 0.56)
            o_r = rng.uniform(0.0, 2.0)
            seq_r = sym_power_sequence(n, p_r, o_r)
            sc_r = fast_score(seq_r)
            if sc_r < best_score:
                best_seq = seq_r.copy()
                best_score = sc_r
                best_exp, best_off = p_r, o_r
                print(f"restart exp={p_r:.3f} off={o_r:.2f} → score={sc_r:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best candidate (plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:                     # very unlikely, but be safe
        best_seq = sym_power_sequence(n, 0.5, 0.0)

    # final clipping to obey the evaluator's safety checks
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return [float(x) for x in best_seq.tolist()]
