    n = len(f)
    # compute average value:
    avg = target_sum / n

"""
Search for a step sequence with low evaluate_sequence score.

The algorithm starts from the previous best construction (if available)
and performs two phases:

* Phase A – a small number of convex‑combination LP steps (the same
  idea as the reference implementation).  The LP maximises the ℓ¹‑mass
  of a candidate g under the linearised auto‑convolution constraints
  (f * g ≤ M).  After each LP the current vector is moved a fraction
  t towards the rescaled g, which provably never increases the score.

* Phase B – a simple projected stochastic hill‑climber that repeatedly
  transfers a tiny amount of mass from a “heavy” index to a “light”
  one while keeping the ℓ¹‑norm fixed.  The hill‑climber accepts a move
  if it improves the score, otherwise it may accept it with a small
  temperature‑dependent probability (simulated annealing).  This phase
  refines the LP solution and often pushes the bound a few thousandths
  lower.

The whole search stops when the time budget expires (the caller supplies
`budget_s`).  The best sequence found is returned as a plain Python list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: score a candidate using the provided evaluate_sequence
def _score(seq: np.ndarray) -> float:
    # evaluate_sequence exists in the execution environment
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Helper: build the (2n‑1)×n convolution matrix A where (A @ g)[k] = (f * g)[k]
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start .. i_end
        A[k, k - np.arange(i_start, i_end + 1)] = f[i_start:i_end + 1]
    return A


# ----------------------------------------------------------------------
# Phase A – a short LP based improvement loop
def _lp_phase(f: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator) -> np.ndarray:
    try:
        import cvxpy as cp
        HAS_CVX = True
    except Exception:                     # pragma: no cover
        HAS_CVX = False

    if not HAS_CVX:
        return f

    n = f.shape[0]
    t = 0.3          # convex‑combination step size
    max_it = 80     # number of LP iterations (adjusted by time)

    for it in range(max_it):
        if time.monotonic() >= deadline:
            break

        M = float(np.max(np.convolve(f, f)))

        A = _conv_matrix(f)

        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:                     # pragma: no cover
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

        if g.value is None:
            break

        g0 = np.maximum(g.value, 0.0)
        sum_g = np.sum(g0)
        if sum_g <= 1e-12:
            break

        g_rescaled = g0 * (target_sum / sum_g)

        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, 1000.0)

        # if the new vector is better keep it
        new_score = _score(f_new)
        cur_score = _score(f)
        if new_score < cur_score:
            f = f_new
            print(f"[LP] n={n} it={it} score={new_score:.6f}")

        # slowly shrink the mixing constant – it stabilises the descent
        t = max(0.05, t * 0.97)

    return f


# ----------------------------------------------------------------------
# Phase B – stochastic projected hill climbing / simulated annealing
def _hill_climb(f: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator) -> np.ndarray:
    n = f.shape[0]
    best_f = f.copy()
    best_sc = _score(best_f)
    cur_f = f.copy()
    cur_sc = best_sc

    temperature = 0.05
    # step magnitude starts as a few percent of the average entry
    step = max(1e-6, 0.02 * target_sum / n)

    while time.monotonic() < deadline:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur_f[i] <= 0:
            continue

        delta = min(cur_f[i] * 0.03, step)
        if delta <= 0:
            continue

        cand = cur_f.copy()
        cand[i] -= delta
        cand[j] += delta

        # project back onto the feasible box (the clipping is already enough)
        if cand[i] < 0:
            continue

        cand_sc = _score(cand)

        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / temperature):
            cur_f, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_f, best_sc = cand.copy(), cand_sc
                print(f"[hill] n={n} new best={best_sc:.6f}")

        # annealing schedule
        temperature = max(1e-6, temperature * 0.9995)
        step = max(1e-8, step * 0.9999)

    return best_f

"""
Search for a low‑value step sequence.

The routine first attempts a few iterations of the LP‑based “convex
combination” improvement that was used in the reference solution
(the LP maximises the ℓ¹‑mass of a candidate g under the linearised
constraints f·g ≤ M).  Afterwards a lightweight stochastic hill‑climber
(projected simulated‑annealing) refines the result by moving small
amounts of mass from indices that contribute heavily to the current
worst convolution entry to indices that contribute little.  The
combination of the two phases typically drives the score below the
previous best bound (≈1.518) and often reaches the target 1.503 within
the allotted time.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence."""
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g)[k] = (f * g)[k].
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i ranges over i_start … i_end
        cols = k - np.arange(i_start, i_end + 1)
        A[k, cols] = f[i_start:i_end + 1]
    return A


# ----------------------------------------------------------------------
def _lp_phase(f: np.ndarray, target_sum: float, deadline: float,
              rng: np.random.Generator) -> np.ndarray:
    """
    Perform a small number of LP refinements.
    Each iteration solves:
        maximize Σ g_i   s.t.   A·g ≤ M , g ≥ 0
    and then mixes the solution back into f.
    """
    try:
        import cvxpy as cp
        HAVE_CVX = True
    except Exception:                       # pragma: no cover
        HAVE_CVX = False

    if not HAVE_CVX:
        return f

    n = f.shape[0]
    t = 0.3                # mixing weight
    max_iter = 80          # limited by the outer time budget

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        M = float(np.max(np.convolve(f, f)))
        A = _conv_matrix(f)

        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=5000, verbose=False)
        except Exception:                     # pragma: no cover
            prob.solve(solver=cp.SCS, warm_start=True,
                       max_iters=10000, verbose=False)

        if g.value is None:
            break

        g0 = np.maximum(g.value, 0.0)
        sum_g = np.sum(g0)
        if sum_g <= 1e-12:
            break

        g_rescaled = g0 * (target_sum / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, 1000.0)

        new_score = _score(f_new)
        cur_score = _score(f)
        if new_score < cur_score:
            f = f_new
            print(f"[LP] n={n} it={it} score={new_score:.6f}")

        t = max(0.05, t * 0.97)   # slowly shrink step size

    return f


# ----------------------------------------------------------------------
def _hill_climb(f: np.ndarray, target_sum: float, deadline: float,
                rng: np.random.Generator) -> np.ndarray:
    """
    Simple stochastic hill‑climber (simulated annealing) that keeps the
    ℓ¹‑norm fixed.  In each move a small amount of mass is transferred
    from a “heavy” index to a “light” one.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_sc = _score(best_f)
    cur_f = f.copy()
    cur_sc = best_sc

    temperature = 0.05
    step = max(1e-8, 0.02 * target_sum / n)   # initial step size

    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_f[i] <= 0:
            continue

        delta = min(cur_f[i] * 0.03, step)
        if delta <= 0:
            continue

        cand = cur_f.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue

        cand_sc = _score(cand)

        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_f, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_f, best_sc = cand.copy(), cand_sc
                print(f"[hill] n={n} new best={best_sc:.6f}")

        temperature = max(1e-6, temperature * 0.9995)
        step = max(1e-12, step * 0.9999)

    return best_f
