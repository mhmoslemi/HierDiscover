"""
Optimiser for the “max‑convolution” score.

The algorithm works in three stages:

1.  Build a small library of analytically‑inspired families
    (arcsine, uniform, symmetric Beta‑α distributions, random Dirichlet).
    All candidates are normalised to the probability simplex
    (sum = 1) and the best of them is kept as a starting point.

2.  Refine the starter with a smoothed‑max gradient descent.
    The true objective is   max(conv(a,a)),
    which is non‑smooth.  We replace it by a log‑sum‑exp
    approximation with a temperature p (≈30).  The resulting gradient
    is a convex combination of the sub‑gradients 2·a_{i‑j},
    efficiently obtained by one FFT‑based convolution per iteration.
    An Adam‑style update together with projection onto the simplex
    yields fast, stable progress.

3.  Periodic diversification:
    - isotropic Gaussian noise every few steps,
    - occasional “block” perturbations (multiplying a random contiguous
      block by a log‑normal factor) to escape shallow minima,
    - a full restart from a random Beta‑α distribution after prolonged stagnation.

The whole routine respects the 1000 s budget, prints progress,
and finally returns the best sequence found as a plain Python list
of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (sum = 1, v ≥ 0).

    Implements the O(n) algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_sequence(n: int) -> np.ndarray:
    """Arcsine (central‑binomial) sequence of length n,
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def beta_like_sequence(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α)–type shape on {0,…,n‑1}.
    a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}, then normalised to sum = 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    # avoid 0⁰ by using idx≥1; (n‑idx+1) is also ≥1
    a = np.power(idx, alpha - 1.0) * np.power(n - idx + 1, alpha - 1.0)
    total = a.sum()
    if total <= 0 or not np.isfinite(total):
        a = np.ones(n, dtype=np.float64)
        total = n
    a /= total
    return a


def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (FFT implementation)."""
    n = x.shape[0]
    size = 2 * n - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(X * X, n=nfft)
    return conv[:size]


def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (returns length len(x)+len(y)-1)."""
    size = x.shape[0] + y.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(X * Y, n=nfft)
    return conv[:size]


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap non‑negative block perturbation used for exploration."""
    cand = base.copy()
    n = cand.size
    # width roughly between 0.5% and 8% of the length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main optimiser – the function that will be called by the judge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
        2·n·max(conv(a,a)) / (sum(a)²)

    Returns the best sequence found (list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)   # spare a little margin

    # ------------------------------------------------------------------
    # Determine problem size (length of the step function)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)

            raise ValueError
        n = int(base.size)
    except Exception:
        base = None
        n = 6000                     # reasonable default

    # ------------------------------------------------------------------
    # 1) Generate a cheap library of analytic families and pick the best
    # ------------------------------------------------------------------
    candidates = []

    # arcsine (known strong baseline)
    candidates.append(arcsine_sequence(n))

    # uniform
    candidates.append(np.ones(n, dtype=np.float64) / n)

    # symmetric Beta‑α families (α from 0.05 to 0.95)
    alphas = np.concatenate([
        np.linspace(0.05, 0.45, 9),   # <0.5
        np.linspace(0.55, 0.95, 9)    # >0.5 (mostly for sanity)
    ])
    for a in alphas:
        candidates.append(beta_like_sequence(n, float(a)))

    # few random Dirichlet draws (concentration between 0.2 and 1.0)
    for _ in range(5):
        conc = rng.uniform(0.2, 1.0)
        g = rng.gamma(shape=conc, scale=1.0, size=n)
        if g.sum() == 0:

        candidates.append(g / g.sum())

    # If a custom starting point is provided, normalise it and add it
    if base is not None:
        tmp = base.copy()
        tmp = np.maximum(tmp, 0.0)
        if tmp.sum() == 0:
            tmp = np.ones_like(tmp)
        candidates.append(tmp / tmp.sum())

    best_seq = None
    best_score = np.inf

    for seq in candidates:
        conv = self_convolution(seq)
        score = float(2.0 * n * np.max(conv))   # sum(seq)=1 ⇒ exact score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    print(f"[init] best score after family scan = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed‑max Adam optimisation starting from the best family
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)          # first moment (Adam)
    v = np.zeros_like(a)          # second moment (Adam)

    # Adam hyper‑parameters
    lr = 0.02                     # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Smoothing parameter for the log‑sum‑exp approximation of max
    p = 30.0

    it = 0
    stagnation = 0               # counts iterations without improvement

    while time.monotonic() < deadline:
        # ----- current convolution and smoothed weights -----
        conv = self_convolution(a)                # length 2n‑1
        max_c = conv.max()
        # log‑sum‑exp weights (numerically stable)
        exp_shift = np.exp(p * (conv - max_c))
        w = exp_shift / exp_shift.sum()           # shape (2n‑1,)

        # ----- gradient of the smoothed objective -----
        # gradient = 2 * (a  *  w_rev)  on the appropriate slice
        tmp = full_convolution_fft(a, w[::-1])    # length 3n‑2
        slice_ = tmp[n - 1: 2 * n - 1]            # exactly n entries
        grad = 2.0 * slice_[::-1]                 # shape (n,)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = proj_simplex(a)

        # ----- evaluate exact score -----

        score = float(2.0 * n * np.max(conv_exact))

            stagnation = 0
            print(f"[iter {it:06d}] new best = {best_score:.6f}")
        else:
            stagnation += 1
            # full restart after a long dry spell

                restart_alpha = rng.uniform(0.1, 0.9)
                a = beta_like_sequence(n, restart_alpha)
                a = proj_simplex(a)
                m.fill(0.0)
                v.fill(0.0)
                stagnation = 0
