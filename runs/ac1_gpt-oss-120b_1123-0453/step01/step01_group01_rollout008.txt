"""
Propose a step‑function (non‑negative sequence) with a low self‑convolution
score.  The algorithm combines three ideas:

1.  A good deterministic start – the *arcsine* sequence (the classic
    √‑law) – and any user supplied sequence.
2.  A simulated‑annealing style stochastic search that repeatedly perturbs a
    random contiguous block, re‑normalises onto the probability simplex and
    accepts the move if it improves the score or with a temperature‑dependent
    probability.
3.  A lightweight Adam‑style gradient descent that uses the *K* largest
    convolution entries as a smooth approximation of the max‑operator.
    The gradient of the smoothed objective is a convex combination of the
    sub‑gradients of those entries.

Both mechanisms run until the allotted time (≈1000 s) expires.  The best
sequence found is returned as a plain ``list[float]`` – suitable for being
passed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest rho where u_i > (cssv_i / i)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_sequence(n: int) -> np.ndarray:
    """
    Return the discrete “arcsine’’ (central‑binomial) distribution of length *n*,
    normalised to unit sum.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # a_i = a_{i‑1} * (2i‑1) / (2i)
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (FFT‑based)."""
    n = x.shape[0]
    size = 2 * n - 1
    # FFT of size ``size`` (the next power‑of‑two is taken automatically by np.fft)
    f = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_kwargs):
    """
    Search for a non‑negative sequence that minimises
        2 * n * max(conv(a, a)) / (sum(a)**2)
    (the denominator is irrelevant after normalising to sum = 1).

    The routine runs for approximately ``budget_s`` seconds and returns the
    best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine the problem size (length of the step function)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        # A reasonable default that balances quality and speed
        n = 4000
        base = None

    # ------------------------------------------------------------------
    # Helper to evaluate the *raw* objective (avoids the extra checks in
    # ``evaluate_sequence``).  The sum of the vector is always normalised to 1,
    # therefore the score reduces to 2 * n * max(conv).
    # ------------------------------------------------------------------
    def raw_score(a: np.ndarray) -> float:
        conv = self_convolution(a)
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # Initial candidates: arcsine, user supplied ``height_sequence_1`` and a few
    # random Dirichlet draws.
    # ------------------------------------------------------------------
    best = arcsine_sequence(n)
    best_score = raw_score(best)

    if base is not None:
        tmp = np.maximum(base, 0.0)
        if tmp.sum() == 0.0:
            tmp = np.ones_like(tmp)
        tmp /= tmp.sum()
        sc = raw_score(tmp)
        if sc < best_score:
            best = tmp.copy()
            best_score = sc

    for _ in range(5):

        cand /= cand.sum()
        sc = raw_score(cand)
        if sc < best_score:

            best_score = sc

    # ------------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------------
    cur = best.copy()
    # Adam moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999

    # learning‑rate schedule (initial value, then ~1/√t decay)
    lr_init = 0.12
    # smoothing parameter: number of top convolution entries used as a proxy

    # Simulated‑annealing temperature (linear cooling from 0.02 to 0.0002)

    it = 0

        # --------------------------------------------------------------

            no_improve = 0

            # Metropolis criterion
            if rng.random() < np.exp(-(cand_score - best_score) / max(T, 1e-12)):

        # --------------------------------------------------------------

            start = int(rng.integers(0, max(1, n - width + 1)))

            pert[block] *= factor

            sc = raw_score(pert)
            if sc < best_score:

                best_score = sc
                cur = pert.copy()
                no_improve = 0

        # 6)  Restart strategy – if we have been stagnating for a long while
        # --------------------------------------------------------------
        if no_improve > 3500:
            # re‑initialise from a fresh Dirichlet sample (α=0.7)
            cur = rng.gamma(0.7, size=n)
            cur /= cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # --------------------------------------------------------------
        # 7)  Progress report
        # --------------------------------------------------------------
        if it % 2000 == 0:

    # ----------------------------------------------------------------------
    for _ in range(1200):

        max_val = conv.max()
        s = int(conv.argmax())
        # exact sub‑gradient for the maximal entry
        lo = max(0, s - (n - 1))
        hi = min(s, n - 1)

        grad = np.zeros_like(best)
        grad[idx] = 2.0 * best[s - idx]
        # tiny fixed step with projection

    print(f"Finished after {it} iterations, best score = {final_score:.8f}")

    # Return as plain Python floats (the evaluator will re‑scale if it wishes)
    return [float(v) for v in best]
