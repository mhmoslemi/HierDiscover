"""
Hybrid linear‑program / projected‑gradient optimizer.

The routine starts from the best known step‑function (the global
`height_sequence_1` if it exists, otherwise the classic arcsine shape).
It repeatedly

1. computes the self‑convolution and the current bound M,
2. solves a linear program that maximises the L1‑norm of a partner
   sequence `g` under the constraint `f * g ≤ M`,
3. rescales `g` to the same L1‑norm as `f` and mixes them
   (`f ← (1‑t)·f + t·g`) – this provably never increases the bound,
4. occasionally performs cheap random block perturbations to escape
   local minima,
5. finally projects onto the simplex (non‑negative, unit sum).

The loop runs until the time budget expires and returns the best
sequence seen.  All operations respect the security checks of
`evaluate_sequence`.  No closures or lambdas are used.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.linalg


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # Width varies between n/300 and n/20 (inclusive)
    min_w = max(1, n // 300)
    max_w = max(2, n // 20)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative log‑normal jitter
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.20)))
    # occasional mirroring – keeps the shape roughly symmetric
    if rng.random() < 0.30:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce the bounds required by evaluate_sequence
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 990.0, **kwargs):
    """
    Search for a low‑score step‑function.

    Returns a plain Python list of non‑negative floats.
    """
    # ------------------------------------------------------------------
    # 1) Choose problem size and a starting point.
    # ------------------------------------------------------------------
    try:
        # use the previously found sequence if it exists
        init = np.array(height_sequence_1, dtype=float)  # pylint: disable=undefined-variable
        # cap size to keep the LP tractable
        max_n = 2000
        if init.size > max_n:
            # simple down‑sampling by averaging blocks
            factor = init.size // max_n
            trimmed = init[: factor * max_n]
            init = trimmed.reshape(-1, factor).mean(axis=1)
        n = init.size
    except Exception:
        # fallback: arcsine shape a_i ∝ 1/√(i+1)
        n = 2000
        idx = np.arange(1, n + 1, dtype=float)
        init = 1.0 / np.sqrt(idx)

    # Normalise to unit L1‑norm (the scale does not affect the score)
    init = np.maximum(init, 0.0)
    if init.sum() == 0.0:
        init = np.ones_like(init) / init.size
    init = init / init.sum()

    # ------------------------------------------------------------------
    # 2) Helper objects and time management.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    best_seq = init.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, score={best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # mixing parameter for LP steps – will be slowly reduced
    mix_t = 0.5
    # step size for the simple sub‑gradient updates
    sub_step = 0.05

    # ------------------------------------------------------------------
    # 3) Main optimisation loop.
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ----- 3.1) current self‑convolution, bound M ------------------
        size_fft = int(2 ** np.ceil(np.log2(2 * n - 1)))
        fa = np.fft.rfft(best_seq, size_fft)
        conv_full = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
        M = conv_full.max()

        # ----- 3.2) Linear‑program step --------------------------------
        # Build the Toeplitz matrix representing the convolution constraints
        col = np.concatenate([best_seq, np.zeros(n - 1, dtype=float)])
        row = np.concatenate(([best_seq[0]], np.zeros(n - 1, dtype=float)))
        A = scipy.linalg.toeplitz(col, row)               # shape (2n‑1, n)

        # Solve   max Σ g_j   s.t.   A·g ≤ M·1,   g ≥ 0
        # → minimise  -1ᵀ g
        lp_res = scipy.optimize.linprog(
            c=-np.ones(n, dtype=float),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M, dtype=float),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.monotonic())},
        )

        if lp_res.success:
            g = lp_res.x
            sum_g = g.sum()
            if sum_g > 0:
                # Rescale g to have the same L1‑norm as best_seq (which is 1)
                g = g / sum_g
                # Mix the current sequence with the partner
                candidate = (1.0 - mix_t) * best_seq + mix_t * g
                candidate = np.maximum(candidate, 0.0)
                candidate /= candidate.sum()   # project onto the simplex
                cand_score = float(evaluate_sequence(candidate.tolist()))
                if cand_score < best_score:
                    best_seq = candidate
                    best_score = cand_score
                    print(f"[LP improve] iter={iteration} score={best_score:.6f}")
        else:
            print(f"[LP] iteration {iteration} – linprog failed, status={lp_res.message}")

        # gradually shrink the mixing factor – later steps become finer
        mix_t = mix_t * 0.97
        mix_t = max(mix_t, 0.01)

        # ----- 3.3) Simple sub‑gradient descent on the max convolution ---
        # Evaluate again (cheap) and locate the peak index
        conv_full = np.convolve(best_seq, best_seq)  # small n → cheap dense conv
        k_star = int(conv_full.argmax())
        # Gradient of max_k (a*a)[k] is 2*a shifted by k_star
        idxs = np.arange(n)
        partner = k_star - idxs
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(best_seq)
        grad[mask] = 2.0 * best_seq[partner[mask]]

        # projected gradient step
        cand = best_seq - sub_step * grad
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:        # avoid degeneration
            cand = best_seq.copy()
        else:
            cand /= cand.sum()
        cand_score = float(evaluate_sequence(cand.tolist()))
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            print(f"[subgrad improve] iter={iteration} score={best_score:.6f}")

        # slowly decay the sub‑gradient step size
        sub_step *= 0.995
        sub_step = max(sub_step, 1e-4)

        # ----- 3.4) Occasional random block perturbation ----------------
        if rng.random() < 0.15:
            pert = _make_block_perturbation(best_seq, rng)
            pert_score = float(evaluate_sequence(pert.tolist()))
            if pert_score < best_score:
                best_seq = pert
                best_score = pert_score
                print(f"[perturb improve] iter={iteration} score={best_score:.6f}")

        # safety break if we already reached an impressive bound
        if best_score < 1.45:
            print("[early stop] reached very low score")
            break

    # ------------------------------------------------------------------
    # 4) Return the best candidate found.
    # ------------------------------------------------------------------
    print(f"[final] best score {best_score:.6f} after {time.monotonic() - start_time:.2f}s")
    return [float(x) for x in best_seq]
