"""
Propose a low‑value step‑function for the given evaluation routine.

Algorithm outline
-----------------
1.  Choose a modest length (default 1500) and start from a power‑law
    (arc‑sine) shape a_i = (i+1)^‑½, scaled so that Σa = √(2·n).  This shape
    already gives an objective ≈ 1.57.

2.  **Linear‑programming refinement** – if SciPy is available:
        * Let M = max(a*a) (the current worst convolution value).
        * Build the full convolution matrix A where
          A[k,i] = a_{k‑i}  (zero when the index is out of range).
          Then (a*g)[k] = A[k]·g.
        * Solve the LP
              maximise   Σ g_i
              subject to A·g ≤ M   (one inequality per convolution index)
                         0 ≤ g_i ≤ 1000.
          The solver (HiGHS) returns a feasible g with a larger ℓ₁‑norm
          than the target Σa = √(2·n).
        * Scale g back to the target sum:
              scale = √(2·n) / Σg ,   new_a = scale·g .
          Because the convolution is linear in the second argument,
          the new maximal value becomes `scale·M < M`.
        * Accept the new sequence if the evaluation improves and repeat a
          few (≤ 5) LP iterations.

3.  **Projected gradient / donor‑receiver moves** – after the LP stage
    perform cheap local updates:
        * locate the index k where the convolution a*a is maximal,
        * compute the contribution a_i·a_{k‑i} of each position,
        * move a small fraction of mass from the donor (largest contribution)
          to the receiver (smallest contribution) while preserving the total
          sum, and keep the move only if the evaluation improves.
        * Occasionally (≈ 1 % of iterations) apply a random multiplicative
          block perturbation to escape plateaus.
        * Restart from a fresh power‑law if no improvement for ≈ 20 s.

4.  Keep track of the best sequence seen and stop as soon as the target
    bound 1.503 is reached or the time budget is exhausted (≈ 1000 s).

All numbers are kept non‑negative and clipped to the range [0, 1000] so that
the security checks inside ``evaluate_sequence`` never trigger.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def make_power_law(n: int, alpha: float) -> np.ndarray:
    """Return a power‑law vector a_i = (i+1)^‑α (i=0..n‑1)."""
    idx = np.arange(n, dtype=np.float64)
    a = np.power(idx + 1.0, -alpha)
    return np.clip(a, 0.0, 1000.0)


def normalize_to_sum(seq: np.ndarray, target: float) -> np.ndarray:
    """Scale seq so that its ℓ₁‑norm equals target (preserving sign)."""
    s = seq.sum()
    if s == 0:
        return seq
    return seq * (target / s)


def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A of shape (2·n‑1, n) such that
    (A @ g)[k] = (f * g)[k] for all k.
    Row k corresponds to convolution index k (0‑based).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A.fill(0.0)
    # The i‑th column contains the vector f shifted down by i positions.
    for i in range(n):
        A[i : i + n, i] = f  # f[0] aligns with row i, f[n‑1] with row i+n‑1
    return A


def contributions_to_max(seq: np.ndarray, k: int) -> np.ndarray:
    """
    For a given convolution index k, return the array
    contrib[i] = seq[i] * seq[k‑i] (zero when the partner index is out of range).
    """
    n = seq.size
    i = np.arange(n, dtype=np.int64)
    j = k - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i[mask]] * seq[j[mask]]
    return contrib


def random_block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a multiplicative log‑normal perturbation to a random contiguous block
    and (with 35 % probability) mirror the whole sequence.
    """
    n = seq.size
    cand = seq.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start : start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

# Main search routine
# ----------------------------------------------------------------------

                last_improve = time.monotonic()
            else:
                # keep the current sequence but give the optimizer a chance to move
                last_improve = time.monotonic()

        # ------- progress output ----------------------------------------
        if iter_counter % 500 == 0:
            print(f"[{time.monotonic():.1f}] iter={iter_counter}  best_score={best_score:.6f}")

        # ------- early exit if target is reached ------------------------
        if best_score < 1.503:
            print("Target bound reached – stopping early.")
            break

    # ------------------------------------------------------------------
    # Final sanity clipping (evaluate_sequence already clips, but we keep it tidy)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(
        f"Search finished – best score = {best_score:.6f} after {iter_counter} iterations."
    )
    return [float(x) for x in best_seq.tolist()]
