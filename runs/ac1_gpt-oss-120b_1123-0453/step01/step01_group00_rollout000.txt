    deadline = time.monotonic() + budget_s - small_epsilon

    best_seq = None; best_score = inf

    # baseline: quick random search (small n) to get initial best
    # or reuse height_sequence_1 if present
    try: initial = height_sequence_1
    except NameError: initial = None
    if initial:
        initial_seq = np.array(initial, dtype=float)
        # maybe truncate or pad to desired n but compute score
        score = _score(initial_seq)
        best_seq = initial_seq; best_score = score

    # Outer loop over n sizes
    for n in [200, 300, 400, 500, 600]:
        if time.monotonic() >= deadline:
            break
        # target sum sqrt(2n)
        target_sum = math.sqrt(2.0 * n)
        # initial f: if best_seq length == n then use it, else random.
        if best_seq is not None and len(best_seq) == n:
            f = best_seq.copy()
            # adjust sum to target_sum
            f = f * (target_sum / max(np.sum(f), 1e-12))
        else:
            # possibly embed previous best sequence
            if best_seq is not None and len(best_seq) < n:
                # embed by zero insertion to new length
                f = np.zeros(n, dtype=float)
                # simple embed: copy best_seq into first part
                f[:len(best_seq)] = best_seq[:len(best_seq)]
                f = f * (target_sum / max(np.sum(f), 1e-12))
            else:
                f = rng.random(n) + 0.1
                f = f / np.sum(f) * target_sum
        t = 0.2
        max_iters = 60
        for it in range(max_iters):
            if time.monotonic() >= deadline:
                break
            M = float(np.max(np.convolve(f, f)))
            A = _conv_matrix(f)
            # define variable
            g = cp.Variable(n)
            constraints = [g >= 0, A @ g <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
                if g.value is None: break
            except Exception:
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=20000)
                if g.value is None: break
            g0 = np.maximum(g.value, 0.0)
            sum_g = np.sum(g0)
            if sum_g < 1e-12: break
            # rescale to target sum
            g_rescaled = g0 * (target_sum / sum_g)
            # combine
            f = np.clip((1-t) * f + t * g_rescaled, 0.0, 1000.0)
            # evaluate
            sc = _score(f)
            if sc < best_score:
                best_score = sc
                best_seq = f.copy()
                print(f"[LP] n={n} it={it} score={sc:.6f}")
            # early stop if improvement small
            if best_score < 1.503: break
        # after inner loop maybe run a few random perturbation steps for refinement
        # using simple hill climbing: small Gaussian changes and acceptance if score improves
        for _ in range(200):
            if time.monotonic() >= deadline: break
            cand = best_seq + rng.normal(scale=0.01, size=len(best_seq))
            cand = np.clip(cand, 0, 1000)
            sc = _score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[perturb] n={len(best_seq)} score={sc:.6f}")
                if best_score < 1.503: break
    # final fallback if still no improvement: simple random search with best n
    if best_seq is None:

    return [float(x) for x in best_seq]

    If CVXPY is unavailable the routine falls back to a simple random walk.

The implementation respects the 1000‑second budget, prints progress and
never uses closures or lambda‑expressions as required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: evaluate a sequence using the provided scorer.
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the objective value for ``seq`` (list or np.ndarray)."""
    # evaluate_sequence is assumed to be globally available.
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Helper: build the convolution matrix A such that (A @ g)[k] = (f * g)[k].
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return a (2n‑1) × n matrix representing convolution with fixed f."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill using vectorised loops – still O(n²) but fast enough for n≤600.
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over indices of f that contribute to output k
        # column index for g is k‑i
        cols = k - np.arange(i_start, i_end + 1)
        A[k, cols] = f[i_start:i_end + 1]
    return A


# ----------------------------------------------------------------------
# Simple random‑search fallback (used when CVXPY cannot be imported).
# ----------------------------------------------------------------------
def _fallback_random_search(budget_s: float, seed: int = 0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    # start with a uniform vector of length 256 (a safe default)
    n = 256
    seq = np.full(n, 1.0, dtype=np.float64)
    best_seq = seq.copy()
    best_score = _score(seq)

    while time.monotonic() < deadline:
        cand = seq + rng.normal(scale=0.02, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            seq = cand
    print("[fallback] best score =", best_score)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging system.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.

    The routine performs a series of LP‑driven refinement steps on several
    problem sizes, followed by short random perturbations.  It respects the
    given time budget (``budget_s`` seconds) and prints intermediate scores.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # give a small safety margin for the outer teardown
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # keep the overall best result
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try to import CVXPY – the LP‑based core heavily depends on it.
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp
        HAS_CVX = True
    except Exception:               # pragma: no cover
        HAS_CVX = False

    # ------------------------------------------------------------------
    # If a previously discovered sequence is present, use it to initialise.
    # It is stored in the global variable ``height_sequence_1`` if supplied
    # by the competition harness.
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        if init_seq.ndim != 1:
            raise ValueError
        init_score = _score(init_seq)
        best_seq = init_seq.copy()
        best_score = init_score
        print("[init] using provided height_sequence_1, score =", init_score)
    except Exception:
        # no usable global, will start from scratch later
        pass

    # ------------------------------------------------------------------
    # Parameter sweep over several lengths.  Bigger n gives more freedom
    # but also larger LPs – we bound n to ≈ 600 to stay well within the
    # time limit.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)          # the normalisation used by the scorer
        # initialise f:
        if best_seq is not None and best_seq.shape[0] == n:
            f = best_seq.copy()
            # re‑scale to the new target ℓ¹‑norm
            current = np.sum(f)
            if current > 1e-12:
                f = f * (target_sum / current)
            else:
                f = rng.random(n) + 0.1
                f = f / np.sum(f) * target_sum
        else:
            # embed a shorter best sequence (if any) by zero‑padding
            if best_seq is not None and best_seq.shape[0] < n:
                f = np.zeros(n, dtype=np.float64)
                f[:best_seq.shape[0]] = best_seq
                cur = np.sum(f)
                if cur > 1e-12:
                    f = f * (target_sum / cur)
                else:
                    f = rng.random(n) + 0.1
                    f = f / np.sum(f) * target_sum
            else:
                # completely fresh start
                f = rng.random(n) + 0.1
                f = f / np.sum(f) * target_sum

        # mixing schedule: start with t=0.3, decay by 5% each iteration
        t = 0.3
        decay = 0.95

        max_iters = 80                # allow many LP refinements for each n
        for it in range(max_iters):
            if time.monotonic() >= deadline:
                break

            # current worst auto‑convolution value (M)
            M = float(np.max(np.convolve(f, f)))
            # build convolution matrix for the constraint (f * g) ≤ M
            A = _conv_matrix(f)

            # ----- linear program -----
            g = cp.Variable(n)
            constraints = [
                g >= 0,
                A @ g <= M
            ]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            # ECOS works well for dense LPs; fall back to SCS if needed.
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
            except Exception:           # pragma: no cover
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000)

            if g.value is None:
                # LP failed – give up on this size
                print(f"[LP] n={n} iteration {it} – solver failed")
                break

            g0 = np.maximum(g.value, 0.0)          # safeguard against tiny negatives
            sum_g = np.sum(g0)
            if sum_g < 1e-12:
                break

            # rescale g to the target ℓ¹‑norm
            g_rescaled = g0 * (target_sum / sum_g)

            # convex combination with decreasing step size
            f = np.clip((1.0 - t) * f + t * g_rescaled, 0.0, 1000.0)

            # evaluate the new candidate
            sc = _score(f)
            if sc < best_score:
                best_score = sc
                best_seq = f.copy()
                print(f"[LP] n={n:3d} it={it:2d} t={t:.3f} score={sc:.6f}")

            # reduce the mixing factor for finer adjustments later
            t = max(0.01, t * decay)

            # early stop if we have already breached the target
            if best_score < 1.503:
                break

        # ------------------------------------------------------------------
        # Short random‑perturbation hill‑climb to polish the current best.
        # ------------------------------------------------------------------
        if best_seq is None:
            continue
        perturb_iters = 200
        for _ in range(perturb_iters):
            if time.monotonic() >= deadline:
                break
            cand = best_seq + rng.normal(scale=0.01, size=best_seq.shape[0])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[perturb] n={len(best_seq)} score={sc:.6f}")
                if best_score < 1.503:
                    break

    # ------------------------------------------------------------------
    # If everything failed (e.g. CVXPY missing) run the simple fallback.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_seq, fallback_score = _fallback_random_search(
            max(0.0, float(budget_s) - (time.monotonic() - start_time)),
            seed=seed
        )
        best_seq = fallback_seq
        best_score = fallback_score

    # final safety: return a plain Python list of floats
    return [float(x) for x in best_seq]
