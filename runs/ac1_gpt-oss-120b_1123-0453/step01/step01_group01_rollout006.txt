    max_val = conv.max()
    exp_arg = beta * (conv - max_val)
    exp_arg = np.clip(exp_arg, -700, 0) # avoid overflow
    w = np.exp(exp_arg)
    w_sum = w.sum()

    wr = w[::-1]
    size = 1 << ((len(a) + len(w) - 1).bit_length())

    start = len(w) - 1

"""
Hybrid search for a low‑max‑convolution step function.

The algorithm works in three stages:

1. **Sub‑gradient descent on the true max** – starts from the arcsine
   distribution (the known analytic construction) and repeatedly moves
   opposite the gradient of the current maximal convolution entry,
   projecting back onto the probability simplex.

2. **Smoothed (soft‑max) gradient descent** – replaces the hard max by a
   temperature‑controlled soft‑max to obtain a smoother objective.
   The temperature β is increased during the run, so the method first
   flattens the whole convolution and then focuses on the remaining peak.

3. **Local random refinements** – cheap pairwise mass swaps and occasional
   block perturbations are performed while time permits.  The best
   sequence seen at any moment is kept and finally returned.

All vectors are kept normalised to sum = 1, which leaves the evaluation
function invariant.  The code respects the 1000 s budget and never writes
to disk.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_sequence(n: int) -> np.ndarray:
    """Return the “arcsine’’ distribution of length n, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def linear_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a via FFT (real‑to‑real)."""
    n = a.shape[0]
    size = 2 * n - 1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a * a (linear convolution)."""
    return float(linear_convolution(a).max())


def score_from_a(a: np.ndarray) -> float:
    """Evaluation score for a vector normalised to sum = 1."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """A cheap O(n) non‑negative perturbation used for exploration."""
    cand = np.copy(base)
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    tot = cand.sum()
    if tot > 0:
        cand /= tot
    else:
        cand[:] = 1.0 / n
    return cand


def softmax_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Uses the identity ∂conv_k/∂a_i = 2·a_{k‑i}.
    """
    n = a.shape[0]
    conv = linear_convolution(a)
    m = conv.max()
    # soft‑max weights (stable formulation)
    exp_arg = beta * (conv - m)
    exp_arg = np.clip(exp_arg, -700, 0)          # avoid overflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w  (valid part length n)
    w_rev = w[::-1]
    size = 1 << ((n + w_rev.size - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    Fw = np.fft.rfft(w_rev, n=size)
    corr = np.fft.irfft(Fa * Fw, n=size)
    start = w_rev.size - 1
    grad = 2.0 * corr[start:start + n]
    return grad
