def conv_matrix(f):
    n = len(f)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A

def lp_max_sum(f, M, cp):
    # Build and solve LP maximize sum(g) s.t. g >=0, A@g <= M
    n = len(f)
    A = conv_matrix(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    return g_val

def project_onto_simplex(v, z=target_sum):
    # similar to Duchi et al.
    # v is vector, we need to find w >= 0, sum w = z minimizing ||w - v||^2
    # algorithm: sort v decreasing, compute cumulative sum.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    # find rho = max i where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""Propose a step sequence by iterative LP refinement with adaptive mixing and occasional gradient descent.

The algorithm:
- Choose a length n (we try several values).
- Initialise a uniform non‑negative vector with ℓ¹‑norm √(2n).
- Repeatedly:
    * Compute M = max(f*f).
    * Solve the linear program   max Σ g_i   s.t. g≥0, (f∗g) ≤ M.
      The LP is solved with SciPy's HiGHS solver (fallback to CVXPY).
    * Rescale g to have the same ℓ¹‑norm as f.
    * Pick a mixing coefficient t (decaying over time, with a small line‑search)
      and set f ← (1‑t)f + t g.
    * Every few iterations apply a projected gradient step that reduces the
      entry of the convolution that currently attains the maximum.
- Keep the best vector seen (lowest value returned by `evaluate_sequence`).
- Stop when the time budget expires and return the best sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helpers (no closures, plain top‑level functions)
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1) × n matrix A such that (A @ g) = f ∗ g."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _solve_lp_scipy(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   maximise Σ g_i   s.t. g≥0, (f∗g) ≤ M   using SciPy (HiGHS)."""
    try:
        import scipy.optimize as opt
    except Exception:
        return None

    n = f.shape[0]
    A = _conv_matrix(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # minimise –Σ g_i  ⇔ maximise Σ g_i

    bounds = [(0.0, None)] * n
    # Use HiGHS; it is very fast for dense problems.
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        return None
    return np.maximum(res.x, 0.0)


def _solve_lp_cvxpy(f: np.ndarray, M: float, cp) -> np.ndarray | None:
    """Solve the same LP with CVXPY (ECOS)."""
    n = f.shape[0]
    A = _conv_matrix(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000)
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)


def _max_autoconv(a: np.ndarray) -> float:
    """Maximum entry of the full autocorrelation a∗a."""
    return float(np.max(np.convolve(a, a)))


def _fast_score(a: np.ndarray) -> float:
    """Compute the scorer value (2·n·max / sum²) without the security wrappers."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    m = np.max(np.convolve(a, a))
    return 2.0 * n * m / (s * s)


def _project_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w≥0, Σw = z}."""
    # Implementation of the algorithm by Duchi et al.
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_at_max(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Return the gradient of the autocorrelation entry that attains the maximum
    together with the index k of that entry.
    Gradient g_i = a_{k-i} + a_i (when the index k−i is inside the array).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    for i in range(i_min, i_max + 1):
        j = k - i
        grad[i] += a[j]
        grad[j] += a[i]
    return grad, k

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: float = 1000.0,
                      **kwargs) -> list[float]:

    Return a list of non‑negative heights that (hopefully) yields a low
    score for the supplied ``evaluate_sequence`` function.

    The routine combines:
        * an LP that maximises the ℓ¹‑mass while keeping the worst
          cross‑convolution fixed,
        * a decaying convex‐combination update,
        * occasional projected‑gradient steps that attack the currently
          maximal autocorrelation entry,
        * restarts with random seeds and a simple random‑perturbation
          fallback when no LP solver is available.

    The algorithm respects the 1000 s time budget and prints progress.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    try:
        import scipy.optimize as _  # noqa: F401
        HAS_SCIPY = True
    except Exception:
        HAS_SCIPY = False

    try:
        import cvxpy as cp
        HAS_CVX = True
    except Exception:
        HAS_CVX = False
        cp = None

    # Helper to compute the official score (calls the provided function).
    def _official_score(seq):
        # ``evaluate_sequence`` is assumed to be present in the global namespace.
        return float(evaluate_sequence(seq.tolist()))

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # Lengths to try – they increase the search space while staying affordable.
    lengths = [200, 400, 600, 800]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise a feasible vector (uniform + tiny noise for diversity)
        # ------------------------------------------------------------------
        f = np.full(n, target_sum / n, dtype=np.float64)
        f += rng.normal(scale=1e-3 * target_sum / n, size=n)
        f = np.clip(f, 0.0, 1000.0)
        f = f / f.sum() * target_sum

        # Mixing schedule
        t = 0.4          # initial mixing weight
        t_min = 0.005    # never go below this
        decay = 0.995    # geometric decay per iteration

        # Counters for stagnation detection
        stagnation = 0
        max_stagnation = 40

        iteration = 0
        while time.time() < deadline:
            iteration += 1

            # --------------------------------------------------------------
            # 1. Current worst autocorrelation value
            # --------------------------------------------------------------
            M = _max_autoconv(f)

            # --------------------------------------------------------------
            # 2. Solve the LP  max Σ g_i  s.t. f∗g ≤ M , g ≥ 0
            # --------------------------------------------------------------
            if HAS_SCIPY:
                g_raw = _solve_lp_scipy(f, M)
            elif HAS_CVX:
                g_raw = _solve_lp_cvxpy(f, M, cp)
            else:
                g_raw = None

            if g_raw is None:
                # LP failure – break out of the inner loop and fall back.
                break

            sum_g = g_raw.sum()
            if sum_g <= 1e-12:
                break

            # --------------------------------------------------------------
            # 3. Rescale g to have the same ℓ¹‑norm as f (which is target_sum)
            # --------------------------------------------------------------
            g = g_raw * (target_sum / sum_g)

            # --------------------------------------------------------------
            # 4. Choose a mixing coefficient (small line‑search)
            # --------------------------------------------------------------
            cand_ts = [t, max(t * 0.5, t_min), min(t * 1.5, 1.0)]
            best_local_f = None
            best_local_M = None
            best_local_t = None
            for tt in cand_ts:
                f_new = (1.0 - tt) * f + tt * g
                f_new = np.clip(f_new, 0.0, 1000.0)
                M_new = _max_autoconv(f_new)
                if best_local_M is None or M_new < best_local_M:
                    best_local_M = M_new
                    best_local_f = f_new
                    best_local_t = tt

            f = best_local_f
            t = max(t * decay, t_min)   # slowly decay mixing weight

            # --------------------------------------------------------------
            # 5. Occasionally perform a cheap projected‑gradient step.
            # --------------------------------------------------------------
            if iteration % 7 == 0:
                grad, _ = _gradient_at_max(f)
                # A modest step size; the factor 0.1 is empirical.
                step = 0.1 * (M / (np.linalg.norm(grad) + 1e-12))
                f_tmp = f - step * grad
                f_tmp = _project_simplex(f_tmp, target_sum)
                f_tmp = np.clip(f_tmp, 0.0, 1000.0)
                # Accept if it does not worsen the worst entry.
                if _max_autoconv(f_tmp) <= _max_autoconv(f):
                    f = f_tmp

            # --------------------------------------------------------------
            # 6. Record the best sequence seen so far.
            # --------------------------------------------------------------
            cur_score = _fast_score(f)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                stagnation = 0
                print(f"[n={n}] it={iteration:04d} t={best_local_t:.4f} "
                      f"score={cur_score:.6f}")
                # Early exit if we have already beaten the target bound.
                if best_score < 1.49:
                    print("Target sufficiently beaten – terminating early.")
                    break
            else:
                stagnation += 1

            # --------------------------------------------------------------
            # 7. Restart if we have been stuck for a while.
            # --------------------------------------------------------------
            if stagnation >= max_stagnation:
                # Create a fresh random feasible vector.
                f = rng.random(n) + 0.1
                f = f / f.sum() * target_sum
                t = 0.4
                stagnation = 0
                print(f"[n={n}] restart after {iteration} iterations (stagnation).")

        # End of while loop for this length

    # ----------------------------------------------------------------------
    # Fallback: simple random perturbation if nothing was found.
    # ----------------------------------------------------------------------
    if best_seq is None:
        print("No LP run succeeded – using simple random search fallback.")
        n = 200
        f = np.full(n, 1.0, dtype=np.float64)
        best_seq = f.copy()
        best_score = _fast_score(f)
        while time.time() < deadline:

            cand = np.clip(cand, 0.0, 1000.0)
            sc = _fast_score(cand)
            if sc < best_score:
                best_seq = cand.copy()
                best_score = sc
                print("[fallback] new best score =", sc)

    # Final safety: ensure list of plain Python floats.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
