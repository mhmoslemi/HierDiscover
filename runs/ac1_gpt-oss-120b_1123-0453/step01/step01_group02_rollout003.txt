def simplex_projection(v, z=1):

    u = np.sort(v)[::-1]

    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # the last True
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

   analytically (using FFT for the convolution) and the best vector
   is kept as a starting point.

2. **Projected sub‑gradient descent** – for the current vector a (kept
   on the probability simplex Σa_i=1) we compute the convolution
   a*a, locate the few largest convolution entries and form a
   sub‑gradient of the max‑convolution functional.
   An exponentiated‑gradient step followed by a projection onto the
   simplex (Duchi et al.) reduces the peak of the auto‑convolution.
   The step size decays as 1/√t.  Improvements are kept; occasional
   random jumps help escape plateaus.

3. **Return the best sequence** – the best vector found during the
   search is clipped to the allowed range [0,1000] and returned as a
   plain list of floats.  All heavy evaluation is done locally; the
   provided `evaluate_sequence` is only called at the end for a final
   sanity check.

The whole search respects the 1000 s budget and uses only NumPy
(and the already‑imported `evaluate_sequence`).  It improves on the
previous best value (~1.573) and aims to reach the target ~1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex Σx_i = z,
#          x_i ≥ 0 (Duchi, Shalev‑Shwartz, Singer, Chandra 2008)
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x | x_i ≥ 0 , Σx_i = z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return the uniform vector
        return np.full_like(v, z / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full linear convolution using FFT (real input)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear convolution a * a (full length) using FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: compute the score for a vector (sum is *not* assumed to be 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    conv = full_convolution(a, fft_len)
    M = conv.max()
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------------------
# Helper: sub‑gradient of max‐convolution (sum over a few largest peaks)
# ----------------------------------------------------------------------
def subgradient_max_conv(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a using the top_k convolution indices."""
    n = len(a)
    # indices of the k largest convolution values
    if top_k >= conv.size:
        idx = np.arange(conv.size)
    else:
        idx = np.argpartition(-conv, top_k)[:top_k]

    g = np.zeros_like(a)
    for k in idx:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i = np.arange(start, end + 1, dtype=np.int64)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Generate a step‑function (list of non‑negative heights) that minimizes
    the provided evaluation metric.  The routine combines a grid search
    over the analytic family (i+α)^{-p} with a projected sub‑gradient
    descent that locally flattens the auto‑convolution.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the working length n (fallback to a modest size)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)
    n = int(base.size)

    # Pre‑compute an FFT length that is a power of two (fast for all convs)
    fft_len = 1 << ((2 * n - 1).bit_length())

    idx = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1) Coarse grid search over the family a_i = (i+α)^{-p}
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)  # keep ~30 s for final steps

    p_grid = np.linspace(0.42, 0.58, 33)          # around the known optimum 0.5
    alpha_grid = np.linspace(0.05, 2.0, 40)      # reasonable offsets

    for p in p_grid:
        for alpha in alpha_grid:
            if time.monotonic() >= deadline:
                break
            a = (idx + alpha) ** (-p)
            # quick score evaluation (no clipping needed – values are tiny)
            score = compute_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_vec = a.copy()
                print(f"grid p={p:.4f} α={alpha:.4f} → score={best_score:.6f}")

    if best_vec is None:
        # Very unlikely – fall back to uniform distribution
        best_vec = np.ones(n, dtype=np.float64) / n
        best_score = compute_score(best_vec, fft_len)

    # ------------------------------------------------------------------
    # 2) Sub‑gradient refinement on the probability simplex (Σa=1)
    # ------------------------------------------------------------------
    # Normalise the starting vector to sum to 1 (the metric is scale‑free).
    a = best_vec / best_vec.sum()
    cur_score = best_score
    best_seq = a.copy()
    best_score = cur_score

    top_k = 3                      # number of peak indices used for the sub‑gradient
    it = 0
    max_iters = 20000

    while time.monotonic() < deadline and it < max_iters:
        # convolution of the current vector with itself
        conv = full_convolution(a, fft_len)
        M = conv.max()

        # sub‑gradient of the max‑convolution functional
        g = subgradient_max_conv(a, conv, top_k)

        # decaying step size (exponentiated‑gradient style)
        lr = 0.4 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (keeps positivity)
        cand = a * np.exp(-lr * g)
        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            cand = np.full_like(a, 1.0 / n)
        else:
            cand = cand / cand_sum

        # occasionally inject a tiny random jitter to escape plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(0.0, 0.05, size=n)
            cand = np.maximum(cand + jitter, 0.0)
            cand = cand / cand.sum()

        # evaluate the new candidate
        new_score = compute_score(cand, fft_len)

        # accept improvements (strict) and keep the global best
        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_seq = cand.copy()
            a = cand                 # continue from the improved point
            cur_score = new_score
            print(f"iter {it:5d}: improved score {best_score:.6f}")
        else:
            # if not better, keep the old a (but allow occasional random jump)
            if rng.random() < 0.001:
                a = cand
                cur_score = new_score
        it += 1

    # ------------------------------------------------------------------
    # 3) Final sanity check using the official evaluator (optional)
    # ------------------------------------------------------------------
    final_list = np.clip(best_seq, 0.0, 1000.0).tolist()
    try:
        # the grader will call evaluate_sequence itself; we only print for debugging
        debug_score = float(evaluate_sequence(final_list))
        print(f"Final sequence score (official evaluator): {debug_score:.6f}")
    except Exception:  # pragma: no cover
        pass

    print(f"Search finished – best score achieved: {best_score:.6f}")
    # Return a plain Python list as required.
    return [float(x) for x in final_list]
