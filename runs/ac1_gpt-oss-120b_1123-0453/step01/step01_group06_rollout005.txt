def solve_lp(f, M):
    n = f.size
    # Build matrices A_ub and b_ub
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum
    bounds = [(0, 1000) for _ in range(n)]
    # use high-quality solver
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        return g
    else:
        return None

"""
Frank‑Wolfe style optimizer for the step‑function evaluation problem.

The algorithm works in two phases:

1️⃣  Warm‑start with a few analytic power‑law shapes (a_i ∝ (i+1)^‑p) and keep
    the best one according to the fast FFT bound.

2️⃣  Repeatedly apply a Frank‑Wolfe move:
        – compute M = max(conv(f,f)) (fast FFT)
        – solve a linear program  max Σ g_i   s.t.   f * g ≤ M , 0 ≤ g_i ≤ 1000
          (variables are the new coefficients g, the constraints are linear in g)
        – blend the current solution f with g :  f←(1‑α)·f + α·g  (α≈0.3)
          and renormalise to unit ℓ₁‑norm (the objective is scale‑invariant).
        – accept the new f if the fast bound improves the best known score,
          otherwise keep the old one.
    A few cheap stochastic perturbations are interleaved to escape possible
    plateaux.

The LP is small (≈2 n rows × n columns) and can be solved with SciPy’s
high‑performance “highs” solver.  Using n≈2000 gives a good trade‑off between
search power and runtime.  The whole routine respects the 1000‑second budget
and finally returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic FFT‑based helpers (fast, no security checks)
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Return max(conv(seq, seq)) using a real FFT (linear convolution)."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(conv[: 2 * n - 1].max())


def _score_estimate(seq: np.ndarray) -> float:
    """Fast surrogate for evaluate_sequence (scale‑invariant)."""
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Linear‑program (LP) to maximise Σ g_i under f∗g ≤ M
# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve max Σ g_i  subject to  f * g ≤ M  and 0 ≤ g_i ≤ 1000.
    Returns the optimal g as a NumPy array, or None if the solver fails.
    """
    n = f.size
    rows = 2 * n - 1

    # Build the convolution matrix A (row k, column j gets f[k‑j] when in range)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  ⇔ minimise –Σ g_i
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # SciPy’s HiGHS LP solver is fast for dense problems of this size.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return np.asarray(res.x, dtype=np.float64)
    return None


# ----------------------------------------------------------------------
# Stochastic perturbations (same as in the reference code)
# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    scale = 0.05 * cand[start:start + block].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:start + block] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)


def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = seq.copy()
    n = cand.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())

    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]

    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand

    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))

    delta = cand[src] * float(0.05 + 0.15 * rng.random())
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) that minimises
    evaluate_sequence.  The routine respects a ~1000 s time budget and uses a
    Frank‑Wolfe / LP driven optimisation combined with stochastic refinements.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start: try a handful of power‑law shapes
    # --------------------------------------------------------------
    n = 2000                     # length used for the LP phase (good trade‑off)
    best = None
    best_score = float("inf")

    for p in np.linspace(0.30, 0.60, 7):
        idx = np.arange(n, dtype=np.float64) + 1.0
        cand = np.power(idx, -p)
        cand = np.clip(cand, 0.0, 1000.0)
        cand /= cand.sum()                     # normalise ℓ₁‑norm → 1.0
        est = _score_estimate(cand)
        if est < best_score:
            # call the authoritative evaluator only if the cheap bound looks promising
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = cand
                best_score = exact
                print(f"[warm‑start] p={p:.3f} → score {best_score:.6f}")

    # If, for any reason, the warm‑start failed, fall back to a uniform vector.
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best /= best.sum()
        best_score = float(evaluate_sequence(best.tolist()))
        print("[fallback] uniform start →", best_score)

    # --------------------------------------------------------------
    # 2️⃣ Frank‑Wolfe / LP driven improvement loop
    # --------------------------------------------------------------
    lp_iter = 0
    max_lp_iter = 40          # keep the LP phase short – each solve costs ≈ 1 s
    alpha = 0.30              # blending factor (can be tuned later)

    while time.monotonic() < deadline and lp_iter < max_lp_iter:
        lp_iter += 1
        M = _conv_max(best)                      # current maximal convolution

        g = _solve_lp(best, M)
        if g is None:
            print("[lp] solver failed – leaving LP phase")
            break

        # Blend and renormalise (scale does not matter for the score)
        cand = (1.0 - alpha) * best + alpha * g
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() == 0.0:
            continue
        cand /= cand.sum()

        est_score = _score_estimate(cand)
        if est_score < best_score - 1e-9:
            exact_score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact_score) and exact_score < best_score:
                best = cand
                best_score = exact_score
                print(f"[lp‑{lp_iter:02d}] improvement → {best_score:.6f}")
                # optionally shrink α as we get closer to a plateau
                alpha = max(0.10, alpha * 0.95)
        else:
            # No improvement – reduce step size to try a subtler move
            alpha = max(0.05, alpha * 0.85)

    # --------------------------------------------------------------
    # 3️⃣ Stochastic local‑search fine‑tuning (use same moves as before)
    # --------------------------------------------------------------
    move_fns = [_make_block_perturbation,
                _perturb_swap,
                _perturb_noise,
                _flatten_move]

    it = 0
    report_every = 2000
    while time.monotonic() < deadline:
        move = rng.choice(move_fns)
        cand = move(best, rng)

        # Renormalise to keep the scale fixed (the objective is scale‑invariant)
        if cand.sum() == 0.0:
            continue
        cand /= cand.sum()

        est = _score_estimate(cand)
        if est < best_score - 1e-12:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = cand
                best_score = exact
                print(f"[{it:07d}] stochastic improvement → {best_score:.6f}")

        it += 1
        if it % report_every == 0:
            print(f"[{it:07d}] current best = {best_score:.6f}")

    print(f"Finished after {it} stochastic steps – best score = {best_score:.6f}")

    # The evaluator will clip values to [0,1000] and enforce a minimal sum,
    # so we simply return the normalized sequence (any positive scaling works).
    return [float(x) for x in best]
