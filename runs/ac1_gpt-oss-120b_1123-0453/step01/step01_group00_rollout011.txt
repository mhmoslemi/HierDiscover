    import cvxpy as cp

    _HAS_CVX = False

def _score(seq):
    return float(evaluate_sequence(seq.tolist()))

def _conv_matrix(f):
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A

def _softmax_weights(conv, beta):
    maxc = np.max(conv)
    # stable exp
    exp_vals = np.exp(beta * (conv - maxc))
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0:
        # fallback: uniform
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp

def _gradient_step(a, target_sum, beta, step):
    # a: numpy array (len n)
    n = a.shape[0]
    conv = np.convolve(a, a)
    w = _softmax_weights(conv, beta)
    # convolution of w with reversed a
    grad = 2.0 * np.convolve(w, a[::-1])[n-1:n-1+n]
    # descent
    a_new = a - step * grad
    # clip and renorm
    a_new = np.clip(a_new, 0.0, 1000.0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new * (target_sum / s)
    else:
        # if all zero, reinitialize
        a_new = np.full_like(a, target_sum / n)
    return a_new, conv, w

def _lp_improve(f, M, target_sum, t, deadline):
    # solve LP to find g maximizing sum g under constraints A @ g <= M
    n = f.shape[0]
    A = _conv_matrix(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000)
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    sum_g = np.sum(g0)
    if sum_g <= 1e-12:
        return None
    g_rescaled = g0 * (target_sum / sum_g)
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)
    return f_new

"""
Search for a non‑negative step sequence that minimises `evaluate_sequence`.

The algorithm mixes two complementary heuristics:

1. **Smoothed‑max gradient descent**  
   The objective  max_k (a*a)[k]  is approximated by a soft‑max
   (log‑sum‑exp).  Its gradient can be written in closed form and
   evaluated with two convolutions.  Starting from a random vector,
   a few hundred inexpensive GD steps quickly push mass away from the
   indices that dominate the auto‑convolution.  The temperature
   parameter β is annealed so that the optimisation first smoothes the
   whole profile and then concentrates on the worst peak.

2. **Linear‑programming refinement** (optional, uses CVXPY)  
   After a GD block we solve a small LP:
        maximise   Σ g_i
        s.t.       (f * g)_k ≤ M   (M = max(f*f))
                     g_i ≥ 0
   The optimal g is rescaled to keep the ℓ¹‑norm fixed and mixed with
   the current f via a convex combination.  This step provably never
   worsens the score and frequently provides a noticeable jump.

The search is performed for several lengths (150…500) with a few
random restarts each.  The best sequence found before the time budget
expires is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional CVXPY import – the LP refinement is used only if the library
# is available.
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:                                   # pragma: no cover
    cp = None
    _HAS_CVX = False

# ----------------------------------------------------------------------
def _score(seq):
    """Return the scalar score produced by the provided evaluator."""
    # `evaluate_sequence` is assumed to be already imported in the runner.
    return float(evaluate_sequence(seq.tolist()))


def _conv_matrix(f):
    """
    Build the (2n‑1) × n matrix A such that (A @ g) == (f * g) for any g.
    Used in the LP where the constraints are linear in g.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def _softmax_weights(conv, beta):
    """
    Compute a probability vector proportional to exp(beta * conv).
    The subtraction of the maximum prevents overflow.
    """
    maxc = np.max(conv)
    exp_vals = np.exp(beta * (conv - maxc))
    s = np.sum(exp_vals)
    if s == 0.0:                                     # degenerate – uniform fallback
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def _gradient_step(a, target_sum, beta, step):
    """
    Perform one gradient‑descent step on the smoothed max of the
    auto‑convolution.  Returns the updated vector, the current convolution
    and the soft‑max weights (needed only for diagnostics).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                         # length 2n‑1
    w = _softmax_weights(conv, beta)                # same length
    # grad_i = 2 * Σ_j w_{i+j} * a_j
    # which is the central n entries of the convolution of w with reversed a
    grad = 2.0 * np.convolve(w, a[::-1])[n - 1:n - 1 + n]

    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    s = a_new.sum()
    if s > 0.0:
        a_new = a_new * (target_sum / s)            # renormalise ℓ¹‑norm
    else:                                            # safety net – fall back to uniform
        a_new = np.full_like(a, target_sum / n)

    return a_new, conv, w


def _lp_improvement(f, M, target_sum, mix):
    """
    One LP refinement: maximise Σ g_i with constraints (f * g) ≤ M,
    then blend the result with the current f using mixing weight `mix`.
    Returns the new vector or None on failure.
    """
    n = f.shape[0]
    A = _conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:                               # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000)

    if g.value is None:
        return None

    g0 = np.maximum(g.value, 0.0)
    sum_g = np.sum(g0)
    if sum_g <= 1e-12:
        return None

    g_rescaled = g0 * (target_sum / sum_g)
    f_new = (1.0 - mix) * f + mix * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)
    return f_new
