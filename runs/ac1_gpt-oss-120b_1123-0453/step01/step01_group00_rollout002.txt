global imports of numpy, math, time, scipy.optimize.

Define helper conv_matrix(f): produce a 2n-1 x n matrix A as earlier.

Define helper solve_max_sum_lp(f, M, n, method='highs'): 
   - Build A = conv_matrix(f)
   - b = np.full(A.shape[0], M)
   - c = -np.ones(n)
   - bounds = [(0, None)] * n
   - result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
   - If result.success and result.x is not None: g = result.x
   - else: return None.

Define helper optimal_alpha(f, g): compute conv_f_f, conv_f_g, conv_g_g, etc., evaluate score for alphas.

   - Compute S_f = sum(f), S_g = sum(g)
   - Compute conv arrays: conv_ff = np.convolve(f,f); conv_fg = np.convolve(f,g); conv_gg = np.convolve(g,g)
   - lambda to compute score(alpha):
         f_sum = (1-alpha)*S_f + alpha*S_g
         conv = (1-alpha)**2 * conv_ff + 2*alpha*(1-alpha) * conv_fg + alpha**2 * conv_gg
         M_new = np.max(conv)
         return 2 * len(f) * M_new / (f_sum**2)

   - Do coarse grid: alphas = np.linspace(0, 1, 51) => compute scores.

def find_optimal_alpha(f, g):
    n = len(f)
    sum_f = float(np.sum(f))
    sum_g = float(np.sum(g))
    conv_ff = np.convolve(f, f)
    conv_fg = np.convolve(f, g)
    conv_gg = np.convolve(g, g)

    def score(alpha):
        # Weighted sums
        combined_sum = (1 - alpha) * sum_f + alpha * sum_g
        # Combined convolution
        conv_comb = ((1 - alpha)**2) * conv_ff + (2 * alpha * (1 - alpha)) * conv_fg + (alpha**2) * conv_gg
        M_new = float(np.max(conv_comb))

assistantfinal**Solution Overview**

The scorer evaluates  

\[
\text{score}(a)=\frac{2\,n\;\max_k(f*f)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad 
f=(a_0,\dots ,a_{n-1}),
\]

so we would like a large \(\ell^1\)‑mass \(\sum a_i\) while keeping the worst entry of the auto‑convolution small.
The algorithm below repeatedly

1. **LP step** – given the current vector `f` and its worst convolution value `M`,
   solves the linear program  

   \[
   \max\;\sum_i g_i \;\; \text{s.t.}\;\; (f*g)_k\le M,\; g_i\ge0,\; g_i\le1000 .
   \]

   The optimum `g` is a non‑negative vector that carries **more ℓ¹‑mass** than `f` without
   increasing any entry of the mixed convolution `f*g`.

2. **Adaptive mixing** – instead of a fixed mixing weight `t` we perform a cheap
   line‑search over \(\alpha\in[0,1]\) and minimise the true score of  

   \[
   f_{\text{new}}=(1-\alpha)f+\alpha g .
   \]

   The convolution of the mixture can be written analytically, so a fine grid
   (coarse→refined) gives the best \(\alpha\) cheaply.

3. **Length enlargement** – after a few successful updates the vector is padded
   with zeros (which does not change its current score).  The LP can now put
   extra mass into the new entries, further reducing the ratio.

4. **Final polishing** – if time is still left a short random‑perturbation
   hill‑climb is applied.

All heavy work (LP solving, matrix assembly, line‑search) is done with NumPy
and SciPy’s fast interior‑point LP solver (`highs`).  The whole routine obeys
the 1000 s budget and returns a plain Python list of floats.

```python
"""
Propose a low‑value step sequence.

The algorithm starts from a known good sequence (or a uniform vector) and
repeatedly performs:
  * a linear programme that maximises the ℓ¹‑mass of a candidate g under the
    constraint (f * g)_k ≤ M, where M is the current worst auto‑convolution.
  * an adaptive line‑search over the mixing coefficient α to minimise the
    true score 2·n·max(f_new*f_new)/sum(f_new)² of the convex combination
    f_new = (1‑α)·f + α·g.
Optionally the length of the vector is increased (padding with zeros) to give
the LP more freedom.  When the time budget almost expires a short random
perturbation hill‑climb is performed as final polishing.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build the (2n‑1) × n matrix A such that (A @ g)[k] = (f * g)[k]."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each convolution entry is a sum of f[i]·g[k‑i]
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


def solve_max_sum_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        max  Σ g_i
        s.t. (f * g)_k ≤ M   for all k
             0 ≤ g_i ≤ 1000
    Returns the optimal g (numpy array) or None on failure.
    """
    n = f.shape[0]
    A = conv_matrix(f)                     # (2n‑1) × n
    b = np.full(A.shape[0], M, dtype=np.float64)

    # linprog minimises cᵀx, therefore minimise –Σg_i
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, 1000.0) for _ in range(n)]

    # use the fast HiGHS interior‑point solver
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success or res.x is None:
        return None
    # numerical safety
    g = np.maximum(res.x, 0.0)
    return g


def find_optimal_alpha(f: np.ndarray, g: np.ndarray) -> tuple[float, float, np.ndarray]:
    """
    Perform a cheap line‑search over α∈[0,1] to minimise the true score of
    (1‑α)·f + α·g.

    Returns (best_alpha, best_score, f_new).
    """
    n = len(f)
    sum_f = float(np.sum(f))
    sum_g = float(np.sum(g))

    conv_ff = np.convolve(f, f)
    conv_fg = np.convolve(f, g)
    conv_gg = np.convolve(g, g)

    def score(alpha: float) -> float:
        """Score for a given mixing coefficient."""
        s = (1.0 - alpha) * sum_f + alpha * sum_g
        # convolution of the mixture (quadratic in α)
        conv = ((1.0 - alpha) ** 2) * conv_ff \
               + (2.0 * alpha * (1.0 - alpha)) * conv_fg \
               + (alpha ** 2) * conv_gg
        M_new = float(np.max(conv))
        return 2.0 * n * M_new / (s ** 2)

    # ----- coarse sweep -------------------------------------------------
    alphas = np.linspace(0.0, 1.0, 51)          # step ≈0.02
    scores = np.array([score(a) for a in alphas])
    best_idx = int(np.argmin(scores))
    best_alpha = alphas[best_idx]
    best_score = scores[best_idx]

    # ----- refined sweep ------------------------------------------------
    lo = max(0.0, best_alpha - 0.04)
    hi = min(1.0, best_alpha + 0.04)
    alphas_ref = np.linspace(lo, hi, 41)       # step ≈0.002
    for a in alphas_ref:
        sc = score(a)
        if sc < best_score:
            best_score = sc
            best_alpha = a

    # construct the new vector
    f_new = (1.0 - best_alpha) * f + best_alpha * g
    f_new = np.clip(f_new, 0.0, 1000.0)

    return best_alpha, best_score, f_new

    # Helper to evaluate a candidate (global evaluate_sequence is provided)
    def _score(seq: np.ndarray) -> float:
        return float(evaluate_sequence(seq.tolist()))

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # Initialise the starting sequence
    if 'height_sequence_1' in globals():
        f0 = np.array(height_sequence_1, dtype=np.float64)
        f0 = np.clip(f0, 0.0, 1000.0)
    else:
        # a simple uniform start (score = 2.0)
        n0 = 200
        f0 = np.full(n0, 1.0, dtype=np.float64)

    best_seq = f0.copy()
    best_score = _score(best_seq)
    print("[init] length =", len(best_seq), "score =", best_score)

    # ------------------------------------------------------------------
    # Main optimisation loop – enlarge length step by step
    rng = np.random.default_rng(seed + 999)
    max_len = 800                     # safety cap (still fast for the LP)
    step_len = 100                    # how many zeros we add each enlargement

    f = best_seq.copy()
    while time.monotonic() < deadline and len(f) < max_len:
        # ------------------------------------------------------------------
        # One phase on the current length
        n = len(f)
        max_iter = 30                 # can be increased if time permits
        for it in range(max_iter):
            if time.monotonic() >= deadline:
                break

            # current worst auto‑convolution value
            conv_ff = np.convolve(f, f)
            M = float(np.max(conv_ff))

            # solve the LP
            g = solve_max_sum_lp(f, M)
            if g is None:
                print("[LP] failed at n =", n, "it =", it)
                break

            # adaptive mixing
            alpha, cand_score, f_new = find_optimal_alpha(f, g)

            # evaluate the candidate (tiny numerical differences may appear)
            eval_score = _score(f_new)

            if eval_score < best_score - 1e-12:
                best_score = eval_score
                best_seq = f_new.copy()
                print(f"[update] n={n} it={it} α={alpha:.4f} score={eval_score:.6f}")

            # stop early if we already reached the target
            if best_score < 1.503:
                print("[goal] reached target score", best_score)
                return [float(x) for x in best_seq]

            # prepare next iteration
            f = f_new

        # ------------------------------------------------------------------
        # Enlarge the vector (pad with zeros) if we still have time
        if time.monotonic() + 0.5 >= deadline:
            break
        new_len = min(max_len, n + step_len)
        if new_len > n:
            padding = np.zeros(new_len - n, dtype=np.float64)
            f = np.concatenate([f, padding])
            best_seq = np.concatenate([best_seq, padding])
            print("[extend] new length =", new_len)

    # ------------------------------------------------------------------
    # Final random local search (short hill‑climb) if time allows
    if time.monotonic() < deadline:
        print("[final] random refinement")
        f = best_seq.copy()
        n = len(f)
        while time.monotonic() < deadline:
            cand = f + rng.normal(scale=0.02, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = _score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                f = cand
                print("[refine] new best", best_score)

    print("[done] best score =", best_score, "length =", len(best_seq))
    # Convert to a plain Python list of floats as required by the framework
    return [float(x) for x in best_seq]
```
