# Helper functions – no closures, no lambdas
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Maximum of the linear convolution seq * seq using FFT."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(conv[: 2 * n - 1].max())


def _score_estimate(seq: np.ndarray) -> float:
    """Fast surrogate of evaluate_sequence (uses FFT for max‑convolution)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                         # occasional symmetry boost
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray,
                  rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray,
                   rng: np.random.Generator) -> np.ndarray:
    """Add small zero‑mean Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)


def _flatten_move(seq: np.ndarray,
                  rng: np.random.Generator) -> np.ndarray:
    """Shift a modest amount of mass away from the pair that creates the
    current maximal convolution value."""
    cand = seq.copy()
    n = cand.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())

    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]
    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand

    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))
    delta = cand[src] * float(0.05 + 0.15 * rng.random())  # 5‑%‑20 %
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimize the
    objective defined by ``evaluate_sequence``.  The routine respects the
    supplied time budget (in seconds) and prints periodic progress information.
    """
    rng = np.random.default_rng(seed)
    # give the optimiser a small safety margin (the caller trims it)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start: dense scan of a simple analytic family
    # ------------------------------------------------------------------
    # pick a length – use the provided construction if available, otherwise a
    # default of 4096 (still fast for FFT).
    if 'height_sequence_1' in globals():
        max_len = int(len(height_sequence_1))
    else:
        max_len = 4096

    # we also try a shorter version (≈ half) because the factor 2·n in the
    # objective sometimes favours a smaller n.
    lengths = [max_len]
    if max_len > 1500:
        lengths.append(max_len // 2)

    # grids for the two parameters
    p_grid = np.linspace(0.45, 0.65, 41)          # step ≈0.005
    offset_grid = np.arange(0.0, 5.1, 0.5)       # 0 … 5

    best_seq = None
    best_score = float("inf")
    # store a few promising candidates (estimate only)
    promising = []                               # (est, seq)

    for n in lengths:
        idx = np.arange(1, n + 1, dtype=np.float64)   # i = 1 … n
        for p in p_grid:
            for off in offset_grid:
                a = np.power(idx + off, -p)
                a = np.clip(a, 0.0, 1000.0)
                est = _score_estimate(a)
                # keep the 30 best estimates overall
                if len(promising) < 30:
                    promising.append((est, a.copy()))
                elif est < max(promising, key=lambda x: x[0])[0]:
                    # replace the worst entry
                    worst_idx = max(range(len(promising)),
                                    key=lambda i: promising[i][0])
                    promising[worst_idx] = (est, a.copy())

    # sort and evaluate the very best few with the true (expensive) evaluator
    promising.sort(key=lambda x: x[0])
    for est, cand in promising[:12]:
        # cheap check that sum is not tiny (should never happen here)
        if cand.sum() < 0.01:
            continue
        exact = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(exact) and exact < best_score:
            best_score = exact
            best_seq = cand.copy()
            print(f"[warm‑start] exact {exact:.6f} (est {est:.6f})")

    # If for any reason the warm‑start failed, fall back to a uniform vector.
    if best_seq is None:
        best_seq = np.ones(max_len, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using uniform start, score =", best_score)

    # ------------------------------------------------------------------
    # 2️⃣ Stochastic local optimisation (the heavy part)
    # ------------------------------------------------------------------
    it = 0
    report_interval = 2000
    move_functions = [_make_block_perturbation,
                      _perturb_swap,
                      _perturb_noise,
                      _flatten_move]

    # keep the original total mass – it never changes (scaling is neutral)
    target_sum = float(best_seq.sum())

    while time.monotonic() < deadline:
        # pick a random move
        move = rng.choice(move_functions)
        candidate = move(best_seq, rng)

        # renormalise the total mass (preserves the objective)
        s = float(candidate.sum())
        if s < 0.01:
            continue
        candidate *= (target_sum / s)

        # cheap bound – only evaluate exactly when the bound looks promising
        est_score = _score_estimate(candidate)
        if est_score < best_score - 1e-12:
            exact_score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(exact_score) and exact_score < best_score:
                best_seq = candidate
                best_score = exact_score
                print(f"[{it:07d}] improvement: {best_score:.6f}")

        it += 1
        if it % report_interval == 0:
            print(f"[{it:07d}] current best = {best_score:.6f}")

    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    # Return a plain Python list of floats as required
    return [float(x) for x in best_seq]
