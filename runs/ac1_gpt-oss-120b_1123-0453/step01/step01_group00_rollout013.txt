import time, math, numpy as np

def _score(seq):
    return float(evaluate_sequence(seq.tolist()))

def _conv_matrix(f):
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A

def _lp_improve(f, target_sum, t, rng):
    # compute M
    M = float(np.max(np.convolve(f, f)))
    A = _conv_matrix(f)
    try:
        import cvxpy as cp
    except Exception:
        return None, None, None  # cvxpy not available
    
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-6, reltol=1e-6)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000)
        except Exception:
            return None, None, None
    if g.value is None:
        return None, None, None
    g0 = np.maximum(g.value, 0.0)
    sumg = np.sum(g0)
    if sumg <= 1e-12:
        return None, None, None
    g_rescaled = g0 * (target_sum / sumg)
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)
    sc = _score(f_new)
    return f_new, sc, M

def _simulated_annealing(seq, best_score, deadline, rng):
    cur_seq = seq.copy()
    cur_score = best_score
    best_seq = seq.copy()
    best_sc = best_score
    start = time.monotonic()
    # parameters
    temp_start = 1.0
    temp_end = 1e-4
    # iterate until deadline
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        temp = temp_start * (temp_end / temp_start) ** frac
        # propose mutation
        idx = rng.integers(len(cur_seq))
        # step size based on temp
        delta = rng.normal(scale=0.5) * temp * 5  # adapt
        cand = cur_seq.copy()
        cand[idx] = float(np.clip(cand[idx] + delta, 0.0, 1000.0))
        cand_score = _score(cand)
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_sc:
                best_sc = cand_score
                best_seq = cand.copy()
                print(f"[SA] improvement: {best_sc:.6f}")
        # occasionally try small random perturb to escape
        if rng.random() < 0.001:
            # random reset a few entries
            pert = np.maximum(cur_seq + rng.normal(scale=0.1, size=len(cur_seq)), 0.0)
            pert = np.clip(pert, 0.0, 1000.0)
            cur_seq = pert
            cur_score = _score(cur_seq)
    return best_seq, best_sc

"""
Search for a low‑value step‑function sequence.

The strategy consists of three stages:

1️⃣  **Initialisation** – use the previously found best construction
    (global variable `height_sequence_1`) if it exists; otherwise start
    from a uniform sequence whose ℓ¹‑norm equals √(2·n).

2️⃣  **LP refinement** – for several lengths `n` (150 … 400) we repeatedly
    solve the linear programme

        maximise   Σ g_i
        subject to (f * g)_k ≤ M   for all k,
                     g_i ≥ 0,

    where `M = max(f * f)`.  The solution `g` is rescaled to the same
    ℓ¹‑norm as `f` and mixed with `f` (parameter `t`).  Repeating this
    step monotonically reduces the evaluation score
    `2·n·max(f*f)/‖f‖₁²`.

3️⃣  **Local optimisation** – a simulated‑annealing hill‑climber
    explores the neighbourhood of the best sequence found by the LP
    phase, accepting occasional uphill moves to escape plateaus.

The whole routine stops after the allotted time (default 1000 s) and
returns the best sequence as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _score(seq: np.ndarray) -> float:
    """Evaluate a candidate sequence using the provided scorer."""
    # `evaluate_sequence` is supplied in the execution environment.
    return float(evaluate_sequence(seq.tolist()))


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix `A` such that (A @ g)[k] == (f * g)[k].
    Each row k contains the coefficients f[i] for those i that pair with
    g[k‑i] in the convolution.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i ranges over indices of f that contribute to output k
        for i in range(i_start, i_end + 1):
            col = k - i               # the index of the g‑entry that pairs with f[i]
            A[k, col] = f[i]
    return A


def _lp_improve(
    f: np.ndarray, target_sum: float, mix: float, rng: np.random.Generator
) -> tuple[np.ndarray | None, float | None]:
    """
    Perform one LP‑based improvement step.

    Parameters
    ----------
    f : np.ndarray
        Current sequence (non‑negative, ℓ¹‑norm = ``target_sum``).
    target_sum : float
        Desired ℓ¹‑norm after rescaling.
    mix : float
        Convex‑combination weight `t` (0 < t ≤ 1).
    rng : np.random.Generator
        Random generator (only needed for a warm start in CVXPY).

    Returns
    -------
    (new_f, new_score)
        The updated sequence and its score, or ``(None, None)`` if the LP
        failed.
    """
    # current worst auto‑convolution entry
    M = float(np.max(np.convolve(f, f)))
    A = _conv_matrix(f)

    try:
        import cvxpy as cp
    except Exception:                 # pragma: no cover
        return None, None

    n = f.shape[0]
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is usually fastest for dense small‑to‑medium LPs.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=8000,
                   abstol=1e-6, reltol=1e-6)
    except Exception:                 # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=15000)

    if g.value is None:
        return None, None

    g0 = np.maximum(g.value, 0.0)
    sum_g = float(np.sum(g0))
    if sum_g <= 1e-12:
        return None, None

    # rescale to keep the ℓ¹‑norm fixed
    g_rescaled = g0 * (target_sum / sum_g)

    # convex combination step
    f_new = (1.0 - mix) * f + mix * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    new_score = _score(f_new)
    return f_new, new_score


def _simulated_annealing(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple SA optimiser that perturbs a single coordinate at a time.
    The schedule cools from `temp_start` to `temp_end` over the remaining
    time budget.
    """
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.monotonic()
    temp_start = 1.0
    temp_end = 1e-4

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        temp = temp_start * (temp_end / temp_start) ** frac

        # propose a change to a random index
        idx = rng.integers(len(cur_seq))
        delta = rng.normal(scale=0.5) * temp * 5.0
        cand = cur_seq.copy()
        cand[idx] = float(np.clip(cand[idx] + delta, 0.0, 1000.0))

        cand_score = _score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA] new best score {best_score:.6f}")

        # occasional tiny random re‑shuffle to avoid getting stuck
        if rng.random() < 0.001:
            pert = np.maximum(cur_seq + rng.normal(scale=0.1, size=len(cur_seq)),
                              0.0)
            cur_seq = np.clip(pert, 0.0, 1000.0)
            cur_score = _score(cur_seq)

    return best_seq, best_score

    if best_seq is None:
