The typical algorithm:

def simplex_projection(v, s):
    # v is 1D np array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (s - cssv) / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho_star = rho[-1]
        theta = (cssv[rho_star] - s) / (rho_star + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate
=================

Search for a non‑negative step‑function (a list of heights) that minimizes
``evaluate_sequence``.  The objective after fixing the L1‑norm to
sqrt(2·n) is simply the maximum value of the self‑convolution of the
vector, i.e. ``max(conv)``.  The program:

1.  Determines the sequence length ``n`` from the global ``height_sequence_1``
    (fallback 256) and fixes the target L1‑norm ``S = sqrt(2·n)``.
2.  Builds a few deterministic start‑vectors:
    * a geometric progression with an optimised ratio ``r`` (golden‑section
      search);
    * a uniform vector;
    * a random sparse vector.
3.  Keeps the best of those as the current solution.
4.  Enters a hybrid optimisation loop (≈ 1000 s):
    * **Mode 0** – projected sub‑gradient descent on the index that attains
      the maximal convolution value.
    * **Mode 1** – multiplicative perturbation of a random block (as in the
      reference code).
    * **Mode 2** – transfer a random fraction of mass between two positions.
    * **Mode 3** – replace the whole vector by a fresh sparse random candidate.
    * **Mode 4** – generate a power‑law candidate with a random exponent.
    After each mutation the fast FFT‑based scorer evaluates the candidate;
    improvements are kept as the new best vector.
5.  When the time budget expires the best vector is returned as a plain
    Python list.  A final call to the official ``evaluate_sequence`` guarantees
    that all security checks are applied.

The heavy part – the convolution – is performed with an FFT, turning the
O(n²) naïve convolution into O(n log n) and allowing many more iterations
within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast convolution (FFT based) and fast scorer (no security checks)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a (self‑convolution) using real FFT.

    The result has length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv[:size]


def _fast_score(seq: np.ndarray, target_sum: float) -> float:
    """Score identical to ``evaluate_sequence`` but without checks.

    ``seq`` is assumed to be a 1‑D float array; it is clipped to [0,1000],
    its L1‑norm is *target_sum* (ensured by the caller).  Returns the
    value 2·n·max_b / (target_sum²) which, with the chosen normalisation,
    equals the maximal convolution coefficient.
    """
    arr = np.clip(seq, 0.0, 1000.0)
    if arr.sum() < 0.01:
        return float('inf')
    conv = _conv_self_fft(arr)
    maxb = float(conv.max())
    n = arr.shape[0]
    return float(2.0 * n * maxb / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum x = s}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the scaled simplex."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_star = rho[-1]
        theta = cssv[rho_star] / (rho_star + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _geometric_candidate_r(n: int, target_sum: float, r: float) -> np.ndarray:
    """Geometric progression a_i = r**i, scaled to have L1 = target_sum."""
    if r <= 0.0:
        raw = np.zeros(n, dtype=np.float64)
        raw[0] = 1.0
    else:
        idx = np.arange(n, dtype=np.float64)
        raw = np.power(r, idx)
    raw_sum = raw.sum()
    if raw_sum == 0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    return raw * (target_sum / raw_sum)


def _optimal_geometric_candidate(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Golden‑section search for the ratio r∈(0,1) that minimises the score."""
    lo, hi = 0.0, 0.999999  # r cannot be exactly 1 (that would be uniform)
    phi = (1 + 5 ** 0.5) / 2  # golden ratio
    for _ in range(30):
        r1 = hi - (hi - lo) / phi
        r2 = lo + (hi - lo) / phi
        cand1 = _geometric_candidate_r(n, target_sum, r1)
        cand2 = _geometric_candidate_r(n, target_sum, r2)
        s1 = _fast_score(cand1, target_sum)
        s2 = _fast_score(cand2, target_sum)
        if s1 < s2:
            hi = r2
        else:
            lo = r1
    r_opt = (lo + hi) * 0.5
    return _geometric_candidate_r(n, target_sum, r_opt)


def _sparse_random_candidate(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Random sparse vector (5‑%–25 % density) scaled to the target L1‑norm."""
    k = rng.integers(max(1, int(n * 0.05)), max(2, int(n * 0.25) + 1))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0:
        cand *= (target_sum / total)
    else:
        cand = np.full(n, target_sum / n, dtype=np.float64)
    return cand


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray) -> tuple[np.ndarray, int, float]:
    """Return (gradient, argmax_index, max_convolution) for the current vector."""
    conv = _conv_self_fft(a)
    k = int(conv.argmax())
    max_val = float(conv[k])
    n = a.shape[0]
    grad = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad, k, max_val


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function with a low ``evaluate_sequence`` score.
    The routine runs until ``budget_s`` seconds have elapsed and returns
    the best list of heights found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # keep a small safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(base.size)
    except Exception:
        n = 256
    target_sum = np.sqrt(2.0 * n)          # the normalisation used in the literature

    # ------------------------------------------------------------------
    # Initialise a small pool of deterministic candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1) Optimised geometric progression
    cand_geo = _optimal_geometric_candidate(n, target_sum, rng)
    candidates.append(cand_geo)

    # 2) Uniform vector (baseline)
    cand_uni = np.full(n, target_sum / n, dtype=np.float64)
    candidates.append(cand_uni)

    # 3) Random sparse start
    cand_sparse = _sparse_random_candidate(n, target_sum, rng)
    candidates.append(cand_sparse)

    # Pick the best of the initial pool
    best = candidates[0].copy()
    best_score = _fast_score(best, target_sum)
    for cand in candidates[1:]:
        sc = _fast_score(cand, target_sum)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    print(f"[init] n={n}, target_sum={target_sum:.3f}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop (Hybrid: sub‑gradient + random mutations)
    # ------------------------------------------------------------------
    cur = best.copy()
    it = 0
    step0 = target_sum * 0.5                 # initial step size for sub‑gradient

    while time.monotonic() < deadline:
        it += 1
        mode = rng.integers(0, 5)            # 0…4 different mutation strategies

        if mode == 0:
            # ---- Projected sub‑gradient step on the max‑convolution entry ----
            grad, max_idx, max_val = _max_conv_gradient(cur)
            eta = step0 / np.sqrt(it)       # diminishing step size
            cand = cur - eta * grad
            cand = _simplex_projection(cand, target_sum)

        elif mode == 1:
            # ---- Block multiplicative perturbation (as in the reference code) ----
            cand = cur.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.12))
            cand[start:start + width] *= factor
            if rng.random() < 0.2:           # occasional symmetrisation
                cand = 0.5 * (cand + cand[::-1])
            cand = _simplex_projection(cand, target_sum)

        elif mode == 2:
            # ---- Transfer a small fraction of mass between two indices ----
            cand = cur.copy()
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * 0.4
            delta = eps * min(cand[i], cand[j])
            if cand[i] > cand[j]:
                cand[i] -= delta
                cand[j] += delta
            else:
                cand[j] -= delta
                cand[i] += delta
            # Projection is not needed – the L1 sum is unchanged and negativity
            # cannot happen because delta ≤ min(a_i,a_j).

        elif mode == 3:
            # ---- Fresh sparse random candidate (restart) ----
            cand = _sparse_random_candidate(n, target_sum, rng)

        else:
            # ---- Power‑law shaped candidate with a random exponent ----
            p = rng.uniform(0.2, 2.0)
            cand = np.arange(1, n + 1, dtype=np.float64) ** (-p)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand = cand * (target_sum / cand_sum)
            else:
                cand = np.full(n, target_sum / n, dtype=np.float64)

        # ------------------------------------------------------------------
        # Evaluate the new candidate (fast scorer)
        # ------------------------------------------------------------------
        cur_score = _fast_score(cand, target_sum)

        # Keep the global best
        if cur_score < best_score:
            best = cand.copy()
            best_score = cur_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:5.1f}s] new best = {best_score:.6f}")

        # Choose the next current vector
        # With 20 % probability we jump back to the current best to avoid drift.
        if rng.random() < 0.2:
            cur = best.copy()
        else:
            cur = cand

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    final = [float(x) for x in best.tolist()]
    # Let the official evaluator apply its security wrapper and final clipping.
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f}")
    return final
