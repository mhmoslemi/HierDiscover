def _flatten_move_advanced(seq, rng):
    cand = seq.copy()
    n = cand.size
    # compute convolution
    size = _next_pow_two(2*n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n - 1]
    max_idx = int(conv.argmax())
    # compute contributions
    i_min = max(0, max_idx - (n - 1))
    i_max = min(n - 1, max_idx)
    i_vals = np.arange(i_min, i_max + 1)
    partners = max_idx - i_vals
    contrib = cand[i_vals] * cand[partners]
    # get source indices sorted by contribution descending
    src_order = np.argsort(-contrib)
    # set number of transfers
    num_transfers = min(3, src_order.size)
    for k in range(num_transfers):
        src_i = i_vals[src_order[k]]
        # find dst index with low contribution (prefer zeros)
        # compute contributions for all indices or sample
        # pick random index not partner of src_i
        # ensure cand[src_i] > 0
        if cand[src_i] <= 0:
            continue
        # randomly choose destination
        cand_dst_candidates = np.delete(np.arange(n), [src_i, max_idx - src_i])
        if cand_dst_candidates.size == 0:
            continue
        dst_i = int(rng.choice(cand_dst_candidates))
        # compute transfer fraction
        frac = float(0.05 + 0.15 * rng.random())
        delta = cand[src_i] * frac
        cand[src_i] -= delta
        cand[dst_i] += delta
    # clip
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _lp_step(seq, rng, t=0.2):
    n = seq.size
    # compute max conv
    max_conv = _conv_max(seq)
    M = max_conv
    # Set up linear program: maximize sum(b) subject to A @ b <= M, b >=0.
    # A is (2n-1) x n, with A[k,i] = seq[i] if 0 <= k-i < n else 0.
    # Build sparse matrix using scipy.sparse
    from scipy.sparse import csr_matrix
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    # This row vector may exceed 2n-2 but we can filter
    # Actually need to compute each column i contributes to rows i..i+n-1
    # We can pre-allocate arrays of length n*n.
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    data = np.empty(n*n, dtype=np.float64)
    idx = 0
    for i in range(n):
        rows[idx:idx+n] = i + np.arange(n)
        cols[idx:idx+n] = i
        data[idx:idx+n] = seq[i]
        idx += n
    # filter rows <=2n-2
    mask = rows < 2*n - 1
    rows = rows[mask]
    cols = cols[mask]
    data = data[mask]
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Solve LP
    from scipy.optimize import linprog
    c = -np.ones(n)  # maximize sum(b) => minimize -sum(b)
    # bounds for b: (0, None)
    bounds = [(0, None)] * n
    # inequality: A @ b <= M
    # Convert to standard form for linprog, with left side <=
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None
    # Normalize to sum = 1 (same as seq sum)
    g = b / sum_b
    # optionally mix
    cand = (1 - t) * seq + t * g
    # clip
    cand = np.clip(cand, 0.0, 1000.0)
    # ensure sum ~1
    cand = cand / cand.sum()
    return cand
```

"""
Stochastic Simulated‑Annealing optimiser for the step‑function score.

The algorithm works with a single normalised candidate (∑a_i = 1) and
repeatedly applies cheap stochastic perturbations:

* multiplicative scaling of a random block,
* pairwise mass exchange,
* small Gaussian noise on a sub‑block,
* “flatten” moves that explicitly reduce the contribution of the
  currently largest convolution entry,
* smoothing by a tiny moving‑average kernel,
* random permutation of a sub‑block,
* occasional heavy re‑distribution of mass (reset move).

Each generated sequence is normalised, its maximum auto‑convolution is
estimated with a fast FFT‑based routine, and the move is accepted by a
temperature‑controlled simulated‑annealing rule.  Whenever the fast
estimate beats the best exact score found so far, the authoritative
``evaluate_sequence`` function is called.  The search stops after the
supplied time budget and returns the best sequence discovered.

All helper functions are defined at top‑level – no closures or lambdas.
"""

import numpy as np
import time
import math
from typing import List

# ----------------------------------------------------------------------
# FFT‑based helpers (fast upper bound)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the linear convolution seq*seq (FFT implementation)."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(conv[: 2 * n - 1].max())


def _score_estimate(seq: np.ndarray) -> float:
    """Fast proxy for the true objective (uses the FFT‑based max)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Perturbation operators
# ----------------------------------------------------------------------


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative scaling of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring to encourage symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from i → j."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)  # ≤20 % of source mass
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add modest zero‑mean Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the contribution of the worst convolution index.
    Performs up to three relocations of mass from high‑contributing
    positions to low‑contributing ones.
    """
    cand = seq.copy()
    n = cand.size

    # full convolution to locate the offending index
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())

    # indices that actually take part in this sum
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]

    # order sources by decreasing contribution
    src_order = np.argsort(-contrib)

    # up to three transfers
    for s in src_order[:3]:
        src = idx[s]
        if cand[src] <= 0.0:
            continue
        # choose a destination far from the offending pair
        forbid = {src, partner[s]}
        dst_candidates = np.setdiff1d(np.arange(n), list(forbid))
        if dst_candidates.size == 0:
            continue
        dst = int(rng.choice(dst_candidates))
        # transfer a modest‑to‑large fraction (5 % … 35 %)
        frac = float(0.05 + 0.30 * rng.random())
        delta = cand[src] * frac
        cand[src] -= delta
        cand[dst] += delta

    # occasional symmetrisation
    if rng.random() < 0.25:
        cand = 0.5 * (cand + cand[::-1])

    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _smooth_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a tiny moving‑average filter (kernel [0.25,0.5,0.25])."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    # same mode keeps length unchanged
    smooth = np.convolve(seq, kernel, mode="same")
    np.clip(smooth, 0.0, 1000.0, out=smooth)
    return smooth


def _permute_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Randomly shuffle a contiguous sub‑block."""
    cand = seq.copy()
    n = cand.size
    block_len = int(rng.integers(max(2, n // 100), max(5, n // 20)))
    start = int(rng.integers(0, n - block_len + 1))
    idx = np.arange(start, start + block_len)
    rng.shuffle(idx)
    cand[start:start + block_len] = cand[idx]
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _reset_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Zero out a small random set of entries and redistribute its mass
    uniformly over the remaining positions – a strong exploratory jump.
    """
    cand = seq.copy()
    n = cand.size
    # up to 5 % of the indices are cleared
    num = int(rng.integers(1, max(2, n // 20)))
    idx = rng.choice(n, size=num, replace=False)
    lost = cand[idx].sum()
    cand[idx] = 0.0
    if lost > 0.0:
        cand += lost / (n - num)
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> List[float]:
    """
    Simulated‑annealing optimiser for the step‑function evaluation problem.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the caller subtracts a small safety margin).

    Returns
    -------
    list[float]
        The best non‑negative sequence found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # a few seconds are subtracted as a safety margin
    deadline = start_time + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start: try a few analytic families and the supplied global.
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0:
            raise Exception
    except Exception:
        # fallback to a power‑law shape (p≈0.45 works well)
        n0 = 2000
        idx = np.arange(1, n0 + 1, dtype=np.float64)
        base_seq = np.power(idx, -0.45)

    base_seq = np.clip(base_seq, 0.0, 1000.0)
    base_seq = base_seq / float(base_seq.sum())  # normalise to unit sum

    best_seq = base_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    best_est = _score_estimate(best_seq)

    # try a handful of power‑law exponents
    n = base_seq.size
    for p in np.linspace(0.30, 0.60, 9):
        cand = np.power(np.arange(1, n + 1, dtype=np.float64), -p)
        cand = cand / cand.sum()
        est = _score_estimate(cand)
        if est < best_est - 1e-12:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_seq, best_score, best_est = cand, exact, est
                print(f"[warm‑start] p={p:.3f} → new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Simulated‑annealing loop
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_est = best_est

    it = 0
    report_interval = 5000

    # temperature schedule (high → low)
    max_temp = 0.02   # allows occasional uphill moves early on
    min_temp = 1e-6

    move_fns = [
        _make_block_perturbation,
        _perturb_swap,
        _perturb_noise,
        _flatten_move,
        _smooth_move,
        _permute_move,
        _reset_move,
    ]

    while time.monotonic() < deadline:
        # temperature linearly annealed
        elapsed = time.monotonic() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max_temp * (1.0 - frac) + min_temp * frac

        move = rng.choice(move_fns)
        cand = move(cur_seq, rng)

        # normalise to unit sum (scaling does not affect the true objective)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        est = _score_estimate(cand)
        delta = est - cur_est

        # Metropolis acceptance rule
        if delta < 0.0 or rng.random() < math.exp(-delta / temp):
            cur_seq = cand
            cur_est = est

            # If we have a chance to beat the global best, evaluate exactly
            if est < best_est - 1e-12:
                exact = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best_seq = cand.copy()
                    best_score = exact
                    best_est = est
                    print(f"[{it:07d}] NEW BEST exact {best_score:.6f} (est {est:.6f})")

        # periodic status report
        if it % report_interval == 0 and it > 0:
            print(f"[{it:07d}] best = {best_score:.6f} (est {best_est:.6f})")
        it += 1

    print(f"Finished after {it} iterations – best exact score = {best_score:.6f}")
    # Convert to plain Python list of floats for the caller
    return [float(x) for x in best_seq]
