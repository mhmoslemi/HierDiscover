def _fast_score(seq):
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return np.inf
    # Use FFT: compute conv length n*2-1.
    n = arr.size
    m = 2*n - 1
    # Compute FFT length as next power of two for efficiency
    fft_len = 1 << (m-1).bit_length()
    fa = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(fa*fa, n=fft_len)[:m]
    maxb = conv.max()
    return float(2.0 * n * maxb / (s * s))

"""
Propose a step‑function with a very low `evaluate_sequence` score.

Algorithm
---------
* The problem is scale‑invariant, therefore we fix the L¹‑norm to
  sqrt(2·n) (the same normalisation used in the reference paper).
* A family of symmetric power‑law shapes
      a_i ∝ (min(i, n‑1‑i) + offset)^{-p}
  contains the analytically known near‑optimal shape
  p = ½, offset = 1 (a_i ≈ 1/√i) which yields a score ≈ π/2 ≈ 1.5708.
  Small adjustments of the exponent *p* and the offset allow the
  convolution to become even flatter, pushing the score below the
  π/2‑bound.
* We keep a small population (≈10) of candidates and evolve it with
  several mutation operators:
      – block multiplicative perturbations,
      – additive Gaussian noise in log‑space,
      – transfer of mass between two positions,
      – fresh random sparse vectors,
      – regeneration of a shape candidate with new (p, offset).
* A simple simulated‑annealing schedule (temperature decreasing linearly)
  is used to occasionally accept a worse candidate, helping the search
  to escape local minima.
* All evaluations are performed with a fast FFT‑based scorer that
  mimics the official `evaluate_sequence` (clipping, sum‑check, etc.).
  The final best vector is returned as a plain Python list; the outer
  harness will call the official evaluator once more for safety.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast scorer – identical to the official evaluator (but without type checks)
# ----------------------------------------------------------------------
def _fast_score(arr: np.ndarray) -> float:
    """Return the same value that evaluate_sequence would compute."""
    a = np.clip(arr, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.shape[0]
    # FFT based convolution (real → real). Length = 2n‑1.
    m = 2 * n - 1
    fft_len = 1 << (m - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:m]
    # Small negative round‑off can appear; clamp to zero.
    conv = np.maximum(conv, 0.0)
    maxb = conv.max()
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Candidate factories
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """Geometric progression a_i = r**i, scaled to the target L1 sum."""
    # Solve exp(t) = 1 + 2 t  (t≈1.256) and set δ = t/n, r = 1‑δ.
    t = 1.0
    for _ in range(20):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    return raw * (sum_target / raw_sum)


def _sparse_random_candidate(n: int, sum_target: float, rng: np.random.Generator) -> np.ndarray:
    """Few random non‑zero entries (5‑25 % sparsity)."""
    k = rng.integers(int(n * 0.05), int(n * 0.25) + 1)
    if k <= 0:
        return np.full(n, sum_target / n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    total = cand.sum()
    if total > 0.0:
        cand *= (sum_target / total)
    return cand


def _sym_powerlaw_candidate(
    n: int,
    sum_target: float,
    p: float,
    offset: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Symmetric power‑law shape:
        a_i ∝ (min(i, n‑1‑i) + offset)^(-p)
    """
    idx = np.arange(n, dtype=np.float64)
    d = np.minimum(idx, n - 1 - idx)
    raw = np.power(d + offset, -p, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    return raw * (sum_target / raw_sum)


def _powerlaw_candidate(
    n: int,
    sum_target: float,
    p: float,
    offset: float,
) -> np.ndarray:
    """One‑sided power‑law a_i ∝ (i+offset)^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(idx + offset, -p, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    return raw * (sum_target / raw_sum)


# ----------------------------------------------------------------------
# Mutation operators
# ----------------------------------------------------------------------
def _block_mul(parent: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = parent.size
    cand = parent.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:start + width] *= factor
    # occasional symmetrisation
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])
    return cand


def _log_noise(parent: np.ndarray, rng: np.random.Generator, sigma: float) -> np.ndarray:
    """Additive Gaussian noise in log‑space."""
    # avoid log(0)
    safe = np.maximum(parent, 1e-12)
    logs = np.log(safe)
    logs += rng.normal(0.0, sigma, size=parent.shape)
    cand = np.exp(logs)
    return cand


def _transfer_mass(parent: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a fraction of the smaller entry from i to j."""
    cand = parent.copy()
    n = cand.size
    i, j = rng.choice(n, size=2, replace=False)
    eps = rng.random() * 0.4
    delta = eps * min(cand[i], cand[j])
    if cand[i] > cand[j]:
        cand[i] -= delta
        cand[j] += delta
    else:
        cand[j] -= delta
        cand[i] += delta
    return cand


def _crossover(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Simple arithmetic crossover (average with a random mask)."""
    mask = rng.random(p1.shape) < 0.5
    child = np.where(mask, p1, p2)
    return child.astype(np.float64)


def _shape_mutation(n: int, sum_target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a fresh symmetric power‑law candidate with random parameters."""
    p = rng.uniform(0.4, 0.6)            # exponent near the critical 0.5
    offset = rng.uniform(1.0, max(2.0, n * 0.05))  # offset ≥ 1 to keep finite
    return _sym_powerlaw_candidate(n, sum_target, p, offset, rng)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The routine runs for roughly the supplied time budget
    (default ≈ 1100 s, leaving a few seconds margin) and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (fallback to 256 if the global is unavailable)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)   # safe fallback
    n = int(base.size)
    sum_target = np.sqrt(2.0 * n)               # normalisation used in the paper

    # ------------------------------------------------------------------
    # Initialise a small population with diverse seeds
    # ------------------------------------------------------------------
    POP_SIZE = 12
    population = []          # list of (vector, score)
    def add_to_pop(vec, scr):
        nonlocal population
        if np.isnan(scr) or np.isinf(scr):
            return
        population.append((vec, scr))
        population.sort(key=lambda x: x[1])
        if len(population) > POP_SIZE:
            population = population[:POP_SIZE]

    # deterministic seeds
    cand_geometric = _geometric_candidate(n, sum_target)
    add_to_pop(cand_geometric, _fast_score(cand_geometric))

    cand_rev = cand_geometric[::-1].copy()
    add_to_pop(cand_rev, _fast_score(cand_rev))

    # symmetric power‑law with p≈0.5, offset=1 (the analytical starter)
    cand_sym = _sym_powerlaw_candidate(n, sum_target, p=0.5, offset=1.0, rng=rng)
    add_to_pop(cand_sym, _fast_score(cand_sym))

    # random sparse vectors
    for _ in range(3):
        cand_sp = _sparse_random_candidate(n, sum_target, rng)
        add_to_pop(cand_sp, _fast_score(cand_sp))

    # a few random one‑sided power‑law vectors
    for _ in range(2):
        p = rng.uniform(0.3, 1.2)
        offset = rng.uniform(1.0, 10.0)
        cand_pl = _powerlaw_candidate(n, sum_target, p, offset)
        add_to_pop(cand_pl, _fast_score(cand_pl))

    best_vec, best_score = population[0]
    print(f"[init] n={n}, target‑sum={sum_target:.3f}, pop‑size={len(population)}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop (simulated‑annealing / evolutionary style)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)   # keep 3 s margin
    total_time = deadline - start_time

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # Linear temperature schedule: high at start, near zero at the end.
        elapsed = time.monotonic() - start_time
        T = max(0.001, 0.2 * (1.0 - elapsed / total_time))

        # Pick a parent (biased towards the best ones)
        # simple roulette based on rank: weight = POP_SIZE - rank
        ranks = np.arange(len(population))
        probs = (len(population) - ranks) / np.sum(len(population) - ranks)
        parent_idx = rng.choice(len(population), p=probs)
        parent_vec, _ = population[parent_idx]

        # Choose a mutation strategy
        mode = rng.integers(0, 7)

        if mode == 0:
            cand = _block_mul(parent_vec, rng)

        elif mode == 1:
            # Gaussian log‑noise (sigma adapts with temperature)
            sigma = 0.06 + 0.15 * T          # larger sigma early, finer later
            cand = _log_noise(parent_vec, rng, sigma)

        elif mode == 2:
            cand = _transfer_mass(parent_vec, rng)

        elif mode == 3:
            # Crossover of two random parents
            i, j = rng.choice(len(population), size=2, replace=False)
            cand = _crossover(population[i][0], population[j][0], rng)

        elif mode == 4:
            # Fresh symmetric power‑law shape
            cand = _shape_mutation(n, sum_target, rng)

        elif mode == 5:
            # Replace entirely by a new sparse random candidate
            cand = _sparse_random_candidate(n, sum_target, rng)

        else:
            # Simple additive Gaussian noise on the raw values (not log‑space)
            cand = parent_vec.copy()
            cand += rng.normal(0.0, 0.02 * sum_target, size=n)

        # Clip to allowed interval (the fast scorer does the same)
        cand = np.clip(cand, 0.0, 1000.0)

        # Fast evaluation
        cur_score = _fast_score(cand)

        # Simulated‑annealing acceptance (keeps the search mobile)
        if cur_score < best_score or rng.random() < np.exp(-(cur_score - best_score) / T):
            # keep in population (replace worst if better)
            add_to_pop(cand, cur_score)
            if cur_score < best_score:
                best_vec, best_score = cand, cur_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:5.1f}s] new best = {best_score:.6f}")

        # Occasionally (every few seconds) prune the population to retain diversity
        # (already done inside add_to_pop).

    # ------------------------------------------------------------------
    # Return the best sequence (as a plain list)
    # ------------------------------------------------------------------
    final = [float(x) for x in best_vec.tolist()]
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f} after {iteration} iterations")
    return final
