    # hyperparameters

    alpha0 = 0.5     # initial step size factor
    decay = 0.001    # step size decay factor

    while time.monotonic() < deadline:
        iteration += 1

        max_conv = conv.max()

        # compute active set

        # Actually we want those near max: conv >= max_conv - epsilon*max_conv (or absolute)

        # compute gradient (subgradient)

        # compute step size
        step = alpha0 / (1.0 + decay * iteration)

        # perform gradient update
        cand = best_seq - step * grad
        # ensure non-negative
        cand = np.maximum(cand, 0.0)

        # enforce isotonic decreasing (optional)
        if rng.random() < 0.7:   # 70% of steps apply isotonic projection
            cand = _isotonic_decreasing(cand)

        # renormalize sum to 1 (or keep scaling)
        s = cand.sum()
        if s < 1e-8:

            cand = best_seq * (1 + 0.01 * rng.normal(size=best_seq.shape))
            cand = np.maximum(cand, 0.0)

        # optional random perturbation to escape:
        if rng.random() < 0.05:
            # add small Gaussian noise
            noise_scale = 0.02 * cand.mean()
            cand += rng.normal(0.0, noise_scale, size=n)

def _isotonic_decreasing(seq):
    # isotonic decreasing using PAVA on reversed seq
    y = seq[::-1].copy()
    n = y.shape[0]
    # Initialize blocks
    avg = y.tolist()
    weight = [1] * n
    i = 0
    while i < len(avg) - 1:
        if avg[i] <= avg[i + 1] + 1e-12:
            i += 1
        else:
            # merge
            total = avg[i] * weight[i] + avg[i + 1] * weight[i + 1]
            w = weight[i] + weight[i + 1]
            new_avg = total / w
            avg[i] = new_avg
            weight[i] = w
            # delete i+1 entry
            del avg[i + 1]
            del weight[i + 1]
            i = max(i - 1, 0)
    # expand
    out = np.empty(n, dtype=np.float64)
    idx = 0
    for a_val, w in zip(avg, weight):
        out[idx: idx + w] = a_val
        idx += w
    return out[::-1]

"""
Stochastic optimizer for the step‑function evaluation problem.

The algorithm combines two complementary ideas:

1. **Active‑set sub‑gradient descent** –  
   For the current sequence we compute the auto‑convolution via FFT, locate
   the (few) indices where the convolution is close to its maximum and
   build a sub‑gradient of the objective  (the maximum of the convolution).  
   A small step opposite to this gradient reduces the offending peak.
   Afterwards the sequence is mirrored (enforcing symmetry), projected onto the
   non‑negative orthant, optionally isotonic‑regressed (non‑increasing) and
   normalised (the objective is scale‑invariant).

2. **Occasional “flatten” moves** –  
   A specialised move that directly transfers mass from the entry that
   contributes most to the worst convolution index to a far‑away entry.
   This helps escaping local plateaus.

Both moves are tried at random; a cheap FFT‑based upper bound is computed for
each candidate and the expensive `evaluate_sequence` function is called only
when the bound promises an improvement.  The loop runs until the supplied time
budget is exhausted and returns the best sequence found.

The implementation is fully self‑contained, uses only NumPy and the already
available `evaluate_sequence`, and respects the security checks of the
evaluation function.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of `seq` with itself (full length 2·n‑1) using FFT."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _conv_max_fft(seq: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution (computed via FFT)."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(conv[: 2 * n - 1].max())


def _score_estimate(seq: np.ndarray) -> float:
    """
    Fast proxy for `evaluate_sequence`.  Uses the FFT max‑convolution.
    The objective is scale‑invariant; we keep the true sum to avoid the
    “sum_a < 0.01” guard.
    """
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max_fft(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _subgradient(seq: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the objective at `seq`.

    `active` contains the convolution indices k that are (almost) maximal.
    For each such k the gradient of conv[k] w.r.t. a_i is:
        a_{k-i}      if i = k-i  (i.e. k even and i = k/2)
        2·a_{k-i}    otherwise
    The returned gradient is the average over the active set.
    """
    n = seq.size
    grad = np.zeros_like(seq)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        partner = k - idx
        same = idx == partner
        # diagonal contribution (i == partner)
        if np.any(same):
            grad[idx[same]] += seq[idx[same]]
        # off‑diagonal contributions (i ≠ partner)
        off = idx[~same]
        if off.size:
            grad[off] += 2.0 * seq[partner[~same]]
    if active.size:
        grad /= float(active.size)
    return grad


# ----------------------------------------------------------------------
# Isotonic regression (non‑increasing) – PAVA implementation
# ----------------------------------------------------------------------


def _isotonic_decreasing(seq: np.ndarray) -> np.ndarray:
    """
    Return the isotonic regression of `seq` under the constraint
    x[0] ≥ x[1] ≥ … ≥ x[n‑1] (L2 optimal).
    """
    # Work on the reversed array to obtain a non‑decreasing problem
    y = seq[::-1].copy()
    n = y.size
    # Initialise blocks
    block_vals = y.tolist()
    block_weights = [1] * n
    i = 0
    while i < len(block_vals) - 1:
        if block_vals[i] <= block_vals[i + 1] + 1e-12:
            i += 1
        else:
            # Merge blocks i and i+1
            total = block_vals[i] * block_weights[i] + block_vals[i + 1] * block_weights[i + 1]
            w = block_weights[i] + block_weights[i + 1]
            avg = total / w
            block_vals[i] = avg
            block_weights[i] = w
            del block_vals[i + 1]
            del block_weights[i + 1]
            i = max(i - 1, 0)

    # Expand back to full length
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val, w in zip(block_vals, block_weights):
        out[pos: pos + w] = val
        pos += w
    return out[::-1]  # re‑reverse to original order


# ----------------------------------------------------------------------
# Flatten move – directly reduces the currently worst convolution entry
# ----------------------------------------------------------------------


def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a modest fraction of mass from the index that contributes most
    to the worst convolution entry to a far‑away index."""
    cand = seq.copy()
    n = cand.size

    # Full convolution and worst index
    conv = _conv_full_fft(cand)
    k_max = int(conv.argmax())

    # Indices that actually participate in this sum
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]

    # Source = index with largest contribution
    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand

    # Destination = index far from the offending pair, with smallest contribution
    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))

    # Transfer a modest fraction (5‑20 %)
    delta = cand[src] * float(0.05 + 0.15 * rng.random())
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Gradient‑step helper
# ----------------------------------------------------------------------


def _gradient_step(seq: np.ndarray, rng: np.random.Generator,
                   it: int, eps: float = 1e-3,
                   step0: float = 0.12, decay: float = 2e-4) -> np.ndarray:
    """
    Perform one sub‑gradient descent step.
    - `eps` controls the width of the active set (relative to the max).
    - `step0` and `decay` set a decreasing step size: step = step0 / (1+decay·it).
    """
    n = seq.size
    conv = _conv_full_fft(seq)
    max_conv = conv.max()
    # Active set: indices whose value is within eps·max of the peak
    thresh = max_conv - eps * max_conv
    active = np.where(conv >= thresh)[0]

    grad = _subgradient(seq, active)

    # Diminishing step size
    step = step0 / (1.0 + decay * it)

    cand = seq - step * grad
    cand = np.maximum(cand, 0.0)

    # Enforce symmetry (helps to keep the convolution flat)
    cand = 0.5 * (cand + cand[::-1])

    # Occasionally apply isotonic regression on the whole vector
    if rng.random() < 0.45:
        cand = _isotonic_decreasing(cand)

    # Normalise (scale does not affect the score)
    s = cand.sum()
    if s > 0:
        cand /= s
    return np.clip(cand, 0.0, 1000.0)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------

    """

    # ------------------------------------------------------------------

    if "height_sequence_1" in globals():

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)  # safety margin
