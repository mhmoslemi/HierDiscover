"""
Optimise a step‑function (non‑negative height sequence) for the metric

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm has three stages:

1.  A fast grid search over the two‑parameter analytic family
        a_i = (i + α)^{-p}
    (the continuous analogue gives a flat auto‑convolution when p≈0.5).
    The best candidate from this search is taken as a starting point.

2.  A long hill‑climbing refinement.
    In each iteration the index that contributes the most to the current
    maximal convolution entry (the “donor”) transfers a small amount of
    mass to the index that contributes the least (the “receiver”).
    The move is kept only if the evaluation score improves.
    Occasional random block perturbations are performed to escape plateaus.

3.  The best sequence found is returned as a plain Python list of floats.

All steps respect the constraints of ``evaluate_sequence`` (non‑negative,
clipped to [0,1000]).  The routine works for any vector length; a modest
length (≈1024) is used so that many hill‑climbing steps fit well inside the
1000 s time budget.
"""

import numpy as np
import time

# ------------------------------------------------------------
# Helper: cheap random block perturbation (used in hill‑climb)
# ------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Return a new candidate that multiplies a random contiguous block of ``base``
    by a log‑normal factor and (with some probability) mirrors the whole vector.
    Values are clipped to the interval [0,1000].
    """
    cand = base.copy()
    n = cand.size
    # block width – at least one element, at most roughly a fifth of the vector
    width = int(rng.integers(1, max(2, n // 5)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetry (helps escape plateaus)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

# ------------------------------------------------------------

    index that contributes most to the current maximal convolution entry to the
    index that contributes least.  Stops when the allotted ``time_budget`` (in
    seconds) expires.  Returns the best sequence encountered and its score.

    cur_score = float(evaluate_sequence(cur.tolist()))
    best_seq = cur.copy()
    best_score = cur_score

        conv = np.convolve(cur, cur)
        k_max = int(np.argmax(conv))

        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i[mask]] * cur[j[mask]]

        donor = int(np.argmax(contrib))
        receiver = int(np.argmin(contrib))

        if donor == receiver or cur[donor] <= 0.0:
            # fall back to a random block move
            cand = make_block_perturbation(cur, rng)
        else:
            # move a modest fraction of the donor’s mass
            delta = float(rng.uniform(0.0, min(0.5, 0.20) * cur[donor]))
            if delta < 1e-12:
                delta = cur[donor] * 0.01
            cand = cur.copy()
            cand[donor] -= delta
            cand[receiver] += delta
            cand = np.clip(cand, 0.0, 1000.0)

        new_score = float(evaluate_sequence(cand.tolist()))

        # keep any strict improvement
        if np.isfinite(new_score) and new_score < cur_score:
            cur = cand
            cur_score = new_score
            if new_score < best_score:
                best_seq = cand.copy()
                best_score = new_score
                print(f"Hill‑climb iter {it}: improved score to {best_score:.6f}")
        else:
            # occasional uphill jump to avoid stagnation
            if rng.random() < 0.001:
                cur = cand
                cur_score = new_score

        # very occasional large random jump
        if it % 7_000 == 0 and rng.random() < 0.02:
            cur = make_block_perturbation(cur, rng)
            cur_score = float(evaluate_sequence(cur.tolist()))
        it += 1

    return best_seq, best_score


# ------------------------------------------------------------
# Main entry point required by the grader
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950, **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights that (hopefully) yields a
    ``evaluate_sequence`` score lower than the current best bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a working length – keep it modest for speed.
    # Use the supplied ``height_sequence_1`` when it is well‑formed,
    # otherwise fall back to a default size (≈1024).
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = min(1024, base.size)
    except Exception:
        n = 1024

    # ------------------------------------------------------------------
    # 1) Grid search over a_i = (i + α)^{-p}
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # keep ~5 s for hill‑climb

    idx = np.arange(n, dtype=np.float64)

    # Coarse grid centred on the analytically promising region
    p_grid = np.linspace(0.45, 0.55, 21)          # 0.45 … 0.55
    alpha_grid = np.linspace(0.05, 1.00, 41)     # 0.05 … 1.00

    best_p = None
    best_alpha = None

    for p in p_grid:
        for alpha in alpha_grid:
            if time.monotonic() >= deadline:
                break
            seq = (idx + alpha) ** (-p)
            # scaling to avoid the hard‑coded 1000‑clip (does not affect the score)
            maxv = np.max(seq)
            if maxv > 1000.0:
                seq *= 1000.0 / maxv
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p = p
                best_alpha = alpha
                print(f"grid p={p:.3f} α={alpha:.3f} → score={best_score:.6f}")

        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Fine‑grained search around the current best parameters.
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        p_fine = np.linspace(max(0.01, best_p - 0.02), min(0.99, best_p + 0.02), 21)
        alpha_fine = np.linspace(max(0.01, best_alpha - 0.4), best_alpha + 0.4, 41)

        for p in p_fine:
            for alpha in alpha_fine:
                if time.monotonic() >= deadline:
                    break
                seq = (idx + alpha) ** (-p)
                maxv = np.max(seq)
                if maxv > 1000.0:
                    seq *= 1000.0 / maxv
                score = float(evaluate_sequence(seq.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_p = p
                    best_alpha = alpha
                    print(f"fine p={p:.4f} α={alpha:.4f} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Hill‑climbing refinement (using the remaining time).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen
        best_seq = np.ones(n, dtype=np.float64)

    # remaining time for the hill‑climb
    remaining = deadline - time.monotonic()
    if remaining > 0.5:            # need a tiny cushion for the final return
        best_seq, best_score = hill_climb(best_seq, remaining)

    print(f"Finished search – best score achieved: {best_score:.6f}")
    # Return a plain Python list as required by the grader
    return [float(x) for x in best_seq]
