from scipy.sparse import csr_matrix

def make_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for s in range(2*n - 1):
        j_start = max(0, s - (n - 1))
        j_end = min(n - 1, s)
        # for j in [j_start...j_end]
        # i = s - j
        # coefficient = f[i]
        for j in range(j_start, j_end + 1):
            i = s - j
            rows.append(s)
            cols.append(j)
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    return A

"""
Propose a step‑function sequence with a low evaluation score.

The search works in two stages.

1️⃣  LP‑interpolation stage (the “active‑set” idea from the editorial):
    • For a fixed f we compute M = max(f⋆f).
    • We solve a linear program:
          maximise   Σ g_j
          subject to Σ_i f_i·g_{s‑i} ≤ M   for all shifts s
                     g_j ≥ 0
      (the constraints are linear in g because f is fixed).
      The solution g tries to push as much ℓ₁‑mass as possible while
      respecting the current peak M.
    • We replace the current iterate by a convex combination
          f←(1–t)·f + t·g   (t≈0.3…0.5) and keep the change only if it
      strictly reduces the peak.  Repeating this quickly drives the
      maximum convolution down.

2️⃣  Stochastic refinement stage:
    • Starting from the best f obtained in stage 1 we perform a cheap
      hill‑climbing with random block perturbations (multiplicative
      log‑normal jitter, optional mirroring and a final ℓ₁‑renormalisation).
    • Only moves that improve the official score are kept.

Both stages are time‑bounded (the total budget is given by *budget_s*).
The code uses only NumPy, SciPy (sparse matrix + HiGHS LP solver) and the
provided ``evaluate_sequence`` function.
"""

def _max_conv_fft(a: np.ndarray) -> float:
    """Fast max of the full convolution a⋆a using FFT."""
    n = a.size
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A·g)[s] = Σ_i f_i·g_{s‑i}
    (i.e. the convolution f⋆g).  The matrix is sparse with n² non‑zeros.
    """
    n = f.size
    rows = []
    cols = []
    data = []
    # For each shift s we add the appropriate f[i] on the columns j = s‑i.
    for s in range(2 * n - 1):
        j_start = max(0, s - (n - 1))
        j_end = min(n - 1, s)
        # i runs from s‑j_end … s‑j_start, i.e. i = s‑j
        for j in range(j_start, j_end + 1):
            rows.append(s)
            cols.append(j)
            data.append(f[s - j])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP: maximise Σ g_j  subject to  f⋆g ≤ M,  g ≥ 0.
    Returns a normalised vector g (ℓ₁‑norm = 1). If the LP fails,
    the original f is returned unchanged.
    """
    n = f.size
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)           # maximise Σg ⇔ minimise –Σg
    bounds = [(0.0, None)] * n

    # HiGHS (default) handles sparse matrices efficiently.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success:
        # LP failed – fall back to the current iterate.
        return f.copy()

    g = res.x
    total = g.sum()
    if total <= 0.0:
        return f.copy()
    return g / total


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator,
                             sigma: float = 0.12,
                             mirror_prob: float = 0.35) -> np.ndarray:
    """
    Return a copy of ``base`` with a cheap non‑negative perturbation:
    a random contiguous block is multiplied by a log‑normal factor,
    optionally mirrored with its reverse.
    """
    cand = base.copy()
    n = cand.size
    # Choose a block width (at least 1, at most n//4)
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative jitter (log‑normal)
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:stop] *= factor
    # optional mirroring – helps keep the shape symmetric
    if rng.random() < mirror_prob:
        cand = 0.5 * (cand + cand[::-1])
    # Enforce the hard limits required by ``evaluate_sequence``.
    cand = np.clip(cand, 0.0, 1000.0)
    # Re‑normalise the ℓ₁‑norm (scale does not affect the score).
    s = cand.sum()
    if s > 0:
        cand /= s
    return cand


def _initial_shapes(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a small collection of plausible starting shapes.
    Each shape is ℓ₁‑normalised.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)

    # 1. Classical arcsine (p = 0.5)
    arcsine = 1.0 / np.sqrt(idx)

    # 2. Slightly flatter power‑law (p = 0.45)
    flatter = idx ** (-0.45)

    # 3. Slightly steeper power‑law (p = 0.55)
    steeper = idx ** (-0.55)

    # 4. Exponential decay (α tuned randomly)
    alpha = rng.uniform(0.5, 2.0) / n
    expo = np.exp(-alpha * (idx - 1))

    # 5. Linear decreasing (clipped to stay >0)
    linear = np.maximum(0.0, 1.0 - (idx - 1) / n)

    shapes = [arcsine, flatter, steeper, expo, linear]
    # Normalise each shape to ℓ₁ = 1
    for i, s in enumerate(shapes):
        total = s.sum()
        if total > 0:
            shapes[i] = s / total
    return shapes

    best_f = None
    best_score = float('inf')
    for shape in _initial_shapes(n, rng):
        # compute fast (the sum is 1 by construction)
        M = _max_conv_fft(shape)
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_f = shape
    if best_f is None:
        # fallback – should never happen
        best_f = np.ones(n, dtype=np.float64) / n
        best_score = 2.0 * n * _max_conv_fft(best_f)

    # Keep track of the globally best sequence.
    global_best = best_f.copy()
    global_best_score = float(evaluate_sequence(global_best.tolist()))
    print(f"[init] length={n}, initial fast score={best_score:.6f}, "
          f"official score={global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) LP‑interpolation phase (time‑limited).
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline_lp = start_time + 0.7 * budget_s   # 70 % time for LP phase
    iteration = 0
    f = best_f.copy()                         # current iterate (ℓ₁‑norm = 1)

    while time.monotonic() < deadline_lp:
        iteration += 1
        M = _max_conv_fft(f)

        # Solve LP to obtain a promising direction g.
        g = _solve_lp_step(f, M)

        # Determine a step size.  Larger steps early, smaller later.
        if iteration < 5:
            t = 0.5
        elif iteration < 15:
            t = 0.3
        else:
            t = 0.15

        # Candidate after mixing.
        f_candidate = (1.0 - t) * f + t * g
        # Re‑normalise explicitly (numerical drift may appear).
        f_candidate /= f_candidate.sum()

        # Evaluate the new peak.
        M_new = _max_conv_fft(f_candidate)
        if M_new < M - 1e-12:               # strict improvement
            f = f_candidate
            # official score check – cheap but ensures we respect clipping etc.
            official = float(evaluate_sequence(f.tolist()))
            if official < global_best_score:
                global_best = f.copy()
                global_best_score = official
                print(f"[LP] iter {iteration:3d}  new official score {official:.6f} "
                      f"t={t:.2f}")
        else:
            # No improvement – try a smaller step size once.
            if t > 0.05:
                t *= 0.5
                f_candidate = (1.0 - t) * f + t * g
                f_candidate /= f_candidate.sum()
                M_new = _max_conv_fft(f_candidate)
                if M_new < M - 1e-12:
                    f = f_candidate
                    official = float(evaluate_sequence(f.tolist()))
                    if official < global_best_score:
                        global_best = f.copy()
                        global_best_score = official
                        print(f"[LP‑backtrack] iter {iteration:3d}  new official score {official:.6f}")
            # If still no progress, skip this iteration.
            # (the loop will try again with a fresh LP solution.)

    # ------------------------------------------------------------------
    # 4) Stochastic local‑search refinement (remaining time).
    # ------------------------------------------------------------------
    deadline = start_time + budget_s
    best_seq = global_best.copy()
    best_score = global_best_score
    width_sigma = 0.12
    mirror_prob = 0.35

    while time.monotonic() < deadline:
        cand = _make_block_perturbation(best_seq, rng,
                                        sigma=width_sigma,
                                        mirror_prob=mirror_prob)
        # Fast evaluation (ℓ₁‑norm stays ≈1)
        cand_sum = cand.sum()
        if cand_sum < 1e-12:
            continue
        cand_score_fast = 2.0 * n * _max_conv_fft(cand) / (cand_sum ** 2)

        if cand_score_fast < best_score - 1e-6:
            # Verify with the official, heavyweight evaluator.
            official = float(evaluate_sequence(cand.tolist()))
            if official < best_score:
                best_seq = cand
                best_score = official
                print(f"[refine] improved to {best_score:.6f} at t={time.monotonic():.2f}")
            # Occasionally shrink the jitter to focus on fine details.
            width_sigma = max(0.04, width_sigma * 0.995)

    # ------------------------------------------------------------------
    # 5) Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"[final] best official score {best_score:.6f} after "
          f"{time.monotonic() - start_time:.2f}s (length={n})")
    return [float(x) for x in best_seq]
