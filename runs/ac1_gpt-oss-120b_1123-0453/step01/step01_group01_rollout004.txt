"""
Propose a better step‑function sequence.

The evaluation score (for a non‑zero sum) equals

        score = 2 * n * max(conv(a, a))

because the sum cancels out.  Hence we only need to minimise the maximal
entry of the linear convolution of the vector with itself.

This optimiser works with a *soft‑max* surrogate of the max‑operator.
For a given sequence `a` we compute its self‑convolution `c`,
form weights `w  ∝  exp(β·c)` (soft‑max), and take the gradient of the
weighted sum `∑ w_k (a*a)_k`.  The gradient can be written as a
cross‑correlation `g_i = 2·∑_k w_k·a_{k-i}` and is obtained via a single
FFT‑based convolution, i.e. in O(n log n) time.

The update `a ← a – η·g` is followed by a projection onto the probability
simplex (non‑negative, sum = 1) – this does not change the score because
the objective is scale‑invariant.  A small isotropic Gaussian noise,
occasional block‑perturbations, and restarts when progress stalls help
the search escape local minima.

The algorithm runs until the supplied time budget (≈ 1000 s) expires and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level (no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_sequence(n: int) -> np.ndarray:
    """
    Return the “arcsine’’ sequence of length n:
        a_i = C(2i,i) / 4^i , then normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself, via FFT."""
    n = x.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    A cheap O(n) non‑negative perturbation used as a diversification step.
    A random contiguous block is multiplied by a log‑normal factor and, with
    probability 0.35, the vector is symmetrised.
    """
    cand = np.array(base, copy=True)
    n = cand.size
    # block width: between n/200 and n/12
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def compute_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (sum(a) = 1 ⇒ score = 2·n·max(conv))."""
    conv = self_convolution(a)
    return float(2.0 * a.shape[0] * conv.max())


def softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate S(a)=∑_k w_k·(a*a)_k
    where w_k ∝ exp(β·conv_k).  The gradient is
        g_i = 2·∑_k w_k·a_{k-i}
    and is computed via a single FFT‑based convolution.
    """
    n = a.shape[0]
    # soft‑max weights, numerically stabilised
    scaled = beta * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w  (linear, not circular)
    w_rev = w[::-1]                     # reverse for correlation
    size = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2
    a_fft = np.fft.rfft(a, size)
    w_rev_fft = np.fft.rfft(w_rev, size)
    corr_full = np.fft.irfft(a_fft * w_rev_fft, n=size)

    mid = len(w_rev) - 1               # = 2·n‑2
    # g_i = 2·corr_full[mid - i]  (take reversed slice)
    g = 2.0 * corr_full[mid : mid - n : -1]
    return g

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises the
    max‑convolution bound.  The routine respects the provided time budget
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Determine the problem size (length of the step function)
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
        # Normalise to the simplex (scale does not affect the score)
        a = np.maximum(base, 0.0)
        if a.sum() == 0.0:
            a = np.ones_like(a) / a.size
        else:
            a = a / a.sum()
    except Exception:
        # Fallback: a reasonably large size that still fits in the budget
        n = 6000
        a = arcsine_sequence(n)

    # --------------------------------------------------------------
    # Initial best solution
    # --------------------------------------------------------------
    best = a.copy()
    best_score = compute_score(best)

    # Occasionally try a random start to diversify the search
    if rng.random() < 0.15:
        random_start = rng.random(n)
        random_start = proj_simplex(random_start)
        rs_score = compute_score(random_start)
        if rs_score < best_score:
            best = random_start
            best_score = rs_score
            a = random_start.copy()

    # --------------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------------
    step = 0.25               # initial learning rate
    step_decay = 0.9997
    beta = 5.0                # soft‑max temperature (large → tighter max)
    beta_factor = 1.015       # multiplicative increase each 200 iters
    noise_std = 0.006
    noise_decay = 0.999
    it = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # 1) current self‑convolution and score
        conv = self_convolution(a)
        cur_score = float(2.0 * n * conv.max())

        # 2) soft‑max gradient
        grad = softmax_gradient(a, conv, beta)

        # 3) stochastic gradient step + tiny isotropic noise
        cand = a - step * grad
        if it % 7 == 0:
            cand += rng.normal(scale=noise_std, size=n)

        # 4) project back onto the probability simplex (keeps sum=1)
        cand = proj_simplex(cand)

        # 5) evaluate candidate (fast score computation)
        cand_score = compute_score(cand)

        # 6) keep the best solution seen so far
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            a = cand.copy()
            no_improve = 0
        else:
            no_improve += 1
            # occasional restart if we are stuck too long
            if no_improve > 3000:
                a = arcsine_sequence(n)               # clean restart
                a = proj_simplex(a)
                no_improve = 0

        # 7) every few hundred iterations try an explicit block perturbation
        if it % 500 == 0 and it > 0:
            pert = make_block_perturbation(best, rng)
            pert = pert / pert.sum()
            pert_score = compute_score(pert)
            if pert_score < best_score:
                best = pert.copy()
                best_score = pert_score
                a = pert.copy()
                no_improve = 0

        # 8) schedule updates
        step *= step_decay
        beta *= beta_factor if (it % 200 == 0 and it > 0) else 1.0
        noise_std *= noise_decay

        it += 1

        # 9) periodic progress report
        if it % 2000 == 0:
            print(
                f"Iter {it:6d} | best_score = {best_score:.6f} | "
                f"step = {step:.6f} | beta = {beta:.2f}"
            )

    print(f"Finished after {it} iterations, best_score = {best_score:.6f}")
    # Return as plain Python floats (the judge expects a list)
    return [float(x) for x in best]
