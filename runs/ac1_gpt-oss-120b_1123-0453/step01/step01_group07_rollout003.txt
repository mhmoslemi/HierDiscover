    # compute grad: grad[i] = 2*seq[idx - i] if 0 <= idx - i < n else 0
    low = max(0, idx - (n - 1))

    grad[i_vals] = 2.0 * seq[j_vals]

    new_seq = np.maximum(new_seq, 0.0)

def solve_lp(a, M, target_sum):

    # Build A
    # each column i contributes a[i] on rows i..i+n-1
    row_idx = np.concatenate([np.arange(i, i + n, dtype=np.int32) for i in range(n)])

"""
Hybrid search for a low‑peak self‑convolution step function.

The algorithm starts from a power‑law kernel (a_i ≈ (i+1)^‑½) which is known
to give a near‑constant convolution.  It then repeatedly applies several
kinds of stochastic moves:

* a gradient step that reduces the current maximum convolution peak,
* a multiplicative perturbation of a random contiguous block,
* a tiny global multiplicative noise.

Every few iterations a linear‑programming re‑balancing is performed:
given the current shape a and its peak M = max(a*a), we solve the LP

    maximise   Σ b_j
    subject to  (a * b)_k ≤ M   for all k,
                b_j ≥ 0.

The optimal b has a larger ℓ¹‑norm, so after scaling it to the required
normalisation we mix it with a.  This is the same “active‑set” idea used in
the reference implementation but is carried out explicitly with CVXPY.

The search runs until the supplied time budget expires (≈ 1000 s) and
returns the best sequence found.  All entries are kept non‑negative and
the sum is constantly normalised to √(2 n), which makes the evaluation
function equal to the peak of the convolution.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def generate_power_sequence(n: int, exponent: float = 0.5,
                            offset: float = 0.0) -> np.ndarray:
    """Return a non‑negative sequence a_i = (i+offset+1)^(-exponent)."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.06)))   # ≈ 6 % change
    new = seq.copy()
    new[start:stop] *= factor
    if rng.random() < 0.1:                       # occasional symmetrisation
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)


def global_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small multiplicative log‑normal noise applied to the whole vector."""
    noise = rng.normal(0.0, 0.02, size=seq.shape)
    factor = np.exp(noise)
    new = seq * factor
    return np.clip(new, 0.0, 1000.0)


def gradient_step(seq: np.ndarray, lr: float) -> np.ndarray:
    """
    One gradient‑descent step aimed at reducing the current convolution peak.
    The sub‑gradient of max_k (a*a)_k is 2·a reflected about the arg‑max index.
    """
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))          # index where the peak occurs
    n = seq.size
    grad = np.zeros_like(seq)

    # indices i where 0 ≤ k‑i < n
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i_vals = np.arange(lo, hi + 1, dtype=np.int64)
    j_vals = k - i_vals               # satisfies 0 ≤ j_vals < n
    grad[i_vals] = 2.0 * seq[j_vals]

    new = seq - lr * grad
    new = np.maximum(new, 0.0)
    return np.clip(new, 0.0, 1000.0)


def solve_lp(a: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve
        maximise   Σ b_j
        subject to  (a * b)_k ≤ M   for all k,
                    b_j ≥ 0 .
    Returns the optimal b scaled to have ℓ¹‑norm = target_sum,
    or None if the problem fails.
    """
    n = a.shape[0]

    # Build the sparse convolution matrix A where (A b)_k = (a * b)_k .
    # Column i contributes a_i on rows i … i+n‑1.
    row_blocks = [np.arange(i, i + n, dtype=np.int64) for i in range(n)]
    row_idx = np.concatenate(row_blocks)
    col_idx = np.repeat(np.arange(n, dtype=np.int64), n)
    data = np.repeat(a, n)
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, abstol=1e-6, reltol=1e-6)
    except Exception as exc:               # pragma: no cover
        print("LP solve exception:", exc)
        return None

    if b.value is None:
        return None

    b_opt = np.asarray(b.value).flatten()
    s = b_opt.sum()
    if s <= 0.0:
        return None

    scale = target_sum / s
    return b_opt * scale


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Hybrid stochastic / LP search for a step sequence with minimal
    self‑convolution supremum.  Returns the best list of non‑negative
    floats found before the time budget runs out.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the wrapper that enforces the budget
    deadline = start + max(0.0, float(budget_s) - 2.0)

    # -----------------------------------------------------------------
    # Choose a reasonable length – big enough to be expressive, small
    # enough that the LP (which we call periodically) stays tractable.
    # -----------------------------------------------------------------
    desired_n = 2000
    try:
        # height_sequence_1 is provided by the surrounding harness
        init = np.array(height_sequence_1, dtype=np.float64)
        n = min(init.size, desired_n)
        cur_seq = init[:n].copy()
    except Exception:                     # pragma: no cover
        n = desired_n
        cur_seq = generate_power_sequence(n, exponent=0.5, offset=0.0)

    target_sum = np.sqrt(2.0 * n)

    # Normalise the initial vector (the objective is scale‑invariant)
    s = cur_seq.sum()
    if s > 0.0:
        cur_seq = cur_seq * (target_sum / s)
    else:
        cur_seq = np.full(n, target_sum / n, dtype=np.float64)

    best_seq = cur_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"initial score {best_score:.6f}  (n={n})")

    cur_score = best_score
    iteration = 0
    t_mix = 0.30                     # mixing factor for the LP step

    while time.monotonic() < deadline:
        # -------------------------------------------------------------
        #  Choose a move type
        # -------------------------------------------------------------
        move_type = rng.random()
        if move_type < 0.25:
            # Gradient descent toward lowering the peak
            lr = 10.0 ** rng.uniform(-4.0, -1.0)   # 1e‑4 … 1e‑1
            cand = gradient_step(cur_seq, lr)
        elif move_type < 0.75:
            # Block‑wise multiplicative perturbation
            cand = block_perturb(cur_seq, rng)
        else:
            # Tiny global log‑normal noise
            cand = global_noise(cur_seq, rng)

        # Re‑normalise to the canonical ℓ¹‑norm (scale does not affect the score)
        s = cand.sum()
        if s > 0.0:
            cand = cand * (target_sum / s)
        else:
            cand = np.full_like(cand, target_sum / n)

        # -------------------------------------------------------------
        #  Evaluate and possibly accept
        # -------------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand.copy()
            best_score = score
            cur_seq = cand.copy()
            cur_score = score
            print(f"new best {best_score:.6f}  it={iteration}  t={time.monotonic()-start:.1f}s")
        else:
            # Simulated‑annealing acceptance of a worse move (helps to escape plateaus)
            # Temperature decays linearly from 0.05 to 0.01.
            frac = (time.monotonic() - start) / (deadline - start)
            temp = max(0.01, (1.0 - frac) * 0.05)
            if rng.random() < np.exp(-(score - cur_score) / (temp + 1e-12)):
                cur_seq = cand.copy()
                cur_score = score

        # -------------------------------------------------------------
        #  Periodic LP re‑balancing (only when n is modest)
        # -------------------------------------------------------------
        if (iteration % 30 == 0) and (n <= 2500):
            conv = np.convolve(cur_seq, cur_seq)
            M = float(conv.max())
            g = solve_lp(cur_seq, M, target_sum)
            if g is not None:
                mixed = (1.0 - t_mix) * cur_seq + t_mix * g
                mixed = mixed * (target_sum / mixed.sum())
                mixed_score = float(evaluate_sequence(mixed.tolist()))
                if np.isfinite(mixed_score) and mixed_score < best_score:
                    best_seq = mixed.copy()
                    best_score = mixed_score
                    cur_seq = mixed.copy()
                    cur_score = mixed_score
                    print(f"LP improvement {best_score:.6f}  it={iteration}")

        # -------------------------------------------------------------
        #  Very rare random restart
        # -------------------------------------------------------------
        if rng.random() < 0.001:
            exp = rng.uniform(0.4, 0.6)
            off = rng.uniform(0.0, 2.0)
            cand = generate_power_sequence(n, exponent=exp, offset=off)
            cand = cand * (target_sum / cand.sum())
            cand_score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(cand_score) and cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                cur_seq = cand.copy()
                cur_score = cand_score
                print(f"restart improved to {best_score:.6f}")

        iteration += 1

    elapsed = time.monotonic() - start
    print(f"finished after {elapsed:.1f}s, best score {best_score:.6f}")
    return [float(x) for x in best_seq]
