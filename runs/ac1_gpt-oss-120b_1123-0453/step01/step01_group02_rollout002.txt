def make_pair_perturbation(base, rng):
    candidate = base.copy()
    n = len(candidate)
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j: return candidate
    mass = candidate[i]
    if mass <= 0:
        return candidate
    delta = rng.uniform(0, min(0.5*mass, mass))
    candidate[i] -= delta
    candidate[j] += delta
    return np.clip(candidate,0,1000)

"""
Propose a near‑optimal step function for the evaluation metric.

The algorithm consists of three stages:

1.  **Analytic family search** – a dense grid search over the
    family a_i = (i+α)^(-p) with p≈0.5.  The grid is first coarse and
    then refined around the best (p,α) pair; the best vector from this
    stage is kept as the current incumbent.

2.  **Simulated annealing** – starting from the incumbent we perform a
    long stochastic search.  Two perturbation operators are used:
        * a multiplicative log‑normal change on a random contiguous block,
        * a small transfer of mass from one index to another.
    Worse proposals are accepted with a temperature‑controlled probability,
    allowing the search to escape shallow local minima.

3.  **Deterministic contribution hill‑climb** – finally we polish the
    best vector found so far.  At each iteration the convolution entry
    with maximal value is located; the element that contributes most to
    that entry donates a small fraction of its mass to the element that
    contributes the least.  The step size gradually shrinks, leading to
    a fine‑grained flattening of the auto‑convolution.

All operations respect the constraints of ``evaluate_sequence`` (non‑negative
entries, clipping to [0, 1000]) and the whole routine stops before the
provided time budget expires (default 1000 s).  Progress and intermediate
best scores are printed to the console.
"""

import time
import math
import numpy as np


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a cheap O(n) multiplicative perturbation to a contiguous block
    of ``base`` and optionally symmetrise the whole vector.
    The result is clipped to the interval [0, 1000].
    """
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and 10 % of the whole vector
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width

    # multiplicative log‑normal factor (≈ 1 ± 12 %)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor

    # occasional mirror symmetrisation – helps to cross plateaus
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])

    return np.clip(cand, 0.0, 1000.0)


def make_pair_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Transfer a small amount of mass from a randomly chosen donor index
    to a randomly chosen receiver index.  The move preserves the total
    L1‑mass and respects the clipping bounds.
    """
    cand = base.copy()
    n = cand.size
    donor = int(rng.integers(0, n))
    receiver = int(rng.integers(0, n))
    if donor == receiver or cand[donor] <= 0.0:
        return cand

    # move at most 5 % of the donor's current value
    delta = float(rng.uniform(0.0, min(0.05 * cand[donor], cand[donor])))
    cand[donor] -= delta
    cand[receiver] += delta
    return np.clip(cand, 0.0, 1000.0)

    deadline = start_time + max(0.0, budget_s - 10.0)

    # coarse grid
    alphas_coarse = np.geomspace(1e-3, 4.0, num=40)          # log‑spaced
    p_coarse = np.linspace(0.45, 0.55, 21)                   # fine around 0.5

    for p in p_coarse:
        for alpha in alphas_coarse:
            if time.monotonic() >= deadline:
                break
            cand = (idx + alpha) ** (-p)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[grid] p={p:.4f} α={alpha:.4f} → {best_score:.6f}")

    # refined grid around the current best (if we have a candidate)
    if best_seq is not None:
        # extract the parameters that produced the current best
        # (we cannot recover them exactly, so we simply zoom around the centre)
        p_center = 0.5
        alpha_center = 0.4
        p_fine = np.linspace(p_center - 0.02, p_center + 0.02, 41)
        alpha_fine = np.linspace(alpha_center - 0.1, alpha_center + 0.1, 81)

        for p in p_fine:
            for alpha in alpha_fine:
                if alpha <= 0:
                    continue
                if time.monotonic() >= deadline:
                    break
                cand = (idx + alpha) ** (-p)
                score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    print(f"[refine] p={p:.5f} α={alpha:.5f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated annealing phase
    # ------------------------------------------------------------------
    if best_seq is None:                     # safety net
        best_seq = np.ones(n, dtype=np.float64)

    current = best_seq.copy()
    cur_score = best_score

    # temperature schedule (linear cooling)
    temp_start = max(0.001, cur_score * 0.05)
    temp_end = 1e-6

    iter_count = 0
    while time.monotonic() < deadline:
        # linear interpolation of temperature based on elapsed fraction
        elapsed = time.monotonic() - start_time
        fraction = min(1.0, elapsed / (deadline - start_time))
        T = temp_start * (1.0 - fraction) + temp_end * fraction

        # 60 % block move, 40 % pair move
        if rng.random() < 0.6:
            candidate = make_block_perturbation(current, rng)
        else:
            candidate = make_pair_perturbation(current, rng)

        score = float(evaluate_sequence(candidate.tolist()))
        if not np.isfinite(score):
            continue

        delta = score - cur_score
        # Metropolis acceptance
        if delta < 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            current = candidate
            cur_score = score
            if score < best_score:
                best_score = score
                best_seq = candidate.copy()
                print(f"[anneal] iter {iter_count:>6d} → new best {best_score:.6f}")

        iter_count += 1

    # ------------------------------------------------------------------
    # 3) Deterministic contribution hill‑climbing (fine polishing)
    # ------------------------------------------------------------------
    current = best_seq.copy()
    cur_score = best_score
    max_iters = 30_000                     # generous limit, time will stop us

    while time.monotonic() < deadline and max_iters > 0:
        conv = np.convolve(current, current)
        k_max = int(np.argmax(conv))

        # contributions of each entry to the peak at k_max
        i = np.arange(n, dtype=np.int64)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(current)
        contrib[mask] = current[i[mask]] * current[j[mask]]

        donor = int(np.argmax(contrib))
        receiver = int(np.argmin(contrib))

        if donor != receiver and current[donor] > 0.0:
            # transfer a tiny fraction (≤ 2 %) of the donor's mass
            delta = float(rng.uniform(0.0, min(0.02 * current[donor], current[donor])))
            candidate = current.copy()
            candidate[donor] -= delta
            candidate[receiver] += delta
            candidate = np.clip(candidate, 0.0, 1000.0)
        else:
            candidate = make_block_perturbation(current, rng)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < cur_score:
            current = candidate
            cur_score = score
            if score < best_score:
                best_score = score
                best_seq = candidate.copy()
                print(f"[hill] iter {max_iters:>6d} → improved {best_score:.6f}")

        max_iters -= 1

    print(f"Search finished – best score achieved: {best_score:.6f}")
    # Return a plain Python list as required by the grader.
    return [float(x) for x in best_seq]
