def propose_candidate(budget_s=1000, seed=12345):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.5
    best_seq = None
    best_score = np.inf
    best_params = None

    # First: random sampling of parameters
    while time.monotonic() < deadline and iterations < max_iter:
         n = int(rng.integers(2000, 12001)) # up to 12000?
         exp = rng.uniform(0.43, 0.58)
         off = rng.uniform(0.0, 3.0)
         lam_factor = rng.uniform(0.0, 5.0)  # lam = lam_factor / n
         seq = generate_seq(n, exp, off, lam_factor / n)
         score = fast_evaluate(seq)
         if score < best_score:
              best_seq = seq
              best_score = score
              best_params = (n, exp, off, lam_factor)

    # Local refinement using simulated annealing:
    # Starting from best_params:
    temp = 1.0
    while time.monotonic() < deadline:
         # propose neighbor
         n, exp, off, lam_factor = best_params
         # Perturb each with some probability
         if rng.random() < 0.3:
              n = max(500, int(n + rng.integers(-200, 201)))   # small variation
         if rng.random() < 0.5:
              exp = np.clip(exp + rng.normal(0, 0.005), 0.3, 0.8)
         if rng.random() < 0.5:
              off = np.clip(off + rng.normal(0, 0.2), 0.0, 5.0)
         if rng.random() < 0.5:
              lam_factor = np.clip(lam_factor + rng.normal(0, 0.5), 0.0, 10.0)
         seq = generate_seq(n, exp, off, lam_factor / n)
         score = fast_evaluate(seq)
         # Accept if better or if random with prob exp((best - score)/temp)
         if score < best_score or rng.random() < np.exp((best_score - score)/max(temp, 1e-8)):
              best_seq = seq
              best_score = score
              best_params = (n, exp, off, lam_factor)
              # optionally reduce temperature gradually
         temp *= 0.9999

    # Finally, apply block perturb hill climbing to best_seq
    # for some time
    while time.monotonic() < deadline:
          cand = block_perturb(best_seq, rng)
          score = fast_evaluate(cand)
          if score < best_score:
               best_seq = cand
               best_score = score

    # Convert to list
    return [float(x) for x in best_seq]

def fast_evaluate(seq):
    n = seq.size
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

"""
Algorithm
---------

We search a low‑dimensional family of step functions

    a_i = (i + offset + 1)^(-exponent) * exp(-lam * i),

with lam = lam_factor / n (n = length of the sequence).  The exponential
factor damps the tail of the classic (i+1)^(-1/2) kernel and reduces the
peak of the self‑convolution, allowing the evaluation
    2·n·max(a*a) / (sum(a)^2)
to drop well below the historic bound π/2 ≈ 1.571.

The optimisation proceeds in three stages, all respecting the 1000 s
budget:

1. **Coarse grid search** over (exponent, offset, lam_factor) to obtain a
   good starting point.

2. **Simulated‑annealing refinement** that perturbs the three parameters
   (and occasionally the length n) and accepts both improving moves
   and, with a slowly decreasing temperature, occasional worsening moves.

3. **Fine‑grained hill‑climbing** on the concrete vector using
   multiplicative block perturbations and small global log‑normal noise.
   This phase smoothes the discrete sequence beyond the low‑dimensional model.

The best sequence encountered is returned as a plain Python list of
floats.  All helper functions are top‑level, no closures are used, and the
search uses only NumPy and the supplied ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def generate_seq(n: int,
                 exponent: float = 0.5,
                 offset: float = 0.0,
                 lam: float = 0.0) -> np.ndarray:
    """
    Produce a non‑negative sequence

        a_i = (i + offset + 1)^(-exponent) * exp(-lam * i)

    The result is clipped to the range [0, 1000] to satisfy the evaluator.
    """
    i = np.arange(n, dtype=np.float64)
    seq = (i + offset + 1.0) ** (-exponent) * np.exp(-lam * i)
    return np.clip(seq, 0.0, 1000.0)


def fast_evaluate(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence – no type‑checks, just the formula."""
    n = seq.size
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.06)))   # ~6 % log‑normal change
    new = seq.copy()
    new[start:stop] *= factor
    # occasional symmetrisation – helps flatten the convolution
    if rng.random() < 0.1:
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)


def global_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Very small multiplicative log‑normal noise applied to the whole vector."""
    noise = rng.normal(0.0, 0.02, size=seq.shape)
    factor = np.exp(noise)
    new = seq * factor
    return np.clip(new, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list:
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a short safety margin so that the platform can finish gracefully
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Choose a base length.  Prefer the length of the previously found
    # construction (height_sequence_1) if it exists.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 5000

    best_seq = None
    best_score = np.inf
    best_params = None  # (exponent, offset, lam_factor, n)

    # ------------------------------------------------------------------
    # 1) Coarse grid search
    # ------------------------------------------------------------------
    exponent_grid = [0.45, 0.475, 0.5, 0.525, 0.55]
    offset_grid = [0.0, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 2.5, 3.0]
    lam_factor_grid = [0.0, 0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5,
                       4.0, 4.5, 5.0, 6.0, 7.0]

    for exp in exponent_grid:
        for off in offset_grid:
            for lam_f in lam_factor_grid:
                if time.monotonic() >= deadline:
                    break
                lam = lam_f / n
                seq = generate_seq(n, exponent=exp, offset=off, lam=lam)
                score = fast_evaluate(seq)
                if score < best_score:
                    best_seq = seq
                    best_score = score
                    best_params = (exp, off, lam_f, n)
                    print(f"grid: exp={exp:.3f}, off={off:.1f}, lamF={lam_f:.1f} → "
                          f"score={score:.6f}")

    # If, for any reason, the grid never produced a candidate, fall back.
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = fast_evaluate(best_seq)
        best_params = (0.5, 0.0, 0.0, n)

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing style refinement of the three parameters
    # ------------------------------------------------------------------
    # Temperature schedule – linear decay from 1.0 to 1e-3 over the remaining time.
    temp_start = 1.0
    temp_end = 1e-3

    while time.monotonic() < deadline:
        # unpack current best parameters
        cur_exp, cur_off, cur_lam_f, cur_n = best_params

        # propose a new point by Gaussian perturbation
        new_exp = np.clip(cur_exp + rng.normal(0.0, 0.01), 0.3, 0.8)
        new_off = np.clip(cur_off + rng.normal(0.0, 0.25), 0.0, 5.0)
        new_lam_f = np.clip(cur_lam_f + rng.normal(0.0, 0.5), 0.0, 12.0)

        # occasionally change the length (kept within reasonable bounds)
        if rng.random() < 0.1:
            delta_n = int(rng.integers(-400, 401))
            new_n = max(1000, cur_n + delta_n)
        else:
            new_n = cur_n

        lam = new_lam_f / new_n
        cand_seq = generate_seq(new_n, exponent=new_exp,
                                offset=new_off, lam=lam)
        cand_score = fast_evaluate(cand_seq)

        # acceptance rule
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            # compute a temperature that linearly decays with elapsed time
            elapsed = time.monotonic() - start_time
            frac = min(1.0, elapsed / budget_s)
            temperature = temp_start * (1.0 - frac) + temp_end * frac
            # Metropolis probability
            prob = math.exp((best_score - cand_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            best_seq = cand_seq
            best_score = cand_score
            best_params = (new_exp, new_off, new_lam_f, new_n)
            print(f"anneal → score={best_score:.6f} "
                  f"(exp={new_exp:.3f}, off={new_off:.3f}, lamF={new_lam_f:.2f}, n={new_n})")

    # ------------------------------------------------------------------
    # 3) Fine‑grained hill climbing on the concrete vector
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        if rng.random() < 0.7:
            cand_seq = block_perturb(best_seq, rng)
        else:
            cand_seq = global_noise(best_seq, rng)

        cand_score = fast_evaluate(cand_seq)
        if cand_score < best_score:
            best_seq = cand_seq
            best_score = cand_score
            print(f"fine‑tune → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (optional, cheap)
    # ------------------------------------------------------------------
    try:
        official_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"final official score = {official_score:.6f}")
    except Exception:
        # If evaluate_sequence is not available we simply return the best we have.
        pass

    # Return a plain Python list as required by the outside harness.
    return [float(x) for x in best_seq]
