"""
Search for a step function with low evaluate_sequence score.

The algorithm consists of three stages:
1) coarse grid search over the 2‑parameter family a_i = (i+α)^{-p}
   to obtain a good "analytic" baseline.
2) systematic perturbation that adds a small constant ε to a tail
   of the sequence (a block of indices at the end).  For several
   tail lengths (as a fraction f of n) and ε values the score is
   evaluated; the best combination is kept.  This exploits the fact
   that adding mass far out in the tail increases the total sum
   faster than it raises the maximal auto‑convolution, thereby
   lowering the evaluation metric.
3) a short hill‑climbing phase that repeatedly moves a tiny amount
   of mass from the element that contributes most to the current
   maximal convolution entry to the element that contributes least.
   Improvements are kept; occasional random jumps help escape plateaus.

All steps respect the evaluator’s constraints (non‑negative,
clipped to [0,1000]) and terminate before the 1000‑second budget.
"""

import numpy as np
import time

def base_sequence(p: float, alpha: float, n: int) -> np.ndarray:
    """Return a_i = (i + alpha)^{-p} for i = 0..n-1."""
    idx = np.arange(n, dtype=np.float64)
    return (idx + alpha) ** (-p)

def evaluate(seq: np.ndarray) -> float:
    """Convenient wrapper that returns the float score."""
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate():
    """Return a list of floats that (hopefully) beats the target score."""
    rng = np.random.default_rng(12345)

    # ------------------------------------------------------------------
    # Determine length from the supplied global sequence (if any)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 256

    deadline = time.monotonic() + 970.0   # keep a safety margin

    # ------------------------------------------------------------------
    # 1) Coarse grid over the analytic family (p, α)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    idx = np.arange(n, dtype=np.float64)

    # ranges – we know the optimum is close to p=0.5, α≈0.4
    p_grid = np.linspace(0.45, 0.55, 21)      # step 0.005
    alpha_grid = np.linspace(0.05, 2.0, 81)   # step ~0.025

    for p in p_grid:
        for alpha in alpha_grid:
            if time.monotonic() >= deadline:
                break
            cand = (idx + alpha) ** (-p)
            score = evaluate(cand)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"grid p={p:.3f} α={alpha:.3f} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Tail‑mass perturbation.
    #    For several tail fractions f and several ε values we add the
    #    same ε to the last f·n entries and keep the best result.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)

    base_seq = best_seq.copy()
    # pre‑compute a rough estimate of the maximal allowed ε (so that
    # ε²·m does not dominate the convolution).  0.05 is a safe ceiling.
    eps_max = 0.05

    tail_fractions = np.linspace(0.3, 0.95, 14)   # 14 values
    eps_grid = np.linspace(0.0, eps_max, 51)     # 51 values

    for f in tail_fractions:
        m = int(round(f * n))
        if m == 0:
            continue
        tail_start = n - m
        head_sum_est = np.sum(base_seq[:tail_start])    # cheap estimate
        for eps in eps_grid:
            if time.monotonic() >= deadline:
                break
            cand = base_seq.copy()
            # add a constant bump ε to the tail
            cand[tail_start:] = np.clip(cand[tail_start:] + eps, 0.0, 1000.0)
            score = evaluate(cand)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"tail f={f:.2f} eps={eps:.4f} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Light hill‑climbing from the best candidate found so far.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    cur_score = best_score

    while time.monotonic() < deadline and max_iters > 0:

        k_max = int(np.argmax(conv))

        # contribution of each element to that convolution entry
        i = np.arange(n, dtype=np.int64)

            if available <= 0.0:

            cur_score = score

                best_seq = cand.copy()
                print(f"hill‑climb improved score to {best_score:.6f}")
        else:
            # occasional random jump
            if rng.random() < 0.001:
                current = cand
                cur_score = score

        max_iters -= 1

    print(f"Search finished – best score: {best_score:.6f}")

    return [float(x) for x in best_seq]

"""
Search for a step‑function with a low evaluate_sequence score.

The method works in three stages:

1. **Analytic baseline** – a quick grid over the 2‑parameter family
   a_i = (i+α)^{-p} (p≈0.5, α≈0.4) to obtain a good starting vector.
2. **Tail‑mass perturbation** – for several tail‑fractions f and several
   tiny constants ε we add ε to the last f·n entries of the baseline.
   Adding a small uniform bump far out in the tail increases the total
   sum much faster than it raises the maximal auto‑convolution,
   therefore the evaluation metric, 2·n·max(conv)/sum², drops.
   The (f,ε) grid is tiny (≈700 candidates) and is evaluated
   directly with `evaluate_sequence`.
3. **Fine hill‑climbing** – starting from the best candidate we
   repeatedly move a small amount of mass from the index that contributes
   most to the current maximal convolution entry to the index that
   contributes least.  Improvements are kept; occasional random jumps
   avoid plateaus.

All generated numbers respect the evaluator’s rules
(non‑negative, clipped to [0, 1000]).  The routine stops well before the
1000‑second time limit and returns the best sequence found.
"""

    return (idx + alpha) ** (-p)
