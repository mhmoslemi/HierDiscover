""" 
Propose a candidate step‑function sequence using a high‑resolution arcsine
distribution. The continuous density f(t)=1/(2√t) on [0,1] has the
property that its self‑convolution is constant (=π/4).  After normalising
the discrete analogue a_i ∝ 1/√(i+1) the evaluation score tends to
π/2≈1.570796.  By using a much larger number of steps (≈10⁵) the
discretisation error of the previous 6 k‑element construction disappears
and the score approaches the theoretical optimum.  A short, cheap
hill‑climbing loop with random block perturbations is added to polish the
solution within the remaining time budget.
"""

import time, math, numpy as np

def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a cheap multiplicative block perturbation (log‑normal jitter)."""
    cand = base.copy()
    n = cand.size
    # block width between 0.2 % and 5 % of the length
    lo = max(1, n // 500)
    hi = max(2, n // 20)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # log‑normal multiplicative factor (small jitter)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.08)))
    # occasional global mirroring – keeps the shape roughly symmetric
    if rng.random() < 0.25:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the bounds required by evaluate_sequence
    return np.clip(cand, 0.0, 1000.0)

    """

    # ------------------------------------------------------------------
    # If a previous construction is present we keep its length as a lower
    # bound and enlarge it, otherwise we start with a fairly large default.
    try:
        prev_len = len(height_sequence_1)               # pylint: disable=undefined-variable
    except Exception:
        prev_len = 0
    # target length – a power of two is convenient for the FFT, but it is
    # not required.  We use at least 100 000 points; if a previous solution
    # is longer we keep that length.
    n = max(prev_len * 2, 100_000)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
