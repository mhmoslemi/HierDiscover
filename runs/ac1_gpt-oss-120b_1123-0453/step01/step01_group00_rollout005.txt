    M = max_convolution(f)
    # build conv matrix A
    A = conv_matrix(f)
    # variable g

    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # solve
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return f, M
    g0 = np.maximum(g.value, 0.0)
    sum_g = np.sum(g0)

    # rescale g to have same l1 norm

    f_new = (1 - t) * f + t * g_rescaled

"""
Propose a low‑value step sequence.
The algorithm is a hybrid of stochastic local search (simulated‑annealing‑style)
and a few linear‑programming refinement steps (if CVXPY is available).

* A random start with the required ℓ¹‑norm (√(2·n)) is refined by
  repeatedly moving a small amount of mass from one coordinate to another.
  Moves that lower the maximal auto‑convolution are always accepted, otherwise
  they are accepted with a temperature‑dependent probability (simulated annealing).

* After the stochastic phase we optionally run a short LP “mass‑inflation”
  routine similar to the reference code: maximise ℓ¹‑mass under the current
  worst convolution bound, then mix the result with the current vector.
  This often decreases the bound further.

The search is performed for several lengths (n ∈ {200,300,…,1000}) and many
random restarts until the time budget (≈ 1000 s) expires. The best sequence
found overall is returned as a plain Python list of floats.
"""

import time, math, numpy as np

# ----------------------------------------------------------------------
# Convolution utilities -------------------------------------------------
# ----------------------------------------------------------------------


def max_conv_fft(seq: np.ndarray, size: int = None) -> float:
    """
    Fast (FFT‑based) computation of max_i (seq * seq)[i].
    size may be pre‑computed as the next power of two ≥ 2·len(seq)−1.
    """
    n = seq.size
    if size is None:
        size = 1 << ((2 * n - 1).bit_length())
    # real FFT is enough because seq is real
    f̂ = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f̂ * f̂, size)[: 2 * n - 1]
    # numerical noise may produce tiny negatives → clip
    conv = np.maximum(conv, 0.0)
    return float(conv.max())


def max_conv_exact(seq: np.ndarray) -> float:
    """Exact (but O(n²)) max convolution using NumPy’s direct convolution."""
    return float(np.max(np.convolve(seq, seq)))


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional, requires cvxpy) -------------
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return a matrix A (shape (2n‑1, n)) such that (A @ g)[k] = (f * g)[k].
    It is the Toeplitz‑like matrix formed by f.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # for i in [i_start, i_end] we have j = k - i
        A[k, k - np.arange(i_start, i_end + 1)] = f[i_start : i_end + 1]
    return A


def lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple:
    """One LP refinement step (maximise ℓ¹‑mass under the current max‑conv bound)."""
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return f, max_conv_exact(f)

    n = f.size
    M = max_conv_exact(f)

    # Build convolution matrix once
    A = build_conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is fast for the dense problems we generate
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

    if g.value is None:
        return f, M

    g_val = np.maximum(g.value, 0.0)
    sum_g = g_val.sum()
    if sum_g < 1e-12:
        return f, M

    # Rescale to the target ℓ¹‑norm
    g_rescaled = g_val * (target_sum / sum_g)

    # Mix with a modest step size
    t = 0.3
    f_new = (1.0 - t) * f + t * g_rescaled
    f_new = np.clip(f_new, 0.0, 1000.0)

    return f_new, max_conv_exact(f_new)


# ----------------------------------------------------------------------
# Stochastic (simulated‑annealing) local search -------------------------
# ----------------------------------------------------------------------


def sa_search(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 200_000,
) -> tuple:
    """
    Greedy‑with‑temperature local search.
    Returns (best_sequence, best_max_conv).
    """
    n = f0.size
    # Pre‑compute FFT size for speed
    fft_size = 1 << ((2 * n - 1).bit_length())

    f = f0.copy()
    cur_max = max_conv_fft(f, fft_size)
    best_f = f.copy()
    best_max = cur_max

    # cooling schedule: start at T0, linear down to ~0
    T0 = 0.02
    start = time.monotonic()
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # linear temperature based on elapsed fraction
        elapsed = time.monotonic() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # amount that can be transferred from j to i
        max_transfer = min(f[j], 1000.0 - f[i])
        if max_transfer <= 1e-12:
            continue

        # typical step magnitude ≈ half of the average entry
        avg = target_sum / n
        step = min(max_transfer, avg * 0.5)
        delta = rng.random() * step
        if delta <= 1e-12:
            continue

        f_new = f.copy()
        f_new[i] += delta
        f_new[j] -= delta

        new_max = max_conv_fft(f_new, fft_size)

        dE = new_max - cur_max
        if dE < -1e-12 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            f = f_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_f = f_new.copy()
                print(f"[SA] n={n} iter={it} new best max={best_max:.6f}")

        # occasional tiny random jitter to escape plateaus
        if it % 5000 == 0:
            k, l = rng.integers(0, n, size=2)
            if k != l:
                tiny = min(f[l], 1e-2)
                if tiny > 0:
                    f[k] += tiny
                    f[l] -= tiny
                    cur_max = max_conv_fft(f, fft_size)

    return best_f, best_max


# ----------------------------------------------------------------------
# Helper to enlarge a known good sequence to a larger length ------------
# ----------------------------------------------------------------------


def expand_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Stretch ``seq`` into ``new_len`` places by simple nearest‑index mapping
    and re‑normalise to the appropriate ℓ¹‑norm (√(2·new_len)).
    """
    old_len = seq.size
    out = np.zeros(new_len, dtype=np.float64)
    for i, val in enumerate(seq):
        idx = int(i * new_len / old_len)
        if idx >= new_len:
            idx = new_len - 1
        out[idx] += val
    total = out.sum()
    if total > 0:
        target = math.sqrt(2.0 * new_len)
        out = out * (target / total)
    return out


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a step sequence with a low evaluation score.
    The function respects the 1000 s time limit (≈ budget_s‑2 s for safety)
    and returns the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction is available, use it as a seed.
    # ``height_sequence_1`` is expected to be a global list/array.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim == 1 and init.size > 0:
            n0 = init.size
            target0 = math.sqrt(2.0 * n0)
            if init.sum() > 0:
                init = init * (target0 / init.sum())
                sc0 = max_conv_exact(init)
                best_seq = init.copy()
                best_score = sc0
                print(f"[init] using height_sequence_1 (n={n0}) – score={sc0:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives a larger search space.
    # The list can be tuned, but these values work well within the time limit.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 600, 800, 1000]

    # ------------------------------------------------------------------
    # Main outer loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        fft_size = 1 << ((2 * n - 1).bit_length())

        # Decide how many random restarts we can afford for this length.
        # We budget roughly equal time per length.
        restarts = 0
        while time.monotonic() < deadline and restarts < 30:
            restarts += 1

            # ------------------------------------------------------------------
            # Initial point: either a fresh random vector or an expanded previous best.
            # ------------------------------------------------------------------
            if best_seq is not None and best_seq.size < n and rng.random() < 0.5:
                f0 = expand_sequence(best_seq, n)
            else:
                f0 = rng.random(n)
                f0 = f0 / f0.sum() * target_sum

            # ------------------------------------------------------------------
            # Stochastic local search
            # ------------------------------------------------------------------
            f_local, max_local = sa_search(f0, target_sum, deadline, rng)

            # ------------------------------------------------------------------
            # Optional LP refinement (only if cvxpy is installed)
            # ------------------------------------------------------------------
            try:
                import cvxpy as cp  # noqa: F401
                has_cvx = True
            except Exception:  # pragma: no cover
                has_cvx = False

            if has_cvx:
                # a few cheap LP iterations often improve the bound
                for lp_it in range(5):
                    if time.monotonic() >= deadline:
                        break
                    f_local, max_local = lp_refine(f_local, target_sum, deadline)
                # recompute exact max after LP
                max_local = max_conv_exact(f_local)

            # ------------------------------------------------------------------
            # Record global best
            # ------------------------------------------------------------------
            if max_local < best_score - 1e-12:
                best_score = max_local
                best_seq = f_local.copy()
                print(
                    f"[global] n={n} restart={restarts} "
                    f"new best score={best_score:.6f}"
                )
            # early exit if we already beat the target 1.503
            if best_score < 1.503:
                print("[stop] reached target bound")
                break

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed, but guarantees a return.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = max_conv_exact(best_seq)
        print("[fallback] uniform sequence used, score=", best_score)

    # Final sanity: convert to plain Python list of floats.
    return [float(x) for x in best_seq]
