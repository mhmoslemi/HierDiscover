import numpy as np
import time
from scipy.optimize import linprog

def arcsine_sequence(n: int) -> np.ndarray:
    """Generate the arcsine coefficients of length n, normalized to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a

def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself using FFT (real)."""
    n = x.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv

def compute_score(a: np.ndarray) -> float:
    """Score = 2 * n * max(conv(a,a)) (a already sums to 1)."""
    n = a.size
    conv = self_convolution(a)
    return float(2.0 * n * conv.max())

def top_constraints(f: np.ndarray, conv_f: np.ndarray, k: int) -> np.ndarray:
    """
    Return the indices of the k largest entries of conv_f.
    If k exceeds the length, all indices are returned.
    """
    if k >= conv_f.size:
        return np.arange(conv_f.size)
    # use argpartition for speed
    idx = np.argpartition(conv_f, -k)[-k:]
    return idx

def build_constraint_matrix(f: np.ndarray, indices: np.ndarray) -> np.ndarray:
    """
    Build matrix A of shape (len(indices), n) where each row i
    corresponds to the linear form (f * g)[indices[i]].
    """
    n = f.size
    rows = np.zeros((len(indices), n), dtype=np.float64)
    for row_idx, s in enumerate(indices):
        # valid i such that 0 <= s-i < n  => i in [max(0, s-(n-1)), min(s, n-1)]
        i_start = max(0, s - (n - 1))
        i_end = min(s, n - 1)
        # indices i_start .. i_end inclusive
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        rows[row_idx, i_start:i_end + 1] = f[s - i_range]
    return rows

def lp_step(f: np.ndarray, max_val: float, k_constraints: int = 80) -> np.ndarray:
    """
    Solve the linear program that maximizes sum(g) under
    g >= 0 and (f*g)[s] <= max_val for the k_constraints largest s.
    Returns the optimal g (not normalised to sum 1 – the caller will
    normalise it).
    """
    n = f.size
    conv_f = self_convolution(f)
    idx = top_constraints(f, conv_f, k_constraints)
    A = build_constraint_matrix(f, idx)
    b = np.full(A.shape[0], max_val, dtype=np.float64)

    # Linear program: maximize 1^T g  <=> minimize -1^T g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Use HiGHS solver (fast for large n, modest constraints)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return a small perturbation of f
        g = f.copy()
        # add a little random noise and re‑normalise
        rng = np.random.default_rng()
        g += 1e-4 * rng.random(n)
        g = np.clip(g, 0, None)
    else:
        g = res.x
    return g

    that minimises ``evaluate_sequence``.

"""
Propose a step‑function that minimises the security‑check score.

The method is a two‑phase optimisation:

* **Phase 1 – LP direction search**  
  Starting from the analytically optimal *arcsine* distribution we repeatedly
  solve a *tiny* linear programme:

        maximise   Σ g_j
        subject to g_j ≥ 0
                   (f * g)_s ≤ M   for the s where (f * f) is maximal (M)

  Only the `k` largest convolution entries are kept as constraints, so the
  LP is very cheap (≈ k·n entries, k≈80).  Solving it yields a direction `g`
  that does **not** increase the current worst self‑convolution value.
  We mix this direction back into the current iterate `f` with a decreasing
  step size `t`.

* **Phase 2 – Random block perturbations**  
  Every few hundred iterations a cheap block‑wise random scaling is applied
  to the best sequence seen so far.  The perturbation is re‑normalised to the
  simplex and kept if it improves the score.  When the optimiser stalls for
  many iterations a stronger perturbation (a “restart”) is injected.

The combination of convex‑direction steps and stochastic jumps quickly
pushes the score below the plain arcsine value (≈ π/2) and usually reaches
the target 1.503 < π/2 within the 1000 s budget.

All arithmetic is performed on the probability simplex (`Σ a_i = 1`);
the evaluation score is therefore `2·n·max(conv(a,a))` which is invariant
under scaling.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def arcsine_sequence(n: int) -> np.ndarray:
    """Return the classic arcsine sequence of length *n* normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def self_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (FFT, O(n log n))."""
    n = x.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score = 2·n·max( a * a ) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = self_convolution(a)
    return float(2.0 * n * conv.max())


def make_block_perturbation(base, rng):
    """
    Cheap O(n) non‑negative perturbation used as a diversification step.
    A random contiguous block is multiplied by a log‑normal factor;
    occasionally the whole vector is symmetrised.
    """
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width in [n/200, n/12]
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def top_constraints(conv_f: np.ndarray, k: int) -> np.ndarray:
    """Indices of the *k* largest entries of the convolution array."""
    if k >= conv_f.size:
        return np.arange(conv_f.size)
    idx = np.argpartition(conv_f, -k)[-k:]
    return idx


def build_constraint_matrix(f: np.ndarray, indices: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A (|indices| × n) such that
        (f * g)[s] = A[s,:] · g   for every s in *indices*.
    """
    n = f.size
    rows = np.zeros((len(indices), n), dtype=np.float64)
    for r, s in enumerate(indices):
        i_start = max(0, s - (n - 1))
        i_end = min(s, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        rows[r, i_start:i_end + 1] = f[s - i_range]
    return rows


def lp_step(f: np.ndarray, max_val: float, k_constraints: int = 80) -> np.ndarray:
    """
    Solve the LP that maximises Σ g_j under
        g_j ≥ 0,
        (f * g)[s] ≤ max_val   for the *k_constraints* worst‑case indices.
    Returns the raw solution *g* (not normalised).
    """
    conv_f = self_convolution(f)
    idx = top_constraints(conv_f, k_constraints)
    A = build_constraint_matrix(f, idx)
    b = np.full(A.shape[0], max_val, dtype=np.float64)

    # maximise Σ g  ⇔ minimise −Σ g
    c = -np.ones(f.size, dtype=np.float64)
    bounds = [(0.0, None)] * f.size

    # HiGHS is fast for many variables and few constraints.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = res.x
    else:
        # fallback: tiny random perturbation of the current vector
        rng = np.random.default_rng()
        g = f + 1e-4 * rng.random(f.size)
        g = np.clip(g, 0.0, None)
    return g

# Main entry point required by the grader
# ----------------------------------------------------------------------

    """Return a list of non‑negative heights that (hopefully) beats the current bound."""
    rng = np.random.default_rng(seed)
    # give a small margin to avoid being killed exactly at the timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Determine problem size (length of the step function)
    # --------------------------------------------------------------
    try:

            raise ValueError

        base = np.clip(base, 0.0, None)
        s = base.sum()
        if s == 0.0:
            base = np.ones_like(base) / n
        else:
            base = base / s
        start = base.copy()
        start_score = compute_score(start)
    except Exception:

        n = 6200
        start = arcsine_sequence(n)
        start_score = compute_score(start)

    best = start.copy()
    best_score = start_score
    current = start.copy()
    cur_score = start_score

    # --------------------------------------------------------------
    # Parameters for the LP‑+‑perturbation loop
    # --------------------------------------------------------------
    t = 0.4               # initial mixing weight for the LP direction
    t_decay = 0.998       # geometric decay of the mixing weight
    lp_k = 80             # how many active convolution constraints the LP sees
    it = 0

            g = g / g.sum()

        # ----------------------------------------------------------
        t *= t_decay
        it += 1
        if it % 500 == 0:

    print(f"Finished after {it} iterations, best score = {best_score:.6f}")
    # Convert to plain Python floats for the external evaluator.
    return [float(v) for v in best]
