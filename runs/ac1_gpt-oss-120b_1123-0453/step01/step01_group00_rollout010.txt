def local_search(f, max_iter, deadline, initial_step_factor=0.2):
    # assumes sum(f) is fixed
    n = len(f)
    conv = np.convolve(f, f)
    sum_f = np.sum(f)   # constant
    best_f = f.copy()
    best_max = conv.max()
    best_score = 2*n*best_max/(sum_f*sum_f)
    step_factor = initial_step_factor
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        # maybe adapt step_factor: reduce over time.
        # Choose i random weighted by f[i] maybe.
        i = rng.integers(n)
        if f[i] <= 1e-12:
            continue
        # choose j random else distinct
        j = rng.integers(n)
        if i == j: continue
        # delta = step_factor * f[i] * rng.random()
        delta = step_factor * f[i] * rng.random()
        # optionally set max delta absolute
        delta = min(delta, f[i])
        # Compute contributions
        # Use broadcasting: indices = np.arange(2*n-1)
        idx = np.arange(2*n-1)
        # Compute f_shift_i = f[idx - i] where valid
        shift_i = idx - i
        # Using np.take with mode='clip' maybe easier.
        # We'll use an array of zeros same length.
        contrib_i = np.zeros_like(conv)
        valid_i = (shift_i >= 0) & (shift_i < n)
        contrib_i[valid_i] = f[shift_i[valid_i]]
        shift_j = idx - j
        contrib_j = np.zeros_like(conv)
        valid_j = (shift_j >= 0) & (shift_j < n)
        contrib_j[valid_j] = f[shift_j[valid_j]]
        delta_conv = 2 * delta * (contrib_j - contrib_i)
        new_conv = conv + delta_conv
        new_max = new_conv.max()
        # Score unchanged denominator, compare new_max with best.
        if new_max < best_max * (1 - 1e-12):  # strict improvement
            # Accept move
            f[i] -= delta
            f[j] += delta
            conv = new_conv
            best_max = new_max
            best_f = f.copy()
        else:
            # maybe accept with probability if better in other metrics?
            pass
        # optionally occasionally recompute conv from scratch to maintain numerical stability
        if (it % 1000) == 0:
            conv = np.convolve(f, f)
    return best_f, best_max

    import cvxpy as cp
    n = len(f)
    A = conv_matrix(f)   # shape (2*n-1, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)

    g0 = np.maximum(g.value, 0.0)
    sum_g = g0.sum()
    if sum_g <= 1e-12:
        return None
    g_rescaled = g0 * (target_sum / sum_g)

def generate_triangle(n, target_sum):
    """Return a symmetric triangular non‑negative sequence of length n whose
    ℓ¹‑norm equals ``target_sum``. Height linearly rises from zero at the ends to
    a maximum in the centre.
    """
    center = (n - 1) / 2.0
    # distance from centre
    d = np.abs(np.arange(n) - center)
    heights = np.maximum(0.0, (n / 2.0) - d)
    s = heights.sum()
    if s == 0:
        return np.zeros(n, dtype=float)
    return heights * (target_sum / s)

def lp_optimize(f, M, target_sum):
    import cvxpy as cp
    n = f.shape[0]
    A = conv_matrix(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000)
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    sum_g = g0.sum()
    if sum_g <= 1e-12:
        return None
    # rescale to target sum (should equal target_sum)
    g_rescaled = g0 * (target_sum / sum_g)
    return g_rescaled

def local_search(f, target_sum, deadline, rng):
    n = len(f)
    conv = np.convolve(f, f)
    best_f = f.copy()
    best_max = conv.max()
    best_score = 2 * n * best_max / (target_sum * target_sum)
    # parameters
    step_factor = 0.2
    it = 0
    # precompute arr for indices
    while time.monotonic() < deadline:
        # check iteration count to stop to avoid infinite
        if it >= 200000:
            break
        k_star = int(np.argmax(conv))
        # compute derivative contributions
        idx = np.arange(n)
        paired = k_star - idx
        valid = (paired >= 0) & (paired < n)
        d = np.zeros(n)
        d[valid] = f[paired[valid]]
        # indices with high derivative, require f[i] > 0
        high_mask = (d > np.percentile(d, 80)) & (f > 1e-12)
        low_mask = (d < np.percentile(d, 20))
        # pick i
        if np.any(high_mask):
            i_candidates = np.where(high_mask)[0]
            i = int(rng.choice(i_candidates))
        else:
            # fallback to argmax derivative with positive mass
            i = int(np.argmax(d * (f > 1e-12)))
            if f[i] <= 1e-12:
                # random fallback
                i = int(rng.integers(n))
                while f[i] <= 1e-12:
                    i = int(rng.integers(n))
        # pick j
        if np.any(low_mask):
            j_candidates = np.where(low_mask)[0]
            j = int(rng.choice(j_candidates))
        else:
            j = int(rng.integers(n))
            while j == i:
                j = int(rng.integers(n))
        # delta
        max_delta = f[i]
        if max_delta <= 1e-12:
            continue
        delta = step_factor * max_delta * rng.random()
        if delta <= 0:
            continue
        # construct contributions for i and j
        # contributions vectors length 2n-1
        contrib_i = np.zeros(2*n-1)
        contrib_i[i:i+n] = f
        contrib_j = np.zeros(2*n-1)
        contrib_j[j:j+n] = f
        new_conv = conv + 2 * delta * (contrib_j - contrib_i)
        new_max = new_conv.max()
        if new_max < best_max - 1e-12:
            # accept
            f[i] -= delta
            f[j] += delta
            conv = new_conv
            best_max = new_max
            best_f = f.copy()
            # maybe increase step_factor slightly
            step_factor = min(0.4, step_factor * 1.02)
        else:
            # optional reduce step_factor
            step_factor = max(0.01, step_factor * 0.99)
        it += 1
        # periodic exact recompute
        if (it % 5000) == 0:
            conv = np.convolve(f, f)
    return best_f, best_max

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Hybrid LP + greedy local refinement to minimise the auto‑convolution
    upper‑bound used by ``evaluate_sequence``.
    The routine tries several sequence lengths, starts from triangular or
    uniform vectors, performs a few LP‑based ℓ¹‑mass enlargements while keeping
    the worst convolution entry fixed, and finally applies a greedy pairwise
    mass‑transfer local search.
    The best sequence found before the time budget expires is returned.
    """
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float('inf')

    # Determine whether CVXPY is available
    try:
        import cvxpy  # noqa: F401
        HAS_CVX = True
    except Exception:
        HAS_CVX = False

    # Candidate lengths (increasing). Larger n give finer lower bound but LP is slower.
    candidate_lengths = [200, 300, 400, 500, 600, 800]

    for n in candidate_lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # Try a few different initialisations
        init_functions = [
            lambda: generate_triangle(n, target_sum),
            lambda: generate_uniform(n, target_sum),
            lambda: generate_random(n, target_sum, rng)
        ]

        for init_idx, init_func in enumerate(init_functions):
            if time.monotonic() >= deadline:
                break
            f = init_func()
            # Ensure values respect the clipping range used by evaluate_sequence
            np.clip(f, 0.0, 1000.0, out=f)

            # Baseline score
            score = evaluate_sequence(f.tolist())
            if score < best_score:
                best_score = score
                best_seq = f.copy()
                print(f"[init] n={n} init={init_idx} score={score:.6f}")

            # LP phase (only if cvxpy is available)
            if HAS_CVX:
                # number of LP iterations limited by remaining time
                max_lp_iters = max(5, int(30 * 200 / n))   # scale down for large n
                # mixing factor (starting)
                for lp_it in range(max_lp_iters):
                    if time.monotonic() >= deadline:
                        break
                    M = np.max(np.convolve(f, f))
                    g = lp_optimize(f, M, target_sum)
                    if g is None:
                        break
                    # try different mixing weights, pick the first that does not increase M
                    accepted = False
                    for t in (0.4, 0.3, 0.2, 0.15, 0.1, 0.05):
                        f_candidate = (1.0 - t) * f + t * g
                        M_candidate = np.max(np.convolve(f_candidate, f_candidate))
                        if M_candidate <= M + 1e-12:
                            f = f_candidate
                            accepted = True
                            break
                    if not accepted:
                        # fallback to a small step
                        f = (1.0 - 0.05) * f + 0.05 * g
                    # evaluate
                    sc = evaluate_sequence(f.tolist())
                    if sc < best_score:
                        best_score = sc
                        best_seq = f.copy()
                        print(f"[LP] n={n} init={init_idx} it={lp_it} score={sc:.6f}")
                    # early exit if we are already below target
                    if best_score < 1.503:
                        break
                # after LP, maybe break if target reached
                if best_score < 1.503:
                    break

            # Local greedy refinement – run until time runs out or modest iteration budget
            # Estimate remaining time and allocate a share for this instance
            time_left = deadline - time.monotonic()
            # Allow up to 0.4 * remaining time for local search of this instance,
            # but cap the number of iterations to keep runtime reasonable.
            max_local_iters = int(1e6)  # large cap, actual stop by time
            f_refined, max_conv = local_search(f.copy(), target_sum, deadline, rng)
            # compute official score
            final_score = evaluate_sequence(f_refined.tolist())
            if final_score < best_score:
                best_score = final_score
                best_seq = f_refined.copy()
                print(f"[local] n={n} init={init_idx} score={final_score:.6f}")
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    if best_seq is None:
        # safety fallback: uniform sequence of length 256
        n = 256
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    # Convert to plain Python list of floats
    return [float(x) for x in best_seq]

1.  For several candidate lengths `n` (200 … 800) a few distinct
    initial sequences are built – a symmetric triangular shape, a uniform
    vector and a random vector – all normalised so that the ℓ¹‑norm equals
    √(2 n).  This normalisation makes the evaluation function equal to the
    worst entry of the auto‑convolution, i.e.  `max(f*f)`.

2.  If CVXPY (and a linear‑program solver) is available we perform a short
    LP‑based “mass‑inflation’’ phase.  For the current `f` we compute
    `M = max(f*f)` and solve

        maximise   Σ g_i
        subject to A·g ≤ M,   g ≥ 0

    where `A` is the convolution matrix (`(A·g)[k] = (f*g)[k]`).  The
    solution `g` is rescaled to have the same ℓ¹‑norm as `f` and mixed
    with `f` by a convex combination.  The mixing factor is chosen
    adaptively so that the new worst auto‑convolution entry does not
    increase.  This step typically lowers the bound quickly.

3.  A greedy local‑search phase follows.  At each iteration we look at the
    index `k*` where the current convolution attains its maximum.
    Moving a small amount of mass from a position `i` that contributes
    strongly to that entry to a position `j` that contributes little
    reduces the peak.  The move is accepted only if it strictly decreases
    the worst convolution value; otherwise it is rejected (with a tiny
    step‑size decay).  The convolution is updated incrementally in O(n)
    time, and the exact convolution is recomputed periodically for
    numerical stability.

4.  Throughout the search the best sequence (lowest score) is recorded.
    The algorithm aborts as soon as the target value 1.503 is reached or
    the allocated time budget expires.

The function `propose_candidate` returns the best sequence found as a
plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: build the convolution matrix A such that (A @ g)[k] = (f * g)[k]
def conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            col = k - i
            A[k, col] = f[i]
    return A


# ----------------------------------------------------------------------
# LP sub‑routine: given f and the current worst convolution M, maximise ℓ¹‑norm
def lp_optimize(f: np.ndarray, M: float, target_sum: float):
    import cvxpy as cp

    n = f.shape[0]
    A = conv_matrix(f)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is the default fast dense LP solver
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000)

    if g.value is None:
        return None

    g0 = np.maximum(g.value, 0.0)
    sum_g = g0.sum()
    if sum_g <= 1e-12:
        return None
    # Rescale so that the ℓ¹‑norm matches the target ( √(2 n) )
    g_rescaled = g0 * (target_sum / sum_g)
    return g_rescaled


# ----------------------------------------------------------------------
# Generate a symmetric triangular sequence (height 0 at the ends)
def generate_triangle(n: int, target_sum: float) -> np.ndarray:
    """Triangular shape, ℓ¹‑norm = target_sum."""
    centre = (n - 1) / 2.0
    d = np.abs(np.arange(n) - centre)
    heights = np.maximum(0.0, (n / 2.0) - d)          # zero at the borders
    s = heights.sum()
    if s == 0:
        return np.zeros(n, dtype=float)
    return heights * (target_sum / s)


# ----------------------------------------------------------------------
def generate_uniform(n: int, target_sum: float) -> np.ndarray:
    """Uniform vector, ℓ¹‑norm = target_sum."""
    return np.full(n, target_sum / n, dtype=float)


# ----------------------------------------------------------------------
def generate_random(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Random non‑negative vector, ℓ¹‑norm = target_sum."""
    a = rng.random(n)
    a = a / a.sum() * target_sum
    return a


# ----------------------------------------------------------------------
# Greedy local search that preserves the ℓ¹‑norm.
def local_search(f: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 rng: np.random.Generator):
    """Reduce max(f*f) by moving tiny amounts of mass from indices that
    heavily contribute to the current worst convolution entry to indices
    that contribute little.
    """
    n = f.shape[0]
    conv = np.convolve(f, f)                    # full convolution, length 2n‑1
    best_f = f.copy()
    best_max = conv.max()
    step_factor = 0.2                           # controls the size of each move
    it = 0

    # pre‑allocate a buffer for the contribution vectors (2n‑1 long)
    buf_len = 2 * n - 1

    while time.monotonic() < deadline:
        if it >= 200_000:                       # safety guard
            break

        k_star = int(np.argmax(conv))

        # derivative d_i = f[k*‑i] (zero outside the range)
        idx = np.arange(n)
        paired = k_star - idx
        valid = (paired >= 0) & (paired < n)
        d = np.zeros(n, dtype=float)
        d[valid] = f[paired[valid]]

        # pick a source index i with large derivative and positive mass
        high_mask = (d > np.percentile(d, 80)) & (f > 1e-12)
        if np.any(high_mask):
            cand_i = np.where(high_mask)[0]
            i = int(cand_i[rng.integers(cand_i.size)])
        else:
            # fallback: the highest derivative among positions that have mass
            mask = f > 1e-12
            if np.any(mask):
                cand_i = np.where(mask)[0]
                i = int(cand_i[rng.integers(cand_i.size)])
            else:
                break

        # pick a sink index j with small derivative
        low_mask = d < np.percentile(d, 20)
        if np.any(low_mask):
            cand_j = np.where(low_mask)[0]
            j = int(cand_j[rng.integers(cand_j.size)])
        else:
            # any index different from i
            j = int(rng.integers(n))
            while j == i:
                j = int(rng.integers(n))

        max_delta = f[i]
        if max_delta <= 1e-12:
            continue

        delta = step_factor * max_delta * rng.random()
        if delta <= 0:
            continue

        # build the contribution vectors for i and j (length 2n‑1)
        contrib_i = np.zeros(buf_len, dtype=float)
        contrib_i[i:i + n] = f
        contrib_j = np.zeros(buf_len, dtype=float)
        contrib_j[j:j + n] = f

        new_conv = conv + 2.0 * delta * (contrib_j - contrib_i)
        new_max = new_conv.max()

        if new_max < best_max - 1e-12:
            # accept the move
            f[i] -= delta
            f[j] += delta
            conv = new_conv
            best_max = new_max
            best_f = f.copy()
            step_factor = min(0.4, step_factor * 1.02)   # slowly enlarge step
        else:
            # reject – shrink step size a bit
            step_factor = max(0.01, step_factor * 0.99)

        it += 1

        # occasional exact recomputation to avoid drift
        if (it % 5000) == 0:
            conv = np.convolve(f, f)

    return best_f, best_max


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """Return a sequence of non‑negative heights with a low evaluation score.

    The algorithm explores several lengths, uses an LP‑based ℓ¹‑mass
    inflation phase (when CVXPY is available) and finishes with a greedy
    local refinement that moves mass between positions to lower the peak
    of the auto‑convolution.  The best sequence seen before the time budget
    expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # Detect CVXPY once
    try:
        import cvxpy  # noqa: F401
        HAS_CVX = True
    except Exception:
        HAS_CVX = False

    # Lengths to be explored – larger n give a finer bound but cost more
    candidate_lengths = [200, 300, 400, 500, 600, 800]

    for n in candidate_lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # Three different initialisers
        initialisers = [
            lambda: generate_triangle(n, target_sum),
            lambda: generate_uniform(n, target_sum),
            lambda: generate_random(n, target_sum, rng)
        ]

        for init_id, init_func in enumerate(initialisers):
            if time.monotonic() >= deadline:
                break

            f = init_func()
            np.clip(f, 0.0, 1000.0, out=f)               # respect the evaluator limits

            # baseline score
            scr = evaluate_sequence(f.tolist())
            if scr < best_score:
                best_score = scr
                best_seq = f.copy()
                print(f"[init] n={n} init={init_id} score={scr:.6f}")

            # ------- LP phase -------------------------------------------------
            if HAS_CVX:
                # number of LP iterations – scale down for larger n
                max_lp_iters = max(5, int(30 * 200 / n))
                for lp_it in range(max_lp_iters):
                    if time.monotonic() >= deadline:
                        break
                    M = np.max(np.convolve(f, f))
                    g = lp_optimize(f, M, target_sum)
                    if g is None:
                        break

                    # try several mixing weights; keep the first that does not increase M
                    mixed = None
                    for t in (0.4, 0.3, 0.2, 0.15, 0.1, 0.05):
                        candidate = (1.0 - t) * f + t * g
                        M_cand = np.max(np.convolve(candidate, candidate))
                        if M_cand <= M + 1e-12:
                            mixed = candidate
                            break
                    if mixed is None:
                        # fallback to a very small step
                        mixed = (1.0 - 0.05) * f + 0.05 * g
                    f = mixed

                    sc = evaluate_sequence(f.tolist())
                    if sc < best_score:
                        best_score = sc
                        best_seq = f.copy()
                        print(f"[LP] n={n} init={init_id} it={lp_it} score={sc:.6f}")

                    if best_score < 1.503:                # target reached – stop early
                        break
                if best_score < 1.503:
                    break

            # ------- Greedy local refinement ---------------------------------
            # Run until the global deadline is reached (or a reasonable number of
            # iterations is performed inside the helper).
            f_refined, max_conv = local_search(f.copy(), target_sum, deadline, rng)
            final_score = evaluate_sequence(f_refined.tolist())
            if final_score < best_score:
                best_score = final_score
                best_seq = f_refined.copy()
                print(f"[local] n={n} init={init_id} score={final_score:.6f}")

            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    # Fallback – deterministic safe sequence
    if best_seq is None:
        n = 256
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    # Return a plain Python list of floats as required
    return [float(x) for x in best_seq]
