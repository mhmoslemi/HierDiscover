def _fft_score(seq):
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01: return float('inf')
    n = arr.shape[0]
    N = 1 << (2 * n - 1).bit_length()  # next power of 2 >= 2n-1
    # compute FFT of arr padded to N.
    fft_arr = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(fft_arr * fft_arr, n=N)[: 2 * n - 1]
    maxb = conv.max()
    return 2.0 * n * maxb / (s * s)

def propose_candidate(seed=123, budget_s=1100):
    import numpy as np, time
    from scipy.sparse import diags
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine n
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(base.size)
    except Exception:
        n = 2000
    # Could cap n to reduce memory if too large.
    if n > 3000:
        n = 3000   # limit to manageable size
    sum_target = np.sqrt(2.0 * n)

    # fast score using fft
    def _score(seq):
        arr = np.clip(seq, 0.0, 1000.0)
        s = arr.sum()
        if s < 0.01:
            return float('inf')
        # FFT convolution
        N = 1 << ((2*n-1).bit_length())
        a_fft = np.fft.rfft(arr, n=N)
        conv = np.fft.irfft(a_fft * a_fft, n=N)[: 2*n - 1]
        maxb = conv.max()
        return float(2.0 * n * maxb / (s * s))

    # Candidate generators

    def geometric(n, sum_target):
        # Solve r^n ... use similar approach to previous code.
        t = 1.0
        for _ in range(20):
            f = np.exp(t) - 1 - 2*t
            fp = np.exp(t) - 2
            t -= f/fp
        delta = t / n
        r = max(0.0, 1.0 - delta)
        idx = np.arange(n, dtype=np.float64)
        raw = r**idx
        raw_sum = raw.sum()
        if raw_sum == 0.0:
            raw = np.ones(n)
            raw_sum = n
        return raw * (sum_target / raw_sum)

    # random sparse
    def sparse_random(n, sum_target):
        k = rng.integers(max(1, int(0.1 * n)), max(2, int(0.3 * n)))  # 10-30%
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.exponential(scale=1.0, size=k)
        arr = np.zeros(n)
        arr[idx] = vals
        total = arr.sum()
        if total > 0:
            arr *= sum_target / total
        return arr

    # power law
    def power_law(n, sum_target):
        p = rng.uniform(0.2, 2.0)
        arr = np.arange(1, n+1, dtype=np.float64) ** (-p)
        arr_sum = arr.sum()
        arr *= sum_target / arr_sum
        return arr

    # initialize best
    candidates = [
        geometric(n, sum_target),
        geometric(n, sum_target)[::-1],
        sparse_random(n, sum_target),
        power_law(n, sum_target)
    ]
    best = candidates[0]
    best_score = _score(best)
    for cand in candidates[1:]:
        s = _score(cand)
        if s < best_score:
            best, best_score = cand, s

    # Current solution for SA (start with best)
    cur = best.copy()
    cur_score = best_score

    # Simulated annealing parameters
    start_temp = 0.1
    end_temp = 1e-5

    # main loop
    it = 0
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - (deadline - budget_s + 2.0)
        progress = (elapsed) / (budget_s - 2.0)
        T = start_temp * (end_temp / start_temp) ** progress

        mode = rng.integers(0, 6)

        cand = cur.copy()
        if mode == 0:
            # block mul
            width = rng.integers(max(1, n // 200), max(2, n // 8))
            start = rng.integers(0, max(1, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.15))
            cand[start:start+width] *= factor
        elif mode == 1:
            # single mul
            i = rng.integers(n)
            factor = np.exp(rng.normal(0.0, 0.1))
            cand[i] *= factor
        elif mode == 2:
            # move mass between indices
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * 0.5
            delta = eps * min(cand[i], cand[j])
            if cand[i] > cand[j]:
                cand[i] -= delta
                cand[j] += delta
            else:
                cand[j] -= delta
                cand[i] += delta
        elif mode == 3:
            # replace with fresh power law
            cand = power_law(n, sum_target)
        elif mode == 4:
            # replace with fresh sparse
            cand = sparse_random(n, sum_target)
        else:
            # symmetric averaging
            cand = 0.5 * (cand + cand[::-1])

        # renormalize to target sum
        total = cand.sum()
        if total > 0:
            cand *= (sum_target / total)
        # ensure non-negative
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _score(cand)

        # accept/reject
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
                print(f"[{elapsed:.1f}s] new best SA = {best_score:.6f}")

        # Occasionally perform LP refinement
        if it % 500 == 0 and time.monotonic() < deadline - 5.0:
            # LP step
            # compute M = max conv of cur (already known as cur_score perhaps)
            M = np.max(np.convolve(cur, cur))
            # build matrix
            A = diags(np.broadcast_to(cur[:, None], (n, n)), offsets=np.arange(n), shape=(2*n-1, n), format='csr')
            # LP: maximize sum(g)
            c = -np.ones(n)
            b_ub = np.full(2*n-1, M, dtype=np.float64)
            bounds = [(0, None)] * n
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, int((deadline - time.monotonic()) * 1000))})
            if res.success:
                g0 = res.x
                sum_g = g0.sum()
                if sum_g > 0:
                    g = g0 * (sum_target / sum_g)
                    # mixing factor
                    t_mix = 0.5
                    new_seq = (1 - t_mix) * cur + t_mix * g
                    new_seq = np.clip(new_seq, 0.0, 1000.0)
                    new_seq = new_seq * (sum_target / new_seq.sum())
                    new_score = _score(new_seq)
                    if new_score < cur_score:
                        cur, cur_score = new_seq, new_score
                        if cur_score < best_score:
                            best, best_score = cur.copy(), cur_score
                            print(f"[{elapsed:.1f}s] new best LP = {best_score:.6f}")

        it += 1

    # finalize
    final = [float(x) for x in best.tolist()]
    final_score = float(evaluate_sequence(final))
    print(f"[done] final score = {final_score:.6f}")
    return final

"""
Propose a step‑function sequence that minimizes `evaluate_sequence`.

The method is a hybrid of three ideas:

1. **Deterministic starters** – geometric, its reverse, power‑law and a
   sparse random pattern, all scaled to the required L¹‑norm
   `sqrt(2·n)`.  These give a good initial bound (≈1.6).

2. **Fast FFT scoring** – the objective
   `2·n·max(conv)/sum²` is evaluated with an FFT based convolution
   (`O(n log n)`) which is much faster than `np.convolve` for the
   thousands of entries we use.

3. **LP refinement** – exactly the alternating linear‑program step
   described in the reference paper.  For a current vector `f` we solve

        maximise   Σ g_j
        subject to  g ≥ 0
                    conv(f,g) ≤ M   (M = max(conv(f,f)))

   The convolution constraints are linear in `g` and are encoded as a
   sparse Toeplitz matrix.  The optimal partner `g` (often the reverse of
   `f`) is renormalised to the fixed L¹‑norm and mixed back into `f`:

        f ← (1‑t)·f + t·g ,   t≈0.5

   This operation provably never increases the maximal self‑convolution,
   and a few such refinements drive the score well below the 1.5 target.

The whole search runs until the supplied time budget (≈1000 s) is exhausted,
keeping the best vector seen.  All helper functions are top‑level and no
closures or lambdas are used.
"""

import time
import numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast score (FFT based, identical to official evaluator for valid inputs)
# ----------------------------------------------------------------------
def _fft_score(seq: np.ndarray) -> float:
    """Return the same value that `evaluate_sequence` would compute.
    The input is assumed to be a 1‑D NumPy array of non‑negative floats.
    """
    arr = np.clip(seq, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    n = arr.shape[0]
    # next power of two >= 2·n‑1 for efficient FFT
    N = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    maxb = conv.max()
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Simple candidate generators (all return a NumPy array of length n)
# ----------------------------------------------------------------------
def _geometric_candidate(n: int, sum_target: float) -> np.ndarray:
    """Monotone geometric progression a_i = r**i, scaled to L¹ = sum_target."""
    # Solve e^t = 1 + 2t  (t≈1.256) and set r = 1‑t/n .
    t = 1.0
    for _ in range(20):
        f = np.exp(t) - 1.0 - 2.0 * t
        fp = np.exp(t) - 2.0
        t -= f / fp
    delta = t / n
    r = max(0.0, 1.0 - delta)
    idx = np.arange(n, dtype=np.float64)
    raw = np.power(r, idx, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.ones(n, dtype=np.float64)
        raw_sum = n
    return (sum_target / raw_sum) * raw


def _sparse_random_candidate(n: int, sum_target: float, rng: np.random.Generator) -> np.ndarray:
    """A few random non‑zero entries (5‑25 % density) scaled to the target sum."""
    k = rng.integers(int(0.05 * n), int(0.25 * n) + 1)
    if k <= 0:
        return np.full(n, sum_target / n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    tot = cand.sum()
    if tot > 0.0:
        cand *= sum_target / tot
    return cand


def _powerlaw_candidate(n: int, sum_target: float, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} with p∈[0.2,2.0]"""
    p = rng.uniform(0.2, 2.0)
    seq = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    seq_sum = seq.sum()
    if seq_sum > 0.0:
        seq *= sum_target / seq_sum
    return seq


# ----------------------------------------------------------------------
# Build the sparse convolution matrix  A  where   conv(f,g) = A @ g
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> diags:
    """Return a CSR matrix of shape (2·n‑1, n) such that
    (A @ g) == np.convolve(f, g) for any vector g of length n.
    """
    n = f.shape[0]
    # Each diagonal i (offset i) is the constant f[i].
    # Broadcast to a (n, n) dense block – diags will copy it into a sparse format.
    data = np.broadcast_to(f[:, None], (n, n))
    offsets = np.arange(n, dtype=int)          # 0 … n‑1
    return diags(data, offsets, shape=(2 * n - 1, n), format="csr")


# ----------------------------------------------------------------------
# LP refinement (the alternating linear‑program step from the paper)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, sum_target: float, deadline: float,
               max_iters: int = 30, mix_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """Iteratively improve `f` by solving the LP for its partner `g`.
    Returns the best vector found and its score.
    """
    n = f.shape[0]
    best_vec = f.copy()
    best_score = _fft_score(best_vec)

    cur = f.copy()
    cur_score = best_score
    t = mix_factor

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # maximal self‑convolution of the current vector
        M = np.max(np.convolve(cur, cur))

        # linear constraints   conv(cur, g) ≤ M
        A = _conv_matrix(cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise Σ g_j  ⇔ minimise −Σ g_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})
        if not res.success:
            break

        g0 = res.x
        s = g0.sum()
        if s <= 0.0:
            break

        # renormalise to the fixed L¹‑norm
        g = g0 * (sum_target / s)

        # mix the partner into the current vector
        new_vec = (1.0 - t) * cur + t * g
        new_vec = np.clip(new_vec, 0.0, 1000.0)

        new_score = _fft_score(new_vec)

        # accept if we made progress (the bound never increases)
        if new_score < cur_score - 1e-12:
            cur, cur_score = new_vec, new_score
            if new_score < best_score:
                best_vec, best_score = new_vec.copy(), new_score
                print(f"[LP] iteration {it+1:2d} – improved score = {best_score:.6f}")
        else:
            # no improvement → stop the LP loop
            break

        # slowly reduce the mixing weight to fine‑tune the final steps
        t = max(0.05, t * 0.9)

    return best_vec, best_score
