```
def propose_candidate(seed=123, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Determine n

    iteration = 0

    # Precompute constants
    len_w = 2*n - 1
    size_conv = 1 << ((2*n-1).bit_length())
    size_corr = 1 << ((n + len_w - 1).bit_length())

    # Setup parameters
    step_init = 0.5
    step_decay = 0.9995
    lambda_init = 5.0
    lambda_inc = 0.02   # per iteration increase

    while time.monotonic() < deadline:
        # Choose initialization method depending on iteration
        if iteration % 5000 == 0:
            # start new random candidate
            a = rng.random(n)
            a = a / a.sum()
        # Gradient descent loop for a limited number of inner steps maybe?

        # Here we apply a certain number of inner updates before reinitialization
        # but we manage iteration counting.

        # compute conv via FFT
        A = np.fft.rfft(a, n=size_conv)
        conv = np.fft.irfft(A * A, n=size_conv)[:len_w]

        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration} | new best score {best_score:.6f}")

        # compute smoothed max weights
        lam = lambda_init + lambda_inc * iteration
        # but keep lam not too large for numerical stability
        lam = min(lam, 200.0)
        exp_weights = np.exp(lam * (conv - max_conv))

        # gradient via correlation
        A_large = np.fft.rfft(a, n=size_corr)
        W_large = np.fft.rfft(w, n=size_corr)
        corr = np.fft.irfft(A_large * np.conj(W_large), n=size_corr)
        g = 2.0 * corr[len_w-1 : len_w-1 + n]

        step = step_init * (step_decay ** iteration)
        # exponentiated gradient update
        a = a * np.exp(-step * g)
```

The score to minimise is  

\[
F(a)=\frac{2n\;\max\limits_{s}(a*a)[s]}{\bigl(\sum a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the score is homogeneous we may normalise the vector to the
probability simplex (`∑a_i = 1`).  
`(a*a)[s]` is the distribution of the sum of two independent draws from
`a`.  The problem is therefore to make this sum‑distribution as flat as
possible.

The algorithm below builds on the previous gradient‑descent approach but
uses a *smoothed* version of the `max` function:

*   For the current convolution `c = a*a` we compute soft‑max weights  

    \[
    w_s = \frac{e^{\lambda (c_s-\max c)}}{\sum_t e^{\lambda (c_t-\max c)}},
    \]

    which concentrate on the largest entry for large `λ` but are smooth
    for smaller `λ`.  
*   The gradient of the smoothed objective is  

    \[
    g_i = 2\sum_{s} w_s\,a_{s-i},
    \]

    i.e. a correlation of `a` with the weight vector `w`.  Both the
    convolution and the correlation are computed with a single FFT per
    iteration (complexity `O(n log n)`).
*   We perform an **exponentiated‑gradient (mirror descent) update**  

    \[
    a_i \leftarrow a_i\;e^{-\eta g_i},\qquad a\leftarrow\frac{a}{\sum a},
    \]

    which automatically keeps the vector non‑negative and on the simplex.
*   To avoid stagnation we occasionally
    * inject a cheap block perturbation,
    * restart from a random Dirichlet vector after a long non‑improving
      streak, and
    * keep the best sequence seen so far.

All operations are vectorised and use only NumPy/FFT, so thousands of
iterations fit comfortably into the 1000 s budget.

```python
"""
Exponentiated‑gradient optimiser for the max‑convolution score.

The algorithm works on the probability simplex (∑a = 1).  Each iteration
computes

    c = a * a                     (full convolution via FFT)
    w = soft‑max(c)               (λ‑smooth approximation of max)
    g = 2 * correlation(a, w)     (gradient of the smoothed objective)

and updates a by a mirror‑descent step
    a ← a·exp(−η·g) ,  a ← a/∑a.

Occasional block perturbations and random restarts improve exploration.
The best vector found before the time limit is returned as a plain list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def arcsine_sequence(n: int) -> np.ndarray:
    """Discrete arcsine distribution, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) perturbation that multiplies a random block by a factor."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (inclusive)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:start + width] *= factor
    # occasional mirroring to keep the shape balanced
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return cand


def next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Main optimisation routine expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    max‑convolution score.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size – use the previously discovered sequence if
    # it exists, otherwise fall back to a moderate length.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = int(base.size)
    except Exception:
        base = None
        n = 6000       # a generous size that still runs fast

    # Lengths required for FFTs
    len_w = 2 * n - 1                      # length of a*a
    size_fft = next_pow2(n + len_w - 1)    # ≥ n + (2n‑1) – 1 = 3n‑2

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    step_init = 0.30            # initial mirror‑descent step size η
    step_decay = 0.9997         # exponential decay of η
    lambda_const = 30.0         # smoothing parameter λ (soft‑max)
    max_no_improve = 5000       # restart after this many dead iterations

    # ------------------------------------------------------------------
    # Initial candidate – blend arcsine with the provided solution (if any)
    # ------------------------------------------------------------------
    a = arcsine_sequence(n)
    if base is not None and base.shape == a.shape:
        cur = np.maximum(base, 0.0)
        if cur.sum() > 0:
            cur = cur / cur.sum()
            a = 0.5 * a + 0.5 * cur
            a = a / a.sum()

    # --------------------------------------------------------------
    # Helper to evaluate the current score (fast, avoids calling the
    # grader's evaluate_sequence repeatedly)
    # --------------------------------------------------------------
    def current_score(vec: np.ndarray) -> float:
        """2·n·max(conv(vec,vec)) – vec is already normalised to sum = 1."""
        A = np.fft.rfft(vec, n=size_fft)
        conv = np.fft.irfft(A * A, n=size_fft)[:len_w]
        return float(2.0 * n * conv.max())

    # initialise best record
    best = a.copy()
    best_score = current_score(best)
    print(f"Initial best score {best_score:.6f}")

    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:
        # ---- 1) convolution of the current vector --------------------
        A_fft = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(A_fft * A_fft, n=size_fft)[:len_w]
        max_conv = conv.max()
        score = float(2.0 * n * max_conv)

        # ---- 2) record improvement ----------------------------------
        if score < best_score:
            best = a.copy()
            best_score = score
            no_improve = 0
            print(f"Iter {it:6d} | new best {best_score:.6f}")
        else:
            no_improve += 1

        # ---- 3) soft‑max weights for a smoothed max -----------------
        lam = lambda_const                     # fixed λ (could be annealed)
        # exp is safe because (conv - max_conv) ≤ 0
        exp_w = np.exp(lam * (conv - max_conv))
        w = exp_w / exp_w.sum()               # ∑ w = 1, all w ≥ 0

        # ---- 4) gradient = 2 * correlation(a , w) -----------------
        W_fft = np.fft.rfft(w, n=size_fft)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), n=size_fft)
        # linear correlation: indices len_w‑1 … len_w‑1+n‑1 correspond to
        # shifts 0 … n‑1
        g = 2.0 * corr[len_w - 1 : len_w - 1 + n]

        # ---- 5) mirror‑descent (exponentiated gradient) update -----
        η = step_init * (step_decay ** it)    # decreasing step size
        a = a * np.exp(-η * g)               # stay non‑negative
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s                          # back on the simplex

        # ---- 6) occasional diversification ---------------------------
        if it % 1000 == 0 and it > 0:
            pert = make_block_perturbation(best, rng)
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            pert_score = current_score(pert)
            if pert_score < best_score:
                best = pert.copy()
                best_score = pert_score
                a = pert.copy()
                no_improve = 0
                print(f"Iter {it:6d} | block‑perturb improved to {best_score:.6f}")

        # ---- 7) random restart if stuck for a long time ------------
        if no_improve > max_no_improve:
            a = rng.random(n)
            a = a / a.sum()
            no_improve = 0
            print(f"Iter {it:6d} | random restart")

        # ---- 8) bookkeeping -----------------------------------------
        it += 1
        if it % 2000 == 0:
            print(f"Iter {it:6d} | best {best_score:.6f} | step {η:.6f}")

    # ------------------------------------------------------------------
    # Final report and conversion to plain Python list
    # ------------------------------------------------------------------
    print(f"Finished after {it} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best]
```
