"""
Stochastic optimizer for the step‑function evaluation problem.

Features:
- Warm‑start from several parametric families (power‑law, exponential,
  symmetric mirroring).
- Main loop uses a sub‑gradient descent step derived from the
  objective's sub‑gradient (the index where the convolution attains its
  maximum) combined with occasional random perturbations (block scaling,
  swap, noise, flatten) for exploration.
- After each move the sequence is re‑scaled to keep the total sum
  constant (the objective is homogeneous), and is forced to be symmetric
  – this reduces the search space and often improves the bound.
- Fast FFT‑based estimator is used to filter candidates; the exact
  evaluate_sequence() is called only when a candidate appears promising.
- The algorithm respects the supplied time budget, printing progress
  periodically and finally returning the best sequence found.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _conv_max(seq: np.ndarray) -> float:
    """Fast computation of max(conv(seq, seq)) using FFT."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # keep only the linear part
    return float(conv[: 2 * n - 1].max())

def _score_estimate(seq: np.ndarray) -> float:
    """Quick estimator of the evaluation function using FFT."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)

# ----------------------------------------------------------------------
# Random perturbation moves (same as previous solution, kept for diversity)
# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one position to another."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)

def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add small Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)

def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Reduce the contribution of the currently worst convolution index."""
    cand = seq.copy()
    n = cand.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]
    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand
    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))
    delta = cand[src] * float(0.05 + 0.15 * rng.random())
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)

def _gradient_descent_step(seq: np.ndarray, step: float) -> np.ndarray:
    """One sub‑gradient step w.r.t. the objective (keeping total sum constant)."""
    n = seq.size
    sum_seq = float(seq.sum())
    if sum_seq < 0.01:
        return seq
    # compute convolution via FFT (real)
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    k_max = int(conv.argmax())

    # compute partner contributions r_i = a_{k_max-i} where defined, else 0
    r = np.zeros_like(seq)
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    if i_max >= i_min:
        idx = np.arange(i_min, i_max + 1, dtype=int)
        partner = k_max - idx
        r[idx] = seq[partner]

    const = max_b / sum_seq
    factor = (4.0 * n) / (sum_seq * sum_seq)
    grad = factor * (r - const)   # for indices outside the interval r=0 already
    # gradient step (subtract gradient)
    cand = seq - step * grad
    # keep within allowed domain
    cand = np.clip(cand, 0.0, 1000.0)

    # renormalise to the original sum (objective is homogeneous)
    new_sum = float(cand.sum())
    if new_sum < 0.01:
        # discard the move if it collapses mass
        return seq
    cand *= (sum_seq / new_sum)
    # enforce symmetry (helps to keep convolution flat)
    cand = 0.5 * (cand + cand[::-1])
    return cand

"""
Stochastic optimiser for the step‑function evaluation problem.

The optimiser works in three stages:

1. **Warm‑start** – several analytic families (power‑law, exponential,
   Gaussian‑like) are tested.  The best candidate is kept as the initial
   sequence.  The sequence is mirrored to enforce symmetry and normalised
   to √(2 n) (the objective is homogeneous, any scaling leaves the score
   unchanged).

2. **Main loop** – a mix of a sub‑gradient descent step (derived from the
   sub‑gradient of the objective at the index where the self‑convolution
   attains its maximum) and a set of cheap random perturbations
   (block scaling, swap, noise, flatten).  After each move the sequence is
   re‑scaled to keep its total sum constant and forced to be symmetric.

   A fast FFT‑based estimator is used to compute the objective cheaply.
   Candidates that improve the current estimate are accepted; a mild
   simulated‑annealing rule also allows occasional uphill moves.
   Whenever a candidate beats the best exact score seen so far,
   ``evaluate_sequence`` is called and the best sequence is updated.

3. **Time‑budget handling** – the loop runs until the supplied time budget
   (minus a small safety margin) is exhausted, printing progress
   periodically.  The best sequence found is returned as a plain list of
   floats.

All helper functions are defined at module level, there are no closures,
and the code respects the constraints of the challenge.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Fast max of the linear convolution seq * seq using FFT."""
    n = seq.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(conv[: 2 * n - 1].max())


def _score_estimate(seq: np.ndarray) -> float:
    """FFT‑based estimate of the objective (the exact evaluate_sequence is O(n²))."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.size
    max_b = _conv_max(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Random perturbation moves (kept for exploration)
# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # optional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _perturb_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    cand = seq.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * float(rng.random() * 0.2)
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def _perturb_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add small zero‑mean Gaussian noise to a random sub‑block."""
    cand = seq.copy()
    n = cand.size
    block = int(rng.integers(max(1, n // 100), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - block + 1)))
    stop = start + block
    scale = 0.05 * cand[start:stop].mean()
    if scale <= 0.0:
        scale = 0.01
    cand[start:stop] += rng.normal(0.0, scale, size=block)
    return np.clip(cand, 0.0, 1000.0)


def _flatten_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Reduce the contribution of the currently worst convolution index."""
    cand = seq.copy()
    n = cand.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k_max = int(conv.argmax())
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k_max - idx
    contrib = cand[idx] * cand[partner]
    src = idx[int(np.argmax(contrib))]
    if cand[src] <= 0.0:
        return cand
    forbidden = {src, k_max - src}
    dst_candidates = np.setdiff1d(np.arange(n), list(forbidden))
    if dst_candidates.size == 0:
        return cand
    dst = int(rng.choice(dst_candidates))
    delta = cand[src] * float(0.05 + 0.15 * rng.random())
    cand[src] -= delta
    cand[dst] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Sub‑gradient descent step
# ----------------------------------------------------------------------
def _gradient_descent_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    Perform one sub‑gradient step for the objective

        F(a) = 2 n * max_k (a*a)[k] / (sum a)^2 .

    The sub‑gradient is taken at the index k where the convolution attains
    its maximum.  After the step the total sum is re‑scaled to its original
    value and symmetry is enforced.
    """
    n = seq.size
    sum_seq = float(seq.sum())
    if sum_seq < 0.01:
        return seq

    # convolution via FFT (linear part)
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    k_max = int(conv.argmax())

    # partner values r_i = a_{k_max - i} where defined, otherwise 0
    r = np.zeros_like(seq)
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    if i_max >= i_min:
        idx = np.arange(i_min, i_max + 1, dtype=int)
        partner = k_max - idx
        r[idx] = seq[partner]

    const = max_b / sum_seq
    factor = (4.0 * n) / (sum_seq * sum_seq)
    grad = factor * (r - const)               # outside the interval r is 0

    cand = seq - step * grad
    cand = np.clip(cand, 0.0, 1000.0)

    new_sum = float(cand.sum())
    if new_sum < 0.01:
        return seq
    cand *= (sum_seq / new_sum)               # keep the sum unchanged

    # Enforce symmetry – this often flattens the convolution.
    cand = 0.5 * (cand + cand[::-1])
    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the allotted time budget, prints progress, and
    returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # a small safety margin to avoid overrunning the allocated time
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start: analytic families
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_seq = None

    if base_seq is None or base_seq.ndim != 1 or base_seq.size == 0 or not np.isfinite(base_seq).all():
        # fallback: a modest power‑law of length 2048
        n0 = 2048
        base_seq = np.power(np.arange(n0, dtype=np.float64) + 1.0, -0.45)

    base_seq = np.clip(base_seq, 0.0, 1000.0)

    # Choose a manageable length (original construction is 4806)
    target_len = max(2000, min(5000, base_seq.size))
    if base_seq.size != target_len:
        if base_seq.size > target_len:
            base_seq = base_seq[:target_len].copy()
        else:
            pad = np.zeros(target_len - base_seq.size, dtype=np.float64)
            base_seq = np.concatenate([base_seq, pad])

    n = base_seq.size
    desired_sum = math.sqrt(2.0 * n)
    cur_sum = float(base_seq.sum())
    if cur_sum < 0.01:
        base_seq[:] = 1.0
        cur_sum = float(base_seq.sum())
    base_seq *= (desired_sum / cur_sum)

    # enforce symmetry from the start
    base_seq = 0.5 * (base_seq + base_seq[::-1])

    best_seq = base_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    best_est = _score_estimate(best_seq)

    # ------------------------------------------------------------------
    # Test a few analytical families (power‑law, exponential, Gaussian‑like)
    # ------------------------------------------------------------------
    families = []

    # power‑law a_i = (i+1)^{-p}
    for p in np.linspace(0.30, 0.65, 9):
        idx = np.arange(n, dtype=np.float64) + 1.0
        families.append(np.power(idx, -p))

    # exponential a_i = exp(-p * i)
    for p in np.linspace(0.001, 0.015, 8):
        idx = np.arange(n, dtype=np.float64)
        families.append(np.exp(-p * idx))

    # Gaussian‑like a_i = exp(-p * i^2)
    for p in np.linspace(0.0005, 0.005, 7):
        idx = np.arange(n, dtype=np.float64)
        families.append(np.exp(-p * idx * idx))

    for cand in families:
        cand = np.clip(cand, 0.0, 1000.0)
        s = float(cand.sum())
        if s < 0.01:
            continue
        cand *= (desired_sum / s)
        cand = 0.5 * (cand + cand[::-1])
        est = _score_estimate(cand)
        if est < best_est - 1e-9:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_seq = cand.copy()
                best_score = exact
                best_est = est
                print(f"[warm‑start] new best exact {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Main optimisation loop
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_est = best_est

    step = 0.5                # initial gradient step size
    min_step = 1e-4
    iteration = 0
    report_interval = 5000
    move_funcs = [_make_block_perturbation,
                  _perturb_swap,
                  _perturb_noise,
                  _flatten_move]

    # simple temperature schedule for occasional uphill moves
    total_budget = float(budget_s)

    while time.monotonic() < deadline:
        # ------- generate a candidate -------
        if rng.random() < 0.65:
            cand = _gradient_descent_step(cur_seq, step)
        else:
            cand = rng.choice(move_funcs)(cur_seq, rng)

        # enforce symmetry and keep the total sum constant
        cand = 0.5 * (cand + cand[::-1])
        s = float(cand.sum())
        if s < 0.01:
            iteration += 1
            continue
        cand *= (desired_sum / s)

        # fast estimate of the objective
        est = _score_estimate(cand)

        # simulated‑annealing style acceptance for the *current* sequence
        temperature = 0.05 + 0.25 * max(0.0, (deadline - time.monotonic())) / total_budget
        accept = False
        if est < cur_est:
            accept = True
        else:
            # probability of accepting a worsening move
            delta = est - cur_est
            # guard against overflow in exp()
            if delta / temperature < 700:
                prob = math.exp(-delta / temperature)
            else:
                prob = 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = cand
            cur_est = est

        # ------- improvement over the *best* sequence -------
        if est < best_est - 1e-12:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_seq = cand.copy()
                best_score = exact
                best_est = est
                print(f"[{iteration:07d}] new best exact {best_score:.6f}")

        # decay the gradient step size periodically
        if iteration % 20000 == 0 and iteration > 0:
            step = max(min_step, step * 0.85)

        iteration += 1
        if iteration % report_interval == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{iteration:07d}] time {elapsed:.1f}s – best est {best_est:.6f}, best exact {best_score:.6f}")

    total_elapsed = time.monotonic() - start_time
    print(f"Finished after {iteration} iterations ({total_elapsed:.1f}s) – best exact = {best_score:.6f}")
    return [float(x) for x in best_seq]
