"""
Iterative LP flattening + stochastic fine‑tuning.

The algorithm starts from a 1/√i (arcsine) shape, which already yields a
ratio ≈ π/2.  It then repeatedly solves a linear program that, for the
current sequence f, finds a non‑negative sequence g with the largest
possible ℓ¹‑norm under the constraint (f∗g) ≤ M, where M is the current
maximum of the self‑convolution f∗f.  Scaling g back to the original
ℓ¹‑norm and mixing it with f strictly reduces the maximum convolution.
After a few LP steps we continue with cheap block‑perturbation hill‑climbing
to squeeze out any remaining improvement.

The LP is linear because (f∗g) is linear in g (f is fixed).  The
convolution matrix A has shape (2n‑1)×n and is built on the fly.
All work respects the 1000 s budget and the contest’s safety checks.
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper: fast max of self‑convolution (uses FFT, works for any non‑negative seq)
def _max_self_convolution(seq: np.ndarray) -> float:
    """Return max((seq * seq) convolution) for a real sequence."""
    n = seq.size
    # Length needed for linear convolution
    size = int(2 ** math.ceil(math.log2(2 * n - 1)))
    fft_seq = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_seq * fft_seq, size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Helper: build the convolution matrix A such that A @ g = f ∗ g
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return dense matrix A (shape (2n‑1, n)) with (A @ g) = f ∗ g."""
    n = f.size
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n), dtype=np.float64)
    # Each column i of A is f shifted by i positions
    for i in range(n):
        A[i : i + n, i] = f
    return A


# ----------------------------------------------------------------------
# Helper: cheap block perturbation (same as in the reference code)
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # block width between n/200 and n/12 (inclusive)
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal jitter
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring – helps retain symmetry
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce the bounds required by evaluate_sequence
    return np.clip(candidate, 0.0, 1000.0)
