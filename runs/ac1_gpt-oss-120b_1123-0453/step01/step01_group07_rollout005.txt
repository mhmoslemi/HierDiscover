def propose_candidate(seed=0, budget_s=1100):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0, budget_s-2)

    # Choose length n; use existing height_sequence_1 length or 5000.
    n = 5000 (or dynamic). Use vector a.

    # Initialize a as power law with exponent random between 0.3 and 0.6 to explore.
    exp0 = rng.uniform(0.35, 0.55)
    off0 = rng.uniform(0., 2.)
    a = generate_power_sequence(n, exponent=exp0, offset=off0)
    a = a / np.sum(a)  # normalize to sum=1

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    iteration = 0

    while time.monotonic() < deadline:
        # subgradient descent step
        b = np.convolve(a, a)
        maxb = b.max()
        # find active shifts within tolerance relative to maxb
        eps = maxb * 1e-12 + 1e-12  # absolute tolerance
        active = np.where(b >= maxb - eps)[0]
        # compute gradient g
        g = np.zeros_like(a)
        for k in active:
            # shift +k
            if k > 0:
                g[:n - k] += a[k:]  # a[i+k] term for i from 0 to n-k-1
                g[k:] += a[:n - k]  # a[i-k] term for i from k to n-1
            else:
                # k=0 case: derivative = 2*a_i
                g += 2 * a
        # Scale gradient maybe by 1/len(active) to average
        if len(active) > 0:
            g = g / len(active)
        # Choose step size decreasing: eta = eta0 / sqrt(iter+1)
        eta0 = 0.1
        eta = eta0 / np.sqrt(iteration + 1)
        # Update
        a_new = a - eta * g
        # Project to nonnegative
        a_new = np.maximum(a_new, 0.0)
        # Renormalize sum to 1
        sum_a = a_new.sum()
        if sum_a < 1e-12:
            # reinitialize if collapse
            a_new = generate_power_sequence(n, exponent=exp0, offset=off0)
            a_new = a_new / np.sum(a_new)
        else:
            a_new /= sum_a

        # Evaluate
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"iter {iteration} new best score {score:.6f}")
        # Accept new a even if not improvement? For descent maybe only accept if improvement to avoid divergence
        if np.isfinite(score) and score <= best_score + 1e-12:
            a = a_new
        # else maybe keep old a; we can also occasionally accept if not improvement with probability (annealing)
        else:
            # with small prob accept worse to escape
            if rng.random() < 0.01:
                a = a_new

        # occasional random restart
        if rng.random() < 0.002:
            exp0 = rng.uniform(0.3, 0.6)
            off0 = rng.uniform(0., 2.)
            a = generate_power_sequence(n, exponent=exp0, offset=off0)
            a /= np.sum(a)

        iteration += 1

    # return best sequence as list
    return best_a.tolist()

"""
Propose a step‑function sequence that (hopefully) beats the previous
bound of ≈1.576.  The search combines three complementary ideas:

1.  A broad random sweep over power‑law tails (different exponents
    and offsets) to find a good seed.
2.  A projected sub‑gradient descent on the *maximum* autocorrelation
    (the true quantity minimised by ``evaluate_sequence`` after normalising
    the ℓ¹‑norm to 1).  The gradient is obtained analytically for the
    active shift(s) where the convolution peak occurs.
3.  A long stochastic hill‑climbing phase that applies several
    perturbation operators (block rescaling, global log‑normal noise and
    a simple “mass‑swap” move).  Improvements are kept and occasional
    random restarts keep the search from stagnating.

All steps respect the 1000 s time budget, use only NumPy (no I/O)
and return a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def generate_power_sequence(
    n: int, exponent: float = 0.5, offset: float = 0.0
) -> np.ndarray:
    """Return a non‑negative sequence a_i = (i+offset+1)^(-exponent)."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    seq = idx ** (-exponent)
    # the evaluator clips to [0,1000]; we keep the raw values.
    return seq


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.06)))  # ~6 % log‑normal change
    new = seq.copy()
    new[start:stop] *= factor
    if rng.random() < 0.1:                     # occasional symmetrisation
        new = 0.5 * (new + new[::-1])
    return new


def global_noise(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small multiplicative log‑normal noise applied to the whole vector."""
    noise = rng.normal(0.0, 0.02, size=seq.shape)
    factor = np.exp(noise)
    return seq * factor


def mass_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one index to another while
    preserving the ℓ¹‑norm.
    """
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j or seq[i] == 0.0:
        return seq.copy()
    # move at most 30 % of the source entry (still tiny in absolute terms)
    delta = float(rng.uniform(0.0, min(seq[i], 0.3 * seq[i])))
    new = seq.copy()
    new[i] -= delta
    new[j] += delta
    return new


def fast_score(seq: np.ndarray, two_n: float) -> float:
    """
    Fast proxy for ``evaluate_sequence`` when the ℓ¹‑norm of *seq* is 1.
    Returns 2·n·max(a*a) which equals the official score in that case.
    """
    b = np.convolve(seq, seq)
    return two_n * b.max()


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # safety margin – leave a couple of seconds for the caller
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # Choose a problem size.  Prefer the length of the previously
    # discovered sequence (if present) otherwise fall back to 6000.
    # --------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))   # type: ignore[name-defined]
    except Exception:
        n = 6000

    two_n = 2.0 * n

    # --------------------------------------------------------------
    # 1) Initialise from a random power‑law seed.
    # --------------------------------------------------------------
    exp0 = rng.uniform(0.35, 0.55)
    off0 = rng.uniform(0.0, 5.0)
    a = generate_power_sequence(n, exponent=exp0, offset=off0)
    a /= a.sum()                     # ℓ¹‑norm = 1  (scale‑invariant)

    best_seq = a.copy()
    best_score = fast_score(best_seq, two_n)
    print(f"initial score {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Broad random sweep over exponent/offset (cheap fast_score).
    # --------------------------------------------------------------
    # Roughly 10 % of the time budget is spent here.
    sweep_deadline = start_time + 0.1 * (deadline - start_time)
    while time.monotonic() < sweep_deadline:
        for _ in range(30):          # batch of proposals
            exp = rng.uniform(0.30, 0.60)
            off = rng.uniform(0.0, 6.0)
            cand = generate_power_sequence(n, exponent=exp, offset=off)
            cand /= cand.sum()
            sc = fast_score(cand, two_n)
            if sc < best_score:
                best_seq = cand.copy()
                best_score = sc
                print(f"grid improve {best_score:.6f} exp={exp:.4f} off={off:.2f}")
        # periodically check the global deadline
        if time.monotonic() >= deadline:
            break

    # --------------------------------------------------------------
    # 3) Projected sub‑gradient descent on the peak autocorrelation.
    # --------------------------------------------------------------
    # We keep this phase short (≈15 % of the total budget) but it
    # often drops the peak substantially.
    subgrad_deadline = start_time + 0.25 * (deadline - start_time)
    it = 0
    while time.monotonic() < subgrad_deadline:
        # compute convolution and the set of active shifts
        b = np.convolve(best_seq, best_seq)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = np.where(b >= maxb - eps)[0]

        # build sub‑gradient of max(b) (see analysis above)
        g = np.zeros_like(best_seq)
        for k in active:
            if k == 0:
                g += 2.0 * best_seq
            else:
                g[: n - k] += best_seq[k:]
                g[k:] += best_seq[: n - k]

        # average over all active shifts
        g = g / max(1, len(active))

        # simple decreasing step size
        eta = 0.2 / np.sqrt(it + 1.0)
        cand = best_seq - eta * g
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            # fallback to a fresh power‑law candidate
            cand = generate_power_sequence(n, exponent=rng.uniform(0.3, 0.6),
                                          offset=rng.uniform(0.0, 5.0))
        cand /= cand.sum()

        sc = fast_score(cand, two_n)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            elapsed = time.monotonic() - start_time
            print(f"sub‑grad improve {best_score:.6f} it={it} ({elapsed:.1f}s)")
        # always keep the iterate (helps to escape flat plateaus)
        best_seq = cand
        it += 1

    # --------------------------------------------------------------
    # 4) Stochastic hill‑climbing with assorted perturbations.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # choose a perturbation
        r = rng.random()
        if r < 0.45:
            cand = block_perturb(best_seq, rng)
        elif r < 0.80:
            cand = global_noise(best_seq, rng)
        else:
            cand = mass_swap(best_seq, rng)

        # re‑normalise to ℓ¹‑norm = 1
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        sc = fast_score(cand, two_n)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            elapsed = time.monotonic() - start_time
            print(f"new best {best_score:.6f} after {elapsed:.2f}s")

        # occasional random restart with a fresh power‑law seed
        if rng.random() < 0.002:
            exp = rng.uniform(0.30, 0.60)
            off = rng.uniform(0.0, 6.0)
            cand = generate_power_sequence(n, exponent=exp, offset=off)
            cand /= cand.sum()
            sc = fast_score(cand, two_n)
            if sc < best_score:
                best_seq = cand.copy()
                best_score = sc
                elapsed = time.monotonic() - start_time
                print(f"restart improve {best_score:.6f} ({elapsed:.2f}s)")

    # --------------------------------------------------------------
    # Final verification with the official scorer (guards against
    # possible clipping differences) and return the result.
    # --------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"final official score {final_score:.6f}")
    return [float(x) for x in best_seq]
