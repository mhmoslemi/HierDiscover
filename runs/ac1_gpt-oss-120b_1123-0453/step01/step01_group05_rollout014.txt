Search for a non‑negative step‑function sequence that minimizes

    2·n·max(conv(a,a)) / (∑a)²

using a projected sub‑gradient descent on the max‑convolution, combined
with occasional random block perturbations.  The algorithm works on a
fixed length (taken from the global `height_sequence_1` if present) and
starts from the analytically motivated arcsine shape `a_i∝1/√(i+1)`.
It repeatedly:
   • evaluates the autocorrelation by FFT,
   • builds a sub‑gradient of the supremum using the active
     convolution indices,
   • takes a diminishing step in the opposite direction,
   • projects back onto the probability simplex (non‑negative, sum=1),
   • occasionally injects a random block jitter to escape local minima.
The best sequence encountered is returned as a plain list of floats.
"""

import time
import numpy as np

# helper: projection onto the probability simplex {x>=0, sum(x)=1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    # Based on algorithm by Wang & Carreira‑Perpinán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(seq: np.ndarray, size: int) -> np.ndarray:
    """Return the linear convolution a*a of length 2·n‑1 using FFT."""
    fa = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fa * fa, size)
    n = seq.shape[0]
    return conv[: 2 * n - 1]

def _compute_score(seq: np.ndarray, n: int, size: int) -> float:
    """Fast computation of the objective 2·n·max(conv)/sum²."""
    conv = _conv_fft(seq, size)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a == 0.0:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap multiplicative jitter on a random contiguous block."""
    n = base.size
    # block width between n/300 and n/20

    hi = max(2, n // 20)

        cand = 0.5 * (cand + cand[::-1])

    # Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”.
