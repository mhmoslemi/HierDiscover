### Simplex projection

Given vector v (maybe after gradient step) we need to project onto the simplex {x >=0, sum x = 1}. Use algorithm:

```
def proj_simplex(v):
    # v can be any real vector
    # get sorted v descending
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.where(u + (1 - sv) / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = (sv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid continuation‑gradient solver for
    F(a) = 2·n·max(a*a) / (sum a)^2

The algorithm proceeds in three phases:

1.   Generate a diverse pool of warm‑starts (uniform, Dirichlet,
     Beta, exponential, power‑law, double‑Gaussian edges, spikes …).

2.   For each candidate run a *continuation projected‑gradient* (CPG)
     descent on a smooth‑max surrogate.
     The surrogate is  smoothed_max = (1/λ)·log∑exp(λ·conv),
     where conv = a*a.  λ is increased slowly (continuation) and the
     step size decays as 1/√t.  After each gradient step the vector is
     projected onto the probability simplex.

     The CPG routine also tracks the best actual objective
     (2·n·max(conv)) and returns the best found vector.

3.   The globally best vector is then refined by a short exact‑max
     sub‑gradient (exponentiated‑gradient) pass and finally by a fast
     simulated‑annealing local search that updates the convolution in
     O(n) per mass‑transfer move.

All heavy convolutions use FFTs; the whole search respects the 1000 s
budget and always returns a non‑negative normalized sequence.
"""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Reference: Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical fallback
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    lam = lam_init
    it = 0

    while time.time() < deadline and it < max_iters:
        # current convolution and true objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[CPG it={it:05d}] new best {cur_val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        # shift by max_c for numerical stability
        exps = np.exp(lam * (conv - max_c))
        w_sum = exps.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exps / w_sum

        # gradient of the smoothed objective w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size (decaying)
        lr = lr_init / np.sqrt(it + 1)

        # try a few back‑tracking reductions if no improvement
        accepted = False
        lr_try = lr
        for _ in range(3):
            a_proposed = a - lr_try * grad
            a_proj = _proj_simplex(a_proposed)

            conv_p = _full_convolution(a_proj, conv_fft_len)
            val_p = 2.0 * n * conv_p.max()
            if val_p < cur_val:
                a = a_proj
                accepted = True
                break
            lr_try *= 0.5

        # if none of the trial steps improved, keep current a (still
        # advance the continuation parameter)
        lam = min(lam * lam_factor, lam_max)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # sub‑gradient (active set)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)   # multiplicative update
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that works directly on the convolution (O(n) per move)."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # gently linear cooling
        T = max(1e-7, 0.15 * (1.0 - fraction))

        # move magnitude shrinks as we cool
        move_scale = max(0.01, 0.25 * (1.0 - fraction))

        # pick a source index proportionally to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        # apply move on copies (O(n) update of convolution)
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function that minimizes
        2·n·max(a∗a) / (sum a)^2

    The routine creates a pool of diverse warm‑starts, runs a
    continuation projected‑gradient optimiser on each, refines the
    global best with a sub‑gradient pass and a short simulated‑annealing
    local search, and finally returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm start from a previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    candidates = _candidate_pool(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    cand_idx = 0

    # --------------------------------------------------------------
    # Optimise each candidate with a bounded local budget
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand_idx += 1

        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()

        # allocate a slice of the remaining budget (≈30 s per candidate)
        remaining = deadline - time.time()
        local_deadline = time.time() + min(30.0, remaining * 0.6)
        if local_deadline - time.time() < 5.0:
            break

        # ---------- Continuation Projected Gradient ----------
        a_cpg, val_cpg = _cpg_optimize(a0, n, conv_fft_len, corr_fft_len,
                                        local_deadline,
                                        max_iters=180_000,
                                        lam_init=0.1,
                                        lam_factor=1.015,
                                        lam_max=5e4,
                                        lr_init=0.9)

        if val_cpg < best_val:
            best_val = val_cpg
            best_seq = a_cpg.copy()
            print(f"[cand {cand_idx}] CPG improved best to {val_cpg:.6f}")

        # ---------- Exact‑max sub‑gradient refinement ----------
        if time.time() < deadline:
            sub_deadline = min(deadline, time.time() + 8.0)
            a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                   conv_fft_len, corr_fft_len,
                                                   sub_deadline,
                                                   max_iters=80_000,
                                                   eta0=0.7)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cand_idx}] Sub‑grad improved best to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Heavier refinement on the global best via simulated annealing
    if best_seq is not None and time.time() < deadline - 5:
        sa_deadline = deadline - 5.0
        a_sa, val_sa = _simulated_annealing(best_seq, n, conv_fft_len,
                                            sa_deadline, best_val)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] best improved to {val_sa:.6f}")

    # --------------------------------------------------------------
    # Fallback: guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Report final objective (using the judge's routine if available)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
