"""
Propose a low‑evaluation step function.

The algorithm is a multi‑stage search that improves on the previous
implementation:

1. **Rich parametric seeding** – generate a large pool of candidate shapes
   (uniform, Dirichlet, exponentials, power‑laws, betas, sinusoid, Gaussian,
   arcsine, etc.) and evaluate them quickly with an FFT‑based convolution.
   The best ≈20 candidates are kept.

2. **Two‑phase optimisation** – each start is refined by
   * Adam on a smooth‑max surrogate of the max‑convolution (soft‑max
     weighting of the convolution values).  The learning‑rate decays and the
     temperature (λ) grows, so the method moves quickly at first and then
     performs a fine‑grained descent.
   * A sub‑gradient (exponentiated‑gradient) stage that works on the true
     max‑convolution.  This stage uses a decreasing step‑size and keeps the
     iterate on the probability simplex.

3. **Post‑processing** – the current best vector is further improved by:
   * convex mixing with the uniform distribution, with its reversed and
     several rolled copies (different shifts);
   * low‑pass Gaussian smoothing of a few bandwidths;
   * a short simulated‑annealing hill‑climb that transfers tiny amounts of
     mass between two random indices with a Metropolis acceptance rule.

4. **Optional up‑sampling** – if time permits the best vector is linearly
   interpolated to a larger grid (n = 8192) and the whole optimisation is
   repeated.

All operations respect the 1000 s deadline; the function returns the best
sequence found (list of floats).  The search never produces negative values
and clips to the evaluator’s allowed range [0, 1000].

"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (sum≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray, n: int, conv_fft_len: int,
                     corr_fft_len: int, deadline: float,
                     global_best_seq: np.ndarray | None,
                     global_best_val: float) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp (smooth‑max) surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr0 = 0.08                               # base learning rate
    lam = 0.2                                # initial temperature
    lam_factor = 1.02
    lam_cap = 1e6

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 250_000

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- evaluate (exact) objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient wrt a -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule to logits (soft‑max) -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update (with decaying LR) -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / (1.0 + it / 5000.0)          # gentle decay
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- occasional jitter if stuck -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(a0: np.ndarray, n: int, conv_fft_len: int,
                            corr_fft_len: int, deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the true max‑convolution."""
    a = init_best_seq if init_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    max_iter = 30_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:05d} new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        it += 1

    return best_seq, best_val


def _deep_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """First Adam (smooth‑max) then exact sub‑gradient refinement."""
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a large, diverse pool of symmetric and non‑symmetric shapes."""
    candidates: list[np.ndarray] = []
    i = np.arange(n, dtype=np.float64)

    # warm‑start and simple cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws – a richer set than before
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):                     # three draws per concentration
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # edge‑exponential families (with optional uniform mix)
    for alpha in np.logspace(-2, 1.0, num=20):   # 0.01 … 10 (finer grid)
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=20):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # symmetric beta‑shapes  (a_i ∝ (i+1)^{α‑1}(n‑i)^{α‑1})
    for alpha in (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6, 8):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

def _local_anneal(a: np.ndarray, n: int, conv_fft_len: int,
                  deadline: float, cur_best_val: float) -> tuple[np.ndarray, float]:
    """Short simulated‑annealing hill‑climb on the exact objective."""
    cur_a = a.copy()
    cur_val = cur_best_val
    best_seq = cur_a.copy()
    best_val = cur_val

    start_time = time.time()
    # schedule: start temperature ≈ 1 % of current value, decay to 0
    T0 = max(1e-5, 0.01 * cur_val)

    it = 0
    while time.time() < deadline:
        # linear decay of temperature
        elapsed = time.time() - start_time
        fraction = elapsed / (deadline - start_time)
        T = T0 * max(0.0, 1.0 - fraction)

        # propose a tiny mass transfer between two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # decide transfer direction randomly
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        max_transfer = cur_a[src]
        if max_transfer < 1e-12:
            continue
        delta = (np.random.rand() - 0.5) * 0.4 * max_transfer   # up to ±20 % of src mass
        delta = np.clip(delta, -cur_a[dst], cur_a[src])

        # apply move
        cur_a[src] -= delta
        cur_a[dst] += delta

        # evaluate
        conv = _full_convolution(cur_a, conv_fft_len)
        max_c = conv.max()
        new_val = 2.0 * n * max_c / (cur_a.sum() ** 2)

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probability of accepting a worse move
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_a.copy()
        else:
            # revert
            cur_a[src] += delta
            cur_a[dst] -= delta

        it += 1
        if it % 5000 == 0:
            print(f"[{n}] Anneal it={it:06d} cur best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Base resolution – 4096 offers a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        ws = np.asarray(warm, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        else:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # Stage 1 – generate & pre‑select a large candidate pool
    cand_pool = _generate_parametric_candidates(n, ws)

    evaluated = []
    for idx, cand in enumerate(cand_pool):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        # optional progress
        # print(f"[cand {idx}] val≈{val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(20, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: selected {top_k} promising starts (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on each start
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    print(f"After deep optimisation best ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – post‑processing: mixing, smoothing and simple annealing
    if best_seq is not None:
        # ----- mixing with uniform -------------------------------------------------
        uniform = np.full(n, 1.0 / n)
        for alpha in np.linspace(0.0, 1.0, 11):
            mixed = alpha * best_seq + (1.0 - alpha) * uniform
            mixed = mixed / mixed.sum()
            val = _objective(mixed, n, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, mixed.copy()
                print(f"Mix‑uniform α={alpha:.2f} improved to {val:.6f}")

        # ----- mixing with reversed and rolled copies --------------------------------
        rev = best_seq[::-1]
        for shift in (n // 4, n // 2, 3 * n // 4):
            rolled = np.roll(best_seq, shift)
            for alpha in np.linspace(0.1, 0.9, 9):
                mixed = alpha * best_seq + (1.0 - alpha) * rolled
                mixed = mixed / mixed.sum()
                val = _objective(mixed, n, conv_fft_len)
                if val < best_val:
                    best_val, best_seq = val, mixed.copy()
                    print(f"Mix‑roll shift={shift} α={alpha:.2f} improved to {val:.6f}")

        # ----- mixing with reversed copy --------------------------------------------
        for alpha in np.linspace(0.1, 0.9, 9):
            mixed = alpha * best_seq + (1.0 - alpha) * rev
            mixed = mixed / mixed.sum()
            val = _objective(mixed, n, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, mixed.copy()
                print(f"Mix‑rev α={alpha:.2f} improved to {val:.6f}")

        # ----- Gaussian smoothing ----------------------------------------------------
        for sigma in (0.5, 1.0, 2.0, 4.0):
            k_sz = int(4 * sigma + 1)
            x = np.arange(k_sz) - k_sz // 2
            kernel = np.exp(-0.5 * (x / sigma) ** 2)
            kernel /= kernel.sum()
            smoothed = np.convolve(best_seq, kernel, mode='same')
            smoothed = np.maximum(smoothed, 0.0)
            smoothed /= smoothed.sum()
            val = _objective(smoothed, n, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, smoothed.copy()
                print(f"Gaussian σ={sigma:.1f} improved to {val:.6f}")

        # ----- short simulated‑annealing hill‑climb ----------------------------------
        if time.time() < deadline - 5.0:
            print("--- Local annealing (Metropolis) ---")
            best_seq, best_val = _local_anneal(
                best_seq, n, conv_fft_len, deadline, best_val
            )
            print(f"After annealing best ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – up‑sample to a larger grid if time permits
    if time.time() < deadline - 30.0:
        n_big = 8192
        if n_big > n:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq if best_seq is not None else np.full(n, 1.0 / n))
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.full(n_big, 1.0 / n_big)

        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq, best_val
        )
        n, conv_fft_len, corr_fft_len = n_big, conv_fft_len_big, corr_fft_len_big

    # ------------------------------------------------------------------
    # Fallback – uniform if everything failed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
