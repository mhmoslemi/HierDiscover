"""
Propose a step‑function (non‑negative vector) with a low evaluation score.

Algorithm
---------

1. **Fast parametric sweep** – generate many structured candidates
   (edge‑exponential, edge‑polynomial, double‑Gaussian, powered sinusoid,
   random Dirichlet, sparse spikes).  For each candidate we compute the
   exact objective  F = 2·n·max(a*a) / (sum(a)^2)  using an FFT‑based
   convolution.  The best candidate from this cheap sweep becomes the
   warm‑start for the refinement phase.

2. **Adam on a smooth‑max surrogate** – we minimise a log‑sum‑exp
   approximation of the maximum convolution.  The variables are the
   logits of a soft‑max (mirror‑descent on the simplex).  Adam with a
   slowly increasing temperature drives the surrogate towards the true
   max.

3. **Exact‑max sub‑gradient (exponentiated‑gradient)** – after the Adam
   phase we switch to the true non‑smooth objective.  The gradient of
   the active convolution entries is used and the update is performed
   in the KL‑geometry (×exp) with a step‑size ηₜ = η₀/√t.

4. **Cheap random local moves** – a final hill‑climbing stage that
   transfers a random fraction of mass between two coordinates and
   keeps the improvement.

All phases respect the global deadline (default 1000 s) and the best
vector found so far is always returned as a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    Returns the full convolution (no truncation).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Return max(a*a) using FFT (no extra allocations)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the linear part of length 2·n‑1
    conv = conv[: 2 * a.shape[0] - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector seen and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # start on simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.05          # temperature for smooth‑max
    lam_factor = 1.025

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- simplex via soft‑max ------------------------------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective -----------------------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weight ---------------------------------------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate -----------------------------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- chain rule through soft‑max (mirror‑descent form) -------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ------------------------------------
        lam = min(lam * lam_factor, 1e7)

        # tiny random jitter near the deadline to escape possible plateaus
        if it % 4000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation (exponentiated‑gradient).
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size ηₜ = η₀ / √(t+1)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against numerical collapse
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 70_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best

    deadline = time.time() + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # fast parametrised sweep ------------------------------------------------
    best_seq = None
    best_val = float("inf")

    def update_best(vec: np.ndarray):
        nonlocal best_seq, best_val
        M = _conv_max(vec, conv_fft_len)          # max of a*a
        val = 2.0 * n * M / (vec.sum() ** 2)       # sum(vec)==1 → 2·n·M
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
            print(f"[Sweep] new best {val:.6f}")

    for gamma in np.geomspace(0.05, 10.0, num=48):
        if time.time() >= deadline:

        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        update_best(shape)

    # 2) Edge‑polynomial / power‑law family
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, num=40):
        if time.time() >= deadline:
            break
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        s = shape.sum()
        if s == 0:
            continue
        shape /= s
        update_best(shape)

    # 3) Double‑Gaussian (vary σ)
    i = np.arange(n, dtype=np.float64)
    for sigma in np.linspace(0.02, 0.5, num=30):
        if time.time() >= deadline:
            break
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        update_best(shape)

    # 4) Powered half‑wave sinusoid
    base = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    base = np.maximum(base, 0.0)
    for p in np.linspace(0.5, 5.0, num=30):
        if time.time() >= deadline:
            break
        shape = base ** p
        shape /= shape.sum()
        update_best(shape)

    # 5) Random Dirichlet seeds (moderate number)
    for _ in range(80):
        if time.time() >= deadline:
            break
        shape = np.random.dirichlet(np.ones(n))
        update_best(shape)

    # 6) Sparse spiked patterns
    for k in (32, 64, 128, 256, 512):
        if time.time() >= deadline:
            break
        shape = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        shape[pos] = 1.0 / k
        update_best(shape)

    # --------------------------------------------------------------
    # if nothing survived (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0 * n * _conv_max(best_seq, conv_fft_len)

    # --------------------------------------------------------------
    # refinement phases – split the remaining time proportionally
    remaining = deadline - time.time()
    if remaining > 5.0:
        # ----- Adam ----- (≈35 % of the leftover budget)
        adam_deadline = time.time() + remaining * 0.35
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Refine‑Adam] improved to {val_adam:.6f}")

        # ----- Sub‑gradient ----- (≈50 % of the leftover budget)
        remaining = deadline - time.time()
        if remaining > 5.0:
            sub_deadline = time.time() + remaining * 0.55
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=400_000,
                eta0=0.85,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Refine‑Sub] improved to {val_sub:.6f}")

        # ----- cheap random local search ----- (remaining time)
        remaining = deadline - time.time()
        if remaining > 5.0:
            a_loc, val_loc = _local_random_search(
                best_seq,
                n,
                conv_fft_len,
                deadline,
                max_moves=120_000,
            )
            if val_loc < best_val:
                best_val = val_loc
                best_seq = a_loc.copy()
                print(f"[Refine‑Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # final safety clip (non‑negative, ≤1000) and return as list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
