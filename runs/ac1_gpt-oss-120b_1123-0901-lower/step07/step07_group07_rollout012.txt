    best_seq = global_best_seq; best_val = global_best_val

    it=0; max_iter=200000

    while time.time()<deadline and it<max_iter:

        # softmax
        max_log = logits.max()
        exp_log = np.exp(logits - max_log)
        a = exp_log / exp_log.sum()

        # objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        val = 2.0*n*max_c/(a.sum()**2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            # maybe print

        # compute weights for smooth max
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule for logits: grad_log = a * (grad_a - np.dot(a, grad_a))
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m / (1-beta1**(it+1))
        v_hat = v / (1-beta2**(it+1))

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter if stuck
        if it % 1000 == 0 and time.time() - deadline < 0: pass

        # optional early break if no improvement for certain time

        it += 1

    return best_seq, best_val

def _local_hill_climb(a, n, conv_fft_len, deadline, best_val):
    # start from a, iterate random transfers

    best_seq = a.copy()
    best_local_val = best_val
    it = 0
    max_iter = 20000
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # small random delta relative to min(a[i], a[j])
        max_transfer = min(a[i], a[j])
        if max_transfer <= 0: continue
        delta = (np.random.rand() - 0.5) * 0.5 * max_transfer
        # ensure positivity
        if a[i] + delta < 0 or a[j] - delta < 0:
            continue
        a[i] += delta
        a[j] -= delta
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0*n*max_c/(a.sum()**2)
        if val < best_local_val:
            best_local_val = val
            best_seq = a.copy()
        else:
            # revert
            a[i] -= delta
            a[j] += delta
        it += 1
    return best_seq, best_local_val

"""
Search for a non‑negative step‑function that minimizes
``evaluate_sequence`` (i.e. 2·n·max(a∗a) / (sum(a)^2) ).

Algorithm
---------
1. **Generate a diverse pool of candidates** (parametric families + random
   Dirichlet draws) at resolution n = 8192, normalised to sum = 1.
2. **Select the best few** after a cheap FFT‑based evaluation.
3. **Deep optimisation** for each selected start
   a) Adam on a smooth‑max surrogate of the convolution.
   b) Projected‑gradient descent on the true max‑convolution
      (sub‑gradient) keeping the vector on the probability simplex.
   c) A short random “mass‑transfer’’ hill‑climb.
4. Keep the globally best sequence and return it as a plain Python list.

All heavy loops respect a global deadline (budget_s seconds).  The code
uses only NumPy and the provided ``evaluate_sequence`` (only at the very
end, if you want to double‑check the value).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) using a real FFT
    of length ``fft_len`` (must be ≥ 2·len(a)-1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) / (sum(a)^2).  a is assumed non‑negative."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # sum(a) should be 1 but we keep the denominator for safety
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Rich pool of symmetric and random candidates, all normalised to sum≈1."""
    candidates: list[np.ndarray] = []

    # warm start + cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families (with uniform mix)
    for alpha in np.logspace(-2, 1.0, num=14):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # symmetric beta‑shapes
    for alpha in (0.2, 0.4, 0.5, 0.8, 1.2, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # a few completely random (non‑symmetric) vectors – they can be useful
    for _ in range(15):
        cand = np.random.rand(n)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        candidates.append(cand)

    return candidates


# ----------------------------------------------------------------------
def _adam_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth‑max surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise in soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05                               # a bit smaller for stability
    lam = 0.4                               # temperature (smooth‑max)
    lam_factor = 1.03
    lam_cap = 1e6

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 150_000

    while time.time() < deadline and it < max_iter:
        # --- simplex via soft‑max ---
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # --- objective (exact max) for monitoring ---
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n}] it={it:06d} new best {val:.6f}")

        # --- smooth‑max weights (log‑sum‑exp) ---
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the surrogate ---
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- temperature schedule ---
        lam = min(lam * lam_factor, lam_cap)

        # --- occasional jitter if stuck for a while ---
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _projgrad_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the true max‑convolution."""
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # ensure we stay on the simplex

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    max_iter = 300_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n}] it={it:07d} new best {val:.6f}")

        # --- active set where convolution attains its maximum ---
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            # numerical safety – fall back to uniform weight
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        # --- gradient of the true max (sub‑gradient) ---
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        # gradient step + projection onto the simplex
        a = _project_to_simplex(a - eta * grad)

        # stop early if we have not improved for a while
        if time.time() - last_improve > 8.0:
            break

        it += 1

    return best_seq, best_val


def _local_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_val: float,
) -> tuple[np.ndarray, float]:
    """Very small random mass‑transfer hill‑climb (pure exploitation)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = init_val
    it = 0
    max_iter = 12_000

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose a tiny transfer between the two positions
        max_transfer = min(a[i], a[j])
        if max_transfer < 1e-14:
            continue
        delta = (np.random.rand() - 0.5) * 0.5 * max_transfer   # up to ±25 % of the smaller mass
        if delta > a[j]:
            delta = a[j]
        if -delta > a[i]:
            delta = -a[i]
        if abs(delta) < 1e-15:
            continue

        a[i] += delta
        a[j] -= delta

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
        else:
            # revert
            a[i] -= delta
            a[j] += delta

        it += 1

    print(f"[HillClimb] finished {it} attempts, best = {best_val:.6f}")
    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Run the three optimisation phases and return the best vector found."""
    best_seq, best_val = _adam_opt(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _projgrad_opt(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    # final local hill‑climb (only if we still have a few seconds left)
    if time.time() < deadline - 5.0:
        best_seq, best_val = _local_hill_climb(
            best_seq, n, conv_fft_len, deadline, best_val
        )
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    # leave a small safety margin for the final return
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Grid size – 8192 provides a good trade‑off between quality and speed
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered vector, if available
    warm_start = globals().get("height_sequence_1")
    ws = None
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws /= ws.sum()

    # ------------------------------------------------------------------
    # ---- Stage 1: generate candidates and keep the most promising ----------
    cand_pool = _generate_parametric_candidates(n, ws)

    evaluated = []
    for idx, cand in enumerate(cand_pool):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    evaluated.sort(key=lambda x: x[0])
    top_k = min(6, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"[Stage 1] selected {top_k} promising starts (n={n})")

    # ------------------------------------------------------------------
    # ---- Stage 2: deep optimisation of the selected starts -----------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        remaining = len(top_candidates) - start_id
        # give each remaining start roughly the same share of the remaining time
        sub_deadline = time.time() + (deadline - time.time()) / remaining
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # ---- Final fallback: if nothing succeeded, use uniform ------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # ---- Clip to evaluator’s allowed range and return ----------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
