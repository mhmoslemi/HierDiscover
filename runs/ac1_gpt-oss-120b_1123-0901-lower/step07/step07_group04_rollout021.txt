def _adam_opt(p0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    p = p0.copy()
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr = 0.5
    tau = 0.1
    tau_factor = 1.001

    best_local_seq = None
    best_local_val = float('inf')
    it=0
    while time.time() < deadline:
        # compute a from p
        a = np.exp(p - np.max(p))
        a = a / a.sum()
        # conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)  # a.sum() =1
        if val < best_local_val:
            best_local_val = val
            best_local_seq = a.copy()
            # update global best
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam n={n}] new global best {val:.6f}")
        # compute weights
        # to avoid overflow, subtract max conv for softmax:
        max_conv = conv.max()
        w = np.exp(tau * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum
        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # gradient wrt p
        grad_p = a * (grad_a - np.dot(a, grad_a))
        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad_p
        v = beta2 * v + (1 - beta2) * (grad_p ** 2)
        # bias correction
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step
        # schedule
        tau *= tau_factor
        # optional: reduce lr gradually
        if it % 10000 == 0:
            lr *= 0.9
        # optionally break if convergence stall
        if it % 2000 == 0:
            # maybe random perturb
            if time.time() - start_time > 30 and best_local_val < best_val * 0.999:
                pass
    return best_seq, best_val
```

def _adam_opt(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    # Input a0 is np.ndarray, assumed non-negative
    eps = 1e-12
    # Initialize p as log of a0 + eps
    p = np.log(a0 + eps)
    # Adam states
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    lr = 0.5
    # Temperature schedule
    tau = 0.01
    tau_factor = 1.001
    # improvement tracking
    last_improve_time = time.time()
    it = 0
    # Preallocate arrays? Not necessary.
    while time.time() < deadline:
        # obtain a
        maxp = np.max(p)
        a = np.exp(p - maxp)
        sum_a = np.sum(a)
        if sum_a == 0:
            # fallback to uniform
            a = np.full(n, 1.0 / n)
            sum_a = 1.0
        a = a / sum_a

        # convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (sum_a * sum_a)   # sum_a is 1

        # update best
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve_time = time.time()
            print(f"[Adam n={n} iter={it}] new best {val:.6f}")

        # compute softmax weights over conv
        # shift for numerical stability
        conv_shift = conv - max_c
        # compute weight exponent
        w = np.exp(tau * conv_shift)
        w_sum = w.sum()
        if w_sum == 0 or np.isinf(w_sum) or np.isnan(w_sum):
            # fallback to uniform active set
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.where(active, 1.0 / np.sum(active), 0.0)
        else:
            w = w / w_sum

        # gradient w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule to gradient w.r.t p
        # dot product a·grad_a
        a_dot_g = np.dot(a, grad_a)
        grad_p = a * (grad_a - a_dot_g)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad_p
        v = beta2 * v + (1 - beta2) * (grad_p * grad_p)

        # bias correction
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        p = p - step

        # schedule updates
        tau = tau * tau_factor
        if it % 5000 == 0:
            lr *= 0.9

        # stagnation: if no improvement for a while, add random noise
        if time.time() - last_improve_time > 5.0:
            # add perturbation to p
            p = p + (np.random.rand(n) - 0.5) * 0.01   # small jitter
            last_improve_time = time.time()

        # optional break if tau becomes huge causing overflow; safe bound
        if tau > 1e7:
            tau = 1e7

    return best_seq, best_val

"""
Adaptive Adam‑based mirror descent with temperature‑controlled
soft‑max smoothing of the max‑convolution.  The algorithm

* builds a diverse pool of parametric seeds (uniform, Dirichlet,
  edge‑exponential/power‑law, Gaussian‑edge, sinusoid) and a warm‑start
  from the global ``height_sequence_1`` if present;
* evaluates these seeds quickly (FFT‑based convolution) and keeps the
  best few;
* runs a fast Adam optimiser on each start.  The optimiser works on
  the log‑probability simplex (exponentiated‑gradient) so the iterates
  stay non‑negative and automatically satisfy the sum‑to‑one constraint.
  A temperature τ is increased slowly, turning the smooth soft‑max
  approximation of the max into the true max as optimisation proceeds.
* finally, if time permits, interpolates the best sequence to a finer
  grid (n=8192) and refines it again.

The routine respects the 1000 s budget, prints progress and returns the
best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with

        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _first_item(pair):
    """Key function for sorting a list of (value, vector) pairs."""
    return pair[0]


def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a diverse set of candidate vectors using several families:
    uniform, Dirichlet, edge‑exponential, edge‑power‑law, double‑Gaussian,
    half‑wave sinusoid.  All candidates are normalised to sum≈1.
    """
    candidates = []

    # Warm‑start and a few rotations (helps if a good shape is already known)
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet seeds with a range of concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential (optionally mixed with uniform)
    for alpha in np.logspace(-2, 1.0, num=16):          # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Edge‑power‑law (i+1)^(-p) + (n-i)^(-p)
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


# ----------------------------------------------------------------------
# Adam‑based optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Adam mirror‑descent on the smoothed max‑convolution.
    Maintains probabilities on the simplex via the log‑softmax
    parametrisation p (a = softmax(p)).  A temperature τ that
    multiplies the convolution values is slowly increased so that
    the soft‑max gradually turns into the true max.  The function
    returns the best vector/value observed during the run.
    """
    eps = 1e-12
    # initialise log‑probabilities (avoid log(0))
    p = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    lr = 0.5

    # temperature schedule (smooth → hard)
    tau = 0.01
    tau_factor = 1.001

    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # ---- obtain current probability vector on the simplex ----
        maxp = np.max(p)
        a = np.exp(p - maxp)
        sum_a = np.sum(a)
        if sum_a < eps:
            a = np.full(n, 1.0 / n)
            sum_a = 1.0
        a = a / sum_a

        # ---- full convolution a*a ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (sum_a * sum_a)

        # ---- keep global best ----
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it}] new best {val:.6f}")

        # ---- soft‑max weights over the convolution entries ----
        # shift by max for numerical stability
        shifted = conv - max_c
        w = np.exp(tau * shifted)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback to the hard active set
            active = np.isclose(conv, max_c, atol=1e-12)
            cnt = np.sum(active)
            if cnt == 0:
                w = np.zeros_like(conv)
            else:
                w = np.where(active, 1.0 / cnt, 0.0)
        else:
            w = w / w_sum

        # ---- gradient of the weighted convolution ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- chain rule to obtain gradient w.r.t. logits p ----
        a_dot_g = np.dot(a, grad_a)
        grad_p = a * (grad_a - a_dot_g)

        # ---- Adam update on the logits ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p * grad_p)

        # bias‑corrected estimates (helps early iterations)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        p = p - step

        # ---- schedule adjustments ----
        tau = tau * tau_factor
        if it % 5000 == 0:
            lr *= 0.9               # gentle decay of the learning rate

        # ---- occasional random jitter to escape plateaus ----
        if time.time() - last_improve > 5.0:
            p = p + (np.random.rand(n) - 0.5) * 0.01
            last_improve = time.time()

        # safety: keep τ from exploding
        if tau > 1e7:
            tau = 1e7

    return best_seq, best_val

        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        conv = _full_convolution(cand, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (cand.sum() ** 2)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fallback: uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0
    else:
        cand_vals.sort(key=_first_item)
        top_k = min(6, len(cand_vals))
        top_candidates = [c for _, c in cand_vals[:top_k]]

        best_seq = None
        best_val = float("inf")

        # --------------------------------------------------------------
        # Deep optimisation of the selected starts
        for start_id, a0 in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            print(
                f"--- Optimising start {start_id + 1}/{len(top_candidates)} "
                f"(initial val≈{_first_item(cand_vals[start_id]):.6f})"
            )
            best_seq, best_val = _adam_opt(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )
            print(f"--- after start {start_id + 1}, best_val={best_val:.6f}")

    # --------------------------------------------------------------
    # If time permits, upscale to a finer grid and refine further
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best onto the bigger grid
        x_small = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
        a_big = np.maximum(a_big, 0.0)
        if a_big.sum() < 0.01:
            a_big = np.full(n_big, 1.0 / n_big)
        else:
            a_big = a_big / a_big.sum()

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _adam_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big  # update length for final reporting

    # --------------------------------------------------------------
    # Final safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
