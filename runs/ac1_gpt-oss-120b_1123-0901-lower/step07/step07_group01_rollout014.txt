"""
Propose a step‑function minimizing
    F(a) = 2·n·max(a∗a) / (sum a)² .

Algorithm (three phases, respecting a 1000 s budget)

1️⃣  Small grid (n≈2048) – a diverse pool of warm‑starts (Dirichlet,
    power‑law, beta‑type, Gaussian edges, spikes, sinusoidal …) is
    optimised with:
    – Adam on a smooth‑max (log‑sum‑exp) surrogate.
    – Exact‑max exponentiated‑gradient (sub‑gradient) descent.
    The best shape is kept.

2️⃣  Upscaling – the best small‑grid shape is linearly interpolated to a
    larger grid (n≈8192) and refined again with Adam + sub‑gradient,
    now using the full time budget.

3️⃣  Final polishing – a fast simulated‑annealing (O(n) mass‑transfer)
    pass explores the remaining neighbourhood.

All vectors are kept on the simplex (non‑negative, sum = 1), which makes
the objective scale‑invariant.  The routine always returns a valid
non‑negative list; the final value is printed via the supplied
`evaluate_sequence` function.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse pool of start vectors."""
    cand = []

    # warm start and a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet – uniform and spiky flavours
    cand.append(np.random.dirichlet(np.ones(n)))
    cand.append(np.random.dirichlet(np.full(n, 0.2)))
    cand.append(np.random.dirichlet(np.full(n, 5.0)))
    cand.append(np.random.dirichlet(np.full(n, 0.05)))

    # Power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.5, -2.0, -1.5, -1.0, -0.5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Beta‑type (arcsine, symmetric, left/right‑skewed)
    xs = (np.arange(n) + 0.5) / n
    for a_, b_ in (
        (0.5, 0.5),  # arcsine
        (2.0, 2.0),
        (0.3, 0.3),
        (5.0, 1.0),
        (1.0, 5.0),
    ):
        shape = (xs ** (a_ - 1)) * ((1 - xs) ** (b_ - 1))
        shape /= shape.sum()
        cand.append(shape)

    # Sinusoidal / half‑sine
    shape = np.sin(np.pi * xs) ** 2
    shape /= shape.sum()
    cand.append(shape)

    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Gaussian centred, and double‑edge Gaussians
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.2, 0.5, 0.9):
        shape = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    for gamma in (0.3, 0.7, 1.5):
        shape = np.exp(-gamma * i ** 2) + np.exp(-gamma * (n - 1 - i) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Spike patterns (k equally spaced non‑zero entries)
    for k in (64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.04,
    lam_start: float = 0.15,
    lam_factor: float = 1.025,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max re‑parameterisation
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter near the deadline
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    cur_best_val: float,
) -> tuple[np.ndarray, float]:
    """Fast SA that works directly on the convolution (O(n) per move)."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start + 1e-12)

        # linear cooling
        T = max(1e-6, 0.12 * (1.0 - frac))
        move_scale = max(0.01, 0.35 * (1.0 - frac))

        # pick source proportional to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue

        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        # apply move on copies
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution in O(n) using the *old* a
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val

            best_seq,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=250_000,
            eta0=0.8,
        )
        if val_ref_big < best_val:
            best_val = val_ref_big
            best_seq = a_ref_big.copy()
            print(f"[big] Sub‑grad improved best to {val_ref_big:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final simulated‑annealing polish
    if best_seq is not None and time.time() < deadline - 2:
        n = best_seq.shape[0]
        fft_len = _next_pow2(2 * n - 1)
        a_sa, val_sa = _simulated_annealing(
            best_seq, n, fft_len, deadline, best_val
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] final best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        n = 4096
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by `evaluate_sequence`
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Report final objective (if the judge's routine is available)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        print(f"Internal best value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
