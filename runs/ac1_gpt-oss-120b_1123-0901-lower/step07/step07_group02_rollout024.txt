def _frank_wolfe_optimize(initial, n_small, iters=10, step=0.2):
    f = initial.copy()
    for it in range(iters):
        # compute M
        conv = np.convolve(f, f)
        M = conv.max()
        # build A matrix
        A = np.zeros((2 * n_small - 1, n_small), dtype=np.float64)
        for k in range(2 * n_small - 1):
            i_start = max(0, k - (n_small - 1))
            i_end = min(k, n_small - 1)
            # fill row slice
            idx = np.arange(i_start, i_end + 1)
            A[k, idx] = f[k - idx]
        # solve LP
        c = -np.ones(n_small)  # maximize sum(g)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n_small - 1, M), bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x
        # normalize
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g
        # update f
        f = (1 - step) * f + step * g
    return f
```

"""
Propose a non‑negative step‑function (list of heights) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm consists of five stages:

1️⃣  A very cheap Frank‑Wolfe style LP on a tiny problem (n≈256) that
    reduces the max‑convolution bound without any gradient steps.
    The solution is interpolated to the full grid.

2️⃣  A broad parametric sweep (power‑law edges, double‑exponential,
    beta‑shapes, sinusoid, random Dirichlet, spike patterns).  The
    best few candidates are kept.

3️⃣  For the best candidate we run a long Adam optimisation on the
    smooth‑max (log‑sum‑exp) surrogate.  The parameters are kept on
    the simplex via a soft‑max.

4️⃣  A multiplicative sub‑gradient phase with a 1/√t step‑size,
    starting from the Adam result.

5️⃣  A cheap hill‑climbing random‑move phase for a final fine‑tune.

All phases respect the global time budget (default 1000 s) and the
routine always returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) / (sum(a)^2).  a is assumed non‑negative."""
    n = a.shape[0]
    conv = np.convolve(a, a)          # dense but fast for n≈4k
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _frank_wolfe_small(n_small: int = 256,
                       iters: int = 12,
                       step: float = 0.2) -> np.ndarray:
    """
    Very cheap Frank‑Wolfe‑style LP on a tiny problem.
    Starts from uniform, repeatedly solves a linear program that
    maximises the ℓ1‑norm of g subject to (f∗g)≤M, and moves a fraction
    step towards the solution.
    """
    f = np.full(n_small, 1.0 / n_small)      # initialise with uniform mass
    for it in range(iters):
        # current bound
        conv = np.convolve(f, f)
        M = conv.max()

        # Build constraint matrix A (2·n‑1 rows, n cols):
        # (f∗g)[k] = Σ_j f[k‑j]·g[j]  → row k contains the appropriate slice of f
        rows = []
        for k in range(2 * n_small - 1):
            i_start = max(0, k - (n_small - 1))
            i_end = min(k, n_small - 1)
            row = np.zeros(n_small, dtype=np.float64)
            idx = np.arange(i_start, i_end + 1, dtype=np.intp)
            row[idx] = f[k - idx]
            rows.append(row)
        A = np.vstack(rows)

        # maximise sum(g)  ⇒  minimise -sum(g)
        c = -np.ones(n_small, dtype=np.float64)
        b = np.full(2 * n_small - 1, M, dtype=np.float64)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b,
                      bounds=[(0, None)] * n_small,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            # LP failed – abort the FW loop
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g                      # re‑normalise to ℓ1‑norm 1

        # convex combination
        f = (1 - step) * f + step * g

    return f


# ----------------------------------------------------------------------
def _initial_candidates(n: int,
                       warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse seed set (parametric families + warm start)."""
    cand = []

    # ------------------------------------------------------------------
    # Warm start (if present) – several rotations to diversify
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # ------------------------------------------------------------------
    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # Random Dirichlet draws (different concentration)
    for alpha in (0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # Edge‑concentrated power‑law shapes   (i·(n‑i+1))^α
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-4.5, -0.1, 28):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # ------------------------------------------------------------------
    # Double‑exponential (symmetric) shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.05, 8.0, 24):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)

    # ------------------------------------------------------------------
    # Beta‑like shapes  x^p·(1‑x)^q  (allowing negative exponents)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    p_vals = np.linspace(-2.5, 2.5, 13)
    q_vals = np.linspace(-2.5, 2.5, 13)
    for p in p_vals:
        for q in q_vals:
            if abs(p) < 1e-12 and abs(q) < 1e-12:
                continue
            shape = np.power(np.maximum(x, eps), p) * np.power(np.maximum(1 - x, eps), q)
            if shape.sum() > 0:
                shape /= shape.sum()
                cand.append(shape)

    # ------------------------------------------------------------------
    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # ------------------------------------------------------------------
    # Sparse “spike’’ patterns with k equally spaced non‑zero entries
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector seen (ℓ1‑norm ≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                     # slightly smaller for stability
    lam = 0.3                     # start temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress print
            if it % 20_000 == 0:
                print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights – log‑sum‑exp
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient (chain rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # tiny random jitter every few thousand steps
        if it % 9_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 6e-3

        it += 1

    return best_a, best_val


def _subgradient_opt(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 600_000,
                     eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent (exponentiated gradient).
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active constraints (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update + renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # occasional shake if we have been stuck a while
        if stall > 12_000:
            a = np.maximum(a + 0.015 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: relocate a random fraction of mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move up to 35 % of the source mass
        delta = a[i] * (np.random.rand() * 0.35)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – tiny Frank‑Wolfe LP (n≈256) to obtain a decent shape
    n_small = 256
    print("[Phase 0] Frank‑Wolfe on small problem")
    small_shape = _frank_wolfe_small(n_small=n_small, iters=14, step=0.25)

    # Upsample to the main grid (n=4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    x_small = np.linspace(0, 1, n_small, endpoint=False)
    x_big = np.linspace(0, 1, n, endpoint=False)
    warm_from_small = np.interp(x_big, x_small, small_shape)
    warm_from_small = np.maximum(warm_from_small, 0.0)
    warm_from_small /= warm_from_small.sum()

    # ------------------------------------------------------------------
    # Warm start from previous run if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – generate a diverse pool of candidates
    print("[Phase 1] Generating candidates")
    candidates = _initial_candidates(n, warm) + [_spike_candidates for _ in []]  # placeholder
    candidates.append(warm_from_small)        # the FW‑seed
    # evaluate all candidates quickly and retain a short list
    evaluated = []
    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a)
        evaluated.append((val, a))
        if time.time() >= deadline - 20:
            break
    # keep the best few (e.g. 12)
    evaluated.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [a for _, a in evaluated[:top_k]]
    best_val = evaluated[0][0]
    best_seq = evaluated[0][1].copy()
    print(f"[Phase 1] best after sweep {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam refinement on the current best
    if time.time() < deadline - 20:
        print("[Phase 2] Adam refinement")
        adam_deadline = deadline - 10
        a_adam, val_adam = _adam_optimize(best_seq,
                                          n,
                                          conv_fft_len,
                                          corr_fft_len,
                                          adam_deadline,
                                          max_iters=300_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase 2] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient refinement (can be long)
    if time.time() < deadline - 10:
        print("[Phase 3] Sub‑gradient refinement")
        sub_deadline = deadline - 5
        a_sub, val_sub = _subgradient_opt(best_seq,
                                          n,
                                          conv_fft_len,
                                          corr_fft_len,
                                          sub_deadline,
                                          max_iters=500_000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase 3] Sub‑gradient improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap hill‑climbing random moves
    if time.time() < deadline - 5:
        print("[Phase 4] Local random‑move search")
        a_loc, val_loc = _local_random_search(best_seq,
                                              n,
                                              conv_fft_len,
                                              deadline,
                                              max_moves=180_000)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Phase 4] Random moves improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – upscale to 8192 if time permits
    if time.time() + 25 < deadline:
        print("[Phase 5] Upscaling to 8192")
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # Adam on the larger grid (short)
        big_deadline = min(deadline - 5, time.time() + 20)
        a_big_adam, val_big_adam = _adam_optimize(a_big,
                                                  n_big,
                                                  conv_fft_len_big,
                                                  corr_fft_len_big,
                                                  big_deadline,
                                                  max_iters=200_000)
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[Big‑8192] Adam improved to {val_big_adam:.6f}")

        # Sub‑gradient on the larger grid
        if time.time() < deadline:
            a_big_sub, val_big_sub = _subgradient_opt(best_seq,
                                                      n_big,
                                                      conv_fft_len_big,
                                                      corr_fft_len_big,
                                                      deadline,
                                                      max_iters=250_000)
            if val_big_sub < best_val:
                best_val = val_big_sub
                best_seq = a_big_sub.copy()
                print(f"[Big‑8192] Sub‑gradient improved to {val_big_sub:.6f}")

        # Local random moves on 8192
        if time.time() < deadline - 5:
            a_big_loc, val_big_loc = _local_random_search(best_seq,
                                                          n_big,
                                                          conv_fft_len_big,
                                                          deadline,
                                                          max_moves=150_000)
            if val_big_loc < best_val:
                best_val = val_big_loc
                best_seq = a_big_loc.copy()
                print(f"[Big‑8192] Random moves improved to {val_big_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – make sure we always return a valid vector
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's safety limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
