def _generate_candidates(n, warm_start, rng):
    cands = []
    if warm_start is not None:
        cands.append(np.copy(warm_start))
        # also rotated versions maybe.
    # Uniform
    cands.append(np.full(n, 1.0/n))
    # Dirichlet draws
    for conc in [0.01, 0.05, 0.1, 0.5, 1.0, 5.0]:
        for _ in range(5):
            c = rng.dirichlet(np.full(n, conc))
            cands.append(c)
    # Single side exponential
    i = np.arange(n, dtype=float)
    for alpha in np.logspace(-2, 1.5, num=15):
        shape = np.exp(-alpha*i)
        shape = shape / shape.sum()
        cands.append(shape)
    # Single side power law
    for p in np.linspace(0.2, 5.0, num=15):
        shape = (i+1)**(-p)
        shape = shape / shape.sum()
        cands.append(shape)
    # Decreasing monotone random
    for _ in range(20):
        # generate random decreasing shape
        r = rng.rand(n)
        r = np.sort(r)[::-1]  # decreasing
        r = r / r.sum()
        cands.append(r)
    # Two-sided exponential (symmetric)
    for alpha in np.logspace(-2, 0.5, num=10):
        shape = np.exp(-alpha*i) + np.exp(-alpha*(n-1-i))
        shape = shape / shape.sum()
        cands.append(shape)
    # Beta families
    for a in [0.2,0.5,1.0,2.0,5.0]:
        for b in [0.2,0.5,1.0,2.0,5.0]:
            shape = (i+1)**(a-1) * (n-i)**(b-1)
            shape = shape / shape.sum()
            cands.append(shape)
    # Random mixture of uniform and edge
    for w in np.linspace(0, 1, 5):
        # uniform + edge
        edge = np.exp(-0.1*i) + np.exp(-0.1*(n-1-i))
        edge = edge / edge.sum()
        shape = w*edge + (1-w)/n
        shape = shape / shape.sum()
        cands.append(shape)
    return cands

"""
Propose a step‑function with a low evaluation score.

Algorithm (≈ 1000 s budget)
--------------------------------
1.  Build a rich set of *asymmetric* parametric candidates
    (single‑sided exponential / power‑law, monotone Dirichlet, mixtures
    with a uniform component, plus the previous best if any).  All candidates
    are normalised to sum = 1.

2.  Evaluate the candidates on a modest grid (n = 4096) with a fast FFT‑based
    convolution and keep the top K (K≈10).

3.  Upsample the survivors to a larger grid (n = 8192) by linear interpolation
    and run a three‑stage optimisation on each start:
        – Adam on a smooth‑max (log‑sum‑exp) surrogate,
        – Exponentiated‑gradient sub‑gradient on the true max‑convolution,
        – Projected‑gradient refinement on the exact objective.
   The optimiser respects the simplex (non‑negative, unit sum).

4.  With the remaining time, perform cheap stochastic local perturbations
    (Gaussian jitter followed by a short sub‑gradient run) to possibly
    escape shallow local minima.

5.  Return the best vector found (clipped to the evaluator’s allowed range).

The code never exceeds the time budget, prints progress and finally returns
the best sequence as a plain list of floats.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective when the vector is already on the
    probability simplex (sum(a)=1).  The true score is
        2·n·max(conv) / (sum(a)²), and here sum(a)=1.
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate of max(conv).
    The vector is stored in log‑space (logits) and normalised with a soft‑max.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)            # logits → soft‑max → a
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                # temperature for smooth‑max
    lam_factor = 1.02        # slower growth – gives a finer approximation
    lam_cap = 1e9

    last_improve = time.time()
    it = 0
    max_iter = 600_000

    while time.time() < deadline and it < max_iter:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate evaluation
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # add a tiny random nudge if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the *true* max‑convolution.
    """
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    last_improve = time.time()
    max_iter = 300_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:            # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early exit if no improvement for a while
        if time.time() - last_improve > 6.0:
            break

        eta *= 0.9999            # slowly decay learning rate

    return best_seq, best_val


def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the exact objective with Euclidean projection
    onto the simplex after each step.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.2
    last_improve = time.time()
    max_iter = 200_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # dM/da
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            break

        lr *= 0.99998

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Three‑stage optimiser (Adam → sub‑gradient → projected gradient)."""
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        global_best_seq, global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline,

    )
    best_seq, best_val = _projected_gradient_refine(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val,
    )
    return best_seq, best_val

def _generate_candidates(n: int, warm_start: np.ndarray | None,
                         rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a diverse library of *asymmetric* shapes.
    All returned vectors are normalised to sum≈1 and contain only non‑negative
    entries.
    """
    cand = []

    # Warm‑start and a few simple shifts (asymmetric shifts are cheap)
    if warm_start is not None:
        cand.append(warm_start.copy())
        for shift in (n // 3, 2 * n // 3):
            cand.append(np.roll(warm_start, shift))

    # Uniform baseline
    cand.append(np.full(n, 1.0 / n))

    # Random Dirichlet draws (different concentrations)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.7, 1.5):
        for _ in range(3):
            cand.append(rng.dirichlet(np.full(n, conc)))

    # Single‑sided exponential families (edge‑heavy, monotone)
    i = np.arange(n, dtype=float)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i)                     # left‑hand tail
        edge = edge / edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape = shape / shape.sum()
            cand.append(shape)

    # Single‑sided power‑law families
    for p in np.linspace(0.2, 4.0, num=12):
        edge = (i + 1.0) ** (-p)
        edge = edge / edge.sum()
        for w in (0.0, 0.4, 0.8):
            shape = w * edge + (1.0 - w) / n
            shape = shape / shape.sum()
            cand.append(shape)

    # Monotone decreasing Dirichlet (sorted)
    for _ in range(20):
        vec = rng.dirichlet(np.ones(n))
        vec = np.sort(vec)[::-1]      # decreasing order
        cand.append(vec)

    # Two‑sided edge families (included for completeness)
    for alpha in np.logspace(-2, 0.5, num=8):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge = edge / edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape = shape / shape.sum()
            cand.append(shape)

    # Beta‑shaped families (flexible left‑right bias)
    for a in (0.2, 0.5, 1.0, 2.0, 5.0):
        for b in (0.2, 0.5, 1.0, 2.0, 5.0):
            beta = (i + 1) ** (a - 1) * (n - i) ** (b - 1)
            beta = np.maximum(beta, 0.0)
            beta /= beta.sum()
            cand.append(beta)

    # Simple triangular / quadratic
    tri = np.minimum(i + 1, n - i)
    tri = tri / tri.sum()
    cand.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad / quad.sum()
    cand.append(quad)

    return cand

def _local_perturbations(
    base_seq: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic local search: jitter + a short sub‑gradient refinement.
    Runs until the deadline is reached.
    """
    a0 = base_seq.copy()
    while time.time() < deadline:
        eps = 0.015
        a_pert = a0 + eps * (np.random.rand(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-12:
            continue
        a_pert /= s

        a_ref, val_ref = _subgradient_refinement(
            a_pert, n, conv_fft_len, corr_fft_len,
            deadline, best_seq, best_val,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            a0 = best_seq.copy()
            print(f"[Perturb] new best {best_val:.6f}")

    return best_seq, best_val

            start,
            n_big,
            conv_fft_big,
            corr_fft_big,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Final stochastic perturbations around the current best
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Local stochastic perturbations ---")
        best_seq, best_val = _local_perturbations(
            best_seq,
            n_big,
            conv_fft_big,
            corr_fft_big,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Fallback – uniform vector if everything failed
    if best_seq is None:
        best_seq = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    # Clip to the evaluator's allowed range (0…1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
