"""
Propose a low‑evaluation step function.

Algorithm
---------
1.  **Candidate pool** – Generate a large and diverse set of vectors
    (uniform, Dirichlet, symmetric/asymmetric exponentials & power‑laws,
    linear/triangular, half‑sine, sparse “spike” vectors, logistic
    shapes and monotone exponentials).  All candidates are projected on the
    probability simplex (sum = 1).

2.  **Screening** – Compute the exact objective (2·n·max(conv)/(∑a)²)
    for every candidate (fast FFT convolution) and keep the best *K*
    (default 48).

3.  **Deep optimisation** – Each of the *K* seeds is refined by two
    stages:
        * Adam on a smooth‑max surrogate of the convolution.  The
          temperature (lam) grows slowly, the learning‑rate is
          annealed and a momentum term is used.
        * Exponentiated‑gradient sub‑gradient descent on the *exact*
          max‑convolution (active‑set gradient).

    Both stages run until the global deadline (1000 s) is reached or
    progress stalls.

4.  **Local stochastic hill‑climbing** – From the current best vector a
    we repeatedly perturb the log‑odds (Gaussian noise) and keep any
    improvement.

5.  **Optional larger grid** – If > 45 s remain after the above steps we
    upscale the best vector to length 8192 by linear interpolation and
    repeat the deep optimisation on the finer grid.

All intermediate results are printed.  The function finally returns a
plain Python list of non‑negative heights (already clipped to the
evaluator’s limits).  The optimiser works purely with NumPy and respects
the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective: 2·n·max(a*a) / (sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    return float(2.0 * n * max_c / (s * s))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    """
    eps = 1e-12
    logits = np.log(a0 + eps)          # stay on simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr0 = 0.07                         # initial learning rate
    lam = 1.5                          # start temperature
    lam_factor = 1.01                  # gradual sharpening
    lam_cap = 1e7

    best_local_seq = best_seq
    best_local_val = best_val

    it = 0
    max_iter = 400_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ---- map logits → probabilities ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- evaluate surrogate objective ----
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_local_val:
            best_local_val = val
            best_local_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of smooth max (chain rule) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)           # ∂/∂logit

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / (1.0 + it * 5e-7)            # gentle decay
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional random kick when stuck ----
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    return best_local_seq, best_local_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the *exact* max‑convolution.
    """
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s                                 # stay on the simplex

    eta0 = 0.85
    decay = 0.99994
    eta = eta0

    best_local_seq = a.copy()
    best_local_val = best_val

    it = 0
    max_iter = 150_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_local_val:
            best_local_val = val
            best_local_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active set where convolution reaches its maximum
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps non‑negativity)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 7.0:
            break

    return best_local_seq, best_local_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run the Adam‑smooth‑max phase followed by the sub‑gradient phase.
    Updates the globally best sequence/value.
    """
    seq, val = _adam_smooth_max(
        a0, n, fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    seq, val = _subgradient_refinement(
        a0, n, fft_len, corr_fft_len, deadline, seq, val
    )
    # keep global best
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
    return global_best_seq, global_best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse pool of candidate vectors.
    Families:
        – uniform
        – Dirichlet draws (various concentrations)
        – symmetric edge‑exponential / power‑law
        – asymmetric two‑sided exponentials & power‑laws
        – linear / triangular shapes
        – half‑sine
        – sparse spike vectors
        – monotone exponential
        – monotone power‑law
        – logistic (sigmoid) decays
    All candidates are normalised to sum ≈ 1.
    """
    candidates = []

    # --------------------------------------------------------------
    # Warm‑start and shifted copies
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # --------------------------------------------------------------
    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # --------------------------------------------------------------
    # Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(10):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # --------------------------------------------------------------
    # Index vector
    i = np.arange(n, dtype=np.float64)

    # ----- symmetric edge‑exponential & power‑law -----
    for alpha in np.logspace(-2, 1.0, num=10):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    for p in np.linspace(0.2, 6.0, num=10):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # ----- asymmetric two‑sided exponentials -----
    alphas = np.logspace(-2, 1.0, num=8)
    for wL in np.linspace(0.0, 1.0, 5):
        for wR in np.linspace(0.0, 1.0, 5):
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ----- asymmetric power‑law -----
    p_vals = np.linspace(0.2, 5.0, 7)
    for wL in np.linspace(0.0, 1.0, 5):
        for wR in np.linspace(0.0, 1.0, 5):
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ----- simple linear / triangular shapes -----
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate(
        [np.linspace(0, 1, n // 2, endpoint=False), np.linspace(1, 0, n - n // 2)]
    )
    for base in (lin, tri):
        for w in (0.0, 0.4, 0.8):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ----- half‑sine wave -----
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sine = np.maximum(sine, 0.0)
    sine /= sine.sum()
    candidates.append(sine)

    # ----- sparse spike vectors -----
    for spikes in (3, 5, 8):
        for _ in range(15):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    # ----- monotone exponential (left‑only) -----
    for scale in np.logspace(-2, 2.0, num=12):
        exp_shape = np.exp(-scale * i)
        exp_shape /= exp_shape.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * exp_shape + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ----- monotone power‑law (left‑only) -----
    for p in np.linspace(0.2, 5.0, 12):
        pl_shape = (i + 1.0) ** (-p)
        pl_shape /= pl_shape.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * pl_shape + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ----- logistic (sigmoid) decays -----
    centers = np.linspace(0.2, 0.8, 5) * (n - 1)
    scales = np.logspace(-2, 1.0, 9)          # smaller scale => steeper
    for c in centers:
        for sc in scales:
            logistic = 1.0 / (1.0 + np.exp((np.arange(n) - c) / sc))
            logistic = np.maximum(logistic, 0.0)
            logistic /= logistic.sum()
            for w in (0.0, 0.4, 0.8):
                cand = w * logistic + (1.0 - w) / n
                cand /= cand.sum()
                candidates.append(cand)

    return candidates


# ----------------------------------------------------------------------
def _local_random_search(
    a_best: np.ndarray,
    n: int,
    fft_len: int,
    best_val: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: Gaussian perturbation of log‑probabilities,
    accept only strict improvements.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.028
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] new best {val:.6f}")

    return a_best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – search for a non‑negative step function that
    minimises ``evaluate_sequence``.  Returns the best vector found as a
    plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Base resolution
    n = 4096
    fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start from previously stored sequence, if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a large pool and keep the top ones
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 48
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"Stage 1: evaluated {len(cand_vals)} candidates, best so far = {cand_vals[0][0]:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation on the selected seeds
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – cheap stochastic hill‑climbing from the global best
    if best_seq is not None:
        print("--- Local stochastic refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq, n, fft_len, best_val, deadline
        )

    # --------------------------------------------------------------
    # Stage 4 – optional refinement on a finer grid (n = 8192)
    time_left = deadline - time.time()
    if time_left > 45.0:
        n_big = 8192
        fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )

        # keep finer vector only if it actually improves the bound
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # --------------------------------------------------------------
    # Fallback – if nothing succeeded, return the uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator limits (≥0, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
