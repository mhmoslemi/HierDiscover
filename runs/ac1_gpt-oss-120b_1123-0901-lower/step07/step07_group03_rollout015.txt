"""
search for a low‐value non‑negative step function.

The approach mixes several ideas that extend the previous runner:

1. **Diversified warm‑starts** – a large pool of candidate shapes
   (warm start, uniform, Dirichlet samples, anti‑pair ordering,
   edge‑focused exponentials, triangular) is built and the three best
   (lowest exact objective) are kept.

2. **Pure sub‑gradient descent on the exact max** – the exact objective
   `max(a*a)` is tackled directly with exponentiated‑gradient updates.
   A simple “anti‑pair” re‑ordering (largest opposite smallest) is
   applied every few thousand iterations: for a fixed multiset this
   ordering provably reduces the convolution maximum, so it consistently
   pushes the objective down.

3. **Focused Ada­grad/Adam polishing** – after the stochastic phase a
   short Adam run on a smooth‑max surrogate (log‑sum‑exp) refines the
   solution.

4. **Final local hill‑climb** – cheap simulated‑annealing moves
   (swap or fractional mass transfer) finish the search.

The whole routine respects the 1000 s budget, constantly checks the
deadline and falls back to a uniform distribution if nothing else is
found.  All numbers are kept non‑negative and normalised to unit sum,
so the evaluator’s safety checks pass automatically.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no safety checks)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate factories ----------------------------------------------------
def _warm_start(n: int):
    """Extract a warm start from the global variable if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm = warm / warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Anti‑pair order – largest opposite smallest, second largest opposite
    second smallest, etc.  Input must be sorted descending.
    """
    n = len(vals)
    # vals already sorted descending
    a = np.empty(n, dtype=vals.dtype)
    half = n // 2
    a[:half] = vals[:half]                     # biggest half on the left
    a[-half:] = vals[-half:][::-1]             # smallest half on the right
    if n % 2 == 1:
        a[half] = vals[half]                    # median stays in the centre
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw followed by anti‑pair ordering."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    vals = np.sort(vals)[::-1]                 # descending
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the centre)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    """Sum of two Gaussians glued to the two ends."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _linear_edges_candidate(n: int, alpha: float = 0.2) -> np.ndarray:
    """
    Exponential decay from each end (U‑shaped).  The decay rate alpha
    controls how much mass is pushed to the borders.
    """
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i).astype(np.float64)
    shape = np.exp(-alpha * d)
    shape = shape / shape.sum()
    return shape

# Optimisation primitives -------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid stagnation
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.9,
    reorder_interval: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase with multiplicative (exponentiated‑gradient)
    updates.  Every `reorder_interval` iterations the vector is forced into
    anti‑pair order – for a fixed multiset this never hurts the max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # weights for active constraints (maximal entries of the convolution)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex, protecting against numerical slip‑up
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # optional anti‑pair re‑ordering (helps keep max low)
        if (it + 1) % reorder_interval == 0:
            a = _anti_pair_arrangement(np.sort(a)[::-1])

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climber:
      * swap two entries or
      * move a random fraction from i to j.
    Accept if improvement or with probability exp(-(Δ)/T) where T
    decays linearly from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        # move type
        if np.random.rand() < 0.5:      # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                           # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the time budget, tries several diverse starts,
    runs a long sub‑gradient phase (with anti‑pair re‑ordering) and ends
    with a short Adam polishing and a stochastic local search.
    The best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size (chosen as a compromise between resolution & speed)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a rich pool of starting vectors
    pool = []

    # warm start (if any) and a few rotations of it
    warm = _warm_start(n)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform baseline
    pool.append(_uniform_candidate(n))

    # Dirichlet samples with a range of concentrations
    for alpha in (0.1, 0.3, 0.5, 0.7, 1.0, 2.0):
        pool.append(_random_dirichlet_candidate(n, alpha))
        pool.append(_anti_pair_candidate(n, alpha))

    # edge‑focused exponential families
    for alpha in (0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0):
        pool.append(_linear_edges_candidate(n, alpha))

    # Gaussian‑edge families (already used in the original runner)
    for gamma in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0):
        pool.append(_gaussian_edges_candidate(n, gamma))

    # simple tent
    pool.append(_triangle_candidate(n))

    # ------------------------------------------------------------------
    # 2) Quick evaluation – keep the three best starters (lowest exact value)
    best_starters = []   # list of tuples (value, array)
    for idx, cand in enumerate(pool):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                     # fall back to a fresh Dirichlet if needed
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        print(f"[Init {idx:02d}] value {val:.6f}")
        if len(best_starters) < 3:
            best_starters.append((val, cand))
            best_starters.sort(key=lambda x: x[0])
        else:
            if val < best_starters[-1][0]:
                best_starters[-1] = (val, cand)
                best_starters.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3) Dedicated sub‑gradient runs on the top starters
    best_seq = best_starters[0][1].copy()
    best_val = best_starters[0][0]

    for rank, (init_val, init_seq) in enumerate(best_starters):
        remaining = deadline - time.time()
        if remaining < 20.0:
            break
        # allocate a slice of the remaining budget
        slice_time = min(remaining * 0.35, 250.0) if rank == 0 else min(remaining * 0.25, 180.0)
        sub_deadline = time.time() + slice_time
        print(f"[Subgrad‑stage {rank}] time slice {slice_time:.1f}s")
        a_opt, val_opt = _subgradient_optimize(
            init_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=600_000,
            eta0=1.0,
            reorder_interval=4000,
        )
        print(f"   candidate {rank} best {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"   >>> new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional Adam polishing (smooth‑max) on the current best
    remaining = deadline - time.time()
    if remaining > 15.0:
        adam_deadline = time.time() + min(180.0, remaining * 0.30)
        print("[Adam polishing] start")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=300_000,
            lr=0.08,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Adam] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Second, longer sub‑gradient refinement on the best so far
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + min(remaining * 0.6, 400.0)
        print("[Final sub‑gradient] start")
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=1_200_000,
            eta0=0.9,
            reorder_interval=3000,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Final subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Light stochastic local search for any last wiggle‑room
    remaining = deadline - time.time()
    if remaining > 5.0:
        local_deadline = deadline - 2.0
        max_moves = int(remaining * 1500)   # heuristics: ~1500 moves per second
        print("[Local search] start")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=max_moves,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
