"""
Propose a step‑function (non‑negative vector) with minimal evaluation score.

Algorithm
---------
1.  Gather a large and diverse set of parametric candidates (uniform,
    Dirichlet, edge‑exponential, edge‑power‑law, Beta, triangular,
    sinusoidal, random mixtures, plus the previous best if available).

2.  Normalise every candidate to have unit sum (probability simplex) and
    evaluate the exact objective   f(a)=2·n·max(a*a)   using an FFT‑based
    convolution.  Keep the top‑K candidates.

3.  Refine each top candidate with a *exact* gradient descent on the true
    max‑convolution:
        •  Compute the active set where the convolution attains its maximum.
        •  The gradient of f on the simplex is
           ∇f = 4·n·cross_corr(w, a)   with w uniform on the active set.
        •  Optimise the logits (soft‑max parameters) with L‑BFGS‑B
           (SciPy) which automatically performs line searches and handles
           the simplex constraint via the logit‑softmax re‑parameterisation.
        •  Keep the best vector found.

4.  Perform a short stochastic hill‑climbing phase around the current best:
    repeatedly move a random amount of mass from one index to another,
    accept only improvements (with occasional random acceptance to escape
    plateaus) and update the convolution exactly via FFT.

5.  If time allows, interpolate the best 4096‑vector to 8192 points and
    repeat steps 2‑4 on the larger grid.

The function respects the 1000 s budget, prints progress, and finally
returns a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns the projected vector (non‑negative, sum=1).
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) (sum(a)=1)."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())

def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a rich library of parametric vectors (already sum‑normalised)."""
    cand = []

    # warm‑start and simple rotations
    if warm_start is not None:
        cand.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm_start, shift))

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    # Dirichlet draws – wide range of concentrations
    for conc in np.logspace(-2, 1, num=12):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    alphas = np.logspace(-3, 2, num=30)

# ----------------------------------------------------------------------
def _lbfgsb_refine(a0: np.ndarray,
                   n: int,
                   fft_len: int,
                   corr_len: int,
                   deadline: float,
                   maxfun: int = 800) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector using L‑BFGS‑B on the exact objective.
    Optimisation is performed on logits (soft‑max parameters) so that the
    simplex constraint is automatically satisfied.
    """
    import scipy.optimize

    # initialise logits from a0 (avoid log(0))
    eps = 1e-15
    logits = np.log(a0 + eps)

    def fun_and_grad(logits_vec):
        # soft‑max
        mx = logits_vec.max()
        exp_log = np.exp(logits_vec - mx)
        a = exp_log / exp_log.sum()

        # convolution & objective
        conv = _full_convolution(a, fft_len)
        max_b = conv.max()
        val = 2.0 * n * max_b

        # active set weights (uniform over ties)
        active = np.isclose(conv, max_b, atol=1e-12)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        # gradient w.r.t. a  (∇_a f = 4·n·cross_corr(w,a))
        grad_a = 4.0 * n * _cross_correlation(w, a, corr_len)

        # chain rule for logits (soft‑max Jacobian)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        return val, grad_log.astype(np.float64)

    # time‑aware maxfun
    remaining = max(0.0, deadline - time.time())
    # assume each function evaluation costs about 2 ms → allow a safe budget
    allowed_fun = min(maxfun, int(remaining / 0.003) if remaining > 0 else 0)

    if allowed_fun < 5:            # not enough time – return early
        return a0, _objective(a0, n, fft_len)

    res = scipy.optimize.minimize(fun_and_grad,
                                  logits,
                                  method='L-BFGS-B',
                                  jac=True,
                                  options={'maxfun': allowed_fun,
                                           'maxiter': allowed_fun,
                                           'disp': False})
    # recover probability vector
    x = res.x - res.x.max()
    a_opt = np.exp(x)
    a_opt /= a_opt.sum()
    val_opt = _objective(a_opt, n, fft_len)
    return a_opt, val_opt


# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                fft_len: int,
                deadline: float,
                best_val: float,
                max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Short stochastic hill‑climbing phase.
    Moves a random amount of mass from one index to another,
    keeps the move only if the exact objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_val = best_val

    it = 0
    while time.time() < deadline and it < max_iters:
        # pick source index proportionally to its mass (ensures a[i]>0)
        src = np.random.choice(n, p=a)
        # pick a distinct destination uniformly
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        ai = a[src]
        if ai <= 1e-14:
            it += 1
            continue

        # random fraction of the source mass (up to 50 %)
        frac = np.random.rand() * 0.5
        delta = ai * frac
        if delta <= 0.0:
            it += 1
            continue

        # propose new vector
        a_candidate = a.copy()
        a_candidate[src] = ai - delta
        a_candidate[dst] = a[dst] + delta

        # evaluate exact objective (full FFT convolution)
        val_candidate = _objective(a_candidate, n, fft_len)

        if val_candidate + 1e-12 < cur_val:          # improvement
            a = a_candidate
            conv = _full_convolution(a, fft_len)
            cur_val = val_candidate
        it += 1

    return a, cur_val

    # --------------------------------------------------------------
    # warm‑start from the previous best if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        if ws.sum() >= 0.01:
            ws /= ws.sum()
            warm_start = ws
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a large pool of parametric candidates
    print("[Stage 1] generate candidates")
    all_candidates = _generate_candidates(n, warm_start)

    # quick evaluation for each candidate; keep the top‑K
    K = 20
    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, fft_len)

        cand_vals.append((val, cand))
        if (idx + 1) % 200 == 0:
            print(f"  evaluated {idx + 1}/{len(all_candidates)} candidates, "
                  f"current best {best_val:.6f}")

    # keep the most promising starts
    cand_vals.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in cand_vals[:K]]
    if cand_vals:
        best_val, best_seq = cand_vals[0]

    # --------------------------------------------------------------
    # Stage 2 – refine each top candidate with L‑BFGS‑B (exact max)
    print("[Stage 2] L‑BFGS‑B refinement")
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"  refining start #{idx + 1}/{len(top_candidates)}")
        refined_seq, refined_val = _lbfgsb_refine(
            start_vec, n, fft_len, corr_len, deadline, maxfun=800
        )
        if refined_val < best_val:
            best_val, best_seq = refined_val, refined_seq.copy()
            print(f"  ** new global best {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – stochastic hill‑climbing around the current best
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Stage 3] hill‑climbing local search")
        best_seq, best_val = _hill_climb(
            best_seq,
            n,
            fft_len,
            deadline,
            best_val,
            max_iters=5000,
        )
        print(f"  after hill‑climb best {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – upscale to a larger grid if time permits
    time_left = deadline - time.time()
    if time_left > 35.0:          # enough time for a second pass
        n_big = 8192
        fft_len_big = _next_pow2(2 * n_big - 1)
        corr_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[Stage 4] optimisation on larger grid (n=8192)")
        # reuse the same pipeline on the larger grid (but without another candidate sweep)
        a_big, val_big = _lbfgsb_refine(
            a_big, n_big, fft_len_big, corr_len_big, deadline, maxfun=600
        )
        if val_big < best_val:
            best_val, best_seq = val_big, a_big
            n = n_big   # final vector size becomes the larger one
            print(f"  improved on large grid: {best_val:.6f}")

    # --------------------------------------------------------------
    # fallback – uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)
        best_val = _objective(best_seq, 4096, _next_pow2(2 * 4096 - 1))

    # Clip to evaluator's hard bounds (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
