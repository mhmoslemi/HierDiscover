def _generate_window_candidates(n):
    cand = []
    w = np.hanning(n)  # actually np.hann? but there's np.hanning
    w = w.astype(np.float64)
    if w.sum() > 0:
        w = w / w.sum()
        cand.append(w)
        # mix with uniform
        uniform = np.full(n, 1.0/n)
        for alpha in (0.25,0.5,0.75,1.0):
            cand.append(alpha * w + (1-alpha) * uniform)
    # Hamming
    w = np.hamming(n)
    if w.sum() > 0:
        w = w / w.sum()
        cand.append(w)
        for alpha in (0.25,0.5,0.75,1.0):
            cand.append(alpha * w + (1-alpha) * uniform)
    # Blackman
    w = np.blackman(n)
    if w.sum() > 0:
        w = w / w.sum()
        cand.append(w)
        for alpha in (0.25,0.5,0.75,1.0):
            cand.append(alpha * w + (1-alpha) * uniform)
    # Bartlett (triangular)
    w = np.bartlett(n)
    if w.sum() > 0:
        w = w / w.sum()
        cand.append(w)
        for alpha in (0.25,0.5,0.75,1.0):
            cand.append(alpha * w + (1-alpha) * uniform)
    # Kaiser with various betas
    for beta in (2,4,6,8,10,12):
        w = np.kaiser(n, beta)
        if w.sum() > 0:
            w = w / w.sum()
            cand.append(w)
            for alpha in (0.25,0.5,0.75,1.0):
                cand.append(alpha * w + (1-alpha) * uniform)
    return cand

"""
Propose a low‑evaluation step‑function.

The search proceeds in four stages:

1. **Broad candidate generation** – a diverse pool of symmetric and
   asymmetric families (beta, exponential, power‑law, sinusoidal,
   classical signal‑processing windows, and random Dirichlet draws) is
   built and quickly evaluated.  The best few seeds are kept.

2. **Two‑stage refinement** – each seed is refined by (a) Adam on a
   log‑sum‑exp surrogate of the max‑convolution (smooth‑max) and
   (b) an exponentiated‑gradient sub‑gradient step on the true max
   convolution.  Hyper‑parameters (learning‑rate, temperature schedule,
   iteration limits) are chosen on‑the‑fly from the remaining time
   budget.

3. **Upscaling** – if time permits the current best vector is interpolated
   to a finer grid (n = 8192) and the whole optimisation (Adam + sub‑grad)
   is repeated.  More degrees of freedom usually flatten the
   convolution further.

4. **Final stochastic hill‑climb** – a short simulated‑annealing
   phase makes tiny mass‑transfer moves; worsening moves are occasionally
   accepted to escape shallow traps.

All vectors are kept on the probability simplex (non‑negative and
sum‑to‑one) via a soft‑max / exponentiated‑gradient parameterisation.
The function returns the best sequence found (list of floats) before the
deadline (default 1000 s).  The algorithm uses only NumPy and the
provided ``evaluate_sequence`` for objective evaluation.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution ``a * a`` (length 2·len(a)-1) using real FFT.
    ``fft_len`` must be ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective (scale‑invariant).
    Returns 2·n·max(a*a) / (sum(a)**2).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _generate_basic_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Rich collection of symmetric / asymmetric families (beta, expo, power‑law, …)."""
    candidates: list[np.ndarray] = []

    # Warm‑start and its cyclic shifts (if supplied)
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # Symmetric exponential decays (both ends)
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=12):  # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Symmetric power‑law tails
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Symmetric beta‑shapes
    for alpha in (0.2, 0.4, 0.5, 0.8, 1.2, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6, 8, 12):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Two‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ------------------------------------------------------------------
    # Asymmetric families – pure one‑sided exponential / power‑law
    for alpha in np.logspace(-3, 0.5, num=10):
        a = np.exp(-alpha * i)
        a /= a.sum()
        candidates.append(a)

    for p in np.linspace(0.2, 4.0, num=10):
        a = (i + 1.0) ** (-p)
        a /= a.sum()
        candidates.append(a)

    # Mix asymmetric shapes with a uniform background
    for mix in (0.2, 0.5, 0.8):
        for alpha in (0.05, 0.2, 1.0):
            a = np.exp(-alpha * i)
            a = a / a.sum()
            candidates.append(mix * a + (1 - mix) / n)

    return candidates


def _generate_window_candidates(n: int) -> list[np.ndarray]:
    """Classic signal‑processing windows plus uniform mixes."""
    candidates: list[np.ndarray] = []
    uniform = np.full(n, 1.0 / n)

    # Hann (numpy calls it "hanning")
    w = np.hanning(n).astype(np.float64)
    w /= w.sum()
    candidates.append(w)
    for mix in (0.25, 0.5, 0.75, 1.0):
        candidates.append(mix * w + (1 - mix) * uniform)

    # Hamming
    w = np.hamming(n).astype(np.float64)
    w /= w.sum()
    candidates.append(w)
    for mix in (0.25, 0.5, 0.75, 1.0):
        candidates.append(mix * w + (1 - mix) * uniform)

    # Blackman
    w = np.blackman(n).astype(np.float64)
    w /= w.sum()
    candidates.append(w)
    for mix in (0.25, 0.5, 0.75, 1.0):
        candidates.append(mix * w + (1 - mix) * uniform)

    # Bartlett (triangular)
    w = np.bartlett(n).astype(np.float64)
    w /= w.sum()
    candidates.append(w)
    for mix in (0.25, 0.5, 0.75, 1.0):
        candidates.append(mix * w + (1 - mix) * uniform)

    # Kaiser with several betas
    for beta in (2, 4, 6, 8, 12, 16):
        w = np.kaiser(n, beta).astype(np.float64)
        w /= w.sum()
        candidates.append(w)
        for mix in (0.25, 0.5, 0.75, 1.0):
            candidates.append(mix * w + (1 - mix) * uniform)

    return candidates

def _adam_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 250_000,
    lr: float = 0.06,
    lam_start: float = 0.15,
    lam_factor: float = 1.03,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of the max‑convolution.
    Returns the best vector seen and its objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)           # start in soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start
    lam_cap = 1e6

    best_seq = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # ---- soft‑max projection (ensures simplex) ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                 # ⟨a,∇⟩ term from chain rule
        grad_log = a * (grad_a - avg)           # gradient w.r.t logits

        # ---- Adam step ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- jitter if stuck ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgrad_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 30_000,
    eta0: float = 0.55,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation on the true max‑convolution.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        a_sum = a.sum()
    a /= a_sum

    best_seq = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    last_improve = time.time()

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:05d} new best {val:.6f}")

        # active set (indices where the convolution attains its maximum)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by a sub‑gradient refinement.
    Keeps the global best across all runs.
    """
    # Adam – allocate a generous iteration budget relative to remaining time
    remaining = deadline - time.time()
    max_iter_adam = int(min(250_000, remaining * 400))   # heuristic: ~0.0025 s per iter
    best_seq, best_val = _adam_opt(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iter=max_iter_adam,
    )

    # Sub‑gradient – shorter but still substantial
    remaining = deadline - time.time()
    max_iter_sub = int(min(40_000, remaining * 200))
    best_seq, best_val = _subgrad_opt(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iter=max_iter_sub,
    )

    # Update the global best
    if best_val < global_best_val:
        global_best_val = best_val
        global_best_seq = best_seq.copy()
    return global_best_seq, global_best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """

    ``evaluate_sequence``.  Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – coarse grid (n = 2048) for rapid screening
    n0 = 2048
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # Warm‑start from previous run (if available)
    warm_start = globals().get("height_sequence_1")
    ws0: np.ndarray | None = None
    if warm_start is not None:
        ws_arr = np.asarray(warm_start, dtype=np.float64)
        if ws_arr.shape[0] > n0:
            ws_arr = ws_arr[:n0]
        elif ws_arr.shape[0] < n0:
            ws_arr = np.pad(ws_arr, (0, n0 - ws_arr.shape[0]))
        ws_arr = np.clip(ws_arr, 0.0, None)
        if ws_arr.sum() >= 0.01:
            ws_arr /= ws_arr.sum()
            ws0 = ws_arr

    # ------------------------------------------------------------------
    # Build candidate pool
    candidates = []
    candidates.extend(_generate_basic_candidates(n0, ws0))
    candidates.extend(_generate_window_candidates(n0))

    # ------------------------------------------------------------------
    # Quick evaluation of the raw pool
    evaluated = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n0))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n0, conv_fft_len0)
        evaluated.append((val, cand))
        # (optional) print progress occasional
        if idx % 50 == 0:
            print(f"[stage1] candidate {idx:04d} → val≈{val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1 (n={n0}): selected {top_k} seeds, best raw val {evaluated[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on a finer grid (n = 4096)

    corr_fft_len = _next_pow2(3 * n - 2)

        a_big = np.interp(x_new, x_old, cand)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
        interpolated.append(a_big)

    # also include the warm‑start (if any) after interpolation

        ws_big = np.interp(x_new, x_old, ws0)

        ws_big /= ws_big.sum()
        interpolated.append(ws_big)

        if time.time() >= deadline:

        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – upscale to n = 8192 if time permits
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Stage 4 – final stochastic hill‑climb (simulated annealing style)
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local_val = best_val
        it = 0
        max_local = 20_000
        T0 = 5e-4
        Tf = 5e-8
        while time.time() < deadline and it < max_local:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # propose a small mass transfer (up to 25 % of the smaller bucket)
            max_transfer = min(a[i], a[j])
            if max_transfer < 1e-14:
                continue
            delta = (np.random.rand() - 0.5) * 0.5 * max_transfer
            # apply
            a[i] += delta
            a[j] -= delta
            if a[i] < 0 or a[j] < 0:
                a[i] -= delta
                a[j] += delta
                continue

            conv = _full_convolution(a, conv_fft_len)
            val = 2.0 * n * conv.max() / (a.sum() ** 2)

            # temperature schedule (geometric cooling)
            T = T0 * ((Tf / T0) ** (it / max_local))

            if val < best_local_val or np.random.rand() < np.exp(-(val - best_local_val) / max(T, 1e-12)):
                if val < best_local_val:
                    best_local_val = val
                    best_seq = a.copy()
                # accept move (already applied)
            else:
                # reject – revert
                a[i] -= delta
                a[j] += delta
            it += 1
        best_val = best_local_val
        print(f"Final SA refinement finished after {it} attempts, best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
