"""
Simple deterministic max‑convolution optimizer.

The objective is
    F(a) = 2·n·max(a*a) / (sum a)^2

Because the objective is scale‑invariant we fix sum(a)=1.
The algorithm proceeds as follows:

1. Generate a diverse pool of warm‑starts (uniform, Dirichlet,
   arcsine, power‑law, double‑Gaussian, spike patterns and random
   mixtures).  If a previous solution (`height_sequence_1`) exists it
   is added to the pool.

2. For each candidate we run a *gradient‑guided mass‑transfer* loop.
   The convolution of the current vector is kept explicitly.
   At each iteration we
   * locate the positions where the convolution attains its current
     maximum,
   * compute the sub‑gradient of the (exact) max via a cross‑correlation
     (O(n log n) with FFT),
   * move a fraction of mass from the index that contributes most to
     the peak to the index that contributes least,
   * update the convolution in O(n) using the analytic formula for a
     single mass‑transfer move,
   * keep the best vector seen.

   The loop stops when the time budget is exhausted, when a preset
   number of moves is reached or when no improving move can be found.

3. After all candidates have been processed, a short simulated‑annealing
   (SA) refinement is performed on the global best (the SA code is
   the same as in the previous baseline).  Finally the best vector is
   clipped to the limits enforced by `evaluate_sequence` and returned.

The whole routine respects the 1000 s budget, never returns an empty
list and works with any `evaluate_sequence` that follows the specification
given in the problem statement.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _update_convolution(conv: np.ndarray, a: np.ndarray,
                        src: int, tgt: int, delta: float) -> np.ndarray:
    """
    Return a copy of `conv` after moving `delta` mass from `src` to `tgt`.
    Uses the exact O(n) update formulas derived from
    (a–delta·e_src + delta·e_tgt) * (a–delta·e_src + delta·e_tgt).
    """
    new_conv = conv.copy()
    n = a.shape[0]

    # the linear part (every entry where the moved index participates)
    new_conv[src : src + n] -= 2.0 * delta * a
    new_conv[tgt : tgt + n] += 2.0 * delta * a

    # quadratic corrections (self‑pair terms)
    new_conv[2 * src] += delta * delta
    new_conv[2 * tgt] += delta * delta
    new_conv[src + tgt] -= 2.0 * delta * delta

    return new_conv


def _candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a rich set of starting vectors."""
    cand = []

    # warm start (and a few cyclic shifts)
    if warm is not None:
        w = warm.copy()
        w /= w.sum()
        cand.append(w)
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(w, shift))

    # pure uniform & Dirichlet
    cand.append(np.full(n, 1.0 / n))
    cand.append(np.random.dirichlet(np.ones(n)))

    # arcsine‑type (Beta(½,½) analogue)
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64) *
                         np.arange(n, 0, -1, dtype=np.float64))
    arcs /= arcs.sum()
    cand.append(arcs)

    # power‑law families (both decreasing and symmetric)
    idx = np.arange(n, dtype=np.float64) + 1.0
    for alpha in (0.5, 1.0, 1.5, 2.0):
        dec = idx ** (-alpha)
        dec /= dec.sum()
        cand.append(dec)

        sym = ((idx) * (n - idx + 1.0)) ** (-alpha)
        if sym.sum() > 0:
            sym /= sym.sum()
            cand.append(sym)

    # double‑Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        g = np.exp(-((i) ** 2) / (2 * sigma ** 2))
        g += np.exp(-((i - (n - 1)) ** 2) / (2 * sigma ** 2))
        g /= g.sum()
        cand.append(g)

    # half‑sine (positive half‑wave)
    half_sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    half_sine = np.maximum(half_sine, 0.0)
    half_sine /= half_sine.sum()
    cand.append(half_sine)

    # spike patterns (k equally spaced points)
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # random mixtures of already generated shapes
    base = list(cand)
    for _ in range(12):
        s1, s2 = np.random.choice(base, size=2, replace=False)
        alpha = np.random.rand()
        mix = alpha * s1 + (1.0 - alpha) * s2
        mix = np.maximum(mix, 0.0)
        mix /= mix.sum()
        cand.append(mix)

    return cand


def _max_convolution_optimize(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              max_moves: int = 5000) -> tuple[np.ndarray, float]:
    """
    Deterministic max‑convolution descent.
    Returns the best vector found and its objective value.
    """
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n - 2)

    a = a0.copy()
    conv = _full_convolution(a, fft_len_conv)
    M = conv.max()
    best_val = 2.0 * n * M
    best_a = a.copy()
    moves = 0

    while time.time() < deadline and moves < max_moves:
        # --- locate current peak(s) ---
        atol = max(1e-12, M * 1e-12)
        active = np.isclose(conv, M, atol=atol)
        if not active.any():
            break

        # smooth weights: uniform over active positions
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # sub‑gradient of the exact max
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # pick a source index (large gradient, positive mass)
        eligible = np.where(a > 1e-14)[0]
        if eligible.size == 0:
            break
        src = eligible[np.argmax(grad[eligible])]

        # pick a target index (smallest gradient)
        tgt = np.argmin(grad)
        if src == tgt:
            # pick any different target if the obvious one coincides
            others = np.arange(n)
            others = others[others != src]
            tgt = np.random.choice(others)

        # how much mass to move?  start with half of the source mass,
        # but never larger than an absolute cap (helps stability)
        delta = min(0.5 * a[src], 0.03)

        # try to shrink delta until an improvement is observed
        improved = False
        while delta > 1e-14:
            new_conv = _update_convolution(conv, a, src, tgt, delta)
            new_M = new_conv.max()
            if new_M < M - 1e-15:
                # accept the move
                a[src] -= delta
                a[tgt] += delta
                conv = new_conv
                M = new_M
                moves += 1
                improved = True

                cur_val = 2.0 * n * M
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[GD it={moves:05d}] new best {cur_val:.6f}")

                break
            else:
                delta *= 0.5

        if not improved:
            # --- stochastic escape: tiny random transfer ---
            i = np.random.choice(eligible)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            tiny = a[i] * np.random.rand() * 0.02
            if tiny <= 0:
                break
            new_conv = _update_convolution(conv, a, i, j, tiny)
            new_M = new_conv.max()
            if new_M < M:
                a[i] -= tiny
                a[j] += tiny
                conv = new_conv
                M = new_M
                # note: we do not count this as a full move for the limit
            # continue the main loop regardless

        # early exit if the improvement becomes negligible
        if best_val < 1.48:   # a safety stop far below the target
            break

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA (identical to the baseline) for final polishing."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)
        T = max(1e-6, 0.1 * (1.0 - fraction))
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution (O(n))
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * conv_new.max()
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimizes
    `2·n·max(a∗a)/(sum a)^2`.  The routine combines a handcrafted
    candidate pool with a deterministic gradient‑descent mass‑transfer
    optimizer and a final simulated‑annealing polish.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    n = 4096                       # primary grid size
    fft_len = _next_pow2(2 * n - 1)

    # ------ warm start (if a previous solution exists) ----------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    candidates = _candidate_pool(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Process each candidate with the deterministic optimizer
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()                     # enforce sum = 1

        # allocate a modest slice of the remaining budget to this candidate
        cand_deadline = min(deadline - 0.5, time.time() + 30.0)

        a_opt, val_opt = _max_convolution_optimize(a0, n, cand_deadline,
                                                   max_moves=5000)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[cand {cid}] GD improved best to {val_opt:.6f}")

        # optional quick SA refinement (very cheap)
        if time.time() < deadline:
            a_sa, val_sa = _simulated_annealing(best_seq, n, fft_len,
                                                deadline, best_val)
            if val_sa < best_val:
                best_val = val_sa
                best_seq = a_sa.copy()
                print(f"[cand {cid}] SA improved best to {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final polishing on the global best (if time permits)
    if best_seq is not None and time.time() < deadline - 5:
        a_final, val_final = _simulated_annealing(best_seq, n, fft_len,
                                                  deadline, best_val)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Final SA] best improved to {val_final:.6f}")

    # --------------------------------------------------------------
    # Upscale to a longer grid (8192) if we still have slack time
    if best_seq is not None and time.time() + 20 < deadline:
        n_big = 8192
        x_small = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        fft_len_big = _next_pow2(2 * n_big - 1)

        # a short run of the deterministic optimizer on the larger grid
        big_deadline = deadline
        a_big_opt, val_big_opt = _max_convolution_optimize(a_big, n_big,
                                                          big_deadline,
                                                          max_moves=3000)
        if val_big_opt < best_val:
            best_val = val_big_opt
            best_seq = a_big_opt.copy()
            print(f"[Big] final best {val_big_opt:.6f}")

    # --------------------------------------------------------------
    # Fallback – guarantee a valid (non‑empty) return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by `evaluate_sequence`
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Report the final objective using the provided evaluator
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
