def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # Determine n
    n = 4096
    conv_len = _next_pow2(2*n-1)
    corr_len = _next_pow2(3*n-2) # maybe needed for cross correlation in gradient.
    # Warm start
    warm_start_seq = globals().get('height_sequence_1')
    if warm_start_seq is not None:
        ws = np.asarray(warm_start_seq, dtype=np.float64)
        if ws.shape[0] != n:
            if ws.shape[0] > n: ws = ws[:n]
            else: ws = np.pad(ws, (0, n-ws.shape[0]), mode='constant')
        ws = np.clip(ws, 0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # Stage 1: generate candidates & keep top k
    top_k = 6
    best_candidates = []  # list of (val, cand)

    def consider(cand):
        # Ensure nonneg and normalized
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_len)
        if len(best_candidates) < top_k:
            best_candidates.append((val, cand))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if val < best_candidates[-1][0]:
                best_candidates[-1] = (val, cand)
                best_candidates.sort(key=lambda x: x[0])

    # Add warm start and its shifts
    if ws is not None:
        consider(ws)
        for shift in (n//4, n//2, 3*n//4):
            consider(np.roll(ws, shift))

    # Uniform
    consider(np.full(n, 1.0/n))

    # Dirichlet draws
    for conc in (0.05,0.1,0.2,0.5,1.0,2.0):
        consider(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # Edge exponential families
    for alpha in np.logspace(-3, 1.5, num=80):  # 0.001 to ~31.6
        edge = np.exp(-alpha*i) + np.exp(-alpha*(n-1-i))
        edge = edge / edge.sum()
        for w in np.linspace(0,1,21):
            cand = w * edge + (1-w) * (1.0/n)
            consider(cand)

    # Edge power law
    for p in np.linspace(0.2,5.0,80):
        edge = (i+1.0)**(-p) + (n-i)**(-p)
        edge = edge / edge.sum()
        for w in np.linspace(0,1,21):
            cand = w*edge + (1-w)*(1.0/n)
            consider(cand)

    # Beta shapes
    for beta in np.linspace(0.2,3.0,80):
        shape = (i+1.0)**(beta-1.0) * (n-i)**(beta-1.0)
        shape = shape / shape.sum()
        consider(shape)

    # Raised-cosine / sine power families
    for p in [1,2,4,6,8]:
        shape = np.sin(np.pi*(i+0.5)/n)**p
        shape = shape / shape.sum()
        consider(shape)

    # Two Gaussian edges
    for gamma in np.logspace(-2, 1, num=40):
        shape = np.exp(-gamma*i**2) + np.exp(-gamma*(n-1-i)**2)
        shape = shape / shape.sum()
        consider(shape)

    # Arcsine shape
    shape = 1.0/np.sqrt((i+0.5)*(n-i-0.5))
    shape = shape / shape.sum()
    consider(shape)

    # At this point we have top_k candidates

def _project_simplex(v):
    # Credit to Duchi et al. 2008.
    n = v.shape[0]

def _adam_smooth_max(a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.07
    lam = 0.1
    lam_factor = 1.03
    lam_cap = 1e6
    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 500_000
    while time.time() < deadline and it < max_iter:
        # compute softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)
        # jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
        it += 1
    return best_seq, best_val

"""
Propose a low‑evaluation step‑function.

Strategy
---------

1. **Parametric seeding** – generate a large pool of analytic
   families (exponential edge, power‑law edge, symmetric beta,
   raised‑cosine, Gaussian edge, arcsine, random Dirichlet draws,
   the uniform baseline and optional warm‑starts).  All candidates are
   normalised to a probability simplex and evaluated by a fast FFT‑based
   convolution.  The best few (default 6) are retained.

2. **Deep optimisation** – each retained start is refined by
   *Adam* on a smooth‑max surrogate of the convolution (log‑sum‑exp
   of the convolution entries) followed by a *projected sub‑gradient*
   descent on the true Chebyshev norm (max‑convolution).  Both phases
   run until the per‑start time budget expires.

3. **Random hill‑climbing** – with the remaining time a simple
   mass‑transfer Monte‑Carlo search (pairwise transfers that preserve
   the total sum) is performed, accepting improving moves or occasional
   uphill moves according to a cooling schedule.

All helper functions are top‑level, no closures or lambdas are used.
The algorithm respects the 1000 s deadline and returns the best sequence
found (as a plain Python list).  The final sequence is stored in the
global variable ``height_sequence_1`` for warm‑starts in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution / correlation utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using a real FFT of length
    ``fft_len`` (must be ≥ 2·len(a)-1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the evaluation function (assuming sum(a)≈1).
    Returns 2·n·max(a*a) / (sum(a)**2).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are <=0 – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp (smooth‑max) surrogate of the
    convolution max.  The parameters are tuned for a fast yet robust
    descent.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.12           # initial temperature
    lam_factor = 1.03    # grow temperature relatively fast
    lam_cap = 1e6

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 500_000

    while time.time() < deadline and it < max_iter:
        # soft‑max → probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # current objective (exact max)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (smooth objective)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)               # ⟨a,∇⟩ term from chain rule
        grad_log = a * (grad_a - avg)         # gradient w.r.t. logits

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses the active set (indices attaining the maximum) to form the
    sub‑gradient and mirrors the iterate back onto the simplex.
    """
    a = init_best_seq.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.8
    it = 0
    last_improve = time.time()
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:06d} new best {val:.6f}")

        # active set where the convolution is maximal
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)

        a = _project_simplex(a - eta * grad_a)

        if time.time() - last_improve > 6.0:
            break
        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by a sub‑gradient refinement.
    Updates the globally best vector/value.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refine(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


def _random_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Simple Monte‑Carlo mass‑transfer hill‑climb.
    Moves preserve the total sum (no renormalisation required) and keep
    all entries non‑negative.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_local_val = best_val
    it = 0
    max_iter = 200_000
    step_factor = 0.4   # up to 40 % of the smaller of the two involved masses

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_transfer = min(a[i], a[j])
        if max_transfer <= 0.0:
            continue
        delta = (np.random.rand() - 0.5) * step_factor * max_transfer
        if a[i] + delta < 0.0 or a[j] - delta < 0.0:
            continue

        # apply move (sum stays 1)
        a[i] += delta
        a[j] -= delta

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_local_val:
            best_local_val = val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[{n}] Hill‑climb it={it:06d} new best {val:.6f}")
        else:
            # revert move
            a[i] -= delta
            a[j] += delta

        it += 1

    return best_seq, best_local_val


# ----------------------------------------------------------------------
# Candidate generation (dense parametric grid)
# ----------------------------------------------------------------------
def _populate_top_k(
    n: int,
    conv_fft_len: int,
    top_k: int,
    warm_start: np.ndarray | None,
) -> list[tuple[float, np.ndarray]]:
    """
    Generate a rich pool of analytic candidates, evaluate them and keep the
    ``top_k`` best (lowest objective) candidates.
    """
    best = []  # list of (val, cand)

    def consider(cand: np.ndarray):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                     # avoid near‑zero sum
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        if len(best) < top_k:
            best.append((val, cand))
            best.sort(key=lambda x: x[0])
        else:
            if val < best[-1][0]:
                best[-1] = (val, cand)
                best.sort(key=lambda x: x[0])

    # Warm‑start and simple cyclic shifts
    if warm_start is not None:
        consider(warm_start)
        for shift in (n // 4, n // 2, 3 * n // 4):
            consider(np.roll(warm_start, shift))

    # Uniform baseline
    consider(np.full(n, 1.0 / n))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        consider(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential families
    for alpha in np.logspace(-3, 1.5, num=80):          # 0.001 … ~31.6
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in np.linspace(0.0, 1.0, 21):
            consider(w * edge + (1.0 - w) / n)

    # Edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=80):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in np.linspace(0.0, 1.0, 21):
            consider(w * edge + (1.0 - w) / n)

    # Symmetric beta‑shapes
    for beta in np.linspace(0.2, 3.0, num=80):
        shape = (i + 1.0) ** (beta - 1.0) * (n - i) ** (beta - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        consider(shape)

    # Raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6, 8):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        consider(shape)

    # Two‑Gaussian edge shapes
    for gamma in np.logspace(-2, 1.0, num=50):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        consider(shape)

    # Arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    consider(shape)

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Primary grid size – 4096 offers a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (previous best stored globally)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        ws = np.asarray(warm, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]), mode="constant")
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # Stage 1 – parametric seeding
    top_k = 6
    top_candidates = _populate_top_k(n, conv_fft_len, top_k, ws)
    print(f"Stage 1: kept {len(top_candidates)} promising starts (n={n})")

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on the retained starts
    for idx, (init_val, cand) in enumerate(top_candidates):
        if time.time() >= deadline - 5.0:
            break
        print(f"--- Deep optimisation start #{idx + 1}/{len(top_candidates)} (init={init_val:.6f}) ---")
        # Allocate a fair share of the remaining time to this start
        remaining = deadline - time.time()
        remaining_starts = len(top_candidates) - idx
        per_start_deadline = time.time() + remaining * 0.9 / remaining_starts
        best_seq, best_val = _deep_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            per_start_deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – final random hill‑climbing with any time left
    if best_seq is not None and time.time() < deadline - 2.0:
        print(f"--- Final random hill‑climbing (time left {deadline - time.time():.1f}s) ---")
        best_seq, best_val = _random_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Fallback to a safe uniform distribution if everything failed
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Store globally for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
