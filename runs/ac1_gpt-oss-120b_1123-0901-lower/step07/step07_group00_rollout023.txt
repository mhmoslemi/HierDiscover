"""
Improved step‑function search for the evaluation function

The algorithm works in several stages:

1.  Warm‑start from the previously found best sequence (if any) and build a
    diverse pool of candidates (Dirichlet, edge‑exponential, double‑Gaussian,
    uniform, spike patterns and a dense family of Beta‑shaped PDFs).

2.  Each candidate is briefly refined by Adam on a smooth‑max (log‑sum‑exp)
    surrogate.  The best candidate from this pool is kept.

3.  A long exponentiated‑gradient (sub‑gradient) phase on the exact max
    convolution is run on the current best vector – this is the main
    optimisation stage.

4.  A cheap random‑move hill‑climbing phase explores the neighbourhood of the
    best solution.

5.  If time permits a final up‑sampling to a finer grid (8192 points) is
    performed, followed by one more Adam + sub‑gradient refinement.

All phases respect the overall time budget (default 1000 s) and the routine
always returns the best sequence found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # warm start (if any) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (full support)
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑exponential families (mass concentrated at the left/right)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with a small number of equally spaced spikes."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑distribution PDF discretised on n points.
    Parameters must be > 0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid numerical problems at the boundaries
    shape = np.exp((alpha - 1) * np.log(np.clip(x, 1e-16, 1.0)) +
                   (beta - 1) * np.log(np.clip(1.0 - x, 1e-16, 1.0)))
    if shape.sum() == 0 or not np.isfinite(shape).all():
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)  # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.05
    lam_factor = 1.08               # warm up temperature a little faster

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- simplex projection (soft‑max) ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake when we are close to the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay (very gentle)
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Step size η_t = η₀ / √(t+1).
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The routine respects the time budget and always returns the best
    sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # build a diverse pool of start vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # add a family of Beta‑shaped PDFs (deterministic grid + random draws)
    alphas = np.logspace(-1, 1, 12)   # 0.1 … 10
    betas   = np.logspace(-1, 1, 12)
    for a in alphas:
        for b in betas:
            pool.append(_beta_shape(n, a, b))

    # random Beta shapes (log‑uniform sampling)
    for _ in range(30):
        a = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
        b = 10 ** np.random.uniform(-2, 1)
        pool.append(_beta_shape(n, a, b))

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # short exploration of the pool
    remaining = deadline - time.time()
    # devote roughly 15 % of the remaining budget to the exploratory phase
    explore_budget = remaining * 0.15
    per_candidate = max(0.01, explore_budget / max(1, len(pool)))

    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a0 = np.clip(cand, 0.0, 1000.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # quick exact evaluation (no optimisation)
        conv0 = _full_convolution(a0, conv_fft_len)
        val0 = 2.0 * n * conv0.max() / (a0.sum() ** 2)
        if val0 < best_val:
            best_val = val0
            best_seq = a0.copy()
            print(f"[cand {idx}] raw score {val0:.6f}")

        # a very short Adam run (at most per_candidate seconds)
        local_deadline = min(deadline - 0.5, time.time() + per_candidate)
        a_adam, val_adam = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=80_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {idx}] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Deep refinement of the current best candidate
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] sub‑grad best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Cheap random local search
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sample to a finer grid (8192) if time permits
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.clip(a_big, 0.0, 1000.0)
        a_big = a_big / a_big.sum()

        # short Adam on the larger grid
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=150_000,
        )

        # longer sub‑gradient refinement on the large grid
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=150_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
