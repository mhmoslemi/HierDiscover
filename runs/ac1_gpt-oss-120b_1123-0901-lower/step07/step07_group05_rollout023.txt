```python
"""Block‑level Adam search for low‑autocorrelation step functions.

The algorithm first explores a pool of simple parametric families
(uniform, Dirichlet, edge‑exponential, power‑law, sparse spikes) on a
coarse block representation (e.g. 64 blocks).  Each candidate is refined
by Adam optimisation of a smooth‑max surrogate of the exact objective.
The best coarse solution is expanded to the full resolution (4096
entries) and further polished by a second Adam phase, a sub‑gradient
exponentiated‑gradient phase and a cheap stochastic hill‑climbing.
The whole search respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective without any scaling."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_block_smooth_max(m0: np.ndarray,
                           n: int,
                           B: int,
                           conv_fft_len: int,
                           corr_fft_len: int,
                           deadline: float,
                           global_best_seq: np.ndarray | None,
                           global_best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on a block‑wise probability vector.

    m0 – initial block masses (must sum to 1)
    Returns the best full‑resolution sequence seen during optimisation
    (as an array of length n) together with its exact objective.
    """
    eps = 1e-15
    logits = np.log(m0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.1

    # temperature for smooth‑max
    lam = 0.5
    lam_factor = 1.005
    lam_cap = 5e5

    block_len = n // B

    best_seq = global_best_seq
    best_val = global_best_val

    it = 0
    max_iter = 400_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # soft‑max over blocks to get a probability distribution
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        m_vec = exp_log / exp_log.sum()            # length B, sum=1

        # up‑sample to full resolution
        a = np.repeat(m_vec, block_len)            # length n, sum=1

        # exact objective (used only for tracking best)
        conv_full = _full_convolution(a, conv_fft_len)
        max_c = conv_full.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Block‑Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv_full - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # aggregate to block level
        grad_m = np.empty(B, dtype=np.float64)
        for b in range(B):
            start = b * block_len
            end = start + block_len
            grad_m[b] = grad_a[start:end].sum() / block_len

        # gradient w.r.t logits (soft‑max chain rule)
        avg = np.dot(m_vec, grad_m)
        grad_log = m_vec * (grad_m - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule – slowly focus on the true maximum
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake if stuck for a while
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(B) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     init_best_seq: np.ndarray | None,
                     init_best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on the full‑resolution vector (identical to the
    version used in the reference implementation, but with a more
    patient temperature schedule)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 2.0
    lam_factor = 1.012
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    max_iter = 300_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, lam_cap)

        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient refinement on the exact objective."""
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    eta0 = 0.8
    decay = 0.99995
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    max_iter = 80_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val

def _local_random_search(a_best: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """Cheap stochastic hill‑climbing in the log‑space of the vector."""
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val

def _generate_block_candidates(B: int, warm_start: np.ndarray | None):
    """Create a diverse pool of block‑level candidate probability vectors."""

The algorithm works in two stages:

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _adam_block_smooth_max(m0: np.ndarray,
                           n: int,
                           B: int,
                           conv_fft_len: int,
                           corr_fft_len: int,
                           deadline: float,
                           global_best_seq: np.ndarray | None,
                           global_best_val: float) -> tuple[np.ndarray, float]:
    """Adam on a block‑wise probability vector.

    m0 – initial block masses (must sum to 1).  Returns the best full‑resolution
    sequence seen during optimisation together with its exact objective.
    """
    eps = 1e-15
    logits = np.log(m0 + eps)
    m1 = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.1

    lam = 0.5               # start temperature for smooth‑max
    lam_factor = 1.005
    lam_cap = 5e5

    block_len = n // B
    best_seq = global_best_seq
    best_val = global_best_val

    it = 0
    max_iter = 400_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ---- soft‑max over blocks ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        m_vec = exp_log / exp_log.sum()          # length B, sums to 1

        # up‑sample to full resolution (each block repeats block_len times)
        a = np.repeat(m_vec, block_len)          # length n, sum = block_len

        # exact objective (just for bookkeeping)
        conv_full = _full_convolution(a, conv_fft_len)
        max_c = conv_full.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Block‑Adam] iter={it:06d} new best {val:.6f}")

        # ---- smooth‑max weights ----
        w = np.exp(lam * (conv_full - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # aggregate gradient to block level (sum over each block)
        grad_m = grad_a.reshape(B, block_len).sum(axis=1)

        # chain rule through the soft‑max (logits → m_vec)
        avg = np.dot(m_vec, grad_m)
        grad_log = m_vec * (grad_m - avg)

        # Adam update on logits
        m1 = beta1 * m1 + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m1 / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # slowly increase temperature – we focus more on the true max
        lam = min(lam * lam_factor, lam_cap)

        # random shake if we have been idle for a while
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(B) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     init_best_seq: np.ndarray | None,
                     init_best_val: float) -> tuple[np.ndarray, float]:
    """Adam on the full‑resolution vector (same structure as the reference)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m1 = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 2.0
    lam_factor = 1.012
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    max_iter = 300_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m1 = beta1 * m1 + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m1 / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, lam_cap)

        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient refinement on the exact objective."""
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    eta0 = 0.8
    decay = 0.99995
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    max_iter = 80_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _local_random_search(a_best: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """Cheap stochastic hill‑climbing in log‑space."""
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


def _generate_block_candidates(B: int,
                               warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of block‑level probability vectors."""
    candidates = []

    # warm‑start projected onto B blocks
    if warm_start is not None:
        n = warm_start.shape[0]
        block_len = n // B
        proj = np.array([warm_start[i * block_len:(i + 1) * block_len].mean()
                         for i in range(B)], dtype=np.float64)
        proj = np.maximum(proj, 0.0)
        s = proj.sum()
        if s > 0:
            proj /= s
            candidates.append(proj)

    # uniform
    candidates.append(np.full(B, 1.0 / B, dtype=np.float64))

    # Dirichlet draws with various concentrations
    for conc in (0.01, 0.03, 0.07, 0.15, 0.3, 0.7, 1.5):
        for _ in range(6):
            cand = np.random.dirichlet(np.full(B, conc))
            candidates.append(cand)

    i = np.arange(B, dtype=np.float64)

    # symmetric edge‑exponential families
    for alpha in np.logspace(-2, 1.0, num=10):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (B - 1 - i))
        edge /= edge.sum()
        for w in (0.3, 0.5, 0.8, 1.0):
            cand = w * edge + (1.0 - w) / B
            cand /= cand.sum()
            candidates.append(cand)

    # symmetric power‑law families
    for p in np.linspace(0.2, 5.0, num=10):
        edge = (i + 1.0) ** (-p) + (B - i) ** (-p)
        edge /= edge.sum()
        for w in (0.4, 0.7, 1.0):
            cand = w * edge + (1.0 - w) / B
            cand /= cand.sum()
            candidates.append(cand)

    # asymmetric two‑sided exponentials (block level)
    alphas = np.logspace(-2, 1.0, num=8)
    w_vals = np.linspace(0.0, 1.0, 5)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (B - 1 - i))
                    shape = wL * left + wR * right + wU / B
                    shape /= shape.sum()
                    candidates.append(shape)

    # sparse spike blocks
    for spikes in (2, 3, 5):
        for _ in range(12):
            idx = np.random.choice(B, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(B, dtype=np.float64)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.  Returns the best
    sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Base resolution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm‑start from a previously found construction if it exists
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – explore a coarse block grid
    B = 64                         # blocks (4096 = 64 × 64)
    block_len = n // B
    block_candidates = _generate_block_candidates(B, warm_start)

    cand_vals = []
    for cand in block_candidates:
        if time.time() >= deadline:
            break
        # up‑sample to full resolution and normalise to sum = 1 (for objective)
        a_full = np.repeat(cand, block_len)
        a_full = np.maximum(a_full, 0.0)
        s = a_full.sum()
        if s < 0.01:
            a_full = np.random.dirichlet(np.ones(n))
            s = a_full.sum()
        a_full /= s
        val = _objective(a_full, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        return best_seq.tolist()

    cand_vals.sort(key=lambda x: x[0])
    top_k = 8
    top_block_cands = [c for _, c in cand_vals[:top_k]]
    print(f"Stage 1: evaluated {len(cand_vals)} block candidates, best = {cand_vals[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – block‑level Adam optimisation on the selected seeds
    best_seq = None
    best_val = float("inf")
    for idx, blk_start in enumerate(top_block_cands):
        if time.time() >= deadline:
            break
        print(f"--- Block Adam run #{idx + 1}/{len(top_block_cands)} ---")
        best_seq, best_val = _adam_block_smooth_max(
            blk_start,
            n,
            B,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – full‑resolution refinement (Adam → sub‑gradient → local search)
    if best_seq is not None:
        print("--- Full‑resolution refinement ---")
        best_seq, best_val = _adam_smooth_max(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        best_seq, best_val = _subgradient_refinement(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4 – if enough time remains, repeat everything on a larger grid
    time_left = deadline - time.time()
    if time_left > 50.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)
        B_big = 128
        block_len_big = n_big // B_big

        # initialise a big‑grid vector by linear interpolation of the current best
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        # generate a few block candidates on the larger grid and optimise one of them
        block_cands_big = _generate_block_candidates(B_big, None)
        chosen = np.random.choice(len(block_cands_big), size=5, replace=False)
        best_seq_big = None
        best_val_big = float("inf")
        for idx in chosen:
            if time.time() >= deadline:
                break
            blk_start = block_cands_big[idx]
            best_seq_big, best_val_big = _adam_block_smooth_max(
                blk_start,
                n_big,
                B_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                best_seq_big,
                best_val_big,
            )
        if best_seq_big is not None:
            best_seq_big, best_val_big = _adam_smooth_max(
                best_seq_big,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                best_seq_big,
                best_val_big,
            )
            best_seq_big, best_val_big = _subgradient_refinement(
                best_seq_big,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                best_seq_big,
                best_val_big,
            )
            best_seq_big, best_val_big = _local_random_search(
                best_seq_big,
                n_big,
                conv_fft_len_big,
                deadline,
                best_val_big,
            )
            if best_val_big < best_val:
                best_seq = best_seq_big
                best_val = best_val_big
                n = n_big

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe default
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
