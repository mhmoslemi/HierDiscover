def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # Choose n based on remaining budget: start with 8192 will be heavy
    # But we can try n=8192 from start, ensures high dimensional space for better optimum.
    # However, if time low, fallback to 4096.

    # Determine if time enough: if budget_s > 800? maybe choose 8192 else 4096.

    if budget_s > 600:
        n = 8192
    else:
        n = 4096

    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    # If warm start exists in globals, use it (maybe length differs)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # interpolate up or down
            x_old = np.linspace(0,1,warm.shape[0],endpoint=False)
            x_new = np.linspace(0,1,n,endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.clip(warm,0,1000)
        warm = warm / max(warm.sum(),1e-12)

    # Build initial candidates
    # Start with a mixture of families:
    # - Dirichlet (full support)
    # - Edge exponential left and right
    # - Edge polynomial power
    # - Beta distributions (various alpha,beta)
    # - Two Gaussian mixture (different sigma)
    # - Sinusoid
    # - Uniform
    # - Sparse spikes (k = n//64 to n//4)
    # Provide list.

    # We'll generate maybe 30-40 candidates.

    candidates = []
    # Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))  # full support

    # Edge exponential
    for gamma in (0.1, 0.3, 0.6, 1.0, 3.0):
        left = np.exp(-gamma * np.arange(n))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        candidates.append(shape)

    # Edge polynomial
    idx = np.arange(1, n+1)
    prod = idx * (n - idx + 1)
    for alpha in np.linspace(-2.5, -0.5, 4):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        shape = shape / shape.sum()
        candidates.append(shape)

    # Beta
    for a in (0.2, 0.5, 0.8, 1.2, 2.0):
        for b in (0.2, 0.5, 0.8, 1.2, 2.0):
            x = (np.arange(n) + 0.5)/n
            shape = (x**(a-1)) * ((1-x)**(b-1))
            shape = shape / shape.sum()
            candidates.append(shape)

    # Two Gaussian
    i = np.arange(n)
    for sigma in (0.05, 0.12, 0.25, 0.45):
        g1 = np.exp(-0.5*((i-n/2)/ (sigma*n))**2) # centered
        shape = g1
        shape = shape / shape.sum()
        candidates.append(shape)
        # mixture with edges
        edge = np.exp(-0.5*((i)/(sigma*n))**2) + np.exp(-0.5*((i - (n-1))/(sigma*n))**2)
        shape = edge
        shape = shape / shape.sum()
        candidates.append(shape)

    # Sinusoid half-wave
    s = np.sin(np.pi * (np.arange(n) + 0.5)/ n)
    s = np.maximum(s,0)
    s /= s.sum()
    candidates.append(s)

    # Uniform
    candidates.append(np.full(n, 1/n))

    # Spike patterns
    for k in (16, 32, 64, 128, 256):
        a_spike = np.zeros(n)
        pos = np.round(np.linspace(0, n-1, k)).astype(int)
        a_spike[pos] = 1.0/k
        candidates.append(a_spike)

    # Add warm if exists
    if warm is not None:
        candidates.append(warm)
        # also shifted variants
        for shift in (n//8, n//4, n//2):
            candidates.append(np.roll(warm, shift))

    # Random Dirichlet with concentration <=1: draws many sparser patterns
    for conc in (0.05,0.2,0.5):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # Possibly we generate a large set but we need to limit.

    # Evaluate them quickly to keep top 5
    top_candidates = []
    best_val = float('inf')
    for cand in candidates:
        # ensure non-negative and normalized sum=1
        cand = np.clip(cand,0,1000)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s
        # compute objective
        conv = _full_convolution(cand, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (cand.sum()**2)  # sum=1
        if val < best_val:
            best_val = val
        if len(top_candidates) < 8:
            top_candidates.append((cand, val))
            # sort after inserted
            top_candidates.sort(key=lambda x: x[1])
        else:
            # if val better than worst in top list
            if val < top_candidates[-1][1]:
                top_candidates[-1] = (cand, val)
                top_candidates.sort(key=lambda x: x[1])

    print(f"Initial random pool best {best_val:.6f}")

    # Keep top candidates
    # Now for each top candidate, run a refinement: Adam + subgradient.

    best_seq = None
    best_val = float('inf')
    # We need to allocate time per candidate.
    # Let's compute remaining time.
    now = time.time()
    remaining = deadline - now
    # Use about 70% of remaining for refinement
    refine_budget = remaining * 0.7
    # allocate equal slice per candidate
    per_cand_budget = refine_budget / len(top_candidates) if top_candidates else 0

    for idx, (cand, val0) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"Refining candidate {idx} initial val {val0:.6f}")

        # set a local deadline for this candidate
        cand_deadline = min(deadline - 0.5, time.time() + per_cand_budget)

        # Adam phase
        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            cand_deadline,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"Candidate {idx} Adam improved to {val_adam:.6f}")

        # Subgradient phase
        if time.time() < deadline - 0.5:
            # tail subgrad up to remaining time for that candidate
            sub_deadline = deadline - 0.5
            a_sub, val_sub = _subgradient_optimize(
                best_seq if best_seq is not None else a_adam,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=300_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"Candidate {idx} Subgrad improved to {val_sub:.6f}")

    # After all candidate refinement, do final random local search
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"Final local search improved to {val_loc:.6f}")

    # Final fallback: ensure non-empty valid sequence.
    if best_seq is None:

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm proceeds in three stages:

1. **Diverse parametric warm‑starts** – a pool of vectors is built from many
   analytic families (exponential edges, power‑law, Beta densities,
   Gaussians, sinusoid, uniform, sparse spikes, Dirichlet draws and the
   optional previous best).  All are normalised to sum = 1 and the best few
   (by exact objective) are retained.

2. **Intensive continuous optimisation** – each retained candidate is refined
   by (a) an Adam optimiser on a smooth‑max surrogate (log‑sum‑exp) with a
   temperature that is slowly increased, and (b) a long exact‑max
   sub‑gradient phase implemented as exponentiated‑gradient (mirror descent)
   on the simplex.  The time budget is split among the candidates; the best
   overall solution is kept.

3. **Cheap local moves** – a final hill‑climbing stage swaps small fractions
   of mass between coordinates; any improvement updates the best vector.

All operations respect the global deadline (default 1000 s).  The returned
list is guaranteed to be non‑empty, non‑negative and within the evaluator’s
range, and the best value seen is printed for logging.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr0: float = 0.07,
    lam0: float = 0.01,
    lam_factor: float = 1.015,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector seen (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # works on the simplex via softmax
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    lr = lr0
    lam = lam0

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- projection onto simplex (soft‑max) -----
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()

        # ----- exact objective (for bookkeeping) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() is 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through the soft‑max (mirror‑descent form) -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e7)

        # ----- learning‑rate decay (gentle) -----
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        # ----- occasional shake near the end -----
        if it % 4000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Step size η_t = η₀ / sqrt(t+1).
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-14)
        if np.any(active):
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            cand_deadline,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Refine #{idx}] Adam improved to {val_adam:.6f}")

        # ----- Sub‑gradient phase -----
        if time.time() < deadline - 0.5:
            sub_deadline = deadline - 0.5
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=400_000,
                eta0=0.85,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Refine #{idx}] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # final cheap random hill‑climbing
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=180_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Final] local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # ensure a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
