"""
Propose a non‑negative step‑function (vector) minimizing

        F(a) = 2·n·max(a∗a) / (sum(a)²)

Algorithm
---------

1. **Generate a diversified pool of shapes** – exponential edges,
   power‑law edges, double‑Gaussian edges, a half‑wave sinusoid,
   uniform, Dirichlet samples (with several concentration values) and
   sparse “spike’’ patterns (both equally spaced and random).  All
   vectors are normalised to sum = 1 and clipped to the evaluator limits.

2. **Fast screening** – evaluate the exact objective for every candidate
   (using an FFT‑based convolution) and keep the best `K` (default 5).

3. **Refinement of the top candidates** – for each of the `K` vectors
   run a long Adam optimisation on a smooth‑max surrogate (log‑sum‑exp)
   with a slowly increasing temperature, followed by an exact‑max
   exponentiated‑gradient (sub‑gradient) phase.  The best vector seen
   during the run is retained.

4. **Deep polishing** – a final, longer sub‑gradient phase (≈ 400 k
   iterations) on the current best solution, then a cheap random‑move
   hill‑climber that swaps a fraction of mass between two coordinates.

5. **Optional up‑sampling** – if time permits, interpolate the best
   solution to a finer grid (8192 points) and repeat steps 3‑4.

All phases respect the global 1000 s deadline; the routine always
returns the best sequence found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective_val(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum(a)²)."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                                 # base learning rate
    lam = 0.05                                 # start temperature
    lam_factor = 1.035                        # slower increase than before

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex projection (soft‑max)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) chain rule
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 70_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    coordinate to another, keeping the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Shape generators (all return a normalised vector that sums to 1)
def _shape_exponential_edge(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


def _shape_power_edge(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    # safeguard against zero (should not happen)
    a = np.where(prod > 0, prod ** alpha, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


def _shape_double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    a = left + right
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


def _shape_sinusoid(n: int) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet(np.full(n, alpha))
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


def _shape_spike_equidist(n: int, k: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


def _shape_spike_random(n: int, k: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = 1.0 / k
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if previously available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a rich pool of candidate shapes
    pool = []

    # warm start and a few cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for sh in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, sh))

    # exponential edge family
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 1.5, 2.5, 5.0):
        pool.append(_shape_exponential_edge(n, gamma))

    # power‑law edge family (negative exponents)
    for alpha in (-0.2, -0.5, -0.8, -1.2, -1.5, -2.0):
        pool.append(_shape_power_edge(n, alpha))

    # double‑Gaussian family
    for sigma in (0.02, 0.04, 0.06, 0.1, 0.2, 0.4):
        pool.append(_shape_double_gaussian(n, sigma))

    # sinusoid, uniform
    pool.append(_shape_sinusoid(n))
    pool.append(_shape_uniform(n))

    # Dirichlet samples with varied concentration
    for alpha in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        pool.append(_shape_dirichlet(n, alpha))

    # Equally spaced spikes
    for k in (16, 32, 64, 128, 256):
        pool.append(_shape_spike_equidist(n, k))

    # Random sparse spikes
    for _ in range(4):
        k = np.random.choice([8, 16, 32, 64, 128])
        pool.append(_shape_spike_random(n, k))

    # ------------------------------------------------------------------
    # Quick evaluation of the whole pool
    cand_vals = []
    for vec in pool:
        val = _objective_val(vec, n, conv_fft_len)
        cand_vals.append((vec, val))

    # keep the top K promising seeds
    cand_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_candidates = [c for c, _ in cand_vals[:top_k]]

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Refine each top candidate with Adam + sub‑gradient
    for cid, cand in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[Main] Optimising top candidate {cid}")

        # Adam phase (smooth‑max)
        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam new best {val_adam:.6f}")

        # Sub‑gradient refinement (short)
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=120_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad new best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Deep sub‑gradient polishing of the overall best
    if best_seq is not None and time.time() < deadline - 5:
        print("[Refine] Long sub‑gradient run")
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Cheap random local search
    if best_seq is not None and time.time() < deadline - 5:
        print("[Local] Random hill‑climb")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=90_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a finer grid (8192) if time permits
    if time.time() + 30 < deadline:
        print("[Upsample] Interpolating to 8192 points")
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.clip(a_big, 0.0, 1000.0)
        a_big = a_big / a_big.sum()

        # short Adam on larger grid
        a_big, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=180_000,
        )

        # longer sub‑gradient on large grid
        a_big, val_big = _subgradient_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=150_000,
            eta0=0.8,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Upsample] improved best to {val_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Final safety clipping (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
