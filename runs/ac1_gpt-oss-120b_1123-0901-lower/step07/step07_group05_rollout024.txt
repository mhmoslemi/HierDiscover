def _fine_subgradient(a0, n, conv_fft_len, corr_fft_len, deadline, init_best_val):
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()

        a_sum = a.sum()

    eta = eta0

    best_val = init_best_val
    last_improve = time.time()

            best_val = val
            best_seq = a.copy()
            last_improve = time.time()

        active = np.isclose(conv, max_c, atol=1e-12)

        a /= s

"""
Propose a step‑function (non‑negative vector) that minimises the
evaluation function 2·n·max(conv(a,a))/sum(a)^2.

Algorithm
---------
1. **Generate a diverse parametric pool** (uniform, Dirichlet, exponentials,
   power‑laws, logistic decays, triangular / sinusoidal shapes, sparse spikes,
   and shifted warm‑starts).  All vectors are normalised to sum ≈ 1.

2. **Quick evaluation** of every candidate using FFT‑based convolution.
   Keep the top K (default 12) for intensive optimisation.

3. **Deep optimisation** – for each of the top candidates run two phases:
      • Adam on a smooth‑max surrogate of the convolution (high temperature,
        long schedule, Adam learning‑rate = 0.07).
      • An extended exponentiated‑gradient sub‑gradient descent on the exact
        objective (≈ 300 k iterations, slowly decreasing step‑size).

   The best vector found so far is tracked globally.

4. **Fine‑grained refinement** – a final long sub‑gradient phase on the
   current global best (no active‑set cutoff, tiny learning‑rate).

5. **Local stochastic hill climb** – cheap Gaussian perturbations of the
   log‑probabilities that are kept if they improve the exact bound.

6. **Optional up‑sampling** – if enough time remains, interpolate the best
   vector to a larger grid (n = 8192) and repeat the deep optimisation.

All loops respect a global deadline (the supplied budget) and stop early if
necessary.  The returned list is clipped to the range required by the
evaluator.  The code uses only NumPy and standard library modules.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective: 2·n·max(a*a)/(sum(a)^2)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                     # slightly larger step‑size than before
    lam = 0.5                     # start with a gentle temperature
    lam_factor = 1.005            # grow slowly → smoother optimisation
    lam_cap = 1e6

    it = 0
    max_iter = 600_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ------- simplex via soft‑max -------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ------- surrogate convolution -------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional shake if we are stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 300_000,
    eta0: float = 0.5,
    decay: float = 0.99995,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the *exact* max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    eta = eta0
    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[SubGrad] iter={it:07d} new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # early stop if no improvement for a while
        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam‑smooth‑max followed by an extended sub‑gradient phase.
    Both phases update the global best.
    """
    # Adam phase
    seq, val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Sub‑gradient phase (more iterations than the original 80 k)
    seq, val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        seq,
        val,
        max_iter=300_000,
        eta0=0.5,
    )
    return seq, val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a rich pool of candidate vectors.
    Families:
        – warm‑start + circular shifts
        – uniform
        – Dirichlet draws (various concentrations)
        – symmetric edge‑exponential / power‑law
        – asymmetric two‑sided exponentials / power‑laws
        – linear / triangular
        – logistic decays
        – half‑wave sine
        – truncated power‑law blocks
        – sparse spike vectors.
    All candidates are normalised to sum ≈ 1.
    """
    candidates = []

    # ----- warm‑start & shifts -----
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ----- uniform -----
    candidates.append(np.full(n, 1.0 / n))

    # ----- Dirichlet draws -----
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(15):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # ----- symmetric edge exponentials -----
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ----- symmetric power‑law -----
    for p in np.linspace(0.2, 6.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ----- asymmetric two‑sided exponentials -----
    alphas = np.logspace(-2, 1.0, num=10)
    w_vals = np.linspace(0.0, 1.0, 6)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ----- asymmetric power‑law -----
    p_vals = np.linspace(0.2, 5.0, num=8)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ----- simple linear / triangular -----
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        shape = base / base.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * shape + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ----- logistic decays -----
    for beta in np.logspace(-2, 1.0, num=8):
        logistic = 1.0 / (1.0 + np.exp(beta * (i - n / 2)))
        logistic /= logistic.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * logistic + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # ----- half‑wave sinusoid -----
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ----- truncated power‑law blocks -----
    for p in np.linspace(0.3, 4.0, num=6):
        for frac in np.linspace(0.3, 0.9, 4):
            cut = int(frac * n)
            tmp = np.zeros(n)
            tmp[:cut] = (np.arange(1, cut + 1, dtype=np.float64)) ** (-p)
            tmp += 1e-12               # avoid exact zeros
            tmp /= tmp.sum()
            candidates.append(tmp)

    # ----- sparse spike vectors -----
    for spikes in (3, 5, 8, 12):
        for _ in range(25):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates

def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: add Gaussian noise to the logits of the current
    best vector, evaluate the exact objective and keep the improvement.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.025
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


def _fine_subgradient(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    A long, low‑temperature exponentiated‑gradient sub‑gradient run on the
    current global best.  This is intended to push the bound a few
    10⁻⁴ lower.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    eta = 0.25                    # start smaller than the earlier phases
    decay = 0.99997
    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[FineSub] iter={it:07d} new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # stop if we see no progress for several seconds
        if time.time() - last_improve > 7.0:
            break

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.  Returns the best sequence
    found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Base dimension (the value 4096 gave the previous best)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of cheap candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for cand in all_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"Stage 1: evaluated {len(cand_vals)} candidates, best so far = {cand_vals[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    for idx, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – fine‑grained sub‑gradient on the global best
    if best_seq is not None:
        print("--- Fine‑grained sub‑gradient refinement ---")
        best_seq, best_val = _fine_subgradient(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4 – cheap stochastic hill‑climbing from the current best
    if best_seq is not None:
        print("--- Local stochastic refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 5 – if generous time remains, try a larger grid (n = 8192)
    time_left = deadline - time.time()
    if time_left > 45.0:               # need a meaningful budget
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # Keep the larger vector if it improves the bound
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
