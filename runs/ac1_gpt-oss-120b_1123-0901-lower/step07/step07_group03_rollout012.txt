"""
Hybrid random‑restart optimizer for the step‑function bound.

The algorithm builds a pool of diverse warm‑starts (uniform, Dirichlet,
anti‑pair ordering, edge‑focused Gaussians, triangle), picks the best
initial shape and then refines it by three stages:

1. **Smooth‑max Adam** – minimises a log‑sum‑exp surrogate of
   max(a∗a) with an annealing temperature (λ) that grows slowly.
   Adam’s momentum and adaptive steps accelerate convergence.

2. **Exact‑max sub‑gradient** – after the surrogate has sharpened,
   a classic exponentiated‑gradient sub‑gradient descent drives the
   true max‑convolution down.

3. **Stochastic hill‑climbing** – a simulated‑annealing style local
   search (swap or fractional transfer moves) further reduces the
   exact objective.

If time permits the solution is interpolated to a finer grid
(8192 points) and the three stages are repeated on the larger
representation, exploiting the higher degrees of freedom.

All operations stay on the simplex (non‑negative, sum = 1) and all
vectors are clipped to the evaluator’s bounds before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) via FFT.
    Returns the first 2·len(a)‑1 entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Both arguments are real, w has length 2·len(a)‑1.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum = 1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) (sum of a is assumed to be 1)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Candidate factories ----------------------------------------------------
def _warm_start(n: int):
    """Extract a warm start from the global variable if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm = warm / warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Anti‑pair ordering:
    largest opposite smallest, 2nd largest opposite 2nd smallest, …
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]           # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]              # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]      # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet sample followed by anti‑pair ordering."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape – large at the edges, small in the middle."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    """Two symmetric exponential‑type bumps at the ends."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape

# Optimisation primitives -------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate of
    max(a∗a).  `lam` plays the role of inverse temperature and
    increases slowly.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # work in the log‑space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # map logits → simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for book‑keeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp gradient)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # centre the gradient (log‑space gradient for simplex)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to help escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using the classic exponentiated‑gradient
    update (mirror descent on the simplex).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update
        a = a * np.exp(-eta * grad_a)

        # re‑project onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing.
    Moves:
        * swap two positions,
        * transfer a random fraction (up to 40 %) of mass from i → j.
    Accept improvement always; otherwise accept with probability exp(−Δ/T)
    where the temperature decays linearly from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap move
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                               # transfer move
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 %
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.
    Uses a hybrid random‑restart + gradient + annealing strategy to
    produce a low‑value step‑function within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Base resolution
    n0 = 4096
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # ------------------------------------------------------------------
    # 1) Assemble a diverse set of starters and pick the best one
    starters = []

    warm = _warm_start(n0)
    if warm is not None:
        starters.append(warm.copy())
        for shift in (n0 // 4, n0 // 2, 3 * n0 // 4):
            starters.append(np.roll(warm, shift))

    starters.append(_uniform_candidate(n0))

    for alpha in (0.3, 0.7, 1.0, 2.0):
        starters.append(_random_dirichlet_candidate(n0, alpha))

    for alpha in (0.3, 0.7, 1.0):
        starters.append(_anti_pair_candidate(n0, alpha))

    for gamma in (0.2, 0.5, 1.0):
        starters.append(_gaussian_edges_candidate(n0, gamma))

    starters.append(_triangle_candidate(n0))

    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(starters):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n0))
        cand /= cand.sum()
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"Initial best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Core optimisation on the 4096‑point grid
    # Reserve some seconds for the fine‑grid phase
    final_upscale_deadline = deadline - 30.0

    # ---- Adam phase (smooth‑max)
    adam_deadline = min(deadline - 20.0, time.time() + 220.0)   # ~220 s max
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n0,
        conv_fft_len0,
        corr_fft_len0,
        adam_deadline,
        max_iters=250_000,
        lr=0.07,
        lam_start=0.5,
        lam_factor=1.05,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # ---- Sub‑gradient refinement
    sub_deadline = min(deadline - 10.0, time.time() + 260.0)   # ~260 s max
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n0,
        conv_fft_len0,
        corr_fft_len0,
        sub_deadline,
        max_iters=260_000,
        eta0=0.8,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Subgrad] improved to {val_sub:.6f}")

    # ---- Local stochastic hill‑climbing
    if time.time() < deadline - 5.0:
        local_deadline = deadline - 5.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n0,
            conv_fft_len0,
            local_deadline,
            max_moves=300_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional up‑sampling to a finer grid (8192 points)
    if time.time() < final_upscale_deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n0, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # ---- Adam on the larger grid
        remaining = deadline - time.time()
        ad_big_deadline = time.time() + remaining * 0.45
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            ad_big_deadline,
            max_iters=250_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        # ---- Sub‑gradient refinement on the large grid
        sub_big_deadline = deadline
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_big_deadline,
            max_iters=250_000,
            eta0=0.8,
        )
        # ---- Final local search on the large grid (if any time left)
        if time.time() < deadline - 2.0:
            local_big_deadline = deadline - 2.0
            a_big_loc, val_big_loc = _local_random_search(
                a_big_sub,
                n_big,
                conv_fft_len_big,
                local_big_deadline,
                max_moves=200_000,
            )
            if val_big_loc < val_big_sub:
                a_big_sub, val_big_sub = a_big_loc, val_big_loc

        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Big] improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0, dtype=np.float64)

    # Clip to the range accepted by the evaluator (0…1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
