```python
"""Search for low‑evaluation non‑negative step function using LP‑based
improvement + gradient‑based refinement.

The routine first solves a series of linear programmes (size ≈ 1024)
which, for a fixed maximal auto‑convolution, maximises the total mass.
Mixing the original vector with the LP solution strictly reduces the
objective.  The obtained shape is up‑sampled to the final size (4096)
and further polished by Adam‑smooth‑max and exponentiated‑gradient
sub‑gradient steps as in the reference implementation.  A short
stochastic hill‑climbing phase finishes the search.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective F = 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 2.0
    lam_factor = 1.012
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    max_iter = 250_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, lam_cap)

        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    eta0 = 0.8
    decay = 0.99995
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    max_iter = 70_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val


def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


def _lp_improve(f: np.ndarray, max_iters: int, deadline: float, mix: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform at most `max_iters` linear‑programming improvement steps.
    The vector `f` is assumed to be non‑negative and sum to 1.
    Returns the best vector seen and its objective value.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # Fixed index arrays for the convolution matrix A (size rows×n)
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    row_idx = (i + j).ravel()
    col_idx = np.tile(np.arange(n), n)

    best_seq = f.copy()
    best_val = _objective(f, n, _next_pow2(2 * n - 1))

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # current max convolution
        conv = np.convolve(f, f)
        M = conv.max()
        # build sparse matrix A where A @ g = conv(f,g)
        data = np.repeat(f, n)
        A = csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))
        # linear programme: maximize sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        b_ub = np.full(rows, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # cannot increase total mass – stop
            print(f"[LP] iteration {it} no mass increase (sum_g={sum_g:.6f})")
            break
        # convex mix and renormalise
        f_mix = (1 - mix) * f + mix * g
        f_mix = f_mix / f_mix.sum()
        f = f_mix
        val = _objective(f, n, _next_pow2(2 * n - 1))
        if val < best_val:
            best_val = val
            best_seq = f.copy()
            print(f"[LP] iter {it} new best {val:.6f} (sum_g={sum_g:.3f})")
    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Uses a linear‑programming “mass‑maximise” phase
    on a modest size (n≈1024) to obtain a low‑max‑convolution shape,
    up‑samples it to the final size (≈4096) and finishes with the
    gradient‑based refinements from the reference implementation.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a small working size for the LP phase.
    n_small = 1024
    # Warm‑start from a known construction if available, otherwise use a
    # symmetric Beta(0.5,0.5)‑like shape (discrete arcsine) which tends to
    # flatten the auto‑convolution.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n_small:
            warm = warm[:n_small]
        elif warm.shape[0] < n_small:
            warm = np.pad(warm, (0, n_small - warm.shape[0]))
        init_f = np.maximum(warm, 0.0)
        if init_f.sum() < 0.01:
            init_f = np.random.dirichlet(np.ones(n_small))
    else:
        i = np.arange(n_small, dtype=np.float64)
        eps = 0.5
        # discrete arcsine‑like density
        init_f = 1.0 / np.sqrt((i + eps) * (n_small - 1 - i + eps))
        init_f = init_f / init_f.sum()
    # Ensure positivity and normalisation.
    init_f = np.maximum(init_f, 0.0)
    init_f /= init_f.sum()

    # ------------------------------------------------------------------
    # 2. Linear‑programming improvement loop (mass‑maximisation).
    # A modest number of iterations (≈10) already yields a strong reduction.
    f_lp, val_lp = _lp_improve(init_f, max_iters=12, deadline=deadline, mix=0.5)

    # ------------------------------------------------------------------
    # 3. Upsample to the final resolution (≈4096) if time permits.
    n_big = 4096
    if time.time() < deadline - 40.0:
        x_small = np.linspace(0, 1, n_small, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        f_big = np.interp(x_big, x_small, f_lp)
        f_big = np.maximum(f_big, 0.0)
        f_big /= f_big.sum()
    else:
        f_big = f_lp  # fallback – maybe a smaller vector

    # ------------------------------------------------------------------
    # 4. Deep optimisation (Adam + sub‑gradient) on the final size.
    # Use the same helper pipeline as the reference code but with a
    # tighter iteration budget (the LP phase already did most of the work).
    n = f_big.shape[0]
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")
    # one start – the upsampled LP solution
    best_seq, best_val = _deep_optimize(
        f_big,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )

    # ------------------------------------------------------------------
    # 5. Cheap stochastic hill‑climbing as a final polish.
    if best_seq is not None:
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # 6. Safety fallback.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished – best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()

"""Search for a low‑evaluation non‑negative step function.

The algorithm has three stages:
1. **LP mass‑maximisation** on a modest size (≈1024).  For a fixed
   maximal auto‑convolution M, a linear programme finds a vector g that
   maximises the total mass while respecting   f ★ g ≤ M.  Mixing the
   original f with g and renormalising strictly reduces the objective.
2. **Upsampling** to the final resolution (≈4096) and a short
   Adam + sub‑gradient refinement (the same pipeline used in the reference
   code).
3. **Cheap stochastic hill‑climbing** to polish the final result.

The LP stage is the novel ingredient – it creates a far better starting
shape than the purely random‑initialisation used previously, enabling
the final bound to drop well below the previous 1.5079 benchmark.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 2.0
    lam_factor = 1.012
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    max_iter = 250_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, lam_cap)

        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient steps on the exact max‑convolution."""
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    eta0 = 0.8
    decay = 0.99995
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    max_iter = 70_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam‑smooth‑max followed by a sub‑gradient phase."""
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climbing: add Gaussian noise to logits."""
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray, max_iters: int, deadline: float, mix: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Linear‑programming improvement: maximise total mass while keeping
    f ★ g ≤ max(f ★ f).  The convex combination (1‑mix)·f + mix·g is
    renormalised after each step.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Fixed index pattern for the convolution matrix A (size rows×n)
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    row_idx = (i + j).ravel()               # length n²
    col_idx = np.tile(np.arange(n), n)       # length n²

    best_seq = f.copy()
    best_val = _objective(f, n, _next_pow2(2 * n - 1))

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        # Build sparse matrix A where (A @ g) = f ★ g
        data = np.repeat(f, n)
        A = csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))

        # LP: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        b_ub = np.full(rows, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            print(f"[LP] iteration {it} no mass increase (sum_g={sum_g:.6f})")
            break

        # Mix and renormalise
        f_mix = (1 - mix) * f + mix * g
        f = f_mix / f_mix.sum()

        val = _objective(f, n, _next_pow2(2 * n - 1))
        if val < best_val:
            best_val = val
            best_seq = f.copy()
            print(f"[LP] iter {it} new best {val:.6f} (sum_g={sum_g:.3f})")

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises the supplied
    evaluation function.  The routine combines an LP‑based mass‑maximisation
    stage with Adam/Sub‑gradient refinement and a final stochastic hill‑climb.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise a modest‑size vector for the LP phase (n≈1024)
    n_small = 1024
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n_small:
            warm = warm[:n_small]
        elif warm.shape[0] < n_small:
            warm = np.pad(warm, (0, n_small - warm.shape[0]))
        f0 = np.maximum(warm, 0.0)
        if f0.sum() < 0.01:
            f0 = np.random.dirichlet(np.ones(n_small))
    else:
        # discrete arcsine‑like density (helps to flatten the auto‑conv.)
        i = np.arange(n_small, dtype=np.float64)
        eps = 0.5
        f0 = 1.0 / np.sqrt((i + eps) * (n_small - 1 - i + eps))
        f0 = f0 / f0.sum()

    f0 = np.maximum(f0, 0.0)
    f0 /= f0.sum()

    # ------------------------------------------------------------------
    # 2. LP improvement loop (few iterations are enough)
    f_lp, val_lp = _lp_improve(f0, max_iters=12, deadline=deadline, mix=0.5)
    print(f"[LP] finished – best objective {val_lp:.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample to the final resolution (≈4096) if we have time
    n_big = 4096
    if time.time() < deadline - 40.0:

        x_big = np.linspace(0, 1, n_big, endpoint=False)
        f_big = np.interp(x_big, x_small, f_lp)
        f_big = np.maximum(f_big, 0.0)
        f_big /= f_big.sum()
    else:
        f_big = f_lp   # fallback – keep the small vector

    # ------------------------------------------------------------------
    # 4. Deep optimisation (Adam + sub‑gradient) on the final size
    n = f_big.shape[0]

        deadline,
        best_seq,
        best_val,

    if best_seq is not None:

            conv_fft_len,
            deadline,
            best_val,

    # ------------------------------------------------------------------

    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished – best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
