def _flatten_exponential_optimize(a0, n, conv_len, corr_len, deadline, max_iters=100_000, eta0=0.5):
    # exponentiated gradient descent with flatten objective, symmetrize.
    a = a0.copy()

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

Algorithm
---------
1. Build a modest, diverse set of start vectors (Dirichlet, edge‑exponential,
   edge‑polynomial, two‑Gaussian, sinusoid, uniform and sparse spikes).  If a
   previous solution is available it is added as a warm‑start.

2. **Flatten phase** – for each start vector run a cheap exponentiated‑gradient
   descent that minimizes the L2 deviation of the auto‑convolution from its
   average value (i.e. makes the convolution as flat as possible).  The update
   is
          a ← a·exp(−η·grad)   with   grad = 4·cross_corr(conv−avg, a)
   and the vector is re‑normalised after each step.  This phase quickly finds
   shapes with a low maximal convolution value.

3. Keep the few best candidates from the flatten phase and run a longer
   exact‑max sub‑gradient refinement (the same mirror‑descent update used in
   the previous state‑of‑the‑art).  This drives the true objective down.

4. Perform a cheap random‑move hill‑climbing (swap a random fraction of mass
   between two coordinates) to polish the solution.

5. Return the best sequence found.  All steps respect the global deadline
   (default 1000 s) and the routine always returns a valid non‑negative list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (full support)
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with a small number of equally spaced spikes."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


# ----------------------------------------------------------------------
def _flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent that flattens the auto‑convolution.
    The loss is Σ (conv - avg)^2 where avg = 1/(2 n‑1).  The update
    uses gradient = 4·cross_corr(conv-avg, a).  After each step the vector
    is symmetrised (a←(a+a[::-1])/2) and re‑normalised.
    Returns the best vector (by the original evaluation metric) seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    avg = 1.0 / (2 * n - 1)          # constant average of the convolution
    best_a = a.copy()
    best_val = float("inf")

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[flatten it={it:05d}] new best {val:.6f}")

        # gradient of the flattening loss
        err = conv - avg
        grad = 4.0 * _cross_correlation(err, a, corr_fft_len)

        # step size schedule
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad)

        # optional symmetry enforcement (helps to keep the convolution balanced)
        a = (a + a[::-1]) / 2.0

        # renormalise – protect against under‑/overflow
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Step size η_t = η₀ / √t.
    Returns the best vector seen and its exact objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: randomly move a fraction of mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # build candidate pool
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # ------------------------------------------------------------------
    # Phase 1 – flatten optimisation on many starts (quick)
    flatten_deadline = min(deadline, time.time() + 0.15 * budget_s)
    flatten_results = []          # (value, vector)
    for cid, cand in enumerate(pool):
        if time.time() >= flatten_deadline:
            break

        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        a_flat, val_flat = _flatten_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            flatten_deadline,
            max_iters=6_000,      # cheap per‑candidate run
            eta0=0.6,
        )
        flatten_results.append((val_flat, a_flat))
        print(f"[flatten cand {cid}] best val {val_flat:.6f}")

    # keep the top few flatten candidates
    flatten_results.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in flatten_results[:5]]

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement on the best few
    best_seq = None
    best_val = float("inf")
    for idx, a_start in enumerate(top_candidates):
        if time.time() >= deadline - 5:
            break
        a_ref, val_ref = _subgradient_optimize(
            a_start,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=150_000,
            eta0=0.85,
        )
        print(f"[refine {idx}] subgrad val {val_ref:.6f}")
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()

    # ------------------------------------------------------------------
    # Phase 3 – cheap local random search
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[local] improved best to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # fallback – ensure a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
