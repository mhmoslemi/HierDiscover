def _smooth_max_and_grad(logits, n, conv_fft_len, corr_fft_len, lam):
    # compute a via softmax
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()
    # compute conv via FFT
    conv = _full_convolution(a, conv_fft_len)
    # compute log-sum-exp
    # compute max_c = conv.max()
    max_c = conv.max()
    # w normalized exponent
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        # fallback uniform weights
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    else:
        w /= w_sum
    # gradient with respect to a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    # convert to gradient wrt logits (softmax)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    # smooth max value:
    smooth_val = max_c + np.log(w_sum) / lam   # equals log-sum-exp/lam
    return smooth_val, grad_log

"""
Propose a non‑negative step‑function that minimizes
        F(a) = 2·n·max(a∗a) / (sum(a)^2)

Algorithm
---------

*   The problem is tackled on several resolutions (4096 → 8192 → 16384)
    using a *continuation* of smooth‑max optimisation.
*   A soft‑max parametrisation (logits → a = softmax(logits)) keeps the
    vector on the probability simplex.
*   For a given temperature λ we minimise the smooth approximation
    of the max‑convolution, i.e.  (1/λ)·log∑exp(λ·(a∗a)),
    whose gradient w.r.t. logits is cheap to compute via FFT
    (full convolution + cross‑correlation).
*   The smooth problem is solved by L‑BFGS‑B (scipy) – a few hundred
    iterations are enough because the surrogate is smooth.
*   λ is increased geometrically (5 → 10 → 30 → … → 3·10⁴) so the
    optimiser gradually focuses on the true max.
*   After each resolution the best vector is **interpolated**
    to a finer grid and the optimisation continues.
*   A cheap hill‑climbing random‑move phase (mass transfer between
    two indices) is applied as a final polish.
*   The routine respects the global budget (default 1000 s) and always
    returns the best sequence found.  All helper objects are top‑level
    (no closures or lambda‑functions).

The implementation uses only NumPy, SciPy and the provided
`evaluate_sequence` for final scoring.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _smooth_max_and_grad(logits: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         corr_fft_len: int,
                         lam: float):
    """
    Returns (smooth_max, gradient_wrt_logits) for the
    log‑sum‑exp surrogate of max(a*a) with temperature lam.
    """
    # soft‑max → simplex
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    # convolution of a with itself
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()

    # log‑sum‑exp / lam  (smooth approximation of max)
    w = np.exp(lam * (conv - max_c))          # non‑positive exponents
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    else:
        w /= w_sum

    smooth_val = max_c + np.log(w_sum) / lam   # = (log‑sum‑exp)/lam

    # gradient w.r.t. a : Σ w_k ∂conv_k/∂a = 2·(w ★ a)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through soft‑max
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)

    return smooth_val, grad_log


# ----------------------------------------------------------------------
class _LBFGSFunc:
    """Callable object required by scipy.optimize for L‑BFGS."""
    def __init__(self, n, conv_fft_len, corr_fft_len, lam):
        self.n = n
        self.conv_fft_len = conv_fft_len
        self.corr_fft_len = corr_fft_len
        self.lam = lam

    def __call__(self, logits):
        return _smooth_max_and_grad(logits,
                                    self.n,
                                    self.conv_fft_len,
                                    self.corr_fft_len,
                                    self.lam)


class _LBFGSCallback:
    """Callback that tracks the best exact objective and enforces the deadline."""
    def __init__(self, n, conv_fft_len, deadline):
        self.n = n
        self.conv_fft_len = conv_fft_len
        self.deadline = deadline
        self.best_val = float("inf")
        self.best_a = None

    def __call__(self, xk):
        if time.time() > self.deadline:
            raise KeyboardInterrupt

        # compute exact value from the current logits
        mx = xk.max()
        exp_log = np.exp(xk - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, self.conv_fft_len)
        val = 2.0 * self.n * conv.max()
        if val < self.best_val:
            self.best_val = val
            self.best_a = a.copy()
            print(f"[L‑BFGS] new best exact {val:.6f}")


def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    corr_fft_len: int,
                    lam: float,
                    maxiter: int,
                    deadline: float):
    """
    One L‑BFGS optimisation stage for a fixed smoothing parameter lam.
    Returns the best vector seen and its exact objective.
    """
    logits0 = np.log(np.maximum(a0, 1e-15))

    func = _LBFGSFunc(n, conv_fft_len, corr_fft_len, lam)
    cb = _LBFGSCallback(n, conv_fft_len, deadline)

    try:
        minimize(func,
                 logits0,
                 method='L-BFGS-B',
                 jac=True,
                 callback=cb,
                 options={'maxiter': maxiter, 'disp': False})
    except KeyboardInterrupt:
        pass    # timeout – best solution stored in the callback

    if cb.best_a is not None:
        best_a = cb.best_a
        best_val = cb.best_val
    else:
        # fallback: use final point
        mx = logits0.max()
        exp_log = np.exp(logits0 - mx)
        best_a = exp_log / exp_log.sum()
        conv = _full_convolution(best_a, conv_fft_len)
        best_val = 2.0 * n * conv.max()
    return best_a, best_val


def _continuation_optimize(a_start: np.ndarray,
                           n: int,
                           conv_fft_len: int,
                           corr_fft_len: int,
                           lam_seq,
                           iters_per,
                           deadline: float):
    """
    Run several L‑BFGS stages while slowly increasing the smoothing
    parameter lam.  Returns the best vector and its exact objective.
    """
    a = a_start.copy()
    best_val = float("inf")
    best_a = a.copy()

    for lam in lam_seq:
        if time.time() >= deadline:
            break
        a, val = _lbfgs_optimize(a,
                                 n,
                                 conv_fft_len,
                                 corr_fft_len,
                                 lam,
                                 maxiter=iters_per,
                                 deadline=deadline)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Continuation] lam={lam:.0e} → best {val:.6f}")
    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 150_000):
    """
    Very cheap hill‑climbing: transfer a random fraction of mass from
    one index to another and keep the move if it reduces the exact
    objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local] move {it} improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Base resolution (n = 4096)
    n0 = 4096

    warm = globals().get('height_sequence_1')
    if warm is not None:

        if warm.size > n0:
            warm = warm[:n0]
        elif warm.size < n0:
            warm = np.pad(warm, (0, n0 - warm.size))
        warm = np.maximum(warm, 0.0)
        s = warm.sum()
        if s >= 0.01:
            warm = warm / s

    # ------------------------------------------------------------------

    starts = []
    if warm is not None:

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Optimise each start with a continuation of L‑BFGS
    lam_schedule0 = [5, 20, 80, 300, 1_000, 3_000, 10_000, 30_000]

    # ------------------------------------------------------------------
