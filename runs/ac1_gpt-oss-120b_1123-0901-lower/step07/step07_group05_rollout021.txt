"""
Optimise a non‑negative step‑function (a probability vector) so that
    F(a) = 2·n·max(a * a) / (sum a)^2
is as small as possible.
The strategy mixes a huge, diverse parametric pool (beta‑shapes,
Gaussian, Kaiser, exponential / power‑law, sinusoid, Dirichlet, …)
with a long Adam optimisation on a smooth‑max surrogate followed by
an extended exponentiated‑gradient sub‑gradient phase and a final
cheap stochastic hill‑climber.  The whole search stops after the
global 1000 s budget, always keeping the best sequence seen.
"""

import time
import math
import numpy as np
from scipy.special import i0   # Kaiser window

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (size = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective (sum(a) is assumed to be 1)."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------

def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     fft_len: int,
                     corr_len: int,
                     deadline: float,
                     best_seq: np.ndarray | None,
                     best_val: float) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp (smooth‑max) surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02                     # a bit smaller than before – more stable

    lam = 0.5                     # start temperature (smoothness)
    lam_factor = 1.005            # slower growth → smoother transition
    lam_cap = 2e6

    it = 0
    max_iter = 600_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ------- soft‑max onto simplex -------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ------- smooth‑max surrogate -------
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # because sum(a)=1

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] it={it:06d}  new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            fft_len: int,
                            corr_len: int,
                            deadline: float,
                            best_seq: np.ndarray | None,
                            best_val: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution."""
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    eta0 = 0.7
    decay = 0.99994
    eta = eta0

    it = 0
    max_iter = 200_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Sub] it={it:06d}  new best {val:.6f}")

        # active set where convolution reaches its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # stop if stagnating for a while
        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_val


def _deep_optimize(start_vec: np.ndarray,

def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^p · (n‑i)^q (p,q>-1)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    left = i ** p
    right = (n - np.arange(n, dtype=np.float64)) ** q
    shape = left * right
    shape[shape <= 0] = 0.0
    s = shape.sum()
    if s < 1e-12:
        return np.full(n, 1.0 / n)
    return shape / s


def _gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = shape.sum()
    return shape / s


def _kaiser_shape(n: int, beta: float) -> np.ndarray:
    """Kaiser window (β≥0)."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0               # in [-1,1]
    w = i0(beta * np.sqrt(1.0 - x * x))
    shape = w / w.sum()
    return shape


def _sin_power_shape(n: int, power: float) -> np.ndarray:
    """(sin)^{power} window."""
    i = np.arange(n, dtype=np.float64) + 0.5
    shape = np.sin(np.pi * i / n) ** power
    shape[shape <= 0] = 0.0
    return shape / shape.sum()


def _exponential_mix(n: int, alphaL: float, alphaR: float, wL: float, wR: float) -> np.ndarray:
    """Two‑sided exponential mixture (asymmetric)."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-alphaL * i)
    right = np.exp(-alphaR * (n - 1 - i))
    shape = wL * left + wR * right + (1.0 - wL - wR) / n
    return shape / shape.sum()

        # random order of indices improves chances
        indices = np.random.permutation(n)
        for idx in indices:
            # compute prospective new sums
            new_sums = {idx + s for s in S}
            new_sums.add(2 * idx)           # self‑sum
            if sums.isdisjoint(new_sums):
                S.append(idx)
                sums.update(new_sums)
                if len(S) >= target_m:
                    break
        if len(S) >= target_m:
            vec = np.zeros(n)
            vec[S] = 1.0 / len(S)
            cand.append(vec)
        attempts += 1

    return cand


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climbing ----------------------------------------

def _local_random_search(a_best: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """Gaussian jitter on the log‑probabilities, keep improvements."""
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.025
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        val = _objective(a, n, fft_len)
        if val < best_val:
            best_val = val
            a_best = a
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")
    return a_best, best_val


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (non‑negative height vector) that minimises
    ``evaluate_sequence``.  Returns the best vector found as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Problem size – 4096 works well for the time budget
    n = 4096
    fft_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previously discovered construction, if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        s = ws.sum()
        if s >= 0.01:
            ws /= s
        else:
            ws = None
        warm_start = ws
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich pool of cheap candidates
    pool = _generate_candidates(n, warm_start)
    print(f"[Stage 1] generated {len(pool)} raw candidates")

    cand_vals = []
    for cand in pool:
        if time.time() >= deadline:
            break
        # safe‑guard against pathological vectors
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 24
    top_candidates = [c for _, c in cand_vals[:top_k]]
    if cand_vals:
        print(f"[Stage 1] best raw value = {cand_vals[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start {idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            start_vec, n, fft_len, corr_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # Stage 3 – final cheap stochastic refinement (extra 2 s budget)
    if best_seq is not None and time.time() < deadline - 2.0:
        print("--- Final local refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq, n, fft_len, deadline, best_val
        )

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return the uniform vector
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator's allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check (optional – can be removed to save a few ms)
    try:
        # evaluate_sequence is assumed to be pre‑imported in the environment
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"*** Search finished: best objective ≈ {final_score:.6f} (n={len(best_seq)})")
    except Exception:            # pragma: no cover
        # If evaluate_sequence is unavailable, just report our internal estimate
        print(f"*** Search finished (internal estimate) ≈ {best_val:.6f}")

    return best_seq.tolist()
