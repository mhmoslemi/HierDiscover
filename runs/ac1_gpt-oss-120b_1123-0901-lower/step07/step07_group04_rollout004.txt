def _smooth_weights(conv, T):
    # conv: np.ndarray of length 2n-1
    # compute stable softmax on conv * T
    maxc = conv.max()
    exp_vals = np.exp(T * (conv - maxc))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    return w

        a = np.random.dirichlet(np.ones(n))

    # but we'll use Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    T = 5.0
    T_max = 2000.0
    T_inc = 1.001 # per iteration multiply
    t = 0
    last_improve = time.time()
    while time.time()<deadline and t<max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Smooth iter={t:07d}] new best {val:.6f}")
        # smooth weights
        w = _softmax_weights(conv, T)   # returns array length 2n-1
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)  # shape n
        
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**(t+1))

        # exponentiated gradient update
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

            last_improve = time.time()

"""
Hybrid smooth‑max exponentiated‑gradient optimizer.

The objective to minimise is

    F(a) = 2·n·max(a∗a) / (sum(a)²),

where a is a non‑negative step‑function (vector).  Scaling a does not change
F, therefore we work with a normalised to have sum = 1.

The core of the search is a mirror‑descent (exponentiated‑gradient) loop
that minimises a *smooth* approximation of the max‑convolution.
We replace the hard active‑set (binary) weights used in the reference
implementation by a softmax (temperature T) over all convolution entries:

    w_k(T) = exp(T·c_k) / Σ_j exp(T·c_j)   with  c = a∗a.

When T is small the weights are spread, giving a useful gradient that can
move mass away from the peak; T is increased gradually so that the method
converges to the true max‑objective.  An Adam‑style momentum term is used
to adapt the step‑size, and occasional random jitter prevents stagnation.
The algorithm runs a multi‑start search: several diverse parametric seeds
are generated (uniform, Dirichlet, edge‑exponential, power‑law, Gaussian,
sinusoid).  The best few are refined with the smooth optimiser; if time
remains the best solution is interpolated to a finer grid (n=8192) and
refined again.

All vectors are kept non‑negative and normalised after each update, and
the final result is clipped to the range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)],   i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _softmax_weights(conv: np.ndarray, T: float) -> np.ndarray:
    """
    Compute softmax weights w_k ∝ exp(T * conv_k) in a numerically stable way.
    """
    # shift by max to avoid overflow
    shift = conv.max()
    scaled = T * (conv - shift)
    # For very large T the exponent may underflow to 0; clip to safe range
    # (numpy already handles extreme negatives by returning 0).
    exp_vals = np.exp(scaled)
    total = exp_vals.sum()
    if total == 0.0:
        # fallback to hard max (single active index)
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exp_vals / total


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective for a (assumed normalised)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of candidate vectors."""
    candidates = []

    # Warm start & shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet draws
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential families
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


# ----------------------------------------------------------------------
# Optimisation core (smooth‑max exponentiated‑gradient)
# ----------------------------------------------------------------------


def _smooth_subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Mirror‑descent on a smoothed max‑convolution.
    Returns the best sequence/value observed before the deadline.
    """
    # initialise
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    # hyper‑parameters
    eta0 = 0.5            # initial step size for exponentiated update
    decay = 0.99995       # multiplicative decay per iteration
    min_eta = 0.001

    T = 5.0               # start temperature for softmax
    T_inc = 1.001         # multiplicative increase each iteration
    T_max = 2000.0

    last_improve = time.time()
    it = 0
    max_iter = 10_000_000

    while time.time() < deadline and it < max_iter:
        # current objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Smooth iter={it:07d}] new best {val:.6f}")

        # soft‑max weights over convolution entries
        w = _softmax_weights(conv, T)

        # gradient of the smooth max (2 * cross‑correlation)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = max(min_eta, eta0 * (decay ** it))

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # anneal temperature
        T = min(T_max, T * T_inc)

        # occasional random jitter if no progress for a few seconds
        if time.time() - last_improve > 5.0:
            perturb = (np.random.rand(n) - 0.5) * 0.02  # log‑space jitter
            a = a * np.exp(perturb)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑evaluation‑score step function.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # primary grid size (balance quality / speed)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start from a previously stored global (if any)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a broad pool of seeds
    candidates = _generate_parametric_candidates(n, warm_start)

    # add a few extra Dirichlet draws for diversity
    for conc in (0.005, 0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(12):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # ------------------------------------------------------------------
    # Fast evaluation of all seeds (FFT based). Keep the top few.
    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fallback: uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    cand_vals.sort(key=lambda x: x[0])
    top_k = 6
    top_candidates = [c for _, c in cand_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of selected starts
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:

        init_val = _objective(a0, n, conv_fft_len)
        print(

        best_seq, best_val = _smooth_subgradient_opt(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # Stage 3 – if time permits, upscale to a finer grid and refine briefly
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto larger grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _smooth_subgradient_opt(

            n_big,
            conv_fft_len_big,

            deadline,
            best_seq,
