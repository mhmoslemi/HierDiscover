from scipy.optimize import linprog

def _lp_optimize(a, M):
    n = len(a)
    # Build matrix C of shape (2*n-1, n)
    rows = []
    cols = []
    data = []
    for i in range(n):
        col_val = a[i]
        start = i
        end = i+n  # exclusive end
        for k in range(start, end):
            rows.append(k)
            cols.append(i)
            data.append(col_val)
    # Build sparse matrix? but linprog expects dense np.ndarray for A_ub in SciPy v1.8?
    # We can convert to dense using np.zeros.
    # For small n (<=512), fine.
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        C[i:i+n, i] = a[i]
    res = linprog(c=-np.ones(n), A_ub=C, b_ub=np.full(2*n - 1, M), bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: keep current a
        return a
    b = res.x
    # Ensure nonnegative
    b = np.maximum(b, 0)
    # Normalize
    s = b.sum()
    if s <= 0:
        return a
    b = b / s
    return b
```

def _lp_refine(a0, max_iters, deadline):
    a = a0.copy()
    t = 0.3
    best_a = a.copy()
    best_val = float('inf')
    for i in range(max_iters):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        val = 2 * len(a) * M
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # if not improved, maybe decrease t
        b = _lp_optimize(a, M)
        a = (1 - t) * a + t * b
        # ensure sum=1
        a = a / a.sum()
        t = t * 0.9  # reduce step size
    return best_a, best_val

"""
Hybrid search for a non‑negative step function that minimises
    F(a) = 2·n·max(a∗a) / (sum a)^2 .
The method consists of three phases

1.  Very small problem (n≈256‑512).  Starting from a handful of
    analytically motivated shapes we run a few iterations of a
    linear‑program that maximises the total mass of a new vector
    under the same convolution bound.  This simple LP (solved with
    scipy.linprog) tends to flatten the convolution quickly.

2.  Upscaling – the best small‑scale solution is interpolated to the
    target size (n=4096).  The interpolated vector is already a good
    warm‑start for the more expensive continuous optimisation.

3.  Continuous refinement – a combination of
        * Adam on a smooth‑max (log‑sum‑exp) surrogate,
        * exact‑max sub‑gradient descent (exponentiated‑gradient),
        * a fast simulated‑annealing (SA) local search.
    These steps are identical to the earlier state‑of‑the‑art code
    but start from a much better point, so they converge to a lower
    objective within the same time budget.

The whole routine respects the 1000 s budget and always returns a
valid non‑negative list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                # modest learning rate
    lam = 0.3                # start temperature for the smooth‑max
    lam_factor = 1.015       # slow anneal

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that works directly on the convolution (O(n) per move)."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling from 0.1 down to 1e‑6
        T = max(1e-6, 0.1 * (1.0 - fraction))
        # move magnitude shrinks as we cool
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        # pick source index proportionally to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        # apply the move on copies
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution using the *old* a (the one before the move)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val

#   Small‑scale linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_optimize(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max   sum(b)
    s.t.    a * b  ≤ M   (convolution bound)
            b ≥ 0
    Returns the optimal b (normalised to sum 1).
    """
    n = a.shape[0]
    # Build convolution matrix C where C[k,i] = a[i] if k-i in [0,n-1]
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        C[i:i + n, i] = a[i]

    res = linprog(
        c=-np.ones(n),
        A_ub=C,
        b_ub=np.full(2 * n - 1, M),
        bounds=[(0.0, None)] * n,
        method="highs",
    )
    if not res.success:
        # fallback – keep the current vector
        return a
    b = np.maximum(res.x, 0.0)
    s = b.sum()
    if s < 1e-12:
        return a
    return b / s


def _lp_refine(a0: np.ndarray,
               max_iters: int,
               deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve a by solving the LP described above
    and mixing a fraction of the solution back.
    """
    a = a0.copy()
    t = 0.3
    best_a = a.copy()
    best_val = float("inf")
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)          # tiny size – direct convolution is fine
        M = conv.max()
        val = 2.0 * n * M
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP it={it:03d}] new best {val:.6f}")

        b = _lp_optimize(a, M)
        a = (1.0 - t) * a + t * b
        a /= a.sum()
        t *= 0.9                # slowly shrink the step size

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function that minimises
        2·n·max(a∗a) / (sum a)^2 .
    The routine runs a tiny LP‑based pre‑optimiser on a small grid,
    upsamples the result and finishes with Adam + sub‑gradient + SA.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Small‑scale LP pre‑optimisation (n≈256‑512)
    # ------------------------------------------------------------------
    n_small = 256
    # build a diverse pool of starting vectors (all already normalised)
    small_cands = []

    # optional warm‑start from previous best
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] >= n_small:
            small_cands.append(w[:n_small] / w[:n_small].sum())
        else:
            pad = np.zeros(n_small)
            pad[: w.shape[0]] = w
            small_cands.append(pad / pad.sum())

    # uniform
    small_cands.append(np.full(n_small, 1.0 / n_small))

    # Dirichlet (moderate concentration)
    small_cands.append(np.random.dirichlet(np.ones(n_small)))

    # arcsine‑type (Beta(½,½))
    arcs = 1.0 / np.sqrt(np.arange(1, n_small + 1) *
                         np.arange(n_small, 0, -1))
    arcs /= arcs.sum()
    small_cands.append(arcs)

    # power‑law edge‑concentrated (negative exponent)
    idx = np.arange(1, n_small + 1, dtype=np.float64)
    prod = idx * (n_small - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.75, -0.5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            small_cands.append(shape)

    # simple exponential decay
    exp_shape = np.exp(-np.arange(n_small) / (n_small / 4.0))
    exp_shape /= exp_shape.sum()
    small_cands.append(exp_shape)

    best_small = None
    best_small_val = float("inf")

    # run a handful of LP refinements – each gets at most ~30 s
    for cid, cand in enumerate(small_cands):
        if time.time() >= deadline - 5.0:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n_small))
        a0 /= a0.sum()

        local_deadline = min(deadline - 1.0, time.time() + 30.0)
        a_opt, val_opt = _lp_refine(a0, max_iters=12, deadline=local_deadline)

        if val_opt < best_small_val:
            best_small_val = val_opt
            best_small = a_opt.copy()
            print(f"[cand {cid}] LP improved best to {val_opt:.6f}")

    # If the LP stage never produced a candidate (unlikely), fall back to uniform
    if best_small is None:
        best_small = np.full(n_small, 1.0 / n_small)

    # ------------------------------------------------------------------
    # 2️⃣  Upscale to the target size (n = 4096) and prepare FFT lengths
    # ------------------------------------------------------------------
    n_big = 4096
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, best_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len = _next_pow2(2 * n_big - 1)
    corr_fft_len = _next_pow2(3 * n_big - 2)

    # ------------------------------------------------------------------
    # 3️⃣  Continuous refinement on the full grid
    # ------------------------------------------------------------------
    # Adam optimisation (smooth‑max surrogate)
    a_adam, _ = _adam_optimize(a_big, n_big, conv_fft_len,
                               corr_fft_len, deadline,
                               max_iters=180_000)

    # Sub‑gradient descent (exact max)
    a_sub, _ = _subgradient_optimize(a_adam, n_big,
                                     conv_fft_len, corr_fft_len,
                                     deadline, max_iters=120_000)

    # Simulated‑annealing local search (if time permits)
    if time.time() < deadline - 5.0:
        a_sa, _ = _simulated_annealing(a_sub, n_big,
                                       conv_fft_len,
                                       deadline,
                                       cur_best_val=float("inf"))
        final_a = a_sa
    else:
        final_a = a_sub

    # ------------------------------------------------------------------
    # Final safety – clip to the limits that evaluate_sequence applies
    # ------------------------------------------------------------------
    final_a = np.clip(final_a, 0.0, 1000.0)

    # Report the final objective using the judge's routine (if present)
    try:
        final_val = evaluate_sequence(final_a.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(final_a)})")
    except Exception:
        # fallback – compute the same quantity locally
        conv = np.convolve(final_a, final_a)
        final_val = 2.0 * n_big * conv.max() / (final_a.sum() ** 2)
        print(f"Final internal value {final_val:.6f} (length {len(final_a)})")

    return final_a.tolist()
