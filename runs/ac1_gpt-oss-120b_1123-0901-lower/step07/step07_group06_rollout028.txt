def build_toeplitz(f):
    n = len(f)
    data = []
    offsets = []
    for d in range(n):
        # diagonal offset d (i=j+d)
        # each entry on this diagonal is f[d]
        data.append(np.full(n - d, f[d], dtype=np.float64))
        offsets.append(d)
    A = scipy.sparse.diags(data, offsets, shape=(2*n-1, n), format='csr')
    return A

"""
Propose a step function with a small evaluation score.

Algorithm
---------
1.  Build a modest (n≈1024) one‑sided exponential family and pick the
    best start (lowest surrogate objective).

2.  Iteratively improve this start by solving a linear program:
        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for every lag k,
                    g_i ≥ 0,
    where M = max_k (f*f)_k and * denotes linear convolution.
    The LP is solved with scipy.optimize.linprog on a sparse Toeplitz
    matrix that encodes the convolution constraints.  After each LP,
    g is scaled to keep the ℓ₁‑norm equal to √(2·n) and mixed into f
    (convex combination).  This reduces the maximal autocorrelation.

3.  Upsample the refined small vector to n=4096 by linear interpolation
    and renormalise to the same ℓ₁‑norm.

4.  Apply a light‑weight gradient‑based refinement (Adam on a smooth‑max
    surrogate → exponentiated‑gradient sub‑gradient → projected gradient)
    to polish the large vector.

5.  Return the best sequence found before the time budget expires.
"""

import time
import numpy as np
import scipy.sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns the projected vector (non‑negative, sum=1).
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the score (ignores clipping etc.)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix that implements
    (f * g)_k = Σ_i f_i·g_{k‑i}.
    """
    n = f.shape[0]
    data = []
    offsets = []
    for d in range(n):
        # diagonal offset d (row = col + d) – constant value f[d]
        data.append(np.full(n - d, f[d], dtype=np.float64))
        offsets.append(d)
    A = scipy.sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_refine(f: np.ndarray,
               target_sum: float,
               conv_fft_len: int,
               deadline: float,
               max_outer: int = 12,
               mix: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the linear program that maximises Σ g_i under the
    current convolution bound and mix the result back into f.
    Returns the improved vector and its objective value.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_val = _objective(best_f, n, conv_fft_len)

    for it in range(max_outer):
        if time.time() >= deadline:
            break

        conv_f = _full_convolution(best_f, conv_fft_len)
        M = conv_f.max()

        A = _build_conv_matrix(best_f)

        c = -np.ones(n)                      # maximise sum → minimise -sum
        b = np.full(A.shape[0], M)           # all convolution constraints ≤ M
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=(0, None),
                      method="highs",
                      options={"presolve": True})

        if not res.success:
            print(f"[LP] iteration {it} – linprog failed, terminating LP phase.")
            break

        g = res.x
        sum_g = g.sum()

        # If the LP cannot increase the ℓ₁‑norm, we have converged.
        if sum_g <= target_sum * (1 + 1e-12):
            print(f"[LP] iteration {it} – no ℓ₁‑norm increase (sum={sum_g:.6f}).")
            break

        # Rescale g to keep the ℓ₁‑norm equal to target_sum.
        scale = target_sum / sum_g
        g_scaled = g * scale

        # Convex combination (preserves the ℓ₁‑norm).
        new_f = (1.0 - mix) * best_f + mix * g_scaled

        # Evaluate.
        val = _objective(new_f, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_f = new_f
            print(f"[LP] iter={it:02d} new best {val:.6f}")
        else:
            # Reduce the mixing factor if we are not improving.
            mix = max(mix * 0.7, 0.05)

    return best_f, best_val

def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     best_seq: np.ndarray | None,
                     best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp surrogate of max(f*f)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.03
    lam_cap = 1e8

    last_improve = time.time()
    it = 0
    max_iter = 250_000

    while time.time() < deadline and it < max_iter:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, lam_cap)

        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            best_seq: np.ndarray | None,
                            best_val: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient steps on the true max‑convolution."""
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    last_improve = time.time()
    max_iter = 120_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 5.0:
            break

        eta *= 0.9999

    return best_seq, best_val


def _projected_gradient_refine(a0: np.ndarray,
                               n: int,
                               conv_fft_len: int,
                               corr_fft_len: int,
                               deadline: float,
                               best_seq: np.ndarray | None,
                               best_val: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact objective."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.2
    last_improve = time.time()
    max_iter = 100_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            break

        lr *= 0.99998

    return best_seq, best_val


def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """Full optimisation pipeline for a single start."""
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    best_seq, best_val = _projected_gradient_refine(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – returns a list of non‑negative heights.
    The search uses a small LP‑based refinement followed by a
    lightweight gradient‑based polishing stage.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) small grid for the LP phase
    n_small = 1024
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)
    target_sum_small = np.sqrt(2.0 * n_small)

    # ------------------------------------------------------------------
    # 2) build a few one‑sided exponential candidates and pick the best
    i = np.arange(n_small, dtype=np.float64)

    candidates = []

    # uniform (baseline)
    candidates.append(np.full(n_small, 1.0 / n_small))

    # one‑sided exponential with a wide range of decay rates
    for alpha in np.logspace(-2, 2.0, num=30):
        shape = np.exp(-alpha * i)
        shape = np.maximum(shape, 0.0)
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        shape *= target_sum_small
        candidates.append(shape)

    # one‑sided power‑law (i+1)^(-p)
    for p in np.linspace(0.2, 4.0, num=25):
        shape = (i + 1.0) ** (-p)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        shape *= target_sum_small
        candidates.append(shape)

    # warm start from a previous construction, if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n_small:
            # interpolate / crop to the required length
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n_small, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        warm_sum = warm.sum()
        if warm_sum >= 0.01:
            warm = warm / warm_sum * target_sum_small
            candidates.append(warm)

    # quick evaluation of the small candidates
    best_small = None
    best_val_small = float("inf")
    for cand in candidates:
        if time.time() >= deadline:
            break
        val = _objective(cand, n_small, conv_fft_len_small)
        if val < best_val_small:
            best_val_small = val
            best_small = cand.copy()
            print(f"[Init] new best small objective {val:.6f}")

    # ------------------------------------------------------------------
    # 3) LP based refinement on the small grid
    if best_small is not None:
        best_small, best_val_small = _lp_refine(
            best_small,
            target_sum_small,
            conv_fft_len_small,
            deadline,
            max_outer=15,
            mix=0.5,
        )
    else:
        # fallback – uniform
        best_small = np.full(n_small, target_sum_small / n_small)

    # ------------------------------------------------------------------
    # 4) upscale to a larger grid (n=4096)
    n_big = 4096
    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)
    target_sum_big = np.sqrt(2.0 * n_big)

    x_small = np.linspace(0, 1, n_small, endpoint=False)
    x_big = np.linspace(0, 1, n_big, endpoint=False)
    best_big = np.interp(x_big, x_small, best_small)
    best_big = np.maximum(best_big, 0.0)
    best_big = best_big / best_big.sum() * target_sum_big

    # ------------------------------------------------------------------
    # 5) final gradient‑based polishing on the large grid
    best_seq, best_val = _deep_optimize(
        best_big,
        n_big,
        conv_fft_len_big,
        corr_fft_len_big,
        deadline,
        None,
        float("inf"),
    )

    # ------------------------------------------------------------------
    # 6) safety fall‑back
    if best_seq is None:
        best_seq = np.full(n_big, target_sum_big / n_big)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final reporting (using the provided evaluator for sanity‑check)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\n*** Finished search: best objective ≈ {final_score:.6f} (n={len(best_seq)})")
    except Exception:
        print("\n*** Finished search (unable to compute final score).")

    return best_seq.tolist()
