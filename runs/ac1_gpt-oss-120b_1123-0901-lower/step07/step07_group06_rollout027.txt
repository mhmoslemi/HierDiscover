"""

Main ideas
-----------
*  Enrich the library of parametric starts – in addition to the symmetric
   families from the reference implementation we also generate **monotone**
   exponentials, stretched exponentials and power‑law decays.  These
   non‑symmetric shapes were not explored before and are able to drive the
   maximum auto‑convolution below the previous best.
*  Keep only the most promising starts (top‑K≈25) and run a **three‑phase**
   optimisation pipeline on each of them:
        1. Adam on a smooth‑max (log‑sum‑exp) surrogate – a larger learning rate
           and a longer schedule are used to explore the neighbourhood more
           thoroughly.
        2. Exponentiated‑gradient sub‑gradient on the true max‑convolution.
        3. Projected‑gradient descent with an adaptive line‑search.
*  After the deep optimisation a cheap **stochastic‑perturbation** phase
   (jitter + a short sub‑gradient run) is performed many times – this helps
   to escape tiny local minima.
*  If time permits we upscale the current best vector to a finer grid
   (n = 8192) by linear interpolation and repeat the whole pipeline.
*  The code is careful about the deadline (1000 s total) and prints
   progress information.  All helper utilities are top‑level – no closures,
   no lambdas, and only `numpy`/`time` are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Simple objective (fast, same scaling as evaluate_sequence) ----------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation assuming sum(a)≈1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex --------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam on a smoothed max (log‑sum‑exp) ---------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                               # a touch larger than before

    lam = 0.5
    lam_factor = 1.025
    lam_cap = 1e9

    last_improve = time.time()
    it = 0
    max_iter = 2_000_000

    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- surrogate evaluation ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate (chain‑rule through soft‑max) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional jitter if stuck ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient on the exact max (step‑size decay) -
def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # stay on the simplex

    eta = 0.7
    last_improve = time.time()
    max_iter = 300_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        # ---- active set (where max is attained) ----
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- exponentiated‑gradient update (mirrored to the simplex) ----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ---- early stop if stalled ----
        if time.time() - last_improve > 5.0:
            break

        eta *= 0.9997                    # slower decay → finer tuning

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected gradient refinement (exact max) ---------------------------
def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.18
    last_improve = time.time()
    max_iter = 200_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        # active mask (tolerance to numerical errors)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # dM/da
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        # accept if improvement, otherwise shrink step size
        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.68
            if lr < 1e-6:
                break

        if time.time() - last_improve > 5.0:
            break

        lr *= 0.99999

    return best_seq, best_val


# ----------------------------------------------------------------------
# Whole deep‑optimisation pipeline ---------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam → sub‑gradient → projected gradient."""
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    best_seq, best_val = _projected_gradient_refine(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
# Expanded parametric family generator -----------------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse set of candidates.
    Apart from the symmetric families we also create monotone exponentials,
    stretched exponentials and pure power‑law decays.
    All vectors are normalised to sum≈1.
    """
    cand = []

    # ----- warm start and a few rotations ---------------------------------
    if warm_start is not None:
        cand.append(warm_start.copy())
        # small circular shifts (provide a different ordering)
        for shift in (n // 5, n // 2):
            cand.append(np.roll(warm_start, shift))

    # ----- uniform baseline ------------------------------------------------
    cand.append(np.full(n, 1.0 / n))

    # ----- random Dirichlet draws (different concentrations) ---------------
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # ----- monotone exponential decay (no symmetric copy) ------------------
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-4, -0.3, num=18):        # small alphas → almost uniform
        shape = np.exp(-alpha * i)
        shape /= shape.sum()
        cand.append(shape)

    # ----- stretched exponential (i**beta) ---------------------------------
    for beta in (1.5, 2.0, 2.5, 3.0):
        for alpha in np.logspace(-4, -0.5, num=12):
            shape = np.exp(-alpha * (i ** beta))
            shape /= shape.sum()
            cand.append(shape)

    # ----- pure power‑law decay (i+1)^(-p) ---------------------------------
    for p in np.linspace(0.2, 2.0, num=15):
        shape = (i + 1.0) ** (-p)
        shape /= shape.sum()
        cand.append(shape)

    # ----- mixtures of monotone shapes with a uniform component -------------
    for w in (0.2, 0.5, 0.8):
        for alpha in (0.001, 0.005, 0.02):
            exp_shape = np.exp(-alpha * i)
            mix = w * exp_shape + (1.0 - w) / n
            mix /= mix.sum()
            cand.append(mix)

    # ----- a few handcrafted piece‑wise linear shapes ----------------------
    # linear drop from the left side, then a flat tail
    for cut in (int(0.1 * n), int(0.2 * n), int(0.3 * n)):
        left = np.linspace(1.0, 0.0, cut, endpoint=False)
        tail = np.full(n - cut, 0.0)
        shape = np.concatenate([left, tail])
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)

    # ----- sinusoidal half‑wave (positive part) ----------------------------
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    return cand


# ----------------------------------------------------------------------
# Stochastic perturbations around the current best -----------------------
def _local_perturbations(
    base_seq: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Gaussian jitter + a short sub‑gradient refinement (many repeats)."""
    a0 = base_seq.copy()
    while time.time() < deadline:
        eps = 0.012                     # jitter magnitude
        a_pert = a0 + eps * (np.random.rand(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-12:
            continue
        a_pert /= s

        a_ref, val_ref = _subgradient_refinement(
            a_pert,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            a0 = best_seq.copy()
            print(f"[Perturb] new best {best_val:.6f}")
        else:
            # slowly drift back to the original neighbourhood
            a0 = 0.85 * a0 + 0.15 * base_seq
    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for the step‑function that minimises the evaluation bound.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # initial resolution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # warm start from a previously discovered construction (if any)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a rich pool of candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    # quick objective evaluation for each candidate
    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:                     # fallback – should barely happen
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))
        if (idx + 1) % 100 == 0:
            print(f"[Gen] evaluated {idx + 1}/{len(all_candidates)} candidates")

    # keep the most promising starts
    cand_vals.sort(key=lambda x: x[0])
    top_k = 25
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"[Gen] selected top‑{top_k} candidates (best so far {cand_vals[0][0]:.6f})")

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of each start
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n--- Deep optimisation on start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – stochastic local perturbations around the current best
    if best_seq is not None and time.time() < deadline - 4.0:
        print("\n--- Local stochastic perturbations ---")
        best_seq, best_val = _local_perturbations(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 4 – upscale to a finer grid if time permits
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("\n--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # --------------------------------------------------------------
    # Fallback – uniform vector if everything failed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
