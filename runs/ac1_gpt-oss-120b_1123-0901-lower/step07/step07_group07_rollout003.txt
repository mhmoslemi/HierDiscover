"""
Propose a non‑negative step function with a very low evaluate_sequence value.

Algorithm overview
------------------
1. **Warm‑start & parametric seeding** – if a previous best sequence
   (`height_sequence_1`) exists it is used together with a collection of
   synthetic families (uniform, Dirichlet draws, edge‑exponential,
   edge‑power‑law, symmetric beta, raised‑cosine, Gaussian‑edge).  The
   parameters of these families are sampled continuously (log‑uniform for
   rates, uniform for mixture weights).  All candidates are normalised to
   sum‑one and evaluated (fast FFT‑based convolution).  The top few are
   retained.

2. **Two‑stage convex‑like refinement** – each retained start is refined
   by (a) Adam optimisation on a smoothed log‑sum‑exp surrogate of the
   max‑convolution and (b) an exact sub‑gradient exponentiated‑gradient
   step on the true max.  The best sequence over all starts is kept.

3. **Simulated‑annealing fine‑tuning** – starting from the best vector,
   random pairwise mass transfers are performed.  Moves that improve the
   objective are always accepted; uphill moves are accepted with a
   temperature‑dependent probability.  The temperature linearly decays.
   This phase can break symmetry and escape local minima.

4. **Upscaling (optional)** – if time permits the current best is
   interpolated to a finer grid (n = 8192) and the whole refinement
   (steps 2‑3) is repeated.

All heavy work uses real FFTs (`np.fft.rfft`) for speed.  The routine is
careful about the overall 1000 s time budget and prints intermediate
objective values for monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) / (sum(a)**2).  Assumes sum(a)>0."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best sequence and value seen during this run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                               # relatively aggressive
    lam = 0.4                               # initial temperature for smooth‑max
    lam_factor = 1.02
    lam_cap = 1e6

    it = 0
    last_improve = time.time()
    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & surrogate objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- jitter if stuck ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:05d} new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        it += 1

    return best_seq, best_val

    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of candidate sequences."""
    candidates: list[np.ndarray] = []
    i = np.arange(n, dtype=np.float64)

    # warm start and a few cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for shift_frac in (0.125, 0.25, 0.5, 0.75):
            shift = int(shift_frac * n)
            candidates.append(np.roll(ws, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws with random concentration
    for _ in range(30):
        conc = 10 ** np.random.uniform(-2, 1)       # 0.01 … 10
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # edge exponential (with optional uniform mix)
    for _ in range(25):
        alpha = 10 ** np.random.uniform(-2, 1)     # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        w = np.random.rand()
        cand = w * edge + (1.0 - w) / n
        candidates.append(cand)

    # edge power‑law (with optional uniform mix)
    for _ in range(25):
        p = np.random.uniform(0.1, 6.0)
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        w = np.random.rand()
        cand = w * edge + (1.0 - w) / n
        candidates.append(cand)

    # symmetric beta‑shapes
    for _ in range(20):
        alpha = np.random.uniform(0.1, 5.0)
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6, 8):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # two‑Gaussian edge shapes
    for _ in range(20):
        gamma = 10 ** np.random.uniform(-2, 1)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ensure non‑negativity and normalisation
    cleaned = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 0.01:
            c = np.random.dirichlet(np.ones(n))
            s = c.sum()
        cleaned.append(c / s)
    return cleaned

def _simulated_annealing(
    a_start: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 200_000,
    init_best_val: float | None = None,
) -> tuple[np.ndarray, float]:
    """
    Pairwise mass‑transfer simulated annealing.
    Returns the best sequence encountered.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_val = _objective(a, n, conv_fft_len) if init_best_val is None else init_best_val
    cur_val = best_val

    T0 = 0.008                     # starting temperature (tuned empirically)
    it = 0
    while time.time() < deadline and it < max_iter:
        # linear temperature schedule
        T = T0 * (1.0 - it / max_iter) + 1e-8

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_transfer = min(a[i], a[j])
        if max_transfer < 1e-12:
            continue
        # delta up to ±30 % of the smaller mass
        delta = (np.random.rand() - 0.5) * 0.6 * max_transfer
        a_new = a.copy()
        a_new[i] += delta
        a_new[j] -= delta
        if a_new[i] < 0 or a_new[j] < 0:
            continue
        # renormalise (minor numerical drift)
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0:
            continue
        a_new /= s

        new_val = _objective(a_new, n, conv_fft_len)
        d = new_val - cur_val
        if d < 0 or np.random.rand() < math.exp(-d / max(T, 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a_new.copy()
                print(f"[SA] it={it:06d} new best {best_val:.6f}")

        it += 1
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point for the optimisation challenge.
    Returns a list of non‑negative floats (the best step‑function found).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # base grid size (fast enough for many iterations)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start if a previous result is available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # Phase 1 – generate and quickly evaluate a rich set of seeds
    cand_pool = _generate_parametric_candidates(n, ws)

    evaluated = []
    for idx, cand in enumerate(cand_pool):
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        # optional progress print
        # print(f"[seed {idx}] val≈{val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Phase 1: selected {top_k} promising seeds (n={n})")

    # ------------------------------------------------------------------
    # Phase 2 – deep optimisation (Adam + sub‑gradient) on the seeds
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing fine‑tuning (break symmetry, escape local minima)
    if best_seq is not None and time.time() < deadline - 5.0:
        print("--- Simulated‑annealing refinement (n={}) ---".format(n))
        best_seq, best_val = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            init_best_val=best_val,
        )

    # ------------------------------------------------------------------
    # Phase 4 – if time permits, move to a finer grid and repeat phases 2‑3
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.clip(a_big, 0.0, None)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # final SA on the larger grid
        if best_seq is not None and time.time() < deadline - 5.0:
            print("--- Final SA on large grid ---")
            best_seq, best_val = _simulated_annealing(
                best_seq,
                n_big,
                conv_fft_len_big,
                deadline,
                init_best_val=best_val,
            )
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Final random tiny hill‑climb (very cheap, only if we still have a few seconds)
    if best_seq is not None and time.time() < deadline - 2.0:
        a = best_seq.copy()
        best_local_val = best_val
        it = 0
        max_local_iter = 6000
        while time.time() < deadline and it < max_local_iter:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            max_transfer = min(a[i], a[j])
            if max_transfer < 1e-12:
                continue
            delta = (np.random.rand() - 0.5) * 0.5 * max_transfer  # ±25 % of smaller mass
            a[i] += delta
            a[j] -= delta
            if a[i] < 0 or a[j] < 0:
                a[i] -= delta
                a[j] += delta
                continue
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            val = _objective(a, n, conv_fft_len)
            if val < best_local_val:
                best_local_val = val
                best_seq = a.copy()
                best_val = val
                # keep the improvement; no print to avoid clutter
            else:
                # revert move
                a[i] -= delta
                a[j] += delta
            it += 1
        print(f"Final hill‑climb finished after {it} tries, best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback: uniform distribution if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the bounds enforced by evaluate_sequence (0 ≤ x ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
