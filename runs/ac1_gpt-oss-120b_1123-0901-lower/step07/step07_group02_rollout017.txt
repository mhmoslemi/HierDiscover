"""
Propose a non‑negative step function (vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The routine works in three stages:

1.  **Fast parametric sweep** – generate a diverse set of monotone
    and asymmetric families (exponential, power‑law, plateau+exp,
    spike patterns, Dirichlet, uniform, plus the previous best if
    available).  Each candidate is evaluated by an FFT‑based
    convolution; the best few are kept.

2.  **Continuous refinement** – the current best candidate is fed to
    an Adam optimiser that minimises a smooth‑max (log‑sum‑exp)
    surrogate of the convolution peak.  The optimisation runs until
    the global deadline is reached.  Afterwards a long
    exponentiated‑gradient (sub‑gradient) phase further drives the
    true max‑convolution down.

3.  **Cheap local hill‑climbing** – if time remains, a random‑move
    local search (mass transfer between two indices) is performed
    until the budget expires.

All steps respect the 1000 s budget, keep the vector on the simplex,
clip values to the evaluator’s safety limits and finally return the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)] .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (∑a≈1) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise in the log‑domain
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # a modest learning‑rate
    temperature = 0.5
    temp_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum≈1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(temperature * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → approach true max
        temperature = min(temperature * temp_factor, 1e6)

        # tiny random shake every few thousand iterations
        if it % 9_000 == 0:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase with a decaying step size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:07d}] best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update + renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # random perturbation after long stagnation
        if stall > 12_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move up to 40 % of the source mass
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1

    return best_a, best_val

        cand_list.append(a)

    # also add reversed versions of all monotone candidates to increase variety
    revs = [c[::-1].copy() for c in cand_list]
    cand_list.extend(revs)

    # ------------------------------------------------------------------
    # Evaluate all candidates quickly, keep the best one
    best_seq = None
    best_val = float("inf")
    for i, cand in enumerate(cand_list):
        if time.time() >= deadline:
            break
        conv = _full_convolution(cand, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(cand) == 1 by construction
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if i % 20 == 0:
            print(f"[Sweep] candidate {i:03d} → value {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Continuous refinement of the current best
    if best_seq is not None and time.time() < deadline - 5:
        # allocate roughly half of the remaining time to Adam, the rest to sub‑gradient
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.45
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_deadline, max_iters=600_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

        # sub‑gradient phase consumes the rest of the budget
        if time.time() < deadline - 2:
            sub_deadline = deadline
            a_sub, val_sub = _subgradient_opt2(
                best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=1_200_000
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 3. Final cheap local search (if any time left)
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=500_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback: ensure a valid return value
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip for the evaluator's safety limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
