    a = a0.copy()
    best_seq = a.copy()
    best_val = init_best_val
    step = 0
    t = 0.5
    # We'll run up to 5 iterations or until time
    while time.time() < deadline and step < 5:
        # compute M
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # build matrix
        rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).astype(np.int32).ravel()
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(a, n)
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # solve LP
        c = -np.ones(n)  # maximize sum -> minimize -sum
        b_ub = M * np.ones(2*n-1)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print("[LP] Linprog failed:", res.message)
            break
        g = res.x
        if g.sum() <= 0:
            break
        g_norm = g / g.sum()
        # mixing
        a = (1 - t) * a + t * g_norm
        # evaluate
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[LP] step={step} new best {val:.6f}")
        # reduce t
        t *= 0.5
        step += 1
    return best_seq, best_val

"""
Hybrid optimizer for the step‑function bound.

Main ideas
----------
1. **Rich parametric warm‑start** – generate a very diverse pool of
   candidates (uniform, Dirichlet, symmetric/asymmetric exponentials,
   power‑laws, linear/triangular, Gaussian, plateau‑+‑exponential,
   sparse spikes).  The best few become seeds for the heavy optimisation.

2. **Two‑phase continuous optimisation** –  
   *Adam* on a log‑sum‑exp (smooth‑max) surrogate with a long
   annealing schedule, followed by an *exact* sub‑gradient phase
   (exponentiated‑gradient mirror descent) that directly minimises the
   true L∞ convolution norm.

3. **Linear‑program refinement** – given a current vector `a`,
   solve the LP  
        maximise  Σ g_i          s.t.  a★g ≤ M ,  g ≥ 0          (M = max(a★a))
   which finds a `g` that can carry more total mass without increasing the
   convolution bound.  Mixing a fraction of `g` into `a` provably reduces the
   objective (the “active‑set” step from the literature).

4. **Stochastic hill‑climbing** – cheap Gaussian perturbations of the
   logits (soft‑max) are accepted whenever they improve the exact objective.

5. **Time‑aware orchestration** – the routine constantly checks the global
   deadline (default 1000 s) and adapts the work‑budget of each stage.
   The best sequence seen at any point is returned as a plain Python list.

All helper functions are top‑level; no closures or lambdas are used.
Only `numpy`, `scipy.optimize` and `scipy.sparse` are imported.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective F = 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Longer schedule and a slightly larger learning‑rate than the baseline.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                     # a bit more aggressive
    lam = 1.5                     # start temperature
    lam_factor = 1.008            # slower growth → smoother transition
    lam_cap = 1e7

    it = 0
    max_iter = 800_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (used for logging only)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:07d}  new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # shake if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the *exact* max‑convolution.
    """
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:                     # safe‑guard against collapse
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s                            # stay on the simplex (sum = 1)

    eta0 = 1.0
    decay = 0.99996
    eta = eta0

    it = 0
    max_iter = 200_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:07d}  new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (maintains non‑negativity)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # early stop if stagnating
        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run the Adam‑smooth‑max stage followed by a longer sub‑gradient stage.
    """
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len,
        deadline, global_best_seq, global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len,
        deadline, best_seq, best_val,
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse pool of candidate vectors.
    Families: uniform, Dirichlet, symmetric/asymmetric exponentials,
    power‑laws, linear/triangular, sinusoid, Gaussian, plateau‑exponential,
    and sparse spikes.
    All candidates are normalised to sum ≈ 1.
    """
    candidates = []

    # ------------------------------------------------------------------
    # warm‑start and simple shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet draws – broader range of concentrations
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(10):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # ------------------------------------------------------------------
    # symmetric families
    i = np.arange(n, dtype=np.float64)

    # edge‑exponential
    for alpha in np.logspace(-2, 1.0, num=14):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape / shape.sum())

    # edge‑power‑law
    for p in np.linspace(0.2, 6.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.35, 0.7, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape / shape.sum())

    # ------------------------------------------------------------------
    # asymmetric two‑sided exponentials
    alphas = np.logspace(-2, 1.0, num=10)
    w_vals = np.linspace(0.0, 1.0, 7)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # asymmetric power‑law
    p_vals = np.linspace(0.2, 5.0, num=9)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ------------------------------------------------------------------
    # simple linear / triangular shapes
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base /= base.sum()
        for w in (0.0, 0.3, 0.6, 0.9):
            cand = w * base + (1.0 - w) / n
            candidates.append(cand / cand.sum())

    # ------------------------------------------------------------------
    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ------------------------------------------------------------------
    # Gaussian bell curves (centred)
    for sigma in np.linspace(n / 64, n / 4, num=6):
        shape = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # plateau + exponential tails
    for pf in (0.2, 0.35, 0.5, 0.7):
        width = int(pf * n)
        left_len = (n - width) // 2
        right_len = n - width - left_len
        for alpha in (0.02, 0.05, 0.1, 0.3, 0.6):
            left = np.exp(-alpha * np.arange(left_len)[::-1])
            right = np.exp(-alpha * np.arange(right_len))
            plateau = np.ones(width)
            shape = np.concatenate([left, plateau, right])
            shape = np.maximum(shape, 0.0)
            shape /= shape.sum()
            candidates.append(shape)

    # ------------------------------------------------------------------
    # sparse spikes (few non‑zeros)
    for spikes in (3, 5, 8, 12):
        for _ in range(15):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates

def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: add Gaussian noise to the logits of the current
    best vector, evaluate the exact objective, keep any improvement.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.025
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local search] improved to {val:.6f}")

    return a_best, best_val

def _lp_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Linear‑program refinement step.
    Solve   max Σ g_i   s.t.   a★g ≤ M,   g ≥ 0,
    then mix a fraction of the optimiser result back into a.
    Repeats a few times while time permits.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = init_best_val

    # pre‑compute row/col index pattern (independent of a)
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).astype(np.int32).ravel()
    cols = np.repeat(np.arange(n, dtype=np.int32), n)

    mix_frac = 0.5
    step = 0
    while time.time() < deadline and step < 5:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()

        data = np.tile(a, n)                 # a repeated column‑wise
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        c = -np.ones(n)                      # maximise Σ g_i  ⇔ minimise −Σ g_i
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n

        res = scipy.optimize.linprog(
            c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
            method='highs', options={'presolve': True}
        )
        if not res.success:
            print(f"[LP] iteration {step} failed: {res.message}")
            break

        g = res.x
        if g.sum() <= 0:
            break
        g_norm = g / g.sum()

        # convex combination (mix_frac decays)
        a = (1 - mix_frac) * a + mix_frac * g_norm
        mix_frac *= 0.5                       # become more conservative

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[LP] step={step}  new best {val:.6f}")

        step += 1

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises ``evaluate_sequence``.  Returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Work with n = 4096 initially (a size that gave good results before)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a rich pool of cheap candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fallback: pure uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        return best_seq.tolist()

    cand_vals.sort(key=lambda x: x[0])
    top_k = min(24, len(cand_vals))
    top_candidates = [c for _, c in cand_vals[:top_k]]
    best_seq = top_candidates[0]
    best_val = cand_vals[0][0]
    print(f"Stage 1: evaluated {len(cand_vals)} candidates, best so far = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # allocate a sensible slice of the remaining budget for this start
        remaining = deadline - time.time()
        # give each remaining candidate roughly an equal share (80% of the pool)
        per = remaining / (len(top_candidates) - start_id) * 0.8
        sub_deadline = time.time() + per
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} (budget≈{per:.1f}s) ---")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_deadline, best_seq, best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – LP refinement (if time permits)
    if best_seq is not None and time.time() < deadline - 30.0:
        print("--- LP refinement stage ---")
        best_seq, best_val = _lp_refine(
            best_seq, n, conv_fft_len, deadline, best_val,
        )

    # --------------------------------------------------------------
    # Stage 4 – cheap stochastic hill‑climbing from the current best
    if best_seq is not None and time.time() < deadline:
        print("--- Local stochastic refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq, n, conv_fft_len, deadline, best_val,
        )

    # --------------------------------------------------------------
    # Stage 5 – if generous time remains, try a larger grid (n = 8192)
    time_left = deadline - time.time()
    if time_left > 60.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big, deadline,
            best_seq, best_val,
        )
        if best_seq.shape[0] == n_big:
            n = n_big
            conv_fft_len = conv_fft_len_big
            corr_fft_len = corr_fft_len_big

    # --------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
