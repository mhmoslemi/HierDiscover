"""
Hybrid search for the auto‑convolution minimisation problem.

The core ideas

* a rich pool of warm‑starts (including exponential edge‑decays,
  symmetric exponential shapes and Dirichlet draws with many
  concentration parameters);
* an additive projected‑gradient optimiser that works directly on the
  exact max‑convolution (sub‑gradient) and projects onto the simplex
  after each step – this often converges faster than the exponentiated‑
  gradient used before;
* a fast simulated‑annealing local search that updates the linear
  convolution in O(n) per elementary mass‑transfer move;
* optional up‑scaling to a longer grid (8192 points) when time permits.

The routine respects a 1000 s budget, always returns a non‑negative
vector, and prints intermediate progress to aid debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of the vector v onto the simplex
    { x | x_i >= 0, Σ x_i = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors."""
    cand = []

    # warm start & a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet (uniform) and a spiky Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))
    cand.append(np.random.dirichlet(0.2 * np.ones(n)))

    # power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.75):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # arcsine‑type (Beta(½,½) analogue)
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    cand.append(arcs)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # half‑sine (positive half‑wave)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # spike families (k equally‑spaced non‑zero entries)
    for k in (64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand


def _exponential_candidates(n: int) -> list[np.ndarray]:
    """Edge‑decaying exponential families (left‑only and symmetric)."""
    cand = []
    # left‑hand exponential
    for rate in np.geomspace(0.001, 0.2, num=12):
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
        cand.append(a)
    # symmetric double‑exponential (decays from both ends)
    for rate in np.geomspace(0.001, 0.2, num=12):
        dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-rate * dist)
        a /= a.sum()
        cand.append(a)
    return cand


def _dirichlet_family(n: int) -> list[np.ndarray]:
    """Dirichlet draws with a range of concentration parameters."""
    cand = []
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        a = np.random.dirichlet(np.full(n, alpha))
        cand.append(a)
    return cand

def _proj_grad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Projected additive sub‑gradient descent.
    Uses the exact max (active‑set) surrogate and projects onto the simplex.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= s

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 2000 == 0:
                print(f"[ProjGrad it={it:05d}] new best {val:.6f}")

        # build the exact‑max weight vector
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()  # uniform over active indices
        # gradient of (max) w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1.0)

        # additive update + projection onto simplex
        a = a - eta * grad
        a = _simplex_projection(a)

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    cur_best_val: float,
) -> tuple[np.ndarray, float]:
    """Fast SA that updates the linear convolution in O(n) per elementary move."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling schedule
        T = max(1e-6, 0.15 * (1.0 - fraction))
        move_scale = max(0.01, 0.4 * (1.0 - fraction))

        # source index – drawn proportionally to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue
        # distinct target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # convolution update (O(n))
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best so far {best_val:.6f}")

    return best_a, best_val


def _interpolate_and_refine(
    a_small: np.ndarray,
    n_big: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Interpolate to a larger grid and perform a short refinement pass."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # a brief projected‑gradient pass on the larger grid
    a_opt, val_opt = _proj_grad_optimize(
        a_big,
        n_big,
        conv_fft_len_big,
        corr_fft_len_big,
        deadline,
        max_iters=80_000,
        eta0=0.5,
    )
    return a_opt, val_opt

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        2·n·max(a∗a) / (sum a)^2 .
    The routine combines a large diverse warm‑start pool,
    projected gradient descent on the exact max‑convolution,
    and a fast simulated‑annealing local search.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # optional warm start from the previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # build a big but manageable candidate pool
    candidates = _initial_candidates(n, warm)
    candidates.extend(_exponential_candidates(n))
    candidates.extend(_dirichlet_family(n))

    # shuffle to avoid systematic ordering bias
    np.random.shuffle(candidates)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – cheap projected‑gradient optimisation of many starts
    cand_idx = 0
    while time.time() < deadline - 30 and cand_idx < len(candidates):
        a0 = np.maximum(candidates[cand_idx], 0.0)
        cand_idx += 1

        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()

        # allocate a modest budget per candidate (≈12 s)
        local_deadline = min(deadline - 5.0, time.time() + 12.0)

        a_opt, val_opt = _proj_grad_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=80_000,
            eta0=0.9,
        )

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(
                f"[cand {cand_idx-1}] proj‑grad improved best to {val_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 2 – heavy simulated‑annealing starting from the global best
    if best_seq is not None and time.time() < deadline - 10:
        a_sa, val_sa = _simulated_annealing(
            best_seq, n, conv_fft_len, deadline, best_val
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] best improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final short projected‑gradient polish (if time permits)
    if best_seq is not None and time.time() < deadline - 5:
        a_fin, val_fin = _proj_grad_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=60_000,
            eta0=0.5,
        )
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[final PG] improved best to {val_fin:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional upscale to a longer grid (8192) and brief refinement
    if time.time() + 20 < deadline:
        n_big = 8192
        a_big, val_big = _interpolate_and_refine(
            best_seq if best_seq is not None else np.random.dirichlet(np.ones(n_big)),
            n_big,
            deadline,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Upscaled] final best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # clip to the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # report the final objective (using the judge's routine if available)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
