import time
import numpy as np

# Try import SciPy
try:
    import scipy.optimize as opt
except Exception:
    opt = None

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w★a needed for gradient (w and a real)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest pool of diverse start vectors."""
    cand = []
    if warm is not None:
        cand.append(warm.copy())
        # a few rotations of warm start
        for shift in (n // 3, 2 * n // 3):
            cand.append(np.roll(warm, shift))
    # uniform Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))
    # very spiky Dirichlet
    cand.append(np.random.dirichlet(0.1 * np.ones(n)))
    # arcsine‑type (Beta(½,½) analogue)
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    cand.append(arcs)
    # power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.0, -0.5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)
    # half‑sine (positive half‑wave)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)
    return cand

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.3
    lam_factor = 1.012
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {cur_val:.6f}")
        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient of smoothed surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        lam = min(lam * lam_factor, 1e6)
        # occasional jitter near deadline to avoid stagnation
        if it % 5000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3
        it += 1
    return best_a, best_val

def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {cur_val:.6f}")
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)
        # multiplicative update
        a = a * np.exp(-eta * grad_a)
        # renormalise onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that updates the convolution in O(n) per mass‑transfer move."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val
    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)
        # linear cooling
        T = max(1e-6, 0.12 * (1.0 - fraction))
        move_scale = max(0.005, 0.25 * (1.0 - fraction))
        # choose source proportional to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # choose distinct target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * (np.random.rand() * move_scale)
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # update convolution efficiently (O(n))
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta
        new_max = conv_new.max()
        new_val = 2.0 * n * new_max
        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")
        it += 1
        if it % 4000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")
    return best_a, best_val

def build_conv_matrix_dense(f: np.ndarray) -> np.ndarray:
    """Return dense matrix A such that (A @ g) == f * g for linear convolution."""
    n = f.shape[0]
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A

def solve_lp_max_sum_dense(f: np.ndarray, M: float) -> np.ndarray:
    """Linear program: maximise sum(g) s.t. f*g <= M (element‑wise) and g>=0."""
    if opt is None:
        raise RuntimeError("SciPy not available – cannot solve LP")
    A = build_conv_matrix_dense(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)  # minimise -sum(g)
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x

def lp_improve(f: np.ndarray,
               max_iters: int = 12,
               t0: float = 0.5,
               t_decay: float = 0.7,
               conv_fft_len: int | None = None) -> tuple[np.ndarray, float]:
    """Repeated LP step + convex mixing to lower max‑convolution."""
    n = f.shape[0]
    if conv_fft_len is None:
        conv_fft_len = _next_pow2(2 * n - 1)
    best_f = f.copy()
    best_val = float('inf')
    t = t0
    for it in range(max_iters):
        conv = _full_convolution(f, conv_fft_len)
        M = conv.max()
        try:
            g0 = solve_lp_max_sum_dense(f, M)
        except Exception as e:
            print(f"[LP it={it}] solve failed: {e}")
            break
        if g0.sum() == 0:
            break
        g = g0 / g0.sum()
        # convex combination
        f = (1.0 - t) * f + t * g
        f = np.maximum(f, 0.0)
        f /= f.sum()
        cur_val = 2.0 * n * _full_convolution(f, conv_fft_len).max()
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()
            print(f"[LP it={it}] new best {cur_val:.6f}")
        t *= t_decay
    return best_f, best_val

    The routine respects the supplied time budget and always returns a
    non‑negative vector.
    """

            if val_sub < best_val:

"""
Hybrid optimisation for the step‑function problem.

The routine works in three stages:

1.  A modest set of diverse warm‑starts (the previous best, uniform,
    Dirichlet, arcsine, power‑law) is optimised on the full grid
    (n = 8192) using Adam on a smooth‑max surrogate and a short
    exact‑max sub‑gradient pass.

2.  The current best shape is down‑sampled to a small grid (n = 1024) and
    improved by a *linear programme*: maximise total mass of a new
    distribution g subject to the linear constraints f * g ≤ M (where
    M is the current peak of f * f).  A few LP iterations with a convex
    mixing step dramatically flatten the convolution peak.

3.  The refined small‑scale shape is interpolated back to the large grid,
    polished once more with Adam and finally with a fast simulated‑annealing
    local search that updates the convolution in O(n) per move.

All steps respect the supplied time budget (default 1000 s) and the
returned vector is guaranteed to be non‑negative (the judge clips it to
[0, 1000] anyway).  The function prints intermediate bounds so you can see
the progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – required only for the LP stage
try:
    import scipy.optimize as opt
except Exception:   # pragma: no cover
    opt = None

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a (needed for gradients)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        # a couple of rotations of the warm start
        for shift in (n // 3, 2 * n // 3):
            cand.append(np.roll(warm, shift))

    # Dirichlet (uniform) and a very spiky Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))
    cand.append(np.random.dirichlet(0.1 * np.ones(n)))

    # arcsine‑type (Beta(½,½) analogue)
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    cand.append(arcs)

    # power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.0, -0.5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # half‑sine (positive half‑wave)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.3
    lam_factor = 1.012

    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {cur_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # tiny jitter close to the deadline to avoid stagnation
        if it % 5000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1
    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {cur_val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that updates the convolution in O(n) per elementary move."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start + 1e-12)

        # linear cooling of temperature
        T = max(1e-6, 0.12 * (1.0 - frac))
        move_scale = max(0.005, 0.25 * (1.0 - frac))

        # choose source proportional to its current mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # distinct target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # convolution update (O(n))
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 4000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val

# Linear‑programming helpers (small grid only – dense matrix is fine)
def build_conv_matrix_dense(f: np.ndarray) -> np.ndarray:
    """Dense matrix A such that (A @ g) == f * g (linear convolution)."""
    n = f.shape[0]
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]
    return A


def solve_lp_max_sum_dense(f: np.ndarray, M: float) -> np.ndarray:
    """
    Linear programme:
        maximise  sum(g)
        s.t.      f * g ≤ M  (element‑wise)
                  g ≥ 0
    Returns the optimiser g.
    """
    if opt is None:            # pragma: no cover
        raise RuntimeError("SciPy is required for the LP stage")
    A = build_conv_matrix_dense(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)   # minimise -sum(g)
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def lp_improve(f: np.ndarray,
               max_iters: int = 12,
               t0: float = 0.5,
               t_decay: float = 0.75,
               conv_fft_len: int | None = None) -> tuple[np.ndarray, float]:
    """
    Repeated LP step with a convex mixing (parameter t) to drive the
    maximum of the self‑convolution down.
    """
    n = f.shape[0]
    if conv_fft_len is None:
        conv_fft_len = _next_pow2(2 * n - 1)

    best_f = f.copy()
    best_val = float('inf')
    t = t0
    for it in range(max_iters):
        conv = _full_convolution(f, conv_fft_len)
        M = conv.max()
        try:
            g0 = solve_lp_max_sum_dense(f, M)
        except Exception as e:
            print(f"[LP it={it}] solve failed: {e}")
            break
        if g0.sum() == 0:
            break
        g = g0 / g0.sum()
        # convex combination
        f = (1.0 - t) * f + t * g
        f = np.maximum(f, 0.0)
        f /= f.sum()

        cur_val = 2.0 * n * _full_convolution(f, conv_fft_len).max()
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()
            print(f"[LP it={it}] new best {cur_val:.6f}")

        t *= t_decay
    return best_f, best_val

        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:      # pragma: no cover
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
