"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm combines two phases:

1. **Massive parametric sweep** – a cheap evaluation of many analytically
   parametrised families (edge‑exponential, edge‑polynomial, double‑Gaussian,
   sinusoidal, Dirichlet, spike patterns).  The best vector from this sweep
   is used as a warm start for the next phase.

2. **Exact‑max sub‑gradient refinement** – a mirror‑descent (exponentiated‑
   gradient) optimiser on the true objective.  A decreasing step‑size
   ηt = η0 / sqrt(t+1) is used; the routine runs as long as time permits and
   always keeps the best vector seen.

The whole search respects the 1000 s budget; the best sequence found
before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no clamping, a is already ≥0)."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _candidate_from_array(arr: np.ndarray) -> np.ndarray:
    """Normalise a raw array into a probability vector (sum == 1)."""
    a = np.maximum(arr, 0.0)
    s = a.sum()
    if s < 1e-12:
        # fall back to a tiny uniform mass to avoid division by zero
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
def _generate_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a diverse set of start vectors.
    Families include:
        * warm start (if available) and its cyclic shifts
        * uniform
        * Dirichlet random vectors
        * edge‑exponential (dense grid of γ)
        * edge‑polynomial (dense grid of α)
        * double‑Gaussian (few σ)
        * half‑wave sinusoid
        * spike patterns (powers‑of‑two spikes)
    """
    cand = []

    # ------------------------------------------------------------------
    # warm start (global variable “height_sequence_1” if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = _candidate_from_array(w)
        cand.append(w.copy())
        # a few cyclic shifts – they preserve the convolution pattern up to index shift
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(w, shift))

    # ------------------------------------------------------------------
    # simple baselines
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))                     # uniform
    cand.append(_candidate_from_array(np.random.dirichlet(np.ones(n))))   # one Dirichlet

    # ------------------------------------------------------------------
    # edge‑exponential families (left+right, symmetric)
    # γ on a log‑grid – we need a fairly fine sweep because the optimum lies
    # somewhere in the middle of the interval.
    gamma_vals = np.logspace(np.log10(0.01), np.log10(5.0), 150)
    idx = np.arange(n, dtype=np.float64)
    for gamma in gamma_vals:
        left = np.exp(-gamma * idx)
        shape = left + left[::-1]                     # symmetric
        cand.append(_candidate_from_array(shape))

    # ------------------------------------------------------------------
    # edge‑polynomial (product) families:  shape = (i+1)*(n-i) raised to α
    prod = (np.arange(1, n + 1, dtype=np.float64) *
            (np.arange(n, 0, -1, dtype=np.float64)))
    alpha_vals = np.linspace(-5.0, -0.1, 120)
    for alpha in alpha_vals:
        # avoid overflow for very negative α – clamp very small values
        with np.errstate(divide='ignore', invalid='ignore'):
            shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            cand.append(_candidate_from_array(shape))

    # ------------------------------------------------------------------
    # double‑Gaussian (two spikes at the ends)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.05, 0.1, 0.2, 0.3, 0.4):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        cand.append(_candidate_from_array(shape))

    # ------------------------------------------------------------------
    # half‑wave sinusoid (zero at the ends)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    cand.append(_candidate_from_array(shape))

    # ------------------------------------------------------------------
    # sparse spike patterns (powers‑of‑two equally spaced spikes)
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(_candidate_from_array(a))

    return cand


# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max exponentiated‑gradient optimisation.
    The update is a ↦ a·exp(‑η·∇) followed by L1‑normalisation.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # --------------------------------------------------------------
        # exact convolution & objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # --------------------------------------------------------------
        # active set (all indices achieving the max, up to numerical tolerance)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient of the exact‑max objective (mirror‑descent form)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing learning‑rate
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise, protect against under‑/overflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (vector) with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found before the deadline.
    """
    np.random.seed(seed)
    # small safety margin so we finish cleanly
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # generate many diverse candidates
    candidates = _generate_candidates(n)

    best_seq = None
    best_val = float("inf")

    # spend roughly half the time just scanning the cheap candidates
    scan_deadline = time.time() + (deadline - time.time()) * 0.45

    for idx, cand in enumerate(candidates):
        if time.time() > scan_deadline:
            break
        # compute exact objective (very cheap compared with an optimisation step)
        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[scan {idx:04d}] new best {val:.6f}")

    # --------------------------------------------------------------
    # If we didn't find anything (should not happen) fall back on uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # Long exact‑max sub‑gradient refinement on the current best vector
    # This phase consumes the remaining time budget.
    if time.time() < deadline - 5.0:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=800_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] sub‑gradient best {val_ref:.6f}")

    # --------------------------------------------------------------
    # cheap random hill‑climbing (mass swaps) – cheap but sometimes finds
    # a final tiny improvement.
    if time.time() < deadline - 2.0:
        a = best_seq.copy()
        a_sum = a.sum()
        conv = _full_convolution(a, conv_fft_len)
        best_val = 2.0 * n * conv.max() / (a_sum ** 2)

        it = 0
        max_moves = 70_000
        while time.time() < deadline and it < max_moves:
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue

            # move a random fraction (up to 30 %) of the source mass
            delta = a[i] * (np.random.rand() * 0.3)

            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

            if val2 < best_val:
                best_val = val2
                best_seq = a2.copy()
                a = a2
                if it % 5_000 == 0:
                    print(f"[Local {it:05d}] improved to {val2:.6f}")

            it += 1

    # --------------------------------------------------------------
    # final safety – ensure the returned vector respects the evaluator’s limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
