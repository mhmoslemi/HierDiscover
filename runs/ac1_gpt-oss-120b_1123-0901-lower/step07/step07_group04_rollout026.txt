def generate_sidon_set(n, k, rng):
    positions = []
    used_diffs = set()
    attempts = 0
    while len(positions) < k:
        cand = rng.integers(0, n)
        if cand in positions:
            continue
        # Check diff uniqueness
        conflict = False
        for p in positions:
            d = abs(cand - p)
            if d in used_diffs:
                conflict = True
                break
        if conflict:
            attempts += 1
            if attempts > 1000 * k:
                # restart
                positions = []
                used_diffs = set()
                attempts = 0
            continue
        # Add
        positions.append(cand)
        for p in positions[:-1]:
            used_diffs.add(abs(cand - p))
    return sorted(positions)
```

"""
sidon‑based search
------------------

The evaluation metric is homogeneous:
 F(a) = 2·n·max(a∗a)/(∑a)².

We may therefore fix ∑a = 1.  If we can guarantee that every
convolution entry is at most 𝑐, then the objective equals 2·n·c.
The idea is to keep the largest coefficient wₗ as small as possible
while making the *pair sums* i+j unique – a B₂ (Sidon) set.  For such a
set each shift k receives at most one product a_i·a_j, so the
convolution peak is bounded by wₗ².  A classic Bose construction
produces a Sidon set of size p (prime) inside [0,p²) :

    S = { i·p + (i² mod p)  |  i = 0 … p‑1 }.

We choose p ≤ √n, set the “large’’ weight wₗ = √(target/(2·n)),
place it on the Sidon positions, and fill the remaining entries with a
tiny uniform weight wₛ so that wₛ ≤ wₗ.  Then max(a∗a) ≤ wₗ² and the
objective is ≈ target.  The construction is extremely cheap, we simply
enumerate a few primes, evaluate every candidate and keep the best.
Optionally a short exponentiated‑gradient refinement is performed if
time permits.

The function `propose_candidate` returns the best sequence found
within the allotted budget (default 1000 s).  It requires no external
files and uses only NumPy and the given `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _is_prime(x: int) -> bool:
    if x < 2:
        return False
    if x % 2 == 0:
        return x == 2
    r = int(math.isqrt(x))
    for p in range(3, r + 1, 2):
        if x % p == 0:
            return False
    return True


def _prime_list_upto(m: int):
    """Return a list of all primes ≤ m in descending order."""
    primes = [p for p in range(m, 1, -1) if _is_prime(p)]
    return primes


def _sidon_set(p: int, n: int) -> np.ndarray:
    """
    Bose‑Chowla Sidon set of size p.
    All numbers lie in [0, p²) and are distinct.
    """
    vals = np.empty(p, dtype=np.int64)
    for i in range(p):
        vals[i] = i * p + (i * i) % p
    # Guard against the extremely unlikely overflow beyond n
    mask = vals < n
    if not mask.all():
        # keep only those that fit; this may reduce the size a little
        return vals[mask]
    return vals


def _objective_fast(seq: np.ndarray, n: int) -> float:
    """Fast objective (same formula as evaluate_sequence)."""
    conv = np.convolve(seq, seq)
    max_c = conv.max()
    s = seq.sum()
    if s < 0.01:                # should never happen for our constructions
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _refine_exponentiated_gradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Very short mirror‑descent (exponentiated‑gradient) run.
    Used only when a few seconds are still available.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    # pre‑compute FFT length for cross‑correlation
    corr_len = 3 * n - 2
    corr_fft_len = 1 << (corr_len - 1).bit_length()

    eta0 = 0.5
    decay = 0.9999
    min_eta = 0.001
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < 2_000_000:
        # ----- convolution -----
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:conv_len]
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[refine] new best {val:.6f}")

        # active set (indices achieving the maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # ----- gradient of the weighted convolution -----
        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        # ----- update -----
        eta = max(min_eta, eta0 * (decay ** it))
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum < 1e-12 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a = a / a_sum

        # occasional random shake if stuck
        if time.time() - last_improve > 4.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02
            a = a * np.exp(jitter)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimizes
    evaluate_sequence.  The method builds a Sidon set (B₂ set) using a
    Bose‑Chowla construction, places a relatively large equal weight on
    those positions, fills the remaining indices with a tiny uniform
    weight, and (if time permits) runs a short exponentiated‑gradient
    refinement.  The best sequence found before the time budget expires
    is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a convenient grid size – 4096 gives a good trade‑off.
    n = 4096

    # Upper bound we aim for (slightly below the target to stay safe).
    # The construction guarantees max(conv) ≤ wL², therefore objective
    # ≈ 2·n·wL².  We start from 1.49 (target ≈ 1.503) and also try a few
    # nearby scalings.
    target_base = 1.49
    scaling_factors = (0.98, 1.0, 1.02)

    # Pre‑compute primes up to √n (descending – larger Sidon sets first).
    max_prime = int(math.isqrt(n))
    primes = _prime_list_upto(max_prime)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Enumerate constructions based on different primes and scalings.
    for factor in scaling_factors:
        target_obj = target_base * factor
        # weight on the Sidon positions (derived from the target objective)
        wL = math.sqrt(target_obj / (2.0 * n))

        for p in primes:
            if time.time() >= deadline:
                break

            # Build Sidon set; keep only indices that fit into the array.
            S = _sidon_set(p, n)
            if S.size == 0:
                continue
            K = S.size

            # Ensure the large weight does not use up the whole mass.
            sum_large = K * wL
            if sum_large >= 1.0:
                # wL is too large for this K – skip to a smaller prime.
                continue

            # Uniform small weight for the remaining entries.
            wS = (1.0 - sum_large) / (n - K)

            # The construction is safe only if the small weight never exceeds
            # the large one, otherwise a shift containing two small entries
            # could dominate the max‑convolution.
            if wS > wL:
                continue

            # Build the candidate vector.
            a = np.full(n, wS, dtype=np.float64)
            a[S] = wL

            # Quick objective evaluation (no extra clipping needed – values are
            # already non‑negative and well below the evaluator’s limit).
            val = _objective_fast(a, n)

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(
                    f"[seed] p={p:3d}, factor={factor:.3f}, wL={wL:.6f}, "
                    f"val={val:.6f}"
                )

            # Optional short refinement if we still have a few seconds.
            if time.time() < deadline - 5.0:
                refined_seq, refined_val = _refine_exponentiated_gradient(
                    a, n, deadline, best_seq, best_val
                )
                if refined_val < best_val:
                    best_val = refined_val
                    best_seq = refined_seq.copy()
                    print(
                        f"[refine] p={p:3d}, factor={factor:.3f}, refined={refined_val:.6f}"
                    )

    # ------------------------------------------------------------------
    # Fallback – uniform vector if everything else failed.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective_fast(best_seq, n)
        print("[fallback] using uniform vector")

    # Clip to evaluator’s allowed range (0 ≤ x ≤ 1000) just in case.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search – best objective ≈ {best_val:.6f} (n={n})")
    return best_seq.tolist()
