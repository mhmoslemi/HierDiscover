def full_convolution_fft(a, fft_len):
   fa = np.fft.rfft(a, fft_len)
   conv = np.fft.irfft(fa * fa, fft_len)
   return conv[:2*n-1]

def cross_corr_fft(w, a, fft_len):
   rev_a = a[::-1]
   fw = np.fft.rfft(w, fft_len)
   fr = np.fft.rfft(rev_a, fft_len)
   corr_full = np.fft.irfft(fw * fr, fft_len)
   # The full correlation length = len(w)+len(a)-1 = (2n-1)+n-1=3n-2
   start = n-1
   # Output length n
   return corr_full[start:start+n]

def _project_simplex(v):
    # from https://gist.github.com/...
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0
    else:
        rho = ind[cond].max()
        theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    a = np.maximum(a0, 0.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / sum_a
    # Adam params
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2   # initial learning rate
    beta_start = 0.5
    beta_max = 5000.0
    beta_ramp_iters = 2000
    it = 0
    last_improve = time.time()
    while time.time() < deadline:
        # compute conv
        conv = _full_convolution(a, conv_fft_len)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        if obj < best_val - 1e-12:
            best_val = obj
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[adam n={n} iter={it}] new best {obj:.6f}")
        # smoothing weight
        beta = beta_start * (beta_max / beta_start) ** min(it / beta_ramp_iters, 1.0)
        scaled = conv * beta
        # subtract max for stability
        c = scaled - np.max(scaled)
        w = np.exp(c)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum
        # gradient
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = lr0 / (1 + it * 1e-4)  # optional decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_simplex(a)
        # random jitter if stagnated
        if time.time() - last_improve > 5.0:
            noise = (np.random.rand(n) - 0.5) * 0.02
            a = a * np.exp(noise)
            a = _project_simplex(a)
            last_improve = time.time()
        it += 1
    return best_seq, best_val

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    # Reference: (Duchi et al., 2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1] + 1  # +1 because indices start at 0
    theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a diverse set of candidate vectors using several families:
    uniform, Dirichlet, edge‑exponential, edge‑power‑law, Gaussian‑edges,
    half‑wave sinusoid. All candidates are normalised to sum≈1.
    """

    # Warm‑start and simple rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet with a few concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential (plus possible uniform mixture)
    for alpha in np.logspace(-2, 1.0, num=16):  # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Edge‑power‑law (i+1)^(-p) + (n-i)^(-p)
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """Adam optimisation of the max‑convolution objective on the simplex."""
    # initialise a on the simplex
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2  # initial learning rate

    # smoothing (log‑sum‑exp) schedule
    beta_start = 0.5
    beta_max = 5000.0
    beta_ramp = 2000  # iters to reach beta_max

    it = 0
    last_improve = time.time()
    max_iter = 10_000_000

    while time.time() < deadline and it < max_iter:
        # convolution a*a
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        obj = 2.0 * n * max_c / (a.sum() ** 2)

        if obj < best_val - 1e-12:
            best_val = obj
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[adam n={n} iter={it}] new best {obj:.6f}")

        # log‑sum‑exp smoothing weight
        beta = beta_start * (beta_max / beta_start) ** min(it / beta_ramp, 1.0)
        scaled = conv * beta
        # to avoid overflow we shift by max
        shift = np.max(scaled)
        w = np.exp(scaled - shift)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # gradient: 2 * (w ★ a)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # decaying learning rate (optional)
        lr = lr0 / (1.0 + it * 1e-4)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        if np.any(np.isnan(a)):
            a = np.random.dirichlet(np.ones(n))
        else:
            a = _project_simplex(a)

        # enforce symmetry (problem is symmetric)
        a = (a + a[::-1]) / 2.0

        # if we have been idle for a few seconds, inject jitter
        if time.time() - last_improve > 5.0:
            noise = (np.random.rand(n) - 0.5) * 0.02  # log‑space jitter
            a = a * np.exp(noise)
            a = _project_simplex(a)
            a = (a + a[::-1]) / 2.0
            last_improve = time.time()

        it += 1

    return best_seq, best_val

"""
Adam‑based optimiser with log‑sum‑exp smoothing and beta‑annealing.

* Generate a large, diverse set of symmetric candidate step‑functions.
* Evaluate them quickly (FFT based) and keep the few most promising.
* Refine the best seed on the probability simplex using Adam.
  – Gradient is the cross‑correlation of the soft‑max‑weighted
    convolution (a smooth approximation of the max‑convolution).
  – After each update the vector is projected onto the simplex and
    symmetrised (the problem is symmetric, symmetrisation never hurts).
  – A small random shake is injected if no improvement occurs for a few
    seconds.
* If time permits, the best solution is linearly interpolated to a larger
  grid (n = 8192) and refined again.

The routine obeys the 1000 s time budget, prints progress and returns the
best sequence found as a plain Python list.

    # ------------------------------------------------------------------
    # Warm‑start from a possibly existing global variable
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Phase 1 – generate a large, diverse pool of seeds
    candidates = _generate_parametric_candidates(n, warm_start)

    # Add extra Dirichlet draws with a spread of concentration parameters
    for conc in (0.02, 0.07, 0.2, 0.5, 1.0, 2.0):
        for _ in range(30):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # Symmetrise all seeds (the problem is symmetric, this never hurts)
    candidates = [(c + c[::-1]) / 2.0 for c in candidates]

    # ------------------------------------------------------------------
    # Phase 2 – quick evaluation of every seed, keep the best few
    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # Fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    # Keep the top‑k promising seeds
    cand_vals.sort(key=lambda x: x[0])
    top_k = 5
    top_starts = [c for _, c in cand_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 3 – deep optimisation (Adam) of the selected starts.
    # The first seed gets the lion's share of the remaining budget;
    # later ones act as cheap fall‑backs if time permits.
    for idx, start in enumerate(top_starts):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        if idx == 0:
            alloc = remaining * 0.8
        else:
            alloc = remaining * 0.1
        sub_deadline = time.time() + alloc
        print(f"--- Adam optimisation start {idx+1}/{len(top_starts)} "
              f"(initial obj {cand_vals[idx][0]:.6f})")
        best_seq, best_val = _adam_optimize(
            start,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Phase 4 – upscale to a larger grid if we still have time.
    time_left = deadline - time.time()
    if time_left > 20.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()
        a_big = (a_big + a_big[::-1]) / 2.0

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback to a safe uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
