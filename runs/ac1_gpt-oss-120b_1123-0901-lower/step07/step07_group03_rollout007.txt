"""
Propose a low‑value step function.

The algorithm improves the previous runner by

1. **Massive diverse warm‑starts** – we draw hundreds of anti‑paired
   Dirichlet samples with random concentration, plus several low‑dimensional
   families (power‑law, exponential, triangular).  The best few are kept
   as seeds.

2. **Multi‑restart optimisation** – each seed is refined by a long Adam
   phase (smooth‑max surrogate), an aggressive exponentiated‑gradient
   sub‑gradient phase and a final simulated‑annealing hill‑climb.
   Different random seeds and hyper‑parameter settings are used for the
   restarts; the globally best result is retained.

3. **Dynamic time budgeting** – the total 1000 s budget is split adaptively
   between initial seeding, several optimisation restarts and an optional
   fine‑grained up‑sampling to a 8192‑point grid if time permits.

All heavy work is performed with FFT‑based convolutions (O(n log n))
and the code is written to run on a single process (but may exploit two
CPU cores via NumPy's internal threading).  The returned list satisfies
the evaluator’s safety checks (non‑negative, finite, sum ≥ 0.01).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

# ----------------------------------------------------------------------
# Objective ----------------------------------------------------------------

def _objective(a: np.ndarray) -> float:
    """Exact objective  2·n·max(a*a)/(sum(a)^2)  – assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

# ----------------------------------------------------------------------
# Warm‑start --------------------------------------------------------------

def _warm_start(n: int):
    """Return a simple warm start if the global variable `height_sequence_1` exists."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm = warm / warm.sum()
        return warm
    return None

# Candidate families -------------------------------------------------------

def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place the largest half on the left, smallest half on the right (anti‑pair)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]           # median in the centre
    return a


def _anti_pair_dirichlet_candidate(n: int, alpha: float) -> np.ndarray:
    """Dirichlet sample (concentration alpha) + anti‑pair ordering."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    return _anti_pair_arrangement(vals)


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """Power‑law decreasing values (i+1)^{-p} with anti‑pair ordering."""
    raw = (np.arange(n, dtype=np.float64) + 1) ** (-p)
    raw /= raw.sum()
    return _anti_pair_arrangement(raw)


def _exponential_candidate(n: int, gamma: float) -> np.ndarray:
    """Exponential decay γ and anti‑pair ordering."""
    raw = np.exp(-gamma * np.arange(n, dtype=np.float64))
    raw /= raw.sum()
    return _anti_pair_arrangement(raw)


def _triangular_candidate(n: int) -> np.ndarray:
    """Tent shape (largest at the edges, smallest in the centre)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape = shape / shape.sum()
    return shape


def _uniform_candidate(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.03,
                   lam_start: float = 0.5,
                   lam_factor: float = 1.05) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Optimises logits that are exponentiated into a probability simplex.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max -> simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for monitoring
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # centre the gradient (mirror‑gradient for softmax parametrisation)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule – approach true max gradually
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape shallow plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # Identify the active (max) offsets
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # Gradient of the exact max (sub‑gradient)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Step‑size schedule (decaying)
        eta = eta0 / np.sqrt(it + 1)

        # Exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # Renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing.
    Moves:
        * swap two entries;
        * transfer a random fraction (≤40 %) of mass from i to j.
    Acceptance follows a linear temperature schedule from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        # --- propose a move -------------------------------------------------
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                                 # transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # ≤40 % of donor
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        # --- acceptance ----------------------------------------------------
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

    best_seq, best_val = a_adam, val_adam

    # 2) Sub‑gradient phase (exact max)
    t2 = min(deadline - 5.0, time.time() + 300.0)  # up to ~5 min
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        t2,
        max_iters=500_000,
        eta0=0.6,
    )
    if val_sub < best_val:
        best_seq, best_val = a_sub, val_sub
        print(f"[Seed] sub‑grad improved to {best_val:.6f}")

    # 3) Short second Adam (polish)
    t3 = min(deadline - 5.0, time.time() + 80.0)
    a_adam2, val_adam2 = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        t3,
        max_iters=200_000,
        lr=0.02,
        lam_start=1.0,
        lam_factor=1.07,
    )
    if val_adam2 < best_val:
        best_seq, best_val = a_adam2, val_adam2
        print(f"[Seed] second Adam improved to {best_val:.6f}")

    # 4) Local stochastic search
    if time.time() < deadline - 2.0:
        t4 = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            t4,
            max_moves=250_000,
        )
        if val_loc < best_val:
            best_seq, best_val = a_loc, val_loc
            print(f"[Seed] local search improved to {best_val:.6f}")

    return best_seq, best_val

# Main entry point --------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)
    under the 1000 s time budget.

    The routine builds a large diverse pool of warm‑starts,
    runs several independent optimisation restarts, and finally
    returns the best sequence found (as a Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Core parameters
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Gather a massive pool of diverse seeds
    pool = []

    # Warm start (if present)
    warm = _warm_start(n)
    if warm is not None:
        pool.append(warm.copy())

    # Uniform baseline
    pool.append(_uniform_candidate(n))

    # Triangular baseline
    pool.append(_triangular_candidate(n))

    # ---- Random anti‑pair Dirichlet samples (many) ----
    n_dirichlet = 250                     # plenty but still cheap
    for _ in range(n_dirichlet):
        alpha = 10 ** np.random.uniform(-2.0, 0.5)   # α ∈ [0.01, 3.16]
        cand = _anti_pair_dirichlet_candidate(n, alpha)
        pool.append(cand)

    # ---- Power‑law family ----
    for p in np.linspace(0.1, 2.0, 12):
        pool.append(_powerlaw_candidate(n, p))

    # ---- Exponential family ----
    for gamma in np.linspace(0.05, 2.0, 12):
        pool.append(_exponential_candidate(n, gamma))

    # ------------------------------------------------------------------
    # 2) Quick evaluation of the pool – keep the best few
    best_pool = []
    best_val = float("inf")
    for idx, cand in enumerate(pool):
        cand = np.maximum(cand, 0.0)
        # Normalise – avoid pathological near‑zero sums
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        if idx % 50 == 0:
            print(f"[Seed‑eval {idx:04d}] val={val:.6f}")
        if val < best_val:
            best_val = val
            best_pool = [cand.copy()]
            print(f"   *** new pool leader {val:.6f}")
        elif val < best_val * 1.02:  # keep a few near‑optimal seeds
            best_pool.append(cand.copy())

    # If we have gathered many near‑optimal seeds, trim to a manageable size
    if len(best_pool) > 6:
        best_pool = sorted(best_pool, key=_objective)[:6]

    # ------------------------------------------------------------------
    # 3) Run independent optimisation restarts on selected seeds
    global_best_seq = None
    global_best_val = float("inf")
    restart_idx = 0
    while time.time() < deadline - 30.0 and restart_idx < len(best_pool):
        seed_vec = best_pool[restart_idx]
        # Allocate a slice of the remaining budget for this restart
        remaining = deadline - time.time()
        # give each restart roughly the same share, minimum 180 s
        allocated = max(180.0, remaining / (len(best_pool) - restart_idx))
        restart_deadline = time.time() + allocated
        print(f"\n[Restart {restart_idx+1}/{len(best_pool)}] allocated {allocated:.1f}s")
        seq, val = _optimise_one_seed(seed_vec, n, restart_deadline)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Global] new champion {global_best_val:.6f}")
        restart_idx += 1

    # ------------------------------------------------------------------
    # 4) Optional upscale to a finer grid (8192) if time permits
    if time.time() < deadline - 20.0 and global_best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, global_best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # Brief Adam on the larger grid
        t_big_adam = min(deadline - 5.0, time.time() + 120.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            t_big_adam,
            max_iters=250_000,
            lr=0.04,
            lam_start=0.5,
            lam_factor=1.05,
        )
        # Sub‑gradient refinement on the large grid
        t_big_sub = min(deadline, time.time() + 180.0)
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            t_big_sub,
            max_iters=350_000,
            eta0=0.6,
        )
        if val_big_sub < global_best_val:
            global_best_val = val_big_sub
            global_best_seq = a_big_sub.copy()
            print(f"[Upscale] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Safeguard – ensure a valid return
    if global_best_seq is None:
        # fallback to a simple uniform distribution
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to limits accepted by the evaluator
    global_best_seq = np.clip(global_best_seq, 0.0, 1000.0)

    print(f"\nFinal best value {global_best_val:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
