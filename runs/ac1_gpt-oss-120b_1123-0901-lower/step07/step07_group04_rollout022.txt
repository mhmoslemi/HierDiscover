def _lbfgs_optimize(a0, n, conv_fft_len, corr_fft_len, lam_schedule, deadline, best_seq, best_val):
    import scipy.optimize as opt
    # convert a0 to z = log(a0)
    # to avoid log(0), add epsilon
    eps = 1e-15
    z = np.log(a0 + eps)

    def softmax(z):
        # stable
        m = np.max(z)
        exp_z = np.exp(z - m)
        return exp_z / exp_z.sum()

    def obj_and_grad(z_vec, lam):
        a = softmax(z_vec)
        conv = _full_convolution(a, conv_fft_len)
        # compute softmax weights over conv
        # protect numerical stability
        max_c = np.max(conv)
        # compute w = exp(lam * (conv - max_c))
        w = np.exp(lam * (conv - max_c))
        sum_w = w.sum()
        w_norm = w / sum_w
        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w_norm, a, corr_fft_len)
        # gradient wrt z: a * (grad_a - <a, grad_a>)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        # softmax approximated objective: 2*n * (max_c + log(sum_w)/lam)
        # Since max_c negligible? Actually max of conv = max_c, but approx max = max_c + log(sum_w)/lam - ??? Wait:
        # Since we shifted conv by -max_c, we have:
        # log( Σ exp(lam*conv) ) = lam*max_c + log( Σ exp(lam*(conv - max_c)) ) = lam*max_c + log(sum_w)
        # So approx max = (1/lam) * log Σ exp(lam*conv) = max_c + (log(sum_w))/lam
        approx_max = max_c + np.log(sum_w) / lam
        obj = 2.0 * n * approx_max
        return obj, grad_z

    # Loop over lambda schedule
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # define wrapper for minimize that returns (obj, grad)
        def fun(z_vec):
            obj, grad = obj_and_grad(z_vec, lam)
            return obj, grad

        # Use L-BFGS-B
        res = opt.minimize(fun, z, method='L-BFGS-B', jac=True,
                           options={'maxiter': 150, 'ftol': 1e-12, 'gtol':1e-12, 'disp': False})
        z = res.x
        # evaluate exact objective
        a = softmax(z)
        val = evaluate_sequence(a.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            print(f"[lam={lam}] new best {val:.6f}")
        # if time left, continue with next lam
    return best_seq, best_val

"""
Propose a step‑function (non‑negative weight vector) with a low
evaluate_sequence() score.

Algorithm
---------

1. **Parameterised seed generation** – A large, diverse pool of
   candidates is created:
   * uniform, Dirichlet (several concentrations)
   * symmetric edge‑exponential / power‑law / double‑Gaussian families
   * **asymmetric** shapes (single‑sided exponential, mixtures with uniform)
   * ordered Dirichlet samples (largest weight left, second largest right,
     etc.) – this “alternating” placement decorrelates a vector with its
     reversal.

2. **Fast pre‑screen** – All seeds are normalised (sum = 1) and evaluated
   with a fast FFT‑based convolution.  The best few (default 8) are kept.

3. **Smooth max optimisation** – For each selected seed we minimise a
   smooth approximation of the true objective using L‑BFGS‑B.  The
   approximation is the *log‑sum‑exp* of the convolution values:

        approx_max(λ) = max(conv) + (log Σ exp(λ·(conv‑max(conv)))) / λ

   The gradient of this surrogate is obtained analytically:
   * compute the convolution `c = a * a`
   * soft‑max‑weights `w_i ∝ exp(λ·c_i)`
   * gradient w.r.t. the probability vector `a` is  
     `grad_a = 2·cross_correlation(w, a)`
   * the optimisation variables are the logits `z` with `a = softmax(z)`,
     giving `grad_z = a·(grad_a - ⟨a,grad_a⟩)`.

   A homotopy on λ (5 → 15 → 50 → 150 → 500) gently sharpens the
   surrogate, allowing L‑BFGS to escape shallow minima.

4. **Exact evaluation & improvement** – After each λ‑phase the exact
   score (`evaluate_sequence`) is computed; the best sequence seen so far
   is retained.  Finally a few re‑ordering tricks (deterministic
   alternating placement and a handful of random shuffles) are tried,
   because the objective depends on the *order* of the numbers.

The whole routine respects a 1000 s time budget, prints progress, and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective for a (already normalised)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """Generate a diverse set of seed vectors."""
    candidates = []

    # Warm‑start and simple rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet with many concentrations
    for conc in (0.005, 0.02, 0.07, 0.15, 0.4, 0.8, 1.5, 3.0):
        for _ in range(12):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential (symmetric)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Edge‑power‑law (symmetric)
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # --------------------------------------------------------------
    # Asymmetric families (new)
    # --------------------------------------------------------------
    # One‑sided exponential
    for alpha in np.logspace(-3, 0.5, num=10):
        shape = np.exp(-alpha * i)
        shape /= shape.sum()
        candidates.append(shape)
        # mixture with uniform
        for w in (0.25, 0.5, 0.75):
            candidates.append(w * shape + (1.0 - w) / n)

    # Alternating ordered Dirichlet samples
    for conc in (0.01, 0.1, 0.5, 2.0):
        for _ in range(5):
            sample = np.random.dirichlet(np.full(n, conc))
            candidates.append(_alternating_reorder(sample))

    return candidates


def _alternating_reorder(v: np.ndarray) -> np.ndarray:
    """Place values in alternating left–right order (largest‑left, 2nd‑right,…)."""
    n = v.shape[0]
    order = np.argsort(v)[::-1]               # descending indices
    out = np.empty_like(v)
    left, right = 0, n - 1
    toggle = True
    for idx in order:
        if toggle:
            out[left] = v[idx]
            left += 1
        else:
            out[right] = v[idx]
            right -= 1
        toggle = not toggle
    return out


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser
# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector that sums to 1."""
    m = np.max(z)
    exp_z = np.exp(z - m)
    return exp_z / exp_z.sum()


def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    lam_schedule: list[float],
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Optimise a probability vector using smooth‑max L‑BFGS (homotopy on λ).
    Returns the best vector/value observed.
    """
    eps = 1e-15
    z = np.log(a0 + eps)                     # start in log‑space

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        def fun(z_vec):
            """Return (objective, gradient) for the current λ."""
            a = _softmax(z_vec)

            # convolution a*a
            conv = _full_convolution(a, conv_fft_len)

            # soft‑max‑weights over the convolution values
            max_c = conv.max()
            w = np.exp(lam * (conv - max_c))   # shift to avoid overflow
            sum_w = w.sum()
            w_norm = w / sum_w

            # gradient w.r.t. a (subgradient of the smoothed max)
            grad_a = 2.0 * _cross_correlation(w_norm, a, corr_fft_len)

            # gradient w.r.t. logits z (softmax Jacobian)
            dot = np.dot(a, grad_a)
            grad_z = a * (grad_a - dot)

            # smooth approximation of max(conv)
            approx_max = max_c + np.log(sum_w) / lam
            obj = 2.0 * n * approx_max
            return obj, grad_z

        # L‑BFGS‑B – box‑free because logits are unrestricted
        res = opt.minimize(
            fun,
            z,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 150, "ftol": 1e-12, "gtol": 1e-12, "disp": False},
        )
        z = res.x

        # Evaluate exact objective (no approximation) and keep the best
        a = _softmax(z)
        val = evaluate_sequence(a.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            print(f"[λ={lam:.0f}] new best exact score {val:.6f}")

    return best_seq, best_val

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)          # 8192
    corr_fft_len = _next_pow2(3 * n - 2)          # 12288

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored sequence if present
    # ------------------------------------------------------------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] != n:
            if warm_start.shape[0] > n:
                warm_start = warm_start[:n]
            else:
                warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate many diverse seeds
    # ------------------------------------------------------------------
    candidates = _generate_parametric_candidates(n, warm_start)

    # ------------------------------------------------------------------
    # Fast pre‑screen – keep the top few according to the exact objective
    # ------------------------------------------------------------------
    scored = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                         # degenerate – replace by Dirichlet
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)   # fast exact max‑conv score
        scored.append((val, cand))

    if not scored:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    scored.sort(key=lambda x: x[0])
    top_k = 8
    top_candidates = [c for _, c in scored[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max L‑BFGS optimisation from each start
    # ------------------------------------------------------------------
    lam_schedule = [5.0, 15.0, 50.0, 150.0, 500.0]

    for idx, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Optimising start #{idx + 1}/{len(top_candidates)}")
        best_seq, best_val = _lbfgs_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            lam_schedule,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – simple post‑processing (re‑ordering, random shuffles)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # deterministic alternating re‑ordering
        alt_seq = _alternating_reorder(best_seq)
        val_alt = evaluate_sequence(alt_seq.tolist())
        if val_alt < best_val - 1e-12:
            best_seq = alt_seq
            best_val = val_alt
            print(f"post‑reorder improvement: {val_alt:.6f}")

        # a few random permutations
        for _ in range(5):
            if time.time() >= deadline:
                break
            rand_seq = np.random.permutation(best_seq)
            val_rand = evaluate_sequence(rand_seq.tolist())
            if val_rand < best_val - 1e-12:
                best_seq = rand_seq
                best_val = val_rand
                print(f"random‑shuffle improvement: {val_rand:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
