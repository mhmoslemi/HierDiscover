```python
"""
Hybrid optimizer: combines parametric initialization, Adam with smooth max,
exponentiated‑gradient sub‑gradient refinement, symmetry enforcement,
and simulated‑annealing fine‑tuning. Aims to push the max‑convolution
bound below 1.503 for a 4096‑point step function.
"""

import time, math, numpy as np

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2*a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _symmetrize(v: np.ndarray) -> np.ndarray:
    sym = (v + v[::-1]) * 0.5
    # renorm to avoid tiny drift
    s = sym.sum()
    if s <= 0:
        return v
    return sym / s

def _adam_smooth_max(a0: np.ndarray, n: int, conv_fft_len: int,
                     corr_fft_len: int, deadline: float,
                     init_best_seq: np.ndarray | None,
                     init_best_val: float) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04

    lam = 1.5        # start temperature
    lam_factor = 1.008
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    max_iter = 500_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # Softmax to obtain a on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # enforce symmetry
        a = _symmetrize(a)

        # surrogate conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via chain‑rule
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update (on logits)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # random shake if stagnating
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def _subgradient_refinement(a0: np.ndarray, n: int, conv_fft_len: int,
                            corr_fft_len: int, deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float) -> tuple[np.ndarray, float]:
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()  # ensure simplex

    eta0 = 0.6
    decay = 0.99996
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    max_iter = 150_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active set of maximal entries
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # enforce symmetry and renorm
        a = _symmetrize(a)
        s = a.sum()
        if s <= 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_val

def _simulated_annealing(a0: np.ndarray, n: int, conv_fft_len: int,
                         deadline: float, init_best_val: float) -> tuple[np.ndarray, float]:
    """Simple SA on the simplex (log‑softmax representation)."""
    a_curr = a0.copy()
    val_curr = _objective(a_curr, n, conv_fft_len)
    best_seq = a_curr.copy()
    best_val = max(val_curr, init_best_val)  # we want the lower value

    # work in log‑space
    logits_curr = np.log(a_curr + 1e-15)

    start_time = time.time()
    total = deadline - start_time
    if total <= 0:
        return best_seq, best_val

    while time.time() < deadline:
        t_left = deadline - time.time()
        frac = t_left / total if total > 0 else 0.0

        # decay of proposal scale and temperature
        sigma = 0.12 * (frac ** 0.6) + 0.001   # noise std
        temp = max(1e-4, 0.015 * frac)        # temperature for acceptance

        logits_prop = logits_curr + np.random.randn(n) * sigma
        mx = logits_prop.max()
        exp_log = np.exp(logits_prop - mx)
        a_prop = exp_log / exp_log.sum()
        a_prop = _symmetrize(a_prop)   # keep symmetry

        val_prop = _objective(a_prop, n, conv_fft_len)

        # Metropolis acceptance
        if val_prop < val_curr or np.log(np.random.rand()) < -(val_prop - val_curr) / temp:
            logits_curr = logits_prop
            a_curr = a_prop
            val_curr = val_prop

        if val_prop < best_val:
            best_val = val_prop
            best_seq = a_prop.copy()
            print(f"[SA] new best {best_val:.6f}")

    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """Create a modest but diverse pool of parametric shapes."""
    candidates = []

    # warm start and simple shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet with varied concentration
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # symmetric edge exponential family
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=10):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand = w * edge + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # asymmetric two‑sided exponentials (coarse grid)
    alphas = np.logspace(-2, 1.0, num=6)
    w_vals = np.linspace(0.0, 1.0, 5)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    cand = wL * left + wR * right + wU / n
                    cand /= cand.sum()
                    candidates.append(cand)

    # simple linear decay and triangular shape
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base = base / base.sum()
        for w in (0.0, 0.5, 0.9):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # sparse spikes (few non‑zeros)
    for spikes in (3, 5, 8):
        for _ in range(12):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    """

    corr_fft_len = _next_pow2(3 * n - 2)

        warm_start = np.asarray(warm_start, dtype=np.float64)

            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate cheap candidates and keep a few elite ones
    candidates = _generate_parametric_candidates(n, warm_start)

        cand = np.maximum(cand, 0.0)

            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)

    top_k = 8
    top_candidates = [c for _, c in cand_vals[:top_k]]

            start, n, conv_fft_len, corr_fft_len,

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing local refinement
    if best_seq is not None:
        print("--- Simulated‑annealing refinement ---")
        best_seq, best_val = _simulated_annealing(best_seq, n, conv_fft_len,
                                                  deadline, best_val)

    # ------------------------------------------------------------------
    # Stage 4 – optional upscale to 8192 if time permits
    time_left = deadline - time.time()
    if time_left > 60.0:  # need some budget for a meaningful run
        n_big = 8192

        if best_seq is not None:
            # linear interpolation to larger grid

            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)

            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline_big, None, float("inf")
        )
        a_big, val_big = _subgradient_refinement(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline_big, a_big, val_big

"""
Hybrid optimiser for the max‑convolution bound.

The routine builds a modest pool of parametric shapes (including the
previous best if available), refines the top few with a long Adam phase
on a smooth‑max surrogate, follows with an exponentiated‑gradient
sub‑gradient phase, and finally applies a simulated‑annealing walk on the
simplex (log‑softmax representation).  Symmetry of the step function is
enforced throughout – the optimal bound is known to be attained by a
symmetric sequence – which shrinks the search space and improves
convergence.  If time permits the optimiser is lifted to a finer grid
(8192 points) and the whole pipeline is repeated.  The function returns
the best non‑negative sequence found within the allotted budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (zero‑padded) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    for i = 0 … n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """
    Enforce mirror symmetry v[i]=v[n‑1‑i] and renormalise to sum = 1.
    """
    sym = (v + v[::-1]) * 0.5
    s = sym.sum()
    if s <= 0:
        return v
    return sym / s


# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray, n: int, conv_fft_len: int,
                     corr_fft_len: int, deadline: float,
                     init_best_seq: np.ndarray | None,
                     init_best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    A higher temperature and a longer schedule are used.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04

    lam = 1.5            # start temperature
    lam_factor = 1.008   # gentle growth
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    max_iter = 500_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # soft‑max -> current a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        a = _symmetrize(a)                     # keep the sequence symmetric

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, lam_cap)

        # random shake if we have stagnated
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(a0: np.ndarray, n: int, conv_fft_len: int,
                            corr_fft_len: int, deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the *exact* max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    eta0 = 0.6
    decay = 0.99996
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    max_iter = 150_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # enforce symmetry and renormalise
        a = _symmetrize(a)
        s = a.sum()
        if s <= 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_val


def _simulated_annealing(a0: np.ndarray, n: int, conv_fft_len: int,
                         deadline: float, init_best_val: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing walk on the simplex.
    Proposals are made in log‑space and a Metropolis rule with a
    slowly decreasing temperature is used.
    """
    a_curr = a0.copy()
    val_curr = _objective(a_curr, n, conv_fft_len)

    best_seq = a_curr.copy()
    best_val = min(val_curr, init_best_val)

    logits_curr = np.log(a_curr + 1e-15)

    start_time = time.time()
    total = deadline - start_time
    if total <= 0:
        return best_seq, best_val

    while time.time() < deadline:
        t_left = deadline - time.time()
        frac = t_left / total if total > 0 else 0.0

        # scaling of the proposal distribution and temperature
        sigma = 0.12 * (frac ** 0.6) + 0.001
        temp = max(1e-4, 0.015 * frac)

        logits_prop = logits_curr + np.random.randn(n) * sigma
        mx = logits_prop.max()
        exp_log = np.exp(logits_prop - mx)
        a_prop = exp_log / exp_log.sum()
        a_prop = _symmetrize(a_prop)

        val_prop = _objective(a_prop, n, conv_fft_len)

        # Metropolis acceptance
        if val_prop < val_curr or np.log(np.random.rand()) < -(val_prop - val_curr) / temp:
            logits_curr = logits_prop
            a_curr = a_prop
            val_curr = val_prop

        if val_prop < best_val:
            best_val = val_prop
            best_seq = a_prop.copy()
            print(f"[SA] new best {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int,
                                    warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest but diverse pool of candidate vectors.
    Families include: uniform, Dirichlet draws, symmetric edge exponentials,
    asymmetric two‑sided exponentials, simple linear/triangular shapes,
    and sparse spike vectors.
    """
    candidates = []

    # warm‑start and simple cyclic shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # symmetric edge‑exponential family
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=10):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand = w * edge + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # asymmetric two‑sided exponentials (coarse grid)
    alphas = np.logspace(-2, 1.0, num=6)
    w_vals = np.linspace(0.0, 1.0, 5)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    cand = wL * left + wR * right + wU / n
                    cand /= cand.sum()
                    candidates.append(cand)

    # simple linear decay and triangular shape
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base = base / base.sum()
        for w in (0.0, 0.5, 0.9):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # sparse spike vectors
    for spikes in (3, 5, 8):
        for _ in range(12):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates

    """
    np.random.seed(seed)

        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()

        else:

    # ------------------------------------------------------------------

    cand_pool = _generate_parametric_candidates(n, warm_start)

    # ------------------------------------------------------------------

        deadline_big = time.time() + time_left - 5.0

        # short Adam + sub‑gradient on the large grid
        seq_big, val_big = _adam_smooth_max(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,

        )
        seq_big, val_big = _subgradient_refinement(

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
