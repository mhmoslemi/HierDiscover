def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 4096
    conv_len = next_pow2(2*n-1)
    corr_len = next_pow2(3*n-2)

    best_val = inf
    best_seq = None

    # Candidate generation phases:
    # 1. Warm-start if available
    def evaluate_and_update(b):
        a, perm = anti_pair_arrangement_with_perm(b)   # b sorted descending
        val = _objective(a)
        nonlocal best_val, best_seq
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"New best {val:.6f}")

    # Add warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.array(warm, dtype=float)
        # compute sorted descending to get b
        b = np.sort(warm)[::-1]
        b = b / b.sum()

    # Random Dirichlet candidates
    for alpha in [0.2, 0.5, 0.8, 1.0, 2.0]:
        if time.time() > deadline: break

    # Exponential sweep
    for c in np.linspace(0.0, 5.0, 51):
        if time.time() > deadline: break
        idx = np.arange(n)
        b = np.exp(-c * idx)
        b = b / b.sum()
        evaluate_and_update(b)

    # Gaussian edge shape generation similar to earlier but we can just use.

def subgradient_optimize_with_sort(b0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000):
    b = b0.copy()
    # Ensure sorted descending
    b = np.sort(b)[::-1]

    best_b = b.copy()
    # compute a and objective
    a, perm = anti_pair_arrangement_with_perm(b)
    best_val = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute a, perm
        a, perm = anti_pair_arrangement_with_perm(b)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # compute active mask
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # gradient wrt b
        grad_b = grad_a[perm]
        # step-size
        eta = 0.5 / np.sqrt(it + 1)   # can tune
        # multiplicative update (exponentiated gradient)
        b = b * np.exp(-eta * grad_b)
        # renorm
        s = b.sum()
        if s <= 0 or np.isnan(s) or np.isinf(s):
            # reinitialize to uniform
            b = np.full(n, 1.0 / n)
            s = 1.0
        b = b / s
        # ensure sorted descending
        b = np.sort(b)[::-1]

        # evaluate objective for best tracking
        if it % 2000 == 0:
            a_cur, _ = anti_pair_arrangement_with_perm(b)
            val_cur = _objective(a_cur)
            if val_cur < best_val:
                best_val = val_cur
                best_b = b.copy()
                print(f"[Subgrad sort it={it}] new best {val_cur:.6f}")
        it += 1
    # final compute a from best_b
    a_best, _ = anti_pair_arrangement_with_perm(best_b)
    final_val = _objective(a_best)
    return a_best, final_val

"""
Propose a low‑value step‑function by (1) a fast global sweep over
parameterised families (Dirichlet, power‑law, exponential, edge‑Gaussian),
(2) picking the best multiset of values, (3) optimising that multiset on the
simplex with an exponentiated‑gradient sub‑gradient method while enforcing
the “anti‑pair” ordering (the optimal permutation for a given sorted
multiset), and (4) a final short annealing phase.
All operations respect the 1000 s budget and the non‑negative constraint.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers ----------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, no safety wrapper) ---------------------------------

def _objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a) / (sum(a)^2).  Assumes a ≥ 0."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Anti‑pair permutation (independent of the actual values) -------------

def _anti_pair_permutation(n: int) -> np.ndarray:
    """
    For a sorted descending vector b (length n) return an array `perm`
    such that a[perm[i]] = b[i] yields the anti‑pair layout:
    largest half on the left, smallest half reversed on the right.
    """
    half = n // 2
    perm = np.empty(n, dtype=int)

    # largest half (or left side)
    perm[:half] = np.arange(half)

    # middle element for odd length
    if n % 2 == 1:
        perm[half] = half
        start_small = half + 1
    else:
        start_small = half

    # smallest half (reversed)
    small_len = n - start_small
    perm[start_small:] = n - 1 - np.arange(small_len)

    return perm


def _anti_pair_from_sorted(b: np.ndarray, perm: np.ndarray) -> np.ndarray:
    """Produce the anti‑pair arranged vector a from a sorted descending b."""
    a = np.empty_like(b)
    a[perm] = b
    return a


# ----------------------------------------------------------------------
# Sub‑gradient optimisation on the sorted multiset ---------------------

def _subgradient_optimize_sorted(
    b0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient on the sorted values `b`.
    After each update the vector is re‑normalised and re‑sorted,
    guaranteeing a valid anti‑pair layout at every iteration.
    """
    # static permutation for anti‑pair layout
    perm = _anti_pair_permutation(n)

    # start from a descending copy
    b = np.sort(b0)[::-1]
    b = np.maximum(b, 0.0)
    if b.sum() < 1e-12:
        b = np.full(n, 1.0 / n)
    else:
        b = b / b.sum()

    best_b = b.copy()
    a = _anti_pair_from_sorted(b, perm)
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # anti‑pair arrangement
        a = _anti_pair_from_sorted(b, perm)

        # exact convolution & max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        # keep the global best
        if val < best_val:
            best_val = val
            best_b = b.copy()
            if it % 2000 == 0:
                print(f"[Subgrad] it={it:06d} new best {best_val:.6f}")

        # active set of the max (sub‑gradient)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient w.r.t. the sorted multiset b
        grad_b = grad_a[perm]

        # step‑size (decaying)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient multiplicative update
        b = b * np.exp(-eta * grad_b)

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 1e-2
            b = b * (1.0 + jitter)

        # renormalise onto the simplex
        total = b.sum()
        if total <= 0.0 or np.isnan(total):
            b = np.full(n, 1.0 / n)
            total = 1.0
        b = b / total

        # enforce descending order (the permutation is defined for sorted vectors)
        b = np.sort(b)[::-1]

        it += 1

    # final best a
    a_best = _anti_pair_from_sorted(best_b, perm)
    final_val = _objective(a_best)
    return a_best, final_val

    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)

    # Convert the current best (a) back to the sorted multiset b
    b_start = np.sort(best_seq)[::-1]

    sub_deadline = deadline - 5.0
    if time.time() < sub_deadline:
        a_refined, val_refined = _subgradient_optimize_sorted(
            b_start,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=600_000,
            eta0=0.7,
        )
        if val_refined < best_val:
            best_val = val_refined
            best_seq = a_refined.copy()
            print(f"[Refined] improved to {val_refined:.6f}")

    # ------------------------------------------------------------------
    # 4) Very short simulated‑annealing style hill‑climbing on the final a
    #    (swap or transfer moves).  This is cheap and can sometimes shave
    #    a few thousandths off the bound.
    def _local_search(a0: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
        a = a0.copy()
        best_a = a.copy()
        best_obj = _objective(a)
        start_temp = 0.01
        moves = 0
        while time.time() < time_limit and moves < 120_000:
            if np.random.rand() < 0.5:               # swap two entries
                i, j = np.random.randint(n, size=2)
                if i != j:
                    a2 = a.copy()
                    a2[i], a2[j] = a2[j], a2[i]
                else:
                    continue
            else:                                    # transfer a fraction
                i = np.random.randint(n)
                if a[i] < 1e-14:
                    continue
                j = np.random.randint(n)
                if i == j:
                    continue
                delta = a[i] * (np.random.rand() * 0.4)
                a2 = a.copy()
                a2[i] -= delta
                a2[j] += delta

            val2 = _objective(a2)
            if val2 < best_obj:
                best_obj = val2
                best_a = a2.copy()
                a = a2
            else:
                temp = start_temp * (1.0 - moves / 120_000)
                if temp > 0 and np.random.rand() < np.exp(-(val2 - best_obj) / temp):
                    a = a2
            if moves % 5000 == 0:
                print(f"[Local move {moves:06d}] best {best_obj:.6f}")
            moves += 1
        return best_a, best_obj

    if time.time() < deadline - 2.0:
        a_local, val_local = _local_search(best_seq, deadline - 2.0)
        if val_local < best_val:
            best_val = val_local
            best_seq = a_local.copy()
            print(f"[Local] improved to {val_local:.6f}")

    # ------------------------------------------------------------------
    # 5) Final safety clamp (required by evaluate_sequence)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
