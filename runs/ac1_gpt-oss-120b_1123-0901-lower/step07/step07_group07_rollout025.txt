        cand = cand / cand.sum()
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    top_k = min(20, len(evaluated))
    top_candidates=[c for _,c in evaluated[:top_k]]
    best_seq = None
    best_val = float("inf")

    # compute remaining after eval
    time_remaining = deadline - time.time()

    # splitting rest equally among other top-1 candidates.

        main_time = time_remaining * 0.7
        other_time = time_remaining - main_time
        # allocate equally among others (top_k-1)
        per_other = other_time / max(1, (top_k-1))
        # refine main candidate

                                          n,
                                          conv_fft_len,
                                          corr_fft_len,
                                          time.time() + main_time,
                                          best_seq,

        # refine others

            if time.time() >= deadline: break
            best_seq,best_val = _deep_optimize(cand,
                                               n,
                                               conv_fft_len,
                                               corr_fft_len,
                                               time.time() + per_other,
                                               best_seq,
                                               best_val)
    # after deep optimization, maybe upscale if time left
    if time.time() < deadline - 30.0:

        n_big = n*2

        corr_fft_len_big = _next_pow2(3*n_big-2)

            x_new = np.linspace(0,1,n_big,endpoint=False)

        corr_fft_len=corr_fft_len_big

    # Final random hill climbing
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()

            i,j = np.random.randint(n, size=2)
            if i==j: continue

            delta = (np.random.rand()-0.5) * 0.1 * mass  # up to ±5% of smaller
            if delta> a[j]:

            if -delta > a[i]:

            max_c = conv.max()
            val = 2.0 * n * max_c / (a.sum()**2)

                best_local_val = val
                best_seq = a.copy()
                # keep change
            else:
                # revert
                a[i]-=delta
                a[j]+=delta
            it+=1
        best_val = best_local_val

    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")

def _adam_smooth_max(a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 1.0
    lam_factor = 1.015
    lam_cap = 1e5
    best_seq = global_best_seq
    best_val = global_best_val
    it = 0
    max_iter = 500000
    last_improve = time.time()
    while time.time() < deadline and it < max_iter:
        # softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # gradient wrt logits via chain rule: derivative of softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # schedule lam
        lam = min(lam * lam_factor, lam_cap)
        # random jitter if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
        it += 1
    return best_seq, best_val

"""
Propose a low‑evaluation step‑function.

The search proceeds in four stages:

1. **Rich candidate pool** – a mixture of symmetric and
   asymmetric parametric families (uniform, Dirichlet, exponential,
   power‑law, linear ramps, piece‑wise constant, beta‑shapes, etc.) is
   generated and evaluated very quickly using FFT‑based convolution.
   The best dozen candidates are retained.

2. **Deep optimisation** – each retained start undergoes a two‑phase
   refinement:
      * Adam on a log‑sum‑exp surrogate of the max‑convolution
        (soft‑max weights).  The temperature (λ) is increased gradually
        and a small jitter is injected when progress stalls.
      * Exponentiated‑gradient sub‑gradient steps on the true
        max‑convolution (active‑set weights) with a decaying step‑size.
   Hyper‑parameters are tuned for speed and robustness.

3. **Upscaling** – if time permits the current best vector is linearly
   interpolated to a larger grid (doubling the length) and the whole
   deep‑optimisation is repeated.  A larger grid gives more degrees of
   freedom and typically reduces the objective further.

4. **Fine‑grained hill‑climbing** – a short random‑mass‑transfer local
   search finishes the run, accepting only moves that improve the
   objective.

All vectors are kept on the probability simplex (non‑negative,
normalised to sum ≈ 1).  The function respects the 1000 s deadline and
returns the best sequence found as a plain Python list.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using a real FFT
    of length ``fft_len`` (must be ≥ 2·len(a)-1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective (normed sum≈1).
    Returns 2·n·max(a*a) / (sum(a)**2).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:

    Adam optimiser on a log‑sum‑exp surrogate of max(conv).
    Returns the best sequence/value seen during the run.
    """

    logits = np.log(a0 + eps)               # start in soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = 1.0                               # initial temperature
    lam_factor = 1.015                      # geometric increase
    lam_cap = 1e5

    best_val = global_best_val

    max_iter = 500_000

        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate w.r.t. a -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)               # ⟨a,∇⟩ term from chain rule
        grad_log = a * (grad_a - avg)         # gradient w.r.t. logits

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- jitter if stuck -----
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.5
    it = 0
    max_iter = 80_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:06d} new best {val:.6f}")

        # active set where the convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # occasional re‑initialisation
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # occasional random jitter to escape plateaus
        if it % 2000 == 0:
            i, j = np.random.randint(n, size=2)
            delta = (np.random.rand() - 0.5) * 0.01 * min(a[i], a[j])
            a[i] += delta
            a[j] -= delta
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # early stop if stuck for a while
        if time.time() - last_improve > 8.0:
            # keep iterating – additional jitter above may revive progress
            pass

        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by a sub‑gradient refinement.
    Updates the globally best vector/value.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose an initial grid size based on the total budget
    if budget_s > 650:                # enough time for a larger grid
        n = 8192
    else:
        n = 4096

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered vector, if available
    warm_start = globals().get("height_sequence_1")
    ws = None
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] != n:
            # linearly interpolate to the desired length
            x_old = np.linspace(0, 1, ws.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None

    # ------------------------------------------------------------------
    # ---- Stage 1: generate & pre‑select candidates --------------------
    cand_pool = _generate_candidates(n, ws)

    evaluated = []
    for idx, cand in enumerate(cand_pool):
        if time.time() >= deadline:
            break
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        # optional progress print:
        # print(f"[candidate {idx}] val≈{val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(20, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: selected {top_k} promising starts (n={n})")

    # ------------------------------------------------------------------
    # ---- Stage 2: deep optimisation on the selected starts ----------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # allocate the remaining time: 70 % for the very best start,
    # the rest shared equally among the others.
    time_left = deadline - time.time()
    if time_left <= 0:
        time_left = 0.0
    main_time = time_left * 0.70
    other_time = time_left - main_time
    per_other = other_time / max(1, (top_k - 1))

    # optimise the top candidate longer
    if top_candidates:
        best_seq, best_val = _deep_optimize(
            top_candidates[0],
            n,
            conv_fft_len,
            corr_fft_len,
            time.time() + main_time,
            best_seq,
            best_val,
        )
        # then a quick pass for the remaining promising starts
        for cand in top_candidates[1:]:
            if time.time() >= deadline:
                break
            best_seq, best_val = _deep_optimize(
                cand,
                n,
                conv_fft_len,
                corr_fft_len,
                time.time() + per_other,
                best_seq,
                best_val,
            )

    # ------------------------------------------------------------------
    # ---- Stage 3: up‑scale to a larger grid if time permits ----------
    if time.time() < deadline - 30.0:
        n_big = n * 2
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big = a_big / a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print(f"--- Upscaling optimisation to n={n_big} ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # update bookkeeping for the final refinement
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # ---- Stage 4: final random hill‑climbing refinement --------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local_val = best_val
        it = 0
        max_local_iter = 20_000
        while time.time() < deadline and it < max_local_iter:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # transfer a modest fraction of the smaller mass
            delta = (np.random.rand() - 0.5) * 0.1 * min(a[i], a[j])
            if delta > a[j]:
                delta = a[j]
            if -delta > a[i]:
                delta = -a[i]
            a[i] += delta
            a[j] -= delta

            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c / (a.sum() ** 2)

            if val < best_local_val:
                best_local_val = val
                best_seq = a.copy()
            else:
                # revert the move
                a[i] -= delta
                a[j] += delta
            it += 1

        best_val = best_local_val
        print(f"Final hill‑climb finished after {it} attempts, best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
