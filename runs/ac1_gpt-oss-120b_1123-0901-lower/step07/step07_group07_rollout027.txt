"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  

The algorithm consists of four stages:

1. **Broad parametric seeding** – generate a diverse set of candidates,
   especially mixtures of the arcsine shape (which flattens the
   self‑convolution in the continuous limit) with a uniform component.
   A fast FFT‑based surrogate of the objective is used to keep the
   cheapest few.

2. **Refinement** – each retained start is optimised by
   (a) Adam on a smooth log‑sum‑exp approximation of the
   maximum convolution and (b) an exact sub‑gradient step on the true
   max‑convolution.  Both phases work on the simplex via a
   soft‑max / exponentiated‑gradient parametrisation.

3. **Upscaling** – if time permits the current best vector is interpolated
   to a finer grid (n = 8192) and the whole optimisation is repeated,
   exploiting the fact that the target objective tends to a limit as
   n grows.

4. **Random hill‑climbing** – a short, cheap local search that swaps a
   little mass between random indices and keeps improving moves.

All helper functions are defined at module level (no closures) and the
search respects the 1000 s budget.  The implementation uses only NumPy
and runs on a single CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (without the protect‑checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:                # defensive (should never happen)
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generation
def _generate_random_dirichlet_candidates(n: int, concentrations, count: int):
    """Return a list of Dirichlet draws with given concentration parameters."""
    cand = []
    for conc in concentrations:
        for _ in range(count):
            cand.append(np.random.dirichlet(np.full(n, conc)))
    return cand


def _generate_beta_mixture_candidates(
    n: int,
    alphas,
    lambdas,
):
    """
    Symmetric beta density a_i ∝ (i+1)^{α-1}(n-i)^{α-1},
    mixed with a uniform component:
        cand = (1-λ)*uniform + λ*beta .
    Both components are already normalised, so the mixture sums to 1.
    """
    i = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    cands = []
    for α in alphas:
        beta = (i + 1.0) ** (α - 1.0) * (n - i) ** (α - 1.0)
        beta_sum = beta.sum()
        if beta_sum == 0:
            continue
        beta = beta / beta_sum
        for lam in lambdas:
            mix = (1.0 - lam) * uniform + lam * beta
            cands.append(mix)
    return cands


def _generate_arcsine_mixture_candidates(n: int, lambdas):
    """
    arcsine shape a_i ∝ 1/√((i+0.5)(n‑i‑0.5)),
    mixed with a uniform component as above.
    """
    i = np.arange(n, dtype=np.float64)
    arc = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arc = arc / arc.sum()
    uniform = np.full(n, 1.0 / n)
    cands = []
    for lam in lambdas:
        mix = (1.0 - lam) * uniform + lam * arc
        cands.append(mix)
    return cands


# ----------------------------------------------------------------------
# Optimisation primitives
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
):
    """
    Adam on a smoothed log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # start in soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                               # reasonable learning‑rate
    lam = 0.05                               # start temperature low
    lam_factor = 1.025                       # schedule factor
    lam_cap = 1e5

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 300_000

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- surrogate objective (log‑sum‑exp) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n}] it={it:06d} new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)               # ⟨a,∇⟩ term from the chain rule
        grad_log = a * (grad_a - avg)         # gradient wrt logits

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional jitter if stuck ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
):
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # keep on simplex

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    max_iter = 30_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n}] it={it:05d} new best {val:.6f}")

        # active set where the convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        total = a.sum()
        if total < 1e-12:                      # numerical safety‑net
            a = np.random.dirichlet(np.ones(n))
            total = a.sum()
        a /= total

        if time.time() - last_improve > 6.0:
            break
        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
):
    """
    Run Adam (smooth‑max) followed by a sub‑gradient refinement.
    Returns the best vector/value seen (may improve the global best).
    """
    # Phase 1 – Adam on smooth max
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Phase 2 – exact sub‑gradient
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found (plain Python
    list of floats).  The routine respects a 1000 s wall‑clock budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 0 – obtain a warm start if one is available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        ws = np.maximum(ws, 0.0)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # Stage 1 – broad, inexpensive candidate sweep (small grid)
    n0 = 2048
    conv_fft_len_0 = _next_pow2(2 * n0 - 1)
    cand_pool = []

    # uniform baseline
    cand_pool.append(np.full(n0, 1.0 / n0))

    # warm‑start (if any)
    if ws is not None:
        if ws.shape[0] > n0:
            cand_pool.append(ws[:n0])
        elif ws.shape[0] < n0:
            cand_pool.append(np.pad(ws, (0, n0 - ws.shape[0])))
        else:
            cand_pool.append(ws)

    # random Dirichlet draws (various concentrations)
    cand_pool.extend(_generate_random_dirichlet_candidates(
        n0,
        concentrations=(0.05, 0.1, 0.2, 0.5, 1.0, 2.0),
        count=4,
    ))

    # beta‑mixtures (α ∈ [0.55, 1.5])
    alphas = np.linspace(0.55, 1.5, 20)
    lambdas = np.linspace(0.0, 1.0, 11)   # weight of beta component
    cand_pool.extend(_generate_beta_mixture_candidates(n0, alphas, lambdas))

    # arcsine‑uniform mixtures (λ governs the amount of arcsine)
    cand_pool.extend(_generate_arcsine_mixture_candidates(n0, lambdas))

    # ------------------------------------------------------------------
    # Fast evaluation of all candidates (keep the best few)
    evals = []
    for idx, cand in enumerate(cand_pool):
        if time.time() >= deadline:
            break
        # protect against numerical 0
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n0))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n0, conv_fft_len_0)
        evals.append((val, cand))
        # optional progress line
        # print(f"[seed {idx}] val≈{val:.6f}")

    evals.sort(key=lambda x: x[0])
    top_k = min(12, len(evals))
    top_candidates = [c for _, c in evals[:top_k]]
    print(f"Stage 1: selected {top_k} promising starts (n={n0})")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on a finer grid (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # helper to interpolate a candidate to the current grid size
    def _interpolate_to(v: np.ndarray, new_n: int) -> np.ndarray:
        x_old = np.linspace(0.0, 1.0, v.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
        return np.interp(x_new, x_old, v)

    for start_id, cand_small in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        cand_big = _interpolate_to(cand_small, n)
        cand_big = np.maximum(cand_big, 0.0)
        cand_big /= cand_big.sum()
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} (n={n}) ---")
        best_seq, best_val = _deep_optimize(
            cand_big,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        # keep a reference to the best vector for possible later up‑scaling
        # (already stored in best_seq)

    # ------------------------------------------------------------------
    # Stage 3 – upscale to a larger grid if time permits (n = 8192)
    time_left = deadline - time.time()
    if time_left > 40.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            a_big = _interpolate_to(best_seq, n_big)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Stage 4 – final local random hill‑climb (very cheap moves)
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local_val = best_val
        it = 0
        max_local_iter = 8000
        while time.time() < deadline and it < max_local_iter:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # propose a tiny transfer of mass from i to j (or opposite)
            delta = (np.random.rand() - 0.5) * 0.3 * min(a[i], a[j])
            if delta > a[j]:
                delta = a[j]
            if -delta > a[i]:
                delta = -a[i]
            if abs(delta) < 1e-16:
                continue
            a[i] += delta
            a[j] -= delta

            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c / (a.sum() ** 2)

            if val < best_local_val:
                best_local_val = val
                best_seq = a.copy()
                # keep the improvement
            else:
                # revert move
                a[i] -= delta
                a[j] += delta
            it += 1

        best_val = best_local_val
        print(f"Final hill‑climb finished after {it} attempts, best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in case nothing succeeded (should never happen)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
