def _simplex_proj(v):
    """Projects onto the probability simplex."""
    # ensures sum 1, nonnegative
    # if all v <=0, set uniform.
    # Implementation:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _projected_gradient_descent(a0, n, conv_fft_len, corr_fft_len, deadline, init_best_seq=None, init_best_val=float('inf')):
    a = a0.copy()
    # ensure sum 1
    sum_a = a.sum()
    if sum_a <= 0: a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_seq = a if init_best_seq is None else init_best_seq
    best_val = init_best_val if init_best_val != float('inf') else _objective(a, n, conv_fft_len)

    # hyperparameters
    eta = 0.5  # initial step size
    decay = 0.99996
    min_eta = 1e-6
    it = 0
    max_iter = 300_000

    last_improve = time.time()

    while time.time() < deadline and it < max_iter and eta > min_eta:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # sum =1

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[PGD] iter={it:06d} new best {val:.6f}")

        # find active set
        # Use tolerance maybe 1e-12 absolute or relative
        active = (conv >= max_c - 1e-12)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / np.count_nonzero(active)

        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # backtracking line search
        step = eta
        success = False
        for _ in range(5):
            a_candidate = a - step * grad_a
            # project onto simplex
            a_candidate = _simplex_projection(a_candidate)
            # compute objective
            conv_cand = _full_convolution(a_candidate, conv_fft_len)
            max_c_cand = conv_cand.max()
            val_cand = 2.0 * n * max_c_cand / (a_candidate.sum() ** 2)
            if val_cand <= val:
                a = a_candidate
                success = True
                eta = step  # keep step size
                break
            else:
                step *= 0.5

        if not success:
            # no improvement; reduce step size
            eta *= 0.5

        # still drop ETA gradually per iteration
        eta = max(eta * decay, min_eta)

        it += 1

        if time.time() - last_improve > 6.0:
            # stagnation, break
            break

    return best_seq, best_val

"""
Propose a non‑negative step function minimizing the provided objective.

Algorithm
---------
1. **Candidate pool** – a diverse set of parametric families (uniform,
   Dirichlet draws, symmetric/asymmetric exponentials and power‑laws,
   linear/triangular, sinusoidal, sparse spikes) is generated.  Each
   candidate is normalised to sum = 1.

2. **Projected‑gradient descent (PGD)** – starting from the best few
   candidates we run a custom gradient descent on the exact
   ‑max‑convolution objective.  The gradient is the sub‑gradient obtained
   from the active set of indices attaining the current maximum.
   After each step we project onto the probability simplex (non‑negative,
   sum = 1).  A simple back‑tracking line‑search guarantees monotone
   improvement; an exponential step‑size decay provides convergence.

3. **Local stochastic refinement** – cheap Gaussian perturbations of the
   logits are accepted when they improve the exact objective.

4. **Multi‑scale refinement** – if time permits we interpolate the best
   sequence to a finer grid (n = 8192) and repeat the PGD + stochastic
   phases.  Larger grids allow a smoother mass distribution and often
   lower the bound.

All stages respect a global deadline (default 1000 s).  The function
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective  F = 2·n·max(a*a) / (sum(a)²).
    Assumes a is non‑negative.
    """
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are negative – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse pool of candidate vectors.
    Families:
      – uniform
      – random Dirichlet (various concentrations)
      – symmetric edge‑exponential / power‑law
      – asymmetric two‑sided exponentials & power‑laws
      – linear / triangular
      – sinusoidal half‑wave
      – sparse spike vectors
      – cosine‑based bump
    All candidates are normalised to sum ≈ 1.
    """
    candidates = []

    # ----- warm‑start & shifted copies -----
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ----- uniform baseline -----
    candidates.append(np.full(n, 1.0 / n))

    # ----- Dirichlet draws -----
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(8):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # ----- symmetric exponential / power‑law families -----
    for alpha in np.logspace(-2, 1.0, num=10):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    for p in np.linspace(0.2, 6.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ----- asymmetric two‑sided exponentials -----
    alphas = np.logspace(-2, 1.0, num=8)
    w_vals = np.linspace(0.0, 1.0, 6)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ----- asymmetric power‑law -----
    p_vals = np.linspace(0.2, 5.0, num=6)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ----- linear / triangular shapes -----
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base = base / base.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ----- sinusoidal half‑wave -----
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ----- cosine bump (centered) -----
    shape = np.maximum(np.cos(np.pi * (np.arange(n) - n/2) / n), 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ----- sparse spikes -----
    for spikes in (3, 5, 8):
        for _ in range(15):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates

def _projected_gradient_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    After each gradient step we project onto the probability simplex.
    A simple back‑tracking line‑search guarantees monotone improvement.
    """
    # initialise on the simplex
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / s

    best_seq = a.copy() if init_best_seq is None else init_best_seq.copy()
    best_val = init_best_val if init_best_val != float("inf") else _objective(a, n, conv_fft_len)

    eta = 0.5            # initial step size
    decay = 0.99996
    min_eta = 1e-6
    max_iter = 250_000
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter and eta > min_eta:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[PGD] iter={it:06d} new best {val:.6f}")

        # --- sub‑gradient from active set ---
        active = conv >= max_c - 1e-12
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / np.count_nonzero(active)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # --- back‑tracking line search ---
        step = eta
        improved = False
        for _ in range(5):
            a_candidate = a - step * grad
            a_candidate = _simplex_projection(a_candidate)
            conv_c = _full_convolution(a_candidate, conv_fft_len)
            max_c_c = conv_c.max()
            val_c = 2.0 * n * max_c_c / (a_candidate.sum() ** 2)
            if val_c <= val:
                a = a_candidate
                eta = step
                improved = True
                break
            step *= 0.5

        if not improved:
            eta *= 0.5  # shrink step if we cannot improve

        # exponential decay of step size
        eta = max(eta * decay, min_eta)

        it += 1

        # early stop if stuck for a while
        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: perturb logits with Gaussian noise,
    accept if exact objective improves.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.025
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimizing the given evaluation
    function.  Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # 0.5 s safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------- problem size ------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------- warm‑start handling -----------------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------- Stage 1: generate candidates ---------------
    all_cands = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for cand in all_cands:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fallback to uniform
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        cand_vals = [(best_val, best_seq)]

    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"Stage 1: evaluated {len(cand_vals)} candidates, best = {cand_vals[0][0]:.6f}")

    # --------------------- Stage 2: PGD on top starts -----------------
    for idx, start in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- PGD start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _projected_gradient_descent(
            start,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------- Stage 3: local stochastic refinement -------
    if best_seq is not None:
        print("--- Local stochastic refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # --------------------- Stage 4: larger grid if time permits -------
    time_left = deadline - time.time()
    if time_left > 45.0:                     # need a reasonable budget
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # interpolate current best to the finer grid
        if best_seq is not None:
            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _projected_gradient_descent(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # final local search on the fine grid
        if best_seq is not None:
            best_seq, best_val = _local_random_search(
                best_seq,
                n_big,
                conv_fft_len_big,
                deadline,
                best_val,
            )
        n = n_big   # remember final size

    # --------------------- final sanity checks -----------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished: best ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
