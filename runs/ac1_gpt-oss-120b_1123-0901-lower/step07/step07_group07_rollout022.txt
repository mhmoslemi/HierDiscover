"""
search_step_function.py

Multi‑stage optimiser for the “max‑convolution” objective

    F(a) = 2·n·max(a∗a) / (sum(a)^2) ,

where a is a non‑negative vector (the step‑function heights) and * is the
linear convolution.  The problem is homogeneous, so we keep a on the
probability simplex (∑a = 1).

Algorithm
---------

1. **Broad seeding** – a large, diverse pool of candidate shapes is built.
   It contains symmetric beta‑, exponential, power‑law, sinusoidal,
   Gaussian, arcsine families, many random Dirichlet draws (different
   concentrations) and a few asymmetric mixtures obtained by shifting
   the symmetric families.  All candidates are normalised to sum 1 and
   quickly evaluated with an FFT‑based convolution.  The best ~30 seeds
   are retained.

2. **Continuation optimisation** – each retained seed is refined by
   two phases

   * **Adam on a smooth‑max surrogate**:  max(·) is replaced by a
     log‑sum‑exp with temperature λ.  λ is increased exponentially so
     that the surrogate tightens gradually.  Variables are written in
     soft‑max (logits) space, guaranteeing non‑negativity and the
     simplex constraint.  Adam with Nesterov‑style momentum updates the
     logits.

   * **Exact sub‑gradient descent**:  The true max‑convolution is used.
     The active set (indices attaining the maximum) yields a sub‑gradient
     which is applied with an exponentiated‑gradient update (mirror
     descent w.r.t. KL).  The step‑size decays as η₀/√t.

   Both phases keep the best vector/value seen so far.

3. **Aggressive random hill‑climb** – using the globally best vector,
   many tiny random mass‑transfer moves (i↔j) are tried.  After each
   move the exact objective (via FFT) is recomputed; the move is kept
   only if it improves the objective.  The number of attempts scales
   with the remaining time (up to ~200 000 moves).

4. **Upscaling (optional)** – if time permits, the best vector is linearly
   interpolated to a finer grid (n = 8192) and steps 1‑3 are repeated.
   This often yields a further reduction because the extra degrees of
   freedom allow a finer shape.

All updates respect the simplex; the final vector is clipped to the
evaluator’s range [0, 1000] and returned as a plain Python list.

The routine respects the 1000 s wall‑time budget; it prints progress
information but never exceeds the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective (∑a≈1).
    Returns 2·n·max(a*a) / (sum(a)**2).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

# Adam optimisation on a smooth‑max surrogate
def _adam_continuation(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr0 = 0.09                         # modest base learning‑rate
    lam = 0.3                          # start temperature
    lam_factor = 1.03
    lam_cap = 1e5

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 500_000

    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- surrogate objective (only for monitoring) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        # w_i ∝ exp(lam * (conv_i - max_c))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate with respect to a ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                     # ⟨a,∇⟩ term
        grad_log = a * (grad_a - avg)               # chain rule for logits

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / (1.0 + it * 1e-6)                 # tiny decay
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional jitter if stuck ----
        if time.time() - last_improve > 3.0:
            jitter = (np.random.rand(n) - 0.5) * 1e-2
            logits += jitter
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (exponentiated‑gradient)
def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.7
    it = 0
    max_iter = 80_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:06d} new best {val:.6f}")

        # ---- active set for the max ----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- multiplicative (exponentiated‑gradient) update ----
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                    # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:     # stagnation → stop early
            break

        it += 1

    return best_seq, best_val

def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by a sub‑gradient refinement.
    Updates the globally best vector/value.
    """
    # Phase 1 – Adam on smooth‑max
    best_seq, best_val = _adam_continuation(
        a0,
        n,

        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

# Candidate generation – rich and slightly asymmetric families
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a rich pool of candidates (both symmetric and a few asymmetric).
    All vectors are normalised to have sum ≈ 1.
    """
    candidates: list[np.ndarray] = []

    i = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Warm‑start & cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        # a few circular shifts (creates asymmetry)
        for shift in (n // 8, n // 4, 3 * n // 8):
            candidates.append(np.roll(ws, shift))

    # ------------------------------------------------------------------
    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # Random Dirichlet – wide range of concentrations
    for conc in (0.001, 0.005, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # ------------------------------------------------------------------
    # Edge‑exponential families (symmetric)
    for alpha in np.logspace(-2, 1.0, num=12):   # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.7, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # ------------------------------------------------------------------
    # Edge‑power‑law families (symmetric)
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # ------------------------------------------------------------------
    # Symmetric beta‑shapes
    for alpha in (0.2, 0.4, 0.5, 0.8, 1.2, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6, 8):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ------------------------------------------------------------------
    # Asymmetric sinusoidal mixtures (shifted sine‑power)
    for shift in (int(0.1 * n), int(0.3 * n), int(0.5 * n)):
        base = np.sin(np.pi * (i + 0.5 - shift) / n) ** 4
        base = np.clip(base, 0.0, None)
        base /= base.sum()
        candidates.append(base)

    # ------------------------------------------------------------------
    # Product‑power‑law (edges stronger)
    for p in (0.5, 1.0, 1.5, 2.0):
        shape = 1.0 / np.sqrt((i + 1.0) ** p * (n - i) ** p)
        shape = np.clip(shape, 0.0, None)
        shape /= shape.sum()
        candidates.append(shape)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary grid size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # ----- Stage 1 – generate & pre‑select candidates -----------------
    cand_pool = _generate_parametric_candidates(n, ws)

    # quick evaluation (FFT based, cheap)
    evaluated = []
    for idx, cand in enumerate(cand_pool):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        # optional progress output
        if idx % 50 == 0:
            print(f"[candidate {idx:04d}] val≈{val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(30, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: selected {top_k} promising starts (n={n})")

    # ------------------------------------------------------------------
    # ----- Stage 2 – deep optimisation on the selected starts ---------
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # ----- Stage 3 – aggressive local random hill‑climb ---------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local_val = best_val
        it = 0
        # allow up to 200 000 attempts (limited by remaining time)
        max_attempts = 200_000
        while time.time() < deadline and it < max_attempts:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # Transfer a modest fraction of the smaller mass
            max_transfer = min(a[i], a[j])
            if max_transfer < 1e-14:
                continue
            delta = (np.random.rand() - 0.5) * 0.4 * max_transfer  # ±20 % of smaller mass
            # Ensure positivity after the move
            if delta > a[j]:
                delta = a[j]
            if -delta > a[i]:
                delta = -a[i]
            if abs(delta) < 1e-15:
                continue

            a[i] += delta
            a[j] -= delta

            val = _objective(a, n, conv_fft_len)

            if val < best_local_val:
                best_local_val = val
                best_seq = a.copy()
                # keep the improvement
                # reset stagnation counter
            else:
                # revert move
                a[i] -= delta
                a[j] += delta
            it += 1

            if it % 20000 == 0:
                print(f"[hill‑climb] iteration {it}, best {best_local_val:.6f}")

        best_val = best_local_val
        print(f"Final hill‑climb finished after {it} attempts, best={best_val:.6f}")

    # ------------------------------------------------------------------
    # ----- Stage 4 – optional upscaling if time permits ----------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

        # final tiny hill‑climb on the larger grid (if time remains)
        if time.time() < deadline - 5.0:
            a = best_seq.copy()
            best_local_val = best_val
            it = 0
            max_attempts = 150_000
            while time.time() < deadline and it < max_attempts:
                i, j = np.random.randint(n, size=2)
                if i == j:
                    continue
                max_transfer = min(a[i], a[j])
                if max_transfer < 1e-15:
                    continue
                delta = (np.random.rand() - 0.5) * 0.3 * max_transfer
                if delta > a[j]:
                    delta = a[j]
                if -delta > a[i]:
                    delta = -a[i]
                a[i] += delta
                a[j] -= delta
                val = _objective(a, n, conv_fft_len)
                if val < best_local_val:
                    best_local_val = val
                    best_seq = a.copy()
                else:
                    a[i] -= delta
                    a[j] += delta
                it += 1
            best_val = best_local_val
            print(f"Fine hill‑climb on n={n} finished after {it} attempts, best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
