def _lp_update(a, M, n):
    # Build Toeplitz matrix A (size (2n-1, n))
    col = np.concatenate([a, np.zeros(n-1)])
    row = np.concatenate([a[:1], np.zeros(n-1)])

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)

    start = a.shape[0] - 1

"""
LP‑sub‑gradient search for a non‑negative step‑function.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a)²)

where conv is the linear convolution.  The method works in three
stages:

1. Initialise a probability vector a (the sequence) – preferably from
   a warm–start, otherwise a random Dirichlet draw.
2. Repeatedly solve a linear programme

        maximise   sum(g)
        subject to  a * g ≤ M   (all shifts)
                    g ≥ 0

   where M = max(conv(a,a)).  Mixing the feasible g with a yields a
   new a with a non‑increasing maximal convolution.
3. Refine the result with an exponentiated‑gradient sub‑gradient
   descent on the exact max‑convolution.  If time permits the final
   vector is up‑sampled and a short refinement on a larger grid is
   performed.

The whole search stops after the allotted budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.linalg

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σₖ w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns the projected vector (non‑negative, sum=1).
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _lp_update(a: np.ndarray, M: float, n: int):
    """
    Solve the LP that maximises sum(g) under a·g ≤ M.
    Returns a feasible g normalised to unit sum, or None if the LP fails.
    """
    # Build Toeplitz matrix A where A[t,i] = a[t-i] if 0≤t-i<n else 0
    col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    row = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
    A = scipy.linalg.toeplitz(col, row)[: 2 * n - 1, :n]  # shape (2n‑1, n)

    c = -np.ones(n)            # maximise sum(g)  → minimise -sum(g)
    b = np.full(2 * n - 1, M)

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=(0, None),
        method="highs",
    )
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    return g / s                # unit‑sum feasible direction


def _subgrad_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    best_seq: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    Stops when the deadline is reached or when no improvement occurs for a few seconds.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # stay on simplex

    fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    eta = 0.6
    last_improve = time.time()
    max_iter = 200_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-9:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] new best {val:.6f}")

        # active set (where max is attained)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (mirrored to the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if stalled
        if time.time() - last_improve > 5.0:
            break

        # slowly decay eta to fine‑tune
        eta *= 0.9999

    return best_seq, best_val

    best_seq, best_val = _subgrad_refine(

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")

    while time.time() < deadline - 5.0 and lp_steps < max_lp_steps:
        conv = np.convolve(a, a)
        M = conv.max()
        g = _lp_update(a, M, n)
        if g is None:
            print("LP solve failed – leaving LP phase")
            break

        # Mix a and g (theory guarantees non‑increase of M)
        t = 0.5
        while t > 1e-6:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()
            conv_new = np.convolve(a_new, a_new)
            M_new = conv_new.max()
            if M_new <= M * (1.0 + 1e-12):
                break
            t *= 0.5
        if t <= 1e-6:
            # No beneficial mixing factor found – stop LP phase
            break

        a = a_new
        lp_steps += 1

        # The objective (sum(a)=1) is 2·n·M_new
        val = 2.0 * n * M_new
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[LP] step {lp_steps} new best {best_val:.6f}")

        # Stop if we are essentially at the trivial lower bound
        if M_new - 1.0 / (2 * n - 1) < 1e-6:
            break

    # ------------------------------------------------------------------
    # Sub‑gradient refinement on the exact objective
    best_seq, best_val = _subgrad_refine(
        a,
        n,
        deadline,
        best_seq,
        best_val,
    )

    # ------------------------------------------------------------------
    # Upscale to a larger grid if time permits and run a short refinement
    if time.time() < deadline - 30.0:
        n_big = 4096
        x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
        best_seq_big, best_val_big = _subgrad_refine(
            a_big,
            n_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_val_big < best_val:
            best_seq = best_seq_big
            best_val = best_val_big
            print(f"[Upscale] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final clipping (the evaluator also clamps values)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
