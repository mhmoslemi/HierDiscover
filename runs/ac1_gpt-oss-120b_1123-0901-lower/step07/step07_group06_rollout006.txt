"""
Propose a low‑evaluation step‑function (≈1000 s budget).

Algorithm
---------
1. Build a very diverse pool of candidate vectors (uniform, Dirichlet,
   monotone exponential / power‑law, asymmetric Beta‑shaped,
   mixtures with a uniform component, random permutations of a warm‑start).
2. Quickly evaluate all candidates with an FFT‑based convolution and keep
   the top‑K (K≈12) for deep optimisation.
3. For each start run a three‑phase optimiser:
   * Adam on a smooth‑max surrogate (log‑sum‑exp) – gives a fast global
     move and quickly raises the temperature.
   * Exponentiated‑gradient sub‑gradient steps on the exact max‑convolution
     – pushes the active constraints to become tighter.
   * Projected‑gradient refinement on the true objective with Euclidean
     simplex projection.
4. After the deep optimisation, perform a cheap swap‑hill‑climbing
   (random pairwise swaps) while time permits.
5. If a few seconds remain, upscale the current best vector to a larger
   grid (n=8192) by linear interpolation and repeat the deep optimiser.

All steps respect the 1000 s deadline and keep the best sequence found.
The final sequence is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector with non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation (identical to evaluate_sequence for sum=1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # work in log‑space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5               # temperature
    lam_factor = 1.03
    lam_cap = 1e8

    last_improve = time.time()
    it = 0
    max_iter = 500_000

    while time.time() < deadline and it < max_iter:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # evaluate surrogate (same objective, we just need max for weights)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient steps on the true max‑convolution."""
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    last_improve = time.time()
    max_iter = 200_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 5.0:
            break

        eta *= 0.9999

    return best_seq, best_val


# ----------------------------------------------------------------------
def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact objective."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.2
    last_improve = time.time()
    max_iter = 150_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            break

        lr *= 0.99998

    return best_seq, best_val


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Full 3‑phase optimisation for a single start."""
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    best_seq, best_val = _projected_gradient_refine(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a rich, *asymmetric* set of candidate vectors.
    All candidates are normalised to sum≈1 and contain only non‑negative entries.
    """
    cand = []
    i = np.arange(n, dtype=np.float64)

    # warm‑start and a few permutations
    if warm_start is not None:
        cand.append(warm_start.copy())
        for _ in range(3):
            cand.append(np.random.permutation(warm_start))

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    # Dirichlet draws (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # monotone exponential families (increasing & decreasing)
    for alpha in np.logspace(-3, 0, num=12):
        inc = np.exp(alpha * i)
        inc /= inc.sum()
        dec = np.exp(alpha * (n - 1 - i))
        dec /= dec.sum()
        for shape in (inc, dec):
            cand.append(shape)
            for w in (0.2, 0.5, 0.8):
                cand.append(w * shape + (1.0 - w) * np.full(n, 1.0 / n))

    # power‑law families (increasing & decreasing)
    for p in np.linspace(0.2, 5.0, num=10):
        inc = (i + 1) ** p
        inc /= inc.sum()
        dec = (n - i) ** p
        dec /= dec.sum()
        for shape in (inc, dec):
            cand.append(shape)
            for w in (0.2, 0.5, 0.8):
                cand.append(w * shape + (1.0 - w) * np.full(n, 1.0 / n))

    # asymmetric Beta‑like families (alpha!=beta)
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        for beta in (0.2, 0.5, 1.0, 2.0, 5.0):
            shape = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)
            shape = np.maximum(shape, 0.0)
            s = shape.sum()
            if s > 0:
                shape /= s
                cand.append(shape)

    # simple triangular / quadratic (still asymmetric when shifted)
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand.append(quad)

    # random permutations of a uniform vector (just to get sanity checks)
    base_uniform = np.full(n, 1.0 / n)
    for _ in range(5):
        cand.append(np.random.permutation(base_uniform))

    return cand


# ----------------------------------------------------------------------
def _swap_hill_climb(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    current_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: random pairwise swaps.
    Keeps improvements on the exact max‑convolution objective.
    """
    best_seq = a.copy()
    best_val = current_best_val
    attempts = 0
    while time.time() < deadline - 0.05:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # swap a[i] ↔ a[j]
        cand = best_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]

        conv = _full_convolution(cand, conv_fft_len)
        val = float(2.0 * n * conv.max() / (cand.sum() ** 2))
        if val < best_val - 1e-12:
            best_seq = cand
            best_val = val
            print(f"[Swap] new best {best_val:.6f}")
        attempts += 1
        if attempts % 5000 == 0:
            # give a tiny pause to let the timer be checked more often
            if time.time() >= deadline - 0.05:
                break
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – searches for a step‑function with minimal evaluation score.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Base resolution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a very large pool of diverse candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    # Quick evaluation for each candidate
    cand_vals = []
    for cand in all_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-8:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # Keep the most promising starts
    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of each promising start
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n--- Deep optimisation on start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – cheap swap‑based local search on the current best
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\n--- Swap‑hill‑climbing local search ---")
        best_seq, best_val = _swap_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 4 – upscale to a larger grid if time permits
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("\n--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # keep larger grid if it succeeded
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # --------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Ensure values respect the evaluator limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
