"""
Propose a step function with a low evaluation score.

Algorithm overview
------------------
1. **Preparation** – Choose a working grid size `n` (4096 by default,
   upgraded to 8192 if time permits).  All convolutions are performed
   with a real FFT of the next power‑of‑two length for speed.

2. **Diverse seeding** – Build a large pool of *asymmetric* and
   *symmetric* candidates:
   * uniform,
   * random Dirichlet draws (many concentration levels),
   * left‑/right‑exponential and power‑law families,
   * mixtures of the two,
   * beta‑shapes (including arcsine), raised‑cosine / sine‑power
   * and a few random monotone decreasing vectors.
   Every candidate is projected onto the probability simplex
   (`sum = 1`) and its objective value is obtained by a single FFT‑based
   convolution.

3. **Gradient‑based refinement** – The best `K` seeds are refined
   independently by two phases:
   * **Adam on a smooth‑max surrogate** – the maximum of the
     convolution is approximated by a log‑sum‑exp weighted average.
     The parameters are the logits of a soft‑max representation,
     guaranteeing non‑negativity and unit sum.
   * **Exact sub‑gradient phase** – after the temperature has been
     raised and the surrogate is close to the true max, an exponentiated‑gradient
     update based on the true active set of the convolution is performed.

   Both phases run until the global deadline is reached; the best
   sequence seen across all runs is retained.

4. **Scale up** – If a few seconds remain the current best vector is
   interpolated to `n = 8192` and the two‑phase optimisation is repeated.

5. **Local hill‑climbing** – A final cheap random‑perturbation loop
   attempts tiny mass transfers between random coordinates and accepts
   only improving moves.  This often squeezes a few extra digits out
   of the objective.

The whole routine respects the 1000 s time limit (a 0.5 s safety margin)
and returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities ---------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """2·n·max(a*a) / (sum(a)**2); a is assumed to sum to 1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

# Optimisation primitives -----------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of `max(conv)`.
    Parameters
    ----------
    a0 : np.ndarray
        Starting probability vector (sum≈1).
    n, conv_fft_len, corr_fft_len : int
        Grid and FFT lengths.
    deadline : float
        Absolute time after which the loop must stop.
    best_seq, best_val : current global best (may be None / inf).
    Returns
    -------
    tuple(best_seq, best_val) seen during this run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    # Adam hyper‑parameters
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    temp = 0.4                # initial smooth‑max temperature
    temp_mul = 1.02
    temp_cap = 1e6

    it = 0
    max_iter = 400_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # soft‑max re‑parameterisation (guarantees positivity and sum=1)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # true objective (used for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        val = float(2.0 * n * conv.max())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -------------------------
        max_c = conv.max()
        w = np.exp(temp * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (w.r.t logits) ----------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # chain rule for soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule ------------------------------------
        temp = min(temp * temp_mul, temp_cap)

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_seq: np.ndarray | None,
    init_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent using exponentiated‑gradient updates.
    """
    a = init_seq if init_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_val
    it = 0
    max_iter = 40_000
    last_improve = time.time()
    eta0 = 0.6

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:05d} new best {val:.6f}")

        # active set where the convolution reaches its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient step (keeps a ≥ 0, sum = 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:               # numerical safety – re‑sample if everything collapses
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    One full optimisation run: Adam (smooth‑max) → sub‑gradient refinement.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _make_exponential(n: int, alpha: float, side: str = "left") -> np.ndarray:
    """Exponential shape; side='left' decays from index 0, 'right' from the opposite."""
    i = np.arange(n, dtype=np.float64)
    if side == "left":
        arr = np.exp(-alpha * i)
    else:
        arr = np.exp(-alpha * (n - 1 - i))
    return arr / arr.sum()


def _make_powerlaw(n: int, p: float, side: str = "left") -> np.ndarray:
    """Power‑law (inverse) shape."""
    i = np.arange(n, dtype=np.float64) + 1.0
    if side == "left":
        arr = i ** (-p)
    else:
        arr = (n - i + 1.0) ** (-p)
    return arr / arr.sum()


def _make_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta‑shape a_i ∝ (i+1)^{α‑1} (n‑i)^{α‑1}."""
    i = np.arange(n, dtype=np.float64)
    shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
    shape = np.maximum(shape, 0.0)
    return shape / shape.sum()


def _make_raised_cosine(n: int, p: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.sin(np.pi * (i + 0.5) / n) ** p
    shape = np.maximum(shape, 0.0)
    return shape / shape.sum()


def _make_random_monotone(n: int, breaks: int = 4) -> np.ndarray:
    """Random decreasing piecewise‑linear curve."""
    # generate break positions
    pts = np.sort(np.random.choice(n, size=breaks, replace=False))
    values = np.random.rand(breaks) + 0.1   # ensure positives
    values = np.sort(values)[::-1]        # decreasing
    # linear interpolation on the whole grid
    grid = np.arange(n)
    shape = np.interp(grid, pts, values, left=values[0], right=values[-1])
    return shape / shape.sum()


def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Collect a rich (asymmetric + symmetric) pool of seeds."""
    candidates: list[np.ndarray] = []

    # ----- warm‑start & circular shifts ---------------------------------
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        # a few non‑trivial shifts (shifted asymmetric shapes can be useful)
        for shift in (n // 5, n // 2, 3 * n // 5):
            candidates.append(np.roll(ws, shift))

    # ----- uniform baseline ---------------------------------------------
    candidates.append(np.full(n, 1.0 / n))

    # ----- random Dirichlet draws ----------------------------------------
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # ----- left / right exponential families -----------------------------
    for alpha in np.logspace(-3, 0, num=12):         # 0.001 … 1
        left = _make_exponential(n, alpha, "left")
        right = _make_exponential(n, alpha, "right")
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * left + (1.0 - w) * right)

    # ----- left / right power‑law families --------------------------------
    for p in np.linspace(0.2, 3.0, num=12):
        left = _make_powerlaw(n, p, "left")
        right = _make_powerlaw(n, p, "right")
        for w in (0.0, 0.4, 0.8, 1.0):
            candidates.append(w * left + (1.0 - w) * right)

    # ----- symmetric beta shapes (including arcsine) ----------------------
    for alpha in (0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0):
        candidates.append(_make_beta(n, alpha))

    # ----- raised‑cosine / sine‑power families ---------------------------
    for p in (1, 2, 4, 6, 8):
        candidates.append(_make_raised_cosine(n, p))

    # ----- random monotone decreasing vectors ----------------------------
    for _ in range(8):
        candidates.append(_make_random_monotone(n, breaks=np.random.randint(3, 7)))

    return candidates

# Final hill‑climbing refinement -----------------------------------------
def _local_hill_climb(a: np.ndarray, n: int, conv_fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    """Tiny random mass‑transfer moves that accept only improvements."""
    best = a.copy()
    conv = _full_convolution(best, conv_fft_len)
    best_val = float(2.0 * n * conv.max())
    step_scale = 0.25          # up to 25 % of the smaller mass involved
    attempts = 0
    max_attempts = 10_000

    while time.time() < deadline and attempts < max_attempts:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # maximal transfer respecting positivity
        max_transfer = min(best[i], best[j])
        if max_transfer < 1e-14:
            continue
        delta = (np.random.rand() - 0.5) * step_scale * max_transfer
        # enforce non‑negativity explicitly
        if delta > best[j]:
            delta = best[j]
        if -delta > best[i]:
            delta = -best[i]
        if abs(delta) < 1e-16:
            continue

        # propose move
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        conv_cand = _full_convolution(cand, conv_fft_len)
        val_cand = float(2.0 * n * conv_cand.max())

        if val_cand < best_val:
            best = cand
            conv = conv_cand
            best_val = val_cand
        attempts += 1

    return best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises ``evaluate_sequence`` and returns it as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Base grid size ------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2. Warm‑start (if a previous result is available) -----------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # 3. Broad seeding --------------------------------------------------
    raw_candidates = _generate_parametric_candidates(n, ws)

    evaluated = []
    for idx, cand in enumerate(raw_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:          # re‑sample if the vector collapsed numerically
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    evaluated.sort(key=lambda x: x[0])
    top_k = min(20, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: retained {top_k} promising seeds (n={n})")

    # ------------------------------------------------------------------
    # 4. Deep optimisation of the seeds --------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"=== Deep optimisation start #{start_id + 1}/{len(top_candidates)} ===")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # 5. Upscale to a larger grid when time permits --------------------
    time_left = deadline - time.time()
    if time_left > 40.0:               # need a few seconds for the upscale run
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Optimising on enlarged grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # update bookkeeping for possible final local refinement
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # 6. Final cheap hill‑climbing refinement ----------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("--- Local hill‑climb refinement ---")
        best_seq, best_val = _local_hill_climb(best_seq, n, conv_fft_len, deadline)

    # ------------------------------------------------------------------
    # 7. Fallback (should never be needed) -------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
