"""
Propose a low‑evaluation step‑function using a richer parametric pool,
smoothed Adam‑style mirror descent and many short random restarts.

Main ideas
----------
1. Build a diverse set of candidates (uniform, Dirichlet, edge‑exponential,
   edge‑power‑law, Beta‑U‑shape, half‑sinusoid, Gaussian edges).  Parameter
   sweeps are finer than in the reference code.
2. Evaluate every candidate with a cheap FFT‑based convolution and keep the
   best few.
3. For each selected start run a mirror‑descent optimiser that minimises a
   smoothed version of the max‑convolution:
      gβ(c)= (1/β)·log∑exp(β·c_k) .
   The gradient of gβ is a weighted cross‑correlation; we apply an Adam
   update to the logarithms of the variables (exponentiated‑gradient keeps
   non‑negativity) and renormalise to sum = 1 after each step.
   β starts small (≈1) and grows geometrically up to a few hundred, allowing
   a smooth descent that eventually focuses on the true maximum.
4. When progress stalls we inject a small random jitter and continue.
5. While time permits we launch many short Adam runs from fresh random
   Dirichlet seeds (different concentration parameters) – a cheap way to
   explore many basins.
6. The best vector is finally up‑sampled to a larger grid (n = 8192) and
   refined briefly if time remains.

The whole routine respects the 1000 s budget, prints occasional progress
messages and returns the best sequence as a plain Python list.
All values are clipped to the range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv) assuming the vector is already normalised (∑a=1)."""
    return float(2.0 * n * conv.max())


def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a large, diverse pool of candidate vectors."""
    cand = []

    # --- warm start & simple rotations ---------------------------------
    if warm_start is not None:
        warm = np.asarray(warm_start, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            cand.append(warm.copy())
            for shift in (n // 4, n // 2, 3 * n // 4):
                cand.append(np.roll(warm, shift))

    # --- simple baselines ------------------------------------------------
    cand.append(np.full(n, 1.0 / n))                     # uniform
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):   # Dirichlet draws
        cand.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # --- edge‑exponential (α on a log‑scale, plus uniform mixture) ---
    for alpha in np.logspace(-3, 1.5, num=24):           # 0.001 … 31.6
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            cand.append(shape)

    # --- edge‑power‑law (p from mild to very steep) -------------------
    for p in np.linspace(0.1, 10.0, num=30):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            cand.append(shape)

    # --- symmetric Beta‑U‑shape (p<1 gives strong edges) -------------
    for p in np.linspace(0.05, 0.9, num=15):
        # density ∝ x^{p‑1}(1‑x)^{p‑1}
        x = (i + 0.5) / n
        shape = (x ** (p - 1) * (1.0 - x) ** (p - 1))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)

    s = np.sin(np.pi * (i + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

def _adam_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Mirror‑descent with Adam on a smoothed max‑convolution.
    Returns the best vector/value observed before the deadline.
    """
    eps = 1e-12
    a = np.maximum(a0, eps)
    a /= a.sum()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eta = 0.1
    eps_adam = 1e-8

    # smoothing parameter for the max
    beta = 1.0
    beta_inc = 1.02
    max_beta = 500.0

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # convolution & objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:07d}] new best {val:.6f}")

        # smoothed max weights (log‑sum‑exp trick)
        shifted = conv - max_c                 # ≤0
        w = np.exp(beta * shifted)             # ≈ soft‑max
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to hard max if all underflowed
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of the smooth objective
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # shape (n,)

        # Adam on log‑variables (mirror‑descent on simplex)
        grad_log = a * grad_a
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        log_a = np.log(a + eps) - eta * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = np.exp(log_a)
        total = a.sum()
        if total == 0 or np.isnan(total):
            a = np.random.dirichlet(np.ones(n))
            total = a.sum()
        a = a / total

        # increase smoothing
        beta = min(max_beta, beta * beta_inc)

        # occasional random shake if we stagnate
        if time.time() - last_improve > 8.0:
            perturb = (np.random.rand(n) - 0.5) * 0.04   # log‑space jitter
            log_a = np.log(a + eps) + perturb
            a = np.exp(log_a)
            a = a / a.sum()
            last_improve = time.time()

    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Entry point – searches for a step‑function with minimal evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Primary grid (medium size – good trade‑off between quality and speed)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    warm_start = globals().get("height_sequence_1")   # may be None
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)

    # ------------------------------------------------------------------
    # Stage 1 – generate a very broad set of seeds
    candidates = _generate_parametric_candidates(n, warm_start)

    # add a few extra random Dirichlet samples with varied concentrations
    for conc in (0.005, 0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(30):
            candidates.append(np.random.dirichlet(np.full(n, conc)))

    # ------------------------------------------------------------------
    # Fast evaluation of all seeds – keep the best few
    scored = []
    for vec in candidates:
        if time.time() >= deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s < 0.01:
            vec = np.random.dirichlet(np.ones(n))
            s = vec.sum()
        vec = vec / s
        conv = _full_convolution(vec, conv_fft_len)
        val = _objective_from_conv(conv, n)
        scored.append((val, vec))

    if not scored:                     # very unlikely, but be safe
        best_seq = np.full(n, 1.0 / n)
        return best_seq.tolist()

    scored.sort(key=lambda x: x[0])
    top_k = 8
    starts = [v for _, v in scored[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation from the top seeds
    for idx, start_vec in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"--- Adam optimisation from start #{idx + 1}/{len(starts)}")
        best_seq, best_val = _adam_descent(
            start_vec, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # Stage 3 – many short random restarts while time permits
    while time.time() < deadline - 5.0:
        # random Dirichlet with random concentration
        conc = np.random.uniform(0.01, 2.0)
        rnd = np.random.dirichlet(np.full(n, conc))
        rnd = rnd / rnd.sum()
        best_seq, best_val = _adam_descent(
            rnd, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # Stage 4 – upscale to a finer grid (n = 8192) if we still have time
    if best_seq is not None and (deadline - time.time()) > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("*** Refinement on larger grid (n=8192)")
        best_seq, best_val = _adam_descent(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq, best_val
        )
        if best_seq is not None:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded (should never happen)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096)

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best value ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
