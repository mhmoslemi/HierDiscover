"""
Hybrid anti‑pair + linear‑programming + gradient optimisation for the
step‑function bound.  The algorithm proceeds in three stages:
1.  Generate a diverse set of warm‑starts (uniform, Dirichlet, anti‑pair,
    edge‑focused) on a modest size n_small (≈2 k).  Keep the best.
2.  Iteratively solve a small linear program:
        max   sum(g)
        s.t.  a * g ≤ M   (convolution bound of the current a)
              g ≥ 0
    where M = max(a*a).  Scaling g back to sum = 1 shrinks the bound by
    1/ sum(g).  The loop stops when no improvement is possible or time runs
    out.
3.  Upsample the improved small solution to the target length n_big
    (4096) and finish the search with the Adam → sub‑gradient → simulated‑annealing
    pipeline from the previous state‑of‑the‑art.  All phases respect the
    global time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# FFT utilities (same as the reference implementation)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

# ----------------------------------------------------------------------
# Exact objective (used for bookkeeping, not for the surrogate)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

def _warm_start_small(n: int):
    """Extract a (down‑sampled) warm start from the global variable if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        # down‑sample by linear interpolation
        x_big = np.linspace(0.0, 1.0, warm.shape[0], endpoint=False)
        x_small = np.linspace(0.0, 1.0, n, endpoint=False)
        warm = np.interp(x_small, x_big, warm)
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Arrange the entries so that big values sit opposite small ones."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]           # median in the centre
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet vector and apply the anti‑pair layout."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape – larger at the edges, smaller in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape

# Linear‑programming refinement ----------------------------------------
def _build_convolution_matrix(a: np.ndarray):
    """
    Build the matrix A with rows corresponding to convolution constraints
    a * g ≤ M.  Shape: (2·n−1) × n, dense with n² entries (stored as CSR).
    """
    n = a.shape[0]
    # each diagonal offset d (0 ≤ d < n) contains a[d] repeated n times
    diags_data = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)                # 0,1,…,n-1
    A = diags(diags_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_refine(a0: np.ndarray,
               deadline: float,
               max_iters: int = 20,
               eps_improve: float = 1e-9):
    """
    Repeatedly solve the LP
        max   sum(g)
        s.t.  a * g ≤ M
              g ≥ 0
    where M = max(a*a).  Each successful iteration replaces ``a`` by
    g/∑g, which reduces M by a factor 1/∑g.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    it = 0
    n = a.shape[0]

    while it < max_iters and time.time() < deadline:
        # compute current bound M
        conv = _full_convolution(a, _next_pow2(2 * n - 1))
        M = conv.max()

        # build convolution matrix A (a * g ≤ M)
        A = _build_convolution_matrix(a)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # Linear program (maximise sum(g) → minimise -sum(g))
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=b,
                      bounds=[(0, None)] * n,
                      method="highs")
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + eps_improve:
            print(f"[LP] iteration {it} no improvement (sum_g = {sum_g:.6f})")
            break

        # scale back to the simplex
        a_new = g / sum_g
        val_new = _objective(a_new)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            print(f"[LP] iter={it} improved to {best_val:.6f} (sum_g={sum_g:.6f})")
        else:
            # even if the exact objective did not improve we keep the new
            # point because the bound M has certainly decreased.
            print(f"[LP] iter={it} reduced bound (sum_g={sum_g:.6f})")
        a = a_new
        it += 1

    return best_a, best_val

# Optimisation primitives – same as the reference implementation
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.05,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c  # sum(a)=1 throughout
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
      - swap two entries or transfer a fraction of mass.
      - accept iff improvement or with a temperature‑dependent probability.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

"""
Hybrid anti‑pair + linear‑programming + gradient optimisation for the
step‑function bound.  The algorithm proceeds in three stages:

1.  Build a diverse pool of warm‑starts (uniform, Dirichlet, anti‑pair,
    edge‑focused) on a moderate grid (≈2048 points).  Keep the best.

2.  Repeatedly solve a small linear programme:
        max   Σ g_j
        s.t.  a ★ g ≤ M   (convolution with the current a)
              g_j ≥ 0
    where M = max(a ★ a).  Scaling g back to Σ g = 1 shrinks the bound
    by a factor 1/Σ g.  The loop stops when no further improvement is
    possible or the time budget expires.

3.  Upsample the refined small solution to the target length (4096) and
    finish with the Adam → exact sub‑gradient → stochastic local‑search
    pipeline that has proved effective in earlier runs.  All phases respect
    the global time budget supplied to ``propose_candidate``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

# ----------------------------------------------------------------------
# Exact objective (used for bookkeeping)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

# ----------------------------------------------------------------------
# Candidate generation
def _warm_start_small(n: int):
    """Down‑sample a global warm start (if available) to size n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        # Linear interpolation to the smaller size
        x_big = np.linspace(0.0, 1.0, warm.shape[0], endpoint=False)
        x_small = np.linspace(0.0, 1.0, n, endpoint=False)
        warm = np.interp(x_small, x_big, warm)
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Large entries opposite small entries."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half right
    if n % 2 == 1:
        a[half] = sorted_vals[half]           # median in centre
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape

# ----------------------------------------------------------------------
# Linear‑programming utilities
def _build_convolution_matrix(a: np.ndarray):
    """
    Build the matrix A (size (2n‑1)×n) representing the linear constraints
        (a ★ g)_k = Σ_i a_i·g_{k‑i} ≤ M .
    Each diagonal offset d (0 ≤ d < n) is filled with the constant a[d].
    """
    n = a.shape[0]
    diags_data = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)                # 0,1,…,n‑1
    A = diags(diags_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_refine(a0: np.ndarray,
               deadline: float,
               max_iters: int = 20,
               eps_improve: float = 1e-9):
    """
    Repeatedly solve the LP
        max   Σ g_j
        s.t.  a ★ g ≤ M
              g ≥ 0
    where M = max(a ★ a).  Scaling the LP solution back to Σ g = 1 shrinks
    the bound by the factor 1/Σ g.  The routine returns the best vector
    encountered and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    n = a.shape[0]
    it = 0

    while it < max_iters and time.time() < deadline:
        # Current convolution bound
        conv = _full_convolution(a, _next_pow2(2 * n - 1))
        M = conv.max()

        # Build constraint matrix for a ★ g ≤ M
        A = _build_convolution_matrix(a)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # Linear program: maximise Σ g → minimise -Σ g
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=b,
                      bounds=[(0, None)] * n,
                      method="highs")
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + eps_improve:
            print(f"[LP] iteration {it} no improvement (sum_g={sum_g:.6f})")
            break

        a_new = g / sum_g
        val_new = _objective(a_new)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            print(f"[LP] iter={it} improved to {best_val:.6f} (sum_g={sum_g:.6f})")
        else:
            print(f"[LP] iter={it} reduced bound (sum_g={sum_g:.6f})")
        a = a_new
        it += 1

    return best_a, best_val

# Optimisation primitives (Adam → sub‑gradient → local search)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.05,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam optimisation of a smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style hill‑climbing."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # ≤40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
    ``F(a)=2·n·max(a∗a)/(sum(a)²)``.
    The routine combines:
        • diverse anti‑pair Dirichlet warm‑starts,
        • an LP‑based refinement that reduces the ℓ∞ convolution bound,
        • Adam → exact sub‑gradient → stochastic local search,
        • optional up‑sampling to the target length (4096).
    It respects the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem sizes
    n_small = 2048
    n_big = 4096

    # --------------------------------------------------------------
    # Stage 1 – generate starters on the small grid
    cand_list = []

    warm = _warm_start_small(n_small)
    if warm is not None:
        cand_list.append(warm.copy())
        for shift in (n_small // 4, n_small // 2, 3 * n_small // 4):
            cand_list.append(np.roll(warm, shift))

    cand_list.append(_uniform_candidate(n_small))
    for alpha in (0.3, 0.7, 1.0, 2.0):
        cand_list.append(_random_dirichlet_candidate(n_small, alpha))
    for alpha in (0.3, 0.7, 1.0):
        cand_list.append(_anti_pair_candidate(n_small, alpha))
    for gamma in (0.2, 0.5, 1.0):
        cand_list.append(_gaussian_edges_candidate(n_small, gamma))
    cand_list.append(_triangle_candidate(n_small))

    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(cand_list):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n_small))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # Stage 2 – LP refinement on the small grid
    lp_deadline = min(deadline - 150.0, time.time() + 200.0)
    best_seq, best_val = _lp_refine(best_seq, lp_deadline, max_iters=15)

    # --------------------------------------------------------------
    # Stage 3 – brief Adam / sub‑gradient on the small grid (optional)
    if time.time() < deadline - 140.0:
        small_deadline = min(deadline - 130.0, time.time() + 60.0)
        a_ad, v_ad = _adam_optimize(
            best_seq,
            n_small,
            _next_pow2(2 * n_small - 1),
            _next_pow2(3 * n_small - 2),
            small_deadline,
            max_iters=200_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if v_ad < best_val:
            best_val = v_ad
            best_seq = a_ad.copy()
            print(f"[Adam small] improved to {v_ad:.6f}")

        if time.time() < deadline - 120.0:
            sub_deadline = min(deadline - 110.0, time.time() + 80.0)
            a_sub, v_sub = _subgradient_optimize(
                best_seq,
                n_small,
                _next_pow2(2 * n_small - 1),
                _next_pow2(3 * n_small - 2),
                sub_deadline,
                max_iters=200_000,
                eta0=0.7,
            )
            if v_sub < best_val:
                best_val = v_sub
                best_seq = a_sub.copy()
                print(f"[Subgrad small] improved to {v_sub:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – up‑sample to target length and heavy‑weight refinement
    if time.time() < deadline - 30.0:
        x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()
    else:
        a_big = np.pad(best_seq, (0, n_big - n_small), mode="constant")
        a_big = a_big[:n_big]
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

    best_seq_big = a_big.copy()
    best_val_big = _objective(best_seq_big)

    # --------------------------------------------------------------
    # Stage 5 – Adam on the big grid
    if time.time() < deadline - 20.0:
        adam_deadline = min(deadline - 10.0, time.time() + 180.0)
        a_big_adam, v_big_adam = _adam_optimize(
            best_seq_big,
            n_big,
            _next_pow2(2 * n_big - 1),
            _next_pow2(3 * n_big - 2),
            adam_deadline,
            max_iters=300_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if v_big_adam < best_val_big:
            best_val_big = v_big_adam
            best_seq_big = a_big_adam.copy()
            print(f"[Adam big] improved to {v_big_adam:.6f}")

    # --------------------------------------------------------------
    # Stage 6 – exact sub‑gradient on the big grid
    if time.time() < deadline - 5.0:
        sub_deadline = min(deadline, time.time() + 200.0)
        a_big_sub, v_big_sub = _subgradient_optimize(
            best_seq_big,
            n_big,
            _next_pow2(2 * n_big - 1),
            _next_pow2(3 * n_big - 2),
            sub_deadline,
            max_iters=300_000,
            eta0=0.7,
        )
        if v_big_sub < best_val_big:
            best_val_big = v_big_sub
            best_seq_big = a_big_sub.copy()
            print(f"[Subgrad big] improved to {v_big_sub:.6f}")

    # --------------------------------------------------------------
    # Stage 7 – final stochastic local search
    if time.time() < deadline - 2.0:
        local_deadline = deadline - 2.0
        a_big_loc, v_big_loc = _local_random_search(
            best_seq_big,
            n_big,
            _next_pow2(2 * n_big - 1),
            local_deadline,
            max_moves=150_000,
        )
        if v_big_loc < best_val_big:
            best_val_big = v_big_loc
            best_seq_big = a_big_loc.copy()
            print(f"[Local big] improved to {v_big_loc:.6f}")

    # --------------------------------------------------------------
    # Safeguard – always return a valid sequence
    if best_seq_big is None:
        best_seq_big = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    best_seq_big = np.clip(best_seq_big, 0.0, 1000.0)

    print(f"Final best value {best_val_big:.6f} (length {len(best_seq_big)})")
    return best_seq_big.tolist()
