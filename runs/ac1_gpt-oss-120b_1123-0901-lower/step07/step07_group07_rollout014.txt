def cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def generate_candidates(n, warm_start=None):
    candidates = []
    i = np.arange(n, dtype=np.float64)
    # warm start and shifts
    if warm_start is not None:
        ws = np.array(warm_start, dtype=np.float64)
        if ws.shape[0] != n:
            # resize via interpolation
            x_old = np.linspace(0, 1, ws.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.clip(ws, 0, None)
        if ws.sum() > 0:
            ws = ws/ ws.sum()
            candidates.append(ws)
            for shift in (n//4, n//2, 3*n//4):
                candidates.append(np.roll(ws, shift))

    # uniform
    candidates.append(np.full(n, 1.0/n))

    # Dirichlet bunch
    for conc in np.logspace(-2, 1, num=20):  # 0.01 to 10
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # Edge exponential with mix
    for alpha in np.logspace(-2, 1, num=20):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n-1-i))
        edge /= edge.sum()
        for w in (0.0, 0.2, 0.5, 0.8, 1.0):
            candidates.append(w*edge + (1-w)/n)

    # Edge power law
    for p in np.linspace(0.1, 5.0, num=20):
        edge = (i+1.0)**(-p) + (n-i)**(-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            candidates.append(w*edge + (1-w)/n)

    # Beta shapes
    for alpha in np.linspace(0.2, 3.0, num=20):
        shape = (i+1.0)**(alpha-1) * (n-i)**(alpha-1)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Raised cosine / sine power
    for p in (1,2,3,4,5):
        shape = np.sin(np.pi * (i + 0.5) / n)**p
        shape = np.maximum(shape,0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Gaussian edges
    for gamma in np.logspace(-2, 0, num=15):
        shape = np.exp(-gamma * (i**2)) + np.exp(-gamma * ((n-1-i)**2))
        shape /= shape.sum()
        candidates.append(shape)

    # arcsine shape
    shape = 1.0 / np.sqrt((i+0.5)*(n-i-0.5))
    shape /= shape.sum()
    candidates.append(shape)

    # Random local perturbations of some candidates
    # for each of first few candidates add Gaussian noise
    base = candidates[:10]
    for base_vec in base:
        for scale in (0.01, 0.05):
            pert = base_vec + scale * np.random.randn(n) * base_vec
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                pert = pert / s
                candidates.append(pert)

    return candidates

def adam_opt(a0, n, conv_fft_len, corr_fft_len, deadline, init_best_seq, init_best_val):
    logits = np.log(a0 + 1e-15) # shape n
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr = 0.03   # smaller than before for stability
    lam = 0.2    # temperature
    lam_factor = 1.02
    lam_cap = 1e6
    best_seq = None
    best_val = init_best_val
    if best_seq is None:
        best_seq = init_best_seq
    last_improve = time.time()
    it = 0
    max_iter = 300000  # keep high, break on deadline
    while time.time() < deadline and it < max_iter:
        # softmax to get a
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()
        # convolution
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            # optional print
            print(f"[{n}] Adam it={it} new best {val:.6f}")
        # smooth max weights
        # compute w in stable way
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient wrt a: 2 * cross correlation
        grad_a = 2.0 * cross_correlation(w, a, corr_fft_len)
        # gradient wrt logits (softmax)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it+1))
        v_hat = v / (1 - beta2 ** (it+1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)
        # jitter if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()
        it += 1
    return best_seq, best_val

def subgrad_opt(a0, n, conv_fft_len, corr_fft_len, deadline, init_best_val=float('inf')):
    a = np.copy(a0)
    # ensure non-negative and normalized
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_seq = a.copy()
    best_val = init_best_val
    if best_val == float('inf'):
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        best_val = 2.0 * n * max_c / (a.sum() ** 2)
    eta0 = 0.8
    it = 0
    max_iter = 300000
    last_improve = time.time()
    while time.time() < deadline and it < max_iter:
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it} new best {val:.6f}")
        # active set
        # tolerance could be relative to max_c
        atol = max(1e-12, 1e-9 * max_c)
        active = np.isclose(conv, max_c, atol=atol)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * cross_correlation(w, a, corr_fft_len)
        # step size decays with sqrt
        eta = eta0 / np.sqrt(it + 1)
        # exponentiated gradient step
        a = a * np.exp(-eta * grad_a)
        # renormalize
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        # early stop if no improvement for some seconds
        if time.time() - last_improve > 10.0:
            break
        it += 1
    return best_seq, best_val

def hill_climb(a0, n, conv_fft_len, deadline, init_best_val=float('inf')):
    a = a0.copy()
    best_seq = a.copy()
    best_val = init_best_val
    if best_val == float('inf'):
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        best_val = 2.0 * n * max_c / (a.sum() ** 2)
    it = 0
    max_iter = 2000000  # lots; break by deadline
    last_improve = time.time()
    while time.time() < deadline and it < max_iter:
        # randomly pick donor and recipient indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount to transfer relative to min of a[i], a[j]
        delta_max = min(a[i], a[j])
        # choose a random fraction of delta_max, up to 20%
        frac = np.random.rand() * 0.2
        delta = (np.random.rand() - 0.5) * 2 * frac * delta_max  # in [-frac*max, +frac*max]
        # apply transfer
        a[i] += delta
        a[j] -= delta
        # ensure non-negative
        if a[i] < 0:
            a[j] += a[i]  # a[i] negative, correct
            a[i] = 0.0
        if a[j] < 0:
            a[i] += a[j]
            a[j] = 0.0
        # evaluate
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
        else:
            # revert
            a[i] -= delta
            a[j] += delta
        it += 1
    return best_seq, best_val

"""
Propose a step‑function sequence that minimises the evaluator.

Algorithm
---------
1. **Rich parametric seeding** – 200‑odd symmetric shapes (uniform,
   Dirichlet draws, edge‑exponential, power‑law, beta, sine‑power,
   Gaussian‑edge, arcsine) together with a few perturbed copies are
   generated and normalised.  The top ≈12 candidates (by a fast FFT‑based
   objective) are kept.

2. **Long Adam optimisation** – each seed is refined by Adam on a
   log‑sum‑exp (smooth‑max) surrogate of the max‑convolution.
   The temperature (λ) is increased exponentially, the learning‑rate
   decays automatically via Adam’s moments, and a jitter is added if
   the run stalls.

3. **Extensive sub‑gradient refinement** – an exponentiated‑gradient
   method uses the exact sub‑gradient of the max‑convolution.
   The step size decays as η₀/√t and the loop runs for up to several
   hundred thousand iterations (or until no improvement for 10 s).

4. **Upscaling** – if time permits the best vector is interpolated to a
   finer grid (n = 8192) and the whole optimisation (Adam + sub‑gradient)
   is repeated.

5. **Final hill‑climbing** – many tiny random mass‑transfer moves are
   attempted; a move is kept only if it improves the true objective.
   This fine‑tunes the solution beyond the gradient steps.

All steps respect the supplied time budget (default 1000 s).
The returned list is clipped to the evaluator’s allowed range (≥0, ≤1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) / (sum(a)**2)."""
    conv = full_convolution(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def adam_opt(a0: np.ndarray, n: int, conv_fft_len: int,
             corr_fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    a0 = np.maximum(a0, 0.0)
    a0_sum = a0.sum()
    if a0_sum < 1e-12:
        a0 = np.random.dirichlet(np.ones(n))
    else:
        a0 = a0 / a0_sum

    best_seq = a0.copy()
    best_val = objective(a0, n, conv_fft_len)

    logits = np.log(best_seq + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                     # a modest learning rate
    lam = 0.2                     # start temperature
    lam_factor = 1.02
    lam_cap = 1e6

    last_improve = time.time()
    it = 0
    max_iter = 300_000

    while time.time() < deadline and it < max_iter:
        # simplex via soft‑max
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()

        # convolution & real objective (for monitoring)
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * cross_correlation(w, a, corr_fft_len)

        # gradient w.r.t. logits (soft‑max chain rule)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule (smooth‑max becomes true max)
        lam = min(lam * lam_factor, lam_cap)

        # jitter if stalled
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def subgrad_opt(a0: np.ndarray, n: int, conv_fft_len: int,
                corr_fft_len: int, deadline: float,
                init_best_val: float = float("inf")) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_seq = a.copy()
    best_val = init_best_val if init_best_val != float("inf") else objective(a, n, conv_fft_len)

    eta0 = 0.8
    it = 0
    max_iter = 300_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:06d} new best {val:.6f}")

        # active set (indices attaining the maximum)
        atol = max(1e-12, 1e-9 * max_c)
        active = np.isclose(conv, max_c, atol=atol)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * cross_correlation(w, a, corr_fft_len)

        # step size – decays like 1/√t
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop if no improvement for a while
        if time.time() - last_improve > 10.0:
            break

        it += 1

    return best_seq, best_val


def hill_climb(a0: np.ndarray, n: int, conv_fft_len: int,
               deadline: float, init_best_val: float = float("inf")) -> tuple[np.ndarray, float]:
    """
    Simple random mass‑transfer hill climbing.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = init_best_val if init_best_val != float("inf") else objective(a, n, conv_fft_len)

    it = 0
    max_iter = 2_000_000          # large budget, will be cut by deadline
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # transfer up to 20 % of the smaller of a[i],a[j]
        delta_max = min(a[i], a[j])
        if delta_max <= 0:
            continue
        frac = 0.2 * np.random.rand()
        delta = (np.random.rand() - 0.5) * 2.0 * frac * delta_max

        # propose move
        a[i] += delta
        a[j] -= delta

        # keep non‑negative
        if a[i] < 0:
            a[j] += a[i]
            a[i] = 0.0
        if a[j] < 0:
            a[i] += a[j]
            a[j] = 0.0

        # evaluate
        val = objective(a, n, conv_fft_len)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
        else:
            # revert
            a[i] -= delta
            a[j] += delta

        it += 1

    return best_seq, best_val

def generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of symmetric candidate step functions."""
    candidates = []
    i = np.arange(n, dtype=np.float64)

    # warm‑start and its cyclic shifts
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, ws.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.clip(ws, 0.0, None)
        if ws.sum() > 0:
            ws = ws / ws.sum()
            candidates.append(ws)
            for shift in (n // 4, n // 2, 3 * n // 4):
                candidates.append(np.roll(ws, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet draws with a wide range of concentration parameters
    for conc in np.logspace(-2, 1, num=20):          # 0.01 … 10
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # Edge‑exponential families (with optional uniform mix)
    for alpha in np.logspace(-2, 1, num=20):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.2, 0.5, 0.8, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Edge‑power‑law families
    for p in np.linspace(0.1, 5.0, num=20):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Symmetric beta‑shapes (a_i ∝ (i+1)^{α‑1}(n‑i)^{α‑1})
    for alpha in np.linspace(0.2, 3.0, num=20):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Raised‑cosine / sine‑power families
    for p in (1, 2, 3, 4, 5):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Two‑Gaussian edge shapes
    for gamma in np.logspace(-2, 0, num=15):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # Small random perturbations of a few base candidates
    for base in candidates[:10]:
        for scale in (0.01, 0.05):
            pert = base + scale * np.random.randn(n) * base
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                pert = pert / s
                candidates.append(pert)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``. Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary grid size
    n = 4096
    conv_fft_len = next_pow2(2 * n - 1)
    corr_fft_len = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered vector if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
    else:
        warm = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool and keep the most promising starts
    pool = generate_candidates(n, warm)
    evaluated = []
    for cand in pool:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    seeds = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: selected {top_k} seeds (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of each seed
    for idx, start in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"--- Optimising seed {idx + 1}/{len(seeds)} ---")
        seq_adam, val_adam = adam_opt(start, n, conv_fft_len,
                                      corr_fft_len, deadline)
        seq_sub, val_sub = subgrad_opt(seq_adam, n,
                                       conv_fft_len, corr_fft_len,
                                       deadline, init_best_val=val_adam)

        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub.copy()
            print(f">>> New global best after seed {idx + 1}: {best_val:.6f}")

        # Small extra refinement from the current global best (if we still have time)
        if best_seq is not None and time.time() < deadline - 5.0:
            seq_extra, val_extra = adam_opt(best_seq, n, conv_fft_len,
                                            corr_fft_len, deadline)
            seq_extra, val_extra = subgrad_opt(seq_extra, n,
                                               conv_fft_len, corr_fft_len,
                                               deadline, init_best_val=val_extra)
            if val_extra < best_val:
                best_val = val_extra
                best_seq = seq_extra.copy()
                print(f">>> Global best improved by extra refinement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – upscale to a finer grid if sufficient time remains
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = next_pow2(2 * n_big - 1)
        corr_fft_len_big = next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Upscaling to n=8192 and further optimisation ---")
        seq_big, val_big = adam_opt(a_big, n_big, conv_fft_len_big,
                                    corr_fft_len_big, deadline)
        seq_big, val_big = subgrad_opt(seq_big, n_big,
                                       conv_fft_len_big, corr_fft_len_big,
                                       deadline, init_best_val=val_big)

        # Final random hill‑climb on the large grid
        if time.time() < deadline - 5.0:
            seq_big, val_big = hill_climb(seq_big, n_big,
                                          conv_fft_len_big, deadline,
                                          init_best_val=val_big)

        best_seq = seq_big
        best_val = val_big
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Stage 4 – final fine‑grained hill‑climbing on the current best
    if best_seq is not None and time.time() < deadline - 2.0:
        seq_final, val_final = hill_climb(best_seq, n,
                                          conv_fft_len, deadline,
                                          init_best_val=best_val)
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final
            print(f"Final hill‑climb improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback to uniform if nothing succeeded (should never happen)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)
        best_val = objective(best_seq, 4096, next_pow2(2 * 4096 - 1))

    # Clip to evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
