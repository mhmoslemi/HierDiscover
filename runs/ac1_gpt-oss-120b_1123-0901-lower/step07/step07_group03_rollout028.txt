"""
Propose a low‑value step function.

The search consists of many short, independent optimisation runs.
Each run starts from a randomly generated sensible seed
(uniform, Dirichlet, anti‑pair, Gaussian‑edge, triangular …),
then runs three phases:

1. Adam on a smooth‑max (log‑sum‑exp) surrogate.
2. Exact‑max sub‑gradient (exponentiated‑gradient) refinement.
3. Simple simulated‑annealing hill‑climbing (swap / mass transfer).

The phases respect a per‑run time budget.
All runs share the best vector found so far, which is finally
returned (clipped to the limits of ``evaluate_sequence``).

The code is deliberately lightweight: it works for any length,
but the default length 4096 gives a good trade‑off between
solution quality and runtime.  If time allows an extra upscale
to 8192 the best solution is interpolated and refined once more.
"""

import time
import math
import random
import numpy as np


# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, without security checks) ---------------------------

def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  a is assumed non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

# Candidate generators --------------------------------------------------

def _warm_start(n: int):
    """Reuse a previous best sequence if the global variable exists."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm = warm / warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Largest ↔ smallest ordering (anti‑pair).
    Input must be sorted descending and already normalised to sum 1.
    """
    n = len(vals)
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = vals[:half]                 # biggest half on the left
    a[-half:] = vals[-half:][::-1]         # smallest half on the right (reversed)
    if n % 2:
        a[half] = vals[half]               # middle element stays in the centre
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    vals_sorted = np.sort(vals)[::-1]      # descending
    return _anti_pair_arrangement(vals_sorted)


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    """Symmetric tent (peak in the centre)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _linear_ramp_candidate(n: int) -> np.ndarray:
    """Ramp up from 0 to a peak and back down (piecewise linear)."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    shape = 1.0 - np.abs(x - centre) / centre
    shape = np.clip(shape, 0, None)
    shape = shape / shape.sum()
    return shape


def _rotated_candidate(base_func, n):
    """Return a randomly rotated version of a base candidate."""
    a = base_func(n)
    shift = np.random.randint(n)
    return np.roll(a, shift)

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   lr: float = 0.07,
                   lam_start: float = 1.0,
                   lam_factor: float = 1.07,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate (log‑sum‑exp) value – we also monitor the exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        exact = 2.0 * n * max_c / (a.sum() ** 2)
        if exact < best_val:
            best_val = exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          eta0: float = 0.7,
                          max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {best_val:.6f}")

        # weights of the active (max‑attaining) indices
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (decaying)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing.
    Moves:  (i) swap two entries, (ii) transfer a fraction of mass.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01

    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                                          # transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % transfer
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < math.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val
