def _lbfgs_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iter=500, lam=30.0):
    logits0 = np.log(a0 + 1e-15)
    # define closure functions convert using outer variables

    def func_and_grad(logits):
        a = _softmax(logits)
        conv = _full_convolution(a, conv_fft_len)
        # compute stable weights
        max_c = conv.max()
        # compute w
        exp_term = np.exp(lam * (conv - max_c))
        Z = exp_term.sum()
        if Z == 0 or np.isnan(Z):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_term / Z
        # objective
        smooth = max_c + np.log(Z) / lam
        val = 2.0 * n * smooth
        # gradient wrt a
        grad_a = 4.0 * n * _cross_correlation(w, a, corr_fft_len)
        # chain rule to logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        return val, grad_log.astype(np.float64)

    # Use minimize with jac=True
    opt_res = minimize(
        fun=lambda logit: func_and_grad(logit)[0],
        x0=logits0,
        jac=lambda logit: func_and_grad(logit)[1],
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'disp': False}
    )
    # after minimize, compute final a and val
    logits_best = opt_res.x
    a_best = _softmax(logits_best)
    conv_best = _full_convolution(a_best, conv_fft_len)
    val_best = 2.0 * n * conv_best.max()
    return a_best, val_best

"""
Propose a non‑negative step‑function (vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The search proceeds in several stages that are deliberately cheap
but complementary:

* **Diversified warm‑starts** – Dirichlet draws, edge‑concentrated
  power‑law spikes, double‑exponential “tails”, sinusoidal bumps and a
  few deterministic spike patterns.  If a previous solution exists it
  is used as an additional seed.

* **Adam on a smooth‑max surrogate** – the max in the objective is
  replaced by a log‑sum‑exp with a temperature that is slowly increased.
  Adam on the logits of the soft‑max parameterisation keeps the iterates
  on the simplex.

* **Exponentiated‑gradient sub‑gradient phase** – after the warm‑up a
  true sub‑gradient (active‑set) update is applied with a 1/√t decaying
  step‑size.  Random shakes are injected when the method stalls.

* **L‑BFGS refinement** – a few quasi‑Newton steps on the same
  smooth‑max are performed (SciPy’s ``L-BFGS-B``) to accelerate
  convergence once the iterates are already near a good region.

* **Cheap local hill‑climbing** – random mass‑transfer moves that are
  kept only when they improve the exact objective.

* **Resolution up‑scaling** – if time permits the current best vector
  is interpolated to 8192 and then to 16384 points, repeating the
  Adam → sub‑gradient → L‑BFGS → local‑search pipeline on the finer
  grids.

All phases monitor the global time budget (default 1000 s) and the
routine always returns the best sequence found.  The implementation
uses only NumPy, SciPy and a single CPU core.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _softmax(logits: np.ndarray) -> np.ndarray:
    mx = logits.max()
    exp = np.exp(logits - mx)
    return exp / exp.sum()


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse set of start vectors."""
    cand = []

    # warm start + a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (uniform on the simplex)
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes (negative exponents → spikes at ends)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.5, -0.1, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # symmetric double‑exponential tails
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.15, 6.0, 9):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid (smooth bump)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # narrow twin‑Gaussian peaks at the two borders
    sigma = n / 12.0
    x = np.arange(n, dtype=np.float64)
    shape = np.exp(-((x) / sigma) ** 2) + np.exp(-((x - (n - 1)) / sigma) ** 2)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """A few “spike’’ candidates with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.linspace(0, n - 1, k).round().astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of the max‑convolution.
    Returns (best vector, exact objective value).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                     # modest learning rate
    lam = 0.5                     # start temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        a = _softmax(logits)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        exp_term = np.exp(lam * (conv - max_c))
        w_sum = exp_term.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_term / w_sum

        # gradient w.r.t a (ignoring the constant 2·n factor)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule to logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # tiny shake every few thousand steps if we are close to the deadline
        if it % 7000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase with decaying step size.
    Returns (best vector, exact objective value).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set – indices attaining the max (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update + renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # occasional random shake if we have been stuck for a while
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the exact objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move up to 40 % of the source mass
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 400,
    lam: float = 40.0,
) -> tuple[np.ndarray, float]:
    """
    A short L‑BFGS‑B refinement on the log‑sum‑exp (smooth‑max) surrogate.
    The returned value is the *exact* objective (using the true max).
    """
    logits0 = np.log(a0 + 1e-15)

    def fun_and_grad(logits):
        a = _softmax(logits)
        conv = _full_convolution(a, conv_fft_len)

        # smooth‑max (log‑sum‑exp)
        max_c = conv.max()
        exp_term = np.exp(lam * (conv - max_c))
        Z = exp_term.sum()
        if Z == 0 or np.isnan(Z):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_term / Z
        smooth = max_c + np.log(Z) / lam
        val = 2.0 * n * smooth

        # gradient w.r.t a of the *smooth* objective
        grad_a = 4.0 * n * _cross_correlation(w, a, corr_fft_len)

        # chain rule to logits (soft‑max)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        return val, grad_log.astype(np.float64)

    # SciPy requires separate callables for value and gradient when jac=True.
    def fun(logits):
        return fun_and_grad(logits)[0]

    def jac(logits):
        return fun_and_grad(logits)[1]

    res = minimize(
        fun=fun,
        x0=logits0,
        jac=jac,
        method="L-BFGS-B",
        options={"maxiter": max_iter, "disp": False},
    )
    logits_best = res.x
    a_best = _softmax(logits_best)
    conv_best = _full_convolution(a_best, conv_fft_len)
    val_best = 2.0 * n * conv_best.max()
    return a_best, val_best

            )
            if val_big_lb < best_val:
                best_val = val_big_lb
                best_seq = a_big_lb.copy()
                print(f"[Big{n_big} L‑BFGS] best {val_big_lb:.6f}")

        # final cheap hill‑climbing on the larger grid
        if time.time() < deadline - 5:
            a_big_loc, val_big_loc = _local_random_search(
                best_seq, n_big, conv_fft_len_big, deadline, max_moves=200_000
            )
            if val_big_loc < best_val:
                best_val = val_big_loc
                best_seq = a_big_loc.copy()
                print(f"[Big{n_big} Local] improved best {val_big_loc:.6f}")

    # --------------------------------------------------------------
    # Fallback – ensure a valid return value
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's safety limits (already non‑negative)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
