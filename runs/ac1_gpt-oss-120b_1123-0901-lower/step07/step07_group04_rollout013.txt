def _smooth_exponential_gradient_opt(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    a = np.maximum(a0,0)
    a = a / a.sum()

    eta = 1.0
    decay_rate = 0.9998
    mu = 0.001  # smoothing param
    min_eta = 0.001
    last_improve = time.time()
    it=0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val - 1e-12:
            best_val = val; best_seq = a.copy(); last_improve = time.time()
            print(f"new best {val:.6f} iter {it}")

        # compute softmax weights
        # scaling for stability: subtract max
        exp_vals = np.exp((conv - max_c) / mu)
        w = exp_vals / exp_vals.sum()

        # compute gradient
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # update step size
        eta = max(min_eta, eta * decay_rate) # or fixed? Use decayed

        a = a * np.exp(-eta * grad)
        a = a / a.sum()
        # if stall for >5s, random jitter
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n)-0.5) * 0.02
            a = a * np.exp(jitter)
            a = a / a.sum()
            last_improve = time.time()
        it+=1
    return best_seq, best_val

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length fft_len (>=2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # truncate to the linear convolution length
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)],   i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the original objective for a (not necessarily normalised)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (s ** 2))

def _beta_candidate(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shaped density on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n  # centre of each bin
    shape = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum < 1e-12:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape

def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of diverse initial vectors."""
    cand = []

    # warm start and its rotations
    if warm_start is not None:
        cand.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm_start, shift))

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    # random Dirichlet with several concentrations
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # edge‑exponential family
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.5, num=20):  # up to 31.6
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law family
    for p in np.linspace(0.2, 6.0, num=20):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # symmetric Beta shapes – α<1 puts most mass near the ends
    for alpha in (0.15, 0.2, 0.25, 0.3, 0.35, 0.4):
        cand.append(_beta_candidate(n, alpha))

    # positive half‑wave sinusoid (already in original)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # a few pure Dirichlet draws with extreme concentrations
    for conc in (0.005, 0.02, 0.07):
        for _ in range(12):
            cand.append(np.random.dirichlet(np.full(n, conc)))

    return cand

def _exp_gradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Exponentiated‑gradient descent using a single‑index sub‑gradient
    (the active shift where the convolution attains its current maximum).
    The step‑size decays logarithmically; occasional random jitter
    prevents stagnation.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    eta = 0.8          # initial step size
    decay = 0.99992    # per‑iteration decay
    min_eta = 0.001
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # convolution and objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        # Track best
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} iter={it:07d}] new best {val:.6f}")

        # Sub‑gradient: pick a single index achieving the maximum.
        # If there are several, pick one at random.
        max_idxs = np.flatnonzero(conv == max_c)
        idx = max_idxs[0] if max_idxs.size == 1 else np.random.choice(max_idxs)
        w = np.zeros_like(conv)
        w[idx] = 1.0

        # gradient of the weighted convolution (factor 2 from symmetry)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule
        eta = max(min_eta, eta * decay)

        # exponentiated‑gradient update (log‑domain multiplicative)
        a = a * np.exp(-eta * grad)

        # re‑normalise onto the probability simplex
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # jitter if we have not improved for a few seconds
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02   # log‑space
            a = a * np.exp(jitter)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine builds a diverse pool of parametric
    seeds, keeps the most promising few, and refines each of them with a
    multiplicative‑gradient descent that always follows the direction of the
    currently maximal convolution entry.  After an initial optimisation on a
    medium‑size grid (n=8192) the best vector is interpolated to a finer grid
    (n=16384) and refined again if time permits.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Primary grid – a compromise between resolution and speed.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if it exists.
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich set of initial vectors.
    candidates = _generate_candidates(n, warm_start)

    # ------------------------------------------------------------------
    # Quick evaluation of the whole pool (FFT‑based, O(N log N) per candidate).
    evaluated = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    if not evaluated:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    # Keep the top‑k promising seeds (k small to give each sufficient time).
    evaluated.sort(key=lambda x: x[0])
    top_k = 4
    top_candidates = [c for _, c in evaluated[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of every selected seed.
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # allocate roughly equal remaining time to the remaining starts
        remaining = deadline - time.time()
        remaining_per = remaining / (len(top_candidates) - start_id)
        sub_deadline = time.time() + remaining_per
        # (optional) print current start info
        init_val = _objective(a0, n, conv_fft_len)
        print(f"--- Optimising start {start_id + 1}/{len(top_candidates)} "
              f"(init {init_val:.6f})")
        best_seq, best_val = _exp_gradient_opt(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – if there is still time, upscale and fine‑tune.
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 30.0:
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Upscaling to n=16384 for final refinement")
        sub_deadline = time.time() + (time_left - 5.0)
        best_seq, best_val = _exp_gradient_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_deadline,
            best_seq,
            best_val,
        )
        n = n_big

    # ------------------------------------------------------------------
    # Safety clipping (evaluate_sequence expects numbers in [0,1000])
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished – best objective ≈ {best_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()

    # edge‑exponential family
    for alpha in np.logspace(-2, 1.5, num=20):      # up to ≈31.6
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law family
    for p in np.linspace(0.2, 6.0, num=20):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # symmetric Beta shapes (α<1 → strong edge concentration)
    for alpha in (0.15, 0.2, 0.25, 0.3, 0.35, 0.4):
        cand.append(_beta_candidate(n, alpha))

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # a few extra Dirichlet draws with very small concentration
    for conc in (0.005, 0.02, 0.07):
        for _ in range(12):
            cand.append(np.random.dirichlet(np.full(n, conc)))

    return cand


def _exp_gradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Exponentiated‑gradient descent that follows the sub‑gradient of the
    *single* maximal convolution entry.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    eta = 0.8          # initial step size
    decay = 0.99992    # per‑iteration decay
    min_eta = 0.001
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} iter={it:07d}] new best {val:.6f}")

        # pick a single index achieving the maximal convolution value
        max_idxs = np.flatnonzero(conv == max_c)
        idx = max_idxs[0] if max_idxs.size == 1 else np.random.choice(max_idxs)
        w = np.zeros_like(conv)
        w[idx] = 1.0

        # gradient of the weighted convolution (factor 2 from symmetry)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule
        eta = max(min_eta, eta * decay)

        # exponentiated update – clip exponent to avoid overflow
        exp_factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * exp_factor
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # occasional jitter to escape plateaus
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02   # log‑space jitter
            a = a * np.exp(jitter)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point used by the evaluation harness.

    Returns a list of non‑negative heights (a step function) that seeks to
    minimise the given objective within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Primary grid size – 8192 balances resolution and iteration speed.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction if present.
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a broad candidate pool.
    candidates = _generate_candidates(n, warm_start)

    # ------------------------------------------------------------------
    # Quick evaluation of the whole pool (FFT‑based).  Keep the best few.
    evaluated = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    if not evaluated:                     # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    evaluated.sort(key=lambda x: x[0])
    top_k = 4
    top_candidates = [c for _, c in evaluated[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected seeds.
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # allocate roughly equal remaining time among remaining starts
        remaining = deadline - time.time()
        per_start = remaining / (len(top_candidates) - start_id)
        sub_deadline = time.time() + per_start

        init_val = _objective(a0, n, conv_fft_len)
        print(f"--- Optimising start {start_id + 1}/{len(top_candidates)} "
              f"(init obj={init_val:.6f})")
        best_seq, best_val = _exp_gradient_opt(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – if we still have time, upscale to a finer grid and refine.
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 30.0:
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Upscaling to n=16384 for final refinement")
        sub_deadline = time.time() + (time_left - 5.0)
        best_seq, best_val = _exp_gradient_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_deadline,
            best_seq,
            best_val,
        )
        n = n_big

    # ------------------------------------------------------------------
    # Final safety clipping – evaluate_sequence accepts values in [0,1000].
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished – best objective ≈ {best_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
