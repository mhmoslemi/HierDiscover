def _refine_with_softmax(a0, n, conv_fft_len, corr_fft_len, deadline,
                         best_seq, best_val):
    a = a0.copy()
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a/ a.sum()
    # enforce symmetry
    a = (a + a[::-1]) / 2
    # parameters
    eta0 = 0.5
    decay = 0.99995
    min_eta = 0.001
    beta0 = 0.1
    beta_growth = 1.0015
    beta_max = 200.0
    it = 0
    last_improve = time.time()
    while time.time() < deadline and it < 10000000:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)  # sum is 1
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} refine iter={it}] new best {val:.6f}")
        # softmax weights
        beta = beta0 * (beta_growth ** it)
        beta = min(beta, beta_max)
        # compute w = softmax(beta * conv)
        # shift for numeric stability
        # by subtracting max conv value before exponentiation
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # gradient
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # step size
        eta = max(min_eta, eta0 * (decay ** it))
        # update a via exponentiated
        a = a * np.exp(-eta * grad)
        # enforce symmetry
        a = (a + a[::-1]) / 2
        # renorm
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        # random perturbation if idle
        if time.time() - last_improve > 8.0:
            perturb = (np.random.rand(n) - 0.5) * 0.02
            a = a * np.exp(perturb)
            a = (a + a[::-1]) / 2
            a = a / a.sum()
            last_improve = time.time()
        it+=1
    return best_seq, best_val
```

"""
Hybrid search for a low‑evaluation step function.
------------------------------------------------
1.  A fast parametric sweep (edge‑exponential, edge‑power‑law,
    edge‑spike) on a modest grid (n=4096) gathers a handful of good
    shapes.
2.  Each of the top shapes is refined with a soft‑max‑smoothed
    exponentiated‑gradient (mirror‑descent) that enforces symmetry and
    uses an adaptive step‑size / “temperature’’ (β) schedule.
3.  If time permits the best vector is interpolated to a finer grid
    (n=8192) and the refinement is repeated.
All values are clipped to the range required by ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]  for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) / (sum(a)²).
    The vector a is assumed to be normalised (sum≈1); the function does
    not enforce normalisation.
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Parametric candidate generation & sweep
# ----------------------------------------------------------------------
def _sweep_candidates(n: int, conv_fft_len: int, deadline: float):
    """
    Generate a diverse family of parametric shapes, evaluate them quickly,
    and return the best few (sorted by objective value).
    """
    i = np.arange(n, dtype=np.float64)
    candidates = []

    # 1) Edge‑exponential family  a_i ∝ exp(-α·i)+exp(-α·(n‑1‑i))
    alphas = np.logspace(-3, 1.0, num=30)        # 0.001 … 10
    ws = np.linspace(0.0, 1.0, num=21)          # mixing weight with uniform
    for alpha in alphas:
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge_sum = edge.sum()
        edge = edge / edge_sum
        for w in ws:
            shape = w * edge + (1.0 - w) / n
            shape = shape / shape.sum()
            candidates.append(shape)

    # 2) Edge‑power‑law family  a_i ∝ (i+1)^(-p)+(n‑i)^(-p)
    ps = np.linspace(0.2, 5.0, num=30)
    for p in ps:
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge = edge / edge.sum()
        for w in ws:
            shape = w * edge + (1.0 - w) / n
            shape = shape / shape.sum()
            candidates.append(shape)

    # 3) Pure edge spikes (mass p at each extreme, remainder uniform)
    for spike in np.linspace(0.0, 0.45, num=21):
        if spike * 2 >= 0.999:
            continue
        interior_mass = max(0.0, 1.0 - 2 * spike)
        shape = np.full(n, interior_mass / (n - 2) if n > 2 else 0.0)
        shape[0] = spike
        shape[-1] = spike
        shape = shape / shape.sum()
        candidates.append(shape)

    # 4) Uniform (already covered by w=0, but keep for safety)
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Evaluate all candidates (fast FFT‑based) and keep the best 6.
    vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        vals.append((val, cand))

    if not vals:
        # fallback – uniform
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        return [(float('inf'), uniform)]

    vals.sort(key=lambda x: x[0])
    top_k = 6
    return vals[:top_k]


# ----------------------------------------------------------------------
# Soft‑max‑smoothed exponentiated‑gradient refinement
# ----------------------------------------------------------------------
def _refine_softmax(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Mirror‑descent on the true max‑convolution, but the gradient is taken
    w.r.t. a smooth approximation of the max (soft‑max with a growing
    temperature β).  Symmetry a_i = a_{n‑1‑i} is enforced after each step.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    # enforce symmetry from the start
    a = (a + a[::-1]) / 2.0

    # hyper‑parameters
    eta0 = 0.6          # initial step size
    decay = 0.99995     # geometric decay for η
    min_eta = 0.001

    beta0 = 0.1         # initial “temperature’’ for soft‑max
    beta_growth = 1.0015
    beta_max = 200.0

    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < 10_000_000:
        # convolution and true objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # sum≈1 → denominator≈1

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} refine iter={it}] new best {val:.6f}")

        # -------- soft‑max weights ----------
        beta = beta0 * (beta_growth ** it)
        beta = min(beta, beta_max)

        # w_i = exp(β·(conv_i – max)) / Σ_j exp(β·(conv_j – max))
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        # -------- gradient of smooth max ----------
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # -------- exponentiated‑gradient update ----------
        eta = max(min_eta, eta0 * (decay ** it))
        a = a * np.exp(-eta * grad)

        # enforce symmetry and renormalise
        a = (a + a[::-1]) / 2.0
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # occasional random jitter if stuck
        if time.time() - last_improve > 8.0:
            perturb = (np.random.rand(n) - 0.5) * 0.02
            a = a * np.exp(perturb)
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (non‑negative vector) minimising
    `evaluate_sequence`.  Returns the best vector found as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Primary resolution (fast sweep)
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm‑start from a previous run if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] != n:
            if warm_start.shape[0] > n:
                warm_start = warm_start[:n]
            else:
                warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – parametric sweep (fast)
    # ------------------------------------------------------------------
    sweep = _sweep_candidates(n, conv_fft_len, deadline)
    top_candidates = [cand for _, cand in sweep]

    # also inject a few random Dirichlet seeds for diversity
    for _ in range(15):
        if time.time() >= deadline:
            break
        cand = np.random.dirichlet(np.full(n, 0.2))
        top_candidates.append(cand)

    # keep only the top few distinct starts (duplicates are harmless)
    if len(top_candidates) > 8:
        # quick crude scoring to prune
        scored = [( _objective(c, n, conv_fft_len), c) for c in top_candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:8]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2 – soft‑max refined mirror‑descent on each start
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        init_val = _objective(a0, n, conv_fft_len)
        print(f"=== Refinement start {idx + 1}/{len(top_candidates)} "
              f"(initial objective={init_val:.6f})")
        best_seq, best_val = _refine_softmax(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – upscale to a finer grid if time allows
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Upscaling to n=8192 and final refinement")
        best_seq, best_val = _refine_softmax(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # If the best solution lives on the big grid, keep that size.
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback handling
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's accepted range and convert to list
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # expose the result for a possible warm‑start in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
