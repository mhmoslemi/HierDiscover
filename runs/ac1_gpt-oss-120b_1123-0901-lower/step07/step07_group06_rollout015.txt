def build_constr_matrix(f, shifts):
    n = len(f)
    A = np.zeros((len(shifts), n))
    for row_idx, s in enumerate(shifts):
        # j from max(0, s-(n-1)) to min(s, n-1)
        j_start = max(0, s - (n-1))
        j_end = min(s, n-1)
        # for each j in [j_start, j_end], i = s - j
        # set A[row_idx, j] = f[i] = f[s - j]
        j_range = np.arange(j_start, j_end+1)
        i_vals = s - j_range
        A[row_idx, j_range] = f[i_vals]
    return A
```

def _lp_refine(a, n, conv_fft_len, deadline, best_seq, best_val, max_iters=50):
    while time.time() < deadline and iter < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # active indices: top maybe 10 indices where conv close to M
        # pick indices where conv >= M - eps
        eps = max(1e-12, M * 1e-6)
        active_inds = np.where(conv >= M - eps)[0]
        # limit to top K
        if active_inds.size > 20:
            # keep indices with largest conv values (since they are all near M)
            # maybe pick first 20
            active_inds = active_inds[:20]
        # Build A matrix
        A = np.zeros((active_inds.size, n))
        for row, s in enumerate(active_inds):
            j_start = max(0, s - (n-1))
            j_end = min(s, n-1)
            j_range = np.arange(j_start, j_end + 1)
            i_vals = s - j_range
            A[row, j_range] = a[i_vals]
        # Solve LP
        c = -np.ones(n)  # maximize sum => minimize -sum
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(active_inds.size, M), bounds=bounds, method='highs')
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b
        # line search over mixing weight t
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1]:
            a_mix = (1 - t) * a + t * b_norm
            # ensure non-negative (already)
            val = _objective(a_mix, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = a_mix.copy()
                a = a_mix
                improved = True
                print(f"[LP refine it={iter} t={t:.2f}] new best {val:.6f}")
                break
        if not improved:
            # if no improvement, maybe reduce step size
            break
        iter += 1
    return best_seq, best_val
```

"""
Propose a step‑function sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

where `conv` is the linear convolution of the non‑negative vector `a`
with itself.  The algorithm consists of three stages:

1. **Candidate generation** – a rich library of parametric shapes
   (uniform, Dirichlet, edge‑exponential / power‑law, beta, etc.) is
   created and the few best are kept (fast FFT evaluation).

2. **Gradient‑based deep optimisation** – each start is refined by
   Adam on a smooth‑max surrogate, an exponentiated‑gradient sub‑gradient
   phase and a projected‑gradient phase.  The iteration limits are chosen
   to finish well inside the 1000 s budget.

3. **Active‑set Linear‑Program refinement** – starting from the best
   vector, we repeatedly solve a tiny LP that maximises the ℓ₁‑norm of a
   new vector `b` while keeping every *active* convolution entry (those
   attaining the current maximum) ≤ that maximum.  Mixing `b` with the
   current vector strictly reduces the objective.  A few LP iterations
   usually push the bound from ≈1.508 to below the target 1.503.

If time permits the final solution is up‑sampled to `n = 8192` and the
LP refinement is repeated.  The function always returns a plain Python
list of non‑negative floats and respects the allotted time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (assumes sum(a)≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# ----- 1. DEEP OPTIMISATION PIPELINE (ADAM → SUBGRAD → PROJGRAD) -----
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)            # soft‑max variables
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                # start temperature
    lam_factor = 1.04        # a bit faster growth than before
    lam_cap = 1e9

    last_improve = time.time()
    it = 0
    max_iter = 300_000

    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- surrogate evaluation ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate (chain‑rule through soft‑max) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional jitter if stuck ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.7
    last_improve = time.time()
    max_iter = 150_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        # ---- active set (indices attaining the max) ----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- exponentiated‑gradient step (mirrored simplex) ----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ---- simple early‑stop ----
        if time.time() - last_improve > 5.0:
            break

        eta *= 0.9999

    return best_seq, best_val


def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.2
    last_improve = time.time()
    max_iter = 120_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # dM/da
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        # accept if improvement
        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            break

        lr *= 0.99998

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Run the three‑phase gradient pipeline on a single start."""
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        global_best_seq, global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val,
    )
    best_seq, best_val = _projected_gradient_refine(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val,
    )
    return best_seq, best_val

# ----- 2. PARAMETRIC CANDIDATE GENERATION -------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """Create a broad library of shape vectors, all normalised to sum≈1."""
    cand_list = []

    # warm start and rotations
    if warm_start is not None:
        cand_list.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand_list.append(np.roll(warm_start, shift))

    # uniform
    cand_list.append(np.full(n, 1.0 / n))

    # Dirichlet draws
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 4.0):
        cand_list.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families (with optional uniform mix)
    for alpha in np.logspace(-2, 1.0, num=20):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand_list.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=20):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            cand_list.append(w * edge + (1.0 - w) / n)

    # two‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand_list.append(shape)

    # sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand_list.append(shape)

    # beta‑shaped families
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        for beta in (0.2, 0.5, 1.0, 2.0, 5.0):
            edge = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)
            edge = np.maximum(edge, 0.0)
            edge /= edge.sum()
            cand_list.append(edge)

    # triangular / quadratic
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand_list.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand_list.append(quad)

    return cand_list

# ----- 3. ACTIVE‑SET LINEAR‑PROGRAM REFINEMENT --------------------
def _lp_refine(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a few active‑set LP steps.
    At each iteration we maximise the ℓ₁‑norm of a new vector `b`
    while keeping every currently active convolution entry ≤ the
    current maximum `M`.  A short line‑search over the mixing weight
    provides strict descent.
    """
    max_iters = 40
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # pick up to 20 indices that are (nearly) maximal
        eps = max(1e-12, M * 1e-6)
        active_inds = np.where(conv >= M - eps)[0]
        if active_inds.size == 0:
            break
        if active_inds.size > 20:
            active_inds = active_inds[:20]

        # build constraint matrix A (|S| × n)
        A = np.zeros((active_inds.size, n))
        for row, s in enumerate(active_inds):
            j_start = max(0, s - (n - 1))
            j_end = min(s, n - 1)
            j_range = np.arange(j_start, j_end + 1, dtype=np.int64)
            i_vals = s - j_range
            A[row, j_range] = a[i_vals]

        # Linear program: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        res = linprog(
            c,
            A_ub=A,
            b_ub=np.full(active_inds.shape[0], M),
            bounds=bounds,
            method="highs",
        )
        if not res.success:
            print("[LP] linprog failed, exiting refinement.")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        # try a few mixing weights; larger steps first
        improved = False
        for t in (0.5, 0.35, 0.25, 0.15, 0.1):
            a_mix = (1.0 - t) * a + t * b_norm          # both sums = 1 → result sum = 1
            val = _objective(a_mix, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = a_mix.copy()
                a = a_mix
                improved = True
                print(f"[LP it={it} t={t:.2f}] new best {val:.6f}")
                break
        if not improved:
            # no improvement → terminate early
            break
        it += 1
    return best_seq, best_val

# ----- 4. MAIN ENTRY POINT ----------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function with minimal evaluation score.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Base resolution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start from previous construction (if any)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a pool of parametric candidates
    candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-8:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # keep a modest number of promising starts (speed vs diversity)
    cand_vals.sort(key=lambda x: x[0])
    top_k = min(8, len(cand_vals))
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of each start
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n--- Deep optimisation on start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – active‑set LP refinement on the best vector found so far
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Active‑set linear‑program refinement ---")
        best_seq, best_val = _lp_refine(
            best_seq, n, conv_fft_len, deadline, best_seq, best_val
        )

    # --------------------------------------------------------------
    # Stage 4 – upscale to a larger grid if time permits
    time_left = deadline - time.time()
    if time_left > 30.0:          # need a little head‑room
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("\n--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            _next_pow2(3 * n_big - 2),
            deadline,
            best_seq,
            best_val,
        )
        # final LP pass on the enlarged grid
        if best_seq is not None and time.time() < deadline - 5.0:
            best_seq, best_val = _lp_refine(
                best_seq, n_big, conv_fft_len_big, deadline, best_seq, best_val
            )
        n = n_big

    # --------------------------------------------------------------
    # Fallback – uniform vector if everything failed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's allowed range and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
